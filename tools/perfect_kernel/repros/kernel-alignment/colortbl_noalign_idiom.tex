% colortbl's (and tabu's) idiom `\ifdim\baselineskip=\z@\noalign\fi{…}` (colortbl.sty:158, :163) chooses between a
% `\noalign` group between rows and a plain group inside a cell by testing `\baselineskip`: array.sty's `\@array`
% sets `\baselineskip\z@` for the table body, so between rows the test is true. The tabular binding never zeroes
% `\baselineskip`, so the group becomes a cell, and the `\hline` after it meets "\noalign cannot be used here" and two
% stray `&` (review 60j r1 side finding). SHARED (Perl alike, KNOWN_PERL_ERRORS #445). Now `\@array`'s zero
% `\lineskip`/`\baselineskip` (latex.ltx:16580) and `\@arrayparboxrestore`'s restore in vertical cells are kept.
% status:  GREEN 60l; guard perfect_kernel_batch60::colortbl_noalign_idiom
% oracle:  pdflatex 0 errors: "a b" / "c d" / "e f" with rules / "Final."

% expect:  0 errors; a three-row table with rules, then "Final."
\documentclass{article}
\usepackage{colortbl,xcolor}
\makeatletter
\def\myarc#1{\ifdim\baselineskip=\z@\noalign\fi{\gdef\CT@arc@{\color{#1}}}}
\makeatother
\begin{document}
\begin{tabular}{|l|l|}\hline a&b\\\arrayrulecolor{red}\hline c&d\\\myarc{blue}\hline e&f\\\hline\end{tabular}

Final.
\end{document}
