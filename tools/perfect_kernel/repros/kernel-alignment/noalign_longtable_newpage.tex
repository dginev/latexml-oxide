% mechanism: a longtable is NOT a box — its rows sit in outer vertical mode, so vertical

\documentclass{article}
\usepackage{longtable}
\begin{document}
\begin{longtable}{|l|c|}

\hline
\newpage

\end{longtable}
\end{document}
