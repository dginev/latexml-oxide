\newif\ifmore
\def\peel{\global\n=0
  \loop
    \setbox\pb=\vbox{\unvbox\pb \global\setbox\lb=\lastbox}%
    \global\advance\n by 1
    \ifvoid\lb \morefalse\else
      \xdef\seen{\seen\the\n:\ifhbox\lb h\else v\fi,}%
      \ifnum\n<6 \moretrue\else\morefalse\fi
    \fi
  \ifmore\repeat}
