\documentclass{article}
\makeatletter
\AddToHook{label}{\xdef\lastctr{\@currentcounter/\csname the\@currentcounter\endcsname}}
\makeatother
\newtheorem{thm}{Theorem}
\newtheorem{lem}[thm]{Lemma}
\begin{document}
\section{One}\label{s1}
Section: \lastctr.
\begin{enumerate}\item a \item b\label{i2}\end{enumerate}
Item: \lastctr.
\begin{equation}x\label{e1}\end{equation}
Equation: \lastctr.
\begin{eqnarray}y\label{e2}\end{eqnarray}
Eqnarray: \lastctr.
\begin{lem}L\label{l1}\end{lem}
Lemma: \lastctr.
\begin{figure}\caption{C}\label{g1}\end{figure}
Figure: \lastctr.
{\refstepcounter{table}}\label{g}
Group: \lastctr.
\end{document}
