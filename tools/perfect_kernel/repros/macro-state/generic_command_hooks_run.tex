\documentclass{article}
\newcounter{foo}
\newcommand\hits{}
\newcommand\mycmd[1]{<#1>}
\AddToHook{cmd/mycmd/before}{\xdef\hits{\hits b}}
\AddToHook{cmd/mycmd/after}{\xdef\hits{\hits a}}
\AddToHook{cmd/refstepcounter/after}{\xdef\hits{\hits r}}
\begin{document}
X\mycmd{1}\refstepcounter{foo}[\hits].
\end{document}
