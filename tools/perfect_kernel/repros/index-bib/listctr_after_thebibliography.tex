%   counter in `\@listctr`; asl.cls composes each entry's number from it (`\addtocounter{\@listctr}{1}\the\value

\documentclass{article}
\makeatletter
\def\composelist{\addtocounter{\@listctr}{1}\the\value{\@listctr}}
\def\@wrap@list#1{[#1]}
\def\bibritem#1{\item[\@wrap@list{\composelist}]\ignorespaces}
\makeatother
\begin{document}
Text.
\begin{thebibliography}{9}
\bibritem{a} First entry.
\bibritem{b} Second entry.
\end{thebibliography}
\end{document}
