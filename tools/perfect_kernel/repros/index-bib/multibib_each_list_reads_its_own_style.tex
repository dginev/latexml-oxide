% declares. The review's order: both declared before both lists.
% witness: the review of M2 (int140a `alpha` on both, M2 `plain` on both).
% oracle:  pdflatex + bibtex (TL 2025) 0 errors: main list [Zet01], web list [1].
% engines: int140a bibstyle="alpha" on both; M2 bibstyle="plain" on both.
% expect:  0 errors, 0 warnings; bibstyle="alpha" citestyle="AY" on the main list, bibstyle="plain" citestyle="numbers" on web.

\usepackage{multibib}
\newcites{web}{Web}
\begin{document}

\end{document}
