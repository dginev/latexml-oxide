\documentclass{article}
\usepackage{glosmathtools}
\makeglossaries
\setglossarystyle{nomencl-L1L2}
\newglossaryentry{latin}{name={latin},description={Latin symbols},descseclang={Symboles latins},sort=1}
\newglosentrymath{m}{m}{description={mass},descseclang={masse},sort=m,parent=latin}
\newglosentrymath{d}{d}{description={diameter},descseclang={diametre},sort=d,parent=latin}
\newglossaryentry{greek}{name={greek},description={Greek symbols},descseclang={Symboles grecs},sort=2}
\newglosentrymath{mu}{\mu}{description={viscosity},descseclang={viscosite},sort=12,parent=greek}
\newglossaryentry{vectors}{name={vectors},description={Vectors},descseclang={Vecteurs},sort=3}
\newglosentrymath{v}{v}{description={velocity},descseclang={vitesse},sort=v,parent=vectors}
\newglossaryentry{operator}{name={operator},description={Operators},descseclang={Operateurs},sort=5}
\newglosentrymath{op.dot}{\dot{\bullet}}{description={flow rate},descseclang={debit},parent=operator}
\begin{document}
Symbols \gls{m}, \gls{d}, \gls{mu} and \gls{v}.
\printglossary[title=Nomenclature]
\end{document}
