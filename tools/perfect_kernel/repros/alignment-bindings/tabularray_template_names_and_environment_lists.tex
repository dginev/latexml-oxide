\documentclass{article}
\usepackage{tabularray}
\SetTblrTemplate{caption}{nosuch}
\SetTblrTemplate{caption-tag,note-tag}{simple}
\SetTblrTemplate{caption}{simple}
\SetTblrTemplate{nosuchelement}{normal}
\SetTblrTemplate{head}{default}
\SetTblrInner[]{colspec={r}}
