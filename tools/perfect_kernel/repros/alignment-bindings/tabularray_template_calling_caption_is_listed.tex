\documentclass{article}
\usepackage{tabularray,caption}
\ExplSyntaxOn
\newcommand\mycap{\caption{\InsertTblrText{caption}}}
\ExplSyntaxOff
\NewTblrTheme{viamacro}{\DefTblrTemplate{firsthead}{default}{\addtocounter{table}{-1}\captionsetup{type=table}\mycap}%
  \SetTblrTemplate{caption-lot}{empty}}
\let\capb\caption
\def\capx{\caption}
\def\pre{\addtocounter{table}{-1}\captionsetup{type=table}}
\NewTblrTheme{letcap}{\DefTblrTemplate{firsthead}{default}{\pre\capb{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{csname}{\DefTblrTemplate{firsthead}{default}{\pre\csname caption\endcsname{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{capof}{\DefTblrTemplate{firsthead}{default}{\addtocounter{table}{-1}\captionof{table}{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{trailstar}{\DefTblrTemplate{firsthead}{default}{\captionsetup{type=table}\capx*{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{headempty}{\SetTblrTemplate{head}{empty}%
  \DeclareTblrTemplate{caption}{default}{\pre\caption{\InsertTblrText{caption}}}\SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{capfig}{\DefTblrTemplate{firsthead}{default}{\captionof{figure}{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{spacecs}{\DefTblrTemplate{firsthead}{default}{\csname caption \endcsname{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{usecap}{\DefTblrTemplate{firsthead}{default}{\UseTblrTemplate{caption}{default}}%
  \DeclareTblrTemplate{caption}{default}{\pre\caption{\InsertTblrText{caption}}}\SetTblrTemplate{caption-lot}{empty}}
\def\ya{\yb}\def\yb{\yc}\def\yc{\xx}\def\xx{\yy}\def\yy{\caption}
\def\cof{\captionof}
\NewTblrTheme{deepfirst}{\DefTblrTemplate{firsthead}{default}{\iffalse\ya\fi\pre\xx{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{cofmacro}{\DefTblrTemplate{firsthead}{default}{\pre\cof{table}{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
\NewTblrTheme{starred}{\DefTblrTemplate{firsthead}{default}{\captionsetup{type=table}\caption*{\InsertTblrText{caption}}}%
  \SetTblrTemplate{caption-lot}{empty}}
