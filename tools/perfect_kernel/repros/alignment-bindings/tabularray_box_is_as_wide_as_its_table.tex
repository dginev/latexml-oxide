% line width (scale 1).

\usepackage{graphicx,tabularray}
\begin{document}
Before.
\resizebox{\linewidth}{!}{\begin{talltblr}[caption={Resized}, note{a}={Resized note.}]{colspec={ll}}
 A & B\TblrNote{a} \\
\end{talltblr}}

\end{talltblr}}
\end{table}
