\usepackage{newverbs}
\newenvironment{codeobj}{[}{]}
\makeatletter
\let\originalverb\verb
\renewcommand{\verb}{\begin{codeobj}\originalverb}

\renewcommand{\verb@egroup}{\originalverb@egroup\end{codeobj}}
\makeatother
