% `<anchor>O.</anchor>P.` (also an empty name, `\hyper@@anchor`). The constructor is digested in vertical
% mode, so the space after it is skipped there; in TeX the anchor's text starts the paragraph first
% (hyperref.sty:4805-4811 → `\hyper@@anchor` typesets #2 in horizontal mode) and the space follows in the
% paragraph. `\mbox{Q.} R.` keeps its space (enter_horizontal). Pre-existing (57co2 identical).

\documentclass{article}
\usepackage{hyperref}
\makeatletter
\def\emptyname{}
\begin{document}
\hypertarget{}{K.} L.

\hyper@@anchor{\emptyname}{M.} N.

\hypertarget{x}{O.} P.

\hypertarget{y}{}

\mbox{Q.} R.
\makeatother
\end{document}
