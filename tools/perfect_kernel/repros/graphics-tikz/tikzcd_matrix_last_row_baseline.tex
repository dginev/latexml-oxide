% Issue 785: a TikZ matrix (tikz-cd here) is placed with its LAST row's baseline at the
% matrix origin -- Perl openTikzAlignment: y = (h + d) - rowdepths[-1]. Dropping the
% last row's depth lifts every cell by that depth, off its arrows.
\documentclass{article}
\usepackage{tikz-cd}
\begin{document}
\begin{tikzcd}
  A \arrow[r, "f"] \arrow[d, "g"'] & B \arrow[d, "h"] \\
  C \arrow[r, "k"'] & D
\end{tikzcd}

\begin{tikzcd}
  g \arrow[r] & y
\end{tikzcd}

% The issue's own example: a rotated, coloured last-row cell, and a \large last row.
\begin{tikzcd}
A \arrow[r, "\phi"] \arrow[d, red]
& B \arrow[d, "\psi" red] \\
C \arrow[r, red, "\eta" blue]
& |[blue, rotate=-15]| D
\end{tikzcd}
\begin{tikzcd}
  A \arrow[r, "\phi\eta" near start, "\phi" near start, "\psi"',"\normalsize \eta x_i^2" near end] & {\hbox{\large B}}
\end{tikzcd}
\end{document}
