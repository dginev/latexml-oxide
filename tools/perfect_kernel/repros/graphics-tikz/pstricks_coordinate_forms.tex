\documentclass{article}
\usepackage{pstricks,pst-node}
\begin{document}
\begin{pspicture}(0,0)(4,3)
\rput(1,1){\rnode{A}{X}}
\pscircle(A){0.3}
\psline(1,0)(1;45)
\psline(!1 2)(2,2)
\psline(A|B)(2,2)
\psline(1,2|3,4)(2,2)
\psline([angle=45,nodesep=1]A)(2,2)
\rput([angle=45,nodesep=1]A){L}
\radians\pswedge(0,0){1}{0}{3.14159}\degrees
\end{pspicture}
\end{document}
