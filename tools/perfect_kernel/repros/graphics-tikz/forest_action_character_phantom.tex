% mechanism: `\bracketset{action character=@}` (forest.sty:1419) makes `@` an action of the
%   bracket parser (:1491): `@+` expands every later token (:1612-1614), and `\x`'s `@@`
%   (:1622-1623) stops the parse until `\bracketResume` (:1450) resumes it on `\xtemp`, so
%   each `\x{c}` builds the root's children `[$\times_n$[c]]`. The binding's token-level
%   parser performs no action and reads all of it as the root's label; a tree whose text
%   holds the action character at brace depth 0 therefore does not apply `phantom`, which
%   hides only the (empty) root in forest.
\documentclass{article}
\usepackage{forest}
\begin{document}
\bracketset{action character=@}
\newcount\xcount
\def\x#1{@@\advance\xcount1
  \edef\xtemp{[$\noexpand\times_{\the\xcount}$[#1]]}%
  \expandafter\bracketResume\xtemp
}
\begin{forest}
  phantom,
  delay={where level=1{content={\strut #1}}{}}
  @+
  [\x{f}\x{o}\x{r}\x{e}\x{s}\x{t}]
\end{forest}

% CONTROL: an action character set as a LETTER `@` (prooftrees.sty:940, neoschool.cls:8568)
% does not match the document's OTHER `@` (`\ifx`, forest.sty:1491), so this tree is read as
% written and its phantom root stays hidden: pdflatex prints only "kid".
\makeatletter\bracketset{action character=@}\makeatother
\begin{forest}[root @x, phantom [kid]]\end{forest}
\end{document}
