\documentclass{article}
\usepackage{amsthm,thmtools,thm-restate}
\makeatletter
\newtheorem{@theorem}{Theorem}[section]
\newenvironment{theorem}{\begin{@theorem}}{\end{@theorem}}
\makeatother
\begin{document}
\section{One}
\begin{restatable}{theorem}{mainthm}\label{thm:main}
Every graph is fine.
\end{restatable}
\begin{theorem}\label{thm:next}
Every tree is fine.
\end{theorem}
\section{Two}
Restated:
\mainthm*
\begin{theorem}\label{thm:after}
Every forest is fine.
\end{theorem}
See Theorems~\ref{thm:main}, \ref{thm:next} and~\ref{thm:after}.
\end{document}
