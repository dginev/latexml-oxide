% mdframed sets a `frametitle=` (the env's options, `\newmdenv`'s preset, `\mdfsetup`) in its own box on top of the
% frame, in `\normalfont\mdf@frametitlefont` (default `\bfseries`, mdframed.sty:702-711, :858-859); a frame nested in a
% titled one inherits its title, as the keys are set in the frame's group. Kept as the frame's first paragraph.

% oracle:  pdflatex 0 errors: "Outer Title / Outer body. / Outer Title / Inner body. / Preset Title / Boxed body. /

% engines: rust 67int28 = 0 errors, 1 warning (the stub notice), every title absent; perl not run

\documentclass{article}
\usepackage{mdframed}
\newmdenv[frametitle=Preset Title]{boxed}
\begin{document}
\begin{mdframed}[frametitle=Outer Title,linecolor=red]
Outer body.
\begin{mdframed}
Inner body.
\end{mdframed}
\end{mdframed}
\begin{boxed}
Boxed body.
\end{boxed}
\mdfsetup{frametitlefont=\itshape}
\begin{mdframed}[frametitle={Styled $x^2$ Title}]
Third body.
\end{mdframed}
\end{document}
