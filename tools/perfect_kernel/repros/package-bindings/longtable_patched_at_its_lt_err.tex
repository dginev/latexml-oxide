%   `\LT@err` reports a two-column document) and has no `\LT@output` (:487-...), so a preamble patch that splits

% witness: 1409.6572, 1503.02372, 1702.08827, 1909.09200, 2503.01096 (the "longtable in twocolumn" preamble hack)
% oracle:  pdflatex (TL 2025, -no-shell-escape) 0 errors: "Text before. A Text after."
% engines: INT081RC6 and DIGFATAL: 1 error ("Missing argument Until:\LT@err at end of input"), 1 fatal error, a 39-byte
%   document; Perl: 0 errors
% expect:  0 errors; "Text before.", the table, "Text after."
% status:  GREEN (EOFBIND)
\documentclass{article}
\usepackage{longtable}
\makeatletter
\long\def\foo#1\LT@err#2#3#4!!{\def\longtable{#1#4}}
\expandafter\foo\longtable!!
\long\def\foo#1\@outputpage#2\@outputpage#3!!{\def\LT@output{#1\@opcol#2\@opcol#3}}
\expandafter\foo\LT@output!!
\makeatother
\begin{document}
Text before.
\begin{longtable}{l}
A\\
\end{longtable}
Text after.
\end{document}
