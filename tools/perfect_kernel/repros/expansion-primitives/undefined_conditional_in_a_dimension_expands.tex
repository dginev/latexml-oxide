% to be typeset or assigned.
% witness: none (probe; the `\ifcompact\columnwidth\else\textwidth\fi` toggle idiom).

%          "C:yes." (pdflatex differs: TeX drops an undefined token).

\documentclass{article}
\begin{document}
\newdimen\da \newskip\sa \sa=7pt plus 1pt
A:\setlength{\da}{\ifcompact\columnwidth\else\textwidth\fi}[\the\da]

B:\da=\ifdraft\else\sa\fi[\the\da]

C:\ifdim\iflong\else\da\fi>1pt yes\else no\fi.
