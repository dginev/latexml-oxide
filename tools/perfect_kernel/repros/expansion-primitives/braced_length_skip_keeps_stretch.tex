\documentclass{article}
\newlength\mylen
\begin{document}
\setlength{\mylen}{1pt plus 2pt minus 3pt}[\the\mylen]

\setlength{\parskip}{3pt plus 1pt minus 1pt}[\the\parskip]

\addtolength{\mylen}{1pt plus 1fil}[\the\mylen]

A\hspace{0pt plus 1fil}B\hspace{\stretch{1}}C\hspace{\fill}D

\setlength{\parindent}{0pt plus 1pt}[\the\parindent]
\end{document}
