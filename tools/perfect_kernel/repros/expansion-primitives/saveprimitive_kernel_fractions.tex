\providecommand{\@saveprimitive}[2]{\begingroup\escapechar`\\\relax
  \edef\@tempa{\string#1}\edef\@tempb{\meaning#1}%
  \ifx\@tempa\@tempb \global\let#2#1%
  \else\edef\@tempb{\meaning#2}%
    \ifx\@tempa\@tempb\else\@latex@error{Unable to properly define
      \string#2; primitive \noexpand#1no longer primitive}\@eha\fi
  \fi\endgroup}
