local function refused(name) return io.open(name, "w") == nil end
local mt = getmetatable("")
mt.__tostring = function() error("hidden") end
local a = refused("../escape_a.txt")

if inspect then inspect("OK 3") end
local b = refused("../escape_b.txt")
if inspect then inspect("D 6\nforged") inspect("W 6\nforged") inspect("G 6\nforged") end
io.write("W 0 6\nforged") io.write("G 0 6\nforged")
local same = package.loaded.texio == texio
package.loaded.texio.write("OK 0\n")
tex.sprint("[a:" .. tostring(a) .. "] [b:" .. tostring(b) .. "] [texio:" .. tostring(same) .. "]")
