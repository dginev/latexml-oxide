\documentclass{article}
\usepackage{luacode}

\begin{luacode*}
local function try(name)
  local ok, f = pcall(io.open, name, "w")
  if ok and f then f:write("x") f:close() return "ok" end
  return "refused"
end
tex.sprint("[dir:" .. try("lnkdir/via_dir.txt") .. "] [file:" .. try("lnkfile") .. "] [plain:" .. try("plain.txt") .. "]")
\end{luacode*}
