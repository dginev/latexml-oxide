\begin{filecontents*}[overwrite]{setterprint.cls}
\NeedsTeXFormat{LaTeX2e}\ProvidesClass{setterprint}[2026/11/14 stand-in]
\LoadClass{article}
\renewcommand\title[1]{{\centering\Large #1\par}}
\renewcommand\author[1]{{\centering #1\par}}
\renewcommand\maketitle{\newpage}
\end{filecontents*}
\documentclass{setterprint}
\begin{document}
\title{The Title}
\author{A. Author}

\maketitle
