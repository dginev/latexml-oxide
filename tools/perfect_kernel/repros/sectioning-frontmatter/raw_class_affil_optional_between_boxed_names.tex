% mechanism: a class's in-place `\affil[1]{…}` between boxed names inside `\author` is the API's labelled entry, and the
%   marks printed after it are still the author's (OD #928 round 8, review rev94a r7 SF3; `queue_digest::
%   with_pending_entry_attr`, `digested_authors_requesting`): the labelled call left its deferred stub after the author
%   being digested, so the next `\textsuperscript{2}` requested its label of no entry — the 2 was lost and `\affil[2]`
%   found no author printing its key (printed inside the name). The key is read against the flush's authors' marks.
% witness: review rev94a r7 o/o10
% oracle:  pdflatex 0 errors: "ANN ABLE1, 1Univ A / BOB BAKER2, 2Univ B"
% engines: rust 94b r3: one creator, both lines in the name and `<sup>1,2</sup>`; rust 94a r7: Univ A the creator's,
%   ", Univ B" in the name, the 2 gone; perl: not run
% expect:  0 errors, 0 warnings; ANN ABLE Univ A, BOB BAKER Univ B (the boxed names re-parted from the reading); eager
%   and --streaming the same
% status:  RED (94a r7) -> GREEN (94a r8)
% preload: ar5iv.sty
\begin{filecontents*}[overwrite]{affilopt.cls}
\ProvidesClass{affilopt}
\LoadClass{article}
\def\author#1{\gdef\@author{#1}}
\newcommand\affil[2][]{{\reset@font\small\unskip,\ \textsuperscript{#1}#2\endgraf}}
\def\maketitle{\par{\Large\@title\par}{\noindent\@author\par}}
\end{filecontents*}
\documentclass{affilopt}
\begin{document}
\title{A Title}
\author{\mbox{ANN ABLE\textsuperscript{1}} \affil[1]{Univ A} \mbox{BOB BAKER\textsuperscript{2}} \affil[2]{Univ B}}
\maketitle
Body.
\end{document}
