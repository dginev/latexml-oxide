% A stand-in for IOS-Book-Article.cls's author markup (:1146-1205), written for a latexml-oxide test: `\author` prints

\LoadClass{article}
\def\author{\@ifnextchar[{\author@optarg}{\author@optarg[]}}
\def\author@optarg[#1]#2{\par\begingroup
  \def\fnms##1{##1}\def\snm##1{\MakeUppercase{##1}}%
  \def\orcid##1{\global\@namedef{orcid@#1}{##1}}%
  \centering\hbox{#2}\par\endgroup}
