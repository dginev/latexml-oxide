\begin{filecontents*}{fcs.cls}
% A minimal stand-in for fcs.cls (Front. Comput. Sci., 2020, Zhao Chongbin), written for a latexml-oxide test: its
% \author (:268-290), \address (:293-307) and the title block they print.
\NeedsTeXFormat{LaTeX2e}\ProvidesClass{fcs}[2020/01/01 stand-in]
\LoadClass{article}\RequirePackage{xparse}
\def\fcs@authors{}\def\fcs@addresses{}\newcounter{fcs@address}
\DeclareDocumentCommand\author{s o m}{\g@addto@macro\fcs@authors{#3\IfBooleanT{#1}{(\textsc{b})}%
  \IfNoValueF{#2}{\textsuperscript{\@for\fcs@r:=#2\do{\ref{\fcs@r} }}} }}
\DeclareDocumentCommand\address{o m}{\g@addto@macro\fcs@addresses{\par
  \IfNoValueTF{#1}{#2}{\refstepcounter{fcs@address}\label{#1}\arabic{fcs@address}\quad #2}}}
\renewcommand\maketitle{{\centering\LARGE\@title\par\normalsize\fcs@authors\par\fcs@addresses\par}}
\end{filecontents*}
