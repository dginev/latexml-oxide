\begin{filecontents*}[overwrite]{lxmaketitle.cls}
\NeedsTeXFormat{LaTeX2e}
\ProvidesClass{lxmaketitle}
\LoadClass{article}
% (a class's own `\maketitle`, as KOMA's: the binding's stays, and leaves the setters live)
\renewcommand\maketitle{\par}
\end{filecontents*}
\documentclass{lxmaketitle}
\begin{document}
