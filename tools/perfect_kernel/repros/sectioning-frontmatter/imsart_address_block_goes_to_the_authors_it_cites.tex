\begin{filecontents*}{imsart.sty}
\NeedsTeXFormat{LaTeX2e}\ProvidesFile{imsart.sty}[2012/01/01 IMS article style, shortened and modified]
\DeclareRobustCommand\ead[2][label= ,email]{\expandafter\gdef\csname ead@\detokenize{#1}\endcsname{#2}}
\def\printead#1{e-mail: \texttt{\csname ead@label=#1\endcsname}}
\newenvironment{aug}{}{\par}
\newenvironment{frontmatter}{\begin{center}}{\end{center}}
\def\and{\unskip~and~}
\def\author{\@ifnextchar[{\author@fmt}{\author@fmt[]}}
\def\author@fmt[#1]#2{\bgroup\def\fnms##1{##1}\def\snm##1{##1}#2\egroup}
\def\address{\@ifnextchar[{\address@fmt}{\address@fmt[]}}
\def\address@fmt[#1]#2{\par\begin{tabular}[t]{@{}l@{}}#2\end{tabular}\par}
\end{filecontents*}
\documentclass{article}
\usepackage{imsart}
