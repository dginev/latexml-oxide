% A minimal stand-in for stacs.cls (2008, on amsart), written for a latexml-oxide test: its \address (:294-298), the
% same store for \curraddr, \email and \urladdr (rmmcart.cls), \@maketitle (:351-364), \@setaddresses (:384-407) and
% \enddoc@text (:468).
\NeedsTeXFormat{LaTeX2e}\ProvidesClass{stacs}[2008/01/01 stand-in]
\LoadClass[11pt,reqno]{amsart}
\renewcommand{\address}[2][]{\g@addto@macro\addresses{\address{#1}{#2}}}
\renewcommand{\curraddr}[2][]{\g@addto@macro\addresses{\curraddr{#1}{#2}}}
\renewcommand{\email}[2][]{\g@addto@macro\addresses{\email{#1}{#2}}}
\renewcommand{\urladdr}[2][]{\g@addto@macro\addresses{\urladdr{#1}{#2}}}
\def\@setaddresses{\par\begingroup\def\address##1##2{##2\par}%
  \def\curraddr##1##2{Current address: ##2\par}\def\email##1##2{E-mail address: ##2\par}%
  \def\urladdr##1##2{URL: ##2\par}\addresses\endgroup}
\def\@maketitle{\@settitle\ifx\@empty\authors\else\@setauthors\fi\@setaddresses}
\def\enddoc@text{}
\end{filecontents*}
