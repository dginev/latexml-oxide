\begin{filecontents*}[overwrite]{vgtc.cls}
\ProvidesClass{vgtc}
\LoadClass{article}
\newcommand{\vgtc@empty}{}
\newcommand{\vgtc@teaser}{}
\newcommand{\teaser}[1]{\renewcommand{\vgtc@teaser}{\begin{minipage}{\linewidth}#1\end{minipage}}}
\newcommand{\vgtc@abstxt}{}
\let\origabstract\abstract
\let\endorigabstract\endabstract
\renewcommand{\abstract}[1]{\renewcommand{\vgtc@abstxt}{#1}}
\renewcommand{\@maketitle}{%
  \begin{center}%
    {\LARGE\@title\par}%
    \vspace{1\baselineskip}%
    \begin{tabular}[t]{c}\@author\end{tabular}\par
    \ifx\vgtc@teaser\vgtc@empty\else
      {\begingroup\def\@captype{figure}\vgtc@teaser\endgroup\par}%
    \fi
  \end{center}\par}
\let\vgtc@origmaketitle\maketitle
\renewcommand{\maketitle}{%
  \vgtc@origmaketitle
  \ifx\vgtc@abstxt\vgtc@empty\else
    \begingroup\begin{origabstract}\vgtc@abstxt\end{origabstract}\endgroup\par
  \fi}
\end{filecontents*}
