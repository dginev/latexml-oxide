% mechanism: a class's own `\@maketitle` (vgtc.cls:799-850, mimicked here) prints its teaser, a `\def\@captype{figure}`
%   minipage, under the authors and ABOVE the abstract it prints from its store; the deposit's figure is in place
%   (`IN_PLACE_FLOAT`), but the abstract is no late one: the frontmatter places it with the head, ahead of the deposit,
%   and where the class printed it relative to the figure is not read (the store fail-safe, `from_the_title_page`).
%   Witness 2609.25700 (vgtc, 18 of a 30k census sample). Deferred: row R TITLEFIG.
% oracle:  pdflatex 0 errors; pdftotext order: "Teaser Title", "Ann Alpha", "Figure 1: The teaser.",
%   "Abstract—Abstract text.", "1 Introduction"
% engines: rust INT081F and TITLEFIG: title, creator, abstract, figure.ltx_minipage, para "Abstract—"
% expect:  0 errors, 0 warnings; title, creator, titlepage.ltx_titlepage_continued > figure, abstract
% status:  RED (catalog, no guard)
\documentclass{article}
\makeatletter
% vgtc.cls's \@maketitle (:799-850): the teaser, a minipage set with \def\@captype{figure}, under the authors and
% above the abstract, all in the title block.
\newcommand{\teaser}[1]{\gdef\my@teaser{#1}}
\renewcommand{\abstract}[1]{\gdef\my@abstract{#1}}
\renewcommand{\@maketitle}{%
  \begin{center}%
    {\LARGE\@title\par}%
    \vspace{1em}%
    \begin{tabular}[t]{c}\@author\end{tabular}\par
    \vspace{1em}%
    {\begingroup\def\@captype{figure}%
     \begin{minipage}{.8\linewidth}\my@teaser\end{minipage}%
     \endgroup\par}%
  \end{center}\par
  {\small\textbf{Abstract}---\my@abstract\par}}
\makeatother
\title{Teaser Title}
\author{Ann Alpha}
\abstract{Abstract text.}
\teaser{\centering\rule{4cm}{1cm}\caption{The teaser.}\label{fig:teaser}}
\begin{document}
\maketitle
\section{Introduction}
See Figure~\ref{fig:teaser}.
\end{document}
