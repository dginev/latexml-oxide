\begin{filecontents*}{imsart.sty}
\NeedsTeXFormat{LaTeX2e}\ProvidesFile{imsart.sty}[2012/01/01 IMS article style, shortened and modified]
\RequirePackage{keyval}
\newif\if@addressatend \@addressatendtrue
\define@key{ead}{email}[true]{\def\ead@type{email}}
\define@key{ead}{url}[true]{\def\ead@type{url}}
\define@key{ead}{label}{\def\ead@label{#1}}
\def\ead@url{url}
\DeclareRobustCommand\ead[2][label= ,email]{{\def\ead@type{email}\setkeys{ead}{#1}%
  \expandafter\gdef\csname ead@a@\ead@label\endcsname{#2}%
  \expandafter\global\expandafter\let\csname ead@t@\ead@label\endcsname\ead@type}}
\def\printead#1{\expandafter\ifx\csname ead@t@#1\endcsname\ead@url url: \else e-mail: \fi
  {\ttfamily\csname ead@a@#1\endcsname}}
\newenvironment{aug}{}{\par}
\newenvironment{frontmatter}{\begin{center}}{\end{center}}
\def\author{\@ifnextchar[{\author@fmt}{\author@fmt[]}}
\def\author@fmt[#1]#2{\bgroup\def\fnms##1{##1}\def\snm##1{##1}\noindent#2\egroup}
\newcount\c@address
\def\address{\@ifnextchar[{\address@fmt}{\address@fmt[default]}}
\def\address@fmt[#1]#2{\global\advance\c@address1
  \expandafter\gdef\csname @address@\the\c@address\endcsname{#2}}
\newcount\addr@n
\def\printaddresses{\addr@n=0
  \loop\ifnum\addr@n<\c@address\global\advance\addr@n1
    \par\noindent\begin{tabular}[t]{@{}l@{}}\csname @address@\the\addr@n\endcsname\end{tabular}\repeat\par}
\let\old@enddocument\enddocument
\def\enddocument{\printaddresses\old@enddocument}
\end{filecontents*}
\documentclass{article}
\usepackage{imsart}
\begin{document}
\begin{frontmatter}
\title{A Title}
\begin{aug}
\author{\fnms{Ann} \snm{Able}\ead[label=e1]{ann@x.edu}\ead[label=u1,url]{http://www.zebra.edu/$\sim$ann}}
\end{aug}
\end{frontmatter}
\address{Ann Able\\ Univ A\\ \printead{e1}\\ \printead{u1}}
Body.
\end{document}
