\NeedsTeXFormat{LaTeX2e}\ProvidesClass{astlb}[2012/02/04 stand-in]
\LoadClass{article}
\def\email#1{\textsuperscript{*}\gdef\@email{#1}}
\def\@@email#1{\let\oldthefootnote\thefootnote\renewcommand\thefootnote{\relax}%
  \footnotetext{\hspace{-\parindent}\textsuperscript{*}e-mail: #1}\let\thefootnote\oldthefootnote
  \setcounter{footnote}{0}}
\def\author#1{\gdef\@author{#1}}
\renewcommand\thempfootnote{\arabic{mpfootnote}}
\def\@maketitle{\begin{center}\begin{minipage}{0.885\linewidth}%
  \def\address##1{\textsuperscript{##1}}%
  \def\addresstext##1##2{\footnotetext[##1]{\itshape ##2}}%
  \begin{center}{\large\bfseries\@title\par}{\bfseries\@author\par}\end{center}%
  \end{minipage}\end{center}}
\renewenvironment{abstract}{\par\textbf{Abstract ---}}{\par\maketitle\@@email{\@email}}
\end{filecontents*}
