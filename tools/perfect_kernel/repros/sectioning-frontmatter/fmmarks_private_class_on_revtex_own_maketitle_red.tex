\ProvidesClass{myrev}

\renewcommand\maketitle{\begingroup\renewcommand\thefootnote{\arabic{footnote}}%
 \def\thanks##1{\footnotemark\protected@xdef\my@thanks{\my@thanks\protect\footnotetext[\the\c@footnote]{##1}}}%
 \begin{center}\@myauthor\end{center}\my@thanks\endgroup}
\renewcommand\author[1]{\gdef\@myauthor{#1}}
\end{filecontents}
