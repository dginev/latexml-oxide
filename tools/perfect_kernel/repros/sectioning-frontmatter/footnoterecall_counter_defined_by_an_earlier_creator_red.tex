\documentclass[a4paper]{scrarticle}
\newcommand{\footnoteremember}[2]{
	\footnote{#2}
	\newcounter{#1}
	\setcounter{#1}{\value{footnote}}
}
\newcommand{\footnoterecall}[1]{
	\footnotemark[\value{#1}]
}
\begin{document}
\title{T}
\author{
 Ann One\footnoteremember{uni}{The University}
		\and  Bob Two \footnoterecall{uni}
  \and
	Cy Three\footnoterecall{uni}
}
\maketitle
Text.
\end{document}
