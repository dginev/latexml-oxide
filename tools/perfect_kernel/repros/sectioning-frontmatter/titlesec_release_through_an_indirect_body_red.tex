\documentclass{article}
\usepackage{titlesec}
\makeatletter
\newcommand\mysection{\@startsection{section}{1}{\z@}{-3.5ex}{2.3ex}{\normalfont\Large\bfseries}}
\renewcommand\section{\mysection}
\let\oldsubsection\subsection
\renewcommand\subsection{\@startsection{subsection}{2}{\z@}{-3ex}{1ex}{\normalfont\bfseries}}
\let\subsection\oldsubsection
\makeatother
\titleformat{\section}{\itshape}{[\thesection]}{1em}{}
\titleformat{\subsection}{\itshape}{[\thesubsection]}{1em}{}
\begin{document}
\section{Intro}
A.
\subsection{Next}

\end{document}
