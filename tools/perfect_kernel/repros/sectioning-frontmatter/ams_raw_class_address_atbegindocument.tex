\NeedsTeXFormat{LaTeX2e}\ProvidesClass{stcsatbegin}[2008/01/01 stand-in]
\LoadClass[11pt,reqno]{amsart}
\AtBeginDocument{\renewcommand{\address}[2][]{\g@addto@macro\addresses{\address{#1}{#2}}}}
\def\@setaddresses{\par\begingroup\def\address##1##2{##2\par}%
  \def\curraddr##1##2{Current address: ##2\par}\def\email##1##2{E-mail address: ##2\par}%
  \def\urladdr##1##2{URL: ##2\par}\addresses\endgroup}
\def\@maketitle{\@settitle\ifx\@empty\authors\else\@setauthors\fi\@setaddresses}
\def\enddoc@text{}
\end{filecontents*}
