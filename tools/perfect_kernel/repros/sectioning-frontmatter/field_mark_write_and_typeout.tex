% expect:  0 errors, 0 warnings; "Read back: Written Title Detok: Written Title."; no mark name in the log or the XML.

\makeatletter
\title{Written Title}
\begin{document}
\newwrite\tout

\immediate\write\tout{\@title}
\immediate\closeout\tout
\typeout{TITLE=\@title}
\message{\meaning\@title}
Read back: \input{\jobname.ttl} Detok: \detokenize\expandafter{\@title}.

Text.
