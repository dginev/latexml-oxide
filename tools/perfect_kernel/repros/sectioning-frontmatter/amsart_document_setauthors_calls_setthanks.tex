\documentclass{amsart}
\makeatletter
\def\@setauthors{%
  \begingroup
    \let\MakeUppercase\relax
    \begin{center}\normalfont\normalsize\authors\end{center}%
  \endgroup
  \@setthanks
}
\makeatother
\begin{document}
\title{A Title}
\author{Ann Able}
\email{ann@x.edu}
\author{Bob Baker}
\thanks{Supported by grant 42.}
\maketitle
Body.
\end{document}
