\begin{filecontents*}{wrapaddr.sty}
% A minimal stand-in for epigamath.sty:227-232 and MyCommA.sty:530-533, written for a latexml-oxide test.
\ProvidesPackage{wrapaddr}[2022/01/01 stand-in]
\def\displayaddresses{}
\let\@pigaaddress\address
\renewcommand{\address}[1]{\g@addto@macro{\displayaddresses}{\par\noindent#1}\@pigaaddress{#1}}
\let\orgemail=\email
\renewcommand{\email}[1]{\orgemail{\href{mailto:#1}{#1}}}
\end{filecontents*}
