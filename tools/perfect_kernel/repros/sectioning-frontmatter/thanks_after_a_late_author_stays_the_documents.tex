% restored the kernel `\thanks` for the rest of the document, and the note moved into the title block
% as a `<pubnote>`.
% witness: none (review of 59e, probe p4).

% oracle:  pdflatex 0 errors: "Text. More text[Bracket note] after."

\begin{filecontents*}[overwrite]{lxaddr.cls}
\NeedsTeXFormat{LaTeX2e}
\ProvidesClass{lxaddr}
\LoadClass{article}
\def\address#1{\gdef\@address{#1}}
\def\email#1{\gdef\@email{#1}}
\let\@address\@empty
\renewcommand\maketitle{\par{\centering\@title\par\@author\par\@address\par\@email\par}}
\end{filecontents*}
\documentclass{lxaddr}
\begin{document}
\title{T0}\author{Max M}\maketitle
Text.
\def\thanks#1{[#1]}
\author{Bob B}
More text\thanks{Bracket note} after.
\section{S}
Body.
\end{document}
