\begin{filecontents*}[overwrite]{nmadr.cls}
\ProvidesClass{nmadr}\LoadClass{article}
\newcounter{adr}
\def\address#1{\gdef\@address{\it #1}}
\def\printaddress{\begingroup\setcounter{adr}{1}%
  \def\\{\par\stepcounter{adr}\noindent$^{\theadr}$\enspace}%
  \noindent$^{1}$\enspace\@address\par\endgroup}
\def\@maketitle{\begin{center}{\LARGE\@title\par}\vskip1em{\large\@author\par}\end{center}
  {\small\printaddress}\vskip1em}
