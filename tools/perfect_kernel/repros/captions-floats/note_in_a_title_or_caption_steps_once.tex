% the caption (`toccaption`) digests the argument again, so the note steps its counter twice and every
% later note's number shifts (here A is 3 and B is 6 for pdflatex's 2 and 4); the caption's note also
% appears twice. The same with endnotes.sty's and enotez's `\endnote`.

% expect:  0 errors, 0 warnings.
\documentclass{article}
\begin{document}
\section{One\protect\footnote{In title.}}
A\footnote{After.}
\begin{table}\caption{Cap\protect\footnote{In caption.}}\end{table}

\end{document}
