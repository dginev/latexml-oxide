74/// Derive the STRING form of a stored value (page `<title>`, `title=` tooltip).
75/// Perl `CrossRef::getTextContent` (`CrossRef.pm` L853-859). A `Value::Xml` title
76/// is flattened tag-aware and math-aware (via [`title_text_content`], which routes
77/// `ltx:Math` through `unicodemath`); any other value uses its plain string form.
78/// Either way the result is whitespace-collapsed like Perl: trim both ends, then
79/// `s/\s+/ /g` — so a multi-line math serialization cannot bloat a `title=`
80/// tooltip (issue #761).
81fn value_text(doc: &PostDocument, val: &Value) -> String {
82  let raw = match val {
83    Value::Xml(node) => title_text_content(doc, node),
84    other => other.to_string(),
85  };
86  // Perl `getTextContent`: `s/^\s+//; s/\s+$//; s/\s+/ /g`. `split_whitespace`
87  // does exactly this (drops leading/trailing runs, collapses interior runs).
88  raw.split_whitespace().collect::<Vec<_>>().join(" ")
89}

91/// Build the child nodes of an `<ltx:ref>` from a stored value, as stored.
92///
93/// Port of Perl `CrossRef::prepRefText` = `cloneNodes(trimChildNodes($value))`
94/// without its `fillInTitle`, which [`CrossRef::ref_text_children`] adds:
95/// deep-clone the title's child nodes — `<ltx:Math>` included — trimming
96/// whitespace at the two edges. Element children become [`NodeData::XmlNode`]
97/// (deep-copied at materialization by `PostDocument::add_xml_node`, which
98/// uniquifies their `xml:id`s); text children become [`NodeData::Text`]. A
99/// plain-string value keeps the single flat-text child.
100fn ref_content_children(val: &Value) -> Vec<NodeData> {
101  match val {
102    Value::Xml(node) => trim_child_data(child_nodes_data(node)),
103    other => vec![NodeData::Text(other.to_string())],
104  }
105}

107/// Perl `Post::Document::trimChildNodes` on ref content: left-trim the first text child, right-trim the last; drop
108/// either if it becomes empty.
109fn trim_child_data(mut out: Vec<NodeData>) -> Vec<NodeData> {
110  if let Some(NodeData::Text(s)) = out.first_mut() {
111    let t = s.trim_start().to_string();
112    if t.is_empty() {
113      out.remove(0);
114    } else {
115      *s = t;
116    }
117  }
118  if let Some(NodeData::Text(s)) = out.last_mut() {
119    let t = s.trim_end().to_string();
120    if t.is_empty() {
121      out.pop();
122    } else {
123      *s = t;
124    }
125  }
126  out
127}

129/// Strip `fragid` from everything inside an `<ltx:ref>` (TOC entries, inline
130/// refs, navigation).
131///
132/// Reference content is a non-anchor DISPLAY copy of a title, so it must not
133/// carry a `fragid` — the XSLT `add_id` template emits the HTML `id` from
134/// `fragid`, and a display copy with an `id` would spuriously duplicate the
135/// real target's anchor. Perl gets this for free: its ref content is cloned
136/// from Scan's `cleanNode` snapshot, taken before `fragid` is assigned
137/// (`Scan.pm` L290 / `CrossRef.pm` fillInFrags). We clone the live (already
138/// `fragid`'d) title, so we drop `fragid` here to match (issue #356). The
139/// uniquified `xml:id` is kept, as in Perl's snapshot.
140fn strip_ref_display_fragids(doc: &PostDocument) {
141  for mut n in doc.findnodes("//ltx:ref//*[@fragid]") {
142    let _ = n.remove_attribute("fragid");
143  }
144}

146/// URL style for cross-references (Perl `--urlstyle`; `Config.pm` accepts
147/// `server`, `negotiated`, `file`). Selects how [`CrossRef`]'s URL generation
148/// rewrites a generated cross-reference URL for the serving environment.
149#[derive(Debug, Clone, Copy, PartialEq, Eq)]
150pub enum UrlStyle {
151  /// Keep the full `file.html#fragment` — nothing stripped. Correct for
152  /// `file://` viewing and servers that do not rewrite `index.html`.
153  File,
154  /// Strip a trailing `index.ext` so a directory index links as `dir/`
155  /// (Perl's `latexml` default; for servers that serve `dir/` → `dir/index.ext`).
156  Server,
157  /// Content negotiation: strip the `.ext` extension AND a trailing `index`
158  /// (for servers that hide the extension, e.g. BookML's `--urlstyle=negotiated`).
159  Negotiated,
160}
161
162impl UrlStyle {
163  /// Parse a CLI `--urlstyle` value. Returns `None` for an unrecognized value
164  /// (the caller reports it, mirroring Perl `_checkOptionValue`).
165  pub fn from_cli(s: &str) -> Option<Self> {
166    match s {
167      "file" => Some(UrlStyle::File),
168      "server" => Some(UrlStyle::Server),
169      "negotiated" => Some(UrlStyle::Negotiated),
170      _ => None,
171    }
172  }
173
174  /// The canonical CLI tag — round-trips through [`UrlStyle::from_cli`]. Used to
175  /// serialize the style across the parallel page-render worker manifest.
176  pub fn as_cli(self) -> &'static str {
177    match self {
178      UrlStyle::File => "file",
179      UrlStyle::Server => "server",
180      UrlStyle::Negotiated => "negotiated",
181    }
182  }
183}

185/// Rewrite a generated cross-reference `url` for the given [`UrlStyle`], mirroring
186/// Perl `CrossRef::generateURL` (CrossRef.pm L656-663) verbatim — including its
187/// `(^|\/)` path boundary: a trailing `index[.ext]` is stripped only at the very
188/// start of the URL or right after a `/`, so a filename like `myindex.html` is
189/// left intact. `extension` is the output file extension (e.g. `html`).
190fn apply_url_style(url: &str, style: UrlStyle, extension: &str) -> String {
191  match style {
192    // Perl: s/(^|\/)index.\Q$ext\E$/($1 ? $1 : '.\/')/e
193    UrlStyle::Server => {
194      let index_suffix = format!("index.{extension}");
195      if let Some(prefix) = url.strip_suffix(&index_suffix) {
196        if prefix.is_empty() {
197          return "./".to_string(); // matched at start ($1 empty → './')
198        } else if prefix.ends_with('/') {
199          return prefix.to_string(); // matched after '/' ($1 = '/', kept)
200        } // else: no path boundary before `index` → leave url unchanged
201      }
202      url.to_string()
203    },
204    // Perl: s/\.\Q$ext\E$// then s/(^|\/)index$/$1/
205    UrlStyle::Negotiated => {
206      let stripped = url.strip_suffix(&format!(".{extension}")).unwrap_or(url);
207      if stripped == "index" {
208        String::new() // matched at start ($1 empty)
209      } else if let Some(prefix) = stripped.strip_suffix("index") {
210        if prefix.ends_with('/') {
211          prefix.to_string() // matched after '/' ($1 = '/', kept)
212        } else {
213          stripped.to_string() // no path boundary → leave (e.g. `myindex`)
214        }
215      } else {
216        stripped.to_string()
217      }
218    },
219    UrlStyle::File => url.to_string(),
220  }
221}

289impl BibciteDatum {
290  /// The entry's number, which a sorting citation orders by: its `number` (else `refnum`) when that is a whole
291  /// number, as natbib's `\NAT@ifcat@num` and cite.sty's `\@cite@posnumtest` take it; none for a missing entry.

293    let text = self.sort_label()?;

297  }

306      return None;
307    }
308    let label = if self.number.is_empty() {
309      &self.refnum

317/// cite.sty's sort number of a citation label (cite.sty:127-222 `\@addto@cite@list`): a number `n` is `256n`; a
318/// `[prefix]number[suffix]` label, one character each, `(16384p + n)·256 + s` (the characters' codes, `p` zero
319/// without a prefix: "2a" sorts after 2, before 3); a number, a separator and a number (`1.2`) `((16384p + n)·256 +
320/// sep)·4096 + m`, while the part before `m` is below 262144. Any other label (`Knu84`) has none: cite.sty prints it
321/// at once, before the sorted ones (`\@cite@gobbledump@now`).
322fn cite_sty_sort_key(label: &str) -> Option<u64> {
323  let chars: Vec<char> = label.chars().collect();
324  if chars.is_empty() {
325    return None;
326  }
327  if chars.iter().all(char::is_ascii_digit) {
328    return label.parse::<u64>().ok().map(|n| n * 256);
329  }
330  let (prefix, rest) = if chars[0].is_ascii_digit() {
331    (0, &chars[..])
332  } else {
333    (u64::from(u32::from(chars[0])), &chars[1..])
334  };
335  let digits = rest.iter().take_while(|c| c.is_ascii_digit()).count();
336  if digits == 0 {
337    return None;
338  }
339  let number: u64 = rest[..digits].iter().collect::<String>().parse().ok()?;
340  let mut key = (prefix * 16384 + number) * 256;
341  let Some((mark, tail)) = rest[digits..].split_first() else {
342    return Some(key);
343  };
344  key += u64::from(u32::from(*mark));
345  if tail.is_empty() {
346    return Some(key);
347  }
348  if tail.iter().all(char::is_ascii_digit) && key < 262144 {
349    let last: u64 = tail.iter().collect::<String>().parse().ok()?;
350    return Some(key * 4096 + last);
351  }
352  None
353}

375/// A bibliography entry's text, inline (show `FullEntry`), as biblatex's `\fullcite` typesets the entry's driver
376/// where it cites (biblatex.def:2481-2495): the `ltx:bibblock`s of its `ltx:bibitem` but the "Cited by" one, one space
377/// apart. `dropped_year` is the year an author-year entry's first block left to its label (MakeBibliography
378/// `drop_first_year`); it comes back after the first block with content (the author's, else the title's), as a numeric
379/// entry's block carries it. The closing period
380/// goes: `\usedriver` lets `\finentry` set no punctuation (biblatex.sty:4135, :4245 `\blx@finentry@usedrv`);
381/// MakeBibliography sets it as a text node `.` of its own (field content is always in an `ltx:text`), and only such
382/// a node is dropped. Empty for a bibitem with no blocks.
383fn full_entry(bibitem: &Node, dropped_year: Option<&str>) -> Vec<NodeData> {
384  let mut text = Vec::new();
385  let blocks = crate::document::element_children(bibitem)
386    .into_iter()
387    .filter(|block| {
388      block.get_name() == "bibblock"
389        && !block
390          .get_attribute("class")
391          .is_some_and(|class| class.split_whitespace().any(|c| c == "ltx_bib_cited"))
392    });
393  let mut dropped_year = dropped_year;
394  for (i, block) in blocks.enumerate() {
395    if i > 0 {
396      text.push(NodeData::Text(" ".to_string()));
397    }
398    text.extend(child_nodes_data(&block));
399    // After the first block with content, as MakeBibliography left it out (`format_blocks`).
400    if !crate::document::element_children(&block).is_empty()
401      && let Some(year) = dropped_year.take()
402    {
403      text.push(NodeData::Element {
404        tag:        "ltx:text".to_string(),
405        attributes: Some(HashMap::from_iter([(
406          "class".to_string(),
407          "ltx_bib_year".to_string(),
408        )])),
409        children:   vec![NodeData::Text(year.to_string())],
410      });
411    }
412  }
413  if matches!(text.last(), Some(NodeData::Text(last)) if last.trim() == ".") {
414    text.pop();
415  }
416  text
417}

419/// A node's children as insertable data, untrimmed: text by value, elements by
420/// reference (deep-copied on insertion). Perl hands `addNodes` the live
421/// `childNodes`, which it copies the same way.
422fn child_nodes_data(node: &Node) -> Vec<NodeData> {
423  let mut out: Vec<NodeData> = Vec::new();
424  let mut child = node.get_first_child();
425  while let Some(c) = child {
426    match c.get_type() {
427      Some(NodeType::TextNode) => out.push(NodeData::Text(c.get_content())),
428      Some(NodeType::ElementNode) => out.push(NodeData::XmlNode(c.clone())),
429      _ => {},
430    }
431    child = c.get_next_sibling();
432  }
433  out
434}

436/// Whether a stored title holds what Perl `CrossRef::fillInTitle` (CrossRef.pm:882-904) replaces before the title
437/// is reused: an `ltx:ref` never filled — Scan stored the title before CrossRef ran, so a `\ref`, `\eqref`,
438/// `\autoref`, `\cref` or `\nameref` in a section title is still empty there (1011.3492, 1111.3672) — or an
439/// `ltx:break`. Math is not entered: its copy keeps the ids its own references point to.
440fn needs_fill_in(node: &Node) -> bool {
441  let mut child = node.get_first_child();
442  while let Some(c) = child {
443    if c.get_type() == Some(NodeType::ElementNode) {
444      match c.get_name().as_str() {
445        "ref" if is_unfilled_ref(&c) => return true,
446        "break" => return true,
447        "Math" => {},
448        _ if needs_fill_in(&c) => return true,
449        _ => {},
450      }
451    }
452    child = c.get_next_sibling();
453  }
454  false
455}

464/// The text of ref content as Perl `getTextContent` reads a filled title (`CrossRef.pm` L861-879): a tag with its
465/// `open`/`close`, math in its Unicode reading.
466fn node_data_text(doc: &PostDocument, data: &[NodeData]) -> String {
467  let mut out = String::new();
468  for datum in data {
469    match datum {
470      NodeData::Text(text) => out.push_str(text),
471      NodeData::XmlNode(node) => out.push_str(&title_node_text(doc, node)),
472      NodeData::Element { tag, attributes, children } => {
473        let attr = |key: &str| {
474          attributes
475            .as_ref()
476            .and_then(|attributes| attributes.get(key))
477            .map_or("", String::as_str)
478        };
479        let tagged = tag == "ltx:tag";
480        if tagged {
481          out.push_str(attr("open"));
482        }
483        out.push_str(&node_data_text(doc, children));
484        if tagged {
485          out.push_str(attr("close"));
486        }
487      },
488    }
489  }
490  out
491}

502/// Perl `make_bibcite`'s show walk (`CrossRef.pm` L564-643) for one cited
503/// entry, returning `(stuff, didref)`. The show string is a sequence of role
504/// words — lowercased with one trailing `s` stripped (L585), so `Authors`,
505/// `author` and `authors` are one role — among `{literal}` text, `~` (a
506/// no-break space), whitespace and other punctuation, which pass through.
507/// `author`/`fullauthor`/`title`/`refnum`/`phraseN` clone their values;
508/// `year`/`number`/`super` make the `<ltx:ref>` link themselves (`didref`),
509/// and absorb the following entries of the same authors when `checkdups`
510/// (`Smith (2001a, b)`). Without a link made inside, the caller wraps the
511/// whole label in one.
512fn bibcite_show_walk(
513  saveshow: &str,
514  datum: &BibciteDatum,
515  data: &mut VecDeque<BibciteDatum>,
516  preformatted: &[NodeData],
517  phrases: &[Node],
518  yysep: &str,
519  checkdups: bool,
520) -> (Vec<NodeData>, bool) {
521  static YEAR_DELIM: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)(\w)year").unwrap());
522  static PHRASE_DELIM: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)(\w)phrase").unwrap());
523  static PHRASE_NUM_DELIM: LazyLock<Regex> =
524    LazyLock::new(|| Regex::new(r"(?i)phrase(\d)(\w)").unwrap());
525  static ROLE: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"^\w+").unwrap());
526  static LITERAL: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"^\{([^}]*)\}").unwrap());
527  static SPACES: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"^\s+").unwrap());
528  static PUNCT: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"^\W+").unwrap());
529  static EMPTY_YEAR_PHRASE2: LazyLock<Regex> =
530    LazyLock::new(|| Regex::new(r"(?i)^\{\}year\{\}phrase2").unwrap());
531
532  let mut didref = false;
533  let mut stuff: Vec<NodeData> = Vec::new();
534  let mut show = saveshow;
535  if show == "none" && !preformatted.is_empty() {
536    stuff = preformatted.to_vec();
537    show = "";
538  } else if datum.missing {
539    stuff = vec![bib_ref(&datum.attr, vec![NodeData::Text(
540      datum.key.clone(),
541    )])];
542    didref = true;
543    show = "";
544  }
545  // Add delimiters for parsing (L579-582): `Phrase1Year` → `Phrase1{}year`.
546  let show = YEAR_DELIM.replace_all(show, "${1}{}year");
547  let show = PHRASE_DELIM.replace_all(&show, "${1}{}phrase");
548  let show = PHRASE_NUM_DELIM.replace_all(&show, "phrase${1}{}${2}");
549  let mut rest: &str = &show;
550  // The same-author run after `datum` (L610, L620, L627): the entries
551  // `year`/`number`/`super` absorb.
552  let same_authors = |data: &VecDeque<BibciteDatum>| {
553    data
554      .front()
555      .is_some_and(|n| n.authortext == datum.authortext)
556  };
557  while !rest.is_empty() {
558    if let Some(m) = ROLE.find(rest) {
559      let mut role = m.as_str().to_lowercase();
560      rest = &rest[m.end()..];
561      if role.ends_with('s') {
562        role.pop();
563      }
564      match role.as_str() {
565        "author" => stuff.extend(datum.authors.iter().cloned()),
566        "fullauthor" => stuff.extend(datum.fullauthors.iter().cloned()),
567        "title" => stuff.extend(datum.title.iter().cloned()),
568        "refnum" => stuff.extend(datum.refnum.iter().cloned()),
569        // An entry with no text to copy (no blocks) links its number instead, as `refnum` would.
570        "fullentry" if datum.fullentry.is_empty() => {
571          stuff.push(bib_ref(&datum.attr, datum.refnum.clone()));
572          didref = true;
573        },
574        "fullentry" => {
575          stuff.extend(datum.fullentry.iter().cloned());
576          didref = true;
577        },
578        "year" => {
579          // (L605-606's "Date for citation" warning is unreachable: the year
580          // list is always defined, possibly empty.)
581          if !datum.year.is_empty() {
582            stuff.push(bib_ref(&datum.attr, datum.year.clone()));
583            didref = true;
584            while checkdups && same_authors(data) {
585              let next = data.pop_front().unwrap();
586              stuff.push(NodeData::Text(yysep.to_string()));
587              stuff.push(NodeData::Text(" ".to_string()));
588              let same_year = datum.rawyear.as_deref().unwrap_or("no_year_1")
589                == next.rawyear.as_deref().unwrap_or("no_year_2");
590              match next.suffix {
591                Some(ref suffix) if same_year => {
592                  stuff.push(bib_ref(&next.attr, vec![NodeData::Text(suffix.clone())]))
593                },
594                _ => stuff.push(bib_ref(&next.attr, next.year.clone())),
595              }
596            }
597          }
598        },
599        "number" | "super" => {
600          let mut r = vec![bib_ref(&datum.attr, datum.number.clone())];
601          didref = true;
602          while checkdups && same_authors(data) {
603            let next = data.pop_front().unwrap();
604            r.push(NodeData::Text(yysep.to_string()));
605            r.push(NodeData::Text(" ".to_string()));
606            r.push(bib_ref(&next.attr, next.number.clone()));
607          }
608          if role == "super" {
609            stuff.push(NodeData::Element {
610              tag:        "ltx:sup".to_string(),
611              attributes: None,
612              children:   r,
613            });
614          } else {
615            stuff.extend(r);
616          }
617        },
618        _ => match role.strip_prefix("phrase").and_then(|d| {
619          let mut ds = d.chars();
620          ds.next()
621            .and_then(|c| c.to_digit(10))
622            .filter(|_| ds.next().is_none())
623        }) {
624          // L594-603. HACK (Perl's): an author-year show frozen before the
625          // entry had a year drops the empty `( )` of `Phrase1{}Year{}Phrase2`.
626          Some(n) => {
627            let short = |i: usize| {
628              phrases
629                .get(i)
630                .is_none_or(|p| p.get_content().chars().count() <= 1)
631            };
632            if n == 1
633              && datum.year.is_empty()
634              && short(0)
635              && short(1)
636              && let Some(m) = EMPTY_YEAR_PHRASE2.find(rest)
637            {
638              rest = &rest[m.end()..];
639            } else {
640              // `$phrases[$n - 1]`: `phrase0` is Perl's index -1, the last.
641              let phrase = match n {
642                0 => phrases.last(),
643                n => phrases.get(n as usize - 1),
644              };
645              if let Some(phrase) = phrase {
646                stuff.extend(child_nodes_data(phrase));
647              }
648            }
649          },
650          None => Info!("unexpected", role, "CITE ignoring show key '{}'", role),
651        },
652      }
653    } else if let Some(c) = LITERAL.captures(rest) {
654      // Pass-thru literal, quoted with {}.
655      if !c[1].is_empty() {
656        stuff.push(NodeData::Text(c[1].to_string()));
657      }
658      rest = &rest[c[0].len()..];
659    } else if let Some(r) = rest.strip_prefix('~') {
660      if !stuff.is_empty() {
661        stuff.push(NodeData::Text("\u{A0}".to_string()));
662      }
663      rest = r;
664    } else if let Some(m) = SPACES.find(rest) {
665      if !stuff.is_empty() {
666        stuff.push(NodeData::Text(m.as_str().to_string()));
667      }
668      rest = &rest[m.end()..];
669    } else if let Some(m) = PUNCT.find(rest) {
670      stuff.push(NodeData::Text(m.as_str().to_string()));
671      rest = &rest[m.end()..];
672    } else {
673      // Unreachable: every character is `\w` or `\W`.
674      break;
675    }
676  }
677  (stuff, didref)
678}

754  /// Generate a title string for a referenced ID, traversing parents for context.
755  ///
756  /// Port of `CrossRef::generateTitle`.
757  fn generate_title(&self, doc: &PostDocument, id: &str, shown: &str) -> Option<String> {
758    let mut current_id = id.to_string();
759    let mut result = String::new();
760    let mut prefix = String::new();
761    let mut shown_so_far = shown.to_string();
762
763    while let Some(entry) = self.db.lookup(&format!("ID:{}", current_id)) {
764      let mut pieces = Vec::new();
765      let mut is_dup = false;
766
767      // Try title, then typerefnum, then refnum
768      if let Some(title_val) = entry.get_value("title") {
769        if title_val.is_truthy() {
770          is_dup = shown_so_far.contains("title");
771          // The title is stored as a NODE (`Value::Xml`) for sections; derive
772          // its string form tag-aware (Perl `getTextContent`). A plain-string
773          // title (e.g. abstract/bibliography names) is used verbatim.
774          pieces.push(self.filled_value_text(doc, &current_id, title_val));
775        }
776      }
777      if pieces.is_empty() {
778        let has_type = entry
779          .get_value("tag:creftypecap")
780          .or_else(|| entry.get_value("tag:creftype"));
781        let has_refnum = entry.get_value("refnum");
782        // (cleveref's `creflabel` is the refnum in its label format: OXIDIZED_DESIGN_DIVERGENCES #845)
783        let shows_refnum = shown_so_far.contains("refnum") || shown_so_far.contains("creflabel");
784        if has_type.is_some() && has_refnum.is_some() {
785          is_dup = shown_so_far.contains("type") && shows_refnum;
786          if let Some(t) = has_type {
787            pieces.push(tag_text(doc, t));
788          }
789          if let Some(r) = has_refnum {
790            pieces.push(tag_text(doc, r));
791          }
792        } else if let Some(tr) = entry.get_value("typerefnum") {
793          is_dup = shown_so_far.contains("type") && shows_refnum;
794          pieces.push(tag_text(doc, tr));
795        } else if let Some(r) = has_refnum {
796          is_dup = shows_refnum;
797          pieces.push(tag_text(doc, r));
798        }
799      }
800
801      if is_dup {
802        prefix = "In ".to_string();
803        shown_so_far.clear();
804      } else {
805        let title = pieces.join(" ");
806        let title = title.trim();
807        if !title.is_empty() {
808          result.push_str(&prefix);
809          prefix = self.ref_join.clone();
810          result.push_str(title);
811        }
812      }
813
814      // Walk to parent for more context
815      match entry.get_string("parent").map(String::from) {
816        Some(pid) => current_id = pid,
817        None => break,
818      }
819    }
820
821    if result.is_empty() {
822      None
823    } else {
824      Some(result)
825    }
826  }

831  fn generate_document_title(&self, doc: &PostDocument) -> Option<String> {
832    // Try to generate from the document's root ID. Use `get_xml_id` so we

854      }
855    }

860  ///
861  /// Port of `CrossRef::generateGlossaryRefTitle` (CrossRef.pm:906-925): the
862  /// entry's `phrase:<show>` — failing that, `<base>-plural` appends an `s` and
863  /// `<base>-indefinite` prefixes `a`/`an` to `phrase:<base>` — wrapped in an

869    };

907  }

1217          .chars()

1238  fn fill_in_relations(&mut self, doc: &mut PostDocument) {
1239    // Same get_xml_id trick as generate_document_title: Scan stores ids
1240    // in the xml namespace by default; without this, sub-docs would skip
1241    // relation filling and never gain the prev/next/up navigation.
1242    let page_id = match doc.get_document_element().as_ref().and_then(get_xml_id) {
1243      Some(id) => id,
1244      None => return,
1245    };
1246
1247    // 1. up / "up up" / "up up up" — walk ancestors that have a title.
1248    let mut current_id = page_id.clone();
1249    let mut rel = "up".to_string();
1250    let mut topmost = current_id.clone();
1251    loop {
1252      let parent_id = self
1253        .db
1254        .lookup(&format!("ID:{}", current_id))
1255        .and_then(|e| e.get_string("parent").map(String::from));
1256      match parent_id {
1257        Some(pid) => {
1258          let has_title = self
1259            .db
1260            .lookup(&format!("ID:{}", pid))
1261            .and_then(|e| e.get_value("title"))
1262            .map(|v| v.is_truthy())
1263            .unwrap_or(false);
1264          if has_title {
1265            doc.add_navigation(&rel, &pid);
1266            rel = format!("{} up", rel);
1267          }
1268          current_id = pid.clone();
1269          topmost = pid;
1270        },
1271        None => break,
1272      }
1273    }
1274
1275    // 2. start — the topmost ancestor (root page), if different from us.
1276    if topmost != page_id {
1277      if let Some(top_pageid) = self
1278        .db
1279        .lookup(&format!("ID:{}", topmost))
1280        .and_then(|e| e.get_string("pageid").map(String::from))
1281      {
1282        doc.add_navigation("start", &top_pageid);
1283      }
1284    }
1285
1286    // 3. prev / next — walk the page tree.
1287    if let Some(prev) = self.find_previous_page_id(&page_id) {
1288      doc.add_navigation("prev", &prev);
1289    }
1290    if let Some(next) = self.find_next_page_id(&page_id) {
1291      doc.add_navigation("next", &next);
1292    }

1295    // interesting related documents": the sibling pages of each ancestor (walking
1296    // up), then this page's own child pages. Each is keyed by the page's own
1297    // element-name relation (`chapter`/`section`/`subsection`/…) if it is a

1299    // `rel="chapter"`/`rel="section"`/… head links; the whole block was unported.
1300    let mut xentry = page_id.clone();

1304          continue;
1305        }
1306        self.add_typed_navigation(doc, sib);
1307      }
1308      xentry = parent;
1309    }
1310    for child in self.child_pages(&page_id).ids.iter() {
1311      self.add_typed_navigation(doc, child);
1312    }

1408  /// Page immediately preceding `page_id` in tree order, restricted to
1409  /// `primary` pages. Port of Perl `CrossRef::findPreviousPage`: previous
1410  /// sibling if any, drilled into rightmost descendant.
1411  fn find_previous_page_id(&self, page_id: &str) -> Option<String> {
1412    let parent_id = self.get_parent_page_id(page_id)?;
1413    let siblings = self.child_pages(&parent_id);
1414    // Our position among the parent's child pages (None = "broken database").

1417    // drop self, keep primaries, take the last one). If there is NONE, Perl's

1419    // itself (e.g. the first `\section` of a `\chapter` → the chapter page).

1422      .iter()
1423      .rev()
1424      .find(|s| self.is_primary_page(s))
1425    {
1426      Some(sib) => sib.clone(),
1427      None => return Some(parent_id),
1428    };

1430    loop {
1431      let kids = self.child_pages(&current);

1433        Some(deepest) => current = deepest.clone(),
1434        None => break,
1435      }
1436    }

1470  fn fill_in_tocs(&mut self, doc: &mut PostDocument) {
1471    // Perl Post.pm L946-948: Document::findnodes defaults the XPath
1472    // context to documentElement. oxide's `findnodes(None)` defaults to
1473    // the XML document node, where libxml2's `descendant::` axis evaluates
1474    // differently — `descendant::ltx:TOC` matches zero from the doc node

1476    // user's `\tableofcontents` placeholder is reachable.

1478      Some(root) => doc.findnodes_at("descendant::ltx:TOC[not(ltx:toclist)]", Some(&root)),
1479      None => Vec::new(),
1480    };
1481    for toc in &tocs {
1482      // Use the unified get_xml_id helper: Scan's `Document` fallback
1483      // assigns xml:id via the xml namespace, which is invisible to a
1484      // bare `get_attribute("xml:id")` lookup but is found by
1485      // `get_attribute_ns("id", XML_NS)` (which get_xml_id tries first).

1487        .get_document_element()
1488        .as_ref()
1489        .and_then(get_xml_id)

1493      // *page* hierarchy (parent → its pageid → …) and taking the root page's
1494      // `pageid`. Default scope (`current` or absent) keeps the current-page id,
1495      // so the inline `\tableofcontents` placeholder stays page-local.

1598    selfid: Option<&str>,

1602      None => return vec![],
1603    };

1606    // entry lives on the target page.

1610    };

1612      entry
1613        .get_children()
1614        .iter()
1615        .flat_map(|child_id| self.gen_toc(child_id, show, types, lists, localto, selfid))
1616        .collect()
1617    } else {

1619    };

1622    // gentoc L255-256: include this entry iff its type passes the `select`
1623    // filter (no `select` ⇒ unrestricted) AND its `inlist` shares a list with
1624    // the TOC's `lists`. This is what makes `\setcounter{tocdepth}` (#291) take
1625    // effect — the level filter rides on `select`.

1627    let in_toc = entry
1628      .get_value("inlist")

1630        Value::Hash(h) => lists.iter().any(|l| h.contains_key(*l)),

1632      })

1642  /// Perl CrossRef.pm gentocentry L268-283. Build one `ltx:tocentry` for an
1643  /// entry: the `before < show > after` split (`generateRef_simple` for the
1644  /// before/after halves), the `ltx:ref` body, the `ltx_ref_self` marker when
1645  /// this is the `selfid`, and a nested `ltx:toclist` of `children`.
1646  fn gen_tocentry(
1647    &self,
1648    entry: &Entry,
1649    selfid: Option<&str>,
1650    show: &str,
1651    children: Vec<NodeData>,
1652  ) -> NodeData {
1653    let id = entry
1654      .get_string("id")
1655      .or_else(|| entry.get_key().strip_prefix("ID:"))
1656      .unwrap_or("")
1657      .to_string();
1658    let entry_type = entry.get_string("type").unwrap_or("");
1659    let type_name = entry_type.strip_prefix("ltx:").unwrap_or(entry_type);
1660

1663    let mut show_mid = show;
1664    if let Some((b, rest)) = show_mid.split_once('<') {
1665      before = Some(b);
1666      show_mid = rest;
1667    }
1668    if let Some((mid, a)) = show_mid.split_once('>') {
1669      show_mid = mid;
1670      after = Some(a);
1671    }

1674      " ltx_ref_self"
1675    } else {

1684      tag:        "ltx:ref".to_string(),

1686        ("show".to_string(), show_mid.to_string()),
1687        ("idref".to_string(), id.clone()),
1688      ])),
1689      children:   vec![],
1690    });
1691    if let Some(a) = after.filter(|a| !a.is_empty()) {
1692      kids.extend(self.generate_ref_simple(&id, a));
1693    }
1694    if !children.is_empty() {
1695      kids.push(NodeData::Element {
1696        tag: "ltx:toclist".to_string(),
1697        attributes: Some(HashMap::from_iter([(
1698          "class".to_string(),
1699          format!("ltx_toclist_{}", type_name),
1700        )])),
1701        children,
1702      });
1703    }
1704
1705    NodeData::Element {
1706      tag:        "ltx:tocentry".to_string(),
1707      attributes: Some(HashMap::from_iter([(
1708        "class".to_string(),
1709        format!("ltx_tocentry_{}{}", type_name, self_class),
1710      )])),
1711      children:   kids,
1712    }
1713  }

1728  /// Perl CrossRef.pm gentoc_context L288-311. A "context" TOC: the current
1729  /// page's own contents (downward, page-local), enclosed upward within its
1730  /// ancestors and their sibling sections — the navigation-bar breadcrumb.
1731  fn gen_toc_context(
1732    &self,
1733    id: &str,
1734    show: &str,
1735    types: Option<&HashSet<&str>>,
1736    lists: &HashSet<&str>,
1737  ) -> Vec<NodeData> {
1738    let start = match self.db.lookup(&format!("ID:{}", id)) {
1739      Some(e) => e,
1740      None => return vec![],
1741    };
1742
1743    // Downward TOC covering items WITHIN the current page (localto = this page's
1744    // location; selfid = this id so the current entry is marked ltx_ref_self).
1745    let location = start.get_string("location").unwrap_or("").to_string();
1746    let mut navtoc = self.gen_toc(id, show, types, lists, Some(&location), Some(id));
1747
1748    // Enclose it upward, along with siblings & ancestors. `came_from` is the id
1749    // of the child we ascended through; its slot in each parent's sibling row is
1750    // replaced by the accumulated `navtoc` subtree.
1751    let mut came_from = id.to_string();
1752    let mut parent_id = start.get_string("parent").map(String::from);
1753
1754    while let Some(pid) = parent_id {
1755      let parent = match self.db.lookup(&format!("ID:{}", pid)) {
1756        Some(e) => e,
1757        None => break,
1758      };
1759
1760      // gentoc_context L297-303: the parent's normal-type children become plain
1761      // tocentries, except the one we came from (spliced with `navtoc`).
1762      let mut row: Vec<NodeData> = Vec::new();
1763      for child_id in parent.get_children() {
1764        let child = match self.db.lookup(&format!("ID:{}", child_id)) {
1765          Some(e) => e,
1766          None => continue,
1767        };
1768        if !NORMAL_TOC_TYPES.contains(&child.get_string("type").unwrap_or("")) {
1769          continue;
1770        }
1771        let child_id_val = child.get_string("id").unwrap_or(&child_id);
1772        if child_id_val == came_from {
1773          row.append(&mut navtoc);
1774        } else {
1775          row.push(self.gen_tocentry(child, None, show, Vec::new()));
1776        }
1777      }
1778      navtoc = row;
1779
1780      // gentoc_context L304-306: wrap in the parent's own tocentry, but only if
1781      // the parent passes the type filter AND is itself nested (never wrap the
1782      // top-level document).
1783      let parent_type = parent.get_string("type").unwrap_or("");
1784      let parent_ok = types.map(|t| t.contains(parent_type)).unwrap_or(true);
1785      let parent_has_parent = parent
1786        .get_string("parent")
1787        .map(|s| !s.is_empty())
1788        .unwrap_or(false);
1789      if parent_ok && parent_has_parent {
1790        navtoc = vec![self.gen_tocentry(parent, None, show, navtoc)];
1791      }
1792
1793      came_from = pid;
1794      parent_id = parent.get_string("parent").map(String::from);
1795    }
1796
1797    navtoc
1798  }

1800  fn fill_in_frags(&self, doc: &PostDocument) {
1801    // Perl (CrossRef.pm L312-324) walks the page's own `//@xml:id` nodes and
1802    // sets `fragid` on any that have a DB entry. Iterating the ObjectDB
1803    // instead (one lookup per DB key) wins ONLY when a single page has far
1804    // more id-nodes than the DB has entries (math-heavy single documents:
1805    // ~60K XM* ids vs ~1K DB entries). On a *split* document the DB holds
1806    // every page's entries (tens of thousands) while each page has a handful
1807    // of id-nodes, so that inverted loop is O(db_keys) per page = O(n²)
1808    // overall. Pick whichever loop is bounded by the smaller set — both
1809    // assign fragids to exactly the same nodes, so the output is identical.
1810    if doc.idcache_len() <= self.db.len() {
1811      // Perl semantics: iterate the page's id-nodes (bounded by page size).
1812      for (id, node) in doc.idcache_iter() {
1813        if let Some(entry) = self.db.lookup(&format!("ID:{}", id)) {
1814          if let Some(fragid) = entry.get_string("fragid") {
1815            let mut n = node.clone();
1816            n.set_attribute("fragid", fragid).ok();
1817          }
1818        }
1819      }
1820    } else {
1821      // Inverted loop: fewer DB entries than page id-nodes. `find_node_by_id`
1822      // restricts to this page's nodes, so only page-local fragids are set.
1823      for key in self.db.keys_iter() {
1824        let id = match key.strip_prefix("ID:") {
1825          Some(rest) => rest,
1826          None => continue,
1827        };
1828        let entry = match self.db.lookup(key) {
1829          Some(e) => e,
1830          None => continue,
1831        };
1832        let fragid = match entry.get_string("fragid") {
1833          Some(f) => f,
1834          None => continue,
1835        };
1836        if let Some(node) = doc.find_node_by_id(id) {
1837          let mut n = node.clone();
1838          n.set_attribute("fragid", fragid).ok();
1839        }
1840      }
1841    }
1842  }

2031  /// into `about`, and `resourceidref`/`resourcelabelref` into `resource`.

2039  /// `LaTeXML-common.rnc` L301 note, "it will be converted to `aboutidref` and
2040  /// `about` during post-processing". Without this pass that conversion never
2041  /// happened, so `\lxRDFa{about=#thm1}` produced an `aboutidref` that no
2042  /// consumer reads and **no `about` at all** — the RDFa triple lost its subject.
2043  /// Visible on math once `outer_wrapper` began copying `about` onto `<m:math>`:
2044  /// Perl emits `about="#thm1"` there and Rust emitted nothing.
2045  ///
2046  /// An id that the ObjectDB knows becomes a real (possibly cross-page) URL via
2047  /// `generate_url`; an id it does not know still becomes a bare `#id` fragment,
2048  /// because — as Perl's comment puts it — "RDF 'id' need not be real, valid,
2049  /// ids!!!": an author may name a subject that is not a document node at all.
2050  ///
2051  /// Perl also re-runs `set_RDFa_prefixes` at the end. Not ported, and it cannot
2052  /// matter here: this pass only ever writes `about`/`resource` values that are
2053  /// absolute URLs or `#id` fragments, never prefixed CURIEs, so there is no new
2054  /// prefix to declare. (Prefix management itself already happens core-side, in
2055  /// `latexml_core::document::set_rdfa_prefixes`, as it does in Perl's

2066        // `aboutidref=""` counts as absent rather than resolving to `about="#"`.
2067        let mut id = ref_node
2068          .get_attribute(&idref_attr)
2069          .filter(|v| !v.is_empty());

2072        // written back so the `if let Some(id)` below treats both spellings
2073        // alike (Perl L379-387).
2074        if id.is_none()
2075          && let Some(label) = ref_node.get_attribute(&format!("{key}labelref"))
2076        {

2078            && let Some(resolved) = entry.get_string("id")

2634/// The `|`-separated alternatives of a ref's `show`, outside its `{…}` literals ([`CrossRef::generate_ref_aux`]).
2635fn show_alternatives(show: &str) -> Vec<&str> {
2636  let mut alternatives = Vec::new();
2637  let (mut depth, mut start) = (0usize, 0);
2638  for (at, c) in show.char_indices() {
2639    match c {
2640      '{' => depth += 1,
2641      '}' => depth = depth.saturating_sub(1),
2642      '|' if depth == 0 => {
2643        alternatives.push(&show[start..at]);
2644        start = at + 1;
2645      },
2646      _ => {},
2647    }
2648  }
2649  alternatives.push(&show[start..]);
2650  alternatives
2651}

2653/// How a citation's entry joins the one printed before it ([`CrossRef::make_bibcite`]).
2654#[derive(Clone, Copy, Debug, PartialEq, Eq)]
2655enum CiteJoint {
2656  /// After the separator (the first entry: alone).
2657  Separator,
2658  /// Inside a compressed range: not printed.
2659  Covered,
2660  /// The last of a compressed range: after an en dash.
2661  RangeEnd,
2662}
2663
2664/// The joints of a sorted citation's entries, by their numbers: each run of three or more consecutive numbers prints
2665/// its first and last joined by an en dash, natbib's `--` (natbib.sty:412-417) and cite.sty's `\citedash`
2666/// (cite.sty:288-290); a run of two keeps the separator, as both print "[1, 2]".
2667fn compressed_joints(numbers: &[Option<u64>], steps: &[u64]) -> Vec<CiteJoint> {
2668  let mut joints = vec![CiteJoint::Separator; numbers.len()];
2669  let follows = |at: usize, step: u64| matches!((numbers[at], numbers.get(at + 1).copied().flatten()), (Some(a), Some(b)) if b == a + step);
2670  let mut start = 0;
2671  while start < numbers.len() {
2672    let mut end = start;
2673    // (the run's step, the first its second number takes)
2674    if let Some(&step) = steps.iter().find(|&&step| follows(start, step)) {
2675      while end + 1 < numbers.len() && follows(end, step) {
2676        end += 1;
2677      }
2678    }
2679    if end >= start + 2 {
2680      joints[start + 1..end].fill(CiteJoint::Covered);
2681      joints[end] = CiteJoint::RangeEnd;
2682    }
2683    start = end + 1;
2684  }
2685  joints
2686}

2688/// Append the last entry of a compressed range after an en dash: inside the superscript the range's first entry
2689/// opened, for a superscript citation ("¹⁻³"), else as text.
2690fn append_range_end(refs: &mut Vec<NodeData>, mut item: Vec<NodeData>) {
2691  let is_sup = |node: &NodeData| matches!(node, NodeData::Element { tag, .. } if tag == "ltx:sup");
2692  if item.len() == 1
2693    && is_sup(&item[0])
2694    && let Some(NodeData::Element { children, .. }) = refs.last_mut().filter(|last| is_sup(last))
2695  {
2696    if let Some(NodeData::Element { children: inner, .. }) = item.pop() {
2697      children.push(NodeData::Text("\u{2013}".to_string()));
2698      children.extend(inner);
2699    }
2700    return;
2701  }
2702  refs.push(NodeData::Text("\u{2013}".to_string()));
2703  refs.extend(item);
2704}

2706/// The phrases-and-number part of a textual numeric `show` (`Authors Phrase1NumberPhrase2`, `FullAuthors …`): what
2707/// an entry with no name to print keeps of it ([`CrossRef::make_bibcite`]). `None` for any other form, which turns
2708/// the citation `refnum` as in Perl (CrossRef.pm:541).
2709fn nameless_numeric_show(show: &str) -> Option<String> {
2710  let lower = show.to_ascii_lowercase();
2711  let after_full = lower.strip_prefix("full").unwrap_or(&lower);
2712  let rest = after_full
2713    .strip_prefix("authors")
2714    .or_else(|| after_full.strip_prefix("author"))
2715    .map(|rest| show[show.len() - rest.len()..].trim_start())?;
2716  let rest_lower = rest.to_ascii_lowercase();
2717  (rest_lower.contains("number") && rest_lower.contains("phrase") && !rest_lower.contains("author"))
2718    .then(|| rest.to_string())
2719}
