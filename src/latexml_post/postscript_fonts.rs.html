1//! The fonts a MetaPost figure leaves to dvips.
2//!
3//! MetaPost writes PostScript with `prologues:=0` (its default) that is not self-contained: the text is
4//! `(S) cmmi10 9.96265 fshow`, the fonts are named by their TeX names, and the file only lists them in `%*Font:`
5//! header comments. dvips reads those comments when it includes the figure, downloads the fonts and supplies `fshow`
6//! (`finclude.pro`), which is how arXiv's PDF build renders them; Ghostscript alone stops at `/undefined in cmmi10`, so
7//! the figure is missing (Perl LaTeXML's ImageMagick path fails the same way). Before converting such a file we do
8//! what dvips does: a copy gets, in its prolog, each listed font's Type1 program (found through TeX Live's
9//! `pdftex.map`), the TeX name bound to the font, and MetaPost's own `fshow` (what it writes with `prologues:=1`).
10//! OXIDIZED_DESIGN #587; witnesses gr-qc0001079 (`foliation.ps`), math0102124 (`sch.13`).
11use std::{
12  collections::HashMap,
13  path::{Path, PathBuf},
14  sync::OnceLock,
15};
16
17/// The header comment dvips reads for a font a figure uses: `%*Font: <texname> <size> <designsize> <chars>`.
18const FONT_COMMENT: &str = "%*Font:";
19
20/// How much of a PostScript file is read to look for `%*Font:` comments.
21const HEADER_BYTES: u64 = 64 * 1024;
22
23/// MetaPost's `fshow` for `prologues:=1` output (string, font name, size).
24const FSHOW: &str = "/fshow {exch findfont exch scalefont setfont show} bind def\n";
25
26/// A TeX font's PostScript font name and its Type1 file, from `pdftex.map`.
27struct MappedFont {
28  ps_name: String,
29  file:    String,
30}
31
32/// `pdftex.map`'s plain entries (`cmmi10 CMMI10 <cmmi10.pfb`), read once. Entries that re-encode or slant a font are
33/// left out: their glyphs are not at the TeX positions MetaPost's strings use without the encoding applied.
34fn font_map() -> &'static HashMap<String, MappedFont> {
35  static MAP: OnceLock<HashMap<String, MappedFont>> = OnceLock::new();
36  MAP.get_or_init(|| {
37    latexml_core::util::pathname::kpsewhich(&["pdftex.map"])
38      .and_then(|path| std::fs::read_to_string(path).ok())
39      .map(|text| parse_font_map(&text))
40      .unwrap_or_default()
41  })
42}
43
44fn parse_font_map(text: &str) -> HashMap<String, MappedFont> {
45  let mut map = HashMap::new();
46  for line in text.lines() {
47    let line = line.trim();
48    if line.is_empty()
49      || line.starts_with(['%', '#'])
50      || line.contains('"')
51      || line.contains(".enc")
52    {
53      continue;
54    }
55    let mut fields = line.split_whitespace();
56    let (Some(tex_name), Some(ps_name)) = (fields.next(), fields.next()) else {
57      continue;
58    };
59    let file = fields
60      .filter_map(|f| {
61        f.trim_start_matches(['<', '['])
62          .strip_suffix(".pfb")
63          .map(|stem| format!("{stem}.pfb"))
64      })
65      .next();
66    if let Some(file) = file
67      && !ps_name.starts_with('<')
68    {
69      map.entry(tex_name.to_string()).or_insert(MappedFont {
70        ps_name: ps_name.to_string(),
71        file,
72      });
73    }
74  }
75  map
76}

78/// The TeX names in a PostScript file's `%*Font:` header comments, in order, each once.
79fn declared_fonts(header: &str) -> Vec<String> {
80  let mut names: Vec<String> = Vec::new();
81  for line in header.lines() {
82    if let Some(rest) = line.strip_prefix(FONT_COMMENT)
83      && let Some(name) = rest.split_whitespace().next()
84      && !names.iter().any(|n| n == name)
85    {
86      names.push(name.to_string());
87    }
88  }
89  names
90}

92/// A `.pfb` font program as the PFA text PostScript reads inline: ASCII segments as they are, binary (eexec)
93/// segments as hex (PFB segment headers: 0x80, type 1 ASCII / 2 binary / 3 end, 4-byte little-endian length).
94fn pfb_to_pfa(bytes: &[u8]) -> Option<String> {
95  let mut out = String::new();
96  let mut at = 0;
97  while at < bytes.len() {
98    if bytes[at] != 0x80 || at + 2 > bytes.len() {
99      return None;
100    }
101    let kind = bytes[at + 1];
102    if kind == 3 {
103      break;
104    }
105    let len_bytes = bytes.get(at + 2..at + 6)?;
106    let len = u32::from_le_bytes([len_bytes[0], len_bytes[1], len_bytes[2], len_bytes[3]]) as usize;
107    let segment = bytes.get(at + 6..at + 6 + len)?;
108    match kind {
109      1 => out.push_str(&String::from_utf8_lossy(segment).replace('\r', "\n")),
110      2 => {
111        for (i, b) in segment.iter().enumerate() {
112          out.push_str(&format!("{b:02x}"));
113          if i % 32 == 31 {
114            out.push('\n');
115          }
116        }
117        out.push('\n');
118      },
119      _ => return None,
120    }
121    at += 6 + len;
122  }
123  Some(out)
124}

126/// The prolog dvips would give a figure that uses `fonts`: each font's program, its TeX name bound to its PostScript
127/// name, and `fshow` unless the file defines its own. `None` when a font cannot be found — the figure would still fail.
128fn download_prolog(fonts: &[String], defines_fshow: bool) -> Option<String> {
129  let map = font_map();
130  let mut prolog = String::from("%%BeginProcSet: latexml-dvips-fonts\n");
131  for name in fonts {
132    let mapped = map.get(name)?;
133    let path = latexml_core::util::pathname::kpsewhich(&[mapped.file.as_str()])?;
134    prolog.push_str(&pfb_to_pfa(&std::fs::read(path).ok()?)?);
135    prolog.push_str(&format!("/{name} /{} def\n", mapped.ps_name));
136  }
137  if !defines_fshow {
138    prolog.push_str(FSHOW);
139  }
140  prolog.push_str("%%EndProcSet\n");
141  Some(prolog)
142}

144/// A converted figure's temporary copy, removed when dropped — on every return and on panic unwinding (the canvas
145/// `catch_unwind`). A `panic = "abort"` build (`maxperf`) or a process killed by a timeout runs no drop and leaves the
146/// copy behind.
147pub(crate) struct FontCopy(PathBuf);
148
149impl FontCopy {
150  pub(crate) fn path(&self) -> &Path { &self.0 }
151}
152
153impl Drop for FontCopy {
154  fn drop(&mut self) { let _ = std::fs::remove_file(&self.0); }
155}
156
157/// A copy of the PostScript file `source`, beside `dest`, holding the fonts its `%*Font:` comments ask dvips for —
158/// `None` when it declares none or one is not available. The caller converts the copy; dropping it removes it.
159pub(crate) fn with_downloaded_fonts(source: &str, dest: &str) -> Option<FontCopy> {
160  // Only the header is read for the check; a figure that declares fonts is read whole.
161  let mut head = Vec::new();
162  std::io::Read::read_to_end(
163    &mut std::io::Read::take(std::fs::File::open(source).ok()?, HEADER_BYTES),
164    &mut head,
165  )
166  .ok()?;
167  if !head.starts_with(b"%!")
168    || !head
169      .windows(FONT_COMMENT.len())
170      .any(|w| w == FONT_COMMENT.as_bytes())
171  {
172    return None;
173  }
174  let bytes = std::fs::read(source).ok()?;
175  // The header comments end at `%%EndProlog` or at the first line that is not a comment.
176  let mut header_end = 0;
177  let mut insert_at = None;
178  for line in bytes.split_inclusive(|b| *b == b'\n') {
179    if line.starts_with(b"%%EndProlog") {
180      insert_at = Some(header_end);
181      break;
182    }
183    if !line.starts_with(b"%") {
184      break;
185    }
186    header_end += line.len();
187  }
188  let insert_at = insert_at.unwrap_or(header_end);
189  let header = String::from_utf8_lossy(&bytes[..insert_at]);
190  let fonts = declared_fonts(&header);
191  if fonts.is_empty() {
192    return None;
193  }
194  let defines_fshow = bytes.windows(6).any(|w| w == b"/fshow");
195  let prolog = download_prolog(&fonts, defines_fshow)?;
196  let parent = Path::new(dest).parent().unwrap_or_else(|| Path::new("."));
197  let stem = Path::new(dest)
198    .file_stem()
199    .and_then(|s| s.to_str())
200    .unwrap_or("figure");
201  let unique = std::time::SystemTime::now()
202    .duration_since(std::time::UNIX_EPOCH)
203    .map(|d| d.as_nanos())
204    .unwrap_or(0);
205  let copy = parent.join(format!(".{stem}.{unique}.fonts.eps"));
206  let mut out = Vec::with_capacity(bytes.len() + prolog.len());
207  out.extend_from_slice(&bytes[..insert_at]);
208  out.extend_from_slice(prolog.as_bytes());
209  out.extend_from_slice(&bytes[insert_at..]);
210  let copy = FontCopy(copy);
211  std::fs::write(copy.path(), out).ok()?;
212  Some(copy)
213}

215#[cfg(test)]
216mod tests {
217  use super::*;

219  #[test]
220  fn declared_fonts_read_from_the_header_comments() {
221    let header = "%!PS\n%%BoundingBox: 0 0 10 10\n%*Font: cmmi10 9.96265 9.96265 53:a\n%*Font: cmr5 4.98 4.98 31:c\n\
222                  %*Font: cmmi10 6.97 9.96 1c:8\n";
223    assert_eq!(declared_fonts(header), vec!["cmmi10", "cmr5"]);
224  }
225
226  #[test]
227  fn plain_map_entries_only() {
228    let map = parse_font_map(
229      "cmmi10 CMMI10 <cmmi10.pfb\nptmr8r NimbusRomNo9L-Regu \" TeXBase1Encoding ReEncodeFont \" <8r.enc <utmr8a.pfb\n\
230       % comment\nmsam10 MSAM10 <msam10.pfb\n",
231    );
232    assert_eq!(map.len(), 2);
233    assert_eq!(map["cmmi10"].ps_name, "CMMI10");
234    assert_eq!(map["msam10"].file, "msam10.pfb");
235    assert!(!map.contains_key("ptmr8r"));
236  }
237
238  #[test]
239  fn pfb_segments_become_pfa_text() {
240    let mut pfb = vec![0x80, 1, 4, 0, 0, 0];
241    pfb.extend_from_slice(b"%!a\n");
242    pfb.extend_from_slice(&[0x80, 2, 2, 0, 0, 0, 0xab, 0x01]);
243    pfb.extend_from_slice(&[0x80, 3]);
244    assert_eq!(pfb_to_pfa(&pfb).as_deref(), Some("%!a\nab01\n"));
245  }
246  /// A MetaPost figure (`prologues:=0`, gr-qc0001079 `foliation.ps`) gets, before its `%%EndProlog`, the program of each
247  /// font its `%*Font:` comments list, its TeX name bound to the font, and `fshow`; the original stays untouched.
248  #[test]
249  fn metapost_figure_gets_the_fonts_dvips_would_download() {
250    let dir = std::env::temp_dir().join(format!("lx_mpfonts_{}", std::process::id()));
251    std::fs::create_dir_all(&dir).unwrap();
252    let source = dir.join("fig.1");
253    let figure = "%!PS\n%%BoundingBox: -1 -1 20 12\n%%Creator: MetaPost\n%%Pages: 1\n\
254                  %*Font: cmmi10 9.96265 9.96265 53:a\n%*Font: cmr5 4.98132 4.98132 31:c\n%%EndProlog\n%%Page: 1 1\n\
255                  0 0 moveto\n(S) cmmi10 9.96265 fshow\n(1) cmr5 4.98132 fshow\nshowpage\n%%EOF\n";
256    std::fs::write(&source, figure).unwrap();
257    let dest = dir.join("fig.png");
258    let copy = with_downloaded_fonts(source.to_str().unwrap(), dest.to_str().unwrap())
259      .expect("cmmi10 and cmr5 are found through pdftex.map");
260    let text = String::from_utf8_lossy(&std::fs::read(copy.path()).unwrap()).into_owned();
261    let prolog_at = text.find("%%BeginProcSet: latexml-dvips-fonts").unwrap();
262    let end_prolog_at = text.find("%%EndProlog").unwrap();
263    assert!(
264      text.starts_with("%!PS\n%%BoundingBox: -1 -1 20 12\n"),
265      "{text}"
266    );
267    assert!(prolog_at < end_prolog_at, "{text}");
268    assert_eq!(text.matches("currentfile eexec").count(), 2);
269    assert!(text.contains("/FontName /CMMI10 def"));
270    assert!(text.contains("/cmmi10 /CMMI10 def\n"));
271    assert!(text.contains("/cmr5 /CMR5 def\n"));
272    assert!(text.contains(FSHOW));
273    assert!(text.ends_with("(S) cmmi10 9.96265 fshow\n(1) cmr5 4.98132 fshow\nshowpage\n%%EOF\n"));
274    assert_eq!(std::fs::read_to_string(&source).unwrap(), figure);
275    let copy_path = copy.path().to_path_buf();
276    drop(copy);
277    assert!(!copy_path.exists());
278    // A figure with no `%*Font:` comments, or one that is not PostScript, is converted as it is.
279    std::fs::write(
280      &source,
281      "%!PS\n%%BoundingBox: 0 0 1 1\n%%EndProlog\nshowpage\n",
282    )
283    .unwrap();
284    assert!(with_downloaded_fonts(source.to_str().unwrap(), dest.to_str().unwrap()).is_none());
285    std::fs::write(
286      &source,
287      "\\documentstyle{article}\n%*Font: cmmi10 9.96 9.96 53:a\n",
288    )
289    .unwrap();
290    assert!(with_downloaded_fonts(source.to_str().unwrap(), dest.to_str().unwrap()).is_none());
291    let _ = std::fs::remove_dir_all(&dir);
292  }
