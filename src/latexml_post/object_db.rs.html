1//! Object database for cross-document data sharing.
2//!
3//! Port of `LaTeXML::Util::ObjectDB` + `ObjectDB::Entry`.
4//! A key-value store used by Scan, CrossRef, MakeIndex, and MakeBibliography
5//! to share structural information across documents and processing phases.
6//!
7//! Keys follow conventions:
8//! - `ID:<xml:id>` — element data (type, parent, children, labels, location, etc.)
9//! - `LABEL:<label>` — label → ID mapping
10//! - `DOCUMENT:<path>` — document location → root ID mapping
11//! - `SITE_ROOT` — root document of the site
12//! - `BIBLABEL:<list>:<key>` — bibliography key → item ID
13//! - `GLOSSARY:<list>:<key>` — glossary entries
14//! - `INDEX:<phrase1>:<phrase2>:...` — index entries
15//! - `DECLARATION:(global|local):<name>` — declared symbols
16//! - `NOTATION:<name>` — notation entries
17
18use libxml::tree::Node;
19use rustc_hash::FxHashMap as HashMap;

21/// A single entry in the ObjectDB.
22///
23/// Port of `LaTeXML::Util::ObjectDB::Entry`.
24#[derive(Debug, Clone)]
25pub struct Entry {
26  /// The key this entry is stored under.
27  pub key: String,
28  /// Attribute-value pairs.
29  values:  HashMap<String, Value>,
30}
31
32/// A value stored in an Entry.
33///
34/// Values can be scalars, lists, nested hashes, or XML node references.
35#[derive(Debug, Clone)]
36pub enum Value {
37  /// A simple string value.
38  String(String),
39  /// An integer value.
40  Int(i64),
41  /// A boolean value.
42  Bool(bool),
43  /// A list of values.
44  List(Vec<Value>),
45  /// A nested hash (for associations like referrers).
46  Hash(HashMap<String, Value>),
47  /// An XML node (cloned from the document).
48  Xml(Node),
49  /// Null/undefined.
50  Null,
51}
52
53impl Value {
54  /// Get as string, if possible.
55  pub fn as_str(&self) -> Option<&str> {
56    match self {
57      Value::String(s) => Some(s),
58      _ => None,
59    }
60  }
61
62  /// Get as string, converting if needed.
63  pub fn as_string(&self) -> String {
64    match self {
65      Value::String(s) => s.clone(),
66      Value::Int(n) => n.to_string(),
67      Value::Bool(b) => b.to_string(),
68      Value::Xml(node) => node.get_content(),
69      Value::Null => String::new(),
70      _ => String::new(),
71    }
72  }
73
74  /// Check if the value is truthy (non-null, non-empty).
75  pub fn is_truthy(&self) -> bool {
76    match self {
77      Value::Null => false,
78      Value::String(s) => !s.is_empty(),
79      Value::Bool(b) => *b,
80      Value::List(v) => !v.is_empty(),
81      Value::Hash(h) => !h.is_empty(),
82      _ => true,
83    }
84  }
85}
86
87impl From<&str> for Value {
88  fn from(s: &str) -> Self { Value::String(s.to_string()) }
89}
90
91impl From<String> for Value {
92  fn from(s: String) -> Self { Value::String(s) }
93}
94
95impl std::fmt::Display for Value {
96  fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
97    match self {
98      Value::String(s) => write!(f, "{}", s),
99      Value::Int(n) => write!(f, "{}", n),
100      Value::Bool(b) => write!(f, "{}", b),
101      Value::Null => Ok(()),
102      Value::Xml(node) => write!(f, "{}", node.get_content()),
103      Value::List(_) | Value::Hash(_) => Ok(()),
104    }
105  }
106}

108// NOTE deliberately NO `impl From<Node> for Value`: a bare node handle points
109// into whichever document it came from, and storing one couples the DB's
110// validity to that document's lifetime (fatal under streaming, where page DOMs
111// are freed as soon as they are processed). Nodes enter the DB only through
112// `ObjectDB::adopt_xml`, which copies them into DB-owned storage.
113
114impl From<bool> for Value {
115  fn from(b: bool) -> Self { Value::Bool(b) }
116}
117
118impl From<Vec<String>> for Value {
119  fn from(v: Vec<String>) -> Self { Value::List(v.into_iter().map(Value::String).collect()) }
120}

122impl Entry {
123  /// Create a new entry with the given key.
124  pub fn new(key: &str) -> Self {
125    Entry {
126      key:    key.to_string(),
127      values: HashMap::default(),
128    }
129  }
130
131  /// Get the entry's key.
132  pub fn get_key(&self) -> &str { &self.key }
133
134  /// Check if the entry has a value for the given attribute.
135  pub fn has_value(&self, attr: &str) -> bool { self.values.contains_key(attr) }
136
137  /// Get a value by attribute name.
138  pub fn get_value(&self, attr: &str) -> Option<&Value> { self.values.get(attr) }
139
140  /// Get a string value by attribute name.
141  pub fn get_string(&self, attr: &str) -> Option<&str> {
142    self.values.get(attr).and_then(|v| v.as_str())
143  }
144
145  /// Get an XML node value by attribute name.
146  pub fn get_xml(&self, attr: &str) -> Option<&Node> {
147    match self.values.get(attr) {
148      Some(Value::Xml(n)) => Some(n),
149      _ => None,
150    }
151  }
152
153  /// Get a children list (as string IDs).
154  pub fn get_children(&self) -> Vec<String> {
155    match self.values.get("children") {
156      Some(Value::List(items)) => items
157        .iter()
158        .filter_map(|v| v.as_str().map(String::from))
159        .collect(),
160      _ => vec![],
161    }
162  }
163
164  /// Set multiple attribute-value pairs.
165  ///
166  /// Port of `Entry::setValues`.
167  pub fn set_values(&mut self, pairs: Vec<(&str, Value)>) {
168    for (key, value) in pairs {
169      match value {
170        Value::Null => {
171          self.values.remove(key);
172        },
173        _ => {
174          self.values.insert(key.to_string(), value);
175        },
176      }
177    }
178  }
179
180  /// Set a single value.
181  pub fn set_value(&mut self, attr: &str, value: Value) {
182    match value {
183      Value::Null => {
184        self.values.remove(attr);
185      },
186      _ => {
187        self.values.insert(attr.to_string(), value);
188      },
189    }
190  }
191
192  /// Push values onto a list attribute.
193  ///
194  /// Port of `Entry::pushValues`.
195  pub fn push_values(&mut self, attr: &str, values: Vec<Value>) {
196    let list = self
197      .values
198      .entry(attr.to_string())
199      .or_insert_with(|| Value::List(Vec::new()));
200    if let Value::List(items) = list {
201      for v in values {
202        items.push(v);
203      }
204    }
205  }
206
207  /// Push values onto a list attribute, skipping duplicates.
208  ///
209  /// Port of `Entry::pushNew`.
210  pub fn push_new(&mut self, attr: &str, values: Vec<Value>) {
211    let list = self
212      .values
213      .entry(attr.to_string())
214      .or_insert_with(|| Value::List(Vec::new()));
215    if let Value::List(items) = list {
216      for v in values {
217        let s = v.to_string();
218        if !items.iter().any(|existing| existing.to_string() == s) {
219          items.push(v);
220        }
221      }
222    }
223  }
224
225  /// Create nested hash association.
226  ///
227  /// Port of `Entry::noteAssociation`.
228  /// `noteAssociation("referrers", "parent_id")` creates `{referrers => {parent_id => 1}}`
229  pub fn note_association(&mut self, keys: &[&str]) {
230    if keys.is_empty() {
231      return;
232    }
233    if keys.len() == 1 {
234      self.values.insert(keys[0].to_string(), Value::Bool(true));
235      return;
236    }
237
238    // Navigate/create nested hash structure
239    let first = keys[0];
240    let rest = &keys[1..];
241
242    let hash = self
243      .values
244      .entry(first.to_string())
245      .or_insert_with(|| Value::Hash(HashMap::default()));
246
247    if let Value::Hash(h) = hash {
248      let mut current = h;
249      for (i, &key) in rest.iter().enumerate() {
250        if i == rest.len() - 1 {
251          // Last key: set to true
252          current.insert(key.to_string(), Value::Bool(true));
253        } else {
254          // Intermediate: navigate/create hash
255          let entry = current
256            .entry(key.to_string())
257            .or_insert_with(|| Value::Hash(HashMap::default()));
258          if let Value::Hash(inner) = entry {
259            current = inner;
260          } else {
261            break;
262          }
263        }
264      }
265    }
266  }
267}

269/// The Object Database.
270///
271/// Port of `LaTeXML::Util::ObjectDB`.
272/// In-memory key-value store. For now, no external DB persistence
273/// (the Perl version uses Berkeley DB via DB_File).
274pub struct ObjectDB {
275  /// In-memory entry storage.
276  objects:    HashMap<String, Entry>,
277  /// The ONE document owning every [`Value::Xml`] node the DB stores. A
278  /// stored node used to be a handle into the scanned page's own DOM, which
279  /// made the DB's lifetime silently depend on every page document staying
280  /// resident — a use-after-free the moment a streaming pipeline frees a
281  /// processed page. [`ObjectDB::adopt_xml`] deep-copies into this document
282  /// instead, so stored XML lives exactly as long as the DB, whatever happens
283  /// to the source.
284  ///
285  /// **One document, not one per value.** The first cut retained a whole fresh
286  /// `xmlDoc` per adopted node, and Scan adopts a `title` and `toctitle` for
287  /// every object it registers: on a 614 MB core XML (200,403 objects) that is
288  /// tens of thousands of documents — each with its own dictionary and
289  /// structure — to hold a few MB of title markup. Measured, that regressed
290  /// the split post-processing of that input from ~21 GB peak (completing over
291  /// 40,201 pages) to **67 GB and a memory-ceiling kill with zero pages
292  /// written**. Titles really are small; `xmlDoc`s are not.
293  xml_holder: Option<libxml::tree::Document>,
294}
295
296impl ObjectDB {
297  /// Create a new empty ObjectDB.
298  pub fn new() -> Self {
299    ObjectDB {
300      objects:    HashMap::default(),
301      xml_holder: None,
302    }
303  }
304
305  /// The DB's holding document, created on first adoption (so a DB that
306  /// stores no XML allocates none), with a root element to parent the copies
307  /// under — an unparented node would not be reached by the holder's own
308  /// `xmlFreeDoc` either.
309  fn xml_holder_mut(&mut self) -> Option<&mut libxml::tree::Document> {
310    if self.xml_holder.is_none() {
311      let mut doc = libxml::tree::Document::new().ok()?;
312      let root = Node::new("_objectdb_", None, &doc).ok()?;
313      doc.set_root_element(&root);
314      self.xml_holder = Some(doc);
315    }
316    self.xml_holder.as_mut()
317  }
318
319  /// Adopt an XML node for storage: deep-copy it into a document the DB owns
320  /// and return the [`Value::Xml`] wrapping the copy. This is the ONLY way a
321  /// node enters the DB (`From<Node> for Value` was removed on purpose), so
322  /// no stored value can dangle into a page document that was freed.
323  ///
324  /// Two hops, because the fork's copy primitives pull in opposite
325  /// directions: `dup_node_into_new_doc` copies from a LINKED source but only
326  /// into a fresh document, while `import_node` copies into an EXISTING
327  /// document but rejects a linked source. So: copy out to a scratch
328  /// document, detach, copy into the holder, and free the scratch copy —
329  /// an unlinked doc-owned node is freed by nobody (the rust-libxml `Linkage`
330  /// rule behind `Node::free_subtree`), so dropping the scratch document
331  /// alone would leak it. A one-hop version wants a fork method that copies
332  /// from a linked source into an existing document; that is a publish + dep
333  /// bump, and this is a regression fix.
334  ///
335  /// Returns `None` when the copy fails; callers should degrade to a string
336  /// form rather than store nothing, and never crash.
337  pub fn adopt_xml(&mut self, node: &Node) -> Option<Value> {
338    let scratch = libxml::tree::Document::dup_node_into_new_doc(node).ok()?;
339    let mut extracted = scratch.get_root_element()?;
340    extracted.unlink_node();
341    let holder = self.xml_holder_mut()?;
342    let mut adopted = holder.import_node(&mut extracted).ok()?;
343    let mut root = holder.get_root_element()?;
344    root.add_child(&mut adopted).ok()?;
345    extracted.free_subtree();
346    Some(Value::Xml(adopted))
347  }
348
349  /// Look up an entry by key.
350  ///
351  /// Port of `ObjectDB::lookup`.
352  pub fn lookup(&self, key: &str) -> Option<&Entry> { self.objects.get(key) }
353
354  /// Look up an entry by key (mutable).
355  pub fn lookup_mut(&mut self, key: &str) -> Option<&mut Entry> { self.objects.get_mut(key) }
356
357  /// Register an entry: create if new, or return existing.
358  /// Sets the given properties on the entry.
359  ///
360  /// Port of `ObjectDB::register`.
361  pub fn register(&mut self, key: &str, props: Vec<(&str, Value)>) -> &mut Entry {
362    let entry = self
363      .objects
364      .entry(key.to_string())
365      .or_insert_with(|| Entry::new(key));
366    if !props.is_empty() {
367      entry.set_values(props);
368    }
369    self.objects.get_mut(key).unwrap()
370  }
371
372  /// Remove an entry.
373  ///
374  /// Port of `ObjectDB::unregister`.
375  pub fn unregister(&mut self, key: &str) { self.objects.remove(key); }
376
377  /// Get all keys, sorted.
378  ///
379  /// Port of `ObjectDB::getKeys`.
380  pub fn get_keys(&self) -> Vec<&String> {
381    let mut keys: Vec<_> = self.objects.keys().collect();
382    keys.sort();
383    keys
384  }
385
386  /// Number of registered entries. O(1); avoids `get_keys().len()`'s
387  /// sort + allocation when only the count is needed.
388  pub fn len(&self) -> usize { self.objects.len() }
389
390  /// True when no entries are registered.
391  pub fn is_empty(&self) -> bool { self.objects.is_empty() }
392
393  /// Iterate keys in arbitrary order without allocating/sorting. Use when
394  /// the traversal order does not matter (e.g. per-node fill-ins).
395  pub fn keys_iter(&self) -> impl Iterator<Item = &String> { self.objects.keys() }
396
397  /// Return a status string.
398  ///
399  /// Port of `ObjectDB::status`.
400  pub fn status(&self) -> String { format!("{} objects", self.objects.len()) }
401}

403impl Default for ObjectDB {
404  fn default() -> Self { Self::new() }
405}

499  #[test]
500  fn many_adoptions_share_one_holding_document() {
501    // The regression this guards: one `xmlDoc` per adopted value. Scan adopts
502    // a title (and toctitle) per registered object, so a book-scale document
503    // adopted tens of thousands of times — measured, that took split
504    // post-processing of a 614 MB core XML from ~21 GB to 67 GB and a
505    // memory-ceiling kill. Every copy must live in ONE document, and all of
506    // them must still be readable after the sources are gone.
507    let parser = libxml::parser::Parser::default();
508    let mut db = ObjectDB::new();
509    let mut sources = Vec::new();
510    for i in 0..64 {
511      let doc = parser
512        .parse_string(format!("<title>Section <em>{i}</em></title>").as_bytes())
513        .expect("parse source");
514      let root = doc.get_root_element().expect("root");
515      let adopted = db.adopt_xml(&root).expect("adopt");
516      db.register(&format!("ID:S{i}"), vec![("title", adopted)]);
517      sources.push(doc);
518    }
519    drop(sources); // every page DOM goes away
520    for i in 0..64 {
521      let entry = db.lookup(&format!("ID:S{i}")).expect("entry");
522      let node = entry.get_xml("title").expect("stored node");
523      assert_eq!(node.get_content(), format!("Section {i}"));
524    }
525    // All 64 copies are children of the single holder root, which is what
526    // keeps the retained cost constant instead of linear in objects.
527    let holder = db
528      .xml_holder
529      .as_ref()
530      .expect("holder exists after adoption");
531    let root = holder.get_root_element().expect("holder root");
532    assert_eq!(root.get_child_elements().len(), 64);
533  }

535  #[test]
536  fn adopted_xml_survives_source_document_drop() {
537    // The streaming contract: a stored XML value must stay valid after the
538    // page document it was scanned from is freed. Under the old
539    // `Value::Xml(source_node)` representation this was a use-after-free.
540    let parser = libxml::parser::Parser::default();
541    let source = parser
542      .parse_string(
543        "<title xmlns:m=\"http://www.w3.org/1998/Math/MathML\">\
544         The <m:mi>\u{03b1}</m:mi> section</title>",
545      )
546      .expect("parse source");
547    let title = source.get_root_element().expect("root");
548
549    let mut db = ObjectDB::new();
550    let adopted = db.adopt_xml(&title).expect("adopt");
551    db.register("ID:S1", vec![("title", adopted)]);
552    drop(source); // the page DOM goes away, as it will under streaming
553
554    let entry = db.lookup("ID:S1").expect("entry");
555    let node = entry.get_xml("title").expect("stored node");
556    assert_eq!(node.get_name(), "title");
557    assert_eq!(node.get_content(), "The \u{03b1} section");
558    // Markup survives adoption, not just flattened text: the MathML child is
559    // still an element in its namespace.
560    let mi = node
561      .get_child_nodes()
562      .into_iter()
563      .find(|c| c.get_name() == "mi")
564      .expect("m:mi child survives");
565    assert_eq!(
566      mi.get_namespace().map(|ns| ns.get_href()),
567      Some(String::from("http://www.w3.org/1998/Math/MathML"))
568    );
569  }
