24/// Set libxslt's global template-recursion cap to Perl's value (1000) exactly
25/// once per process. Mirrors `XML::LibXSLT->max_depth(1000)` in
26/// `LaTeXML::Post::XSLT`. Prevents deeply-recursive stylesheet templates from
27/// exhausting the C call stack (SIGSEGV) or RAM on pathological documents,
28/// aborting the transform gracefully like Perl instead.
29fn set_xslt_max_depth() {
30  static SET_MAX_DEPTH: std::sync::Once = std::sync::Once::new();
31  SET_MAX_DEPTH.call_once(|| {
32    // SAFETY: `xsltMaxDepth` is libxslt's process-global recursion cap
33    // (a plain C `int`, declared below with its C type). The libxslt crate
34    // exposes no safe setter. `Once` guarantees a single writer; libxslt only
35    // ever READS this value (when creating each transform context), so there
36    // is no data race with concurrent transforms.
37    #[cfg(any(unix, windows))]
38    unsafe {
39      xsltMaxDepth = 1000;
40    }
41    // Any other platform: skip the write; libxslt's built-in default cap
42    // of 3000 still bounds recursion, just above Perl's 1000.
43  });
44}

46// libxslt's recursion cap (`XSLTPUBVAR int xsltMaxDepth`), declared here rather
47// than taken from the crate's `libxslt::bindings::xsltMaxDepth`. Those
48// pregenerated (bindgen-on-Linux) bindings pin the raw ELF symbol name with
49// `#[link_name = "\u{1}xsltMaxDepth"]`, which fails to LINK on Mach-O where the
50// C symbol is `_xsltMaxDepth` (macOS probe 2026-06-07 — the sole undefined
51// symbol in the whole workspace link; see
52// docs/archive/PORTABILITY_MACOS_PROBE_2026-06-07.md, WISDOM #56). A plain
53// declaration gets the platform's own C-symbol decoration from rustc on ELF and
54// Mach-O, dynamic or static, and on x64 COFF for the static vcpkg libxslt we
55// link there (a DLL's data would need a `dllimport`, `#[link(kind = "dylib")]`);
56// the libxslt crate's build script already puts the library on the link line.
57//
58// The reference is resolved at LINK time. The runtime lookup it replaces on
59// unix, `dlsym(RTLD_DEFAULT, "xsltMaxDepth")`, only searches DYNAMIC symbol
60// tables, so it silently skipped the write — leaving libxslt's 3000 — whenever
61// the symbol was not in one: libxslt linked statically (the `LIBXSLT_STATIC`
62// release binary, where the executable does not export the global), or not
63// loaded at all (the `latexml_post` unit-test binary with dependencies at
64// opt-level 1, whose live code has no other libxslt reference, so the linker's
65// `--as-needed` dropped the library). A static libxslt 1.1.42 (the release
66// build's, tools/build_static_libxml.sh) linked into a C program: `dlsym` NULL,
67// the linked global 3000. `GetProcAddress` fails on Windows' static vcpkg
68// libxslt for the same reason (WINDOWS_COMPATIBILITY_PLAN Phase 2.3).
69#[cfg(any(unix, windows))]
70#[allow(non_upper_case_globals)]
71unsafe extern "C" {
72  static mut xsltMaxDepth: std::os::raw::c_int;
73}

93  /// A named template that calls itself `$n` times, applied through the
94  /// production `XSLT::process` path (which sets the cap under `XSLT_LOCK`).
95  /// libxslt spends two depth units per level of this template (measured:
96  /// `xsltproc --maxdepth 1000` aborts from n = 499, the default 3000 from
97  /// n = 1499).
98  fn apply_recursion(n: u32) -> bool {
99    const RECURSION_XSL: &str = r#"<xsl:stylesheet version="1.0" xmlns:xsl="http://www.w3.org/1999/XSL/Transform">
100  <xsl:param name="n" select="0"/>
101  <xsl:template match="/">
102    <out><xsl:call-template name="down"><xsl:with-param name="k" select="$n"/></xsl:call-template></out>
103  </xsl:template>
104  <xsl:template name="down">
105    <xsl:param name="k"/>
106    <xsl:if test="$k &gt; 0">
107      <xsl:call-template name="down"><xsl:with-param name="k" select="$k - 1"/></xsl:call-template>
108    </xsl:if>
109  </xsl:template>
110</xsl:stylesheet>
111"#;
112    let dir = tempfile::tempdir().expect("tempdir");
113    let xsl = dir.path().join("recursion.xsl");
114    std::fs::write(&xsl, RECURSION_XSL).expect("write recursion.xsl");
115    let mut params = rustc_hash::FxHashMap::default();
116    params.insert("n".to_string(), n.to_string());
117    let mut xslt =
118      super::XSLT::new(xsl.to_str().unwrap(), params, true, None, vec![]).expect("XSLT processor");
119    let doc =
120      PostDocument::new_from_string("<a/>", PostDocumentOptions::default()).expect("parse <a/>");
121    xslt.process(doc, vec![]).is_ok()
122  }

147  /// Run `f` with file descriptor 2 pointed at a temporary file, and return what
148  /// was written there: libxslt reports the aborted recursion (and its template
149  /// stack) with C `fprintf(stderr, …)`, below any Rust output capture, so without
150  /// this the report would reach the terminal of whoever runs the suite.
151  #[cfg(unix)]
152  fn with_fd2_captured<T>(f: impl FnOnce() -> T) -> (T, Option<String>) {
153    use std::{
154      io::{Read, Seek},
155      os::fd::AsRawFd,
156    };
157    /// Points fd 2 back at its saved copy when dropped — on a panic in `f` too.
158    struct Restore(std::ffi::c_int);
159    impl Drop for Restore {
160      fn drop(&mut self) {
161        // SAFETY: `self.0` is the `dup(2)` taken below, still open.
162        unsafe {
163          libc::dup2(self.0, 2);
164          libc::close(self.0);
165        }
166      }
167    }
168    let mut sink = tempfile::tempfile().expect("tempfile");
169    // SAFETY: plain descriptor calls on fd 2 and descriptors this function owns.
170    let saved = unsafe { libc::dup(2) };
171    assert!(saved >= 0, "dup(2)");
172    let restore = Restore(saved);
173    assert!(
174      unsafe { libc::dup2(sink.as_raw_fd(), 2) } >= 0,
175      "dup2 onto fd 2"
176    );
177    let out = std::panic::catch_unwind(std::panic::AssertUnwindSafe(f));
178    drop(restore);
179    let mut text = String::new();
180    sink.rewind().expect("rewind");
181    sink.read_to_string(&mut text).expect("read captured fd 2");
182    match out {
183      Ok(out) => (out, Some(text)),
184      Err(panic) => {
185        // The panic message went to the captured fd 2: show it before unwinding on.
186        eprint!("{text}");
187        std::panic::resume_unwind(panic)
188      },
189    }
190  }
191  #[cfg(not(unix))]
192  fn with_fd2_captured<T>(f: impl FnOnce() -> T) -> (T, Option<String>) { (f(), None) }
193}

350  pub fn new(
351    stylesheet: &str,
352    parameters: HashMap<String, String>,
353    no_resources: bool,
354    resource_directory: Option<String>,
355    searchpaths: Vec<String>,
356  ) -> Result<Self, PostError> {
357    if stylesheet.is_empty() {
358      // Perl XSLT.pm:36 — Error('expected', 'stylesheet', undef,
359      //   "No stylesheet specified!")
360      Error!("expected", "stylesheet", "No stylesheet specified!");
361      return Err(PostError::Processing(
362        "No stylesheet specified!".to_string(),
363      ));
364    }
365
366    // Find the stylesheet file
367    let stylesheet_path = match find_stylesheet(stylesheet, &searchpaths) {
368      Ok(p) => p,
369      Err(e) => {
370        // Perl XSLT.pm:42 — Error('missing-file', $stylesheet, undef,
371        //   "No stylesheet '$stylesheet' found!")
372        Error!(
373          "missing-file",
374          stylesheet,
375          "No stylesheet '{}' found!",
376          stylesheet
377        );
378        return Err(e);
379      },
380    };
381
382    Ok(XSLT {
383      name: format!(
384        "XSLT[using {}]",
385        stylesheet
386          .strip_prefix(embedded_xslt::URL_PREFIX)
387          .unwrap_or(stylesheet)
388      ),
389      stylesheet_path: Some(stylesheet_path),
390      parameters,
391      no_resources,
392      resource_directory,
393      searchpaths,
394    })
395  }

397  /// Copy a resource file and return the path relative to the destination.
398  ///
399  /// Port of `XSLT::copyResource`.
400  fn copy_resource(&self, doc: &PostDocument, src: &str, resource_type: Option<&str>) -> String {
401    // If it's a URL, return as-is
402    if src.starts_with("http://") || src.starts_with("https://") || src.starts_with("//") {
403      return src.to_string();
404    }
405
406    let info = match resource_type {
407      Some("text/css") => Some(&RESOURCE_CSS),
408      Some("text/javascript") => Some(&RESOURCE_JS),
409      _ => None,
410    };
411
412    // Try to find the file
413    let search_paths: Vec<&str> = doc
414      .get_search_paths()
415      .iter()
416      .chain(self.searchpaths.iter())
417      .map(String::as_str)
418      .collect();
419
420    let basename = Path::new(src)
421      .file_name()
422      .and_then(|f| f.to_str())
423      .unwrap_or(src);
424
425    // The path (relative to the destination base) the resource is copied to.
426    // Perl `copyResource`: with a `resource_directory`, flatten into it; otherwise
427    // PRESERVE the src's path relative to the source directory — provided it stays
428    // BELOW the destination (a plain relative path, not escaping via `../` or
429    // absolute), else flatten to the basename. So `subdir/foo.css` lands at
430    // `<dest>/subdir/foo.css`, matching the `<link>`/`<script>` href, instead of a
431    // flattened `foo.css` the tag can no longer resolve (#662).
432    let rel_dest: String = if self.resource_directory.is_some() {
433      basename.to_string()
434    } else {
435      let relpath = latexml_core::util::pathname::relative(src, doc.get_source_directory());
436      if relpath.starts_with("../") || Path::new(&relpath).is_absolute() {
437        basename.to_string()
438      } else {
439        relpath
440      }
441    };
442
443    // Determine destination once — same logic regardless of whether
444    // the resource ends up on disk or comes from the embedded table.
445    let dest = if let Some(ref rd) = self.resource_directory {
446      if let Some(site_dir) = doc.get_site_directory() {
447        format!("{}/{}/{}", site_dir, rd, rel_dest)
448      } else {
449        format!("{}/{}", rd, rel_dest)
450      }
451    } else if let Some(base) = doc
452      .get_site_directory()
453      .or_else(|| doc.get_destination_directory())
454    {
455      format!("{}/{}", base, rel_dest)
456    } else {
457      rel_dest
458    };
459    let ensure_parent = |dest: &str| {
460      if let Some(parent) = Path::new(dest).parent() {
461        let _ = fs::create_dir_all(parent);
462      }
463    };
464
465    match find_resource_file(src, info, &search_paths) {
466      Some(path) => {
467        if same_file(&path, &dest) {
468          // The only search-path match IS the destination file itself — the dest
469          // dir is on the search path and a stale/empty resource from an earlier
470          // (possibly failed) run is shadowing the bundled copy (#312). A plain
471          // `fs::copy` here copies the file onto itself and TRUNCATES it to empty.
472          // Rewrite the embedded canonical bytes when we have them, repairing the
473          // stale/empty leftover; otherwise it is genuinely already in place.
474          if let Some(bytes) = embedded_resources::lookup(basename) {
475            ensure_parent(&dest);
476            if let Err(e) = fs::write(&dest, bytes) {
477              Warn!(
478                "I/O",
479                dest,
480                "Couldn't rewrite embedded {} to {}: {}",
481                basename,
482                dest,
483                e
484              );
485            }
486          }
487        } else {
488          ensure_parent(&dest);
489          if let Err(e) = fs::copy(&path, &dest) {
490            Warn!("I/O", dest, "Couldn't copy {} to {}: {}", path, dest, e);
491          }
492        }
493      },
494      None => {
495        // Not on disk — try the embedded table. CSS/JS assets are
496        // baked into the binary at build time; we materialize them
497        // straight to the destination, no temp dir round-trip.
498        if let Some(bytes) = embedded_resources::lookup(basename) {
499          ensure_parent(&dest);
500          if let Err(e) = fs::write(&dest, bytes) {
501            Warn!(
502              "I/O",
503              dest,
504              "Couldn't write embedded resource {} to {}: {}",
505              basename,
506              dest,
507              e
508            );
509          }
510        } else {
511          Warn!(
512            "missing_file",
513            src,
514            "Couldn't find resource file {} in paths {:?}",
515            src,
516            search_paths
517          );
518          return src.to_string();
519        }
520      },
521    }
522
523    // Return path relative to destination directory.
524    if let Some(dest_dir) = doc.get_destination_directory() {
525      relative_path(&dest, dest_dir)
526    } else {
527      dest
528    }
529  }

531  /// Copy each resource named in a `"a|b|c"` quoted pipe-list stylesheet
532  /// parameter (`CSS`, `JAVASCRIPT`, `ICON`) to the destination so the
533  /// `<link>`/`<script>` the stylesheet emits actually resolve.
534  ///
535  /// Port of `XSLT::process` L71-78 (the param-resource copy, distinct from the
536  /// embedded `<ltx:resource>` handling): every `--css` / `--javascript` /
537  /// icon entry is searched on the path (then the embedded table), copied, and
538  /// a missing entry warns `missing_file`. This runs **regardless of
539  /// `no_resources`** — `--nodefaultresources` only governs embedded
540  /// `<ltx:resource>` nodes, not CLI-specified resources (Perl L62-78: the CSS/
541  /// JAVASCRIPT/ICON copies sit *outside* the `noresources` guard).
542  ///
543  /// The copy targets the **site directory** (the root the relativized links in
544  /// [`relativize_resource_params`] point at, via `../` for split sub-pages),
545  /// falling back to the destination directory when no site dir is set. Link
546  /// relativization itself stays in `relativize_resource_params`; this method
547  /// only performs the copy, so split/`--splitat` link paths are unchanged.
548  fn copy_param_resources(&self, doc: &PostDocument, value: &str, info: Option<&ResourceInfo>) {
549    let dest_root = match doc
550      .get_site_directory()
551      .or_else(|| doc.get_destination_directory())
552    {
553      Some(d) => d.to_string(),
554      None => return,
555    };
556    let search_paths: Vec<&str> = doc
557      .get_search_paths()
558      .iter()
559      .chain(self.searchpaths.iter())
560      .map(String::as_str)
561      .collect();
562    // Only CSS recurses into its `@import`-ed sub-resources (below).
563    let is_css = info.is_some_and(|i| i.extension == "css");
564    // Cycle/dedup guard shared across every entry's @import graph.
565    let mut seen: HashSet<PathBuf> = HashSet::new();
566    for entry in value.trim_matches('"').split('|') {
567      let entry = entry.trim();
568      // Skip empties and URLs (nothing to copy — same guard as copy_resource).
569      if entry.is_empty()
570        || entry.starts_with("http://")
571        || entry.starts_with("https://")
572        || entry.starts_with("//")
573      {
574        continue;
575      }
576      let basename = Path::new(entry)
577        .file_name()
578        .and_then(|f| f.to_str())
579        .unwrap_or(entry);
580      let dest = format!("{}/{}", dest_root, basename);
581      let ensure_parent = || {
582        if let Some(parent) = Path::new(&dest).parent() {
583          let _ = fs::create_dir_all(parent);
584        }
585      };
586      match find_resource_file(entry, info, &search_paths) {
587        Some(path) => {
588          if same_file(&path, &dest) {
589            // Only search-path match IS the destination — avoid the self-copy
590            // truncate; rewrite the embedded canonical bytes if bundled (#312).
591            if let Some(bytes) = embedded_resources::lookup(basename) {
592              ensure_parent();
593              if let Err(e) = fs::write(&dest, bytes) {
594                Warn!(
595                  "I/O",
596                  dest,
597                  "Couldn't rewrite embedded {} to {}: {}",
598                  basename,
599                  dest,
600                  e
601                );
602              }
603            }
604          } else {
605            ensure_parent();
606            if let Err(e) = fs::copy(&path, &dest) {
607              Warn!("I/O", dest, "Couldn't copy {} to {}: {}", path, dest, e);
608            }
609          }
610          // Beyond Perl: follow the copied CSS's local @import chain so split
611          // stylesheets (e.g. ar5iv.css -> ./ar5iv/*.css layer files) bring
612          // their sub-files along. Gated to CSS; no-op for JS/icon.
613          if is_css {
614            copy_css_imports(Path::new(&path), Path::new(&dest), &mut seen);
615          }
616        },
617        None => {
618          // Not on the path — try the binary's embedded resource table
619          // (same fallback as copy_resource, so bundled CSS/JS still land).
620          if let Some(bytes) = embedded_resources::lookup(basename) {
621            ensure_parent();
622            if let Err(e) = fs::write(&dest, bytes) {
623              Warn!(
624                "I/O",
625                dest,
626                "Couldn't write embedded resource {} to {}: {}",
627                basename,
628                dest,
629                e
630              );
631            }
632          } else {
633            Warn!(
634              "missing_file",
635              entry,
636              "Couldn't find resource file {} in paths {:?}",
637              entry,
638              search_paths
639            );
640          }
641        },
642      }
643    }
644  }

646  /// Build a per-doc parameter map with `CSS`, `JAVASCRIPT`, and `ICON`
647  /// relativized so each split sub-page references the resource at the
648  /// correct relative path.
649  ///
650  /// The raw values are constructed as `"foo.css|bar.css"` (quoted,
651  /// pipe-separated basenames) by the binary's `run_post_processing`.
652  /// They are interpreted as paths relative to the site root, so
653  /// sub-pages need `../foo.css` etc.
654  fn relativize_resource_params(&self, doc: &PostDocument) -> HashMap<String, String> {
655    let mut out = self.parameters.clone();
656    let (Some(site), Some(dest)) = (doc.get_site_directory(), doc.get_destination_directory())
657    else {
658      return out;
659    };
660    let prefix = match relative_dir_prefix(site, dest) {
661      Some(p) => p,
662      None => return out,
663    };
664    if prefix.is_empty() {
665      return out;
666    }
667    for key in ["CSS", "JAVASCRIPT", "ICON"] {
668      if let Some(value) = out.get(key).cloned() {
669        out.insert(
670          key.to_string(),
671          relativize_quoted_pipe_list(&value, &prefix),
672        );
673      }
674    }
675    out
676  }
677}

679/// Walk-up prefix from `dest_dir` to `site_dir`. Returns `Some("")` when
680/// they're identical, `Some("../")` when `dest_dir` is one level deeper,
681/// `Some("../../")` two levels, etc. Returns `None` if `dest_dir` is not
682/// inside `site_dir`.
683fn relative_dir_prefix(site_dir: &str, dest_dir: &str) -> Option<String> {
684  let site = Path::new(site_dir);
685  let dest = Path::new(dest_dir);
686  let rel = dest.strip_prefix(site).ok()?;
687  let depth = rel.components().count();
688  Some("../".repeat(depth))
689}

691/// Apply `prefix` to every basename in a `"a|b|c"` quoted pipe-list, but
692/// only when the entry doesn't already look absolute or scheme-prefixed.
693fn relativize_quoted_pipe_list(value: &str, prefix: &str) -> String {
694  let inner = value.trim_matches('"');
695  let parts: Vec<String> = inner
696    .split('|')
697    .map(|p| {
698      let p = p.trim();
699      if p.is_empty()
700        || p.starts_with('/')
701        || p.starts_with("./")
702        || p.starts_with("../")
703        || p.contains("://")
704      {
705        p.to_string()
706      } else {
707        format!("{}{}", prefix, p)
708      }
709    })
710    .collect();
711  format!("\"{}\"", parts.join("|"))
712}

714/// Extract the targets of CSS `@import` rules from stylesheet source.
715///
716/// Block comments are stripped first (so a commented-out `@import` is ignored),
717/// then the common forms are matched: `@import "x.css";`, `@import url("x");`,
718/// `@import url('x');`, and bare `@import url(x);`, each with an optional
719/// trailing `layer(...)` / media / `supports(...)` tail (only the URL token is
720/// captured). One target per `@import` (CSS allows a single URL per rule).
721fn parse_css_imports(css: &str) -> Vec<String> {
722  static COMMENT: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?s)/\*.*?\*/").unwrap());
723  static IMPORT: LazyLock<Regex> =
724    LazyLock::new(|| Regex::new(r#"(?i)@import\s+(?:url\(\s*)?["']?([^"')\s;]+)"#).unwrap());
725  let stripped = COMMENT.replace_all(css, "");
726  IMPORT
727    .captures_iter(&stripped)
728    .filter_map(|c| c.get(1).map(|m| m.as_str().to_string()))
729    .collect()
730}

732/// True iff both paths exist and canonicalize to the **same file** on disk.
733/// Used to avoid `fs::copy`ing a file onto itself — which truncates it to empty
734/// — when a resource's only search-path match is the destination file itself
735/// (the dest dir is on the search path; #312). A plain string `!=` on the two
736/// path spellings (relative vs absolute, `./` prefixes) misses this.
737fn same_file(a: impl AsRef<Path>, b: impl AsRef<Path>) -> bool {
738  match (fs::canonicalize(a), fs::canonicalize(b)) {
739    (Ok(ca), Ok(cb)) => ca == cb,
740    _ => false,
741  }
742}

744/// After a CSS file is copied from `src` to `dest`, recursively copy the LOCAL
745/// resources it `@import`s, preserving each import's relative subpath so the
746/// cascade still resolves at the destination.
747///
748/// BEYOND PERL: `copyResource` copies only the named file. ar5iv-style
749/// stylesheets split themselves across `@import url("./layer/part.css")`
750/// sub-files; without following those, the copied top-level CSS renders
751/// unstyled. Each target is resolved relative to the importing file — for both
752/// the on-disk source and the destination, so the same relative path is
753/// recreated under the destination — copied, and (if itself a `.css`) recursed
754/// into. Remote (`http(s)://`, `//`, `data:`, any `scheme://`) and absolute
755/// (`/…`) targets are left untouched. `seen` (keyed on the absolute source
756/// path) guards against import cycles and redundant copies.
757fn copy_css_imports(src: &Path, dest: &Path, seen: &mut HashSet<PathBuf>) {
758  let content = match fs::read_to_string(src) {
759    Ok(c) => c,
760    Err(_) => return,
761  };
762  let (Some(src_dir), Some(dest_dir)) = (src.parent(), dest.parent()) else {
763    return;
764  };
765  for target in parse_css_imports(&content) {
766    let t = target.trim();
767    // Skip remote / non-copyable targets — only local relative paths are ours.
768    if t.is_empty()
769      || t.starts_with('/')
770      || t.starts_with("//")
771      || t.starts_with("data:")
772      || t.contains("://")
773    {
774      continue;
775    }
776    let import_src = src_dir.join(t);
777    let import_dest = dest_dir.join(t);
778    // Skip if already handled (shared import or cycle).
779    if !seen.insert(import_src.clone()) {
780      continue;
781    }
782    if !import_src.is_file() {
783      Warn!(
784        "missing_file",
785        t,
786        "Couldn't find @import target {} referenced by {}",
787        t,
788        src.display()
789      );
790      continue;
791    }
792    if !same_file(&import_src, &import_dest) {
793      if let Some(parent) = import_dest.parent() {
794        let _ = fs::create_dir_all(parent);
795      }
796      if let Err(e) = fs::copy(&import_src, &import_dest) {
797        Warn!(
798          "I/O",
799          t,
800          "Couldn't copy @import {} to {}: {}",
801          import_src.display(),
802          import_dest.display(),
803          e
804        );
805        continue;
806      }
807    }
808    // Recurse into nested CSS imports only (not imported fonts/images).
809    if import_src
810      .extension()
811      .and_then(|e| e.to_str())
812      .is_some_and(|e| e.eq_ignore_ascii_case("css"))
813    {
814      copy_css_imports(&import_src, &import_dest, seen);
815    }
816  }
817}

819impl Processor for XSLT {
820  fn get_name(&self) -> &str { &self.name }
821
822  fn process(&mut self, doc: PostDocument, _nodes: Vec<Node>) -> ProcessResult {
823    let stylesheet_path = match &self.stylesheet_path {
824      Some(p) => p.clone(),
825      None => return Ok(vec![doc]),
826    };
827
828    Info!(
829      "xslt",
830      "stylesheet",
831      "Applying XSLT stylesheet: {}",
832      stylesheet_path
833    );
834
835    // Handle resource elements first (before transformation removes them)
836    let resource_nodes = doc.findnodes("//ltx:resource[@src]");
837    if self.no_resources {
838      // Perl L64-65: remove resource nodes so XSLT won't generate CSS/JS links
839      for mut node in resource_nodes {
840        node.unlink_node();
841      }
842    } else {
843      for node in &resource_nodes {
844        if let Some(src) = node.get_attribute("src") {
845          let resource_type = node.get_attribute("type");
846          let path = self.copy_resource(&doc, &src, resource_type.as_deref());
847          // Perl XSLT.pm:70 — rewrite `@src` to where the file was actually copied,
848          // so the `<link>`/`<script>` the stylesheet emits resolves (#662). Node is
849          // a shared handle, so mutating the clone edits the real node.
850          if path != src {
851            node.clone().set_attribute("src", &path).ok();
852          }
853        }
854      }
855    }
856
857    // Copy CLI-specified --css/--javascript/icon resources to the destination
858    // and warn on any that can't be found. Port of XSLT::process L71-78 — the
859    // param-resource copy that the binary's `--css`/`--javascript` flow needs
860    // (those flags only set the CSS/JAVASCRIPT stylesheet params; without this
861    // the link is emitted but the file is never searched on --path or copied).
862    // Deliberately OUTSIDE the `no_resources` guard above: --nodefaultresources
863    // suppresses only the bundled defaults' <ltx:resource> nodes, not these.
864    if let Some(css) = self.parameters.get("CSS") {
865      self.copy_param_resources(&doc, css, Some(&RESOURCE_CSS));
866    }
867    if let Some(js) = self.parameters.get("JAVASCRIPT") {
868      self.copy_param_resources(&doc, js, Some(&RESOURCE_JS));
869    }
870    if let Some(icon) = self.parameters.get("ICON") {
871      self.copy_param_resources(&doc, icon, None);
872    }
873
874    // Serialize the entire libxslt-touching critical section process-wide.
875    // libxslt/libxml2 keep NON-thread-safe process-global state — the input-
876    // callback + EXSLT registries, the generic error context, and the
877    // namespace-internalisation / dictionary caches that `xsltApplyStylesheet
878    // User` and stylesheet parsing mutate (the hidden mutation this file's
879    // per-thread-cache note already anticipates). Two conversion threads
880    // transforming concurrently DEADLOCK on that state: witnessed as the
881    // `52_source_map` XSLT tests hanging forever on a futex under
882    // `cargo test --tests` (all threads `futex_do_wait`, 0% CPU). The thread-
883    // local stylesheet cache below removes cross-thread *cache* sharing but not
884    // this shared C-library state, so a process-global lock is required for
885    // correctness. Cost: NONE in production — the CLI and the cortex fleet run
886    // one conversion per process (single thread), so this is never contended;
887    // only the multi-threaded test harness (or a hypothetical in-process pool)
888    // ever serializes here. Poison-tolerant: a transform that panicked while
889    // holding the lock didn't corrupt anything we read, so recover the guard.
890    static XSLT_LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
891    let _xslt_guard = XSLT_LOCK
892      .lock()
893      .unwrap_or_else(|poisoned| poisoned.into_inner());
894
895    // Register EXSLT extension functions (str:tokenize, math:*, etc.)
896    // used by LaTeXML stylesheets. Safe-wrapped upstream in
897    // rust-libxslt — `register_exslt()` is Once-guarded.
898    libxslt::register_exslt();
899
900    // Faithful port of Perl `XML::LibXSLT->max_depth(1000)`
901    // (LaTeXML::Post::XSLT.pm L48): cap libxslt's template-recursion depth.
902    // libxslt's compiled-in default is 3000; lowering to Perl's 1000 makes a
903    // runaway / deeply-recursive stylesheet apply ABORT gracefully (matching
904    // Perl) instead of growing the C call stack until SIGSEGV/OOM on
905    // pathological input. `xsltMaxDepth` is a process-global libxslt static
906    // read when each transform context is created, so setting it once is
907    // sufficient. See docs/performance/STABILITY_WITNESSES.md (Cluster A, hypothesis 3).
908    set_xslt_max_depth();
909
910    // Hand the source tree to libxslt WITHOUT a deep copy. `transform()`
911    // takes its `Document` by value (the moved handle's Drop would free the
912    // tree), so earlier code `dup()`'d (xmlCopyDoc — a full deep copy) to keep
913    // this PostDocument's own tree alive. On a large-math document the DOM is
914    // multi-GB, and that deep copy TRANSIENTLY DOUBLES peak RSS during the
915    // transform — the dominant driver of post-processing OOM on the canvas
916    // sweep (docs/performance/STABILITY_WITNESSES.md, Cluster A / hypothesis 1).
917    //
918    // `libxml::Document` is `Rc<RefCell<_Document>>` and the underlying
919    // `xmlDoc` is freed only when the LAST handle drops, so an Rc `clone()`
920    // (a refcount bump, no copy) is the right tool: libxslt reads the shared
921    // tree to build a SEPARATE result tree (`xsltApplyStylesheetUser` does not
922    // free its source), the moved clone's Drop just decrements the count, and
923    // `doc`'s own handle (dropped at function end) performs the single real
924    // free. We never read `doc`'s tree again after this point — only its
925    // string metadata below — so libxslt mutating the shared source while
926    // applying is harmless. This mirrors Perl, which passes
927    // `$doc->getDocument` straight to `transform` with no pre-copy
928    // (LaTeXML::Post::XSLT.pm L79).
929    let transform_doc = doc.get_document().clone();
930
931    // Build parameters, relativizing path-valued ones (CSS, JAVASCRIPT,
932    // ICON) for the current doc's destination. The crate-level params
933    // hold basenames in site-relative form; split sub-pages live in a
934    // subdirectory and need `../foo.css` etc. to resolve correctly.
935    let per_doc_params = self.relativize_resource_params(&doc);
936    let params: Vec<(&str, &str)> = per_doc_params
937      .iter()
938      .map(|(k, v)| (k.as_str(), v.as_str()))
939      .collect();
940
941    // Apply the transformation. The parsed `Stylesheet` lives in a
942    // per-thread cache (`with_cached_stylesheet`) — `libxslt::parser::
943    // parse_file` runs once per (thread, stylesheet path) instead of
944    // once per conversion. The cache is thread-local, not shared, so
945    // we don't lean on libxslt's undocumented thread-safety (mirroring
946    // the caution that resolved KWARC/rust-libxslt issue #6).
947    let result_doc = with_cached_stylesheet(&stylesheet_path, |stylesheet| {
948      stylesheet
949        .transform(transform_doc, params)
950        .map_err(|e| PostError::Processing(format!("XSLT transformation failed: {}", e)))
951    })?;
952    // Transform done — the libxslt-global critical section is over. Release the
953    // lock BEFORE wrapping the result: `result_doc` is the transform's fresh,
954    // unshared output tree, so building the `PostDocument` around it touches no
955    // shared libxslt/libxml2 state. Narrowing the hold keeps a hypothetical
956    // in-process pool serialized only over the actual transform, not the cheap
957    // wrapping. (No effect on the one-conversion-per-process production path,
958    // where the lock is never contended anyway.)
959    drop(_xslt_guard);
960
961    // XSLT returns a libxml `Document` directly — wrap it into a
962    // PostDocument without the serialize → reparse roundtrip the
963    // earlier code did. Saves ~10-30 ms on a typical mid-size paper
964    // (XML serialize + libxml2 reparse of ~100-500 KB markup).
965    if result_doc.get_root_element().is_none() {
966      return Err(PostError::Processing(
967        "XSLT produced empty output".to_string(),
968      ));
969    }
970
971    let result_doc = PostDocument::new(result_doc, PostDocumentOptions {
972      destination: doc.destination.clone(),
973      destination_directory: doc.destination_directory.clone(),
974      site_directory: doc.site_directory.clone(),
975      source: doc.source.clone(),
976      source_directory: doc.source_directory.clone(),
977      searchpaths: Some(doc.searchpaths.clone()),
978      ..PostDocumentOptions::default()
979    });
980
981    Ok(vec![result_doc])
982  }
983}

986// Per-thread cache of parsed stylesheets.
987//
988// `libxslt::parser::parse_file` reads the .xsl from disk and compiles
989// it. For LaTeXML-html5.xsl that's ~5–10 ms including its xsl:imports.
990// On a single CLI run that's once per process — fine. On a daemon-mode
991// `cortex_worker` chewing through 10 000 papers from a thread pool of
992// 8 workers, naive code re-parses once per paper. With this cache,
993// each worker thread parses each unique stylesheet path *once* and
994// reuses the compiled artefact for the rest of its lifetime.
995//
996// ## Why thread-local (and not process-wide + Arc/Mutex)?
997//
998// libxslt is not documented as thread-safe. `xsltApplyStylesheetUser`
999// is not audited to be read-only on the stylesheet — it may write
1000// back into namespace-internalisation caches, error context fields,
1001// or other internal state. This is the same kind of hidden mutation
1002// that issue KWARC/rust-libxslt#6 punctured for the input `Document`
1003// (libxslt silently mutates docs during whitespace stripping). A
1004// process-wide cache shared across worker threads via `Arc` would
1005// either need a `Mutex` (serialising transforms — defeats the
1006// throughput benefit) or rely on libxslt's undocumented thread-safety
1007// (the same bet that #6 retired).
1008//
1009// Thread-local keeps the safety story simple: each thread owns its
1010// own `Stylesheet` for its lifetime, no cross-thread sharing, and the
1011// `&mut Stylesheet` requirement is satisfied by `RefCell::borrow_mut`.
1012// Worst case: 8 worker threads × 1 parse per unique stylesheet =
1013// 8 parses per process, instead of N parses per N papers.
1014
1015fn cache_key(path: &str) -> String {
1016  // Canonicalise so `./resources/XSLT/foo.xsl` and
1017  // `/abs/.../resources/XSLT/foo.xsl` hit the same entry. Falls back
1018  // to the raw path on canonicalisation failure (the file might not
1019  // exist yet — let parse_file emit its own error in that case).
1020  fs::canonicalize(path)
1021    .map(|p| p.to_string_lossy().into_owned())
1022    .unwrap_or_else(|_| path.to_string())
1023}

1025thread_local! {
1026  static STYLESHEET_CACHE: RefCell<HashMap<String, libxslt::stylesheet::Stylesheet>> =
1027    RefCell::new(HashMap::default());
1028}
1029
1030/// Borrow a `&mut Stylesheet` from the per-thread cache, parsing on
1031/// miss. The closure runs while the cache is mutably borrowed, so
1032/// nested calls (which the LaTeXML pipeline never makes) would
1033/// `RefCell::borrow_mut`-panic — a deliberate single-borrow contract.
1034fn with_cached_stylesheet<F, R>(path: &str, f: F) -> Result<R, PostError>
1035where F: FnOnce(&mut libxslt::stylesheet::Stylesheet) -> Result<R, PostError> {
1036  let key = cache_key(path);
1037  // A user `--stylesheet` (parsed from disk, below) may `xsl:import` the engine
1038  // by `urn:x-LaTeXML:XSLT:` — install the embedded-XSLT input callback for
1039  // every parse, not just the embed:/// fallback path (issue #292). Idempotent.
1040  embedded_xslt::install_callback_once();
1041  STYLESHEET_CACHE.with(|cache| {
1042    let mut map = cache.borrow_mut();
1043    if !map.contains_key(&key) {
1044      let parsed = if let Some(name) = path.strip_prefix(embedded_xslt::URL_PREFIX) {
1045        // `embed:///<name>` sentinel from `find_stylesheet`. Parse the
1046        // root stylesheet from the embedded byte table; libxslt's
1047        // `xsl:import` machinery will then re-enter our libxml2 input
1048        // callback for every referenced URL.
1049        let bytes = embedded_xslt::lookup(name).ok_or_else(|| {
1050          PostError::Processing(format!("Embedded XSLT stylesheet {} not found", name))
1051        })?;
1052        libxslt::parser::parse_bytes(bytes.to_vec(), path)
1053          .map_err(|e| PostError::Processing(format!("Failed to parse embedded XSLT: {}", e)))?
1054      } else {
1055        libxslt::parser::parse_file(path)
1056          .map_err(|e| PostError::Processing(format!("Failed to parse XSLT stylesheet: {}", e)))?
1057      };
1058      map.insert(key.clone(), parsed);
1059    }
1060    let entry = map
1061      .get_mut(&key)
1062      .expect("cache entry just inserted is missing");
1063    f(entry)
1064  })
1065}

1172  /// Map any URL libxml2 hands us to an embedded XSLT file, or `None` if we
1173  /// don't serve it. We key purely on the **final path segment**, because a
1174  /// user `--stylesheet` imports the engine by the LaTeXML-canonical
1175  /// `urn:x-LaTeXML:XSLT:X` scheme (Perl resolves it via an XML catalog), and
1176  /// libxml2 versions then resolve that root's *relative* child imports
1177  /// differently against the opaque `urn:` base: Linux composes
1178  /// `urn:LaTeXML-all-xhtml.xsl`, macOS drops to a bare `LaTeXML-all-xhtml.xsl`
1179  /// (CI witness, issue #292). Keying on the basename resolves `embed:///X`,
1180  /// `urn:x-LaTeXML:XSLT:X`, `urn:X`, and bare `X` uniformly. Only our
1181  /// distinctive `LaTeXML-*.xsl` names are in [`FILES`], so a user's own
1182  /// relative import (a different basename) still loads from disk.
1183  ///
1184  /// An absolute path or `file:` URL naming a file on disk is that file (#806): only a stylesheet
1185  /// the user named is made absolute (`find_stylesheet`), so it and an engine-named sibling it
1186  /// imports load from disk as in Perl, while a relative or bare engine name never reads the cwd.
1187  pub fn resolve(url: &str) -> Option<&'static [u8]> {
1188    if disk_path(url).is_some() {
1189      return None;
1190    }
1191    let name = url.rsplit(['/', ':']).next().unwrap_or(url);
1192    lookup(name)
1193  }

1195  /// The file an absolute path or a `file:` URL (`file:///p`, `file://localhost/p`, `file:/p`,
1196  /// percent-encoded; `file://server/share/p` on Windows) names, when it exists. A plain path is
1197  /// taken as written: its `%` is a character, not an escape (`/w/100%/LaTeXML-html5.xsl`).
1198  pub fn disk_path(url: &str) -> Option<std::path::PathBuf> {
1199    let path = match url.strip_prefix("file:") {
1200      None => url.to_string(),
1201      Some(rest) => {
1202        let path = match rest.strip_prefix("//") {
1203          Some(auth) => match auth.strip_prefix("localhost").unwrap_or(auth) {
1204            local if local.starts_with('/') => local,
1205            // `file://server/share/p` is the UNC path `//server/share/p`.
1206            _ if cfg!(windows) => rest,
1207            _ => return None,
1208          },
1209          None => rest,
1210        };
1211        let path = percent_decode(path)?;
1212        // `file:///C:/x` names `C:/x`.
1213        match path.strip_prefix('/') {
1214          Some(rest) if cfg!(windows) && rest.as_bytes().get(1) == Some(&b':') => rest.to_string(),
1215          _ => path,
1216        }
1217      },
1218    };
1219    let path = std::path::PathBuf::from(path);
1220    (path.is_absolute() && path.is_file()).then_some(path)
1221  }
1222
1223  fn percent_decode(s: &str) -> Option<String> {
1224    if !s.contains('%') {
1225      return Some(s.to_string());
1226    }
1227    let mut bytes = Vec::with_capacity(s.len());
1228    let mut it = s.bytes();
1229    while let Some(b) = it.next() {
1230      if b == b'%' {
1231        let hex = [it.next()?, it.next()?];
1232        bytes.push(u8::from_str_radix(std::str::from_utf8(&hex).ok()?, 16).ok()?);
1233      } else {
1234        bytes.push(b);
1235      }
1236    }
1237    String::from_utf8(bytes).ok()
1238  }

1240  /// Install the libxml2 input callback that serves `embed:///`
1241  /// URLs from [`FILES`]. Called once per process; subsequent calls
1242  /// are no-ops. The callback fires whenever libxml2 itself opens a
1243  /// URL — including `xsl:import` / `xsl:include` resolution from
1244  /// inside `libxslt::parser::parse_bytes`. Result: every stylesheet
1245  /// (root + imports) is loaded from the binary's own `.rodata`
1246  /// section, no disk extraction required.
1247  pub fn install_callback_once() {
1248    static INSTALLED: OnceLock<()> = OnceLock::new();
1249    INSTALLED.get_or_init(|| {
1250      libxml::io::register_input_callback(
1251        // Serve any URL whose basename is one of our embedded engine files —
1252        // `embed:///X`, a user stylesheet's `urn:x-LaTeXML:XSLT:X`, and the
1253        // relative child imports libxml2 composes from it (`urn:X` on Linux, a
1254        // bare `X` on macOS). A non-engine basename, or a user-named file on disk (#806),
1255        // resolves to None, so the callback declines it and libxml2 loads it from disk (#292).
1256        |url| resolve(url).is_some(),
1257        |url| resolve(url).map(|s| s.to_vec()),
1258      );
1259    });
1260  }

1355/// The stylesheet a name resolves to: an `embed:///` name (the default for a format) is served from
1356/// the binary; a file the user named, found as given or under a search path, is made absolute, the
1357/// form [`embedded_xslt::resolve`] reads from disk with its imports (#806).
1358fn find_stylesheet(stylesheet: &str, searchpaths: &[String]) -> Result<String, PostError> {
1359  if let Some(name) = stylesheet.strip_prefix(embedded_xslt::URL_PREFIX)
1360    && embedded_xslt::lookup(name).is_some()
1361  {
1362    embedded_xslt::install_callback_once();
1363    return Ok(stylesheet.to_string());
1364  }
1365  if let Some(path) = embedded_xslt::disk_path(stylesheet) {
1366    return Ok(path.to_string_lossy().into_owned());
1367  }
1368  let candidates = std::iter::once(PathBuf::from(stylesheet))
1369    .chain(searchpaths.iter().map(|sp| Path::new(sp).join(stylesheet)));
1370  for candidate in candidates {
1371    if candidate.is_file() {
1372      let path = std::path::absolute(&candidate).unwrap_or(candidate);
1373      return Ok(path.to_string_lossy().into_owned());
1374    }
1375  }
1376  // 3. Fallback: serve from the embedded table via the libxml2 input callback. We return an
1377  //    `embed:///<basename>` URL sentinel that `with_cached_stylesheet` routes through
1378  //    `libxslt::parser:: parse_bytes`; subsequent `xsl:import` references inside that stylesheet
1379  //    compose against this base URI and re-enter our callback, so the whole chain stays in memory.
1380  let filename = Path::new(stylesheet)
1381    .file_name()
1382    .and_then(|f| f.to_str())
1383    .unwrap_or(stylesheet);
1384  if embedded_xslt::lookup(filename).is_some() {
1385    embedded_xslt::install_callback_once();
1386    return Ok(format!("{}{}", embedded_xslt::URL_PREFIX, filename));
1387  }
1388  Err(PostError::Processing(format!(
1389    "No stylesheet '{}' found!",
1390    stylesheet
1391  )))
1392}

1394/// Disk-only lookup for a CSS/JS/icon resource — searches the literal
1395/// path, then `info.subdir`-prefixed variants, then each `search_paths`
1396/// entry. Embedded (compile-time-bundled) assets are handled by the
1397/// caller via `embedded_resources::lookup`; this function deliberately
1398/// does NOT check the embed, so on-disk overrides always win and the
1399/// "couldn't find" branch can fall through to the embed cleanly.
1400fn find_resource_file(
1401  src: &str,
1402  info: Option<&ResourceInfo>,
1403  search_paths: &[&str],
1404) -> Option<String> {
1405  let name = Path::new(src).file_name()?.to_str()?;
1406  let mut candidates = vec![src.to_string()];
1407  if let Some(info) = info {
1408    candidates.push(format!("{}/{}", info.subdir, name));
1409    candidates.push(format!("{}/{}", info.subdir, src));
1410  }
1411  for candidate in &candidates {
1412    if Path::new(candidate).is_file() {
1413      return Some(candidate.clone());
1414    }
1415    for sp in search_paths {
1416      let p = format!("{}/{}", sp, candidate);
1417      if Path::new(&p).is_file() {
1418        return Some(p);
1419      }
1420    }
1421  }
1422  None
1423}
