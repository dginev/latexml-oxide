1//! EPUB manifest creation.
2//!
3//! Port of `LaTeXML::Post::Manifest::Epub` (252 lines of Perl).
4//! Creates the EPUB 3.2 package structure:
5//! - `mimetype` file
6//! - `META-INF/container.xml`
7//! - `OPS/content.opf` (spine + manifest)
8//! - Indexes all content files with correct media types
9
10use std::{
11  fs,
12  path::{Path, PathBuf},
13};
14
15/// The OPF package namespace (EPUB 3).
16const OPF_NS: &str = "http://www.idpf.org/2007/opf";
17/// Dublin Core, the metadata vocabulary an OPF's `<metadata>` carries.
18const DC_NS: &str = "http://purl.org/dc/elements/1.1/";
19
20/// EPUB 3.2 container.xml content.
21const CONTAINER_XML: &str = r#"<?xml version="1.0"?>
22<container version="1.0" xmlns="urn:oasis:names:tc:opendocument:xmlns:container">
23    <rootfiles>
24        <rootfile full-path="OPS/content.opf" media-type="application/oebps-package+xml"/>
25   </rootfiles>
26</container>"#;

28/// Error context for a libxml call.
29///
30/// Generic over the error type on purpose: `rust-libxml` returns
31/// `Box<dyn Error>` from some constructors and `String` from others, so one
32/// fixed-argument closure cannot serve both.
33fn ctx<E>(what: impl Into<String>) -> impl Fn(E) -> String {
34  let what = what.into();
35  move |_| format!("couldn't create {what}")
36}

38/// Core Media Types as per EPUB 3.2 spec.
39fn core_media_type(ext: &str) -> &'static str {
40  match ext.to_lowercase().as_str() {
41    "gif" => "image/gif",
42    "jpg" | "jpeg" => "image/jpeg",
43    "png" => "image/png",
44    "svg" => "image/svg+xml",
45    "mp3" => "audio/mpeg",
46    "mp4" | "mpg4" => "audio/mp4",
47    "css" => "text/css",
48    "ttf" => "font/ttf",
49    "otf" => "font/otf",
50    "woff" => "font/woff",
51    "woff2" => "font/woff2",
52    "xhtml" => "application/xhtml+xml",
53    "js" => "text/javascript",
54    "ncx" => "application/x-dtbncx+xml",
55    "smi" | "smil" => "application/smil+xml",
56    "pls" => "application/pls+xml",
57    _ => "application/octet-stream",
58  }
59}

77  /// Initialize the EPUB directory structure.
78  ///
79  /// Port of `Epub::initialize`.
80  pub fn initialize(&mut self) -> Result<(), String> {
81    let dir = &self.site_directory;
82
83    // 1. Create mimetype file
84    let mime_path = format!("{}/mimetype", dir);
85    fs::write(&mime_path, "application/epub+zip")
86      .map_err(|e| format!("Couldn't write mimetype: {}", e))?;
87
88    // 2. Create META-INF/container.xml
89    let meta_inf = format!("{}/META-INF", dir);
90    fs::create_dir_all(&meta_inf).map_err(|e| format!("Couldn't create META-INF: {}", e))?;
91    fs::write(format!("{}/container.xml", meta_inf), CONTAINER_XML)
92      .map_err(|e| format!("Couldn't write container.xml: {}", e))?;
93
94    // 3. Create OPS directory
95    let ops_dir = format!("{}/OPS", dir);
96    fs::create_dir_all(&ops_dir).map_err(|e| format!("Couldn't create OPS: {}", e))?;
97
98    // Generate a UUID for the publication
99    self.unique_identifier = Some(format!("urn:uuid:{}", generate_uuid()));
100
101    Ok(())
102  }

104  /// Port of `Epub::process`: the package for the content documents `documents` (their destinations, in reading
105  /// order) under `OPS/`, every other file there a resource, `content.opf` written unless the user supplied one.
106  pub fn process(&mut self, documents: &[PathBuf], title: Option<&str>) -> Result<(), String> {
107    self.initialize()?;
108    let ops = Path::new(&self.site_directory).join("OPS");
109    let spine: Vec<SpineEntry> = documents
110      .iter()
111      .map(|doc| {
112        let (has_math, has_svg, has_nav) = content_properties(doc);
113        let relative = doc.strip_prefix(&ops).unwrap_or(doc);
114        self.add_document(&relative.to_string_lossy(), has_math, has_svg, has_nav)
115      })
116      .collect();
117    // Epub::finalize: every non-XHTML file under OPS/, in name order.
118    let mut files = Vec::new();
119    collect_files(&ops, &mut files);
120    files.sort();
121    let resources: Vec<ResourceEntry> = files
122      .iter()
123      .filter_map(|path| {
124        let href = path.strip_prefix(&ops).ok()?.to_string_lossy().into_owned();
125        let ext = path.extension().and_then(|e| e.to_str()).unwrap_or("");
126        if ext == "xhtml" || href == "content.opf" || path.ends_with("LaTeXML.cache") {
127          return None;
128        }
129        let media_type = core_media_type(ext);
130        if media_type == "application/octet-stream" {
131          crate::Info!(
132            "unexpected",
133            ext.to_lowercase(),
134            "resource '{}' is not of a core media type, assigning type application/octet-stream",
135            href
136          );
137        }
138        Some(ResourceEntry {
139          id: url_to_id(&href),
140          href,
141          media_type: media_type.to_string(),
142        })
143      })
144      .collect();
145    let content_path = ops.join("content.opf");
146    if content_path.is_file() {
147      crate::Info!(
148        "note",
149        "content.opf",
150        "using the manifest supplied by the user"
151      );
152      return Ok(());
153    }
154    // Perl reads the title from the ObjectDB's SITE_ROOT entry, which carries no authors or language.
155    let opf = self.generate_opf(title.unwrap_or("No Title"), &[], "en", &spine, &resources);
156    fs::write(&content_path, opf).map_err(|e| format!("Couldn't write content.opf: {e}"))
157  }

159  /// Add a document to the EPUB spine; `href` is its path relative to `OPS/`.
160  ///
161  /// Port of `Epub::process` per-document loop.
162  pub fn add_document(
163    &self,
164    href: &str,
165    has_math: bool,
166    has_svg: bool,
167    has_nav: bool,
168  ) -> SpineEntry {
169    let href = href.replace(std::path::MAIN_SEPARATOR, "/");
170    let mut properties = Vec::new();
171    if has_math {
172      properties.push("mathml");
173    }
174    if has_svg {
175      properties.push("svg");
176    }
177    if has_nav {
178      properties.push("nav");
179    }
180
181    SpineEntry {
182      id: url_to_id(&href),
183      href,
184      media_type: "application/xhtml+xml".to_string(),
185      properties: if properties.is_empty() {
186        None
187      } else {
188        Some(properties.join(" "))
189      },
190    }
191  }

193  /// Generate the content.opf package document.
194  ///
195  /// Port of `Epub::finalize`. Built through the libxml DOM rather than by
196  /// `push_str`, because an OPF is parsed by strict readers: one unescaped `&`
197  /// in a single `href` invalidates the whole package, not one entry.
198  ///
199  /// That was reachable. `href` is `format!("{}.{}", name, ext)` over a split
200  /// document's file stem, and `--splitnaming=label` takes that stem from the
201  /// author's `\label{...}` — `\label{Fisher&Yates}` really does produce a file
202  /// named `Fisher&Yates.xhtml`. The old builder escaped 2 of 12 interpolated
203  /// values and `href` was not one of them. libxml escapes on set, so the
204  /// question no longer arises for any of them. (Issue 386 item 2.)
205  pub fn generate_opf(
206    &self,
207    title: &str,
208    authors: &[String],
209    language: &str,
210    spine: &[SpineEntry],
211    resources: &[ResourceEntry],
212  ) -> String {
213    let uid = self
214      .unique_identifier
215      .as_deref()
216      .unwrap_or("urn:uuid:00000000-0000-0000-0000-000000000000");
217
218    match self.build_opf_dom(title, authors, language, uid, spine, resources) {
219      Ok(xml) => xml,
220      // A DOM allocation failure is not something a caller can act on, and an
221      // empty package is a clearer failure than a half-built one.
222      Err(e) => {
223        crate::Error!(
224          "epub",
225          "opf",
226          "Couldn't build the EPUB package document: {}",
227          e
228        );
229        String::new()
230      },
231    }
232  }

234  fn build_opf_dom(
235    &self,
236    title: &str,
237    authors: &[String],
238    language: &str,
239    uid: &str,
240    spine: &[SpineEntry],
241    resources: &[ResourceEntry],
242  ) -> Result<String, String> {
243    use libxml::tree::{Document as XmlDoc, Namespace, Node};
244
245    let mut doc = XmlDoc::new().map_err(|_| "couldn't create the OPF document".to_string())?;
246
247    let mut package = Node::new("package", None, &doc).map_err(ctx("<package>"))?;
248    let opf_ns = Namespace::new("", OPF_NS, &mut package).map_err(ctx("the OPF namespace"))?;
249    package
250      .set_namespace(&opf_ns)
251      .map_err(ctx("the OPF namespace"))?;
252    package
253      .set_attribute("unique-identifier", "pub-id")
254      .map_err(ctx("@unique-identifier"))?;
255    package
256      .set_attribute("version", "3.0")
257      .map_err(ctx("@version"))?;
258    doc.set_root_element(&package);
259
260    // ---- metadata ----
261    let mut metadata = Node::new("metadata", None, &doc).map_err(ctx("<metadata>"))?;
262    // `dc:` is declared HERE, matching the shape readers expect, and the
263    // Dublin Core children below are created in it.
264    let dc_ns = Namespace::new("dc", DC_NS, &mut metadata).map_err(ctx("the dc namespace"))?;
265    package
266      .add_child(&mut metadata)
267      .map_err(ctx("<metadata>"))?;
268
269    // A closure would hold `metadata` borrowed across every call, so the
270    // Dublin Core children are appended inline through one small helper.
271    fn dc_child(
272      doc: &XmlDoc,
273      dc_ns: &Namespace,
274      metadata: &mut Node,
275      name: &str,
276      text: &str,
277    ) -> Result<Node, String> {
278      let mut n = Node::new(name, Some(dc_ns.clone()), doc).map_err(ctx(name))?;
279      n.set_content(text).map_err(ctx(name))?;
280      metadata.add_child(&mut n).map_err(ctx(name))?;
281      Ok(n)
282    }
283    dc_child(&doc, &dc_ns, &mut metadata, "title", title)?;
284    for author in authors {
285      dc_child(&doc, &dc_ns, &mut metadata, "creator", author)?;
286    }
287    dc_child(&doc, &dc_ns, &mut metadata, "language", language)?;
288
289    let mut modified = Node::new("meta", None, &doc).map_err(ctx("<meta>"))?;
290    modified
291      .set_attribute("property", "dcterms:modified")
292      .map_err(ctx("@property"))?;
293    modified
294      .set_content(&chrono_like_now())
295      .map_err(ctx("<meta>"))?;
296    metadata.add_child(&mut modified).map_err(ctx("<meta>"))?;
297
298    let mut identifier = dc_child(&doc, &dc_ns, &mut metadata, "identifier", uid)?;
299    identifier
300      .set_attribute("id", "pub-id")
301      .map_err(ctx("@id"))?;
302
303    // ---- manifest ----
304    let mut manifest = Node::new("manifest", None, &doc).map_err(ctx("<manifest>"))?;
305    package
306      .add_child(&mut manifest)
307      .map_err(ctx("<manifest>"))?;
308    let mut item =
309      |id: &str, href: &str, media_type: &str, properties: Option<&str>| -> Result<(), String> {
310        let mut n = Node::new("item", None, &doc).map_err(ctx("<item>"))?;
311        n.set_attribute("id", id).map_err(ctx("@id"))?;
312        n.set_attribute("href", href).map_err(ctx("@href"))?;
313        n.set_attribute("media-type", media_type)
314          .map_err(ctx("@media-type"))?;
315        if let Some(props) = properties {
316          n.set_attribute("properties", props)
317            .map_err(ctx("@properties"))?;
318        }
319        manifest.add_child(&mut n).map_err(ctx("<item>"))
320      };
321    for entry in spine {
322      item(
323        &entry.id,
324        &entry.href,
325        &entry.media_type,
326        entry.properties.as_deref(),
327      )?;
328    }
329    for res in resources {
330      item(&res.id, &res.href, &res.media_type, None)?;
331    }
332
333    // ---- spine ----
334    let mut spine_el = Node::new("spine", None, &doc).map_err(ctx("<spine>"))?;
335    package.add_child(&mut spine_el).map_err(ctx("<spine>"))?;
336    for entry in spine {
337      let mut itemref = Node::new("itemref", None, &doc).map_err(ctx("<itemref>"))?;
338      itemref
339        .set_attribute("idref", &entry.id)
340        .map_err(ctx("@idref"))?;
341      spine_el.add_child(&mut itemref).map_err(ctx("<itemref>"))?;
342    }
343
344    Ok(doc.to_string())
345  }
346}

365/// Convert a URL/filename to a valid NCName for use as an XML id.
366///
367/// Port of `url_id`: a character outside NCName becomes `_xN_` (N its uppercase hex codepoint), and an `_` that
368/// opens such a sequence is itself escaped first.
369fn url_to_id(name: &str) -> String {
370  let chars: Vec<char> = name.chars().collect();
371  let mut result = String::from("_");
372  for (i, &ch) in chars.iter().enumerate() {
373    if ch == '_' && opens_escape(&chars[i + 1..]) {
374      result.push_str("_x5F_");
375    } else if is_ncname_char(ch) {
376      result.push(ch);
377    } else {
378      result.push_str(&format!("_x{:X}_", ch as u32));
379    }
380  }
381  result
382}

384/// Whether `rest` (the text after an `_`) reads `x<HEX>+_`, Perl's `_(x[0-9A-F]+)(?=_)`.
385fn opens_escape(rest: &[char]) -> bool {
386  let hex = rest
387    .iter()
388    .skip(1)
389    .take_while(|c| matches!(c, '0'..='9' | 'A'..='F'))
390    .count();
391  rest.first() == Some(&'x') && hex > 0 && rest.get(1 + hex) == Some(&'_')
392}

394/// The NCName character class of Perl's `url_id`.
395fn is_ncname_char(ch: char) -> bool {
396  matches!(ch,
397    'A'..='Z' | '_' | 'a'..='z' | '-' | '.' | '0'..='9' | '\u{B7}'
398    | '\u{C0}'..='\u{D6}' | '\u{D8}'..='\u{F6}' | '\u{F8}'..='\u{2FF}' | '\u{300}'..='\u{36F}'
399    | '\u{370}'..='\u{37D}' | '\u{37F}'..='\u{1FFF}' | '\u{200C}'..='\u{200D}' | '\u{203F}'..='\u{2040}'
400    | '\u{2070}'..='\u{218F}' | '\u{2C00}'..='\u{2FEF}' | '\u{3001}'..='\u{D7FF}' | '\u{F900}'..='\u{FDCF}'
401    | '\u{FDF0}'..='\u{FFFD}' | '\u{10000}'..='\u{EFFFF}')
402}

404/// The OPF item properties of a content document: (MathML, SVG, the navigation TOC), Perl `Epub::process`.
405fn content_properties(path: &Path) -> (bool, bool, bool) {
406  let Ok(doc) = libxml::parser::Parser::default().parse_file(&path.to_string_lossy()) else {
407    return (false, false, false);
408  };
409  let Ok(mut xpath) = libxml::xpath::Context::new(&doc) else {
410    return (false, false, false);
411  };
412  let mut has = |query: &str| {
413    xpath
414      .findnodes(query, None)
415      .is_ok_and(|nodes| !nodes.is_empty())
416  };
417  (
418    has(r#"//*[local-name() = "math"]"#),
419    has(r#"//*[local-name() = "svg"]"#),
420    has(r#"//*[@class="ltx_toclist"]"#),
421  )
422}

424/// Every file under `dir`, recursively.
425fn collect_files(dir: &Path, files: &mut Vec<PathBuf>) {
426  let Ok(entries) = fs::read_dir(dir) else {
427    return;
428  };
429  for entry in entries.flatten() {
430    let path = entry.path();
431    if path.is_dir() {
432      collect_files(&path, files);
433    } else if path.is_file() {
434      files.push(path);
435    }
436  }
437}

439/// The text content of `node` without its `ltx:indexmark`s (Perl Scan.pm `cleanNode`, then `textContent`); the
440/// serializer's indentation (blank text in element-only content) is not text.
441fn text_without_indexmarks(node: &libxml::tree::Node) -> String {
442  use libxml::tree::NodeType;
443  let is_text = |n: &libxml::tree::Node| {
444    matches!(
445      n.get_type(),
446      Some(NodeType::TextNode | NodeType::CDataSectionNode)
447    )
448  };
449  let mixed = node
450    .get_child_nodes()
451    .iter()
452    .any(|n| is_text(n) && !n.get_content().trim().is_empty());
453  let mut text = String::new();
454  let mut child = node.get_first_child();
455  while let Some(current) = child {
456    if is_text(&current) && !mixed {
457      child = current.get_next_sibling();
458      continue;
459    }
460    match current.get_type() {
461      Some(NodeType::ElementNode) if current.get_name() != "indexmark" => {
462        text.push_str(&text_without_indexmarks(&current));
463      },
464      Some(NodeType::TextNode | NodeType::CDataSectionNode) => {
465        text.push_str(&current.get_content());
466      },
467      _ => {},
468    }
469    child = current.get_next_sibling();
470  }
471  text
472}

474/// The parts of the core XML an EPUB needs before post-processing.
475pub struct EpubSource {
476  /// The core XML with a navigation TOC, when `check_toc` had to add one.
477  pub xml:   Option<String>,
478  /// The document's title text (Scan's `cleanNode`: its index marks removed).
479  pub title: Option<String>,
480}

482/// Port of `LaTeXML.pm` `check_TOC` (an EPUB requires a TOC: add a full, undisplayed one unless the document has
483/// one) together with the title the manifest reads from the document.
484pub fn prepare_source(core_xml: &str) -> Result<EpubSource, String> {
485  const LTX_NS: &str = "http://dlmf.nist.gov/LaTeXML";
486  let doc = libxml::parser::Parser::default()
487    .parse_string(core_xml)
488    .map_err(|e| format!("couldn't parse the document: {e:?}"))?;
489  let mut root = doc
490    .get_root_element()
491    .ok_or("the document has no root element")?;
492  let mut xpath =
493    libxml::xpath::Context::new(&doc).map_err(|_| "couldn't create an XPath context")?;
494  xpath
495    .register_namespace("ltx", LTX_NS)
496    .map_err(|_| "couldn't register the ltx namespace")?;
497  let title = xpath
498    .findnodes("/*/ltx:title", None)
499    .ok()
500    .and_then(|nodes| nodes.into_iter().next())
501    .map(|title| text_without_indexmarks(&title));
502  let has_toc = xpath
503    .findnodes(r#"//ltx:TOC[@lists="toc"]"#, None)
504    .is_ok_and(|nodes| !nodes.is_empty());
505  let xml = if has_toc {
506    None
507  } else {
508    let ns = root.get_namespace();
509    let mut toc = libxml::tree::Node::new("TOC", ns, &doc).map_err(ctx("<TOC>"))?;
510    for (name, value) in [
511      ("lists", "toc"),
512      ("scope", "global"),
513      (
514        "select",
515        "ltx:part | ltx:chapter | ltx:section | ltx:subsection | ltx:subsubsection | ltx:paragraph | \
516         ltx:subparagraph | ltx:appendix | ltx:index | ltx:bibliography",
517      ),
518      ("class", "ltx_nodisplay"),
519    ] {
520      toc
521        .set_attribute(name, value)
522        .map_err(ctx("a TOC attribute"))?;
523    }
524    match root.get_first_child() {
525      Some(mut first) => first.add_prev_sibling(&mut toc).map_err(ctx("<TOC>"))?,
526      None => root.add_child(&mut toc).map(|_| ()).map_err(ctx("<TOC>"))?,
527    };
528    Some(doc.to_string())
529  };
530  Ok(EpubSource { xml, title })
531}

533/// Generate a UUID v4 string.
534fn generate_uuid() -> String {
535  // Simple random UUID v4 (no external dependency)
536  use std::time::{SystemTime, UNIX_EPOCH};
537  let seed = SystemTime::now()
538    .duration_since(UNIX_EPOCH)
539    .unwrap_or_default()
540    .as_nanos();
541  // LCG-based pseudo-random for simplicity
542  let mut state = seed as u64;
543  let mut bytes = [0u8; 16];
544  for b in &mut bytes {
545    state = state
546      .wrapping_mul(6364136223846793005)
547      .wrapping_add(1442695040888963407);
548    *b = (state >> 33) as u8;
549  }
550  bytes[6] = (bytes[6] & 0x0F) | 0x40; // version 4
551  bytes[8] = (bytes[8] & 0x3F) | 0x80; // variant 1
552  format!(
553    "{:02x}{:02x}{:02x}{:02x}-{:02x}{:02x}-{:02x}{:02x}-{:02x}{:02x}-{:02x}{:02x}{:02x}{:02x}{:02x}{:02x}",
554    bytes[0],
555    bytes[1],
556    bytes[2],
557    bytes[3],
558    bytes[4],
559    bytes[5],
560    bytes[6],
561    bytes[7],
562    bytes[8],
563    bytes[9],
564    bytes[10],
565    bytes[11],
566    bytes[12],
567    bytes[13],
568    bytes[14],
569    bytes[15]
570  )
571}

573/// Generate an ISO 8601 timestamp (CCYY-MM-DDThh:mm:ssZ), at `SOURCE_DATE_EPOCH` when set (Perl Epub.pm).
574fn chrono_like_now() -> String {
575  use std::time::{SystemTime, UNIX_EPOCH};
576  let secs = std::env::var("SOURCE_DATE_EPOCH")
577    .ok()
578    .and_then(|s| s.trim().parse::<u64>().ok())
579    .unwrap_or_else(|| {
580      SystemTime::now()
581        .duration_since(UNIX_EPOCH)
582        .unwrap_or_default()
583        .as_secs()
584    });
585  // Simple UTC timestamp computation (no chrono dependency)
586  let days = secs / 86400;
587  let time_of_day = secs % 86400;
588  let hours = time_of_day / 3600;
589  let minutes = (time_of_day % 3600) / 60;
590  let seconds = time_of_day % 60;
591  // Rough date from epoch days (good enough for timestamps)
592  let (year, month, day) = days_to_ymd(days as i64);
593  format!(
594    "{:04}-{:02}-{:02}T{:02}:{:02}:{:02}Z",
595    year, month, day, hours, minutes, seconds
596  )
597}
598
599/// Convert days since Unix epoch to (year, month, day).
600fn days_to_ymd(days: i64) -> (i64, u32, u32) {
601  // Algorithm from https://howardhinnant.github.io/date_algorithms.html
602  let z = days + 719468;
603  let era = if z >= 0 { z } else { z - 146096 } / 146097;
604  let doe = (z - era * 146097) as u32;
605  let yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
606  let y = yoe as i64 + era * 400;
607  let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
608  let mp = (5 * doy + 2) / 153;
609  let d = doy - (153 * mp + 2) / 5 + 1;
610  let m = if mp < 10 { mp + 3 } else { mp - 9 };
611  (y + if m <= 2 { 1 } else { 0 }, m, d)
612}

627  /// The package document must be well-formed for **author-controlled** input.
628  ///
629  /// Issue 386 item 2. An OPF is read by strict parsers, so one unescaped `&`
630  /// invalidates the whole book rather than one entry — and `&` is reachable:
631  /// `href` is built from a split document's file stem, and `--splitnaming=label`
632  /// takes that stem from the author's `\label{...}`, so `\label{Fisher&Yates}`
633  /// yields a file named `Fisher&Yates.xhtml`. The old `push_str` builder escaped
634  /// 2 of 12 interpolated values and `href` was not among them.
635  ///
636  /// The assertion is deliberately "it parses", not "it contains `&amp;`":
637  /// well-formedness is the property that matters, and it cannot be satisfied by
638  /// accident.
639  #[test]
640  fn opf_stays_well_formed_when_author_input_carries_xml_specials() {
641    let mut manifest = EpubManifest::new("/tmp/does-not-need-to-exist");
642    manifest.unique_identifier = Some("urn:uuid:test".to_string());
643
644    let spine = vec![
645      spine_entry(
646        "_Fisher_x26_Yates.xhtml",
647        "Fisher&Yates.xhtml",
648        Some("mathml"),
649      ),
650      spine_entry("_quote_x22_.xhtml", "a\"quote\".xhtml", None),
651      spine_entry("_lt_x3C_.xhtml", "a<less>.xhtml", None),
652    ];
653    let resources = vec![ResourceEntry {
654      id:         "_css".to_string(),
655      href:       "LaTeXML&core.css".to_string(),
656      media_type: "text/css".to_string(),
657    }];
658
659    let opf = manifest.generate_opf(
660      "Tom & Jerry: a <study> of \"conflict\"",
661      &["Ampersand & Co.".to_string()],
662      "en",
663      &spine,
664      &resources,
665    );
666
667    let parser = libxml::parser::Parser::default();
668    let doc = parser.parse_string(&opf).unwrap_or_else(|e| {
669      panic!("the OPF is not well-formed XML ({e:?}) — an unescaped special reached it:\n{opf}")
670    });
671    let root = doc.get_root_element().expect("a root element");
672    assert_eq!(root.get_name(), "package");
673
674    // The values must round-trip to their ORIGINAL text, not to an escaped or
675    // truncated form — well-formedness alone would also be satisfied by dropping
676    // the offending characters.
677    let hrefs: Vec<String> = crate::document::element_children(
678      &crate::document::element_children(&root)
679        .into_iter()
680        .find(|n| n.get_name() == "manifest")
681        .expect("a <manifest>"),
682    )
683    .iter()
684    .filter_map(|n| n.get_attribute("href"))
685    .collect();
686    assert!(
687      hrefs.iter().any(|h| h == "Fisher&Yates.xhtml"),
688      "the ampersand href did not survive the round-trip: {hrefs:?}"
689    );
690    assert!(
691      hrefs.iter().any(|h| h == "a\"quote\".xhtml") && hrefs.iter().any(|h| h == "a<less>.xhtml"),
692      "a quote/less-than href did not survive the round-trip: {hrefs:?}"
693    );
694    assert!(
695      hrefs.iter().any(|i| i == "LaTeXML&core.css"),
696      "a RESOURCE href is interpolated by the same code path and must be escaped too: {hrefs:?}"
697    );
698  }
