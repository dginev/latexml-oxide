1//! The resource cache: the names the generated and copied resources of one destination directory take, kept across
2//! the split pages of a run and across runs into the same directory.
3//!
4//! Port of the part of Perl's document cache the resource names live in ([`crate::document::DocCache`], persisted as
5//! [`crate::document::CACHE_FILE`] in the destination directory): `generateResourcePathname`'s counter under
6//! `_max_<subdir>_<prefix>_counter_` (`Post.pm` L192-200) and Graphics' `key → "dest|w|h"` entries
7//! (`Post/Graphics.pm` L245-262, L331). A name's owner is its source (path relative to the destination directory,
8//! page, crop and turn, type; never a size), as a Perl key is: our entries `_owner_<name>`, `_name_<owner>` and
9//! `_source_<name>` keep it, and `_holds_<name>` the bytes the name was last written with, so an edited image is a new
10//! key (the content digest replaces Perl's timestamp check) written again into the name its source owns.
11//! `_content_<bytes>` lets a source with the same bytes share that file while its owner is named in the same run with
12//! them (the pages of one document); another document's file is never shared, as that document may write other bytes
13//! into it later, under a page this run does not rewrite. Divergences (OXIDIZED_DESIGN_DIVERGENCES #1157): an entry
14//! stores the name only (Perl's width and height let it skip a conversion, which the content-addressed
15//! `graphics_cache` serves here); a converted or generated name is kept across an edit (Perl generates a new `xN`);
16//! a kept name (a copy's relative path, a converted image's stem) another source owns falls back to a generated `xN`,
17//! where Perl overwrote it (issue 995); and a stored name is written only when the request could have been given it
18//! ([`Checks`]). Each [`resolve`] is one `IMMEDIATE` transaction, so processes naming into one folder at once through
19//! the file never hand out one name to two sources. A name new to the store is never given over a file in the folder
20//! the request may not write ([`Checks::on_disk`]): with the store on its file (a new one too, made where a cache was
21//! deleted) any file it does not own; in memory in place of its file (it could not be opened, or the transaction
22//! failed and runs again there) a file holding other bytes than the request writes, so a rerun takes its own files
23//! again. One that could not copy the file's entries does not know a file's owner: a file of another document holding
24//! the same bytes is taken as the request's own (OD #1157 (10)).

32/// What a resource's name is first tried as.
33#[derive(Clone, Debug, PartialEq, Eq)]
34pub enum Proposal {
35  /// A copy's path relative to the destination directory, kept when no other source owns it
36  /// (`Post/Graphics.pm` L292-300).
37  Kept(String),
38  /// A converted image's source stem (`plot` for `a/plot.pdf`), with the destination type appended: a readability
39  /// name Perl does not keep (it generates `xN` for every conversion, `Post/Graphics.pm` L318-319; witness
40  /// 2606.30620).
41  Stem(String),
42  /// A generated `<prefix>N` (`generateResourcePathname`).
43  Generated,
44}

46/// One resource to name.
47#[derive(Clone, Debug)]
48pub struct Request {
49  /// Perl's cache key (`ref($self):reldir/name.srctype.type|transform`) with the source's content digest.
50  pub key:      String,
51  /// What the destination file's bytes are made of (the source's bytes, page, type, crop and turn, never a size):
52  /// requests with the same bytes share one file.
53  pub bytes:    String,
54  /// Who owns the names the request is given: its source, transform and type, so an edited source keeps them.
55  pub owner:    String,
56  /// The source, relative to the destination directory, kept with the names it owns ([`Checks::vouch`]).
57  pub source:   String,
58  /// The source's content digest (hex), or its path when unreadable.
59  pub identity: String,
60  pub proposal: Proposal,
61  /// The destination's file type (extension), empty for none.
62  pub ext:      String,
63  /// A vector render `<name>.svg` may be written beside it, so that name is claimed too.
64  pub svg_alt:  bool,
65  /// A copy (named after the conversions of its page when it cannot keep its path) or a conversion.
66  pub copy:     bool,
67  /// What the destination file will hold ([`Checks::on_disk`]).
68  pub output:   Output,
69}

89/// How a request was named.
90#[derive(Clone, Debug)]
91pub struct Resolved {
92  pub name:      ResourceName,
93  /// A name not stored under the request's key: a kept path, a stem, a generated one, or its source's earlier name.
94  pub allocated: bool,

97/// Whether a file of the name exists in the destination directory that the request may not write ([`Checks::on_disk`]).
98pub type OnDisk<'a> = &'a dyn Fn(&Request, &ResourceName) -> bool;
99
100/// What the caller knows of the destination directory.
101pub struct Checks<'a> {
102  /// Perl's `resource_prefix` (`x` unless set).
103  pub prefix:  &'a str,
104  /// Whether a name may be written: a copy's kept path outside this run's site directory may not.
105  pub usable:  &'a dyn Fn(&ResourceName) -> bool,
106  /// Whether `name`, owned by the source at `source` (relative to the destination directory), is the name that source
107  /// takes and the source holds the request's bytes: another source's kept path or stem a request may share. Only a
108  /// relative path to a regular file is read.
109  pub vouch:   &'a dyn Fn(&Request, &ResourceName, &str) -> bool,
110  /// Whether a file of the name exists in the destination directory that the request may not write, asked of a name
111  /// the store does not know: the store may not know every file earlier runs wrote (a deleted cache, an unusable one).
112  /// `None` for a store in memory from the start (`--nocache`), whose names restart and overwrite as Perl's do.
113  pub on_disk: Option<OnDisk<'a>>,
114  /// The cache file a Warning names.
115  pub cache:   &'a str,
116  /// Report a Warning (`unexpected`, `cache`): the caller emits it once the transaction is done, as a transaction run
117  /// again in memory meets the same entries.
118  pub warn:    &'a dyn Fn(String),
119}

121/// The counter Perl's `generateResourcePathname` keeps for `prefix` (`Post.pm` L194-196; no `resource_directory`).
122pub fn counter_key(prefix: &str) -> String { format!("_max__{prefix}_counter_") }

124/// Name every request, page by page, on `conn` (in the caller's `IMMEDIATE` transaction): a name stored for its key,
125/// its source or its content, else, within a page, the copies that keep their path first, then the conversions, then
126/// the copies that take a generated name, each in node order (the order the page's names were handed out in before).
127pub fn resolve(
128  conn: &Connection,
129  checks: &Checks,
130  pages: &[Vec<Request>],
131) -> rusqlite::Result<Vec<Vec<Resolved>>> {
132  let counter = counter_key(checks.prefix);
133  // The sources this run names, with their bytes: only their files are shared, while they hold the same bytes.
134  let now: HashMap<&str, &str> = pages
135    .iter()
136    .flatten()
137    .map(|r| (r.owner.as_str(), r.bytes.as_str()))
138    .collect();
139  let mut out = Vec::with_capacity(pages.len());
140  for page in pages {
141    let mut names: Vec<Option<Resolved>> = vec![None; page.len()];
142    // Copies keeping their path.
143    for (i, r) in page.iter().enumerate() {
144      if let (true, Proposal::Kept(rel)) = (r.copy, &r.proposal) {
145        names[i] = place(conn, checks, &now, r, Some(name_of(rel, false)), true)?;
146      }
147    }
148    // Conversions, then the copies left.
149    for copies in [false, true] {
150      for (i, r) in page.iter().enumerate() {
151        if r.copy != copies || names[i].is_some() {
152          continue;
153        }
154        let proposed = match &r.proposal {
155          Proposal::Stem(stem) if !copies => Some(name_of(&with_ext(stem, &r.ext), r.svg_alt)),
156          _ => None,
157        };
158        // A copy whose kept path is taken looks its stored names up again (a copy since may share its bytes), its
159        // corrupt ones reported already.
160        let warn = !matches!((r.copy, &r.proposal), (true, Proposal::Kept(_)));
161        names[i] = match place(conn, checks, &now, r, proposed, warn)? {
162          Some(found) => Some(found),
163          None => Some(generate(conn, checks, r, &counter)?),
164        };
165      }
166    }
167    out.push(names.into_iter().flatten().collect());
168  }
169  Ok(out)
170}

172fn with_ext(stem: &str, ext: &str) -> String {
173  if ext.is_empty() {
174    stem.to_string()
175  } else {
176    format!("{stem}.{ext}")
177  }
178}

180/// `dest`, and the `.svg` of its stem when a vector render may be written beside it.
181fn name_of(dest: &str, svg_alt: bool) -> ResourceName {
182  let stem = dest.rsplit_once('.').map_or(dest, |(stem, _)| stem);
183  ResourceName {
184    dest: dest.to_string(),
185    alt:  svg_alt.then(|| format!("{stem}.svg")),
186  }
187}

189/// The name stored under the request's key, its source's or its content's, else `proposed` when no other source
190/// owns it, else `None`. A stored name no request could have is reported when `warn`.
191fn place(
192  conn: &Connection,
193  checks: &Checks,
194  now: &HashMap<&str, &str>,
195  r: &Request,
196  proposed: Option<ResourceName>,
197  warn: bool,
198) -> rusqlite::Result<Option<Resolved>> {
199  // Perl's value, `dest|w|h`.
200  let by_key =
201    kv_get(conn, &r.key)?.map(|value| value.split('|').next().unwrap_or_default().to_string());
202  let by_owner = kv_get(conn, &format!("_name_{}", r.owner))?;
203  let by_content = kv_get(conn, &format!("_content_{}", r.bytes))?;
204  // The first two are names this source was given: one no request could have is a corrupt or planted entry.
205  let mut tried: Vec<String> = Vec::new();
206  for (dest, given, keyed) in [
207    (by_key, true, true),
208    (by_owner, true, false),
209    (by_content, false, false),
210  ] {
211    let Some(dest) = dest.filter(|dest| !tried.contains(dest)) else {
212      continue;
213    };
214    tried.push(dest.clone());
215    let name = name_of(&dest, r.svg_alt);
216    let Some(own) = reusable(conn, checks, now, r, &name, warn && given)? else {
217      continue;
218    };
219    if own {
220      take(conn, r, &name)?;
221    } else {
222      share(conn, r, &name)?;
223    }
224    return Ok(Some(Resolved { name, allocated: !keyed }));
225  }
226  let Some(name) = proposed else {
227    return Ok(None);
228  };
229  // A kept path or a stem the store does not know may be a file of another document's (issue 995 review: an unusable
230  // cache, and a second document's `figs/a.png` written over the first's).
231  let taken = claimed(conn, &name, Some(&r.owner))?
232    || checks.on_disk.is_some_and(|on_disk| on_disk(r, &name));
233  if taken {
234    return Ok(None);
235  }
236  take(conn, r, &name)?;
237  Ok(Some(Resolved { name, allocated: true }))
238}

240/// Whether a stored `name` serves `r`: `Some(true)` when `r`'s source owns it (or no source does and `r` could be
241/// given it), `Some(false)` when `r` shares the file of another source of this run holding its bytes, `None` when it
242/// does not serve. A name no request of `r`'s type could have is reported when `warn` (a Warning) and never written.
243fn reusable(
244  conn: &Connection,
245  checks: &Checks,
246  now: &HashMap<&str, &str>,
247  r: &Request,
248  name: &ResourceName,
249  warn: bool,
250) -> rusqlite::Result<Option<bool>> {
251  let given = own_name(r, &name.dest) || generated_name(checks, r, &name.dest);
252  if !given && !shareable_name(r, &name.dest) {
253    if warn {
254      (checks.warn)(format!(
255        "The cache {} names {} for {}, a name no {} image is given; it is ignored",
256        checks.cache, name.dest, r.source, r.ext
257      ));
258    }
259    return Ok(None);
260  }
261  if !(checks.usable)(name) {
262    return Ok(None);
263  }
264  // A source owns only its kept path, its stem or a generated name.
265  let owner = kv_get(conn, &format!("_owner_{}", name.dest))?;
266  let Some(owner) = owner.filter(|owner| *owner != r.owner) else {
267    return Ok(given.then_some(true));
268  };
269  // Another source's file, only while that source is named in this run with these bytes: a source of another
270  // document (or not in this run) may write other bytes into it later, under a page this run does not rewrite.
271  let holds = kv_get(conn, &format!("_holds_{}", name.dest))?;
272  if holds.as_deref() != Some(r.bytes.as_str())
273    || now.get(owner.as_str()) != Some(&r.bytes.as_str())
274  {
275    return Ok(None);
276  }
277  if given {
278    return Ok(Some(false));
279  }
280  let source = kv_get(conn, &format!("_source_{}", name.dest))?.unwrap_or_default();
281  Ok((checks.vouch)(r, name, &source).then_some(false))
282}

284/// Its kept path or its stem.
285fn own_name(r: &Request, dest: &str) -> bool {
286  match &r.proposal {
287    Proposal::Kept(rel) => dest == rel,
288    Proposal::Stem(stem) => dest == with_ext(stem, &r.ext),
289    Proposal::Generated => false,
290  }
291}

293/// A `<prefix>N` of its type.
294fn generated_name(checks: &Checks, r: &Request, dest: &str) -> bool {
295  let stem = if r.ext.is_empty() {
296    Some(dest)
297  } else {
298    dest.strip_suffix(&format!(".{}", r.ext))
299  };
300  stem
301    .and_then(|stem| stem.strip_prefix(checks.prefix))
302    .is_some_and(|n| !n.is_empty() && n.bytes().all(|b| b.is_ascii_digit()))
303}

305/// A name another source of the same bytes may have: of `r`'s type, no hidden file or directory on its path.
306fn shareable_name(r: &Request, dest: &str) -> bool {
307  let family = |ext: &str| match ext.to_ascii_lowercase().as_str() {
308    "jpeg" => "jpg".to_string(),
309    "tiff" => "tif".to_string(),
310    ext => ext.to_string(),
311  };
312  let ext = std::path::Path::new(dest)
313    .extension()
314    .map_or(String::new(), |e| e.to_string_lossy().into_owned());
315  !r.ext.is_empty()
316    && family(&ext) == family(&r.ext)
317    && dest
318      .split(['/', '\\'])
319      .all(|part| !part.is_empty() && (part == ".." || !part.starts_with('.')))
320}

322/// The next `<prefix>N` no source owns (Perl `generateResourcePathname`, which skips none) and whose file the request
323/// may write ([`Checks::on_disk`]). A counter that is no count, or has no next, is reported and counted again from 0.
324fn generate(
325  conn: &Connection,
326  checks: &Checks,
327  r: &Request,
328  counter: &str,
329) -> rusqlite::Result<Resolved> {
330  // A planted `u64::MAX` wrapped to `x0` (issue 995 review, round 7).
331  let unusable = |value: &str| {
332    (checks.warn)(format!(
333      "The cache {} holds {value} for {counter}, no count with a next one; names are counted again from {}1",
334      checks.cache, checks.prefix
335    ))
336  };
337  let stored = kv_get(conn, counter)?;
338  let mut n: u64 = match stored.as_deref().map(str::parse) {
339    None => 0,
340    Some(Ok(n)) => n,
341    Some(Err(_)) => {
342      unusable(stored.as_deref().unwrap_or_default());
343      0
344    },
345  };
346  let name = loop {
347    n = n.checked_add(1).unwrap_or_else(|| {
348      unusable(&n.to_string());
349      1
350    });
351    let name = name_of(
352      &with_ext(&format!("{}{n}", checks.prefix), &r.ext),
353      r.svg_alt,
354    );
355    if !claimed(conn, &name, None)? && !checks.on_disk.is_some_and(|on_disk| on_disk(r, &name)) {
356      break name;
357    }
358  };
359  kv_put(conn, counter, Some(&n.to_string()))?;
360  take(conn, r, &name)?;
361  Ok(Resolved { name, allocated: true })
362}

364/// Whether another source than `owner` (any, for `None`) owns one of `name`'s files.
365fn claimed(conn: &Connection, name: &ResourceName, owner: Option<&str>) -> rusqlite::Result<bool> {
366  for part in std::iter::once(&name.dest).chain(&name.alt) {
367    if let Some(found) = kv_get(conn, &format!("_owner_{part}"))?
368      && owner != Some(found.as_str())
369    {
370      return Ok(true);
371    }
372  }
373  Ok(false)
374}

376/// `r`'s source owns `name`, which holds `r`'s bytes.
377fn take(conn: &Connection, r: &Request, name: &ResourceName) -> rusqlite::Result<()> {
378  share(conn, r, name)?;
379  for part in std::iter::once(&name.dest).chain(&name.alt) {
380    kv_put(conn, &format!("_owner_{part}"), Some(&r.owner))?;
381  }
382  kv_put(conn, &format!("_holds_{}", name.dest), Some(&r.bytes))?;
383  kv_put(conn, &format!("_source_{}", name.dest), Some(&r.source))?;
384  kv_put(conn, &format!("_name_{}", r.owner), Some(&name.dest))
385}

387/// `r`'s key and content name `name`.
388fn share(conn: &Connection, r: &Request, name: &ResourceName) -> rusqlite::Result<()> {
389  kv_put(conn, &r.key, Some(&format!("{}||", name.dest)))?;
390  kv_put(conn, &format!("_content_{}", r.bytes), Some(&name.dest))
391}

461  /// source owns takes the next `xN`; one content, one file; a key met again keeps its name.
462  #[test]

464    let cache = DocCache::new(false);
465    let pages = vec![

468        convert("a/plot.pdf", "a", Some("plot"), true),

476    ];

648  /// Two processes allocating into one cache file at once never hand out one name twice.

650  fn concurrent_tables_on_one_file_never_share_a_name() {
651    let dir = tempfile::tempdir().unwrap();

653    let names: Vec<Vec<String>> = std::thread::scope(|s| {
654      let handles: Vec<_> = (0..8)
