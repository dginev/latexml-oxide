1//! Writer for Rust-native kernel dump files.
2//!
3//! Serializes the State diff (from snapshot/diff_from_snapshot) into a
4//! compact text format that can be loaded back efficiently.
5//!
6//! This is the "DumpFile" equivalent from Perl's TeX_Job.pool.ltxml.
7//! The workflow follows Perl's Makefile.PL "make formats" scaffold:
8//!   1. take_snapshot() — before processing latex.ltx
9//!   2. Process latex.ltx through the engine
10//!   3. diff_snapshot() — find what changed
11//!   4. write_dump() — serialize the diff
12//!   5. At runtime: load_dump() restores the state
13
14use std::{io::Write, path::Path};
15
16use crate::definition::Definition; // trait for get_expansion(), get_cs(), etc.
17use crate::{
18  common::{arena, numeric_ops::NumericOps, store::Stored},
19  state::TableName,
20};
21
22/// Write a state diff to a dump file.
23/// Returns the number of entries successfully written.
24///
25/// Writes in three ordered sections, matching Perl's
26/// `TeX_Job.pool.ltxml::DumpFile` which separates
27/// `@cmds_early` / `@cmds` / `@cmds_late`:
28///
29///   1. **cmds_early** — M:PA / M:MPA let-aliases that are still the **bootstrap snapshot's**
30///      definition of their target (e.g. `\tex_let:D → \let`, `\tex_def:D → \def`,
31///      `\tex_jobname:D → \jobname`). Applied first because their targets are always available
32///      (they're in the bootstrap pool we loaded unconditionally before this dump).
33///   2. **cmds** (regular) — V / M:E / M:T / R / C / LC / UC / SC / MC / DC entries: data,
34///      expandable definitions, registers, codes — including a `\let` copy of a token-bodied
35///      macro whose target was redefined since, written by value.
36///   3. **cmds_late** — M:PA / M:MPA let-aliases that are the target's **current** definition,
37///      defined by this dump (e.g. `\bool_if_exist:NTF → \cs_if_exist:NTF`). Applied last so the
38///      target is installed before the alias fires.
39///
40/// The early/late split requires the caller (ini_tex) to have staged a
41/// snapshot via `state::stage_snapshot("bootstrap")`. If no snapshot is
42/// available, all aliases are emitted as `late` (safe — they just wait
43/// until after regular entries).
44///
45/// `sources` are header lines naming the format files the dump was built from
46/// (`latexml_engine::dump_paths::SourceStamp::header_line`), which a loader checks against the TeX tree
47/// it reads packages from.
48pub fn write_dump(
49  path: &Path,
50  entries: &[(TableName, arena::SymStr, Stored)],
51  sources: &[String],
52) -> Result<usize, String> {
53  let mut file =
54    std::fs::File::create(path).map_err(|e| format!("Failed to create dump file: {}", e))?;
55
56  writeln!(file, "# latexml-oxide kernel dump v3").ok();
57  writeln!(
58    file,
59    "# Generated by dump_writer, {} entries",
60    entries.len()
61  )
62  .ok();
63  for source in sources {
64    writeln!(file, "{source}").ok();
65  }
66  writeln!(file, "# Format: <table>\\t<key>\\t<data>").ok();
67  writeln!(file, "#   <table> = M | V | C | LC | UC | SC | MC | DC").ok();
68  writeln!(file, "#   <data> depends on <table>:").ok();
69  writeln!(
70    file,
71    "#     M/E:   E\\t<cs>\\t<nargs>\\t<flags>\\t<body>\\t<proto>\\t<v3_params>"
72  )
73  .ok();
74  writeln!(
75    file,
76    "#            v3_params encodes each Parameter structurally so Until:/Match:"
77  )
78  .ok();
79  writeln!(
80    file,
81    "#            delimiter tokens round-trip (see docs/archive/DUMP_FORMAT_PERL_ANALYSIS_2026-04-30.md)."
82  )
83  .ok();
84  writeln!(
85    file,
86    "#     M/PA:  PA\\t<target_cs>   (MPA for math primitives)"
87  )
88  .ok();
89  writeln!(file, "#     M/T:   T\\t<meaning_token>").ok();
90  writeln!(
91    file,
92    "#     M/R:   R\\t<cs>\\t<rtype>\\t<value>[\\t<mathglyph>]"
93  )
94  .ok();
95  writeln!(
96    file,
97    "#     M/N:   N                     (None-meaning sentinel)"
98  )
99  .ok();
100  writeln!(
101    file,
102    "#     V:     <type>\\t<value>       (B,I,S,CH,T,TK,N,F,G,MG,D,MD,VD)"
103  )
104  .ok();
105  writeln!(file, "#     C:                  CC\\t<u8>   (catcode)").ok();
106  writeln!(
107    file,
108    "#     DC/LC/UC/SC/MC:     CH\\t<u32>  (char-indexed table entries)"
109  )
110  .ok();
111  writeln!(
112    file,
113    "#   <flags> = subset of {{L, P, T}} (long, protected, checks \\par: tex.web §392)"
114  )
115  .ok();
116  writeln!(
117    file,
118    "#   All keys and proto are url-encoded (%09 for tab etc.); token body"
119  )
120  .ok();
121  writeln!(
122    file,
123    "#   tokens use the form <catcode_int>:<url_encoded_text> joined by commas."
124  )
125  .ok();
126
127  // Read the bootstrap snapshot so we can classify each alias's target
128  // as pre-existing (→ early) or defined-during-load (→ late).
129  let bootstrap_snap = crate::state::get_staged_snapshot("bootstrap");
130
131  // Pre-serialize + partition. Skipped keys (ver@..., unserializable
132  // values) fall out naturally.
133  let mut early_aliases: Vec<(String, String, String)> = Vec::new();
134  let mut regular: Vec<(String, String, String)> = Vec::new();
135  let mut late_aliases: Vec<(String, String, String)> = Vec::new();
136  let mut skipped = 0usize;
137
138  // Perl TeX_Job.pool.ltxml:150,153 walks `sort @tables` and, in each, `sort keys %$table`, so
139  // every section of its dump is in (table name, key) order. The diff arrives in hash-table order,
140  // which differs from one binary to the next (det354: one key set in 55 orders across 173 dumps),
141  // and the dump is applied in file order, so the order also fixes the symbol ids of every
142  // conversion. Sorting here makes the dump a function of its content alone. A table's `Display`
143  // name is Perl's capitalized (`Catcode` .. `Value`), so it sorts as Perl's `@tables` do.
144  let mut ordered: Vec<(String, String, &(TableName, arena::SymStr, Stored))> = entries
145    .iter()
146    .map(|entry| (entry.0.to_string(), arena::to_string(entry.1), entry))
147    .collect();
148  ordered.sort_unstable_by(|a, b| (&a.0, &a.1).cmp(&(&b.0, &b.1)));
149
150  for (_, key_str, (table, _, value)) in ordered {
151    let table_code = table_to_code(*table);
152
153    // Skip \ver@ package version markers (runtime metadata, not definitions)
154    if key_str.starts_with("\\ver@") {
155      skipped += 1;
156      continue;
157    }
158
159    // Skip MAX_ERRORS — `ini_tex.rs:189` raises it to 1_000_000 during
160    // dump-build (so raw latex.ltx + expl3-code.tex can run through
161    // transient errors). It must NOT leak into runtime conversions:
162    // a 1M cap lets runaway error cascades (e.g. AmS-TeX `\cases` mis-
163    // parse → 1M `\hbox`/`&` per paper, math0205073) bypass the
164    // 10000 default. dump_reader has a parallel filter for the
165    // already-shipped dumps; this filter keeps future regenerations
166    // clean.
167    if matches!(*table, TableName::Value) && key_str == "MAX_ERRORS" {
168      skipped += 1;
169      continue;
170    }
171    // Skip the files a run read as content (`binding::content::CONTENT_FILES`): the format's own sources are no
172    // document's text, and a dumped list would stand in a conversion's for the files it reads.
173    if matches!(*table, TableName::Value) && key_str == crate::binding::content::CONTENT_FILES {
174      skipped += 1;
175      continue;
176    }
177
178    // Skip the file-IO bookkeeping family (`is_file_io_bookkeeping`). The
179    // dump is applied WHILE the document's preload is being read (the
180    // LaTeX pool+dump load runs inside `latexml.sty`), so any per-load
181    // state it carries clobbers the live values: a dumped
182    // `\@currnamestack` = empty made `latexml.sty`'s `\@popfilename` pop
183    // nothing, `\catcode`\@` then read an empty number ("Missing number")
184    // and every later `@`-name broke — a4wide 3 → 35 errors, every
185    // siunitx tabular (2026-09-17, cortex dump regeneration).
186    if matches!(*table, TableName::Meaning) && is_file_io_bookkeeping(&key_str) {
187      skipped += 1;
188      continue;
189    }
190
191    // Skip token-list register VALUES whose body contains a self-`\the`
192    // reference. latex.ltx L10564-10580 sets `\frozen@everymath` =
193    // `{... \the\everymath}` then allocates a NEW `\everymath` via
194    // `\newtoks`. Our `\newtoks` doesn't fully detach the new register's
195    // slot from the frozen alias's, so the body lands on `\everymath`'s
196    // value slot — `\the\everymath` then expands recursively at math
197    // entry, blowing the token limit. Until the `\newtoks`/`\let`
198    // slot-aliasing is fixed at the dialect level, the safest cure is
199    // to drop the corrupted self-referential capture; runtime
200    // re-evaluation of latex.ltx's hook chain (or LaTeXML's own
201    // `\everymath`/`\everydisplay` setup) repopulates correctly.
202    if matches!(*table, TableName::Value)
203      && matches!(
204        key_str.as_str(),
205        "\\everymath"
206          | "\\everydisplay"
207          | "\\everyhbox"
208          | "\\everyvbox"
209          | "\\everycr"
210          | "\\everyjob"
211          | "\\everypar"
212          | "\\everyeof"
213      )
214    {
215      // Confirm the loop pattern before dropping (don't suppress
216      // valid hook contributions on engines with sound aliasing).
217      if let Stored::Tokens(tks) = value {
218        let body = tks.unlist_ref();
219        let needle_cs = key_str.clone();
220        let has_self_the = body.iter().any(|t| t.with_str(|s| s == "\\the"))
221          && body.iter().any(|t| t.with_str(|s| s == needle_cs));
222        if has_self_the {
223          skipped += 1;
224          continue;
225        }
226      }
227    }
228
229    // Perl #2771 (2026-03-13) + upstream IGNORED_SYMBOLS: these are
230    // runtime-only Value entries — control-flow counters, and large
231    // registered-rule tables that can't meaningfully round-trip through
232    // the dump (they hold closures, cross-refs, or re-populate during
233    // engine init anyway). Match Perl's TeX_Job.pool.ltxml list.
234    if matches!(*table, TableName::Value)
235      && matches!(
236        key_str.as_str(),
237        "if_stack"
238          | "DOCUMENT_REWRITE_RULES"
239          | "PARAMETER_TYPES"
240          | "TAG_PROPERTIES"
241          | "MATH_LIGATURES"
242          | "TEXT_LIGATURES"
243      )
244    {
245      skipped += 1;
246      continue;
247    }
248
249    // Perl IGNORED_SYMBOLS: meaning:\lnot, meaning:\to are skipped because
250    // pre-2017 TeXlive let-aliased \lnot → \neg and \to → \rightarrow,
251    // which would gratuitously diverge tests across TL versions.
252    // Mirror Perl `TeX_Job.pool.ltxml` IGNORED_SYMBOLS L104-107.
253    if matches!(*table, TableName::Meaning) && matches!(key_str.as_str(), "\\lnot" | "\\to") {
254      skipped += 1;
255      continue;
256    }
257
258    // A font's parameter array (expl3 keeps its intarrays there, ~89k slots
259    // for an initialized expl3 + LaTeX kernel): one `IA` row, its values
260    // run-length encoded. See `docs/parity/DUMP_DESIGN.md` (IA records) and
261    // `docs/parity/WISDOM.md` #53.
262    if let Stored::FontDimens(_) = value
263      && let Some(body) = serialize_stored(value)
264    {
265      regular.push(("IA".to_string(), url_encode(&key_str), body));
266      continue;
267    }
268
269    // Perl TeX_Job.pool.ltxml:185-197: a `\let` copy — a meaning whose own cs is not its key —
270    // that is still the very definition of that cs, before the load (the bootstrap snapshot) or
271    // now, is written `Lt(<key>, <cs>)` and re-`\let` when the dump loads, so it takes the cs's
272    // definition in the run it is loaded into; any other copy is written by value. Closure bodies
273    // and primitives always take this alias form (`PA`, `serialize_stored`); a token body was
274    // written by value, which froze `\tex_jobname:D` (l3names `\cs_new_eq:NN \tex_jobname:D
275    // \jobname`) to the dump run's empty `\jobname` placeholder, so `\c_sys_jobname_str` was
276    // empty in every conversion. Repro expl3/tex_jobname_is_the_jobname.
277    if matches!(*table, TableName::Meaning)
278      && let Stored::Expandable(exp) = value
279      && !matches!(
280        exp.get_expansion(),
281        Some(crate::definition::ExpansionBody::Closure(_))
282      )
283    {
284      let cs = exp.get_cs().into_owned();
285      let cs_name = cs.with_str(ToString::to_string);
286      if cs_name != key_str {
287        let same = |other: Option<&Stored>| match other {
288          Some(Stored::Expandable(o)) => std::rc::Rc::ptr_eq(o, exp),
289          _ => false,
290        };
291        let early = bootstrap_snap
292          .as_ref()
293          .is_some_and(|snap| same(snap.get(&(TableName::Meaning, arena::pin(&cs_name)))));
294        let late = !early && crate::state::with_meaning(&cs, same);
295        if early || late {
296          let row = (
297            table_code.to_string(),
298            url_encode(&key_str),
299            format!("PA\t{}", url_encode(&cs_name)),
300          );
301          if early {
302            early_aliases.push(row);
303          } else {
304            late_aliases.push(row);
305          }
306          continue;
307        }
308      }
309    }
310
311    let Some(serialized) = serialize_stored(value) else {
312      skipped += 1;
313      continue;
314    };
315
316    // A let-alias entry is an M row whose serialized form starts with
317    // `PA\t` or `MPA\t`. Nothing else produces those prefixes.
318    let is_alias = matches!(*table, TableName::Meaning)
319      && (serialized.starts_with("PA\t") || serialized.starts_with("MPA\t"));
320
321    let row = (table_code.to_string(), url_encode(&key_str), serialized);
322    if is_alias {
323      // Extract the target CS from "PA\t<target>" / "MPA\t<target>"
324      // (we already filtered self-aliases in serialize_stored, so the
325      //  target is always a different CS than the key).
326      let target_raw = row.2.split('\t').nth(1).unwrap_or("");
327      let target = crate::dump_reader::url_decode(target_raw);
328      let target_sym = arena::pin(&target);
329      // Target pre-existed iff it was in the bootstrap meaning table.
330      let pre_existed = bootstrap_snap
331        .as_ref()
332        .is_some_and(|snap| snap.contains_key(&(TableName::Meaning, target_sym)));
333      if pre_existed {
334        early_aliases.push(row);
335      } else {
336        late_aliases.push(row);
337      }
338    } else {
339      regular.push(row);
340    }
341  }
342
343  writeln!(
344    file,
345    "# Section 0: early let-aliases ({}) — target in bootstrap",
346    early_aliases.len()
347  )
348  .ok();
349  for (t, k, v) in &early_aliases {
350    writeln!(file, "{}\t{}\t{}", t, k, v).map_err(|e| format!("Write error: {}", e))?;
351  }
352  writeln!(file, "# Section 1: regular entries ({})", regular.len()).ok();
353  for (t, k, v) in &regular {
354    writeln!(file, "{}\t{}\t{}", t, k, v).map_err(|e| format!("Write error: {}", e))?;
355  }
356  writeln!(
357    file,
358    "# Section 2: late let-aliases ({}) — target in Section 1",
359    late_aliases.len()
360  )
361  .ok();
362  for (t, k, v) in &late_aliases {
363    writeln!(file, "{}\t{}\t{}", t, k, v).map_err(|e| format!("Write error: {}", e))?;
364  }
365
366  let count = early_aliases.len() + regular.len() + late_aliases.len();
367  writeln!(file, "# Written: {}, Skipped: {}", count, skipped).ok();
368  eprintln!(
369    "[dump_writer] Wrote {} entries to {} ({} skipped; {} early + {} regular + {} late)",
370    count,
371    path.display(),
372    skipped,
373    early_aliases.len(),
374    regular.len(),
375    late_aliases.len(),
376  );
377
378  Ok(count)
379}

381fn table_to_code(t: TableName) -> &'static str {
382  match t {
383    TableName::Meaning => "M",
384    TableName::Value => "V",
385    TableName::Catcode => "C",
386    TableName::Mathcode => "MC",
387    TableName::Sfcode => "SC",
388    TableName::Lccode => "LC",
389    TableName::Uccode => "UC",
390    TableName::Delcode => "DC",
391    TableName::Stash => "ST",
392    TableName::StashActive => "SA",
393  }
394}

396/// Serialize a Stored value to a type-tag + data string.
397/// Returns None if the value can't be serialized.
398pub(crate) fn serialize_stored(stored: &Stored) -> Option<String> {
399  match stored {
400    Stored::None => Some("N".to_string()),
401    Stored::Bool(b) => Some(format!("B\t{}", if *b { "1" } else { "0" })),
402    Stored::Int(i) => Some(format!("I\t{}", i)),
403    Stored::String(s) => Some(format!(
404      "S\t{}",
405      url_encode(&arena::with(*s, |s| s.to_string()))
406    )),
407    Stored::Charcode(c) => Some(format!("CH\t{}", c)),
408    Stored::Catcode(cc) => Some(format!("CC\t{}", u8::from(*cc))),
409    Stored::Token(t) => Some(format!("T\t{}", serialize_token(t))),
410    Stored::Tokens(tks) => {
411      let tok_strs: Vec<String> = tks.unlist_ref().iter().map(serialize_token).collect();
412      Some(format!("TK\t{}", tok_strs.join(",")))
413    },
414    Stored::Expandable(exp) => {
415      use crate::common::object::Object;
416      let cs_name = exp.get_cs().with_str(url_encode);
417      let nargs = exp.get_num_args();
418      let mut flags = String::new();
419      if exp.is_long {
420        flags.push('L');
421      }
422      if exp.is_protected() {
423        flags.push('P');
424      }
425      // Rust-only: the macro checks its arguments for `\par` (tex.web §392; `Expandable::checks_par`), which a
426      // restored definition cannot recompute (its origin is the dump's). Perl's Dumper stores none (Dumper.pm:353).
427      if exp.checks_par {
428        flags.push('T');
429      }
430      // Two parallel parameter encodings:
431      //
432      //   - `proto` (v2, 5th field) — `Parameters::stringify()`, a flat space-separated prototype
433      //     string. Round-trips simple specs but not `Until:\end{verbatim}` or other
434      //     brace-in-delimiter forms.
435      //   - `v3_params` (v3, 6th field) — per-Parameter structured record carrying (name, spec,
436      //     flags, extras). Modeled on Perl's `P(type, spec, extra=>[T(...)])`; see
437      //     `docs/archive/DUMP_FORMAT_PERL_ANALYSIS_2026-04-30.md`. Bypasses `parse_parameters` at load
438      //     time so delimited params round-trip intact.
439      //
440      // We emit both so older readers still work, and newer readers can
441      // prefer v3 transparently.
442      let proto = exp
443        .get_parameters()
444        .map(|p| p.stringify())
445        .unwrap_or_default();
446      let proto_encoded = url_encode(&proto);
447      let v3_params = exp
448        .get_parameters()
449        .map(serialize_parameters_v3)
450        .unwrap_or_default();
451      match exp.get_expansion() {
452        Some(crate::definition::ExpansionBody::Tokens(tks)) => {
453          let tok_strs: Vec<String> = tks.unlist_ref().iter().map(serialize_token).collect();
454          Some(format!(
455            "E\t{}\t{}\t{}\t{}\t{}\t{}",
456            cs_name,
457            nargs,
458            flags,
459            tok_strs.join(","),
460            proto_encoded,
461            v3_params
462          ))
463        },
464        // No body at all (e.g. `DefMacro!("\\@gobble{}", None)`): serializes
465        // as an E-entry with empty token body. The reader reconstructs an
466        // Expandable with empty Tokens + the original paramlist, which at
467        // expansion time reads and discards its arguments — semantically
468        // identical to the no-body form's runtime behavior (expandable.rs
469        // `invoke` matches `None` → reads args, returns NO_TOKENS).
470        None => Some(format!(
471          "E\t{}\t{}\t{}\t\t{}\t{}",
472          cs_name, nargs, flags, proto_encoded, v3_params
473        )),
474        // Closure-based body — can't serialize the closure itself, BUT
475        // if the dump entry is a `\let`-alias to a closure-Expandable
476        // primitive (e.g. `\let \tex_expandafter:D \expandafter`), we
477        // CAN capture it via PA. Mirrors the `Stored::Primitive` /
478        // `Stored::Conditional` arms below. Without this, every
479        // `\let \X \expandafter`-style alias is silently dropped from
480        // the dump — the cause of `\tex_expandafter:D`,
481        // `\tex_unexpanded:D`, `\tex_the:D` being missing in
482        // `latex.dump.txt` even though expl3-code.tex L357+ aliases them.
483        //
484        // The dump_reader's add-only policy at load-time skips entries
485        // whose key is already defined (the canonical CS lives in the
486        // compiled engine), so self-aliases are no-ops. Real let-aliases
487        // (key != cs) replay via `state::let_i`.
488        Some(crate::definition::ExpansionBody::Closure(_)) => Some(format!("PA\t{}", cs_name)),
489      }
490    },
491    // Closure-based primitives can't be serialized directly — but if the entry's
492    // CS differs from the primitive's own CS, it's a `\let`-alias, and we CAN
493    // capture the alias so the dump reader can replay `\let <key> <target>` at
494    // load time. This is how `\tex_let:D`, `\tex_def:D`, and hundreds of other
495    // expl3-renamed primitives stay reachable across dump boundaries (the
496    // defining assignment `\let \tex_let:D \let` in expl3-code.tex needs to
497    // survive the dump for `\usepackage{expl3}`'s guard `\ifx\csname
498    // tex_let:D\endcsname\relax` to short-circuit).
499    //
500    // NOTE: the KEY we're saving IS the dump entry's first column; the VALUE
501    // we write is the primitive's OWN CS name. Dump reader compares these:
502    // if equal, it's the "primary" primitive (already in bindings) and skipped;
503    // if different, it's an alias — replay `\let` at load time.
504    Stored::Primitive(p) => {
505      // Perl `Core/Dumper.pm::dump_primitive` (L383-389): if a primitive has
506      // a `font` directive AND no replacement closure (i.e. defined by `\font`),
507      // emit `FD(<cs>, <fontID>)` instead of generic Primitive serialization.
508      // `font_id` is set on `\font`-defined primitives by `engine/tex_fonts.rs`'s
509      // post-define hook (the Rust path) — it carries the value-table key under
510      // which the Stored::Font lives. The reader installs a synthesized
511      // Primitive whose before_digest looks up that font and merges, mirroring
512      // `LaTeXML::Core::Definition::FontDef::invoke` (FontDef.pm L38-45).
513      if let Some(fid) = p.font_id {
514        let fid_str = arena::with(fid, |s| s.to_string());
515        return Some(format!("FD\t{}", url_encode(&fid_str)));
516      }
517      let target_cs = p.cs.with_str(url_encode);
518      Some(format!("PA\t{}", target_cs))
519    },
520    // A font's parameter array, the body of its `IA` row: `<len>\t<rle>`.
521    Stored::FontDimens(params) => {
522      let params = params.borrow();
523      Some(format!("{}\t{}", params.len(), rle_encode_i64(&params)))
524    },
525    Stored::Font(f) => {
526      // Perl `dump_font` (Core/Dumper.pm L281-284) emits `F(... components ...)`.
527      // Rust's `Font` is a flat options struct, so we serialize the non-None
528      // fields as `key=value` pairs separated by `\x1f` (unit-separator), with
529      // values url-encoded for tab/newline-safety. Reader inverts in dump_reader.rs.
530      let mut parts = Vec::with_capacity(8);
531      if let Some(ref name) = f.name {
532        parts.push(format!("name={}", url_encode(name)));
533      }
534      if let Some(size) = f.size {
535        parts.push(format!("size={}", size));
536      }
537      if let Some(ref family) = f.family {
538        parts.push(format!("family={}", url_encode(family)));
539      }
540      if let Some(ref series) = f.series {
541        parts.push(format!("series={}", url_encode(series)));
542      }
543      if let Some(ref shape) = f.shape {
544        parts.push(format!("shape={}", url_encode(shape)));
545      }
546      if let Some(ref encoding) = f.encoding {
547        parts.push(format!("encoding={}", url_encode(encoding)));
548      }
549      if let Some(ref language) = f.language {
550        parts.push(format!("language={}", url_encode(language)));
551      }
552      if let Some(ref mathstyle) = f.mathstyle {
553        parts.push(format!("mathstyle={}", url_encode(mathstyle)));
554      }
555      if let Some(ref opacity) = f.opacity {
556        parts.push(format!("opacity={}", url_encode(opacity)));
557      }
558      if let Some(scale) = f.scale {
559        parts.push(format!("scale={}", scale));
560      }
561      if let Some(b) = f.emph {
562        parts.push(format!("emph={}", if b { 1 } else { 0 }));
563      }
564      if let Some(b) = f.scripted {
565        parts.push(format!("scripted={}", if b { 1 } else { 0 }));
566      }
567      if let Some(step) = f.mathstylestep {
568        parts.push(format!("mathstylestep={}", step));
569      }
570      if let Some(flags) = f.flags {
571        parts.push(format!("flags={}", flags));
572      }
573      Some(format!("F\t{}", parts.join("\x1f")))
574    },
575    Stored::MathPrimitive(p) => {
576      let target_cs = p.cs.with_str(url_encode);
577      Some(format!("MPA\t{}", target_cs))
578    },
579    // Conditionals — same logic as Primitives. \ifx, \ifnum, \if etc. are
580    // stored as Stored::Conditional with their canonical CS field. When
581    // expl3-code.tex does `\let \if_meaning:w \ifx`, the new entry shares
582    // the same Conditional Rc, with cs = \ifx. Capturing this as a PA
583    // alias lets the dump reader replay `\let \if_meaning:w \ifx` at load
584    // time. Without this, every expl3 conditional alias is silently lost
585    // from the dump and `\if_meaning:w`-style invocations fail at runtime.
586    Stored::Conditional(c) => {
587      let target_cs = c.cs.with_str(url_encode);
588      Some(format!("PA\t{}", target_cs))
589    },
590    // Constructors — `\cr`, `\noalign`, `\mathchoice`, etc. defined via
591    // `DefConstructor!` carry replacement / before_digest / after_digest
592    // closures that the dump format can't serialize. But the canonical
593    // CS lives in the compiled engine, so a `\let \tex_cr:D \cr`-style
594    // alias can round-trip as `PA\t<cs>` and replay via `state::let_i`.
595    // Self-aliases are no-ops at load (key already defined).
596    Stored::Constructor(c) => {
597      let target_cs = c.cs.with_str(url_encode);
598      Some(format!("PA\t{}", target_cs))
599    },
600    Stored::Register(reg) => {
601      // Register: serialize as R\tCS\tTYPE\tVALUE[\tMATHGLYPH]
602      //
603      // Closure-backed registers can't be serialized directly — but if the
604      // dump entry is a `\let`-alias to a closure-Register primitive
605      // (e.g. `\let \tex_dimexpr:D \dimexpr`), we CAN capture it via PA.
606      // Mirrors the `Stored::Primitive` / closure-`Expandable` arms above.
607      // Without this every `\let \tex_dimexpr:D \dimexpr`-style alias is
608      // silently dropped from the dump — the cause of `\tex_dimexpr:D`,
609      // `\tex_cr:D`, `\tex_dp:D`, `\tex_ht:D`, `\tex_wd:D` (and ~430
610      // engine-specific others) being missing in `latex.dump.txt` even
611      // though expl3-code.tex L280-700 aliases them. The dump_reader's
612      // add-only policy at load-time skips entries whose key is already
613      // defined (the canonical CS lives in the compiled engine), so
614      // self-aliases are no-ops. Real let-aliases (key != cs) replay via
615      // `state::let_i` — so the engine-only register's behavior follows.
616      if reg.getter.is_some() || reg.setter.is_some() {
617        let target_cs = reg.cs.with_str(url_encode);
618        return Some(format!("PA\t{}", target_cs));
619      }
620      let cs_name = reg.cs.with_str(url_encode);
621      let rtype = match reg.register_type {
622        crate::definition::register::RegisterType::Number => "N",
623        crate::definition::register::RegisterType::Dimension => "D",
624        crate::definition::register::RegisterType::Glue => "G",
625        crate::definition::register::RegisterType::MuGlue => "MG",
626        crate::definition::register::RegisterType::Tokens => "TK",
627        crate::definition::register::RegisterType::CharDef => "CD",
628        _ => return None,
629      };
630      let value_str = match &reg.value {
631        Some(crate::definition::register::RegisterValue::Number(n)) => n.value_of().to_string(),
632        Some(crate::definition::register::RegisterValue::Dimension(d)) => d.value_of().to_string(),
633        Some(crate::definition::register::RegisterValue::Glue(g)) => {
634          let mut s = g.skip.to_string();
635          if let Some(p) = g.plus {
636            s.push_str(&format!(",p{}", p));
637          }
638          if let Some(ref pf) = g.pfill {
639            s.push_str(&format!(",pf{}", fillcode_index(pf)));
640          }
641          if let Some(m) = g.minus {
642            s.push_str(&format!(",m{}", m));
643          }
644          if let Some(ref mf) = g.mfill {
645            s.push_str(&format!(",mf{}", fillcode_index(mf)));
646          }
647          s
648        },
649        Some(crate::definition::register::RegisterValue::Tokens(tks)) => {
650          let tok_strs: Vec<String> = tks.unlist_ref().iter().map(serialize_token).collect();
651          tok_strs.join(",")
652        },
653        _ => "0".to_string(),
654      };
655      // Serialize address if different from cs name. Allocated registers
656      // (`\newcount\m@ne`/`\newdimen\p@`/etc.) point at low-level `\count<n>`/
657      // `\dimen<n>` slots whose runtime values live there, NOT at the alias
658      // CS name. Without the address, dump_reader would `assign_internal` at
659      // the CS name's slot (default value), and `\the\m@ne` would yield 0
660      // instead of -1. Mirror Perl's `R(C(...),undef,...,address=>'\\count22')`
661      // serialization (LaTeXML/Core/Dumper.pm). Format extension:
662      // `R\t<cs>\t<rtype>\t<value>\t<mathglyph_or_empty>\t<address>`.
663      // If address == cs_name we still serialize an empty 6th field for
664      // forward-compat parsers; the reader accepts both 4/5/6-field forms.
665      let mut s = format!("R\t{}\t{}\t{}", cs_name, rtype, value_str);
666      let mathglyph_str = match reg.mathglyph {
667        Some(glyph) => format!("{}", glyph as u32),
668        None => String::new(),
669      };
670      let cs_decoded = arena::with(reg.cs.get_sym(), |s| s.to_string());
671      let address_field = if reg.address.is_empty() || reg.address == cs_decoded {
672        String::new()
673      } else {
674        url_encode(&reg.address)
675      };
676      // Only emit extended fields when needed (keep current format size for
677      // the common case)
678      if !mathglyph_str.is_empty() || !address_field.is_empty() {
679        s.push_str(&format!("\t{}", mathglyph_str));
680      }
681      if !address_field.is_empty() {
682        s.push_str(&format!("\t{}", address_field));
683      }
684      Some(s)
685    },
686    // "Nm" (Number) — distinct from "I" (raw Int) so the reader knows
687    // to install Stored::Number, matching the type expected by register
688    // slots. Round-tripping Number through "I" silently downgraded
689    // values to Stored::Int and broke `\count` register reads.
690    Stored::Number(n) => Some(format!("Nm\t{}", n.value_of())),
691    Stored::Float(f) => Some(format!("F\t{}", f.0)),
692    Stored::Dimension(d) => Some(format!("D\t{}", d.0)),
693    Stored::Glue(g) => {
694      let mut s = format!("G\t{}", g.skip);
695      if let Some(p) = g.plus {
696        s.push_str(&format!(",p{}", p));
697      }
698      if let Some(ref pf) = g.pfill {
699        s.push_str(&format!(",pf{}", fillcode_index(pf)));
700      }
701      if let Some(m) = g.minus {
702        s.push_str(&format!(",m{}", m));
703      }
704      if let Some(ref mf) = g.mfill {
705        s.push_str(&format!(",mf{}", fillcode_index(mf)));
706      }
707      Some(s)
708    },
709    Stored::MuDimension(d) => Some(format!("MD\t{}", d.0)),
710    Stored::MuGlue(g) => {
711      let mut s = format!("MG\t{}", g.skip);
712      if let Some(p) = g.plus {
713        s.push_str(&format!(",p{}", p));
714      }
715      if let Some(ref pf) = g.pfill {
716        s.push_str(&format!(",pf{}", fillcode_index(pf)));
717      }
718      if let Some(m) = g.minus {
719        s.push_str(&format!(",m{}", m));
720      }
721      if let Some(ref mf) = g.mfill {
722        s.push_str(&format!(",mf{}", fillcode_index(mf)));
723      }
724      Some(s)
725    },
726    Stored::VecDequeStored(vd) if vd.is_empty() => Some("VD\t".to_string()),
727    _ => None,
728  }
729}

746/// Serialize a `Parameters` list in the v3 structured format.
747///
748/// Layout (ASCII control-byte delimited so nothing collides with
749/// url-encoded printable content):
750/// - `\x1e` between Parameters (RS, Record Separator)
751/// - `\x1f` between fields of a single Parameter (US, Unit Separator)
752/// - `\x1d` between Tokens lists inside `extras` (GS, Group Separator)
753///
754/// Per-parameter fields: `<name>\x1f<spec>\x1f<flags>\x1f<extras>`
755///   - `<name>` — `Parameter.name` (Perl's `type`), url-encoded.
756///   - `<spec>` — `Parameter.spec`, url-encoded.
757///   - `<flags>` — semicolon-separated `n=1` (novalue) / `o=1` (optional).
758///   - `<extras>` — `<Tokens1>\x1d<Tokens2>…` where each `<TokensN>` is the same comma-joined
759///     `<catcode>:<text>` encoding used for E-entry bodies. Empty extras → empty field.
760///
761/// The reader (`dump_reader::parse_parameters_v3`) hands the decoded
762/// `(name, spec, Some(extras))` triple directly to `Parameter::new`,
763/// bypassing `parse_parameters` entirely — so `Until:\end{verbatim}`
764/// and `Match:…` round-trip with their catcoded delimiter tokens intact.
765pub(crate) fn serialize_parameters_v3(params: &crate::parameter::Parameters) -> String {
766  let mut out = String::new();
767  for (i, p) in params.get_parameters().iter().enumerate() {
768    if i > 0 {
769      out.push('\x1e');
770    }
771    let name = arena::with(p.name, url_encode);
772    let spec = arena::with(p.spec, url_encode);
773    let mut flags = String::new();
774    if p.novalue {
775      flags.push_str("n=1");
776    }
777    if p.optional {
778      if !flags.is_empty() {
779        flags.push(';');
780      }
781      flags.push_str("o=1");
782    }
783    let extras = p
784      .extra
785      .iter()
786      .map(|tks| {
787        tks
788          .clone()
789          .unlist()
790          .iter()
791          .map(serialize_token)
792          .collect::<Vec<_>>()
793          .join(",")
794      })
795      .collect::<Vec<_>>()
796      .join("\x1d");
797    out.push_str(&name);
798    out.push('\x1f');
799    out.push_str(&spec);
800    out.push('\x1f');
801    out.push_str(&flags);
802    out.push('\x1f');
803    out.push_str(&extras);
804  }
805  out
806}

808fn url_encode(s: &str) -> String {
809  let mut result = String::with_capacity(s.len());
810  for ch in s.chars() {
811    match ch {
812      '\t' => result.push_str("%09"),
813      '\n' => result.push_str("%0A"),
814      '\r' => result.push_str("%0D"),
815      '%' => result.push_str("%25"),
816      ',' => result.push_str("%2C"),
817      _ if ch.is_ascii_control() => {
818        result.push_str(&format!("%{:02X}", ch as u8));
819      },
820      _ => result.push(ch),
821    }
822  }
823  result
824}

826/// Run-length encode a slice of i64 values as a comma-separated list:
827/// each run is either `<v>` (single) or `<v>x<n>` (count ≥ 2). Decoder
828/// in `dump_reader::rle_decode_i64` is the inverse.
829fn rle_encode_i64(values: &[i64]) -> String {
830  let mut out = String::new();
831  let mut i = 0;
832  while i < values.len() {
833    let v = values[i];
834    let mut count = 1usize;
835    while i + count < values.len() && values[i + count] == v {
836      count += 1;
837    }
838    if !out.is_empty() {
839      out.push(',');
840    }
841    if count == 1 {
842      out.push_str(&v.to_string());
843    } else {
844      out.push_str(&format!("{}x{}", v, count));
845    }
846    i += count;
847  }
848  out
849}

851/// Per-load file-IO state that must never be dumped. `\@currname` /
852/// `\@currext` are assigned per `\input` by `read_input_file_recursive`
853/// (`binding/content.rs`), so at snapshot time they hold the LAST opened
854/// file ("latex.ltx") — Perl's `\input` leaves them at the `\@empty`
855/// baseline its latex_dump records (latex_dump.pool.ltxml:24122-24123).
856/// `\@currnamestack` is the `\@pushfilename`/`\@popfilename` stack
857/// (tex_file_io.rs:20-31) and carries `\the\catcode`\@`; Perl never dumps it
858/// because its serialized form equals the bootstrap's (`diff_from_snapshot`
859/// now makes the same comparison — this list is the belt to that brace).
860/// All describe the file being read at dump time, not the format.
861/// ltfilehook's `\CurrentFile*` token lists are NOT in the family: latex.ltx
862/// (:19582-19585) creates them, so the dump carries them EMPTY, as Perl's
863/// blib latex_dump.pool.ltxml:2716-2719 does.
864pub(crate) fn is_file_io_bookkeeping(key: &str) -> bool {
865  matches!(key, "\\@currname" | "\\@currext" | "\\@currnamestack")
866}
