1//! Windowed cycle-detection infinite-loop guard.
2//!
3//! A general defense layer that complements the coarse size/count limits
4//! (`Gullet::token_limit` / `pushback_limit`) and the outermost RSS soft cap
5//! (`stomach::check_timeout`). Where those catch a runaway only after it has
6//! consumed millions of tokens or gigabytes of RSS, this guard spots the
7//! *structure* of a loop directly: a short window of items that repeats many
8//! times back-to-back.
9//!
10//! Two instances run at different kernel levels (the project's layered-guard
11//! philosophy): one over the gullet's expansion stream (token fingerprints),
12//! one over the stomach's accumulated digest list (box fingerprints). Either
13//! can terminate a runaway with a clean `Fatal` long before the RSS cap.
14//!
15//! Algorithm (per the design directive): record a stream of `u64`
16//! fingerprints in a fixed ring buffer. Periodically check whether the most
17//! recent items are periodic with some period `W` in `1..=MAX_WINDOW`,
18//! repeated at least `REPEAT` times. The check is pure periodicity over the
19//! last `W*REPEAT` items (`item[i] == item[i-W]`), so it is **phase/offset
20//! independent** — the cycle need not align to any buffer boundary. The
21//! smallest matching period is reported.
22//!
23//! Cost: detection is throttled to once per `CHECK_EVERY` pushes and is
24//! `O(MAX_WINDOW^2 * REPEAT)` per check (~5.4k u64 compares for the defaults),
25//! i.e. a few amortized compares per push. Callers further gate activation on
26//! an already-high item count so normal conversions pay nothing.
27
28/// Largest cycle period (in items) we look for.
29pub const MAX_WINDOW: usize = 10;
30/// How many consecutive repetitions of a window constitute "infinite".
31///
32/// Stays at 100 (not thousands): the gullet fingerprints only a
33/// `CYCLE_GUARD_DUTY_ON` = 2 048-token window per duty period, so
34/// `MAX_WINDOW * REPEAT` must fit inside it. Finite constructs that emit a
35/// short window thousands of times (`\prg_replicate:nn {10000} {…}`,
36/// maze manual) are told apart by the backlog gate in [`CycleGuard::push`],
37/// not by a larger count.
38pub const REPEAT: usize = 100;
39/// Ring-buffer capacity: enough to hold `MAX_WINDOW` repeated `REPEAT` times.
40const CAP: usize = MAX_WINDOW * REPEAT;
41/// Run the (cheap but non-trivial) periodicity scan only this often.
42const CHECK_EVERY: usize = 256;
43
44/// Number of check points (every [`CHECK_EVERY`] pushes) whose backlog
45/// samples are kept — enough to cover the largest window
46/// (`MAX_WINDOW * REPEAT` items) plus one.
47const BACKLOG_SAMPLES: usize = CAP / CHECK_EVERY + 2;
48/// Sentinel for a backlog sample not yet taken since the last reset.
49const UNFILLED: usize = usize::MAX;
50
51/// A windowed cycle detector over a stream of `u64` fingerprints.
52pub struct CycleGuard {
53  /// Backlog sampled at the last `BACKLOG_SAMPLES` checks (newest last).
54  backlog:      [usize; BACKLOG_SAMPLES],
55  backlog_head: usize,
56  buf:          Box<[u64; CAP]>,
57  /// next write position (ring)
58  head:         usize,
59  /// number of valid entries (saturates at CAP)
60  len:          usize,
61  /// throttle counter
62  since:        usize,
63}
64
65impl Default for CycleGuard {
66  fn default() -> Self { Self::new() }
67}

76  }

79impl CycleGuard {
80  pub fn new() -> Self {
81    CycleGuard {
82      buf:          Box::new([0u64; CAP]),
83      head:         0,
84      len:          0,
85      since:        0,
86      backlog:      [UNFILLED; BACKLOG_SAMPLES],
87      backlog_head: 0,
88    }
89  }
90
91  /// Drop all recorded history (call at the start of each conversion).
92  pub fn reset(&mut self) {
93    self.head = 0;
94    self.len = 0;
95    self.since = 0;
96    self.backlog = [UNFILLED; BACKLOG_SAMPLES];
97    self.backlog_head = 0;
98  }
99
100  /// Record one fingerprint. Returns `Some(period)` if the recent stream is a
101  /// window of `period` items repeated at least [`REPEAT`] times.
102  ///
103  /// `backlog` is the caller's pending-input measure at this push — the
104  /// gullet passes its pushback length, sampled only at check points. A
105  /// detected period is reported only when the backlog did NOT shrink
106  /// across the repeated span: a genuine cycle re-expands every period, so
107  /// its backlog holds level or grows (`\def\x{a\x}`, `\def\x{\x\x}`),
108  /// while a periodic stream whose backlog drains is pre-built data being
109  /// consumed — `\prg_replicate:nn {10000} {<body>}` unpacking l3intarray's
110  /// `\intarray_gzero:N` body into the pushback and then executing it entry
111  /// by entry (maze manual, a cycle-guard Fatal on a finite construct).
112  /// Callers without a backlog notion (the stomach's box stream) pass a
113  /// constant, which never gates.
114  #[inline]
115  pub fn push(&mut self, fp: u64, backlog: usize) -> Option<usize> {
116    self.buf[self.head] = fp;
117    self.head = if self.head + 1 == CAP {
118      0
119    } else {
120      self.head + 1
121    };
122    if self.len < CAP {
123      self.len += 1;
124      if self.len == 1 {
125        // First push after a reset: a baseline sample, so the first check
126        // (which may already see a `REPEAT`-fold window) can judge the trend.
127        self.backlog[self.backlog_head] = backlog;
128        self.backlog_head = (self.backlog_head + 1) % BACKLOG_SAMPLES;
129      }
130    }
131    self.since += 1;
132    if self.since >= CHECK_EVERY {
133      self.since = 0;
134      self.backlog[self.backlog_head] = backlog;
135      self.backlog_head = (self.backlog_head + 1) % BACKLOG_SAMPLES;
136      return self.detect().filter(|_| !self.backlog_drained());
137    }
138    None
139  }
140
141  /// True when the backlog is falling: the newest sample sits more than
142  /// `MAX_WINDOW` below the previous one (a cycle's backlog oscillates by at
143  /// most its period between samples; consumed data drops by `CHECK_EVERY`).
144  /// The previous sample is the last check point, or the post-reset baseline
145  /// when this is the first check.
146  fn backlog_drained(&self) -> bool {
147    let newest = self.backlog[(self.backlog_head + BACKLOG_SAMPLES - 1) % BACKLOG_SAMPLES];
148    let prev = self.backlog[(self.backlog_head + BACKLOG_SAMPLES - 2) % BACKLOG_SAMPLES];
149    prev != UNFILLED && newest + MAX_WINDOW < prev
150  }
151
152  /// The backlog samples, oldest first (for the `LATEXML_DEBUG_FATAL` trace).
153  pub fn backlog_trace(&self) -> Vec<usize> {
154    (0..BACKLOG_SAMPLES)
155      .map(|k| self.backlog[(self.backlog_head + k) % BACKLOG_SAMPLES])
156      .filter(|&b| b != UNFILLED)
157      .collect()
158  }
159
160  /// The `k`-th most recently pushed item (`k = 0` is newest).
161  #[inline]
162  fn at_from_end(&self, k: usize) -> u64 {
163    // head points one past the newest, modulo CAP.
164    let idx = (self.head + CAP - 1 - k) % CAP;
165    self.buf[idx]
166  }
167
168  fn detect(&self) -> Option<usize> {
169    // Smallest period first, so a 2-cycle reports period 2, not 4/6/…
170    for w in 1..=MAX_WINDOW {
171      let need = w * REPEAT;
172      if self.len < need {
173        // Larger windows need even more history; the loop is monotone in `w`.
174        break;
175      }
176      // The last `need` items are periodic with period `w` iff
177      // item[i] == item[i-w] for every i in the most-recent `need - w`.
178      let mut periodic = true;
179      for k in 0..(need - w) {
180        if self.at_from_end(k) != self.at_from_end(k + w) {
181          periodic = false;
182          break;
183        }
184      }
185      if periodic {
186        // UNIFORM-run suppression (PR #249 review P1-3): a window whose
187        // items are all IDENTICAL describes a long run of one repeated
188        // fingerprint — textually legitimate input (a verbatim `====…`
189        // separator row, dot leaders, repeated rule glyphs), not a loop.
190        // Real macro loops re-read ≥2 distinct tokens per cycle
191        // (`\def\x{a\x}` → a,\x,a,\x…), and the pure single-token
192        // self-expansion `\def\x{\x}` is caught by the Expandable
193        // self-recursion error before any guard. Note a uniform stream
194        // matches EVERY w, so rejecting uniform windows here rejects the
195        // whole stream (each larger window is uniform too) — by design.
196        let first = self.at_from_end(0);
197        let has_distinct = (1..w).any(|k| self.at_from_end(k) != first);
198        if has_distinct {
199          return Some(w);
200        }
201      }
202    }
203    None
204  }
205}

207/// How many recent token spans [`SpanHistory`] keeps.
208const SPAN_CAP: usize = 1024;
209/// The largest input period, in boxes, [`SpanHistory::input_period`] looks for:
210/// eight repetitions fill the ring.
211pub const MAX_SPAN_PERIOD: usize = SPAN_CAP / 8;
212
213/// The recent token spans behind the stomach's boxes: how many tokens the
214/// gullet read for each box the stomach's [`CycleGuard`] records. A box period
215/// alone cannot tell a loop from a long run over data: every point of a pgf
216/// `plot[smooth]` digests the same content-free boxes (tkz-grapheur's
217/// 11,000-point plot read as a 6-box cycle). A loop repeats the same boxes from
218/// the same input; a run over data repeats them from input of varying length.
219/// So a detected box period is a loop only when the spans repeat too, with a
220/// period that is a multiple of the box period ([`Self::input_period`]).
221pub struct SpanHistory {
222  buf:  Box<[u32; SPAN_CAP]>,
223  head: usize,
224  len:  usize,
225}
226
227impl Default for SpanHistory {
228  fn default() -> Self {
229    SpanHistory {
230      buf:  Box::new([0; SPAN_CAP]),
231      head: 0,
232      len:  0,
233    }
234  }
235}
236
237impl SpanHistory {
238  /// Forget every recorded span.
239  pub fn reset(&mut self) {
240    self.head = 0;
241    self.len = 0;
242  }
243
244  /// Record the span of the newest box.
245  #[inline]
246  pub fn push(&mut self, span: usize) {
247    self.buf[self.head] = u32::try_from(span).unwrap_or(u32::MAX);
248    self.head = (self.head + 1) % SPAN_CAP;
249    self.len = (self.len + 1).min(SPAN_CAP);
250  }
251
252  /// The `k`-th most recent span (`k = 0` is newest).
253  #[inline]
254  fn at_from_end(&self, k: usize) -> u32 { self.buf[(self.head + SPAN_CAP - 1 - k) % SPAN_CAP] }
255
256  /// The smallest period, a multiple of `box_period` up to
257  /// [`MAX_SPAN_PERIOD`], with which the recorded spans repeat over the whole
258  /// history (at least eight repetitions of it); `None` when they do not.
259  pub fn input_period(&self, box_period: usize) -> Option<usize> {
260    let n = self.len;
261    (box_period..=MAX_SPAN_PERIOD)
262      .step_by(box_period.max(1))
263      .filter(|&p| n >= 8 * p)
264      .find(|&p| (0..n - p).all(|k| self.at_from_end(k) == self.at_from_end(k + p)))
265  }
266}

272  fn spans(stream: impl IntoIterator<Item = usize>) -> SpanHistory {
273    let mut h = SpanHistory::default();
274    for s in stream {
275      h.push(s);
276    }
277    h
278  }

280  #[test]
281  fn constant_spans_repeat_with_any_box_period() {
282    let h = spans(std::iter::repeat_n(4, 1500));
283    assert_eq!(h.input_period(1), Some(1));
284    assert_eq!(h.input_period(6), Some(6));
285  }
286
287  #[test]
288  fn spans_repeating_at_a_multiple_of_the_box_period_repeat() {
289    // A 2-box window whose input alternates over 12 boxes (two macros).
290    let pattern = [3, 1, 3, 1, 3, 1, 4, 1, 4, 1, 4, 2];
291    let h = spans((0..1500).map(|i| pattern[i % pattern.len()]));
292    assert_eq!(h.input_period(2), Some(12));
293    assert_eq!(h.input_period(4), Some(12));
294    // A counter wrapping every 33 iterations of a 2-box loop: input period 66.
295    let h = spans((0..1500).map(|i| if i % 66 == 65 { 9 } else { 3 }));
296    assert_eq!(h.input_period(2), Some(66));
297  }

299  #[test]
300  fn varying_spans_do_not_repeat() {
301    // Coordinates of different lengths behind the same boxes (a smooth plot):
302    // xorshift output, with no short period.
303    let mut x: u32 = 0x9E37_79B9;
304    let h = spans((0..1500).map(|_| {
305      x ^= x << 13;
306      x ^= x >> 17;
307      x ^= x << 5;
308      5 + (x % 7) as usize
309    }));
310    assert_eq!(h.input_period(6), None);
311    assert_eq!(h.input_period(1), None);
312  }

314  #[test]
315  fn an_input_period_past_the_ring_is_not_seen() {
316    // Eight repetitions must fit the ring: period 160 over 1024 spans is not.
317    let h = spans((0..2000).map(|i| if i % 160 == 0 { 9 } else { 3 }));
318    assert_eq!(h.input_period(2), None);
319  }

321  fn run(stream: &[u64]) -> Option<usize> {
322    let mut g = CycleGuard::new();
323    let mut hit = None;
324    // A level backlog: the guard's core periodicity contract.
325    for &x in stream {
326      if let Some(p) = g.push(x, 0) {
327        hit = Some(p);
328        break;
329      }
330    }
331    hit
332  }

334  #[test]
335  fn periodic_stream_with_draining_backlog_is_data() {
336    // A perfect period-2 stream consumed while the backlog falls is
337    // pre-built data (`\prg_replicate` output), not a loop.
338    let mut g = CycleGuard::new();
339    let n = 9 * REPEAT;
340    let hit = (0..n).find_map(|i| g.push((i % 2) as u64, n - i));
341    assert_eq!(hit, None);
342  }

352  #[test]
353  fn uniform_run_does_not_fire() {
354    // A long run of IDENTICAL fingerprints is textually legitimate input
355    // (a verbatim `====…` separator row, a dot-leader, a repeated rule
356    // glyph) — NOT a loop. Real macro loops re-read at least two distinct
357    // tokens (`\def\x{a\x}` → a, \x, a, \x …), and a pure single-token
358    // self-expansion is caught by the Expandable self-recursion error

360    let s: Vec<u64> = std::iter::repeat_n(7, 20 * REPEAT).collect();
361    assert_eq!(run(&s), None);
362  }
