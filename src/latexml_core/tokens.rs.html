109impl From<Tokens> for Token {
110  fn from(mut ts: Tokens) -> Token {
111    if ts.0.is_empty() {
112      // Match the &Tokens impl below: empty → \relax fallback rather
113      // than panic. Callers that must see the empty case are rare and
114      // should inspect Tokens directly.
115      T_CS!("\\relax")
116    } else if ts.0.len() == 1 {
117      ts.0.remove(0)
118    } else {
119      // Prefer the first token and warn; cascading a panic here usually
120      // means a stringly-typed binding slot received a multi-token value
121      // (e.g. a macro argument coerced into a single-token slot). The
122      // first token preserves TEx's "grab a single token" semantics.
123      emit_warn(
124        "internal",
125        "tokens",
126        &format!("multi-token Tokens cast into single Token: {ts:?}"),
127      );
128      ts.0.remove(0)
129    }
130  }
131}

166impl Display for Tokens {
167  /// to_string is used often, and for more keyword-like reasons,
168  /// NOT for creating valid TeX (use revert or UnTeX for that!)
169  ///
170  /// This is THE token-list → Rust-string boundary (Perl `ToString`). Under
171  /// the pdfTeX byte mouth (KERNEL_CAPABILITIES K10) a non-ASCII name has two
172  /// token spellings — the bytes and the character the octet readers emit —
173  /// that pdfTeX never distinguishes; every Rust-side key derived here (colour
174  /// names, key-value keys, labels) gets the one decoded spelling
175  /// (`mouth::decode_byte_mouth_runs`), so the bindings need no per-site
176  /// normalization. Text runs reach the document as single-character boxes
177  /// and are reassembled there (`Document::open_text_internal`); `\csname`
178  /// builds its name in the gullet (`read_cs_name_inner`).
179  fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
180    if crate::state::lookup_bool_sym(crate::pin!("PDFTEX_BYTE_MOUTH")) {
181      let mut s = String::new();
182      for t in &self.0 {
183        if t.code != Catcode::COMMENT {
184          write!(s, "{t}")?;
185        }
186      }
187      return f.write_str(&crate::mouth::decode_byte_mouth_runs(&s));
188    }
189    for t in &self.0 {
190      if t.code != Catcode::COMMENT {
191        write!(f, "{t}")?;
192      }
193    }
194    Ok(())
195  }
196}

202/// A string of **TeX markup** — text that is safe to hand back to the tokenizer.
203///
204/// It is *not* a path and *not* an input `.tex` file (`source_directory`,
205/// `--source-map` and `docs/performance/SOURCE_PROVENANCE.md` own that sense of
206/// "source"); it is the character content a [`crate::mouth::Mouth`] will read.
207///
208/// # Why the type exists
209///
210/// Flattening [`struct@Tokens`] with [`Display`] **welds control words**. TeX consumes
211/// the space that terminates a control word, so `\v S` tokenizes to `[\v][S]`;
212/// re-emitting that with `Display` gives `\vS`, a control sequence that exists in
213/// no LaTeX. [`Tokens::untex`] re-emits the space, `Display` deliberately does
214/// not — this is faithful to Perl (`Core/Tokens.pm:61 toString` joins the token
215/// strings, and `Core/Token.pm:306` returns a CS name with no trailing space),
216/// whose own comment says the result is "NOT for creating valid TeX (use revert
217/// or UnTeX for that!)".
218///
219/// Perl relies on author discipline there. It has failed three times in this
220/// port — `\bib@@names` (PR #399), `dcolumn`/`overpic` (PR #400), and the
221/// MathSciNet review path (issue 410: `MRREVIEWER = {Fran\c cois\ Digne}` became
222/// `undefined:\ccois`) — each found by a user-visible failure years after the
223/// code was written. `TeXString` makes the mistake unrepresentable instead: the
224/// tokenizing sinks take `impl Into<TeXString>`, and a bare `String` has no way
225/// in.
226///
227/// # The three ways in
228///
229/// * `From<&'static str>` — a string *literal* in a binding is TeX its author
230///   typed by hand, so it converts implicitly and the ~125 literal call sites
231///   stay untouched. There is deliberately **no** `From<String>` and **no**
232///   `From<&str>`: those are exactly the shapes a welded `Tokens::to_string()`
233///   arrives in, and `s!(…)`/`format!(…)` returns the former.
234/// * [`Tokens::untex_string`] — the blessed path from `Tokens`.
235/// * [`TeXString::assembled`] — the explicit escape hatch, for a `format!` of
236///   literal TeX around already-safe pieces. It names the obligation it imposes.
237///
238/// ```
239/// # use latexml_core::tokens::TeXString;
240/// let s: TeXString = r"\relax".into(); // literal: implicit
241/// assert_eq!(s.as_str(), r"\relax");
242/// ```
243///
244/// A `String` cannot get in on its own — this is the guard, and it bites:
245///
246/// ```compile_fail
247/// # use latexml_core::tokens::TeXString;
248/// let welded: String = String::from(r"\vS");
249/// let _: TeXString = welded.into(); // no `From<String>`: does not compile
250/// ```
251#[derive(Debug, Clone, PartialEq, Eq, Hash, Default)]
252pub struct TeXString(Cow<'static, str>);
253
254impl TeXString {
255  /// Assert that an owned `String` is valid TeX markup.
256  ///
257  /// **The caller's obligation**: every interpolated fragment must be either
258  /// literal TeX written at the call site, or a fragment that came from
259  /// [`Tokens::untex_string`] / another `TeXString`. It must **not** be a bare
260  /// `Tokens::to_string()` — that is the welding bug this type exists to
261  /// prevent (`\v S` → `\vS`); use [`Tokens::untex_string`] for those.

317  /// Zero-alloc equivalent of `self.to_string().starts_with(prefix)`.
318  /// Walks tokens byte-by-byte into `prefix` using the same Display
319  /// semantics as `eq_text` (COMMENT skipped, ARG prefixed with `#`).
320  /// Returns `true` once the full prefix has been consumed, even if
321  /// more token text follows.
322  pub fn starts_with_text(&self, prefix: &str) -> bool {
323    let mut remaining = prefix;
324    for t in &self.0 {
325      if remaining.is_empty() {
326        return true;
327      }
328      if t.code == Catcode::COMMENT {
329        continue;
330      }
331      if t.code == Catcode::ARG {
332        if !remaining.starts_with('#') {
333          return false;
334        }
335        remaining = &remaining[1..];
336        if remaining.is_empty() {
337          return true;
338        }
339      }
340      let keep_going = t.with_str(|text| {
341        if text.is_empty() {
342          return true;
343        }
344        if remaining.starts_with(text) {
345          remaining = &remaining[text.len()..];
346          true
347        } else if text.starts_with(remaining) {
348          // This token's text extends past `prefix` — prefix matches
349          // and we're done.
350          remaining = "";
351          true
352        } else {
353          false
354        }
355      });
356      if !keep_going {
357        return false;
358      }
359      if remaining.is_empty() {
360        return true;
361      }
362    }
363    remaining.is_empty()
364  }

366  /// Zero-alloc equivalent of `self.to_string() == target`. Walks the
367  /// contained tokens byte-by-byte, skipping COMMENT tokens (matching
368  /// `Display for Tokens`) and prefixing ARG tokens with `#` (matching
369  /// `Display for Token`). Returns `true` iff the rendered text exactly
370  /// equals `target`. Used by DefMacro bodies that check keyword
371  /// values like `true` / `false` / `swapnumber` without wanting to
372  /// allocate a fresh `String` per invocation.
373  pub fn eq_text(&self, target: &str) -> bool {
374    let mut remaining = target;
375    for t in &self.0 {
376      if t.code == Catcode::COMMENT {
377        continue;
378      }
379      if t.code == Catcode::ARG {
380        if !remaining.starts_with('#') {
381          return false;
382        }
383        remaining = &remaining[1..];
384      }
385      let ok = t.with_str(|text| {
386        if remaining.starts_with(text) {
387          remaining = &remaining[text.len()..];
388          true
389        } else {
390          false
391        }
392      });
393      if !ok {
394        return false;
395      }
396    }
397    remaining.is_empty()
398  }

472  /// Methods for overloaded ops.
473  pub fn equals(&self, other: Tokens) -> bool {
474    let self_tokens: Vec<&Token> = self
475      .0
476      .iter()
477      .filter(|t| t.code != Catcode::COMMENT && t.code != Catcode::MARKER)
478      .collect();
479    let other_tokens: Vec<&Token> = other
480      .0
481      .iter()
482      .filter(|t| t.code != Catcode::COMMENT && t.code != Catcode::MARKER)
483      .collect();
484    if self_tokens.len() != other_tokens.len() {
485      false
486    } else {
487      self_tokens
488        .into_iter()
489        .zip(other_tokens)
490        .all(|(t_self, t_other)| *t_self == *t_other)
491    }
492  }

527  /// [`Tokens::is_balanced`] of a token slice.
528  pub fn slice_is_balanced(tokens: &[Token]) -> bool {
529    let mut level = 0;
530    for t in tokens {
531      level += match t.get_catcode() {
532        Catcode::BEGIN => 1,
533        Catcode::END => -1,
534        _ => 0,
535      };
536      if level < 0 {
537        // a negative level encountered at any point is ill-formed,
538        // return early
539        return false;
540      }
541    }
542    level == 0
543  }

545  /// The body as `macro_call` substitutes it (tex.web §390: a macro's body is stored with each `#n` as an
546  /// `out_param` token carrying `n`): runs of the body's own tokens and the parameters between them, in order.
547  /// Computed once per definition ([`crate::definition::expandable::Expandable`]), so a call copies runs
548  /// instead of testing every token and parsing every parameter's number.
549  pub fn substitution_plan(&self) -> Box<[BodyPiece]> {
550    let mut plan = Vec::new();
551    let mut start = 0;
552    for (at, token) in self.0.iter().enumerate() {
553      if token.get_catcode() == Catcode::ARG {
554        if start < at {
555          plan.push(BodyPiece::Run(start as u32, at as u32));
556        }
557        let index = token.with_str(|ts| ts.parse::<usize>().unwrap_or(0));
558        plan.push(BodyPiece::Parameter(index));
559        start = at + 1;
560      }
561    }
562    if start < self.0.len() {
563      plan.push(BodyPiece::Run(start as u32, self.0.len() as u32));
564    }
565    plan.into_boxed_slice()
566  }

568  // NOTE: Assumes each arg either undef or also Tokens
569  // Using inline accessors on those assumptions
570  /// substitutes the parameters (ARG catcode) in a Tokens list for concrete arguments
571  pub fn substitute_parameters(&self, args: &[Option<Cow<Tokens>>]) -> Self {
572    // Pre-size to the template plus every present argument once: for the
573    // common single-use body that is the result's length plus the parameter
574    // tokens it replaces, so the push/extend loop below never regrows (a
575    // template-only size regrew on every argument-bearing expansion;
576    // expansion bodies can be thousands of tokens in the expl3 kernel). An
577    // empty body (`\@gobble`, `\use_none:n`) returns before any allocation; a
578    // non-empty one reserves for every present argument without scanning the
579    // body for parameters, so one that names none over-reserves, harmlessly:
580    // the hot caller (`Expandable::invoke`) only gets here for a body that has one.
581    if self.0.is_empty() {
582      return Tokens::new(Vec::new());
583    }
584    let args_len: usize = args
585      .iter()
586      .flatten()
587      .map(|arg| arg.as_ref().unlist_ref().len())
588      .sum();
589    let mut result = Vec::with_capacity(self.0.len() + args_len);
590    for token in self.0.iter() {
591      if token.get_catcode() != Catcode::ARG {
592        // Non-match; copy it
593        result.push(*token);
594      } else {
595        let idx = token.with_str(|ts| ts.parse::<usize>().unwrap_or(0));
596        if idx > 0
597          && idx <= args.len()
598          && let Some(ref arg) = args[idx - 1]
599        {
600          // `arg` is `Cow<Tokens>`; iterate via `unlist_ref` + copy
601          // (Tokens is a Vec<Token> of `Copy` tokens). Avoids the
602          // previous `clone().into_owned().unlist()` chain which
603          // double-cloned the Vec when `arg` was `Cow::Borrowed`.
604          result.extend(arg.as_ref().unlist_ref().iter().copied());
605        }
606      }
607    }
608    Tokens::new(result)
609  }

611  /// Consumes a Tokens to a string containing TeX that created it (or could have).
612  /// Note that this is not necessarily the original TeX code; expansions or other substitutions may
613  /// have taken place.
614  ///
615  /// **Design decision:** The Perl `UnTeX` inserts `%\n` line-breaks (TeX comment + newline) when
616  /// a token string would exceed 78 characters. The Rust port deliberately omits this feature.
617  /// Line-break insertion is purely cosmetic and makes test expectations fragile — the `%\n`
618  /// appears verbatim in `tex=` attributes of `ltx:Math` elements, causing test XML files to
619  /// contain `%&#10;` escape sequences that depend on exact token lengths. We instead always
620  /// produce compact, single-line output. Test `.xml` files should not contain `%&#10;`.
621  pub fn untex(self) -> String {
622    // `VecDeque::from(Vec)` reuses the Vec's heap buffer directly
623    // (no second allocation), unlike `.into_iter().collect()`.
624    let mut tokens: VecDeque<Token> = VecDeque::from(self.revert());
625    let mut tex_string = String::new();
626    let mut length = 0;
627    let mut level = 0;
628    let mut prevs = String::new();
629    let mut prevcc = Catcode::COMMENT;
630    while let Some(token) = tokens.pop_front() {
631      let cc = token.get_catcode();
632      if cc == Catcode::COMMENT {
633        continue;
634      }
635      let mut token_string = token.to_string();
636      // Note: \n only-used to fail alphanumeric test
637      let first_char = token_string.chars().next().unwrap_or('\n');
638      if cc == Catcode::LETTER {
639        // keep "words" together, just for aesthetics
640        while !tokens.is_empty() && tokens[0].get_catcode() == Catcode::LETTER {
641          tokens
642            .pop_front()
643            .unwrap()
644            .with_str(|front_str| token_string.push_str(front_str));
645        }
646      }
647
648      let l = token_string.len();
649      if cc == Catcode::BEGIN {
650        level += 1;
651      }
652      //  Seems a reasonable & safe time to line break, for readability, etc.
653      if cc == Catcode::SPACE && token_string == "\n" {
654        // preserve newlines already present
655        if length > 0 {
656          tex_string.push_str(&token_string);
657          length = 0;
658        }
659      // If this token is a letter (or otherwise starts with a letter or digit): space or linebreak
660      } else {
661        let last_prevs = prevs.chars().last().unwrap_or('_');
662        // Perl: $STATE->lookupCatcode($1) == CC_LETTER
663        // Must use actual catcode lookup, not just is_alphabetic(), because
664        // characters like @ may have catcode LETTER in some contexts.
665        let prev_is_letter = crate::state::lookup_catcode(last_prevs)
666          .map(|cc| cc == Catcode::LETTER)
667          .unwrap_or_else(|| last_prevs.is_alphabetic());
668
669        if (cc == Catcode::LETTER || (cc == Catcode::OTHER && first_char.is_alphanumeric()))
670          && prevcc == Catcode::CS
671          && prev_is_letter
672        {
673          // Insert a (virtual) space before a letter if previous token was a CS w/letters
674          // This is required for letters, but just aesthetic for digits (to me?)
675          let space = ' ';
676          tex_string.push(space);
677          tex_string.push_str(&token_string);
678          length += 1 + l;
679        } else {
680          tex_string.push_str(&token_string);
681          length += l;
682        }
683        if cc == Catcode::END {
684          level -= 1;
685        }
686        prevs = token_string;
687        prevcc = cc;
688      }
689    }
690    // Patch up nesting for valid TeX !!!
691    match level {
692      1..=i32::MAX => {
693        for _ in 0..level {
694          tex_string.push('}');
695        }
696      },
697      i32::MIN..=-1 => {
698        // Prepend `-level` opening braces in one alloc (was O(n²) with
699        // String::from("{") + &tex_string per iteration).
700        let n = (-level) as usize;
701        let mut prefixed = String::with_capacity(n + tex_string.len());
702        for _ in 0..n {
703          prefixed.push('{');
704        }
705        prefixed.push_str(&tex_string);
706        tex_string = prefixed;
707      },
708      0 => {},
709    }
710    tex_string
711  }

713  /// [`untex`](Tokens::untex), typed — the blessed way to get TeX markup out of
714  /// a `Tokens` and back into a tokenizing sink.
715  ///
716  /// Prefer this over `untex()` whenever the string is destined for
717  /// `Tokenize!` / `mouth::tokenize` / `mouth::tokenize_internal`: it is the
718  /// only [`struct@Tokens`]→[`TeXString`] conversion, so the sink's signature proves
719  /// the round trip cannot weld a control word (`\v S` stays `\v S`, not
720  /// `\vS`). `untex()` itself is unchanged for the callers that want a plain
721  /// `String` (a `tex=` attribute, a log message, a comparison).
722  pub fn untex_string(self) -> TeXString { TeXString::assembled(self.untex()) }

724  /// Packs repeated CC_PARAM tokens into CC_ARG tokens for use as a macro body (and other token
725  /// lists) Also unwraps \noexpand tokens, since that is also needed for macro bodies
726  /// (but not strictly part of packing parameters)
727  pub fn pack_parameters(self) -> Result<Self> {
728    // Result is at most the same size as input (param-digit pairs
729    // collapse 2→1; other tokens copy 1→1). Pre-sizing avoids the
730    // initial Vec doublings on 1k+ token expansions (common for
731    // expl3 macros).
732    let mut rescanned = Vec::with_capacity(self.0.len());
733    // `VecDeque::from(Vec)` reuses the Vec's heap buffer directly (no
734    // second allocation), unlike `into_iter().collect()` which copies.
735    let mut toks: VecDeque<Token> = VecDeque::from(self.unlist());
736    // tex.web resolves tokens by MEANING during macro-definition scanning
737    // (get_next assigns cur_cmd=mac_param for a CS `\let` to a catcode-6
738    // `#`), so an IMPLICIT parameter token participates in `#1`/`##`
739    // pairing exactly like a literal `#`. Generated code relies on it:
740    // ctexart.cls L1194 `\cs_new_protected:Npn \__ctex_patch_toc_width:n
741    // \c_parameter_token 1 { … \c_parameter_token 1 … }` (docstrip writes
742    // the CS spelling to survive catcode changes) — without this, the CS
743    // leaks literally into hook names → `\csname __hook package/#…` errors
744    // across the 50-doc ctex family.
745    let is_param_tok = |t: &Token| {
746      t.get_catcode() == Catcode::PARAM
747        || (t.get_catcode().is_active_or_cs()
748          && crate::state::with_meaning(t, |m| {
749            matches!(m, Some(crate::common::store::Stored::Token(l))
750                        if l.get_catcode() == Catcode::PARAM)
751          }))
752    };
753    while let Some(mut t) = toks.pop_front() {
754      if t.get_catcode() != Catcode::PARAM && is_param_tok(&t) {
755        t = T_PARAM!();
756      }
757      if t.get_catcode() == Catcode::PARAM && !toks.is_empty() {
758        let next_t = toks.pop_front();
759        let next_cc = next_t.as_ref().map(|t| t.get_catcode());
760        if next_cc == Some(Catcode::OTHER) {
761          // only group clear match token cases
762          rescanned.push(Token!(sym next_t.unwrap().get_sym(), Catcode::ARG));
763        } else if next_cc == Some(Catcode::PARAM) || next_t.as_ref().map(is_param_tok) == Some(true)
764        {
765          rescanned.push(t);
766        } else {
767          // A PARAM (`#`) followed by neither a digit nor another `#` is, in
768          // real documents, almost always a `\halign`/`\valign` alignment-cell
769          // marker embedded in a macro body (e.g. `\def\foo{\halign{#\hfil&...}}`)
770          // or the `#{` end-of-parameter-text delimiter — both VALID TeX where
771          // the catcode-6 `#` must survive losslessly into the template/preamble.
772          // Real TeX resolves the parameter-vs-cell ambiguity during alignment
773          // processing, a lower level than LaTeXML operates at, so we cannot
774          // reliably tell this apart from a genuine typo.
775          //
776          // Perl's packParameters (Tokens.pm L139) emits a *counted* Error here
777          // AND drops both tokens, corrupting the template — but Perl rarely
778          // reaches it (it often can't find the offending package and skips the
779          // raw load). We DO raw-load such packages, so erroring+dropping broke
780          // the error-free target for the common halign-in-macro idiom. Preserve
781          // both tokens and log at Info (non-counted) instead. Documented as a
782          // beneficial divergence in docs/parity/KNOWN_PERL_ERRORS.md item 1. Witness
783          // 2006.02269 (easyeqn.sty `{MATRIX}` env → `$\mathstrut##$` template;
784          // 2 errors → 0).
785          Info!(
786            "misdefined",
787            "expansion",
788            &s!(
789              "Lone # (catcode PARAM) preserved as alignment/template marker. In expansion {}",
790              Tokens::new(toks.clone().into_iter().collect()).to_string()
791            )
792          );
793          rescanned.push(t);
794          if let Some(nt) = next_t {
795            rescanned.push(nt);
796          }
797        }
798      } else {
799        rescanned.push(t);
800      }
801    }
802    Ok(Tokens::new(rescanned))
803  }

805  /// Trims outer braces (if they balance each other).
806  /// Strips exactly 1 layer of matching outer braces by default.
807  /// Should this also trim whitespace? or only if there are braces?
808  pub fn strip_braces(self) -> Self { self.strip_braces_n(1) }
809
810  /// Trims `layers` outer brace pairs (if they balance each other).
811  /// Also trims whitespace *outer to* the removed braces.
812  /// Follows the Perl Tokens.pm algorithm: first collects all balanced
813  /// brace pairs, then strips from outside-in, only removing pairs that
814  /// span the full remaining width.
815  pub fn strip_braces_n(self, mut layers: usize) -> Self {
816    let tokens = self.0;
817    let n = tokens.len();
818    if n <= 1 {
819      return Tokens::new(tokens);
820    }
821
822    let mut i0: usize = 0;
823    let mut i1: usize = n;
824
825    // skip past spaces at ends
826    while i0 < i1 && tokens[i0].get_catcode() == Catcode::SPACE {
827      i0 += 1;
828    }
829    while i1 > i0 && tokens[i1 - 1].get_catcode() == Catcode::SPACE {
830      i1 -= 1;
831    }
832
833    // Collect balanced pairs (innermost first due to stack order)
834    let mut opens: Vec<usize> = Vec::new();
835    let mut pairs: Vec<(usize, usize)> = Vec::new();
836    for i in i0..i1 {
837      match tokens[i].get_catcode() {
838        Catcode::BEGIN => opens.push(i),
839        Catcode::END => {
840          if let Some(j) = opens.pop() {
841            pairs.push((j, i));
842          } else {
843            return Tokens::new(tokens); // Unbalanced: Too many }
844          }
845        },
846        _ => {},
847      }
848    }
849    if !opens.is_empty() {
850      return Tokens::new(tokens); // Unbalanced: Too many {
851    }
852
853    // Strip layers from outside-in.
854    // pairs is ordered innermost-first, so pop() gives outermost pair first.
855    while layers > 0 {
856      layers -= 1;
857      if let Some((j0, j1)) = pairs.pop()
858        && j0 == i0
859        && j1 == i1 - 1
860      {
861        i0 += 1;
862        i1 -= 1;
863      }
864    }
865
866    // Empty after stripping
867    if i0 >= i1 {
868      return Tokens::new(Vec::new());
869    }
870
871    if i0 > 0 || i1 < n {
872      Tokens::new(tokens[i0..i1].to_vec())
873    } else {
874      Tokens::new(tokens)
875    }
876  }
877}

881// `latexml_codegen` proc-macros via `quote!{ ... #tokens_value ... }`
882// splices. Resolver v2 keeps proc-macro feature unification isolated,

893  }

1162  /// (`s!(…)`/`format!(…)` gives the first, `&some_local` the second), so an
1163  /// implicit conversion would silently reopen the bug this type exists to close
1164  /// (`\bib@@names` PR #399, `dcolumn`/`overpic` PR #400, MathSciNet review path
1165  /// issue 410).
1166  ///
1167  /// Mechanism (the `static_assertions::assert_not_impl_any!` trick, hand-rolled
1168  /// to avoid the dependency): a helper trait with a blanket impl for everything
1169  /// plus a second impl gated on `Into<TeXString>`. If BOTH apply the item path is
1170  /// ambiguous and this fails to compile; if only the blanket one applies it
1171  /// resolves. So "still compiles" == "the conversion does not exist".
1172  ///
1173  /// A `&str` whose lifetime is shorter than `'static` is rejected too, but not
1174  /// by this assertion — an unconstrained `&'_ str` here infers to `&'static str`,
1175  /// which is exactly the case that MUST convert. The compiler enforces that half
1176  /// at the call site instead, as an E0521 "borrowed data escapes … `'1` must
1177  /// outlive `'static`".
1178  const _NO_IMPLICIT_STRING_CONVERSION: fn() = || {
1179    trait AmbiguousIfConvertible<A> {
1180      fn some_item() {}
1181    }
1182    impl<T> AmbiguousIfConvertible<()> for T {}
1183    impl<T: Into<TeXString>> AmbiguousIfConvertible<u8> for T {}
1184
1185    let _ = <String as AmbiguousIfConvertible<_>>::some_item;
1186    let _ = <&String as AmbiguousIfConvertible<_>>::some_item;
1187  };
1188
1189  /// The other half: a `&'static str` — i.e. every TeX literal a binding types
1190  /// out — must keep converting implicitly, or ~125 call sites would need
1191  /// ceremony for nothing.
1192  const _LITERALS_STILL_CONVERT: fn() = || {
1193    fn sink(_: impl Into<TeXString>) {}
1194    sink(r"\relax");
1195    sink(TeXString::assembled(String::new()));
1196  };
1197
1198  /// …while the three blessed ways in must all still work. Kept beside the
1199  /// negative assertion so a "fix" that deletes the conversions to satisfy it is
1200  /// caught here.
1201  #[test]
1202  fn the_three_blessed_constructors_reach_the_sink() {
1203    // 1. a literal
1204    assert_eq!(crate::mouth::tokenize(r"\relax").to_string(), r"\relax");
1205    // 2. untex_string — the space that terminates `\v` survives the round trip
1206    let welded = crate::mouth::tokenize(r"\v Spakov");
1207    assert_eq!(
1208      crate::mouth::tokenize(welded.clone().untex_string()).to_string(),
1209      r"\vSpakov",
1210      "re-tokenizing untex_string() output must give back the SAME tokens \
1211       (whose Display is again the welded form) — i.e. \\v stayed \\v"
1212    );
1213    // …which is precisely what the welded string does NOT do:
1214    let welded_again = crate::mouth::tokenize(TeXString::assembled(welded.to_string()));
1215    assert_eq!(
1216      welded_again.unlist().first().map(|t| t.to_string()),
1217      Some(r"\vSpakov".to_string()),
1218      "the welded path collapses \\v + Spakov into the single undefined CS \
1219       \\vSpakov — the bug TeXString exists to make hard to write"
1220    );
1221    // 3. assembled
1222    assert_eq!(
1223      crate::mouth::tokenize(TeXString::assembled(format!(r"\the{}", "section"))).to_string(),
1224      r"\thesection"
1225    );
1226  }
