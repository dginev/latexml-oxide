60/// Reads NOMINAL_FONT_SIZE from state, defaulting to 10.0. Perl uses the value
61/// directly as a float (`Common/Font.pm:44`) — the `11pt` class option is
62/// `10.95` (LaTeX's `\@xipt`), so this must NOT truncate via `lookup_int` (#542).
63pub fn defsize() -> f64 {
64  let v = lookup_float("NOMINAL_FONT_SIZE").map_or(0.0, |f| f.0);
65  if v > 0.0 { v } else { 10.0 }
66}

448/// Find a Font Metric for a given fontname, fallback to 10pt or cmr as needed.
449/// Perl: getMetricForName
450pub fn get_metric_for_name(name: &str) -> &'static MetricData {
451  let base = if let Some(idx) = name.find(|c: char| c.is_ascii_digit()) {
452    &name[..idx]
453  } else {
454    name
455  };
456  // Try exact name first (e.g. "cmr10")
457  if let Some(m) = STDMETRICS.get(name) {
458    return m;
459  }
460  // Try base without size (e.g. "cmr" from "cmr10")
461  if let Some(m) = STDMETRICS.get(base) {
462    return m;
463  }
464  // Try base + "10". Stack-buffer concat avoids the per-call `format!`
465  // heap alloc — `get_metric_for_name` is reached through the
466  // per-character `get_metric` loop, so this is a real allocation
467  // site. Font basenames are short (≤ ~20 ASCII bytes); 32 bytes is
468  // ample padding.
469  let base_bytes = base.as_bytes();
470  if base_bytes.len() + 2 <= 32 {
471    let mut buf = [0u8; 32];
472    buf[..base_bytes.len()].copy_from_slice(base_bytes);
473    buf[base_bytes.len()] = b'1';
474    buf[base_bytes.len() + 1] = b'0';
475    if let Ok(s) = std::str::from_utf8(&buf[..base_bytes.len() + 2])
476      && let Some(m) = STDMETRICS.get(s)
477    {
478      return m;
479    }
480  }
481  // Ultimate fallback to "cmr"
482  STDMETRICS
483    .get("cmr")
484    .expect("STDMETRICS must contain 'cmr'")
485}

487/// The TFM parameters `\font` gives a font of this name, per point of its
488/// size — Perl's `getMetricForName` (Common/Font.pm:551-562): the name's own
489/// metric, else its family at 10pt, else Computer Modern Roman at the name's
490/// size, else cmr10; the size is the name's trailing digits (10 without). This
491/// port's size-less and hand-merged entries (`cmr`, `cmbx`, `cmex`, …) carry
492/// no parameters and are passed over, as Perl has none of them. Perl's Error
493/// on the cmr10 fallback is not ported: it fires for valid fonts (`ecrm1000`).
494pub fn font_parameters_for_name(name: &str) -> &'static [f64] {
495  let base = name.trim_end_matches(|c: char| c.is_ascii_digit());
496  let size = if base.len() < name.len() {
497    &name[base.len()..]
498  } else {
499    "10"
500  };
501  [name, &format!("{base}10"), &format!("cmr{size}"), "cmr10"]
502    .into_iter()
503    .filter_map(|candidate| STDMETRICS.get(candidate))
504    .map(|metric| metric.parameters)
505    .find(|parameters| !parameters.is_empty())
506    .unwrap_or(&[])
507}

975  /// Compute math bearing (inter-atom spacing) between two boxes.
976  /// Perl: math_bearing
977  pub fn math_bearing(&self, thisbox: &Digested, prevbox: &Digested) -> f64 {
978    let r0 = prevbox
979      .get_property("role")
980      .and_then(|s| match s.into_owned() {
981        Stored::String(sym) => Some(arena::with(sym, |s| s.to_string())),
982        _ => None,
983      })
984      .unwrap_or_else(|| "ID".to_string());
985    let r1 = thisbox
986      .get_property("role")
987      .and_then(|s| match s.into_owned() {
988        Stored::String(sym) => Some(arena::with(sym, |s| s.to_string())),
989        _ => None,
990      })
991      .unwrap_or_else(|| "ID".to_string());
992    let t0 = *MATH_ATOM_TYPE.get(r0.as_str()).unwrap_or(&0);
993    let t1 = *MATH_ATOM_TYPE.get(r1.as_str()).unwrap_or(&0);
994    let bearing = MATH_BEARINGS[t0][t1];
995    let style = self
996      .get_mathstyle()
997      .map(|s| s.to_string())
998      .unwrap_or_else(|| "text".to_string());
999    if bearing == 0 || (bearing < 0 && style != "display" && style != "text") {
1000      return 0.0;
1001    }
1002    // Look up the bearing register: 1=thinmuskip, 2=medmuskip, 3=thickmuskip
1003    let reg_cs = match bearing.unsigned_abs() {
1004      1 => T_CS!("\\thinmuskip"),
1005      2 => T_CS!("\\medmuskip"),
1006      3 => T_CS!("\\thickmuskip"),
1007      _ => return 0.0,
1008    };
1009    if let Ok(Some(def)) = lookup_definition(&reg_cs)
1010      && let Some(val) = def.value_of(Vec::new())
1011    {
1012      // Perl: $STATE->lookupDefinition(...)->valueOf->spValue
1013      // MuGlue->spValue = fixpoint($skip/UNITY, font->getMUWidth)
1014      //                 = kround((skip/UNITY) * MUWidth)
1015      // The raw skip is in mu*UNITY units; convert to sp via MUWidth.
1016      let skip = val.value_of();
1017      let mu_width = self.get_mu_width() as f64;
1018      return (skip as f64 / UNITY_F64 * mu_width).trunc();
1019    }
1020    0.0
1021  }

1050  /// Like `merge` but borrows `other` to avoid requiring callers to own or
1051  /// clone a Font just to pass into merge. Clones only the retained fields
1052  /// from `other` (cheap — Option<Cow<'static,str>> clones are free for
1053  /// Borrowed variants, and most Font fields are None in typical uses).
1054  pub fn merge_ref(&self, other: &Font) -> Self {
1055    // A font selected by `\font` replaces `cur_font` only (tex.web §1217); the next font
1056    // selection — a family, series, shape or encoding declaration, which LaTeX ends with
1057    // `\selectfont` (latex.ltx:12576-12579) — starts from NFSS's font again and ends the raw
1058    // one. Colour, a bare size and math style keep it. Guard
1059    // `perfect_kernel_batch58::raw_font_ends_at_a_font_selection`.
1060    if self.nfss_font.is_some()
1061      && other.name.is_none()
1062      && (other.family.is_some()
1063        || other.series.is_some()
1064        || other.shape.is_some()
1065        || other.encoding.is_some()
1066        || other.forcebold.is_some()
1067        || other.emph == Some(true))
1068    {
1069      return self.nfss_restored().merge_ref(other);
1070    }
1071    // Handle forcebold for compatibility (Perl lines 873-874)
1072    let mut series = other.series.clone();
1073    let mut force_series = other.forceseries;
1074    if other.forcebold == Some(true) {
1075      series = Some(Cow::Borrowed("bold"));
1076      force_series = Some(true);
1077    }
1078    // `forcebold` false is the normal math version (`\unboldmath`, `\mathversion{normal}`, latex.ltx:14092-14095,
1079    // :10545): the series medium again, no longer forced. Perl's merge ignores it (Font.pm:934-935, KNOWN_PERL_ERRORS
1080    // #687), so the bold version stayed to the end of the group (witness 1803.08052, Buras's
1081    // `\boldmath\section{…}\unboldmath`). Guard `perfect_kernel_batch78a::unboldmath_restores_the_normal_math_version`.
1082    let unbold = other.forcebold == Some(false);
1083    if unbold {
1084      series = Some(Cow::Borrowed("medium"));
1085    }
1086
1087    // Build flags from force options
1088    let mut flags: u8 = 0;
1089    if other.forcefamily == Some(true) {
1090      flags |= FLAG_FORCE_FAMILY;
1091    }
1092    if force_series == Some(true) {
1093      flags |= FLAG_FORCE_SERIES;
1094    }
1095    if other.forceshape == Some(true) {
1096      flags |= FLAG_FORCE_SHAPE;
1097    }
1098
1099    let oflags = self.flags.unwrap_or(0);
1100    // Perl: fallback to self if not overridden, or if force-flags on self prevent override
1101    let family = if other.family.is_none() || (oflags & FLAG_FORCE_FAMILY != 0) {
1102      self.family.clone()
1103    } else {
1104      other.family.clone()
1105    };
1106    let series = if !unbold && (series.is_none() || (oflags & FLAG_FORCE_SERIES != 0)) {
1107      self.series.clone()
1108    } else {
1109      series
1110    };
1111    let mut shape = if other.shape.is_none() || (oflags & FLAG_FORCE_SHAPE != 0) {
1112      self.shape.clone()
1113    } else {
1114      other.shape.clone()
1115    };
1116    let mut size = other.size.or(self.size);
1117    let color = other.color.or(self.color);
1118    // Perl: $bg = $$self[5] if (!exists $options{background});
1119    // Only override bg if `other` actually specifies one
1120    let bg = if other.bg.is_some() {
1121      other.bg
1122    } else {
1123      self.bg
1124    };
1125    let opacity = other.opacity.clone().or_else(|| self.opacity.clone());
1126    let encoding = other.encoding.clone().or_else(|| self.encoding.clone());
1127    let language = other.language.clone().or_else(|| self.language.clone());
1128    let mut mathstyle = other.mathstyle.clone().or_else(|| self.mathstyle.clone());
1129    flags |= self.flags.unwrap_or(0);
1130    if unbold {
1131      flags &= !FLAG_FORCE_SERIES;
1132    }
1133
1134    // Dynamic adjustment directives
1135    if let Some(scale) = other.scale
1136      && let Some(ref mut sz) = size
1137    {
1138      *sz *= scale;
1139    }
1140
1141    // Scale factor for mathstyle-based sizing
1142    let style_scale = if let Some(sz) = self.size {
1143      let key: &str = self.mathstyle.as_deref().unwrap_or("display");
1144      sz / *STYLE_SIZE.get(key).unwrap_or(&10) as f64
1145    } else {
1146      1.0
1147    };
1148
1149    if other.size.is_some() {
1150      // Explicitly requested size, use it
1151    } else if other.mathstyle.is_some() {
1152      // Set the size from mathstyle
1153      let ms: &str = mathstyle.as_deref().unwrap_or("display");
1154      size = Some(style_scale * *STYLE_SIZE.get(ms).unwrap_or(&10) as f64);
1155    } else if other.scripted == Some(true) {
1156      // Adjust both the mathstyle & size for scripts
1157      let ms: &str = mathstyle.as_deref().unwrap_or("display");
1158      mathstyle = SCRIPT_STYLE_MAP.get(ms).map(|c| Cow::Borrowed(*c));
1159      let new_ms: &str = mathstyle.as_deref().unwrap_or("display");
1160      size = Some(style_scale * *STYLE_SIZE.get(new_ms).unwrap_or(&10) as f64);
1161    } else if other.fraction == Some(true) {
1162      // Adjust both for fractions
1163      let ms: &str = mathstyle.as_deref().unwrap_or("display");
1164      mathstyle = FRAC_STYLE_MAP.get(ms).map(|c| Cow::Borrowed(*c));
1165      let new_ms: &str = mathstyle.as_deref().unwrap_or("display");
1166      size = Some(style_scale * *STYLE_SIZE.get(new_ms).unwrap_or(&10) as f64);
1167    }
1168
1169    // Emphasis handling (Perl lines 909-912)
1170    if other.emph == Some(true) {
1171      shape = if shape.as_deref() == Some("italic") {
1172        Some(Cow::Borrowed("upright"))
1173      } else {
1174        Some(Cow::Borrowed("italic"))
1175      };
1176      flags |= FLAG_EMPH;
1177    }
1178    // Disable emph in math (Perl: $flags &= ~$FLAG_EMPH if $mathstyle)
1179    if mathstyle.is_some() {
1180      flags &= !FLAG_EMPH;
1181    }
1182
1183    let newfont = Font {
1184      family,
1185      series,
1186      shape,
1187      size,
1188      color,
1189      bg,
1190      opacity,
1191      encoding,
1192      language,
1193      mathstyle,
1194      flags: Some(flags),
1195      // Carry over fields that aren't part of Perl's merge:
1196      mathstylestep: other.mathstylestep.or(self.mathstylestep),
1197      name: other.name.clone().or_else(|| self.name.clone()),
1198      nfss_font: self.nfss_font.clone(),
1199      emph: None,
1200      scripted: None,
1201      fraction: None,
1202      forceseries: if flags & FLAG_FORCE_SERIES != 0 {
1203        Some(true)
1204      } else {
1205        None
1206      },
1207      forcefamily: if flags & FLAG_FORCE_FAMILY != 0 {
1208        Some(true)
1209      } else {
1210        None
1211      },
1212      forceshape: if flags & FLAG_FORCE_SHAPE != 0 {
1213        Some(true)
1214      } else {
1215        None
1216      },
1217      forcebold: None,
1218      scale: None,
1219    };
1220    // Note: Perl's merge() has an optional `specialize` option that is passed
1221    // explicitly (e.g. merge(specialize => $text)). It's NOT keyed on the font name.
1222    // Specialize is called at TBox creation time (tbox.rs) with the actual text content.
1223    // Do NOT call specialize here with the font name — it corrupts font properties
1224    // (e.g. resetting series "bold" to "medium" for font names like "cmb10").
1225    newfont
1226  }

1228  /// Instanciate the font for a particular class of symbols.
1229  /// NOTE: This works in `normal' latex, but probably needs some tunability.
1230  /// Depending on the fonts being used, the allowable combinations may be different.
1231  /// Getting the font right is important, since the author probably
1232  /// thinks of the identity of the symbols according to what they SEE in the printed
1233  /// document.  Even though the markup might seem to indicate something else...
1234  ///
1235  /// Use Unicode properties to determine font merging.
1236  pub fn specialize(&self, text: &str) -> Self {

1238    if text.is_empty() {
1239      return new; // ?
1240    }
1241    let (deffamily, defseries, defshape) = self.specialize_defaults();
1242    if LATIN_LETTER_RE.is_match(text) {
1243      // Latin Letter
1244      new.specialize_as_latin_letter();
1245    } else if GREEK_LETTER_RE.is_match(text) {
1246      // Single Greek character?
1247      if UPPER_LETTER_RE.is_match(text) {
1248        // Uppercase
1249        if new.family.is_none() || (new.family.as_ref().unwrap() == "math") {
1250          new.family = Some(deffamily);
1251          if new.shape.is_some() && (new.shape != Some(DEFSHAPE.into())) {
1252            new.shape = Some(defshape); // if ANY shape, must be default
1253          }
1254        }
1255      } else {
1256        // Lowercase
1257        new.specialize_as_lowercase_greek(self, deffamily, defseries);
1258      }
1259    } else if DIGIT_LETTER_RE.is_match(text) {
1260      // Digit
1261      if new.family.is_none() || (new.family.as_ref().unwrap() == "math") {
1262        new.family = Some(deffamily);
1263        new.shape = Some(defshape); // defaults, always.
1264      }
1265    } else {
1266      // Other Symbol
1267      new.family = Some(deffamily);
1268      new.shape = Some(defshape); // defaults, always.
1269      if new.series.is_some() && (new.series.as_ref().unwrap() != DEFSERIES) {
1270        new.series = Some(defseries);
1271      } // defaults, always.
1272    }
1273    new
1274  }

1351  pub fn distance(&self, other: &Font) -> i8 {

1353    // Normalize "math" → "serif" for comparison (Perl lines 436-437)
1354    let fam = self
1355      .family
1356      .as_deref()
1357      .map(|f| if f == "math" { "serif" } else { f });
1358    let ofam = other
1359      .family
1360      .as_deref()
1361      .map(|f| if f == "math" { "serif" } else { f });
1362    if is_diff_opt_str(fam, ofam) {
1363      distance += 1;
1364    }
1365    if is_diff_opt_str(self.series.as_deref(), other.series.as_deref()) {
1366      distance += 1;
1367    }
1368    if is_diff_opt_str(self.shape.as_deref(), other.shape.as_deref()) {
1369      distance += 1;
1370    }
1371    if is_diff_f64(self.size, other.size) {
1372      distance += 1;
1373    }

1375    // Perl's isDiff uses object reference equality: Cmyk(0,0,0,1) ≠ Rgb(0,0,0)
1376    // even though both are visually black.
1377    if is_diff_font_color_ref(self.color.as_ref(), other.color.as_ref()) {
1378      distance += 1;
1379    }

1381      distance += 1;
1382    }
1383    if is_diff_opt_str(self.opacity.as_deref(), other.opacity.as_deref()) {
1384      distance += 1;
1385    }
1386    // Perl does NOT count encoding differences
1387    // Perl does NOT count mathstyle differences
1388    if is_diff_opt_str(self.language.as_deref(), other.language.as_deref()) {
1389      distance += 1;
1390    }
1391    // Perl: ($flags & $FLAG_EMPH) ^ ($oflags & $FLAG_EMPH) ? 1 : 0
1392    let flags = self.flags.unwrap_or(0);
1393    let oflags = other.flags.unwrap_or(0);
1394    if (flags & FLAG_EMPH) ^ (oflags & FLAG_EMPH) != 0 {
1395      distance += 1;
1396    }
1397    distance

1457      diffs.push(family.clone().unwrap());

1459    }
1460    if is_diff(self.series.as_ref(), other.series.as_ref()) {
1461      let series = self.series.clone().unwrap();

1464    }

1466      let shape = self.shape.clone().unwrap();

1468      font_properties.shape.clone_from(&self.shape);
1469    }
1470    let mut result = HashMap::default();
1471
1472    if !diffs.is_empty() {
1473      let font_value = diffs.join(" ");
1474      result.insert(s!("font"), (font_value, font_properties));
1475    }
1476
1477    if is_diff_f64(self.size.as_ref().copied(), other.size.as_ref().copied()) {
1478      result.insert(
1479        "fontsize".to_string(),
1480        (
1481          relative_font_size(self.size.unwrap(), other.size.unwrap()),
1482          Font {
1483            size: self.size,
1484            ..Font::default()
1485          },
1486        ),

1488    }

1494        "color".to_string(),
1495        (effective_color.to_attribute(), Font {
1496          color: Some(effective_color),
1497          ..Font::default()
1498        }),
1499      );
1500    }
1501    if is_diff_color(self.bg.as_ref(), other.bg.as_ref()) {
1502      result.insert(
1503        "backgroundcolor".to_string(),
1504        (self.bg.as_ref().unwrap().to_attribute(), Font {
1505          bg: self.bg,
1506          ..Font::default()
1507        }),
1508      );
1509    }

1511      result.insert(
1512        "opacity".to_string(),
1513        (self.opacity.as_ref().unwrap().to_string(), Font {
1514          opacity: self.opacity.clone(),
1515          ..Font::default()
1516        }),
1517      );
1518    }
1519    // #638: the font `encoding` is a FontMap-lookup key consumed DURING digestion
1520    // (unicode decoding), not a presentation property — it is meaningless as an
1521    // output attribute and no ltx element even declares it, so Perl emits no
1522    // `@encoding` anywhere. Emitting it here only ever surfaced (spuriously) on a
1523    // foreign `xhtml:*` element whose `attribute *` wildcard accepts anything.
1524    // Omit it from the relativized attribute set entirely (Perl `Font.pm` keeps
1525    // the analogous line, but the value never reaches real output there).
1526    if is_diff(self.language.as_ref(), other.language.as_ref()) {
1527      result.insert(
1528        "xml:lang".to_string(),
1529        (self.language.as_ref().unwrap().to_string(), Font {
1530          language: self.language.clone(),
1531          ..Font::default()
1532        }),
1533      );
1534    }
1535    // Emph: (!$mstyle && $flags && ($flags & $FLAG_EMPH) && (!$oflags || !($oflags & $FLAG_EMPH))
1536    let flags = self.flags.unwrap_or(0);
1537    let oflags = other.flags.unwrap_or(0);
1538    if self.mathstyle.is_none()
1539      && flags != 0
1540      && (flags & FLAG_EMPH) != 0
1541      && (oflags == 0 || (oflags & FLAG_EMPH) == 0)
1542    {
1543      result.insert(
1544        "element".to_string(),
1545        ("ltx:emph".to_string(), Font::default()),
1546      );
1547    }
1548    // We do NOT want mathstyle showing up automatically in the attributes
1549    result
1550  }

1617  /// Find a Font Metric corresponding to this font's family_series_shape_size
1618  /// that contains the given `char`, if given.
1619  /// Try to find a fallback metric if `char` is not in the current Font.
1620  /// Perl: getMetric
1621  pub fn get_metric(&self, c_opt: Option<char>) -> &MetricData {
1622    let family = self.family.as_deref().unwrap_or("serif");
1623    let series = self.series.as_deref().unwrap_or("medium");
1624    let shape = self.shape.as_deref().unwrap_or("upright");
1625    let size = self.size.unwrap_or_else(defsize) as i64;
1626    // Stack buffer for char→&str lookup key, reused across paths. Avoids
1627    // one String allocation per character per get_metric call (which is
1628    // called per-character inside compute_string_size).
1629    let mut ch_buf = [0u8; 4];
1630    let ch_key = c_opt.map(|c| c.encode_utf8(&mut ch_buf) as &str);
1631    if let Some(name) = lookup_metric_name(family, series, shape) {
1632      let fullname = format!("{name}{size}");
1633      if let Some(metric) = STDMETRICS.get(fullname.as_str())
1634        && ch_key.is_none_or(|k| metric.sizes.contains_key(k))
1635      {
1636        return metric;
1637      }
1638      // Try base name fallback
1639      let metric = get_metric_for_name(name);
1640      if ch_key.is_none_or(|k| metric.sizes.contains_key(k)) {
1641        return metric;
1642      }
1643    }
1644    // Look for a fallback metric if char given
1645    if let Some(k) = ch_key {
1646      for name in METRIC_FALLBACKS {
1647        let fullname = format!("{name}{size}");
1648        let metric = STDMETRICS
1649          .get(fullname.as_str())
1650          .unwrap_or_else(|| get_metric_for_name(name));
1651        if metric.sizes.contains_key(k) {
1652          return metric;
1653        }
1654      }
1655    }
1656    get_metric_for_name("cmr")
1657  }

1675  pub fn compute_string_size(
1676    &self,
1677    text: &str,
1678    _options: SymHashMap<Stored>,
1679  ) -> (Dimension, Dimension, Dimension) {
1680    if text.is_empty()
1681      || self
1682        .get_family()
1683        .map(|fam| fam == "nullfont")
1684        .unwrap_or(false)
1685    {
1686      return (
1687        Dimension::default(),
1688        Dimension::default(),
1689        Dimension::default(),
1690      );
1691    }
1692    let size = self.get_size().unwrap_or_else(defsize);
1693    let ismath = self.get_family().map(|fam| fam == "math").unwrap_or(false);
1694    let (mut w, mut h, mut d) = (0, 0, 0);
1695    // Iterate via Peekable — no intermediate Vec<char> allocation,
1696    // and we get O(1) lookahead for kerning between consecutive chars.
1697    let mut chars_iter = text.chars().peekable();
1698    // Stack buffers for char→&str and char+char→&str lookups, avoiding
1699    // String::to_string() + String::format!() heap allocations inside
1700    // the per-character hot loop. encode_utf8 writes directly to the
1701    // buffer and returns a borrowed &str slice — no allocation.
1702    let mut ch_buf = [0u8; 4];
1703    let mut kern_buf = [0u8; 8];
1704    while let Some(ch) = chars_iter.next() {
1705      let metric = self.get_metric(Some(ch));
1706      let ch_key = ch.encode_utf8(&mut ch_buf);
1707      let entry_opt = metric.sizes.get(ch_key);
1708      let (cw, ch_sz, cd, ci) = if let Some(entry) = entry_opt {
1709        *entry
1710      } else {
1711        (0.75 * UNITY_F64, 0.7 * UNITY_F64, 0.2 * UNITY_F64, 0.0)
1712      };
1713      w += (cw * size).trunc() as i64;
1714      // Kerning: check kern between this char and next.
1715      if let Some(&next_ch) = chars_iter.peek() {
1716        let first_len = ch.encode_utf8(&mut kern_buf).len();
1717        let second_len = next_ch.encode_utf8(&mut kern_buf[first_len..]).len();
1718        let kern_key = std::str::from_utf8(&kern_buf[..first_len + second_len]).unwrap();
1719        if let Some(kern) = metric.kerns.get(kern_key) {
1720          w += (size * kern).trunc() as i64;
1721        }
1722      }
1723      // Italic correction in math
1724      if ismath && ci != 0.0 {
1725        w += (size * ci).trunc() as i64;
1726      }
1727      h = max(h, (ch_sz * size).trunc() as i64);
1728      d = max(d, (cd * size).trunc() as i64);
1729    }
1730    // The 1 is so that any actual glyph appears to be non-empty.
1731    // This is presumably only necessary to deal with the flawed emptiness heiristics in Alignment?
1732    if w == 0 {
1733      w = 1;
1734    }
1735    (Dimension::new(w), Dimension::new(h), Dimension::new(d))
1736  }

1750  // Here's where I avoid trying to emulate Knuth's line-breaking...
1751  // Mostly for List & Whatsit: compute the size of a list of boxes.
1752  // Options _SHOULD_ include:
1753  //   width:  if given, pretend to simulate line breaking to that width
1754  //   height,depth : ?
1755  //   vattach : top, bottom, center, baseline (...?) affects how the height & depth are
1756  //      allocated when there are multiple lines.
1757  //   layout : horizontal or vertical !!!
1758  // Boxes that arent a Core Box, List, Whatsit or a string are IGNORED
1759  //
1760  // The big problem with width is to have it propogate down from where
1761  // it may have been specified to the actual nested box that will get wrapped!
1762  // Try to mask this (temporarily) by unlisting, and (pretending to ) breaking up too wide items
1763  //
1764  // Another issue; SVG needs (sometimes) real sizes, even if the programmer
1765  // set some dimensions to 0 (eg.)   We may need to distinguish & store
1766  // requested vs real sizes?
1767  /// Perl: Font.pm sub computeBoxesSize (L635-680)
1768  /// Compute the size of a List of boxes, dispatching to helpers based on layout mode.
1769  pub fn compute_boxes_size(
1770    &self,
1771    boxes: &[Digested],
1772    options: SymHashMap<Stored>,
1773  ) -> Result<(Dimension, Dimension, Dimension)> {
1774    // Perl L646-647: `elsif ($ref =~ /^LaTeXML::Core::(?:Box|Whatsit|Alignment)$/) {
1775    // return $boxes->getSize; }` — a single bare Box/Whatsit/Alignment (NOT a
1776    // List) returns its getSize directly, short-circuiting ahead of the
1777    // List/split_words logic. This is essential for an isSpace box carrying an
1778    // explicit height/depth (e.g. `\phantom`'s XMHint): split_words keeps only
1779    // its width as inter-word space and discards height/depth, but getSize honors
1780    // the full width/height/depth. In Perl the dispatch is on the type of the
1781    // measured object (a bare-box body, not a List); a single-element slice here
1782    // is the faithful analogue, since Perl's `List()` returns a lone box of the
1783    // list's mode itself (List.pm:41-44). `List::compute_size` floors a
1784    // horizontal list's height and depth after it, as `hpack` does.
1785    if let [single] = boxes
1786      && !matches!(single.data(), DigestedData::List(_))
1787    {
1788      let mut bx_clone = single.clone();
1789      let (w, h, d, ..) = bx_clone.get_size(None)?;
1790      return Ok((w, h, d));
1791    }
1792    // Perl L638: my $mode = $boxes->getProperty('mode') || 'restricted_horizontal';
1793    let mode_str = match options.get("mode") {
1794      Some(Stored::String(s)) => arena::with(*s, |s| s.to_string()),
1795      _ => "restricted_horizontal".to_string(),
1796    };
1797    // Perl: $vattach = $boxes->getProperty('vattach') || $options{vattach} || 'baseline'
1798    let vattach = match options.get("vattach") {
1799      Some(Stored::String(s)) => arena::with(*s, |s| s.to_string()),
1800      _ => "baseline".to_string(),
1801    };
1802    // Perl #2798: `elsif (my $width = ($mode =~ /horizontal$/) && $boxes->getProperty('width'))`
1803    // — a horizontal list is formatted as a paragraph IFF an explicit width is
1804    // supplied (recorded by S4's repack_horizontal). NO `\hsize` fallback: a
1805    // horizontal List without a width property is restricted_horizontal (a single
1806    // line, no wrapping), matching Perl.
1807    let para_width: Option<i64> = if mode_str.ends_with("horizontal") {
1808      match options.get("width") {
1809        Some(Stored::Dimension(d)) => Some(d.value_of()),
1810        Some(Stored::Int(i)) => Some(*i),
1811        _ => None,
1812      }
1813    } else {
1814      None
1815    };
1816    // Perl #2798: baseline (sp) — from List property (S4) / option, default 12pt.
1817    let baseline: i64 = match options.get("baseline") {
1818      Some(Stored::Dimension(d)) => d.value_of(),
1819      Some(Stored::Int(i)) => *i,
1820      _ => 12 * UNITY,
1821    };
1822    let mut maxwidth: i64 = 0;
1823    // Perl #2798: lines are now [baseline, wd, ht, dp] (per-line baseline; -1 = no
1824    // inter-line adjustment, e.g. \vskip / \hrule).
1825    let mut lines: Vec<[i64; 4]> = Vec::new();
1826    if mode_str.ends_with("vertical") {
1827      // Perl: For vertical, ALL boxes are lines. (Rust makes the box-list match
1828      // Perl's structure BEFORE this point: a text-mode `{...}` group that
1829      // breaks a paragraph repacks the outer paragraph at digestion — see
1830      // `tex_box.rs` `{` primitive R2 / OXIDIZED_DESIGN #100 — so no run of loose
1831      // characters ever reaches here to be mis-counted one-line-per-glyph.)
1832      for bx in boxes {
1833        // A penalty, like glue, makes a top-attached box 0pt high when it comes first (tex.web §1087).
1834        if bx.get_property_bool("isPenalty") {
1835          if lines.is_empty() && vattach == "top" {
1836            lines.push([-1, 0, 0, 0]);
1837          }
1838          continue;
1839        }
1840        if bx.has_property("isEmpty") {
1841          continue;
1842        }
1843        // Perl: a horizontal sub-List WITH a width is formatted as a paragraph.
1844        if matches!(bx.data(), DigestedData::List(_))
1845          && bx
1846            .get_property("mode")
1847            .map(|v| v.to_string())
1848            .unwrap_or_default()
1849            == "horizontal"
1850          && let Some(w) = bx.get_property("width").and_then(|v| match &*v {
1851            Stored::Dimension(d) => Some(d.value_of()),
1852            Stored::Int(i) => Some(*i),
1853            _ => None,
1854          })
1855        {
1856          if w > maxwidth {
1857            maxwidth = w;
1858          }
1859          let sub_baseline = bx
1860            .get_property("baseline")
1861            .and_then(|v| match &*v {
1862              Stored::Dimension(d) => Some(d.value_of()),
1863              Stored::Int(i) => Some(*i),
1864              _ => None,
1865            })
1866            .unwrap_or(baseline);
1867          let items = bx.unlist();
1868          if items.is_empty() {
1869            // A paragraph's empty line (stomach.rs `end_graf`, `\leavevmode\par`): no height, but a line on the
1870            // list, so the next box is a `\baselineskip` below it (tex.web §1096).
1871            lines.push([sub_baseline, w, 0, 0]);
1872          } else {
1873            lines.extend(self.linebreak_paragraph(&items, w, sub_baseline)?);
1874          }
1875          continue;
1876        }
1877        // Perl: single box → one line, with baseline (or -1 for vskip/rule).
1878        // DIVERGENCE from Perl #2798 (upstream candidate): Perl folds vskips
1879        // and rules into one `-1` flag and RESETS prevdepth for both, so any
1880        // glue item between lines silently disables \baselineskip accounting
1881        // — a stack of N verbatim lines interleaved with fancyvrb's interline
1882        // vspace measures as Σ(h+d) ≈ N×6pt instead of N×\baselineskip
1883        // (witness 2605.00468: 49-line Prompt boxes budgeted at half their
1884        // TeX height; content spilled through every following box). TeX truth
1885        // (tex.web vpack): \prevdepth is TRANSPARENT to glue — only a BOX
1886        // updates it (to its depth), and only \hrule disables it (sentinel
1887        // \prevdepth = -1000pt). Encode vskip as -1 (transparent) and rule
1888        // as -2 (reset) so the stack can honor both.
1889        let (w, h, d) = self.compute_boxes_size_box(bx)?;
1890        let bs = if bx.get_property_bool("isHorizontalRule") {
1891          -2
1892        } else if bx.get_property_bool("isVerticalSpace") {
1893          -1
1894        } else {
1895          baseline
1896        };
1897        // A leading skip or kern sets a top-attached box's height to 0pt even when it is empty (tex.web §1087):
1898        // `\vtop{\kern0pt …}`, as `\vspace{0pt}` atop a `[t]` minipage, puts the reference point at the top.
1899        let leading_skip = bs == -1 && lines.is_empty() && vattach == "top";
1900        if w != 0 || h != 0 || d != 0 || leading_skip {
1901          lines.push([bs, w, h, d]);
1902        }
1903      }
1904    } else if let Some(w) = para_width {
1905      // Perl: proper paragraph — flatten, split into words, break into lines.
1906      if w > maxwidth {
1907        maxwidth = w;
1908      }
1909      let flat: Vec<&Digested> = boxes
1910        .iter()
1911        .filter(|b| !b.has_property("isEmpty"))
1912        .collect();
1913      let flat_owned: Vec<Digested> = flat.into_iter().cloned().collect();
1914      lines = self.linebreak_paragraph(&flat_owned, w, baseline)?;
1915    } else {
1916      // Perl: restricted_horizontal or math — one line, no wrapping.
1917      let filtered: Vec<&Digested> = boxes
1918        .iter()
1919        .filter(|b| !b.has_property("isEmpty"))
1920        .collect();
1921      let words = self.compute_boxes_size_words(&filtered)?;
1922      lines = Self::compute_boxes_size_lines(None, baseline, &words);
1923    }
1924    // Perl: stack up the multiple lines; mathaxis = size/4.
1925    let size = self.get_size().unwrap_or_else(defsize) as i64;
1926    let mathaxis = size * UNITY / 4;
1927    static SIZE_TRACE: std::sync::LazyLock<bool> =
1928      std::sync::LazyLock::new(|| std::env::var("LXML_SIZE_TRACE").is_ok());
1929    if *SIZE_TRACE {
1930      eprintln!(
1931        "SIZE mode={mode_str} vattach={vattach} baseline={} nboxes={} lines={:?}",
1932        baseline as f64 / 65536.0,
1933        boxes.len(),
1934        lines
1935          .iter()
1936          .map(|l| [
1937            l[0] as f64 / 65536.0,
1938            l[1] as f64 / 65536.0,
1939            l[2] as f64 / 65536.0,
1940            l[3] as f64 / 65536.0
1941          ])
1942          .collect::<Vec<_>>()
1943      );
1944    }
1945    let skip_option = |key: &str| match options.get(key) {
1946      Some(Stored::Dimension(d)) => Some(d.value_of()),
1947      Some(Stored::Int(i)) => Some(*i),
1948      _ => None,
1949    };
1950    let interline = (skip_option("lineskip"), skip_option("lineskiplimit"));
1951    let (mut wd, mut ht, mut dp) =
1952      Self::compute_boxes_size_stack(&vattach, mathaxis, &lines, interline);

2461/// map slot holds the whole string, so the family's map replaces the encoding's multichar entries too. Perl reads a
2462/// family map only after its encoding's map has loaded; here a family map stands on its own, as `font_decode`
2463/// (content.rs) already reads it: `U` has no map of its own (u_fontmap.rs), and its families' — `U_msa`, `U_msb`,
2464/// `U_eur`, and bbold's `U_blackboard` — are each one font's layout, not the encoding's (bbold_sty.rs;
2465/// OXIDIZED_DESIGN_DIVERGENCES #27).
2466fn decoding_map(encoding: &str, font: Option<&Font>) -> (Option<Fontmap>, SymStr) {
2467  // `load_font_map` preloads; keys are memoized (2026-08-23 audit R6).
2468  let map = load_font_map(encoding);
2469  if let Some(family) = font.and_then(|font| font.get_family()) {
2470    let (family_map_key, family_multichar_key) = fontmap_family_key_syms(encoding, family);
2471    let family_map: Option<Fontmap> = with_value_sym(family_map_key, |v| v.and_then(|s| s.into()));
2472    if family_map.is_some() {
2473      return (family_map, family_multichar_key);
2474    }
2475  }
2476  (map, fontmap_key_syms(encoding).multichar)
2477}

2479pub fn decode(code: u8, encoding_opt: Option<String>, implicit: bool) -> Option<char> {
2480  let mut font = None;
2481  let encoding = match encoding_opt {
2482    Some(enc) => Cow::Owned(enc),
2483    None => {
2484      // Perl `FontDecode`: `$encoding = $font->getEncoding || 'OT1'`
2485      // (Package.pm L2877). The `|| 'OT1'` is NOT shared with
2486      // `FontDecodeString` (L2906), whose port is `decode_string` below and
2487      // which deliberately keeps the empty fallback — do not "align" the two.
2488      //
2489      // This branch is the one `\char` reaches (via `decode_str`, encoding
2490      // `None`). It matters in MATH mode, where `Font::math_default()` sets
2491      // `encoding: None` on purpose: without the default, the lookup ran
2492      // against the empty encoding and `$\char65$` decoded to NOTHING where
2493      // Perl gives `A`. The default lives here, rather than at the call site,
2494      // so the font stays in scope for the `<enc>_<family>_fontmap`
2495      // refinement below — resolving the encoding earlier and passing it in
2496      // would silently drop `\ttfamily`'s `OT1_typewriter` variant.
2497      font = lookup_font();
2498      if let Some(ref font) = font {
2499        match font.get_encoding() {
2500          None => Cow::Borrowed("OT1"),
2501          Some(encoding) => encoding.clone(),
2502        }
2503      } else {
2504        Cow::Borrowed("OT1")
2505      }
2506    },
2507  };
2508
2509  decode_with_map(code, &encoding, font.as_deref(), implicit)
2510}

2512/// Decode a codepoint in `encoding` as `font` draws it: through the font family's own map, when one is declared
2513/// (`decoding_map`). Perl `FontDecode($n, $encoding, $font)`, as `decodeMathChar` calls it (Package.pm:2990): the
2514/// math character's encoding is the font info's, its family the font's — `\mathbb{R}` under bbold is `U` in the
2515/// `blackboard` family, decoded through `U_blackboard` (bbold_sty.rs).
2516pub fn decode_in_font(code: u8, encoding: &str, font: &Font, implicit: bool) -> Option<char> {
2517  decode_with_map(code, encoding, Some(font), implicit)
2518}

2520fn decode_with_map(code: u8, encoding: &str, font: Option<&Font>, implicit: bool) -> Option<char> {
2521  let map = if encoding.is_empty() {
2522    None
2523  } else {
2524    decoding_map(encoding, font).0
2525  };
2526
2527  if implicit {
2528    if let Some(map) = map {
2529      if code < 128 {
2530        match map.get(code as usize) {
2531          None => None,
2532          Some(c) => *c,
2533        }
2534      } else {
2535        Some(code.into())
2536      }
2537    } else {
2538      Some(code.into())
2539    }
2540  } else if let Some(map) = map {
2541    match map.get(code as usize) {
2542      None => None,
2543      Some(c) => *c,
2544    }
2545  } else {
2546    None
2547  }
2548}

2550pub fn decode_string(string: SymStr, encoding_opt: Option<&str>, implicit: bool) -> SymStr {
2551  let empty_sym = pin!("");
2552  if string == empty_sym {
2553    return empty_sym;
2554  }
2555  let mut font = None;
2556  let encoding = match encoding_opt {
2557    None => {
2558      font = lookup_font();
2559      if let Some(ref font) = font {
2560        font.get_encoding().unwrap_or(&Cow::Borrowed(""))
2561      } else {
2562        ""
2563      }
2564    },
2565    Some(encoding) => encoding,
2566  };
2567
2568  // Memoized key syms — this runs per digested character run, and rebuilding
2569  // the "{encoding}_fontmap"-family key strings each call was ~1M allocations
2570  // on a 1.3 s paper (2026-08-23 audit R6). `load_font_map` preloads, so no
2571  // separate `preload_font_map` call is needed.
2572  let (map, multichar_key) = if encoding.is_empty() {
2573    (None, None)
2574  } else {
2575    let (map, multichar_key) = decoding_map(encoding, font.as_deref());
2576    (map, Some(multichar_key))
2577  };
2578
2579  // Load multi-char overrides if available
2580  let multichar_map: Option<HashMap<String, String>> = if let Some(multichar_key) = multichar_key {
2581    with_value_sym(multichar_key, |val_opt| {
2582      if let Some(Stored::HashString(m)) = val_opt {
2583        Some(m.clone())
2584      } else {
2585        None
2586      }
2587    })
2588  } else {
2589    None
2590  };
2591
2592  let mut result_string: String = String::new();
2593  arena::with(string, |str| {
2594    for c in str.chars() {
2595      if implicit {
2596        if let Some(ref map_ref) = map {
2597          let code = c as u16; // u16, so that Unicode chars get cast correctly
2598          if code < 128 {
2599            // Check multi-char override first
2600            if let Some(ref mc) = multichar_map
2601              && let Some(mc_str) = mc.get(&(code as u8).to_string())
2602            {
2603              result_string.push_str(mc_str);
2604              continue;
2605            }
2606            if let Some(Some(mapc_val)) = map_ref.get(code as usize) {
2607              push_fontmap_char(&mut result_string, *mapc_val, code as u8);
2608            }
2609          } else {
2610            result_string.push(c);
2611          }
2612        } else {
2613          result_string.push(c)
2614        }
2615      } else if let Some(ref map_ref) = map {
2616        let code = c as u8;
2617        // Check multi-char override first
2618        if let Some(ref mc) = multichar_map
2619          && let Some(mc_str) = mc.get(&code.to_string())
2620        {
2621          result_string.push_str(mc_str);
2622          continue;
2623        }
2624        if let Some(Some(mapc_val)) = map_ref.get(code as usize) {
2625          push_fontmap_char(&mut result_string, *mapc_val, code);
2626        }
2627      }
2628    }
2629  });
2630  arena::pin(result_string)
2631}

2633/// Convert stanard font size names, such as `tiny`, `Huge`, etc to f64
2634pub fn rationalize_font_size(size: &str) -> f64 {
2635  if let Some(symbolic) = FONT_SIZE.get(size) {
2636    *symbolic * defsize()
2637  } else {
2638    // Perl: return $size — if not a symbolic name, return the numeric value as-is
2639    size.parse::<f64>().unwrap_or_else(|_| defsize())
2640  }
2641}
