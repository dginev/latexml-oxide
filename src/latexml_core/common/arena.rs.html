1//! An arena for interning strings
2//!  (global, mutable, single thread only)
3//!
4//! Note: works under the assumption of a single threaded, short-lived process,
5//! where memory starvation will not be an issue. It would need a hard reset if the same process
6//! does multiple conversions, and a different implementation if one needs a thread-local arena.
7//!
8//! ## Borrow safety
9//!
10//! A lend ([`with`] and its siblings) holds a shared `RefCell` borrow of the arena for the closure's duration, so the
11//! buffer it lends from cannot move under it. A string interned while a lend is live cannot grow that buffer: it goes
12//! to the overflow page, its own `Rc<str>` under a symbol at or above `OVERFLOW_BASE`. One string has one symbol across
13//! both tiers until a [`reset`]: every intern probes the buffer first, then the page, and only a string neither holds
14//! is added (to the buffer outside a lend, to the page under one).
15

17  cell::{Cell, RefCell},

20  rc::Rc,

30type Interner = StringInterner<BufferBackend, BuildHasherDefault<FxHasher>>;

32#[thread_local]
33static ARENA: LazyLock<RefCell<Interner>> = LazyLock::new(|| {
34  // 131,072 = 2^17 — sized to absorb the latex.dump (109,863 entries)
35  // plus a typical conversion's content (~15k more), so the hot path
36  // hits no reallocation. Profiled 2026-05-12: a representative
37  // `\usepackage{glossaries}` + math conversion ends at ~125,628
38  // strings allocated.
39  //
40  // Cost: ~2-3 MB extra startup memory vs the prior 32,768. Worth it
41  // when running batched conversions over a corpus (one fewer
42  // BufferBackend Vec realloc + HashMap rehash per process startup);
43  // negligible on a single short-lived run.
44  RefCell::new(new_interner())
45});
46
47fn new_interner() -> Interner {
48  StringInterner::with_capacity_and_hasher(131_072, BuildHasherDefault::<FxHasher>::default())
49}

51/// The first overflow symbol: the arena's own symbols are byte offsets into its buffer, kept below 2 GiB.
52const OVERFLOW_BASE: usize = 1 << 31;
53
54/// The strings interned while a lend was live, which the arena's buffer could not take without moving.
55#[derive(Default)]
56struct Overflow {
57  texts:   Vec<Rc<str>>,
58  symbols: FxHashMap<Rc<str>, SymStr>,
59  /// New strings since [`take_overflow_interns`] last read the count.
60  fresh:   usize,
61}
62
63#[thread_local]
64static OVERFLOW: LazyLock<RefCell<Overflow>> = LazyLock::new(Default::default);
65
66/// The overflow page's symbol for `text`, if the page holds it.
67#[inline(never)]
68fn page_symbol(text: &str) -> Option<SymStr> { OVERFLOW.borrow().symbols.get(text).copied() }
69
70/// A symbol of the arena's buffer, refusing one past the 2 GiB the overflow symbols start at. Lookups pass theirs
71/// through too: a string interned past 2 GiB stays in the buffer after the panic is caught, never to read as an
72/// overflow symbol.
73#[inline]
74fn in_buffer(sym: SymStr) -> SymStr {
75  assert!(
76    sym.to_usize() < OVERFLOW_BASE,
77    "the string arena's buffer reached 2 GiB, where the overflow symbols start"
78  );
79  sym
80}

82#[inline]
83fn intern(text: &str) -> SymStr {
84  if let Ok(mut arena) = ARENA.try_borrow_mut() {
85    if let Some(sym) = arena.get(text) {
86      return in_buffer(sym);
87    }
88    // A string first interned under a lend keeps its overflow symbol.
89    return page_symbol(text).unwrap_or_else(|| in_buffer(arena.get_or_intern(text)));
90  }
91  intern_cold(text)
92}

94/// [`intern`] while a lend is live: a string either tier holds keeps its symbol (the buffer probed first, as most
95/// interns find their string); a new one goes to the overflow page.
96#[cold]
97#[inline(never)]
98fn intern_cold(text: &str) -> SymStr {
99  if let Some(sym) = ARENA.borrow().get(text) {
100    return in_buffer(sym);
101  }
102  let mut overflow = OVERFLOW.borrow_mut();
103  if let Some(&sym) = overflow.symbols.get(text) {
104    return sym;
105  }
106  let sym = SymStr::try_from_usize(OVERFLOW_BASE + overflow.texts.len())
107    .expect("the arena's overflow page is full");
108  let text: Rc<str> = Rc::from(text);
109  overflow.texts.push(Rc::clone(&text));
110  overflow.symbols.insert(text, sym);
111  overflow.fresh += 1;
112  sym
113}

115/// A symbol's text, from the arena's buffer (under the caller's borrow) or the overflow page (its own `Rc`).
116enum Text<'a> {
117  Buffer(&'a str),
118  Page(Rc<str>),
119}
120
121impl Deref for Text<'_> {
122  type Target = str;
123  #[inline]
124  fn deref(&self) -> &str {
125    match self {
126      Text::Buffer(text) => text,
127      Text::Page(text) => text,
128    }
129  }
130}

132#[inline]
133fn text(arena: &Interner, sym: SymStr) -> Text<'_> {
134  if is_paged(sym) {
135    Text::Page(page_text(sym))
136  } else {
137    Text::Buffer(buffer_text(arena, sym))
138  }
139}

141#[inline]
142fn is_paged(sym: SymStr) -> bool { sym.to_usize() >= OVERFLOW_BASE }
143
144/// The text of a symbol of the buffer (the callers' test of [`is_paged`] lets the compiler drop this one's).
145#[inline]
146fn buffer_text(arena: &Interner, sym: SymStr) -> &str {
147  assert!(
148    !is_paged(sym),
149    "an overflow symbol resolved in the arena's buffer"
150  );
151  // SAFETY: `resolve_unchecked` skips the bounds and UTF-8 checks. Every symbol below `OVERFLOW_BASE` came from
152  // the buffer's `get_or_intern` of a `&str`, and the buffer is append-only between `reset`s, so the symbol names a
153  // whole UTF-8 string in it. Callgrind measured the validating `resolve` at ~3% of total Ir (`str::from_utf8`).
154  unsafe { arena.resolve_unchecked(sym) }
155}

157#[cold]
158#[inline(never)]
159fn page_text(sym: SymStr) -> Rc<str> {
160  Rc::clone(&OVERFLOW.borrow().texts[sym.to_usize() - OVERFLOW_BASE])
161}

163/// Assign a static str into the arena, returning a unique symbol.
164pub fn pin_static(text: &'static str) -> SymStr { intern(text) }

166/// The [`pin!`] call sites that cached a symbol on this thread: [`reset`]
167/// replaces the arena and clears each, so a cached symbol never indexes the
168/// replaced arena. A site registers on its first (cold) pin, so the hot path
169/// stays one thread-local load.
170#[thread_local]
171static PINNED_SITES: RefCell<Vec<&'static std::thread::LocalKey<Cell<Option<SymStr>>>>> =
172  RefCell::new(Vec::new());
173
174/// A [`pin!`] call site's first pin after a [`reset`] (the cold path): pin the
175/// literal, cache it in the site and register the site for the next reset.
176#[doc(hidden)]
177#[cold]
178#[inline(never)]
179pub fn pin_site(
180  site: &'static std::thread::LocalKey<Cell<Option<SymStr>>>,
181  text: &'static str,
182) -> SymStr {
183  let sym = pin_static(text);
184  site.with(|c| c.set(Some(sym)));
185  PINNED_SITES.borrow_mut().push(site);
186  sym
187}

189/// Call-site-cached interning for string literals — the first call on
190/// a thread (after an arena [`reset`]) pins the literal via `pin_static`,
191/// later calls return the cached `SymStr` directly (a thread-local load,
192/// no arena access). Use this from hot state-key lookup sites so you can keep
193/// writing string literals at the call site and still skip the per-call
194/// `pin()` hash probe:
195///
196/// ```ignore
197/// if state::lookup_bool_sym(pin!("groupNonBoxing")) { ... }
198/// ```
199///
200/// Each call site gets its own thread-local cache (no dedicated pub-static
201/// constant per key), so there is no ergonomic cost beyond typing the macro
202/// name; a site registers in `PINNED_SITES` on its first pin so that
203/// [`reset`] can clear it.
204///
205/// Note: the macro `pin!` and the runtime-string function
206/// `arena::pin(s)` share a name but occupy different namespaces in
207/// Rust — `pin!(…)` is the macro, `pin(…)` is the function — so both
208/// remain callable.
209#[macro_export]
210macro_rules! pin {
211  ($s:literal) => {{
212    std::thread_local! {
213      static CACHED: std::cell::Cell<Option<$crate::common::arena::SymStr>> =
214        const { std::cell::Cell::new(None) };
215    }
216    match CACHED.with(std::cell::Cell::get) {
217      Some(sym) => sym,
218      None => $crate::common::arena::pin_site(&CACHED, $s),
219    }
220  }};
221}

223/// Assign a string into the arena, returning a unique symbol.
224///
225/// No overflow guard below 2 GiB: the main-level wall-clock watchdog (watchdog.rs) catches genuinely runaway loops
226/// long before any real-world workload approaches it. Earlier versions had both a call-count and a distinct-symbol
227/// sentinel; the call-count one false-fired on dedup-heavy hot loops, and the distinct-symbol one added a per-call
228/// `arena.len()` read on a hot path (~350k calls per doc). Neither cost was paying for itself.
229pub fn pin<S: AsRef<str>>(text: S) -> SymStr { intern(text.as_ref()) }
230
231/// Probe-only lookup: the symbol for `text` if it was ever interned, without
232/// interning it. For existence-style checks (e.g. `install_definition`'s
233/// `"{cs}:locked"` gate) a `None` here proves the key cannot be bound in any
234/// state table — and skipping the intern avoids permanently growing the arena
235/// with one probe-key twin per defined control sequence (2026-08-23 audit R6).
236pub fn get<S: AsRef<str>>(text: S) -> Option<SymStr> {
237  let text = text.as_ref();
238  let found = ARENA.borrow().get(text).map(in_buffer);
239  found.or_else(|| page_symbol(text))
240}

242/// The buffer [`get_suffixed`] and [`pin_qname`] spell their key in. Freed by [`reset`], as a `#[thread_local]` has no
243/// destructor.
244#[thread_local]
245static SUFFIXED_KEY: RefCell<String> = RefCell::new(String::new());
246
247/// [`get`] of `name` with `suffix` appended, without formatting a string: every definition asks this of its side keys
248/// (`"{cs}:locked"`, `"{cs}:error@stub"`), and almost none of them exists.
249pub fn get_suffixed(name: &str, suffix: &str) -> Option<SymStr> {
250  let mut key = SUFFIXED_KEY.borrow_mut();
251  key.clear();
252  key.push_str(name);
253  key.push_str(suffix);
254  get(key.as_str())
255}

257/// [`pin`] of `prefix:local`, spelled in a reused buffer rather than a new `String`: `get_node_qname` asks it of every
258/// element and attribute it names.
259pub fn pin_qname(prefix: &str, local: &str) -> SymStr {
260  let mut key = SUFFIXED_KEY.borrow_mut();
261  key.clear();
262  key.push_str(prefix);
263  key.push(':');
264  key.push_str(local);
265  pin(key.as_str())
266}

268/// ASCII char-pin cache: every unique ASCII byte resolves to a single
269/// SymStr for the lifetime of the thread's arena (append-only between
270/// [`reset`]s, which clears the cache with it). Cache entries use `u32::MAX` as the "not yet pinned"
271/// sentinel — every symbol value, the overflow page's included, is strictly below that.
272/// Called from `lookup_catcode` / `assign_catcode` on every token,
273/// so the RefCell + hashmap overhead on `pin` is a measurable cost
274/// (1.4% Ir per callgrind on siunitx-heavy fixtures). The fast path
275/// avoids the arena entirely for the common ASCII case.
276#[thread_local]
277static ASCII_CHAR_SYM: [Cell<u32>; 128] = [const { Cell::new(u32::MAX) }; 128];
278
279pub fn pin_char(c: char) -> SymStr {
280  let code = c as u32;
281  if code < 128 {
282    let cached = ASCII_CHAR_SYM[code as usize].get();
283    if cached != u32::MAX {
284      // (a prior `pin` below cached it, and `reset` clears the cache with the arena)
285      return SymStr::try_from_usize(cached as usize).expect("invalid cached ASCII SymStr");
286    }
287  }
288  let sym = {
289    let mut tmp = [0u8; 4];
290    let s = c.encode_utf8(&mut tmp);
291    pin(s)
292  };
293  if code < 128 {
294    ASCII_CHAR_SYM[code as usize].set(sym.to_usize() as u32);
295  }
296  sym
297}

299/// Resolve a symbol and call the closure with a `&str` reference. The closure may intern (`pin()`, `pin_char()`, …):
300/// the text is lent under a shared borrow of the arena, so a string new during the lend goes to the overflow page
301/// instead of growing (and moving) the buffer the text lives in (string-interner 0.20 `buffer.rs:144`).
302/// An overflow symbol's lend holds the borrow too, so every lend is one where a [`reset`] panics.
303#[inline]
304pub fn with<R, FnR>(sym: SymStr, caller: FnR) -> R
305where FnR: FnOnce(&str) -> R {
306  let arena = ARENA.borrow();
307  // (one branch, not `text()`'s two: the hottest lend)
308  let page;
309  let text = if is_paged(sym) {
310    page = page_text(sym);
311    &*page
312  } else {
313    buffer_text(&arena, sym)
314  };
315  caller(text)
316}

319#[inline]
320pub fn first_bytes<const N: usize>(sym: SymStr) -> Option<[u8; N]> {
321  with(sym, |text| text.as_bytes().first_chunk().copied())

325where FnR: FnOnce(&str, &str) -> R {

330pub fn with3<R, FnR>(sym1: SymStr, sym2: SymStr, sym3: SymStr, caller: FnR) -> R
331where FnR: FnOnce(&str, &str, &str) -> R {
332  let arena = ARENA.borrow();
333  caller(
334    &text(&arena, sym1),

340pub fn with_many<R, FnR>(syms: &[SymStr], caller: FnR) -> R
341where FnR: FnOnce(Vec<&str>) -> R {
342  let arena = ARENA.borrow();
343  let texts: Vec<Text> = syms.iter().map(|sym| text(&arena, *sym)).collect();
344  caller(texts.iter().map(|text| &**text).collect())
345}

357/// Free every interned string on this thread, returning the arena to a
358/// fresh, empty state.
359///
360/// **Danger:** this invalidates *every* outstanding [`SymStr`] on the
361/// thread — they become dangling indices that may resolve to unrelated
362/// strings after re-interning. It is only sound when **nothing on the
363/// thread will read a pre-reset `SymStr` again**: i.e. between fully
364/// independent conversions in a reused process (the test harness, where
365/// each test has already serialized its output to owned `String`s and
366/// the thread is about to exit or be re-initialized) or a future daemon
367/// that re-initializes the engine afterward. The single-conversion
368/// `latexml_oxide` binary never calls this — it exits instead.
369///
370/// Needed because the engine's roots are `#[thread_local]` *attribute*
371/// statics, which (unlike the `thread_local!` macro) do **not** run
372/// destructors on thread exit. Without an explicit reset, every reused
373/// thread leaks its interner (~tens of MB for a full document). See
374/// `latexml_core::reset_thread_engine`.
375pub fn reset() {
376  // A lend holds a shared borrow until its closure returns, so a reset from inside one finds the arena borrowed.
377  *ARENA
378    .try_borrow_mut()
379    .expect("arena::reset inside a lend (`arena::with`): the lent text would dangle") =
380    new_interner();
381  *OVERFLOW.borrow_mut() = Overflow::default();
382  // The cached symbols index the replaced arena: a character or literal first
383  // pinned mid-document would otherwise resolve to another string in the next
384  // conversion on this thread.
385  for cached in &ASCII_CHAR_SYM {
386    cached.set(u32::MAX);
387  }
388  for site in PINNED_SITES.borrow_mut().drain(..) {
389    site.with(|c| c.set(None));
390  }
391  *SUFFIXED_KEY.borrow_mut() = String::new();
392}

394/// Eagerly initialize this thread's `#[thread_local]` `ARENA` LazyLock.
395///
396/// `ARENA` is the *leaf* of the engine's thread-local dependency graph:
397/// every other root's `LazyLock` initializer interns symbols via [`pin`], so it
398/// reaches into `ARENA`. Calling this at conversion entry (see
399/// `Core::new`), before any other root is touched, guarantees those later
400/// initializers find a fully-constructed `ARENA` instead of triggering its
401/// initialization *re-entrantly from within their own*. That re-entrant
402/// cross-`#[thread_local]` initialization is benign on Linux/ELF TLS but is
403/// the documented macOS hazard (rust-lang/rust#29594) behind the macOS
404/// worker-thread memory corruption in issue #217. `ARENA`'s own initializer
405/// touches no other thread-local, so forcing it first is always safe.
406/// No behavioral change on Linux.
407pub(crate) fn force_init() { LazyLock::force(&ARENA); }
