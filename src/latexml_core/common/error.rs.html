37/// Depth of diagnostic-macro emission on this thread (see [`macro_diag_guard`]).
38/// A depth counter, not a bool: `Error!` can raise `Fatal!` inside its own
39/// scope (the too-many-errors escalation), nesting two guards.
40#[thread_local]
41static MACRO_DIAG_DEPTH: std::cell::Cell<u32> = std::cell::Cell::new(0);
42
43/// RAII marker: "the log record currently being emitted comes from a
44/// diagnostic macro that already counted itself via [`note_status`]".
45///
46/// The tally has TWO producers which must never overlap: the macros
47/// (`Info!`/`Warn!`/`Error!`/`Fatal!`/`Debug!`) count at RAISE time — even when
48/// output is suppressed, which the `MAX_ERRORS` cap depends on — and the
49/// logger backend counts every OTHER record it prints (raw `log::warn!` and
50/// friends, which previously printed `Warning:` lines that no counter ever
51/// saw: the 131 MB witness logged 12,105 `Warning:` lines and reported
52/// "2 warnings"). This guard is how the logger tells the two apart.
53pub struct MacroDiagGuard(());
54impl Drop for MacroDiagGuard {
55  fn drop(&mut self) { MACRO_DIAG_DEPTH.set(MACRO_DIAG_DEPTH.get().saturating_sub(1)); }
56}
57#[must_use = "the guard must live across the log emission it marks"]
58pub fn macro_diag_guard() -> MacroDiagGuard {
59  MACRO_DIAG_DEPTH.set(MACRO_DIAG_DEPTH.get() + 1);
60  MacroDiagGuard(())
61}

150/// Perl `$LaTeXML::IGNORE_ERRORS` (Common/Error.pm L362/L380/L396: `Error`,
151/// `Warn` and `Info` all `return if $LaTeXML::IGNORE_ERRORS`): inside a probing
152/// scope every diagnostic is dropped WHOLE — neither its line nor its count —
153/// so the log and the status can never disagree about it. This is the only
154/// sanctioned way to silence a scope; muting lines while keeping counts (or
155/// the reverse) is exactly the log/status inconsistency the pipeline must not
156/// produce. Returns the previous value; restore it on every exit path.
157pub fn set_ignore_diagnostics(ignore: bool) -> bool {
158  let prev = IGNORE_DIAGNOSTICS.get();
159  IGNORE_DIAGNOSTICS.set(ignore);
160  prev
161}
162/// Whether an `IGNORE_ERRORS`-style probing scope is active (see
163/// `set_ignore_diagnostics`).
164pub fn diagnostics_ignored() -> bool { IGNORE_DIAGNOSTICS.get() }

171/// A tally of the Error and Fatal records an ignoring scope dropped.
172#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
173pub struct IgnoredRecords {
174  pub errors: usize,
175  pub fatals: usize,
176}

178/// Perl `local $LaTeXML::IGNORE_ERRORS = 1` as a scope: ignores diagnostics
179/// while alive and restores the previous state on drop — panic-safe, unlike a
180/// manual restore after the probed call.
181pub struct IgnoreDiagnosticsScope {
182  previous: bool,
183  before:   IgnoredRecords,
184}
185impl IgnoreDiagnosticsScope {
186  pub fn new() -> Self {
187    Self {
188      previous: set_ignore_diagnostics(true),
189      before:   IGNORED_RECORDS.get(),
190    }
191  }
192  /// The Error and Fatal records dropped while this scope was alive: a scope that mutes a load can still say what
193  /// it muted (#825).
194  pub fn dropped(&self) -> IgnoredRecords {
195    let now = IGNORED_RECORDS.get();
196    IgnoredRecords {
197      errors: now.errors - self.before.errors,
198      fatals: now.fatals - self.before.fatals,
199    }
200  }
201}
202impl Default for IgnoreDiagnosticsScope {
203  fn default() -> Self { Self::new() }
204}
205impl Drop for IgnoreDiagnosticsScope {
206  fn drop(&mut self) { set_ignore_diagnostics(self.previous); }
207}

216/// Per-thread tracker for the most recently emitted error's
217/// `category:object` signature, plus the count of how many
218/// consecutive errors share the same signature. Used to detect
219/// runaway loops where a single pathological control-sequence (like
220/// plain-TeX `\tabalign` invoked in math mode → unbounded `\halign`
221/// cell loop) keeps emitting the same error indefinitely. See
222/// `wisdom_tabalign_math_runaway.md` for the canonical witness.
223#[thread_local]
224static LAST_ERROR_KEY: RefCell<Option<String>> = RefCell::new(None);
225#[thread_local]
226static CONSECUTIVE_ERROR_COUNT: std::cell::Cell<usize> = std::cell::Cell::new(0);
227
228/// Threshold for "same error fired this many times in a row → bail."
229/// Set well above any legitimate same-error pattern (a paper with
230/// 500+ identical errors would already be near-useless output) but
231/// well below the 10000 MAX_ERRORS cap so runaway papers don't
232/// accumulate huge noise logs. Empirically, the pathological
233/// `\tabalign`-in-math-mode runaway hits >9000 consecutive same
234/// errors; this catches that at 500 instead. The threshold was
235/// tightened from an initial 2000 after verifying no test in the
236/// 1112-test suite exceeds it.
237pub const MAX_CONSECUTIVE_ERRORS: usize = 500;
238
239/// Record an error signature; returns the new consecutive count.
240/// Call from the Error! macro after note_status. Resets count to 1
241/// on a different signature, increments on a match.
242pub fn note_consecutive_error(key: &str) -> usize {
243  let mut last = LAST_ERROR_KEY.borrow_mut();
244  if last.as_deref() == Some(key) {
245    let c = CONSECUTIVE_ERROR_COUNT.get() + 1;
246    CONSECUTIVE_ERROR_COUNT.set(c);
247    c
248  } else {
249    *last = Some(key.to_string());
250    CONSECUTIVE_ERROR_COUNT.set(1);
251    1
252  }
253}

255/// The ONE emission primitive every diagnostic flows through (DRY pass,
256/// user directive 2026-08-02): count on the emitting thread's `REPORT`,
257/// then log with the pre-formatted `target`, respecting output suppression —
258/// EXCEPT for `Error` and `Fatal` records, which are emitted UNCONDITIONALLY
259/// (user decision 2026-08-03): frameworks such as cortex aggregate success
260/// rates from `Error:`/`Fatal:` lines, so muting either would hide exactly
261/// the signal they measure. Suppression mutes Debug/Info/Warning only.
262/// The `MacroDiagGuard` marks the emission so the logger backend does not
263/// count it a second time.
264pub fn emit_record(status: LogStatus, target: &str, message: &str) {
265  let _diag_guard = macro_diag_guard();
266  // Perl `IGNORE_ERRORS`: a probing scope drops the record whole (no line, no
267  // count) — see `set_ignore_diagnostics`.
268  if diagnostics_ignored() {
269    let mut tally = IGNORED_RECORDS.get();
270    match status {
271      LogStatus::Error => tally.errors += 1,
272      LogStatus::Fatal => tally.fatals += 1,
273      _ => {},
274    }
275    IGNORED_RECORDS.set(tally);
276    return;
277  }
278  // After a `TooManyErrors` Fatal or a RESOURCE Fatal has latched, drop
279  // further Error-level records entirely (don't log, don't count). Perl dies
280  // at the Fatal so nothing ever logs past it; our recovery machinery keeps
281  // converting, and paths that swallow the `Error!` macro's Err (e.g.
282  // tex_logic::compare inside a bool-returning conditional) otherwise churn
283  // tens of thousands of post-cap records — gckanbun 12.8k, panda-doc 3.6k,
284  // past the tikz 1000-cap (perfect-kernel sweep 13). Once a resource Fatal
285  // (MemoryBudget, Timeout) is latched, the document is aborted and the
286  // recovery pass only drains what is pending: every Error that draining
287  // raises is teardown noise, not a finding — the group closers
288  // (`stomach::egroup` / `endgroup` / `end_mode_opt`, sweep 82) and, past
289  // those six site guards, the gullet's own drain errors (`conditional
290  // fell off end`, `unexpected \fi`, recursion of an interrupted pgf path:
291  // glossaries-extra-manual 5 → 6 → 27 errors from one RSS-fuse trip a
292  // megabyte apart, sweep 96). The Fatal line and every pre-Fatal error
293  // still report; Warnings and Info (the `recovery` notes) still do too.
294  // (The `Error!` macro's escalation bookkeeping still runs at the site — a
295  // drain that trips the consecutive-error or MAX_ERRORS check returns
296  // `Err(TooManyErrors)`, which cannot displace the latched resource Fatal:
297  // `record_last_fatal` keeps only `ErrorTarget::Timeout`.)
298  // After a resource Fatal or the error cap has latched, Error records are
299  // dropped (Perl `Fatal` sets `$LaTeXML::IGNORE_ERRORS = 1` after
300  // `hardYankProcessing`, Common/Error.pm L315, and `Error` returns at L362).
301  // Digestion stops at the Fatal (`core_interface::hard_yank_processing`), so
302  // what still reaches this gate is teardown noise, never a finding.
303  if matches!(status, LogStatus::Error) && (too_many_errors_latched() || resource_fatal_latched()) {
304    return;
305  }
306  let level = match status {
307    LogStatus::Debug => log::Level::Debug,
308    LogStatus::Info => log::Level::Info,
309    LogStatus::Warning => log::Level::Warn,
310    _ => log::Level::Error,
311  };
312  let unconditional = matches!(status, LogStatus::Error | LogStatus::Fatal);
313  note_status(status, None);
314  if unconditional || !is_log_output_suppressed() {
315    log::log!(target: target, level, "{message}");
316  }
317}

319/// The single diagnostic vehicle, function form — for contexts that cannot
320/// use the `Error!`/`Warn!`/`Info!` macros because those are return-based
321/// (`Error!` escalates to `Fatal!`, which `return Err(...)`s, so it only
322/// typechecks in `Result<_, error::Error>` functions).
323///
324/// Perl LaTeXML has exactly one emission vehicle per severity (`Error.pm`),
325/// which is what lets its tally and cortex's aggregation be lossless by
326/// construction. These functions restore that property for Rust's
327/// non-`Result` contexts (post-processing drivers, workers, the LSP server):
328/// they count, emit with a proper `category:object` target, respect output
329/// suppression, and participate in the runaway circuit-breakers — everything
330/// the raw `log::warn!`-family calls they replace silently skipped. Raw
331/// `log::*!` diagnostics are BANNED in workspace crates (see
332/// `tools/lint_raw_log_diag.sh`); the logger backend's own tally
333/// (`note_status_from_logger`) remains only as the net for FOREIGN crates
334/// logging through the `log` facade, which can never use this vehicle.
335pub fn emit_info(category: &str, object: &str, message: &str) {
336  emit_record(LogStatus::Info, &format!("{category}:{object}"), message);
337}

367/// `return`ing — there is no `Err` channel in the contexts it serves — so

369/// emits the `Fatal:TooManyErrors` record and latches the sticky fatal

374  note_error_cap(&format!("{category}:{object}"));

377/// The error cap's bookkeeping after one counted Error (`key` = `category:object`): at the crossing of `MAX_ERRORS` or
378/// of the consecutive-error limit, latch `TooManyErrors` and emit its Fatal (Perl Error.pm L362-372). Shared by
379/// [`emit_error`] and a replayed deferred error (logger.rs `DeferredDiagnostics`).
380pub fn note_error_cap(key: &str) {
381  // Perl `IGNORE_ERRORS` (Error.pm L362): an ignored record does nothing,
382  // not even cap bookkeeping.
383  if is_demote_fatals() || diagnostics_ignored() {
384    return;
385  }
386  let maxerrors = match crate::state::try_lookup_int("MAX_ERRORS") {
387    None => usize::MAX, // STATE contended: skip the check for this error
388    Some(v) if v > 0 => v as usize,
389    Some(_) => 100,
390  };
391  let consec = note_consecutive_error(key);
392  // Once latched, `emit_record` drops the repeats and the count freezes at
393  // the cap, so the crossing test below would stay true and re-fire the
394  // `Fatal:TooManyErrors` line on every later call. One Fatal, like Perl.
395  if too_many_errors_latched() {
396    return;
397  }
398  let over_total = get_status(LogStatus::Error) > maxerrors;
399  let over_consec = consec > MAX_CONSECUTIVE_ERRORS;
400  // Latch exactly at the crossing, not on every error past it.
401  if (over_total && get_status(LogStatus::Error) == maxerrors + 1)
402    || (over_consec && consec == MAX_CONSECUTIVE_ERRORS + 1)
403  {
404    latch_too_many_errors();
405    emit_fatal(
406      "TooManyErrors",
407      "MaxLimit",
408      &format!(
409        "Too many errors (> {})!",
410        if over_total {
411          maxerrors
412        } else {
413          MAX_CONSECUTIVE_ERRORS
414        }
415      ),
416    );
417  }
418}

451/// The whole diagnostic tally of this thread: the [`REPORT`] plus the gates

499pub fn note_status(status: LogStatus, what: Option<&str>) {
500  let mut report = REPORT.borrow_mut();
501  use LogStatus::*;
502  match status {
503    Debug => report.debug += 1,
504    Info => report.info += 1,
505    Warning => report.warning += 1,
506    Error => report.error += 1,
507    Fatal => {
508      // Diagnostic for "phantom fatals" (a fatal counted in the final summary
509      // with no `Fatal:` line in the log — an `Err` raised via `Fatal!` that
510      // some caller swallowed without `log_fatal`): dump a backtrace at the
511      // moment the tally first flips. Witness math0402448.
512      if !report.fatal && debug_fatal_enabled() {
513        eprintln!("[debug-fatal] LogStatus::Fatal first noted here:");
514        eprintln!("{}", std::backtrace::Backtrace::force_capture());
515      }
516      report.fatal = true;
517    },
518    Undefined => {
519      // `what` may borrow the arena buffer; `entry` re-interns via `arena::pin`,
520      // which can REALLOCATE that buffer and invalidate `what` mid-read, then
521      // intern whatever bytes now occupy the slot (e.g. a freshly-interned
522      // `\special_relax` family-token name → phantom undefined). Copy out first.
523      let key = what.unwrap_or_default().to_string();
524      let entry = report.undefined.entry(&key).or_insert(0);
525      *entry += 1;
526    },
527    Missing => {
528      let key = what.unwrap_or_default().to_string();
529      let entry = report.missing.entry(&key).or_insert(0);
530      *entry += 1;
531    },
532  }
533}

590thread_local! {
591  /// Latch for the most recent RESOURCE-class fatal (`ErrorTarget::Timeout`
592  /// with a unit category: token/pushback/if limits, cycle-guard recursion,
593  /// memory budget, conversion deadline). Some layers flatten `Error` into a
594  /// plain string on the way up (the marpa semantics boundary turns it into
595  /// `marpa::error::Error`), destroying the structured identity — which made
596  /// resource fatals indistinguishable from semantic parse rejections and
597  /// produced "phantom fatals" (counted in the summary, never logged, parse
598  /// grinding on). The `Fatal!` macro records here at raise time; consumers
599  /// `take` it to re-classify a flattened error. PR #249 review P1-4.
600  static LAST_RESOURCE_FATAL: RefCell<Option<Error>> = const { RefCell::new(None) };
601  /// Sticky companion of `LAST_RESOURCE_FATAL` (which consumers `take`): has a
602  /// resource fatal been raised in this conversion? Once it has, the
603  /// document is aborted and the recovery pass only drains what is pending;
604  /// the group-closer errors that draining raises (`stomach::egroup` /
605  /// `endgroup` / `end_mode_opt`) are teardown noise, not findings — 11-13
606  /// non-deterministic `Attempt to close a group that switched to mode`
607  /// lines after glossaries-user's MemoryBudget Fatal (sweep 82), a phantom
608  /// regression. Cleared in `initialize_report`.
609  static RESOURCE_FATAL_SEEN: std::cell::Cell<bool> = const { std::cell::Cell::new(false) };
610}

612/// Whether a resource-class Fatal (timeout, memory budget, runaway limits)
613/// has been raised in this conversion — see `RESOURCE_FATAL_SEEN`.
614pub fn resource_fatal_latched() -> bool { RESOURCE_FATAL_SEEN.with(|c| c.get()) }

616/// Record a fatal into the resource-fatal latch — only Timeout-target fatals
617/// with payload-free categories are kept (the latch exists for resource
618/// fatals; payload-carrying `ErrorCategory` variants are not cloneable and
619/// are never resource-class).
620pub fn record_last_fatal(e: &Error) {
621  use ErrorCategory as C;
622  if !matches!(e.target, ErrorTarget::Timeout) {
623    return;
624  }
625  let category = match &e.category {
626    C::TokenLimit => C::TokenLimit,
627    C::PushbackLimit => C::PushbackLimit,
628    C::Recursion => C::Recursion,
629    C::IfLimit => C::IfLimit,
630    C::MemoryBudget => C::MemoryBudget,
631    C::Convert => C::Convert,
632    _ => return,
633  };
634  LAST_RESOURCE_FATAL.with(|c| {
635    *c.borrow_mut() = Some(Error::from(ErrorData {
636      target: ErrorTarget::Timeout,
637      category,
638      message: e.message.clone(),
639    }));
640  });
641  RESOURCE_FATAL_SEEN.with(|c| c.set(true));
642}

644/// Unlatch the latched resource fatal when it is a runaway limit of the expansion in progress — the pushback, a
645/// recursion — for a run the document never makes, discarded with it (`title_page_probe_run`). A limit of the
646/// conversion stays latched as the document's: its deadline, its memory budget, its token budget, its conditional
647/// ids. (The token count itself does not carry past the failed run: `reading_from_mouth`'s error path resets the
648/// gullet's `progress`, gullet.rs.) Whether it was one.
649pub fn unlatch_runaway_fatal() -> bool {
650  use ErrorCategory as C;
651  let runaway = LAST_RESOURCE_FATAL.with(|c| {
652    matches!(c.borrow().as_ref(), Some(e) if matches!(e.category, C::PushbackLimit | C::Recursion))
653  });
654  if runaway {
655    LAST_RESOURCE_FATAL.with(|c| *c.borrow_mut() = None);
656    RESOURCE_FATAL_SEEN.with(|c| c.set(false));
657  }
658  runaway
659}

663/// propagate the real thing.
664pub fn take_last_resource_fatal() -> Option<Error> {
665  LAST_RESOURCE_FATAL.with(|c| c.borrow_mut().take())

693pub fn get_status_message() -> String {
694  let report = REPORT.borrow();
695  let mut parts = Vec::new();
696  if report.warning > 0 {
697    parts.push(format!(
698      "{} warning{}",
699      report.warning,
700      if report.warning > 1 { "s" } else { "" }
701    ));
702  }
703  if report.error > 0 {
704    parts.push(format!(
705      "{} error{}",
706      report.error,
707      if report.error > 1 { "s" } else { "" }
708    ));
709  }
710  if report.fatal {
711    parts.push("1 fatal error".to_string());
712  }
713  // (sorted, and so are the missing files: a hash map's order follows the symbols' interning history, which any
714  // work done before an error changes — the frontmatter observer's rehearsal, a new definition; Perl State.pm:771
715  // lists `keys %{$$status{undefined}}`, a random hash order)
716  let mut undef_keys: Vec<String> = report
717    .undefined
718    .keys()
719    .map(|k| crate::common::arena::to_string(*k))
720    .collect();
721  undef_keys.sort_unstable();
722  if !undef_keys.is_empty() {
723    parts.push(format!(
724      "{} undefined macro{}[{}]",
725      undef_keys.len(),
726      if undef_keys.len() > 1 { "s" } else { "" },
727      undef_keys.join(", ")
728    ));
729  }
730  let mut miss_keys: Vec<String> = report
731    .missing
732    .keys()
733    .map(|k| crate::common::arena::to_string(*k))
734    .collect();
735  miss_keys.sort_unstable();
736  if !miss_keys.is_empty() {
737    parts.push(format!(
738      "{} missing file{}[{}]",
739      miss_keys.len(),
740      if miss_keys.len() > 1 { "s" } else { "" },
741      miss_keys.join(", ")
742    ));
743  }
744  if parts.is_empty() {
745    "No obvious problems".to_string()
746  } else {
747    parts.join("; ")
748  }
749}

1015#[macro_export]
1016macro_rules! Error {
1017  ($category:expr_2021, $object:expr_2021, $message:expr_2021) => {{
1018    $crate::Error!($category,$object,$message,"")
1019  }};
1020 ($category:expr_2021, $object:expr_2021, $message:expr_2021, $($details:expr_2021),*) => {{
1021    $crate::common::error::emit_record(
1022      $crate::common::error::LogStatus::Error,
1023      &format!("{}:{}", $category, $object),
1024      &$crate::generate_message!($message, $($details),*),
1025    );
1026    // In the fatal-demotion scope (bibliography post-processing) the
1027    // too-many/consecutive-error escalations are SKIPPED: their Fatal!
1028    // would demote back into an Error, turning the circuit-breaker into
1029    // an error multiplier (run-233 follow-up: 470 self-feeding
1030    // "Too many errors" lines on 2605.02213). The bib interpreter has its
1031    // own bounded failure latch instead.
1032    // An ignored record (Perl `IGNORE_ERRORS`, Error.pm L362) does nothing at
1033    // all — no count, no cap bookkeeping: 500 identical forward-reference
1034    // errors inside the degraded expl3 raw-load must not trip the consecutive
1035    // cap on the first error after the scope.
1036    if !$crate::common::error::is_demote_fatals() && !$crate::common::error::diagnostics_ignored() {
1037    // Borrow-safe read: an Error! can be raised from inside a `state_mut()`
1038    // scope (e.g. push_value's BUG branch, a constructor's after_digest),
1039    // where a plain `lookup_int` would panic "RefCell already mutably
1040    // borrowed" and abort the conversion (tikz-cd 2001.08973).
1041    let max_from_state = $crate::state::try_lookup_int("MAX_ERRORS");
1042    // Match Perl LaTeXML default of 100 errors before Fatal('too_many_errors').
1043    // Past 100 errors a paper has already failed comprehension; continuing
1044    // produces noise without information. Override via state for tests
1045    // or specific bindings (e.g. tikz_sty raises to 1000, dump-build raises
1046    // to 1_000_000).
1047    let maxerrors = match max_from_state {
1048      // STATE contended: we cannot read the (possibly raised) cap, so skip the
1049      // too-many-errors check for *this* error rather than risk a spurious
1050      // Fatal from a stale default. The next uncontended error re-applies it.
1051      None => usize::MAX,
1052      Some(v) if v > 0 => v as usize,
1053      Some(_) => 100,
1054    };
1055    if $crate::common::error::get_status($crate::common::error::LogStatus::Error) > maxerrors {
1056      $crate::common::error::latch_too_many_errors();
1057      Fatal!(TooManyErrors, MaxLimit(maxerrors), format!("Too many errors (> {maxerrors})!"));
1058    }
1059    // Runaway-loop early-bail: if the same error signature has fired
1060    // MAX_CONSECUTIVE_ERRORS times in a row, we're stuck in a loop
1061    // (the canonical witness is plain-TeX `\tabalign` invoked in math
1062    // mode → unbounded `\halign` cell loop emitting `\hbox` end-mode
1063    // mismatches). Bail before MAX_ERRORS so logs stay short and
1064    // post-processing sees a clear cause. The threshold is well above
1065    // any legitimate same-error pattern (real papers max out at a few
1066    // hundred unique errors).
1067    let __consec_key = format!("{}:{}", $category, $object);
1068    let __consec = $crate::common::error::note_consecutive_error(&__consec_key);
1069    if __consec > $crate::common::error::MAX_CONSECUTIVE_ERRORS {

1429impl Error {
1430  pub fn log_fatal(&self) {
1431    // One primitive does both halves: the `Fatal:<target>:<category>` line
1432    // AND the sticky `LogStatus::Fatal` latch. Without the latch,
1433    // `Fatal:Timeout:MemoryBudget` etc. printed but the runtime status_code
1434    // stayed at 0 — canvas would classify the worker as OK with an empty
1435    // HTML output. R35.A.
1436    //
1437    // Idempotent per fatal: `Fatal!`/`fatal!` log at the raise and the
1438    // converter/post sinks call this again on the propagated `Err`; the
1439    // same `target:category:message` prints once. A fatal built by hand
1440    // (`Error { .. }` literals) is logged here for the first time.
1441    if self.is_logged_fatal() {
1442      return;
1443    }
1444    let key = self.fatal_key();
1445    emit_record(
1446      LogStatus::Fatal,
1447      &s!(
1448        "Fatal:{}:{} ",
1449        self.target.log_name(),
1450        self.category.log_name()
1451      ),
1452      &self.message,
1453    );
1454    LAST_FATAL_LOGGED.with(|c| *c.borrow_mut() = Some(key));
1455  }
1456  fn fatal_key(&self) -> String {
1457    s!(
1458      "{}:{}:{}",
1459      self.target.log_name(),
1460      self.category.log_name(),
1461      self.message
1462    )
1463  }
1464  /// Whether this is the Fatal `log_fatal` printed last: an `Err` carrying an already-reported Fatal.
1465  pub fn is_logged_fatal(&self) -> bool {
1466    let key = self.fatal_key();
1467    LAST_FATAL_LOGGED.with(|c| c.borrow().as_deref() == Some(key.as_str()))
1468  }
