1//! Perl-style local assignments (dynamic scope)
2//!
3//! This module provides stack-based dynamic scoping for global state,
4//! matching Perl's `local` mechanism. Each "localized" field uses a `Vec<T>`
5//! as a stack: push to shadow, pop to restore.
6//!
7//! ## RAII Guards
8//!
9//! Each localized variable has a corresponding `Guard` type that pushes on
10//! creation and pops on `Drop`. This prevents leaked state on early returns
11//! or panics:
12//!
13//! ```rust,no_run
14//! # use latexml_core::common::local_assignments::*;
15//! # use latexml_core::token::Token;
16//! # fn example(token: Token) {
17//! let _guard = local_current_token_guard(token);
18//! // ... do work ...
19//! // guard auto-pops when _guard goes out of scope
20//! # }
21//! ```
22//!
23//! The explicit `set_*/expire_*` pairs still exist for cases where
24//! RAII doesn't fit (e.g., cross-function push/pop with different lifetimes).
25

34/// These are fields realized via Perl's "local" mechanism in LaTeXML,
35/// but (for now) require explicit "expire" calls in Rust.
36/// Ideally their ergonomics gets improved, or we gradually phase them out.
37#[derive(Debug, Default)]
38pub struct Localized {
39  dual_branch:       Vec<&'static str>,
40  if_frames:         Vec<Option<Rc<RefCell<IfFrame>>>>,
41  current_token:     Vec<Token>,
42  align_group_count: Vec<i32>, // was $LaTeXML::ALIGN_STATE
43  reading_alignment: Vec<Digested>,
44  build_template:    Vec<Template>,
45  unlocked:          Vec<bool>,
46}
47
48macro_rules! locals {
49  () => {
50    (*LOCALIZED_VARS).borrow()
51  };
52}
53macro_rules! locals_mut {
54  () => {
55    (*LOCALIZED_VARS).borrow_mut()
56  };
57}
58
59#[thread_local]
60static LOCALIZED_VARS: Lazy<RefCell<Localized>> = Lazy::new(|| RefCell::new(Localized::default()));

98/// Cached snapshot of `LXML_TRACE_ALIGN_STATE`: print every change of the
99/// alignment ledger (tex.web `align_state`) with the current token and a
100/// short backtrace, to bisect a ledger drift (a cell-ending `&`/`\\` is only
101/// recognized at `align_group_count()==0`). Debug aid, off by default.
102static TRACE_ALIGN_STATE: Lazy<bool> =
103  Lazy::new(|| std::env::var("LXML_TRACE_ALIGN_STATE").is_ok());
104
105fn trace_align_state(what: &str, before: i32, after: i32) {
106  if *TRACE_ALIGN_STATE && !locals!().reading_alignment.is_empty() {
107    let tok = get_current_token()
108      .map(|t| t.to_string())
109      .unwrap_or_default();
110    let bt = std::backtrace::Backtrace::force_capture().to_string();
111    let frames: Vec<&str> = bt
112      .lines()
113      .filter(|l| l.contains("latexml_") && !l.contains("local_assignments"))
114      .take(4)
115      .map(str::trim)
116      .collect();
117    eprintln!(
118      "ALIGN_STATE {what}: {before} -> {after} at {tok} | {}",
119      frames.join(" < ")
120    );
121  }
122}

124/// Bump the current `align_state` by `delta` (tex.web §358: `left_brace:
125/// incr(align_state)`, `right_brace: decr(align_state)`), returning the new
126/// value. Every scanned `{`/`}` lands here, so it is one thread-local
127/// borrow; the trace's before/after read is behind the off-by-default switch.
128fn shift_align_group_count(delta: i32) -> i32 {
129  let mut locals = locals_mut!();
130  match locals.align_group_count.last_mut() {
131    Some(v) => {
132      *v += delta;
133      *v
134    },
135    None => {
136      locals.align_group_count.push(delta);
137      delta
138    },
139  }
140}
141pub fn increment_align_group_count() {
142  let after = shift_align_group_count(1);
143  if *TRACE_ALIGN_STATE {
144    trace_align_state("+1", after - 1, after);
145  }
146}
147pub fn decrement_align_group_count() {
148  let after = shift_align_group_count(-1);
149  if *TRACE_ALIGN_STATE {
150    trace_align_state("-1", after + 1, after);
151  }
152}

154pub fn state_is_unlocked() -> bool { locals!().unlocked.last().copied().unwrap_or(false) }
155pub fn local_state_unlocked(v: bool) { locals_mut!().unlocked.push(v); }

158pub fn align_group_count() -> i32 {
159  locals!()
160    .align_group_count
161    .last()
162    .copied()
163    .unwrap_or_default()
164}
165pub fn set_align_group_count(v: i32) {
166  trace_align_state("set", align_group_count(), v);
167  match locals_mut!().align_group_count.last_mut() {
168    Some(gc) => {
169      *gc = v;
170    },
171    _ => {
172      locals_mut!().align_group_count.push(v);
173    },
174  }
175}
176pub fn local_align_group_count(v: i32) { locals_mut!().align_group_count.push(v); }
177/// The alignment brace count as it stands (tex.web's `align_state`, with its stack of saved values), to be put back by
178/// [`restore_align_group_count`] after a digestion whose tokens are not the document's input.
179pub fn mark_align_group_count() -> (usize, Option<i32>) {
180  let locals = locals!();
181  (
182    locals.align_group_count.len(),
183    locals.align_group_count.last().copied(),
184  )
185}
186/// Puts back the alignment brace count [`mark_align_group_count`] read: the stack at its depth then, the current value
187/// as it was. A `{` the abandoned digestion scanned and never closed is no longer counted against the document's `&`s.
188pub fn restore_align_group_count(mark: (usize, Option<i32>)) {
189  let (depth, value) = mark;
190  trace_align_state("restore", align_group_count(), value.unwrap_or_default());
191  let mut locals = locals_mut!();
192  let stack = &mut locals.align_group_count;
193  stack.truncate(depth);
194  if let Some(value) = value {
195    if stack.len() < depth {
196      stack.resize(depth, value);
197    }
198    if let Some(top) = stack.last_mut() {
199      *top = value;
200    }
201  }
202}
203pub fn expire_align_group_count() -> Option<i32> { locals_mut!().align_group_count.pop() }

205/// Disables tab marks for the lifetime of the guard, restoring the previous
206/// count on drop — what ltcmd does around every argument it reads: latex.ltx:2103-2110
207/// (`\__cmd_start:nNNnnn`) opens `\group_align_safe_begin:` (closed at :2128), which raises `align_state`
208/// by the backquote-constant rule (tex.web §442), and physics.sty does it by hand around its fenced reads
209/// (`{\ifnum\z@=`}\fi`, physics.sty:35/74/141). TeX's own `macro_call` (§389-399) does NOT disable tab
210/// marks — the `{disable tab marks, etc.}` of `align_state:=1000000` is `read_toks`, behind `\read`
211/// (tex.web §482) — so a plain macro's delimited argument holding an `&` inside an alignment errs in TeX
212/// too ("Argument of \… has an extra }"); arming this in `Parameters::read_arguments` would be unfaithful.
213/// For an ltcmd/physics reader, without it a `&` in a **delimiter-fenced** argument reaches the alignment as a
214/// cell break: for
215/// `\mqty( b_0 &0 \\ 0 &b_1 )` (physics.sty, witness 2605.05903) the row splits
216/// mid-argument and the alignment then cannot close its own group —
217/// `Error:unexpected:\lx@begin@alignment Attempt to close a group that switched
218/// to mode restricted_horizontal` — truncating the rest of the document. The
219/// brace form `\mqty{…}` was always safe because cell scanning skips balanced
220/// groups; `(…)` is not a group. Perl raises the identical error (11 of them on
221/// the 14-line repro), so this is beyond-Perl and `pdflatex`, which renders the
222/// repro silently, is the ground truth.
223///
224/// Only armed inside an alignment: outside one there are no tab marks to
225/// suppress, which keeps this off the hot path for every ordinary macro call.
226/// 28 papers of the 2026-07-29 bibliography-absence residual truncate here.
227pub struct SuppressedTabMarks(Option<i32>);
228
229impl SuppressedTabMarks {
230  pub fn for_argument_scan() -> Self {
231    if has_reading_alignment() {
232      let saved = align_group_count();
233      set_align_group_count(1_000_000);
234      Self(Some(saved))
235    } else {
236      Self(None)
237    }
238  }
239}
240
241impl Drop for SuppressedTabMarks {
242  fn drop(&mut self) {
243    if let Some(saved) = self.0 {
244      set_align_group_count(saved);
245    }
246  }
247}

271/// Runs `caller` on the alignment template being built, if one is. A column type's `\NC@rewrite@<c>` expanded outside
272/// a preamble (an `\edef` that meets it) builds nothing: in TeX it only yields tokens there (`\NC@find p{…}`), and
273/// unwrapping the missing template panicked (2605.18869: tabularx's `X`, current inside a tabularx since 61s, met by an
274/// `\edef` while an algorithm's unclosed group was open).
275pub fn with_building_template(caller: impl FnOnce(&mut Template)) {
276  if let Some(template) = locals_mut!().build_template.last_mut() {
277    caller(template);
278  }

285/// Generic RAII guard that calls an expire function on drop.
286/// Zero-cost when the expire function is a simple fn pointer.
287pub struct LocalGuard {
288  expire: fn(),
289}
290impl Drop for LocalGuard {
291  fn drop(&mut self) { (self.expire)(); }
292}
293
294/// Push a localized current token; returns guard that pops on drop.
295pub fn local_current_token_guard(token: Token) -> LocalGuard {
296  local_current_token(token);
297  LocalGuard { expire: expire_current_token }
298}
299
300/// Push a localized if-frame; returns guard that pops on drop.
301pub fn local_ifframe_guard(frame: Option<Rc<RefCell<IfFrame>>>) -> LocalGuard {
302  set_ifframe(frame);
303  LocalGuard { expire: expire_ifframe }
304}
305
306/// Push a localized dual branch; returns guard that pops on drop.
307pub fn local_dual_branch_guard(mode: &'static str) -> LocalGuard {
308  set_dual_branch(mode);
309  LocalGuard { expire: expire_dual_branch }
310}

312/// Push a localized align group count; returns guard that pops on drop.

346  #[test]
347  fn test_dual_branch_push_pop() {
348    initialize_localized();
349    assert_eq!(get_dual_branch(), None);
350    set_dual_branch("true");
351    assert_eq!(get_dual_branch(), Some("true"));
352    set_dual_branch("false");
353    assert_eq!(get_dual_branch(), Some("false"));
354    expire_dual_branch();
355    assert_eq!(get_dual_branch(), Some("true"));
356    expire_dual_branch();
357    assert_eq!(get_dual_branch(), None);

360  #[test]
361  fn test_dual_branch_guard_auto_restore() {
362    initialize_localized();
363    assert_eq!(get_dual_branch(), None);
364    {
365      let _guard = local_dual_branch_guard("guarded");
366      assert_eq!(get_dual_branch(), Some("guarded"));
367    }
368    // Guard dropped — value restored
369    assert_eq!(get_dual_branch(), None);
370  }
