49impl Number {
50  /// Perl Number.pm:31-34 `int($number)`: Perl's numeric coercion, truncated toward zero (`"1e3"` is 1000, `"1.5"` 1);
51  /// a string that is not wholly a number Warns as Perl's `isn't numeric` and reads its numeric prefix (`"12abc"` 12).
52  pub fn from_str_value(spec: &str) -> Number {
53    let trimmed = spec.trim();
54    let prefix = numeric_prefix(trimmed);
55    if prefix.is_empty() || prefix.len() != trimmed.len() {
56      Warn!(
57        "perl",
58        "warn",
59        s!("Argument \"{spec}\" isn't numeric in int")
60      );
61    }
62    Number(prefix.parse::<f64>().map_or(0, |f| f.trunc() as i64))
63  }
64}

66/// The longest prefix of `s` Perl's string-to-number conversion reads: `[+-]?\d*(\.\d*)?([eE][+-]?\d+)?`.
67fn numeric_prefix(s: &str) -> &str {
68  let b = s.as_bytes();
69  let digits = |mut i: usize| {
70    while i < b.len() && b[i].is_ascii_digit() {
71      i += 1;
72    }
73    i
74  };
75  let mut i = usize::from(matches!(b.first(), Some(b'+' | b'-')));
76  let start = i;
77  i = digits(i);
78  let mut mantissa = i > start;
79  if b.get(i) == Some(&b'.') {
80    let j = digits(i + 1);
81    mantissa |= j > i + 1;
82    i = j;
83  }
84  if !mantissa {
85    return "";
86  }
87  if matches!(b.get(i), Some(b'e' | b'E')) {
88    let k = i + 1 + usize::from(matches!(b.get(i + 1), Some(b'+' | b'-')));
89    let e = digits(k);
90    if e > k {
91      i = e;
92    }
93  }
94  &s[..i]
95}

140      ("-7", -7),
141      (" 42 ", 42),
142      ("12abc", 12),
143    ] {
144      assert_eq!(Number::from(String::from(input)).value_of(), n, "{input}");
145    }
146  }
