138/// Perl `CheckOptions($operation, $allowed, %options)`: one `Error:misdefined` naming every key
139/// `allowed` lacks. Returns whether all keys were allowed (`Err` only when the error cap escalates); the definition proceeds either way, as
140/// in Perl, with the rejected keys ignored.
141pub fn check_options<'a>(
142  operation: &str,
143  allowed: &[&str],
144  keys: impl IntoIterator<Item = &'a str>,
145) -> crate::common::error::Result<bool> {
146  let bad: Vec<&str> = keys.into_iter().filter(|k| !allowed.contains(k)).collect();
147  if bad.is_empty() {
148    return Ok(true);
149  }
150  crate::Error!(
151    "misdefined",
152    operation,
153    format!("{operation} does not accept options:{}", bad.join(", "))
154  );
155  Ok(false)
156}

158/// A key Perl accepts whose effect the core cannot carry yet: said, never silently dropped.
159pub fn unsupported_option(operation: &str, key: &str) {
160  crate::Warn!(
161    "unsupported",
162    operation,
163    format!("option {key} is not supported yet; ignored")
164  );
