1//! latex.ltx:9521-9522 `\gdef\do#1{\global\let#1\@notprerr}\@preamblecmds`: after `\begin{document}` each listed
2//! command IS `\@notprerr`, which raw code tests with `\ifx` (ltnews23.tex:42-43). LaTeXML also reads a package, class
3//! or sub-file preamble in the body (Perl's recovery for a body `\usepackage`, a binding's load, standalone and
4//! subfiles), which pdflatex never does: there the walk's commands are reopened, and disabled again after it. The walk
5//! keeps each meaning under `\lx@onlypreamble@<name>`; the State holds the list (OXIDIZED_DESIGN_DIVERGENCES #226).
6use crate::{
7  common::store::Stored,
8  gullet::unread_after_current_file,
9  state::{
10    Scope, assign_meaning, assign_value, lookup_bool, lookup_meaning, lookup_tokens, x_equals,
11  },
12  token::Token,
13  tokens::Tokens,
14};
15
16/// The commands the walk let to `\@notprerr`, in the walk's order.
17const DISABLED: &str = "lx@preamble@disabled";
18/// Set while the disabled commands are reopened (a body load or a sub-file preamble), so a nested one leaves them.
19const REOPENED: &str = "lx@preamble@reopened";
20/// Set while a sub-file's preamble holds the reopening ([`reopen_for_subfile_preamble`]).
21const SUBFILE: &str = "lx@preamble@subfile";
22
23fn saved_cs(cs: &Token) -> Token {
24  let name = cs.to_string();
25  T_CS!(s!("\\lx@onlypreamble@{}", name.trim_start_matches('\\')))
26}
27
28/// `\global\let<to><from>`, the engine's own write: no file's lock applies to the kernel's walk.
29fn global_let(to: &Token, from: &Token) {
30  assign_meaning(
31    to,
32    lookup_meaning(from).unwrap_or(Stored::None),
33    Some(Scope::Global),
34  );
35}
36
37fn let_notprerr(cs: &Token) {
38  global_let(&saved_cs(cs), cs);
39  global_let(cs, &T_CS!("\\@notprerr"));
40}
41
42/// The walk's `\global\let<cs>\@notprerr`, its meaning kept for a later [`reopen_preamble_commands`].
43pub fn disable_preamble_command(cs: &Token) {
44  // (once: a second `\begin{document}` walks again, sect02.rs)
45  if x_equals(cs, &T_CS!("\\@notprerr")) {
46    return;
47  }
48  let_notprerr(cs);
49  let mut list = lookup_tokens(DISABLED)
50    .map(Tokens::unlist)
51    .unwrap_or_default();
52  list.push(*cs);
53  assign_value(DISABLED, Tokens::new(list), Some(Scope::Global));
54}
55
56/// Give the disabled commands back their meanings (a body load, a sub-file's preamble); false when there is nothing
57/// to reopen (still in the preamble, or reopened by an enclosing load).
58pub fn reopen_preamble_commands() -> bool {
59  if lookup_bool(REOPENED) {
60    return false;
61  }
62  let Some(list) = lookup_tokens(DISABLED) else {
63    return false;
64  };
65  for cs in list.unlist_ref() {
66    global_let(cs, &saved_cs(cs));
67  }
68  assign_value(REOPENED, true, Some(Scope::Global));
69  true
70}
71
72/// The walk again over the reopened commands: what the load (re)defined is kept for the next reopening.
73pub fn close_preamble_commands() {
74  if !lookup_bool(REOPENED) {
75    return;
76  }
77  if let Some(list) = lookup_tokens(DISABLED) {
78    for cs in list.unlist_ref() {
79      let_notprerr(cs);
80    }
81  }
82  assign_value(REOPENED, false, Some(Scope::Global));
83}
84
85/// [`reopen_preamble_commands`] for a sub-file's preamble (standalone, subfiles): closed at its `\begin{document}` or,
86/// without one, where the sub-file ends (`\lx@subfile@preamble@end`, [`close_subfile_preamble`]).
87pub fn reopen_for_subfile_preamble() {
88  if reopen_preamble_commands() {
89    assign_value(SUBFILE, true, Some(Scope::Global));
90    unread_after_current_file(Tokens::new(vec![T_CS!("\\lx@subfile@preamble@end")]));
91  }
92}
93
94/// The close of the reopening a sub-file's preamble holds; false when it holds none (closed at its `\begin{document}`).
95pub fn close_subfile_preamble() -> bool {
96  if !lookup_bool(SUBFILE) {
97    return false;
98  }
99  assign_value(SUBFILE, false, Some(Scope::Global));
100  close_preamble_commands();
101  true
102}
103
104/// `\lx@subfile@preamble@end`: a sub-file that ends with no `\begin{document}` ends its preamble there (pdflatex:
105/// "File ended while scanning use of \sa@gobble").
106pub fn end_subfile_preamble() {
107  if close_subfile_preamble() {
108    assign_value("inPreamble", false, None);
109    assign_value("lx@subpreamble@begindocument", false, None);
110  }
111}

113/// [`reopen_preamble_commands`] for the extent of a load, closed when dropped (also on an Error's early return).
114pub struct ReopenedForLoad(bool);
115
116impl ReopenedForLoad {
117  pub fn new() -> Self { ReopenedForLoad(reopen_preamble_commands()) }
118}
119
120impl Default for ReopenedForLoad {
121  fn default() -> Self { Self::new() }
122}
123
124impl Drop for ReopenedForLoad {
125  fn drop(&mut self) {
126    if self.0 {
127      close_preamble_commands();
128    }
129  }
