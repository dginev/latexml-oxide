35/// configuration for new_counter
36#[derive(Default)]
37pub struct NewCounterOptions<'ct> {
38  /// specifies a prefix to be used in formatting ID's for document structure elements
39  /// counted by this counter.  Ie. subsection 3 in section 2 might get: id="S2.SS3"
40  pub idprefix: &'ct str,
41  /// specifies that the ID is composed from $idwithin's ID,, even though
42  /// the counter isn't numbered within it.  (mainly to avoid duplicated ids)
43  pub idwithin: &'ct str,
44  /// a list of counters that correspond to scopes which are "inside" this one.
45  /// Whenever any definitions scoped to this counter are deactivated,
46  /// the inner counter's scopes are also deactivated.
47  // NOTE: I'm not sure this is even a sensible implementation,
48  // or why inner should be different than the counters reset by incrementing this counter.
49  pub nested: Vec<&'ct str>,
50}

52/// Perl `addtoCounterReset` (latex_constructs.pool.ltxml:3005-3012): put `\<ctr>` and `\UN<ctr>`
53/// at the front of `\cl@<within>`, globally, as latex.ltx:10156's `\@cons` `\xdef`s — a reset
54/// registered inside a group outlives it — and re-derive the list's macro.
55/// Repro macro-state/counter_reset_lists_are_global.
56pub fn add_to_counter_reset(ctr: &str, within: &str) -> Result<()> {
57  if copy_provides_keeps_counter(ctr) {
58    return Ok(());
59  }
60  let reg = s!("\\cl@{within}");
61  let mut toks = vec![T_CS!(ctr), T_CS!(s!("UN{ctr}"))];
62  if let Some(prev) = lookup_tokens(&reg) {
63    toks.extend(prev.unlist());
64  }
65  assign_value(&reg, Stored::Tokens(Tokens::new(toks)), Some(Scope::Global));
66  sync_reset_list_macro(within)?;
67  define_hyper_counter_name(ctr, within)
68}

70/// The `\theH<ctr>` hyperref builds a counter's anchor from (`\hyper@makecurrent`), as the
71/// kernel makes it since 2024-11-01: latex.ltx:10145 `\@definecounter` xdefs it as
72/// `\the\value{<ctr>}`, and :10157 `\@addtoreset` as `\theH<within>.\the\value{<ctr>}`, so
73/// hyperref.sty:6607 skips its own `\providecommand` list. Perl's counters have none. A package
74/// building on them needs them: pgfplots' `\theHpgfplotslink{\theHsection.…}`
75/// (pgfplots.code.tex:6138, through `\hyper@makecurrent{pgfplotslink}` at :6123). LaTeXML's
76/// pseudo-counter `document` is no parent: no `\theHdocument` prefix.
77/// Guard: `perfect_kernel_batch64::kernel_counters_have_hyperref_names`.
78fn define_hyper_counter_name(ctr: &str, within: &str) -> Result<()> {
79  let mut body: Vec<Token> = Vec::new();
80  if !within.is_empty() && within != "document" {
81    body.push(T_CS!(s!("\\theH{within}")));
82    body.push(T_OTHER!("."));
83  }
84  body.push(T_CS!("\\the"));
85  body.push(T_CS!("\\value"));
86  body.push(T_BEGIN!());
87  body.extend(ExplodeText!(ctr));
88  body.push(T_END!());
89  def_macro(
90    T_CS!(s!("\\theH{ctr}")),
91    None,
92    ExpansionBody::from(Tokens::new(body)),
93    Some(ExpandableOptions {
94      scope: Some(Scope::Global),
95      ..ExpandableOptions::default()
96    }),
97  )
98}

100/// Perl `remfromCounterReset` (latex_constructs.pool.ltxml:3014-3022): drop `\<ctr>` and
101/// `\UN<ctr>` from `\cl@<within>`, globally (latex.ltx:10170 `\xdef`), when there is a list.
102pub fn remove_from_counter_reset(ctr: &str, within: &str) -> Result<()> {
103  let reg = s!("\\cl@{within}");
104  if let Some(prev) = lookup_tokens(&reg) {
105    let (t, unt) = (T_CS!(ctr), T_CS!(s!("UN{ctr}")));
106    let kept: Vec<Token> = prev
107      .unlist()
108      .into_iter()
109      .filter(|x| *x != t && *x != unt)
110      .collect();
111    assign_value(&reg, Stored::Tokens(Tokens::new(kept)), Some(Scope::Global));
112    sync_reset_list_macro(within)?;
113  }
114  Ok(())
115}

117/// Mirror the reset list into the `\cl@<ctr>` MACRO latex.ltx exposes.
118///
119/// latex.ltx:10140-10142 `\@definecounter` lets `\cl@<ctr>` to `\@empty` and
120/// :10156 `\@addtoreset` `\@cons`es `\@elt{<child>}` onto it, so the reset list
121/// is a macro raw code expands and rewrites: contract.sty:336
122/// `\edef\cl@Clause{\cl@Clause\cl@contractClause}` (contract-example-de/-en,
123/// 44 errors each), afthesis.cls:44-49's `\@removefromreset` re-`\edef` with a
124/// filtering `\@elt` (usethesis: "`\cl@chapter` expands into itself"). LaTeXML
125/// keeps the list as the State VALUE `\cl@<ctr>` (Perl Package.pm:674, entries
126/// `\<child>`/`\UN<child>`), which stays authoritative for stepping; this
127/// re-derives the macro (`\@elt{child}` per non-`UN` entry) after every
128/// mutation so both faces agree. Perl shares the gap (sectioning topic).
129/// Guard: `perfect_kernel_batch54::reset_list_is_an_expandable_cl_macro`.
130pub fn sync_reset_list_macro(ctr: &str) -> Result<()> {
131  let clctr = s!("\\cl@{ctr}");
132  let mut body: Vec<Token> = Vec::new();
133  if let Some(list) = lookup_tokens(&clctr) {
134    for t in list.unlist() {
135      let name = t.with_str(|s| s.to_string());
136      if name.starts_with("UN") {
137        continue;
138      }
139      body.push(T_CS!("\\@elt"));
140      body.push(T_BEGIN!());
141      body.extend(ExplodeText!(name));
142      body.push(T_END!());
143    }
144  }
145  def_macro(
146    T_CS!(&clctr),
147    None,
148    ExpansionBody::from(Tokens::new(body)),
149    Some(ExpandableOptions {
150      scope: Some(Scope::Global),
151      ..ExpandableOptions::default()
152    }),
153  )
154}

156/// Defines a new counter named $ctr.
157/// If `within` is defined, `ctr` will be reset whenever `within` is incremented.
158pub fn new_counter(ctr: &str, within: &str, options_opt: Option<NewCounterOptions>) -> Result<()> {
159  // (a stand-in's copy keeps a counter the binding owns as the binding numbers it: `state::copy_provides_keeps_counter`)
160  if copy_provides_keeps_counter(ctr) {
161    return Ok(());
162  }
163  // (any other counter a stand-in's copy declares is declared for real — its `\c@`, `\cl@`, `\the`, `\p@` names outlive
164  // the copy's overlay, as the real class's do: `state::copy_provides_keeps_counter`)
165  let _declared = (crate::definition::origin::copy_provides() != 0)
166    .then(|| crate::definition::origin::CopyProvidesGuard::new(0));
167  let unctr = s!("UN{ctr}"); // UNctr is counter for generating ID's for UN-numbered items.
168  if !within.is_empty()
169    && within != "document"
170    && lookup_definition(&T_CS!(s!("\\c@{within}")))?.is_none()
171  {
172    new_counter(within, "", None)?;
173  }
174  let cctr = s!("\\c@{ctr}");
175  let clctr = s!("\\cl@{ctr}");
176  let cunctr = s!("\\c@{unctr}");
177  let clunctr = s!("\\cl@{unctr}");
178  // Perl Package.pm L660-672: Check if counter already defined. Skip register if already a
179  // Register. Warn if previously defined as something other than \relax.
180  let cs_cctr = T_CS!(&cctr);
181  let prev_defn = lookup_definition(&cs_cctr)?;
182  if let Some(ref defn) = prev_defn {
183    if defn.is_register() {
184      // Counter already exists as a register — fine, just continue (may change within/nesting)
185    } else {
186      // Warn unless the previous definition was \relax
187      let relax_meaning = lookup_meaning(&T_RELAX!());
188      let prev_meaning = lookup_meaning(&cs_cctr);
189      if prev_meaning != relax_meaning {
190        Warn!(
191          "unexpected",
192          &cctr,
193          s!("Counter {} was already defined; redefining", cctr)
194        );
195      }
196      def_register(
197        cs_cctr,
198        None,
199        Number::new(0),
200        Some(RegisterOptions {
201          allocate: Some(String::from("\\count")),
202          ..RegisterOptions::default()
203        }),
204      )?;
205    }
206  } else {
207    def_register(
208      cs_cctr,
209      None,
210      Number::new(0),
211      Some(RegisterOptions {
212        allocate: Some(String::from("\\count")),
213        ..RegisterOptions::default()
214      }),
215    )?;
216  }
217  // An alias counter (`register_counter_alias`) given a register of its own (`\let\c@case\relax\newcounter{case}`
218  // after llncs's `\@spothm{case}[theorem]`: 2303.08726) is an alias no more: `\refstepcounter{case}` steps the new
219  // register, as in LaTeX, where the alias was only the `\let` of `\c@case`. A counter whose `\c@` is still the alias's
220  // register keeps it (LaTeX refuses that `\newcounter`, "Command \c@case already defined").
221  if !prev_defn.as_ref().is_some_and(|defn| defn.is_register()) && is_counter_alias(ctr) {
222    assign_mapping::<Stored>("counter_for_type", ctr, None);
223    assign_mapping::<Stored>("counter_alias", ctr, None);
224  }
225  after_assignment();
226  if !has_value(&clctr) {
227    assign_value(&clctr, Tokens!(), Some(Scope::Global));
228  }
229  sync_reset_list_macro(ctr)?;
230  def_register(T_CS!(&cunctr), None, Number::new(0), None)?;
231  if !has_value(&clunctr) {
232    assign_value(&clunctr, Tokens!(), Some(Scope::Global));
233  }
234
235  if !within.is_empty() {
236    let clwithin = s!("\\cl@{within}");
237    let clunwithin = s!("\\cl@UN{within}");
238    let x = if let Some(cl) = lookup_tokens(&clwithin) {
239      cl.unlist()
240    } else {
241      Vec::new()
242    };
243    let mut clwithin_tokens = vec![T_CS!(ctr), T_CS!(&unctr)];
244    clwithin_tokens.extend(x);
245    assign_value(
246      &clwithin,
247      Stored::Tokens(Tokens::new(clwithin_tokens)),
248      Some(Scope::Global),
249    );
250
251    let mut unx = if let Some(clun) = lookup_tokens(&clunwithin) {
252      clun.unlist()
253    } else {
254      Vec::new()
255    };
256    let mut clunwithin_tokens = vec![T_CS!(unctr)];
257    clunwithin_tokens.append(&mut unx);
258
259    assign_value(
260      &clunwithin,
261      Stored::Tokens(Tokens::new(clunwithin_tokens)),
262      Some(Scope::Global),
263    );
264    sync_reset_list_macro(within)?;
265  }
266  define_hyper_counter_name(ctr, within)?;
267
268  // default is equivalent to \arabic{ctr}, but w/o using the LaTeX macro:
269  // Perl Package.pm:686 `\the<ctr>` = `\lx@counter@arabic{<ctr>}`, a Token
270  // body the format dump can carry (a closure is opaque to it).
271  let ctr_string = ctr.to_string();
272  def_macro(
273    T_CS!(s!("\\the{}", ctr)),
274    None,
275    Some(ExpansionBody::Tokens(mouth::tokenize_internal(
276      TeXString::assembled(s!("\\lx@counter@arabic{{{ctr_string}}}")),
277    ))),
278    Some(ExpandableOptions {
279      scope: Some(Scope::Global),
280      ..ExpandableOptions::default()
281    }),
282  )?;
283  let p_ctr_cs = T_CS!(&s!("\\p@{}", ctr));
284  if lookup_definition(&p_ctr_cs)?.is_none() {
285    def_macro(
286      p_ctr_cs,
287      None,
288      Tokens::default(),
289      Some(ExpandableOptions {
290        scope: Some(Scope::Global),
291        ..ExpandableOptions::default()
292      }),
293    )?;
294  }
295
296  // (a standard-class derivative re-declaring a counter its standard class's binding declared — article.cls's
297  // `\newcounter{figure}` in a full copy — keeps the binding's ids, LaTeXML's markup: `S1.F1`, not `F1`; the register,
298  // its reset list and `\the<ctr>` are the copy's, as TeX makes them. OXIDIZED_DESIGN_DIVERGENCES #731 rule 13;
299  // 2402.12725's article_saj.cls; guard `perfect_kernel_batch77w::standard_class_derivative_patches_stand`)
300  let ids = T_CS!(s!("\\the{ctr}@ID"));
301  if crate::binding::content::standard_class_binding_assigned(meaning_key(&ids)) {
302    return Ok(());
303  }
304  define_counter_ids(ctr, within, options_opt.as_ref())
305}

307/// The id half of [`new_counter`] (Perl Package.pm:689-705, `NewCounter`'s `idprefix`/`idwithin`/`nested`): records
308/// `@ID@prefix@<ctr>` and `nested_counters_<ctr>`, and defines `\the<ctr>@ID` and `\@<ctr>@ID`. A binding calls it
309/// alone for a counter a raw class declared, to give it the ids its standard-class binding gives (the KOMA classes,
310/// `koma_script.rs`: `S1.SS1`, not `section1.subsection1`); the register and its reset list stay the class's.
311pub fn define_counter_ids(
312  ctr: &str,
313  within: &str,
314  options_opt: Option<&NewCounterOptions>,
315) -> Result<()> {
316  if let Some(options) = options_opt
317    && !options.nested.is_empty()
318  {
319    assign_value(
320      &s!("nested_counters_{}", ctr),
321      options.nested.clone(),
322      Some(Scope::Global),
323    )
324  }
325
326  let mut prefix = match options_opt {
327    None => String::new(),
328    Some(opt) => opt.idprefix.to_string(),
329  };
330  if !prefix.is_empty() {
331    assign_value(
332      &s!("@ID@prefix@{}", ctr),
333      prefix.clone(),
334      Some(Scope::Global),
335    );
336  } else {
337    prefix = lookup_string(&s!("@ID@prefix@{}", ctr));
338    if prefix.is_empty() {
339      prefix = ctr.to_string();
340    }
341  }
342  prefix = clean_id(&prefix);
343
344  if !prefix.is_empty() {
345    let idwithin = match options_opt {
346      Some(opts) => {
347        if opts.idwithin.is_empty() {
348          within
349        } else {
350          opts.idwithin
351        }
352      },
353      None => within,
354    }
355    .to_string();
356
357    let ctr_string = ctr.to_string();
358    let thectrid = s!("\\the{}@ID", ctr);
359    // Token bodies, as Perl's `DefMacroI` string bodies (Package.pm:695-703):
360    // a closure here is opaque to the format dump (recorded as an unresolvable
361    // self-alias), so every kernel counter's `\the<ctr>@ID` was lost on the
362    // dump path and a class with no binding (ptptex) generated ids from an
363    // undefined formatter — `subequations` children `.1`/`X.1`, colliding.
364    if !idwithin.is_empty() {
365      def_macro(
366        T_CS!(thectrid),
367        None,
368        Some(ExpansionBody::Tokens({
369          // Perl Package.pm L696 probes `\lx@empty`, NOT `\@empty` — and
370          // deliberately so: `\@empty` is LaTeX-pool-only (latex_base.rs
371          // aliases it to `\lx@empty`), while `\lx@empty` is engine-level
372          // (base_schema.rs) and exists in PLAIN TeX documents too. With
373          // `\@empty` here, a plain-TeX document's `\the<ctr>@ID` probe
374          // compared `\thedocument@ID` (an alias of `\lx@empty`) against an
375          // UNDEFINED `\@empty` → unequal → the \else branch re-expands the
376          // parent's @ID formatter, spinning the gullet into a 4-token
377          // expansion cycle (caught by the cycle guard as a phantom
378          // `Fatal:Timeout:Recursion`, swallowed inside math-parser
379          // semantics — `create_xmrefs`/`get_xmarg_id`). Witness
380          // math0402448 (plain TeX + 3464 formulae): "Conversion failed:
381          // 1 fatal error" with no Fatal: line in the log.
382          // The names are spliced in as characters (`id_name_chars`, as `define_hyper_counter_name` does), never
383          // re-read from a string: a counter named `claim_nb` or `c_n` (2606.16543, 1912.04210) made the `_` a
384          // subscript, digested for the id ("Script _ can only appear in math mode"; Perl Package.pm:695-702 alike,
385          // KNOWN_PERL_ERRORS #621). Guard `perfect_kernel_batch65m::counter_name_with_underscore_is_its_id`.
386          let mut body = vec![T_CS!("\\expandafter"), T_CS!("\\ifx"), T_CS!("\\csname")];
387          body.extend(id_name_chars(&s!("the{idwithin}@ID")));
388          body.extend([
389            T_CS!("\\endcsname"),
390            T_CS!("\\lx@empty"),
391            T_CS!("\\else"),
392            T_CS!("\\csname"),
393          ]);
394          body.extend(id_name_chars(&s!("the{idwithin}@ID")));
395          body.extend([T_CS!("\\endcsname"), T_OTHER!("."), T_CS!("\\fi")]);
396          body.extend(id_name_chars(&prefix));
397          body.extend(counter_id_number(&ctr_string));
398          Tokens::new(body)
399        })),
400        Some(ExpandableOptions {
401          scope: Some(Scope::Global),
402          ..ExpandableOptions::default()
403        }),
404      )?;
405    } else {
406      def_macro(
407        T_CS!(thectrid),
408        None,
409        Some(ExpansionBody::Tokens({
410          let mut body = id_name_chars(&prefix);
411          body.extend(counter_id_number(&ctr_string));
412          Tokens::new(body)
413        })),
414        Some(ExpandableOptions {
415          scope: Some(Scope::Global),
416          ..ExpandableOptions::default()
417        }),
418      )?;
419    }
420    def_macro(
421      T_CS!(s!("\\@{}@ID", ctr)),
422      None,
423      Some(ExpansionBody::Tokens(Tokens!(T_OTHER!("0")))),
424      Some(ExpandableOptions {
425        scope: Some(Scope::Global),
426        ..ExpandableOptions::default()
427      }),
428    )?;
429  }
430
431  Ok(())
432}
433/// A name in a `\the<ctr>@ID` body as characters: letters and `@` letters, the rest other, as `tokenize_internal`
434/// reads a name made of them (the format dump keeps those bodies), but a `_`, `^` or `&` in the name is a character too.
435fn id_name_chars(name: &str) -> Vec<Token> {
436  name
437    .chars()
438    .map(|c| {
439      let mut buf = [0u8; 4];
440      let c_str: &str = c.encode_utf8(&mut buf);
441      if c == ' ' {
442        T_SPACE!()
443      } else if c == '@' || c.is_alphabetic() {
444        T_LETTER!(c_str)
445      } else {
446        T_OTHER!(c_str)
447      }
448    })
449    .collect()
450}
451/// `\csname @<ctr>@ID\endcsname`, the counter's number in an id, the name spliced in as characters.
452fn counter_id_number(ctr: &str) -> Vec<Token> {
453  let mut toks = vec![T_CS!("\\csname")];
454  toks.extend(id_name_chars(&s!("@{ctr}@ID")));
455  toks.push(T_CS!("\\endcsname"));
456  toks
457}
458/// Fetches the value associated with the counter C<$ctr>.

466    },
467    Some(value) => Ok(Number::new(value.value_of())),
468  }

507/// Analog of `\stepcounter`, steps the counter and returns the expansion of
508/// `\the$ctr`  Usually you should use `ref_step_counter(ctr)` instead.
509pub fn step_counter(ctr: &str, noreset: bool) -> Result<()> {
510  let value = counter_value(ctr)?;
511  let newvalue = value.add(Number::new(1));
512  let c_ctr = s!("\\c@{ctr}");
513  assign_register(&c_ctr, newvalue.into(), Some(Scope::Global), Vec::new())?;
514  after_assignment();
515  let token_value = Tokens::new(Explode!(newvalue.value_of()));
516  def_macro(
517    T_CS!(s!("\\@{ctr}@ID")),
518    None,
519    token_value,
520    Some(ExpandableOptions {
521      scope: Some(Scope::Global),
522      ..ExpandableOptions::default()
523    }),
524  )?;
525
526  // and reset any within counters!
527  if !noreset && let Some(nested) = lookup_tokens(&s!("\\cl@{ctr}")) {
528    for c in nested.unlist() {
529      reset_counter(&c)?;
530    }
531  }
532  Ok(())
533}

535/// Analog of `\refstepcounter`, steps the counter and returns a hash
536/// containing the keys `refnum` and `id`.
537///
538/// This makes it
539/// suitable for use in a `properties` option to constructors.
540/// The `id` is generated in parallel with the reference number
541/// to assist debugging.
542// TODO: Maybe these should be specialized types in Rust, rather than hashmaps?
543pub fn ref_step_counter(ctype: &str, noreset: bool) -> Result<HashMap<Stored>> {
544  let ctype = strip_counter_type_sentinel(ctype);
545  let ctr = counter_for_type(ctype);

834/// with its current meaning (latex.ltx:17391-17399 `\refstepcounter\@captype`, :15737-15738), so a
835/// rebinding applies — floatrow measures each floatbox under `\FR@loc@`, whose `\refstepcounter`

841/// direct; otherwise `\refstepcounter{<counter>}` is digested — the counter, as LaTeX passes it
842/// (cleveref's `\cref@constructprefix{#1}`, fncylab's `\stepcounter{#1}` read it) — with `ctype`
843/// left for the kernel's label to make the tags of (a shared-counter theorem's own type); the props
844/// are the ones it recorded, else (a rebinding that never reaches the kernel) the label of the
845/// counter's value as it now stands. Perl steps the counter directly (`RefStepCounter`; KPE #412).
846/// Witness kaytannollista-latexia (floatrow), 2605.17685 (continued floats).
847/// Guards `perfect_kernel_batch58::{caption_steps_through_refstepcounter,
848/// floatrow_floatbox_steps_its_counter_once, floatbox_steps_its_equation_once,
849/// steps_keep_their_type_and_continuation_through_a_wrapper}`.

851  ctype: &str,
852  before: Vec<Token>,
853  continued: bool,
854) -> Result<HashMap<Stored>> {
855  let ctype = strip_counter_type_sentinel(ctype);
856  let ctr = counter_for_type(ctype);
857  if !before.is_empty() {
858    stomach::digest(Tokens::new(before))?;
859  }
860  let refstep = T_CS!("\\refstepcounter");
861  if x_equals(&refstep, &T_CS!("\\lx@kernel@refstepcounter")) {
862    if continued {
863      // The suppressed `\stepcounter`: the value stays, no counter within is reset, and the
864      // continuation is used up (caption's next step clears its flags, caption.sty:590-599).
865      assign_value("lx@float@continued", String::new(), Some(Scope::Global));
866      add_to_counter(&ctr, Number::new(-1))?;
867      return ref_step_counter(ctype, true);
868    }
869    return ref_step_counter(ctype, false);
870  }
871  PENDING_STEPS.with(|steps| steps.borrow_mut().insert(ctr.clone(), ctype.to_string()));
872  REFSTEP_PROPS.with(|slot| slot.borrow_mut().remove(&ctr));
873  let mut tokens = Vec::new();
874  if continued {
875    // caption.sty:557-568 `\caption@@refcounter`.
876    tokens.extend([
877      T_CS!("\\let"),
878      T_CS!("\\lx@caption@stepcounter@ORI"),
879      T_CS!("\\stepcounter"),
880      T_CS!("\\def"),
881      T_CS!("\\stepcounter"),
882      T_PARAM!(),
883      T_OTHER!("1"),
884      T_BEGIN!(),
885      T_CS!("\\lx@caption@stepcounter@unless"),
886      T_BEGIN!(),
887    ]);
888    tokens.extend(Explode!(&ctr));
889    tokens.extend([
890      T_END!(),
891      T_BEGIN!(),
892      T_PARAM!(),
893      T_OTHER!("1"),
894      T_END!(),
895      T_END!(),
896    ]);
897  }
898  tokens.extend([refstep, T_BEGIN!()]);
899  tokens.extend(Explode!(&ctr));
900  tokens.push(T_END!());
901  if continued {
902    tokens.extend([
903      T_CS!("\\let"),
904      T_CS!("\\stepcounter"),
905      T_CS!("\\lx@caption@stepcounter@ORI"),
906    ]);
907  }
908  let digested = stomach::digest(Tokens::new(tokens));
909  take_pending_step(&ctr);
910  digested?;
911  match REFSTEP_PROPS.with(|slot| slot.borrow_mut().remove(&ctr)) {
912    Some(props) => Ok(props),
913    None => label_stepped_counter(ctype, &ctr),
914  }
915}

917/// Does the `\the<start>@ID` formatter chain reach `\the<target>@ID`?
918/// `begin_itemize` defines each list's `\the<usecounter>@ID` through its
919/// own `\the<listcounter>@ID`, which is defined through the enclosing
920/// list's `\the<outerusecounter>@ID` — so the open lists form a chain of
921/// `\the…@ID` macros. Walking the bodies (never expanding) is bounded by
922/// the nesting depth.
923pub fn list_id_chain_reaches(start: &str, target: &str) -> Result<bool> {
924  let target_cs = T_CS!(s!("\\the{target}@ID"));
925  let mut frontier = vec![T_CS!(s!("\\the{start}@ID"))];
926  let mut seen: Vec<Token> = Vec::new();
927  while let Some(cs) = frontier.pop() {
928    if cs == target_cs {
929      return Ok(true);
930    }
931    if seen.contains(&cs) || seen.len() > 64 {
932      continue;
933    }
934    seen.push(cs);
935    if let Some(defn) = lookup_definition(&cs)?
936      && defn.is_expandable()
937      && let Some(ExpansionBody::Tokens(body)) = defn.get_expansion()
938    {
939      for t in body.unlist_ref() {
940        if t.get_catcode() == Catcode::CS
941          && t.with_cs_name(|n| n.starts_with("\\the") && n.ends_with("@ID"))
942        {
943          frontier.push(*t);
944        }
945      }
946    }
947  }
948  Ok(false)
949}

977/// `<ctype>` without a trailing sentinel control sequence the parameter reader may have pulled in.
978pub fn strip_counter_type_sentinel(ctype: &str) -> &str {
979  // Defensive: under some upstream conditions the {} parameter reader pulls a
980  // trailing `\par` (or similar trailing CS) into a counter-type identifier
981  // before it reaches us (Cluster A: math0010095, hep-ph0204075). Strip it
982  // here so the downstream `\csname @<ctype>...@ID\endcsname` and similar
983  // constructions stay well-formed. We strip the same well-known sentinels
984  // as latex_constructs::strip_trailing_cs.
985  for tail in ["\\par", "\\@startsection@hook", "\\relax"] {
986    if let Some(stripped) = ctype.strip_suffix(tail) {
987      return stripped;
988    }

1032/// The labelling half of `\refstepcounter` (latex.ltx:14956-14967, after its `\stepcounter`):
1033/// `\@currentlabel`, `\@currentID`, the tags and the counter's scope, for the counter `ctr` just
1034/// stepped for `ctype`. Returns the `tags` and `id` properties.
1035pub fn label_stepped_counter(ctype: &str, ctr: &str) -> Result<HashMap<Stored>> {
1036  let ctr = ctr.to_string();
1037  // A step is the current label's now (`note_raw_refstep` records it again for a raw one); global, as this port's
1038  // `\@currentlabel` is.
1039  if matches!(lookup_value(RAW_REFSTEP_TAGS), Some(Stored::Digested(_))) {
1040    assign_value(RAW_REFSTEP_TAGS, Stored::None, Some(Scope::Global));
1041  }
1042  maybe_preempt_refnum(&ctr, false);
1043
1044  let the_ctr_id = s!("\\the{ctr}@ID");
1045  let the_ctr = s!("\\the{ctr}");
1046
1047  let has_id: bool = match lookup_definition(&T_CS!(&the_ctr_id))? {
1048    Some(iddef) => {
1049      if let Some(params) = iddef.get_parameters() {
1050        params.get_num_args() == 0
1051      } else {
1052        true
1053      }
1054    },
1055    _ => false,
1056  };
1057
1058  let the_ctr_cs = T_CS!(&the_ctr);
1059  let the_ctr_id_cs = T_CS!(&the_ctr_id);
1060  def_macro(
1061    T_CS!("\\@currentlabel"),
1062    None,
1063    the_ctr_cs,
1064    Some(ExpandableOptions {
1065      scope: Some(Scope::Global),
1066      ..ExpandableOptions::default()
1067    }),
1068  )?;
1069  if has_id {
1070    def_macro(
1071      T_CS!("\\@currentID"),
1072      None,
1073      the_ctr_id_cs,
1074      Some(ExpandableOptions {
1075        scope: Some(Scope::Global),
1076        ..ExpandableOptions::default()
1077      }),
1078    )?;
1079  }
1080
1081  let id = if has_id {
1082    digest_literal(Tokens!(T_CS!(&the_ctr_id)))?.to_string()
1083  } else {
1084    String::new()
1085  };
1086
1087  let digest_label = || -> Result<(crate::Digested, crate::Digested)> {
1088    let refnum = digest_text(Tokens!(T_CS!(&the_ctr)))?;
1089    let invocation = build_invocation(T_CS!("\\lx@make@tags"), vec![Some(Tokens!(T_OTHER!(
1090      ctype
1091    )))])?;
1092    Ok((refnum, stomach::digest(invocation)?))
1093  };
1094  // TeX keeps the label unexpanded (`\@currentlabel`, latex.ltx:14963-14967) and typesets it only where `\ref`
1095  // prints it; Perl digests it as text here (Package.pm:770 `DigestText`). A step made in math whose label text
1096  // cannot set (`\renewcommand{\theconsta}{{\kappa_{\arabic{consta}}}}`, `$\refstepcounter{consta}…$`: three "Script _
1097  // can only appear in math mode" per step, 1609.03070, 1904.00290) is the author's math: its digest is discarded and
1098  // the label set as `\ensuremath{\the<ctr>}`, local to the label (OXIDIZED_DESIGN #748; guard
1099  // `perfect_kernel_batch73b::refstepcounter_in_math_sets_a_math_label`). A label text sets keeps Perl's digest.
1100  let (refnum, tags) = if lookup_bool("IN_MATH") {
1101    let hold = crate::util::logger::DiagnosticsHold::begin();
1102    match digest_label() {
1103      Ok(label) if hold.errors_raised() == 0 => {
1104        hold.commit();
1105        label
1106      },
1107      Err(err) => {
1108        hold.commit();
1109        return Err(err);
1110      },
1111      Ok(_) => {
1112        hold.discard();
1113        push_frame();
1114        let saved = T_CS!(s!("\\lx@the@in@math@{ctr}"));
1115        let math_label = Tokens::new(vec![T_CS!("\\ensuremath"), T_BEGIN!(), saved, T_END!()]);
1116        // (a label whose digest left a group open — a `\p@equation` taking nine arguments ate the closing braces,
1117        // 1902.09345 — keeps the frame this pushed, and with it the math label: saved again, the label would call
1118        // itself, `\@ensuremath` for ever)
1119        let label = if matches!(lookup_meaning(&the_ctr_cs), Some(Stored::Expandable(ref d))
1120          if matches!(d.get_expansion(), Some(ExpansionBody::Tokens(body)) if *body == math_label))
1121        {
1122          digest_label()
1123        } else {
1124          let_i(&saved, &the_ctr_cs, Some(Scope::Local));
1125          def_macro(
1126            the_ctr_cs,
1127            None,
1128            math_label,
1129            Some(ExpandableOptions {
1130              scope: Some(Scope::Local),
1131              ..ExpandableOptions::default()
1132            }),
1133          )
1134          .and_then(|()| digest_label())
1135        };
1136        pop_frame()?;
1137        label?
1138      },
1139    }
1140  } else {
1141    digest_label()?
1142  };
1143
1144  // Any scopes activated for previous value of this counter (& any nested counters) must be
1145  // removed. This may also include scopes activated for \label
1146  deactivate_counter_scope(arena::pin(&ctr));
1147
1148  // And install the scope (if any) for this reference number.
1149  assign_value("current_counter", ctr.clone(), Some(Scope::Local));
1150  // The step's serial: in scope (local, as LaTeX's `\@currentlabel` is, latex.ltx:14963) and the latest one made
1151  // (global). A `\label` whose scope's step is not the latest comes after a step a group has since undone
1152  // ([`label_follows_an_undone_step`]).
1153  let serial = match lookup_value(LATEST_REFSTEP_SERIAL) {
1154    Some(Stored::Int(serial)) => serial + 1,
1155    _ => 1,
1156  };
1157  assign_value(
1158    CURRENT_REFSTEP_SERIAL,
1159    Stored::Int(serial),
1160    Some(Scope::Local),
1161  );
1162  assign_value(
1163    LATEST_REFSTEP_SERIAL,
1164    Stored::Int(serial),
1165    Some(Scope::Global),
1166  );
1167
1168  let scope = arena::pin(format!("{ctr}:{refnum}"));
1169  let mut receiver = VecDeque::new();
1170  receiver.push_front(Stored::String(scope));
1171  assign_value(
1172    &s!("scopes_for_counter:{ctr}"),
1173    receiver,
1174    Some(Scope::Local),
1175  );
1176  activate_scope(scope);
1177
1178  Ok(stored_map!(
1179    "tags" => Stored::Digested(tags),
1180    "id" => Stored::String(arena::pin(id))
1181  ))
1182}

1184/// Internal: Use a label-derived reference number and/or ID
1185/// instead of the traditional counter based ones.
1186/// Since the \label{} determins the reference number and ID,
1187/// we MUST sniff out the label BEFORE we call RefStepCounter/RefStepID !!!!!
1188/// (see MaybePeekLabel below; and also MaybeNoteLabel for use within
1189/// captions & certain equation environments)
1190/// Assign a sub to LABEL_MAPPING_HOOK: &sub($label,$counter,$norefnum)
1191/// to return the desired refnum and id for a given object.
1192fn maybe_preempt_refnum(ctr: &str, norefnum: bool) {
1193  if let Some(mapper) = get_label_mapping_hook() {
1194    let hj_refnum = T_CS!(s!("\\_PREEMPTED_REFNUM_{ctr}"));
1195    let hj_id = T_CS!(s!("\\_PREEMPTED_ID_{ctr}"));
1196    // First, restore the \the<ctr> and \the<ctr>@ID macros to defaults
1197    if !norefnum && has_meaning(&hj_refnum) {
1198      let_i(&T_CS!(s!("\\the{ctr}")), &hj_refnum, Some(Scope::Global));
1199    }
1200    if has_meaning(&hj_id) {
1201      let_i(&T_CS!(s!("\\the{ctr}@ID")), &hj_id, Some(Scope::Global));
1202    }
1203    let label = lookup_string("PEEKED_LABEL");
1204    let (fixedrefnum, fixedid) = mapper(&label, ctr, norefnum);
1205    if let Some(refnum) = fixedrefnum
1206      && !norefnum
1207    {
1208      if !has_meaning(&hj_refnum) {
1209        // Save for later
1210        let_i(&hj_refnum, &T_CS!(s!("\\the{ctr}")), Some(Scope::Global));
1211      }
1212      let _ = def_macro(
1213        T_CS!(s!("\\the{ctr}")),
1214        None,
1215        ExpansionBody::Tokens(Tokens::new(Explode!(&refnum))),
1216        Some(ExpandableOptions {
1217          scope: Some(Scope::Global),
1218          ..Default::default()
1219        }),
1220      );
1221    }
1222    if let Some(id) = fixedid {
1223      if !has_meaning(&hj_id) {
1224        // Save for later
1225        let_i(&hj_id, &T_CS!(s!("\\the{ctr}@ID")), Some(Scope::Global));
1226      }
1227      let _ = def_macro(
1228        T_CS!(s!("\\the{ctr}@ID")),
1229        None,
1230        ExpansionBody::Tokens(Tokens::new(Explode!(&id))),
1231        Some(ExpandableOptions {
1232          scope: Some(Scope::Global),
1233          ..Default::default()
1234        }),
1235      );
1236    }
1237    remove_value("PEEKED_LABEL"); // CONSUME the label
1238    assign_value(
1239      "PROCESSED_LABEL",
1240      Stored::String(arena::pin(label)),
1241      Some(Scope::Global),
1242    );
1243  }
1244}

1296fn deactivate_counter_scope(ctr: SymStr) {
1297  let (scopes_for_counter, nested_counters) = arena::with(ctr, |cstr| {
1298    (
1299      s!("scopes_for_counter:{cstr}"),
1300      s!("nested_counters_{cstr}"),
1301    )
1302  });
1303  // with_value avoids the outer Stored::clone by reading through a
1304  // borrow; we still collect the scope SymStrs (Copy) or panic-pointers
1305  // into owned Vecs to outlive the borrow.
1306  let scope_syms: Vec<SymStr> = with_value(&scopes_for_counter, |v| match v {
1307    Some(Stored::VecDequeStored(stored_scopes)) => stored_scopes
1308      .iter()
1309      .map(|s| match s {
1310        Stored::String(scope) => *scope,
1311        _ => panic!("assignment scopes should be stored as strings, got: {s:?}"),
1312      })
1313      .collect(),
1314    _ => Vec::new(),
1315  });
1316  for scope in scope_syms {
1317    deactivate_scope(scope);
1318  }
1319
1320  // TODO: if we ever want to unshift from the nested_counters, we'll need to also use
1321  // Stored::VecDequeStored for them.
1322  let inner_ctrs: Vec<SymStr> = with_value(&nested_counters, |v| match v {
1323    Some(Stored::Strings(stored_counters)) => stored_counters.iter().copied().collect(),
1324    _ => Vec::new(),
1325  });
1326  for inner_ctr in inner_ctrs {
1327    deactivate_counter_scope(inner_ctr);
1328  }

1388/// Resets the counter `ctr` to zero.
1389pub fn reset_counter(ctr: &Token) -> Result<()> {
1390  let (c_ctr, c_un_ctr, ctr_id) =
1391    ctr.with_str(|ctr| (s!("\\c@{ctr}"), s!("\\c@UN{ctr}"), s!("\\@{ctr}@ID")));
1392  let exists = |cs: &str| lookup_register(cs, Vec::new()).ok().flatten().is_some();
1393  // A reset list names each child and its `UN` companion (Package.pm:674); a
1394  // companion never allocated (below) has nothing to reset.
1395  if ctr.with_str(|cstr| cstr.starts_with("UN")) && !exists(&c_ctr) {
1396    return Ok(());
1397  }
1398  assign_register(
1399    &c_ctr,
1400    Number::new(0).into(),
1401    Some(Scope::Global),
1402    Vec::new(),
1403  )?;
1404  // but not UN — and only a companion that exists: `\c@UN<ctr>` is LaTeXML's
1405  // (`new_counter`), so a raw `\newcount\c@CodelineNo` (doc.sty:870) in a reset
1406  // list (l3doc.cls:463 `\@addtoreset{CodelineNo}{part}`) has none, and
1407  // assigning it warned "not a register" at every `\part` (ltx-talk-code, 20).
1408  // Perl `ResetCounter` (Package.pm:886) assigns unconditionally and warns
1409  // too; the same existence test guards `ref_step_id` above (Package.pm:863).
1410  // OXIDIZED_DESIGN #268.
1411  if !ctr.with_str(|cstr| cstr.starts_with("UN")) && exists(&c_un_ctr) {
1412    assign_register(
1413      &c_un_ctr,
1414      Number::new(0).into(),
1415      Some(Scope::Global),
1416      Vec::new(),
1417    )?;
1418  }
1419  def_macro(
1420    T_CS!(ctr_id),
1421    None,
1422    Tokens!(T_OTHER!("0")),
1423    Some(ExpandableOptions {
1424      scope: Some(Scope::Global),
1425      ..ExpandableOptions::default()
1426    }),
1427  )?;
1428  // and reset any within counters!
1429  if let Some(nested) = lookup_tokens(&s!("\\cl@{ctr}")) {
1430    for c in nested.unlist() {
1431      reset_counter(&c)?;
1432    }
1433  }
1434  Ok(())
1435}

1499    // (latex_constructs.pool.ltxml:1383 `\let\the<counter>\@empty`), the enclosing item's label only while `\p@enumii`
1500    // is the kernel's `\theenumi`: enumitem empties it below the first level (enumitem.sty:551-556), and the reference
1501    // was lost (2603.02347, 7 items; covington's `\item[b.]` in an `{example}`, 2405.01972). So when the last numbered
1502    // item in force is an enclosing list's (`ITEM_LABEL_COUNTER` names another counter), the refnum is that item's
1503    // `\p@<ctr>\the<ctr>`, read before `\the<counter>` is emptied. After a numbered item of the same list LaTeX's
1504    // label is that sibling's. Perl's form keeps the enclosing item's label (`\p@enumii` = `\theenumi`, "1" where
1505    // LaTeX has "1a"), and nothing at all once enumitem has emptied `\p@<ctr>`: there a nested list's refnum is the
1506    // sibling's `\p@<ctr>\the<ctr>`, LaTeX's `\@currentlabel` ("a"). A top-level list keeps Perl's form (no refnum,
1507    // the reference reads the item's own label): its `\p@enumi` is empty too, but its authors replace the sibling's
1508    // label with their own (`\def\@currentlabel{(P2')}\label{…}`, 2502.11630; 2409.02896), and `\@currentlabel` is
1509    // not read here; so is a top-level `\item[…]` with no numbered item in force, where LaTeX's label is the last
1510    // section's or equation's. A nested list's author can override it too, and there the sibling's label wins
1511    // (2609.14520: "ii)" for the author's "ii′)"); reading an author-set `\@currentlabel` is deferred (user ruling
1512    // 2026-10-20, SYNC_STATUS 77u residual). KNOWN_PERL_ERRORS #1003; guards `perfect_kernel_batch77u::{
1513    // item_with_its_own_label_takes_the_enclosing_label, enumitem_item_with_its_own_label_takes_the_enclosing_label}`.
1514    let ctr = counter_for_type(&counter);
1515    let labelled = lookup_string(ITEM_LABEL_COUNTER);
1516    let sibling_label = labelled == ctr
1517      && !lookup_string(ITEM_LABEL_ENCLOSING).is_empty()
1518      && !prints_something(&s!("\\p@{ctr}"))?;
1519    if !labelled.is_empty() && (labelled != ctr || sibling_label) {
1520      for (alias, cs) in [
1521        ("\\lx@item@label@p@", s!("p@{labelled}")),
1522        ("\\lx@item@label@the@", s!("the{labelled}")),
1523      ] {
1524        tag_tokens.extend([
1525          T_CS!("\\expandafter"),
1526          T_CS!("\\let"),
1527          T_CS!("\\expandafter"),
1528          T_CS!(alias),
1529          T_CS!("\\csname"),
1530        ]);
1531        tag_tokens.extend(Explode!(cs));
1532        tag_tokens.push(T_CS!("\\endcsname"));
1533      }
1534      tag_tokens.extend([
1535        T_CS!("\\def"),
1536        T_CS!(s!("\\p@{ctr}")),
1537        T_BEGIN!(),
1538        T_CS!("\\lx@item@label@p@"),
1539        T_CS!("\\lx@item@label@the@"),
1540        T_END!(),
1541      ]);
1542    }
1543    tag_tokens.extend([
1544      T_CS!("\\let"),
1545      T_CS!(s!("\\the{counter}")),
1546      T_CS!("\\@empty"),
1547      T_CS!("\\def"),
1548      T_CS!(s!("\\fnum@{counter}")),
1549      T_BEGIN!(),
1550      formatter,
1551      T_BEGIN!(),
1552    ]);
1553    // TODO: Another iffy clone...
1554    let reverted_tag = (*tag).clone().revert();
1555    tag_tokens.extend(reverted_tag.clone());
1556    tag_tokens.extend(vec![
1557      T_END!(),
1558      T_END!(),
1559      T_CS!("\\def"),
1560      T_CS!(s!("\\typerefnum@{counter}")),
1561      T_BEGIN!(),
1562      typename,
1563      T_SPACE!(),
1564    ]);
1565    tag_tokens.extend(reverted_tag);
1566    tag_tokens.push(T_END!());
1567    tag_tokens.extend(
1568      build_invocation(T_CS!("\\lx@make@tags"), vec![Some(Tokens!(T_OTHER!(
1569        counter
1570      )))])?
1571      .unlist(),
1572    );
1573    tag_tokens.push(T_END!());
1574
1575    let tags = stomach::digest(tag_tokens)?;
1576    if !tags.is_empty()? {
1577      props.insert("tags", tags.into());
1578    }
1579    props
1580  } else {
1581    let props = ref_step_counter_by_meaning(&counter)?;
1582    // The item set `\@currentlabel` (latex.ltx `\refstepcounter` in `\@item`) when its list is numbered — its
1583    // `\the<counter>` prints something; an itemize's `\the@item<n>` and a description's are empty (sect06.rs), as
1584    // LaTeX's `\@nmbrlistfalse` lists never step, so an `\item[…]` after an itemize sibling still takes the enclosing
1585    // enumerate item's label ("1"). Local to the list, as the `\protected@edef` is; the first numbered item records
1586    // the enclosing list's counter it replaces.
1587    let ctr = counter_for_type(&counter);
1588    if prints_something(&s!("\\the{ctr}"))? {
1589      let in_force = lookup_string(ITEM_LABEL_COUNTER);
1590      if in_force != ctr {
1591        assign_value(ITEM_LABEL_ENCLOSING, in_force, None);
1592      }
1593      assign_value(ITEM_LABEL_COUNTER, ctr, None);
1594    }
1595    props
1596  };
1597  for (k, v) in attr.into_iter() {
1598    result.insert_sym(k, v);
1599  }
1600  Ok(result)
1601}

1637/// Prepare for an list (itemize/enumerate/description/etc)
1638/// by determining the right counter (level)
1639/// and binding the right \item ( \$type@item, if $type is defined)
1640pub fn begin_itemize(
1641  itype: &str,
1642  counter: Option<&str>,
1643  options: BeginItemizeOptions,
1644) -> Result<HashMap<Stored>> {
1645  // The list-type and level of the *containing* list (if any!)
1646  let outercounter = lookup_string("itemcounter");
1647  let outerlevel = if !outercounter.is_empty() {
1648    lookup_int(&s!("{outercounter}level"))
1649  } else {
1650    0
1651  };
1652  let counter = counter.unwrap_or("@item");
1653  let listlevel = lookup_int("itemization_level") + 1; // level for this list overall
1654  let level = lookup_int(&s!("{counter}level")) + // level for lists of specific type
1655    (if options.nolevel { 0 } else { 1 });
1656  AssignRegister!(
1657    "\\itemsep",
1658    lookup_dimension("\\lx@default@itemsep")
1659      .unwrap_or_default()
1660      .into()
1661  );
1662  assign_value("itemization_level", listlevel, None);
1663  assign_value(&s!("{counter}level"), level, None);
1664  // latex.ltx's `\itemize`/`\enumerate` advance `\@itemdepth`/`\@enumdepth` inside their group
1665  // (as do enumitem's and paralist's inline forms; a `\newlist` list has its own
1666  // `\enitdp@<name>`), and every list but an in-paragraph one opens a `\list` that advances
1667  // `\@listdepth` globally until `\endlist` (latex.ltx:15852, :15913; enumitem.sty:1188-1191,
1668  // :1254 for its inline lists). Packages read them (enumitem's `list<depth>` keys,
1669  // enumitem.sty:977-978). The kernel `\list` binding keeps its own `\@listdepth`, and
1670  // `\trivlist` opens no `\list`. Perl's beginItemize (latex_constructs.pool.ltxml:1296) leaves all
1671  // three at 0 (SHARED).
1672  let depth_register = match itype {
1673    "itemize" | "inline@itemize" => Some("\\@itemdepth"),
1674    "enumerate" | "inline@enumerate" => Some("\\@enumdepth"),
1675    _ => None,
1676  };
1677  let register_depth = |cs: &str| match lookup_register_quiet(cs) {
1678    Some(RegisterValue::Number(n)) => n.0,
1679    _ => 0,
1680  };
1681  let mut item_depth = None;
1682  if let Some(cs) = depth_register {
1683    let depth = register_depth(cs) + 1;
1684    assign_register(cs, RegisterValue::Number(Number(depth)), None, vec![])?;
1685    if cs == "\\@itemdepth" && counter == "@item" {
1686      item_depth = Some(depth);
1687    }
1688  }
1689  if !options.opens_no_list && !matches!(itype, "list" | "trivlist") {
1690    let depth = register_depth("\\@listdepth");
1691    assign_register(
1692      "\\@listdepth",
1693      RegisterValue::Number(Number(depth + 1)),
1694      Some(Scope::Global),
1695      vec![],
1696    )?;
1697    let mut restore = vec![T_CS!("\\lx@listdepth@restore")];
1698    restore.extend(Explode!(&depth.to_string()));
1699    restore.push(T_CS!("\\relax"));
1700    push_value("afterGroup", Stored::Tokens(Tokens::new(restore)))?;
1701  }
1702  assign_value("itemization_items", 0, None);
1703  let listpostfix = roman!(listlevel).to_string();
1704  let postfix = roman!(level).to_string();
1705  let mut usecounter = counter.to_string();
1706  if !options.nolevel && !postfix.is_empty() {
1707    usecounter.push_str(&postfix);
1708  }
1709  // The `@item` levels past the six the kernel declares (sect06.rs) are declared as they are reached, in the same
1710  // id chain: lists nest that deep where a source opens trivlists it ends at once (doc.sty's `\@doc@env` opens one
1711  // per documented name; frankenstein, source2e).
1712  if counter == "@item" && level > 6 && lookup_definition(&T_CS!(s!("\\c@{usecounter}")))?.is_none()
1713  {
1714    let within = s!("@item{}", roman!(level - 1));
1715    new_counter(
1716      &usecounter,
1717      "",
1718      Some(NewCounterOptions {
1719        idprefix: "i",
1720        idwithin: &within,
1721        ..Default::default()
1722      }),
1723    )?;
1724    // As the kernel's six: an unnumbered item, named as an item (Perl pool:1514-1519, :1622).
1725    let global = || {
1726      Some(ExpandableOptions {
1727        scope: Some(Scope::Global),
1728        ..Default::default()
1729      })
1730    };
1731    def_macro(
1732      T_CS!(s!("\\the{usecounter}")),
1733      None,
1734      Tokens::new(Vec::new()),
1735      global(),
1736    )?;
1737    def_macro(
1738      T_CS!(s!("\\{usecounter}name")),
1739      None,
1740      Tokens!(T_CS!("\\itemtyperefname")),
1741      global(),
1742    )?;
1743  }
1744  if !itype.starts_with("inline@") {
1745    run_trivlist_hook()?;
1746  }
1747  if !itype.is_empty() {
1748    let itype_cs = T_CS!(s!("\\{itype}@item"));
1749    let_i(&T_CS!("\\item"), &itype_cs, None);
1750  }
1751  // In case within odd environment.
1752  let_i(&T_CS!("\\par"), &T_CS!("\\lx@normal@par"), None);
1753  def_macro(
1754    T_CS!("\\@listctr"),
1755    None,
1756    Tokens!(Explode!(usecounter)),
1757    None,
1758  )?;
1759  // latex.ltx:16068-16075 `\itemize` labels by `\labelitem\romannumeral\the\@itemdepth`, which only
1760  // itemizes advance; the `@item` level (the counter the item ids hang on) also counts kernel
1761  // `\list`s and `\trivlist`s (Perl pool:1639 `beginItemize('list')`), so an itemize inside a
1762  // `\list` took `\labelitemii` where pdflatex prints `\labelitemi`. The label follows the depth; the
1763  // counter, and so every id, stays Perl's. PERL-ORIGIN, pdflatex the oracle. A caller's own label
1764  // (enumitem's `label=`, paralist's `[<label>]` through `set_itemization_style`) is defined after
1765  // this. Repros: list-structure/{itemize_in_a_list_takes_the_first_label,
1766  // itemize_label_follows_the_itemize_depth}.
1767  if let Some(depth) = item_depth
1768    && depth != level
1769    && (1..=4).contains(&depth)
1770  {
1771    def_macro(
1772      T_CS!(s!("\\label{usecounter}")),
1773      None,
1774      Tokens!(T_CS!(s!("\\labelitem{}", roman!(depth)))),
1775      None,
1776    )?;
1777  }
1778  // latex.ltx:16057 `\enumerate` also does
1779  // `\edef\@enumctr{enum\romannumeral\the\@enumdepth}`; packages that reuse
1780  // the kernel enumerate read it back (bullenum.sty:58/61
1781  // `\csname the\@enumctr\endcsname`). `usecounter` is already
1782  // `enum` + roman(level), so this is the same value; enumerate-only, as in
1783  // the kernel. Perl's beginItemize (pool:1314) omits it (SHARED).
1784  // Witness: bullcntr/bullcntr-man (100-error cascade).
1785  if itype == "enumerate" {
1786    def_macro(
1787      T_CS!("\\@enumctr"),
1788      None,
1789      Tokens!(Explode!(usecounter)),
1790      None,
1791    )?;
1792  }
1793  // Now arrange that this list's id's are relative to the current (outer) item (if any)
1794  // And that the items within this list's id's are relative to this (new) list.
1795  assign_value("itemcounter", Stored::String(arena::pin(&usecounter)), None);
1796  // …and the environment this itemization is (natbib's `\bibitem` keeps its place only among this list's own `\item`s)
1797  assign_value(
1798    "itemization_environment",
1799    Stored::String(arena::pin(lookup_string("current_environment"))),
1800    None,
1801  );
1802  let listcounter = s!("@itemize{listpostfix}");
1803  if lookup_definition(&T_CS!(s!("\\c@{listcounter}")))?.is_none() {
1804    //Create new list counters as needed
1805    new_counter(&listcounter, "", None)?;
1806  }
1807  let mut shared = false;
1808  if !outercounter.is_empty() {
1809    // Make this list's ID relative to outer list's ID
1810    let outerusecounter = s!("{outercounter}{}", roman!(outerlevel).to_string());
1811    let thectr = s!("\\the{listcounter}@ID");
1812    // Explicit tokens, as `useexp` below: a counter name with a digit (`steps2i`) must stay one
1813    // control sequence (KPE #414).
1814    let mut expansion = vec![
1815      T_CS!(s!("\\the{outerusecounter}@ID")),
1816      T_OTHER!(".I"),
1817      T_CS!("\\arabic"),
1818      T_BEGIN!(),
1819    ];
1820    expansion.extend(Explode!(&listcounter));
1821    expansion.push(T_END!());
1822    // A list numbered by a counter an enclosing list's id hangs on — `\list{}{\usecounter{enumi}}`
1823    // around an `enumerate`, directly or through lists in between — redefines that counter's id
1824    // below through this list's, so the formatters would expand one another
1825    // (Fatal:Timeout:PushbackLimit; Perl loops too, latex_constructs.pool.ltxml:1323-1335). Then
1826    // the list hangs on the enclosing item, its id expanded once (the item's `\@<counter>@ID` is put
1827    // back when a list on the shared counter closes, below), and the shared counter's steps (this
1828    // list's own items) no longer reset the open lists' counters. Only inputs that looped meet the
1829    // condition. Guards `perfect_kernel_batch58::{
1830    // numbered_list_with_a_nested_enumerate_converts, list_numbered_by_an_enclosing_counter}`.
1831    shared = list_id_chain_reaches(&outerusecounter, &usecounter)?;
1832    if shared {
1833      let outer_id = crate::gullet::do_expand(Tokens!(expansion[0]))?;
1834      let mut frozen = outer_id.unlist();
1835      frozen.extend(expansion.drain(1..));
1836      expansion = frozen;
1837    }
1838    def_macro(T_CS!(thectr), None, Tokens::new(expansion), None)?;
1839    if shared {
1840      let open: Vec<Token> = (1..=listlevel)
1841        .map(|k| T_CS!(s!("@itemize{}", roman!(k))))
1842        .collect();
1843      let cl_use = s!("\\cl@{usecounter}");
1844      let kept = with_value(&cl_use, |v| match v {
1845        Some(Stored::Tokens(tks)) => Some(
1846          tks
1847            .clone()
1848            .unlist()
1849            .into_iter()
1850            .filter(|t| !open.contains(t))
1851            .collect::<Vec<_>>(),
1852        ),
1853        _ => None,
1854      });
1855      if let Some(kept) = kept {
1856        assign_value(
1857          &cl_use,
1858          Stored::Tokens(Tokens::new(kept)),
1859          Some(Scope::Global),
1860        );
1861      }
1862    }
1863
1864    // AND reset this list's counter when the outer item is stepped
1865    let mut cl_toks = if shared && outerusecounter == usecounter {
1866      Vec::new()
1867    } else {
1868      vec![T_CS!(&listcounter)]
1869    };
1870    let cl_name = s!("\\cl@{outerusecounter}");
1871    let existing = with_value(&cl_name, |v| match v {
1872      Some(Stored::Tokens(tks)) => tks.clone().unlist(),
1873      _ => Vec::new(),
1874    });
1875    cl_toks.extend(existing);
1876    assign_value(
1877      &cl_name,
1878      Stored::Tokens(Tokens::new(cl_toks)),
1879      Some(Scope::Global),
1880    );
1881  }
1882  // format the id of \item's relative to the id of this list.
1883  // Perl: Tokens(T_CS('\the' . $listcounter . '@ID'), T_OTHER('.i'),
1884  //              T_CS('\@' . $usecounter . '@ID'))
1885  // — build the Tokens array directly from 3 explicit tokens rather than
1886  // round-tripping through a string tokenizer. This matters when `usecounter`
1887  // contains digits (e.g. "count1" from \usecounter{count1}): `tokenize_internal`
1888  // on a literal string "\@count1@ID" would split into \@count + "1" + "@" + "ID"
1889  // because TeX's CS reader stops at digits (digits are catcode OTHER even in
1890  // the style table). Perl uses T_CS($name) to build a single CS token directly
1891  // by name, bypassing tokenization.
1892  let useexp = Tokens::new(vec![
1893    T_CS!(s!("\\the{listcounter}@ID")),
1894    T_OTHER!(".i"),
1895    T_CS!(s!("\\@{usecounter}@ID")),
1896  ]);
1897  def_macro(T_CS!(s!("\\the{usecounter}@ID")), None, useexp, None)?;
1898
1899  let mut series = if let Some(s) = options.series {
1900    s.to_string()
1901  } else {
1902    String::new()
1903  };
1904  // The shared counter's `\@<counter>@ID` is the enclosing item's id while this list is open
1905  // (`shared`, above): this list resets and steps it, so its value as the enclosing item had it is
1906  // put back, globally as the step set it, when the list closes — a later sibling list, or the
1907  // enclosing list's next items, hang on that item again.
1908  if shared {
1909    let at_id = T_CS!(s!("\\@{usecounter}@ID"));
1910    if let Some(defn) = lookup_definition(&at_id)?
1911      && let Some(ExpansionBody::Tokens(body)) = defn.get_expansion()
1912    {
1913      let mut restore = vec![T_CS!("\\gdef"), at_id, T_BEGIN!()];
1914      restore.extend(body.clone().unlist());
1915      restore.push(T_END!());
1916      push_value("afterGroup", Stored::Tokens(Tokens::new(restore)))?;
1917    }
1918  }
1919  if let Some(start) = options.start {
1920    SetCounter!(usecounter, start);
1921    add_to_counter(&usecounter, Number(-1))?;
1922  } else if let Some(s) = match options.resume {
1923    Some(s) => Some(s),
1924    None => options.resume_star,
1925  } {
1926    if s != "noseries" {
1927      series = s.clone();
1928      let last_val = lookup_int(&s!("enumitem_series_{s}_last"));
1929      if last_val != 0 {
1930        SetCounter!(usecounter, Number(last_val));
1931      }
1932    }
1933  } else {
1934    reset_counter(&T_OTHER!(&usecounter))?;
1935  }
1936
1937  let mut rsc = ref_step_counter(&s!("@itemize{listpostfix}"), false)?;
1938  rsc.insert("counter", usecounter.into());
1939  rsc.insert("series", series.into());
1940  // Perl latex_constructs.pool L1354-1356: the list carries its TeX vertical
1941  // padding (\topsep + \parskip + \partopsep) as padtop/padbottom sizing
1942  // properties; Box::computeSizeStore adds them to the computed height/depth
1943  // (compute_size_and_cache here). Omitting them under-measured every list,
1944  // clipping tcolorbox frames drawn from the estimate (2605.02240).
1945  let pad = lookup_dimension_cs("\\topsep", false)
1946    .unwrap_or_default()
1947    .add(lookup_dimension_cs("\\parskip", false).unwrap_or_default())
1948    .add(lookup_dimension_cs("\\partopsep", false).unwrap_or_default());
1949  rsc.insert("padtop", Stored::Dimension(pad));
1950  rsc.insert("padbottom", Stored::Dimension(pad));
1951  Ok(rsc)
1952}

1979/// Parses the style tokens to detect A/a/I/i/1 patterns
1980/// and defines \theenum$level and \labelenum$level accordingly.
1981pub fn set_enumeration_style(stuff: Option<&Tokens>, level: Option<i32>) -> Result<()> {
1982  if let Some(stuff) = stuff {
1983    if stuff.is_empty() {
1984      return Ok(());
1985    }
1986    let level = level.unwrap_or_else(|| lookup_int("enumlevel").max(0) as i32);
1987    let level_str = roman_aux(level);
1988    // Iterate the borrowed token slice — no clone needed, only reads.
1989    let tokens = stuff.unlist_ref();
1990    let mut out: Vec<Token> = Vec::with_capacity(tokens.len());
1991    let ctr = T_OTHER!(s!("enum{level_str}"));
1992    let mut i = 0;
1993    while i < tokens.len() {
1994      let t = tokens[i];
1995      if t.get_catcode() == Catcode::BEGIN {
1996        // Copy braced groups verbatim
1997        out.push(t);
1998        let mut brlevel = 1i32;
1999        i += 1;
2000        while brlevel > 0 && i < tokens.len() {
2001          let tt = tokens[i];
2002          if tt.get_catcode() == Catcode::BEGIN {
2003            brlevel += 1;
2004          } else if tt.get_catcode() == Catcode::END {
2005            brlevel -= 1;
2006          }
2007          out.push(tt);
2008          i += 1;
2009        }
2010      } else {
2011        let ch = char::from_u32(t.get_charcode()).unwrap_or('\0');
2012        let cat = t.get_catcode();
2013        match (ch, cat) {
2014          ('A', Catcode::LETTER) => {
2015            // \Alph{enum$level}
2016            def_macro(
2017              T_CS!(s!("\\theenum{level_str}")),
2018              None,
2019              Tokens::new(vec![T_CS!("\\Alph"), T_BEGIN!(), ctr, T_END!()]),
2020              None,
2021            )?;
2022            out.push(T_CS!(s!("\\theenum{level_str}")));
2023          },
2024          ('a', Catcode::LETTER) => {
2025            // \alph{enum$level}
2026            def_macro(
2027              T_CS!(s!("\\theenum{level_str}")),
2028              None,
2029              Tokens::new(vec![T_CS!("\\alph"), T_BEGIN!(), ctr, T_END!()]),
2030              None,
2031            )?;
2032            out.push(T_CS!(s!("\\theenum{level_str}")));
2033          },
2034          ('I', Catcode::LETTER) => {
2035            // \Roman{enum$level}
2036            def_macro(
2037              T_CS!(s!("\\theenum{level_str}")),
2038              None,
2039              Tokens::new(vec![T_CS!("\\Roman"), T_BEGIN!(), ctr, T_END!()]),
2040              None,
2041            )?;
2042            out.push(T_CS!(s!("\\theenum{level_str}")));
2043          },
2044          ('i', Catcode::LETTER) => {
2045            // \roman{enum$level}
2046            def_macro(
2047              T_CS!(s!("\\theenum{level_str}")),
2048              None,
2049              Tokens::new(vec![T_CS!("\\roman"), T_BEGIN!(), ctr, T_END!()]),
2050              None,
2051            )?;
2052            out.push(T_CS!(s!("\\theenum{level_str}")));
2053          },
2054          ('1', Catcode::OTHER) => {
2055            // \arabic{enum$level}
2056            def_macro(
2057              T_CS!(s!("\\theenum{level_str}")),
2058              None,
2059              Tokens::new(vec![T_CS!("\\arabic"), T_BEGIN!(), ctr, T_END!()]),
2060              None,
2061            )?;
2062            out.push(T_CS!(s!("\\theenum{level_str}")));
2063          },
2064          _ => {
2065            out.push(t);
2066          },
2067        }
2068        i += 1;
2069      }
2070    }
2071    // Define \labelenum$level = { out }
2072    let mut label_tokens = vec![T_BEGIN!()];
2073    label_tokens.extend(out);
2074    label_tokens.push(T_END!());
2075    def_macro(
2076      T_CS!(s!("\\labelenum{level_str}")),
2077      None,
2078      Tokens::new(label_tokens),
2079      None,
2080    )?;
2081  }
2082  Ok(())
2083}
