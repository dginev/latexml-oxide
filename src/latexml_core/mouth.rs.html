273/// Decode raw input bytes as text when no encoding has been declared:
274/// valid UTF-8 is taken as-is, anything else is a Latin-1 passthrough
275/// (byte → char).
276///
277/// Mirrors Perl `Mouth.pm` L75-80: when `PERL_INPUT_ENCODING` is undef Perl
278/// never decodes, so the bytes pass through untouched and the read cannot
279/// fail. The point is that a non-UTF-8 file is **never lost** — only decoded
280/// conservatively. `std::fs::read_to_string` gives the opposite behaviour
281/// (hard error on the first stray byte), which silently cost witness
282/// 2605.00490 its entire bibliography: a JabRef-written `.bib` self-declaring
283/// `% Encoding: Cp1252`. Real `bibtex` 0.99d is 8-bit clean and reads it fine.
284///
285/// Latin-1 (rather than `from_utf8_lossy`) is the better fallback here
286/// because it is lossless byte → char: legacy `.bib` files are overwhelmingly
287/// Latin-1/Cp1252, whose accented names survive intact instead of collapsing
288/// to U+FFFD.
289///
290/// The fallback is applied **per line**, not per buffer. `raw` is a single
291/// line when the Mouth calls this, but a whole file when a `.bib` reader does,
292/// and decoding a whole file as Latin-1 because of one stray byte would
293/// mojibake every correctly-UTF-8-encoded name in it (`é` → `Ã©`). Per-line
294/// keeps the damage to the offending line and matches the Mouth's own
295/// granularity. The all-valid-UTF-8 case (the overwhelming majority) still
296/// costs exactly one `from_utf8` SIMD validation of the whole buffer.
297/// KERNEL_CAPABILITIES K10: under the pdfTeX byte mouth a non-ASCII string
298/// has two spellings — the bytes (a run of chars in U+0080..U+00FF, from
299/// `\string`/`\detokenize`/`\csname` of the active bytes, or a stringified
300/// token list) and the character the octet readers emit. pdfTeX knows only
301/// the bytes, so the two never meet there; here they meet at every string
302/// boundary (a color name defined from a detokenized list and used from the
303/// text — cncolours.sty:21 vs pgfornament-han-doc; hyperref's `pdftitle`;
304/// a theorem tag). Decode every maximal byte run that is valid UTF-8 back
305/// into its characters; anything else stays as it is. A no-op unless the
306/// byte mouth is on.
307pub fn decode_byte_mouth_runs(s: &str) -> std::borrow::Cow<'_, str> {
308  if !s.chars().any(is_byte_spelling) || !lookup_bool_sym(crate::pin!("PDFTEX_BYTE_MOUTH")) {
309    return std::borrow::Cow::Borrowed(s);
310  }
311  let mut out = String::with_capacity(s.len());
312  let mut run: Vec<u8> = Vec::new();
313  for c in s.chars() {
314    if is_byte_spelling(c) {
315      run.push(c as u32 as u8);
316    } else {
317      decode_byte_run(&mut run, &mut out);
318      out.push(c);
319    }
320  }
321  decode_byte_run(&mut run, &mut out);
322  std::borrow::Cow::Owned(out)
323}

329/// Decode one maximal byte run SEQUENCE by sequence: every valid UTF-8
330/// sequence becomes its character, an invalid byte stays its Latin-1 image,
331/// and a trailing INCOMPLETE sequence stays as bytes — it is still being
332/// assembled when the run is a text node the bytes reach one token at a time
333/// (`byte_mouth_pending_tail`). An all-or-nothing `from_utf8` of the run
334/// mis-read a decoded Latin-1 character (é = U+00E9, a 3-byte lead's image)
335/// next to a fresh CJK sequence as one invalid run and left both undecoded.
336fn decode_byte_run(run: &mut Vec<u8>, out: &mut String) {
337  let mut rest: &[u8] = run;
338  while !rest.is_empty() {
339    match str::from_utf8(rest) {
340      Ok(text) => {
341        out.push_str(text);
342        break;
343      },
344      Err(e) => {
345        let (valid, bad) = rest.split_at(e.valid_up_to());
346        out.push_str(str::from_utf8(valid).unwrap_or_default());
347        let n = e.error_len().unwrap_or(bad.len());
348        out.extend(bad[..n].iter().map(|&b| b as char));
349        rest = &bad[n..];
350      },
351    }
352  }
353  run.clear();
354}

356/// Byte offset in `s` where a trailing INCOMPLETE UTF-8 sequence in the byte
357/// spelling begins (`s.len()` when there is none). The text-run merge in
358/// `Document::open_text_internal` re-decodes only from there: the bytes of
359/// one character arrive one token at a time (tex.web §1034: one char node per
360/// byte), so a run is complete only once its last continuation byte is on,
361/// and everything before the pending lead byte is already decoded (or
362/// invalid and final). Invariant relied on: a decoded character in
363/// U+00C2..U+00F4 (é, Ã) is never followed by raw continuation bytes without
364/// their own lead — well-formed input never produces that shape, so the one
365/// preceding character this looks back at is only ever re-merged when it IS
366/// the pending lead byte.
367pub fn byte_mouth_pending_tail(s: &str) -> usize {
368  let mut continuations = 0usize;
369  for (i, c) in s.char_indices().rev() {
370    let u = c as u32;
371    if (0x80..=0xBF).contains(&u) {
372      continuations += 1;
373      if continuations > 3 {
374        return s.len();
375      }
376      continue;
377    }
378    let needed = match u {
379      0xC2..=0xDF => 1,
380      0xE0..=0xEF => 2,
381      0xF0..=0xF4 => 3,
382      _ => return s.len(),
383    };
384    return if continuations < needed { i } else { s.len() };
385  }
386  s.len()
387}

389pub fn decode_input_bytes(raw: &[u8]) -> String {
390  match str::from_utf8(raw) {
391    Ok(s) => s.to_string(),
392    Err(_) => {
393      let mut out = String::with_capacity(raw.len());
394      for (i, line) in raw.split(|&b| b == b'\n').enumerate() {
395        if i > 0 {
396          out.push('\n');
397        }
398        match str::from_utf8(line) {
399          Ok(s) => out.push_str(s),
400          Err(_) => out.extend(line.iter().map(|&b| b as char)),
401        }
402      }
403      out
404    },
405  }
406}

408/// The character an 8-bit input byte stands for, when it reaches the tokenizer as a
409/// plain letter or other character instead of through its active inputenc
410/// definition.
411///
412/// inputenc makes every upper byte active (`\DeclareInputText{199}{\CYRZ}`,
413/// cp1251.def:54), and the active definition yields the character. A package that
414/// hands the bytes back to the letter class (russ.sty:58-63 `\catcode…=11`, so
415/// Cyrillic words can form control-sequence names) bypasses that definition: in TeX
416/// the byte then goes straight to the font as a slot, and T2A's letter block is laid
417/// out like cp1251, so the glyph is still the letter the author typed. LaTeXML is
418/// Unicode-native — a byte is decoded where it enters, as utf8 input already is — so
419/// the byte becomes the character its own inputenc declaration produces: the active
420/// definition followed through single-token macros (`\IeC{…}` unwrapped) to one
421/// character (`\CYRZ` → `З`, fontenc_sty.rs). Only for a genuine input byte — a
422/// character of a file line that is not UTF-8 (decoded as its bytes), or a `^^xx`
423/// code — under a declared 8-bit input encoding, outside the pdfTeX byte mouth (K10);
424/// a valid UTF-8 line keeps its characters (é stays é in an `\input` UTF-8 file). The
425/// catcode stays the byte's. Everything downstream is then Unicode, so the font map
426/// never has to guess whether a code in 128–255 is a slot or a character.
427/// `None` keeps the byte (no declaration, or one that is not a single character).
428/// OXIDIZED_DESIGN #266.
429fn eight_bit_input_char(byte: char) -> Option<char> {
430  use crate::definition::Definition;
431  if lookup_bool_sym(crate::pin!("PDFTEX_BYTE_MOUTH")) {
432    return None;
433  }
434  let encoding = lookup_string_from_sym(crate::pin!("INPUT_ENCODING"));
435  if encoding.is_empty() || encoding == "utf8" {
436    return None;
437  }
438  let mut meaning = lookup_meaning(&T_ACTIVE!(byte))?;
439  for _ in 0..6 {
440    let Stored::Expandable(ref defn) = meaning else {
441      return None;
442    };
443    let Some(crate::definition::ExpansionBody::Tokens(body)) = defn.get_expansion() else {
444      return None;
445    };
446    let toks: Vec<Token> = body
447      .unlist_ref()
448      .iter()
449      .copied()
450      .filter(|t| {
451        !matches!(t.get_catcode(), Catcode::BEGIN | Catcode::END) && !t.with_str(|s| s == "\\IeC")
452      })
453      .collect();
454    let [t] = toks.as_slice() else {
455      return None;
456    };
457    match t.get_catcode() {
458      Catcode::LETTER | Catcode::OTHER => {
459        let mut it = t.with_str(|s| s.chars().collect::<Vec<char>>()).into_iter();
460        return match (it.next(), it.next()) {
461          (Some(c), None) if c != byte => Some(c),
462          _ => None,
463        };
464      },
465      Catcode::CS => {
466        // A LaTeX text command (`\CYRZ` = the `\cf@encoding` dispatch,
467        // latex.ltx `\DeclareTextSymbol`): its glyph is `\<encoding>\CYRZ` for
468        // the font encoding current when the byte is READ (TeX resolves it when
469        // the letter is typeset), a one-character primitive decoded from that
470        // encoding's font map (sect08.rs) — T2A slot 199 → `З`.
471        if let Some(c) = text_symbol_char(t) {
472          return Some(c);
473        }
474        meaning = lookup_meaning(t)?;
475      },
476      Catcode::ACTIVE => meaning = lookup_meaning(t)?,
477      _ => return None,
478    }
479  }
480  None
481}

483/// The single character `\<current encoding><cs>` typesets, if it is a
484/// `\DeclareTextSymbol` glyph (a primitive with a one-character string body).
485pub fn text_symbol_char(cs: &Token) -> Option<char> {
486  use crate::definition::PrimitiveBody;
487  // `\cf@encoding` is the current font's encoding (a computed macro here).
488  let encoding = lookup_font()?.get_encoding()?.to_string();
489  let glyph_cs = T_CS!(format!("\\{}{}", encoding.trim(), cs.to_string()));
490  let Some(Stored::Primitive(prim)) = lookup_meaning(&glyph_cs) else {
491    return None;
492  };
493  let Some(PrimitiveBody::String(sym)) = prim.replacement else {
494    return None;
495  };
496  let mut it = crate::common::arena::with(sym, |s| s.chars().collect::<Vec<char>>()).into_iter();
497  match (it.next(), it.next()) {
498    (Some(c), None) => Some(c),
499    _ => None,
500  }

506  // options are
507  //  quiet,

511  // DG: For now we are using a `foodtype` field instead of subclassing mouth, as it feels more

513  // application logic, with a capacity of reading different kinds of sources
514  pub fn create(source: &str, mut options: MouthOptions) -> Result<Self> {
515    if let Some(content) = options.content.take() {
516      // we've cached the content of this source
517      let (_dir, name, ext) = pathname::split(source);
518      options.source = Some(source.to_string());
519      options.shortsource = Some(s!("{}.{}", name, ext));
520      options.lazy_decode = true;
521      options.file_level = true;
522      // Read-log: a named cached-content open (filecontents / LSP overlay).
523      record_opened_source(crate::common::arena::pin(source));
524      Mouth::new(&content, Some(options))
525    } else if source.starts_with("literal:") {
526      let source = source.replacen("literal:", "", 1);
527      // we've supplied literal data
528      options.source = None; // the source does not have a corresponding file name
529      options.foodtype = FoodType::opt_from_str("literal");
530      Mouth::new(&source, Some(options))
531    } else if source.is_empty() {
532      Mouth::new("", Some(options))
533    } else {
534      let (_dir, name, ext) = pathname::split(source);
535      options.foodtype = FoodType::opt_from_str(&pathname::protocol(source));
536      options.file_level = true;
537      options.source = Some(source.to_string());
538      if options.shortsource.is_none() {
539        options.shortsource = Some(if ext.is_empty() {
540          name
541        } else {
542          s!("{}.{}", name, ext)
543        });
544      }
545      // Read-log: a named file open (recorded even when the open then
546      // fails — a pinned-but-missing path that later APPEARS must read
547      // as a dependency change).
548      record_opened_source(crate::common::arena::pin(source));
549      Mouth::new(source, Some(options))
550    }

590  /// Read `% & #` as ordinary characters (catcode 12) instead of comment,
591  /// alignment tab and parameter, for the whole life of this mouth.
592  ///
593  /// **Treatment 1 of two** (see `OXIDIZED_DESIGN #74`): this is "be `bibtex`".
594  /// BibTeX's lexer interprets only braces and the entry/field delimiters — it
595  /// has no comment syntax inside an entry (`%` is significant only in the junk
596  /// BETWEEN entries, `Pre::BibTeX::skipJunk`), no alignment and no parameters.
597  /// So a field value it hands back is a string in which all three are ordinary
598  /// characters: a percent-encoded URL, a publisher's name ("Taylor &
599  /// Francis"), an issue number.
600  ///
601  /// Re-injected as TeX source (BibTeX.pool's `\bibentry@create`) under the
602  /// default catcodes, each misfires: `%` (14) comments out the rest of its
603  /// line — the field's own closing brace included — so the entry's group never
604  /// closes; `&` (4) is a stray alignment tab and is dropped; `#` (6) reaches
605  /// the Stomach as a parameter token. Reading the injected text with all three
606  /// neutralized preserves the value BibTeX actually parsed, **without altering
607  /// a byte of it**.
608  ///
609  /// **`_` is deliberately NOT in this set**, and the reason is the boundary
610  /// between the two treatments. A catcode is decided at tokenization, before
611  /// anything knows whether it is inside `$…$` — and a subscript in a `.bib`
612  /// title's math (`title = {Bounds on $x_1+x_2$}`) is *legitimate TeX* that
613  /// must keep working. `_` therefore belongs to treatment 2
614  /// (`bibtex.rs::escape_bib_data_specials`), which walks the value and skips
615  /// math spans. Measured: putting `_` here silently flattened every
616  /// subscript in a bibliography title. The other three have no legitimate
617  /// meaning inside a `.bib` field, in math or out.
618  ///
619  /// A `\catcode` in the injected text cannot do this job either: the catcode
620  /// would still be a State assignment, so a raw `.sty` opened from inside a
621  /// field handler would inherit it — and so would the document. Scoping to the
622  /// Mouth keeps the rule attached to the *text that BibTeX lexed*, which is
623  /// exactly where it belongs.
624  ///
625  /// Only the TeX-special meaning is removed: a character that has been given
626  /// some other catcode (LETTER, say) keeps it. And `\%`, `\&`, `\#` still
627  /// work, because the backslash is untouched.
628  pub fn with_bib_data_literals(mut self) -> Self {
629    self.bib_data_literals = true;
630    self
631  }

646  fn open_file(&mut self, pathname: &str) -> Result<()> {
647    if self.foodtype == FoodType::File {
648      // Perl: check readable, then check binary (non-empty), then open
649      let metadata = std::fs::metadata(pathname);
650      match &metadata {
651        Err(e) if e.kind() == io::ErrorKind::NotFound => {
652          // Perl Mouth/file.pm:32-41 `openFile`: an unreadable/unopenable file
653          // is a recoverable `Error('I/O', …)` and the mouth stays empty —
654          // callers pre-check with FindFile, so this is a race or a stale
655          // path, not a reason to abort the document. (It was a `fatal!`
656          // here, which since 56fc logs at the raise: a probing `\openin`
657          // then surfaced `Fatal:Mouth:MissingFile` — s108, four docs.)
658          Error!(
659            "I/O",
660            pathname,
661            s!("Can't find file {}", pathname),
662            "Reading it as empty"
663          );
664          return Ok(());
665        },
666        Err(e) if e.kind() == io::ErrorKind::PermissionDenied => {
667          Error!(
668            "I/O",
669            "unreadable",
670            s!("File {} is not readable. Ignoring.", pathname),
671            "",
672            "",
673            self.get_location()
674          );
675          return Ok(());
676        },
677        Err(e) => {
678          return Err(io::Error::new(e.kind(), e.to_string()).into());
679        },
680        Ok(meta) => {
681          // Every opened source file raises the runaway-token backstop in
682          // proportion (see `gullet::scale_token_limit_to_source`).
683          crate::gullet::scale_token_limit_to_source(meta.len() as usize);
684          // Check for binary file (non-empty and appears binary)
685          // Perl's -B heuristic: check first block for high proportion of non-text bytes
686          if meta.len() > 0
687            && let Ok(mut f) = File::open(pathname)
688          {
689            let mut buf = [0u8; 512];
690            if let Ok(n) = f.read(&mut buf)
691              && n > 0
692            {
693              let non_text = buf[..n]
694                .iter()
695                .filter(|&&b| {
696                  b == 0 || (b < 0x20 && b != b'\n' && b != b'\r' && b != b'\t' && b != 0x1b)
697                })
698                .count();
699              if non_text * 3 > n {
700                // High ratio of non-text bytes — likely binary
701                Error!(
702                  "invalid",
703                  "binary",
704                  s!("Input file {} appears to be binary. Ignoring.", pathname),
705                  "",
706                  "",
707                  self.get_location()
708                );
709                return Ok(());
710              }
711            }
712          }
713        },
714      }
715      let f = match File::open(pathname) {
716        Ok(f) => f,
717        Err(e) => {
718          Error!(
719            "I/O",
720            "open",
721            s!("Can't open {} for reading: {}", pathname, e),
722            "",
723            "",
724            self.get_location()
725          );
726          return Err(e.into());
727        },
728      };
729      let reader = BufReader::new(f);
730      self.reader = Some(reader);
731      self.reads_input_bytes = true;
732      self.buffer = VecDeque::new();
733      self.raw_buffer = VecDeque::new();
734    }
735    Ok(())
736  }

862  /// Split raw bytes into lines without decoding, splitting on \r\n, \r, or \n.
863  fn split_raw_lines(bytes: &[u8]) -> VecDeque<Vec<u8>> {
864    let mut lines = VecDeque::new();
865    let mut start = 0;
866    let mut i = 0;
867    while i < bytes.len() {
868      if bytes[i] == b'\r' {
869        lines.push_back(bytes[start..i].to_vec());
870        if i + 1 < bytes.len() && bytes[i + 1] == b'\n' {
871          i += 1; // skip \n after \r
872        }
873        start = i + 1;
874      } else if bytes[i] == b'\n' {
875        lines.push_back(bytes[start..i].to_vec());
876        start = i + 1;
877      }
878      i += 1;
879    }
880    // Add remaining bytes (last line without trailing newline)
881    if start < bytes.len() {
882      lines.push_back(bytes[start..].to_vec());
883    }
884    lines
885  }

887  /// Decode a raw byte line using the current encoding setting.
888  /// Matches Perl's per-line decode behavior. The flag says whether the line's
889  /// characters are the Latin-1 image of its bytes (a byte encoding, or a line
890  /// that is not UTF-8) rather than UTF-8-decoded characters.
891  fn decode_bytes(raw_line: &[u8], location: impl FnOnce() -> String) -> (String, bool) {
892    if let Some(ref encoding_sym) = get_input_encoding() {
893      // Probe the encoding without allocating — this fires per input
894      // line, so even a small heap alloc per call adds up on large
895      // documents. Only resolve the symbol to an owned String when we
896      // actually need it for the misdefined-encoding Info! message.
897      // `bytes` is the pdfTeX byte mouth (KERNEL_CAPABILITIES K10): pdfTeX
898      // reads a file byte by byte (tex.web §30-31 `buffer: array of
899      // ASCII_code`, §341 `cur_chr:=buffer[loc]`) against a 256-entry
900      // `\catcode` table, so a UTF-8 syllable is three char tokens whose codes
901      // are its bytes — the model cjkutf8-josa.sty:176 (`\DeclareRobustCommand*
902      // \^^ea[2]`, a control symbol over the lead byte) and dhucs.sty:44
903      // (`\ifx 가가`, two different bytes) are built on. Byte→char over
904      // U+0000..U+00FF is exactly the Latin-1 decode, and the catcode table's
905      // U+0000..U+00FF entries ARE that 256-entry table (`\catcode"EA` and a
906      // `^^ea` both address U+00EA). Enabled package-scoped by
907      // `inputenc_sty::enable_pdftex_byte_mouth`, never globally.
908      let is_latin1 = crate::common::arena::with(*encoding_sym, |s| {
909        s.eq_ignore_ascii_case("iso-8859-1")
910          || s.eq_ignore_ascii_case("latin1")
911          || s.eq_ignore_ascii_case("latin-1")
912          || s.eq_ignore_ascii_case("bytes")
913      });
914      let file_str = if is_latin1 {
915        raw_line.iter().map(|&b| b as char).collect::<String>()
916      } else {
917        // Fast path for valid UTF-8 (overwhelming majority of TeX
918        // source under `inputenc[utf8]`). `str::from_utf8` validates
919        // the whole slice in a tight SIMD loop and returns a
920        // borrow on success — no `Utf8Chunks::next` iteration
921        // looking for invalid bytes. Fall back to `from_utf8_lossy`
922        // only when validation fails.
923        match str::from_utf8(raw_line) {
924          Ok(s) => s.to_string(),
925          Err(_) => String::from_utf8_lossy(raw_line).into_owned(),
926        }
927      };
928      // Replace the U+FFFD inserted by lossy decode with space. For
929      // valid-UTF-8 inputs (no FFFD), skip the replace+log scan
930      // entirely. The original logic compared `replaced.len()` to
931      // `file_str.len()` to detect FFFD presence indirectly; the
932      // explicit `contains` is cheaper and lets us avoid the
933      // unconditional `replace` walk on every input line.
934      let has_fffd = file_str.contains('\u{FFFD}');
935      let file_str = if has_fffd {
936        let encoding_name = crate::common::arena::to_string(*encoding_sym);
937        Info!(
938          "misdefined",
939          &encoding_name,
940          s!("input isn't valid under encoding {}", &encoding_name),
941          "",
942          "",
943          location()
944        );
945        file_str.replace('\u{FFFD}', " ")
946      } else {
947        file_str
948      };
949      (file_str, is_latin1)
950    } else {
951      // No encoding set — interpret as UTF-8, falling back to a Latin-1
952      // passthrough for non-UTF-8 bytes. This happens after inputenc
953      // disables PERL_INPUT_ENCODING and the remaining file lines contain
954      // high bytes. (`decode_input_bytes` for one line.)
955      //
956      // The decode stays the Latin-1 passthrough (OXIDIZED_DESIGN_DIVERGENCES
957      // #481: the bytes are kept as text, where Perl blanks them), but the
958      // report is Perl's: with no `inputenc` declaring a non-UTF-8 encoding,
959      // the file is read as UTF-8 and an undecodable line raises
960      // `Info:misdefined` "input isn't valid under encoding utf-8"
961      // (Core/Mouth/file.pm:72-77). An Info, as in Perl; pdflatex's own
962      // "Invalid UTF-8 byte" Error (utf8.def:113) is not raised here.
963      match str::from_utf8(raw_line) {
964        Ok(s) => (s.to_string(), false),
965        Err(_) => {
966          let declared = lookup_string("lx@inputenc@encoding");
967          if declared.is_empty() || declared.starts_with("utf8") {
968            let encoding_name = lookup_string("PERL_INPUT_ENCODING");
969            let encoding_name = if encoding_name.is_empty() {
970              s!("utf-8")
971            } else {
972              encoding_name
973            };
974            Info!(
975              "misdefined",
976              &encoding_name,
977              s!("input isn't valid under encoding {}", &encoding_name),
978              "",
979              "",
980              location()
981            );
982          }
983          (raw_line.iter().map(|&b| b as char).collect(), true)
984        },
985      }
986    }
987  }

989  /// Original LaTeXML:
990  /// This is (hopefully) a correct way to split a line into "chars",
991  /// or what is probably more desired is "Grapheme clusters" (even "extended")
992  /// These are unicode characters that include any following combining chars, accents & such.
993  /// I am thinking that when we deal with unicode this may be the most correct way?
994  /// If it's not the way XeTeX does it, perhaps, it must be that ALL combining chars
995  /// have to be converted to the proper accent control sequences!
996  fn get_next_line(&mut self) -> Option<String> {
997    if self.buffer.is_empty() && !self.raw_buffer.is_empty() {
998      // Decode the next raw byte line lazily using the current encoding.
999      // This matches Perl's approach: each line is decoded with the encoding
1000      // that is active at the time the line is read, allowing inputenc to
1001      // change encoding mid-file.
1002      if let Some(raw_line) = self.raw_buffer.pop_front() {
1003        let (decoded, is_bytes) = Mouth::decode_bytes(&raw_line, || self.get_location());
1004        self.buffered_line_is_bytes = is_bytes;
1005        self.buffer.push_back(decoded);
1006      }
1007    }
1008    if self.buffer.is_empty()
1009      && let Some(ref mut reader) = self.reader
1010    {
1011      // file mouth case — read all bytes, split into raw lines, decode lazily
1012      let mut file_bytes = Vec::new();
1013      let _num_bytes = match reader.read_to_end(&mut file_bytes) {
1014        Ok(count) => count,
1015        Err(e) => {
1016          let message = s!("BufReader::read_to_end returned an error: {:?}", e);
1017          Warn!("mouth", "io", message, "", "", self.get_location());
1018          0
1019        },
1020      };
1021      // remove the now exhausted reader
1022      self.reader.take();
1023      // Split raw bytes into lines without decoding (preserving raw bytes).
1024      // Each line is decoded lazily via decode_bytes() using the CURRENT encoding.
1025      self.raw_buffer = Mouth::split_raw_lines(&file_bytes);
1026      // Decode the first line now
1027      if let Some(raw_line) = self.raw_buffer.pop_front() {
1028        let (decoded, is_bytes) = Mouth::decode_bytes(&raw_line, || self.get_location());
1029        self.buffered_line_is_bytes = is_bytes;
1030        self.buffer.push_back(decoded);
1031      }
1032    }
1033    // A raw line is decoded into an empty `buffer` and popped here at once, so
1034    // the flag set by the decode belongs to this line.
1035    self.line_is_bytes = std::mem::take(&mut self.buffered_line_is_bytes);
1036    self.caret_bytes.clear();
1037    self.buffer.pop_front()
1038  }

1040  /// Get the next character & it's catcode from the current line of input, even ignored chars,
1041  /// handling TeX's "^^" encoding.
1042  /// Note that this is the only place where catcode lookup is done (well almost),
1043  /// and that it is somewhat `inlined'.
1044  fn get_next_char(&mut self) -> Option<(char, Catcode)> {
1045    if self.colno >= self.nchars {
1046      return None;
1047    };
1048    let ch_opt = self.chars.get(self.colno);
1049    self.colno += 1;
1050    if let Some(ch) = ch_opt {
1051      let mut ch = *ch;
1052      let mut cc = self.catcode_of(ch);
1053      // Possible convert ^^x
1054      // Perl: (cc == CC_SUPER) && (colno + 1 < nchars) && (ch == chars[colno])
1055      if cc == Catcode::SUPER
1056        && self.colno + 1 < self.nchars
1057        && Some(&ch) == self.chars.get(self.colno)
1058      {
1059        // XeTeX/LuaTeX extended caret notation, longest-match-first:
1060        // ^^^^^^hhhhhh (6 hex) and ^^^^hhhh (4 hex) produce one Unicode
1061        // scalar. This engine is Unicode-native (same precedent as
1062        // providing \Ucharcat despite pdfTeX lacking it), and packages
1063        // PROBE for a Unicode engine with exactly this notation —
1064        // newunicodechar.sty L52-56 `\edef\next{\@gobble^^^^0021}` takes
1065        // its broken 8-bit branch without it (9-doc corpus cluster,
1066        // witnesses eigo, verifica, tikz-trackschematic, uspace).
1067        let is_lowerhex_c = |c: char| -> bool { matches!(c, '0'..='9' | 'a'..='f') };
1068        for (extra_carets, ndigits) in [(4usize, 6usize), (2, 4)] {
1069          // self.colno-1 is the first ^; require extra ^s then ndigits hex.
1070          let carets_ok =
1071            (0..extra_carets).all(|k| self.chars.get(self.colno + 1 + k) == Some(&ch));
1072          let dig_start = self.colno + 1 + extra_carets;
1073          if carets_ok
1074            && dig_start + ndigits <= self.nchars
1075            && (0..ndigits).all(|k| {
1076              self
1077                .chars
1078                .get(dig_start + k)
1079                .is_some_and(|c| is_lowerhex_c(*c))
1080            })
1081          {
1082            let hex: String = (0..ndigits)
1083              .filter_map(|k| self.chars.get(dig_start + k))
1084              .collect();
1085            if let Some(newch) = u32::from_str_radix(&hex, 16).ok().and_then(char::from_u32) {
1086              let removed = 2 + extra_carets + ndigits;
1087              self.splice(self.colno - 1..self.colno - 1 + removed, &[newch]);
1088              self.nchars -= removed - 1;
1089              let cc2 = self.catcode_of(newch);
1090              return Some((newch, cc2));
1091            }
1092          }
1093        }
1094        let c1_opt = self.chars.get(self.colno + 1);
1095        let c2_opt = self.chars.get(self.colno + 2);
1096        let mut two_hex = false;
1097        // ^^ followed by TWO LOWERCASE Hex digits???
1098        if let Some(c1) = c1_opt
1099          && let Some(c2) = c2_opt
1100        {
1101          // Perf: avoid per-char String alloc + regex match by using
1102          // direct ASCII class check. LOWERHEX_REGEX = ^[0-9a-f]$, i.e.
1103          // lowercase hex digits only.
1104          let is_lowerhex = |c: char| -> bool { matches!(c, '0'..='9' | 'a'..='f') };
1105          if (self.colno + 2 < self.nchars) && is_lowerhex(*c1) && is_lowerhex(*c2) {
1106            // TODO: Maybe Result type warranted here?
1107            let hex = u8::from_str_radix(&s!("{}{}", c1, c2), 16).unwrap();
1108            ch = hex as char;
1109            self.splice(self.colno - 1..self.colno + 3, &[ch]);
1110            self.nchars -= 3;
1111            two_hex = true;
1112          }
1113        }
1114        if !two_hex {
1115          // OR ^^ followed by a SINGLE Control char type code???
1116          let c = self.chars[self.colno + 1];
1117          let cn = c as i16;
1118
1119          ch = (cn + if cn >= 64 { -64 } else { 64 }) as u8 as char;
1120          self.splice(self.colno - 1..self.colno + 2, &[ch]);
1121          self.nchars -= 2;
1122        }
1123        cc = self.catcode_of(ch);
1124        self.caret_bytes.push(self.colno - 1);
1125      }
1126      // Only an input byte: a character of a byte-decoded line, or a `^^xx` /
1127      // `^^X` code (tex.web §355 — a byte even on a UTF-8 line).
1128      if self.reads_input_bytes
1129        && ('\u{80}'..='\u{FF}').contains(&ch)
1130        && (self.line_is_bytes || self.caret_bytes.contains(&(self.colno - 1)))
1131        && matches!(cc, Catcode::LETTER | Catcode::OTHER)
1132        && let Some(decoded) = eight_bit_input_char(ch)
1133      {
1134        return Some((decoded, cc));
1135      }
1136      Some((ch, cc))
1137    } else {
1138      None
1139    }
1140  }

1186  /// Read the next token, or undef if exhausted.
1187  /// Note that this also returns COMMENT tokens containing source comments,
1188  /// and also locator comments (file, line# info).
1189  /// LaTeXML::Core::Gullet intercepts them and passes them on at appropriate times.
1190  pub fn read_token(&mut self) -> Option<Token> {
1191    loop {
1192      // Iterate till we find a token, or run out. (use return)
1193      // ===== Get next line, if we need to.
1194      if self.colno >= self.nchars {
1195        self.lineno += 1;
1196        self.colno = 0;
1197        let line_opt = self.get_next_line();
1198        // For \read, we have to return something for EOL, and handle implicit final newline
1199        let read_mode = lookup_int("PRESERVE_NEWLINES") > 1;
1200        let eolch = endline_char();
1201        if line_opt.is_none() {
1202          // Exhausted the input.
1203          let eolcc = if let Some(ch) = eolch {
1204            lookup_catcode(ch).unwrap_or(Catcode::OTHER)
1205          } else {
1206            Catcode::OTHER
1207          };
1208          let eoftoken = if let Some(eolch_content) = eolch {
1209            if read_mode && !self.at_eof && !self.source.is_empty() {
1210              // The synthetic final line is empty + `\endlinechar`, read in
1211              // state N like any other line (tex.web §345-349): an EOL char
1212              // is `\par`, a SPACE or IGNORE char is dropped (§349 "goto
1213              // switch" — it can never become a token), anything else is a
1214              // char token. Perl Mouth.pm:303-307 emits the IGNORE char and
1215              // it reaches the Stomach as `misdefined` (KNOWN_PERL_ERRORS
1216              // #125); witness liftarm (pgfmanual `codeexample` sets
1217              // `\catcode`\^^M=9` around `\scantokens`, and animate.sty's
1218              // `\@anim@buildtmln` `\read`s the timeline to EOF inside it —
1219              // 501 errors capped).
1220              match eolcc {
1221                Catcode::EOL => Some(T_CS!("\\par")),
1222                Catcode::SPACE | Catcode::IGNORE => None,
1223                _ => Some(CharToken!(eolch_content, eolcc)),
1224              }
1225            } else {
1226              None
1227            }
1228          } else {
1229            None
1230          };
1231          self.at_eof = true;
1232          self.chars = VecDeque::new();
1233          self.nchars = 0;
1234          return eoftoken;
1235        }
1236        // Remove trailing spaces from external sources
1237        let mut line = line_opt.unwrap();
1238        if !self.source.is_empty() && line.ends_with(' ') {
1239          line = TRAILING_SPACE_CHARS.replace(&line, "").to_string();
1240        }
1241        // Then append the appropriate \endlinechar, or "\r";
1242        if let Some(ch) = eolch {
1243          line.push(ch);
1244        }
1245
1246        self.chars = line.chars().collect::<VecDeque<char>>();
1247        self.nchars = self.chars.len();
1248        // In state N, skip leading spaces & ignored, possibly decoding (trailing space removed
1249        // above)
1250        while let Some((_ch, cc)) = self.get_next_char() {
1251          match cc {
1252            Catcode::SPACE | Catcode::IGNORE => {},
1253            Catcode::EOL => {
1254              // Eolch already? empty line!
1255              self.colno = self.nchars; // ignore rest of line.
1256              return Some(T_CS!("\\par"));
1257            },
1258            _ => break,
1259          }
1260        }
1261        if self.nchars == 0 || self.colno > self.nchars {
1262          // Past end of line?
1263          // If upcoming line is empty, and there is no recognizable EOL, fake one
1264          if read_mode && eolch != Some('\r') {
1265            return Some(T_MARKER!("EOL"));
1266          }
1267        } else {
1268          // Back up over peeked char
1269          self.colno -= 1;
1270        }
1271        // Sneak a comment out, every so often.
1272        if self.lineno.is_multiple_of(READLINE_PROGRESS_QUANTUM) && lookup_bool("INCLUDE_COMMENTS")
1273        {
1274          // Perl T_COMMENT prepends '%' (Token.pm L81)
1275          return Some(T_COMMENT!(s!(
1276            "%**** {} Line {} ****",
1277            &self.shortsource,
1278            &self.lineno.to_string()
1279          )));
1280        }
1281      }
1282      // In state::S, skip spaces — and ignored characters, which leave the state as it is (tex.web §344
1283      // `any_state_plus(ignore)`): under `\ExplSyntaxOn` the (ignored) space of `\foo ~` kept TeX skipping, so the
1284      // `~` (a spacer) is skipped too, where stopping at the ignored character made it a space token
1285      // (`\str_if_eq:nnTF { \foo ~ } { \foo }` false). Perl Mouth.pm skips CC_SPACE only (KNOWN_PERL_ERRORS #1177).
1286      if self.skipping_spaces {
1287        let mut cc = None;
1288        // This is very awkward as a loop,
1289        //  but I had to port the Perl logic without going crazy...
1290        // tokenizer/verb.tex depends on it.
1291        while let Some((_, ncc)) = self.get_next_char() {
1292          cc = Some(ncc);
1293          if !matches!(ncc, Catcode::SPACE | Catcode::IGNORE) {
1294            break;
1295          }
1296        }
1297        if self.colno <= self.nchars
1298          && cc.is_some()
1299          && !matches!(cc, Some(Catcode::SPACE | Catcode::IGNORE))
1300        {
1301          self.colno -= 1;
1302        }
1303        if cc == Some(Catcode::EOL) {
1304          // If we've got an EOL
1305          self.get_next_char();
1306          if self.colno < self.nchars {
1307            self.colno -= 1;
1308          }
1309        }
1310        self.skipping_spaces = false;
1311      }
1312      // ==== Extract next token from line.
1313      // §1 (docs/performance/SOURCE_PROVENANCE.md): record the token's source start now —
1314      // after all inter-token skips (line fetch, leading / skipping spaces) and
1315      // before `get_next_char` advances past its first char. `colno` is 0-indexed
1316      // here; the +1 to 1-indexed columns happens in `get_locator`.
1317      self.last_token_start = (self.lineno, self.colno);
1318      if let Some((ch, cc)) = self.get_next_char() {
1319        #[cfg(not(feature = "token-locators"))]
1320        if let Some(token) = Mouth::dispatch_char(self, ch, cc) {
1321          return Some(token);
1322        } // Else, repeat till we get something or run out.
1323        // token-locators: stamp the token with an origin handle into the side
1324        // arena, using `last_token_start` (the token's first char — captured
1325        // above, before `dispatch_char` reads the rest, e.g. a CS name). This is
1326        // what survives expansion to digestion (Experiments 1–3 showed the mouth
1327        // position at digest time cannot recover it). See SOURCE_PROVENANCE §3.1.1.
1328        #[cfg(feature = "token-locators")]
1329        if let Some(mut token) = Mouth::dispatch_char(self, ch, cc) {
1330          let (line, col0) = self.last_token_start;
1331          token.loc =
1332            crate::token::push_token_origin(self.source_sym, line as u32, (col0 + 1) as u32);
1333          return Some(token);
1334        } // Else, repeat till we get something or run out.
1335      }
1336    }
1337  }

1341  /// Returns an empty Tokens list, if there is no input
1342  pub fn read_tokens(&mut self) -> Tokens {
1343    // Pre-size to skip the early doubling reallocations of the per-token push
1344    // loop below (a `grow_one` site in the allocation profile); 16 covers a
1345    // typical line/group in one allocation.
1346    let mut tokens = Vec::with_capacity(16);
1347    while let Some(token) = self.read_token() {
1348      tokens.push(token);
1349    }
1350    while let Some(Token { code: Catcode::SPACE, .. }) = tokens.last() {
1351      // Remove trailing space
1352      tokens.pop();
1353    }
1354    Tokens::new(tokens)
1355  }

1424  /// The source text this mouth will read next — the rest of the current line, then up to `max_lines` lines after it,
1425  /// joined by `\n`, at most `max_chars` characters in all — without reading it: nothing is consumed, tokenized or
1426  /// decoded in place. A look-ahead for a decision about what follows (stomach.rs `formula_continues_past_par`,
1427  /// OXIDIZED_DESIGN #553); `max_chars` keeps its cost bounded on a source whose current line is the whole document.
1428  /// Lines not yet decoded are read as UTF-8, lossily; their structure (`$`, `\int`, `_`) is all a look-ahead needs.
1429  /// A current line read to its end contributes nothing: its end-of-line character is read already, and an empty line
1430  /// in its place would tokenize as one more (a `\par`, or under `\obeylines` an active `^^M`; hep-ph9809328).
1431  pub fn peek_text(&self, max_lines: usize, max_chars: usize) -> String {
1432    // Whether a line is begun in `text` that the next must be joined to.
1433    let mut joined = self.colno < self.nchars;
1434    let mut text = String::new();
1435    let mut room = max_chars;
1436    for ch in self.chars.iter().skip(self.colno).take(max_chars) {
1437      text.push(*ch);
1438      room -= 1;
1439    }
1440    if room == 0 {
1441      return text;
1442    }
1443    if let Some(eol) = endline_char()
1444      && text.ends_with(eol)
1445    {
1446      text.pop();
1447    }
1448    let decoded = self
1449      .buffer
1450      .iter()
1451      .map(|line| std::borrow::Cow::Borrowed(line.as_str()));
1452    let raw = self
1453      .raw_buffer
1454      .iter()
1455      .map(|line| String::from_utf8_lossy(line));
1456    for line in decoded.chain(raw).take(max_lines) {
1457      if room == 0 {
1458        break;
1459      }
1460      if joined {
1461        text.push('\n');
1462      }
1463      joined = true;
1464      let mut count = 0;
1465      text.extend(line.chars().take(room).inspect(|_| count += 1));
1466      room = room.saturating_sub(count + 1);
1467    }
1468    text
1469  }

1471  pub fn read_raw_line(&mut self, noread: bool) -> Option<String> {
1472    let mut line = String::new();
1473    if self.colno < self.nchars {
1474      line = self.chars.iter().skip(self.colno).collect();
1475      // Strip the final carriage return, if it has been added back (Perl: s/\r$//s)
1476      if line.ends_with('\r') {
1477        line.pop();
1478      }
1479      self.colno = self.nchars;
1480    } else if !noread {
1481      match self.get_next_line() {
1482        None => {
1483          // We've exhausted this mouth
1484          self.at_eof = true;
1485          self.chars = VecDeque::new();
1486          self.nchars = 0;
1487          self.colno = 0;
1488          return None;
1489        },
1490        Some(next_line) => {
1491          // Strip trailing spaces (Perl: s/ *$//s)
1492          line = next_line.trim_end_matches(' ').to_string();
1493          self.lineno += 1;
1494          self.chars = line.chars().collect();
1495          self.nchars = self.chars.len();
1496          self.colno = self.nchars;
1497        },
1498      }
1499    }
1500    Some(line)
1501  }

1503  fn dispatch_char(&mut self, ch: char, cc: Catcode) -> Option<Token> {
1504    // Possibly want to think about caching (common) letters, etc to keep from
1505    // creating tokens like crazy... or making them more compact... or ???
1506    use crate::token::Catcode::*;
1507    match cc {
1508      ESCAPE => self.handle_escape(), // T_ESCAPE
1509      BEGIN => {
1510        if ch == '{' {
1511          Some(T_BEGIN!())
1512        } else {
1513          Some(CharToken!(ch, BEGIN))
1514        }
1515      },
1516      END => {
1517        if ch == '}' {
1518          Some(T_END!())
1519        } else {
1520          Some(CharToken!(ch, END))
1521        }
1522      },
1523      MATH => {
1524        if ch == '$' {
1525          Some(T_MATH!())
1526        } else {
1527          Some(CharToken!(ch, MATH))
1528        }
1529      },
1530      ALIGN => {
1531        if ch == '&' {
1532          Some(T_ALIGN!())
1533        } else {
1534          Some(CharToken!(ch, ALIGN))
1535        }
1536      },
1537      EOL => Some(self.handle_end_of_line()),
1538      PARAM => {
1539        if ch == '#' {
1540          Some(T_PARAM!())
1541        } else {
1542          Some(CharToken!(ch, PARAM))
1543        }
1544      }, // T_PARAM
1545      SUPER => {
1546        if ch == '^' {
1547          Some(T_SUPER!())
1548        } else {
1549          Some(CharToken!(ch, SUPER))
1550        }
1551      }, // T_SUPER
1552      SUB => {
1553        if ch == '_' {
1554          Some(T_SUB!())
1555        } else {
1556          Some(CharToken!(ch, SUB))
1557        }
1558      }, // T_SUB
1559      SPACE => self.handle_space(),
1560      LETTER => Some(CharToken!(ch, LETTER)),
1561      OTHER => Some(CharToken!(ch, OTHER)),
1562      ACTIVE => Some(T_ACTIVE!(ch)),
1563      COMMENT => self.handle_comment(),
1564      INVALID => Some(CharToken!(ch, OTHER)), // T_INVALID (we could get unicode!)
1565      _ => None,                              // IGNORE, others
1566    }
1567  }

1578  fn handle_space(&mut self) -> Option<Token> {
1579    // Skip any following spaces!
1580    while let Some((_ch, cc)) = self.get_next_char() {
1581      if (cc != Catcode::SPACE) && (cc != Catcode::EOL) {
1582        // backup at nonspace/eol
1583        if self.colno <= self.nchars {
1584          self.colno -= 1;
1585        }
1586        break;
1587      }
1588    }
1589    Some(T_SPACE!())
1590  }

1592  fn handle_comment(&mut self) -> Option<Token> {
1593    let n = self.colno;
1594    self.colno = self.nchars;
1595    let mut comment = String::new();
1596    for c in self.chars.iter().skip(n).take(self.nchars - n) {
1597      comment.push(*c);
1598    }
1599    let trimmed_comment = comment.trim();

1602      Some(T_COMMENT!(s!("%{}", trimmed_comment)))
1603    } else if lookup_int("PRESERVE_NEWLINES") > 1 {
1604      Some(T_MARKER!("EOL")) // Required EOL during \read
1605    } else {
1606      None
1607    }
1608  }

1614  /// Read control sequence
1615  fn handle_escape(&mut self) -> Option<Token> {
1616    // NOTE: We're using control sequences WITH the \ prepended!!!
1617    if let Some((ch, mut cc)) = self.get_next_char() {
1618      // Knuth, p.46 says that Newlines are converted to spaces,
1619      // Bit I believe that he does NOT mean within control sequences
1620      let mut cs = s!("\\{}", ch);
1621      if cc == Catcode::LETTER {
1622        // For letter, read more letters for csname.
1623        while let Some((nch, ncc)) = self.get_next_char() {
1624          cc = ncc;
1625          if ncc == Catcode::LETTER {
1626            cs.push(nch);
1627          } else {
1628            break;
1629          }
1630        }
1631        // We WILL skip spaces, but not till next token is read (in case catcode changes!!!!)
1632        self.skipping_spaces = true;
1633        if cc != Catcode::LETTER {
1634          self.colno -= 1;
1635        }
1636      }
1637      Some(T_CS!(cs))
1638    } else {
1639      None
1640    }

1666  pub fn is_eol(&mut self) -> bool {
1667    let savecolno = self.colno;
1668    // We have to peek past any ignored tokens & also spaces, if skipping
1669    let mut cc = None;
1670    while let Some((_, ncc)) = self.get_next_char() {
1671      if ncc != Catcode::IGNORE && (!self.skipping_spaces || ncc != Catcode::SPACE) {
1672        cc = Some(ncc);
1673        break;
1674      }
1675    }
1676    if self.colno <= self.nchars && cc.is_some() {
1677      // Back-up if too far.
1678      self.colno -= 1;
1679    }
1680    // If skipping spaces (really, reading for input (\endinput) ?), jump to end of EOL or comments
1681    if self.skipping_spaces && (cc == Some(Catcode::EOL) || cc == Some(Catcode::COMMENT)) {
1682      // If we've got an EOL | COMMENT
1683      self.colno = self.nchars
1684    }
1685    let eol = self.colno >= self.nchars;
1686    self.colno = savecolno;
1687    eol
1688  }

1710      from_line as u32,
1711      (from_col0 + 1) as u32,
1712      self.lineno as u32,
1713      (self.colno + 1) as u32,
1714    )
1715  }

1723/// Tokenize a string under the **standard** catcode table — Perl
1724/// `Package.pm:Tokenize` L1019-1023.
1725///
1726/// "Standard" is the document-level table: `@` is an ordinary letter-less
1727/// character, so this is how user-facing text should be read. The current
1728/// state's catcodes are deliberately NOT consulted; the table is swapped in for
1729/// the duration and restored afterwards, exactly as Perl's `local $STATE =
1730/// $STD_CATTABLE` does, so a document that has been playing with catcodes
1731/// cannot change what a binding's own string means.
1732///
1733/// See [`tokenize_internal`] for the `.sty`-style table that treats `@` as a
1734/// letter.
1735///
1736/// The argument is an `impl Into<`[`TeXString`]`>`, not a `&str`: a string
1737/// literal converts implicitly, but a `String` — the shape a control-word-welding
1738/// `Tokens::to_string()` arrives in — must declare itself via
1739/// [`Tokens::untex_string`] or [`TeXString::assembled`]. See the [`TeXString`]
1740/// docs for why.
1741pub fn tokenize(text: impl Into<TeXString>) -> Tokens {
1742  let text = text.into();
1743  // special case! empty input is empty Tokens
1744  if text.is_empty() {
1745    return NO_TOKENS;
1746  }
1747  use_std_state();
1748  let result = Mouth::new(text.as_str(), None).unwrap().read_tokens();
1749  use_main_state();
1750  result
1751}
1752/// Tokenize a string under the standard catcode table, reading `% & #` as
1753/// ordinary characters rather than comment, alignment tab and parameter.
1754///
1755/// For text that came out of the BibTeX lexer, which has none of those
1756/// constructs, so all three are data — treatment 1 of `OXIDIZED_DESIGN #74`,
1757/// see [`Mouth::with_bib_data_literals`] (including why `_` is not in the set).
1758/// Plain [`tokenize`] would let a `%` comment out the rest of the string, which
1759/// for a `.bib` field means losing its closing brace and leaving whatever it
1760/// opened unclosed, and would make the `&` in "Taylor & Francis" a stray
1761/// alignment tab.
1762///
1763/// This exists because the handlers that re-read a raw field — `\bib@@title`
1764/// recasing, name splitting, date/pages assembly — build their tokens from the
1765/// stored string and never pass through the per-entry mouth.
1766///
1767/// Takes an `impl Into<`[`TeXString`]`>` for the same reason as [`tokenize`] —
1768/// and it is the sink that most needs it: the bibliography is where the
1769/// control-word weld has surfaced three times (PR #399, PR #400, issue 410).
1770pub fn tokenize_bib_literal(text: impl Into<TeXString>) -> Tokens {
1771  let text = text.into();
1772  // special case! empty input is empty Tokens
1773  if text.is_empty() {
1774    return NO_TOKENS;
1775  }
1776  use_std_state();
1777  let result = Mouth::new(text.as_str(), None)
1778    .unwrap()
1779    .with_bib_data_literals()
1780    .read_tokens();
1781  use_main_state();
1782  result
1783}

1786/// `Package.pm:TokenizeInternal` L1026-1030.
1787///
1788/// Same swap-and-restore discipline as [`tokenize`], but with `@` a letter, so
1789/// internal control sequences (`\@ifnextchar`, `\@currentlabel`, …) tokenize as

1798    return NO_TOKENS;
1799  }
1800  use_sty_state();

1802  use_main_state();

1806/// The character `\endlinechar` ends each input line with (tex.web §362: none when it is outside 0..255).
1807fn endline_char() -> Option<char> {
1808  match lookup_definition(&T_CS!("\\endlinechar")).unwrap() {
1809    Some(defn) => {
1810      if defn.is_register()
1811        && let Some(eol) = defn.value_of(Vec::new())
1812      {
1813        let eol = eol.value_of() as i16;
1814        if eol > 0 && eol <= 255 {
1815          return Some((eol as u8) as char);
1816        }
1817      }
1818      None
1819    },
1820    _ => Some('\r'),
1821  }
1822}
