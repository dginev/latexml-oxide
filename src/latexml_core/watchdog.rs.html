1//! Wall-clock watchdog that forcibly aborts the process after a deadline.
2//!
3//! The existing `stomach::check_timeout()` is a cooperative mechanism — it only
4//! fires when the digestion loop polls it. That leaves tight native loops
5//! (Marpa precompute / parse, libxml2 post-processing, FFI calls into libxslt,
6//! ...) completely unguarded: a 60-second timeout can easily turn into 10
7//! minutes if control never returns to the digestion loop.
8//!
9//! This module provides a main-level `Watchdog` that spawns a dedicated thread
10//! at construction, wakes after the specified number of seconds, and — if the
11//! watchdog has not yet been cancelled — prints a message and exits the process
12//! ([`EXIT_TIMEOUT`]). The process dies within `timeout + poll_interval`,
13//! regardless of what the main thread is doing, plus the time one-time setup
14//! spent [`off_the_clock`] (the format build): only its own bound ends that setup.
15//! `--timeout 0` arms no time guard: no deadline, and no bound on the setup (a non-zero `--max-memory` still starts
16//! the watchdog thread for its memory guard).
17//!
18//! # Design notes
19//!
20//! - Uses `Arc<AtomicBool>` for cancellation. Polling every 100 ms keeps the cancellation latency
21//!   low without burning CPU.
22//! - `Drop` on the `Watchdog` handle cancels the watchdog thread, so RAII usage (`let _wd =
23//!   Watchdog::new(secs)`) is sufficient.
24//! - We exit ([`exit_now`]) rather than `panic!` because panic may unwind or be caught by a
25//!   surrounding `catch_unwind`, which would defeat the safety guarantee.
26//! - The existing cooperative `stomach::check_timeout()` path is retained: on most conversions it
27//!   fires before the hard abort, giving callers a nice `Err(Fatal)` with proper error propagation.
28//!   The watchdog is a safety net for the pathological cases where cooperative polling doesn't
29//!   happen.
30//!
31//! # Resource limits
32//!
33//! [`Watchdog::with_limits`] guards
34//! **both** a wall-clock deadline and a
35//! resident-memory ceiling — the two defenses any executable that converts
36//! arbitrary input needs. It is the shared guard reused by both
37//! `cortex_worker` (in-process, one paper per process) and the
38//! `latexml_oxide --server` LSP (run inside each forked body child, which
39//! self-terminates on breach so the parent reaps it via pipe EOF). The exit
40//! codes are distinct so a supervising parent can tell them apart:
41//! `124` = wall-clock timeout, `137` = memory ceiling.
42//!
43//! # Portability
44//!
45//! The wall-clock guard is portable (`std::thread` + `Instant`).
46//!
47//! **Portable:** [`total_memory_bytes`] and [`available_disk_bytes`] answer on
48//! Linux, macOS and Windows — `sysconf(_SC_PHYS_PAGES)`/`statvfs` are POSIX, and
49//! Windows uses `GlobalMemoryStatusEx`/`GetDiskFreeSpaceExW`. They back the
50//! machine-derived default ceiling ([`default_ceiling_mib`]) and the
51//! spill-headroom check, so those behave identically on every supported OS.
52//!
53//! **Enforcement, also portable:** [`process_rss_kb`] — the half that actually
54//! checks live usage against the ceiling — answers on all three: Linux samples
55//! `/proc/self/status`, macOS asks libproc (`proc_pidinfo`), and Windows uses
56//! `GetProcessMemoryInfo`. So the memory ceiling is both computed *and* checked
57//! on every supported OS; other targets fall back to `None` (time guard only).

68/// Current resident set size of this process in KiB, or `None` if it can't be
69/// determined. Linux reads `VmRSS` from `/proc/self/status`; macOS asks
70/// libproc (`proc_pidinfo`/`PROC_PIDTASKINFO`) — without it the whole
71/// `--max-memory` ceiling silently did not exist on macOS, and the
72/// `115_streaming_cli` Fatal-contract guard caught exactly that on macOS CI
73/// (an over-budget run exited 0). Cheap enough to poll a few times a second.
74#[cfg(target_os = "linux")]
75pub fn process_rss_kb() -> Option<u64> {
76  let status = std::fs::read_to_string("/proc/self/status").ok()?;
77  for line in status.lines() {
78    if let Some(rest) = line.strip_prefix("VmRSS:") {
79      return rest.split_whitespace().next()?.parse::<u64>().ok();
80    }
81  }
82  None
83}

85/// macOS: libproc task info; `pti_resident_size` is in BYTES.
86#[cfg(target_os = "macos")]
87pub fn process_rss_kb() -> Option<u64> {
88  let mut info: libc::proc_taskinfo = unsafe { std::mem::zeroed() };
89  let size = size_of::<libc::proc_taskinfo>() as libc::c_int;
90  // SAFETY: proc_pidinfo fills at most `size` bytes of the zeroed struct we
91  // own; a short or negative return is handled below.
92  let got = unsafe {
93    libc::proc_pidinfo(
94      std::process::id() as libc::c_int,
95      libc::PROC_PIDTASKINFO,
96      0,
97      (&raw mut info).cast::<libc::c_void>(),
98      size,
99    )
100  };
101  (got == size).then(|| info.pti_resident_size / 1024)
102}

104/// Windows: `GetProcessMemoryInfo`'s `WorkingSetSize` is this process's resident
105/// set — the RSS analog the cooperative ceiling needs. Without it the whole
106/// `--max-memory` ceiling silently did not exist on Windows (exactly as it did
107/// not on macOS before the libproc arm), and the `115_streaming_cli`
108/// Fatal-contract guard caught it: an over-budget run exited 0. `windows-sys` is
109/// already linked, and the `K32`-prefixed forwarder lives in kernel32, so this
110/// adds no new DLL import to the self-contained release binary.
111#[cfg(windows)]
112pub fn process_rss_kb() -> Option<u64> {
113  use windows_sys::Win32::System::{
114    ProcessStatus::{K32GetProcessMemoryInfo, PROCESS_MEMORY_COUNTERS},
115    Threading::GetCurrentProcess,
116  };
117  let mut counters: PROCESS_MEMORY_COUNTERS = unsafe { std::mem::zeroed() };
118  counters.cb = size_of::<PROCESS_MEMORY_COUNTERS>() as u32;
119  // SAFETY: `counters` is a correctly sized, zeroed struct with `cb` set, as the
120  // API requires; `GetCurrentProcess` returns a pseudo-handle needing no close.
121  if unsafe { K32GetProcessMemoryInfo(GetCurrentProcess(), &mut counters, counters.cb) } != 0 {
122    return Some(counters.WorkingSetSize as u64 / 1024);
123  }
124  None
125}

200/// The end-of-run memory report — emitted ONLY when a memory-budget Fatal
201/// fired during the run (user directive 2026-08-03: alert when needed, stay
202/// quiet on clean runs). `None` otherwise, or when no peak is measurable.
203///
204/// It reports the kernel-tracked peak and says the document needs MORE — never
205/// a specific sufficient figure, because none is knowable from a truncated
206/// run: the fuse clamped the peak at 75% of the ceiling, and the streaming
207/// spill watermark itself derives from the ceiling, so the true requirement
208/// can only be found by rerunning higher. A document's need scales with macro
209/// expansion and math density, not source bytes (the 131 MB math-dense
210/// witness needs ~23 GB resident just to stream through core), so this
211/// measured figure is the only honest lower bound there is.
212pub fn peak_memory_report() -> Option<String> {
213  if !MEMORY_FATAL_SEEN.load(Ordering::Relaxed) {
214    return None;
215  }
216  let peak_kb = process_peak_rss_kb()?;
217  Some(format!(
218    "peak memory {} MB was not enough: this document needs a higher --max-memory ceiling \
219     (0 disables the limit) and enough free RAM",
220    peak_kb / 1024,
221  ))
222}

224/// Total physical RAM on this machine in bytes, or `None` if it cannot be
225/// determined.
226///
227/// Portable by construction: `sysconf(_SC_PHYS_PAGES) * sysconf(_SC_PAGE_SIZE)`
228/// is POSIX and answers on both Linux and macOS, and Windows has
229/// `GlobalMemoryStatusEx`. No new dependency — `libc` and `windows-sys` are
230/// already in the tree.
231///
232/// Deliberately NOT `/proc/meminfo`: that would repeat the Linux-only mistake
233/// this module already carries in [`process_rss_kb`], which returns `None`
234/// everywhere else and so silently deactivates the memory ceiling on
235/// macOS/Windows.
236pub fn total_memory_bytes() -> Option<u64> {
237  #[cfg(unix)]
238  {
239    // SAFETY: `sysconf` is a pure query with no pointer arguments; a negative
240    // return means "unavailable", which we map to `None` rather than trusting.
241    let pages = unsafe { libc::sysconf(libc::_SC_PHYS_PAGES) };
242    // `sysconf(_SC_PAGESIZE)` itself: latexml_core enables rustix's `use-libc-auxv`, so `page_size` reads the libc
243    // auxiliary vector and cannot fail (the raw backend would read `/proc/self/auxv` before Linux 6.4 and panic
244    // without it).
245    let page_size = rustix::param::page_size();
246    if pages > 0 {
247      return (pages as u64).checked_mul(page_size as u64);
248    }
249    None
250  }
251  #[cfg(windows)]
252  {
253    // `MEMORYSTATUSEX` is what windows-sys calls this, mirroring the Win32
254    // header. This arm originally said `GLOBAL_MEMORY_STATUS_EX` — a spelling
255    // from a different generation of the bindings — while Cargo.toml pinned
256    // `windows-sys = "0.61"`, so it never matched the version it declared and
257    // has NEVER compiled. It survived because CI builds Linux and macOS only:
258    // the first Windows compile of this file was the 0.7.5-rc4 RELEASE
259    // workflow, on a tag, days after the code landed. See task #164.
260    use windows_sys::Win32::System::SystemInformation::{GlobalMemoryStatusEx, MEMORYSTATUSEX};
261    let mut status: MEMORYSTATUSEX = unsafe { std::mem::zeroed() };
262    status.dwLength = size_of::<MEMORYSTATUSEX>() as u32;
263    // SAFETY: `status` is a correctly sized, zeroed struct with `dwLength` set,
264    // exactly as the API requires.
265    if unsafe { GlobalMemoryStatusEx(&mut status) } != 0 {
266      return Some(status.ullTotalPhys);
267    }
268    None
269  }
270  #[cfg(not(any(unix, windows)))]
271  {
272    None
273  }
274}

276/// The default per-conversion memory ceiling in MiB, derived from the machine
277/// **as it is right now**.
278///
279/// `min(64 GiB, 90 % of AVAILABLE RAM at startup)`, never above the
280/// cgroup-capped total, floored at [`MIN_DEFAULT_CEILING_MIB`]. When
281/// availability cannot be probed the rule falls back to half of the
282/// (cgroup-capped) total, and to [`FALLBACK_CEILING_MIB`] when nothing can be
283/// probed at all. One sentence for users: *a conversion may use most of the
284/// RAM that is actually free when it starts, never more than 64 GiB.*
285///
286/// History of the rule (each step measured, none guessed):
287/// - flat 6144 MiB — absurd on a 256 GB host, over-generous on a laptop;
288/// - 90 % of TOTAL RAM — laptop-hostile: blind to what the session already
289///   uses, it let one conversion push a busy 16 GB machine deep into swap;
290/// - half of TOTAL RAM — laptop-safe but wasteful the other way: on the 31 GB
291///   witness laptop it derives ~15.5 GiB while the machine sits idle with
292///   ~28 GB free, and the 131 MB witness (24.2 GB core peak) then dies at a
293///   fuse it did not need to meet (user directive 2026-08-01: "we can
294///   comfortably use up to 24 GB here");
295/// - 90 % of AVAILABLE — self-adjusting: on that idle laptop it derives
296///   ~25 GiB, and on the same laptop with a browser session holding 6 GB it
297///   derives roughly what the half-of-total rule chose. `MemAvailable` is the
298///   kernel's own estimate of what is reclaimable without swapping, so "the
299///   session starts swapping long before the guard fires" — the failure that
300///   killed the 90 %-of-total rule — cannot recur by construction.
301///
302/// The **64 GiB cap** is not about this machine but about the *others*: in a
303/// parallel fleet the aggregate is `N_processes x ceiling`, so an uncapped
304/// fraction-of-RAM rule on a big host would let a busy fleet OOM it. The
305/// `cortex_worker` fleet overrides this with its own per-child ceiling anyway;
306/// the cap keeps the single-process default from being reckless.
307///
308/// Startup-derived, deliberately: the figure is resolved once and becomes the
309/// run's `--max-memory`, so a neighbours' later allocations cannot move a
310/// running conversion's fuse mid-flight.
311pub fn default_ceiling_mib() -> u64 {
312  derive_ceiling_mib(available_memory_bytes(), effective_memory_bytes())
313}
314
315/// The pure arithmetic behind [`default_ceiling_mib`], split out so the rule
316/// is unit-testable without controlling the test host's actual memory state.
317fn derive_ceiling_mib(available: Option<u64>, effective_total: Option<u64>) -> u64 {
318  const MIB: u64 = 1024 * 1024;
319  let candidate = match (available, effective_total) {
320    // Headroom rule: 90 % of what is free right now, never above what the
321    // process may use at all (the cgroup-capped total).
322    (Some(avail), Some(total)) => (avail / MIB * 9 / 10).min(total / MIB),
323    (Some(avail), None) => avail / MIB * 9 / 10,
324    // Availability unknown (exotic platform): the conservative half-of-total
325    // rule this default previously shipped.
326    (None, Some(total)) => total / MIB / 2,
327    (None, None) => return FALLBACK_CEILING_MIB,
328  };
329  candidate.clamp(MIN_DEFAULT_CEILING_MIB, MAX_DEFAULT_CEILING_MIB)
330}

332/// RAM available for new allocations right now, in bytes, or `None` when the
333/// platform offers no honest answer.
334///
335/// "Available" here means the OS's own estimate of what a new consumer can
336/// take **without pushing the machine into swap** — not merely "free", which
337/// undercounts by excluding reclaimable caches:
338/// - Linux: `MemAvailable` from `/proc/meminfo`, the kernel's purpose-built
339///   estimate (free + reclaimable page cache − watermarks). This is the one
340///   legitimate use of `/proc/meminfo` in this module — availability has no
341///   `sysconf` spelling — and it degrades safely: `None` on any other Unix,
342///   which the ceiling rule answers with the half-of-total fallback.
343/// - Windows: `GlobalMemoryStatusEx`'s `ullAvailPhys`, the direct analog.
344/// - macOS: `host_statistics64` free + inactive pages — inactive is macOS's
345///   reclaimable class, the moral equivalent of Linux's page cache share.
346pub fn available_memory_bytes() -> Option<u64> {
347  #[cfg(target_os = "linux")]
348  {
349    let meminfo = std::fs::read_to_string("/proc/meminfo").ok()?;
350    for line in meminfo.lines() {
351      if let Some(rest) = line.strip_prefix("MemAvailable:") {
352        let kb = rest.split_whitespace().next()?.parse::<u64>().ok()?;
353        return kb.checked_mul(1024);
354      }
355    }
356    None
357  }
358  #[cfg(target_os = "macos")]
359  {
360    let mut stats: libc::vm_statistics64 = unsafe { std::mem::zeroed() };
361    let mut count = (size_of::<libc::vm_statistics64>() / size_of::<libc::integer_t>())
362      as libc::mach_msg_type_number_t;
363    // SAFETY: `stats` is a correctly sized, zeroed struct and `count` names
364    // its capacity in `integer_t` units, exactly as the API requires;
365    // `mach_host_self` returns a send right that need not be deallocated for
366    // a one-shot query in a short-lived process. (`mach2`, not libc's deprecated
367    // `mach_host_self`: libc points Mach APIs to that crate.)
368    let kr = unsafe {
369      libc::host_statistics64(
370        mach2::mach_init::mach_host_self(),
371        libc::HOST_VM_INFO64,
372        (&raw mut stats).cast::<libc::integer_t>(),
373        &mut count,
374      )
375    };
376    if kr != libc::KERN_SUCCESS {
377      return None;
378    }
379    let page_size = rustix::param::page_size();
380    (stats.free_count as u64 + stats.inactive_count as u64).checked_mul(page_size as u64)
381  }
382  #[cfg(windows)]
383  {
384    use windows_sys::Win32::System::SystemInformation::{GlobalMemoryStatusEx, MEMORYSTATUSEX};
385    let mut status: MEMORYSTATUSEX = unsafe { std::mem::zeroed() };
386    status.dwLength = size_of::<MEMORYSTATUSEX>() as u32;
387    // SAFETY: `status` is a correctly sized, zeroed struct with `dwLength`
388    // set, exactly as the API requires.
389    if unsafe { GlobalMemoryStatusEx(&mut status) } != 0 {
390      return Some(status.ullAvailPhys);
391    }
392    None
393  }
394  #[cfg(not(any(target_os = "linux", target_os = "macos", windows)))]
395  {
396    None
397  }
398}

400/// Memory this PROCESS may actually use: physical RAM, capped by the cgroup
401/// limit when one is in force.
402///
403/// `total_memory_bytes` reports the HOST's RAM (`sysconf(_SC_PHYS_PAGES)`),
404/// which is blind to containers — so `docker run -m 4g` on a 258 GB host chose
405/// a 64 GiB ceiling and a 48 GiB cooperative fuse, neither of which the process
406/// could ever reach. The kernel killed it at 4 GB instead: exit 137, no output,
407/// no `Fatal:` line — exactly the failure the graceful guard exists to replace.
408/// The cortex fleet is only accidentally safe here, because it pins every child
409/// with `--max-rss-mb`; a plain containerized CLI user is not.
410fn effective_memory_bytes() -> Option<u64> {
411  let physical = total_memory_bytes();
412  match (physical, cgroup_limit_bytes()) {
413    (Some(p), Some(c)) => Some(p.min(c)),
414    (p, None) => p,
415    (None, c) => c,
416  }
417}

419/// The cgroup memory limit, v2 then v1, or `None` when unlimited/unreadable.
420///
421/// Both spell "unlimited" in their own way: v2 writes the literal `max`, v1
422/// writes a huge sentinel (`u64::MAX` rounded down to a page multiple), so a
423/// value at or above the host's RAM is treated as no limit rather than as a
424/// bound worth honouring.
425fn cgroup_limit_bytes() -> Option<u64> {
426  let v2 = std::fs::read_to_string("/sys/fs/cgroup/memory.max").ok();
427  let v1 = || std::fs::read_to_string("/sys/fs/cgroup/memory/memory.limit_in_bytes").ok();
428  let raw = v2.or_else(v1)?;
429  interpret_cgroup_limit(&raw, total_memory_bytes())
430}
431
432/// Decide what a cgroup limit file MEANS. Split out from the file read so the
433/// rules are testable without a container: the reachable spellings of
434/// "unlimited" are what make this subtle.
435fn interpret_cgroup_limit(raw: &str, physical: Option<u64>) -> Option<u64> {
436  let text = raw.trim();
437  // cgroup v2 spells unlimited literally.
438  if text == "max" {
439    return None;
440  }
441  let limit = text.parse::<u64>().ok()?;
442  // cgroup v1 spells it as a huge sentinel, and a "limit" at or above the
443  // host's RAM bounds nothing — treat both as no limit rather than as a
444  // ceiling worth honouring.
445  if limit == 0 || physical.is_some_and(|phys| limit >= phys) {
446    return None;
447  }
448  Some(limit)
449}

463/// Free space in bytes on the filesystem holding `path`, or `None` if it cannot
464/// be determined. Used to check headroom before spilling intermediates to disk.
465pub fn available_disk_bytes(path: &std::path::Path) -> Option<u64> {
466  #[cfg(unix)]
467  {
468    // (a path holding a NUL is an `Err`, as `CString::new` refused it)
469    let stat = rustix::fs::statvfs(path).ok()?;
470    // `f_bavail` is blocks available to unprivileged users — the honest figure,
471    // as `f_bfree` includes the root-reserved slice we cannot use.
472    stat.f_bavail.checked_mul(stat.f_frsize)
473  }
474  #[cfg(windows)]
475  {
476    use std::os::windows::ffi::OsStrExt;
477
478    use windows_sys::Win32::Storage::FileSystem::GetDiskFreeSpaceExW;
479    let mut wide: Vec<u16> = path.as_os_str().encode_wide().collect();
480    wide.push(0);
481    let mut free_to_caller: u64 = 0;
482    // SAFETY: `wide` is NUL-terminated and outlives the call; the two unused
483    // out-parameters are passed as null, which the API permits.
484    let ok = unsafe {
485      GetDiskFreeSpaceExW(
486        wide.as_ptr(),
487        &mut free_to_caller,
488        std::ptr::null_mut(),
489        std::ptr::null_mut(),
490      )
491    };
492    (ok != 0).then_some(free_to_caller)
493  }
494  #[cfg(not(any(unix, windows)))]
495  {
496    let _ = path;
497    None
498  }
499}

506/// Handle to a watchdog thread. Cancels on drop.

509/// lets call-sites set a watchdog conditionally without special-casing the
510/// "no timeout" branch.

512/// exit, called with the exit code it is about to take ([`EXIT_TIMEOUT`] or

590impl KillOnExit {
591  /// Ends the child and says how it ended: its own exit status when it exits within `grace` (a child whose pipe
592  /// closed is ending), `None` when it is still running then (the guard's drop kills it) or is already gone.
593  pub fn finish(self, grace: Duration) -> Option<std::process::ExitStatus> {
594    let deadline = Instant::now() + grace;
595    loop {
596      let status = {
597        let mut children = children();
598        let (_, child) = children.iter_mut().find(|(id, _)| *id == self.0)?;
599        child.try_wait().ok().flatten()
600      };
601      if status.is_some() || Instant::now() >= deadline {
602        return status;
603      }
604      thread::sleep(Duration::from_millis(5));
605    }
606  }
607}

609impl Drop for KillOnExit {
610  fn drop(&mut self) {
611    let child = {
612      let mut children = children();
613      let index = children.iter().position(|(id, _)| *id == self.0);
614      index.map(|index| children.swap_remove(index).1)
615    };
616    if let Some(mut child) = child {
617      let _ = child.kill();
618      let _ = child.wait();
619    }
620  }
621}

764/// Set by the first allocation failure that reaches a binary's allocation-failure hook.
765static ALLOC_FAILURE: AtomicBool = AtomicBool::new(false);

768  /// Whether this thread is inside the allocation-failure hook (const-initialized, no destructor:
769  /// reading it allocates nothing).
770  static IN_ALLOC_FAILURE_HOOK: std::cell::Cell<bool> = const { std::cell::Cell::new(false) };

773/// The first call of a binary's allocation-failure hook. The hook then runs alone and without
774/// allocating: it writes its line ([`write_stderr_unlocked`]), sweeps the run's trees
775/// ([`try_remove_registered_paths`]) and ends the process ([`exit_now`]).
776///
777/// When memory is spent, anything the hook allocates fails as well and calls the hook again, on the
778/// same thread. Run again, it would allocate again: a fleet child printed 6,048 nested
779/// `Fatal:oom:alloc_failed` lines in a millisecond (each re-entry failing at reading
780/// `RUST_BACKTRACE`, 1 byte). Or it would wait forever. The old hook formatted a backtrace while
781/// holding the standard library's backtrace lock and the stderr lock. Symbolizing the backtrace
782/// allocated, the re-entered hook waited on the backtrace lock, and the watchdog's own
783/// `Fatal:timeout` waited on stderr. The child stayed parked until the harness killed it as
784/// unresponsive and re-leased its paper (run 351, 2026-10-14; 1804.07640 among the papers). So a
785/// re-entry on the same thread ends the process at once. An allocation failure on another thread
786/// meanwhile waits for the first thread's exit, and exits itself if that has not come in ten seconds.
787pub fn enter_alloc_failure_hook() {
788  if IN_ALLOC_FAILURE_HOOK
789    .try_with(|inside| inside.replace(true))
790    .unwrap_or(true)
791  {
792    exit_now(EXIT_OOM);
793  }
794  if ALLOC_FAILURE.swap(true, Ordering::AcqRel) {
795    for _ in 0..10 {
796      thread::sleep(Duration::from_secs(1));
797    }
798    exit_now(EXIT_OOM);
799  }
800}

802/// Ends the process now with `code`: `_exit`, without the atexit handlers, destructors and stream
803/// flushes of [`std::process::exit`], any of which can allocate or wait on a lock held by a thread
804/// that will never release it. Every exit taken while another thread may still run goes through it:
805/// `exit` runs the libraries' destructors under that thread (libxml2's `.fini_array` calls
806/// `xmlCleanupParser`, freeing the catalog, encoding aliases, dict mutexes and global-state key a
807/// running render reads: a render worker's SIGSEGV with ip == fault address in freed heap, on Nasser's
808/// 2,400-problem slice under concurrent load, issue #777, not reproduced). What
809/// must land is written first, unbuffered: the stderr line ([`write_stderr_unlocked`]), the log
810/// mirror (a raw descriptor), a pre-exit hook's file. Only a partial line still in stdout's
811/// line buffer is lost.
812pub fn exit_now(code: i32) -> ! {
813  #[cfg(unix)]
814  // SAFETY: `_exit` takes no pointers and does not return.
815  unsafe {
816    libc::_exit(code)
817  }
818  #[cfg(not(unix))]
819  std::process::exit(code)
820}

822/// Writes `args` to standard error in one `write(2)`, formatted into a stack buffer (cut at 1 KiB).
823/// Unlike `eprintln!` it neither allocates nor takes the standard library's stderr lock. That makes
824/// it safe in an allocation-failure hook, and for the watchdog when a stuck thread holds that lock.
825/// It opens a fresh line when a progress note left stderr mid-line, as the logger's records do: a
826/// `Fatal:timeout:wallclock` glued onto `(Processing definitions …l3backend-pdftex.def... )` escaped
827/// every `^Fatal:` count.
828pub fn write_stderr_unlocked(args: std::fmt::Arguments) {
829  struct StackLine {
830    bytes: [u8; 1024],
831    len:   usize,
832  }
833  impl std::fmt::Write for StackLine {
834    fn write_str(&mut self, text: &str) -> std::fmt::Result {
835      let take = text.len().min(self.bytes.len() - self.len);
836      self.bytes[self.len..self.len + take].copy_from_slice(&text.as_bytes()[..take]);
837      self.len += take;
838      Ok(())
839    }
840  }
841  let mut line = StackLine { bytes: [0; 1024], len: 0 };
842  if !crate::util::logger::stderr_at_line_start() {
843    line.bytes[0] = b'\n';
844    line.len = 1;
845  }
846  let _ = std::fmt::write(&mut line, args);
847  if line.len > 0 {
848    crate::util::logger::set_stderr_at_line_start(line.bytes[line.len - 1] == b'\n');
849  }
850  #[cfg(unix)]
851  // SAFETY: a write of `len` initialized bytes from a stack buffer to the standard error descriptor.
852  unsafe {
853    libc::write(2, line.bytes.as_ptr().cast(), line.len);
854  }
855  #[cfg(not(unix))]
856  {
857    use std::io::Write;
858    let _ = std::io::stderr().write_all(&line.bytes[..line.len]);
859  }
860}

862/// The exit code claimed when the watchdog (or an allocation-failure hook) begins to exit, 0 while none is. Its sweep
863/// deletes the run's trees while the other threads still run, and one finishing in that window must neither exit first
864/// with its own code (the first `exit` wins) nor hand on output built over the deleted trees: [`park_if_exiting`] holds
865/// it until the claimed exit ends the process. [`NORMAL_END`] while the normal path hands on its result.
866static EXITING: AtomicI32 = AtomicI32::new(0);
867
868/// [`EXITING`] once the normal path has claimed the process's end ([`claim_normal_end`]).
869const NORMAL_END: i32 = -1;
870
871/// Claims the exit with `code` (the first claim stands); false once the normal path claimed the end.
872fn claim_exit(code: i32) -> bool {
873  match EXITING.compare_exchange(0, code, Ordering::AcqRel, Ordering::Acquire) {
874    Ok(_) => true,
875    Err(held) => held != NORMAL_END,
876  }
877}

879/// Claims the process's end for the normal path before a single-run binary hands on its result: a breach after it
880/// stands down, where its pre-exit hook would overwrite the written result with its placeholder (cortex_worker
881/// `--standalone --output`). When a breach claimed first, parks ([`park_if_exiting`]) until its exit.
882pub fn claim_normal_end() {
883  if EXITING
884    .compare_exchange(0, NORMAL_END, Ordering::AcqRel, Ordering::Acquire)
885    .is_err()
886  {
887    park_if_exiting();
888  }
889}

891/// Parks the calling thread once the watchdog has begun to exit, until that exit ends the process. Called before every
892/// exit and result hand-off a timed-out run could race (the binaries' final exits, cortex_worker's results). Bounded: if
893/// the exit has not come within a minute (a pre-exit hook that blocked, or a sweep slower than that, which it then cuts
894/// short) the thread exits itself, with the claimed code, through [`exit_now`]: the watchdog thread still runs.
895pub fn park_if_exiting() {
896  let code = EXITING.load(Ordering::Acquire);
897  if code > 0 {
898    let deadline = Instant::now() + Duration::from_secs(60);
899    while let Some(left) = deadline.checked_duration_since(Instant::now()) {
900      thread::park_timeout(left);
901    }
902    exit_now(code);
903  }
904}

906/// Setup kept off every deadline ([`off_the_clock`]), read without a lock (`stomach::check_timeout` is on the hot path):
907/// the time the finished setups took, and the running one's start (nanoseconds from [`epoch`], plus one; 0: none).
908static SETUP_DONE_NS: AtomicU64 = AtomicU64::new(0);
909static SETUP_SINCE_NS: AtomicU64 = AtomicU64::new(0);

911/// The running setup's nesting depth, what it is and its bound (the outermost call's): taken by the setup's calls and
912/// the watchdog's poll, never by a deadline check.
913static SETUP_RUNNING: std::sync::Mutex<(u32, &str, Duration)> =
914  std::sync::Mutex::new((0, "", Duration::ZERO));
915
916fn setup_running() -> std::sync::MutexGuard<'static, (u32, &'static str, Duration)> {
917  SETUP_RUNNING
918    .lock()
919    .unwrap_or_else(std::sync::PoisonError::into_inner)
920}

922fn epoch() -> Instant {
923  static EPOCH: std::sync::OnceLock<Instant> = std::sync::OnceLock::new();
924  *EPOCH.get_or_init(Instant::now)
925}

927fn nanos_since_epoch(at: Instant) -> u64 {
928  u64::try_from(at.saturating_duration_since(epoch()).as_nanos()).unwrap_or(u64::MAX)
929}

931/// Runs `work`, one-time setup that is not the conversion's (`what`: the format build for the host's TeX tree), off
932/// every deadline: the watchdog's and the cooperative one (`stomach::set_timeout`) move back by the time it takes,
933/// and while it runs only `bound` ends it, a watchdog's breach (exit 124). Calls nest: the outermost one's time and
934/// bound count. Without a watchdog (`--timeout 0`) nothing bounds it.
935pub fn off_the_clock<T>(what: &'static str, bound: Duration, work: impl FnOnce() -> T) -> T {
936  let _running = OffTheClock::enter(what, bound);
937  work()
938}

940/// One [`off_the_clock`] call's share of the running setup; its drop ends the setup when it is the outermost, a
941/// panic's unwinding included.
942struct OffTheClock;
943
944impl OffTheClock {
945  fn enter(what: &'static str, bound: Duration) -> Self {
946    let mut running = setup_running();
947    if running.0 == 0 {
948      *running = (0, what, bound);
949      SETUP_SINCE_NS.store(nanos_since_epoch(Instant::now()) + 1, Ordering::Release);
950    }
951    running.0 += 1;
952    OffTheClock
953  }
954}
955
956impl Drop for OffTheClock {
957  fn drop(&mut self) {
958    let mut running = setup_running();
959    running.0 -= 1;
960    if running.0 == 0 {
961      let since = SETUP_SINCE_NS.load(Ordering::Acquire).saturating_sub(1);
962      let took = nanos_since_epoch(Instant::now()).saturating_sub(since);
963      // Added before the start is cleared: a reader that sees no running setup sees its time (one between may count
964      // it twice, a later deadline, never an early one).
965      SETUP_DONE_NS.fetch_add(took, Ordering::AcqRel);
966      SETUP_SINCE_NS.store(0, Ordering::Release);
967    }
968  }
969}

971/// The time [`off_the_clock`] setup has taken in this process so far, the running one's included.
972pub fn time_off_the_clock() -> Duration {
973  let since = SETUP_SINCE_NS.load(Ordering::Acquire);
974  let done = SETUP_DONE_NS.load(Ordering::Acquire);
975  let running = since.checked_sub(1).map_or(0, |since| {
976    nanos_since_epoch(Instant::now()).saturating_sub(since)
977  });
978  Duration::from_nanos(done.saturating_add(running))
979}

981/// Whether `deadline`, set when [`time_off_the_clock`] read `off_at_set`, has passed once the setup since is taken off
982/// the clock: the one test of both the watchdog and the cooperative deadline.
983pub fn deadline_passed(deadline: Instant, off_at_set: Duration) -> bool {
984  let now = Instant::now();
985  // The setup only defers a deadline: before it, no atomic is read.
986  now > deadline && now > deadline + time_off_the_clock().saturating_sub(off_at_set)
987}

989/// The running setup and its bound, once it has run past that bound (what the watchdog's breach line names).
990pub fn setup_past_bound() -> Option<(&'static str, Duration)> {
991  let (depth, what, bound) = *setup_running();
992  let since = SETUP_SINCE_NS.load(Ordering::Acquire).checked_sub(1)?;
993  let ran = Duration::from_nanos(nanos_since_epoch(Instant::now()).saturating_sub(since));
994  (depth > 0 && ran >= bound).then_some((what, bound))
995}

1006  /// Create a watchdog guarding a wall-clock deadline **and** a resident-memory

1012  /// [`EXIT_OOM`]. The memory guard is inactive where [`process_rss_kb`]
1013  /// returns `None` (non-Linux); see the module portability note.
1014  pub fn with_limits(timeout_secs: u64, max_rss_kb: u64) -> Self {
1015    let cancelled = Arc::new(AtomicBool::new(false));
1016    if timeout_secs > 0 || max_rss_kb > 0 {
1017      let c = cancelled.clone();
1018      thread::Builder::new()
1019        .name("latexml-watchdog".to_string())
1020        .spawn(move || Self::run(c, timeout_secs, max_rss_kb))
1021        .expect("watchdog thread spawn failed");
1022    }
1023    Self { cancelled }
1024  }
1025
1026  fn run(cancelled: Arc<AtomicBool>, timeout_secs: u64, max_rss_kb: u64) {
1027    let deadline = (timeout_secs > 0).then(|| {
1028      (
1029        Instant::now() + Duration::from_secs(timeout_secs),
1030        time_off_the_clock(),
1031      )
1032    });
1033    let poll_interval = Duration::from_millis(100);
1034    loop {
1035      if cancelled.load(Ordering::Relaxed) {
1036        return; // cancelled: graceful exit.
1037      }
1038      let breach = deadline.and_then(|(deadline, off_at_set)| match setup_past_bound() {
1039        Some((what, bound)) => Some(format!(
1040          "{what} ran past its own bound of {}s (kept off the {timeout_secs}s timeout)",
1041          bound.as_secs()
1042        )),
1043        None => deadline_passed(deadline, off_at_set)
1044          .then(|| format!("main-level wall-clock timeout after {timeout_secs}s")),
1045      });
1046      if let Some(breach) = breach {
1047        // Claimed first: a normal end claimed before it (the result being written) stands the breach down.
1048        if cancelled.load(Ordering::Relaxed) || !claim_exit(EXIT_TIMEOUT) {
1049          return;
1050        }
1051        // Unlocked: a thread stuck while holding the stderr lock must not hold up the exit.
1052        write_stderr_unlocked(format_args!(
1053          "Fatal:timeout:wallclock latexml-oxide: {breach} — exiting process\n"
1054        ));
1055        // Run the optional pre-exit hook (e.g. cortex_worker writing a
1056        // structured Status:conversion:3 placeholder to its --output path)
1057        // BEFORE exiting. The hook is invoked at most once per process.
1058        run_pre_exit_hook(EXIT_TIMEOUT);
1059        kill_registered_children();
1060        remove_registered_paths();
1061        // Exit 124 (the standard timeout code), not `abort()`: the worker thread is presumed wedged
1062        // and won't observe a cancel, and a SIGABRT reads to the harness as a crash ("Aborted (core
1063        // dumped)"; witnesses 2602.11915, 2604.11500, 2604.13944, hep-ph9205242, q-alg9604005,
1064        // q-alg9605003, q-alg9605028, the 7 "Aborted" rows of the 2026-05-13 588-paper sweep).
1065        // Through `exit_now`, as that thread still runs: `exit`'s library destructors would free
1066        // libxml2's globals under it (issue #777's render-worker SIGSEGV, not reproduced).
1067        exit_now(EXIT_TIMEOUT);
1068      }
1069      if max_rss_kb > 0
1070        && let Some(rss) = process_rss_kb()
1071        && rss > max_rss_kb
1072      {
1073        if cancelled.load(Ordering::Relaxed) || !claim_exit(EXIT_OOM) {
1074          return;
1075        }
1076        write_stderr_unlocked(format_args!(
1077          "Fatal:oom:rss latexml-oxide: resident memory {}MB exceeded the {}MB ceiling — exiting \
1078           process. This document needs a larger ceiling: rerun with a higher --max-memory on a \
1079           machine with enough free RAM.\n",
1080          rss / 1024,
1081          max_rss_kb / 1024
1082        ));
1083        run_pre_exit_hook(EXIT_OOM);
1084        kill_registered_children();
1085        remove_registered_paths();
1086        exit_now(EXIT_OOM);
1087      }
1088      thread::sleep(poll_interval);
1089    }
1090  }
