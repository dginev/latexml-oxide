1//! Wall-clock watchdog that forcibly aborts the process after a deadline.
2//!
3//! The existing `stomach::check_timeout()` is a cooperative mechanism — it only
4//! fires when the digestion loop polls it. That leaves tight native loops
5//! (Marpa precompute / parse, libxml2 post-processing, FFI calls into libxslt,
6//! ...) completely unguarded: a 60-second timeout can easily turn into 10
7//! minutes if control never returns to the digestion loop.
8//!
9//! This module provides a main-level `Watchdog` that spawns a dedicated thread
10//! at construction, wakes after the specified number of seconds, and — if the
11//! watchdog has not yet been cancelled — prints a message and calls
12//! `std::process::abort()`. That guarantees the process dies within
13//! `timeout + poll_interval` of the configured deadline, regardless of what
14//! the main thread is doing.
15//!
16//! # Design notes
17//!
18//! - Uses `Arc<AtomicBool>` for cancellation. Polling every 100 ms keeps the cancellation latency
19//!   low without burning CPU.
20//! - `Drop` on the `Watchdog` handle cancels the watchdog thread, so RAII usage (`let _wd =
21//!   Watchdog::new(secs)`) is sufficient.
22//! - We use `std::process::abort()` rather than `panic!` because panic may unwind or be caught by a
23//!   surrounding `catch_unwind`, which would defeat the safety guarantee. `abort()` delivers
24//!   `SIGABRT` and always terminates the process.
25//! - The existing cooperative `stomach::check_timeout()` path is retained: on most conversions it
26//!   fires before the hard abort, giving callers a nice `Err(Fatal)` with proper error propagation.
27//!   The watchdog is a safety net for the pathological cases where cooperative polling doesn't
28//!   happen.
29//!
30//! # Resource limits
31//!
32//! [`Watchdog::with_limits`](crate::watchdog::Watchdog::with_limits) guards
33//! **both** a wall-clock deadline and a
34//! resident-memory ceiling — the two defenses any executable that converts
35//! arbitrary input needs. It is the shared guard reused by both
36//! `cortex_worker` (in-process, one paper per process) and the
37//! `latexml_oxide --server` LSP (run inside each forked body child, which
38//! self-terminates on breach so the parent reaps it via pipe EOF). The exit
39//! codes are distinct so a supervising parent can tell them apart:
40//! `124` = wall-clock timeout, `137` = memory ceiling.
41//!
42//! # Portability
43//!
44//! The wall-clock guard is portable (`std::thread` + `Instant`).
45//!
46//! **Portable:** [`total_memory_bytes`](crate::watchdog::total_memory_bytes) and [`available_disk_bytes`](crate::watchdog::available_disk_bytes) answer on
47//! Linux, macOS and Windows — `sysconf(_SC_PHYS_PAGES)`/`statvfs` are POSIX, and
48//! Windows uses `GlobalMemoryStatusEx`/`GetDiskFreeSpaceExW`. They back the
49//! machine-derived default ceiling ([`default_ceiling_mib`](crate::watchdog::default_ceiling_mib)) and the
50//! spill-headroom check, so those behave identically on every supported OS.
51//!
52//! **Still Linux-only:** [`process_rss_kb`](crate::watchdog::process_rss_kb) samples `/proc/self/status`, so the
53//! *enforcement* half of the RAM guard is inactive elsewhere — the ceiling is
54//! computed correctly but never checked, and only the time guard bites. Closing
55//! this needs macOS `task_info(TASK_BASIC_INFO)` and Windows
56//! `GetProcessMemoryInfo`; both are reachable through the crates this module
57//! already links, so it is a bounded follow-up rather than a new dependency
58//! question.

69/// Current resident set size of this process in KiB, or `None` if it can't be
70/// determined. Linux reads `VmRSS` from `/proc/self/status`; macOS asks
71/// libproc (`proc_pidinfo`/`PROC_PIDTASKINFO`) — without it the whole
72/// `--max-memory` ceiling silently did not exist on macOS, and the
73/// `115_streaming_cli` Fatal-contract guard caught exactly that on macOS CI
74/// (an over-budget run exited 0). Cheap enough to poll a few times a second.
75#[cfg(target_os = "linux")]
76pub fn process_rss_kb() -> Option<u64> {
77  let status = std::fs::read_to_string("/proc/self/status").ok()?;
78  for line in status.lines() {
79    if let Some(rest) = line.strip_prefix("VmRSS:") {
80      return rest.split_whitespace().next()?.parse::<u64>().ok();
81    }
82  }
83  None
84}

86/// macOS: libproc task info; `pti_resident_size` is in BYTES.
87#[cfg(target_os = "macos")]
88pub fn process_rss_kb() -> Option<u64> {
89  let mut info: libc::proc_taskinfo = unsafe { std::mem::zeroed() };
90  let size = std::mem::size_of::<libc::proc_taskinfo>() as libc::c_int;
91  // SAFETY: proc_pidinfo fills at most `size` bytes of the zeroed struct we
92  // own; a short or negative return is handled below.
93  let got = unsafe {
94    libc::proc_pidinfo(
95      std::process::id() as libc::c_int,
96      libc::PROC_PIDTASKINFO,
97      0,
98      (&raw mut info).cast::<libc::c_void>(),
99      size,
100    )
101  };
102  (got == size).then(|| info.pti_resident_size / 1024)
103}

109/// Total physical RAM on this machine in bytes, or `None` if it cannot be
110/// determined.
111///
112/// Portable by construction: `sysconf(_SC_PHYS_PAGES) * sysconf(_SC_PAGE_SIZE)`
113/// is POSIX and answers on both Linux and macOS, and Windows has
114/// `GlobalMemoryStatusEx`. No new dependency — `libc` and `windows-sys` are
115/// already in the tree.
116///
117/// Deliberately NOT `/proc/meminfo`: that would repeat the Linux-only mistake
118/// this module already carries in [`process_rss_kb`], which returns `None`
119/// everywhere else and so silently deactivates the memory ceiling on
120/// macOS/Windows.
121pub fn total_memory_bytes() -> Option<u64> {
122  #[cfg(unix)]
123  {
124    // SAFETY: `sysconf` is a pure query with no pointer arguments; a negative
125    // return means "unavailable", which we map to `None` rather than trusting.
126    let pages = unsafe { libc::sysconf(libc::_SC_PHYS_PAGES) };
127    let page_size = unsafe { libc::sysconf(libc::_SC_PAGESIZE) };
128    if pages > 0 && page_size > 0 {
129      return (pages as u64).checked_mul(page_size as u64);
130    }
131    None
132  }
133  #[cfg(windows)]
134  {
135    // `MEMORYSTATUSEX` is what windows-sys calls this, mirroring the Win32
136    // header. This arm originally said `GLOBAL_MEMORY_STATUS_EX` — a spelling
137    // from a different generation of the bindings — while Cargo.toml pinned
138    // `windows-sys = "0.61"`, so it never matched the version it declared and
139    // has NEVER compiled. It survived because CI builds Linux and macOS only:
140    // the first Windows compile of this file was the 0.7.5-rc4 RELEASE
141    // workflow, on a tag, days after the code landed. See task #164.
142    use windows_sys::Win32::System::SystemInformation::{GlobalMemoryStatusEx, MEMORYSTATUSEX};
143    let mut status: MEMORYSTATUSEX = unsafe { std::mem::zeroed() };
144    status.dwLength = std::mem::size_of::<MEMORYSTATUSEX>() as u32;
145    // SAFETY: `status` is a correctly sized, zeroed struct with `dwLength` set,
146    // exactly as the API requires.
147    if unsafe { GlobalMemoryStatusEx(&mut status) } != 0 {
148      return Some(status.ullTotalPhys);
149    }
150    None
151  }
152  #[cfg(not(any(unix, windows)))]
153  {
154    None
155  }
156}

158/// The default per-conversion memory ceiling in MiB, derived from the machine.
159///
160/// `min(64 GiB, HALF of physical RAM)`, floored at
161/// [`MIN_DEFAULT_CEILING_MIB`], falling back to [`FALLBACK_CEILING_MIB`] when
162/// the machine cannot be probed. One sentence for users: *a conversion uses at
163/// most half your RAM, never more than 64 GiB.*
164///
165/// The default was a flat 6144 MiB regardless of hardware — absurd on a 256 GB
166/// host and over-generous on a laptop — and then 90 % of RAM, which is
167/// laptop-hostile for a different reason: the cooperative fuse rides at 75 % of
168/// the ceiling, so on a 16 GB machine one conversion was allowed to reach
169/// **10.8 GiB** before we so much as complained, by which point the user's
170/// session has been swapping for a long time. Half of RAM is the design point
171/// for a 16 GB baseline laptop:
172///
173/// | RAM | ceiling | fuse (75 %) | spill watermark |
174/// |---|---|---|---|
175/// | 16 GB | 8 GiB | 6 GiB | 2 GiB |
176/// | 32 GB | 16 GiB | 12 GiB | 4 GiB |
177/// | 96 GB | 48 GiB | 36 GiB | 12 GiB |
178///
179/// The 96 GB row is not a guess: 48 GiB is the ceiling under which the 131 MB
180/// witness was measured to convert end-to-end (28.1 GB peak), so the default
181/// reproduces a validated configuration on the reporter's own hardware.
182///
183/// The **64 GiB cap** is not about this machine but about the *others*: in a
184/// parallel fleet the aggregate is `N_processes x ceiling`, so an uncapped
185/// fraction-of-RAM rule on a big host would let a busy fleet OOM it. The
186/// `cortex_worker` fleet overrides this with its own per-child ceiling anyway;
187/// the cap keeps the single-process default from being reckless.
188pub fn default_ceiling_mib() -> u64 {
189  const MIB: u64 = 1024 * 1024;
190  match effective_memory_bytes() {
191    Some(total) => (total / MIB / 2).clamp(MIN_DEFAULT_CEILING_MIB, MAX_DEFAULT_CEILING_MIB),
192    None => FALLBACK_CEILING_MIB,
193  }
194}

196/// Memory this PROCESS may actually use: physical RAM, capped by the cgroup
197/// limit when one is in force.
198///
199/// `total_memory_bytes` reports the HOST's RAM (`sysconf(_SC_PHYS_PAGES)`),
200/// which is blind to containers — so `docker run -m 4g` on a 258 GB host chose
201/// a 64 GiB ceiling and a 48 GiB cooperative fuse, neither of which the process
202/// could ever reach. The kernel killed it at 4 GB instead: exit 137, no output,
203/// no `Fatal:` line — exactly the failure the graceful guard exists to replace.
204/// The cortex fleet is only accidentally safe here, because it pins every child
205/// with `--max-rss-mb`; a plain containerized CLI user is not.
206fn effective_memory_bytes() -> Option<u64> {
207  let physical = total_memory_bytes();
208  match (physical, cgroup_limit_bytes()) {
209    (Some(p), Some(c)) => Some(p.min(c)),
210    (p, None) => p,
211    (None, c) => c,
212  }
213}

215/// The cgroup memory limit, v2 then v1, or `None` when unlimited/unreadable.
216///
217/// Both spell "unlimited" in their own way: v2 writes the literal `max`, v1
218/// writes a huge sentinel (`u64::MAX` rounded down to a page multiple), so a
219/// value at or above the host's RAM is treated as no limit rather than as a
220/// bound worth honouring.
221fn cgroup_limit_bytes() -> Option<u64> {
222  let v2 = std::fs::read_to_string("/sys/fs/cgroup/memory.max").ok();
223  let v1 = || std::fs::read_to_string("/sys/fs/cgroup/memory/memory.limit_in_bytes").ok();
224  let raw = v2.or_else(v1)?;
225  interpret_cgroup_limit(&raw, total_memory_bytes())
226}
227
228/// Decide what a cgroup limit file MEANS. Split out from the file read so the
229/// rules are testable without a container: the reachable spellings of
230/// "unlimited" are what make this subtle.
231fn interpret_cgroup_limit(raw: &str, physical: Option<u64>) -> Option<u64> {
232  let text = raw.trim();
233  // cgroup v2 spells unlimited literally.
234  if text == "max" {
235    return None;
236  }
237  let limit = text.parse::<u64>().ok()?;
238  // cgroup v1 spells it as a huge sentinel, and a "limit" at or above the
239  // host's RAM bounds nothing — treat both as no limit rather than as a
240  // ceiling worth honouring.
241  if limit == 0 || physical.is_some_and(|phys| limit >= phys) {
242    return None;
243  }
244  Some(limit)
245}

259/// Free space in bytes on the filesystem holding `path`, or `None` if it cannot
260/// be determined. Used to check headroom before spilling intermediates to disk.
261pub fn available_disk_bytes(path: &std::path::Path) -> Option<u64> {
262  #[cfg(unix)]
263  {
264    use std::os::unix::ffi::OsStrExt;
265    let c_path = std::ffi::CString::new(path.as_os_str().as_bytes()).ok()?;
266    // SAFETY: zeroed `statvfs` is a valid initial state; `c_path` is a
267    // NUL-terminated string that outlives the call.
268    let mut stat: libc::statvfs = unsafe { std::mem::zeroed() };
269    if unsafe { libc::statvfs(c_path.as_ptr(), &mut stat) } != 0 {
270      return None;
271    }
272    // `f_bavail` is blocks available to unprivileged users — the honest figure,
273    // as `f_bfree` includes the root-reserved slice we cannot use.
274    (stat.f_bavail as u64).checked_mul(stat.f_frsize as u64)
275  }
276  #[cfg(windows)]
277  {
278    use std::os::windows::ffi::OsStrExt;
279
280    use windows_sys::Win32::Storage::FileSystem::GetDiskFreeSpaceExW;
281    let mut wide: Vec<u16> = path.as_os_str().encode_wide().collect();
282    wide.push(0);
283    let mut free_to_caller: u64 = 0;
284    // SAFETY: `wide` is NUL-terminated and outlives the call; the two unused
285    // out-parameters are passed as null, which the API permits.
286    let ok = unsafe {
287      GetDiskFreeSpaceExW(
288        wide.as_ptr(),
289        &mut free_to_caller,
290        std::ptr::null_mut(),
291        std::ptr::null_mut(),
292      )
293    };
294    (ok != 0).then_some(free_to_caller)
295  }
296  #[cfg(not(any(unix, windows)))]
297  {
298    let _ = path;
299    None
300  }
301}

304pub const EXIT_TIMEOUT: i32 = 124;
305/// Exit code used when the memory ceiling is exceeded (128 + SIGKILL).
306pub const EXIT_OOM: i32 = 137;

308/// Handle to a watchdog thread. Cancels on drop.
309///
310/// `Watchdog::new(0)` is a no-op — produces a handle that does nothing. This
311/// lets call-sites set a watchdog conditionally without special-casing the
312/// "no timeout" branch.
313/// Optional hook to run from the watchdog thread immediately before
314/// `exit(124)`. Used by `cortex_worker --standalone` to write a
315/// structured `Status:conversion:3` placeholder to `--output` so the
316/// timeout produces a usable failure artifact instead of a missing
317/// file. Set once at startup via `set_pre_exit_hook`; the hook is
318/// invoked exactly once. Zero overhead on the happy path — only the
319/// watchdog firing reads it.
320type PreExitHook = Box<dyn FnOnce() + Send + 'static>;
321
322static PRE_EXIT_HOOK: std::sync::OnceLock<std::sync::Mutex<Option<PreExitHook>>> =
323  std::sync::OnceLock::new();
324
325pub fn set_pre_exit_hook(hook: PreExitHook) {
326  let cell = PRE_EXIT_HOOK.get_or_init(|| std::sync::Mutex::new(None));
327  if let Ok(mut guard) = cell.lock() {
328    *guard = Some(hook);
329  }
330}
331
332fn run_pre_exit_hook() {
333  if let Some(cell) = PRE_EXIT_HOOK.get()
334    && let Ok(mut guard) = cell.lock()
335    && let Some(hook) = guard.take()
336  {
337    hook();
338  }
339}

341pub struct Watchdog {
342  cancelled: Arc<AtomicBool>,
343}
344
345impl Watchdog {
346  /// Create a wall-clock-only watchdog. `timeout_secs = 0` disables it.
347  /// Equivalent to [`Watchdog::with_limits(timeout_secs, 0)`].
348  pub fn new(timeout_secs: u64) -> Self { Self::with_limits(timeout_secs, 0) }
349
350  /// Create a watchdog guarding a wall-clock deadline **and** a resident-memory
351  /// ceiling. `timeout_secs = 0` disables the time guard; `max_rss_kb = 0`
352  /// disables the memory guard. With both `0` this is a no-op handle.
353  ///
354  /// The thread polls `cancelled`, the deadline, and RSS every `poll_interval`.
355  /// On a time breach it exits [`EXIT_TIMEOUT`]; on a memory breach,
356  /// [`EXIT_OOM`]. The memory guard is inactive where [`process_rss_kb`]
357  /// returns `None` (non-Linux); see the module portability note.
358  pub fn with_limits(timeout_secs: u64, max_rss_kb: u64) -> Self {
359    let cancelled = Arc::new(AtomicBool::new(false));
360    if timeout_secs > 0 || max_rss_kb > 0 {
361      let c = cancelled.clone();
362      thread::Builder::new()
363        .name("latexml-watchdog".to_string())
364        .spawn(move || Self::run(c, timeout_secs, max_rss_kb))
365        .expect("watchdog thread spawn failed");
366    }
367    Self { cancelled }
368  }
369
370  fn run(cancelled: Arc<AtomicBool>, timeout_secs: u64, max_rss_kb: u64) {
371    let deadline = (timeout_secs > 0).then(|| Instant::now() + Duration::from_secs(timeout_secs));
372    let poll_interval = Duration::from_millis(100);
373    loop {
374      if cancelled.load(Ordering::Relaxed) {
375        return; // cancelled: graceful exit.
376      }
377      if let Some(deadline) = deadline
378        && Instant::now() >= deadline
379      {
380        if cancelled.load(Ordering::Relaxed) {
381          return;
382        }
383        eprintln!(
384          "Fatal:timeout:wallclock latexml-oxide: main-level wall-clock timeout after {timeout_secs}s — exiting process"
385        );
386        // Run the optional pre-exit hook (e.g. cortex_worker writing a
387        // structured Status:conversion:3 placeholder to its --output path)
388        // BEFORE exiting. The hook is invoked at most once per process.
389        run_pre_exit_hook();
390        // `std::process::exit(124)` instead of `abort()`: the watchdog must
391        // terminate the whole process (the worker thread is presumed wedged
392        // in a tight loop that won't observe a cooperative cancel), but
393        // `abort()` produces a "Aborted (core dumped)" SIGABRT trace from
394        // the shell. `exit(124)` (standard timeout exit code) runs atexit
395        // handlers, flushes stderr, and leaves a clean exit signal the
396        // parent harness can interpret as "paper timed out" without
397        // conflating it with a Rust panic / memory corruption. Witnesses:
398        // 2602.11915, 2604.11500, 2604.13944, hep-ph9205242, q-alg9604005,
399        // q-alg9605003, q-alg9605028 — the 7 "Aborted" rows in the
400        // 2026-05-13 588-paper sweep.
401        std::process::exit(EXIT_TIMEOUT);
402      }
403      if max_rss_kb > 0
404        && let Some(rss) = process_rss_kb()
405        && rss > max_rss_kb
406      {
407        if cancelled.load(Ordering::Relaxed) {
408          return;
409        }
410        eprintln!(
411          "Fatal:oom:rss latexml-oxide: resident memory {}MB exceeded the {}MB ceiling — exiting process",
412          rss / 1024,
413          max_rss_kb / 1024
414        );
415        run_pre_exit_hook();
416        std::process::exit(EXIT_OOM);
417      }
418      thread::sleep(poll_interval);
419    }
420  }
421
422  /// Explicitly cancel the watchdog. Idempotent.
423  pub fn cancel(&self) { self.cancelled.store(true, Ordering::Relaxed); }
424}
425
426impl Drop for Watchdog {
427  fn drop(&mut self) { self.cancel(); }
428}

434  /// The cgroup limit decides the ceiling in a container, so each way of
435  /// spelling "unlimited" has to be recognised — otherwise a sentinel is
436  /// mistaken for a 8 EiB budget, or `max` for a parse failure.
437  #[test]
438  fn cgroup_limit_recognises_every_unlimited_spelling() {
439    const PHYS: Option<u64> = Some(258 * 1024 * 1024 * 1024);
440
441    // cgroup v2: the literal word.
442    assert_eq!(interpret_cgroup_limit("max\n", PHYS), None);
443    // cgroup v1: a huge sentinel (u64::MAX rounded to a page multiple).
444    assert_eq!(interpret_cgroup_limit("9223372036854771712", PHYS), None);
445    // A limit at or above physical RAM bounds nothing (258 GiB exactly, and
446    // above).
447    assert_eq!(interpret_cgroup_limit("277025390592", PHYS), None);
448    assert_eq!(interpret_cgroup_limit("281474976710656", PHYS), None);
449    // Garbage is not a limit.
450    assert_eq!(interpret_cgroup_limit("", PHYS), None);
451    assert_eq!(interpret_cgroup_limit("nonsense", PHYS), None);
452    assert_eq!(interpret_cgroup_limit("0", PHYS), None);

454    // A REAL limit is honoured — this is the `docker run -m 4g` case that was
455    // being ignored, leaving the process to be OOM-killed at 4 GB while its
456    // cooperative fuse sat at 48 GiB.

538    assert!(
539      total < 100 * 1024 * 1024 * 1024 * 1024,
540      "implausibly large total RAM ({total} bytes) — check the unit conversion"
541    );
542  }

544  /// `min(64 GiB, 90 % of RAM)`, and never zero — a zero ceiling would mean
545  /// "no limit" downstream (`resolve_rss_cap`), inverting the intent.
546  #[test]
547  fn default_ceiling_respects_both_halves_of_the_rule() {
548    let ceiling = default_ceiling_mib();
549    assert!(
550      ceiling > 0,
551      "a derived ceiling of 0 would read as 'unlimited'"
552    );
553    assert!(
554      ceiling <= MAX_DEFAULT_CEILING_MIB,
555      "derived {ceiling} MiB exceeds the {MAX_DEFAULT_CEILING_MIB} MiB cap; on a \
556       large host an uncapped fraction-of-RAM default would let a parallel fleet \
557       (N_processes x ceiling) OOM the machine"
558    );
559    if let Some(total) = total_memory_bytes() {
560      let half = total / (1024 * 1024) / 2;
561      assert_eq!(
562        ceiling,
563        half.clamp(MIN_DEFAULT_CEILING_MIB, MAX_DEFAULT_CEILING_MIB),
564        "ceiling should be min(cap, HALF of RAM), floored"
565      );
566      // Half of RAM, not 90 %: the cooperative fuse rides at 75 % of the
567      // ceiling, so at 90 % a 16 GB machine let one conversion reach 10.8 GiB
568      // before complaining — long after the session started swapping. This
569      // assertion is hardware-dependent by nature (it failed only on the
570      // smaller macOS runner, where the 64 GiB cap does not bind and the
571      // fraction is what shows).
572      assert!(
573        ceiling <= total / (1024 * 1024) / 2 || ceiling == MIN_DEFAULT_CEILING_MIB,
574        "the ceiling must leave the OS the larger share of physical RAM"
575      );
576    } else {
577      assert_eq!(ceiling, FALLBACK_CEILING_MIB);
578    }
579  }
