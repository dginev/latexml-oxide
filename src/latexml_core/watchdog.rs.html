1//! Wall-clock watchdog that forcibly aborts the process after a deadline.
2//!
3//! The existing `stomach::check_timeout()` is a cooperative mechanism — it only
4//! fires when the digestion loop polls it. That leaves tight native loops
5//! (Marpa precompute / parse, libxml2 post-processing, FFI calls into libxslt,
6//! ...) completely unguarded: a 60-second timeout can easily turn into 10
7//! minutes if control never returns to the digestion loop.
8//!
9//! This module provides a main-level `Watchdog` that spawns a dedicated thread
10//! at construction, wakes after the specified number of seconds, and — if the
11//! watchdog has not yet been cancelled — prints a message and calls
12//! `std::process::abort()`. That guarantees the process dies within
13//! `timeout + poll_interval` of the configured deadline, regardless of what
14//! the main thread is doing.
15//!
16//! # Design notes
17//!
18//! - Uses `Arc<AtomicBool>` for cancellation. Polling every 100 ms keeps the cancellation latency
19//!   low without burning CPU.
20//! - `Drop` on the `Watchdog` handle cancels the watchdog thread, so RAII usage (`let _wd =
21//!   Watchdog::new(secs)`) is sufficient.
22//! - We use `std::process::abort()` rather than `panic!` because panic may unwind or be caught by a
23//!   surrounding `catch_unwind`, which would defeat the safety guarantee. `abort()` delivers
24//!   `SIGABRT` and always terminates the process.
25//! - The existing cooperative `stomach::check_timeout()` path is retained: on most conversions it
26//!   fires before the hard abort, giving callers a nice `Err(Fatal)` with proper error propagation.
27//!   The watchdog is a safety net for the pathological cases where cooperative polling doesn't
28//!   happen.
29//!
30//! # Resource limits
31//!
32//! [`Watchdog::with_limits`](crate::watchdog::Watchdog::with_limits) guards
33//! **both** a wall-clock deadline and a
34//! resident-memory ceiling — the two defenses any executable that converts
35//! arbitrary input needs. It is the shared guard reused by both
36//! `cortex_worker` (in-process, one paper per process) and the
37//! `latexml_oxide --server` LSP (run inside each forked body child, which
38//! self-terminates on breach so the parent reaps it via pipe EOF). The exit
39//! codes are distinct so a supervising parent can tell them apart:
40//! `124` = wall-clock timeout, `137` = memory ceiling.
41//!
42//! # Portability
43//!
44//! The wall-clock guard is portable (`std::thread` + `Instant`).
45//!
46//! **Portable:** [`total_memory_bytes`](crate::watchdog::total_memory_bytes) and [`available_disk_bytes`](crate::watchdog::available_disk_bytes) answer on
47//! Linux, macOS and Windows — `sysconf(_SC_PHYS_PAGES)`/`statvfs` are POSIX, and
48//! Windows uses `GlobalMemoryStatusEx`/`GetDiskFreeSpaceExW`. They back the
49//! machine-derived default ceiling ([`default_ceiling_mib`](crate::watchdog::default_ceiling_mib)) and the
50//! spill-headroom check, so those behave identically on every supported OS.
51//!
52//! **Still Linux-only:** [`process_rss_kb`](crate::watchdog::process_rss_kb) samples `/proc/self/status`, so the
53//! *enforcement* half of the RAM guard is inactive elsewhere — the ceiling is
54//! computed correctly but never checked, and only the time guard bites. Closing
55//! this needs macOS `task_info(TASK_BASIC_INFO)` and Windows
56//! `GetProcessMemoryInfo`; both are reachable through the crates this module
57//! already links, so it is a bounded follow-up rather than a new dependency
58//! question.

69/// Current resident set size of this process in KiB, or `None` if it can't be
70/// determined (non-Linux, or `/proc` unavailable). Reads `VmRSS` from
71/// `/proc/self/status`. Cheap enough to poll a few times a second.
72pub fn process_rss_kb() -> Option<u64> {
73  let status = std::fs::read_to_string("/proc/self/status").ok()?;
74  for line in status.lines() {
75    if let Some(rest) = line.strip_prefix("VmRSS:") {
76      return rest.split_whitespace().next()?.parse::<u64>().ok();
77    }
78  }
79  None
80}

82/// Total physical RAM on this machine in bytes, or `None` if it cannot be
83/// determined.
84///
85/// Portable by construction: `sysconf(_SC_PHYS_PAGES) * sysconf(_SC_PAGE_SIZE)`
86/// is POSIX and answers on both Linux and macOS, and Windows has
87/// `GlobalMemoryStatusEx`. No new dependency — `libc` and `windows-sys` are
88/// already in the tree.
89///
90/// Deliberately NOT `/proc/meminfo`: that would repeat the Linux-only mistake
91/// this module already carries in [`process_rss_kb`], which returns `None`
92/// everywhere else and so silently deactivates the memory ceiling on
93/// macOS/Windows.
94pub fn total_memory_bytes() -> Option<u64> {
95  #[cfg(unix)]
96  {
97    // SAFETY: `sysconf` is a pure query with no pointer arguments; a negative
98    // return means "unavailable", which we map to `None` rather than trusting.
99    let pages = unsafe { libc::sysconf(libc::_SC_PHYS_PAGES) };
100    let page_size = unsafe { libc::sysconf(libc::_SC_PAGESIZE) };
101    if pages > 0 && page_size > 0 {
102      return (pages as u64).checked_mul(page_size as u64);
103    }
104    None
105  }
106  #[cfg(windows)]
107  {
108    use windows_sys::Win32::System::SystemInformation::{
109      GLOBAL_MEMORY_STATUS_EX, GlobalMemoryStatusEx,
110    };
111    let mut status: GLOBAL_MEMORY_STATUS_EX = unsafe { std::mem::zeroed() };
112    status.dwLength = std::mem::size_of::<GLOBAL_MEMORY_STATUS_EX>() as u32;
113    // SAFETY: `status` is a correctly sized, zeroed struct with `dwLength` set,
114    // exactly as the API requires.
115    if unsafe { GlobalMemoryStatusEx(&mut status) } != 0 {
116      return Some(status.ullTotalPhys);
117    }
118    None
119  }
120  #[cfg(not(any(unix, windows)))]
121  {
122    None
123  }
124}

126/// The default per-conversion memory ceiling in MiB, derived from the machine.
127///
128/// `min(64 GiB, 90 % of physical RAM)`, falling back to
129/// [`FALLBACK_CEILING_MIB`] when the machine cannot be probed.
130///
131/// The previous default was a flat 6144 MiB regardless of hardware, which is
132/// simultaneously absurd on a 256 GB host (a conversion that would comfortably
133/// fit is refused) and over-generous on an 8 GB laptop (the box starts swapping
134/// before the guard ever fires). Both halves of the rule matter:
135///
136/// * **90 % of RAM** leaves the OS and everything else on the machine a share.
137///   A ceiling at 100 % is a promise the kernel will not keep.
138/// * **the 64 GiB cap** is not about this machine but about the *others*: in a
139///   parallel fleet the aggregate is `N_processes x ceiling`, so an uncapped
140///   fraction-of-RAM rule on a big host would let a busy fleet OOM it. The
141///   `cortex_worker` fleet overrides this with its own per-child ceiling
142///   anyway; the cap keeps the single-process default from being reckless.
143pub fn default_ceiling_mib() -> u64 {
144  const MIB: u64 = 1024 * 1024;
145  match total_memory_bytes() {
146    Some(total) => {
147      let ninety_percent = total / MIB * 9 / 10;
148      ninety_percent.clamp(1, MAX_DEFAULT_CEILING_MIB)
149    },
150    None => FALLBACK_CEILING_MIB,
151  }
152}
153
154/// Upper bound on the machine-derived default ceiling (64 GiB in MiB).
155pub const MAX_DEFAULT_CEILING_MIB: u64 = 64 * 1024;
156
157/// Ceiling used when the machine's RAM cannot be probed — the historical flat
158/// default, kept so an unprobeable platform behaves exactly as before rather
159/// than losing its guard entirely.
160pub const FALLBACK_CEILING_MIB: u64 = 6144;

162/// Free space in bytes on the filesystem holding `path`, or `None` if it cannot
163/// be determined. Used to check headroom before spilling intermediates to disk.
164pub fn available_disk_bytes(path: &std::path::Path) -> Option<u64> {
165  #[cfg(unix)]
166  {
167    use std::os::unix::ffi::OsStrExt;
168    let c_path = std::ffi::CString::new(path.as_os_str().as_bytes()).ok()?;
169    // SAFETY: zeroed `statvfs` is a valid initial state; `c_path` is a
170    // NUL-terminated string that outlives the call.
171    let mut stat: libc::statvfs = unsafe { std::mem::zeroed() };
172    if unsafe { libc::statvfs(c_path.as_ptr(), &mut stat) } != 0 {
173      return None;
174    }
175    // `f_bavail` is blocks available to unprivileged users — the honest figure,
176    // as `f_bfree` includes the root-reserved slice we cannot use.
177    (stat.f_bavail as u64).checked_mul(stat.f_frsize as u64)
178  }
179  #[cfg(windows)]
180  {
181    use std::os::windows::ffi::OsStrExt;
182
183    use windows_sys::Win32::Storage::FileSystem::GetDiskFreeSpaceExW;
184    let mut wide: Vec<u16> = path.as_os_str().encode_wide().collect();
185    wide.push(0);
186    let mut free_to_caller: u64 = 0;
187    // SAFETY: `wide` is NUL-terminated and outlives the call; the two unused
188    // out-parameters are passed as null, which the API permits.
189    let ok = unsafe {
190      GetDiskFreeSpaceExW(
191        wide.as_ptr(),
192        &mut free_to_caller,
193        std::ptr::null_mut(),
194        std::ptr::null_mut(),
195      )
196    };
197    (ok != 0).then_some(free_to_caller)
198  }
199  #[cfg(not(any(unix, windows)))]
200  {
201    let _ = path;
202    None
203  }
204}

206/// Exit code used when the wall-clock deadline is exceeded (standard `timeout`).
207pub const EXIT_TIMEOUT: i32 = 124;
208/// Exit code used when the memory ceiling is exceeded (128 + SIGKILL).
209pub const EXIT_OOM: i32 = 137;

211/// Handle to a watchdog thread. Cancels on drop.
212///
213/// `Watchdog::new(0)` is a no-op — produces a handle that does nothing. This
214/// lets call-sites set a watchdog conditionally without special-casing the
215/// "no timeout" branch.
216/// Optional hook to run from the watchdog thread immediately before
217/// `exit(124)`. Used by `cortex_worker --standalone` to write a
218/// structured `Status:conversion:3` placeholder to `--output` so the
219/// timeout produces a usable failure artifact instead of a missing
220/// file. Set once at startup via `set_pre_exit_hook`; the hook is
221/// invoked exactly once. Zero overhead on the happy path — only the
222/// watchdog firing reads it.
223type PreExitHook = Box<dyn FnOnce() + Send + 'static>;
224
225static PRE_EXIT_HOOK: std::sync::OnceLock<std::sync::Mutex<Option<PreExitHook>>> =
226  std::sync::OnceLock::new();
227
228pub fn set_pre_exit_hook(hook: PreExitHook) {
229  let cell = PRE_EXIT_HOOK.get_or_init(|| std::sync::Mutex::new(None));
230  if let Ok(mut guard) = cell.lock() {
231    *guard = Some(hook);
232  }
233}
234
235fn run_pre_exit_hook() {
236  if let Some(cell) = PRE_EXIT_HOOK.get()
237    && let Ok(mut guard) = cell.lock()
238    && let Some(hook) = guard.take()
239  {
240    hook();
241  }
242}

244pub struct Watchdog {
245  cancelled: Arc<AtomicBool>,
246}
247
248impl Watchdog {
249  /// Create a wall-clock-only watchdog. `timeout_secs = 0` disables it.
250  /// Equivalent to [`Watchdog::with_limits(timeout_secs, 0)`].
251  pub fn new(timeout_secs: u64) -> Self { Self::with_limits(timeout_secs, 0) }
252
253  /// Create a watchdog guarding a wall-clock deadline **and** a resident-memory
254  /// ceiling. `timeout_secs = 0` disables the time guard; `max_rss_kb = 0`
255  /// disables the memory guard. With both `0` this is a no-op handle.
256  ///
257  /// The thread polls `cancelled`, the deadline, and RSS every `poll_interval`.
258  /// On a time breach it exits [`EXIT_TIMEOUT`]; on a memory breach,
259  /// [`EXIT_OOM`]. The memory guard is inactive where [`process_rss_kb`]
260  /// returns `None` (non-Linux); see the module portability note.
261  pub fn with_limits(timeout_secs: u64, max_rss_kb: u64) -> Self {
262    let cancelled = Arc::new(AtomicBool::new(false));
263    if timeout_secs > 0 || max_rss_kb > 0 {
264      let c = cancelled.clone();
265      thread::Builder::new()
266        .name("latexml-watchdog".to_string())
267        .spawn(move || Self::run(c, timeout_secs, max_rss_kb))
268        .expect("watchdog thread spawn failed");
269    }
270    Self { cancelled }
271  }
272
273  fn run(cancelled: Arc<AtomicBool>, timeout_secs: u64, max_rss_kb: u64) {
274    let deadline = (timeout_secs > 0).then(|| Instant::now() + Duration::from_secs(timeout_secs));
275    let poll_interval = Duration::from_millis(100);
276    loop {
277      if cancelled.load(Ordering::Relaxed) {
278        return; // cancelled: graceful exit.
279      }
280      if let Some(deadline) = deadline
281        && Instant::now() >= deadline
282      {
283        if cancelled.load(Ordering::Relaxed) {
284          return;
285        }
286        eprintln!(
287          "Fatal:timeout:wallclock latexml-oxide: main-level wall-clock timeout after {timeout_secs}s — exiting process"
288        );
289        // Run the optional pre-exit hook (e.g. cortex_worker writing a
290        // structured Status:conversion:3 placeholder to its --output path)
291        // BEFORE exiting. The hook is invoked at most once per process.
292        run_pre_exit_hook();
293        // `std::process::exit(124)` instead of `abort()`: the watchdog must
294        // terminate the whole process (the worker thread is presumed wedged
295        // in a tight loop that won't observe a cooperative cancel), but
296        // `abort()` produces a "Aborted (core dumped)" SIGABRT trace from
297        // the shell. `exit(124)` (standard timeout exit code) runs atexit
298        // handlers, flushes stderr, and leaves a clean exit signal the
299        // parent harness can interpret as "paper timed out" without
300        // conflating it with a Rust panic / memory corruption. Witnesses:
301        // 2602.11915, 2604.11500, 2604.13944, hep-ph9205242, q-alg9604005,
302        // q-alg9605003, q-alg9605028 — the 7 "Aborted" rows in the
303        // 2026-05-13 588-paper sweep.
304        std::process::exit(EXIT_TIMEOUT);
305      }
306      if max_rss_kb > 0
307        && let Some(rss) = process_rss_kb()
308        && rss > max_rss_kb
309      {
310        if cancelled.load(Ordering::Relaxed) {
311          return;
312        }
313        eprintln!(
314          "Fatal:oom:rss latexml-oxide: resident memory {}MB exceeded the {}MB ceiling — exiting process",
315          rss / 1024,
316          max_rss_kb / 1024
317        );
318        run_pre_exit_hook();
319        std::process::exit(EXIT_OOM);
320      }
321      thread::sleep(poll_interval);
322    }
323  }
324
325  /// Explicitly cancel the watchdog. Idempotent.
326  pub fn cancel(&self) { self.cancelled.store(true, Ordering::Relaxed); }
327}
328
329impl Drop for Watchdog {
330  fn drop(&mut self) { self.cancel(); }
331}

337  #[test]
338  fn watchdog_zero_timeout_is_noop() {
339    // timeout_secs=0 should NOT spawn a thread and NOT abort.
340    let w = Watchdog::new(0);
341    assert!(
342      !w.cancelled.load(Ordering::Relaxed),
343      "initial cancelled state is false"
344    );
345    // Dropping is safe — there's no live thread to interact with.
346    drop(w);
347  }

349  #[test]
350  fn watchdog_cancel_is_idempotent() {
351    let w = Watchdog::new(60);
352    w.cancel();
353    assert!(w.cancelled.load(Ordering::Relaxed));
354    // Calling again is a no-op.
355    w.cancel();
356    assert!(w.cancelled.load(Ordering::Relaxed));
357  }

392  /// The ceiling is derived from the machine, so the machine must be probeable
393  /// on every platform we ship. A `None` here means the default silently falls
394  /// back to a flat number and the ceiling stops tracking the host — the exact
395  /// failure `process_rss_kb` already has on non-Linux.
396  #[test]
397  #[cfg(any(unix, windows))]
398  fn total_memory_is_probeable_and_plausible() {
399    let total = total_memory_bytes().expect("physical RAM must be probeable on unix/windows");
400    // No real machine we support has under 256 MiB, and none has over 100 TiB;
401    // a value outside that says the units are wrong, not that the host is odd.
402    assert!(
403      total > 256 * 1024 * 1024,
404      "implausibly small total RAM ({total} bytes) — check the unit conversion"
405    );
406    assert!(
407      total < 100 * 1024 * 1024 * 1024 * 1024,
408      "implausibly large total RAM ({total} bytes) — check the unit conversion"
409    );
410  }
411
412  /// `min(64 GiB, 90 % of RAM)`, and never zero — a zero ceiling would mean
413  /// "no limit" downstream (`resolve_rss_cap`), inverting the intent.
414  #[test]
415  fn default_ceiling_respects_both_halves_of_the_rule() {
416    let ceiling = default_ceiling_mib();
417    assert!(
418      ceiling > 0,
419      "a derived ceiling of 0 would read as 'unlimited'"
420    );
421    assert!(
422      ceiling <= MAX_DEFAULT_CEILING_MIB,
423      "derived {ceiling} MiB exceeds the {MAX_DEFAULT_CEILING_MIB} MiB cap; on a \
424       large host an uncapped fraction-of-RAM default would let a parallel fleet \
425       (N_processes x ceiling) OOM the machine"
426    );
427    if let Some(total) = total_memory_bytes() {
428      let ninety = total / (1024 * 1024) * 9 / 10;
429      assert_eq!(
430        ceiling,
431        ninety.clamp(1, MAX_DEFAULT_CEILING_MIB),
432        "ceiling should be min(cap, 90 % of RAM)"
433      );
434      assert!(
435        ceiling <= total / (1024 * 1024),
436        "the ceiling must leave the OS a share of physical RAM"
437      );
438    } else {
439      assert_eq!(ceiling, FALLBACK_CEILING_MIB);
440    }
441  }
