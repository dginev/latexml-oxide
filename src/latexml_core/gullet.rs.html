64fn csname_trace_lookahead(tag: &str, cs: &str, bad: &Token) -> Result<()> {
65  let mut ahead = Vec::new();
66  for _ in 0..40 {
67    match read_token()? {
68      Some(t) => ahead.push(t),
69      None => break,
70    }
71  }
72  eprintln!(
73    "CSNAME-TRACE({tag}) partial={cs:?} bad={bad} ahead={}",
74    Tokens::new(ahead.clone())
75  );
76  unread_vec(ahead);
77  Ok(())

120/// What a balanced read ([`read_balanced`]) does when it exhausts a mouth before
121/// the braces balance.
122///
123/// Perl has no such distinction: `Gullet.pm` L465-472 reads `$$self{mouth}
124/// ->readToken()` — the *current* mouth only — and `last`s at the boundary, so
125/// every Perl mouth is [`Opaque`](BalancedBoundary::Opaque). Rust crosses the
126/// boundary for token-level injections, which is a deliberate surpass-Perl
127/// divergence (xint, see [`read_balanced`]) and must stay narrow: crossing from a
128/// mouth that is *its own input* lets one runaway argument swallow everything
129/// after it.
130#[derive(PartialEq, Eq, Debug, Clone, Copy)]
131pub enum BalancedBoundary {
132  /// The mouth is a token-level continuation of the enclosing stream
133  /// (`\scantokens`, RawTeX): its `}` may legitimately live in the parent, so a
134  /// balanced read drains it and resumes there.
135  Transparent,
136  /// The mouth is a self-contained input. A balanced read stops at its end, as
137  /// Perl always does — an unbalanced argument loses the rest of *this* mouth
138  /// and nothing more. An Opaque mouth that is also a file level (catchfile's
139  /// opener) gets TeX's file-end recovery, like any file
140  /// ([`recover_at_file_end`]).
141  Opaque,
142}

144/// What the scanner is reading, for the recovery TeX makes when a file ends in
145/// the middle of it: tex.web §305 `scanner_status` (with `warning_index`, the
146/// command it names). A scan that owns a status sets it for its duration with
147/// [`set_scanner_status`] and restores the enclosing one after, as tex.web
148/// saves and restores it (§389 `macro_call`, §473 `scan_toks`); every nested
149/// read that sets none inherits it. [`recover_at_file_end`] consults it.
150///
151/// tex.web's `skipping` (conditional text) is not modelled: a skip that runs
152/// off an input level stops there, as Perl's does (`read_next_conditional`).
153#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
154pub enum ScannerStatus {
155  /// Not inside any of the scans below: a file's end is not a runaway.
156  #[default]
157  Normal,
158  /// `defining`: a definition body, `\def`/`\edef` (§473 `scan_toks(true, _)`).
159  Defining,
160  /// `matching`: a macro's arguments (§389-391 `macro_call`).
161  Matching,
162  /// `aligning`: an alignment preamble (§777 `init_align`).
163  Aligning,
164  /// `absorbing`: a balanced text, `\message`, `\write`, `\toks=`,
165  /// `\detokenize` … (§473 `scan_toks(false, _)`, eTeX `scan_general_text`).
166  Absorbing,
167}

192/// Restores the enclosing [`ScannerStatus`] (and `warning_index`) when it
193/// drops, on every exit path of the scan that set it, `?` included.
194/// The runaway argument ([`argument_runaway`]) is scoped the same way: a scan
195/// starts without one, and the enclosing scan's comes back when it ends.
196#[must_use = "the scanner status is restored when this guard drops"]
197pub struct ScannerStatusGuard {
198  status:            ScannerStatus,
199  warning_index:     Option<Token>,
200  runaway_argument:  Option<Runaway>,
201  par_ends_argument: bool,
202  ended:             bool,
203}

205impl ScannerStatusGuard {
206  /// End the scan now and hand back the macro argument that ran off a file
207  /// inside it, if one did ([`argument_runaway`]): the tokens it had read,
208  /// for [`abandon_runaway_call`]. A `macro_call` abandons itself then
209  /// (tex.web §392; `Expandable::read_call_arguments`).
210  pub fn end_taking_runaway_argument(mut self) -> Option<Runaway> {
211    let mut gullet = GULLET.borrow_mut();
212    gullet.scanner_status = self.status;
213    gullet.warning_index = self.warning_index.take();
214    PAR_ENDS_ARGUMENT.with(|c| c.set(self.par_ends_argument));
215    self.ended = true;
216    std::mem::replace(&mut gullet.runaway_argument, self.runaway_argument.take())
217  }
218}

220impl Drop for ScannerStatusGuard {
221  fn drop(&mut self) {
222    if self.ended {
223      return;
224    }
225    // `try_`: a guard dropped while unwinding from a panic that holds the
226    // gullet must not panic again; `initialize_gullet` resets the status.
227    if let Ok(mut gullet) = GULLET.try_borrow_mut() {
228      gullet.scanner_status = self.status;
229      gullet.warning_index = self.warning_index.take();
230      gullet.runaway_argument = self.runaway_argument.take();
231    }
232    PAR_ENDS_ARGUMENT.with(|c| c.set(self.par_ends_argument));
233  }
234}

288/// The input marks' watch on one mouth's pushback ([`input_mark`], [`input_unread_since`]): for each mark taken on the
289/// mouth, the shortest the pushback has been since — no token below a mark's length was read while that is no less
290/// than the length. Kept as runs: the marks from a run's serial up to the next run's share its shortest length, the
291/// serials increasing and the lengths never decreasing towards the top, so a mark costs at most one run and a
292/// shortening of the pushback one comparison, with the top run's length ([`floor`](Self::floor)); no token is copied.
293/// A run whose every mark is read — the pushback more than [`MARK_HELD`] shorter than its shortest mark — is dropped:
294/// its marks answer from the run below, no longer, or from none (read), and the reads below it are fast again.
295#[derive(PartialEq, Debug, Default)]
296struct PushbackWatch {
297  runs:  Vec<WatchRun>,
298  /// the top run's length, 0 when there is none: only a pushback shorter than it changes a run, so the fast read
299  /// (`read_internal_token_checked`) pops only above it and every other shortening tells [`note_len`](Self::note_len)
300  floor: usize,
301}
302
303/// A run of input marks on a [`PushbackWatch`].
304#[derive(PartialEq, Debug, Clone, Copy)]
305struct WatchRun {
306  /// the serial of its first mark
307  serial: u64,
308  /// the shortest the pushback has been since that mark
309  low:    usize,
310  /// the shortest of its marks' lengths: the run is live while the pushback has been at most [`MARK_HELD`] shorter
311  least:  usize,
312}
313
314impl WatchRun {
315  fn live(&self) -> bool { self.low + MARK_HELD >= self.least }
316}
317
318impl PushbackWatch {
319  /// The pushback is `len` tokens long now, shorter than it was.
320  #[inline(always)]
321  fn note_len(&mut self, len: usize) {
322    if len < self.floor {
323      self.lower(len);
324    }
325  }
326
327  /// [`note_len`](Self::note_len) below the floor: the runs longer than `len` become one at `len` (joined to the run
328  /// below when that is at `len` already), dropped when every mark of it is read.
329  #[cold]
330  #[inline(never)]
331  fn lower(&mut self, len: usize) {
332    let keep = self.runs.partition_point(|run| run.low <= len);
333    let mut lowered = self.runs.drain(keep..).reduce(|first, run| WatchRun {
334      least: first.least.min(run.least),
335      ..first
336    });
337    if let Some(run) = lowered.as_mut() {
338      run.low = len;
339    }
340    if let Some(run) = lowered.filter(WatchRun::live) {
341      match self.runs.last_mut() {
342        Some(below) if below.low == len => below.least = below.least.min(run.least),
343        _ => self.runs.push(run),
344      }
345    }
346    self.floor = self.runs.last().map_or(0, |run| run.low);
347  }
348
349  /// A mark `serial` taken with the pushback `len` tokens long: it joins the top run when that run is at `len` (the
350  /// top run is never longer than the pushback).
351  fn mark(&mut self, serial: u64, len: usize) {
352    match self.runs.last_mut() {
353      Some(top) if top.low == len => top.least = len,
354      _ => self.runs.push(WatchRun { serial, low: len, least: len }),
355    }
356    self.floor = len;
357  }
358
359  /// No longer than the shortest the pushback has been since the mark `serial` — that, while the mark's run is live —,
360  /// if a run of this mouth's holds a mark at or before it.
361  fn shortest_since(&self, serial: u64) -> Option<usize> {
362    let after = self.runs.partition_point(|run| run.serial <= serial);
363    after.checked_sub(1).map(|run| self.runs[run].low)
364  }
365}

624/// What [`read_budget`] began; the budget is off once this is ended or dropped.
625pub struct ReadBudget {
626  start: usize,
627  reads: usize,
628}

630impl ReadBudget {
631  /// The run is over: how many tokens it read, and whether it read all it may. A run the budget ended counts as read
632  /// what it read, as a run that finished does: the failed reads, and an inner [`reading_from_mouth`]'s reset of the
633  /// count on the way out, leave no mark on [`progress`](Gullet::progress). A run that ended otherwise keeps the count
634  /// as it found it. (A failed run's reset is that path's own.)
635  pub fn end(self) -> ReadBudgetEnd {
636    let mut g = gullet_mut!();
637    let at = g.read_budget_at.unwrap_or(usize::MAX);
638    let left = at.saturating_sub(g.progress);
639    let reads = self.reads.saturating_sub(left);
640    let over = g.read_budget_spent;
641    if over {
642      g.progress = self.start.saturating_add(reads);
643    }
644    drop(g);
645    ReadBudgetEnd { reads, over }
646  }
647}

721/// Scale the runaway-token backstop to the SOURCE size — cumulatively, over
722/// every file the conversion opens. The [`DEFAULT_TOKEN_LIMIT`] baseline is
723/// sized for arXiv-scale inputs; a book-scale source legitimately expands
724/// past it (witness: a 131 MB flat-index compilation died at the then-400M
725/// ceiling mid-digestion with no loop in sight, at a measured ~3+
726/// tokens/byte and paper-class documents measured up to ~160 tokens/byte;
727/// l3doc's crossref pass over source2e's `.dtx`s ~1000/byte). ×1000/byte
728/// keeps the ceiling finite — a true infinite loop still trips, and opens
729/// no new files while it spins — while never LOWERING the baseline for
730/// small documents. The bytes are summed over the main source AND
731/// everything it `\input`s/`\DocInclude`s (every `Mouth::open_file`):
732/// source2e.tex is a 15 KB driver over ~40 `.dtx` files (~2 MB), so the
733/// driver-only sizing never engaged. An explicit `LATEXML_TOKEN_LIMIT`
734/// wins unchanged (including 0 = disabled).
735pub fn scale_token_limit_to_source(source_bytes: usize) {
736  if std::env::var_os("LATEXML_TOKEN_LIMIT").is_some() {
737    return;
738  }
739  let mut gullet = gullet_mut!();
740  gullet.source_bytes_total = gullet.source_bytes_total.saturating_add(source_bytes);
741  let scaled = gullet
742    .source_bytes_total
743    .saturating_mul(1000)
744    .max(DEFAULT_TOKEN_LIMIT);
745  if let Some(limit) = gullet.token_limit.as_mut() {
746    *limit = (*limit).max(scaled);
747  }
748  gullet.full_read_at = 0;
749}

796/// Get the current location of input getting read
797pub fn get_locator() -> Locator {
798  let gullet = gullet!();
799  let mut runtime_opt = gullet.runtime.as_ref();
800  let mut mouthstack_iter = gullet.mouthstack.iter();
801  while runtime_opt.is_some() && runtime_opt.as_ref().unwrap().mouth.get_source().is_empty() {
802    runtime_opt = mouthstack_iter.next();
803  }
804  // The free fn stays `-> Locator` ("where the parser is now" — always a real
805  // position during digestion; the workhorse for errors + box creation). A
806  // Mouth's `get_locator` is `Option` (per the `Object` trait) but is always
807  // `Some`, so unwrap to the default only in the no-mouth backup.
808  if let Some(runtime) = runtime_opt {
809    // First exit condition: we found a mouth with a source, and asked it for a locator
810    runtime.mouth.get_locator().unwrap_or_default()
811  } else if let Some(runtime) = gullet.mouthstack.front() {
812    // Backup strategy: return the first locator in the mouthstack:
813    runtime.mouth.get_locator().unwrap_or_default()
814  } else {
815    // Final backup -- the default locator
816    Locator::default()
817  }
818}

890/// Push EXPANSION OUTPUT (macro bodies, conditional-branch replays, `\the`
891/// results) to the start of the token stream WITHOUT touching the alignment
892/// brace ledger.
893///
894/// tex.web distinguishes two insertion flavors: `back_input` (§325) RETRACTS
895/// `align_state` for a brace token because that token was already counted
896/// when scanned (`get_next` §342-344 counts every scanned brace, token lists
897/// included §357); `begin_token_list` — how an expansion enters the input —
898/// adjusts NOTHING, because expansion output was never scanned and its braces
899/// will be counted when they are. Retracting never-scanned braces here is not
900/// merely a transient skew: expl3 brace-tricks (`\if… { \else: } \fi:` under
901/// `\exp_after:wN`, the l3tl replace machinery) legitimately push UNBALANCED
902/// fragments whose skipped half was counted by the conditional skipper — the
903/// Perl-style retract-everything protocol (Gullet.pm L343-358, which Perl
904/// LaTeXML still has) then drifts `ALIGN_STATE` permanently negative and every
905/// later `&` in the alignment goes stray (l3doc `{function}` manuals; repro:
906/// `\tl_greplace_all:Nnn` + a protected macro emitting `&` inside a tabular
907/// cell). See `cluster_package_guards::alignment_ledger_expansion_pushback`.
908pub fn unread_expansion(tokens: Tokens) {
909  if let Some(ref mut runtime) = gullet_mut!().runtime {
910    let tokens = tokens.unlist();
911    runtime.pushback.reserve(tokens.len());
912    for token in tokens.into_iter().rev() {
913      runtime.pushback.push(token);
914    }
915  }
916}

987/// Unreads a `Vec<Token>` to the start of the token stream
988/// Perl: also adjusts ALIGN_STATE by retracting scanned braces (Gullet.pm lines 343-358).
989/// Correct ONLY for tokens that were previously READ (and hence counted) —
990/// for expansion output use [`unread_expansion`] (tex.web `begin_token_list`
991/// flavor, which adjusts nothing).
992pub fn unread_vec(tokens: Vec<Token>) {
993  let mut level: i64 = 0;
994  if let Some(ref mut runtime) = gullet_mut!().runtime {
995    // Reserve once, push each token in reverse-iteration order so the
996    // first element of `tokens` ends up at the stack top. Same
997    // semantics as the old VecDeque push_front loop, but without
998    // per-element head-pointer arithmetic.
999    runtime.pushback.reserve(tokens.len());
1000    for token in tokens.into_iter().rev() {
1001      match token.get_catcode() {
1002        Catcode::BEGIN => level -= 1, // Retract scanned braces
1003        Catcode::END => level += 1,
1004        _ => {},
1005      }
1006      runtime.pushback.push(token);
1007    }
1008  }

1077pub fn close_mouth(forced: bool) -> Result<()> {
1078  let mut shift_from_mouthstack = false;
1079  let mut error_has_more_input = false;
1080  if let Some(ref mut runtime) = runtime!()
1081    && !forced
1082    && (!runtime.pushback.is_empty() || runtime.mouth.has_more_input())
1083  {
1084    error_has_more_input = true
1085  }
1086  if error_has_more_input {
1087    let next = match read_token()? {
1088      Some(t) => t.stringify(),
1089      None => String::from("Empty"),
1090    };
1091    let message = s!("Closing mouth with input remaining '{}'", next);
1092    Error!("unexpected", next, message);
1093  }
1094  let mut wants_everyeof = false;
1095  {
1096    let mut gullet = gullet_mut!();
1097    if let Some(ref mut runtime) = gullet.runtime {
1098      runtime.mouth.finish();
1099      // A FORCED close is error-recovery/cleanup (`reading_from_mouth`
1100      // teardown) — never a file end, so it must not inject the payload
1101      // into a recovery stream.
1102      wants_everyeof = runtime.insert_everyeof && !forced && !runtime.eof_seen;
1103      shift_from_mouthstack = true;
1104    }
1105    if shift_from_mouthstack {
1106      gullet.runtime = gullet.mouthstack.pop_front();
1107      note_current_mouth(&gullet);
1108    }
1109  }
1110  // Insurance for a marked mouth closed before it ever ran dry (a forced
1111  // teardown is excluded above, and `\endinput` sets `eof_seen`): insert the
1112  // payload where reading continues. Token-identity preserved — expansion
1113  // flavor push.
1114  if wants_everyeof
1115    && let Ok(Some(RegisterValue::Tokens(eof_toks))) = lookup_register("\\everyeof", Vec::new())
1116    && !eof_toks.is_empty()
1117  {
1118    unread_expansion(eof_toks);
1119  }
1120  Ok(())
1121}
1122/// A file has just ended under the current read: tex.web §362 closes it and
1123/// calls §336 `check_outer_validity`, which, when the [`ScannerStatus`] is not
1124/// normal, reports the runaway (§338 "File ended while scanning definition /
1125/// use / preamble / text of \cs") and inserts the tokens that should lead to
1126/// recovery (§339).
1127///
1128/// Only an autoclose file level with an enclosing input ends like this: an
1129/// `\input` file, a file held in memory (filecontents, an `\openout` file),
1130/// catchfile's opener ([`crate::mouth::Mouth::is_file_level`]). A package or
1131/// class read through `reading_from_mouth` (not autoclose) and the main
1132/// document keep Perl's stop at their end; string mouths (`\scantokens`,
1133/// RawTeX) are no file levels (`read_balanced` crosses them).
1134///
1135/// - `defining`, `absorbing`: a `}` — the definition or text ends at the
1136///   file's end and the enclosing input goes on;
1137/// - `aligning`: `\cr` and `}` — the preamble, and with it the alignment, ends;
1138/// - `matching`: TeX inserts `\par` with `long_state:=outer_call`, and
1139///   `macro_call` abandons the call and drops its arguments on reading it
1140///   (§392 "Report a runaway argument and abort", which consumes the `\par`).
1141///   Modelled without the token: [`FileEndRecovery::Abandoned`] marks
1142///   [`argument_runaway`] and the reader stops; the report and the file's
1143///   close wait for the call ([`abandon_runaway_call`]), which alone knows
1144///   whether TeX would have stopped earlier, at a `\par`. Until then the ended
1145///   file stays current, so a reader loop that does not look at the mark meets
1146///   the same end again rather than reading on into the enclosing input.
1147///
1148/// The inserted tokens enter as `ins_list` inserts them, as input never
1149/// scanned (`unread_expansion`): the brace ledger counts the `}` when it is
1150/// read, which balances the definition's opening `{`.
1151pub fn recover_at_file_end() -> Result<FileEndRecovery> {
1152  let (status, warning_index, file_end, runaway) = {
1153    let gullet = gullet!();
1154    let file_end = !gullet.mouthstack.is_empty()
1155      && gullet
1156        .runtime
1157        .as_ref()
1158        .is_some_and(|r| r.autoclose && r.mouth.is_file_level());
1159    (
1160      gullet.scanner_status,
1161      gullet.warning_index,
1162      file_end,
1163      gullet.runaway_argument.is_some(),
1164    )
1165  };
1166  if status == ScannerStatus::Normal || !file_end {
1167    return Ok(FileEndRecovery::NotARunaway);
1168  }
1169  if status == ScannerStatus::Matching {
1170    // Met again by a reader that went on, the same end is the same runaway.
1171    if !runaway {
1172      gullet_mut!().runaway_argument = Some(Runaway::FileEnd(Vec::new()));
1173    }
1174    return Ok(FileEndRecovery::Abandoned);
1175  }
1176  let of = warning_index.map(|cs| s!(" of {cs}")).unwrap_or_default();
1177  let message = s!("File ended while scanning {}{of}", status.scanning_what());
1178  Error!("expected", "}", message);
1179  close_mouth(false)?;
1180  unread_expansion(if status == ScannerStatus::Aligning {
1181    // TeX inserts `frozen_cr` (§339), the primitive whatever `\cr` means now:
1182    // the engine's `\lx@frozen@cr`, a `\let` copy of the primitive taken when
1183    // `\cr` is defined (tex_tables.rs), so an author's `\def\cr` does not run
1184    // here (`is_column_end` takes the copy for `\cr`). Guard:
1185    // `perfect_kernel_batch65::file_end_in_a_preamble_inserts_the_frozen_cr`.
1186    let frozen_cr = T_CS!("\\lx@frozen@cr");
1187    let cr = if with_meaning(&frozen_cr, |defn| defn.is_some()) {
1188      frozen_cr
1189    } else {
1190      T_CS!("\\cr")
1191    };
1192    Tokens!(cr, T_END!())
1193  } else {
1194    Tokens!(T_END!())
1195  });
1196  Ok(FileEndRecovery::Inserted)
1197}

1209/// Abandon the call of `cs` whose argument ran off the end of a file, given
1210/// the `runaway` tokens it had read ([`ScannerStatusGuard::end_taking_runaway_argument`]).
1211///
1212/// TeX checks every token of a non-`\long` macro's argument for `\par`
1213/// (tex.web §392, §399) and at the first one abandons the call, "Paragraph
1214/// ended before \cs was complete", putting the `\par` back (§396
1215/// `back_error`): the call and its argument up to there are dropped, and the
1216/// rest is read again. The readers make that check while they read for a
1217/// macro whose long-ness is TeX's ([`Runaway::Paragraph`]); for LaTeXML's own
1218/// macros (Perl's `isLong` is stored, Expandable.pm:46, and never read) it is
1219/// made here, on the runaway only: the argument ran to the file's end, past
1220/// the `\par` where TeX stopped, and without the check the call would drop
1221/// everything after it too. The tokens from that `\par` on go back into the
1222/// file, which then ends later, under the enclosing scan.
1223///
1224/// Otherwise (a `\long` macro, or no `\par`) it is the file's end: "File ended
1225/// while scanning use of \cs" (§338-339), and the file is closed now, as
1226/// tex.web §362 closes it (`end_file_reading`) before recovering, so the next
1227/// read is from the enclosing input (`\expandafter\a\foo{…` gives `\a` the
1228/// next token after `\input`).
1229pub fn abandon_runaway_call(
1230  cs: Token,
1231  is_long: bool,
1232  checks_par: bool,
1233  runaway: Runaway,
1234) -> Result<()> {
1235  let runaway = match runaway {
1236    // The reader stopped at the `\par` (tex.web §392, §399): report it and put back only the `\par` (§396
1237    // `back_error`); the arguments read so far are dropped with the call.
1238    // The token goes back as TeX's `cur_tok`: a `\noexpand`ed `\par` without its mark (§358).
1239    Runaway::Paragraph(par) => {
1240      Error!(
1241        "unexpected",
1242        "\\par",
1243        s!("Paragraph ended before {cs} was complete")
1244      );
1245      let par = if par.noexpand_shadowed().is_some() {
1246        T_CS!("\\par")
1247      } else {
1248        par
1249      };
1250      unread_expansion(Tokens!(par));
1251      return Ok(());
1252    },
1253    Runaway::FileEnd(tokens) => tokens,
1254  };
1255  let par = T_CS!("\\par");
1256  // A macro that checks while it reads met no `\par` it should stop at (only one a delimiter accepted): the
1257  // file's end. The others are checked here, on the runaway only.
1258  match runaway
1259    .iter()
1260    .position(|t| !is_long && !checks_par && *t == par)
1261  {
1262    Some(at) => {
1263      Error!(
1264        "unexpected",
1265        "\\par",
1266        s!("Paragraph ended before {cs} was complete")
1267      );
1268      // Never scanned, as far as the brace ledger knows: the readers
1269      // retracted the argument's braces when they stopped.
1270      unread_expansion(Tokens::new(runaway[at..].to_vec()));
1271    },
1272    None => {
1273      Error!(
1274        "expected",
1275        "\\par",
1276        s!("File ended while scanning use of {cs}")
1277      );
1278      // Only the file that ended: still current, nothing left in it.
1279      let ended_file = {
1280        let mut gullet = gullet_mut!();
1281        let has_parent = !gullet.mouthstack.is_empty();
1282        gullet.runtime.as_mut().is_some_and(|r| {
1283          has_parent
1284            && r.autoclose
1285            && r.mouth.is_file_level()
1286            && r.pushback.is_empty()
1287            && !r.mouth.has_more_input()
1288        })
1289      };
1290      if ended_file {
1291        close_mouth(false)?;
1292      }
1293    },
1294  }
1295  Ok(())
1296}

1298/// This flushes a mouth so that it will be automatically closed, next time it's read
1299/// Corresponds to TeX's \endinput
1300pub fn flush_mouth() {
1301  if let Some(ref mut runtime) = runtime!() {
1302    // Collect remaining mouth tokens in mouth order (t1, t2, t3, …),
1303    // then splice them into the stack's BOTTOM in reverse order so
1304    // that after the stack's existing top is popped, the mouth tokens
1305    // come out in the original mouth order (t1 first, then t2, …).
1306    // Only the current line's: a comment ends it, and with source comments off (OD #2) the Mouth reads past a `%` to
1307    // the next line, whose tokens TeX never reads (tex.web §362). atbegshi.sty:557's
1308    // `\AtBegShi@AbortIfUndefined{AtBeginDocument}%`, whose `\endinput` comes from the macro's expansion, read on
1309    // into line 558 under the restored `@` catcode (`\AtBegShi` undefined, the file's definitions unbalanced: math0305287, a
1310    // plain TeX paper loading pstricks through pgfutil-plain.def; guard
1311    // `perfect_kernel_batch65f::endinput_from_a_macro_ends_the_file_at_the_line`).
1312    let line = runtime.mouth.line_number();
1313    let mut trailer: Vec<Token> = Vec::new();
1314    while !runtime.mouth.is_eol() {
1315      if let Some(token) = runtime.mouth.read_token() {
1316        if runtime.mouth.line_number() != line {
1317          break;
1318        }
1319        trailer.push(token);
1320      }
1321    }
1322    if !trailer.is_empty() {
1323      trailer.reverse();
1324      // (every token the pushback held moves up a slot: as if read, for the input marks)
1325      runtime.watch.note_len(0);
1326      runtime.pushback.splice(0..0, trailer);
1327    }
1328    // Stop reading (clear buffers, close file) but do NOT restore catcodes.
1329    // Catcodes are restored by close_mouth → finish() when the mouth is
1330    // properly popped from the stack.
1331    runtime.mouth.stop_reading();
1332    // eTeX inserts `\everyeof` only when a file really ends: `\endinput` sets
1333    // `force_eof`, and §362 then ends the file without the list (pdflatex
1334    // `[X]` for `X\endinput`; the payload came out as `[X[EOF]]`). Guard:
1335    // `noexpand_input_ends::endinput_does_not_insert_everyeof`.
1336    runtime.eof_seen = true;
1337  }
1338}

1402/// Outcome of [`read_internal_token_checked`].
1403enum CheckedRead {
1404  /// The gullet has no runtime — early shutdown / recovery from a fatal
1405  /// error. Treated as end-of-input instead of panicking. Driver: 2404.06289
1406  /// (natbib \NAT@@wrout cascade landed here after the conversion was already

1408  NoRuntime,

1411  Exhausted,

1416/// Duty cycle for the ACTIVE cycle guard (`progress` above the activation
1417/// floor): fingerprint + scan only [`CYCLE_GUARD_DUTY_ON`] of every
1418/// [`CYCLE_GUARD_DUTY_PERIOD`] tokens. A genuine infinite expansion loop is
1419/// *persistent*, so scanning periodic windows still detects it — worst case
1420/// one duty period plus a ring fill later (~17k tokens), four orders of
1421/// magnitude before the 1G `token_limit` backstop — while a legitimately
1422/// huge healthy stream (pgfplots data plots routinely read 200M+ tokens, past
1423/// even the raised graphics floor) stops paying a per-token fingerprint tax
1424/// for the whole remainder of the run (`cycle_guard_checkpoint` was 6.07% of
1425/// self-time on witness 2405.14114 before duty-cycling). The ring is reset at
1426/// each ON-window start so a window never straddles an OFF gap (no cross-gap
1427/// phase artifacts). The ON width covers the ring capacity
1428/// (`MAX_WINDOW × REPEAT = 1000`) plus detection cadence (`CHECK_EVERY`) for
1429/// every period, so any loop already running at an ON-window start is caught
1430/// within that same window. Bursts that end on their own inside an OFF window
1431/// are, by definition, not infinite loops — the guard's only target.
1432const CYCLE_GUARD_DUTY_PERIOD: usize = 16_384; // power of two (masked below)
1433const CYCLE_GUARD_DUTY_ON: usize = 2_048;
1434
1435/// Per-token combined step for all FOUR reader loops (`read_token`,
1436/// `read_x_token`, `read_balanced`, `read_next_conditional`): resource
1437/// checkpoint, low-level pushback/mouth read, and (duty-cycled) cycle-guard
1438/// fingerprint in ONE `GULLET` borrow. The loops are siblings, NOT a
1439/// delegation chain, so each runs this same step — otherwise full-expansion
1440/// paths (`\edef`, csname construction, conditional skipping) bypass every
1441/// gullet guard and a runaway grinds to the watchdog (gap found on
1442/// math0402448). Previously three separate borrows per token
1443/// (`read_resource_checkpoint` → `read_internal_token` →
1444/// `cycle_guard_checkpoint`, together ~10% of self-time on the pgfplots
1445/// witness 2405.14114); the single borrow is the point of the merge.
1446///
1447/// Resource accounting:
1448/// - `progress` counts UNCONDITIONALLY: the cycle guard's activation gate
1449///   feeds off it, so nesting the increment inside the token-limit branch
1450///   made `LATEXML_TOKEN_LIMIT=0` (and `set_token_limit(None)` during format
1451///   init) silently disable the cycle guard as well — exactly when an
1452///   operator disables the limit to let a big document through is when the
1453///   loop guard matters most. (PR #249 review P2-5.) Only the limit
1454///   COMPARISON stays conditional.
1455/// - Limit breaches Fatal via the `#[cold]` outlined helpers below, with the
1456///   borrow dropped first (`Fatal!` runs error machinery). A breach consumes
1457///   no token: the read is skipped, exactly as when the old checkpoint fired
1458///   before the read.
1459///
1460/// Cycle guard: fingerprints the token AS READ (before alignment /
1461/// `\dont_expand` special-casing), so the guard sees the same stream
1462/// regardless of which loop reads it. The reading-context serial is mixed
1463/// into the fingerprint (see `Gullet::ctx_serial`) so windows never match
1464/// across `reading_from_mouth` contexts; a token read from the Mouth (input,
1465/// not expansion output) also carries its source position. Activation is gated on
1466/// `progress > cycle_guard_activate` AND the duty window above.
1467#[inline]
1468fn read_internal_token_checked(sink: CommentSink) -> Result<CheckedRead> {
1469  {
1470    let mut borrow = gullet_mut!();
1471    let g: &mut Gullet = &mut borrow;
1472    // The usual read: nothing is due at this count (`full_read_at`), and the token on top of the pushback is an
1473    // ordinary one. It does exactly what the full read does in that case, without testing each thing that isn't due.
1474    // (The input marks' floor, `PushbackWatch`, is its emptiness test: 0 with no live mark; a read at or below it is
1475    // the full read's, which tells the watch.)
1476    if g.progress + 1 < g.full_read_at
1477      && (matches!(sink, CommentSink::Pending) || g.pending_comments.is_empty())
1478      && let Some(rt) = g.runtime.as_mut()
1479      && g
1480        .pushback_limit
1481        .is_none_or(|limit| rt.pushback.len() <= limit)
1482      && rt.pushback.len() > rt.watch.floor
1483      && let Some(&token) = rt.pushback.last()
1484      && !matches!(token.get_catcode(), Catcode::COMMENT | Catcode::MARKER)
1485    {
1486      rt.pushback.pop();
1487      g.progress += 1;
1488      return Ok(CheckedRead::Tok(token));
1489    }
1490  }
1491  read_internal_token_full(sink)
1492}

1494/// [`read_internal_token_checked`] in full, whatever is due: see that function's documentation above.
1495#[inline(never)]
1496fn read_internal_token_full(mut sink: CommentSink) -> Result<CheckedRead> {
1497  enum Breach {
1498    TokenLimit(usize),
1499    PushbackLimit(usize),
1500    Cycle(usize, Token),
1501    Deadline,
1502    ReadBudget,
1503  }
1504  let mut breach: Option<Breach> = None;
1505  let mut outcome = CheckedRead::Exhausted;
1506  {
1507    let mut borrow = gullet_mut!();
1508    let g: &mut Gullet = &mut borrow;
1509    if g.runtime.is_none() {
1510      return Ok(CheckedRead::NoRuntime);
1511    }
1512    g.progress += 1;
1513    if g.progress >= g.full_read_at {
1514      g.full_read_at = g.next_full_read();
1515    }
1516    // (`next_due_read` stands in for these tests on the fast path, and `token_limit_tests::is_due` spells them: change
1517    // all three together.)
1518    let rt = g.runtime.as_mut().unwrap();
1519    if let Some(limit) = g.token_limit
1520      && g.progress > limit
1521    {
1522      breach = Some(Breach::TokenLimit(limit));
1523    } else if let Some(limit) = g.pushback_limit
1524      && rt.pushback.len() > limit
1525    {
1526      breach = Some(Breach::PushbackLimit(limit));
1527    } else if g.progress & DEADLINE_CHECK_MASK == 0 && crate::stomach::conversion_deadline_passed()
1528    {
1529      // The cooperative conversion deadline (`stomach::set_timeout`), checked here too: an expansion that never
1530      // returns to a digestion loop never met it, and only a process watchdog, where one is armed, ended the run
1531      // (OD #538).
1532      breach = Some(Breach::Deadline);
1533    } else if g.read_budget_at.is_some_and(|at| g.progress > at) {
1534      g.read_budget_spent = true;
1535      breach = Some(Breach::ReadBudget);
1536    } else {
1537      // Duty-cycled guard activation, computed BEFORE the read: the phase —
1538      // and the ON-window-start ring reset — depends only on `progress`,
1539      // which is already final for this iteration. An Exhausted read landing
1540      // exactly on phase 1 must still reset the ring, or the next window
1541      // would splice onto the previous ON-window's history (adversarial-
1542      // review finding, 2026-07-29).
1543      let duty_on = g.progress > g.cycle_guard_activate && {
1544        let phase = (g.progress - g.cycle_guard_activate) & (CYCLE_GUARD_DUTY_PERIOD - 1);
1545        if phase == 1 {
1546          // ON-window start: drop any pre-gap history.
1547          g.cycle_guard.reset();
1548        }
1549        (1..=CYCLE_GUARD_DUTY_ON).contains(&phase)
1550      };
1551      if let CommentSink::Into(ref mut out) = sink {
1552        // read_balanced keeps comments in its result: flush comments stashed
1553        // by earlier reads before reading fresh ones (same order as the old
1554        // inline loop's `pending_comments` drain at its top).
1555        if !g.pending_comments.is_empty() {
1556          out.extend(g.pending_comments.drain(..));
1557        }
1558      }
1559      // The raw read: pushback first, then the current Mouth. COMMENT tokens
1560      // go to the sink, MARKER tokens are handled inline (handle_marker only
1561      // touches the align-count thread-locals, never GULLET — it already ran
1562      // under this borrow in the old `read_internal_token`). `route` is the
1563      // shared per-token filter: `Some(t)` = a real token, `None` = consumed
1564      // (comment/marker).
1565      let pending = &mut g.pending_comments;
1566      let mut route = |t: Token| -> Option<Token> {
1567        match t.get_catcode() {
1568          Catcode::COMMENT => {
1569            match sink {
1570              CommentSink::Pending => pending.push_back(t),
1571              CommentSink::Into(ref mut out) => out.push(t),
1572            }
1573            None
1574          },
1575          Catcode::MARKER => {
1576            handle_marker(t);
1577            None
1578          },
1579          _ => Some(t),
1580        }
1581      };
1582      let mut next_token: Option<Token> = None;
1583      while let Some(t) = rt.pushback.pop() {
1584        if let Some(t) = route(t) {
1585          next_token = Some(t);
1586          break;
1587        }
1588      }
1589      rt.watch.note_len(rt.pushback.len());
1590      // Whether the token is expansion output (the pushback, `\everyeof`'s tokens included) rather than the current
1591      // Mouth's own input, for the cycle guard below.
1592      let mut expanded = next_token.is_some();
1593      if next_token.is_none() {
1594        while let Some(t) = rt.mouth.read_token() {
1595          if let Some(t) = route(t) {
1596            next_token = Some(t);
1597            break;
1598          }
1599        }
1600      }
1601      // eTeX file end (tex.web §362 + etex.ch `every_eof`): the first time a
1602      // marked mouth runs dry, its `\everyeof` tokens are read AS THE FILE'S
1603      // LAST TOKENS — pushed at this level, before the mouth closes. Token
1604      // identity preserved (l3tl's rescan quarks, xint's captures). spreadtab
1605      // 's `\ST_scan_func_arg` (spreadtab.sty:1493-1504) ends its `\scantokens`
1606      // scan on `\everyeof{\ST_nil}`; a parent-level insertion after the close
1607      // (the earlier designs) reached a nested scan in the enclosing stream.
1608      if next_token.is_none() && rt.insert_everyeof && !rt.eof_seen {
1609        rt.eof_seen = true;
1610        if let Ok(Some(RegisterValue::Tokens(eof_toks))) = lookup_register("\\everyeof", Vec::new())
1611          && !eof_toks.is_empty()
1612        {
1613          rt.pushback.extend(eof_toks.unlist().into_iter().rev());
1614          while let Some(t) = rt.pushback.pop() {
1615            if let Some(t) = route(t) {
1616              next_token = Some(t);
1617              expanded = true;
1618              break;
1619            }
1620          }
1621          rt.watch.note_len(rt.pushback.len());
1622        }
1623      }
1624      if let Some(token) = next_token {
1625        // Record BEFORE the activation gate: the ring is also what the
1626        // token-limit fatal dumps, and that fatal fires precisely for
1627        // runaways the cycle guard never recognised — which includes ones
1628        // that trip a lowered `LATEXML_TOKEN_LIMIT` before the guard ever
1629        // activates. Gating the ring on activation left it empty in exactly
1630        // that case. Debug-only, so free in production.
1631        if *DEBUG_FATAL {
1632          DEBUG_RECENT_TOKENS.with(|ring| {
1633            let mut ring = ring.borrow_mut();
1634            if ring.len() >= 512 {
1635              ring.pop_front();
1636            }
1637            ring.push_back(format!("{token:?}"));
1638          });
1639        }
1640        if duty_on {
1641          // Mix the reading-context serial into the fingerprint (see the
1642          // `Gullet::ctx_serial` field doc): tokens read in different
1643          // `reading_from_mouth` contexts can then never form a matching
1644          // window, scoping cycle detection to ONE expansion context
1645          // without destroying the outer context's history. The multiplier
1646          // spreads the serial across the hash bits (splitmix64's odd
1647          // constant).
1648          //
1649          // A token read from the Mouth is input, which advances: its fingerprint also carries where it starts in
1650          // the file (line, column), so a file's repeated text is no period. Fingerprinted by meaning alone, it was a
1651          // loop once the activation floor was passed: 1009.1698's gnuplot label file, 626 lines of
1652          // `\color{black}%` read as one argument with the pushback empty, a window of 8 tokens "repeated 100+ times"
1653          // (repro expansion-primitives/cycle_guard_repeated_file_input.tex). A loop that re-reads the same text
1654          // through a new Mouth reads it at the same places every period, and is still one (a nesting one, as
1655          // `\def\x{\scantokens{\x}}`, stops first at TeX's input-level limit, `mark_everyeof_mouth`).
1656          let mut fp = token.cycle_fingerprint() ^ g.ctx_serial.wrapping_mul(0x9E37_79B9_7F4A_7C15);
1657          if !expanded {
1658            let (line, column) = rt.mouth.last_token_start();
1659            fp ^= (((line as u64) << 32) ^ column as u64).wrapping_mul(0xBF58_476D_1CE4_E5B9);
1660          }
1661          if let Some(period) = g.cycle_guard.push(fp, rt.pushback.len()) {
1662            breach = Some(Breach::Cycle(period, token));
1663          }
1664        }
1665        if breach.is_none() {
1666          outcome = CheckedRead::Tok(token);
1667        }
1668      }
1669    }
1670  }
1671  // Borrow dropped; cold outlined fatal paths only from here on.
1672  match breach {
1673    None => Ok(outcome),
1674    Some(Breach::TokenLimit(limit)) => token_limit_fatal(limit),
1675    Some(Breach::PushbackLimit(limit)) => pushback_limit_fatal(limit),
1676    Some(Breach::Cycle(period, token)) => cycle_trip_fatal(period, &token),
1677    Some(Breach::Deadline) => fatal!(Timeout, Convert, "Conversion timed out!"),
1678    Some(Breach::ReadBudget) => read_budget_error(),
1679  }
1680}

1708/// The first count past `read` at which [`read_internal_token_full`]'s tests have something to do, by the count alone:
1709/// past the token `limit`, a multiple of the deadline's period, or in one of the cycle guard's windows, phases
1710/// 1..=`CYCLE_GUARD_DUTY_ON` of a period past its `floor`.
1711fn next_due_read(read: usize, limit: Option<usize>, floor: usize) -> usize {
1712  let deadline = (read | DEADLINE_CHECK_MASK) + 1;
1713  let limit = limit.map_or(usize::MAX, |limit| limit.saturating_add(1));
1714  let guard = match read.checked_sub(floor) {
1715    None => floor + 1,
1716    Some(past) => {
1717      let phase = past & (CYCLE_GUARD_DUTY_PERIOD - 1);
1718      if phase < CYCLE_GUARD_DUTY_ON {
1719        read + 1
1720      } else {
1721        read + (CYCLE_GUARD_DUTY_PERIOD - phase) + 1
1722      }
1723    },
1724  };
1725  deadline.min(limit).min(guard)
1726}

1825      return Ok(Some(token));
1826    }
1827    let crossable = {
1828      let gullet = gullet!();
1829      gullet.runtime.as_ref().is_some_and(|r| r.autoclose) && !gullet.mouthstack.is_empty()
1830    };
1831    if !crossable {
1832      return Ok(None);
1833    }
1834    close_mouth(false)?;
1835  }

1838/// Read a token as a primitive reads one: tex.web §365 `get_token` under the
1839/// current [`ScannerStatus`] (`\let`, `\futurelet`, `\afterassignment`,
1840/// `\aftergroup`, `\expandafter`, a `Token` parameter). An input level
1841/// that ends here closes (§362 `end_file_reading`) and §336
1842/// `check_outer_validity` looks at the status:
1843/// - `normal`: no runaway; the read goes on in the enclosing input, as
1844///   [`read_token_across_input_ends`] reads, so a primitive that is the last
1845///   token of an `\input` file takes its token from the input that follows;
1846/// - `defining`, `absorbing`, `aligning`: the scan in progress ran off the file,
1847///   and the primitive's read meets it: the runaway is reported there and
1848///   TeX's recovery tokens are read ([`recover_at_file_end`]), so an
1849///   `\edef\x{\expandafter` at a file's end reads the inserted `}` and
1850///   `\x` is empty, as in pdflatex;
1851/// - `matching`: TeX reads a macro's arguments without expanding, so it gives
1852///   no primitive's read this status, but Rust does: an expandable primitive
1853///   with typed parameters (`\csname CSName`, `\number Number`, `\readline …
1854///   Token`) reads them at `matching`, set by `Expandable::read_call_arguments`.
1855///   The read stops at the file's end (a residual: `\csname a\expandafter` as
1856///   a file's last tokens does not cross), and so does a macro's `Token`
1857///   argument (the macro gets `\relax`, the `Token` parameter's end value).
1858///
1859/// `\ifx` and `\ifdefined` read at `normal` whatever the status (§507), with
1860/// [`read_token_across_input_ends`]. The status is consulted only once the
1861/// current input has run out.
1862///
1863/// Guards: `token_kernel_gaps::*`.
1864pub fn read_primitive_token() -> Result<Option<Token>> {
1865  if let Some(token) = read_token()? {
1866    return Ok(Some(token));
1867  }
1868  match scanner_status() {
1869    ScannerStatus::Normal => read_token_across_input_ends(),
1870    ScannerStatus::Matching => Ok(None),
1871    _ => loop {
1872      if let Some(token) = read_token()? {
1873        return Ok(Some(token));
1874      }
1875      if recover_at_file_end()? != FileEndRecovery::Inserted {
1876        return Ok(None);
1877      }
1878    },
1879  }
1880}

1882/// The control sequence a definition names: tex.web §1215 `get_r_token`,
1883/// `repeat get_token until cur_tok<>space_token` (`\def`, `\let`, `\futurelet`,
1884/// `\chardef`, `\countdef` …). Each read is [`read_primitive_token`], so a
1885/// bare `\let` or `\def` as a file's last token takes its name from the input
1886/// after the file, skipping the spaces there.
1887///
1888/// A name that is neither a control sequence nor an active character
1889/// (`cur_cs=0`) is "Missing control sequence inserted": the token is read
1890/// again (`back_input`) and the frozen `\inaccessible` (§222
1891/// `frozen_protection`) is named instead, so `\gdef{}X{Y}` defines it and
1892/// typesets `X{Y}`. yquant-doc.tex:3832's `\gdef{…` (an `\edef`-expanded
1893/// `\pgfsyssoftpath@flushbuffers`) had installed a macro on the Begin catcode,
1894/// and every later `{` read a delimited argument: 1001 errors and a Fatal.
1895/// Perl reads the name as a `Token` (TeX_Macro.pool.ltxml:174-177) and shares
1896/// that. A frozen control sequence as the name (`cur_cs>frozen_control_sequence`,
1897/// dropped with the same error) is not modelled: Rust has none.
1898///
1899/// A `\noexpand`ed name is the control sequence it marks: §358 `get_next`
1900/// sets `cur_cs` to the shadowed token, so `\expandafter\def\noexpand\foo`
1901/// defines `\foo`. Rust's marker is a `\special_relax` family token, which
1902/// had defined the family's shared `\relax` meaning instead.
1903///
1904/// Guards: `rtoken_patchcmd::def_of_a_non_cs_defines_inaccessible`,
1905/// `rtoken_patchcmd::let_family_of_a_non_cs_reads_the_token_again`,
1906/// `rtoken_patchcmd::noexpand_marked_name_is_its_control_sequence`.
1907pub fn read_redefinable_token() -> Result<Option<Token>> {
1908  loop {
1909    match read_primitive_token()? {
1910      Some(token) if token.get_catcode() == Catcode::SPACE => {},
1911      Some(token) if token.is_noexpand_family() => {
1912        return Ok(Some(token.noexpand_shadowed().unwrap_or(token)));
1913      },
1914      Some(token) if !token.get_catcode().is_active_or_cs() => {
1915        Error!(
1916          "expected",
1917          "<cs>",
1918          "Missing control sequence inserted",
1919          s!("{} cannot be a definition's name", token.stringify()),
1920          "Please don't say `\\def cs{...}', say `\\def\\cs{...}'.\n\tI've inserted an \
1921           inaccessible control sequence so that your definition will be completed without \
1922           mixing me up too badly."
1923        );
1924        unread_one(token);
1925        // `\inaccessible ` is how TeX prints it (§262 `print_cs`); the
1926        // trailing space keeps it apart from a document's `\inaccessible`
1927        // (not from `\csname inaccessible\space\endcsname`, nor from a
1928        // `\DeclareRobustCommand\inaccessible`'s inner macro).
1929        return Ok(Some(T_CS!("\\inaccessible ")));
1930      },
1931      other => return Ok(other),
1932    }
1933  }
1934}

1954      format!("input ended while scanning use of {what}")

1956  }
1957  Ok(tok)

1966    && !(token.code == Catcode::CS && token.text == pin!("\\dont_expand"))

1969      Catcode::BEGIN => increment_align_group_count(),
1970      Catcode::END => decrement_align_group_count(),

1973    return Ok(Some(token));

1978/// [`read_token`] given what its first read found, in full: the alignment and `\dont_expand` cases.
1979#[inline(never)]
1980fn read_token_after(first: CheckedRead) -> Result<Option<Token>> {
1981  let mut next_token: Option<Token>;
1982  let mut read = Some(first);
1983  loop {
1984    // Combined checkpoint + raw read + cycle fingerprint, one borrow.
1985    let this_read = match read.take() {
1986      Some(this_read) => this_read,
1987      None => read_internal_token_checked(CommentSink::Pending)?,
1988    };
1989    next_token = match this_read {
1990      CheckedRead::NoRuntime => return Ok(None),
1991      // A marked eTeX pseudo-file (`\scantokens`) end is NOT crossed here:
1992      // its `\everyeof` payload was already read at the file's own level
1993      // (`read_internal_token_checked`, `eof_seen`), so this is the true end
1994      // — a delimited scan whose marker misses must not escape into the
1995      // enclosing LIVE stream (inside an alignment cell it swallowed the
1996      // column-after program as data and poisoned l3tl rescan results: l3doc
1997      // `\tl_replace_all` 5-token loop, witnesses spath3/litetable-zh/

1999      // crosses via read_x_token's autoclose drain.
2000      CheckedRead::Exhausted => None,
2001      CheckedRead::Tok(t) => Some(t),
2002    };

2004

2006    //   alignment column-end check → \dont_expand check → break

2008    //   token to be returned (Perl L320-324).
2009    if let Some(ref nextt) = next_token {
2010      if (align_group_count() == 0)
2011        && has_reading_alignment()

2021          handle_template(data.borrow_mut(), *nextt, atype, ahidden)?;
2022        } else {
2023          return Err("reading_alignment should always contain DigestedData::Alignment".into());
2024        }
2025        continue; // Perl: handleTemplate then continue while(1) loop
2026      }
2027      if nextt.code == Catcode::CS && nextt.text == pin!("\\dont_expand") {
2028        // `\noexpand <tok>`: collapse to the per-token `\special_relax` family,
2029        // encoding <tok>'s identity in the name (faithful to TeX's
2030        // `no_expand_flag`, which keeps `cur_cs`). End-of-input ⇒ bare
2031        // `\special_relax` (nothing to shadow).
2032        next_token = Some(match read_token()? {
2033          Some(tok) => crate::token::noexpand_family(&tok),
2034          None => T_CS!("\\special_relax"),
2035        });
2036      }
2037      break;
2038    } else {
2039      break;
2040    }
2041  }
2042  // Perl Gullet.pm L320-324: ALIGN_STATE tracking happens AFTER the loop,
2043  // applied only to the FINAL returned token. Previously this was inside
2044  // the loop BEFORE the alignment check, which prevented column-end
2045  // template handling on `{` tokens (count became 1 before the check).
2046  if let Some(ref nextt) = next_token {
2047    match nextt.get_catcode() {
2048      Catcode::BEGIN => increment_align_group_count(),
2049      Catcode::END => decrement_align_group_count(),
2050      _ => {},
2051    }
2052  }
2053  Ok(next_token)
2054}

2116/// RAII depth counter for `read_x_token`: `enter` increments (Fatals past the
2117/// limit), drop decrements — so every return path stays balanced.

2126      return Self::too_deep(d, limit);
2127    }
2128    Ok(ExpandDepthGuard)
2129  }

2142  }

2275/// Whether none of the input `mark` held has been read since it was taken ([`input_mark`]): its mouth is still open,

2277/// was read, or what a mouth opened since holds). Read-only, for an observer telling apart what a call's expansion
2278/// did from what the input after the call did (the frontmatter observer's ledger, the title-page check's document
2279/// calls). The mouth's watch tells how short its pushback has been since the mark (`PushbackWatch`): no shorter
2280/// than the mark's length, nothing it held was read; shorter by at most `MARK_HELD` tokens, those read were put
2281/// back if the slots hold the very tokens they held (`same_token`: a peek leaves the input unread); shorter still,
2282/// it was read. Tokens read below the mark and others pushed in their place read as read, though the token now at
2283/// the mark's last slot equals the one it held — IEEEtran's `\IEEEauthorblockN{…}⏎\IEEEauthorblockA{…}`, whose
2284/// line-end space was read with `\IEEEauthorblockA` and its argument and whose expansion, longer, put another space
2285/// there: its affiliation calls read as the author block's own and were lost by the frontmatter emitter (2401.11731,

2292    return false;
2293  }

2295    gullet.runtime.as_ref()
2296  } else {
2297    gullet.mouthstack.get(depth - mark.depth - 1)
2298  };
2299  let Some(rt) = runtime else {
2300    return mark.len == 0 && mark.mouth == (0, 0, 0, 0);

2302  if rt.mouth.read_position() != mark.mouth || rt.pushback.len() < mark.len {
2303    return false;
2304  }

2307  };

2310    || (shortest >= held_from
2311      && rt.pushback[shortest..mark.len]
2312        .iter()
2313        .zip(&mark.held[shortest - held_from..])
2314        .all(|(now, then)| then.is_some_and(|then| same_token(now, &then))))

2393/// Expand `token` (defined as `defn`) ONE level and push the expansion back
2394/// onto the input, as `read_x_token` does for each expandable token it meets.
2395#[inline]
2396fn invoke_expansion(token: Token, defn: &std::rc::Rc<dyn Definition>) -> Result<()> {
2397  local_current_token(token);
2398  // Grow the native stack ahead of deep expansion recursion (xint
2399  // `\XINT_…` number-arg chains nest tens of thousands deep) so
2400  // finite-deep recursion completes instead of overflowing the conversion
2401  // thread's 256 MB stack → SIGABRT (Perl degrades via `$MAXSTACK`). This
2402  // only grows the stack; the depth CAP is `ExpandDepthGuard` at the top
2403  // of `read_x_token` / `expand_once_partial`. Same idiom as the recursive walks in `document.rs`
2404  // / the math parser; params in `crate::stack_guard`.
2405  // The expansion goes straight onto the input (`Definition::invoke_onto_input`, tex.web §390) unless it is
2406  // inspected on the way: token-locators fill its origins, `TRACE_GROUP_END` counts its groups.
2407  // A token of a frontmatter field's stored value passes its provenance to its expansion (`Token::prov`).
2408  if cfg!(not(feature = "token-locators")) && !*TRACE_GROUP_END && token.prov == 0 {
2409    crate::stack_guard::maybe_grow(|| defn.invoke_onto_input(false))?;
2410    expire_current_token();
2411    return Ok(());
2412  }
2413  let mut invoked = crate::stack_guard::maybe_grow(|| match defn.native_expansion()? {
2414    Some(expansion) => Ok(expansion),
2415    None => defn.invoke(false),
2416  })?;
2417  if token.prov != 0 {
2418    // the macro's position and its arguments are covered by the boxes its expansion digests to (or by the next ones,
2419    // when it expands to nothing): a field's print is whole with them (OXIDIZED_DESIGN #715)
2420    note_field_expansion(&token, &invoked);
2421    crate::token::inherit_field_prov(token.prov, &mut invoked);
2422  }
2423  // token-locators: fill-only origin inheritance. A macro that
2424  // expands into synthesized tokens with no origin — e.g.
2425  // `\today → ExplodeText!(Today!())` yielding "May 25, 2026" —
2426  // would leave its output unlocatable. Attribute such tokens to
2427  // the invocation site so the rendered text is source-mapped.
2428  // The inherited handle is flagged `inherited` (one push per
2429  // expansion, shared by every result token), so `child_span`'s
2430  // genuine-origin-first scan never lets a macro's structural body
2431  // literals widen its arguments' content-exact span. We also never
2432  // overwrite a token that already carries an origin. See
2433  // SOURCE_PROVENANCE.md §3.1.3.
2434  #[cfg(feature = "token-locators")]
2435  {
2436    let inv_loc = token.loc;
2437    if inv_loc != 0 {
2438      let mut inherited = 0u32;
2439      for t in invoked.unlist_mut() {
2440        if t.loc == 0 {
2441          if inherited == 0 {
2442            inherited = crate::token::push_inherited_origin(inv_loc);
2443          }
2444          t.loc = inherited;
2445        }
2446      }
2447    }
2448  }
2449  if *TRACE_GROUP_END {
2450    // Print per-event {macro, delta} so post-processing can sum
2451    // by-macro to find which expandable CS contributes net +/- 1
2452    // imbalance across the run. Format: TRACE_GE delta CS
2453    let (mut begs, mut ends) = (0, 0);
2454    for t in invoked.unlist_ref() {

2464        begs - ends,
2465        begs,
2466        ends,
2467        token
2468      );
2469    }
2470  }
2471  unread_expansion(invoked);
2472  expire_current_token();
2473  Ok(())
2474}

2536/// Read the next non-expandable token, expanding until one appears. Hot path —
2537/// `read_token` is folded in. `toplevel` (default true): on mouth exhaustion,
2538/// step to the containing mouth. `fully_expand` (default = toplevel): expand
2539/// even protected defns ("for execution"). Unlike `read_balanced`, does NOT
2540/// defer `\the` & friends; `\noexpand`'d tokens act like `\relax`. For `\if`/
2541/// `\ifx` arguments pass `for_conditional=true` (handles `\noexpand` and CS
2542/// `\let` to tokens specially).
2543pub fn read_x_token(
2544  toplevel_opt: Option<bool>,
2545  for_conditional: bool,
2546  fully_expand_opt: Option<bool>,
2547) -> Result<Option<Token>> {
2548  // toplevel should be true by default
2549  let toplevel = toplevel_opt.unwrap_or(true);
2550  let fully_expand = fully_expand_opt.unwrap_or(toplevel);
2551  let _depth_guard = ExpandDepthGuard::enter()?; // Cluster F expansion-depth cap
2552  loop {
2553    // Combined checkpoint + raw read: this loop reads via the low-level
2554    // `read_internal_token_checked` (NOT `read_token`), so it runs the same
2555    // guards itself — otherwise every full-expansion read path (`\edef`,
2556    // `read_balanced`, csname construction) bypasses the token/pushback
2557    // limits and the expansion cycle guard entirely, and a `\def\x{a\x}`
2558    // runaway grinds to the multi-GB watchdog instead of a clean Fatal.
2559    let next_token = match read_internal_token_checked(CommentSink::Pending)? {
2560      // No runtime ≡ nothing more to read. (The pre-merge code fell through
2561      // to the exhausted branch below, whose autoclose peek — runtime gone ⇒
2562      // not autoclose — concluded Ok(None); return that directly.)
2563      CheckedRead::NoRuntime => return Ok(None),
2564      CheckedRead::Exhausted => None,
2565      CheckedRead::Tok(t) => Some(t),
2566    };
2567    //ProgressStep() if ($$self{progress}++ % $TOKEN_PROGRESS_QUANTUM) == 0;
2568    let Some(token) = next_token else {
2569      if read_x_token_crosses_input_end()? {
2570        continue;
2571      }
2572      return Ok(None);
2573    };
2574    if token.get_catcode() == Catcode::CS && token.text == pin!("\\dont_expand") {
2575      return read_x_token_dont_expand(for_conditional);
2576    }
2577    // Wow!!!!! See TeX the Program \S 309
2578    // SHOULD count nesting of { }!!! when SCANNED (not digested)
2579    if has_reading_alignment() {
2580      match read_x_token_column_end(token)? {
2581        ColumnEndRead::Worked => continue,
2582        ColumnEndRead::EndsNumberScan => return Ok(Some(token)),
2583        ColumnEndRead::NotOne => {},
2584      }
2585    }
2586    if token.get_catcode().is_active_or_cs() {
2587      // Read the meaning via closure so we can branch on the borrowed
2588      // Stored without cloning (Stored::clone was ~1% of total on
2589      // siunitx-heavy profiles; this site fires on every CS/ACTIVE
2590      // expansion — the hottest lookup_meaning caller).
2591      enum Outcome {
2592        LetTo(Token),
2593        Undefined,
2594        NonExpandable,
2595        Invoke(std::rc::Rc<dyn Definition>),
2596      }
2597      let outcome = with_meaning(&token, |defn_opt| match defn_opt {
2598        Some(Stored::Token(t)) => Outcome::LetTo(*t),
2599        Some(Stored::None) | None => Outcome::Undefined,
2600        Some(other) => match other.to_definition() {
2601          Some(defn) => {
2602            if expands_now(&defn, fully_expand) {
2603              Outcome::Invoke(defn)
2604            } else {
2605              Outcome::NonExpandable
2606            }
2607          },
2608          None => Outcome::Undefined,
2609        },
2610      });
2611      match outcome {
2612        Outcome::LetTo(let_token) => {
2613          return Ok(Some(if for_conditional { let_token } else { token }));
2614        },
2615        Outcome::Undefined => match read_x_token_undefined(token)? {
2616          Some(result) => return Ok(Some(result)),
2617          None => continue,
2618        },
2619        Outcome::NonExpandable => {
2620          return Ok(Some(token));
2621        },
2622        Outcome::Invoke(defn) => {
2623          invoke_expansion(token, &defn)?;
2624          continue;
2625        },
2626      }
2627    } else {
2628      // Perl Gullet.pm L421-422: track { and } at scan level for ALIGN_STATE
2629      match token.get_catcode() {
2630        Catcode::BEGIN => increment_align_group_count(),
2631        Catcode::END => decrement_align_group_count(),
2632        _ => {},
2633      }
2634      return Ok(Some(token));
2635    }
2636  }
2637}

2639/// [`read_x_token`] at the end of its input: whether a transparent injection closed and reading goes on.
2640#[cold]
2641#[inline(never)]
2642fn read_x_token_crosses_input_end() -> Result<bool> {
2643  {
2644    let gullet = gullet!();
2645    let current_is_autoclose = gullet
2646      .runtime
2647      .as_ref()
2648      .map(|r| r.autoclose)
2649      .unwrap_or(false);
2650    // Drain a *transparent autoclose injection* (\scantokens, raw_tex) and
2651    // resume the enclosing mouth even for a BOUNDED reader (toplevel==false,
2652    // e.g. the InputDefinitions file loop) — these are part of the current
2653    // logical stream. Faithful to tex.web `get_next` §362-365 (exhausting any
2654    // input level resumes the enclosing one; \scantokens is a pseudo-file
2655    // level). DIVERGES from Perl `Gullet.pm` readXToken, which gates on
2656    // `autoclose = toplevel` and so returns at the first exhausted mouth —
2657    // truncating a `.sty` whenever `\scantokens` runs mid-load (witness
2658    // 1906.03240: real babel.sty `\selectlanguage` dropped every later def →
2659    // undefined-CS cascade; Perl's hand-written babel.ltxml dodges it).
2660    // A non-autoclose boundary is left to its owner (return None).
2661    if !current_is_autoclose || gullet.mouthstack.is_empty() {
2662      return Ok(false);
2663    }
2664  }
2665  close_mouth(false)?; // Drain the autoclose injection; resume the parent.
2666  Ok(true)
2667}

2669/// [`read_x_token`] of `\dont_expand`.
2670#[cold]
2671#[inline(never)]
2672fn read_x_token_dont_expand(for_conditional: bool) -> Result<Option<Token>> {
2673  let unexpanded = match read_token()? {
2674    Some(t) => t,
2675    None => return Ok(Some(T_CS!("\\special_relax"))), // \dont_expand at end-of-input
2676  };

2678    Ok(Some(unexpanded))
2679  } else {
2680    // `\noexpand <tok>`: per-token `\special_relax` family encoding <tok>'s

2682    // `cur_cs` while giving relax meaning for this one access. (Perl

2688    // back its plain self (`unread_scanned`, KNOWN_PERL_ERRORS #255).)
2689    Ok(Some(crate::token::noexpand_family(&unexpanded)))
2690  }

3137          }

3141        // an argument read when a column end fires at alignment count 0 meets it (the cell's `}` closed the

3157          // Perl Gullet.pm L476: track ALIGN_STATE for } inside readBalanced

3165        },

3171        },

3196            // definition — both are cheap (bool + Rc-clone).

3264          }
3265        },
3266      },
3267    }
3268  }

3272    // a FILE boundary deliberately is not (TeX/Perl parity — "file ended

3275    // my $loc_message = $startloc ? ("Started at " . ToString($startloc)) : ("Ended at " .

3279      "}",

3285  } else {
3286    Ok((
3287      if pack_parameters {
3288        Tokens::new(tokens).pack_parameters()?
3289      } else {

3291      },
3292      close,
3293    ))
3294  }

3297/// Match the input against a set of keywords; Similar to readMatch, but the keywords are strings,
3298/// and Case and catcodes are ignored; additionally, leading spaces are skipped.
3299/// AND, macros are expanded.
3300///
3301/// Perf: zero-allocation char-wise comparison against each keyword.
3302/// The previous version allocated two Strings per char-match (via `to_uppercase()`
3303/// and `char::to_string()`), which was expensive in hot parameter parsing loops.
3304pub fn read_keyword(keywords: &[&str]) -> Result<Option<String>> {
3305  skip_spaces()?;
3306  for keyword in keywords.iter() {
3307    // Pre-size to the keyword length — `matched` holds one token per
3308    // matched char, and we unread them on no-match. Keyword-match
3309    // runs on every parameter/keyword read; small win per call.
3310    let mut matched = Vec::with_capacity(keyword.len());
3311    let mut ok = true;
3312    for expected in keyword.chars() {
3313      let Some(tok) = read_x_token(Some(false), false, None)? else {
3314        ok = false;
3315        break;
3316      };
3317      // Compare char-by-char against the token's text, case-insensitively.
3318      let eq = tok.with_str(|s| {
3319        let mut it = s.chars();
3320        match it.next() {
3321          Some(c) if it.next().is_none() => {
3322            // single-char token: case-insensitive compare
3323            c.to_uppercase().eq(expected.to_uppercase())
3324          },
3325          _ => false,
3326        }
3327      });
3328      matched.push(tok);
3329      if !eq {
3330        ok = false;
3331        break;
3332      }
3333    }
3334    if ok {
3335      return Ok(Some(keyword.to_string()));
3336    } else {
3337      // scan_keyword backs up the plain tokens it read (tex.web §407): a
3338      // `\noexpand`'d one goes back as its plain self (see `unread_scanned`).
3339      for token in &mut matched {
3340        if let Some(plain) = token.noexpand_shadowed() {
3341          *token = plain;
3342        }
3343      }
3344      unread(matched.into());
3345    }
3346  }
3347  Ok(None)
3348}

3386        .skip(ring.len() + 1 - k)

3388  })

3391pub fn read_until(delim: &Tokens) -> Result<Option<Tokens>> {
3392  // Pre-size like `read_balanced`: the accumulator is grown one token at a time
3393  // in the loops below, so an unsized `Vec::new()` pays the 0→1→2→4→8 doubling
3394  // reallocations on every call. 16 covers the common short delimited read in a
3395  // single allocation (a top `grow_one` site in the allocation profile).
3396  let mut tokens: Vec<Token> = Vec::with_capacity(16);
3397  let mut nbraces = 0;
3398  let want = delim.unlist_ref();
3399  let ntomatch = want.len();
3400  let mut has_matched;
3401
3402  if ntomatch == 1 {
3403    let want = &want[0];
3404    loop {
3405      let token = match read_token()? {
3406        Some(t) => t,
3407        None => {
3408          // A delimited macro argument that runs off a file's end abandons
3409          // its call (`recover_at_file_end`), which drops the argument.
3410          if argument_ran_off_a_file()? {
3411            note_runaway_tokens(tokens);
3412            return Ok(Some(Tokens!()));
3413          }
3414          // Ran out! Unread and report the distinguishable EOF. A marked
3415          // `\scantokens` mouth has already delivered its `\everyeof`
3416          // payload as its last tokens (`eof_seen`), so this is the true end.
3417          unread(Tokens::new(tokens));
3418          return Ok(None);
3419        },
3420      };
3421      // Perl: check direct match OR \special_relax smuggling (Gullet.pm line 662)
3422      if token == *want || special_relax_matches(&token, want) {
3423        break;
3424      }
3425      // After the delimiter test, as tex.web §392 orders them: a `\par` delimiter is matched, any other `\par`
3426      // ends a non-`\long` macro's call.
3427      if is_par_ending_argument(&token) {
3428        abandon_argument_at_par(token);
3429        return Ok(Some(Tokens!()));
3430      }
3431      match token.get_catcode() {
3432        Catcode::MARKER => {
3433          // would have been handled by readToken, but we're bypassing
3434          handle_marker(token);
3435        },
3436        Catcode::BEGIN => {
3437          // And if it's a BEGIN, copy till balanced END
3438          nbraces += 1;
3439          tokens.push(token);
3440          let (balanced_arg, close) = read_balanced_with_close(ExpansionLevel::Off, false, false)?;
3441          if argument_runaway() {
3442            note_runaway_tokens(tokens);
3443            return Ok(Some(Tokens!()));
3444          }
3445          if !balanced_arg.is_empty() {
3446            tokens.extend(balanced_arg.unlist());
3447          }
3448          // The group's OWN close (tex.web §392), not a canonical `}`.
3449          tokens.push(close.unwrap_or_else(|| T_END!()));
3450        },
3451        _ => {
3452          tokens.push(token);
3453        },
3454      }
3455    }
3456  } else {
3457    let mut ring = VecDeque::new();
3458    loop {
3459      // prefill the required number of tokens
3460      while ring.len() < ntomatch {
3461        let token = match read_token()? {
3462          Some(t) => t,
3463          None => {
3464            if argument_ran_off_a_file()? {
3465              tokens.extend(ring);
3466              note_runaway_tokens(tokens);
3467              return Ok(Some(Tokens!()));
3468            }
3469            // Ran out! Unread and report the distinguishable EOF.
3470            // The partial ring is DROPPED, not unread — replaying a
3471            // half-matched delimiter prefix into the stream re-tokenizes
3472            // it under current catcodes (l3doc `_`-laden names became
3473            // double-subscript cascades: spath3 1→1001, 2026-09-01).
3474            unread(Tokens::new(tokens));
3475            return Ok(None);
3476          },
3477        };
3478        // A `\par` that continues a partial match of the delimiter is accepted quietly (tex.web §397); any other
3479        // ends a non-`\long` macro's call (§392).
3480        if is_par_ending_argument(&token) && !continues_partial_match(&ring, want) {
3481          abandon_argument_at_par(token);
3482          return Ok(Some(Tokens!()));
3483        }
3484        // Perl: $$token[1] == CC_BEGIN — direct catcode check. A `{` that completes the delimiter (one ending in `{`,
3485        // from a parameter text's closing `#{`) is its last token: tex.web §392 compares the token with the delimiter
3486        // before it takes a `{` as a group.
3487        if token.get_catcode() == Catcode::BEGIN
3488          && !(ring.len() + 1 == ntomatch
3489            && token == want[ntomatch - 1]
3490            && ring
3491              .iter()
3492              .zip(want.iter())
3493              .all(|(have, wanted)| have == wanted))
3494        {
3495          // read balanced, and refill ring.
3496          nbraces += 1;
3497          // the ring empties into the tokens: the scan retries with a fresh one
3498          for r_token in std::mem::take(&mut ring) {
3499            tokens.push(r_token);
3500          }
3501          tokens.push(token);
3502          let (balanced_arg, close) = read_balanced_with_close(ExpansionLevel::Off, false, false)?;
3503          if argument_runaway() {
3504            note_runaway_tokens(tokens);
3505            return Ok(Some(Tokens!()));
3506          }
3507          if !balanced_arg.is_empty() {
3508            tokens.append(&mut balanced_arg.unlist());
3509          }
3510          tokens.push(close.unwrap_or_else(|| T_END!())); // the group's own close
3511        } else {
3512          ring.push_back(token);
3513        }
3514      }
3515      has_matched = &ring == want; // Test match
3516      if has_matched {
3517        break;
3518      } // Matched all!
3519      if let Some(ring_token) = ring.pop_front() {
3520        tokens.push(ring_token);
3521      }
3522    }
3523  }
3524  // Notice that IFF the arg looks like {balanced}, the outer braces are stripped
3525  // so that delimited arguments behave more similarly to simple, undelimited arguments.
3526  // Perl: ($nbraces == 1) && ($tokens[0][1] == CC_BEGIN) && ($tokens[-1][1] == CC_END)
3527  if nbraces == 1
3528    && tokens.first().unwrap().get_catcode() == Catcode::BEGIN
3529    && tokens.last().unwrap().get_catcode() == Catcode::END
3530  {
3531    tokens.remove(0);
3532    tokens.pop();
3533  }
3534  Ok(Some(Tokens::new(tokens)))
3535}

3643fn read_cs_name_inner(quiet: bool) -> Result<Token> {
3644  CSNAME_DEPTH.set(CSNAME_DEPTH.get() + 1);
3645  let _csname_scope = CsnameScope;
3646  // TeX does NOT store the csname with the leading `\`, BUT stores active chars with a flag
3647  // However, so long as the Mouth's CS and \string properly respect \escapechar, all's well!
3648
3649  // Safety bound: a real CS name fits in well under 256 chars. We've seen
3650  // pathological cases (lipsum.sty with malformed \cs_set_nopar:Npe expansion,
3651  // or expl3 raw-load before \endcsname is bound) where `\csname` reads
3652  // thousands of tokens accumulating into `cs`, eventually OOMing the
3653  // `read_x_token` pushback Vec. Cap at 4096 bytes — beyond that there's no
3654  // legitimate CS name, just a runaway. Emit one clear error and break.
3655  const MAX_CS_NAME_BYTES: usize = 4096;
3656  let mut cs = String::from("\\");
3657  // keep newlines from having \n inside!
3658  while let Some(token) = read_x_token(Some(true), false, None)? {
3659    if token.defined_as(&TOKEN_ENDCSNAME) {
3660      break;
3661    }
3662    // tex.web §370: an undefined control sequence inside `\csname…\endcsname`
3663    // is reported by `expand` and DISCARDED — the name scan continues. The
3664    // `read_x_token` above just stubbed it (`generate_error_stub`, error
3665    // emitted), so without this check it read as a defined non-expandable
3666    // token and took the §373 `back_error` path, ending the name early and
3667    // leaving the real `\endcsname` stray ("Extra \endcsname" ×1693 lines /
3668    // 65 manuals: babel's `\csname l@\beamer@torinoth@language\endcsname`
3669    // in beamer2thesis, gckanbun's pgf arrow declarations …). Perl consumes
3670    // the `\endcsname` but also emits a second error; pdflatex emits one.
3671    if token.get_catcode() == Catcode::CS && is_error_stub(&token) {
3672      continue;
3673    }

3675      Error!(
3676        "runaway",
3677        "csname",
3678        format!(
3679          "CS-name read exceeded {MAX_CS_NAME_BYTES} bytes; aborting at partial cs: {:?}",
3680          // Truncate by CHARS, not bytes — a byte slice at 200 can split a
3681          // multi-byte UTF-8 char (e.g. 'ä') and panic. Witness 2601.03403.
3682          cs.chars().take(200).collect::<String>()
3683        )
3684      );
3685      break;
3686    }

3688      Catcode::CS => {
3689        // Soft-substitute the underlying char for a character-equivalent CS
3690        // token in the \csname stream — a documented divergence from Knuth TeX
3691        // (tex.web L7745-7758 hard-errors "Missing \endcsname inserted" for any
3692        // CS that isn't \endcsname). Our expansion pipeline surfaces PA-aliased
3693        // `Stored::Token` CSes (expl3 `\exp_stop_f:` = frozen space, `\lx@NBSP`
3694        // from CLUSTER-NBSP) into the csname stream where real TeX wouldn't reach
3695        // this state; erroring like Knuth would break real expl3/mhchem/glossaries
3696        // loads, so we substitute the char the author meant. Witnesses: `\lx@NBSP`
3697        // (CLUSTER-NBSP, 18 papers; `~`→U+00A0 in `\csname r@LABEL\endcsname`),
3698        // `\exp_stop_f:` (mhchem raw-load). The `\lx@NBSP` carve-out below stays
3699        // for clarity; the general `Stored::Token` case handles the rest uniformly.

3702        // pdflatex's csname-stream interpretation. The `DefPrimitive!(name,
3703        // "char")` body is a closure that wraps a Tbox — not statically
3704        // inspectable from here — so we maintain an explicit table for the
3705        // canonical set. Witnesses (stage-1..3 of 100k warning corpus):
3706        //   \\textquoteright surfacing in `\twemoji flag: Côte d` cluster
3707        //   (≥4 papers across 2603.08303, 2604.13899, 2604.17338, 2604.20621).

3709          "\\lx@NBSP" | "\\lx@nobreakspace" | "\\nobreakspace" => Some('\u{00A0}'),
3710          "\\textquoteright" => Some('\u{2019}'),
3711          "\\textquoteleft" => Some('\u{2018}'),
3712          "\\textquotedblright" => Some('\u{201D}'),
3713          "\\textquotedblleft" => Some('\u{201C}'),
3714          "\\textquotedbl" => Some('"'),
3715          "\\textemdash" => Some('\u{2014}'),
3716          "\\textendash" => Some('\u{2013}'),
3717          "\\textbackslash" => Some('\u{005C}'),

3719          "\\textbraceleft" => Some('{'),

3734          // (U+0131 / U+0237) so the constructed csname carries the

3746          // `\lx@applyaccent <accent> <combining> <standalone> {<letter>}`

3800                  cs

3851  };

3855/// reads and discards tokens, until it encounters a conditional, if any.
3856/// Perl: skipConditionalBody inner loop (Conditional.pm L127-133) reads tokens directly
3857/// from pushback/mouth (NOT through readToken) and manually tracks
3858/// `$LaTeXML::ALIGN_STATE` for `{` and `}`. This loop does not call `read_token` either:
3859/// its expansion and comment handling belong to digestion, and calling it from
3860/// here once let pmatrix's `&` get treated as the OUTER alignment's
3861/// column-end during `\ifx.#1.\else…\fi` skip when `#1` contains
3862/// `\begin{pmatrix}…&…\end{pmatrix}`. (REG-2 / math-ph0501074: the 9-line
3863/// `\nonumber+\lefteqn+\pmatrix` repro.) It does take a column end at a zero count
3864/// for the cell's end, as tex.web's `pass_text` does through `get_next` (§342) and
3865/// Perl does not (KNOWN_PERL_ERRORS #1183).
3866pub fn read_next_conditional() -> Result<Option<(Token, ConditionalType)>> {
3867  loop {
3868    // The FOURTH reader loop over the combined checked read (the `\else`/`\fi`
3869    // skipper — it must not delegate to read_token, whose alignment trigger
3870    // misfires during conditional-skip; see the fn doc). It still needs the
3871    // resource/cycle checkpoints: a skip over a pathologically large stream
3872    // must count progress and remain loop-detectable like every other read
3873    // path (PR #249 review P2-9).
3874    match read_internal_token_checked(CommentSink::Pending)? {
3875      CheckedRead::NoRuntime => return Ok(None),
3876      CheckedRead::Tok(token) => {
3877        let cc = token.get_catcode();
3878        // Perl L128-130: manual ALIGN_STATE tracking for `{` / `}` (avoids
3879        // the trigger check that read_token applies).
3880        match cc {
3881          Catcode::BEGIN => increment_align_group_count(),
3882          Catcode::END => decrement_align_group_count(),
3883          _ => {},
3884        }
3885        // tex.web §342: `pass_text` reads through `get_next`, so a tab or `\cr` met at align_state 0 while a false
3886        // branch is skipped ends the cell all the same: the column's v-part goes in ahead of it (§789,
3887        // `handle_template` raises the count to 1000000) and the skip reads on through it, where its `\fi` can end
3888        // the skip — the hide-column template `>{\iffalse}c<{\fi}` (witness 1504.08100). A column end inside a brace
3889        // group of the skipped text is at a nonzero count and is skipped, as the pmatrix `&` above is. The hidden
3890        // ends (`\lx@hidden@cr`, `\lx@hidden@align`) stand for LaTeX's `\\` and amsmath's tabs, whose `\cr` or `&`
3891        // TeX reaches only by expansion, behind `\relax\iffalse{\fi` (latex.ltx `\@arraycr`), so a skip passes
3892        // them (`\ifnum1<\nb\\\hline\fi`, tabularcalc.sty:428). Guard
3893        // `perfect_kernel_batch80a::a_false_conditional_in_a_template_ends_at_its_v_part`.
3894        if align_group_count() == 0
3895          && has_reading_alignment()
3896          && let Some((_, atype, false)) = is_column_end(&token)
3897          && let Some(reading_alignment) = get_reading_alignment()
3898          && let DigestedData::Alignment(data) = reading_alignment.data()
3899        {
3900          handle_template(data.borrow_mut(), token, atype, false)?;
3901          continue;
3902        }
3903        if cc.is_active_or_cs()
3904          && let Some(cond_type) = lookup_conditional(&token)
3905        {
3906          return Ok(Some((token, cond_type)));
3907        }
3908      },
3909      CheckedRead::Exhausted => {
3910        // The current mouth is exhausted: the skip ENDS here, like Perl's
3911        // `skipConditionalBody` (Core/Definition/Conditional.pm:127), which
3912        // reads `$$gullet{mouth}->readToken()` — the current mouth only —

3931/// Read and return a "normal" TeX argument
3932///
3933/// The next Token or Tokens (if surrounded by braces).
3934/// `expansion_level` controls expansion as if the argument were read
3935///  and then expanded in isolation:
3936///
3937/// In the case of a single unbraced expandable token,
3938///  it will **not** read any macro arguments from the following input!
3939pub fn read_arg(expansion_level: ExpansionLevel) -> Result<Tokens> {
3940  match read_non_space()? {
3941    None => {
3942      // `\foo` as a file's last token, `\foo` taking an argument: a runaway
3943      // argument (tex.web §392), which abandons the call.
3944      argument_ran_off_a_file()?;
3945      Ok(Tokens!())
3946    },
3947    // An undelimited argument that is `\par` ends a non-`\long` macro's call (tex.web §392).
3948    Some(token) if is_par_ending_argument(&token) => {
3949      abandon_argument_at_par(token);
3950      Ok(Tokens!())
3951    },
3952    Some(token) => {
3953      // Perl: $$token[1] == CC_BEGIN — checks actual catcode, NOT defined_as.
3954      // \bgroup (catcode CS) does NOT match here; only literal { does.
3955      if token.get_catcode() == Catcode::BEGIN {
3956        read_balanced(expansion_level, false, false)
3957      } else if matches!(expansion_level, ExpansionLevel::Off) {
3958        // A `\noexpand`'d token captured as an (undelimited) macro argument
3959        // reverts to its plain shadowed identity — faithful to TeX, where the
3960        // `no_expand_flag` is transient and never stored in the captured arg
3961        // (`#1` is `cur_cs`, the plain token). This lets `\ifx\X#1` match (xint's
3962        // `\def\XINTfstop{\noexpand\XINTfstop}` f-stop, witness 1804.01117).
3963        // `\let`/`\ifx` of the LIVE token read via `read_token` (the `Token`
3964        // param), NOT here, so their relax-meaning capture is unaffected.
3965        Ok(Tokens!(token.noexpand_shadowed().unwrap_or(token)))
3966      } else {
3967        // Perl Gullet.pm `readArg`:
3968        //   return $self->readingFromMouth(Tokens(T_BEGIN, $token, T_END), sub {
3969        //       readBalanced($self, $expanded, 0, 1); });
3970        // Use an isolated mouth so leftover tokens (e.g. an extra `}` when
3971        // `$token` itself happens to be T_END) cannot leak back into the
3972        // caller's stream. `unread_vec` here would pollute the parent mouth.
3973        let synth = Tokens::new(vec![T_BEGIN!(), token, T_END!()]);
3974        reading_from_mouth(Mouth::default(), move || -> Result<Tokens> {
3975          unread(synth);
3976          read_balanced(expansion_level, false, true)
3977        })
3978      }
3979    },
3980  }

4349/// Match the input against one of the Token or Tokens in @choices; return the matching one or
4350/// undef.
4351pub fn read_match(choices: &[&Tokens]) -> Result<Option<Tokens>> {
4352  for choice in choices {
4353    let mut to_match: Vec<&Token> = choice.unlist_ref().iter().rev().collect();
4354    // `matched` accumulates tokens read so far, bounded by `choice.len()`.
4355    // Pre-size to avoid reallocations on multi-token match attempts.
4356    let mut matched = Vec::with_capacity(choice.unlist_ref().len());
4357    while !to_match.is_empty() {
4358      match read_token()? {
4359        None => break,
4360        Some(token) => {
4361          // Perl: also check smuggled \special_relax token (Gullet.pm line 612)
4362          let was_last_match = if let Some(&&want) = to_match.last() {
4363            token == want || special_relax_matches(&token, &want)
4364          } else {
4365            false
4366          };
4367          matched.push(token);
4368          if was_last_match {
4369            to_match.pop();
4370          } else {
4371            break;
4372          }
4373          // A space in the delimiter matches exactly one space token (tex.web §392:
4374          // the delimiter tokens are matched one for one). Perl then skips every
4375          // following space (Gullet.pm:614-617, KNOWN_PERL_ERRORS #222), which eats the
4376          // space xint's `\XINT_zapsp_b`-style macros leave for the next call: a `\def`
4377          // prefix delimiter `\B<space>` met by `\B<space><space>X` must leave
4378          // `<space>X` (pdflatex `| X|`, Perl `|X|`).
4379        },
4380      }
4381    }
4382    if to_match.is_empty() {
4383      return Ok(Some((*choice).clone())); // All matched!!!
4384    } else {
4385      // Put 'em back and try next — use unread_vec for proper agc adjustment
4386      unread_vec(matched);

4404  // consumed its tokens, so the next kind is tried on a fresh token (as Perl's chain of reads).

4416      match register_value_from(token, RegisterType::Dimension, false)? {

4425        Scanned::Value(None) => {},

4427      }
4428    }
4429    match hand.take() {
4430      Some(token) if attempt == 0 && became_expandable(&token)? => unread_one(token),
4431      other => {
4432        hand = other;
4433        break;
4434      },
4435    }
4436  }
4437  {
4438    let next = match hand {
4439      Some(token) => Some(token),
4440      None => read_token()?,
4441    };
4442    // Perl Gullet.pm:904-905: the primary message is just "Missing number,
4443    // treated as zero"; the processing context and the unexpected-token
4444    // (showUnexpected) are SEPARATE Error details rendered on their own
4445    // lines. Render the tokens with ToString/Stringify, not Rust-Debug
4446    // (which leaks `Some("\\relax")` into user-facing diagnostics).
4447    let current = get_current_token()
4448      .map(|t| t.to_string())
4449      .unwrap_or_default();
4450    let unexpected = match next {
4451      Some(t) => s!("Next token is {}", t.stringify()),
4452      None => s!("Input is empty"),
4453    };
4454    Warn!(
4455      "expected",
4456      "<number>",
4457      "Missing number, treated as zero",
4458      s!("while processing {current}"),
4459      unexpected
4460    );
4461    if let Some(next) = next {
4462      unread_one(next);
4463    }
4464    Ok(Number::new(0))
4465  }
4466}

4486/// tex.web §445: an integer constant (decimal, octal or hexadecimal) of 2^31 or
4487/// more is "Number too big", and its value is `infinity` = 2^31 − 1. An Error,
4488/// as TeX's: the recovered value is not the author's (OXIDIZED_DESIGN #519).
4489fn number_too_big(n: i64) -> Result<i64> {

4493      "<number>",
4494      "Number too big.",
4495      s!("{n} clamped to {MAX_INTEGER} (tex.web §445)")
4496    );
4497    Ok(MAX_INTEGER)
4498  } else {
4499    Ok(n)
4500  }
4501}

4503/// tex.web §460 `attach_sign`: a scanned dimension whose magnitude reaches
4504/// 2^30 sp (`arith_error`, or `abs(cur_val) >= @'10000000000`) is "Dimension
4505/// too large", and its value is `max_dimen` (16383.99998pt), negated only for
4506/// the scan's own `<optional signs>` (`negative`) — a negative internal unit
4507/// (`2000000000\dimen0` with `\dimen0` < 0) gives +max_dimen, as pdfTeX
4508/// prints it. `scaled` is the fixed-point product, `Err(_)` when it left the
4509/// i64 range. An Error, as for [`number_too_big`], a Warning in a box measured by our approximation
4510/// ([`approximated_measure`]).
4511fn dimension_too_large(scaled: std::result::Result<i64, i64>, negative: bool) -> Result<i64> {
4512  match scaled {
4513    Ok(v) if v.abs() <= MAX_DIMEN => Ok(v),
4514    _ => {
4515      let detail = "clamped to max_dimen (tex.web §460)";
4516      match approximated_measure() {
4517        Some((ours, true)) => Warn!("expected", "<dimen>", "Dimension too large.", detail, ours),
4518        Some((ours, false)) => Info!("expected", "<dimen>", "Dimension too large.", detail, ours),
4519        None => Error!("expected", "<dimen>", "Dimension too large.", detail),
4520      }
4521      Ok(if negative { -MAX_DIMEN } else { MAX_DIMEN })
4522    },
4523  }
4524}

4526/// <normal integer> from the token in hand (TeX's `cur_tok`, read with expansion), or the token
4527/// back when it starts none.
4528fn normal_integer_from(token: Token) -> Result<Scanned<Number>> {
4529  let cc = token.get_catcode();
4530  // tex.web §440-448 `scan_int` inspects the token's command/char codes
4531  // and never stringifies it; the decimal arm is the only one that needs
4532  // the text, so the common internal-quantity case (`\pgf@x`, a count
4533  // register) pays no String — this ran once per number read.
4534  let is_decimal_start = cc == Catcode::OTHER
4535    && token.with_str(|s| !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit()));
4536  if is_decimal_start {
4537    // Read decimal literal, accumulating the digits as tex.web §444-445
4538    // does (no string). The accumulation saturates at i64::MAX; the
4539    // constant is then held to TeX's `infinity` (`number_too_big`).
4540    let mut n = token.with_str(|s| s.parse::<i64>().unwrap_or(i64::MAX));
4541    let end = loop {
4542      let Some(next) = read_x_token(None, false, None)? else {
4543        break None;
4544      };
4545      let digit = next.with_str(|s| {
4546        let mut chars = s.chars();
4547        match (chars.next(), chars.next()) {
4548          (Some(c), None) => c.to_digit(10),
4549          _ => None,
4550        }
4551      });
4552      match digit {
4553        Some(d) => {
4554          n = n
4555            .checked_mul(10)
4556            .and_then(|v| v.checked_add(i64::from(d)))
4557            .unwrap_or(i64::MAX)
4558        },
4559        None => break Some(next),
4560      }
4561    };
4562    // <one optional space> (`read_digits`' skip).
4563    if let Some(end) = end
4564      && !is_space_or_implicit_space(&end)
4565    {
4566      unread_scanned(end);
4567    }
4568    Ok(Scanned::Value(Some(Number::new(number_too_big(n)?))))
4569  } else if token == T_OTHER!("'") {
4570    // Read Octal literal. Perl: `Number(oct(readDigits(...)))`, and
4571    // Perl's `oct("")` is 0 — so a `'` with no octal digit following
4572    // yields 0 (TeX's "Missing number, treated as zero"), NOT a fatal
4573    // error. Mirror that, and clamp overflow to i64::MAX like the
4574    // decimal arm rather than propagating a ParseIntError.
4575    let digits = read_digits(is_octal_digit, true)?;
4576    let decimal = if digits.is_empty() {
4577      0
4578    } else {
4579      i64::from_str_radix(&digits, 8).unwrap_or(i64::MAX)
4580    };
4581    Ok(Scanned::Value(Some(Number::new(number_too_big(decimal)?))))
4582  } else if token == T_OTHER!("\"") {
4583    //  Read Hex literal. Perl: `Number(hex(readDigits(...)))`, and
4584    // Perl's `hex("")` is 0 — so a `"` with no hex digit following
4585    // yields 0, NOT a fatal error. (Witness 2008.10843: mdwmath.sty
4586    // raw-load reads a bare `"` with no hex digit → previously a
4587    // `Fatal:Document:Generic(ParseIntError)` aborting the run.)
4588    let digits = read_digits(is_hex_digit, true)?;
4589    let decimal = if digits.is_empty() {
4590      0
4591    } else {
4592      i64::from_str_radix(&digits, 16).unwrap_or(i64::MAX)
4593    };
4594    Ok(Scanned::Value(Some(Number::new(number_too_big(decimal)?))))
4595  } else if token == T_OTHER!("`") {
4596    //  Read Charcode: `<character token><one optional space>
4597    // The leading `\` is stripped only from a control-sequence token
4598    // (`\a` → 'a', `\\` → '\'). Perl (Gullet.pm L926 `s/^\\//`) strips
4599    // it from a catcode-12 backslash *character* too, so `` `\ `` after
4600    // `\detokenize`/`\string` reads as 0 instead of TeX's 92 — see
4601    // KNOWN_PERL_ERRORS "backquote charcode of a detokenized backslash"
4602    // (witness bibleref-parse.sty L481-486 `\brp@ifcs`).
4603    let (mut s, is_cs) = match read_token()? {
4604      None => (String::new(), false),
4605      Some(next) => {
4606        // tex.web §442: a brace read as a character constant is NOT a
4607        // group for ALIGN_STATE — `get_token` counted it, so undo that.
4608        // This is what makes the `\iffalse{\fi\ifnum0=`}\fi` idiom
4609        // (expl3 `\group_align_safe_begin:`, amsmath) leave align_state
4610        // +1 with no group open, so an alignment-catcode token inside
4611        // a delimited-macro definition in a tabular cell (l3tl
4612        // `\tl_replace_all` with a rescanned `_`(4), l3doc `\marg` in
4613        // `syntax`) is not mistaken for the cell end. Perl Gullet.pm:926
4614        // omits the undo (SHARED).
4615        match next.get_catcode() {
4616          Catcode::BEGIN => decrement_align_group_count(),
4617          Catcode::END => increment_align_group_count(),
4618          _ => {},
4619        }
4620        // A space token's character code is 32 whatever its text
4621        // (tex.web §289; an end-of-line space carries "\n" — batch 56gc,
4622        // the same normalization as `Token::get_charcode`).
4623        if next.get_catcode() == Catcode::SPACE {
4624          (" ".to_string(), false)
4625        } else {
4626          (next.to_string(), next.get_catcode() == Catcode::CS)
4627        }
4628      },
4629    };
4630    if is_cs && s.starts_with('\\') {
4631      s.remove(0);
4632    }
4633    let s_char = s.chars().next().unwrap_or('\0');
4634    // Perl: skip1Space($self, 1); — expanded space-skip after charcode
4635    skip_one_space(true)?;
4636    Ok(Scanned::Value(Some(Number::new(s_char as i64)))) //  Only a character token!!! NOT expanded!!!!
4637  } else {
4638    Ok(
4639      match register_value_from(token, RegisterType::Number, false)? {
4640        Scanned::Value(value) => Scanned::Value(value.map(Number::from)),
4641        Scanned::Declined(token) => Scanned::Declined(token),
4642      },
4643    )
4644  }
4645}

4649/// Similar to factor, but does NOT accept comma!
4650/// This is NOT part of TeX, but is convenient.
4651pub fn read_float() -> Result<Float> {

4653  let is_negative = read_optional_signs()?;
4654  let s = if is_negative { -1.0 } else { 1.0 };
4655  let mut string = read_digits(is_decimal_digit, true)?;
4656  let mut token = read_x_token(None, false, None)?;
4657  if token.is_some() && token.as_ref().unwrap().get_sym() == pin!(".") {
4658    string = s!("{string}.{}", read_digits(is_decimal_digit, true)?);
4659    token = read_x_token(None, false, None)?;
4660  }
4661  let n_opt: Option<f64> = if !string.is_empty() {
4662    if let Some(t) = token
4663      && t.get_catcode() != Catcode::SPACE
4664    {
4665      unread_one(t);
4666    }
4667    // Same rationale as read_normal_integer above: malformed float
4668    // literals (e.g. very long digit runs, "1e" without exponent)
4669    // should degrade to 0.0 rather than panic.

4671  } else {

4673      unread_one(t); // Unread
4674    }
4675    read_normal_integer()?.map(|v| v.value_of() as f64)
4676  };
4677
4678  if let Some(n) = n_opt {
4679    Ok(Float::new_f64(s * n))
4680  } else {
4681    Ok(Float::new_f64(0.0))
4682  }
4683}

4705// Dimensions
4706//======================================================================
4707// ```
4708// <dimen> = <optional signs><unsigned dimen>
4709// <unsigned dimen> = <normal dimen> | <coerced dimen>
4710// <coerced dimen> = <internal glue>
4711// ```
4712pub fn read_dimension() -> Result<Dimension> {
4713  let _scan = NumberScan::begin();
4714  // tex.web §448 scan_dimen: the token after the signs is tested in hand for an internal
4715  // dimension, then glue (as `read_number`), and put back once for the factor.
4716  let (is_negative, mut hand) = scan_optional_signs()?;
4717  for attempt in 0..2 {
4718    if let Some(token) = in_hand_or_read(hand.take())? {
4719      match register_value_from(token, RegisterType::Dimension, false)? {
4720        Scanned::Value(Some(d)) => {
4721          let d = Dimension::from(d);
4722          return Ok(if is_negative { d.negate() } else { d });
4723        },
4724        Scanned::Value(None) => {},
4725        Scanned::Declined(token) => hand = Some(token),
4726      }
4727    }
4728    if let Some(token) = in_hand_or_read(hand.take())? {
4729      match register_value_from(token, RegisterType::Glue, false)? {
4730        Scanned::Value(Some(d)) => {
4731          let d = Glue::from(d);
4732          return Ok(Dimension::new(if is_negative {
4733            d.negate().value_of()
4734          } else {
4735            d.value_of()
4736          }));
4737        },
4738        Scanned::Value(None) => {},
4739        Scanned::Declined(token) => hand = Some(token),
4740      }
4741    }
4742    match hand.take() {
4743      Some(token) if attempt == 0 && became_expandable(&token)? => unread_one(token),
4744      other => {
4745        hand = other;
4746        break;
4747      },
4748    }
4749  }
4750  if let Some(token) = hand {
4751    unread_one(token);
4752  }
4753  if let Some(d) = read_factor()? {
4754    let (num, den) = match read_unit()? {
4755      Some(ratio) => ratio,
4756      None => {
4757        Warn!(
4758          "expected",
4759          "<unit>",
4760          "Illegal unit of measure (pt inserted)."
4761        );
4762        (1, 1)
4763      },
4764    };
4765    let d_signed = if is_negative { -d } else { d };
4766    // tex.web §460 attach_sign: "Dimension too large" at 2^30 sp, the value
4767    // clamped to max_dimen (the i128 product once wrapped to garbage past the
4768    // i64 range: `200000000000\dimen0`, chinesechess.sty:2016-2020).
4769    Ok(Dimension::new(dimension_too_large(
4770      fixpoint_unit_checked(d_signed, num, den),
4771      is_negative,
4772    )?))
4773  } else {
4774    // Perl Gullet.pm:972: the type is named in the primary message
4775    // ("(Dimension)") and "while processing X" is a separate detail
4776    // (ToString of the current token, not Rust-Debug).
4777    let cur = get_current_token()
4778      .map(|t| t.to_string())
4779      .unwrap_or_default();

4781      "expected",
4782      "<number>",
4783      "Missing number (Dimension), treated as zero.",
4784      s!("while processing {cur}")
4785    );
4786    Ok(Dimension::new(0))
4787  }

4798/// Read a unit, returning the exact TeX `(num, den)` conversion fraction (see
4799/// [`convert_unit_ratio`] / `numeric_ops::fixpoint_unit`). Internal/coerced units
4800/// (`\wd0`, `\dimen`, glue) yield `(value_sp, 65536)` — the `floor(fix·v/65536)`
4801/// path of tex.web §8983, exact in integer arithmetic.
4802pub fn read_unit() -> Result<Option<(i64, i64)>> {

4806  } else if let Some(u) = read_internal_integer()? {

4809    Some((u.value_of(), UNITY))
4810  } else if let Some(u) = read_internal_glue()? {
4811    Some((u.value_of(), UNITY))
4812  } else {
4813    read_keyword(&["true"])?; // But ignore, we're not bothering with mag...
4814    if let Some(u) = read_keyword(&["pt", "pc", "in", "bp", "cm", "mm", "dd", "cc", "sp", "px"])? {
4815      skip_one_space(true)?;
4816      Some(convert_unit_ratio(&u))
4817    } else {
4818      None
4819    }
4820  };

4859pub fn read_rubber(mu: bool) -> Result<(Option<i64>, Option<FillCode>)> {
4860  let is_negative = read_optional_signs()?;
4861  let s = if is_negative { -1 } else { 1 };
4862  match read_factor()? {
4863    None => {
4864      let f = if mu {
4865        read_mu_dimension()?.value_of()
4866      } else {
4867        read_dimension()?.value_of()
4868      };
4869      Ok((Some(f * s), None))
4870    },
4871    Some(f) => match read_keyword(&["filll", "fill", "fil"])? {
4872      Some(fil) => Ok((Some(fixpoint(s as f64 * f, None)), FillCode::from(&fil))),
4873      None => {
4874        let ratio = if mu {
4875          match read_mu_unit()? {
4876            None => {
4877              Warn!(
4878                "expected",
4879                "<unit>",
4880                "Illegal unit of measure (mu inserted)."
4881              );
4882              None
4883            },
4884            some => some,
4885          }
4886        } else {
4887          match read_unit()? {
4888            None => {
4889              Warn!(
4890                "expected",
4891                "<unit>",
4892                "Illegal unit of measure (pt inserted)."
4893              );
4894              None
4895            },
4896            some => some,
4897          }
4898        };
4899        let val = s as f64 * f;
4900        // The <dimen> of a glue component is `scan_dimen`'s (§460).
4901        let sp = match ratio {
4902          Some((num, den)) => dimension_too_large(fixpoint_unit_checked(val, num, den), s < 0)?,
4903          None => fixpoint(val, None),
4904        };
4905        Ok((Some(sp), None))
4906      },
4907    },
4908  }

5233// ```
5234// <factor> = <normal integer> | <decimal constant>
5235// <decimal constant> = . | , | <digit><decimal constant> | <decimal constant><digit>
5236// ```
5237/// Return a number (Rust f64 number)
5238pub fn read_factor() -> Result<Option<f64>> {
5239  // scan_dimen tests the token that ended the digits (`cur_tok`) for the
5240  // decimal point without reading it again (tex.web §448, §452), then puts it
5241  // back once, a `\noexpand`'d token as its plain self (`unread_scanned`).
5242  // An undefined control sequence ending the digits installed its error stub on that read;
5243  // Perl's `readFactor` reads the end token again (Gullet.pm:876-880), which expands it.
5244  let reread_stub = |token_opt: Option<Token>| -> Result<Option<Token>> {
5245    match token_opt {
5246      Some(token) if became_expandable(&token)? => {
5247        unread_one(token);
5248        read_x_token(None, false, None)
5249      },
5250      other => Ok(other),
5251    }
5252  };
5253  let (mut factor, token_opt) = scan_digits(is_decimal_digit)?;
5254  let mut token_opt = reread_stub(token_opt)?;
5255  if let Some(ref token) = token_opt {
5256    let sym = token.get_sym();
5257    if sym == pin!(".") || sym == pin!(",") {
5258      let (fraction, end) = scan_digits(is_decimal_digit)?;
5259      factor = s!("{factor}.{fraction}");
5260      token_opt = reread_stub(end)?;
5261    }
5262  }
5263
5264  // Note: zero is an edge case with the unwrap_or fallback, handle it
5265  if !factor.is_empty() {
5266    // The integer part is `scan_int`'s (§448), held to `infinity` (§445).
5267    let (whole, fraction) = factor.split_once('.').unwrap_or((factor.as_str(), ""));
5268    let value = whole
5269      .parse::<i64>()
5270      .unwrap_or(if whole.is_empty() { 0 } else { i64::MAX });
5271    if number_too_big(value)? != value {
5272      factor = s!("{MAX_INTEGER}.{fraction}");
5273    }
5274    let factor_f64: f64 = factor.parse::<f64>().unwrap_or(0.0);
5275    if let Some(token) = token_opt
5276      && token.get_catcode() != Catcode::SPACE
5277    {
5278      unread_scanned(token);
5279    }
5280    Ok(Some(factor_f64))
5281  } else {
5282    // No digits: back_input, and scan_int reads the token again (§448).
5283    if let Some(token) = token_opt {
5284      unread_scanned(token);
5285    }
5286    match read_normal_integer()? {
5287      None => Ok(None),
5288      Some(n) => Ok(Some(n.value_of() as f64)),
5289    }
5290  }
5291}

5397/// Do something, while reading tokens from a specific Mouth.
5398///
5399/// This reads ONLY from that mouth (or any mouth openned by code in that source),
5400/// and the mouth should end up empty afterwards, and only be closed here.
5401pub fn reading_from_mouth<R, FnR>(mouth: Mouth, reader: FnR) -> Result<R>
5402where FnR: FnOnce() -> Result<R> {
5403  let context_mouth_source = arena::pin(mouth.get_source());
5404  let outer_mouths = mouth_depth();
5405  // A cycle is only a cycle WITHIN one expansion context. Rather than
5406  // resetting the guard history here (an earlier fix that also BLINDED the
5407  // guard to outer loops whose body calls `do_expand` each iteration), give
5408  // this reading context a fresh SERIAL that `read_internal_token_checked`
5409  // mixes into every fingerprint: windows can never match across contexts, so
5410  // consecutive identical short expansions — the math0402448 xymatrix
5411  // per-cell `get_xmarg_id` stream that false-positived as a "loop" — stay
5412  // inert, while the OUTER context's serial (restored on exit) keeps outer
5413  // periodicity intact and detectable. PR #249 review P2-7.
5414  {
5415    let mut g = gullet_mut!();
5416    let outer = g.ctx_serial;
5417    g.ctx_stack.push(outer);
5418    g.ctx_next += 1;
5419    g.ctx_serial = g.ctx_next;
5420  }
5421  open_mouth(mouth, false); // only allow mouth to be explicitly closed here.
5422  let reader_result = reader();
5423  // Reading in this context is over (whether Ok or Err): restore the outer
5424  // context's serial before any cleanup/return path below.
5425  {
5426    let mut g = gullet_mut!();
5427    let restored = g.ctx_stack.pop().unwrap_or(0);
5428    g.ctx_serial = restored;
5429  }
5430  // If the reader returned an error (e.g., Fatal from token limit),
5431  // we STILL need to clean up the mouth to preserve the caller's state.
5432  let results: R = match reader_result {
5433    Ok(v) => v,
5434    Err(e) => {
5435      // Force-close our mouth and any mouth above it, by depth: another of the same name above ours (an "Anonymous
5436      // String", a `\scantokens`'s in a discarded run) is not ours, and stopping at it left ours open and took the
5437      // document's text next (`perfect_kernel_batchr1::a_run_over_budget_leaves_no_trace`).
5438      while mouth_depth() > outer_mouths {
5439        close_mouth(true).ok();
5440      }
5441      // Reset progress counter so subsequent processing isn't immediately killed
5442      let mut g = gullet_mut!();
5443      // (what a budgeted run has left stays what it has left: [`read_budget`])
5444      let progress = g.progress;
5445      g.read_budget_at = g.read_budget_at.map(|at| at.saturating_sub(progress));
5446      g.progress = 0;
5447      g.full_read_at = 0;
5448      drop(g);
5449      return Err(e);
5450    },
5451  };
5452  // `mouth` must still be open, with (at worst) empty autoclosable mouths in front of it.
5453  // Rate-limit the "mouth closed" error — when the gullet gets into a state
5454  // where the cleanup loop keeps finding stale mouths above the target, the
5455  // same error can fire on EVERY caller of reading_from_mouth. Arxiv 0906.1883
5456  // (birkmult + local .cls) can trigger 10K+ such firings, one per stack frame.
5457  // Fatal out after 50 repeat firings so the process surfaces a clear "we lost
5458  // the mouth stack" signal instead of filling the log with identical messages.
5459  fn record_mouth_closed_error() { MOUTH_CLOSED_ERRORS.with(|c| c.set(c.get().saturating_add(1))); }
5460  fn should_emit_mouth_closed() -> bool { MOUTH_CLOSED_ERRORS.with(|c| c.get() < 10) }
5461  fn mouth_closed_budget_exhausted() -> bool { MOUTH_CLOSED_ERRORS.with(|c| c.get() >= 50) }
5462  loop {
5463    let mouth_source = gullet!()
5464      .runtime
5465      .as_ref()
5466      .map(|r| arena::pin(r.mouth.get_source()));
5467    if mouth_source == Some(context_mouth_source) {
5468      close_mouth(true)?;
5469      break;
5470    } else if gullet!().mouthstack.is_empty() {

5473        // BEFORE we hand it to Error! — a following `arena::pin` triggered

5479          "unexpected",
5480          "<closed>",
5481          "Mouth is unexpectedly already closed",
5482          s!("Reading from {src}, but it has already been closed.")
5483        );
5484      }
5485      record_mouth_closed_error();

5487        Fatal!(
5488          Stomach,
5489          Recursion,
5490          "Too many unexpectedly-closed mouth errors (>50); gullet mouth-stack state is inconsistent"
5491        );
5492      }
5493      break;
5494    } else {
5495      let is_autoclosable = gullet!()
5496        .runtime
5497        .as_ref()
5498        .map(|r| r.autoclose)
5499        .unwrap_or(false);
5500      if is_autoclosable {
5501        // Auto-closable mouth (e.g. from \scantokens, raw_tex) — safe to close
5502        close_mouth(true)?;
5503      } else {
5504        // Non-autoclosable mouth that isn't our target — this means our target
5505        // mouth was already consumed. Don't close this mouth (it belongs to an
5506        // outer reading_from_mouth call). Just error and stop.
5507        if should_emit_mouth_closed() {
5508          let src = arena::to_string(context_mouth_source);
5509          Error!(
5510            "unexpected",
5511            "<closed>",
5512            "Mouth is unexpectedly already closed",
5513            s!(
5514              "Reading from {src}, but it has already been closed (found different non-closable mouth on top)."
5515            )
5516          );
5517        }
5518        record_mouth_closed_error();
5519        if mouth_closed_budget_exhausted() {
5520          Fatal!(
5521            Stomach,
5522            Recursion,
5523            "Too many unexpectedly-closed mouth errors (>50); gullet mouth-stack state is inconsistent"
5524          );
5525        }
5526        break;
5527      }
5528    }
5529  }

5543/// own, such as a constructor's argument `{$x}`, which may come wrapped in a further brace pair): only `}`s left in
5544/// its pushback and its mouth exhausted. A `}` read here is the argument's own closing brace, which TeX strips with the

5565/// The tokens the current input holds next, without reading them: the pushback, next-read first, then the current
5566/// mouth's next `max_lines` source lines (at most `max_chars` characters of them, [`Mouth::peek_text`]) tokenized by
5567/// a scratch mouth under the current catcodes as the iterator is drawn on, `max_tokens` in all. Nothing is read,
5568/// expanded or moved, so a catcode change the look-ahead runs past still applies to the text when it is really read.
5569/// Only the current mouth: the input levels outside it are not reached. The cost is bounded by the limits, not by the
5570/// length of the line or of the pending expansion. A look-ahead for a decision about what follows (stomach.rs
5571/// `formula_continues_past_par`, OXIDIZED_DESIGN #553).
5572pub fn peek_upcoming_tokens(
5573  max_lines: usize,
5574  max_chars: usize,
5575  max_tokens: usize,
5576) -> impl Iterator<Item = Token> {
5577  let (pending, text) = {
5578    let gullet = gullet!();
5579    match gullet.runtime.as_ref() {
5580      Some(runtime) => (
5581        runtime
5582          .pushback
5583          .iter()
5584          .rev()
5585          .take(max_tokens)
5586          .copied()
5587          .collect::<Vec<Token>>(),
5588        runtime.mouth.peek_text(max_lines, max_chars),
5589      ),
5590      None => (Vec::new(), String::new()),
5591    }
5592  };
5593  let mut scratch = if pending.len() < max_tokens && !text.is_empty() {
5594    Mouth::new(&text, None).ok()
5595  } else {
5596    None
5597  };
5598  pending
5599    .into_iter()
5600    .chain(std::iter::from_fn(move || scratch.as_mut()?.read_token()))
5601    .take(max_tokens)
5602}
