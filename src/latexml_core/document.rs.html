301  /// [`crate::sxml::FragmentReader`]) as a `Document`, so the pass-2 phases —
302  /// rewrites, math parsing, per-fragment finalize — run on it with the same
303  /// machinery the eager path uses. The insertion point starts at the root;

308    crate::ensure_libxml_init();

310      Some(root) => root,
311      None => doc.as_node(),
312    };
313    let mut document = Document {
314      document: doc,
315      node,
316      node_fonts,
317      ..Document::new_empty_fields()
318    };
319    if let Some(root) = document.document.get_root_element() {
320      document.record_node_ids(&root)?;
321    }
322    Ok(document)
323  }

412  // This should be called before returning the final XML::LibXML::Document to the
413  // outside world.  It resolves the fonts for each node relative to it's
414  // ancestors. It removes the `helper' attributes that store fonts, source
415  // box, etc.
416  pub fn finalize(&mut self) -> Result<()> {
417    // Belt-and-suspenders idstore rebuild before prune_xmduals.
418    // Originally guarded a SIGSEGV on arxiv 1605.08055 where
419    // `mark_xmnode_visibility` dereferenced dangling lookup_id
420    // entries while recursing through XMRef nodes. Cycle 72 audited
421    // the 5 hazard call sites the earlier comment listed (math-parser
422    // `replace_tree` at parser.rs:456/690, `unbind_node` loops at
423    // parser.rs:639/856 and rewrite.rs:522); all 5 now have proper
424    // unrecord_node_ids / remove_node-cascade coverage. The rebuild
425    // call is retained as a safety net pending empirical
426    // 10k-sandbox verification on 1605.08055 — see SYNC_STATUS.md
427    // D3b [~] entry. A fresh DOM walk drops any surviving dangling
428    // entries; duplicates in DOM get modify_id via record_node_ids.
429    self.rebuild_idstore_from_dom()?;
430    // Sweep dangling XMRefs that were specifically created by
431    // amsmath::rearrange_ams_split (tagged with `_split_ref="1"`).
432    // The math parser later absorbs some XMArray cells (inserted
433    // MULOPs, etc.) and the parallel XMWrap refs end up pointing
434    // at vanished targets, cascading through Warning:expected:node
435    // (here) and Error:expected:id (post-process) for ~1500 wp3
436    // canvas papers. Restricting the sweep to `_split_ref` avoids
437    // breaking declare_test's renamed-id case (XMRefs pointing to
438    // `S1.Ex1.m1.1`-style ids that resolve through Perl-faithful
439    // idstore staleness; those don't carry the marker).
440    self.prune_dangling_split_xmrefs()?;
441    self.prune_xmduals()?;
442    if let Some(mut root) = self.document.get_root_element() {
443      self.set_local_font(Rc::new(Font::text_default()));
444      self.finalize_rec(&mut root)?;
445      self.set_rdfa_prefixes();
446      self.apply_document_namespace_declarations(&mut root);
447      self.expire_local_font();
448    }
449    Ok(())
450  }

473    result

476  /// Apply registered document namespace declarations to the root element.
477  /// Perl's RegisterDocumentNamespace stores prefix→URI mappings in the model.
478  /// These must appear as xmlns:prefix="URI" on the root element during serialization.
479  /// Only emit namespaces that are actually used in the document (Perl behavior).
480  fn apply_document_namespace_declarations(&self, root: &mut Node) {
481    let prefixes = model::get_document_namespace_prefixes();
482    // Collect which prefixes are actually used in the document
483    let nsnodes = root.get_namespace_declarations();
484    let existing_prefixes: Vec<String> = nsnodes.iter().map(|ns| ns.get_prefix()).collect();
485    for (prefix, ns_uri) in prefixes {
486      // Skip internal/default namespaces
487      if prefix.is_empty() || prefix == "ltx" || prefix == "xml" {
488        continue;
489      }
490      // Skip namespaces containing "DEFAULT#" (internal model entries)
491      if ns_uri.contains("DEFAULT#") {
492        continue;
493      }
494      // Only add if this prefix appears as a namespace declaration on some child node
495      // (meaning it's actually used in the document)
496      if existing_prefixes.contains(&prefix) {
497        continue; // already declared on root
498      }
499      // Check if any descendant element uses this namespace prefix
500      // by looking for namespace declarations on descendant elements
501      let has_usage = self.has_namespace_usage(root, &prefix);
502      if has_usage {
503        let attr_name = format!("xmlns:{prefix}");
504        root.set_attribute(&attr_name, &ns_uri).ok();
505      }
506    }
507  }

509  /// Check if any descendant of node uses the given namespace prefix.
510  fn has_namespace_usage(&self, node: &Node, prefix: &str) -> bool {
511    // Check attributes of this node for prefix: usage. The
512    // allocation-free check `starts_with(prefix) + byte-at-prefix.len()`
513    // replaces `format!("{prefix}:")` which would heap-allocate per node
514    // visited during the recursive descent.
515    let plen = prefix.len();
516    for (key, _) in node.get_attributes() {
517      if key.len() > plen && key.starts_with(prefix) && key.as_bytes()[plen] == b':' {
518        return true;
519      }
520    }
521    // Check children recursively
522    for child in node.get_child_nodes() {
523      if child.get_type() == Some(NodeType::ElementNode) {
524        // Check if element itself is in this namespace
525        for ns in child.get_namespace_declarations() {
526          if ns.get_prefix() == prefix {
527            return true;
528          }
529        }
530        if self.has_namespace_usage(&child, prefix) {
531          return true;
532        }
533      }
534    }
535    false

540  /// during parsing. After finalization (which includes prune_xmduals), some ids

558        let _ = tok.remove_attribute("xml:id");
559        let _ = tok.remove_attribute_ns("id", xml_ns);

564  /// Iterative implementation of finalize_rec to avoid stack overflow.
565  /// Uses an explicit heap-allocated work stack instead of call-stack recursion.
566  /// Resolves fonts for each node relative to its ancestors, and removes
567  /// helper attributes (_font, _standalone_font, etc).
568  fn finalize_rec(&mut self, node: &mut Node) -> Result<()> {
569    // Work items for the iterative traversal.
570    // Enter: process font declarations and children (text children handled inline,
571    //        element children deferred to the stack).
572    // PostElement: after finalizing an element child, check for font wrapper collapse.
573    // PostWork: after all children processed, remove bookkeeping attrs and expire font.
574    #[allow(clippy::enum_variant_names)]
575    enum Work {
576      Enter(Node),
577      PostElement {
578        child:         Node,
579        parent_qname:  SymStr,
580        was_forcefont: bool,
581      },
582      PostWork {
583        node: Node,
584      },
585    }
586
587    let mut stack: Vec<Work> = Vec::new();
588    // Track nodes that have no visible attributes after bookkeeping removal.
589    // Used in PostElement to decide collapse without calling get_attributes()
590    // (which can crash on corrupted libxml2 attribute lists after replace_node).
591    let mut empty_attr_nodes: HashSet<usize> = HashSet::default();
592    // Defer collapse operations to avoid libxml2 memory corruption.
593    // Collapsing nodes during tree traversal corrupts attribute linked lists
594    // of ancestor nodes. By collecting collapses and running them after
595    // the entire traversal completes, we avoid accessing corrupted state.
596    let mut deferred_collapses: Vec<(Node, SymStr)> = Vec::new();
597    stack.push(Work::Enter(node.clone()));
598
599    while let Some(work) = stack.pop() {
600      match work {
601        Work::Enter(mut current) => {
602          let raw_qname = get_node_qname(&current);
603          // Streaming pass 2: judge the fragment's parse wrapper as the REAL
604          // parent it splices back under, so schema decisions for top-level
605          // fragment content match the eager walk (see the field docs).
606          let qname = match self.fragment_parent_qname {
607            Some(parent) if raw_qname == pin!("ltx:_lxfragment") => parent,
608            _ => raw_qname,
609          };
610          let local_font = self.get_local_font().unwrap();
611          // _standalone_font is typically for metadata that gets extracted out of context
612          let mut declared_font = if current.has_attribute("_standalone_font") {
613            Cow::Borrowed(&*FONT_TEXT_DEFAULT)
614          } else {
615            Cow::Borrowed(&*local_font)
616          };
617
618          // TODO: _pre_comment / _comment insertion requires create_comment support in libxml
619          // wrapper Perl: parent.insertBefore(XML::LibXML::Comment.new(comment), node)
620          // Perl: parent.insertAfter(XML::LibXML::Comment.new(comment), node)
621
622          // Use boxed HashMap to reduce work item size — Font is ~500 bytes per entry
623          let mut pending_declaration: Box<HashMap<String, (String, Font)>> = Box::default();
624
625          if self.has_node_font(&current) {
626            let desired_font = self.get_node_font(&current);
627            *pending_declaration = desired_font.relative_to(&declared_font);
628            if (current.get_first_child().is_some() || current.has_attribute("_force_font"))
629              && !pending_declaration.is_empty()
630            {
631              let mut keys_to_remove: Vec<SymStr> = Vec::new();
632              let mut attrs_to_set: Vec<(SymStr, SymStr)> = Vec::new();
633              for (key, (value, properties)) in pending_declaration.iter() {
634                if model::can_have_attribute(qname, arena::pin(key)) {
635                  let key_sym = arena::pin(key);
636                  attrs_to_set.push((key_sym, arena::pin(value)));
637                  // Merge to set the font currently in effect
638                  declared_font = Cow::Owned(declared_font.merge_ref(properties));
639                  keys_to_remove.push(key_sym);
640                }
641              }
642
643              for (key, mut value) in attrs_to_set {
644                if key == pin!("class") {
645                  // Merge and sort class values alphabetically, matching Perl's behavior
646                  if let Some(ovalue) = current.get_attribute("class") {
647                    let new_s = arena::with(value, |s| s.to_string());
648                    let mut classes: Vec<&str> = new_s
649                      .split_whitespace()
650                      .chain(ovalue.split_whitespace())
651                      .collect();
652                    classes.sort_unstable();
653                    classes.dedup();
654                    value = arena::pin(classes.join(" "));
655                  }
656                }
657                // Resolve to owned Strings before calling set_attribute,
658                // to avoid holding an arena borrow while set_attribute may need arena::pin
659                // for schema-based attribute filtering (canHaveAttribute check).
660                let key_s = arena::with(key, |s| s.to_string());
661                let value_s = arena::with(value, |s| s.to_string());
662                self.set_attribute(&mut current, &key_s, &value_s)?;
663              }
664              for key in keys_to_remove {
665                arena::with(key, |key_str| pending_declaration.remove(key_str));
666              }
667            }
668          }
669          // Optionally add ids to all nodes (AFTER all parsing, rearrangement, etc)
670          // (gate on the RAW name: a fragment wrapper standing in for its
671          // real parent must not mint an id — it is never serialized).
672          if raw_qname != pin!("ltx:document")
673            && raw_qname != pin!("ltx:_lxfragment")
674            && state::lookup_bool("GENERATE_IDS")
675            && !current.has_attribute("xml:id")
676            && !current.has_attribute("id") // SVG elements with plain id don't need xml:id
677            && arena::with(qname, |qname_str| can_have_attribute(qname_str, "xml:id"))
678          {
679            self.generate_id(&mut current, "")?;
680          }
681          self.set_local_font(Rc::new(declared_font.into_owned()));
682
683          // Process children using the snapshot from get_child_nodes().
684          // Element children are deferred to the stack; text children are handled inline.
685          // PostWork is pushed first (runs after all children), then children in reverse.
686          //
687          // No attribute snapshot is taken here: PostWork deliberately
688          // re-derives the bookkeeping set at its own time (see the comment
689          // there — descendant `generate_id` calls write `_ID_counter_*` onto
690          // this node AFTER Enter). Capturing it here cost a full
691          // `get_attributes()` HashMap plus a String per attribute on every
692          // node — ~10M nodes on the streaming witness — and the result was
693          // never read.
694          stack.push(Work::PostWork { node: current.clone() });
695
696          let children = current.get_child_nodes();
697          // Collect work items forward, then push in reverse for left-to-right processing
698          let mut child_work: Vec<Work> = Vec::new();
699          for child in &children {
700            let child_type = child.get_type();
701            if child_type == Some(NodeType::ElementNode) {
702              let was_forcefont = child.has_attribute("_force_font");
703              // Enter first, PostElement after (reversed when pushed to stack)
704              child_work.push(Work::Enter(child.clone()));
705              child_work.push(Work::PostElement {
706                child: child.clone(),
707                parent_qname: qname,
708                was_forcefont,
709              });
710            } else if child_type == Some(NodeType::TextNode) {
711              // Text node: wrap with font element if needed (handled inline)
712              let mut text_keys_to_remove = Vec::new();
713              for key in pending_declaration.keys() {
714                if !can_have_attribute(FONT_ELEMENT_NAME, key) {
715                  text_keys_to_remove.push(key.clone());
716                }
717              }
718              for key in text_keys_to_remove {
719                pending_declaration.remove(&key);
720              }
721              if can_contain(&current, FONT_ELEMENT_NAME)
722                && !pending_declaration.is_empty()
723                && let Some(mut text) = self.wrap_nodes(FONT_ELEMENT_NAME, vec![child.clone()])?
724              {
725                for (key, (value, _properties)) in pending_declaration.iter() {
726                  self.set_attribute(&mut text, key, value)?;
727                }
728                // Text wrapper finalization is shallow (only text content), push to stack
729                child_work.push(Work::Enter(text));
730              }
731            }
732          }
733          // Push in reverse so left-to-right processing order is maintained
734          for work_item in child_work.into_iter().rev() {
735            stack.push(work_item);
736          }
737        },
738        Work::PostElement {
739          child,
740          parent_qname,
741          was_forcefont,
742        } => {
743          // After finalizing a child element, check if it should be collapsed.
744          // Use empty_attr_nodes set instead of child.get_attributes().is_empty()
745          // to avoid traversing the attribute linked list, which can be corrupted
746          // by prior replace_node operations in libxml2.
747          // Defer the actual collapse to after the traversal completes, since
748          // replace_node can corrupt ancestor attribute lists in libxml2.
749          if (get_node_qname(&child) == pin!("ltx:text"))
750            && !was_forcefont
751            && empty_attr_nodes.contains(&child.to_hashable())
752          {
753            let grandchildren = child.get_child_nodes();
754            if grandchildren
755              .iter()
756              .all(|gchild| can_contain_qsym(parent_qname, get_node_qname(gchild)))
757            {
758              deferred_collapses.push((child, parent_qname));
759            }
760          }
761        },
762        Work::PostWork { mut node } => {
763          // Mirrors Perl `Document.pm:452`: at finalize time, ANY attribute
764          // whose name starts with `_` is internal bookkeeping and gets
765          // stripped. The set MUST be derived here, at PostWork time, and not
766          // captured back at Enter, because descendant
767          // `generate_id` calls can write `_ID_counter_<prefix>_` attributes
768          // ONTO the current node (their nearest-id-bearing ancestor) during
769          // child traversal — those late-added attrs were missing from the
770          // Enter snapshot and would otherwise leak into the output XML,
771          // causing duplicate xml:id collisions in the post-processing
772          // libxml2 validator (1312.5864 cluster: 70× `S8.T5.m2241 already
773          // defined`, where the Math element carried `_ID_counter__="1"`
774          // populated post-Enter).
775          //
776          // EXCEPT the streaming pass-2 parse wrapper: it is never
777          // serialized, and its `_ID_counter_*` attrs are how id numbering
778          // carries ACROSS segments (read back after each segment — see
779          // `streaming_pass2`).
780          if get_node_qname(&node) != pin!("ltx:_lxfragment") {
781            let attrs_now = node.get_attributes();
782            let total_attrs = attrs_now.len();
783            let bookkeeping_attrs: Vec<&String> = attrs_now
784              .keys()
785              .filter(|name| name.starts_with('_'))
786              .collect();
787            let bookkeeping_count = bookkeeping_attrs.len();
788            for name in bookkeeping_attrs {
789              let _ = node.remove_attribute(name);
790            }
791            // If all attributes were bookkeeping, the node now has empty attrs.
792            if total_attrs <= bookkeeping_count {
793              empty_attr_nodes.insert(node.to_hashable());
794            }
795          }
796          self.expire_local_font();
797        },
798      }
799    }
800    // Execute deferred font wrapper collapses now that the entire tree has been
801    // finalized. Process from deepest (last) to shallowest (first) to avoid
802    // corrupting ancestor attribute lists during the replacement operations.
803    // This deferred approach prevents libxml2 memory corruption that occurs
804    // when replace_node runs during tree traversal.
805    for (child, _parent_qname) in deferred_collapses.into_iter().rev() {
806      let grandchildren = child.get_child_nodes();
807      if grandchildren.is_empty() {
808        // Empty font wrapper — just remove it
809        self.remove_node(child);
810      } else {
811        Debug!(
812          "will replace {} grandchildren nodes in finalize_rec (deferred)",
813          grandchildren.len()
814        );
815        self.replace_node(child, grandchildren)?;
816      }
817    }
818    Ok(())
819  }

821  /// Document construction at the Current Insertion Point.
822  ///
823  /// absorb the given $box into the DOM (called from constructors).
824  /// This will return a list of whatever nodes were created.
825  /// Note that this may include nodes that are children of other nodes in the list
826  /// or nodes that are no longer in the document.
827  /// Also, note that when a text nodes is appended to, the complete text node is in the list,
828  /// not just the portion that was added.
829  /// [Note that recording the nodes being constructed isn't all that costly,
830  /// but filtering them for parent/child relations IS, particularly since it usually isn't needed]
831  ///
832  /// A box that is a TBox, or List, or Whatsit, is responsible for carrying out
833  /// its own insertion, but it should ultimately call methods of Document
834  /// that will record the nodes that were created.
835  /// $box can also be a plain string (Digested::Postponed)
836  /// which will be inserted according to whatever
837  /// font, mode, etc, are in %props.
838  pub fn absorb(&mut self, object: &Digested, props_opt: Option<SymHashMap<Stored>>) -> Result<()> {
839    use DigestedData::*;
840    let props = props_opt.unwrap_or_default();
841    let mut boxes = vec![Cow::Borrowed(object)];
842    while let Some(front_box) = boxes.pop() {
843      // Cooperative guard tick — the SAME one the digestion loops run
844      // (`stomach.rs`, three sites). Build had none, and Build is where a large
845      // document actually peaks: measured 2026-07-29 on 800k words of plain
846      // prose, digest ends under 2 GB while Build takes it to 6.4 GB (54 % of
847      // wall, ~70 % of peak RSS). So `--max-memory` guarded only the cheap
848      // phase, and what a user hit was the HARD watchdog: SIGKILL, exit 137, no
849      // `Fatal:` line, no partial document — the 0-byte output reported against
850      // rc4 on a 131 MB source. The tick makes the ceiling cooperative here
851      // too, so an over-budget document degrades to a graceful Fatal that the
852      // caller can salvage from (`core_interface::convert_document`).
853      //
854      // Cost is negligible: `check_timeout` only samples RSS every ~1024 calls,
855      // and this loop already does far more work per iteration than a counter
856      // increment.
857      crate::stomach::check_timeout()?;
858      match front_box.data() {
859        List(list) => {
860          // Simply unwind Lists to avoid unneccessary recursion; This occurs quite frequently!
861          for tbox in list.borrow().unlist().into_iter().rev() {
862            boxes.push(Cow::Owned(tbox));
863          }
864        },
865        // A Proper Box or Whatsit? Absorb it.
866        TBox(digested) => {
867          self.set_box_to_absorb(Some((*front_box).clone()));
868          self.init_constructed_nodes();
869          digested.borrow().be_absorbed(self)?;
870          // record these for OUTER caller, but return only the most recent set
871          self.close_constructed_nodes();
872          self.expire_box_to_absorb();
873        },
874        Whatsit(digested) => {
875          self.set_box_to_absorb(Some((*front_box).clone()));
876          self.init_constructed_nodes();
877          digested.borrow().be_absorbed(self)?;
878          self.close_constructed_nodes();
879          self.expire_box_to_absorb();
880        },
881        Alignment(alignment) => {
882          self.set_box_to_absorb(Some((*front_box).clone()));
883          self.init_constructed_nodes();
884          // A self-referential alignment (transitively contains itself) would
885          // re-enter `borrow_mut` here and panic ("already borrowed"). Guard like
886          // the size-computation traversal (digested.rs) and skip the re-entrant
887          // absorption to break the cycle. Witness: astro-ph0310145 et al.
888          match alignment.try_borrow_mut() {
889            Ok(mut a) => {
890              a.be_absorbed_mut(self)?;
891            },
892            Err(_) => {
893              // Degrading MUST be loud (fail-toward-flagging): the whole
894              // alignment's content is being dropped from the output.
895              Error!(
896                "unexpected",
897                "self_referential_alignment",
898                "Skipping absorption of a self-referential alignment (its content is lost)"
899              );
900            },
901          }
902          self.close_constructed_nodes();
903          self.expire_box_to_absorb();
904        },
905        Comment(comment) => {
906          comment.be_absorbed(self)?;
907        },
908        Postponed(tokens) => {
909          let text_font_opt = if let Some(Stored::Font(prop_font)) = props.get("font") {
910            Some(Rc::clone(prop_font))
911          } else {
912            match self.box_to_absorb {
913              Some(ref thisbox) => thisbox.get_font()?,
914              None => None,
915            }
916          };
917          let text_font = text_font_opt.unwrap_or_default();
918          if let Some(new_text) = self.open_text(&tokens.to_string(), &text_font)? {
919            self.record_constructed_node(&new_text);
920          }
921        },
922        KeyVals(kv) => {
923          // When KeyVals appear in body absorption (e.g. #1 for RequiredKeyVals),
924          // convert them to text representation matching Perl's stringify behavior.
925          let text = kv.to_string();
926          if !text.is_empty() {
927            let text_font = match self.box_to_absorb {
928              Some(ref thisbox) => thisbox.get_font()?.unwrap_or_default(),
929              None => Rc::default(),
930            };
931            if let Some(new_text) = self.open_text(&text, &text_font)? {
932              self.record_constructed_node(&new_text);
933            }
934          }
935        },
936        RegisterValue(_) => {
937          // RegisterValue should not normally appear in the absorption pipeline.
938        },
939      }
940    }
941    Ok(())
942  }

944  fn init_constructed_nodes(&mut self) {
945    // Pop a buffer from the free-list (retaining its previously-grown capacity);
946    // fall back to a fresh empty Vec if the pool is dry. Swap it in as the new
947    // inner frame; the outgoing outer frame goes onto the save stack.
948    let fresh = self.reusable_node_buffers.pop().unwrap_or_default();
949    let prev = std::mem::replace(&mut self.constructed_nodes, fresh);
950    self.localized_constructed_nodes.push(prev);
951  }
952
953  /// Close the current constructed-nodes frame, restoring the outer frame and
954  /// re-recording the inner frame's nodes into it. The drained inner buffer
955  /// is returned to `reusable_node_buffers` (empty but with capacity intact)
956  /// so the next `init_constructed_nodes` can reuse it without allocating.
957  fn close_constructed_nodes(&mut self) {
958    let outer = self.localized_constructed_nodes.pop().unwrap_or_default();
959    let mut inner = std::mem::replace(&mut self.constructed_nodes, outer);
960    for n in inner.drain(..) {
961      self.record_constructed_node(&n);
962    }
963    // `inner` is now empty; its capacity is preserved. Return it to the pool.
964    self.reusable_node_buffers.push(inner);
965  }

968  /// This is a refactored `else` cases from the main absorb routine, to allow for better type
969  /// hygiene
970  pub fn absorb_string(
971    &mut self,
972    object: &str,
973    props: &SymHashMap<Stored>,
974  ) -> Result<Option<Node>> {
975    // Else, plain string in text mode.
976    let ismath: bool = match props.get("isMath") {
977      Some(v) => v.into(),
978      None => false,
979    };
980    if !ismath {
981      // Perf: avoid cloning Rc<Font> into owned Font in the common case.
982      // We pull out the Rc (shared reference is fine since open_text only
983      // borrows the Font, not self) and go through Cow<Font>.
984      let font_opt: Option<Rc<Font>> = match props.get("font") {
985        Some(Stored::Font(fnt)) => Some(Rc::clone(fnt)),
986        Some(Stored::FontDirective(FontDirective::Asset(fnt))) => Some(Rc::clone(fnt)),
987        _ => None,
988      };
989      if let Some(fnt) = font_opt {
990        return self.open_text(object, &fnt);
991      }
992      if let Some(Stored::FontDirective(FontDirective::Closure(code))) = props.get("font") {
993        let fnt = code(None)?;
994        return self.open_text(object, &fnt);
995      }
996      // Fallback to box_to_absorb font.
997      let fnt = self.box_to_absorb.as_ref().unwrap().get_font()?.unwrap();
998      self.open_text(object, &fnt)
999    } else if get_node_qname(&self.node) == pin!("ltx:XMTok") {
1000      // Or plain string in math mode.
1001      // Note text nodes can ONLY appear in <XMTok> or <text>!!!
1002      // Have we already opened an XMTok? Then insert into it.
1003      Ok(Some(self.open_math_text_internal(object)?))
1004    // Else create the XMTok now.
1005    } else {
1006      // Odd case: constructors that work in math & text can insert raw strings in Math mode.
1007      let font_math_opt = match props.get("font") {
1008        Some(Stored::Font(fnt)) => Some(Cow::Borrowed(&**fnt)),
1009        Some(Stored::FontDirective(FontDirective::Asset(fnt))) => Some(Cow::Borrowed(&**fnt)),
1010        Some(Stored::FontDirective(FontDirective::Closure(code))) => Some(Cow::Owned(code(None)?)),
1011        _ => None,
1012      };
1013      if let Some(font_math) = font_math_opt {
1014        Ok(Some(self.insert_math_token(
1015          object,
1016          HashMap::default(),
1017          Some(&font_math),
1018        )?))
1019      } else {
1020        Ok(Some(self.insert_math_token(
1021          object,
1022          HashMap::default(),
1023          None,
1024        )?))
1025      }
1026    }
1027  }

1052  /// Shorthand for open,absorb,close, but returns the new node.
1053  pub fn insert_element(
1054    &mut self,
1055    qname: &str,
1056    content: Vec<&Digested>,
1057    attrib: Option<HashMap<String, String>>,
1058  ) -> Result<Node> {
1059    // TODO: Quickly hacked together, needs a careful refactor with all .clone()
1060    // calls removed
1061    let node = self.open_element(qname, attrib, None)?;
1062    // Debug!("Inserting element {:?} with body: {:?}", qname, content);
1063    for digested in content {
1064      self.absorb(digested, None)?;
1065    }
1066
1067    // Walk up from the current insertion point to learn whether `node` is still
1068    // an open ancestor (so it can be closed below). `self.node` may be the
1069    // parentless document root, and an intermediate node may be detached — a
1070    // `None` parent there simply means `node` is NOT an ancestor (stop the walk),
1071    // so don't `.unwrap()`-panic on it. Witnesses: hep-th0201062, 1009.0637.
1072    let mut c = self.node.get_parent();

1074      && c.as_ref() != Some(&node)
1075      && c.as_ref().unwrap().get_type() != Some(NodeType::DocumentNode)

1081          Some(_) => Some(parent),
1082        },
1083      };
1084    }

1087    // close it if it is still open.

1094  /// Insert a ProcessingInstruction of the form <?op attr=value ...?>
1095  /// Does NOT move the current insertion point to the PI,
1096  /// but may move up past a text node.
1097  // Rust note: attrib would have been best as Vec<(String,String)> but
1098  // currently quote!() doesn't work out of the box on them
1099  pub fn insert_pi(
1100    &mut self,
1101    op: &str,
1102    attributes_opt: Option<HashMap<String, String>>,
1103  ) -> Result<()> {
1104    let mut attr_data = Vec::new();
1105    if let Some(attributes) = attributes_opt {
1106      let mut keys = vec!["class", "package", "options"];
1107      let other_keys = attributes
1108        .keys()
1109        .filter(|k| k.as_str() != "class" && k.as_str() != "package" && k.as_str() != "options")
1110        .map(String::as_str)
1111        .collect::<Vec<_>>();
1112      keys.extend(other_keys);
1113      for key in keys {
1114        if let Some(value) = attributes.get(key) {
1115          attr_data.push(s!("{}=\"{}\"", key, value));
1116        }
1117      }
1118    }
1119    // self.close_text_internal();  // Close any open text node
1120    let mut pi_node = self
1121      .document
1122      .create_processing_instruction(op, &attr_data.join(" "))
1123      .unwrap();
1124    // Perl (Core/Document.pm insertPI): a PI always lands before the root
1125    // element. When the root already exists — a mid-body or document-final PI
1126    // (#683's `<?latexml nominal-font-size?>` is emitted at finalization) —
1127    // insert directly before it, whatever the current insertion point. Only
1128    // when NO root has been opened yet (the usual pre-`\begin{document}` case)
1129    // queue the PI in `pending`, to be flushed the moment the root opens
1130    // (`open_element_at`). The old code keyed solely on the insertion point
1131    // being the DocumentNode and so re-queued a post-root PI into `pending`,
1132    // which drains once and never again — the PI was silently lost.
1133    if let Some(mut root) = self.document.get_root_element() {
1134      root.add_prev_sibling(&mut pi_node)?;
1135    } else if self.node.get_type() == Some(NodeType::DocumentNode) {
1136      self.pending.push(pi_node);
1137    } else {
1138      self.node.add_prev_sibling(&mut pi_node)?;
1139    }
1140    Ok(())
1141  }

1179  /// Stamp a freshly-opened element with its source range as a
1180  /// `data-sourcepos` attribute, for the `--source-map` feature (issues
1181  /// #47/#92). The range comes from the construct currently being absorbed
1182  /// (`box_to_absorb`); the integer file `tag` is resolved through the
1183  /// document-level `sources` table (`state::source_tag`) so no path is
1184  /// inlined. See `docs/performance/SOURCE_PROVENANCE.md` §0/§2.
1185  ///
1186  /// Math is kept **opaque** per the MVP scope: the `ltx:Math` wrapper is
1187  /// stamped, but its `ltx:XM*` MathML internals are skipped (the Marpa
1188  /// math parser has no locator awareness — §7 A.3). Only invoked when the
1189  /// source-map switch is on (the caller gates it).
1190  fn stamp_source_locator(&mut self, node: &Node, qname: &str) {
1191    // Math internals: stamp only the leaf token elements (`ltx:XMTok` — the
1192    // operators / identifiers / numbers) when token-locators gives them a real
1193    // located box locator (the math char's source origin). This is the per-token
1194    // in-equation provenance step (§7 A.3). The structural XM* (XMApp/XMDual/
1195    // XMArray) are rebuilt by the Marpa parser — created directly, not via
1196    // `open_element` — so they never reach here; the remaining digestion-built
1197    // wrappers (XMArg/XMHint/XMText/XMRef/XMWrap) stay opaque. The `data:sourcepos`
1198    // rides the XMTok element through the parser's restructuring (attribute on a
1199    // reparented node) and through the XMath→MathML XSLT.
1200    //
1201    // Gated at compile time: feature-OFF keeps math fully opaque (the MVP scope
1202    // and the golden's math-opacity assertion); only the token-locators build
1203    // exposes the located XMTok leaves.
1204    #[cfg(not(feature = "token-locators"))]
1205    if qname.starts_with("ltx:XM") {
1206      return;
1207    }
1208    #[cfg(feature = "token-locators")]
1209    if qname.starts_with("ltx:XM") && qname != "ltx:XMTok" {
1210      return;
1211    }
1212    // Read the pre-captured Copy locator — never re-borrow the box here.
1213    let Some(loc) = self.current_box_locator else {
1214      return;
1215    };
1216    // Skip locators with no real source position (default/synthetic).
1217    if loc.from_line == 0 {
1218      return;
1219    }
1220    // User-source only (§7.B): emit a navigable locator only into an editable
1221    // user document — `.tex`/`.ltx`, plus the bibliography sources `.bbl` (the
1222    // BibTeX-generated, but author-editable, list of `\bibitem`s) and `.bib`
1223    // (BibTeX database entries). All four are files the editor may legitimately
1224    // scroll into. This skips both synthetic default locators (whose source is
1225    // `…/locator.rs`, from `Locator::default()`'s `file!()`) and foreign
1226    // package/class files (`.sty`/`.cls`/…) — the editor must never scroll into
1227    // those. Foreign/unstamped elements inherit their nearest user-source
1228    // ancestor's range client-side (DOM walk-up). (MVP heuristic; a tracked
1229    // user-input set would be more precise.)
1230    let src = loc.get_source();
1231    let is_user_source = arena::with(src, |s| {
1232      let s = s.to_ascii_lowercase();
1233      s.ends_with(".tex") || s.ends_with(".ltx") || s.ends_with(".bbl") || s.ends_with(".bib")
1234    });
1235    if !is_user_source {
1236      return;
1237    }
1238    let tag = state::source_tag(src);
1239    // Emit in LaTeXML's `data:` namespace (`http://dlmf.nist.gov/LaTeXML/data`,
1240    // registered in `base_schema.rs:19`). The post XSLT's `copy_foreign_attributes`
1241    // path converts a `data:`-prefixed *foreign-namespaced* attribute to the HTML
1242    // `data-sourcepos` attribute (`LaTeXML-common.xsl`: `data:` prefix → `data-…`
1243    // when `USE_DATA_ATTRIBUTES` = true, i.e. HTML5). Faithful to Perl LaTeXML's
1244    // foreign-attribute convention; no XSLT change needed. The general namespaced-
1245    // attribute binding lives in `set_attribute` (shared with `aria:` etc.).
1246    let mut n = node.clone();
1247    let _ = self.set_attribute(&mut n, "data:sourcepos", &loc.to_sourcepos(tag));
1248  }

1252  /// Since this is an "explicit request", we're currently skipping over those nodes,
1253  /// ie. we're automatically closing them, even if they're the same type as we're asking to
1254  /// close!!! This is kinda risky! Maybe we should try to request closing of specific nodes.
1255  pub fn close_element(&mut self, qname: &str) -> Result<Option<Node>> {
1256    Debug!(
1257      "document",
1258      "close_element",
1259      s!(
1260        "Close element {:?} at {:?}",
1261        qname,
1262        self.document.node_to_string(&self.node)
1263      )
1264    );
1265    let qsym = arena::pin(qname);
1266    self.close_text_internal()?;
1267    let mut node = self.node.clone();
1268    let mut cant_close = Vec::new();
1269    while node.get_type() != Some(NodeType::DocumentNode) {
1270      let t = get_node_qname(&node);
1271      // autoclose until node of same name BUT also close nodes opened' for font
1272      // switches!
1273      if t == qsym && !(t == pin!("ltx:text") && node.has_attribute("_fontswitch")) {
1274        break;
1275      }
1276      if !can_auto_close(&node) {
1277        cant_close.push(node.clone());
1278      }
1279      match node.get_parent() {
1280        Some(parent) => {
1281          node = parent;
1282        },
1283        None => break, // detached node — treat as not found
1284      }
1285    }
1286
1287    if node.get_type() == Some(NodeType::DocumentNode) {
1288      // Didn't find $qname at all!!
1289      let qname_msg: String = match qname {
1290        "#PCDATA" => qname.to_owned(),
1291        _ => s!("</{qname}>"),
1292      };
1293      let message = s!(
1294        "Attempt to close {}, which isn't open. Currently in {}",
1295        qname_msg,
1296        self.get_insertion_context(None)?
1297      );
1298      Error!("malformed", qname, message);
1299      Ok(None)
1300    } else {
1301      // Found node.
1302      if !cant_close.is_empty() {
1303        // Intervening non-auto-closeable nodes!!
1304        let message = s!(
1305          "Closing tag {:?} whose open descendents do not auto-close. Descendants are {:?}",
1306          qname,
1307          cant_close
1308            .into_iter()
1309            .map(|n| n.get_name())
1310            .collect::<Vec<String>>()
1311            .join(",")
1312        );
1313        Error!("malformed", qname, message);
1314      }
1315      // So, now close up to the desired node.
1316      self.close_node_internal(&node)?;
1317      Ok(Some(node))
1318    }
1319  }

1339  /// Check whether it is possible to close each element in @tags,
1340  /// any intervening nodes must be autocloseable.
1341  /// returning the last Some(node) that would be closed if it is possible,
1342  /// otherwise None
1343  pub fn is_closeable<T: IntoVDQS>(&self, tags: T) -> Option<Node> {
1344    let mut tags: VecDeque<SymStr> = tags.into_vdqs();
1345    let mut node_opt = if self.node.get_type() == Some(NodeType::TextNode) {
1346      self.node.get_parent()
1347    } else {
1348      Some(self.node.clone())
1349    };
1350    while let Some(qname) = tags.pop_front() {
1351      'inner: loop {
1352        let node: &Node = match node_opt {
1353          None => break,
1354          Some(ref n) => n,
1355        };
1356        let node_type = node.get_type();
1357        if node_type == Some(NodeType::DocumentNode) || node_type.is_none() {
1358          return None;
1359        }
1360        let this_qname = get_node_qname(node);
1361        if this_qname == qname {
1362          break 'inner;
1363        }
1364        if !can_auto_close(node) {
1365          Debug!(
1366            "It was impossible to autoclose node: {:?}",
1367            self.document.node_to_string(node)
1368          );
1369          return None;
1370        }
1371        node_opt = node.get_parent();
1372      }
1373      if !tags.is_empty()
1374        && let Some(node) = node_opt
1375      {
1376        node_opt = node.get_parent();
1377      }
1378    }
1379    node_opt
1380  }

1574  /// get the actions that should be performed on afterOpen or afterClose
1575  pub fn get_tag_action_list(
1576    &self,
1577    tag: SymStr,
1578    when: TagOptionName,
1579  ) -> Vec<TagConstructionClosure> {
1580    use self::tag::TagOptionName::*;
1581    // my ($p, $n) = (undef, $tag);
1582    // if ($tag =~ /^([^:]+):(.+)$/) {
1583    //   ($p, $n) = ($1, $2); }
1584    let mut when_early = None;
1585    let mut when_late = None;
1586
1587    match when {
1588      AfterOpen => {
1589        when_early = Some(AfterOpenEarly);
1590        when_late = Some(AfterOpenLate);
1591      },
1592      AfterClose => {
1593        when_early = Some(AfterCloseEarly);
1594        when_late = Some(AfterCloseLate);
1595      },
1596      _ => {},
1597    };
1598
1599    let mut actions = Vec::new();
1600    // Borrow the per-tag and `ltx:*` option hashes in place instead of cloning
1601    // the whole `TagOptions` map twice per call (this runs on every element
1602    // open/close — ~8.8M calls across the witness corpus). We only need to copy
1603    // the matched action lists, and those hold `Rc<>` closures, so cloning each
1604    // is just a pointer + refcount bump. `with_tag_property` returns `None` for
1605    // an absent tag, equivalent to an empty `TagOptions` for a read (the only
1606    // difference vs `get_tag_property` is it skips vivifying an empty default,
1607    // which has no observable effect).
1608    state::with_tag_property(tag, |tag_hash| {
1609      state::with_tag_property(pin!("ltx:*"), |all_hash| {
1610        let mut collect = |opts: Option<&tag::TagOptions>, key: &TagOptionName| {
1611          if let Some(v) = opts.and_then(|o| o.get(key)) {
1612            actions.extend(v.iter().cloned());
1613          }

1672  /// Splice a processed segment's text into `out`, resolving NESTED
1673  /// placeholders recursively (final assembly only). A segment spilled by a
1674  /// closing ancestor keeps its children's `<_spilled_ ref=…/>` markers as
1675  /// literal elements — inlining their text at spill time rebuilt multi-GB
1676  /// segments that pass 2 could not re-materialize. Each literal marker
1677  /// occupies either a full line (`indent + element + newline`, the
1678  /// indenting serialization) or an exact element span (noindent contexts);
1679  /// the inner segment text carries its own indentation/newline, so the
1680  /// marker's surrounding whitespace is dropped — but ONLY when the marker
1681  /// is line-positioned, so content spaces in noindent contexts survive.
1682  /// `<` is escaped everywhere in serialized text/attribute content, so the
1683  /// scan cannot match anything but real markers.
1684  fn splice_segment_text(&self, out: &mut String, text: &str) {
1685    let mut rest = text;
1686    while let Some(pos) = rest.find("<_spilled_ ") {
1687      let (before, from) = rest.split_at(pos);
1688      let Some(elem_end) = from.find("/>").map(|e| e + 2) else {
1689        break; // malformed marker: emit as-is below
1690      };
1691      let elem = &from[..elem_end];
1692      let seg_opt = elem
1693        .split_once("ref=\"")
1694        .and_then(|(_, tail)| tail.split_once('"'))
1695        .and_then(|(num, _)| num.parse::<u32>().ok())
1696        .map(crate::sxml::SegmentId);
1697      let inner = seg_opt.and_then(|seg| {
1698        self
1699          .spill_store
1700          .as_ref()
1701          .and_then(|store| store.read_segment(seg).ok())
1702      });
1703      let Some(inner) = inner else {
1704        emit_error(
1705          "spill",
1706          "unresolved",
1707          &format!("a nested disk-staged segment could not be spliced back ({elem})"),
1708        );
1709        out.push_str(before);
1710        out.push_str("<!-- latexml-oxide: LOST staged segment -->\n");
1711        rest = &from[elem_end..];
1712        continue;
1713      };
1714      // Drop the marker's own line whitespace when line-positioned.
1715      let mut cut = before.len();
1716      while cut > 0 && before.as_bytes()[cut - 1] == b' ' {
1717        cut -= 1;
1718      }
1719      let line_positioned = cut == 0 || before.as_bytes()[cut - 1] == b'\n';
1720      if line_positioned {
1721        out.push_str(&before[..cut]);
1722      } else {
1723        out.push_str(before);
1724      }
1725      self.splice_segment_text(out, &inner);
1726      rest = &from[elem_end..];
1727      if line_positioned && rest.starts_with('\n') {
1728        rest = &rest[1..];
1729      }
1730    }
1731    out.push_str(rest);
1732  }

1734  /// Buffer-threaded core of [`serialize_aux`]. Writes the subtree directly into
1735  /// `out` instead of building (and re-copying) a fresh `String` per node — the
1736  /// recursive return-a-String pattern was the single largest byte source in the
1737  /// allocation profile (~2.8 GB across the witness corpus). Output is identical.
1738  fn serialize_into(
1739    &self,
1740    out: &mut String,
1741    node: &Node,
1742    depth: usize,
1743    noindent: bool,
1744    heuristic: bool,
1745  ) {
1746    // Push `depth` levels of two-space indent without the `"  ".repeat(depth)`
1747    // allocation the old code paid on every call (including text nodes).
1748    fn push_indent(out: &mut String, depth: usize) {
1749      for _ in 0..depth {
1750        out.push_str("  ");
1751      }
1752    }
1753    // `spill_flat` means "emit no decorative whitespace", which is exactly what
1754    // `noindent` already means at every emission site — so fold it in ONCE here
1755    // rather than testing it at each. It suppresses formatting only; the
1756    // schema-driven `noindent_children` still governs mixed content, whose
1757    // whitespace is meaningful and was never indented anyway.
1758    let noindent = noindent || self.spill_flat;
1759
1760    match node.get_type() {
1761      Some(NodeType::DocumentNode) => {
1762        out.push_str("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n");
1763        if let Some(child) = node.get_first_child() {
1764          self.serialize_into(out, &child, depth, noindent, heuristic);
1765          let mut current_child = child;
1766          while let Some(sibling) = current_child.get_next_sibling() {
1767            self.serialize_into(out, &sibling, depth, noindent, heuristic);
1768            current_child = sibling;
1769          }
1770        }
1771      },
1772      Some(NodeType::ElementNode) => {
1773        // Get the qualified name (prefix:localname) for namespace-prefixed elements
1774        let local_name = node.get_name();
1775        // Streaming assembly: a spill placeholder serializes as the processed
1776        // segment's text, spliced verbatim from disk (same depth/noindent by
1777        // construction — the segment was serialized with the exact values this
1778        // recursion carries here). A placeholder that cannot be resolved is
1779        // LOST CONTENT: flag it loudly in the log AND the output, never drop
1780        // it silently (fail toward flagging).
1781        if local_name == SPILL_PLACEHOLDER && !self.literal_placeholders {
1782          let resolved = self
1783            .spill_store
1784            .as_ref()
1785            .zip(node.get_attribute("ref"))
1786            .and_then(|(store, r)| {
1787              let seg = crate::sxml::SegmentId(r.parse::<u32>().ok()?);
1788              store.read_segment(seg).ok()
1789            });
1790          match resolved {
1791            Some(text) => self.splice_segment_text(out, &text),
1792            None => {
1793              // `Error!` can early-return and this serializer returns `()`,
1794              // so raise through the logger; the marker keeps the loss
1795              // visible in the output itself.
1796              emit_error(
1797                "spill",
1798                "unresolved",
1799                &format!(
1800                  "a disk-staged segment could not be spliced back (ref={:?})",
1801                  node.get_attribute("ref")
1802                ),
1803              );
1804              out.push_str("<!-- latexml-oxide: LOST staged segment -->\n");
1805            },
1806          }
1807          return;
1808        }
1809        // (a literal_placeholders doc falls through: the placeholder is an
1810        // ordinary childless element and serializes as itself)
1811        let tag = if let Some(ns) = node.get_namespace() {
1812          let prefix = ns.get_prefix();
1813          if prefix.is_empty() {
1814            local_name
1815          } else {
1816            s!("{}:{}", prefix, local_name)
1817          }
1818        } else {
1819          local_name
1820        };
1821        let children = node.get_child_nodes();
1822        let mut open_tag = s!("<{tag}");
1823
1824        let nsnodes = node.get_namespace_declarations();
1825        for ns in nsnodes {
1826          let prefix = ns.get_prefix();
1827          let prefix_declaration = if prefix.is_empty() {
1828            s!("xmlns")
1829          } else {
1830            s!("xmlns:{}", prefix)
1831          };
1832          let href = ns.get_href();
1833          write!(open_tag, " {prefix_declaration}=\"{href}\"").ok();
1834        }
1835
1836        let anodes = node.get_attributes();
1837        // `get_attributes()` reports LOCAL names, so an `xml:id` attribute comes
1838        // back keyed as "id" -- by name alone indistinguishable from a genuine
1839        // plain `id`, which the model really does declare (see
1840        // `LaTeXML-bib.rnc:335,353`: `ltx:bib-identifier/@id`,
1841        // `ltx:bib-review/@id`). `get_node_document_qname` resolves the
1842        // attribute's NAMESPACE, so it returns "xml:id" for the former and "id"
1843        // for the latter; SERIALIZE that qname. Ordering stays keyed on the
1844        // local name, which is what the goldens encode (`xml:lang` files under
1845        // "lang", so it precedes `role`), with `xml:id` forced last -- the
1846        // previous code got that by appending it separately, which is also what
1847        // rewrote every plain `id` into an invalid-NCName `xml:id`
1848        // (witness arXiv 2508.17585).
1849        let mut anodes_keys: Vec<(SymStr, &String)> = anodes
1850          .keys()
1851          .map(|key| {
1852            let qname = node
1853              .get_attribute_node(key)
1854              .map(|a| model::get_node_document_qname(&a))
1855              .unwrap_or_else(|| arena::pin(key));
1856            (qname, key)
1857          })
1858          .collect();
1859        // Rank: xmlns:* declarations first (matching Perl's output order), then
1860        // the ordinary attributes by local name, then xml:id last.
1861        let rank = |qname: SymStr, key: &String| -> u8 {
1862          if key.starts_with("xmlns:") {
1863            0
1864          } else if arena::with(qname, |q| q == "xml:id") {
1865            2
1866          } else {
1867            1
1868          }
1869        };
1870        anodes_keys.sort_by(|(a_sym, a_key), (b_sym, b_key)| {
1871          rank(*a_sym, a_key)
1872            .cmp(&rank(*b_sym, b_key))
1873            .then_with(|| a_key.cmp(b_key))
1874        });
1875        for (key_sym, key) in anodes_keys {
1876          // Reuse the value already in `anodes` instead of re-fetching it with
1877          // `get_property` — `get_attributes()` read each value from the
1878          // attribute node directly, so a by-name `xmlGetProp` re-scan (plus a
1879          // fresh `CString` and result `String`) here is pure redundant FFI
1880          // churn (top `get_property` caller in the alloc profile).
1881          let val_serialized = serialize_attr(anodes.get(key).map(String::as_str).unwrap_or(""));
1882          arena::with(key_sym, |key_str| {
1883            write!(open_tag, " {key_str}=\"{val_serialized}\"")
1884          })
1885          .ok();
1886        }
1887
1888        let noindent_children: bool = if heuristic {
1889          // libxml2's heuristic: inline (noindent) if ANY direct child is a text node.
1890          // Crucially, this does NOT propagate the parent's noindent — each element
1891          // independently checks its own children for text nodes.
1892          children
1893            .iter()
1894            .any(|e| e.get_type() == Some(NodeType::TextNode))
1895        } else {
1896          // This is the "Correct" way to determine whether to add indentation
1897          let node_qname = get_node_qname(node);
1898          model::can_contain_sym(node_qname, pin!("#PCDATA"))
1899        };
1900
1901        let noindent_children = noindent_children || self.spill_flat;
1902        if !noindent {
1903          push_indent(out, depth);
1904        }
1905        out.push_str(&open_tag);
1906        // Perl serializes elements with children (including empty text nodes) as
1907        // <tag>...</tag>, and truly childless elements as <tag/>. Match this behavior.
1908        if !children.is_empty() {
1909          // with contents.
1910          out.push('>');
1911          if !noindent_children {
1912            out.push('\n');
1913          }
1914          for child in children {
1915            self.serialize_into(out, &child, depth + 1, noindent_children, heuristic);
1916          }
1917          if !noindent_children {
1918            push_indent(out, depth);
1919          }
1920          write!(out, "</{tag}>").ok();
1921        } else {
1922          // empty element.
1923          out.push_str("/>");
1924        }
1925        if !noindent {
1926          out.push('\n');
1927        }
1928      },
1929      Some(NodeType::TextNode) => {
1930        out.push_str(&serialize_string(&node.get_content()));
1931      },
1932      Some(NodeType::PiNode) => {
1933        // should code this by hand, as well...
1934        if !noindent {
1935          push_indent(out, depth);
1936        }
1937        out.push_str(&self.document.node_to_string(node));
1938        if !noindent {
1939          out.push('\n');
1940        }
1941      },
1942      Some(NodeType::CommentNode) => {
1943        write!(out, "<!-- {}-->", serialize_string(&node.get_content())).ok();
1944      },
1945      _ => {},
1946    }
1947  }

1949  /// Move the current insertion point to `node` — Perl `Document.pm:setNode`
1950  /// L74-87.
1951  ///
1952  /// A document fragment is not itself an insertion point: as in Perl, a
1953  /// single-child fragment is descended into, and a fragment with several
1954  /// children is an error (reported, not fatal — see the body comment on why
1955  /// this returns `()`).
1956  pub fn set_node(&mut self, node: &Node) {
1957    // Perl Document.pm:setNode L74-87: if the candidate is a
1958    // DOCUMENT_FRAG_NODE, validate that it has exactly one child and
1959    // descend to that child. The original Rust port had this check
1960    // commented-out with a wrong-node-type marker (`DocumentNode`
1961    // instead of `DocumentFragNode`); revived with the correct enum.
1962    let mut chosen = node.clone();
1963    if chosen.get_type() == Some(NodeType::DocumentFragNode) {
1964      let children = chosen.get_child_nodes();
1965      // Wrap the Error!/note_status side-effects in an IIFE to swallow the
1966      // `Result<()>` the macro returns (it can early-return Err if the
1967      // MAX_ERRORS cap is hit). `set_node` returns `()` and is called from
1968      // 32 call-sites; threading `Result` through all of them is out of
1969      // scope for an audit-fix. The hot path stays the same; the rare
1970      // hit-the-cap case loses one escape attempt but the next Error!
1971      // anywhere else will trigger the cap regardless.
1972      let _ = (|| -> Result<()> {
1973        if children.len() > 1 {
1974          Error!(
1975            "unexpected",
1976            "multiple-nodes",
1977            "Cannot set insertion point to a DOCUMENT_FRAG_NODE"
1978          );
1979        } else if children.is_empty() {
1980          Error!(
1981            "unexpected",
1982            "empty-nodes",
1983            "Cannot set insertion point to an empty DOCUMENT_FRAG_NODE"
1984          );
1985        }
1986        Ok(())
1987      })();
1988      if let Some(first) = children.into_iter().next() {
1989        chosen = first;
1990      }
1991    }
1992    self.node = chosen;
1993  }

1998  fn set_rdfa_prefixes(&mut self) {

2000    let prefix_map: HashMap<String, String> = state::with_mapping_keys("RDFa_prefixes", |keys| {
2001      let mut map = HashMap::default();

2005          map.insert(key_str, stored.to_string());

2007      }
2008      map
2009    });
2010    if prefix_map.is_empty() {
2011      return;
2012    }
2013
2014    let non_rdf_prefixes: HashSet<&str> = ["http", "https", "ftp"].iter().copied().collect();
2015    let rdf_term_attrs = [
2016      "about", "resource", "property", "typeof", "rel", "rev", "datatype",
2017    ];
2018
2019    // Build XPath to find elements with any RDFa term attribute
2020    let xpath = format!(
2021      "descendant::*[{}]",
2022      rdf_term_attrs
2023        .iter()
2024        .map(|a| format!("@{a}"))
2025        .collect::<Vec<_>>()
2026        .join(" or ")
2027    );
2028
2029    let mut used_prefixes: BTreeSet<String> = BTreeSet::new();
2030

2035          for term in value.split_whitespace() {
2036            if let Some(colon_pos) = term.find(':') {
2037              let prefix = &term[..colon_pos];
2038              if !non_rdf_prefixes.contains(prefix) && prefix_map.contains_key(prefix) {
2039                used_prefixes.insert(prefix.to_string());
2040              }
2041            }
2042          }
2043        }
2044      }
2045    }
2046
2047    for prefix in &self.extra_rdfa_prefixes {
2048      if !non_rdf_prefixes.contains(prefix.as_str()) && prefix_map.contains_key(prefix) {
2049        used_prefixes.insert(prefix.clone());
2050      }
2051    }
2052
2053    if !used_prefixes.is_empty()
2054      && let Some(mut root) = self.document.get_root_element()
2055    {
2056      let prefix_str = used_prefixes
2057        .iter()
2058        .map(|p| format!("{}: {}", p, prefix_map[p]))
2059        .collect::<Vec<_>>()
2060        .join(" ");
2061      let _ = root.set_attribute("prefix", &prefix_str);
2062    }

2152  /// Does NOT move the current insertion point to the Comment,
2153  /// but may move up past a text node.
2154  /// Perl: Document.pm lines 678-698
2155  pub fn insert_comment(&mut self, text: &str) -> Result<Node> {
2156    let trimmed = text.trim_end();
2157    let clean = DASHES_RE.replace_all(trimmed, "__");
2158    // Perl does NOT close the text node here — it uses getElement() to find
2159    // the nearest element, then inserts the comment relative to element children.
2160    // This preserves self.node as the current text node, so subsequent text
2161    // appends correctly and ligatures fire on the full text run.
2162
2163    let comment_text = s!(" {} ", clean);
2164
2165    if self.node.get_type() == Some(NodeType::DocumentNode) {
2166      Self::add_comment(
2167        &self.document,
2168        &self.node,
2169        &comment_text,
2170        Placement_::AppendChild,
2171      );
2172    } else {
2173      if let Some(node) = self.get_element() {
2174        // Get the nearest element node (Perl: getElement)
2175        let prev = node.get_last_child();

2436  /// If we're closing a node that can take font switches and it contains

2443      // "font") BUT, it isn"t being forced somehow
2444      if c.len() == 1
2445        && (get_node_qname(&c[0]) == pin!("ltx:text"))

2481    &mut self,

2483    to: &mut Node,

2487      // Skip internal attributes
2488      if key.starts_with('_') {
2489        continue;

2492      // Check both "xml:id" and bare "id" with XML namespace for the special case.
2493      let is_xml_id = key.as_str() == "xml:id"
2494        || (key.as_str() == "id"
2495          && from
2496            .get_attribute_ns("id", "http://www.w3.org/XML/1998/namespace")
2497            .is_some());

2503        // deduplicated id when a DIFFERENT node already claims the
2504        // same one; must use the return value, not the original `val`.
2505        let to_has_id = to.has_attribute("xml:id")
2506          || to
2507            .get_attribute_ns("id", "http://www.w3.org/XML/1998/namespace")
2508            .is_some();

2510          self.unrecord_id(val);
2511          let deduped = self.record_id_with_node(val, to);
2512          to.set_attribute("xml:id", &deduped)?;
2513        }
2514      } else if MERGE_ATTRIBUTE_SPACEJOIN.contains(key.as_str()) {
2515        self.add_ss_values(to, key, val)?;
2516      } else if MERGE_ATTRIBUTE_SEMICOLONJOIN.contains(key.as_str()) {
2517        if let Some(existing) = to.get_attribute(key) {
2518          let merged = format!("{existing}; {val}");
2519          to.set_attribute(key, &merged)?;
2520        } else {
2521          to.set_attribute(key, val)?;
2522        }
2523      } else if MERGE_ATTRIBUTE_SUMLENGTH.contains(key.as_str()) {
2524        if let Some(val2) = to.get_attribute(key) {
2525          // Parse and sum pt values
2526          let v1 = val.trim_end_matches("pt").parse::<f64>().unwrap_or(0.0);

2529        } else {
2530          to.set_attribute(key, val)?;
2531        }
2532      } else if !to.has_attribute(key) || is_forced {
2533        // Else if attribute not present on $to, or if we specifically override it, just copy
2534        to.set_attribute(key, val)?;
2535      }
2536    }
2537    Ok(())
2538  }

2542    // taking it here keeps the flag from leaking into unrelated calls.
2543    let force_whitespace = std::mem::take(&mut self.verbatim_space_pending);
2544    if text.is_empty() {
2545      return Ok(self.node.clone());
2546    }
2547    // Sibling guard to open_math_text_internal: libxml's append_text uses
2548    // CString and panics on embedded NULs (forbidden in XML text per spec).
2549    // Strip them — and the rest of the XML-invalid control class — up-front
2550    // so all downstream append_text calls are safe and the output stays
2551    // well-formed (shared policy with `set_attribute`, see `xml_sanitize`).

2577    // a text node, swap them to avoid splitting text runs, then recurse.
2578    // This avoids libxml text-node merging which would bypass ligature processing.

2615      //
2616      // Detect the merge WITHOUT dereferencing the possibly-freed node:
2617      // after `add_child` our text is `point`'s last child in both cases

2627        self.set_node(&point);
2628      }
2629    }

2633  /// Perl lines 1139-1144: Avoid splitting text runs across comments.
2634  /// If the current node's lastChild is a comment and the previous sibling is a text node,
2635  /// swap them so the text node is last, set it as current, and recurse to append new text.
2636  /// Returns true if the swap+append was performed, false otherwise.
2637  fn swap_comment_text_if_needed(&mut self, text: &str) -> Result<bool> {
2638    if self.node.get_type() != Some(NodeType::ElementNode) {
2639      return Ok(false);
2640    }
2641    if let Some(last_child) = self.node.get_last_child()
2642      && last_child.get_type() == Some(NodeType::CommentNode)
2643      && let Some(mut prev_text) = last_child.get_prev_sibling()
2644      && prev_text.get_type() == Some(NodeType::TextNode)
2645    {
2646      // Swap: move text node after comment node
2647      let mut comment_node = last_child;
2648      comment_node.add_next_sibling(&mut prev_text)?;
2649      // Set current node to the moved text node and recurse. Use
2650      // pointer-identity, not the `prev_text` handle directly:
2651      // `add_next_sibling` of a text node can MERGE it into an adjacent
2652      // text sibling and FREE it (the same libxml2 hazard fixed in
2653      // open_text_internal — benign on glibc, a use-after-free on macOS
2654      // libmalloc, issue #217). The moved/merged text is `comment_node`'s
2655      // next sibling either way; if that is still `prev_text` it survived,
2656      // otherwise `prev_text` was freed — fall back to the live sibling.
2657      match comment_node.get_next_sibling() {
2658        Some(moved) => self.set_node(&moved),
2659        None => self.set_node(&prev_text),
2660      }
2661      self.open_text_internal(text)?;
2662      return Ok(true);
2663    }
2664    Ok(false)
2665  }

2796      // zero-length match, spinning `get_prev_sibling().unwrap()` until it
2797      // panics. Saturate: a 0/1-length match removes no prior siblings.

2800        // (the mirror of the zero-length underflow above) — stop instead of

2803          Error!(
2804            "unexpected",
2805            "ligature",
2806            "Math ligature matched more siblings than exist; truncating the merge"

2808          break;
2809        };
2810        if let Some(b) = self.get_node_box(&remove) {
2811          boxes.push_front(b);
2812        }
2813        self.remove_node(remove);
2814      }
2815      // This fragment replaces the node's box by the composite boxes it replaces
2816      // HOWEVER, this gets things out of sync because parent lists of boxes still

2887  /// Return a string indicating the path to the current insertion point in the document.
2888  /// if $levels is defined, show only that many levels
2889  pub fn get_insertion_context(&self, levels_opt: Option<usize>) -> Result<String> {
2890    let mut levels = match levels_opt {
2891      None => {
2892        // Default depth is based on verbosity
2893        if state::current_verbosity() <= 1 {
2894          Some(5)
2895        } else {
2896          None
2897        }
2898      },
2899      Some(t) => Some(t),
2900    };
2901    let mut node = self.node.clone();
2902    let node_type = node.get_type();
2903    if node_type != Some(NodeType::TextNode)
2904      && node_type != Some(NodeType::ElementNode)
2905      && node_type != Some(NodeType::DocumentNode)

2907      let message = s!(

2909        self.document.node_to_string(&node)
2910      );
2911      Error!("internal", "context", message);
2912      return Ok(String::new());
2913    }

2915    // ancestors and prepending each element's qname. Mirrors Perl's
2916    // `Stringify($node)` chain with a depth cap from `levels_opt`.
2917    //
2918    // Cap each qname at 80 chars. Pathological inputs (e.g. xy-pic

2925    // Witness papers: math0203082, math0402448 (R35.B, sandbox).
2926    const QNAME_CAP: usize = 80;
2927    let truncate_qname = |qname: &str| -> String {
2928      if qname.len() <= QNAME_CAP {
2929        qname.to_string()
2930      } else {
2931        let mut s: String = qname.chars().take(QNAME_CAP).collect();
2932        s.push('…');
2933        s
2934      }
2935    };

2937      match n.get_type() {
2938        Some(NodeType::ElementNode) => {

2940        },

2942        Some(NodeType::DocumentNode) => "#document".to_string(),
2943        _ => "?".to_string(),
2944      }
2945    };

2947    while let Some(parent_node) = node.get_parent() {
2948      node = parent_node;
2949      if let Some(levels_val) = levels {
2950        levels = Some(levels_val - 1);
2951        if levels_val <= 1 {
2952          path = format!("...{path}");
2953          break;
2954        }
2955      }
2956      path = format!("{}{}", qn_for(&node), path);
2957    }
2958    Ok(path)
2959  }

3590  // Streaming (fragmented) conversion: spill closed subtrees to disk.
3591
3592  /// Spill closed subtrees to the segment store, freeing their DOM memory
3593  /// (streaming pass 1).
3594  ///
3595  /// Walks the SPINE — root element down to the current insertion element —
3596  /// and, at every level, spills the runs of closed element children that
3597  /// precede the open (spine) child: serialize in the pre-finalize form with
3598  /// the exact `(depth, noindent)` the eager serializer would use, write to
3599  /// `store`, record every spilled `xml:id`/label in `index`, purge the
3600  /// pointer-keyed registries (`idstore` via `unrecord_id`, `node_boxes` by
3601  /// key — freed nodes' pointers can be REUSED, so a stale entry is not just
3602  /// dangling but can mis-associate a box with a future node), replace the
3603  /// run with one `<_spilled_ ref="N"/>` placeholder, and unlink+free the
3604  /// nodes. Serialization at assembly time splices the processed segment
3605  /// where the placeholder sits, so document order, nesting and
3606  /// spine-attribute finality are preserved by construction.
3607  ///
3608  /// Two deliberate exclusions:
3609  /// * ROOT-level children spill only when sectional (`ROOT_SPILLABLE`):
3610  ///   the frontmatter fallback and `maybe_promote_leading_title` operate on
3611  ///   the leading non-sectional root children at end-of-build.
3612  /// * Children of the insertion element itself never spill: in-flight
3613  ///   construction state may still reference recently built nodes.
3614  ///
3615  /// Returns the number of runs spilled.
3616  pub fn spill_closed_subtrees(&mut self, index: &mut crate::sxml::FragmentIndex) -> Result<usize> {
3617    debug_assert!(
3618      self.spill_store.is_some(),
3619      "spill_closed_subtrees needs the store attached (set_spill_store)"
3620    );
3621    let Some(root) = self.document.get_root_element() else {
3622      return Ok(0);
3623    };
3624    // The spine, root-first. The insertion point may be a text node; start
3625    // from its nearest element.
3626    let mut spine: Vec<Node> = Vec::new();
3627    let mut cursor = if self.node.get_type() == Some(NodeType::ElementNode) {
3628      Some(self.node.clone())
3629    } else {
3630      self.node.get_parent()
3631    };
3632    while let Some(n) = cursor {
3633      if n.get_type() != Some(NodeType::ElementNode) {
3634        break;
3635      }
3636      cursor = n.get_parent();
3637      spine.push(n);
3638    }
3639    spine.reverse();
3640    if spine.is_empty() && self.node.get_type() == Some(NodeType::DocumentNode) {
3641      // The ROOT itself has closed (`\end{document}` absorbed): everything
3642      // under it is a closed subtree now, so the spine is just the root and
3643      // every eligible child may spill. Without this case the FINAL spill of
3644      // a streaming conversion silently no-ops — the witness kept 27,400
3645      // formulae live into the spine tail this way and died there.
3646      spine.push(root.clone());
3647    }
3648    if spine.first() != Some(&root) {
3649      // The insertion point is outside the tree under the root: nothing is
3650      // safely classifiable as closed.
3651      return Ok(0);
3652    }

3654      .get_namespace_declarations()

3657      .collect();

3661    // yield seam the insertion point routinely IS the root (between
3662    // top-level constructs), where a levels-above-only walk sees an empty
3663    // spine range and spills NOTHING — the witness leaked two thirds of its
3664    // chapters into the spine tail exactly this way. The insertion element's
3665    // children are complete, closed subtrees at a seam (the constructed-node
3666    // stacks expire between absorbs); its level simply has no barrier child.

3671      // nearest section at or above this spill parent, mirroring the
3672      // ancestor walk `apply_lx_declarations` performs.

3674        .iter()
3675        .rev()

3678      // The serializer's recursion contract for children of `parent`
3679      // (`serialize_into`: depth+1, parent's schema-driven noindent).

3681        let parent_qname = get_node_qname(&parent);
3682        model::can_contain_sym(parent_qname, pin!("#PCDATA"))
3683      };
3684      let mut run: Vec<Node> = Vec::new();
3685      for child in parent.get_child_nodes() {
3686        if Some(&child) == barrier.as_ref() {
3687          break;
3688        }
3689        let eligible = child.get_type() == Some(NodeType::ElementNode)
3690          && child.get_name() != SPILL_PLACEHOLDER
3691          && (level > 0 || ROOT_SPILLABLE.contains(&child.get_name().as_str()))

3696            .findnodes("descendant-or-self::ltx:bibliography", Some(&child))
3697            .is_empty();

3702          // each placeholder replaces exactly the contiguous nodes it stands
3703          // for, and document order is preserved around the interloper.

3736  /// pass 1. The witness's first RSS-triggered yield once spilled half the
3737  /// document as ONE 512 MB segment, and pass 2 died re-parsing it.
3738  fn segment_chunk_bytes() -> usize {
3739    // Calibration override (`LATEXML_SEGMENT_CHUNK_MIB`), deliberately env-only
3740    // and NOT a CLI flag — the same reasoning as `LATEXML_SPILL_AT_MIB`, and
3741    // for the same reason: MAX below is an ARGUED constant, not a measured one.
3742    //
3743    // Its stated basis is a TIME claim ("the point past which a single
3744    // segment's re-parse dominates pass 2"), but the pass-2 tail is linear in
3745    // fragment size — `prune_dangling_split_xmrefs`, `prune_xmduals`,
3746    // `mark_xmnode_visibility` and `cleanup_unreferenced_xmtok_ids` are each
3747    // one `findnodes` plus an O(1)-per-node loop, `finalize_subtree` is a
3748    // recursive walk, and the rewrite rule set is fixed. Larger segments
3749    // should also SUIT the allocator, which is the largest single profile
3750    // category (~16.5%): fewer, bigger alloc/free cycles.
3751    //
3752    // Note what the derivation below wants: at --max-memory 48000 it computes
3753    // watermark/72 = 166 MB and MAX then clamps it to 32 MB. This knob exists
3754    // so that ceiling can be swept rather than argued.
3755    if let Some(mib) = std::env::var("LATEXML_SEGMENT_CHUNK_MIB")
3756      .ok()
3757      .and_then(|v| v.parse::<usize>().ok())
3758      .filter(|mib| *mib > 0)
3759    {
3760      return mib.saturating_mul(1024 * 1024);
3761    }
3762    /// Validated on the 131 MB witness. The accompanying claim that a single
3763    /// segment's re-parse dominates pass 2 past this point is NOT substantiated
3764    /// by the pass-2 tail, which is linear — see the override above.
3765    const MAX: usize = 32 * 1024 * 1024;
3766    /// Below this, per-segment overhead (parse, index merge, reserialize)
3767    /// outweighs the memory saved.
3768    const MIN: usize = 4 * 1024 * 1024;
3769    /// Serialized bytes → live DOM, measured: 18.6 MB of core XML
3770    /// materialized as ~1,346 MB of libxml2 DOM on a witness slice.
3771    const DOM_EXPANSION: u64 = 72;
3772    match crate::stomach::spill_watermark_bytes() {
3773      // Pass 2 re-materializes ONE segment whole, so a chunk must fit the
3774      // same watermark pass 1 respects — otherwise the phase that exists to
3775      // bound memory becomes the phase that blows it on a small machine.
3776      Some(watermark) => ((watermark / DOM_EXPANSION) as usize).clamp(MIN, MAX),
3777      None => MAX,
3778    }
3779  }

3781  fn spill_run(
3782    &mut self,
3783    run: &mut Vec<Node>,
3784    depth: usize,
3785    noindent: bool,
3786    namespaces: &[(String, String)],
3787    section_id: &Option<String>,
3788    index: &mut crate::sxml::FragmentIndex,
3789  ) -> Result<usize> {
3790    use crate::sxml::SegmentMeta;
3791    // An element whose children are exclusively EMPTY text nodes serializes
3792    // as `<p></p>` — indistinguishable, after a parse round-trip, from a
3793    // childless `<p/>`. Mark such elements before serializing so pass 2 can
3794    // restore the empty text child and keep the round-trip exact (sweep
3795    // witness tests/fonts/abxtest.tex).
3796    fn mark_empty_text_elements(node: &Node) {
3797      if node.get_type() != Some(NodeType::ElementNode) {
3798        return;
3799      }
3800      let children = node.get_child_nodes();
3801      if !children.is_empty()
3802        && children
3803          .iter()
3804          .all(|c| c.get_type() == Some(NodeType::TextNode) && c.get_content().is_empty())
3805      {
3806        let mut n = node.clone();

3808      }
3809      for child in children {

3817    // their boxes are purged: the FLOATSUPERSCRIPT rewrite (tex_math.rs)
3818    // decides `<text font="italic">` wrapping from the box font's family,
3819    // and a fragment re-materialized in pass 2 has no boxes. An attr carries
3820    // the EXACT verdict (an ancestor-`_font` approximation mis-wrapped
3821    // text-mode superscripts — sweep witness tests/math/niceunits.tex).
3822    for node in run.iter() {
3823      for mut xmarg in self.findnodes("descendant-or-self::ltx:XMArg", Some(node)) {

3827            .get_family()

3829            .unwrap_or(false)
3830        {
3831          let _ = xmarg.set_attribute("_boxfont_math", "1");
3832        }
3833      }
3834    }
3835
3836    // Chunk the run so each segment stays under `segment_chunk_bytes()` of
3837    // serialized text; every chunk gets its own segment + placeholder, so
3838    // document order is preserved chunk-by-chunk.
3839    let chunk_ceiling = Self::segment_chunk_bytes();
3840    let mut runs_spilled = 0usize;
3841    let mut chunk: Vec<Node> = Vec::new();
3842    let mut xml = String::new();
3843    let all: Vec<Node> = std::mem::take(run);
3844    let total = all.len();
3845    for (i, node) in all.into_iter().enumerate() {

3847      // spills (a closing section swallows its paragraphs' placeholders).
3848      // They serialize LITERALLY (`literal_placeholders` is set for all of
3849      // pass 1) and the final assembly's splice resolves them recursively —
3850      // inlining them here instead once rebuilt an 841 MB and a 1.85 GB
3851      // segment out of chapter shells on the 131 MB witness, and pass 2
3852      // died re-materializing them.
3853      debug_assert!(
3854        self.literal_placeholders,
3855        "spill_run must serialize placeholders literally (set for all of pass 1)"
3856      );
3857      self.serialize_into(&mut xml, &node, depth, noindent, false);
3858      chunk.push(node);
3859      let last = i + 1 == total;
3860      if xml.len() >= chunk_ceiling || last {
3861        let meta = SegmentMeta {
3862          depth,
3863          noindent,
3864          font: None,
3865          namespaces: namespaces.to_vec(),
3866          section_id: section_id.clone(),
3867          parent: chunk
3868            .first()
3869            .and_then(|n| n.get_parent())
3870            .map(|p| arena::to_string(get_node_qname(&p))),
3871          ancestors: {
3872            // Every ancestor xml:id: a scope rooted at one of these covers
3873            // the whole fragment (see SegmentMeta::ancestors).
3874            let mut ids = Vec::new();
3875            let mut cur = chunk.first().and_then(|n| n.get_parent());
3876            while let Some(n) = cur {
3877              // Both attribute forms — constructed ancestors carry the plain
3878              // one (see `node_xml_id_any_form`); a namespace-only read left
3879              // these lists near-empty, so label/id-scoped rules could not
3880              // cover their fragments.
3881              if let Some(id) = Self::node_xml_id_any_form(&n) {
3882                ids.push(id);
3883              }
3884              cur = n.get_parent();
3885            }
3886            ids
3887          },
3888        };
3889        let store = self
3890          .spill_store
3891          .as_mut()
3892          .expect("checked at spill_closed_subtrees entry");
3893        let seg = store.write_segment(&xml, meta)?;
3894        xml.clear();
3895        // Index + purge while the nodes are still alive, then placeholder +
3896        // free.
3897        for node in chunk.iter() {
3898          self.index_and_purge_spilled(node, seg, index);
3899        }
3900        let spill_error = |details: String| Error {
3901          target:   ErrorTarget::Document,
3902          category: ErrorCategory::Libxml,
3903          message:  s!("segment staging: {}", details),
3904        };
3905        let mut placeholder = Node::new(SPILL_PLACEHOLDER, None, &self.document)
3906          .map_err(|()| spill_error(s!("cannot create the placeholder node")))?;
3907        placeholder
3908          .set_attribute("ref", &seg.to_string())
3909          .map_err(|e| spill_error(s!("cannot mark the placeholder: {e}")))?;
3910        // Record the LAST spilled sibling's qname: build-time hooks that
3911        // consult a previous element sibling (prune_empty_para's
3912        // was-I-first test — sweep witness tests/alignment/colortbls.tex)
3913        // would otherwise see the placeholder where eager sees that node.
3914        // Never serialized — assembly splices the segment text instead.
3915        if let Some(last) = chunk.last() {
3916          placeholder
3917            .set_attribute("last", &arena::to_string(get_node_qname(last)))
3918            .map_err(|e| spill_error(s!("cannot mark the placeholder: {e}")))?;
3919        }
3920        chunk
3921          .first_mut()
3922          .expect("a flushed chunk is non-empty")
3923          .add_prev_sibling(&mut placeholder)
3924          .map_err(|e| spill_error(s!("cannot insert the placeholder: {e}")))?;
3925        for node in chunk.drain(..) {
3926          // Actually FREE the spilled subtree (unlink alone leaks — see
3927          // Node::free_subtree). Its ids were unrecorded by
3928          // index_and_purge_spilled above; discard_subtree re-purges
3929          // node_boxes (cheap) and reclaims the C memory, which is the whole
3930          // point of the spill.
3931          self.discard_subtree(node);
3932        }
3933        SPILLED_SEGMENTS.set(SPILLED_SEGMENTS.get() + 1);
3934        runs_spilled += 1;
3935      }
3936    }
3937    Ok(runs_spilled)
3938  }

4270  //  In an XMDual, in each branch (content, presentation) there will be atoms
4271  //  that correspond to the input (one will be real, the other an XMRef to the first).
4272  //  But also there will be additional "decoration" (delimiters, punctuation, etc on the
4273  // presentation  side; other symbols, bindings, whatever, on the content side).
4274  //  These decorations should NOT be subject to rewrite rules,
4275  //  and in cross-linked parallel markup, they should be attributed to the
4276  //  upper containing object's ID, rather than left dangling.
4277  //
4278  //  To determine this, we mark all math nodes as to whether they are "visible" from
4279  //  presentation, content or both (the default top-level being both).
4280  //  Decorations are the nodes that are visible to only one mode.
4281  //  Note that nodes that are not visible at all CAN occur (& do currently when the parser
4282  //  creates XMDuals), pruneXMDuals (below) gets rid of them.

4310  fn mark_xmnode_visibility_aux_inner(
4311    &self,
4312    mut node: Node,
4313    cvis: bool,
4314    mut pvis: bool,
4315  ) -> Result<()> {
4316    if (!cvis || node.has_attribute("_cvis")) && (!pvis || node.has_attribute("_pvis")) {
4317      return Ok(());
4318    }
4319    let qname = get_node_qname(&node);
4320    // Special case: for XMArg used to wrap "formal" arguments on the content side,
4321    // mark them as visible as presentation as well.
4322    if cvis && (qname == pin!("ltx:XMArg")) {
4323      pvis = true;
4324    }
4325    if cvis {
4326      node.set_attribute("_cvis", "1")?;
4327    }
4328    if pvis {
4329      node.set_attribute("_pvis", "1")?;
4330    }

4333      // XMDual should have exactly 2 element children (content + presentation),
4334      // but a malformed math parse (e.g. semantic action producing empty pair
4335      // during deep ambiguity collapse) can leave an empty XMDual. Skip
4336      // visibility-marking rather than panicking on `children.remove(0)` —
4337      // see wp5 sandbox 2110.10033 and 4 sibling papers.
4338      if children.len() >= 2 {
4339        let c = children.remove(0);
4340        let p = children.remove(0);

4343        }
4344        if pvis {
4345          self.mark_xmnode_visibility_aux(p, false, true)?;
4346        }
4347      }
4348    } else if qname == pin!("ltx:XMRef") {
4349      match node.get_attribute("idref") {
4350        None => {
4351          let key = node.get_attribute("_xmkey");

4353            "expected",
4354            "id",
4355            "Missing idref on ltx:XMRef",
4356            s!("_xmkey is `{}`", key.unwrap_or_default())
4357          );
4358        },
4359        Some(id) => match self.lookup_id(&id) {
4360          None => {
4361            Warn!(
4362              "expected",
4363              "node",
4364              s!("No node found with id='{id}' (referred to from ltx:XMRef)")
4365            );
4366          },
4367          Some(reffed) => {
4368            self.mark_xmnode_visibility_aux(reffed.clone(), cvis, pvis)?;
4369          },
4370        },
4371      }
4372    } else {
4373      for child in xml::element_nodes(&node) {

4380  /// Remove `ltx:XMRef[@_split_ref="1"]` whose `idref` no longer
4381  /// resolves. These are the XMRefs minted by
4382  /// `amsmath::rearrange_ams_split` to mirror the flattened cell
4383  /// sequence inside an `XMDual(XMWrap(refs), XMArray(cells))`. The
4384  /// math parser can later absorb some cells (typically inserted
4385  /// MULOP times-ops on `\mathcal{L}\rho` chains) into wrapping
4386  /// XMApps, dropping their xml:id from the live DOM and leaving
4387  /// the sibling XMRefs dangling. Left in place, each dangling
4388  /// XMRef trips three separate diagnostics later — math parser's
4389  /// read_xmref Warn, finalize's mark_xmnode_visibility Warn, and
4390  /// post-process's mark_xm_node_visibility Error.
4391  ///
4392  /// We restrict the sweep to the `_split_ref` marker so refs from
4393  /// other provenance (base_xmath `\lx@dual`, renamed-id cases like
4394  /// declare_test's `S1.Ex1.m1.1` → `.1a` rename) stay untouched.
4395  ///
4396  /// Content-preserving: XMRefs are structural cross-references,
4397  /// not author body, and the math parser has already absorbed the
4398  /// referenced cell into the visible XMArray branch — no glyph or
4399  /// formula material is lost.
4400  pub fn prune_dangling_split_xmrefs(&mut self) -> Result<()> {
4401    let xmrefs = self.findnodes("//ltx:XMRef[@_split_ref or @_mf_ref]", None);
4402    for xmref in xmrefs {
4403      let idref = match xmref.get_attribute("idref") {
4404        Some(id) => id,
4405        None => continue,
4406      };
4407      if self.lookup_id(&idref).is_none() && xmref.get_parent().is_some() {
4408        self.remove_node(xmref);
4409      }
4410    }
4411    // Broader sweep: any XMRef pointing to a canonical math node id
4412    // `S<N>.E<M>.m1.<K>...` that no longer resolves. These are minted
4413    // by base_xmath::add_column_to_math_fork during rearrange_ams_*
4414    // (align/gather/multline), but unlike `_split_ref` they aren't
4415    // marked. The math parser absorbs cells later, leaving the refs
4416    // dangling and triggering the `Error:expected:id` cascade in
4417    // post-processing.
4418    //
4419    // We restrict the regex to the equation-numbered form (E<digit>,
4420    // not Ex<digit>) so declare_test's renamed-id case
4421    // (`S1.Ex1.m1.1` → `.1a`) stays untouched. canvas papers using
4422    // `\begin{equation}` produce `E1`/`E2`/... ids.
4423    static RE_MATH_ID: Lazy<Regex> = Lazy::new(|| Regex::new(r"^S\d+\.E\d+\.m\d+\.").unwrap());
4424    let xmrefs2 = self.findnodes("//ltx:XMRef[@idref]", None);
4425    for xmref in xmrefs2 {
4426      // Skip if already pruned via _split_ref sweep above.
4427      if xmref.get_parent().is_none() {
4428        continue;
4429      }
4430      let idref = match xmref.get_attribute("idref") {
4431        Some(id) => id,
4432        None => continue,
4433      };
4434      if RE_MATH_ID.is_match(&idref) && self.lookup_id(&idref).is_none() {
4435        // Safety guard: NEVER drop a ref that is an OPERAND of an XMApp.
4436        // Removing it would corrupt the application — e.g. an `\lx@dual`
4437        // content arm `XMApp(probability, XMRef)` would lose its argument,
4438        // emitting a malformed `apply(probability)` with no operand (silent
4439        // content-MathML corruption; witnessed on `aligned`+`\Pr(a,b|c)`,
4440        // docs/reproducers/class_b_aligned_pr_xmref.tex). Such content-arm
4441        // refs are essential shared links — they are NEVER the redundant
4442        // MathFork/`_split_ref` mirrors this sweep targets (those live in
4443        // XMWrap/XMArray, not as XMApp operands). Leave it dangling so the
4444        // faithful "No node found" Warn (`mark_xmnode_visibility`) fires
4445        // rather than silently corrupting the content tree.
4446        if let Some(parent) = xmref.get_parent()
4447          && get_node_qname(&parent) == pin!("ltx:XMApp")
4448        {
4449          let kids = parent.get_child_elements();
4450          let is_operand = kids
4451            .iter()
4452            .position(|k| *k == xmref)
4453            .is_some_and(|pos| pos > 0);
4454          if is_operand {
4455            continue;
4456          }
4457        }
4458        self.remove_node(xmref);
4459      }
4460    }
4461    Ok(())
4462  }

4513  fn compact_xmdual(
4514    &mut self,
4515    dual: Node,
4516    content: Node,
4517    presentation: Option<Node>,
4518  ) -> Result<()> {
4519    // Perl: our $content_transfer_overrides = { decl_id, meaning, name, omcd };
4520    // Perl: our $dual_transfer_overrides = { decl_id, meaning, name, omcd, xml:id, role };
4521    static CONTENT_TRANSFER: Lazy<HashSet<&'static str>> =
4522      Lazy::new(|| HashSet::from_iter(["decl_id", "meaning", "name", "omcd"]));
4523    static DUAL_TRANSFER: Lazy<HashSet<&'static str>> =
4524      Lazy::new(|| HashSet::from_iter(["decl_id", "meaning", "name", "omcd", "xml:id", "role"]));
4525
4526    let presentation = match presentation {
4527      Some(p) => p,
4528      None => return Ok(()),
4529    };
4530    let c_name = with_node_qname(&content, |n| n.to_string());
4531    let p_name = with_node_qname(&presentation, |n| n.to_string());
4532
4533    // Case 1: Quick fix — merge two tokens (Perl compactXMDual L1588-1593).
4534    // Perl has NO id cleanup here: mergeAttributes deliberately TRANSFERS the
4535    // content/dual xml:id onto the surviving presentation token (unRecordID +
4536    // recordID, L1308-1313), keeping refs to that id resolvable. A former
4537    // Rust-only "remove the leaked content id" block contradicted that — and
4538    // was dead anyway (its bare has/get/remove_attribute("xml:id") calls
4539    // always missed the namespaced attribute).
4540    if c_name == "ltx:XMTok" && p_name == "ltx:XMTok" {
4541      let mut pres = presentation;
4542      self.merge_attributes(&content, &mut pres, Some(&CONTENT_TRANSFER))?;
4543      self.merge_attributes(&dual, &mut pres, Some(&DUAL_TRANSFER))?;
4544      // Unlink presentation from dual before replacing, since presentation is a child of dual
4545      pres.unlink();
4546      self.replace_node(dual, vec![pres])?;
4547      return Ok(());
4548    }
4549
4550    // Case 2: Compact mirror XMApp nodes
4551    if c_name != "ltx:XMApp" || p_name != "ltx:XMApp" {
4552      return Ok(());
4553    }
4554    let content_args = xml::element_nodes(&content);
4555    let pres_args = xml::element_nodes(&presentation);
4556    if content_args.len() != pres_args.len() {
4557      return Ok(());
4558    }
4559    let n_args = content_args.len();
4560
4561    // Walk the corresponding children, double-check they are referenced in the same order
4562    enum NewArg {
4563      Single(Node),
4564      Pair(Node, Node), // (content_arg, pres_arg) — to be merged
4565    }
4566    let mut new_args: Vec<NewArg> = Vec::with_capacity(n_args);
4567    for (c_arg, p_arg) in content_args.into_iter().zip(pres_args) {
4568      if let Some(c_idref) = c_arg.get_attribute("idref")
4569        && c_idref == p_arg.get_attribute_ns("id", XML_NS).unwrap_or_default()
4570      {
4571        new_args.push(NewArg::Single(p_arg));
4572        continue;
4573      }
4574      if let Some(p_idref) = p_arg.get_attribute("idref")
4575        && p_idref == c_arg.get_attribute_ns("id", XML_NS).unwrap_or_default()
4576      {
4577        new_args.push(NewArg::Single(c_arg));
4578        continue;
4579      }
4580      // We can handle content-side XMToks to any XM* presentation subtree
4581      let c_arg_name = with_node_qname(&c_arg, |n| n.to_string());
4582      if c_arg_name != "ltx:XMTok" {
4583        return Ok(()); // Can't compact this structure
4584      }
4585      new_args.push(NewArg::Pair(c_arg, p_arg));
4586    }
4587
4588    // If we made it here, this dual has two mirrored applications — compact it.
4589    let mut parent = match dual.get_parent() {
4590      Some(p) => p,
4591      None => return Ok(()),
4592    };
4593    let mut compact_apply = self.open_element_at(&mut parent, "ltx:XMApp", None, None)?;
4594    for n_arg in new_args {
4595      let mut node = match n_arg {
4596        NewArg::Single(n) => n,
4597        NewArg::Pair(c_arg, mut p_arg) => {
4598          self.merge_attributes(&c_arg, &mut p_arg, Some(&CONTENT_TRANSFER))?;
4599          p_arg
4600        },
4601      };
4602      node.unlink();
4603      compact_apply.add_child(&mut node)?;
4604    }
4605    // Migrate dual attributes to the new XMApp
4606    self.merge_attributes(&dual, &mut compact_apply, Some(&DUAL_TRANSFER))?;
4607    // Direct DOM swap: replace dual with compact_apply without re-creating nodes.
4608    // Perl uses replaceChild which is a direct swap; replace_tree calls append_tree
4609    // which re-creates elements and fires afterOpen/afterClose hooks a second time.
4610    compact_apply.unlink();
4611    let mut dual_mut = dual;
4612    dual_mut.add_prev_sibling(&mut compact_apply).ok();
4613    self.remove_node(dual_mut);
4614    Ok(())
4615  }

4618  fn collapse_xmdual(&mut self, dual: Node, mut branch: Node) -> Result<()> {
4619    // The other branch is not visible, nor referenced,
4620    // but the dual may have an id and be referenced
4621    if let Some(dualid) = dual.get_attribute_ns("id", XML_NS) {
4622      self.unrecord_id(&dualid); // We'll move or remove the ID from the dual
4623      if let Some(branchid) = branch.get_attribute_ns("id", XML_NS) {
4624        // branch has id too!
4625        for mut tref in self.findnodes(&s!("//*[@idref='{}']", dualid), None) {
4626          tref.set_attribute("idref", &branchid)?;
4627        } // Change dualid refs to branchid
4628      } else {
4629        // Assign the dual's id to the branch. Record first so we
4630        // receive a deduplicated id if something else claimed it
4631        // between the `unrecord_id` above and now — write the
4632        // deduped value, not the original.
4633        let deduped = self.record_id_with_node(&dualid, &branch);
4634        branch.set_attribute("xml:id", &deduped)?;
4635      }
4636    }
4637    // Direct DOM swap: Perl uses replaceChild (no re-creation, no hooks fired twice)
4638    let mut dual_mut = dual;
4639    dual_mut.add_prev_sibling(&mut branch).ok();
4640    self.remove_node(dual_mut);
4641    Ok(())
4642  }

4776          current = parent;
4777          node_ref = &current;
4778        },
4779        _ => {

4977  fn open_element_internal(
4978    &mut self,
4979    point: &mut Node,
4980    ns_opt: Option<String>,
4981    tag: &str,
4982  ) -> Result<Node> {
4983    // TODO:
4984    //
4985    // I am seriously irritated by the XML namespace and the confusion of the "default#"
4986    // tricks and libxml2's custom decisions about namespace interactions
4987    //
4988    // I have "hacked together" a working flow for now, but I expect to
4989    // encounter bugs related to the shortcuts taken here. I would welcome a
4990    // redesign       that simplifies the namespace logic dramatically.
4991    let new_ns = match ns_opt {
4992      Some(ns_uri) => {
4993        match point.lookup_namespace_prefix(&ns_uri) {
4994          // namespace not already declared?
4995          None => {
4996            if let Some(prefix) = model::get_document_namespace_prefix(&ns_uri, false, false) {
4997              if prefix != pin!("") {
4998                let mut root = self.document.get_root_element().unwrap();
4999                match arena::with(prefix, |prefix_str| {
5000                  Namespace::new(prefix_str, &ns_uri, &mut root)
5001                }) {
5002                  Ok(ns) => Some(ns),
5003                  Err(_) => {
5004                    // The namespace already exists on root (declared by an
5005                    // earlier element of the same namespace — e.g. a prior
5006                    // tikz/SVG picture) but `lookup_namespace_prefix` did not
5007                    // find it from this deeply-nested insertion point. Recover
5008                    // by reusing the root declaration (or creating it on the
5009                    // insertion point), exactly as the already-declared branch
5010                    // below — do NOT drop the namespace. Witness 1802.00756:
5011                    // a `tikzpicture` inside a nested `gather*`/`minipage`/
5012                    // `figure*` emitted 14× "failed to create namespace: svg"
5013                    // and the `<svg:svg>`/`<svg:g>` lost their namespace.
5014                    arena::with(prefix, |prefix_str| {
5015                      let found = root
5016                        .get_namespace_declarations()
5017                        .into_iter()
5018                        .find(|ns| ns.get_prefix() == prefix_str);
5019                      if found.is_none() {
5020                        Namespace::new(prefix_str, &ns_uri, point).ok()
5021                      } else {
5022                        found
5023                      }
5024                    })
5025                  },
5026                }
5027              } else {
5028                // default namespace?
5029                None
5030              }
5031            } else {
5032              // default namespace?
5033              None
5034            }
5035          },
5036          Some(prefix) => {
5037            if !prefix.is_empty() {
5038              let mut root = self.document.get_root_element().unwrap();
5039              match Namespace::new(&prefix, &ns_uri, &mut root) {
5040                Ok(ns) => Some(ns),
5041                Err(_) => {
5042                  // Namespace already exists on root — find and reuse it.
5043                  // We search declarations then fall back to creating on the
5044                  // insertion point (which inherits from root).
5045                  let found = root
5046                    .get_namespace_declarations()
5047                    .into_iter()
5048                    .find(|ns| ns.get_prefix() == prefix);
5049                  if found.is_none() {
5050                    // Try creating on the insertion point instead
5051                    Namespace::new(&prefix, &ns_uri, point).ok()
5052                  } else {
5053                    found
5054                  }
5055                },
5056              }
5057            } else {
5058              // default namespace?
5059              None
5060            }
5061          },
5062        }
5063      },
5064      None => None,
5065    };
5066
5067    let no_ns = new_ns.is_none();
5068    let mut newnode = match Node::new(tag, new_ns.clone(), &self.document) {
5069      Ok(n) => n,
5070      Err(_) => {
5071        // libxml2 rejected the tag (e.g. NUL byte, malformed name).
5072        // Bail out of element creation rather than aborting; caller
5073        // can recover. Driver: 2304.07380 panic at Node::new unwrap.
5074        let message = s!("failed to create element {:?}", tag);
5075        Error!("document", "open_element_internal", message);
5076        return Err(message.into());
5077      },
5078    };
5079    point.add_child(&mut newnode)?;
5080    if no_ns {
5081      // When no explicit namespace was determined (default namespace element),
5082      // try to find the root's default namespace first. This prevents inheriting
5083      // the parent's namespace when inside a different namespace context (e.g.,
5084      // SVG elements getting svg: prefix on LaTeXML elements like Math/XMath).
5085      let root_ns = self
5086        .document
5087        .get_root_element()
5088        .and_then(|r| r.get_namespace());
5089      let parent_ns = point.get_namespace();
5090      if let Some(ref rns) = root_ns {
5091        // Use root's namespace for default-namespace elements
5092        let _ = newnode.set_namespace(rns);
5093      } else if let Some(ns) = parent_ns {
5094        // Fallback: inherit from parent (original behavior)
5095        newnode.set_namespace(&ns)?;
5096      }
5097    } else if let Some(ref ns) = new_ns {
5098      // For explicitly namespaced elements (e.g., svg:svg), ensure the namespace
5099      // is set after add_child — Node::new may not properly bind the namespace
5100      // when the Namespace was retrieved from get_namespace_declarations().
5101      let _ = newnode.set_namespace(ns);
5102    }
5103
5104    self.record_constructed_node(&newnode);
5105    Ok(newnode)
5106  }

5160  // Inserting clones of nodes into the document.
5161  // Nodes that exist in some other part of the document (or some other document)
5162  /// Append COPIES of `new_children` under `node`.
5163  ///
5164  /// Cloning rather than moving is what makes this usable on nodes that belong
5165  /// to another document — moving them would remove them from it. Three things
5166  /// are repaired on the way in: document fragments are expanded to their
5167  /// children, the namespace structure is rebuilt clean (libxml2 otherwise has
5168  /// a tendency to introduce annoying "default" namespace prefix
5169  /// declarations), and every `xml:id` in the copy is rewritten to a fresh id,
5170  /// with internal references remapped to match — otherwise the copy would
5171  /// duplicate the original's ids.
5172  // # Should have variants here for prepend, insert before, insert after.... ???
5173  pub fn append_clone(&mut self, node: &mut Node, new_children: Vec<Node>) -> Result<()> {
5174    // Expand any document fragments
5175    let new_children = new_children
5176      .into_iter()
5177      .flat_map(|child| {
5178        if child.get_type() == Some(NodeType::DocumentFragNode) {
5179          child.get_child_nodes()
5180        } else {
5181          vec![child]
5182        }
5183      })
5184      .collect::<Vec<Node>>();
5185    // Now find all xml:id's in the new_children and record replacement id's for them
5186    let mut id_map = HashMap::default();
5187    // Find all id's defined in the copy and change the id.
5188    // Note: XPath ".//@xml:id" can fail to find namespace-qualified attributes.
5189    // Use DOM walking as fallback to ensure all ids are found.
5190    for child in new_children.iter() {
5191      let mut xpath_ids: Vec<String> = self.findvalues(".//@xml:id", Some(child));
5192      if xpath_ids.is_empty() {
5193        // Fallback: walk DOM to find xml:id attributes
5194        Self::collect_xml_ids_from(child, &mut xpath_ids);
5195      }
5196      for id in xpath_ids {
5197        id_map.insert(id.clone(), self.modify_id(id));
5198      }
5199    }
5200    // Now do the cloning (actually copying) and insertion.
5201    self.append_clone_aux(node, new_children, &mut id_map)
5202  }

5218  fn append_clone_aux(
5219    &mut self,
5220    node: &mut Node,
5221    new_children: Vec<Node>,
5222    id_map: &mut HashMap<String, String>,
5223  ) -> Result<()> {
5224    for child in new_children.into_iter() {
5225      match child.get_type() {
5226        Some(NodeType::ElementNode) => {
5227          let mut new = self.open_element_internal(
5228            node,
5229            child.get_namespace().map(|ns| ns.get_href()),
5230            &child.get_name(),
5231          )?;
5232          for (key, val) in child.get_attributes() {
5233            match key.as_str() {
5234              "xml:id" | "id" => {
5235                // Use the replacement id. The pre-walk
5236                // (findvalues/collect_xml_ids_from) normally populates
5237                // id_map with every `xml:id` it can see, but the
5238                // namespace/prefix of the attribute key returned by
5239                // libxml2's `get_attributes()` can differ from what the
5240                // XPath / DOM walk picked up (e.g. when the incoming
5241                // node's tree came through a cloneNode that stripped
5242                // the xml namespace). Fall back to minting a fresh
5243                // replacement id on-the-fly rather than panicking —
5244                // 1410.8508 hit this when the pre-walk found zero ids
5245                // but attributes on a child node did carry xml:id.
5246                let fresh;
5247                let mapped_id = match id_map.get(&val) {
5248                  Some(id) => id,
5249                  None => {
5250                    fresh = self.modify_id(val.clone());
5251                    id_map.insert(val.clone(), fresh.clone());
5252                    id_map.get(&val).unwrap()
5253                  },
5254                };
5255                let newid = self.record_id_with_node(mapped_id, &new);
5256                // Write the literal "xml:id" key (not the bare "id" local-
5257                // name returned by libxml's get_attributes). Otherwise the
5258                // cloned node only gets a plain `id` attribute, and the
5259                // subsequent `after_open` chain's `has_attribute_ns("id",
5260                // XML_NS)` check returns false, causing `generate_id` to
5261                // mint a fresh `.<parent>.N` xml:id that doesn't match the
5262                // sibling XMRef idrefs (which were rewritten from id_map).
5263                // Witness: arXiv:2509.07628 — MathFork mainfork emitted
5264                // 154 XMRefs with `.mf` idrefs while the cloned target
5265                // nodes received parent-scoped `.m2.N` xml:ids, leaving
5266                // every XMRef dangling and triggering 4
5267                // `Error:expected:id` per equation during post-processing
5268                // visibility marking. Same shape applies anywhere a
5269                // cloned subtree carries xml:id attributes — MathFork,
5270                // tabular-cell clone, _Capture_ flush.
5271                new.set_attribute("xml:id", &newid)?;
5272                // Update id_map so subsequent idref lookups use the ACTUAL recorded id.
5273                // record_id_with_node may change the id (e.g., if there are conflicts),
5274                // so the mapped_id and newid may differ.
5275                if *mapped_id != newid {
5276                  id_map.insert(val.clone(), newid);
5277                }
5278              },
5279              "idref" => {
5280                // Refer to the replacement id if it was replaced
5281                let id = id_map.get(&val).unwrap_or(&val);
5282                new.set_attribute(&key, id)?;
5283              },
5284              other_key =>

5314  /// Wrap `nodes` with an element named `qname`, making the new element replace
5315  /// the first `node`, and all `nodes` becomes the child of the new node.
5316  /// \[this makes most sense if `nodes` are a sequence of siblings\]
5317  ///
5318  /// Returns `None` if `qname` isn't allowed in the parent, or if `nodes`
5319  /// aren't allowed in `qname`, otherwise the newly created `qname` — so a
5320  /// caller must treat "wrapped" as a request the model may decline, not as a
5321  /// guarantee. `None` also covers a first node with no parent (already
5322  /// detached, or the root): there is nothing to wrap it in place of. Witness
5323  /// 1804.09736.
5324  ///
5325  /// The wrapper inherits the parent's font and box, so wrapping does not
5326  /// change how the enclosed material renders. [`unwrap_nodes`](Self::unwrap_nodes)
5327  /// is the inverse.
5328  pub fn wrap_nodes(&mut self, qname: &str, nodes: Vec<Node>) -> Result<Option<Node>> {
5329    if nodes.is_empty() {
5330      return Ok(None);
5331    }
5332    let first_node = &nodes[0];
5333    // Can't wrap a node that has no parent (already detached / is the root) —
5334    // return None like the other "can't wrap here" paths. Witness: 1804.09736.
5335    let Some(mut parent) = first_node.get_parent() else {
5336      return Ok(None);
5337    };
5338    let (ns, tag) = model::decode_qname(qname)?;
5339    let mut new = self.open_element_internal(&mut parent, ns, &tag)?;
5340    self.after_open(&mut new)?;
5341    parent.replace_child_node(new.clone(), first_node.clone())?;
5342
5343    self.copy_node_font(&parent, &mut new)?;
5344
5345    if let Some(tbox) = self.get_node_box(&parent) {
5346      self.set_node_box(&new, tbox);
5347    }
5348    for mut node in nodes.into_iter() {
5349      node.unlink();
5350      new.add_child(&mut node)?;
5351    }
5352    self.after_close(&mut new)?;
5353    Ok(Some(new))
5354  }

5362  /// Replace `node` by `nodes` (presumably descendants of some kind?)
5363  // DG: Don't return the replaced `node`, as it is groudns for memory management trouble
5364  //     with the low-level libxml layer. I've encountered segfaults here.
5365  pub fn replace_node(&mut self, mut node: Node, with: Vec<Node>) -> Result<()> {
5366    if let Some(_parent) = node.get_parent() {
5367      // libxml2's xmlAddNextSibling merges consecutive text nodes: when both
5368      // the reference sibling and the new node are TextNode, it appends the
5369      // new node's content to the reference node and frees the new node. The
5370      // Rust wrapper doesn't surface the merged result, so naively advancing
5371      // `c0_opt` to `with_node` would capture a pointer to freed memory,
5372      // producing silent data loss for the third+ insertion and eventually
5373      // a libxml2 SIGSEGV when the dangling pointer is re-traversed (e.g.
5374      // during a later XPath evaluation). Detect the text-text case and
5375      // coalesce in-place instead.
5376      let mut c0_opt: Option<Node> = None;
5377      for mut with_node in with.into_iter() {
5378        with_node.unlink();
5379        let is_text = with_node.get_type() == Some(NodeType::TextNode);
5380        if let Some(mut c0) = c0_opt {
5381          let c0_is_text = c0.get_type() == Some(NodeType::TextNode);
5382          if is_text && c0_is_text {
5383            let existing = c0.get_content();
5384            let added = with_node.get_content();
5385            c0.set_content(&format!("{existing}{added}"))?;
5386            // with_node is still a standalone (unlinked) text node; drop it.
5387            c0_opt = Some(c0);
5388            continue;
5389          }
5390          c0.add_next_sibling(&mut with_node)?;
5391        } else {
5392          // first node, swap in
5393          node.add_next_sibling(&mut with_node)?;
5394        }
5395        c0_opt = Some(with_node);
5396      }
5397      self.remove_node(node);
5398    }
5399    Ok(())
5400  }

5423  fn rename_node_internal(
5424    &mut self,
5425    mut node: Node,
5426    newname: SymStr,
5427    ns: Option<String>,
5428    tag: String,
5429    reinsert: bool,
5430  ) -> Result<Node> {
5431    let mut parent = node
5432      .get_parent()
5433      .expect("rename should never be called on an orphan or root node.");
5434    let mut new = self.open_element_internal(&mut parent, ns, &tag)?;
5435    // Move to the position AFTER node
5436    node.add_next_sibling(&mut new)?;
5437    // Copy ALL attributes from `node` to `newnode`
5438    let mut id = None;
5439    for (key, value) in node.get_attributes() {
5440      // `get_attributes()` returns the `xml:id` attribute under its LOCAL name
5441      // `"id"` (it lives in the XML namespace), NOT the prefixed `"xml:id"`.
5442      // Capture it here so it can be re-registered on `new` AFTER `remove_node`
5443      // unrecords it below — and re-set it through `Document::set_attribute`
5444      // (which routes "id"/"xml:id" through `record_id_with_node` + the XML
5445      // namespace), rather than the raw `Node::set_attribute` copy used for
5446      // ordinary attributes (that would drop the namespace AND the idstore
5447      // registration). Missing this stranded the equation refnum id across
5448      // `rearrange_lone_ams_aligned`'s equation→equationgroup rename, leaving
5449      // the group with a generic paragraph id and dangling intra-math XMRefs
5450      // (witness 2311.01600; see docs/parity/diagnostics/EXPECTED_ID_XMREF_DESIGN_2026-06-08.md).
5451      if key == "xml:id" || key == "id" {
5452        id = Some(value);
5453        continue;
5454      }
5455      let can_have = model::can_have_attribute(newname, arena::pin(&key));
5456      if can_have {
5457        new.set_attribute(&key, &value)?;
5458      }
5459    }
5460    // AND move all content from `node` to `newnode`
5461    if !reinsert {
5462      for mut child in node.get_child_nodes() {
5463        child.unbind();
5464        new.add_child(&mut child)?;
5465      }
5466    } else {
5467      std::mem::swap(&mut self.node, &mut new);
5468      for mut child in node.get_child_nodes() {
5469        child.unbind();
5470        if child.get_type() == Some(NodeType::TextNode) {
5471          self.open_text_internal(&child.get_content())?;
5472          self.close_text_internal()?;
5473        } else {
5474          let child_qname = get_node_qname(&child);
5475          let mut point = self.find_insertion_point_qsym(child_qname, None)?;
5476          point.add_child(&mut child)?;
5477        }
5478      }
5479      std::mem::swap(&mut self.node, &mut new);
5480    }
5481    // THEN call afterOpen... ?
5482    //   It would normally be called before children added,
5483    //   but how can we know if we're duplicated auto-added stuff?
5484    self.after_open(&mut new)?;
5485    self.after_close(&mut new)?;
5486    // Finally, remove the old node
5487    self.remove_node(node);
5488
5489    // and FINALLY, we can register the new node under the id.
5490    // `Document::set_attribute("xml:id", …)` routes through
5491    // `record_id_with_node` (idstore registration + dedup) and sets the
5492    // correctly-namespaced `xml:id` attribute. Only set it when the new qname
5493    // can carry an id (mirrors the per-attribute `can_have_attribute` gate).
5494    if let Some(id) = id
5495      && model::can_have_attribute(newname, arena::pin("xml:id"))
5496    {
5497      self.set_attribute(&mut new, "xml:id", &id)?;
5498    }
5499
5500    Ok(new)
5501  }

5503  /// Whitespace inside an explicit TYPEWRITER text wrapper is verbatim
5504  /// CONTENT (code indentation, measured by the sizing pass) — the p-edge
5505  /// trim recursion must not descend into it. During construction the
5506  /// public `font` attribute may not be written yet, so also decode the
5507  /// internal `_font`. `ltx:verbatim` itself is EXEMPT: Perl's edge trim
5508  /// descends into an inline `\verb` at a paragraph edge and trims its
5509  /// leading space (tokenize/verb.t), and we keep that parity.

5512      return false;

5514    if node
5515      .get_attribute("font")
5516      .is_some_and(|f| f.contains("typewriter"))
5517    {
5518      return true;
5519    }
5520    node
5521      .get_attribute("_font")
5522      .and_then(|id| self.decode_font(&id))
5523      .is_some_and(|f| f.family.as_deref() == Some("typewriter"))
5524  }

5526  pub fn trim_node_whitespace(&mut self, node: &Node) -> Result<()> {
5527    // A paragraph in a typewriter CONTEXT keeps its whitespace: verbatim
5528    // content is line-mapped into ltx:p's (fancyvrb et al.), where leading
5529    // spaces ARE the code indentation (2605.00468 JSON schemas flush-left)
5530    // and a space-only line is real content. Keyed on the PARENT's font
5531    // context — NOT on the node's own computed font, which is stamped from
5532    // whichever box auto-opened it (a prose paragraph STARTING with inline
5533    // \verb reads as typewriter and would wrongly keep its trailing prose
5534    // whitespace; tokenize/verb.t). Perl's own {verbatim} lands in
5535    // ltx:verbatim (no trim hook) so it never faces this; the raw-fancyvrb
5536    // constructs that do cannot be converted by Perl at all (fvextra
5537    // breaklines exceeds 7 min on a 6-line file) — surpass-Perl scope.
5538    if let Some(parent) = node.get_parent()
5539      && self.get_node_font(&parent).family.as_deref() == Some("typewriter")
5540    {
5541      return Ok(());
5542    }
5543    self.trim_node_left_whitespace(node)?;
5544    self.trim_node_right_whitespace(node)?;
5545    Ok(())
5546  }

5548  fn trim_node_left_whitespace(&self, node: &Node) -> Result<()> {
5549    if let Some(mut first_child) = node.get_first_child() {
5550      match first_child.get_type() {
5551        Some(NodeType::TextNode) => {
5552          let content = first_child.get_content();
5553          // Perl: s/^ +// — only trim ASCII spaces, preserve unicode spaces (nbsp, em-space, etc.)
5554          let trimmed_content = content.trim_start_matches(' ');
5555          if !content.is_empty() && (trimmed_content != content) {
5556            first_child.set_content(trimmed_content)?;
5557          }
5558        },
5559        Some(NodeType::ElementNode) if !self.is_typewriter_wrapper(&first_child) => {
5560          self.trim_node_left_whitespace(&first_child)?
5561        },
5562        _ => {},
5563      };
5564    }
5565    Ok(())
5566  }

5568  fn trim_node_right_whitespace(&self, node: &Node) -> Result<()> {
5569    // Skip trailing empty <text> font wrapper elements to find the real last content.
5570    // These are artifacts of font change tracking during alignment absorption.
5571    let mut candidate = node.get_last_child();
5572    while let Some(ref child) = candidate {
5573      if child.get_type() == Some(NodeType::ElementNode)
5574        && child.get_name() == "text"
5575        && child.get_first_child().is_none()
5576        && child.has_attribute("_noautoclose")
5577      {
5578        candidate = child.get_prev_sibling();
5579      } else {
5580        break;
5581      }
5582    }
5583    if let Some(mut last_child) = candidate {
5584      match last_child.get_type() {
5585        Some(NodeType::TextNode) => {
5586          let content = last_child.get_content();
5587          // Perl: s/\s+$// — but we can't trim all Unicode whitespace because some
5588          // tests have significant thin spaces (U+2009) from DimensionToSpaces.
5589          // Trim: ASCII whitespace, nbsp (U+00A0), em-space (U+2003), en-space (U+2002).
5590          let trimmed_content = content.trim_end_matches(|c: char| {
5591            c.is_ascii_whitespace() || c == '\u{00A0}' || c == '\u{2003}' || c == '\u{2002}'
5592          });
5593          if !content.is_empty() && (trimmed_content != content) {
5594            if trimmed_content.is_empty() {
5595              // Remove the entirely-whitespace text node
5596              last_child.unlink();
5597            } else {
5598              last_child.set_content(trimmed_content)?;
5599            }
5600          }
5601        },
5602        Some(NodeType::ElementNode) if !self.is_typewriter_wrapper(&last_child) => {
5603          self.trim_node_right_whitespace(&last_child)?
5604        },
5605        _ => {},
5606      };
5607    }
5608    Ok(())
5609  }

5671  /// Find a node in the document that can contain an element `qname`
5672  pub fn float_to_element(&mut self, qname: &str, closeifpossible: bool) -> Result<Option<Node>> {
5673    let mut candidates: VecDeque<Node> = VecDeque::from(self.get_insertion_candidates(&self.node));
5674    let mut closeable = true;
5675    // If the current node can contain already, we're fine right here - just return
5676    if !candidates.is_empty() && can_contain(&candidates[0], qname) {
5677      // Edge case: Don't resume at a text node, if it is current.
5678      // Don't append more to it after other insertions.
5679      if self.node.get_type() == Some(NodeType::TextNode) {
5680        self.set_node(&candidates[0]);
5681      }
5682      return Ok(candidates.pop_front());
5683    }
5684    while !candidates.is_empty() && !can_contain(&candidates[0], qname) {
5685      if closeable {
5686        closeable = can_auto_close(&candidates[0]);
5687      }
5688      candidates.pop_front();
5689    }
5690    match candidates.pop_front() {
5691      Some(n) => {
5692        if closeifpossible && closeable {
5693          self.close_to_node(&n, false)?;
5694        } else {
5695          let savenode = self.node.clone();
5696          self.set_node(&n);
5697          // Debug!("Floating from " . Stringify($savenode) . " to " . Stringify($n) . " for
5698          // $qname")   if ($$savenode ne $$n) && $LaTeXML::DEBUG{document};
5699          return Ok(Some(savenode));
5700        }
5701      },
5702      _ => {
5703        if can_contain_node_somehow(&self.node, qname).is_none() {
5704          Warn!(
5705            "malformed",
5706            qname,
5707            s!("No open node can contain element '{}'", qname)
5708          );
5709          // self.get_insertion_context())
5710        }
5711      },
5712    }
5713    Ok(None)
5714  }

5717  // A bit more than just whether the element can have the attribute, but
5718  // whether it has an id (and ideally either a refnum or title)
5719  pub fn float_to_label(&mut self) -> Option<Node> {
5720    let key = "labels";
5721    // Perl: start from lastChild of current node if it's an element
5722    let start = if self.node.get_type() == Some(NodeType::ElementNode) {
5723      self
5724        .node
5725        .get_last_child()
5726        .unwrap_or_else(|| self.node.clone())
5727    } else {
5728      self.node.clone()
5729    };

5731      .get_insertion_candidates(&start)
5732      .into_iter()
5733      .filter(|node| node.get_type() == Some(NodeType::ElementNode))
5734      .collect();
5735    let mut candidates: VecDeque<&Node> = ancestors.iter().collect();
5736    // Should we only accept a node that already has an id, or should we create an id?
5737    let mut node_opt: Option<Cow<Node>> = None;
5738    while let Some(candidate) = candidates.pop_front() {
5739      if can_node_have_attribute(candidate, key) && candidate.has_attribute_ns("id", XML_NS) {
5740        node_opt = Some(Cow::Borrowed(candidate));
5741        break;
5742      }
5743    }
5744
5745    if node_opt.is_none() {
5746      // No appropriate ancestor?
5747      let sib: Option<Node> = match ancestors.first() {
5748        Some(n) => n.get_last_child(),
5749        None => None,
5750      };
5751      if let Some(sibling) = sib {
5752        if can_node_have_attribute(&sibling, key) && sibling.has_attribute_ns("id", XML_NS) {
5753          node_opt = Some(Cow::Owned(sibling));
5754        } else if !ancestors.is_empty() {
5755          // just take root element?
5756          node_opt = Some(Cow::Borrowed(ancestors.last().as_ref().unwrap()));
5757        }
5758      } else if !ancestors.is_empty() {
5759        // just take root element?
5760        node_opt = Some(Cow::Borrowed(ancestors.last().as_ref().unwrap()));
5761      }
5762    }
5763    if let Some(node) = node_opt {
5764      let savenode = self.node.clone();
5765      self.set_node(&node);
5766      Some(savenode)
5767    } else {
5768      let message = s!("No open node with an xml:id can get attribute {:?}", key);
5769      Warn!("malformed", key, message);
5770      //  $self->getInsertionContext());
5771      None
5772    }
5773  }

5863  /// This function computes an xml:id for a node, if it hasn't already got one.
5864  /// It is suitable for use in Tag afterOpen as
5865  ///  `Tag('ltx:para',afterOpen=>sub { GenerateID(@_,'p'); });`
5866  /// It generates an id of the form `<parentid>.<prefix><number>`
5867  /// The parent node (the one with `ID=<parentid>`) also maintains a counter
5868  /// stored in an attribute `_ID_counter_<prefix>` recording the last used
5869  /// `number` for `prefix` amongst its descendents.
5870  pub fn generate_id(&mut self, node: &mut Node, mut prefix: &str) -> Result<()> {
5871    // If node doesn't already have an id, and can
5872    // but isn't a _Capture_ node (which ultimately should disappear)
5873    let qname = get_node_qname(node);
5874    if !node.has_attribute_ns("id", XML_NS)
5875      && model::can_have_attribute(qname, pin!("xml:id"))
5876      && (qname != pin!("ltx:_Capture_"))
5877    {
5878      let mut ancestor = self
5879        .findnode("ancestor::*[@xml:id][1]", Some(node))
5880        .unwrap_or_else(|| self.get_document().get_root_element().unwrap());
5881      //// Old versions don't like ancestor.getAttribute('xml:id');
5882      let ancestor_id = ancestor.get_attribute_ns("id", XML_NS);
5883      // If we've got no ancestor_id, then we've got no ancestor (no document yet!),
5884      // or ancestor IS the root element (but without an id);
5885      // If we also have no prefix, we'll end up with an illegal id (just digits)!!!
5886      // We'll use "id" for an id prefix; this will work whether or not we have an ancestor.
5887      if prefix.is_empty() && ancestor_id.is_none() {
5888        prefix = "id";
5889      }
5890
5891      // Perl `Package.pm:939` (`'_ID_counter_' . ($prefix ? $prefix . '_' : '')`)
5892      // — empty prefix uses `_ID_counter_` with a single trailing underscore,
5893      // not `_ID_counter__`. Matters for interop with code that reads the
5894      // attribute by exact name (e.g. `Base_XMath.pool.ltxml:940` reads
5895      // `_ID_counter_` for the empty-prefix counter).
5896      let ctrkey = if prefix.is_empty() {
5897        s!("_ID_counter_")
5898      } else {
5899        s!("_ID_counter_") + prefix + "_"
5900      };
5901      let a_ctr = ancestor.get_attribute(&ctrkey).unwrap_or_else(|| s!("0"));
5902
5903      let ctr_int = 1 + a_ctr.parse::<u32>().unwrap_or(0);
5904      let ctr = ctr_int.to_string();
5905
5906      let id = match ancestor_id {
5907        Some(aid) => aid + ".",
5908        None => String::new(),
5909      } + prefix
5910        + &ctr;
5911
5912      ancestor.set_attribute(&ctrkey, &ctr)?;
5913      self.set_attribute(node, "xml:id", &id)?;
5914    }
5915    Ok(())
5916  }

5925  pub fn replace_tree(&mut self, new: Node, old: Node) -> Result<Option<Node>> {
5926    match old.get_parent() {
5927      Some(mut parent) => {
5928        let mut following = VecDeque::new(); // Collect the matching and following nodes
5929        while let Some(mut sib) = parent.get_last_child() {
5930          if sib == old {
5931            break;
5932          }
5933          // parent.remove_child(sib); // We're putting these back, in a moment!
5934          sib.unlink();
5935          following.push_front(sib);
5936        }
5937        // Unrecord old's ids BEFORE the copy, so the copy's freshly
5938        // recorded entries (append_tree → open_element_at) survive — the
5939        // copy typically re-uses the same id strings.
5940        self.unrecord_node_ids(&old);
5941        // COPY FIRST, detach `old` after: `new` may sit INSIDE `old`
5942        // (parse_single's single-node shortcut returns an original child;
5943        // tex_box's foreignObject cleanup replaces a node with its own
5944        // grandchild), so `old` must stay intact while append_tree
5945        // serializes the copy. With `following` detached, the copy lands in
5946        // old's slot either way.
5947        self.append_tree(&mut parent, vec![new])?;
5948        let inserted = parent.get_last_child();
5949        // Keep the insertion point out of the discarded subtree (the
5950        // `chopped` half of remove_node; its id bookkeeping happened above).
5951        let mut cursor = Some(self.node.clone());
5952        while let Some(cur) = cursor {
5953          if cur == old {
5954            self.set_node(&parent);
5955            break;
5956          }
5957          cursor = cur.get_parent();
5958        }
5959        {
5960          let mut old = old;
5961          old.unlink();
5962        }
5963        for mut child in following {
5964          parent.add_child(&mut child)?; // No need for clone
5965        }
5966        Ok(inserted)
5967      },
5968      _ => Ok(None),
5969    }
5970  }

5972  /// `replace_tree` for a caller that owns BOTH trees as garbage-after-copy
5973  /// (the math parser's rebuild sites): the replacement is copied into
5974  /// place exactly as `replace_tree` does (Perl appendTree parity —
5975  /// elements are re-created, sources abandoned), and then the sources are
5976  /// FREED: `old`'s subtree, plus `new`'s detached root when `new` is a
5977  /// standalone built tree rather than a node inside `old`. Without the
5978  /// frees every replaced formula leaks its pre-parse tree AND the built
5979  /// parse tree (see `discard_subtree`).
5980  ///
5981  /// On the `None` return (old had no parent) nothing was copied and
5982  /// NOTHING is freed — the caller keeps using `new` as-is.
5983  pub fn replace_tree_free(&mut self, new: Node, old: Node) -> Result<Option<Node>> {
5984    // Resolve new's root BEFORE any freeing (walking afterwards would read
5985    // freed memory). A chain ending at a Document node means `new` is
5986    // inside a live tree — either inside `old` (freed below with it) or
5987    // elsewhere (not ours to free).
5988    let new_root = xml::detached_root(&new);
5989    let inserted = self.replace_tree(new, old.clone())?;
5990    if inserted.is_some() {
5991      self.discard_subtree(old);
5992      if let Some(root) = new_root {
5993        // Standalone source tree; disjoint from old's subtree by
5994        // construction (a detached root has no parent, every node inside
5995        // old's subtree has one).
5996        self.discard_subtree(root);
5997      }
5998    }
5999    Ok(inserted)
6000  }

6002  pub fn append_tree(&mut self, node: &mut Node, data: Vec<Node>) -> Result<()> {
6003    for child in data {
6004      match child.get_type() {
6005        Some(NodeType::ElementNode) => {
6006          let mut attributes: HashMap<String, String> =
6007            child.get_attributes().into_iter().collect();
6008          // Perl appendTree: REMOVE xml:id from source node before re-creation.
6009          // This prevents duplicate ID registration. The ID will be re-registered
6010          // by open_element_at when the new node is created with the same ID.
6011          if let Some(xmlid) = child.get_attribute_ns("id", XML_NS) {
6012            attributes
6013              .entry("xml:id".to_string())
6014              .or_insert_with(|| xmlid.clone());
6015            // Unrecord before re-creation (Perl: $child->removeAttribute('xml:id') + unRecordID)
6016            self.unrecord_id(&xmlid);
6017          }
6018
6019          // `get_FOREIGN_node_qname`: `data` may be a tree parsed elsewhere
6020          // (`Document::insert_xml`), whose elements can sit in a DEFAULT
6021          // namespace that is not ours — `<p xmlns="…/1999/xhtml">`. Only here is
6022          // that shape possible, and only here do we pay to resolve it.
6023          let tag_sym = model::get_foreign_node_qname(&child);
6024          let tag = arena::to_string(tag_sym);
6025          let mut new = self.open_element_at(node, &tag, Some(attributes), None)?;
6026          self.append_tree(&mut new, child.get_child_nodes())?;
6027          self.close_element_at(&mut new)?;
6028        },
6029        Some(NodeType::DocumentFragNode) => {
6030          self.append_tree(node, child.get_child_nodes())?;
6031        },
6032        Some(NodeType::TextNode) => {
6033          node.append_text(&child.get_content())?;
6034        },
6035        other => {
6036          log::debug!("append_tree: unhandled libxml NodeType {other:?}");
6037        },
6038      }
6039    }
6040    Ok(())
6041  }

6043  /// Parse an XML / (X)HTML markup string and splice the resulting subtree into
6044  /// the document at the current insertion point.
6045  ///
6046  /// Named as the markup counterpart to [`Document::insert_element`] (Perl
6047  /// `insertElement`): both insert an already-FINISHED thing at the current point.
6048  /// Deliberately NOT an `absorb*` name — in Perl `absorb` consumes a digested
6049  /// Box and has no `XML::LibXML` branch at all (it would die on a node), so
6050  /// borrowing that verb here would imply a kinship that does not exist.
6051  ///
6052  /// This is the Rust analog of Perl BookML's `\bmlRawHTML` idiom, which composes
6053  /// two mechanisms core never chains itself: `XML::LibXML->parse_string`
6054  /// (`LaTeXML::Common::XML::Parser` `parseChunk`, `Common/XML/Parser.pm:36-39`)
6055  /// and `$document->appendTree`
6056  /// (`Document.pm:2093`, foreign-node branch `:2105-2124`). Both halves already
6057  /// exist here — libxml's parser (a direct `latexml_core` dep, used in
6058  /// `common/relaxng/scan.rs`) and [`Document::append_tree`] — so this is pure
6059  /// glue.
6060  ///
6061  /// The markup must be WELL-FORMED, but need not be a single root: several
6062  /// sibling nodes, or bare text, are accepted as a document fragment
6063  /// (OXIDIZED_DESIGN #66 — Perl's `parseChunk` is single-node only). The parsed
6064  /// nodes are re-created one by one through `append_tree`'s model-aware path, so
6065  /// namespaces declared in the snippet (e.g. xhtml) are preserved and `xml:id`s
6066  /// re-registered. Malformed markup is REJECTED, never salvaged — see
6067  /// [`crate::common::xml::parse_chunk`] for what libxml's recovery mode
6068  /// destroys — and surfaces as a clean `Error:` that inserts nothing, degrading
6069  /// the offending binding rather than aborting the conversion (the
6070  /// runtime-bindings failure-isolation contract).
6071  pub fn insert_xml(&mut self, xml: &str) -> Result<()> {
6072    // The parsed Document OWNS the nodes `append_tree` re-creates from, so it
6073    // must stay alive across the call below (bound here, dropped at fn end).
6074    let parsed = match xml::parse_fragment(xml) {
6075      Ok(doc) => doc,
6076      Err(e) => {
6077        // Quote the offending snippet: a binding may insert markup from several
6078        // places, and the reason alone would not say WHICH one to go fix. Capped
6079        // so a runaway string cannot flood the log with a single message.
6080        const SNIPPET_CAP: usize = 200;
6081        let mut snippet: String = xml.chars().take(SNIPPET_CAP).collect();
6082        if xml.chars().nth(SNIPPET_CAP).is_some() {
6083          snippet.push('…');
6084        }
6085        Error!(
6086          "malformed",
6087          "insertXML",
6088          format!("could not parse XML markup: {e}"),
6089          format!("in: {snippet}")
6090        );
6091        return Ok(());
6092      },
6093    };
6094    if parsed.is_empty() {
6095      Error!(
6096        "malformed",
6097        "insertXML",
6098        "XML markup parsed to an empty document".to_string()
6099      );
6100      return Ok(());
6101    }
6102    // `parsed` stays bound until this returns, so it keeps owning the nodes.
6103    self.insert_nodes(parsed.nodes())
6104  }

6123/// the other C0 controls except TAB/LF/CR, plus the non-characters
6124/// U+FFFE/U+FFFF. libxml's `CString`-based APIs *panic* on interior NULs

6129fn xml_sanitize(value: &str) -> Cow<'_, str> {
6130  fn invalid(c: char) -> bool {
6131    (c < '\u{20}' && c != '\t' && c != '\n' && c != '\r') || c == '\u{FFFE}' || c == '\u{FFFF}'
6132  }
6133  if value.chars().any(invalid) {
6134    Cow::Owned(value.chars().filter(|c| !invalid(*c)).collect())
6135  } else {
6136    Cow::Borrowed(value)
6137  }
6138}
