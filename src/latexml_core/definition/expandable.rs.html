14/// Returns true when `\protect` currently has no meaning, or is
15/// `\let`-equivalent to `\relax`. Used by the recursion guard in
16/// `Expandable::invoke` to distinguish `\def\foo{\protect\foo}`
17/// definitions under safe (`\@unexpandable@protect`/`\string`/…)
18/// vs. unsafe (`\relax`/undefined) `\protect` regimes. Only the
19/// unsafe regime actually runaways at full expansion.
20/// True when `\protect` currently means `\@typeset@protect` — i.e. normal
21/// typesetting, where real LaTeX's `\@protected@testopt` DOES expand an
22/// optional-argument `\newcommand`. Under any other regime
23/// (`\protected@edef` sets `\@unexpandable@protect`, `\protected@write`
24/// sets `\@unexpandable@protect`, etc.) the real macro emits
25/// `\protect\cs` unexpanded. Used by the gullet's partial-expansion gate
26/// for binding-defined optional-argument macros (see
27/// `wisdom_replayed_tokens_param_packing` sibling entry; titlecaps OOM,
28/// perfect-kernel sweep 16).
29pub fn protect_is_typeset() -> bool {
30  let protect = lookup_meaning(&T_CS!("\\protect"));
31  let typeset = lookup_meaning(&T_CS!("\\@typeset@protect"));
32  match (protect, typeset) {
33    (Some(p), Some(t)) => p == t,
34    // No \protect regime established yet (early bootstrap): behave as
35    // typesetting (expand), the historical behavior.
36    _ => true,
37  }
38}

40pub(crate) fn protect_is_relax_or_undefined() -> bool {
41  let protect = T_CS!("\\protect");
42  match lookup_meaning(&protect) {
43    None => true,
44    // Meaning equal to \relax's meaning ⇒ unsafe.
45    Some(stored) => match lookup_meaning(&T_CS!("\\relax")) {
46      None => false,
47      Some(relax) => stored == relax,
48    },
49  }
50}

257  /// Expand the expandable control sequence. This should be carried out by the Gullet.
258  fn invoke(&self, once_only: bool) -> Result<Tokens> {
259    if self.may_be_watched()
260      && let Some(watched) = self.watched_call()
261    {
262      return self.expand_watched(once_only, watched);
263    }
264    self.expand(once_only)
265  }

267  /// [`Self::invoke`] onto the input, a token-list body without building the expansion: the parameterless
268  /// body is pushed as stored, a body with parameters through its substitution plan
269  /// ([`crate::gullet::unread_substituted`]). Closures, argument-only definitions and a watched call (whose
270  /// arguments are kept, as [`Self::invoke`] keeps them) take the general path.
271  fn invoke_onto_input(&self, once_only: bool) -> Result<()> {
272    if self.may_be_watched()
273      && let Some(watched) = self.watched_call()
274    {
275      crate::gullet::unread_expansion(self.expand_watched(once_only, watched)?);
276      return Ok(());
277    }
278    match self.special_call {
279      SpecialCall::None => {},
280      SpecialCall::YieldsCalls => {
281        if let Some(owner) = self.call_owner()? {
282          crate::gullet::unread_expansion(Tokens!(owner));
283          return Ok(());
284        }
285      },
286      SpecialCall::Native => {
287        if let Some(expansion) = native_call()? {
288          crate::gullet::unread_expansion(expansion);
289          return Ok(());
290        }
291      },
292    }
293    let Some(ExpansionBody::Tokens(tokens)) = &self.expansion else {
294      crate::gullet::unread_expansion(self.invoke(once_only)?);
295      return Ok(());
296    };
297    let Some(parms) = &self.paramlist else {
298      let body = self.parameterless_body(tokens, once_only)?;
299      crate::gullet::unread_substituted(body, &[BodyPiece::Run(0, body.len() as u32)], &[]);
300      return Ok(());
301    };
302    // A call that does not match its `\def` (tex.web §398) or whose argument ran off a file's end (§392)
303    // expands to nothing.
304    let Some(args) = self.read_call_arguments(parms)? else {
305      return Ok(());
306    };
307    if !self.has_cc_arg {
308      crate::gullet::unread_substituted(
309        tokens.unlist_ref(),
310        &[BodyPiece::Run(0, tokens.len() as u32)],
311        &[],
312      );
313      return Ok(());
314    }
315    self.substitute_onto_input(tokens, &args)
316  }

318  fn native_expansion(&self) -> Result<Option<Tokens>> {
319    match self.special_call {
320      SpecialCall::Native => native_call(),
321      _ => Ok(None),
322    }
323  }

338/// What a setter watch observes of a macro call ([`Expandable::watched_call`]).
339#[derive(Clone, Copy)]
340struct WatchedCall {
341  /// a document call's: its arguments kept ([`note_document_call`])
342  recorded: bool,
343  /// a binding macro's while the document's frontmatter entry is digested: the arguments it drops kept
344  /// ([`note_dropped_argument`])
345  dropping: bool,
346  /// a raw class's or own-directory file's setter's while the document's text is read: the arguments it stores kept
347  /// ([`note_store_call`])
348  stored:   bool,
349}
350
351impl Expandable {
352  /// Can a watch that is on see this call at all ([`crate::definition::origin::call_watch`])? [`Self::watched_call`] says
353  /// what it observes, on a cold path.
354  #[inline(always)]
355  fn may_be_watched(&self) -> bool {
356    crate::definition::origin::call_watch().is_some_and(|only_owned_with_parameters| {
357      !only_owned_with_parameters
358        || (self.paramlist.is_some() && (self.made_in_own_file || self.origin.is_latexml_owned()))
359    })
360  }
361
362  /// [`SpecialCall::YieldsCalls`]?
363  #[inline]
364  pub fn yields_calls(&self) -> bool { matches!(self.special_call, SpecialCall::YieldsCalls) }
365
366  /// What a setter watch (`definition::origin::setter_watch`) observes of this call, if anything: a watched setter's
367  /// ([`note_setter_call`], a stand-in copy's default replaced), a call of LaTeXML's own macro that reads its arguments from the
368  /// document's text (`origin::document_call_watch`, [`crate::gullet::reading_document_text`]: its arguments kept,
369  /// [`note_document_call`]), and a binding's macro called while a frontmatter entry the document gave is digested
370  /// (`origin::dropped_argument_watch`, `DefinitionOrigin::is_binding_made`: the arguments it drops kept,
371  /// [`note_dropped_argument`]); and a setter a raw class or a file of the document's own directory made, while the
372  /// current mouth reads the document's text, before the title page (`origin::store_call_watch`, off from the title page
373  /// on: the arguments it writes into a store kept, [`note_store_call`]) — its arguments may come back through
374  /// `\@ifnextchar` (nj_style's `\ead`, CUP-JNL-DTM's `\address`), so no [`crate::gullet::reading_document_text`]: a class's
375  /// own literal is no word the document's source writes. Not while a raw class's setter replays or a speculative
376  /// run digests.
377  #[cold]
378  #[inline(never)]
379  fn watched_call(&self) -> Option<WatchedCall> {
380    if crate::definition::origin::copy_default_watch() {
381      note_setter_call(&self.cs);
382    }
383    let recorded = crate::definition::origin::document_call_watch()
384      && self.paramlist.is_some()
385      && self.origin.is_latexml_owned()
386      && crate::gullet::reading_document_text();
387    let dropping = crate::definition::origin::dropped_argument_watch()
388      && self.paramlist.is_some()
389      && self.origin.is_binding_made()
390      && !matches!(self.expansion, Some(ExpansionBody::Closure(_)));
391    let stored = crate::definition::origin::store_call_watch()
392      && self.paramlist.is_some()
393      && self.made_in_own_file
394      && self.stored_parameters().numbers != 0
395      && !lookup_bool("lx_replaying_class_setter")
396      && !lookup_bool("lx_depositing_class_maketitle");
397    (recorded || dropping || stored).then_some(WatchedCall { recorded, dropping, stored })
398  }
399
400  /// What this macro's token body writes into a store ([`stored_parameters`]), once per definition; nothing of a closure.
401  fn stored_parameters(&self) -> &StoredParameters {
402    self.stored.get_or_init(|| {
403      Box::new(match self.expansion.as_ref() {
404        Some(ExpansionBody::Tokens(body)) => stored_parameters(body.unlist_ref()),
405        _ => StoredParameters::default(),
406      })
407    })
408  }

410  /// [`Self::expand`] for a watched call ([`Self::watched_call`]): a document call's arguments kept, each with whether its parameter is optional and whether the body names it (the title-page
411  /// check); and the required arguments a binding's macro drops while a document's frontmatter entry is digested.
412  #[cold]
413  #[inline(never)]
414  fn expand_watched(&self, once_only: bool, watched: WatchedCall) -> Result<Tokens> {
415    let expansion = if watched.recorded || watched.dropping || watched.stored {
416      let mut captured = None;
417      let expansion = self.expand_capturing::<true>(once_only, &mut captured)?;
418      if let (Some(args), Some(parms)) = (captured, self.paramlist.as_ref()) {
419        let parameters = parms.get_parameters();
420        // (an optional argument, or one set in brackets by a `\def`'s parameter text, is a store's key or label:
421        // `\@@address[#1]#2`, `\cortext[cor1]{…}` of an elsarticle copy, 1603.09660, 1005.1545)
422        let keyed: Vec<bool> = parameters
423          .iter()
424          .filter(|parameter| !parameter.novalue)
425          .map(|parameter| parameter.optional || parameter.spec == crate::pin!("Until:]"))
426          .collect();
427        let optional = parameters
428          .iter()
429          .filter(|parameter| !parameter.novalue)
430          .map(|parameter| parameter.optional);
431        // (a parameter a token body never names, `#n`, is dropped by the call: quantumarticle's `\ead[]{}` →
432        // `\email{#1}` drops the address; a closure's use is unknown, so its arguments are taken as handed on; one the
433        // binding declares dropped by design is accounted for — hyperref's `\texorpdfstring` #2, 2306.06741; and only
434        // a binding's drop can lose what pdflatex prints: the kernel's is TeX's own, as the format's `\@firstoftwo{T}{F}`
435        // drops the branch ifthen's `\ifthenelse` does not take around a document's `\title`, 2202.08832)
436        let used = |at: usize| {
437          !self.origin.is_binding_made()
438            || self.dropped_by_design & (1 << at) != 0
439            || match self.expansion.as_ref() {
440              Some(ExpansionBody::Tokens(body)) => names_parameter(body, at),
441              Some(ExpansionBody::Closure(_)) => true,
442              None => false,
443            }
444        };
445        let args: Vec<DocumentArgument> = optional
446          .zip(args.iter())
447          .enumerate()
448          .filter_map(|(at, (optional, arg))| match arg {
449            ArgWrap::Tokens(tokens) => Some(DocumentArgument {
450              number: at + 1,
451              optional,
452              used: used(at + 1),
453              tokens: tokens.clone(),
454            }),
455            _ => None,
456          })
457          .collect();
458        if watched.dropping {
459          for arg in args.iter().filter(|arg| !arg.optional && !arg.used) {
460            note_dropped_argument(self.cs, arg);
461          }
462        }
463        if watched.stored {
464          let stored = self.stored_parameters();
465          let (keys, values): (Vec<_>, Vec<_>) = args
466            .iter()
467            .filter(|arg| stored.numbers & (1 << arg.number) != 0 && !arg.tokens.is_empty())
468            .partition(|arg| keyed.get(arg.number - 1).copied().unwrap_or(false));
469          let tokens =
470            |args: Vec<&DocumentArgument>| args.into_iter().map(|arg| arg.tokens.clone()).collect();
471          note_store_call(self.cs, stored, tokens(values), tokens(keys));
472        }
473        if watched.recorded {
474          note_document_call(self.cs, args);
475        }
476      }
477      expansion
478    } else {
479      self.expand(once_only)?
480    };
481    Ok(expansion)
482  }

490  /// [`Self::expand`], its arguments copied into `captured` when `CAPTURE` ([`Self::expand_watched`]).
491  #[inline(always)]
492  fn expand_capturing<const CAPTURE: bool>(
493    &self,
494    once_only: bool,
495    captured: &mut Option<Vec<ArgWrap>>,
496  ) -> Result<Tokens> {
497    // Perl shortcut for "trivial" macros that were tracing- or
498    // profiling-aware. Neither tracing nor profiling is implemented
499    // in the Rust port (the returned `_tracing` / `_profiled` values
500    // were discarded), so the two state lookups were pure overhead
501    // on every macro expansion (\~350k calls in si.tex alone per
502    // callgrind). Removed — re-introduce only alongside the actual
503    // tracing/profiling features if/when they land.
504    if self.yields_calls()
505      && let Some(owner) = self.call_owner()?
506    {
507      return Ok(Tokens!(owner));
508    }
509    match &self.expansion {
510      Some(ExpansionBody::Closure(closure)) => {
511        // Harder to emulate \tracingmacros here.
512        let args = if let Some(ref parms) = self.paramlist {
513          // An argument that ran off a file's end abandons the call.
514          match self.read_call_arguments(parms)? {
515            Some(args) => args,
516            None => return Ok(Tokens!()),
517          }
518        } else {
519          Vec::new()
520        };
521        if CAPTURE {
522          *captured = Some(args.clone());
523        }
524        // Profiling: not implemented (Perl: startProfiling($profiled, 'expand'))
525        let result = closure(args)?;
526        // Tracing: Perl prints tracingCSName ==> tracetoString(result)
527        // Not implemented — silently skip to avoid panic on \tracingmacros=1
528        Ok(result)
529      },
530      Some(ExpansionBody::Tokens(tokens)) => {
531        let result = if self.paramlist.is_none() {
532          // Case: Trivial macro
533          // Profiling: not implemented (Perl: startProfiling($profiled, 'expand'))
534          // Tracing: Perl prints tracingCSName -> tracetoString(expansion)
535          // Not implemented — silently skip to avoid panic on \tracingmacros=1
536          Tokens::new(self.parameterless_body(tokens, once_only)?.to_vec())
537        } else {
538          let args = if let Some(ref parms) = self.paramlist {
539            // A call that does not match its `\def` (tex.web §398): reported,
540            // and the macro is ignored; so is one whose argument ran off a
541            // file's end (§392).
542            match self.read_call_arguments(parms)? {
543              Some(args) => args,
544              None => return Ok(Tokens!()),
545            }
546          } else {
547            Vec::new()
548          };
549          if CAPTURE {
550            *captured = Some(args.clone());
551          }
552          if self.has_cc_arg {
553            // Do we actually need to substitute the args in?
554            // Pre-size: one entry per argument; avoids Vec doublings on
555            // macros with many args.
556            let mut args_tks = Vec::with_capacity(args.len());
557            for arg in args.iter() {
558              args_tks.push(arg.as_tokens()?);
559            }
560            tokens.substitute_parameters(args_tks.as_slice())
561          } else {
562            tokens.clone()
563          }
564        };
565        // Profiling: Perl appends T_MARKER(profiled) for exclusive profiling
566        // Not implemented — silently skip
567        Ok(result)
568      },
569      None => {
570        // we always need to read the arguments, for e.g. things like \@gobble
571        if let Some(ref parms) = self.paramlist {
572          let args = self.read_call_arguments(parms)?;
573          if CAPTURE {
574            *captured = args;
575          }
576        }
577        Ok(NO_TOKENS)
578      },
579    }
580  }

583  /// call of it — an explicit catcode-1 `{`, or a `[`/`*` its signature opens with
584  /// ([`crate::state::call_openers`]) — and `None` for any other use, which expands the override. The peek reads
585  /// without expanding and puts the token back. Like TeX's argument scan it may read past the end of the macro body
586  /// the override was met in: `\def\wrap{\text}\wrap{x}` runs the binding on `{x}`. In an expansion-only context
587  /// (`\edef`, `\write`) the kept name `\lx@locked@<cs>` is what remains (OXIDIZED_DESIGN #500 residual).
588  fn call_owner(&self) -> Result<Option<Token>> {
589    let Some(next) = crate::gullet::peek_token()? else {

593    let opens_call = next.get_catcode() == Catcode::BEGIN
594      || next.get_catcode() == Catcode::OTHER && call_openers(&owner).contains(&next);
595    Ok(opens_call.then_some(owner))

598  /// The expansion of a parameterless body: the body, or nothing when it expands into itself
599  /// ([`Self::expands_into_itself`]), with `Error:recursion` (Perl Expandable.pm L81-89). The one place for
600  /// both [`Definition::invoke`] and [`Definition::invoke_onto_input`].
601  fn parameterless_body<'a>(&self, tokens: &'a Tokens, once_only: bool) -> Result<&'a [Token]> {
602    // For trivial expansion, make sure we don't get \cs or
603    // \relax\cs direct recursion!  Perl: Expandable.pm L81-89.
604    //   if (!$onceonly && $$self{cs}) {
605    //     my ($t0, $t1) = ($$expansion[0], $$expansion[1]);
606    //     if ($t0 && ($t0->equals($$self{cs})
607    //         || ($t1 && $t1->equals($$self{cs})
608    //              && $t0->equals(T_CS('\protect'))))) {
609    //       Error('recursion', $$self{cs}, …,
610    //         "Token X expands into itself!", "defining as empty");
611    //       $expansion = TokensI(); } }
612    //
613    // Detect `\def\foo{\foo}` and `\def\foo{\protect\foo}`. Both
614    // are runaway-expansion landmines under any full-expansion
615    // context (`\edef`, `\xdef`, `\write`, `\message`). Perl
616    // reports an `Error:recursion` and substitutes an empty
617    // expansion for this invocation; the stored definition is
618    // unchanged (subsequent invocations re-detect and re-error).
619    //
620    // A previous Rust port tried to re-install the CS as
621    // `Stored::Token(self.cs)` to preserve `\ifx` identity for
622    // expl3 quarks (`\q_no_value`, `\q_nil`, …) and PGF keys
623    // (`\pgfkeys@mainstop`). That was a no-op: `assign_meaning`'s
624    // `token == mt` short-circuit (state.rs:1918-1922) rejects
625    // the `\foo → \foo` self-let, so the Expandable definition
626    // stayed in place and the recursion guard re-fired forever.
627    // Witness: cleveref × algorithmicx × hyperref on 2403.15855,
628    // where `\xdef\cref@currentprefix{\cref@currentprefix}` hung
629    // at the 60 s wall-clock guard.
630    //
631    // Identity for expl3 quarks is independent of this path: the
632    // quarks are defined `\cs_new_protected:Npn`, so they are
633    // protected expandables. Under partial expansion (the normal
634    // path) protected expandables aren't expanded at all — the
635    // recursion guard never fires, and the stored body keeps the
636    // CS as its first token, so `\ifx`-by-meaning comparisons
637    // remain distinct. Under full expansion the Error+empty
638    // recovery matches Perl exactly.
639    if self.expands_into_itself(tokens, once_only) {
640      Error!(
641        "recursion",
642        &self.cs.to_string(),
643        s!("Token {} expands into itself!", self.cs)
644      );
645      Ok(&[])
646    } else {
647      Ok(tokens.unlist_ref())
648    }
649  }

651  /// The substitution half of [`Definition::invoke_onto_input`], out of line: its argument arrays stay out of
652  /// the frame of the argument reading, which recurses through `read_x_token`.
653  #[inline(never)]
654  fn substitute_onto_input(&self, tokens: &Tokens, args: &[ArgWrap]) -> Result<()> {
655    // A macro's arguments are read as tokens, so they're there to read in place (tex.web §390).
656    if args.iter().all(|arg| matches!(arg, ArgWrap::Tokens(_))) {
657      let plan = self.plan.get_or_init(|| tokens.substitution_plan());
658      crate::gullet::unread_substituted_args(tokens.unlist_ref(), plan, args);
659      return Ok(());
660    }
661    // The arguments as token lists; TeX's nine at most (§476), so on the stack.
662    let mut owned: [Option<Cow<'_, Tokens>>; 9] = Default::default();
663    if args.len() > owned.len() {
664      crate::gullet::unread_expansion(
665        tokens.substitute_parameters(
666          &args
667            .iter()
668            .map(|arg| arg.as_tokens())
669            .collect::<Result<Vec<_>>>()?,
670        ),
671      );
672      return Ok(());
673    }
674    for (slot, arg) in owned.iter_mut().zip(args.iter()) {
675      *slot = arg.as_tokens()?;
676    }
677    let mut slices: [&[Token]; 9] = [&[]; 9];
678    for (slice, arg) in slices.iter_mut().zip(owned.iter()) {
679      if let Some(arg) = arg {
680        *slice = arg.unlist_ref();
681      }
682    }
683    let plan = self.plan.get_or_init(|| tokens.substitution_plan());
684    crate::gullet::unread_substituted(tokens.unlist_ref(), plan, &slices[..args.len()]);
685    Ok(())
686  }

688  /// A parameterless body that begins with the control sequence being expanded (or with `\protect` and it,
689  /// while `\protect` is `\relax`) expands into itself: `Error:recursion`, and the call expands to nothing
690  /// (Perl Expandable.pm L81-89; the cases below).
691  fn expands_into_itself(&self, tokens: &Tokens, once_only: bool) -> bool {
692    if !once_only {
693      let token_vec = tokens.unlist_ref();
694      let t0_opt = token_vec.first();
695      let t1_opt = token_vec.get(1);
696      // OXIDIZED_DESIGN #185: anchor on the token actually being
697      // expanded, not the definition's home CS: a `\let` alias shares the Expandable
698      // (`self.cs` = the original), and musixlyr.tex:709-722 legitimately
699      // stores a body beginning with `\cont@<verse>` that is invoked
700      // through the alias `\der@kontext` after the original was cleared
701      // (Perl Expandable.pm:84 uses `$$self{cs}` and errs the same way:
702      // recorder-fingering, undar-digitacion-doc). A genuine
703      // `\def\x{\x}` invoked as `\x` still fires; through an alias the
704      // loop is caught one step later when `\x` itself expands.
705      let invoker = get_current_token().unwrap_or(self.cs);
706      if let Some(t0) = t0_opt {
707        if t0 == &self.cs && *t0 == invoker {
708          true
709        } else if let Some(t1) = t1_opt {
710          // `\protect\foo` is only an actual runaway when
711          // `\protect` currently expands to `\relax` (or is
712          // undefined). Under `\protected@edef` it is `\let`
713          // to `\@unexpandable@protect`, which turns the body
714          // into `\noexpand\protect\noexpand\foo` — both tokens
715          // become un-expandable and the loop terminates after
716          // one expansion. msg.sty (loaded transitively from
717          // french.sty, czech.sty, … under INCLUDE_STYLES=true)
718          // uses exactly this idiom for `\msgheader`, so the
719          // earlier blanket `\protect\foo`-is-runaway check
720          // fired ~3 errors per language-style paper. Witness:
721          // math9903002, gr-qc9511021, alg-geom9611022,
722          // math9807030/.../math9810088 (8 papers).
723          t1 == &self.cs && t0 == &T_CS!("\\protect") && protect_is_relax_or_undefined()
724        } else {
725          false
726        }
727      } else {
728        false
729      }
730    } else {
731      false
732    }
733  }

735  /// Read this macro's arguments as tex.web §389-391 `macro_call` does, at
736  /// `matching` status naming the invoked token (an alias's own name, as
737  /// TeX's `warning_index:=cur_cs`). `None` abandons the call: a delimiter
738  /// that does not match (§398), or an argument that ran off the end of a file
739  /// ("File ended while scanning use of \foo", §338-339 and §392; or, for a
740  /// non-`\long` macro whose argument held a `\par`, "Paragraph ended before
741  /// \foo was complete", §396: [`crate::gullet::abandon_runaway_call`]).
742  /// Guards: `scanner_status::*`.
743  fn read_call_arguments(&self, parms: &Parameters) -> Result<Option<Vec<ArgWrap>>> {
744    let cs = get_current_token().unwrap_or(self.cs);
745    let matching = crate::gullet::set_scanner_status_checking_par(
746      crate::gullet::ScannerStatus::Matching,
747      Some(cs),
748      self.checks_par,
749    );
750    let args = parms.read_macro_arguments(Some(self))?;
751    match matching.end_taking_runaway_argument() {
752      None => Ok(args),
753      Some(runaway) => {
754        crate::gullet::abandon_runaway_call(cs, self.is_long, self.checks_par, runaway)?;
755        Ok(None)
756      },
757    }
758  }

760  pub fn new(
761    cs: Token,
762    paramlist: Option<Parameters>,
763    mut expansion_opt: Option<ExpansionBody>,
764    traits: Option<ExpandableOptions>,
765  ) -> Result<Self> {
766    let traits = traits.unwrap_or_default();
767    if !traits.nopack_parameters
768      && let Some(ExpansionBody::Tokens(expansion_tokens)) = expansion_opt
769    {
770      // Perl Core/Definition/Expandable.pm:35: FATAL if the expansion is
771      // unbalanced (mismatched {/}). An Error-and-store-as-is let
772      // jarticle.cls:94-97's `\ds@tate` (an ISO-2022-JP byte pair whose `%`
773      // eats `\message`'s closing brace) proceed into japanese-otf's kanji
774      // scanners, an aperiodic 250 s `\advance` loop the cycle guard cannot
775      // see (platexcheat sample/platexsheet; Perl aborts in 1.3 s). Fatal
776      // stays Fatal. Guard: `perfect_kernel_batch56::unbalanced_expansion_is_fatal`.
777      if !expansion_tokens.is_balanced() {
778        Fatal!(
779          Stomach,
780          Misdefined,
781          s!("Expansion of '{}' has unbalanced {{}}", cs)
782        );
783      } else {
784        expansion_opt = Some(ExpansionBody::Tokens(expansion_tokens.pack_parameters()?));
785      }
786    }
787    let has_cc_arg = match expansion_opt {
788      Some(ExpansionBody::Tokens(ref tks)) => tks
789        .unlist_ref()
790        .iter()
791        .any(|t| t.get_catcode() == Catcode::ARG),
792      _ => false,
793    };
794    // simplify: treat empty tokens as None
795    let expansion = match expansion_opt {
796      Some(ExpansionBody::Tokens(tks)) if tks.is_empty() => None,
797      real_body => real_body,
798    };
799
800    // A `\newcommand` optional argument is read by `\@protected@testopt` →
801    // `\kernel@ifnextchar` (latex.ltx:1249, 1259-1261), a futurelet peek.
802    let peeks_by_futurelet = traits.peeks_by_futurelet
803      || paramlist.as_ref().is_some_and(|params| {
804        params
805          .get_parameters()
806          .first()
807          .is_some_and(|param| param.testopt)
808      });
809    let is_long = traits.long || get_prefix_sym(crate::pin!("long"));
810    let reads_arguments = paramlist
811      .as_ref()
812      .is_some_and(|params| !params.get_parameters().is_empty());
813    Ok(Expandable {
814      cs,
815      paramlist,
816      peeks_by_futurelet,
817      expansion,
818      // locator           => $source->getLocator,
819      // Hot path: Expandable::new fires on every \def/\edef; pin!-cached keys
820      // skip the per-call arena probe (same policy as Conditional::invoke).
821      is_protected: traits.protected || get_prefix_sym(crate::pin!("protected")),
822      is_outer: traits.outer || get_prefix_sym(crate::pin!("outer")),
823      is_long,
824      checks_par: !is_long
825        && reads_arguments
826        && (traits.short
827          || traits.tex_declared
828            && matches!(
829              crate::definition::origin::current_origin(),
830              crate::definition::origin::DefinitionOrigin::Format
831                | crate::definition::origin::DefinitionOrigin::File
832                | crate::definition::origin::DefinitionOrigin::Document
833            )),
834      has_cc_arg,
835      alias: traits.alias,
836      dropped_by_design: traits.drops.iter().fold(0, |bits, &number| {
837        // (a macro's arguments are #1-#9: a bit each in the u16)
838        debug_assert!(number <= 9, "a dropped argument number past #9: {number}");
839        bits | 1 << number
840      }),
841      made_in_own_file: crate::definition::origin::reading_own_file(),
842      ..Expandable::default()
843    })
844  }
845}

994/// The parameters the token body `body` writes into a store: inside the value `{…}` of a writer of a named store
995/// ([`WRITERS_OF_NAMES`] but `\let`, which takes one name; `\renewcommand`, `\newcommand`, `\providecommand`), of a token
996/// register's assignment, or of a `\write` to a named stream. CUP-JNL-DTM.cls's `\g@addto@macro\auaddress{…#2…}`
997/// (2305.00867), aeb_emulateapj's `\global\alt@affil@toks\expandafter{\the\alt@affil@toks … #2}` (1011.2770),
998/// naverlabseurope.cls's `\def\@authors{#1}` (2408.05088), bosonai_techreport.sty's
999/// `\renewcommand{\@affiliation}{#1}` (2605.02209), nj_style's `\immediate\write\@auxout{…{#1}…}` (1401.7920). A store the
1000/// body names by a parameter (`\gdef#1{#2}`) is a shorthand maker's, none; a `\csname` name is one, by no name. A name
1001/// written earlier in the body hands its parameters on to a value that names it (`\toks@{#1}\xdef\x{\the\toks@}`,
1002/// `\def\@tempa{#1}\global\let\x\@tempa`, SCIS2026.cls's temporary `\@authoremail`); a scratch name itself
1003/// ([`SCRATCH_NAMES`]) is no store.
1004fn stored_parameters(body: &[Token]) -> StoredParameters {
1005  const WRITERS_OF_COMMANDS: &[&str] = &["\\renewcommand", "\\newcommand", "\\providecommand"];
1006  let mut stored = StoredParameters {
1007    replaces: true,
1008    ..StoredParameters::default()
1009  };
1010  // (what each name written so far holds: its parameters, and every token of its value)
1011  let mut written: Vec<(Token, u16, Vec<Token>)> = Vec::new();
1012  let mut at = 0;
1013  while at < body.len() {
1014    let token = &body[at];
1015    at += 1;
1016    if token.get_catcode() != Catcode::CS {
1017      continue;
1018    }
1019    let is = |set: &[&str]| token.with_str(|name| set.contains(&name));
1020    let register = at >= 2 && body[at - 2].with_str(|name| name == "\\the");
1021    let is_let = token.with_str(|name| name == "\\let");
1022    // (the store's name, and from where the search for its value starts)
1023    let (store, value_search) = if is(WRITERS_OF_COMMANDS) || is(&WRITERS_OF_NAMES) {
1024      match written_store_name(body, at) {
1025        Some(found) => found,
1026        None => continue,
1027      }
1028    } else if token.with_str(|name| name == "\\write") {
1029      match body.get(at) {
1030        Some(stream) if stream.get_catcode() == Catcode::CS => (Some(*stream), at + 1),
1031        _ => continue,
1032      }
1033    } else if !register && is_token_register(token) {
1034      (Some(*token), at)
1035    } else {
1036      continue;
1037    };
1038    let scratch = store.is_some_and(|store| store.with_str(is_scratch_name));
1039    // (a value: a group's inside, or `\let`'s one token)
1040    let range = if is_let {
1041      let mut from = value_search;
1042      while body
1043        .get(from)
1044        .is_some_and(|t| t.get_catcode() == Catcode::SPACE || t.with_str(|text| text == "="))
1045      {
1046        from += 1;
1047      }
1048      (from < body.len()).then_some((from, from + 1))
1049    } else {
1050      value_group(body, value_search, token)
1051    };
1052    let Some((from, to)) = range else {
1053      continue;
1054    };
1055    let value = &body[from..to];
1056    // (a write that reads a store this body wrote first: `\gdef\@tmpz{#1}\g@addto@macro\x\@tmpz`, so a later call does
1057    // not supersede this one. Two stores side by side do: acm's `\conferenceinfo`'s `\global\conf={#1}\global\confinfo{#2}`,
1058    // called twice, 1612.06202)
1059    if value.iter().any(|inner| stored.stores.contains(inner)) {
1060      stored.replaces = false;
1061    }
1062    let mut named = 0u16;
1063    let mut holds: Vec<Token> = value.to_vec();
1064    for inner in value {
1065      if inner.get_catcode() == Catcode::ARG
1066        && let Some(number) = inner.with_str(|text| text.parse::<u16>().ok())
1067      {
1068        named |= 1 << number;
1069      } else if let Some((_, numbers, tokens)) = written.iter().find(|(name, ..)| name == inner) {
1070        named |= numbers;
1071        holds.extend_from_slice(tokens);
1072      }
1073    }
1074    if named != 0 && !scratch {
1075      stored.numbers |= named;
1076      let appends = is(APPENDERS) || token.with_str(|name| name == "\\write");
1077      match store {
1078        // (a `\csname` store: which one a later call writes, its counter's next, cannot be told)
1079        None => stored.replaces = false,
1080        Some(store) => {
1081          if appends || holds.contains(&store) {
1082            stored.replaces = false;
1083          }
1084          if token.with_str(|name| name == "\\write") {
1085            stored.written_out = true;
1086          } else {
1087            stored.stores.push(store);
1088          }
1089        },
1090      }
1091    }
1092    if let Some(store) = store
1093      && !(token.with_str(|name| name == "\\write"))
1094    {
1095      written.retain(|(name, ..)| *name != store);
1096      written.push((store, named, holds));
1097    }
1098    at = to;
1099  }
1100  stored
1101}

1103/// The store a writer at `at - 1` names at `at` (`\x`, `{\x}`, `*\x`, `\expandafter\x`, a `\csname…\endcsname` or the
1104/// group `\@namedef` and etoolbox's `\csgdef` take: none known then), and where the search for its value starts; `None`
1105/// when it names no store (`\gdef#1{#2}`).
1106pub fn written_store_name(body: &[Token], mut at: usize) -> Option<(Option<Token>, usize)> {
1107  while let Some(token) = body.get(at)
1108    && (token.get_catcode() == Catcode::SPACE
1109      || token.with_str(|text| text == "*" || text == "\\expandafter"))
1110  {
1111    at += 1;
1112  }
1113  let token = body.get(at)?;
1114  match token.get_catcode() {
1115    Catcode::CS if token.with_str(|name| name == "\\csname") => {
1116      let end = body[at..]
1117        .iter()
1118        .position(|t| t.with_str(|name| name == "\\endcsname"))?;
1119      Some((None, at + end + 1))
1120    },
1121    Catcode::CS | Catcode::ACTIVE => Some((Some(*token), at + 1)),
1122    Catcode::BEGIN => {
1123      let name = body.get(at + 1)?;
1124      let close = body.get(at + 2)?;
1125      if name.get_catcode() == Catcode::CS && close.get_catcode() == Catcode::END {
1126        return Some((Some(*name), at + 3));
1127      }
1128      // (a name built in a group)
1129      let mut depth = 0usize;
1130      for (offset, token) in body[at..].iter().enumerate() {
1131        match token.get_catcode() {
1132          Catcode::BEGIN => depth += 1,
1133          Catcode::END => {
1134            depth -= 1;
1135            if depth == 0 {
1136              return Some((None, at + offset + 1));
1137            }
1138          },
1139          _ => {},
1140        }
1141      }
1142      None
1143    },
1144    _ => None,
1145  }
1146}
1147
1148/// The inside of the value group of `writer`, searched from `at`: the next `{…}`, balanced. A `\def`'s parameter text may
1149/// stand before it, a `\newcommand`'s `[n]`; a register's assignment or a `\write` has only `=`, spaces and `\expandafter`
1150/// there (`\toks@\the\x` assigns no group; `\@auxout\foo`).
1151pub fn value_group(body: &[Token], mut at: usize, writer: &Token) -> Option<(usize, usize)> {
1152  let by_group_only = !writer.with_str(|name| {
1153    WRITERS_OF_NAMES.contains(&name)
1154      || matches!(name, "\\renewcommand" | "\\newcommand" | "\\providecommand")
1155  });
1156  while let Some(token) = body.get(at) {
1157    match token.get_catcode() {
1158      Catcode::BEGIN => break,
1159      Catcode::END => return None,
1160      Catcode::SPACE => {},
1161      _ if !by_group_only => {},
1162      _ if token.with_str(|text| text == "=" || text == "\\expandafter") => {},
1163      _ => return None,
1164    }
1165    at += 1;
1166  }
1167  let from = at + 1;
1168  let mut depth = 0usize;
1169  for (offset, token) in body.get(at..)?.iter().enumerate() {
1170    match token.get_catcode() {
1171      Catcode::BEGIN => depth += 1,
1172      Catcode::END => {
1173        depth -= 1;
1174        if depth == 0 {
1175          return Some((from, at + offset));
1176        }
1177      },
1178      _ => {},
1179    }
1180  }
1181  None
1182}

1184/// Does the token body `body` name its parameter `#number` (an argument token, `Catcode::ARG`)?
1185fn names_parameter(body: &Tokens, number: usize) -> bool {
1186  body.unlist_ref().iter().any(|token| {
1187    token.get_catcode() == Catcode::ARG && token.with_str(|text| text == number.to_string())
1188  })
1189}
