64  /// rebuild an equivalent merge-font Primitive.

66  pub origin:        crate::definition::origin::DefinitionOrigin,
67  /// The mode options `def_primitive` compiled into `before_digest` (K13).
68  pub declared_mode: Option<crate::definition::DeclaredMode>,
69  /// The `font` option `def_primitive` compiled into `before_digest` ([`Definition::declared_font`]).
70  pub font:          Option<FontDirective>,
71}

100impl Primitive {
101  /// Perl Primitive.pm:75 `Tokens($self->getCSorAlias, $parms->revertArguments(@args))`: the
102  /// box's reversion, each argument as its parameter reverts it (braces, optional brackets; an
103  /// absent optional reverts to nothing).
104  fn box_reversion(&self, args: &[ArgWrap]) -> Result<Tokens> {
105    let cs_token = self
106      .alias
107      .as_ref()
108      .map(|alias| Token::from(alias.as_str()))
109      .unwrap_or(self.cs);
110    let mut box_tokens = vec![cs_token];
111    if let Some(ref params) = self.paramlist {
112      let reverted = args
113        .iter()
114        .map(|arg| {
115          if arg.is_none() {
116            Ok(None)
117          } else {
118            arg.revert().map(Some)
119          }
120        })
121        .collect::<Result<Vec<_>>>()?;
122      box_tokens.extend(params.revert_arguments(reverted)?);
123    }
124    Ok(Tokens::new(box_tokens))
125  }
126}

137  fn invoke(&self, _once_only: bool) -> Result<Tokens> { Ok(Tokens!()) }
138  fn invoke_primitive(&self) -> Result<Vec<Digested>> {
139    Debug!("primitive invoke for {:?}", self.cs);
140    // my $profiled = $state->lookupValue('PROFILING') && ($LaTeXML::CURRENT_TOKEN || $$self{cs});
141    // my $tracing = $state->lookupValue('tracingcommands');
142    // LaTeXML::Core::Definition::startProfiling($profiled, 'digest') if $profiled;
143    // print STDERR '{' . $self->tracingCSName . "}\n" if $tracing;
144    let mut invoked_boxes: Vec<Digested> = self.execute_before_digest()?;
145    let args = self.read_arguments()?;
146    // print STDERR $self->tracingArgs(@args) . "\n" if $tracing && @args;
147    match self.replacement {
148      Some(PrimitiveBody::Closure(ref closure)) => invoked_boxes.extend(closure(args)?),
149      Some(PrimitiveBody::String(symbol)) => {
150        // Perl L73: $stomach->enterHorizontal if defined $replacement (def_primitive also puts it
151        // in the prologue, ahead of a bounded group; here it finds horizontal mode already)
152        crate::stomach::enter_horizontal();
153        let box_tokens = self.box_reversion(&args)?;
154        let box_props = SymHashMap::default();
155        invoked_boxes.push(Digested::from(Tbox::new(
156          symbol, None, None, box_tokens, box_props,
157        )));
158      },
159      None => {
160        // Perl L74-76: an undef replacement is an empty box (`isEmpty`) reverting as a string's
161        // does, so a font switch (`\rm`) shows in `tex=` and `\mark{x}` keeps its argument.
162        let box_tokens = self.box_reversion(&args)?;
163        let mut box_props = SymHashMap::default();
164        box_props.insert("isEmpty", Stored::Bool(true));
165        invoked_boxes.push(Digested::from(Tbox::new(
166          pin!(""),
167          None,
168          None,
169          box_tokens,
170          box_props,
171        )));
172      },
173    }
174    if !self.after_digest.is_empty() {
175      // optimize to avoid needless generation of whatsits
176      let mut w = Whatsit::default();
177      let after_boxes = self.execute_after_digest(&mut w)?;
178      invoked_boxes.extend(after_boxes);
179    }
180
181    // LaTeXML::Core::Definition::stopProfiling($profiled, 'digest') if $profiled;
182    Ok(invoked_boxes)
183  }

230  #[test]
231  fn primitive_partial_eq_by_cs() {
232    // PartialEq compares by cs only — Perl parity (closures can't
233    // be structurally compared).
234    // Primitive doesn't derive Debug, so assert_eq! / assert_ne!
235    // can't format it on failure — use plain equality checks.
236    let mut a = Primitive::default();
237    let mut b = Primitive::default();
238    a.cs = T_CS!("\\foo");
239    b.cs = T_CS!("\\foo");
240    assert!(a == b, "same cs should compare equal");
241    b.cs = T_CS!("\\bar");
242    assert!(!(a == b), "different cs should not be equal");
243  }
