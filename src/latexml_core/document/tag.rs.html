25/// The attribute key of the author block's own creator, which the frontmatter emitter writes below a flush's names: no
26/// annotation's parent by place ([`TagList::parent_indices`]). The same key as the engine's `frontmatter_api::BLOCK_CREATOR`,
27/// which declares it for the frontmatter API.
28pub const BLOCK_CREATOR: &str = "_blockcreator";

30/// One tag's frontmatter entries, Perl's `@{$frontmatter{$tag}}` (Base_Utility.pool.ltxml), with the indices of the
31/// entries an annotation may attach to kept as the list changes — Perl's `grep { ($$_[1]{role} || '') ne 'pending' }`
32/// (Base_Utility.pool.ltxml:492), read per annotation instead of rescanned over every entry. The
33/// entries are a slice to read; they change only through the list's own methods, which keep the index exact: `push`,
34/// `remove`, `retain`, and [`TagList::get_mut`]/[`TagList::last_mut`], whose [`EntryMut`] re-files its entry when an edit
35/// changed its kind (debug builds also check the index against the scan at every read).
36#[derive(Clone, Default, PartialEq)]
37pub struct TagList {
38  entries: Vec<TagData>,
39  /// the entries not `role=pending`, in order, without the block creators
40  parents: Vec<usize>,
41  /// the entries not `role=pending` with the [`BLOCK_CREATOR`] key, in order
42  block:   Vec<usize>,
43}
44
45/// Which of [`TagList`]'s indices an entry goes to: none (a pending stub), the parents, or the block creators.
46fn parent_kind(entry: &TagData) -> Option<bool> {
47  (entry.attr.get("role").map(String::as_str) != Some("pending"))
48    .then(|| entry.attr.contains_key(BLOCK_CREATOR))
49}

51impl TagList {
52  /// The list of `entries`, indexed.
53  pub fn from_entries(entries: Vec<TagData>) -> Self {
54    let mut list = TagList {
55      entries,
56      parents: Vec::new(),
57      block: Vec::new(),
58    };
59    list.reindex();
60    list
61  }
62  fn reindex(&mut self) {
63    self.parents.clear();
64    self.block.clear();
65    for (i, entry) in self.entries.iter().enumerate() {
66      match parent_kind(entry) {
67        Some(false) => self.parents.push(i),
68        Some(true) => self.block.push(i),
69        None => {},
70      }
71    }
72  }
73  /// Append `entry` (Perl's `push(@{$frontmatter{$tag}}, $entry)`).
74  pub fn push(&mut self, entry: TagData) {
75    let i = self.entries.len();
76    match parent_kind(&entry) {
77      Some(false) => self.parents.push(i),
78      Some(true) => self.block.push(i),
79      None => {},
80    }
81    self.entries.push(entry);
82  }
83  /// Remove and return the entry at `index` (panics if out of bounds, as `Vec::remove`).
84  pub fn remove(&mut self, index: usize) -> TagData {
85    let entry = self.entries.remove(index);
86    for indices in [&mut self.parents, &mut self.block] {
87      if let Ok(at) = indices.binary_search(&index) {
88        indices.remove(at);
89      }
90      let from = indices.partition_point(|&i| i < index);
91      for i in &mut indices[from..] {
92        *i -= 1;
93      }
94    }
95    entry
96  }
97  /// Keep only the entries `keep` holds (`Vec::retain`).
98  pub fn retain(&mut self, keep: impl FnMut(&TagData) -> bool) {
99    self.entries.retain(keep);
100    self.reindex();
101  }
102  /// The entries an annotation may attach to: those not `role=pending`, in order — without the block creators when
103  /// `exclude_block` (once a flush wrote one).
104  pub fn parent_indices(&self, exclude_block: bool) -> Vec<usize> {
105    debug_assert!(
106      {
107        let scan = TagList::from_entries(self.entries.clone());
108        scan.parents == self.parents && scan.block == self.block
109      },
110      "TagList index out of step with its entries"
111    );
112    if exclude_block || self.block.is_empty() {
113      return self.parents.clone();
114    }
115    let mut all = Vec::with_capacity(self.parents.len() + self.block.len());
116    all.extend_from_slice(&self.parents);
117    all.extend_from_slice(&self.block);
118    all.sort_unstable();
119    all
120  }
121  /// The entries, as a `Vec`.
122  pub fn into_entries(self) -> Vec<TagData> { self.entries }
123  /// The entry at `index` to edit in place, `None` out of bounds (`slice::get_mut`): the index follows whatever the
124  /// edit does to its role or [`BLOCK_CREATOR`] key.
125  pub fn get_mut(&mut self, index: usize) -> Option<EntryMut<'_>> {
126    let entry = self.entries.get_mut(index)?;
127    Some(EntryMut {
128      kind: parent_kind(entry),
129      entry,
130      index,
131      parents: &mut self.parents,
132      block: &mut self.block,
133    })
134  }
135  /// The last entry to edit in place, as [`TagList::get_mut`].
136  pub fn last_mut(&mut self) -> Option<EntryMut<'_>> {
137    let last = self.entries.len().checked_sub(1)?;
138    self.get_mut(last)
139  }
140}

142/// One entry of a [`TagList`] edited in place ([`TagList::get_mut`]): when it is dropped, an entry whose kind (pending
143/// stub, parent, block creator) the edit changed moves to the index of its new kind — an O(1) check when unchanged.
144/// The re-filing runs in `Drop`, so a guard passed to `std::mem::forget` would leave the index stale (no caller does;
145/// debug builds check the index against a scan at every read: review revI7).
146pub struct EntryMut<'a> {
147  entry:   &'a mut TagData,
148  index:   usize,
149  kind:    Option<bool>,
150  parents: &'a mut Vec<usize>,
151  block:   &'a mut Vec<usize>,
152}
153
154impl std::ops::Deref for EntryMut<'_> {
155  type Target = TagData;
156  fn deref(&self) -> &TagData { self.entry }
157}
158
159impl std::ops::DerefMut for EntryMut<'_> {
160  fn deref_mut(&mut self) -> &mut TagData { self.entry }
161}
162
163impl Drop for EntryMut<'_> {
164  fn drop(&mut self) {
165    let kind = parent_kind(self.entry);
166    if kind == self.kind {
167      return;
168    }
169    let index = self.index;
170    match self.kind {
171      Some(false) => sorted_remove(self.parents, index),
172      Some(true) => sorted_remove(self.block, index),
173      None => {},
174    }
175    match kind {
176      Some(false) => sorted_insert(self.parents, index),
177      Some(true) => sorted_insert(self.block, index),
178      None => {},
179    }
180  }
181}
182
183fn sorted_remove(indices: &mut Vec<usize>, index: usize) {
184  if let Ok(at) = indices.binary_search(&index) {
185    indices.remove(at);
186  }
187}
188
189fn sorted_insert(indices: &mut Vec<usize>, index: usize) {
190  if let Err(at) = indices.binary_search(&index) {
191    indices.insert(at, index);
192  }
193}

231// Specify the properties of a Node tag.
232pub enum TagOptionName {
233  AfterOpen,
234  AfterOpenEarly,
235  AfterOpenLate,
236  AfterClose,
237  AfterCloseEarly,
238  AfterCloseLate,
239}
240
241impl TagOptionName {
242  pub fn all() -> Vec<TagOptionName> {
243    use self::TagOptionName::*;
244    vec![
245      AfterOpen,
246      AfterOpenEarly,
247      AfterOpenLate,
248      AfterClose,
249      AfterCloseEarly,
250      AfterCloseLate,
251    ]
252  }
253  pub fn is_prepend(&self) -> bool {
254    use self::TagOptionName::*;
255    matches!(*self, AfterOpenEarly | AfterCloseEarly)
256  }
257  pub fn is_append(&self) -> bool {
258    use self::TagOptionName::*;
259    matches!(
260      *self,
261      AfterOpen | AfterClose | AfterOpenLate | AfterCloseLate
262    )
263  }

266/// How readily an element is auto-opened to hold a child it can contain — Perl's
267/// `Tag($tag, autoOpen => $x)`, where any true value counts as 1 and a fraction
268/// (`ltx:picture`'s 0.5, latex_constructs.pool.ltxml:4943) makes the element's paths
269/// lose to those through a full auto-opener (Document.pm:196-199, 220). Kept as a
270/// percentage (100 = 1, 50 = 0.5) for the indirect model's integer arithmetic
271/// (`compute_indirect_model`).
272#[derive(Clone, Copy, Debug, PartialEq, Eq)]
273pub struct AutoOpen(pub u32);
274
275impl From<bool> for AutoOpen {
276  fn from(open: bool) -> Self { AutoOpen(if open { 100 } else { 0 }) }
277}
278
279impl From<f64> for AutoOpen {
280  fn from(fraction: f64) -> Self { AutoOpen((fraction.clamp(0.0, 1.0) * 100.0).round() as u32) }
281}

283#[derive(Clone, Default)]
284pub struct TagOptions {
285  pub auto_open:         Option<AutoOpen>,
286  pub auto_close:        Option<bool>,
287  pub after_open:        Option<Vec<TagConstructionClosure>>,
288  pub after_close:       Option<Vec<TagConstructionClosure>>,
289  pub after_open_early:  Option<Vec<TagConstructionClosure>>,
290  pub after_close_early: Option<Vec<TagConstructionClosure>>,
291  pub after_open_late:   Option<Vec<TagConstructionClosure>>,
292  pub after_close_late:  Option<Vec<TagConstructionClosure>>,
293}
294
295impl TagOptions {
296  pub fn get(&self, name: &TagOptionName) -> Option<&Vec<TagConstructionClosure>> {
297    use self::TagOptionName::*;
298    match *name {
299      AfterOpen => self.after_open.as_ref(),
300      AfterOpenEarly => self.after_open_early.as_ref(),
301      AfterOpenLate => self.after_open_late.as_ref(),
302      AfterClose => self.after_close.as_ref(),
303      AfterCloseEarly => self.after_close_early.as_ref(),
304      AfterCloseLate => self.after_close_late.as_ref(),
305    }
306  }
307
308  pub fn get_mut(&mut self, name: &TagOptionName) -> Option<&mut Vec<TagConstructionClosure>> {
309    use self::TagOptionName::*;
310    match *name {
311      AfterOpen => self.after_open.as_mut(),
312      AfterOpenEarly => self.after_open_early.as_mut(),
313      AfterOpenLate => self.after_open_late.as_mut(),
314      AfterClose => self.after_close.as_mut(),
315      AfterCloseEarly => self.after_close_early.as_mut(),
316      AfterCloseLate => self.after_close_late.as_mut(),
317    }
318  }
319
320  pub fn remove(&mut self, name: &TagOptionName) -> Vec<TagConstructionClosure> {
321    use self::TagOptionName::*;
322    let field = match *name {
323      AfterOpen => &mut self.after_open,
324      AfterOpenEarly => &mut self.after_open_early,
325      AfterOpenLate => &mut self.after_open_late,
326      AfterClose => &mut self.after_close,
327      AfterCloseEarly => &mut self.after_close_early,
328      AfterCloseLate => &mut self.after_close_late,
329    };
330    match field {
331      Some(vec) => std::mem::take(vec),
332      None => Vec::new(),
333    }
334  }
335
336  pub fn set(&mut self, name: &TagOptionName, value: Option<Vec<TagConstructionClosure>>) {
337    use self::TagOptionName::*;
338    match *name {
339      AfterOpen => {
340        self.after_open = value;
341      },
342      AfterOpenEarly => {
343        self.after_open_early = value;
344      },
345      AfterOpenLate => {
346        self.after_open_late = value;
347      },
348      AfterClose => {
349        self.after_close = value;
350      },
351      AfterCloseEarly => {
352        self.after_close_early = value;
353      },
354      AfterCloseLate => {
355        self.after_close_late = value;
356      },
357    }
358  }
359
360  pub fn prepend(&mut self, name: &TagOptionName, mut value: Vec<TagConstructionClosure>) {
361    let drained: Vec<TagConstructionClosure> = match self.get_mut(name) {
362      Some(vec) => std::mem::take(vec),
363      None => Vec::new(),
364    };
365
366    value.extend(drained);
367
368    self.set(name, Some(value));
369  }
370
371  pub fn append(&mut self, name: &TagOptionName, value: Vec<TagConstructionClosure>) {
372    if self.get(name).is_none() {
373      // initialize if needed
374      self.set(name, Some(Vec::new()));
375    }
376
377    if let Some(vec) = self.get_mut(name) {
378      vec.extend(value);
379    }
380  }
381}

387  #[test]
388  fn tag_list_entry_edit_refiles_its_kind() {
389    let entry = |attrs: &[(&str, &str)]| TagData {
390      tag:     "ltx:creator".to_string(),
391      attr:    attrs
392        .iter()
393        .map(|(k, v)| (k.to_string(), v.to_string()))
394        .collect(),
395      content: Vec::new(),
396    };
397    let mut list = TagList::from_entries(vec![
398      entry(&[]),
399      entry(&[("role", "pending")]),
400      entry(&[(BLOCK_CREATOR, "1")]),
401    ]);
402    assert_eq!(list.parent_indices(true), vec![0]);
403    assert_eq!(list.parent_indices(false), vec![0, 2]);
404    // a stub made a parent, a parent made a stub, a block creator made a parent: each re-filed on drop
405    list.get_mut(1).unwrap().attr.remove("role");
406    list
407      .get_mut(0)
408      .unwrap()
409      .attr
410      .insert("role".into(), "pending".into());
411    list.last_mut().unwrap().attr.remove(BLOCK_CREATOR);
412    assert_eq!(list.parent_indices(true), vec![1, 2]);
413    list
414      .get_mut(1)
415      .unwrap()
416      .attr
417      .insert(BLOCK_CREATOR.into(), "1".into());
418    assert_eq!(list.parent_indices(true), vec![2]);
419    assert_eq!(list.parent_indices(false), vec![1, 2]);
420    assert!(list.get_mut(3).is_none());
421  }
