14// Process-once cached env var (see WISDOM #56 — getenv hot-path race).
15static SOURCE_DATE_EPOCH: Lazy<Option<String>> =
16  Lazy::new(|| std::env::var("SOURCE_DATE_EPOCH").ok());

18/// `SOURCE_DATE_EPOCH` (Unix seconds), when set and numeric: the reproducible-build clock Perl reads in
19/// TeX_Job.pool.ltxml:42.
20fn source_date_epoch_env() -> Option<i64> {
21  SOURCE_DATE_EPOCH
22    .as_deref()
23    .and_then(|epoch| epoch.trim().parse::<i64>().ok())
24}

26/// Sets the job's clock — `\year`, `\month`, `\day`, `\time` (globally) — to the instant `epoch` (Unix seconds) on the
27/// UTC clock, as Perl TeX_Job.pool.ltxml:42-46 does for `SOURCE_DATE_EPOCH` (`gmtime`). `\today` and
28/// `\pdfcreationdate` read these registers. Returns whether `epoch` is a representable instant (the clock is left
29/// untouched when it is not).
30pub fn assign_date_registers(epoch: i64) -> bool {
31  match DateTime::from_timestamp(epoch, 0) {
32    Some(utc) => {
33      assign_clock_registers(&utc);
34      true
35    },
36    None => false,
37  }
38}
39
40fn assign_clock_registers<Tz: TimeZone>(dt: &DateTime<Tz>) {
41  AssignValue!("\\day", Number!(dt.day()), Scope::Global);
42  AssignValue!("\\month", Number!(dt.month()), Scope::Global);
43  AssignValue!("\\year", Number!(dt.year()), Scope::Global);
44  AssignValue!(
45    "\\time",
46    Number!(60 * dt.hour() + dt.minute()),
47    Scope::Global
48  );
49}

120  //======================================================================
121  // LaTeX 2.09 compatibility
122  //----------------------------------------------------------------------
123  // Perl latex_constructs.pool.ltxml:97-129 (`\documentstyle` afterDigest).
124  // LaTeX 2.09 compat shim. Three branches mirroring Perl strictly:
125  //
126  //   1. <class>.sty exists  → input_definitions("article", cls, handleoptions=true, options=opts)
127  //                          + require_package(class, as_class=true, after=\compat@loadpackages)
128  //   2. <class>.cls exists  → load_class(class, opts, after=\compat@loadpackages)
129  //   3. neither             → input_definitions("OmniBus", cls, handleoptions=true, options=opts,
130  //      after=\compat@loadpackages)
131  //                          + require_package(class, as_class=true)
132  //
133  // Critical Perl semantics (matching latex_constructs.pool.ltxml:97-129):
134  //   * `handleoptions => 1` makes the cls's `\DeclareOption`/`\ProcessOptions` consume the
135  //     `<opts>` and route leftovers onto `@unusedoptionlist`.
136  //   * `after => \compat@loadpackages` runs *after* the cls finishes its option-processing — at
137  //     that point unmet options sit on the unused list and `\compat@loadpackages`
138  //     (`latex_constructs.rs:2502`) walks them, RequirePackage's any that resolve, and triggers
139  //     OmniBus when anything went unmet. That is what lets `\documentstyle[paspconf] {article}`
140  //     transitively load `aas_macros.sty.ltxml` to define `\affil` / `\altaffilmark` /
141  //     `\acknowledgments` etc.
142  // The latex_dump unconditionally redefines `\documentstyle` with the
143  // kernel-style `\input{latex209.def}\documentclass` form (Perl
144  // latex_dump.pool.ltxml entry for `\documentstyle`). In Perl that
145  // version is itself overridden by latex_constructs.pool.ltxml's
146  // DefConstructor; in our Rust port `latex_constructs.rs` doesn't
147  // redefine `\documentstyle`, so without intervention any autoload-of-
148  // LaTeX-pool path (e.g. `\newcommand` before `\documentstyle`)
149  // replaces our impl with the dump's, breaking the
150  // `\compat@loadpackages` after-hook that dispatches `[epsf]` etc.
151  // from `@unusedoptionlist`. Witness: hep-th9912229
152  // (`\newcommand` before `\documentstyle[12pt,epsf]`).
153  //
154  // Workaround: register the impl under a stable backup name
155  // `\lx@documentstyle@impl` and `\let \documentstyle = \lx@documentstyle@impl`
156  // at the end of `\@load@latex@pool` so we restore our impl after
157  // every LaTeX pool load.
158  DefMacro!("\\lx@documentstyle@impl[]{}", sub[(options_opt, class_tks)] {
159    use latexml_core::binding::content::{
160      find_file, find_file_fallback_exists, FindFileOptions, load_class,
161    };
162    // Perl latex_constructs.pool.ltxml:79 — `$class =~ s/\s+//g;`. Strip ALL
163    // whitespace (not just leading/trailing as a bare `.trim()` would), so a
164    // multi-line `\documentstyle[...]{<newline>article}` — whose post-`{`
165    // newline becomes an embedded space token in the Semiverbatim arg — still
166    // matches its binding registration. Mirrors the sibling `\documentclass`
167    // fix (latex_constructs.rs); a whitespace-bearing name otherwise misses
168    // the binding and falls through to OmniBus.
169    let class: String = class_tks
170      .to_string()
171      .chars()
172      .filter(|c| !c.is_ascii_whitespace())
173      .collect();
174
175    // A second class declaration loads no class: latex.ltx's `\documentclass` has made itself `\@twoclasseserror`
176    // (:18617-18618, :18944-18947), which `\documentstyle`'s calls too (:18622-18624), "Two \documentclass or
177    // \documentstyle commands". A first declaration no binding and no file answered (the OmniBus fallback, branch 3
178    // below; `sect01.rs` `\documentclass`) does not count (main-session ruling 2026-10-19): astro-ph9812142's line 1
179    // `#\documentstyle[psfig,epsf]{/disk/3/roud/home/Texstuff/aa}` names a path no tree has (the `#` is no comment:
180    // pdflatex's "You can't use macro parameter character #", then the missing class stops it), and its line 2
181    // `\documentstyle[psfig,epsf]{aa}` loads the real class (aa's `\thesaurus`, `\offprints`, bibitem tags).
182    if lookup_bool(crate::latex_constructs::CLASS_DECLARED) {
183      return Ok(TokenizeInternal!(
184        r"\@latex@error{Two \noexpand\documentclass or \noexpand\documentstyle commands}{The document may only declare one class.}"));
185    }
186    assign_value(crate::latex_constructs::CLASS_DECLARED, true, Some(Scope::Global));
187
188    let pool = if class == "amsppt" { "AmSTeX" } else { "LaTeX" };
189    input_definitions(pool, InputDefinitionOptions {
190      extension: Some(Cow::Borrowed("pool")),
191      ..InputDefinitionOptions::default()
192    })?;
193
194    assign_value("2.09_COMPATIBILITY", true, Some(Scope::Global));
195
196    // Perl TeX.pool.ltxml:60-65 — when the class triggers the AmSTeX pool
197    // (only `amsppt` today), Perl LoadPool's AmSTeX and *re-emits*
198    // `\documentstyle{class}` so the AmSTeX-pool-defined `\documentstyle`
199    // (amstex.rs L40) takes over. Critically, Perl never loads the LaTeX
200    // pool (`latex_constructs.pool.ltxml`) on the amsppt path, so its
201    // `Let('\magnification', '\@undefined')` (L56) never fires and plain
202    // TeX's `\magnification = \magstep N` keeps working. The Rust
203    // implementation previously fell through to the article.cls +
204    // RequirePackage chain unconditionally, which loads latex.ltx and
205    // re-binds `\magnification` to `\@undefined` — an amsppt-only
206    // regression. Mirror Perl: for amsppt, just RequirePackage(amsppt,
207    // as_class=true) and return. amstex.rs's `\documentstyle` does the
208    // same thing in the same way.
209    if pool == "AmSTeX" {
210      let _ = require_package(&class, RequireOptions {
211        notex: Some(true),
212        as_class: true,
213        ..RequireOptions::default()
214      });
215      return Ok(Tokens!());
216    }
217
218    // latex.ltx:18623 `\documentstyle` inputs latex209.def before the class (`\lx@latex@ix@definitions`), and the
219    // class's end runs latex209.def's `\@documentclasshook` (latex.ltx:18721), ahead of `\compat@loadpackages`.
220    // Once per document: after a first declaration's OmniBus fallback (above), applying it again would save the
221    // settled `\normalsize` as `\original@normalsize` and make the two call each other (astro-ph9812142). The second
222    // class's load is again under the 2e-command traps' `\@gobble`, until its own `\@documentclasshook`.
223    if lookup_bool(LATEX_IX_DEFINITIONS_APPLIED) {
224      digest(TokenizeInternal!(r"\let\@latex@e@error\@gobble"))?;
225    } else {
226      assign_value(LATEX_IX_DEFINITIONS_APPLIED, true, Some(Scope::Global));
227      digest(Tokens!(T_CS!("\\lx@latex@ix@definitions")))?;
228    }
229
230    // Perl L132-135 `compatDefinitions` — pre-bind LaTeX 2.09 dimensions.
231    // Perl helper redefines `\@maxsep` and `\@dblmaxsep`; if these come
232    // from another file and are already defined, redef is harmless.
233    let zero_dim = Stored::Number(Number::new(0));
234    assign_value("\\@maxsep", zero_dim.clone(), Some(Scope::Global));
235    assign_value("\\@dblmaxsep", zero_dim, Some(Scope::Global));
236
237    // Now that LaTeX pool is loaded, undefine `\magnification` so babel
238    // detects "running under LaTeX" correctly. The latex_constructs
239    // pool-load no longer does this unconditionally; deferring to
240    // \documentclass/\documentstyle keeps plain TeX papers (whose first
241    // line is often `\magnification=\magstep1`) functional under the
242    // cortex_worker's eager LaTeX.pool preload. See sibling change in
243    // latex_constructs.rs `\documentclass` after_digest.
244    if let Some(undef_meaning) = lookup_meaning(&T_CS!("\\@undefined")) {
245      assign_meaning(&T_CS!("\\magnification"), undef_meaning, Some(Scope::Global));
246    }
247
248    // Comma-list to Vec<String>. Whitespace-strip per Perl
249    // TrimmedCommaList. Empty entries dropped.
250    let opts_vec: Vec<String> = options_opt.as_ref()
251      .map(|t| t.to_string())
252      .unwrap_or_default()
253      .split(',')
254      .map(|s| s.trim().to_string())
255      .filter(|s| !s.is_empty())
256      .collect();
257
258    // Perl uses `notex => !LookupValue('INCLUDE_CLASSES')`, but its FindFile prefers a `.ltxml` binding whatever
259    // `notex` says (Package.pm:2126-2129; only `noltxml` gates it). These probes ask for bindings, which Rust's
260    // `find_file` consults only under `notex` — so they always set it: with raw classes on (INCLUDE_CLASSES set, the
261    // arXiv profile since 62k) they missed `mn.cls`'s binding and fell to OmniBus (`\ifoldfss` undefined,
262    // `{keywords}` lost; astro-ph0008081, astro-ph0105519). A paper-local raw `.sty` is probed below, only when no
263    // binding answers, so a shipped `mn.sty` yields to the `mn` binding (astro-ph0602372).
264    let notex = true;
265    // Probe `.sty` then `.cls`, AND fall back to version-stripping
266    // (`find_file_fallback`) so e.g. `\documentstyle{aipproc2}` resolves
267    // to aipproc.sty.ltxml — matching Perl's `FindFile` which consults
268    // the binding-name registry plus the `\d+`-suffix fallback. Without
269    // this, versioned class names go to OmniBus → aipproc.sty.ltxml is
270    // never loaded → `\epsfsize` etc. that aipproc.sty's
271    // `RequirePackage('psfig'→'epsfig')` chain provides stay undefined.
272    // Check `<class>.sty` via binding registry (notex), version-strip
273    // fallback, AND disk-only probe (`forbid_ltxml: true`) so paper-local
274    // `<class>.sty` files (e.g. `jinstpub.sty`, with no .cls counterpart)
275    // can be loaded as a class. Disk probe is gated on the absence of
276    // ANY `<class>.cls` binding (exact OR via version-strip fallback) so
277    // that `\documentstyle{mn1}` falls back to `mn.cls.ltxml` (Perl-faithful
278    // priority) instead of the local `mn1.sty` whose raw load is suppressed
279    // by the default `INCLUDE_STYLES=false` gate. Witness: astro-ph0002213
280    // (\documentstyle[epsfig]{mn1} + local mn1.sty + \begin{keywords}/\psfig).
281    let class_sty_binding = find_file(
282      &format!("{}.sty", class),
283      Some(FindFileOptions { notex, ..Default::default() }),
284    ).is_some();
285    // Pre-flight existence checks must NOT eagerly load the fallback
286    // binding (its body would run \LoadClass / \RequirePackage with no
287    // option-pass-through, contaminating the actual `\documentstyle`
288    // load that happens further below). Fix for the `\psfig` cluster
289    // (12 papers) — astro-ph0002213 R=1→0 with this gate change.
290    let class_sty_fallback = find_file_fallback_exists(&class, "sty");
291    let class_cls_binding_exact = find_file(
292      &format!("{}.cls", class),
293      Some(FindFileOptions { notex, ..Default::default() }),
294    ).is_some();
295    let class_cls_via_fallback = find_file_fallback_exists(&class, "cls");
296    let class_sty_via_disk = !class_sty_binding
297      && !class_sty_fallback
298      && !class_cls_binding_exact
299      && !class_cls_via_fallback
300      && find_file(
301        &class,
302        Some(FindFileOptions { ext_type: Some(Cow::Borrowed("sty")), forbid_ltxml: true, ..Default::default() }),
303      ).is_some();
304    // With INCLUDE_CLASSES set (`rawclasses`, the arXiv profile's since 67v, or `localrawclasses`), a raw `<class>.cls`
305    // no binding answers is the class (Perl latex_constructs.pool.ltxml:115-118 `FindFile(type=>'cls', notex=>!INCLUDE_CLASSES)`
306    // takes the binding first, then the file), searched locally only under `searchpaths`.
307    let class_cls_raw = !class_cls_binding_exact
308      && !class_cls_via_fallback
309      && lookup_bool("INCLUDE_CLASSES")
310      && find_file(
311        &class,
312        Some(FindFileOptions {
313          ext_type: Some(Cow::Borrowed("cls")),
314          forbid_ltxml: true,
315          search_paths_only: lookup_string("INCLUDE_CLASSES") == "searchpaths",
316          ..Default::default()
317        }),
318      ).is_some();
319    let class_sty_found = class_sty_binding || class_sty_via_disk || class_sty_fallback;
320    let class_cls_found = !class_sty_found && (class_cls_binding_exact
321      || class_cls_via_fallback || class_cls_raw);
322
323    // (OXIDIZED_DESIGN #179: the "has chapters" answer is stored where the class's load ends, before the style options)
324    crate::latex_constructs::store_chapter_after_latex_ix_class_load(true);
325    let after = Tokens!(
326      T_CS!("\\@documentclasshook"),
327      T_CS!("\\lx@latex@ix@store@chapters"),
328      T_CS!("\\compat@loadpackages")
329    );
330
331    if class_sty_found {
332      // Branch 1 — class is actually a `.sty` (e.g. spackap, aipproc,
333      // kluwer): load article.cls under it, then RequirePackage(class).
334      // Pin extension to "sty" so the version-stripping fallback (e.g.
335      // aipproc2 → aipproc) prefers the binding registry's `.sty` entry
336      // instead of also probing for a same-name `.cls` (which has a
337      // different DefMacro signature for `\author` etc. that conflicts
338      // with the `.sty` body — root cause for nucl-th0010030).
339      // Choose article.cls vs OmniBus.cls as the underlying class. When
340      // the `.sty` was found via binding registry or version-strip
341      // fallback, the user's intent is "load this binding as the
342      // document class" — article.cls is the right base. When the
343      // `.sty` was found ONLY via paper-local disk-probe (no binding /
344      // no fallback), the file is an arbitrary user style that doesn't
345      // know about LaTeXML's frontmatter conventions; OmniBus is the
346      // right base because it provides broad fallback coverage
347      // (`\citeauthoryear` autoload trigger, generic `\affil`, etc.).
348      // Mirrors Perl's flow for paper-local-only `\documentstyle`
349      // classes (verified via `--verbose` on astro-ph0510540 and
350      // astro-ph0008100 — Perl loads OmniBus for both).
351      let underlying = if class_sty_via_disk { "OmniBus" } else { "article" };
352      input_definitions(underlying, InputDefinitionOptions {
353        extension: Some(Cow::Borrowed("cls")),
354        options: opts_vec.clone(),
355        handleoptions: true,
356        noerror: true,
357        ..InputDefinitionOptions::default()
358      })?;
359      // When the .sty was found ONLY via paper-local disk-probe (no
360      // binding, no fallback), pass `notex=false` to allow the raw
361      // load. Otherwise the default `INCLUDE_STYLES=false` gate inside
362      // require_package would force `notex=true` and suppress the
363      // load. Mirrors the `\compat@loadpackages` fix from commit
364      // bb4cf2e17. Witness: astro-ph0008100
365      // (\documentstyle[PASJadd]{PASJ95} + uppercase-named PASJ95.STY).
366      let notex_for_require = if class_sty_via_disk { Some(false) } else { None };
367      require_package(&class, RequireOptions {
368        options: opts_vec,
369        extension: Some(Cow::Borrowed("sty")),
370        notex: notex_for_require,
371        after,
372        ..RequireOptions::default()
373      })?;
374    } else if class_cls_found {
375      // Branch 2 — `<class>.cls` exists: load it as the document class.
376      crate::latex_constructs::note_document_class(&class);
377      load_class(&class, opts_vec, after)?;
378      crate::latex_constructs::make_missing_float_counters()?;
379    } else {
380      // Branch 3 — neither sty nor cls found. Load OmniBus to provide the
381      // wide AAS/elsevier/etc. coverage, then attempt the user-named class
382      // as a package (will likely no-op via missing_file warn).
383      input_definitions("OmniBus", InputDefinitionOptions {
384        extension: Some(Cow::Borrowed("cls")),
385        options: opts_vec.clone(),
386        handleoptions: true,
387        noerror: true,
388        after,
389        ..InputDefinitionOptions::default()
390      })?;
391      require_package(&class, RequireOptions {
392        options: opts_vec,
393        as_class: true,
394        ..RequireOptions::default()
395      })?;
396      // No binding and no file answered: a later `\documentstyle` may still declare the class (above).
397      assign_value(crate::latex_constructs::CLASS_DECLARED, false, Some(Scope::Global));
398    }
399
400    Ok(Tokens!())
401  });

403  // What latex209.def (TL 2025 v0.55, read whole) defines that the native `\documentstyle` path lacks and a 2.09
404  // class or document uses. The size switch first: `\normalsize` is the class's `\@normalsize` (:138-139), which a
405  // 2.09 class defines (`\def\@normalsize{\@setsize\normalsize{12pt}\xpt\@xpt}` then `\normalsize`; witnesses
406  // hep-ex0107077 dp_delphi.cls, 0802.1120 klu.cls, astro-ph0009372 cupconf.cls: "The font size command \normalsize is
407  // not defined"), and at the class's end `\@documentclasshook` (:124-137) settles the pair: `\normalsize` the
408  // class's `\@normalsize`, else `\@normalsize` the class's `\normalsize`, else the kernel's placeholder again.
409  // `\@normalsize` is undefined until then, as in latex.ltx (only its own `\@documentclasshook`, :18348, gives it a
410  // meaning; latex_constructs/sect13.rs's default stands for that hook), so a class defining neither meets
411  // "undefined", not a `\normalsize`/`\@normalsize` loop. Then the 2.09 registers and macros a class reads
412  // (:234-236 `\@cla`, `\@clb`, `\mscount`; :213 `\@lquote`; :229-233 `\SLiTeX`; :342 `\load`; :336-338 the old
413  // font scaling words; :356-367 the old size commands `\vpt`…`\xxvpt`) and latexsym (:307). The LaTeX2e-command
414  // traps come first, as in the file (:79-123): `\usepackage`, `\listfiles`, `\lrbox` and `\@xargdef` call
415  // `\@latex@e@error` before their own meaning, a `\@gobble` until the class's end, where `\@documentclasshook`
416  // makes it latex209.def's "LaTeX2e command … in LaTeX 2.09 document" error (:137) — so a `\usepackage` after
417  // `\documentstyle` errs as in pdflatex and still loads the package (user ruling 2026-10-17; no paper of the 250 `\documentstyle` papers in the census sample
418  // uses it — 0806.0904 redefines `\usepackage` itself); and
419  // the `\@ifdefinable` wrapper lets a 2.09 document define the 2e-only names it lists (`\th`, `\SS`, the guillemets).
420  // The `\ensuremath` trap (:88) is held: our bindings call `\ensuremath` where their 2.09-era originals do not (AASTeX 4
421  // aaspp4.sty `\def\fd{\hbox{$.\!\!^{\rm d}$}}` against aas_support_sty.rs `\ensuremath{\@fd}`; REVTeX 3,
422  // emulateapj), so the trap would fire on binding code pdflatex never runs (astro-ph9606143, astro-ph9701025,
423  // cond-mat9906271, astro-ph9809336, astro-ph9806372) — it lands once binding internals use a kernel-internal
424  // ensure-math (SYNC row). OXIDIZED_DESIGN_DIVERGENCES #725.
425  // Not applied, the native engine standing for them: the missing-file and obsolete-file fallbacks (:140-168, the
