15// Process-once cached env var (see WISDOM #56 — getenv hot-path race).
16static SOURCE_DATE_EPOCH: LazyLock<Option<String>> =
17  LazyLock::new(|| std::env::var("SOURCE_DATE_EPOCH").ok());

19/// `SOURCE_DATE_EPOCH` (Unix seconds), when set and numeric: the reproducible-build clock Perl reads in
20/// TeX_Job.pool.ltxml:42.
21fn source_date_epoch_env() -> Option<i64> {
22  SOURCE_DATE_EPOCH
23    .as_deref()
24    .and_then(|epoch| epoch.trim().parse::<i64>().ok())
25}

27/// Sets the job's clock — `\year`, `\month`, `\day`, `\time` (globally) — to the instant `epoch` (Unix seconds) on the
28/// UTC clock, as Perl TeX_Job.pool.ltxml:42-46 does for `SOURCE_DATE_EPOCH` (`gmtime`). `\today` and
29/// `\pdfcreationdate` read these registers. Returns whether `epoch` is a representable instant (the clock is left
30/// untouched when it is not).
31pub fn assign_date_registers(epoch: i64) -> bool {
32  match DateTime::from_timestamp(epoch, 0) {
33    Some(utc) => {
34      assign_clock_registers(&utc);
35      true
36    },
37    None => false,
38  }
39}
40
41fn assign_clock_registers<Tz: TimeZone>(dt: &DateTime<Tz>) {
42  AssignValue!("\\day", Number!(dt.day()), Scope::Global);
43  AssignValue!("\\month", Number!(dt.month()), Scope::Global);
44  AssignValue!("\\year", Number!(dt.year()), Scope::Global);
45  AssignValue!(
46    "\\time",
47    Number!(60 * dt.hour() + dt.minute()),
48    Scope::Global
49  );
50}

121  //======================================================================
122  // LaTeX 2.09 compatibility
123  //----------------------------------------------------------------------
124  // Perl latex_constructs.pool.ltxml:97-129 (`\documentstyle` afterDigest).
125  // LaTeX 2.09 compat shim. Three branches mirroring Perl strictly:
126  //
127  //   1. <class>.sty exists  → input_definitions("article", cls, handleoptions=true, options=opts)
128  //                          + require_package(class, as_class=true, after=\compat@loadpackages)
129  //   2. <class>.cls exists  → load_class(class, opts, after=\compat@loadpackages)
130  //   3. neither             → input_definitions("OmniBus", cls, handleoptions=true, options=opts,
131  //      after=\compat@loadpackages)
132  //                          + require_package(class, as_class=true)
133  //
134  // Critical Perl semantics (matching latex_constructs.pool.ltxml:97-129):
135  //   * `handleoptions => 1` makes the cls's `\DeclareOption`/`\ProcessOptions` consume the
136  //     `<opts>` and route leftovers onto `@unusedoptionlist`.
137  //   * `after => \compat@loadpackages` runs *after* the cls finishes its option-processing — at
138  //     that point unmet options sit on the unused list and `\compat@loadpackages`
139  //     (`latex_constructs.rs:2502`) walks them, RequirePackage's any that resolve, and triggers
140  //     OmniBus when anything went unmet. That is what lets `\documentstyle[paspconf] {article}`
141  //     transitively load `aas_macros.sty.ltxml` to define `\affil` / `\altaffilmark` /
142  //     `\acknowledgments` etc.
143  // The latex_dump unconditionally redefines `\documentstyle` with the
144  // kernel-style `\input{latex209.def}\documentclass` form (Perl
145  // latex_dump.pool.ltxml entry for `\documentstyle`). In Perl that
146  // version is itself overridden by latex_constructs.pool.ltxml's
147  // DefConstructor; in our Rust port `latex_constructs.rs` doesn't
148  // redefine `\documentstyle`, so without intervention any autoload-of-
149  // LaTeX-pool path (e.g. `\newcommand` before `\documentstyle`)
150  // replaces our impl with the dump's, breaking the
151  // `\compat@loadpackages` after-hook that dispatches `[epsf]` etc.
152  // from `@unusedoptionlist`. Witness: hep-th9912229
153  // (`\newcommand` before `\documentstyle[12pt,epsf]`).
154  //
155  // Workaround: register the impl under a stable backup name
156  // `\lx@documentstyle@impl` and `\let \documentstyle = \lx@documentstyle@impl`
157  // at the end of `\@load@latex@pool` so we restore our impl after
158  // every LaTeX pool load.
159  DefMacro!("\\lx@documentstyle@impl[]{}", sub[(options_opt, class_tks)] {
160    use latexml_core::binding::content::{
161      find_file, find_file_fallback_exists, FindFileOptions, load_class,
162    };
163    // Perl latex_constructs.pool.ltxml:79 — `$class =~ s/\s+//g;`. Strip ALL
164    // whitespace (not just leading/trailing as a bare `.trim()` would), so a
165    // multi-line `\documentstyle[...]{<newline>article}` — whose post-`{`
166    // newline becomes an embedded space token in the Semiverbatim arg — still
167    // matches its binding registration. Mirrors the sibling `\documentclass`
168    // fix (latex_constructs.rs); a whitespace-bearing name otherwise misses
169    // the binding and falls through to OmniBus.
170    let class: String = class_tks
171      .to_string()
172      .chars()
173      .filter(|c| !c.is_ascii_whitespace())
174      .collect();
175
176    // A second class declaration loads no class: latex.ltx's `\documentclass` has made itself `\@twoclasseserror`
177    // (:18617-18618, :18944-18947), which `\documentstyle`'s calls too (:18622-18624), "Two \documentclass or
178    // \documentstyle commands". A first declaration no binding and no file answered (the OmniBus fallback, branch 3
179    // below; `sect01.rs` `\documentclass`) does not count (main-session ruling 2026-10-19): astro-ph9812142's line 1
180    // `#\documentstyle[psfig,epsf]{/disk/3/roud/home/Texstuff/aa}` names a path no tree has (the `#` is no comment:
181    // pdflatex's "You can't use macro parameter character #", then the missing class stops it), and its line 2
182    // `\documentstyle[psfig,epsf]{aa}` loads the real class (aa's `\thesaurus`, `\offprints`, bibitem tags).
183    if lookup_bool(crate::latex_constructs::CLASS_DECLARED) {
184      return Ok(TokenizeInternal!(
185        r"\@latex@error{Two \noexpand\documentclass or \noexpand\documentstyle commands}{The document may only declare one class.}"));
186    }
187    assign_value(crate::latex_constructs::CLASS_DECLARED, true, Some(Scope::Global));
188
189    let pool = if class == "amsppt" { "AmSTeX" } else { "LaTeX" };
190    input_definitions(pool, InputDefinitionOptions {
191      extension: Some(Cow::Borrowed("pool")),
192      ..InputDefinitionOptions::default()
193    })?;
194
195    assign_value("2.09_COMPATIBILITY", true, Some(Scope::Global));
196
197    // Perl TeX.pool.ltxml:60-65 — when the class triggers the AmSTeX pool
198    // (only `amsppt` today), Perl LoadPool's AmSTeX and *re-emits*
199    // `\documentstyle{class}` so the AmSTeX-pool-defined `\documentstyle`
200    // (amstex.rs L40) takes over. Critically, Perl never loads the LaTeX
201    // pool (`latex_constructs.pool.ltxml`) on the amsppt path, so its
202    // `Let('\magnification', '\@undefined')` (L56) never fires and plain
203    // TeX's `\magnification = \magstep N` keeps working. The Rust
204    // implementation previously fell through to the article.cls +
205    // RequirePackage chain unconditionally, which loads latex.ltx and
206    // re-binds `\magnification` to `\@undefined` — an amsppt-only
207    // regression. Mirror Perl: for amsppt, just RequirePackage(amsppt,
208    // as_class=true) and return. amstex.rs's `\documentstyle` does the
209    // same thing in the same way.
210    if pool == "AmSTeX" {
211      let _ = require_package(&class, RequireOptions {
212        notex: Some(true),
213        as_class: true,
214        ..RequireOptions::default()
215      });
216      return Ok(Tokens!());
217    }
218
219    // latex.ltx:18623 `\documentstyle` inputs latex209.def before the class (`\lx@latex@ix@definitions`), and the
220    // class's end runs latex209.def's `\@documentclasshook` (latex.ltx:18721), ahead of `\compat@loadpackages`.
221    // Once per document: after a first declaration's OmniBus fallback (above), applying it again would save the
222    // settled `\normalsize` as `\original@normalsize` and make the two call each other (astro-ph9812142). The second
223    // class's load is again under the 2e-command traps' `\@gobble`, until its own `\@documentclasshook`.
224    if lookup_bool(LATEX_IX_DEFINITIONS_APPLIED) {
225      digest(TokenizeInternal!(r"\let\@latex@e@error\@gobble"))?;
226    } else {
227      assign_value(LATEX_IX_DEFINITIONS_APPLIED, true, Some(Scope::Global));
228      digest(Tokens!(T_CS!("\\lx@latex@ix@definitions")))?;
229    }
230
231    // Perl L132-135 `compatDefinitions` — pre-bind LaTeX 2.09 dimensions.
232    // Perl helper redefines `\@maxsep` and `\@dblmaxsep`; if these come
233    // from another file and are already defined, redef is harmless.
234    let zero_dim = Stored::Number(Number::new(0));
235    assign_value("\\@maxsep", zero_dim.clone(), Some(Scope::Global));
236    assign_value("\\@dblmaxsep", zero_dim, Some(Scope::Global));
237
238    // Now that LaTeX pool is loaded, undefine `\magnification` so babel
239    // detects "running under LaTeX" correctly. The latex_constructs
240    // pool-load no longer does this unconditionally; deferring to
241    // \documentclass/\documentstyle keeps plain TeX papers (whose first
242    // line is often `\magnification=\magstep1`) functional under the
243    // cortex_worker's eager LaTeX.pool preload. See sibling change in
244    // latex_constructs.rs `\documentclass` after_digest.
245    if let Some(undef_meaning) = lookup_meaning(&T_CS!("\\@undefined")) {
246      assign_meaning(&T_CS!("\\magnification"), undef_meaning, Some(Scope::Global));
247    }
248
249    // Comma-list to Vec<String>. Whitespace-strip per Perl
250    // TrimmedCommaList. Empty entries dropped.
251    let opts_vec: Vec<String> = options_opt.as_ref()
252      .map(|t| t.to_string())
253      .unwrap_or_default()
254      .split(',')
255      .map(|s| s.trim().to_string())
256      .filter(|s| !s.is_empty())
257      .collect();
258
259    // Perl uses `notex => !LookupValue('INCLUDE_CLASSES')`, but its FindFile prefers a `.ltxml` binding whatever
260    // `notex` says (Package.pm:2126-2129; only `noltxml` gates it). These probes ask for bindings, which Rust's
261    // `find_file` consults only under `notex` — so they always set it: with raw classes on (INCLUDE_CLASSES set, the
262    // arXiv profile since 62k) they missed `mn.cls`'s binding and fell to OmniBus (`\ifoldfss` undefined,
263    // `{keywords}` lost; astro-ph0008081, astro-ph0105519). A paper-local raw `.sty` is probed below, only when no
264    // binding answers, so a shipped `mn.sty` yields to the `mn` binding (astro-ph0602372).
265    let notex = true;
266    // Probe `.sty` then `.cls`, AND fall back to version-stripping
267    // (`find_file_fallback`) so e.g. `\documentstyle{aipproc2}` resolves
268    // to aipproc.sty.ltxml — matching Perl's `FindFile` which consults
269    // the binding-name registry plus the `\d+`-suffix fallback. Without
270    // this, versioned class names go to OmniBus → aipproc.sty.ltxml is
271    // never loaded → `\epsfsize` etc. that aipproc.sty's
272    // `RequirePackage('psfig'→'epsfig')` chain provides stay undefined.
273    // Check `<class>.sty` via binding registry (notex), version-strip
274    // fallback, AND disk-only probe (`forbid_ltxml: true`) so paper-local
275    // `<class>.sty` files (e.g. `jinstpub.sty`, with no .cls counterpart)
276    // can be loaded as a class. Disk probe is gated on the absence of
277    // ANY `<class>.cls` binding (exact OR via version-strip fallback) so
278    // that `\documentstyle{mn1}` falls back to `mn.cls.ltxml` (Perl-faithful
279    // priority) instead of the local `mn1.sty` whose raw load is suppressed
280    // by the default `INCLUDE_STYLES=false` gate. Witness: astro-ph0002213
281    // (\documentstyle[epsfig]{mn1} + local mn1.sty + \begin{keywords}/\psfig).
282    let class_sty_binding = find_file(
283      &format!("{}.sty", class),
284      Some(FindFileOptions { notex, ..Default::default() }),
285    ).is_some();
286    // Pre-flight existence checks must NOT eagerly load the fallback
287    // binding (its body would run \LoadClass / \RequirePackage with no
288    // option-pass-through, contaminating the actual `\documentstyle`
289    // load that happens further below). Fix for the `\psfig` cluster
290    // (12 papers) — astro-ph0002213 R=1→0 with this gate change.
291    let class_sty_fallback = find_file_fallback_exists(&class, "sty");
292    let class_cls_binding_exact = find_file(
293      &format!("{}.cls", class),
294      Some(FindFileOptions { notex, ..Default::default() }),
295    ).is_some();
296    let class_cls_via_fallback = find_file_fallback_exists(&class, "cls");
297    let class_sty_via_disk = !class_sty_binding
298      && !class_sty_fallback
299      && !class_cls_binding_exact
300      && !class_cls_via_fallback
301      && find_file(
302        &class,
303        Some(FindFileOptions { ext_type: Some(Cow::Borrowed("sty")), forbid_ltxml: true, ..Default::default() }),
304      ).is_some();
305    // With INCLUDE_CLASSES set (`rawclasses`, the arXiv profile's since 67v, or `localrawclasses`), a raw `<class>.cls`
306    // no binding answers is the class (Perl latex_constructs.pool.ltxml:115-118 `FindFile(type=>'cls', notex=>!INCLUDE_CLASSES)`
307    // takes the binding first, then the file), searched locally only under `searchpaths`.
308    let class_cls_raw = !class_cls_binding_exact
309      && !class_cls_via_fallback
310      && lookup_bool("INCLUDE_CLASSES")
311      && find_file(
312        &class,
313        Some(FindFileOptions {
314          ext_type: Some(Cow::Borrowed("cls")),
315          forbid_ltxml: true,
316          search_paths_only: lookup_string("INCLUDE_CLASSES") == "searchpaths",
317          ..Default::default()
318        }),
319      ).is_some();
320    let class_sty_found = class_sty_binding || class_sty_via_disk || class_sty_fallback;
321    let class_cls_found = !class_sty_found && (class_cls_binding_exact
322      || class_cls_via_fallback || class_cls_raw);
323
324    // (OXIDIZED_DESIGN #179: the "has chapters" answer is stored where the class's load ends, before the style options)
325    crate::latex_constructs::store_chapter_after_latex_ix_class_load(true);
326    let after = Tokens!(
327      T_CS!("\\@documentclasshook"),
328      T_CS!("\\lx@latex@ix@store@chapters"),
329      T_CS!("\\compat@loadpackages")
330    );
331
332    if class_sty_found {
333      // Branch 1 — class is actually a `.sty` (e.g. spackap, aipproc,
334      // kluwer): load article.cls under it, then RequirePackage(class).
335      // Pin extension to "sty" so the version-stripping fallback (e.g.
336      // aipproc2 → aipproc) prefers the binding registry's `.sty` entry
337      // instead of also probing for a same-name `.cls` (which has a
338      // different DefMacro signature for `\author` etc. that conflicts
339      // with the `.sty` body — root cause for nucl-th0010030).
340      // Choose article.cls vs OmniBus.cls as the underlying class. When
341      // the `.sty` was found via binding registry or version-strip
342      // fallback, the user's intent is "load this binding as the
343      // document class" — article.cls is the right base. When the
344      // `.sty` was found ONLY via paper-local disk-probe (no binding /
345      // no fallback), the file is an arbitrary user style that doesn't
346      // know about LaTeXML's frontmatter conventions; OmniBus is the
347      // right base because it provides broad fallback coverage
348      // (`\citeauthoryear` autoload trigger, generic `\affil`, etc.).
349      // Mirrors Perl's flow for paper-local-only `\documentstyle`
350      // classes (verified via `--verbose` on astro-ph0510540 and
351      // astro-ph0008100 — Perl loads OmniBus for both).
352      let underlying = if class_sty_via_disk { "OmniBus" } else { "article" };
353      input_definitions(underlying, InputDefinitionOptions {
354        extension: Some(Cow::Borrowed("cls")),
355        options: opts_vec.clone(),
356        handleoptions: true,
357        noerror: true,
358        ..InputDefinitionOptions::default()
359      })?;
360      // When the .sty was found ONLY via paper-local disk-probe (no
361      // binding, no fallback), pass `notex=false` to allow the raw
362      // load. Otherwise the default `INCLUDE_STYLES=false` gate inside
363      // require_package would force `notex=true` and suppress the
364      // load. Mirrors the `\compat@loadpackages` fix from commit
365      // bb4cf2e17. Witness: astro-ph0008100
366      // (\documentstyle[PASJadd]{PASJ95} + uppercase-named PASJ95.STY).
367      let notex_for_require = if class_sty_via_disk { Some(false) } else { None };
368      require_package(&class, RequireOptions {
369        options: opts_vec,
370        extension: Some(Cow::Borrowed("sty")),
371        notex: notex_for_require,
372        after,
373        ..RequireOptions::default()
374      })?;
375    } else if class_cls_found {
376      // Branch 2 — `<class>.cls` exists: load it as the document class.
377      crate::latex_constructs::note_document_class(&class);
378      load_class(&class, opts_vec, after)?;
379      crate::latex_constructs::make_missing_float_counters()?;
380    } else {
381      // Branch 3 — neither sty nor cls found. Load OmniBus to provide the
382      // wide AAS/elsevier/etc. coverage, then attempt the user-named class
383      // as a package (will likely no-op via missing_file warn).
384      input_definitions("OmniBus", InputDefinitionOptions {
385        extension: Some(Cow::Borrowed("cls")),
386        options: opts_vec.clone(),
387        handleoptions: true,
388        noerror: true,
389        after,
390        ..InputDefinitionOptions::default()
391      })?;
392      require_package(&class, RequireOptions {
393        options: opts_vec,
394        as_class: true,
395        ..RequireOptions::default()
396      })?;
397      // No binding and no file answered: a later `\documentstyle` may still declare the class (above).
398      assign_value(crate::latex_constructs::CLASS_DECLARED, false, Some(Scope::Global));
399    }
400
401    Ok(Tokens!())
402  });

404  // What latex209.def (TL 2025 v0.55, read whole) defines that the native `\documentstyle` path lacks and a 2.09
405  // class or document uses. The size switch first: `\normalsize` is the class's `\@normalsize` (:138-139), which a
406  // 2.09 class defines (`\def\@normalsize{\@setsize\normalsize{12pt}\xpt\@xpt}` then `\normalsize`; witnesses
407  // hep-ex0107077 dp_delphi.cls, 0802.1120 klu.cls, astro-ph0009372 cupconf.cls: "The font size command \normalsize is
408  // not defined"), and at the class's end `\@documentclasshook` (:124-137) settles the pair: `\normalsize` the
409  // class's `\@normalsize`, else `\@normalsize` the class's `\normalsize`, else the kernel's placeholder again.
410  // `\@normalsize` is undefined until then, as in latex.ltx (only its own `\@documentclasshook`, :18348, gives it a
411  // meaning; latex_constructs/sect13.rs's default stands for that hook), so a class defining neither meets
412  // "undefined", not a `\normalsize`/`\@normalsize` loop. Then the 2.09 registers and macros a class reads
413  // (:234-236 `\@cla`, `\@clb`, `\mscount`; :213 `\@lquote`; :229-233 `\SLiTeX`; :342 `\load`; :336-338 the old
414  // font scaling words; :356-367 the old size commands `\vpt`…`\xxvpt`) and latexsym (:307). The LaTeX2e-command
415  // traps come first, as in the file (:79-123): `\usepackage`, `\listfiles`, `\lrbox` and `\@xargdef` call
416  // `\@latex@e@error` before their own meaning, a `\@gobble` until the class's end, where `\@documentclasshook`
417  // makes it latex209.def's "LaTeX2e command … in LaTeX 2.09 document" error (:137) — so a `\usepackage` after
418  // `\documentstyle` errs as in pdflatex and still loads the package (user ruling 2026-10-17; no paper of the 250 `\documentstyle` papers in the census sample
419  // uses it — 0806.0904 redefines `\usepackage` itself); and
420  // the `\@ifdefinable` wrapper lets a 2.09 document define the 2e-only names it lists (`\th`, `\SS`, the guillemets).
421  // The `\ensuremath` trap (:88) is held: our bindings call `\ensuremath` where their 2.09-era originals do not (AASTeX 4
422  // aaspp4.sty `\def\fd{\hbox{$.\!\!^{\rm d}$}}` against aas_support_sty.rs `\ensuremath{\@fd}`; REVTeX 3,
423  // emulateapj), so the trap would fire on binding code pdflatex never runs (astro-ph9606143, astro-ph9701025,
424  // cond-mat9906271, astro-ph9809336, astro-ph9806372) — it lands once binding internals use a kernel-internal
425  // ensure-math (SYNC row). OXIDIZED_DESIGN_DIVERGENCES #725.
426  // Not applied, the native engine standing for them: the missing-file and obsolete-file fallbacks (:140-168, the
