1//! latex_constructs_rust_only — Rust-side hotfix overrides for LaTeX-format CSes.
2//!
3//! Holds bindings present in the Rust port but **not** in any of Perl's three
4//! `latex_{base,bootstrap,constructs}.pool.ltxml` files. Anything that lives
5//! here is a hotfix tracked separately so that the corresponding Rust
6//! "engine/latex_*.rs" siblings stay byte-for-byte parity with the Perl
7//! source.
8//!
9//! Loaded LAST in `latex.rs`'s `LoadFormat('latex')` chain, after
10//! `latex_constructs`, so every entry can rely on:
11//! * The dump (or `latex_base.rs` under NODUMP) having installed raw LaTeX-kernel CSes.
12//! * `latex_constructs.rs` having registered its own definitions (which some entries here `Let!`
13//!   against — e.g. `\IfPackageLoadedTF ↦ \@ifpackageloaded`).
14//!
15//! Categories (in source order below):
16//! 1. Modern LaTeX kernel CSes added post-2020 (the `\If…AtLeast/LoadedTF` family). LaTeX2e re-Lets
17//!    these from the kernel; the Perl source predates them, so they need an explicit override here.
18//! 2. LaTeXML-internal helpers that the engine code expects to exist (`\ltx@hard@MessageBreak`,
19//!    `\ltx@ifclassloaded`, `\ltx@ifpackageloaded`).
20//! 3. List internals not in Perl source (`\@bls`, `\@listi`-`\@listvi`, `\@maxlistdepth`). The dump
21//!    captures the values raw `latex.ltx` installs; these are defensive overrides for the NODUMP
22//!    path.
23//! 4. Misc Rust-side stubs (`\@latexbug`, `\maybe@end@title`, `\thebibliography@ID` empty default).
24//!
25//! The file ends with a **retraction record**: the MathSciNet `\cprime` transliteration family was
26//! briefly always-on here and was removed 2026-07-27. The comment is kept because it carries the
27//! witnesses and the reasoning — do not re-add the family.

935  // 8. {filecontents}/{filecontents*} environments — Rust impl
936  //
937  // Perl uses Semiverbatim DefConstructor for begin{filecontents}; Rust
938  // implements via DefPrimitive that reads raw lines until end-marker
939  // and caches the content for later \input. Helper fn defined here so
940  // the migration is self-contained.
941  //======================================================================
942  fn cache_filecontents(header_star: bool) -> Result<()> {
943    // latex.ltx:19047 `\edef\E{\@backslashchar end\string{\@currenvir\string}}`:
944    // the capture ends at `\end{<current environment>}`, NOT at a fixed
945    // `\end{filecontents}`. A wrapper environment that runs the bare command
946    // (latexdemo.sty:97-101 `\csname filecontents*\endcsname` inside
947    // `DefineCode`; democodetools, codedescribe, revtex's ltxdocext.sty) ends
948    // its capture at ITS OWN `\end{DefineCode}`; the fixed marker never
949    // matched and the rest of the document was swallowed silently
950    // (latex4wp: 63% of the manual, sweep 75 S3). Perl binds only
951    // `\begin{filecontents*}`, so it errors on the bare command instead.
952    let currenvir = lookup_value("current_environment")
953      .map(|v| v.to_string())
954      .unwrap_or_default();
955    let end_marker = if currenvir.is_empty() {
956      // Bare command outside any environment: latex.ltx would look for
957      // `\end{document}`; keep the historical marker so a hand-written
958      // `\filecontents{f}…\end{filecontents}` at the top level still closes.
959      if header_star {
960        "\\end{filecontents*}".to_string()
961      } else {
962        "\\end{filecontents}".to_string()
963      }
964    } else {
965      format!("\\end{{{currenvir}}}")
966    };
967    skip_spaces()?;
968    // Real LaTeX `\filecontents` `\edef`s the filename argument, so
969    // `\begin{filecontents}{\jobname-acro.tex}` writes — and a later
970    // `\input`/`\loadglsentries` finds — the file under the EXPANDED name
971    // (e.g. `root-acro.tex`). Read with Full expansion (not Off) so `\jobname`
972    // and friends are resolved; otherwise the content is cached under the
973    // literal key `\jobname-acro.tex_contents` and the lookup for
974    // `root-acro.tex_contents` misses, leaving the file "not found" and any
975    // entries it defines (glossaries acronyms, etc.) undefined. Witness
976    // 1905.05350 (`\begin{filecontents}{\jobname-acro.tex}` …
977    // `\loadglsentries[\acronymtype]{\jobname-acro}`).
978    // filecontents 2019+ takes a leading optional (`[overwrite, nosearch,
979    // noheader]{file}` — algxpar-doc L18); without skipping it the CACHE KEY
980    // became "[" and the payload was lost. Absorb it before the filename.
981    let opts = read_optional(None)?;
982    skip_spaces()?;
983    let filename_toks = read_arg(ExpansionLevel::Full)?;
984    let filename = filename_toks.to_string();
985    // latex.ltx `filecontents` options: `noheader` (and the starred form)
986    // write no `%%` header — a data file read back by `\read`/`\ior_get`
987    // (a CSV for `\ior_map_inline`) must not start with comment lines that
988    // read as empty rows. `overwrite`/`force`/`nosearch`/`nowarn` are
989    // no-ops for the cache.
990    // latex.ltx:18956-18967 `\filec@ntents@opt`: the bracket is a comma list
991    // and each key is matched EXACTLY (`\filec@ntents@noheader`), never by
992    // substring.
993    let has_option = |key: &str| {
994      opts
995        .as_ref()
996        .is_some_and(|o| o.to_string().split(',').any(|k| k.trim() == key))
997    };
998    let noheader = header_star || has_option("noheader");
999    // latex.ltx:18990-19004 `\filec@ntents`: unless `force`/`overwrite` (`\@fileswfalse`, :18967-18968), a file
1000    // `\openin` finds — anywhere on the search path, or in the current directory only under `nosearch` (:18970) —
1001    // is kept: "File `name' already exists on the system. Not generating it from this source", the environment's
1002    // lines read and dropped. Writing it shadowed the `name.tex` a paper ships beside the document (KNOWN_PERL_ERRORS
1003    // #279 residual). Guard `perfect_kernel_batch77s::filecontents_keeps_a_file_on_disk`.
1004    let keep_existing = !has_option("force")
1005      && !has_option("overwrite")
1006      && filecontents_target_exists(&filename, has_option("nosearch"));
1007    let mut lines: Vec<String> = Vec::new();
1008    if !noheader {
1009      lines.push(format!("%% LaTeX2e file `{filename}'"));
1010      lines.push("%% generated by the `filecontents' environment".to_string());
1011      lines.push("%% from source `\\jobname' on YYYY/MM/DD.".to_string());
1012      lines.push("%%".to_string());
1013    }
1014    // Discard remainder of \begin{filecontents} line
1015    read_raw_line();
1016    // latex.ltx:19047-19066 `\reserved@b ##1\E##2\E##3\relax`: each line is split at `\end{<currenvir>}` wherever it
1017    // stands; the text before it is written as the file's last line ("Writing text `…' before \end{…} as last line"),
1018    // the text after it ignored ("Ignoring text `…' after \end{…}"). A whole-line match (fancyvrb's rule, the former
1019    // `capture_raw_lines_until`) never saw `\end{tikzpicture}\end{filecontents*}` or `\end{filecontents}{b.tex}`, and
1020    // captured the rest of the document (Fatal `Document:Malformed`: 1909.03404, 2311.02007, 2503.03764). A prefix of
1021    // spaces only is not written (LaTeX writes it, a blank last line). OXIDIZED_DESIGN #572; guard
1022    // `perfect_kernel_batch66d::filecontents_ends_mid_line`.
1023    while let Some(line) = read_raw_line() {
1024      let Some(at) = line.find(end_marker.as_str()) else {
1025        lines.push(line);
1026        continue;
1027      };
1028      let (before, after) = (&line[..at], line[at + end_marker.len()..].trim_end());
1029      if !before.trim().is_empty() {
1030        Warn!(
1031          "unexpected",
1032          "filecontents",
1033          s!("Writing text `{before}' before {end_marker} as last line of {filename}")
1034        );
1035        lines.push(before.to_string());
1036      }
1037      if !after.is_empty() {
1038        Warn!(
1039          "unexpected",
1040          "filecontents",
1041          s!("Ignoring text `{after}' after {end_marker}")
1042        );
1043      }
1044      break;
1045    }
1046    if keep_existing {
1047      let place = if has_option("nosearch") {
1048        "in current directory"
1049      } else {
1050        "exists on the system"
1051      };
1052      Info!(
1053        "note",
1054        "filecontents",
1055        s!("File `{filename}' already {place}. Not generating it from this source")
1056      );
1057      return Ok(());
1058    }
1059    let n = lines.len();
1060    Info!(
1061      "note",
1062      "filecontents",
1063      s!("Cached filecontents for {filename} ({n} lines)")
1064    );
1065    // latex.ltx:18998/:19023 writes through `\immediate\openout`, which names
1066    // an extension-less file `<name>.tex` (`output_file_name`, tex.web §1374):
1067    // latexdemo's `{filecontents*}{democode}` is `democode.tex` to its own
1068    // `\IfFileExists{democode.tex}` (latex4wp). The header keeps the name as
1069    // given, as LaTeX's `\@curr@file` does.
1070    vfs_store(&output_file_name(&filename), &lines.join("\n"));
1071    Ok(())
1072  }

1083  }

1706/// Is the document's code on `hook`? lthooks labels code by its owner (latex.ltx:5208 `\__hook_currname_or_default:`): the
1707/// document's is `top-level` (:5364-5369) or a label naming no loaded class or package; `\AddToHookNext`'s carries no
1708/// label (:6340), counted as the document's.
1709fn document_hook_code(hook: &str) -> bool {
1710  let body = |cs: &str| match lookup_meaning(&T_CS!(cs)) {
1711    Some(Stored::Expandable(m)) => match m.get_expansion() {
1712      Some(ExpansionBody::Tokens(t)) => Some(t.clone()),
1713      _ => None,
1714    },
1715    _ => None,
1716  };
1717  if ["toplevel", "next"]
1718    .iter()
1719    .any(|code| body(&s!("\\__hook_{code} {hook}")).is_some_and(|t| !t.is_empty()))
1720  {
1721    return true;
1722  }
1723  // l3prop's `\s__prop \__prop_pair:wn <label> \s__prop {<code>}` (l3prop)
1724  let Some(prop) = body(&s!("\\g__hook_{hook}_code_prop")) else {
1725    return false;
1726  };
1727  let (pair, mark) = (T_CS!("\\__prop_pair:wn"), T_CS!("\\s__prop"));
1728  let tokens = prop.unlist();
1729  let mut rest = &tokens[..];
1730  while let Some(at) = rest.iter().position(|t| *t == pair) {
1731    rest = &rest[at + 1..];
1732    let end = rest.iter().position(|t| *t == mark).unwrap_or(rest.len());
1733    let label: String = rest[..end].iter().map(ToString::to_string).collect();
1734    if document_hook_label(&label) {
1735      return true;
1736    }
1737    rest = &rest[end..];
1738  }
1739  false
1740}
