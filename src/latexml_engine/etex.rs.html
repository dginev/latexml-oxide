17/// Resolve a FontDef token to an Rc<Font>: look up fontinfo via definition CS name, fall back to
18/// current font. Perl: `$font = $STATE->lookupValue('font')->merge(%$fontinfo);`
19fn fontchar_lookup_font(font_tok: &Token) -> Option<Rc<Font>> {
20  // Resolve through definition to get actual CS name (e.g. \font -> \tenrm)
21  let key = match lookup_definition(font_tok) {
22    Ok(Some(defn)) => {
23      s!("fontinfo_{}", defn.get_cs_name())
24    },
25    _ => {
26      s!("fontinfo_{}", font_tok)
27    },
28  };
29  with_value(&key, |v| {
30    if let Some(Stored::Font(f)) = v {
31      Some(Rc::clone(f))
32    } else {
33      None
34    }
35  })
36  .or_else(lookup_font)
37}

39/// The glyph coverage of the font a `\iffontchar` font token names, when
40/// its file can be resolved: the current font (`\font`) selected by
41/// fontspec carries `FONTSPEC_FONTFILE` (locally scoped by
42/// `\lx@fontspec@usefile`), any `\font`-declared font carries its TFM name.
43fn fontchar_coverage(
44  font_tok: &Token,
45) -> Option<Rc<::latexml_core::common::font::coverage::GlyphCoverage>> {
46  use ::latexml_core::common::font::coverage::{coverage_for_file, font_file_path};
47  // `\font` itself, or a `\let` alias of it (`\tex_font:D`).
48  let is_current_font = font_tok.defined_as(&T_CS!("\\font"));
49  if is_current_font {
50    let file = lookup_string("FONTSPEC_FONTFILE");
51    if !file.is_empty() {
52      return coverage_for_file(&file);
53    }
54  }
55  let font = fontchar_lookup_font(font_tok)?;
56  let name = font.name.as_deref()?.trim().to_string();
57  if name.is_empty() || name.contains('/') {
58    return None;
59  }
60  let tfm = font_file_path(&format!("{name}.tfm"))?;
61  coverage_for_file(&tfm)
62}

67  // Perl L153-193: etex_readexpr / etex_readexpr_i for \numexpr/\dimexpr/etc.
68  fn is_relax_meaning(token: &Token) -> bool {
69    if *token == *TOKEN_RELAX {
70      return true;
71    }
72    if token.get_catcode() != Catcode::CS {
73      return false;
74    }
75    matches!(lookup_meaning(token),
76      Some(Stored::Primitive(ref p)) if *p.get_cs() == *TOKEN_RELAX)
77  }

79  fn etex_readexpr(rtype: RegisterType) -> Result<RegisterValue> {
80    // etex.ch `scan_expr`: an alignment tab/row end met between operands is
81    // unexpandable (batch 56bi, `gullet::NumberScan`).
82    let _scan = NumberScan::begin();
83    let value = etex_readexpr_i(rtype, 0)?;
84    if let Some(token) = read_token()? {
85      // Skip \relax or token with \relax meaning (\__int_eval_end: etc.)
86      if !is_relax_meaning(&token) {
87        unread_one(token);
88      }
89    }
90    // eTeX fixes the result type by the command (\dimexpr→Dimension, etc.), not
91    // by the operands. The evaluator's "missing number, treated as zero"
92    // recovery / LHS-typed arithmetic can return a mismatched variant (e.g.
93    // Number(0) for \dimexpr), which then panics the register getter's
94    // expect_dimension(). Coerce to the declared type so the getter stays total.
95    let value = value.coerce_to(rtype);
96    // etex.ch `scan_expr`: a result past `infinity` (2^31 − 1, an integer) or
97    // `max_dimen` (any other type) is "Arithmetic overflow", and the
98    // expression's value is zero (pdfTeX: `\dimexpr-16383pt*3\relax` = 0.0pt).
99    // An Error, as TeX's: the zero is not the author's value (OXIDIZED_DESIGN #519); a Warning in a box measured by
100    // our approximation (`approximated_measure`).
101    let ceiling = if matches!(rtype, RegisterType::Number) {
102      MAX_INTEGER
103    } else {
104      MAX_DIMEN
105    };
106    if value.clone().value_of().abs() > ceiling {
107      let detail = "the expression's value is out of range, so it is zero (etex.ch scan_expr)";
108      match approximated_measure() {
109        Some((ours, true)) => Warn!("expected", "<number>", "Arithmetic overflow.", detail, ours),
110        Some((ours, false)) => Info!("expected", "<number>", "Arithmetic overflow.", detail, ours),
111        None => Error!("expected", "<number>", "Arithmetic overflow.", detail),
112      }
113      return Ok(RegisterValue::default().coerce_to(rtype));
114    }
115    Ok(value)
116  }

118  fn etex_readexpr_i(rtype: RegisterType, prec: usize) -> Result<RegisterValue> {
119    let token = match read_x_non_space()? {
120      Some(t) => t,
121      None => return Ok(RegisterValue::default()),
122    };
123    let mut value = if token == T_OTHER!("(") {
124      let i_value = etex_readexpr_i(rtype, 0)?;
125      let close = read_x_token(None, false, None)?;
126      if close.is_none() || close != Some(T_OTHER!(")")) {
127        let got = close
128          .map(|t| t.stringify())
129          .unwrap_or_else(|| "EOF".to_string());
130        let message = format!("Missing close parenthesis in {:?} expr. Got {}", rtype, got);
131        Error!("expected", ")", message);
132      }
133      i_value
134    } else {
135      // etex.ch `scan_expr` ("Scan a factor f of type o or start a
136      // subexpression"): the `(` probe is `get_x_token` + `back_input`, and
137      // `back_input` re-inserts `cur_tok = cs_token_flag + cur_cs` — the PLAIN
138      // control sequence (tex.web §358 keeps `cur_cs` under `no_expand_flag`).
139      // So a `\noexpand`'d macro heading a factor loses its protection and the
140      // following `scan_int` expands it: `\edef…{\the\numexpr\noexpand\@nameuse
141      // {sectiontocdepth}+\@ne\relax}` (tocbasic.sty:2688) yields 2, not
142      // "Missing number" (pdfTeX-probed). Strip the `\special_relax` family
143      // back to the shadowed token before unreading. Guard:
144      // `perfect_kernel_batch53::numexpr_factor_reexpands_noexpanded_macro`.
145      let token = token.noexpand_shadowed().unwrap_or(token);
146      unread_one(token);
147      read_value(rtype)?
148    };
149
150    while let Some(next) = read_x_non_space()? {
151      if next == *TOKEN_RELAX {
152        unread_one(next);
153        break;
154      } else if next == T_OTHER!("+") && prec < 1 {
155        value = value.add(etex_readexpr_i(rtype, 1)?);
156      } else if next == T_OTHER!("-") && prec < 1 {
157        value = value.subtract(etex_readexpr_i(rtype, 1)?);
158      } else if next == T_OTHER!("*") && prec < 2 {
159        value = value.multiply(etex_readexpr_i(RegisterType::Number, 2)?);
160      } else if next == T_OTHER!("/") && prec < 2 {
161        value = value.divideround(etex_readexpr_i(RegisterType::Number, 2)?);
162      } else {
163        unread_one(next);
164        break;
165      }
166    }
167    Ok(value)
168  }

171  fn etex_penalties_getter(name: &str) -> Option<RegisterValue> {
172    let n = read_number().unwrap_or_default().value_of();
173    let n = if n < 0 { 0i64 } else { n } as usize;

179        let idx = n - 1;

182          .and_then(|s| {

184              Some(*num)
185            } else {
186              None
187            }
188          })
189          .map(RegisterValue::Number)
190          .or_else(|| Some(RegisterValue::Number(Number::new(0))))
191      } else {
192        Some(RegisterValue::Number(Number::new(0)))
193      }
194    })

538  // 3.9 Math Formulas — Perl L306
539  // `\middle` (eTeX) emits a separator delimiter inside a balanced
540  // `\left/\right` pair, e.g. `\left\{ x \middle| x<0 \right\}` for a
541  // set-comprehension. The delimiter inherits the role MIDDLE
542  // (already distinct from VERTBAR), so the math grammar's middle_bar
543  // / middle rules can pair it without ambiguity. We also stamp
544  // `role_side="middle"` on the emitted XMTok — symmetry with
545  // \left's `role_side="left"` and \right's `role_side="right"` set
546  // in tex_math.rs:\lx@delim@left and :\lx@delim@right, giving a uniform 3-way side
547  // discriminator on side-aware fence-pair delimiters. Task #263.
548  //
549  // `\middle` is TeX's `left_right` command (tex.web §1188), unexpandable; §1191-1192 require the `math_left_group` a
550  // `\left` opened, and without one ("Extra \middle.") the delimiter is still typeset
551  // (`tex_math::left_right_group`). Perl's `\middle` is the constructor itself (`DefConstructor('\middle Token')`,
552  // eTeX.pool.ltxml:306); here it dispatches to `\lx@delim@middle`, which reverts as `\middle`, as `\right` dispatches
553  // to `\lx@right` (TeX_Math.pool.ltxml:782-791), so that a `\middle` retried after an inserted closer (§1064
554  // `off_save`) is typeset once. OXIDIZED_DESIGN #530. Guard
555  // `perfect_kernel_batch65d::extra_right_and_middle_without_left`.
556  DefPrimitive!("\\middle", sub[_args] {
557    use crate::tex_math::LeftRightGroup;
558    // An "Extra \middle." keeps its delimiter too, as `\right`'s does (user ruling 2026-10-12).
559    if !matches!(crate::tex_math::left_right_group("middle")?, LeftRightGroup::Retried) {
560      unread(Tokens!(T_CS!("\\lx@delim@middle")));
561    }
562  });
563  DefConstructor!("\\lx@delim@middle DelimiterToken", "#1",
564  alias => "\\middle",
565  after_construct => sub[document, _whatsit] {
566    let current = document.get_node().clone();
567    let delim_opt = current.get_child_nodes()
568      .into_iter()
569      .filter(|n| n.get_type() == Some(NodeType::ElementNode))
570      .last();
571    let mut delim = delim_opt.unwrap_or_else(|| current.clone());
572    document.set_attribute(&mut delim, "role", "MIDDLE")?;
573    document.set_attribute(&mut delim, "stretchy", "true")?;
574    document.set_attribute(&mut delim, "role_side", "middle")?;
575  });

627      None => true,
628    }
