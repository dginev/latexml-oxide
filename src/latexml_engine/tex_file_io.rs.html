47  // TeX I/O primitives
48  // The optional `=` is scanned with expansion (tex.web §1275, §405; KNOWN_PERL_ERRORS #398).
49  DefPrimitive!("\\openin Number SkipKeyword:= SkipSpaces TeXFileName",
50  sub[(port, filename)] {
51    let port = port.to_string();
52    let (filename, _) = tex_file_name(&filename.unlist());
53    // tex.web §1275: `\openin` first closes the stream, so a name that cannot be opened leaves it closed (`\ifeof`
54    // true) instead of still reading the previous file. pinlabel.sty:607-623 leaves a figure's EPS open after finding
55    // its `%%BoundingBox`, then probes the next figure's `.ps` (:567): the stale stream answered, so every later
56    // figure read the previous EPS, found no box and printed its file name ("summ\_2 not found", a stray `_`), its
57    // labels lost (1010.6236). Perl alike (TeX_FileIO.pool.ltxml:52,60; KNOWN_PERL_ERRORS #520).
58    close_input_stream(&port);
59    // Rely on FindFile to enforce any access restrictions
60    // Perl: NOT noltxml! \openin is often used to check file existence,
61    // and we SHOULD find .ltxml (binding) versions too.
62    if let Some(path) = find_file(&filename, None) {
63      // `\openin` is a PROBE (tex.web §1275: a file that cannot be opened
64      // just leaves the stream closed, `\ifeof` true) — it must never reach
65      // the mouth's open-failure diagnostics. A virtual file that `\openout`
66      // registered and has not been written yet is a real, EMPTY file (Perl
67      // parity: the raw name resolves to an empty mouth), so `Some("")` is kept
68      // as content; a path that is neither virtual nor on disk (a binding-only
69      // name find_file admits) gets an empty mouth WITHOUT touching the disk —
70      // minitoc's `\openin\@inputcheck\jobname.mtc0`, frenchle's `.aux_LE`,
71      // chapterbib's `.cb` made `Fatal:Mouth:MissingFile` in sweep s108 once
72      // the open-failure `fatal!` logged at the raise (cahierprof-doc,
73      // fepslatex, pst-eucl-docBG, ArsClassica-de; 0 diagnostics in Perl).
74      // Guard: `perfect_kernel_batch56::openin_of_a_missing_or_unwritten_file_is_silent`.
75      let content = match vfs_read(&path) {
76        Some(virtual_content) => Some(virtual_content),
77        None if std::path::Path::new(&path).is_file() => None,
78        None => Some(String::new()),
79      };
80      let mouth = Mouth::create(&path, MouthOptions {
81        content,
82        .. MouthOptions::default()
83      })?;
84      AssignValue!(&s!("input_file:{}", port), mouth, Some(Scope::Global));
85    }
86  });
87  DefPrimitive!("\\closein Number", sub[(port)] {
88    close_input_stream(&port.to_string());
89  });

91  DefPrimitive!("\\read Number SkipKeyword:to RedefinableToken", sub[(port, token)] {
92    // (the Rc of the mouth, not a clone of the Stored envelope around it, and a copy under a speculative run's journal)
93    if let Some(mouth_obj) = input_stream_to_change(&format!("input_file:{port}")) {
94      bgroup();
95      // Special EOL/EOF treatment for \read — bookkeeping for THIS read, undone by
96      // the `egroup` below. Explicitly local: a `\global` prefix is still live here
97      // (prefixes clear after the primitive), and an unscoped assignment would take
98      // it — `\global\read` (csvsimple-legacy.sty:296 `\csvreadnext`) then left the
99      // MAIN mouth in read mode for the rest of the document, and its stray `\par`
100      // tokens landed in every later label and id (`labels="LABEL:x\par\par"`,
101      // `xml:id="S2par"`; RUST-ONLY, Perl clean). The prefix still globalizes the
102      // target macro below. Guard `perfect_kernel_batch56::global_read_leaves_the_main_mouth_alone`.
103      AssignValue!("PRESERVE_NEWLINES", 2, Some(Scope::Local));
104      AssignValue!("INCLUDE_COMMENTS", false, Some(Scope::Local));
105      let mut tokens = Vec::new();
106      let mut level: i32 = 0;
107      let mut discard = false;
108      let mut mouth = mouth_obj.borrow_mut();
109      while let Some(t) = mouth.read_token() {
110        let cc = t.get_catcode();
111        if cc == Catcode::BEGIN { level += 1; }
112        if cc == Catcode::END   { level -= 1; }
113        if level < 0            { discard = true; } // silently discard extra } & read till EOL
114        if !discard && cc != Catcode::MARKER {
115          tokens.push(t);
116        }
117        if (level == 0 || discard) && mouth.is_eol() {
118          break;
119        }
120      }
121      // tex.web §485-486 `input_ln`: the physical line is consumed whole,
122      // whatever catcode its `\endlinechar` had (`Mouth::finish_physical_line`).
123      // Guard: `perfect_kernel_batch54::read_consumes_the_physical_line_across_catcode_regimes`.
124      mouth.finish_physical_line();
125      egroup()?;
126      if level > 0 {
127        Error!("unexpected", "unbalanced",
128          s!("Runaway definition? File ended within \\read with unbalanced {{"));
129        // Append T_END tokens to balance (non-TeX patch to avoid Fatal)
130        for _ in 0..level {
131          tokens.push(T_END!());
132        }
133      }
134      DefMacro!(token, None, Tokens::new(tokens), nopack_parameters => true);
135    } else {
136      // tex.web §484: a stream that is not open reads from the terminal, and
137      // "if interaction > nonstop_mode" TeX prompts — otherwise
138      // `fatal_error("*** (cannot \read from terminal in nonstop modes)")`.
139      // That is the deliberate halt of iftex.sty:51 (`\RequireXeTeX` & co.)
140      // and expl3's `\__msg_fatal_exit:` (`\batchmode\read -1 to …`): a
141      // wrong-engine package stops loading HERE instead of running its body on
142      // undefined engine primitives (the 23-doc `alloc_failed` cluster of sweep
143      // 57: bidi/ucharclasses looping on `\XeTeXcharclass`). In scroll/
144      // errorstop mode (no terminal here) the read stays a no-op, as in Perl.
145      let interaction = lookup_value("INTERACTION_MODE")
146        .and_then(|v| match v { Stored::Number(n) => Some(n.0), _ => None })
147        .unwrap_or(2);
148      if interaction <= 1 {
149        Fatal!(Mouth, EoF, s!("*** (cannot \\read from terminal in nonstop modes) — \\read{port} to {token} under \\batchmode/\\nonstopmode (tex.web §484)"));
150      }
151    }
152  });

157  });

188  // XGeneralText is the TeX <general text> — balanced group read with PARTIAL
189  // expansion (respects `\the`, `\showthe`, `\unexpanded`, `\detokenize`).
190  // Using a raw `{}` followed by `Expand!` over-expands active chars like `~`
191  // → `\lx@NBSP`, whose untex leaks the literal string `"\lx@NBSP"` to the
192  // aux file; when `\input` reads it back with `@` in OTHER catcode, the CS
193  // splits into `\lx`+`@NBSP` — an "undefined \lx" error. arxiv 1112.4846
194  // (harvmac `\listrefs`) triggered this.
195  DefPrimitive!("\\write Number XGeneralText", sub[(port_n, tokens)] {
196    let port = port_n.value_of();
197    let target = with_value(&s!("output_file:{}", port), |val_opt|
198    if let Some(filename) = val_opt {
199       filename.to_string()
200    } else { String::new() });
201    if !target.is_empty() {
202      let text = tokens.untex();
203      vfs_append_line(&target, &text);
204      // An `.aux` file's labels are read back as they are written: LaTeX reads the file at the next run's
205      // `\begin{document}` (sect11.rs `read_back_aux_text`).
206      if target.ends_with(".aux") {
207        crate::latex_constructs::read_back_aux_text(&text)?;
208      }
209    } else if port < 0 {
210      NoteLog!(tokens.untex());
211    } else {
212      Note!(tokens.untex());
213    }

246  // Special output
247  //----------------------------------------------------------------------
248  // \special          c  sends material to the dvi file for special processing.
249  // The text is tex.web §1354 `scan_toks(false, true)`, expanded as it is read
250  // (Perl `\special XGeneralText`, TeX_FileIO.pool.ltxml:177).
251  DefPrimitive!("\\special XGeneralText", sub[(arg)] {
252    let special_str = arg.to_string();
253    // recognize one special graphics inclusion case
254    if let Some(cap) = PSFILE_REGEX.captures(&special_str) {
255      let graphic = cap.get(1).unwrap().as_str();
256      RequirePackage!("graphicx", searchpaths_only => true);
257      let mut kv = Vec::new();
258      for prop in ["voffset","hoffset","hscale","vscale","hsize","vsize","angle"] {
259        let prop_regex = Regex::new(&s!("\\b{prop}=(.+?)(?:\\s|\\}})")).unwrap();
260        if let Some(cap) = prop_regex.captures(&special_str) {
261          let prop_val = cap.get(1).unwrap().as_str();
262          if !kv.is_empty() {
263            kv.push(T_OTHER!(","));
264          }
265          kv.push(T_OTHER!(prop));
266          kv.push(T_OTHER!("="));
267          kv.push(T_OTHER!(prop_val));
268        }
269      }
270      if !kv.is_empty() {
271        let mut wrapped = vec![T_OTHER!("[")];
272        wrapped.extend(kv);
273        wrapped.push(T_OTHER!("]"));
274        kv = wrapped;
275      }
276
277      unread_vec(vec![T_BEGIN!(), T_OTHER!(graphic), T_END!()]);
278      unread_vec(kv);
279      unread_one(T_CS!("\\lx@special@graphics"));
280    } else {
281      // A dvips header (`\special{! …}`) is PostScript run before the page; its numeric constants are recorded as
282      // `dvips_header:<name>`, as PostScript drawing commands may name them in their arguments (axodraw's,
283      // axodraw_sty.rs: 1701.01554 `/Ldensity {0.25} def`, 0901.1195 `/Tdensity {0.10} def`). Nothing is drawn.
284      if special_str.trim_start().starts_with('!') {
285        for cap in PS_HEADER_NUMBER_REGEX.captures_iter(&special_str) {
286          if let Some(value) = cap.get(2).or_else(|| cap.get(3)) {
287            assign_value(&s!("dvips_header:{}", &cap[1]), value.as_str().to_string(), Some(Scope::Global));
288          }
289        }
290      }
291      Info!("ignored", "special", s!("Unrecognized TeX Special: {arg}"));
292    }
293  });

411  // output processing
412  //----------------------------------------------------------------------
413  // \shipout          c  sends the contents of a box to the dvi file.
414  // \output           pt holds the token list used to typeset one page.
415
416  // `\shipout` scans its box operand as `\setbox` does (tex.web §1073
417  // `leader_ship` with `ship_out_flag`, §1084 `scan_box`) and hands the box to
418  // `ship_out` (§1075 `box_end`, §638). LaTeXML has no pages: the shipped-out
419  // box is emitted in place, where `\box` would put it in the current mode.
420  // Perl defines no `\shipout` (TeX_FileIO.pool.ltxml:257 is a placeholder);
421  // LaTeX's `\shipout` is `\__shipout_execute:` (latex.ltx:19911-19917), whose
422  // `\setbox\l_shipout_box` + `\afterassignment` chain ends in
423  // `\tex_shipout:D \box_use:N \l_shipout_box` (latex.ltx:19986), the
424  // expl3-code.tex:531 copy of this primitive. With a braced operand
425  // (`\shipout\vbox{…}`) Perl parks the `\afterassignment` token for the box
426  // body (TeX_Box.pool.ltxml:170-172), so it reaches `\tex_shipout:D`, reports
427  // it undefined and typesets the box in place; with a register operand
428  // (`\shipout\box255`) the token never fires (KNOWN_PERL_ERRORS #253) and the
429  // box is lost silently. Rust fires it for both since batch 56ir, so both
430  // errored until the primitive existed.
431  // Witness: coverpage/SimpleSample (the cover page `\shipout\box255`).
432  // Guard: `shipout_parskip::shipout_emits_the_box_in_place`.
433  DefPrimitive!("\\shipout", {
434    match crate::tex_box::read_box_operand()? {
435      Some(xtoken) => invoke_token(&xtoken)?,
436      None => Vec::new(),
437    }
438  });

442/// Close input stream `port` (`\closein`): finish its mouth, if one is open, and clear it.
443fn close_input_stream(port: &str) {
444  let file_key = s!("input_file:{}", port);
445  if let Some(mouth) = input_stream_to_change(&file_key) {
