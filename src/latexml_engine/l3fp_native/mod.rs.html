1//! A native answer to `\__fp_parse:n` and `\__fp_parse_o:n` (expl3-code.tex:17313-17339), the doors of l3fp: `\fp_set:Nn`,
2//! `\fp_eval:n`, `\fp_compare:nNnTF`, xfp's `\fpeval`, siunitx and l3draw all come through them. It takes one `\fp_set:Nn`
3//! from about 40k token reads to a few hundred (`pgf-interference`'s pixel loop, docs/performance/PERFORMANCE.md).
4//!
5//! **The hook.** Both doors stay the format's own macros. Their expansion is
6//! `… \exp:w \__fp_parse_operand:Nw \c__fp_prec_end_int \__fp_parse_expand:w <expression> \s__fp_expr_mark
7//! \__fp_parse_infix_end:N \s__fp_expr_stop …`, the only such call. `\__fp_parse_operand:Nw` keeps its body too, and carries
8//! a native call (`operand_call`, `SpecialCall::Native`): when the gullet expands it in full and the input is that
9//! shape, the expression is read from the pushback without consuming it. The answer leaves what the macro's own run
10//! leaves for `\__fp_parse_after:ww`, `\exp_end_continue_f:w <fp>@ \__fp_parse_infix_end:N` (`\s__fp_expr_stop` still on the
11//! input), through the ordinary path of a macro's expansion (`invoke_expansion`, whose `inherit_field_prov` a printed
12//! `\@title` needs). On any other call, and on anything the native path does not handle with certainty (`Decline`: an
13//! unknown token, a tuple, `nan`, `rand`, a flag or a message of any kind, a tab mark that ends a cell), nothing has happened and the macro runs as
14//! defined over the input as it stands. So `\meaning`, the prefixes and a one-step `\exp_after:wN` of either door and of
15//! this macro are the format's, and a tuple, a flag or a message is the macro's own.
16//!
17//! **While it may answer.** Only while l3fp is what this module ports: the meanings of the `\__fp_…`, `\c__fp_…`, `\s__fp…`
18//! and `\c_…_fp` names are watched ([`watch_meanings`]), and an assignment that changes one, however local, ends the native
19//! path for the rest of the conversion. A document's new word (`\fp_new_variable:n`) has no meaning to change: the parser
20//! declines any word it does not know. So a control sequence the parser knows by name means what it did at the install.
21//!
22//! **The arithmetic is a port.** `+ - * /` and `sqrt` round correctly in the TeX, so exact integers give the same digits
23//! (`num.rs`). `exp`, `ln`, `^` and the trigonometric functions do not, so `func.rs` and `fixed.rs` translate their block
24//! operations and tables one for one, truncations included. No `f64` comes near a value.
25//!
26//! `LATEXML_L3FP_NATIVE=0` installs nothing. `LATEXML_L3FP_CHECK=1` runs the macro's own run on every answer and reports an
27//! Error naming the argument if the tokens differ; it compares tokens only, not what a run reports (`kit/l3fp_oracle.sh`
28//! holds the arithmetic against pdflatex as well). `LATEXML_L3FP_TRACE=1` counts answers and fallbacks by reason, printed on
29//! stderr at the thread's end.
30//!
31//! Installed at the format's end, after the dump's apply: the dump does not change, and a later raw load of
32//! `expl3-code.tex` simply wins (and trips the watch). Only over the l3fp this port was made from (`native_applies`): another
33//! year's l3fp stays entirely the macros'. No code here relies on unwinding (`maxperf` aborts on a panic). OXIDIZED_DESIGN_DIVERGENCES #977.

75/// Is `name` one of l3fp's own control sequences whose meaning the answers depend on? The marks `\s__fp…` are `\relax` by
76/// meaning (`\scan_new:N`); any other is defined otherwise, so a `\relax` on it comes from a `\csname` over a word still to
77/// define. (Asked only of a name that begins `\__`, `\c_` or `\s_`: [`install`]'s [`MeaningWatch`].)
78fn watched_name(name: &str) -> WatchedName {
79  if name.starts_with("\\s__fp") {
80    return WatchedName::Yes;
81  }
82  let family = name.starts_with("\\__fp_")
83    || (name.starts_with("\\c_") && (name.starts_with("\\c__fp_") || name.ends_with("_fp")));
84  if family && !NOT_WATCHED.contains(&name) {
85    WatchedName::RelaxIsUnset
86  } else {
87    WatchedName::No
88  }
89}

130/// Answers and fallbacks by reason (`LATEXML_L3FP_TRACE=1`).
131#[derive(Default)]
132struct Census {
133  native:   u64,
134  fallback: HashMap<String, u64>,
135  sample:   HashMap<String, String>,
136}
137
138impl Drop for Census {
139  fn drop(&mut self) {
140    if self.native == 0 && self.fallback.is_empty() {
141      return;
142    }
143    eprintln!("L3FP native {}", self.native);
144    for (reason, n) in &self.fallback {
145      eprintln!("L3FP fallback {n} {reason}");
146      eprintln!(
147        "L3FP sample {reason} <- {}",
148        self.sample.get(reason).map_or("", String::as_str)
149      );
150    }
151  }
152}

156/// `input`: the pushback's slice of the argument, its first token last.
157fn note_fallback(decline: Decline, at: Option<Token>, via: Option<Token>, input: &[Token]) {
158  let at = at.map(|t| t.to_string()).unwrap_or_default();
159  let via = via.map(|t| format!(" via {t}")).unwrap_or_default();
160  let key = format!("{} {at}{via}", decline.0);
161  CENSUS.with(|c| {
162    let mut census = c.borrow_mut();
163    *census.fallback.entry(key.clone()).or_insert(0) += 1;
164    census.sample.entry(key).or_insert_with(|| {
165      Tokens::new(input.iter().rev().copied().collect())
166        .to_string()
167        .replace('\n', " ")
168        .chars()
169        .take(160)
170        .collect()
171    });
172  });
173}

175/// A digit, or a `-`, of a number's form.
176fn other_digits(out: &mut Vec<Token>, v: i64, width: usize) {
177  let mut buf = [b'0'; 20];
178  let mut n = 0;
179  let mut u = v.unsigned_abs();
180  while u > 0 || n == 0 {
181    buf[n] = b'0' + (u % 10) as u8;
182    u /= 10;
183    n += 1;
184  }
185  n = n.max(width);
186  if v < 0 {
187    out.push(T_OTHER!("-"));
188  }
189  for i in (0..n).rev() {
190    out.push(digit_token(buf[i]));
191  }
192}

194#[inline]
195fn digit_token(d: u8) -> Token {
196  match d {
197    b'0' => T_OTHER!("0"),
198    b'1' => T_OTHER!("1"),
199    b'2' => T_OTHER!("2"),
200    b'3' => T_OTHER!("3"),
201    b'4' => T_OTHER!("4"),
202    b'5' => T_OTHER!("5"),
203    b'6' => T_OTHER!("6"),
204    b'7' => T_OTHER!("7"),
205    b'8' => T_OTHER!("8"),
206    _ => T_OTHER!("9"),
207  }
208}

210/// The form of a value, as `\__fp_exp_after_o:w` leaves it, onto `out`.
211fn fp_tokens(x: Fp, out: &mut Vec<Token>) {
212  out.reserve(36);
213  out.push(T_CS!("\\s__fp"));
214  out.push(T_CS!("\\__fp_chk:w"));
215  let sign_digit = |s: Sign| {
216    if s == Sign::Neg {
217      T_OTHER!("2")
218    } else {
219      T_OTHER!("0")
220    }
221  };
222  match x {
223    Fp::Zero(s, tag) => {
224      out.push(T_OTHER!("0"));
225      out.push(sign_digit(s));
226      out.push(if tag == Tag::Exact {
227        T_CS!("\\s__fp_exact")
228      } else {
229        T_CS!("\\s__fp_underflow")
230      });
231    },
232    Fp::Inf(s) => {
233      out.push(T_OTHER!("2"));
234      out.push(sign_digit(s));
235      out.push(T_CS!("\\s__fp_exact"));
236    },
237    Fp::Normal { sign, exp, mant } => {
238      out.push(T_OTHER!("1"));
239      out.push(sign_digit(sign));
240      out.push(T_BEGIN!());
241      other_digits(out, i64::from(exp), 0);
242      out.push(T_END!());
243      for b in num::blocks(mant) {
244        out.push(T_BEGIN!());
245        other_digits(out, i64::from(b), 4);
246        out.push(T_END!());
247      }
248    },
249  }
250  out.push(T_CS!("\\__fp_sep:"));
251}

253/// `\__fp_to_decimal:w`: the digits, no exponent, no trailing zeros. An infinity is that function's invalid operation.
254fn to_decimal(x: Fp) -> num::R<String> {
255  let (sign, exp, mant) = match x {
256    Fp::Zero(s, _) => return Ok(String::from(if s == Sign::Neg { "-0" } else { "0" })),
257    Fp::Inf(_) => return Err(Decline("invalid: the decimal of inf")),
258    Fp::Normal { sign, exp, mant } => (sign, i64::from(exp), mant),
259  };
260  let mut out = String::with_capacity(24);
261  if sign == Sign::Neg {
262    out.push('-');
263  }
264  let digits = mant.to_string();
265  if exp >= 16 {
266    // \__fp_to_decimal_huge:wnnnn
267    out.push_str(&digits);
268    for _ in 16..exp {
269      out.push('0');
270    }
271  } else {
272    if exp > 0 {
273      out.push_str(&digits[..exp as usize]);
274      out.push('.');
275      out.push_str(&digits[exp as usize..]);
276    } else {
277      out.push_str("0.");
278      for _ in exp..0 {
279        out.push('0');
280      }
281      out.push_str(&digits);
282    }
283    // \__fp_trim_zeros:w
284    let trimmed = out.trim_end_matches('0').trim_end_matches('.').len();
285    out.truncate(trimmed);
286  }
287  Ok(out)
288}

357/// Is the current l3fp the one this module ports: the expl3 date, and the bodies of the door and of what the answer stands
358/// for?
359fn native_applies() -> Result<bool> {
360  let (Some(door), Some(operand), Some(infix_end), Some(date)) = (
361    body_of(&T_CS!("\\__fp_parse:n"))?,
362    body_of(&T_CS!("\\__fp_parse_operand:Nw"))?,
363    body_of(&T_CS!("\\__fp_parse_infix_end:N"))?,
364    body_of(&T_CS!("\\ExplLoaderFileDate"))?,
365  ) else {
366    return Ok(false);
367  };
368  let date = date.to_string();
369  Ok(
370    operand.unlist_ref() == expected_operand().as_slice()
371      && infix_end.unlist_ref() == expected_infix_end().as_slice()
372      && PORTED.iter().any(|(d, align_safe)| {
373        *d == date && door.unlist_ref() == expected_door(*align_safe).as_slice()
374      }),
375  )
376}

378/// The door's shape on the input: the index of its `\s__fp_expr_mark`, or `None`. The argument is what lies between, its
379/// braces as they are. `\s__fp_expr_mark` in the argument, or `\s__fp_expr_stop` not next, is no door of the macro's.
380fn door_mark(input: &[Token]) -> Option<usize> {
381  let mut depth = 0u32;
382  let mut i = input.len().checked_sub(2)?;
383  while i > 0 {
384    i -= 1;
385    let t = input[i];
386    match t.code {
387      Catcode::BEGIN => depth += 1,
388      Catcode::END => depth = depth.checked_sub(1)?,
389      Catcode::CS if depth == 0 && t == T_CS!("\\s__fp_expr_mark") => {
390        return (i >= 2
391          && input[i - 1] == T_CS!("\\__fp_parse_infix_end:N")
392          && input[i - 2] == T_CS!("\\s__fp_expr_stop"))
393        .then_some(i);
394      },
395      _ => {},
396    }
397  }
398  None
399}

401/// `LATEXML_L3FP_CHECK=1`: the macro's own run over the same input, what it leaves against the native answer. Returns what
402/// the macro's run leaves, `\s__fp_expr_stop` included, and an Error if that differs.
403fn checked(input: &[Token], native: &[Token]) -> Result<Vec<Token>> {
404  let mut toks = vec![
405    T_CS!("\\exp_after:wN"),
406    T_CS!("\\exp_not:n"),
407    T_CS!("\\exp_after:wN"),
408    T_BEGIN!(),
409    T_CS!("\\exp:w"),
410    T_CS!("\\__fp_parse_operand:Nw"),
411  ];
412  toks.extend(input.iter().rev().copied());
413  toks.push(T_END!());
414  CHECKING.with(|c| c.set(true));
415  let raw = do_expand_partially(Tokens::new(toks));
416  CHECKING.with(|c| c.set(false));
417  let raw = raw?.unlist();
418  let stop = T_CS!("\\s__fp_expr_stop");
419  if raw.split_last() != Some((&stop, native)) {
420    let message = s!(
421      "native and raw l3fp differ on {{{}}}: native {} raw {}",
422      Tokens::new(input.iter().rev().copied().collect()),
423      Tokens::new(native.to_vec()),
424      Tokens::new(raw.clone())
425    );
426    Error!("l3fp", "native-mismatch", message);
427  }
428  Ok(raw)
429}

431/// `\__fp_parse_operand:Nw`'s native call (`SpecialCall::Native`). Every other caller passes another precedence:
432/// one comparison, and it runs as defined.
433fn operand_call() -> Result<Option<Tokens>> {
434  let at_door = with_pushback(|input| {
435    matches!(input, [.., expand, prec]
436      if *prec == T_CS!("\\c__fp_prec_end_int") && *expand == T_CS!("\\__fp_parse_expand:w"))
437  });
438  if at_door != Some(true) || CHECKING.with(Cell::get) {
439    return Ok(None);
440  }
441  // With the macro's own run near the recursion limit, what it reports is the answer.
442  let changed = watched_meaning_changed();
443  if changed.is_some() || expand_depth_left() < RAW_DEPTH_RESERVE {
444    left_to_the_macro();
445    if *TRACE {
446      let at = changed.map(|name| with(name, |cs| T_CS!(cs)));
447      note_fallback(
448        Decline("a name of l3fp changed, or near the recursion limit"),
449        at,
450        None,
451        &[],
452      );
453    }
454    return Ok(None);
455  }
456  let length = pushback_len();
457  let read = with_pushback(|input| {
458    let mark = door_mark(input)?;
459    Some((
460      mark,
461      parse::Parser::over_input(&input[mark + 1..input.len() - 2]),
462    ))
463  })
464  .flatten();
465  let Some((mark, mut parser)) = read else {
466    left_to_the_macro();
467    if *TRACE {
468      note_fallback(Decline("no door's marks on the input"), None, None, &[]);
469    }
470    return Ok(None);
471  };
472  let value = parser.parse();
473  let answer = match value {
474    Ok(x) if pushback_len() == length => x,
475    Ok(_) => {
476      left_to_the_macro();
477      return Ok(None);
478    },
479    Err(decline) => {
480      left_to_the_macro();
481      if *TRACE {
482        with_pushback(|input| {
483          note_fallback(
484            decline,
485            parser.at,
486            parser.via,
487            &input[mark + 1..input.len() - 2],
488          );
489        });
490      }
491      return Ok(None);
492    },
493  };
494  COUNTS.with(|c| c.set((c.get().0 + 1, c.get().1)));
495  if *TRACE {
496    note_native();
497  }
498  let mut out = vec![T_CS!("\\exp_end_continue_f:w")];
499  fp_tokens(answer, &mut out);
500  out.push(T_LETTER!("@"));
501  out.push(T_CS!("\\__fp_parse_infix_end:N"));
502  if check_enabled() {
503    let input = with_pushback(|input| input[mark - 2..].to_vec()).unwrap_or_default();
504    let raw = checked(&input, &out[1..])?;
505    out.truncate(1);
506    out.extend(raw);
507    consume_pushback_to(mark - 2);
508  } else {
509    consume_pushback_to(mark - 1);
510  }
511  Ok(Some(Tokens::new(out)))
512}

514fn left_to_the_macro() { COUNTS.with(|c| c.set((c.get().0, c.get().1 + 1))); }

516/// Give `\__fp_parse_operand:Nw` its native call, its definition as it is, and watch l3fp's meanings from now on.
517pub(crate) fn install() -> Result<()> {
518  COUNTS.with(|c| c.set((0, 0)));
519  INSTALLED.with(|c| c.set(false));
520  if !native_enabled() || !native_applies()? {
521    return Ok(());
522  }
523  let operand = T_CS!("\\__fp_parse_operand:Nw");
524  if let Some(Stored::Expandable(defn)) = lookup_meaning(&operand) {
525    let mut hooked = (*defn).clone();

527    set_native_call(operand_call);
528    assign_meaning(
529      &operand,
530      Stored::Expandable(Rc::new(hooked)),
531      Some(Scope::Global),
532    );
533    watch_meanings(MeaningWatch {
534      second_byte: *b"_cs",
535      third_byte:  b'_',
536      select:      watched_name,
537    });
538    INSTALLED.with(|c| c.set(true));
539  }
