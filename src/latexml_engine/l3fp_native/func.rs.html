24/// The digits of a 16-digit mantissa.
25fn mant_digits(mant: u64) -> [u8; 16] {
26  let mut out = [0u8; 16];
27  let mut v = mant;
28  for slot in out.iter_mut().rev() {
29    *slot = (v % 10) as u8;
30    v /= 10;
31  }
32  out
33}

35/// The first 24 of `digits` (zero-padded) as six blocks (`\__fp_pack_twice_four:wNNNNNNNN` three times).
36fn blocks_of(digits: &[u8]) -> Fixed {
37  let mut out = [0i64; 6];
38  for (i, slot) in out.iter_mut().enumerate() {
39    for j in 0..4 {
40      *slot = *slot * 10 + i64::from(digits.get(i * 4 + j).copied().unwrap_or(0));
41    }
42  }
43  out
44}

46/// The blocks of a normal number with two zero blocks more: `#8{0000}{0000}`.
47fn fixed_of_mant(mant: u64) -> Fixed {
48  let b = super::num::blocks(mant);
49  [
50    i64::from(b[0]),
51    i64::from(b[1]),
52    i64::from(b[2]),
53    i64::from(b[3]),
54    0,
55    0,
56  ]
57}

59/// `\__fp_sanitize:Nw` with what `\__fp_ep_to_float_o:wwN` adds: the exponent `base` and the float of `f`.
60fn float_of(sign: Sign, base: i64, f: &Fixed) -> R<Fp> {
61  let Some((e, mant)) = fixed::to_float(f) else {
62    // \__fp_fixed_to_float_zero:w: `- 2 * \c__fp_max_exponent_int`, an underflow's flag
63    return Err(Decline("underflow"));
64  };
65  normal(sign, base + e, mant)
66}

77/// `\__fp_exp_Taylor_loop:www`: `x` the fraction's fixed.
78fn exp_taylor(x: &Fixed) -> Fixed {
79  let mut y = *x;
80  for n in (2..=10).rev() {
81    y = mul(&fixed::add_one(&fixed::div_int(&y, n)), x);
82  }
83  fixed::add_one(&y)
84}

86/// `\__fp_exp_large:NwN` from `level` over `digits`, then `\__fp_exp_large_after:wwn` on the rest: the exponent the table's
87/// rows add and the mantissa `exp(x)` of the digits.
88fn exp_large(level: i64, digits: &[u8]) -> R<(i64, Fixed)> {
89  let mut accum = ONE_FIXED;
90  let mut exponent = 0;
91  let mut pos = 0;
92  let mut l = level;
93  loop {
94    let d = i64::from(digits.get(pos).copied().ok_or(Decline("digits ran out"))?);
95    pos += 1;
96    if d != 0 {
97      // \__fp_exp_intarray:w: item (36*l + 4*d) - 3 is the exponent, the next three the mantissa
98      let row = (9 * l + d - 1) as usize;
99      let [e, a, b, c] = *EXP_TABLE.get(row).ok_or(Decline("exp table"))?;
100      let (a, b, c) = (a - 100_000_000, b - 100_000_000, c - 100_000_000);
101      let t: Fixed = [
102        a / 10000,
103        a % 10000,
104        b / 10000,
105        b % 10000,
106        c / 10000,
107        c % 10000,
108      ];
109      accum = mul(&t, &accum);
110      exponent += e;
111    }
112    if l == 0 {
113      break;
114    }
115    l -= 1;
116  }
117  // `0 #2 0000 0000`: the fraction's digits after the zero
118  let mut rest = [0u8; 40];
119  let tail = &digits[pos..];
120  let n = tail.len().min(rest.len() - 1);
121  rest[1..=n].copy_from_slice(&tail[..n]);
122  let x = blocks_of(&rest);
123  Ok((exponent, mul(&exp_taylor(&x), &accum)))
124}

126/// The float of an `exp`-family result: `#1 \__fp_int_eval:w <exponent> #1 <float function>`.
127fn exp_result(negate: bool, exponent: i64, f: &Fixed) -> R<Fp> {
128  if negate {
129    // \__fp_fixed_inv_to_float_o:wN, its exponent less the table's
130    let (e, g) = ep_div(&(1, [1000, 0, 0, 0, 0, 0]), &(0, *f));
131    let Some((fe, mant)) = fixed::to_float(&g) else {
132      return Err(Decline("underflow"));
133    };
134    normal(Sign::Pos, e + fe - exponent, mant)
135  } else {
136    float_of(Sign::Pos, exponent, f)
137  }
138}

140/// `\__fp_exp_o:w`
141pub(super) fn exp(x: Fp) -> R<Fp> {
142  match x {
143    Fp::Zero(..) => Ok(ONE),
144    Fp::Inf(Sign::Pos) => Ok(super::num::INF),
145    Fp::Inf(Sign::Neg) => Ok(ZERO),
146    Fp::Normal { sign, exp: e, mant } => {
147      let e = i64::from(e);
148      if e > super::num::MAX_EXP_EXPONENT {
149        return Err(Decline(if sign == Sign::Pos {
150          "overflow"
151        } else {
152          "underflow"
153        }));
154      }
155      let m = mant_digits(mant);
156      let negate = sign == Sign::Neg;
157      if e < 0 {
158        // \__fp_decimate:nNnnnn { - #4 }, then `#2#3#4 0000 0000`
159        let zeros = if -e > 16 { 17 } else { (-e) as usize };
160        let mut digits = [0u8; 48];
161        let keep = 16.min(digits.len() - zeros);
162        digits[zeros..zeros + keep].copy_from_slice(&m[..keep]);
163        let f = exp_taylor(&blocks_of(&digits));
164        return exp_result(negate, 0, &f);
165      }
166      // `\int_value:w #3 #4`: the first digit at level e; digits after the (e+1)st, and 00000
167      let mut digits = [0u8; 30];
168      digits[..16].copy_from_slice(&m);
169      let (exponent, f) = exp_large(e, &digits[..21])?;
170      exp_result(negate, exponent, &f)
171    },
172  }
173}

194/// `\__fp_div_significand_calc:wwnnnnnnn` on the seven-digit `v` (`1 d q`), the dividend `b1 b3 b4` and the divisor
195/// `y`: the remainder's four blocks. Exact arithmetic with the shifts of `\__fp_pack_Bigg:NNNNNNw`.
196fn div_calc(v: i64, b: [i64; 3], y: &[i64; 4]) -> [i64; 4] {
197  const BL: i64 = -200000;
198  const BM: i64 = 200000 * 9999;
199  const BT: i64 = 200000 * 10000;
200  let d = (v / 100000) % 10;
201  let q = v % 100000;
202  let big = d == 1;
203  let r4 = BT - q * y[3];
204  let r3 = BM + b[2] - q * y[2] - if big { y[3] * 10 } else { 0 } + r4 / 10000;
205  let r2 = BM + b[1] - q * y[1] - if big { y[2] * 10 } else { 0 } + r3 / 10000;
206  let r1 = BL + b[0] - q * y[0] - if big { concat(y[0], y[1]) * 10 } else { 0 } + r2 / 10000;
207  [r1, r2 % 10000, r3 % 10000, r4 % 10000]
208}

210/// `\__fp_ln_significand:NNNNnnnN`: `-ln` of the mantissa, as a fixed (`\__fp_ln_x_ii:wnnnn` to `\__fp_ln_c:NwNw`).
211fn ln_significand(mant: u64) -> Fixed {
212  let b = super::num::blocks(mant);
213  let [b1, b2, b3, b4] = [
214    i64::from(b[0]),
215    i64::from(b[1]),
216    i64::from(b[2]),
217    i64::from(b[3]),
218  ];
219  let (d1, d2) = (b1 / 1000, (b1 / 100) % 10);
220  let factor = match d1 {
221    1 => {
222      if d2 < 4 {
223        10 - d2
224      } else {
225        6
226      }
227    },
228    2 => 4,
229    3 => 3,
230    4..=6 => 2,
231    _ => 1,
232  };
233  // \__fp_ln_x_ii:wnnnn: X = factor * mantissa, and 1 + X
234  let low = 1_000_000_000 + factor * concat(b3, b4);
235  let high = 99_999_990 + factor * concat(b1, b2) + low / 100_000_000;
236  let high5 = high / 10000;
237  let y_int = high5 + 1;
238  let y = [high5, high % 10000, (low / 10000) % 10000, low % 10000];
239  // \__fp_ln_x_iv:wnnnnnnnn: 2/(1+X), six quotient digits by \__fp_div_significand_calc
240  let mut vals = [0i64; 6];
241  let v1 = 999_999 + rdiv(200_000_000, y_int);
242  let mut r = div_calc(v1, [200_000_000, 0, 0], &y);
243  vals[0] = v1;
244  for val in vals.iter_mut().take(5).skip(1) {
245    let dividend = concat(r[0], r[1]);
246    let v = 999_999 + rdiv(dividend, y_int);
247    r = div_calc(v, [dividend, r[2], r[3]], &y);
248    *val = v;
249  }
250  vals[5] = 1_000_000 + rdiv(concat(r[0], r[1]), y_int);
251  // the carries of \__fp_div_significand_pack:NNN, from the last quotient up
252  let mut t = [0i64; 6];
253  let mut carry = 0;
254  for i in (0..6).rev() {
255    let v = vals[i] + carry;
256    t[i] = v % 10000;
257    carry = v / 10000 - 100;
258  }
259  let t0 = carry;
260  // \__fp_ln_div_after:Nw
261  let (plus, t) = if t0 == 0 {
262    (true, [
263      9999 - t[0],
264      9999 - t[1],
265      9999 - t[2],
266      9999 - t[3],
267      9999 - t[4],
268      10000 - t[5],
269    ])
270  } else {
271    (false, t)
272  };
273  // \__fp_ln_t_large:NNw: t^2 ...
274  let [t1, t2, t3, t4, t5, t6] = t;
275  let s5 = 100_000_000
276    + 2 * t1 * t5
277    + 2 * t2 * t4
278    + t3 * t3
279    + rdiv(2 * t1 * t6 + 2 * t2 * t5 + 2 * t3 * t4, 10000);
280  let s4 = 99_990_000 + 2 * t1 * t4 + 2 * t2 * t3 + s5 / 10000;
281  let s3 = 99_990_000 + 2 * t1 * t3 + t2 * t2 + s4 / 10000;
282  let s2 = 99_990_000 + 2 * t1 * t2 + s3 / 10000;
283  let s1 = 99_990_000 + t1 * t1 + s2 / 10000;
284  // \__fp_ln_square_t_after:w `1 0 #1#2#3 #4`
285  let square: Fixed = [
286    (s1 - 100_000_000) / 10000,
287    s1 % 10000,
288    s2 % 10000,
289    s3 % 10000,
290    s4 % 10000,
291    s5 % 10000,
292  ];
293  // ... and 2t (\__fp_ln_twice_t_pack:Nw)
294  let d6 = 10000 + 2 * t6;
295  let d5 = 9999 + 2 * t5 + d6 / 10000;
296  let d4 = 9999 + 2 * t4 + d5 / 10000;
297  let d3 = 9999 + 2 * t3 + d4 / 10000;
298  let d2 = 9999 + 2 * t2 + d3 / 10000;
299  let d1 = -1 + 2 * t1 + d2 / 10000;
300  let twice: Fixed = [
301    d1,
302    d2 % 10000,
303    d3 % 10000,
304    d4 % 10000,
305    d5 % 10000,
306    d6 % 10000,
307  ];
308  // \__fp_ln_Taylor_loop:www
309  let mut s: Fixed = [0; 6];
310  let mut n = 21;
311  while n != 1 {
312    s = mul(&fixed::add(&fixed::div_int(&ONE_FIXED, n), &s), &square);
313    n -= 2;
314  }
315  let series = mul(&fixed::add_one(&s), &twice);
316  // \__fp_ln_c:NwNw
317  let constant = ln_constant(factor);
318  if plus {
319    sub(&constant, &series)
320  } else {
321    fixed::add(&constant, &series)
322  }
323}

325/// `\__fp_ln_o:w`
326pub(super) fn ln(x: Fp) -> R<Fp> {
327  match x {
328    Fp::Zero(..) => Err(Decline("division by zero: ln(0)")),
329    Fp::Inf(Sign::Pos) => Ok(x),
330    Fp::Inf(Sign::Neg) | Fp::Normal { sign: Sign::Neg, .. } => {
331      Err(Decline("invalid: ln of a negative"))
332    },
333    Fp::Normal { exp: e, mant, .. } => {
334      let e = i64::from(e);
335      let sign = if e < 1 { Sign::Neg } else { Sign::Pos };
336      let l = ln_significand(mant);
337      // \__fp_ln_exponent:wn
338      match e {
339        0 => float_of(sign, 0, &l),
340        1 => float_of(sign, 0, &sub(&LN_TEN, &l)),
341        _ => {
342          let n = e.abs();
343          let product = mul(&LN_TEN, &[n, 0, 0, 0, 0, 0]);
344          let shifted: Fixed = [0, l[0], l[1], l[2], l[3], l[4]];
345          let f = if e > 0 {
346            sub(&product, &shifted)
347          } else {
348            fixed::add(&product, &shifted)
349          };
350          float_of(sign, 4, &f)
351        },
352      }
353    },
354  }
355}

359/// `\__fp_pow_exponent:Nwnnnnnw`: `s * (n ln(10) - l)`.
360fn pow_exponent(l: &Fixed, e: i64) -> Fixed {
361  let s = if e > 0 { 1 } else { -1 };
362  const M: i64 = 50000 * 9999;
363  let v6 = 50000 * 10000 + s * e * 8401 - s * l[4] + s * rdiv(e * 7991 - l[5], 10000);
364  let v5 = M + s * e * 456 - s * l[3] + v6 / 10000;
365  let v4 = M + s * e * 2994 - s * l[2] + v5 / 10000;
366  let v3 = M + s * e * 8509 - s * l[1] + v4 / 10000;
367  let v2 = M + s * e * 23025 - s * l[0] + v3 / 10000;
368  [
369    -50000 + v2 / 10000,
370    v2 % 10000,
371    v3 % 10000,
372    v4 % 10000,
373    v5 % 10000,
374    v6 % 10000,
375  ]
376}

378/// `\__fp_pow_npos_o:Nww`: `|a|^b`, both normal.
379fn pow_npos(ea: i64, ma: u64, sb: Sign, eb: i64, mb: u64) -> R<Fp> {
380  let negate = sb != if ea > 0 { Sign::Pos } else { Sign::Neg };
381  let l = ln_significand(ma);
382  let exponent = pow_exponent(&l, ea);
383  let product = mul(&exponent, &fixed_of_mant(mb));
384  // \__fp_pow_B:wwN: `1` and five digits, the blocks, zeros; shifted by b's exponent
385  let mut digits = [0u8; 80];
386  write_digits(&mut digits[..5], product[0]);
387  for (i, block) in product[1..].iter().enumerate() {
388    write_digits(&mut digits[5 + i * 4..9 + i * 4], *block);
389  }
390  if eb < 0 {
391    // \__fp_pow_C_neg:w: the zeros in front. Past 40 of them the digits are beyond what `\__fp_exp_large:NwN` reads.
392    let zeros = eb.unsigned_abs().min(40) as usize;
393    let mut shifted = [0u8; 120];
394    shifted[zeros..zeros + 60].copy_from_slice(&digits[..60]);
395    return pow_finish(negate, &shifted);
396  }
397  // \__fp_pow_C_pos:w: `eb` leading digits must be zero, and \__fp_pow_B:wwN stops at 22
398  if eb >= 22 || digits[..eb as usize].iter().any(|d| *d != 0) {
399    return Err(Decline(if negate { "underflow" } else { "overflow" }));
400  }
401  pow_finish(negate, &digits[eb as usize..])
402}

404/// `v` as digits filling `out`.
405fn write_digits(out: &mut [u8], mut v: i64) {
406  for slot in out.iter_mut().rev() {
407    *slot = (v % 10) as u8;
408    v /= 10;
409  }
410}

412/// `\__fp_pow_C_pack:w`: `\__fp_exp_large:NwN` from level 5 over `digits`.
413fn pow_finish(negate: bool, digits: &[u8]) -> R<Fp> {
414  let (exponent, f) = exp_large(5, digits)?;
415  exp_result(negate, exponent, &f)
416}

418/// `\__fp_^_o:ww`
419pub(super) fn pow(a: Fp, b: Fp) -> R<Fp> {
420  if b.is_zero() {
421    return Ok(ONE);
422  }
423  let magnitude = pow_magnitude(a, b)?;
424  if a.sign() == Sign::Pos {
425    return Ok(magnitude);
426  }
427  // \__fp_pow_neg:www, \__fp_pow_neg_case:w
428  match pow_neg_case(b) {
429    -1 => Ok(magnitude),
430    0 => Ok(magnitude.with_sign(magnitude.sign().flip())),
431    _ => {
432      if matches!(magnitude, Fp::Normal { .. } | Fp::Inf(_)) {
433        Err(Decline("invalid: a negative base's fractional power"))
434      } else {
435        Ok(magnitude)
436      }
437    },
438  }
439}
440
441fn pow_magnitude(a: Fp, b: Fp) -> R<Fp> {
442  match a {
443    Fp::Normal { exp: ea, mant: ma, .. } => {
444      let ea = i64::from(ea);
445      if ea == 1 && ma == POW10[15] {
446        return Ok(ONE);
447      }
448      match b {
449        Fp::Normal { sign, exp: eb, mant: mb } => pow_npos(ea, ma, sign, i64::from(eb), mb),
450        Fp::Inf(sb) => {
451          let grows = ea > 0;
452          Ok(if grows == (sb == Sign::Pos) {
453            super::num::INF
454          } else {
455            ZERO
456          })
457        },
458        Fp::Zero(..) => Ok(ONE),
459      }
460    },
461    // \__fp_pow_zero_or_inf:ww
462    _ => {
463      let bneg = b.sign() == Sign::Neg;
464      let akind_zero = a.is_zero();
465      if (akind_zero && !bneg) || (!akind_zero && bneg) {
466        return Ok(ZERO);
467      }
468      if !akind_zero {
469        return Ok(super::num::INF);
470      }
471      if matches!(b, Fp::Inf(_)) {
472        return Ok(super::num::INF);
473      }
474      Err(Decline("division by zero: 0 to a negative power"))
475    },
476  }
477}

479/// `\__fp_pow_neg_case:w`: -1 an even integer (or a zero, an infinity), 0 an odd one, 1 not an integer.
480fn pow_neg_case(b: Fp) -> i32 {
481  match b {
482    Fp::Zero(..) | Fp::Inf(_) => -1,
483    Fp::Normal { exp, mant, .. } => {
484      let e = i64::from(exp);
485      if e > 16 {
486        return -1;
487      }
488      let q = 16 - e;
489      if q > 16 {
490        return 1;
491      }
492      let p = POW10[q as usize];
493      if mant % p != 0 {
494        return 1;
495      }
496      if (mant / p) % 2 == 1 { 0 } else { -1 }
497    },
498  }
499}

503/// `\__fp_fact_o:w`
504pub(super) fn fact(x: Fp) -> R<Fp> {
505  match x {
506    Fp::Zero(..) => Ok(ONE),
507    Fp::Inf(Sign::Pos) => Ok(x),
508    Fp::Inf(Sign::Neg) | Fp::Normal { sign: Sign::Neg, .. } => {
509      Err(Decline("invalid: fact of a negative"))
510    },
511    Fp::Normal { .. } => {
512      let n = small_int(x).ok_or(Decline("invalid: fact of a non-integer"))?;
513      if n > 3248 {
514        return Err(Decline("overflow"));
515      }
516      // \__fp_fact_loop_o:w
517      let mut n = n;
518      let mut acc: Ep = (4, [1, 0, 0, 0, 0, 0]);
519      while n >= 12 {
520        acc = ep_mul(&(4, [n * (n - 1), 0, 0, 0, 0, 0]), &acc);
521        n -= 2;
522      }
523      const SMALL: [i64; 12] = [
524        1, 1, 2, 6, 24, 120, 720, 5040, 40320, 362880, 3628800, 39916800,
525      ];
526      let (e, f) = ep_mul(&(4, [SMALL[n as usize], 0, 0, 0, 0, 0]), &acc);
527      float_of(Sign::Pos, e, &f)
528    },
529  }

545/// `\__fp_trigd_large:ww`: an angle in degrees reduced modulo 360 by digits (`(x-4)/9*9`).
546fn trigd_large(octant: i64, e: i64, f: &Fixed) -> (i64, Fixed, Ep) {
547  // zeros, the 24 digits, `0000 0000 0000`
548  let zeros = (22 - e).max(0) as usize;
549  let mut s = [0u8; 80];
550  let mut n = zeros;
551  for block in f {
552    let mut v = *block;
553    for i in (0..4).rev() {
554      s[n + i] = (v % 10) as u8;
555      v /= 10;
556    }
557    n += 4;
558  }
559  let read = |from: usize, to: usize| s[from..to].iter().fold(0i64, |v, d| v * 10 + i64::from(*d));
560  // \__fp_trigd_large_auxi:nnnnwNNNN
561  let (a1, a2) = (read(0, 8), read(8, 16));
562  let (a3, a4_d1, a4_rest) = (read(16, 20), read(20, 21), read(21, 24));
563  let first = a1 + a2 - rdiv(a1 + a2 - 4, 9) * 9;
564  // `#1` of auxii is first and the next four digits side by side; `#2` one digit more
565  let v = (first * 10000 + a3) * 10 + a4_d1;
566  let k = rdiv(v - 4, 9);
567  let mut octant = octant + k * 2;
568  let reduced = v - k * 9;
569  let head = reduced * 1000 + a4_rest;
570  let x: Fixed = [head, read(24, 28), read(28, 32), read(32, 36), 0, 0];
571  let x = if head < 4500 {
572    x
573  } else {
574    octant += 1;
575    sub(&[9000, 0, 0, 0, 0, 0], &x)
576  };
577  trigd_small(octant, (2, x))
578}

580/// `\__fp_trig_large:ww`: an angle in radians, by its digits against those of `\c__fp_trig_intarray`.
581fn trig_large(octant: i64, e: i64, f: &Fixed) -> R<(i64, Fixed, Ep)> {
582  let k = rdiv(e - 4, 8);
583  let eaten = e - k * 8;
584  let start = k as usize;
585  // nine items, each without its leading 1; the first `eaten` digits go
586  let mut digits = [0u8; 72];
587  for i in 0..9 {
588    let item = *TRIG_TABLE
589      .get(start + i)
590      .ok_or(Decline("an angle past the trig table"))?
591      - 100_000_000;
592    let mut v = item;
593    for j in (0..8).rev() {
594      digits[i * 8 + j] = (v % 10) as u8;
595      v /= 10;
596    }
597  }
598  let mut p = [0i64; 16];
599  for (i, slot) in p.iter_mut().enumerate() {
600    let at = eaten as usize + i * 4;
601    *slot = digits[at..at + 4]
602      .iter()
603      .fold(0, |v, d| v * 10 + i64::from(*d));
604  }
605  // \__fp_trig_large_auxvi:wnnnnnnnn: m * p, thirteen blocks
606  const M: i64 = 50000 * 9999;
607  let m = [f[0], f[1], f[2], f[3]];
608  let mut low = [0i64; 13];
609  let mut carry = 0;
610  for i in (0..13).rev() {
611    let base = if i == 12 { 50000 * 10000 } else { M };
612    let v = base + m[0] * p[i + 3] + m[1] * p[i + 2] + m[2] * p[i + 1] + m[3] * p[i] + carry;
613    low[i] = v % 10000;
614    carry = v / 10000;
615  }
616  // \__fp_trig_large_auxvii:w: the first three digits name the octant, \__fp_trig_large_auxviii:ww
617  let first3 = low[0] / 10;
618  let q = rdiv(first3 - 62, 125);
619  let octant = octant + q;
620  let s: i64 = if q.rem_euclid(2) == 1 { -1 } else { 1 };
621  // \__fp_trig_large_auxx:wNNNNN: 8 * the fraction's digits (or their complement), 52 digits
622  let mut d = [0i64; 13];
623  let mut carry = 0;
624  // the digit stream starts with the three digits again, so this runs over the same 13 blocks
625  for i in (0..13).rev() {
626    let base = if i == 12 { 50000 * 10000 } else { M };
627    let v = base + s * 8 * low[i] + carry;
628    d[i] = v % 10000;
629    carry = v / 10000;
630  }
631  // \__fp_trig_large_auxxi:w: \__fp_ep_to_ep_loop:N over them
632  let mut v: u128 = 0;
633  let mut zeros: i64 = 0;
634  let mut seen = false;
635  let mut count = 0;
636  for block in &d {
637    for j in (0..4).rev() {
638      let digit = (block / 10i64.pow(j)) % 10;
639      if !seen && digit == 0 {
640        zeros += 1;
641        continue;
642      }
643      seen = true;
644      if count < 24 {
645        v = v * 10 + digit as u128;
646        count += 1;
647      }
648    }
649  }
650  if !seen || count < 24 {
651    // the sign would be read as a digit: what the TeX does then is not a number
652    return Err(Decline("an angle within 1e-28 of a multiple of pi/4"));
653  }
654  let mut blocks = [0i64; 6];
655  for slot in blocks.iter_mut().rev() {
656    *slot = (v % 10000) as i64;
657    v /= 10000;
658  }
659  let product = ep_mul_raw(&(-zeros, blocks), &(0, MINI_PI));
660  Ok(trig_small(octant, product))
661}

663/// `\__fp_trig:NNNNNwn`'s reduction: octant base, then the fixed and the extended form of the angle.
664fn reduce(unit: Unit, base: i64, exp: i32, mant: u64) -> R<(i64, Fixed, Ep)> {
665  let e = i64::from(exp);
666  let f = fixed_of_mant(mant);
667  let threshold = if unit == Unit::Rad { 0 } else { 1 };
668  Ok(match (e > threshold, unit) {
669    (true, Unit::Rad) => trig_large(base, e, &f)?,
670    (true, Unit::Deg) => trigd_large(base, e, &f),
671    (false, Unit::Rad) => trig_small(base, (e, f)),
672    (false, Unit::Deg) => trigd_small(base, (e, f)),
673  })
674}

699fn sin_series(octant: i64, x: &Fixed, ep: &Ep) -> Ep {

702    let mut v = mul_sub_back(&[0, 0, 0, 1, 5619, 2070], &square, &SIN_COEFFS_COS[0]);

725/// `sin`, `cos`, `csc`, `sec` (`\__fp_sin_o:w` and its siblings).
726pub(super) fn sin_like(unit: Unit, x: Fp, kind: SinKind) -> R<Fp> {
727  let (base, inverse) = match kind {
728    SinKind::Sin => (0, false),
729    SinKind::Cos => (2, false),
730    SinKind::Csc => (0, true),
731    SinKind::Sec => (2, true),
732  };
733  match x {
734    Fp::Zero(..) => match kind {
735      SinKind::Sin => Ok(x),
736      SinKind::Cos | SinKind::Sec => Ok(ONE),
737      SinKind::Csc => Err(Decline("division by zero: csc(0)")),
738    },
739    Fp::Inf(_) => Err(Decline("invalid: a trigonometric function of inf")),
740    Fp::Normal { sign, exp, mant } => {
741      let (octant, f, ep) = reduce(unit, base, exp, mant)?;
742      let sign = sin_sign(
743        octant,
744        if matches!(kind, SinKind::Sin | SinKind::Csc) {
745          sign
746        } else {
747          Sign::Pos
748        },
749      );
750      let r = sin_series(octant, &f, &ep);
751      if inverse {
752        inv_float_of(sign, &r)
753      } else {
754        float_of(sign, r.0, &r.1)
755      }
756    },
757  }
758}

788/// `\__fp_tan_o:w`, `\__fp_cot_o:w` (`\__fp_tan_series_o:NNwwww`).
789pub(super) fn tan_like(unit: Unit, x: Fp, cot: bool) -> R<Fp> {
790  match x {
791    Fp::Zero(..) => {
792      if cot {
793        Err(Decline("division by zero: cot(0)"))
794      } else {
795        Ok(x)
796      }
797    },
798    Fp::Inf(_) => Err(Decline("invalid: a trigonometric function of inf")),
799    Fp::Normal { sign, exp, mant } => {
800      let (first, base) = if cot { (Sign::Neg, 3) } else { (Sign::Pos, 1) };
801      let (octant, f, ep) = reduce(unit, base, exp, mant)?;
802      let square = mul(&f, &f);
803      // the sign: `\if_meaning:w #1#2 2 \else: 0 \fi:`, reversed where `#3 / 2` is odd
804      let same = first == sign;
805      let mut neg = same;
806      if rdiv(octant, 2).rem_euclid(2) == 1 {
807        neg = !neg;
808      }
809      let result_sign = if neg { Sign::Neg } else { Sign::Pos };
810      let mut n = mul_sub_back(&[0, 0, 1527, 3493, 856, 7059], &square, &TAN_NUM[0]);
811      for c in &TAN_NUM[1..] {
812        n = mul_sub_back(&n, &square, c);
813      }
814      let numerator = ep_mul(&(0, n), &ep);
815      let mut d = mul_sub_back(&[0, 7, 258, 681, 9408, 4706], &square, &TAN_DEN[0]);
816      for c in &TAN_DEN[1..] {
817        d = mul_sub_back(&d, &square, c);
818      }
819      let denominator: Ep = (0, d);
820      let (e, r) = if rdiv(octant - 1, 2).rem_euclid(2) == 1 {
821        ep_div(&denominator, &numerator)
822      } else {
823        ep_div(&numerator, &denominator)
824      };
825      float_of(result_sign, e, &r)
826    },
827  }
828}

832/// `\__fp_atan_combine_o:NwwwwwN`
833fn atan_combine(sign: Sign, octant: i64, s: &Fixed, x: &Fixed, ep: &Ep, unit: Unit) -> R<Fp> {
834  let finish = |base: i64, f: &Fixed| -> R<Fp> {
835    match unit {
836      Unit::Rad => float_of(sign, base, f),
837      // \__fp_fixed_to_float_rad_o:wN
838      Unit::Deg => {
839        let g = mul(f, &[5729, 5779, 5130, 8232, 876, 7981]);
840        float_of(sign, base + 2, &g)
841      },
842    }
843  };
844  if octant == 0 {
845    finish(ep.0, &mul(s, &ep.1))
846  } else {
847    let p = mul(s, x);
848    // \__fp_atan_combine_aux:ww
849    let q = fixed::mul_short(&MINI_PI, [rdiv(octant, 2), 0, 0]);
850    let f = if octant.rem_euclid(2) == 1 {
851      sub(&q, &p)
852    } else {
853      fixed::add(&q, &p)
854    };
855    finish(0, &f)
856  }
857}

859/// `\__fp_atan_auxi:ww`, `\__fp_atan_Taylor_loop:www`
860fn atan_series(ep: &Ep) -> (Fixed, Fixed) {
861  let x = ep_to_fixed(ep);
862  let square = mul(&x, &x);
863  let mut s: Fixed = [0; 6];
864  let mut n = 39;
865  while n != -1 {
866    s = mul_sub_back(&s, &square, &fixed::div_int(&ONE_FIXED, n));
867    n -= 2;
868  }
869  (s, x)
870}

872/// `\__fp_atan_test_o:NwwNwwN`: `atan2` of two extended numbers with signs.
873fn atan_test(s1: Sign, y: &Ep, s2: Sign, x: &Ep, unit: Unit) -> R<Fp> {
874  let reversed = fixed::ep_compare(y, x) > 0;
875  let (num, den) = if reversed { (x, y) } else { (y, x) };
876  // \__fp_atan_div:wnwwnw
877  let scale = match den.0 - num.0 {
878    0 => 100,
879    1 => 10,
880    _ => 1,
881  };
882  let near = 41421 * den.1[0] < num.1[0] * 1000 * scale;
883  let (mut octant, quotient) = if near {
884    // \__fp_atan_near:wwwn, \__fp_atan_near_aux:wwn
885    let a = ep_to_fixed(&(num.0 - den.0, num.1));
886    let sum = fixed::add(&a, &den.1);
887    let diff = sub(&den.1, &a);
888    (1, ep_div(&(0, diff), &(0, sum)))
889  } else {
890    (0, ep_div(num, den))
891  };
892  let (s, xfix) = atan_series(&quotient);
893  if reversed {
894    octant = 3 - octant;
895  }
896  if s2 == Sign::Neg {
897    octant = 7 - octant;
898  }
899  atan_combine(s1, octant, &s, &xfix, &quotient, unit)
900}

903fn atan_inf(sign: Sign, octant: i64, x_sign: Sign, unit: Unit) -> R<Fp> {
904  let octant = if x_sign == Sign::Neg {
905    7 - octant
906  } else {
907    octant
908  };

912/// `\__fp_atanii_o:Nww`: `atan(y, x)`.
913pub(super) fn atan2(unit: Unit, y: Fp, x: Fp) -> R<Fp> {
914  match (y, x) {
915    (Fp::Normal { sign: sy, exp: ey, mant: my }, Fp::Normal { sign: sx, exp: ex, mant: mx }) => {
916      let py: Ep = (i64::from(ey), fixed_of_mant(my));

920    _ => {

923        std::cmp::Ordering::Greater => 4,
924        std::cmp::Ordering::Less => 0,
925      };
926      atan_inf(y.sign(), octant, x.sign(), unit)
927    },

931/// `\__fp_asin_auxi_o:NnNww`: the fixed `sqrt(1-x^2)` by `\__fp_ep_isqrt:wwn`, then `atan`.
932fn asin_auxi(unit: Unit, reverse: bool, sign: Sign, ep: &Ep) -> R<Fp> {
933  let x = ep_to_fixed(ep);
934  // \__fp_asin_isqrt:wn
935  let diff = sub(&ONE_FIXED, &x);
936  let sum = fixed::add_one(&x);
937  let product = ep_mul(&(0, sum), &(0, diff));
938  let isqrt = fixed::ep_isqrt(&product);
939  let y = ep_to_ep(&ep_mul(&isqrt, ep));
940  let one: Ep = (1, [1000, 0, 0, 0, 0, 0]);
941  if reverse {
942    atan_test(Sign::Pos, &one, sign, &y, unit)
943  } else {
944    atan_test(sign, &y, Sign::Pos, &one, unit)
945  }
946}
