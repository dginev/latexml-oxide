48  // \documentstyle{<style>} — pretend the style is a class for
49  // RequirePackage purposes; fall back to amsppt (Perl L65-73).
50  //
51  // Perl uses `RequirePackage($style, ...) || RequirePackage('amsppt')`,
52  // where `||` checks the BOOLEAN success of InputDefinitions. The
53  // Rust port previously used `is_ok()` on `require_package`, but
54  // `require_package` returns `Ok(())` even when nothing loaded
55  // (the miss-handler fires `Warning:missing_file` and returns OK), so
56  // the fallback to `amsppt` never ran. Mirror Perl's semantics by
57  // probing the `<style>.sty_loaded` flag (or the raw equivalent)
58  // AFTER the call. Witness: arXiv:gr-qc/9507042 — paper has
59  // `\documentstyle{fic}` (fic.sty is a paper-local AmSTeX style
60  // derived from amsppt). Without the fallback, `\leftheadtext` and
61  // 23 other amsppt CSes stayed undefined.
62  //
63  // amstex.tex:1731 `\def\documentstyle#1{\W@{}\input #1.sty\relax}`: a style of its own with no binding that is on
64  // the search path — the paper's (math0402285's and math9806089's amams.sty, whose `\title#1\endtitle` the amsppt
65  // fallback lacks: a Fatal runaway, 0 words) — is read raw, as `\input` reads it. A style that `\input`s another
66  // style is that style's variation (gr-qc9507042's fic.sty `\input amsppt.sty`, math0406145's grubb-p.sty and
67  // 0803.2668's conm-p-mod.sty `\input ams-p.sty`, not in TeX Live): its changes are to the internals of the style it
68  // builds on, which the amsppt binding stands in for, as before. Perl requires a binding (`notex => 1`,
69  // AmSTeX.pool.ltxml:71) and falls back to amsppt (KNOWN_PERL_ERRORS #1051). Guard
70  // `perfect_kernel_batch78i::amstex_local_style_is_read`.

72    "<?latexml class='#1' amstex='true'?>",
73    after_digest => sub[whatsit] {
74      let style = whatsit.get_arg(1).map(|d| d.to_string()).unwrap_or_default();

77        && !binding_exists(&style, "sty")
78        && let Some(path) = find_file(&s!("{style}.sty"), None)
79        && !inputs_a_style(&path)

81        let mut input = vec![T_CS!("\\input")];

83        input.push(T_CS!("\\relax"));

796  if x_equals(t, &T_CS!("\\cr")) || x_equals(t, &T_CS!("\\crcr")) || x_equals(t, &T_CS!("\\\\")) {
797    return true;
798  }
799  if depth >= 4 {
800    return false;
801  }
802  let Ok(Some(def)) = lookup_definition(t) else {
803    return false;
804  };
805  if !def.is_expandable()
806    || def
807      .get_parameters()
808      .is_some_and(|p| !p.get_parameters().is_empty())
809  {
810    return false;
811  }
812  let Some(ExpansionBody::Tokens(body)) = def.get_expansion() else {
813    return false;
814  };
815  body
816    .unlist_ref()
817    .first()
818    .is_some_and(|first| first.get_catcode() == Catcode::CS && means_row_end(first, depth + 1))
819}

862/// AMS-TeX's `\tag` (amstex.tex:597-605, :629-900): read the tag text, unexpanded and brace-balanced, up to the
863/// display's closing `$$` (`\def\tag#1$$`) or the row's or environment's end (left in the input), and hand it to
864/// amsmath's `\tag` where an equation row holds it, else float it to the display (`\lx@equation@settag@float`). The
865/// `$$` form's `#1` is a non-`\long` argument: a `\par` (at any depth) ends the call with TeX's runaway error, and a
866/// `}` closing a group opened before it raises TeX's "extra }" error, backs the `}` up behind an inserted `\par` and
867/// goes on scanning, so the `\par` raises the runaway error too (tex.web §395-396); TeX drops the call and its tag, and
868/// the input goes on with `\par }`. A row's tag takes no argument (`\def\tag{\global\tag@true…&&}` in `\align`,
869/// amstex.tex:633-635; `\def\tag{&}` in `\gather`, :797; nor does a LaTeX equation row's): there a `\par` outside the
870/// tag's own groups, or such a `}`, ends the tag cell with the text read so far; a `\par` inside a group of the tag
871/// that closes stays in the text; a row's or environment's end read while a group of the tag is open stops the scan
872/// there, and the tag's groups are closed ([`close_runaway_row_tag`]; OXIDIZED_DESIGN_DIVERGENCES #575, where TeX errs
873/// on these shapes). A `$` closing inline math inside a row's tag cell or a group of the tag is never the start of
874/// `$$` (`\tag{$a$$b$}\\`), and the document's end is never read into the tag ([`amstex_tag_document_end`]). In a
875/// LaTeX equation row a braced `\tag{…}` (`\tag*{…}`) is amsmath's,
876/// its argument only (an `\input amstex` LaTeX document, or one whose `\NoBlackBoxes` autoloads the AmSTeX pool,
877/// tex.rs).
878fn amstex_tag() -> Result<Vec<Token>> {
879  let in_equation = in_equation_row();
880  if in_equation && amsmath_braced_tag_follows()? {
881    return Ok(vec![T_CS!("\\lx@amstex@ams@tag")]);
882  }
883  // in a row (`$$\align` is amsmath's `aligned` inside the display, opened in its math) the text runs to the row's
884  // end; in the display itself, to its `$$` (`$$\tag 1 \pmatrix a\\b\endpmatrix$$` puts the matrix in the tag, as TeX
885  // does); a row's tag cell (an AMS-TeX alignment's, or a LaTeX equation row's) is no argument
886  let in_row = in_equation || alignment_opened_in_current_math();
887  let mut text: Vec<Token> = Vec::new();
888  let mut depth = 0usize;
889  let mut delimited = false;
890  // in a row cell, inside a group of the tag: where the first `\par` was read (the index into `text`, and the depth
891  // there), while that depth's group is still open
892  let mut par_at: Option<(usize, usize)> = None;
893  // inside a row's tag cell, or a group of the tag: whether a `$` opened inline math, so the `$` closing it is never
894  // read as the start of `$$` (`\tag{$a$$b$}\\` is two inline maths)
895  let mut inline_math = false;
896  loop {
897    // TeX abandons a call whose argument runs away (tex.web §392, §395-396): the tag and its text are dropped (a row's
898    // tag keeps what it read, `close_runaway_row_tag`)
899    let Some(t) = read_token()? else {
900      Error!("expected", "$$", "File ended while scanning use of \\tag");
901      if in_row {
902        close_runaway_row_tag(&mut text, depth, par_at, None);
903        break;
904      }
905      return Ok(Vec::new());
906    };
907    if t == T_CS!("\\par") {
908      if in_row && depth > 0 {
909        // a `\par` inside the tag's own group stays in its text if the group closes before the row ends
910        // (`\tag{3\par}\\`)
911        par_at.get_or_insert((text.len(), depth));
912        text.push(t);
913        continue;
914      }
915      unread_one(t);
916      if in_row {
917        break;
918      }
919      Error!(
920        "unexpected",
921        "\\par",
922        "Paragraph ended before \\tag was complete"
923      );
924      return Ok(Vec::new());
925    }
926    // the document's end: the `$$` form's argument runs away there (tex.web §336-339: the `\outer` token is backed up
927    // and the call dropped); in a row it ends the row and the environment, as below
928    if !in_row && amstex_tag_document_end(&t) {
929      Error!(
930        "unexpected",
931        &t.to_string(),
932        "Forbidden control sequence found while scanning use of \\tag"
933      );
934      unread_one(t);
935      return Ok(Vec::new());
936    }
937    // where `$` parity is tracked (a row's tag cell, a group of the tag)
938    let math_tracked = in_row || depth > 0;
939    // `$$`, a display end, or (in a row) the row's or environment's end
940    let ends = if t == T_MATH!() {
941      if math_tracked && inline_math {
942        false
943      } else {
944        match read_token()? {
945          Some(next) => {
946            let display_end = next == T_MATH!();
947            unread_one(next);
948            display_end
949          },
950          None => false,
951        }
952      }
953    } else {
954      amstex_tag_display_end(&t)
955        || (in_row && (amstex_tag_row_end(&t) || amstex_tag_document_end(&t)))
956    };
957    if depth == 0 {
958      if ends {
959        if t == T_MATH!() {
960          delimited = true;
961        }
962        unread_one(t);
963        break;
964      } else if t.get_catcode() == Catcode::END {
965        // an enclosing group's `}` ends a row's tag cell; it is never the tag's
966        if in_row {
967          unread_one(t);
968          break;
969        }
970        // §395: back up the `}`, insert `\par`, go on scanning; §396: the `\par` ends the call
971        Error!("unexpected", "}", "Argument of \\tag has an extra }");
972        Error!(
973          "unexpected",
974          "\\par",
975          "Paragraph ended before \\tag was complete"
976        );
977        unread_vec(vec![T_CS!("\\par"), t]);
978        return Ok(Vec::new());
979      }
980    } else if ends && in_row {
981      // a row's or environment's end inside a group of the tag: the group cannot close before the row ends (TeX's
982      // `\cr` there is §1126-1127's "Missing } inserted"; `\align`'s argument runs away, amstex.tex:667, :687), so the
983      // scan stops at once and never reads past it (`\tag{5\\ c&=d\endalign$$`, also inside an enclosing `\vbox{…}`)
984      if par_at.is_some() {
985        Error!(
986          "unexpected",
987          "\\par",
988          "Paragraph ended before \\tag was complete"
989        );
990      } else {
991        // one per group closed, as TeX's `\cr` comes back after each inserted `}`
992        for _ in 0..depth {
993          Error!(
994            "expected",
995            "}",
996            "Missing } inserted before the row's end in \\tag"
997          );
998        }
999      }
1000      close_runaway_row_tag(&mut text, depth, par_at, Some(t));
1001      break;
1002    }
1003    match t.get_catcode() {
1004      Catcode::BEGIN => depth += 1,
1005      Catcode::END => {
1006        depth -= 1;
1007        // a `\par` read in the group just closed is the group's (`\tag{{3\par}5\\`)
1008        if par_at.is_some_and(|(_, at)| at > depth) {
1009          par_at = None;
1010        }
1011        // back at the `$$` form's own level, where `$$` is matched as tokens
1012        if depth == 0 && !in_row {
1013          inline_math = false;
1014        }
1015      },
1016      Catcode::MATH if math_tracked => inline_math = !inline_math,
1017      _ => {},
1018    }
1019    text.push(t);
1020  }
1021  // `\def\tag#1$$`: a delimited argument that is one balanced group loses its braces (tex.web §400), so
1022  // `\tag{"A"}$$` is the quoted form; a row's tag cell is no argument
1023  if delimited && is_one_group(&text) {
1024    text = text[1..text.len() - 1].to_vec();
1025  }
1026  // `\tagform@`'s `\ignorespaces#1\unskip`
1027  while text
1028    .first()
1029    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
1030  {
1031    text.remove(0);
1032  }
1033  while text
1034    .last()
1035    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
1036  {
1037    text.pop();
1038  }
1039  // `\maketag@@@"#1"#2\maketag@`: a quoted tag is set as it is, without parentheses, and the rest dropped
1040  let quoted = text.first().is_some_and(|t| *t == T_OTHER!("\""));
1041  if quoted {
1042    let close = text
1043      .iter()
1044      .skip(1)
1045      .position(|t| *t == T_OTHER!("\""))
1046      .map_or(text.len(), |i| i + 1);
1047    text = text[1..close].to_vec();
1048  } else if x_equals(&T_CS!("\\ifmathtags@"), &T_CS!("\\iftrue")) {
1049    text.insert(0, T_MATH!());
1050    text.push(T_MATH!());
1051  }
1052  let mut out = Vec::new();
1053  if in_equation {
1054    out.push(T_CS!("\\lx@amstex@ams@tag"));
1055    if quoted {
1056      out.push(T_OTHER!("*"));
1057    }
1058    out.push(T_BEGIN!());
1059    out.extend(text);
1060    out.push(T_END!());
1061  } else {
1062    out.extend([T_CS!("\\lx@equation@settag@float"), T_BEGIN!()]);
1063    if quoted {
1064      out.extend([T_CS!("\\let"), T_CS!("\\fnum@equation"), T_CS!("\\relax")]);
1065    } else if !IsDefined!(&T_CS!("\\fnum@equation")) {
1066      // no LaTeX pool (a plain `\input amstex` document): the tag is AMS-TeX's `\tagform@`, `(#1)` (amstex.tex:595),
1067      // as LaTeX's `\fnum@equation` (`\@eqnnum`) sets it
1068      out.extend([
1069        T_CS!("\\def"),
1070        T_CS!("\\fnum@equation"),
1071        T_BEGIN!(),
1072        T_OTHER!("("),
1073      ]);
1074      out.extend([T_CS!("\\theequation"), T_OTHER!(")"), T_END!()]);
1075    }
1076    out.extend([T_CS!("\\lx@ams@tag@text"), T_BEGIN!()]);
1077    out.extend(text);
1078    out.extend([T_END!(), T_CS!("\\lx@make@tags"), T_BEGIN!()]);
1079    out.extend(Explode!("equation"));
1080    out.extend([T_END!(), T_END!()]);
1081  }
1082  Ok(out)
1083}

1085/// The recovery of a row tag cell whose group runs away: a row's or environment's end (`current`) read inside a group
1086/// of the tag, or the file's end. The text from the earlier of `current` and the first `\par` read at a depth still
1087/// open (`par_at`: the index into `text` and the depth there) is put back, and the groups open there are closed, so the
1088/// row, the alignment and every later paragraph are read again as if the tag had ended there. TeX's `\align` reads its
1089/// body as a non-`\long` argument (amstex.tex:667, :687 `\csname align \endcsname#1\endalign`) and drops it ("Paragraph
1090/// ended before \align was complete", then "Missing $ inserted" and "Display math should end with $$" for the `\par`
1091/// in the display), OXIDIZED_DESIGN_DIVERGENCES #575.
1092fn close_runaway_row_tag(
1093  text: &mut Vec<Token>,
1094  depth: usize,
1095  par_at: Option<(usize, usize)>,
1096  current: Option<Token>,
1097) {
1098  let (open, mut rest) = match par_at {
1099    Some((at, open)) => (open, text.split_off(at)),
1100    None => (depth, Vec::new()),
1101  };
1102  rest.extend(current);
1103  if !rest.is_empty() {
1104    unread_vec(rest);
1105  }
1106  // the kept text's unclosed `{`s were counted when read and are counted again when its expansion is read with the
1107  // added `}`s: retract its braces, or the alignment's brace ledger stays raised and the row's `\\` is no row end
1108  retract_scanned_braces(text);
1109  text.extend(std::iter::repeat_n(T_END!(), open));
1110}

1112/// Whether amsmath's `\tag{…}` or `\tag*{…}` follows (its next non-space token a `{`, or a `*` and then one): the
1113/// tokens read are put back either way.
1114fn amsmath_braced_tag_follows() -> Result<bool> {
1115  let mut seen: Vec<Token> = Vec::new();
1116  let mut braced = false;
1117  let mut starred = false;
1118  while let Some(t) = read_token()? {
1119    let catcode = t.get_catcode();
1120    seen.push(t);
1121    match catcode {
1122      Catcode::SPACE => {},
1123      Catcode::BEGIN => {
1124        braced = true;
1125        break;
1126      },
1127      _ if !starred && seen.last().is_some_and(|t| *t == T_OTHER!("*")) => starred = true,
1128      _ => break,
1129    }
1130  }
1131  unread_vec(seen);
1132  Ok(braced)
1133}

1135/// Whether `tokens` is one balanced group: a `{` whose matching `}` is the last token.
1136fn is_one_group(tokens: &[Token]) -> bool {
1137  if tokens.len() < 2 || tokens[0].get_catcode() != Catcode::BEGIN {
1138    return false;
1139  }
1140  let mut depth = 0usize;
1141  for (i, t) in tokens.iter().enumerate() {
1142    match t.get_catcode() {
1143      Catcode::BEGIN => depth += 1,
1144      Catcode::END => {
1145        depth -= 1;
1146        if depth == 0 {
1147          return i == tokens.len() - 1;
1148        }
1149      },
1150      _ => {},
1151    }
1152  }
1153  false
1154}

1156/// Tie the `ltx:tags` `\lx@tags@float` just floated (the last `ltx:tags` child of the nearest ancestor holding one) to
1157/// the math alignment row it was built in, if any: both get the same `_tags` mark.
1158fn tie_floated_tags_to_row(document: &mut Document) -> Result<()> {
1159  let qname = |n: &Node| document::get_node_qname(n);
1160  let mut row = Some(document.get_node().clone());
1161  while let Some(n) = row.take() {
1162    if qname(&n) == pin!("ltx:XMRow") {
1163      row = Some(n);
1164      break;
1165    }
1166    row = n.get_parent();
1167  }
1168  let Some(mut row) = row else {
1169    return Ok(());
1170  };
1171  let mut ancestor = row.get_parent();
1172  while let Some(a) = ancestor {
1173    let last_tags = a
1174      .get_child_elements()
1175      .into_iter()
1176      .rev()
1177      .find(|c| qname(c) == pin!("ltx:tags"));
1178    if let Some(mut tags) = last_tags {
1179      if tags.get_attribute("_tags").is_none() {
1180        let mark = lookup_int("lx@tags@row@mark") + 1;
1181        assign_value("lx@tags@row@mark", Stored::Int(mark), Some(Scope::Global));
1182        let mark = mark.to_string();
1183        document.set_attribute(&mut tags, "_tags", &mark)?;
1184        document.set_attribute(&mut row, "_tags", &mark)?;
1185      }
1186      break;
1187    }
1188    ancestor = a.get_parent();
1189  }
1190  Ok(())
1191}
