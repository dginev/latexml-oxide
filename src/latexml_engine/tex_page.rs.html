5use crate::prelude::*;
6LoadDefinitions!({
7  //%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
8  // Page Family of primitive control sequences
9  //%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
10
11  //======================================================================
12  // Parameters for page layout
13  //----------------------------------------------------------------------
14  // \hoffset          pd is a value added to the default 1-inch left margin.
15  // \voffset          pd is a value added to the default 1-inch top margin.
16  // \topskip          pg is special glue added before the first box on each page.
17  // \pagedepth        iq is the actual depth of the last box on the main page.
18  // \pagetotal        iq is the accumulated height of the current page.
19  // \maxdepth         pd is the maximum depth of boxes on the main page.
20  // \vsize            pd is the desired height of the current page.
21  // \pagegoal         iq is the desired height of the current page.
22  // \pageshrink       iq is the amount of finite shrinkability in the current page.
23  // \pagestretch      iq is the amount of finite stretchability in the current page.
24  // \pagefilllstretch iq is the amount of third-order infinite stretchability in the current page.
25  // \pagefillstretch  iq is the amount of second-order infinite stretchability in the current page.
26  // \pagefilstretch   iq is the amount of first-order infinite stretchability in the current page.
27
28  DefRegister!("\\hoffset", Dimension!("0"));
29  DefRegister!("\\voffset", Dimension!("0"));
30  DefRegister!("\\topskip", Glue!("10pt"));
31  DefRegister!("\\pagedepth", Dimension::new(0));
32  DefRegister!("\\pagetotal", Dimension::new(0));
33  DefRegister!("\\maxdepth", Dimension!("4pt"));
34  DefRegister!("\\vsize", Dimension!("8.9in"));
35
36  // tex.web §982/§987: `page_goal` is `max_dimen` only while the current
37  // page is EMPTY; the moment the page builder contributes a box it becomes
38  // `\vsize` (minus insertions). Our model has no page builder, so a single
39  // standing value must serve every "free space on this page" probe: Perl
40  // uses Dimension(0) (fullwidth.sty:243-273 `\fwd@freepagevspace` computes
41  // 0−0 and retries `\vfill\eject` forever — SHARED loop), batch 28 used
42  // `\maxdimen` (fullwidth's `\ifdimequal{\pagegoal}{\maxdimen}` branch
43  // then took `\vsize`, but fillwith.sty:319 `\dim_until_do … \pagegoal -
44  // \footskip - \pagetotal …` stacked ~1300 line coffins toward a 16384pt
45  // goal and l3coffins' quadratic corner naming ran to the TokenLimit).
46  // `\vsize` — the mid-page value — satisfies both: fullwidth's free space
47  // is `\vsize − \pagetotal`, fillwith fills one page (~50 coffins with a
48  // real `\strutbox`, latex_constructs_rust_only.rs). Guard:
49  // `perfect_kernel_batch54::pagegoal_is_vsize_and_strutbox_is_real`.
50  DefRegister!("\\pagegoal", Dimension!("8.9in"),
51  getter => sub[_args] {
52    // Follow `\vsize` (classes set it) unless the document assigned
53    // `\pagegoal` itself.
54    match lookup_value("pagegoal_assigned") {
55      Some(Stored::Dimension(d)) => Some(RegisterValue::Dimension(d)),
56      _ => lookup_dimension_cs("\\vsize", true).map(RegisterValue::Dimension),
57    }
58  },
59  setter => sub[value, scope, _args] {
60    if let RegisterValue::Dimension(d) = value {
61      assign_value("pagegoal_assigned", Stored::Dimension(d), scope);
62    }
63  });
64  DefRegister!("\\pagestretch", Dimension::new(0));
65  DefRegister!("\\pagefilstretch", Dimension::new(0));
66  DefRegister!("\\pagefillstretch", Dimension::new(0));
67  DefRegister!("\\pagefilllstretch", Dimension::new(0));
68  DefRegister!("\\pageshrink", Dimension::new(0));
69
70  //======================================================================
71  // Usable for things line \clearpage, etc.
72  //
73  // OXIDIZED_DESIGN #178: a shipped page advances `\c@page` (latex.ltx:15271
74  // `\@outputpage`'s `\global\advance\c@page\@ne`; plain.tex `\advancepageno`).
75  // Neither engine has an output routine, and Perl's marker leaves the counter
76  // at 1 forever, so the standard "pad to page N" idiom
77  // `\loop\ifnum\value{page}<N \null\clearpage\repeat` (knowledge.tex:803-809)
78  // never ends (Perl hangs; Rust's box-cycle guard made it a Fatal). Counting
79  // the page markers is the honest model of a page count. Guard:
80  // `perfect_kernel_batch54::clearpage_advances_the_page_counter`.
81  // The register is the page number's owner: LaTeX's `\c@page` once the format
82  // is loaded (`\count0` from the dump, latex.ltx:14891 `\countdef\c@page=0`; a
83  // counter of its own on the NODUMP branch, sect05.rs `NewCounter!("page")`),
84  // plain's `\pageno` otherwise (`\count0` from the dump, plain.tex
85  // `\countdef\pageno=0`; its own register on the NODUMP branch, plain_base.rs).
86  // Spelled `\c@page` in a plain document, `\eject` (plain_constructs.rs)
87  // autoloaded LaTeX.pool mid-document, which rebinds `\end`, and its closing
88  // `\end` raised "`\endgroup` Attempt to close a group that switched to mode
89  // vertical" (stopped full-arXiv run 329: 0911.4241, 1001.3079, 1312.4456,
90  // 1407.6634, hep-th9310069, chao-dyn9706006; 27 of 16,333 papers). Guards
91  // `perfect_kernel_batch61::plain_document_stays_plain`,
92  // `perfect_kernel_batch61::clearpage_advances_the_register_c_at_page_names`.
93  DefMacro!("\\lx@newpage", "\\lx@newpage@mark\\lx@advance@page");
94  DefMacro!("\\lx@advance@page", sub[_args] {
95    let page = if lookup_bool("LaTeX.pool_loaded") { "\\c@page" } else { "\\pageno" };
96    Ok(Tokens::new(vec![
97      T_CS!("\\global"),
98      T_CS!("\\advance"),
99      T_CS!(page),
100      T_OTHER!("1"),
101      T_CS!("\\relax"),
102    ]))
103  });
104  DefConstructor!("\\lx@newpage@mark", "^<ltx:pagination role='newpage'/>");
105});
