15/// tex.web §1083 (`vpack(…, exactly | additional)`): `\vbox to <h>` is `h` high, a `spread <s>` box its contents'
16/// height plus `s` — Perl `setHeight` (TeX_Box.pool.ltxml:549-552, :571-574), which the port had left out: `\vbox to
17/// 50pt{\hbox{A}}` measured 6.83pt. A `\vtop` keeps its first box's height and takes the rest into its depth (§1087:
18/// `d := x + d - h`, the packed box's height and depth less the first item's height): `\vtop to 30pt{\hbox{A}}` is
19/// 6.83pt high and 23.17pt deep (Perl set the height, OXIDIZED_DESIGN_DIVERGENCES #420).
20fn set_specified_height(whatsit: &mut Whatsit, top: bool) {
21  let spec = whatsit.get_arg(1);
22  let to = match GetKeyVal!(spec, "to") {
23    Some(ArgWrap::Dimension(h)) => Some(*h),
24    _ => None,
25  };
26  let spread = match GetKeyVal!(spec, "spread") {
27    Some(ArgWrap::Dimension(s)) => Some(*s),
28    _ => None,
29  };
30  if to.is_none() && spread.is_none() {
31    return;
32  }
33  // `\vbox to` needs no measuring; a `spread` box its natural height, a `\vtop` its packed depth.
34  let (packed_height, packed_depth) = if top || to.is_none() {
35    packed_size(whatsit.get_arg(2))
36  } else {
37    Default::default()
38  };
39  let height = to.unwrap_or_else(|| packed_height.add(spread.unwrap_or_default()));
40  if top {
41    let first = match whatsit.get_height() {
42      Some(RegisterValue::Dimension(h)) => h,
43      _ => Dimension::default(),
44    };
45    whatsit.set_depth(height.add(packed_depth).subtract(first));
46  } else {
47    whatsit.set_height(height);
48  }
49}

51/// The contents of a `\vbox` or `\vtop` packed as `\vbox` packs them (tex.web §668 `vpack`): the height down to the
52/// last item's baseline and that item's depth. A `\vtop`'s contents are measured top-aligned, so they are measured
53/// here as a bottom-aligned copy.
54fn packed_size(contents: Option<&Digested>) -> (Dimension, Dimension) {
55  let dimension = |value: Option<RegisterValue>| match value {
56    Some(RegisterValue::Dimension(d)) => d,
57    _ => Dimension::default(),
58  };
59  match contents.map(Digested::data) {
60    Some(DigestedData::List(list)) => {
61      let mut bottom = list.borrow().clone();
62      bottom
63        .properties
64        .insert("vattach", Stored::String(pin!("bottom")));
65      bottom
66        .compute_size(SymHashMap::default())
67        .map(|(_, height, depth)| (height, depth))
68        .unwrap_or_default()
69    },
70    Some(_) => contents.map_or_else(Default::default, |item| {
71      (dimension(item.get_height()), dimension(item.get_depth()))
72    }),
73    None => Default::default(),
74  }
75}

92/// tex.web §1110 `unpackage`: `\unhbox`/`\unhcopy` put a box's list into the current list. A register filled by
93/// `\setbox0\hbox{…}` (LaTeX's `\sbox`, `\savebox`) holds the `\hbox` whatsit, whose list is its contents (argument 2);
94/// Perl puts the whatsit back whole (`Whatsit::unlist` returns itself), so `\unhbox0\unskip` could not reach the box's
95/// last glue, `\lastbox` after `\unhbox` took the whole box, and `\hbox{\unhcopy0}` kept a `to` width (KNOWN_PERL_ERRORS
96/// #434). Any other box goes back as it is. `\unhcopy` puts back copies (§1110 `copy_node_list`): a later `\lastbox` or
97/// `\wd` acts on them, not on the register's box. Put into a paragraph, the items are paragraph material
98/// (`in_paragraph`): digested inside the `\hbox`, a `\parbox` there — in a group's body — was restricted-box material,
99/// and one spliced at a paragraph's start was set apart as a paragraph of its own; what a box inside holds stays
100/// restricted (`\rotatebox{90}{\parbox…}`). In math mode, where TeX refuses to unpack a box ("Incompatible list
101/// can't be unboxed"), the box goes back whole, as Perl does: LaTeXML is in math mode inside `$\emph{…}$`, where TeX is
102/// not.
103fn hlist_of(stuff: &Digested, copy: bool) -> Vec<Digested> {
104  if lookup_bool_sym(pin!("IN_MATH")) {
105    return stuff.unlist();
106  }
107  let items = match stuff.data() {
108    DigestedData::Whatsit(w)
109      if w
110        .try_borrow()
111        .is_ok_and(|w| w.get_definition().get_cs_name().as_ref() == "\\hbox") =>
112    {
113      w.borrow()
114        .get_arg(2)
115        .map(Digested::unlist)
116        .unwrap_or_default()
117    },
118    _ => stuff.unlist(),
119  };
120  let in_paragraph = lookup_string_from_sym(pin!("MODE")) == "horizontal";
121  items
122    .into_iter()
123    .map(|item| {
124      let mut item = spliced(item, in_paragraph, copy);
125      // In a paragraph each is an item of its line (tex.web §1076): a minipage keeps its own (vertical) mode, and
126      // the paragraph's repack (stomach.rs `repack_paragraph`) stopped at it, stacking tcolorbox's `sidebyside`
127      // halves (`\unhbox\tcb@upperbox\kern…\unhbox\tcb@lowerbox`) instead of setting them side by side.
128      if in_paragraph && matches!(item.data(), DigestedData::Whatsit(_)) {
129        item.set_property("in_hlist", true);
130      }
131      item
132    })
133    .collect()
134}

136/// tex.web §1110 `unpackage` for `\unvbox`/`\unvcopy` (§1109 `vmode+un_vbox`): the box's list goes into the current
137/// vertical list and the box itself — its dimensions, its `to` height, its being one item — is gone. A register
138/// filled by `\setbox0\vbox{…}` or `\vtop{…}` holds the whatsit, whose list is its contents (argument 2). Perl puts the
139/// whatsit back whole (`Whatsit::unlist` returns itself, TeX_Box.pool.ltxml:725-742), so a `\lastbox` after `\unvbox`
140/// took the whole box and a peel loop never ended: `\setbox\pb=\vbox{\unvbox\pb \global\setbox\lb=\lastbox}` took the
141/// box, left an empty `\vbox{}`, and took that again the next round, for ever (infernce.sty's `\@adjustPremises`, under
142/// every semantic.sty `\inference`: ~315 arXiv `oom:alloc_failed` Fatals, among them 1603.01635 and 2506.05832;
143/// KNOWN_PERL_ERRORS #840; the study's other `\lastbox` users, 1209.2208 and math0301370, fail elsewhere). `\unvcopy`
144/// puts back copies (§1110 `copy_node_list`).
145///
146/// The list is put back where it goes into a vertical box's own list (`\vbox`, `\vtop`, `\parbox` contents:
147/// [`in_vertical_box_list`]), where LaTeXML sets the box's lines as lines, bracketed by a pair of boundary marks
148/// ([`splice_mark`]): the box's end closed whatever its contents had opened (§1086 `vbox_group`: `end_graf`; a
149/// minipage's or `\parbox`'s `ltx:p`; a `\noindent` paragraph's `ltx:para`), and the end mark closes it in the new
150/// list, so what follows the spliced material starts a line of its own (without it: "Para two.More.", "MiniA" and
151/// "AfterA" side by side). Elsewhere — the main vertical list, a float's, footnote's or minipage's body — and in math
152/// the box goes back whole, as Perl's: spliced there, its hboxes ran together into one paragraph
153/// (OXIDIZED_DESIGN_DIVERGENCES #637; the shape needs R6, `\hbox` in vertical mode as a line, and a ruling: repro
154/// boxes-groups/box_primitives_unpack cases 8-9).
155fn vlist_of(stuff: &Digested, copy: bool) -> Vec<Digested> {
156  let is_vbox = |w: &Whatsit| {
157    matches!(
158      w.get_definition().get_cs_name().as_ref(),
159      "\\vbox" | "\\vtop"
160    )
161  };
162  match stuff.data() {
163    DigestedData::Whatsit(w)
164      if !lookup_bool_sym(pin!("IN_MATH"))
165        && w.try_borrow().is_ok_and(|w| is_vbox(&w))
166        && in_vertical_box_list() =>
167    {
168      let mut items: Vec<Digested> = w
169        .borrow()
170        .get_arg(2)
171        .map(Digested::unlist)
172        .unwrap_or_default();
173      if copy {
174        items = items
175          .into_iter()
176          .map(|item| spliced(item, false, true))
177          .collect();
178      }
179      // A list of nothing but the marks of earlier splices brackets nothing: put back empty, as the box was.
180      if items.iter().all(is_splice_mark) {
181        return Vec::new();
182      }
183      let serial = SPLICE_SERIAL.get() + 1;
184      SPLICE_SERIAL.set(serial);
185      if let (Some(begin), Some(end)) = (
186        splice_mark(&T_CS!("\\lx@splice@begin"), serial),
187        splice_mark(&T_CS!("\\lx@splice@end"), serial),
188      ) {
189        items.insert(0, begin);
190        items.push(end);
191      }
192      items
193    },
194    _ => stuff.unlist(),
195  }
196}

198/// Whether the current list is a vertical box's own (`VBoxContents`: `\vbox`, `\vtop`, `\parbox`, a `p` column's
199/// cell): the frame its mode was bound in (`begin_mode`'s `BOUND_MODE`) is the one the box reader marked
200/// (`lx@vbox@list`, base_utilities.rs `predigest_list_in_mode`). A group inside the box is still the box's list; a
201/// minipage, a float, a footnote begun in it are lists of their own.
202fn in_vertical_box_list() -> bool {
203  for frame in 0..undo_depth() {
204    if is_value_bound("BOUND_MODE", Some(frame)) {
205      return is_value_bound("lx@vbox@list", Some(frame));
206    }
207  }
208  false
209}

219/// The internal attribute by which a begin mark records the element it was absorbed in — a value, not a node handle,
220/// which `rename_node` or a removal would leave dangling —, removed again by the end mark of the splice that set it
221/// (and stripped from the output with every `_` attribute should that end mark not come).
222const SPLICE_ATTRIBUTE: &str = "_lx_splice";

224/// A boundary mark of a list `\unvbox` spliced ([`vlist_of`]), `cs` `\lx@splice@begin` or `\lx@splice@end`: no node in
225/// TeX, so the list-tail primitives see through it (stomach.rs `is_list_tail_transparent`) and it reverts as nothing.
226/// Absorbed, the begin mark records the element the list starts in; the end mark closes back to it whatever the list
227/// opened and left open — only to that element, only while it is open under the same name, only if what lies between
228/// may auto-close, and nothing at all for an end mark whose begin mark was not absorbed (a `\vsplit` took it).
229fn splice_mark(cs: &Token, serial: i64) -> Option<Digested> {
230  let definition = lookup_definition(cs).ok().flatten()?;
231  let mut mark = Whatsit {
232    definition,
233    ..Whatsit::default()
234  };
235  // What a constructor's invocation gives every whatsit (constructor.rs): its font, mode and math flag; the sizing
236  // pass reads the font.
237  let font = lookup_font().unwrap_or_else(|| Rc::new(Font::text_default()));
238  mark.set_property("font", Stored::Font(font));
239  mark.set_property("isMath", false);
240  mark.set_property("mode", lookup_string_from_sym(pin!("MODE")));
241  mark.set_property("alignmentSkippable", true);
242  mark.set_property("lx@splice", Stored::Int(serial));
243  mark.set_property("reversion", Tokens!());
244  Some(Digested::from(mark))
245}

260/// The begin mark's absorption ([`splice_mark`]): record the current element, by the [`SPLICE_ATTRIBUTE`] it already
261/// carries (an enclosing splice began there) or one it now gets.
262fn absorb_splice_begin(document: &mut Document, serial: i64) -> Result<()> {
263  let Some(mut node) = document.get_element() else {
264    return Ok(());
265  };
266  let value = match node.get_attribute(SPLICE_ATTRIBUTE) {
267    Some(value) => value,
268    None => {
269      let value = serial.to_string();
270      node.set_attribute(SPLICE_ATTRIBUTE, &value)?;
271      value
272    },
273  };
274  let qname = document::get_node_qname(&node);
275  SPLICE_STARTS.with_borrow_mut(|starts| starts.insert(serial, (value, qname)));
276  Ok(())
277}

279/// The end mark's absorption ([`splice_mark`]): close back to the element its begin mark recorded.
280fn absorb_splice_end(document: &mut Document, serial: i64) -> Result<()> {
281  let Some((value, qname)) = SPLICE_STARTS.with_borrow_mut(|starts| starts.remove(&serial)) else {
282    return Ok(());
283  };
284  let Some(current) = document.get_element() else {
285    return Ok(());
286  };
287  let mut node = current.clone();
288  let mut closeable = true;
289  loop {
290    if node.get_attribute(SPLICE_ATTRIBUTE).as_deref() == Some(value.as_str()) {
291      break;
292    }
293    closeable &= document::can_auto_close(&node);
294    match node.get_parent() {
295      Some(parent) if parent.get_type() == Some(NodeType::ElementNode) => node = parent,
296      _ => return Ok(()),
297    }
298  }
299  if closeable && node != current && document::get_node_qname(&node) == qname {
300    document.close_to_node(&node, true)?;
301  }
302  // The begin mark that gave the element its value ends last of those sharing it (the splices nest): the record goes.
303  if value == serial.to_string() {
304    node.remove_attribute(SPLICE_ATTRIBUTE)?;
305  }
306  Ok(())
307}

309/// An item put back by `\unhbox` (`copy` false: itself, unless it must be marked as paragraph material) or `\unhcopy`
310/// (`copy`: a copy, arguments and body included). `in_paragraph` marks the item and the items of a group's body, never
311/// what a constructor's arguments hold, nor an environment's body, which is digested in its own mode (restricted
312/// horizontal unless declared: `{rotate}`'s `\parbox` stays a box's content, as typed; arXiv 2605.20645's
313/// `\rotatebox{90}{\parbox…}` shape).
314fn spliced(item: Digested, in_paragraph: bool, copy: bool) -> Digested {
315  if !copy && !in_paragraph {
316    return item;
317  }
318  match item.data() {
319    DigestedData::Whatsit(w) => match w.try_borrow() {
320      Ok(w) => {
321        let mut whatsit = w.clone();
322        if copy {
323          whatsit.args = std::mem::take(&mut whatsit.args)
324            .into_iter()
325            .map(|arg| arg.map(|arg| spliced(arg, false, true)))
326            .collect();
327        }
328        if let Some(Stored::Digested(body)) = whatsit.properties.get("body").cloned() {
329          // No declared mode: a group (`\lx@hidden@bgroup`), whose body shares the box's list; an environment's body
330          // has its own. Constructors that start a mode in `before_digest` (inline math, notes) pass too — harmless:
331          // only `\parbox` and `{minipage}` read the mark, marked anyway in a note and ignored in math.
332          let shares_list = whatsit
333            .get_definition()
334            .declared_mode()
335            .is_none_or(|declared| declared.mode.is_none());
336          whatsit.properties.insert(
337            "body",
338            Stored::Digested(spliced(body, in_paragraph && shares_list, copy)),
339          );
340        }
341        if in_paragraph {
342          whatsit.set_property("in_paragraph", true);
343        }
344        Digested::from(whatsit)
345      },
346      Err(_) => item.clone(),
347    },
348    DigestedData::List(list) => match list.try_borrow() {
349      Ok(list) => {
350        let mut list = list.clone();
351        list.boxes = std::mem::take(&mut list.boxes)
352          .into_iter()
353          .map(|item| spliced(item, in_paragraph, copy))
354          .collect();
355        Digested::from(list)
356      },
357      Err(_) => item.clone(),
358    },
359    _ => item,
360  }
361}

363/// `\unvbox`/`\unvcopy` of a list LaTeXML does not unpack (a `\vsplit` piece, a list without a mode: R2) goes back
364/// as one item at its contents' natural size: the split height, or a `\ht`/`\dp`/`\wd` set on the box, belongs to
365/// the box, not to the list `unpackage` puts back (tex.web §1110). `\vbox{\unvbox2}` of a piece split `to 12pt`
366/// measured 12pt; TeX's is the line's 6.83pt. Residual (R2): a piece of several lines is measured side by side.
367fn at_natural_size(stuff: Digested) -> Digested {
368  const BOX_SIZE: [&str; 3] = ["height", "depth", "width"];
369  if let DigestedData::List(list) = stuff.data()
370    && let Ok(list) = list.try_borrow()
371    && BOX_SIZE.iter().any(|key| list.properties.contains_key(key))
372  {
373    let mut natural = list.clone();
374    for key in BOX_SIZE
375      .iter()
376      .chain(&["cached_width", "cached_height", "cached_depth"])
377    {
378      natural.properties.remove(key);
379    }
380    return Digested::from(natural);
381  }
382  stuff
383}

385/// Perl: hackVBoxAttachment($box, $valign)
386/// Sets vattach on the box, with special handling for \halign alignment objects.
387///
388/// For \halign inside \vbox/\vtop, vattach must be set on the alignment's XML
389/// attributes (so it becomes an attribute on `<tabular>`), NOT on the box property
390/// (which would end up on `<para>` via insertBlock).
391///
392/// Perl's List() simplification returns single-item vertical lists unwrapped,
393/// so $box->getProperty('alignment') finds the alignment directly. In Rust,
394/// Lists are not simplified, so we must walk into children to find it.
395fn hack_vbox_attachment(whatsit: &mut Whatsit, valign: &'static str) {
396  if let Some(content_box) = whatsit.get_arg_mut(2)
397    && !set_halign_vattach(content_box, valign)
398  {
399    // No \halign alignment found — set vattach as property on the box
400    content_box.set_property("vattach", valign);
401  }
402}

404/// Walk into a Digested tree to find a \halign Whatsit with an 'alignment' property.
405/// If found, set vattach on the alignment's XML attributes and return true.
406/// Returns false if no \halign alignment was found.
407fn set_halign_vattach(digested: &Digested, valign: &str) -> bool {
408  match digested.data() {
409    DigestedData::Whatsit(w) => {
410      let w_ref = w.borrow();
411      if w_ref.get_property("alignment").is_some() {
412        // Check if this whatsit's definition is \halign
413        let def = w_ref.get_definition();
414        let is_halign = *def.get_cs_name() == *"\\halign";
415        if is_halign {
416          // Get the alignment property value and set vattach on it
417          if let Some(Cow::Borrowed(Stored::Digested(alignment_dig))) =
418            w_ref.get_property("alignment")
419            && let DigestedData::Alignment(ref alignment_cell) = *alignment_dig.data()
420          {
421            alignment_cell
422              .borrow_mut()
423              .get_xml_attributes_mut()
424              .insert(String::from("vattach"), String::from(valign));
425            return true;
426          }
427        }
428        // Has alignment but not \halign (e.g. tabular) — don't set vattach
429        return false;
430      }
431      false
432    },
433    DigestedData::List(list_cell) => {
434      // Walk children to find a \halign
435      let children = list_cell.borrow().unlist();
436      for child in children.iter() {
437        if set_halign_vattach(child, valign) {
438          return true;
439        }
440      }
441      false
442    },
443    _ => false,
444  }
445}

447/// Helper to get (width, height, depth) from a Digested node_box without mutable borrows.
448/// This avoids RefCell conflicts when called during the absorption pipeline.
449/// For simple TBox/Whatsit: reads the declared width/height/depth, else the cached ones.
450/// For Lists: sums child box widths and takes max height/depth.
451fn fobj_get_size(digested: &Digested) -> (Dimension, Dimension, Dimension) {
452  // Declared before cached: Perl's `\ht` setter writes `cheight` too (TeX_Box.pool.ltxml:620-628, Box.pm `setHeight`),
453  // so a size measured before it never wins (repro graphics-tikz/node_box_declared_size_over_measured; 2408.03209).
454  // A declared size stored as a String (diagbox, `\rotate[f]`, "1em") is no Dimension: the measured one stands.
455  fn read_dims(d: &Digested) -> (Dimension, Dimension, Dimension) {
456    d.with_properties(|props| {
457      let dim = |declared: &str, cached: &str| {
458        stored_dimension(props.get(declared))
459          .or_else(|| stored_dimension(props.get(cached)))
460          .unwrap_or_default()
461      };
462      (
463        dim("width", "cached_width"),
464        dim("height", "cached_height"),
465        dim("depth", "cached_depth"),
466      )
467    })
468  }
469  let dims = read_dims(digested);
470  if dims.0.value_of() != 0 || dims.1.value_of() != 0 || dims.2.value_of() != 0 {
471    return dims;
472  }
473  // If zero dims: for Lists, sum children's dimensions
474  if let DigestedData::List(list_cell) = digested.data()
475    && let Ok(list) = list_cell.try_borrow()
476  {
477    let mut total_w: i64 = 0;
478    let mut max_h: i64 = 0;
479    let mut max_d: i64 = 0;
480    for child in &list.boxes {
481      let (cw, ch, cd) = fobj_get_size(child);
482      total_w += cw.value_of();
483      max_h = max_h.max(ch.value_of());
484      max_d = max_d.max(cd.value_of());
485    }
486    return (
487      Dimension::new(total_w),
488      Dimension::new(max_h),
489      Dimension::new(max_d),
490    );
491  }
492  dims
493}

518    .get_attributes()
519    .into_iter()
520    .filter(|(k, _)| !k.starts_with('_'))
521    .collect();
522  // Perl L208: skip if clip-path is set
523  if nodeattr.contains_key("clip-path") {
524    return Ok(());

527  let is_svg_g = |n: &Node| -> bool { document::with_node_qname(n, |q| q == "svg:g") };

531  let mut nempty = 0;
532  for c in &children {
533    if is_svg_g(c) && element_nodes(c).is_empty() {
534      document.remove_node(c.clone());
535      nempty += 1;
536    }
537  }
538  if nempty > 0 {
539    children = element_nodes(node);
540  }

542  // Step 2: Pop leading children that completely mask parent attributes (Perl L218-228)
543  // If a leading child svg:g has collapsible attributes covering ALL of parent's attributes,
544  // the child "masks" the parent — move it before parent in the DOM.
545  let nodeattr_count = nodeattr.len();
546  let mut npopped = 0;
547  while !children.is_empty() && is_svg_g(&children[0]) {
548    let c = &children[0];
549    let mut nmasked = 0;
550    for (key, _val) in c.get_attributes() {
551      if !key.starts_with('_') && COLLAPSIBLE.contains(&key.as_str()) && nodeattr.contains_key(&key)
552      {
553        nmasked += 1;
554      }
555    }
556    if nmasked != nodeattr_count {
557      break; // child doesn't completely mask parent
558    }
559    // Move child before node in parent
560    let mut child = children.remove(0);
561    node.add_prev_sibling(&mut child)?;
562    npopped += 1;
563  }
564
565  // Step 3: Push trailing children that completely mask parent attributes (Perl L230-237)
566  let mut npushed = 0;
567  while !children.is_empty() && is_svg_g(children.last().unwrap()) {
568    let c = children.last().unwrap();
569    let mut nmasked = 0;
570    for (key, _val) in c.get_attributes() {
571      if !key.starts_with('_') && COLLAPSIBLE.contains(&key.as_str()) && nodeattr.contains_key(&key)
572      {
573        nmasked += 1;
574      }
575    }
576    if nmasked != nodeattr_count {
577      break; // child doesn't completely mask parent
578    }
579    // Move child after node in parent
580    let mut child = children.pop().unwrap();
581    node.add_next_sibling(&mut child)?;
582    npushed += 1;
583  }
584  if npopped > 0 || npushed > 0 {
585    children = element_nodes(node);
586  }
587
588  // Step 4: Remove redundant svg:g children (same attributes as parent, Perl L239-250)
589  let mut nredundant = 0;
590  for c in &children {
591    if is_svg_g(c) {
592      let mut is_same = true;
593      for (key, val) in c.get_attributes() {
594        if key.starts_with('_') {
595          continue;
596        }
597        // Perl L245-246: different value OR key is 'transform' → not redundant
598        if val != *nodeattr.get(&key).unwrap_or(&String::new()) || key == "transform" {
599          is_same = false;
600          break;
601        }
602      }
603      if is_same {
604        document.unwrap_nodes(c.clone())?;
605        nredundant += 1;
606      }
607    }
608  }
609  if nredundant > 0 {
610    children = element_nodes(node);
611  }
612
613  // Step 5: Merge single svg:g child into parent (Perl L254-270)
614  if children.len() == 1 && is_svg_g(&children[0]) {
615    let c = &children[0];
616    let mut av: HashMap<String, String> = HashMap::default();
617    let mut mergeable = true;
618    for (key, val) in c.get_attributes() {
619      if key.starts_with('_') || COLLAPSIBLE.contains(&key.as_str()) {
620        av.insert(key, val);
621      } else if key == "transform" {
622        // Compose transforms: parent's transform + child's transform
623        let composed = if let Some(parent_t) = nodeattr.get("transform") {
624          format!("{} {}", parent_t, val)
625        } else {
626          val
627        };
628        av.insert(key, composed);
629      } else {
630        mergeable = false;
631        break;
632      }
633    }
634    if mergeable {
635      for (key, val) in &av {
636        node.set_attribute(key, val)?;
637      }
638      document.unwrap_nodes(children[0].clone())?;
639    }
640  }
641
642  Ok(())
643}

661/// Compute the properties required for a framed something.
662///
663/// Port of Perl `framedProperties` (TeX_Box.pool.ltxml L69-89, #2829):
664/// consistent `framed`/`framecolor`/`backgroundcolor`/`cssstyle` attributes,
665/// plus `padtop`/`padbottom`/`padleft`/`padright` Dimension properties
666/// (margin + rule per side) that feed the whatsit size computation.
667/// `cssstyle` carries `padding:` whenever a margin is given, and
668/// `border-width:` only when the rule differs from the 0.4pt default.
669pub fn framed_properties(options: FramedOptions) -> SymHashMap<Stored> {
670  // Perl `$options{margin} && LookupDimension(...)`: an absent OR empty
671  // option is falsy and skips the lookup entirely.
672  let sep = options
673    .margin
674    .as_deref()
675    .filter(|m| !m.is_empty())
676    .and_then(|m| lookup_dimension_cs(m, false));
677  let th = options
678    .rule
679    .as_deref()
680    .filter(|r| !r.is_empty())
681    .and_then(|r| lookup_dimension_cs(r, false));
682  let pad = match (sep, th) {
683    (Some(s), Some(t)) => Some(Dimension::new(s.value_of() + t.value_of())),
684    (s, t) => s.or(t),
685  };
686  let th_pt = th.map(|t| t.to_attribute());
687  let mut style_parts: Vec<String> = Vec::new();
688  if let Some(s) = sep {
689    style_parts.push(s!("padding:{}", s.to_attribute()));
690  }
691  if let (Some(t), Some(tp)) = (th, th_pt.as_deref())
692    && tp != "0.4pt"
693  {
694    style_parts.push(s!("border-width:{}", t.to_attribute()));
695  }
696  let style = style_parts.join(";");
697
698  let mut props: SymHashMap<Stored> = SymHashMap::default();
699  props.insert(
700    "framed",
701    Stored::from(options.frame.unwrap_or_else(|| "rectangle".to_string())),
702  );
703  // Perl `LookupValue('font')->getColor` always yields a color (default
704  // Black), so framecolor is always present.
705  let framecolor = options
706    .color
707    .or_else(|| lookup_font().and_then(|f| f.get_color().map(|c| c.to_attribute())))
708    .unwrap_or_else(|| s!("#000000"));
709  props.insert("framecolor", Stored::from(framecolor));
710  if let Some(bg) = options.backgroundcolor {
711    props.insert("backgroundcolor", Stored::from(bg));
712  }
713  if !style.is_empty() {
714    props.insert("cssstyle", Stored::from(style));
715  }
716  if let Some(p) = pad {
717    props.insert("padtop", Stored::Dimension(p));
718    props.insert("padbottom", Stored::Dimension(p));
719    props.insert("padleft", Stored::Dimension(p));
720    props.insert("padright", Stored::Dimension(p));
721  }
722  props
723}
