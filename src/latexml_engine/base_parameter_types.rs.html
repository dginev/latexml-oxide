7/// The group a style parameter (`DisplayStyle` … `InFractionStyle`) opens to digest its argument carries a group code
8/// (K9, `lx@group@code` = the command whose argument it is): its closer is the parameter's own, never a `}` from the
9/// argument — an author's `}` an inserted one already replaced meets it as a group `}` cannot close and is deleted
10/// (stomach.rs `delete_extra_brace`; `\frac{\left( {a \right)}}{b}`, where LaTeX's `\frac` has `\begingroup`).
11/// OXIDIZED_DESIGN #530.
12fn mark_argument_group() {
13  if let Some(cs) = get_current_token() {
14    assign_frame_value("lx@group@code", Stored::Token(cs));
15  }
16}

18/// Shared reader body of the `Optional` / `OptionalAngled` parameter types:
19/// fold a (possibly absent) delimited `value` into the default / `inner`-reparse
20/// contract. The delimiters themselves are the caller's business — see
21/// `gullet::read_optional` / `read_optional_angled`.
22fn optional_arg_value(
23  value: Option<Tokens>,
24  inner: Option<&Parameters>,
25  default: &[Tokens],
26) -> Result<ArgWrap> {
27  if value.is_none() && !default.is_empty() {
28    // TODO: Is the default really multiple Vec<Tokens> ? Or just a single Tokens?
29    //       the default[0] is suspicious, compared to the original perl "$default"
30    Ok(ArgWrap::Tokens(default[0].clone()))
31  } else if let Some(inner_ps) = inner {
32    let mut reparsed = inner_ps.reparse_argument(value.into())?;
33    Ok(if reparsed.is_empty() {
34      ArgWrap::None
35    } else {
36      reparsed.remove(0)
37    })
38  } else {
39    Ok(value.into())
40  }
41}

43/// Shared reversion of the `Optional` / `OptionalAngled` parameter types:
44/// re-wrap a read argument in the `open`/`close` delimiters it came in.
45fn optional_arg_reversion(
46  arg: Vec<Token>,
47  inner: Option<&Parameters>,
48  open: Token,
49  close: Token,
50) -> Result<Tokens> {
51  // TODO: Same question for the type of "arg": should this be a single
52  //  `Tokens` rather than a `Vec<Token>`?
53  if arg.is_empty() {
54    return Ok(Tokens!());
55  }
56  let mut read_tokens: Vec<Token> = vec![open];
57  read_tokens.extend(match inner {
58    None => arg.into_iter().map(Token::revert).collect(),
59    Some(inner_ps) => inner_ps.revert_arguments(vec![Some(Tokens::new(arg))])?,
60  });
61  read_tokens.push(close);
62  Ok(Tokens::new(read_tokens))
63}

65/// `tokens` as a TeX decimal constant (tex.web §441, `<optional signs>` and
66/// digits with a `.` or `,`), when they are nothing else but spaces around it.
67fn plain_decimal(tokens: &[Token]) -> Option<f64> {
68  let is_space = |t: &&Token| t.get_catcode() == Catcode::SPACE;
69  let start = tokens.iter().position(|t| !is_space(&t))?;
70  let end = tokens.len() - tokens.iter().rev().position(|t| !is_space(&t))?;
71  let mut negative = false;
72  let mut digits = String::new();
73  for token in &tokens[start..end] {
74    if token.get_catcode() != Catcode::OTHER {
75      return None;
76    }
77    match token.with_str(|s| s.chars().next())? {
78      '-' if digits.is_empty() => negative = !negative,
79      '+' if digits.is_empty() => {},
80      c @ '0'..='9' => digits.push(c),
81      '.' | ',' if !digits.contains('.') => digits.push('.'),
82      _ => return None,
83    }
84  }
85  let value: f64 = digits.parse().ok()?;
86  Some(if negative { -value } else { value })
87}

89/// The file a `TeXFileName` argument names, and whether it was braced. TeX Live's `scan_file_name` reads a braced
90/// name's group as the name — `\input{name}`, `\openin\r{name}` — without the braces; the reader keeps them on the
91/// argument because `\input` loads LaTeX.pool for a braced (LaTeX-style) name (TeX_FileIO.pool.ltxml:164-169).
92/// A `"` quotes spaces and is not part of the name (tex.ch `more_name`). One reading for `\input`, `\openin`,
93/// `\openout` and `\font`: an `\openin` of a braced name kept the braces and found no file (pgfmanual's
94/// `\openin\examplesource\expandafter{\codeexamplesource}`, pgf-pie-manual).
95pub fn tex_file_name(name: &[Token]) -> (String, bool) {
96  let (inner, braced) = match name {
97    [open, inner @ .., close]
98      if open.get_catcode() == Catcode::BEGIN && close.get_catcode() == Catcode::END =>
99    {
100      (inner, true)
101    },
102    _ => (name, false),
103  };
104  (
105    Tokens::new(inner.to_vec()).to_string().replace('"', ""),
106    braced,
107  )
108}

112/// Run `body` under LaTeX's `\protected@edef` regime: `\protect` `\let` to
113/// `\@unexpandable@protect` (latex.ltx:1384) for the duration, restored after.
114/// A no-op outside LaTeX, where `\@unexpandable@protect` has no meaning.
115pub fn with_unexpandable_protect<R>(body: impl FnOnce() -> Result<R>) -> Result<R> {
116  let regime = T_CS!("\\@unexpandable@protect");
117  if lookup_meaning(&regime).is_some() {
118    with_let(&T_CS!("\\protect"), &regime, body)
119  } else {
120    body()
121  }
122}

124/// The `.idx` re-read of a `SanitizedVerbatim` argument, and the control words it formed from characters.
125pub(crate) fn reread_sanitized_entry(arg: &Tokens) -> (Tokens, Vec<Token>) {
126  // Now that we have the semiverbatim tokens, retokenize.
127  // This may seem like wasted work, but it avoids very unfortunate error
128  // propagation in cases where the \index argument was malformed for one
129  // reason or another. The strangeness comes from the original TeX
130  // workflow requiring multiple conversion calls, alongside a call to the
131  // `makeidx` binary, which we don't do in latexml. This parameter type
132  // emulates one important aspect implied by those steps.
133  // The string is what `\@wrindex`'s `\write` would put in the `.idx`
134  // file (tex.web §262 print_cs: a space after every control word), so
135  // an argument assembled by a macro — `\index{packages!#1@\texttt{#1}}`
136  // with `#1` = `\TIKZ` (pgfornament usefulcommands.tex:93) — re-reads as
137  // `\TIKZ` + `@`, not the undefined `\TIKZ@`. Perl re-tokenizes its
138  // UnTeX string, which glues a control word to a following non-letter
139  // (KNOWN_PERL_ERRORS #140). Style catcodes (`@` a letter) here: the
140  // entry is EXPANDED first (`\protected@write`, process_index_phrases),
141  // and a package-built entry is made of `@` names — tcolorbox's
142  // `\kvtcb@doc@sortindex\idx@actual…`; the `.ind` re-read with `@`
143  // OTHER is applied to what survives that expansion, in
144  // process_index_phrases.
145  // A control sequence the argument already holds as ONE token, whose
146  // written name would not re-read as itself (`\bool_if:nT`, `_`/`:`
147  // not letters here), is kept whole: TeX expands it in the
148  // `\protected@write`, before any re-read — genealogy-profiles.sty:536
149  // `\index[#1]{#2 \bool_if:nT {…} {| \tl_use:N …}}`, re-read first, gave
150  // `\bool` + `Script _` per entry. Guard:
151  // `perfect_kernel_batch61::index_entry_keeps_expl3_names_whole`.
152  let reread = |run: Vec<Token>| {
153    mouth::tokenize_internal(TeXString::assembled(writable_tokens(&Tokens::new(run)))).unlist()
154  };
155  let held: Vec<Token> = arg
156    .unlist_ref()
157    .iter()
158    .filter(|t| t.get_catcode() == Catcode::CS)
159    .copied()
160    .collect();
161  // A name of letters (or `@`), or of one character, re-reads as itself; only the others need the test.
162  let plain_name = |t: &Token| {
163    t.with_str(|s| {
164      let name = s.strip_prefix('\\').unwrap_or(s);
165      name.chars().count() <= 1 || name.chars().all(|c| c.is_ascii_alphabetic() || c == '@')
166    })
167  };
168  let mut out: Vec<Token> = Vec::new();
169  let mut run: Vec<Token> = Vec::new();
170  for t in arg.unlist_ref().iter().copied() {
171    if t.get_catcode() == Catcode::CS && !plain_name(&t) && reread(vec![t]) != [t] {
172      out.extend(reread(std::mem::take(&mut run)));
173      out.push(t);
174    } else {
175      run.push(t);
176    }
177  }
178  out.extend(reread(run));
179  // A control word formed here from characters was typed after `\@sanitize`
180  // made `\` other (imakeidx.sty:164-168, latex.ltx `\index`): it is written
181  // to the `.idx` verbatim, never expanded — `\printindex` runs it. Only a
182  // control sequence that was a token before (a macro-built entry) is
183  // expanded by the `\protected@write` (process_index_phrases). egpeirce-doc's
184  // visual index `\index[visual]{i@\ontop{…}\shk{1}…}` ran its pstricks
185  // graphs inside the write expansion (timeout; pdflatex prints them in the
186  // index). Guard: `perfect_kernel_batch61::index_entry_typed_in_source_is_written_verbatim`.
187  // A typed `\protect` is not frozen: `process_index_phrases` lets it be `\@unexpandable@protect` for the write, and
188  // the shipout `\write` drops it.
189  let formed: Vec<Token> = out
190    .iter()
191    .filter(|t| t.get_catcode() == Catcode::CS && !held.contains(t) && **t != T_CS!("\\protect"))
192    .copied()
193    .collect();
194  (Tokens::new(out), formed)
195}

318  DefParameterType!(Until, sub[_inner, until_extra] {
319    // A ran-out scan (read_until → None) delivers an EMPTY argument, not a
320    // missing one: many of our mouth ends are ARTIFICIAL (isolated
321    // constructor-arg mouths) where real TeX would find the delimiter in

325    // loops terminate via the stomach cycle guard.

340    // being exhausted, §360 ends the job (`*** (job aborted, no legal \end

355        }

357      },
358      None => {
359        if at_end_of_all_input() {
360          let caller = get_current_token()
361            .map(|t| t.to_string())
362            .unwrap_or_default();
363          Error!(
364            "expected",
365            "Until:",
366            s!("Missing argument Until:{} at end of input", until_extra[0]),
367            s!("while reading the arguments of {caller}")
368          );
369          latch_too_many_errors();
370          Fatal!(Mouth, EoF, s!("File ended while scanning use of a Until:{} argument (job aborted, no legal \\end found)", until_extra[0]));
371        }
372        Ok(ArgWrap::Tokens(Tokens::default()))
373      },
374    }
375  },

379      rev.push(t.revert());
380    }

563  // Perl: ReadPair in latex_constructs.pool.ltxml
564  // Returns ArgWrap::Pair if ( is found, ArgWrap::None otherwise (for Optional).
565  //
566  // Helper: read a float that may be wrapped in braces (e.g. `{36.5}`).
567  // Some authors brace pair coordinates to disambiguate negative-number
568  // tokenization or to keep \multiput / \put pair args together.

571    use latexml_core::common::pair::Pair;
572    use latexml_core::token::Catcode;
573    fn read_pair_float() -> Result<Float> {
574      let _ = skip_spaces();
575      // If next is BEGIN brace, consume to matching END brace and read float inside.
576      if let Some(tok) = read_token()? {
577        if tok.get_catcode() == Catcode::BEGIN {
578          let _ = skip_spaces();
579          let f = read_float()?;
580          let _ = skip_spaces();
581          // Consume matching close brace
582          if let Some(close) = read_token()?
583            && close.get_catcode() != Catcode::END {
584              unread_one(close);
585            }
586          return Ok(f);
587        }
588        unread_one(tok);
589      }
590      read_float()
591    }
592    let _ = skip_spaces();
593    // Kernel picture trampoline (latex.ltx L16768): `\long\def\picture#1
594    // {\pictur@#1}` strips ONE brace group before the `(`-scan, so the
595    // braced form `\begin{axopicture}{(90,140)(-10,0)}` (axodraw2-man,
596    // 101 errs) works exactly like the plain form. Mirror it here: a BEGIN
597    // group ahead of a Pair is unwrapped and its content re-scanned. Perl's
598    // ReadPair lacks this (shared failure with the braced form).
599    // A group that is no pair (`\diaghead{$V_i$}{a}{b}`, makecell's `OptionalPair` with no slope) is put back whole: its
600    // braces are the next argument's (2111.10338).
601    if if_next(T_BEGIN!())? {
602      read_token()?; // consume {  (counted; the group read below balances)
603      let group = read_balanced(ExpansionLevel::Off, false, false)?;
604      let pair = group
605        .unlist_ref()
606        .iter()
607        .find(|t| t.get_catcode() != Catcode::SPACE)
608        .is_some_and(|t| *t == T_OTHER!("("));
609      if pair {
610        unread(group);
611        let _ = skip_spaces();
612      } else {
613        let mut whole = vec![T_BEGIN!()];
614        whole.extend(group.unlist());
615        whole.push(T_END!());
616        unread(Tokens::new(whole));
617      }
618    }
619    if if_next(T_OTHER!("("))? {
620      read_token()?; // consume (
621      let _ = skip_spaces();
622      let x = read_pair_float()?;
623      // Perl latex_constructs.pool.ltxml:ReadPair L4910-4912 uses
624      //   $gullet->skipSpaces; $gullet->readUntil(T_OTHER(',')); $gullet->skipSpaces;
625      //   my $y = ...;
626      //   $gullet->skipSpaces; $gullet->readUntil(T_OTHER(')')); $gullet->skipSpaces;
627      // — `readUntil` is tolerant of extra junk between the float and the
628      // separator. Witness: physics/9709007 line 1594
629      //   \multiput(3.2,3,8)(.3,0){2}{\circle*{.1}}
630      // The user typoed `3,8` for `3.8`; Perl reads x=3.2, swallows nothing
631      // up to the comma, reads y=3, then `readUntil(')')` consumes the
632      // extra `,8` silently. The earlier Rust port read one token after y
633      // and only consumed it if literally `)` — bailing the second pair.
634      let _ = skip_spaces();
635      let _ = read_until(&Tokens!(T_OTHER!(",")));
636      let _ = skip_spaces();
637      let y = read_pair_float()?;
638      let _ = skip_spaces();
639      let _ = read_until(&Tokens!(T_OTHER!(")")));
640      let _ = skip_spaces();
641      Ok(ArgWrap::Pair(Pair::new(x, y)))
642    } else {
643      Ok(ArgWrap::None)
644    }
645  });

1850  // CommaList expects something like {item1,item2,...}; items may be
1851  // `{balanced}` groups or plain token runs. Perl returns a
1852  // `LaTeXML::Core::Array`; Rust has no Array type and follows the
1853  // `DirectoryList` convention — emit a token stream where each item
1854  // is wrapped in its own `{...}`.
1855  //
1856  // Perl supports a parameterised form `CommaList:Type` (e.g.
1857  // `CommaList:Number`) where each item is re-parsed through `Type`'s
1858  // own reader (`$typedef->reparseArgument`). When the parameter is
1859  // declared as `CommaList:Number`, `inner` carries the parsed
1860  // `Number` Parameter; we route each item through
1861  // `inner.reparse_argument(...)` and emit the reverted form so the
1862  // result is canonical-shape tokens of the typed value. Untyped
1863  // `CommaList` (no `inner`) keeps the original brace-delimit form.
1864  DefParameterType!(CommaList, sub[inner, _extra] {
1865    skip_spaces()?;
1866    // Phase 1: gather items as raw Tokens (one Tokens per comma-
1867    // separated piece). Done in a single pass to mirror the Perl
1868    // gullet-reader logic; the typed-reparse step happens after.
1869    let mut items: Vec<Tokens> = Vec::new();
1870    if if_next(T_BEGIN!())? {
1871      read_token()?; // consume outer `{`
1872      let mut current: Vec<Token> = Vec::new();
1873      let comma = T_OTHER!(",");
1874      while let Some(token) = read_token()? {
1875        let cc = token.get_catcode();
1876        if cc == Catcode::END {
1877          items.push(Tokens::new(std::mem::take(&mut current)));
1878          break;
1879        } else if token == comma {
1880          items.push(Tokens::new(std::mem::take(&mut current)));
1881        } else if cc == Catcode::BEGIN {
1882          // Nested `{balanced}` — preserve brace wrapping.
1883          current.push(token);
1884          let balanced = read_balanced(ExpansionLevel::Off, false, false)?;
1885          current.extend(balanced.unlist());
1886          current.push(T_END!());
1887        } else {
1888          current.push(token);
1889        }
1890      }
1891    } else if let Some(token) = read_token()? {
1892      // No outer brace — read a single token as the sole item.
1893      items.push(Tokens::new(vec![token]));
1894    }
1895    // Phase 2: if a typed form was declared, reparse each item via
1896    // the inner type's reader (`Perl reparseArgument` parity). The
1897    // reverted form is emitted so the downstream callsite sees
1898    // canonical tokens of the typed value.
1899    let mut collected: Vec<Token> = Vec::new();
1900    if let Some(inner_ps) = inner {
1901      for item in items {
1902        let reparsed = inner_ps.reparse_argument(ArgWrap::Tokens(item))?;
1903        collected.push(T_BEGIN!());
1904        for wrap in reparsed {
1905          collected.extend(wrap.revert()?.unlist());
1906        }
1907        collected.push(T_END!());
1908      }
1909    } else {
1910      for item in items {
1911        collected.push(T_BEGIN!());
1912        collected.extend(item.unlist());
1913        collected.push(T_END!());
1914      }
1915    }
1916    Tokens::new(collected)
1917  });

1919  // Support for Key / Value arguments.
1920  // The very basic form is
1921  //   RequiredKeyVals: $keyset
1922  //   OptionalKeyVals: $keyset
1923  // to parse Key-Value pairs from a given keyset (see the 'keyval' package
1924  // documentation for more information). These types of KeyVal
1925  // parameters will return a LaTeXML::Core::KeyVals object, which can then be
1926  // used to access the values of the individual items.
1927  // The difference between the two forms is that RequiredKeyVals expects a set of
1928  // key-value pairs wrapped in T_BEGIN T_END, where as OptionalKeyVals optionally
1929  // expects a set of KeyValue pairs wrapped in T_OTHER('[') T_OTHER(']')
1930  //
1931  // Several extension of the keyval package exist, the most common one we support
1932  // is the xkeyval package. This introduces further variations on the keyval
1933  // arguments parsing, in particular it allows to read keys from more than one
1934  // keyset at once. These can be specified by giving comma-separated values in
1935  // the keyset argument. By default, a key will only be set in the **first**
1936  // keyset it occurs in. By using
1937  //   RequiredKeyVals+: $keysets
1938  //   OptionalKeyVals+: $keysets
1939  // the key will be set in all keysets instead.
1940  //
1941  // All keys to be parsed with these arguments should be declared using
1942  // DefKeyVal in LaTeXML::Package. By default, an error is thrown if an unknown
1943  // key is encountered. To surpress this behaviour, and instead store all
1944  // undefined keys, use
1945  //   RequiredKeyVals*: $keysets
1946  //   OptionalKeyVals*: $keysets
1947  // instead. The '*' and '+' modifiers can be combined by using:
1948  //   RequiredKeyVals*+: $keysets
1949  //   OptionalKeyVals*+: $keysets
1950  //
1951  // Furthermore, the xkeyval package supports giving prefixes to keys,
1952  //   RequiredKeyVals[*][+]: $prefix|$keysets
1953  //   OptionalKeyVals[*][+]: $prefix|$keysets
1954  //
1955  // Finally, it is possible to specify specific keys to skip when digesting the
1956  // object. This can be achieved using comma-separated key values in
1957  //   RequiredKeyVals[*][+]: $prefix|$keysets|$skip
1958  //   OptionalKeyVals[*][+]: $prefix|$keysets|$skip
1959
1960  pub fn required_key_vals(
1961    star: bool,
1962    plus: bool,
1963    _inner: Option<&Parameters>,
1964    extra: &[Tokens],
1965  ) -> Result<KeyVals> {
1966    // Skip whitespace between this arg and the previous one. TeX-style
1967    // parameter matching `#1#2{...}` skips spaces before each `{`-delimited
1968    // group; our `RequiredKeyVals` parameter type checks `if_next(T_BEGIN)`
1969    // directly, so without an explicit skip it errors on user input like
1970    //   \newglossaryentry{RIS}\n{ name={...}, ... }
1971    // (\n + indentation between args[0] and args[1]). Driver: 2203.11854
1972    // R=1 → R=0 — Perl raw-loads glossaries.sty so `\newglossaryentry` is
1973    // a 2-arg `\newcommand` whose TeX matching handles this natively.
1974    skip_spaces()?;
1975    if if_next(T_BEGIN!())? {
1976      let mut extra_iter = extra.iter();
1977      // subtle!!! The first extra is the prefix, according to the Perl use.
1978      let prefix = extra_iter.next().map(ToString::to_string);
1979      // TODO: is the last extra field actually a "skip" ? Example?
1980      let keysets = extra_iter.map(ToString::to_string).collect();
1981      keyvals_aux(Some(T_END!()), KVSpec {
1982        star,
1983        plus,
1984        prefix,
1985        keysets,
1986        ..KVSpec::default()
1987      })
1988    } else {
1989      // Perl-faithful lowercase category — matches Base_ParameterTypes.pool.ltxml
1990      // `Error('expected', '{', $gullet, "Missing keyval arguments")`. The
1991      // engine convention is lowercase categories throughout.
1992      Error!("expected", "{", "Missing keyval arguments");
1993      Ok(KeyVals::default())
1994    }
1995  }

2002  });

2004    required_key_vals(true, false, inner, extra)
2005  },
2006  reversion => sub[arg, _inner, _extra] {
2007    Ok(Tokens!(T_BEGIN!(), Tokens::new(arg).revert(), T_END!()))

2011  },
2012  reversion => sub[arg, _inner, _extra] {
