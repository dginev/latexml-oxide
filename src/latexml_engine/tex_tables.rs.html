14/// The alignment property counting the `\omit`s of the current `\multicolumn` cell still to be read by the cell-head

33/// The alignment property of an alignment set without `\halign` in TeX: latex.ltx's `tabbing` builds each line from
34/// boxes (`\@startline`, `\@stopline`, `\@addfield`) and never executes `\halign`, so a document's `\halign` macro
35/// does not run for it (`run_halign_redefinition_prelude`). Every other alignment `\lx@begin@alignment` starts reaches
36/// `\halign` in TeX.
37pub const ALIGNMENT_WITHOUT_HALIGN: &str = "without_halign";

39/// The alignment property of an alignment that is a display by itself in TeX, `$$\halign…$$`: amsmath's align and
40/// gather families (amsmath.sty `\start@align`, `\start@gather`), latex.ltx's `eqnarray` and IEEEtrantools'
41/// `IEEEeqnarray` (IEEEtrantools.sty:2013). Material a document's `\halign` macro typesets before the primitive is an
42/// error there (`run_halign_redefinition_prelude`). Not cases.sty's `numcases`, whose `\halign` is inside a `\vbox`
43/// (cases.sty:73-75). amsmath's `multline` is a display too but is not marked: its material is kept in the formula
44/// (TeX: "Improper \halign inside $$'s"), the safe direction.
45pub const ALIGNMENT_IS_DISPLAY: &str = "display";

47/// Whether `token` means the `\halign` primitive, under whatever name a package saved it (cryptocode.sty:950
48/// `\let\@pc@halign\halign`): a constructor declared `halign_primitive`, the primitive itself or a binding's stand-in
49/// that takes its place — inside a pgf picture its SVG matrix `\lxSVG@halign` (`\lxSVG@installcommands` lets `\halign`
50/// to it, pgfsys_latexml_def.rs; Perl pgfsys-latexml.def L59), which a pseudocode in a TikZ node saves instead (witness
51/// 2107.05054).
52fn means_halign_primitive(token: &Token) -> bool {
53  with_meaning(
54    token,
55    |meaning| matches!(meaning, Some(Stored::Constructor(c)) if c.halign_primitive),
56  )
57}

71/// Run a document's `\halign` redefinition where an alignment the binding builds would reach `\halign` in TeX: the
72/// `before_digest` of `\lx@begin@alignment`, whose boxes go before the alignment.
73///
74/// amsmath's alignment environments and every `\ialign`-based one (latex.ltx `\@array`, plain `\ialign` =
75/// `\everycr{}\tabskip\z@skip\halign`, `\eqalign`, `\matrix`, `\oalign`, longtable, eqnarray, amscd, blkarray,
76/// IEEEtrantools) execute the token `\halign`, so in TeX a macro `\halign` runs first and the primitive it ends with
77/// starts the alignment. Our bindings build those alignments in Rust (`\lx@begin@alignment`), so the macro's
78/// expansion is digested here, in the group the alignment starts in (TeX's: its definitions last as long), and its
79/// final primitive is the alignment already being built. `tabbing` never reaches `\halign`
80/// ([`ALIGNMENT_WITHOUT_HALIGN`]). Only the conservative shape is run: a parameterless macro whose expansion ends with
81/// the primitive (under any name); one that never reaches it, reaches it only through another macro, or reaches it
82/// mid-expansion is not run at all.
83///
84/// Material the expansion typesets precedes the alignment, as on TeX's list. In display math TeX errs instead ("Improper
85/// \halign inside $$'s", tex.web §774) and deletes it: there it is a Warning and dropped.
86///
87/// cryptocode.sty:950-957 (and its `mode=text` branch :760-770) redefines `\halign` inside `\pseudocode` to bind
88/// `\@pc@lb` to the alignment's `\\` and `\\` to its line-numbering wrapper: without the prelude every line after the
89/// first lost its number and `mode=text` erred `undefined:\@pc@lb` (witnesses 1804.05141, 2106.03626, 2107.06049,
90/// 2108.04600, 2202.07253, 2410.11134). Repro kernel-alignment/cryptocode_halign_redefinition_numbers_lines; guards
91/// `perfect_kernel_batch76a`. OXIDIZED_DESIGN_DIVERGENCES #760.
92fn run_halign_redefinition_prelude() -> Result<Vec<Digested>> {
93  let halign = T_CS!("\\halign");
94  let Some(Stored::Expandable(defn)) = lookup_meaning(&halign) else {
95    return Ok(Vec::new());
96  };
97  if defn.get_num_args() != 0 || !matches!(defn.get_expansion(), Some(ExpansionBody::Tokens(_))) {
98    return Ok(Vec::new());
99  }
100  if alignment_property(ALIGNMENT_WITHOUT_HALIGN) {
101    return Ok(Vec::new());
102  }
103  let mut expansion = defn.invoke(false)?.unlist();
104  while expansion
105    .last()
106    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
107  {
108    expansion.pop();
109  }
110  match expansion.pop() {
111    Some(last) if means_halign_primitive(&last) => {},
112    _ => return Ok(Vec::new()),
113  }
114  if expansion.iter().any(means_halign_primitive) {
115    return Ok(Vec::new());
116  }
117  let material = digest(Tokens::new(expansion))?;
118  if material.is_skippable() {
119    return Ok(Vec::new());
120  }
121  if alignment_property(ALIGNMENT_IS_DISPLAY) {
122    Warn!(
123      "unexpected",
124      "\\halign",
125      format!(
126        "Improper \\halign inside $$'s: the redefined \\halign typeset material before its display alignment, which \
127         is deleted: {}",
128        material.untex().unwrap_or_default()
129      )
130    );
131    return Ok(Vec::new());
132  }
133  Ok(vec![material])
134}

916/// 54), and every tikz-cd inside an amsmath cell left the cell's math open.

921    tks.extend(spec.revert()?.unlist());
922  }
923  tks.push(T_BEGIN!());

925    tks.extend_from_slice(template_tks.unlist_ref());
926  }
927  tks.push(T_CS!("\\cr"));
928  if let Some(Stored::Digested(alignment_d)) = whatsit.get_property("alignment").as_deref()
929    && let DigestedData::Alignment(data) = alignment_d.data()
930  {
931    // Mirror line-30 fix: tolerate a re-entrant borrow during
932    // reversion rather than panic. See comment there.
933    if let Ok(d) = data.try_borrow() {
934      tks.extend(d.revert()?.unlist());
935    }
936  }
937  tks.push(T_END!());
938  Ok(Tokens::new(tks))
939}

1142/// Digest an alignment's body, row by row. `runs_everycr`: the alignment is the `\halign` primitive's (tex.web
1143/// §768-812), which inserts `\everycr` (`insert_everycr`) after its preamble and after every row. An alignment a
1144/// binding builds (`\lx@begin@alignment`) stands for a LaTeX/plain macro that starts its `\halign` with an `\everycr`
1145/// of its own — `\ialign`'s `\everycr{}` (latex.ltx:631, `\@array` :16573), array.sty's and longtable's tagging hooks,
1146/// amsmath's `\displ@y` spacing — whose rows the binding already builds, so it runs none.
1147///
1148/// Before (Perl alike, TeX_Tables.pool.ltxml:109 only stores the register): no alignment ran it, so lamstex's
1149/// `\newCD` (math0005042), which counts a diagram's rows with `\everycr{\noalign{\global\advance\rowcount@\@ne…}}`,
1150/// counted none and built its `\ifcase` size lists empty. KNOWN_PERL_ERRORS #1019. Repro
1151/// kernel-alignment/everycr_runs_after_every_cr; guards `perfect_kernel_batch78b`.
1152pub fn digest_alignment_body(whatsit: &mut Whatsit, runs_everycr: bool) -> Result<()> {
1153  // Now read & digest the body.
1154  // Note that the body MUST end with a \cr, and that we've made Special Arrangments
1155  // with \alignment@cr to recognize the end of the \halign
1156  local_align_group_count(0);
1157  let alignment_stored = match lookup_alignment() {
1158    Some(alignment) => alignment,
1159    _ => {
1160      Error!(
1161        "missing",
1162        "alignment",
1163        "There is no open alignment structure here"
1164      );
1165      return Ok(());
1166    },
1167  };
1168  local_reading_alignment(&alignment_stored);
1169  whatsit.set_property("alignment", Stored::Digested(alignment_stored.clone()));
1170  if runs_everycr {
1171    insert_everycr(); // tex.web §774: the preamble's `\cr`
1172  }
1173
1174  // Debug!("Halign {}: BODY Processing...",alignment) if $LaTeXML::DEBUG{halign};
1175  let mut lastwascr = false;
1176  let mut reversion: Vec<Token> = Vec::new();
1177  let mut creversion: Vec<Token> = Vec::new();
1178  // the field-ledger serials of the row-end markers the body consumes (core `document::FIELD_TAKEN_PROPERTY`)
1179  let mut field_taken: Vec<u32> = Vec::new();
1180  let alignment_cell = alignment_stored.alignment_cell().unwrap();
1181  loop {
1182    let (cell_opt, next, vtype, hidden) = digest_alignment_column(alignment_cell, lastwascr)?;
1183    //     Debug("Halign $alignment: BODY got CELL"
1184    //         . "[" . $alignment->currentRowNumber . "," . $alignment->currentColumnNumber . "]"
1185    //         . ToString($cell) . " ended at " . Stringify($next)) if $LaTeXML::DEBUG{halign};
1186
1187    if let Some(cell) = cell_opt {
1188      reversion.extend(
1189        trim_column_template(alignment_cell.borrow_mut(), p_revert(cell.clone())?).unlist(),
1190      );
1191      creversion.extend(
1192        trim_column_template(alignment_cell.borrow_mut(), c_revert(cell.clone())?).unlist(),
1193      );
1194      extract_alignment_column(alignment_cell.borrow_mut(), cell)?;
1195    } else {
1196      // Debug("Halign $alignment: BODY DONE!") if $LaTeXML::DEBUG{halign};
1197      break;
1198    }
1199    lastwascr = false;
1200    // Perl L319-320: $next->defined_as(T_END) — recognizes \egroup as alignment end; any catcode-2 character
1201    // ends it too (tex.web §1068), as `scan_left_brace` opens it on any catcode-1 character (60j)
1202    let next_is_end = next.as_ref().map(is_right_brace).unwrap_or(false)
1203      || next == Some(T_CS!("\\lx@close@alignment"));
1204    if (vtype.is_none() || vtype.as_ref().unwrap().is_empty()) && (next.is_none() || next_is_end) {
1205      // End of alignment
1206      alignment_cell.borrow_mut().end_row()?;
1207      break;
1208    } else if vtype.as_deref() == Some("align") {
1209      alignment_cell.borrow_mut().end_column()?;
1210      alignment_cell.borrow_mut().column_ended();
1211      if !hidden {
1212        reversion.push(next.unwrap()); // and record the &
1213        creversion.push(next.unwrap()); // and record the &
1214      }
1215    } else if vtype.as_deref() == Some("insert") {
1216      alignment_cell.borrow_mut().end_column()?;
1217    } else if matches!(vtype.as_deref(), Some("cr" | "crcr" | "noalign")) {
1218      alignment_cell.borrow_mut().end_row()?;
1219      // A binding's request for the row just ended (`RowDisposition`: longtable's `\kill` and caption row, witness
1220      // 2010.09763), taken after the row end has closed its column boxes (no locked-frame Fatal).
1221      alignment_cell.borrow_mut().dispose_ended_row();
1222      if !hidden {
1223        reversion.push(next.unwrap());
1224        creversion.push(next.unwrap());
1225      } else if vtype.as_deref() == Some("cr") {
1226        let arg_toks = read_arg(ExpansionLevel::Off)?;
1227        let arg = digest(arg_toks)?;
1228        // the row's end marker is the alignment's, never absorbed: a field value's `\\` (`\author{A \\ B}` set in a
1229        // `tabular`) is taken by the print's structure once the alignment is shown (core
1230        // `document::field_ink_taken_serials`, OXIDIZED_DESIGN #715)
1231        document::field_ink_taken_serials(&arg, &mut field_taken);
1232        reversion.extend(p_revert(arg.clone())?.unlist());
1233        creversion.extend(c_revert(arg)?.unlist());
1234      } else if vtype.as_deref() == Some("crcr") {
1235      }
1236      // tex.web §799 `fin_row`: a `\cr` or nonredundant `\crcr` ended the row; a `\noalign`'s material ends none.
1237      if runs_everycr && vtype.as_deref() != Some("noalign") {
1238        insert_everycr();
1239      }
1240      lastwascr = true;
1241    } else if let Some(next_tok) = next {
1242      // Note, in case next is \crcr
1243      Error!("unexpected", next_tok, s!("Column ended with {next_tok}"));
1244    }
1245  }
1246  alignment_cell.borrow_mut().end_row()?;
1247  {
1248    let borrowed = alignment_cell.borrow();
1249    let n_rows = borrowed.rows().len();
1250    let n_cols = borrowed
1251      .rows()
1252      .iter()
1253      .map(|r| r.get_columns().len())
1254      .max()
1255      .unwrap_or(0);
1256    assign_value(
1257      "LAST_ALIGNMENT_ROWS",
1258      Stored::Int(n_rows as i64),
1259      Some(Scope::Global),
1260    );
1261    assign_value(
1262      "LAST_ALIGNMENT_COLS",
1263      Stored::Int(n_cols as i64),
1264      Some(Scope::Global),
1265    );
1266  }
1267  alignment_cell
1268    .borrow_mut()
1269    .set_reversion(Tokens!(reversion));
1270  alignment_cell
1271    .borrow_mut()
1272    .set_content_reversion(Tokens!(creversion));
1273  //   Debug("Halign $alignment: BODY DONE!\n"
1274  //       . "=> " . join(',', map { Stringify($_); } @reversion)) if $LaTeXML::DEBUG{halign};
1275  if !field_taken.is_empty() {
1276    whatsit.set_property(
1277      document::FIELD_TAKEN_PROPERTY,
1278      Stored::VecDequeStored(
1279        field_taken
1280          .into_iter()
1281          .map(|s| Stored::Int(i64::from(s)))
1282          .collect(),
1283      ),
1284    );
1285  }
1286  expire_align_group_count();
1287  expire_reading_alignment();
1288  Ok(())
1289}

1319pub fn digest_alignment_column(alignment: &RefCell<Alignment>, lastwascr: bool) -> DigestedColumn {
1320  new_local_box_list();
1321  let ismath = lookup_bool_sym(pin!("IN_MATH"));
1322  // Scan for leading \omit, skipping over (& saving) \hline.
1323  //   Debug("Halign $alignment: COLUMN starting scan "
1324  //       . "(" . ($ismath ? " math" : " text") . ")") if $LaTeXML::DEBUG{halign};
1325  // Declared without initializer — Perl resets this to undef at the
1326  // start of every OUTER iteration (see L742), so the initial value
1327  // is genuinely dead. The compiler tracks definite-assignment for us.
1328  let mut last_token: Option<Token>;
1329  let mut spanning = false;
1330  loop {

1332    // Scan till we get something NOT \omit, \noalign.
1333    // Perl TeX_Tables.pool.ltxml L371-396: `$token` is set per readXToken call,
1334    // so when the gullet returns undef (mouth exhausted), `!$token` triggers
1335    // the early-return `return (undef, $token, undef, undef);`. In Rust the
1336    // INNER 1 `while let Some(xtoken) = read()` branch only updates last_token
1337    // on Some — when read returns None, last_token would stay at its prior
1338    // value (e.g. a content token from a previous OUTER iteration), the
1339    // `last_token.is_none()` check below would skip, and we'd re-feed the
1340    // (column_before, marker, last_token) bundle into an empty gullet,
1341    // looping infinitely. Reset to None per Perl's per-iteration semantics.

1343    // tex.web §15350 (`init_align`: a display/box `\halign` enters -vmode) and
1344    // §15510 (`align_peek`): the per-column peek that expands the cell-head
1345    // token runs in the alignment's INTER-ROW mode, internal vertical;
1346    // `init_row` §15532 switches to restricted horizontal only afterwards, and
1347    // `\noalign` material stays in -vmode (§15513). LaTeXML digests the whole

1349    // \hbox\fi\bgroup…$…$…\egroup` (abntexto.tex:79-81's active `<…>` in the
1350    // class's `$$\halign{$#$\cr…}$$`, :727-736) dropped its `\hbox`, its inner
1351    // `$` became a math END inside the template's math, and the display, the
1352    // list and the section after it never closed (abntexto 8; Perl 4 — SHARED).
1353    // The override lives only for the peek: restored on EVERY exit of this
1354    // loop (`peek_mode`'s Drop, ended before any row/cell group opens). Guard:
1355    // `perfect_kernel_batch56::alignment_cell_head_peeks_in_internal_vertical_mode`.

1359      // would open the next row (`dropping_argument_tails`).
1360      // The peek stops at a `\newcommand` optional-argument macro (its `\@protected@testopt` reaches `\futurelet`'s
1361      // `\let`): TeX starts the cell there, and its arguments are read inside it (`read_x_token_cell_head`, tex.web
1362      // §785/§791).

1365        dropping_argument_tails(BETWEEN_ALIGNMENT_ROWS, read_x_token_cell_head)?
1366      } else {
1367        read_x_token_cell_head()?
1368      };

1371      let token = last_token.as_ref().unwrap();

1373      // Perl L372-375: `equals` for SPACE/`\par` (literal tokens), but
1374      // `defined_as` for `\crcr`/`\lx@hidden@crcr` (CS that may be `\let`-aliased).

1376        || *token == T_CS!("\\par")

1378          && (token.defined_as(&T_CS!("\\crcr")) || token.defined_as(&T_CS!("\\lx@hidden@crcr"))))

1380      } else if token.defined_as(&T_CS!("\\omit")) {

1382        //         Debug("Halign $alignment: OMIT at " . Stringify($token)) if
1383        // $LaTeXML::DEBUG{halign};

1386          // set it and re-enter inside the row's frame (see `AlignPeekMode`).

1390        }

1393        // its material is boxed as text, as `\lx@alignment@multicolumn` boxes an amsmath body (and keeps an array's

1401            .borrow_mut()
1402            .get_properties_mut()

1405        } else if omit_cell_is_text(&alignment.borrow()) {
1406          alignment.borrow_mut().omit_next_column_as_text();
1407        } else {
1408          alignment.borrow_mut().omit_next_column();
1409        }
1410      } else if token.defined_as(&T_CS!("\\noalign")) {
1411        // \puts something in vertical list. Perl L381 uses `defined_as`, so a
1412        // `\let`-copy of `\noalign` (e.g. expl3's `\tex_noalign:D`, used by the
1413        // `pseudo` package's per-line init `\noalign`) is recognized here rather
1414        // than falling through to the primitive's "cannot be used here" error.
1415        // Debug("Halign $alignment: noalign at " . Stringify($token)) if $LaTeXML::DEBUG{halign};

1417        // material's own -vmode (§15513) is bound inside its group below.

1420          alignment.borrow_mut().end_row()?;
1421        }

1423        alignment.borrow_mut().last_column();
1424        // tex.web §15513: `\noalign` does `scan_left_brace`, opens the
1425        // no_align_group, and the material is EXECUTED up to the `}` that
1426        // closes that group. A token-level pre-scan (`read_arg`) miscounted
1427        // latex.ltx's `\hline` brace hack — `\noalign{\ifnum0=`}\fi\hrule…}`
1428        // whose first `}` is a char constant consumed by `\ifnum` at
1429        // execution — so the body was cut at `\ifnum0=` and the rest leaked
1430        // into the alignment as cell content (boldline `\hlineB`, any raw

1434        // hit it. Guard: `perfect_kernel_batch54::noalign_body_is_executed_to_its_group_end`.
1435        // §785 `align_peek`: `scan_left_brace` — expanded, past spaces and `\relax` (`\noalign\relax{\hrule}`).

1444            },

1446          }
1447        };

1450          && !package_loaded("array")
1451          && matches!(alignment.borrow().get_property("row_end").as_deref(),
1452            Some(Stored::Token(t)) if *t == tabularcr);
1453        if let Some(t) = raw {
1454          unread_one(t);
1455        }
1456        for t in skipped.into_iter().rev() {
1457          unread_one(t);
1458        }
1459        let empty_noalign = match read_non_blank_non_relax()? {
1460          Some(t) => {
1461            let is_cr = t.defined_as(&T_CS!("\\lx@hidden@cr"));
1462            unread_one(t);
1463            is_cr
1464          },
1465          None => false,
1466        };
1467        if empty_noalign && !kernel_tabularcr {
1468          Error!("expected", "{", "Missing { inserted");
1469        }
1470        let mut body = Vec::new();
1471        if !empty_noalign {
1472          scan_left_brace()?;
1473          // The one-frame reader shape of `predigest_box_contents_in_mode`:
1474          // execute the material (so `\ifnum0=`}` consumes its char-constant
1475          // `}` during expansion) and STOP at the `}` that closes the group
1476          // without invoking it — the alignment's own cell machinery must not
1477          // see it as a cell end.
1478          bgroup();
1479          // tex.web §15513: `\noalign` material runs in -vmode; bound in the
1480          // group's own frame, so `egroup` below restores the cell mode.
1481          assign_frame_value_sym(pin!("MODE"), "internal_vertical");
1482          let level = get_frame_depth();
1483          new_local_box_list();

1485            let next = match get_pending_comment() {
1486              Some(comment) => Some(comment),
1487              None => read_x_token(Some(true), false, None)?,
1488            };
1489            let Some(token) = next else { break };
1490            if is_right_brace(&token) && level >= get_frame_depth() {
1491              break;
1492            }
1493            check_timeout()?;
1494            extend_box_list(invoke_token(&token)?);
1495          }
1496          body = expire_local_box_list();
1497          egroup()?;
1498        }
1499        // As `digest` does: a single box stands alone (the rule-only
1500        // `\noalign{\hrule}` becomes the next row's border via that shape).
1501        let r: Digested = if body.len() == 1 {
1502          body.pop().unwrap()
1503        } else {
1504          List::new(body).into()
1505        };
1506        alignment.borrow_mut().end_row()?;
1507        expire_local_box_list();
1508        // A `\noalign`'s material is digested as a row ended by `\cr`, but it ends no row (vtype `noalign`: no
1509        // `\everycr` after it, tex.web §1133).
1510        return Ok((Some(r), Some(T_CS!("\\cr")), some!("noalign"), false)); // Pretend this is a whole
1511      // row???
1512      } else if token.defined_as(&T_CS!("\\lx@hidden@noalign")) {
1513        // \puts something in vertical list. Perl L389 uses `defined_as`.
1514        //         Debug("Halign $alignment: COLUMN invisible noalign") if $LaTeXML::DEBUG{halign};
1515        let invoked = invoke_token(token)?;
1516        extend_box_list(invoked);
1517      } else {
1518        break;
1519      }
1520    }
1521    // The peek is over: back to the cell mode (tex.web `init_row`/`init_col`,
1522    // §15532/§15560) in the frame that set the override, before the cell
1523    // group opens below.
1524    drop(peek_mode.take());
1525    //     Debug("Halign $alignment: COLUMN end scan at " . Stringify($token)) if
1526    // $LaTeXML::DEBUG{halign};
1527    // Perl L395: $token->defined_as(T_END) — recognizes \egroup as column end (any catcode-2 character, §1068)
1528    let last_is_end = last_token.as_ref().map(is_right_brace).unwrap_or(false)
1529      || last_token == Some(T_CS!("\\lx@close@alignment"));
1530    if last_token.is_none() || last_is_end {
1531      // Material between the last row and the alignment's end — a `\label` after the final `\\` (`… \\ \label{c}
1532      // \end{eqnarray*}`, 1206.1393, where `\eqnarray` set `\@currentlabel` and pdflatex resolves it) — has no next
1533      // cell to carry it: the row above keeps what must reach the document (`alignmentPreserve`), as `\noalign`
1534      // material after the last row stays in the list after it. Perl drops it (KNOWN_PERL_ERRORS #1182).
1535      let trailing: Vec<Digested> = expire_local_box_list()
1536        .into_iter()
1537        .filter(|item| item.get_property_bool("alignmentPreserve"))
1538        .collect();

1540        && let Some(row) = alignment
1541          .borrow_mut()
1542          .rows_mut()
1543          .iter_mut()
1544          .rev()
1545          .find(|row| !row.is_pseudo())
1546      {
1547        row.after.extend(trailing);
1548      }
1549      return Ok((None, last_token, None, false));
1550    }
1551    // Next column, unless spanning (then combine columns)
1552    if spanning {
1553      spanning = false;
1554      alignment.borrow_mut().next_column()?;
1555    } else {
1556      alignment.borrow_mut().start_column(false)?;
1557    }
1558    // An `\omit`ted cell that ends at once (`\omit\span`, latex.ltx `\multispan`; an empty `\omit&`) has nothing to
1559    // box as text: no empty box for each spanned column.
1560    if last_token
1561      .as_ref()
1562      .is_some_and(|t| is_column_end(t).is_some())
1563      && let Some(column) = alignment.borrow_mut().current_column()
1564    {
1565      column.omit_text_box = false;
1566    }
1567    // Push before template,  Marker and put the token back
1568    let to_unread = Tokens!(
1569      alignment.borrow_mut().get_column_before(),
1570      T_MARKER!("before-column"),
1571      last_token.unwrap()
1572    );
1573    // eprintln!("Halign: COLUMN preload at {}", to_unread.stringify());

1575    while let Some(token) = read_x_token(Some(true), false, Some(false))? {
1576      if align_error(&token)? {
1577        continue;
1578      }
1579      if let Some((_atoken, vtype, hidden)) = is_column_end(&token) {
1580        if vtype == "span" {
1581          // next column, but continue accumulating
1582          // Debug("Halign $alignment: COLUMN span") if $LaTeXML::DEBUG{halign};
1583          spanning = true;
1584          break;
1585        } else {
1586          // Debug("Halign $alignment: COLUMN ended with " . Stringify($token) . "\n"
1587          //     . "  => " . ToString(List(@LaTeXML::LIST))) if $LaTeXML::DEBUG{halign};
1588          alignment
1589            .borrow_mut()
1590            .get_properties_mut()
1591            .remove(MULTICOLUMN_OMITS);
1592          let current_list = expire_local_box_list();
1593          let mut out_list = List::new(current_list);
1594          out_list.mode = Some(if ismath { TexMode::Math } else { TexMode::Text });
1595          return Ok((
1596            Some(out_list.into()),
1597            Some(token),
1598            Some(String::from(vtype)),
1599            hidden,
1600          ));
1601        }
1602      // DG: Note this block is commented out as clippy warned it has the exact identical logic

1604      //
1605      // } else if token == T_CS!("\\lx@hidden@noalign") { //  \puts something in vertical list

1608      //   stomach.box_list.extend(invoked.into_iter());

1614        // List::new(stomach.box_list.clone()).stringify()); //       if

1621// This attempts to trim off the column template parts from contents of the full column,
1622// leaving only the author supplied part for a sensible reversion.
1623// It's not nearly clever enough, given that macros can be in the template,
1624// but works surprisingly well so far.
1625// A better alternative might be based on sneaking some Marker tokens/boxes through
1626// but they would likely interfere with the macros tehmselves.
1627pub fn trim_column_template(mut alignment: RefMut<Alignment>, tokens: Tokens) -> Tokens {
1628  if let Some(row) = alignment.current_row()
1629    && row.is_pseudo()
1630  {
1631    return tokens;
1632  }
1633  let mut pre = alignment.get_column_before().unlist();
1634  let mut post = alignment.get_column_after().unlist();
1635  //   Debug("Halign $alignment: COLUMN Compare:\n"
1636  //       . "  Column: " . ToString(Tokens(@tokens)) . "\n"
1637  //       . "  Before: " . ToString(Tokens(@pre)) . "\n"
1638  //       . "  After : " . ToString(Tokens(@post)) . "\n") if $LaTeXML::DEBUG{halign};
1639  let mut tks_vec = tokens.unlist();
1640  while !pre.is_empty() && !tks_vec.is_empty() {
1641    let t = pre.remove(0);
1642    if let Some(tks_first) = tks_vec.first()
1643      && t == *tks_first
1644    {
1645      tks_vec.remove(0);
1646    }
1647  }
1648  while !post.is_empty() && !tks_vec.is_empty() {
1649    let t = post.pop().unwrap();
1650    if let Some(tks_last) = tks_vec.last()
1651      && t == *tks_last
1652    {
1653      tks_vec.pop();
1654    }
1655  }
1656  //   Debug("  Trimmed: " . ToString(Tokens(@tokens))) if $LaTeXML::DEBUG{halign};
1657  Tokens::new(tks_vec)
1658}
1659// Given the boxes for an alignment cell,
1660// extract & remove the various fills and rules from the ends to annotate the cell structure
1661/// True when a column's `after` template carries real content *after* its
1662/// trailing fill, so the fill no longer centers the cell.
1663///
1664/// The `array`-package `!{<filler>}` (`array.sty` `DefColumnType('!{}')` →
1665/// `addBetweenColumn`) appends `\lx@intercol <filler> \lx@intercol` past the
1666/// column's `\hfil`; when `<filler>` is real content (e.g. a math `|`) Perl's
1667/// back-scan meets it before the fill and leaves the cell right-aligned
1668/// (dginev#742). A plain `|` rule digests to `\vrule\relax`, which is skippable
1669/// and must NOT count as defeating — that cell stays centered with a `border`.
1670/// The skippable set matches [`intercol_reachable_in_before`]'s left-scan.
1671fn template_after_fill_defeated(after: &Option<Tokens>) -> bool {
1672  let Some(toks) = after else { return false };
1673  let toks = toks.unlist_ref();
1674  // Index of the last fill in the template, read as `has_after_fill` reads it ([`is_fill`]).
1675  let last_fill = toks.iter().rposition(is_fill);
1676  let Some(fill_idx) = last_fill else {
1677    return false;
1678  };
1679  toks.iter().skip(fill_idx + 1).any(|tok| {
1680    let s = tok.to_string();
1681    !(s == "\\lx@intercol"
1682      || s.contains("intercol")
1683      || s == "\\lx@column@trimright"
1684      || s == "\\vrule"
1685      // colortbl's `|` paint: `\lx@CT@paint@live`, or `\lx@CT@paint` and its frozen `\lx@CT@arc@<n>` (colortbl_sty.rs)
1686      || s.starts_with("\\lx@CT@")
1687      || s == "\\relax"
1688      || s == "\\hfil"
1689      || s == "\\hfill"
1690      || s == "\\hskip"
1691      || s.trim().is_empty())
1692  })
1693}

2230      // End the column
2231      if before {

2233        repeated = true;
2234        nonreps = std::mem::take(&mut cols);
2235      } else {
2236        // Finished column spec; add it

2259      }
2260      if column_end != Some(("align", false)) {
2261        break; // \cr or \crcr (explicit or implicit) ends the template
2262      }
2263      tokens.push(t);
2264    } else if before {
2265      // Other random tokens go into the column's pre-template
2266      if !pre.is_empty() || cc != Catcode::SPACE {
2267        pre.push(t);
2268      }
2269      tokens.push(t);
2270    } else {
2271      // Or the post-template
2272      if !post.is_empty() || cc != Catcode::SPACE {
2273        post.push(t);
2274      }
2275      tokens.push(t);
2276    }
2277  }
2278  drop(aligning);
2279  expire_align_group_count();

2281  let template_tokens = Tokens::new(tokens.clone());
2282  whatsit.set_property("template_tokens", Stored::Tokens(template_tokens));
2283  // Now create & return the template object
2284  let template = if repeated {
2285    Template::new(TemplateConfig {
2286      columns: Some(nonreps),
2287      repeated: cols,
2288      tokens: Some(tokens),
2289      ..TemplateConfig::default()
2290    })
2291  } else {
2292    Template::new(TemplateConfig {
2293      columns: Some(cols),
2294      tokens: Some(tokens),
2295      ..TemplateConfig::default()
2296    })
2297  };
2298  Ok(template)
2299}

2341/// tex.web §789: an `\omit` cell is restricted horizontal, its material boxed as text (`Alignment::get_column_before`
2342/// opens the `\hbox{`, `get_column_after` closes it), but in TeX its fill glue is the cell's, setting the cell flush
2343/// right or centered (`\omit\hfill $y$`, `\multispan2\hfil … \hfil`, `\omit{\hfill $x$}`: 2306.14299). The box's
2344/// leading fills (and the spaces before them) go before it and its trailing fills after it, where
2345/// `extract_alignment_column`'s scan reads the cell's alignment from them, as from an unboxed cell. Only `\hfil` and
2346/// `\hfill` are fills there (`\hss` leaves no box). Guard
2347/// `perfect_kernel_batch68z::an_omit_cell_text_box_keeps_the_cell_alignment`.
2348fn hoist_omit_text_box_fills(boxes: &mut VecDeque<Digested>) {
2349  let is_hbox = |item: &Digested| match item.data() {
2350    DigestedData::Whatsit(w) => w
2351      .try_borrow()
2352      .is_ok_and(|w| w.get_definition().get_cs_name().as_ref() == "\\hbox"),
2353    _ => false,
2354  };
2355  let is_space = |item: &Digested| {
2356    item.get_property_bool("isSpace") && !item.get_property_bool("isVerticalSpace")
2357  };
2358  let is_fill = |item: &Digested| item.get_property_bool("isFill");
2359  // The cell's contents, as `\hbox`'s list, the fills cut from its front (`leading`) or back.
2360  let cut = |item: &Digested, leading: bool| -> Vec<Digested> {
2361    let DigestedData::Whatsit(w) = item.data() else {
2362      return Vec::new();
2363    };
2364    let Ok(mut w) = w.try_borrow_mut() else {
2365      return Vec::new();
2366    };
2367    let Some(contents) = w.get_arg_mut(2) else {
2368      return Vec::new();
2369    };
2370    // the box's own list, under the lists a group in it leaves (`\omit{\hfill $x$}`: the group's empty `{`/`}`
2371    // markers stay where they are, for its reversion)
2372    let mut items: VecDeque<Digested> = contents.unlist().into();
2373    let mut passed = Vec::new();
2374    let mut last_fill = 0;
2375    loop {
2376      let Some(item) = (if leading {
2377        items.pop_front()
2378      } else {
2379        items.pop_back()
2380      }) else {
2381        break;
2382      };
2383      if matches!(item.data(), DigestedData::List(_)) {
2384        let inner = item.unlist();
2385        if leading {
2386          for inner in inner.into_iter().rev() {
2387            items.push_front(inner);
2388          }
2389        } else {
2390          items.extend(inner);
2391        }
2392        continue;
2393      }
2394      let fill = is_fill(&item);
2395      let passes = fill || is_space(&item) || item.is_empty().unwrap_or(false);
2396      passed.push(item);
2397      if fill {
2398        last_fill = passed.len();
2399      } else if !passes {
2400        break;
2401      }
2402    }
2403    if last_fill == 0 {
2404      return Vec::new();
2405    }
2406    let rest = passed.split_off(last_fill);
2407    let (mut hoisted, kept): (Vec<Digested>, Vec<Digested>) = passed
2408      .into_iter()
2409      .partition(|item| is_fill(item) || is_space(item));
2410    // what is not hoisted goes back, in order
2411    for item in kept.into_iter().chain(rest).rev() {
2412      if leading {
2413        items.push_front(item);
2414      } else {
2415        items.push_back(item);
2416      }
2417    }
2418    if !leading {
2419      hoisted.reverse();
2420    }
2421    let items: Vec<Digested> = items.into();
2422    let mode = match contents.data() {
2423      DigestedData::List(list) => list.borrow().mode.clone(),
2424      _ => None,
2425    };
2426    let mut list = List::new(items);
2427    list.mode = mode;
2428    *contents = list.into();
2429    hoisted
2430  };
2431  // the box, under the lists the cell's material may be wrapped in
2432  while let Some(front) = boxes.front()
2433    && matches!(front.data(), DigestedData::List(_))
2434  {
2435    let front = boxes.pop_front().unwrap();
2436    for item in front.unlist().into_iter().rev() {
2437      boxes.push_front(item);
2438    }
2439  }
2440  if let Some(front) = boxes.front().cloned()
2441    && is_hbox(&front)
2442  {
2443    for item in cut(&front, true).into_iter().rev() {
2444      boxes.push_front(item);
2445    }
2446  }
2447  while let Some(back) = boxes.back()
2448    && matches!(back.data(), DigestedData::List(_))
2449  {
2450    let back = boxes.pop_back().unwrap();
2451    boxes.extend(back.unlist());
2452  }
2453  if let Some(back) = boxes.back().cloned()
2454    && is_hbox(&back)
2455  {
2456    boxes.extend(cut(&back, false));
2457  }
2458}
