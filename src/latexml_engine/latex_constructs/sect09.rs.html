192  DefPrimitive!("\\@@add@caption@counters", {
193    // Perl: $type = ToString(Digest(T_CS('\@captype')))
194    // Rust port had used `stomach::digest`, but stomach::digest's
195    // `read_x_token` loop in vmode (figure-environment body is vmode)
196    // can leak a trailing `\par` token into the result when the captype
197    // expansion completes — the digester continues reading beyond the
198    // expansion and picks up an environment-emitted `\par`. Use
199    // `do_expand` instead: it expands the macro one level (`\@captype`
200    // → "figure" letter tokens) and stops, mirroring Perl's `ToString`
201    // of the captype's body without invoking stomach digestion.
202    // Witness: math0010095 BoxedEPS+figure+caption produced
203    // `\thefigure\par` undefined errors when captype was "figure\par".
204    let captype = do_expand(T_CS!("\\@captype"))?.to_string();
205    let prekey = s!("PREINCREMENTED_{captype}");
206    let props = match remove_value(&prekey) {
207      Some(Stored::HashStored(pre)) => pre,
208      _ => step_float_counter(&captype)?,
209    };
210    // The list's extension, read as `\@captype` is above: digested, its letters began a paragraph in the float's
211    // vertical list, whose horizontal mode outlived the counters, and the caption's box then ended that paragraph with
212    // a `\par` (an aas `\references` entry in a table, `perfect_kernel_batch64x::bibitem_after_a_float_reopens_the_list`).
213    let inlist = do_expand(T_CS!(s!("\\ext@{}", captype)))?.to_string();
214    assign_value(
215      &s!("{}_tags", captype),
216      props.get("tags"),
217      Some(Scope::Global),
218    );
219    // A sectioning unit in the float's body fixed the float's id first (`pin_float_id_before_its_sectioning`).
220    let pinned = remove_value(&s!("lx@float@pinned@id@{captype}"));
221    assign_value(
222      &s!("{}_id", captype),
223      pinned.as_ref().or(props.get("id")),
224      Some(Scope::Global),
225    );
226    assign_value(&s!("{}_inlist", captype), inlist, Some(Scope::Global));
227    // How many captions of the type were made, for a box that holds one (`end_caption_box`).
228    let count = s!("{CAPTION_COUNT_PREFIX}{captype}");
229    assign_value(&count, lookup_int(&count) + 1, Some(Scope::Global));
230  });

293  // no-`\@captype` path emits — an
294  // `ltx:text class="ltx_caption"` holding the title, minus the counter tag
295  // (which no inline element may carry) — and drop the toc entry. Guard:
296  // `perfect_kernel_batch54::caption_outside_a_float_becomes_its_float`.
297  // Perl names this constructor `\@@caption` (latex_constructs.pool.ltxml:3368), a name LaTeX leaves to classes:
298  // the Cambridge University Press classes keep `\caption` there for their `\contcaption` (fac.cls:803-805,
299  // ijsipaper.cls:876-878, mscs.cls:794-795), and with the kernel's `\@caption@@@` calling `\@@caption` every
300  // caption then called itself (Perl `deep_recursion`, ours `Stomach:Recursion`; witnesses 1209.5152, 1311.2960,
301  // 1410.5131, 1610.02500, 2205.14586, 2212.13027, 1604.04575, 2112.07588). The constructor is `\lx@@caption`, which
302  // the kernel and the bindings call; `\@@caption` stays its alias, so a class's own meaning for the name is the
303  // class's (OXIDIZED_DESIGN #566, KNOWN_PERL_ERRORS #637). Guard `perfect_kernel_batch66b::cup_contcaption_keeps_its_caption`.
304  // The material is a paragraph of a vertical box (`PBoxContents`, as Perl's `\@@caption PBoxContents`,
305  // latex_constructs.pool.ltxml:3368), so a `$$…$$` in it is display math (1005.1555, 1907.01498, 2310.11556,
306  // 1404.2359; OXIDIZED_DESIGN_DIVERGENCES #769); `\@@toccaption` below alike.
307  DefConstructor!("\\lx@@caption PBoxContents", sub[document, args, props] {
308    let body = args[0].clone();
309    // The list entry `\@@toccaption` could not place (below), for the float this caption may become.
310    let unplaced_toccaption = remove_value("lx@unplaced@toccaption");
311    // A caption whose `\@captype` is set in a box that is not a float
312    // (tufte-common.def:1110-1133 `marginfigure`: a minipage with `\def\@captype
313    // {figure}` inside `\marginpar`; the `\def\@captype{figure}` minipage idiom)
314    // IS that type's caption in LaTeX — numbered "Figure 1:", the target of a
315    // following `\label` — so it becomes one: a float element of its type at the
316    // nearest ancestor that admits it (the minipage's `inline-logical-block`, whose
317    // model is `Para.model`), carrying the counter tags and id the caption stepped,
318    // the shape `\captionof` already gives (OXIDIZED_DESIGN #89). Before, #182
319    // degraded it to `ltx:text class="ltx_caption"`: no number, and a `\ref` to
320    // its `\label` dangled (pgfornament ornaments ×40, the tufte manuals). Guard:
321    // `perfect_kernel_batch54::caption_outside_a_float_becomes_its_float`.
322    // A sub-float's caption (`\subcaption`'s `\@captype` is `sub<type>`, one prefix: Perl subcaption.sty.ltxml:50-53)
323    // becomes the element of its parent type, as `{subfigure}` is an `ltx:figure` (subcaption.sty.ltxml:60-68;
324    // 2609.33998's teaser).
325    let float_tag = props.get("captype").map(|t| {
326      let captype = t.to_string();
327      match captype.strip_prefix("sub").unwrap_or(&captype) {
328        "figure" => "ltx:figure",
329        "table" => "ltx:table",
330        _ => "ltx:float",
331      }
332    });
333    if !caption_can_float(document, "ltx:caption")
334      && let Some(float_tag) = float_tag
335      && let Some(Stored::String(id)) = props.get("float_id")
336      && caption_can_float(document, float_tag)
337    {
338      let save = document.float_to_element(float_tag, true)?;
339      let mut attrs = string_map!("xml:id" => with(*id, |s| s.to_string()));
340      if let Some(inlist) = props.get("float_inlist") {
341        let inlist = inlist.to_string();
342        if !inlist.trim().is_empty() {
343          attrs.insert("inlist".to_string(), inlist);
344        }
345      }
346      document.open_element(float_tag, Some(attrs), None)?;
347      if let Some(Stored::Digested(tags)) = props.get("float_tags") {
348        document.absorb(tags, None)?;
349      }
350      if let Some(Stored::Digested(toccaption)) = unplaced_toccaption {
351        document.open_element("ltx:toccaption", None, None)?;
352        document.absorb(&toccaption, None)?;
353        document.close_element("ltx:toccaption")?;
354      }
355      document.open_element("ltx:caption", None, None)?;
356      if let Some(ref body) = body {
357        document.absorb(body, None)?;
358      }
359      document.close_element("ltx:caption")?;
360      document.close_element(float_tag)?;
361      if let Some(save) = save {
362        document.set_node(&save);
363      }
364    } else if caption_can_float(document, "ltx:caption") {
365      // `^^`: float up, closing what can be closed on the way.
366      let save = document.float_to_element("ltx:caption", true)?;
367      document.open_element("ltx:caption", None, None)?;
368      if let Some(ref body) = body {
369        document.absorb(body, None)?;
370      }
371      document.close_element("ltx:caption")?;
372      if let Some(save) = save {
373        document.set_node(&save);
374      }
375    } else {
376      let node = document.open_element(
377        "ltx:text",
378        Some(string_map!("class" => "ltx_caption")),
379        None,
380      )?;
381      if let Some(ref body) = body {
382        absorb_without_tags(document, body)?;
383      }
384      document.maybe_close_node(&node)?;
