225  DefMacro!("\\lx@ifusebbl{}{}{}", sub[(bib_files_tks, bbl_clause, bib_clause)] {
226    // Every `\bibliography` call is counted, for a package that asks whether one followed its own: bibentry's
227    // `\nobibliography*` warns at the end when none did (bibentry.sty:74-79, whose `\def\bibliography` wrapper the
228    // lock refuses; latexml_contrib bibentry_sty.rs). Witnesses 2505.07606 (`\nobibliography*` then the

239      // `return unless $bib_files;`) and so did this port — silently, in both
240      // engines. But real `latex.ltx` ends `\bibliography` with an
241      // UNCONDITIONAL `\@input@{\jobname.bbl}`, so `\bibliography{}` beside a
242      // shipped `<jobname>.bbl` renders the full reference list under

244      // the `\cite` resolves to [1]). Following latex.ltx here recovers the
245      // references instead of dropping them: 7 papers in the 2605+2606
246      // sandboxes, including the GWTC-5 LIGO set, all of which ship a
247      // jobname-matching `.bbl`. Witness 2605.27226 (repro
248      // `docs/parity/bib_absence_2026-07-29/repros/f3_empty_arg_bbl/`).
249      // OXIDIZED_DESIGN #86; audit family F3(a).
250      if FindFile!(&jobname, type => "bbl").is_some() {
251        return Ok(bbl_clause);
252      }
253      return Ok(Tokens!());
254    }
255    let bbl_path = FindFile!(&jobname, type => "bbl");
256    // BIB_CONFIG is a list of phases; with bibconfig=bbl,bib: try bbl first, fall back to bib.
257    // Default (bibtex option) is ['bib', 'bbl']; nobibtex sets ['bbl'].
258    let default_bib_config: Rc<[SymStr]> = Rc::new([pin("bib"), pin("bbl")]);
259    let bib_config = match lookup_value("BIB_CONFIG") {
260      Some(Stored::Strings(v)) => v,
261      _ => default_bib_config,
262    };
263    if bib_config.is_empty() {
264      Info!("missing", "bib_config", "BIB_CONFIG was empty, ignoring bibliography phase.");
265      return Ok(Tokens!());
266    }
267    // Perl `\lx@ifusebbl` (latex_constructs.pool.ltxml L3967-3983) acts on the
268    // FIRST configured phase ONLY (`$$bib_config[0]`); it does NOT fall through
269    // to later phases. The earlier Rust port iterated all phases as a fallback
270    // chain, which — with a `bbl,bib` config and no `\jobname.bbl` on disk —
271    // fell through to the `bib` phase and emitted a spurious empty
272    // `\lx@bibliography` placeholder, i.e. a SECOND, content-less "References"
273    // section, whenever the real entries arrived via a differently-named
274    // `.bbl` processed as content. Witness 2107.03065 (refs.bbl, not
275    // <jobname>.bbl): Perl emits 1 bibliography, the old Rust 2. Mirror Perl:
276    // branch on the first phase only.
277    let first_is_bbl = with(bib_config[0], |s| s == "bbl");
278    if first_is_bbl {
279      if bbl_path.is_some() {
280        return Ok(bbl_clause);
281      }
282      // bbl-first precedence (e.g. ar5iv's bibconfig=bbl,bib) but no
283      // <jobname>.bbl on disk: fall through to a configured 'bib' phase, BUT
284      // only when the .bib files actually exist. Emitting \lx@bibliography
285      // unconditionally would add an empty placeholder "References"; when the
286      // real entries instead arrive via a manual \input{refs.bbl} (witness
287      // 2107.03065, which ships refs.bbl and NO refs.bib) that produces a
288      // duplicate, content-less bibliography. Requiring a real .bib delivers
289      // true "bbl-over-bib" precedence (prefer .bbl, else .bib) without the
290      // double. Witness 2605.16562 (refs.bib, no .bbl) now gets a bibliography.
291      let bib_in_config = bib_config.iter().any(|p| with(*p, |s| s == "bib"));
292      let all_bibs_exist = bib_in_config
293        && bib_files
294          .split(',')
295          .map(str::trim)
296          .filter(|bf| !bf.is_empty())
297          .all(|bf| FindFile!(bf, type => "bib").is_some());
298      if all_bibs_exist {
299        return Ok(bib_clause);
300      }
301      Info!("expected", "bbl", "Couldn't find bbl file, bibliography may be empty.");
302      Ok(Tokens!())
303    } else {
304      // 'bib' phase — check if .bib files exist
305      let mut missing_bibs = String::new();
306      for bf in bib_files.split(',').map(str::trim).filter(|bf| !bf.is_empty()) {
307        let bib_path = FindFile!(bf, type => "bib");
308        if bib_path.is_none() {
309          if !missing_bibs.is_empty() {
310            missing_bibs.push(',');
311          }
312          missing_bibs.push_str(bf);
313        }
314      }
315      if missing_bibs.is_empty() || bbl_path.is_none() {
316        Ok(bib_clause)
317      } else {
318        Info!("expected", missing_bibs, s!("Couldn't find all bib files, using {jobname}.bbl instead"));
319        Ok(bbl_clause)
320      }
321    }
322  });

1760/// A labelled node answers, for each role whose tags are held (`type_tag_deferred`, `\lx@tag@intags@held`; ruling 7e),
1761/// for the tag a reference to its labels prints: its own tag of that role, else the nearest ancestor's — CrossRef's
1762/// walk up for a show word the labelled entry lacks (`\section*` under a numbered chapter prints the chapter's).
1763/// Every label counts, however it was set (`\lx@label`, longtable's `labels=`, a panel merge), as the node closes.
1764/// The held diagnostics are raised if a reference shows the role (logger.rs `replay_shown_deferred`). Witness
1765/// biblatex-gost-examples.
1766fn tie_held_tags(node: &Node, labels: &str) {
1767  let roles = held_tag_roles(with_mapping_keys("type_tag_deferred", |keys| keys));
1768  if roles.is_empty() {
1769    return;
1770  }
1771  // A node's element children, walked lazily (an ancestor may be wide: the document root).

1773    std::iter::successors(parent.get_first_child(), |child| child.get_next_sibling())
1774      .filter(|child| child.get_type() == Some(NodeType::ElementNode))
1775  };

1777    let mut cursor = Some(node.clone());
1778    while let Some(current) = cursor {
1779      let tag = elements(&current)
1780        .filter(|child| child.get_name() == "tags")
1781        .find_map(|tags| {
1782          elements(&tags).find(|tag| tag.get_attribute("role").as_deref() == Some(role))
1783        });

1785        if let Some(id) = tag
1786          .get_attribute("_deferred")
1787          .and_then(|id| id.parse().ok())
1788        {
1789          for label in labels.split_whitespace() {
1790            util::logger::defer_to_label(label, role, id);
1791          }
1792        }
1793        break;
1794      }
1795      cursor = current.get_parent();
1796    }
1797  }

2493/// One `\newlabel{<name>}{<value>}`, the record of a label an `.aux` file holds (latex.ltx `\newlabel` → `\@newl@bel`,
2494/// `\global\@namedef{r@<name>}{<value>}`): the value's first group is the text `\ref` prints, hyperref's fourth the
2495/// anchor; a `<label>@cref` record is cleveref's type and number (cleveref.sty `\cref@label`). `url` is the document
2496/// the anchors are in, where one is given (xr's `\externaldocument[…]{…}[url]`). Records are kept as xr's are
2497/// (`record_external_label`): references to a label this document holds after all print its own.
2498pub fn record_newlabel(name: &str, value: &str, url: Option<&str>) {
2499  let fields: Vec<&str> = brace_groups(value).collect();
2500  let Some(text) = fields.first().map(|text| text.trim()) else {
2501    return;
2502  };
2503  if let Some(key) = name.strip_suffix("@cref") {
2504    if let Some((ty, number)) = cref_record(text) {
2505      record_external_cref(&clean_label(key, None), ty, number);
2506    }
2507    return;
2508  }
2509  let anchor = fields
2510    .get(3)
2511    .map(|anchor| anchor.trim())
2512    .filter(|anchor| !anchor.is_empty());
2513  let href = url.filter(|url| !url.is_empty()).map(|url| match anchor {
2514    Some(anchor) => s!("{url}#{anchor}"),
2515    None => url.to_string(),
2516  });
2517  record_external_label(&clean_label(name, None), text, anchor, href.as_deref());
2518}

2520/// The `\newlabel{<name>}{<value>}` records of text written to an `.aux` file, which LaTeX reads back at the next
2521/// run's `\begin{document}`, before any reference (latex.ltx `\document`, `\@input{\jobname.aux}`). A single run reads
2522/// each as it is written, for the references after it: a document's own label macros write them — 2503.03625's
2523/// `\customlabel{eq:TC-1}{TC-1}` (`\protected@write\@auxout{}{\string\newlabel{#1}{{#2}…}}`), 1004.2872's
2524/// `\phantomequation` (`\immediate\write\@auxout`), theoremref's `\thlabel` — and Perl, which reads no `.aux`, leaves
2525/// their references missing.
2526pub fn read_back_aux_text(text: &str) -> Result<()> {
2527  let mut rest = text;
2528  while let Some(at) = rest.find("\\newlabel") {
2529    rest = &rest[at + "\\newlabel".len()..];
2530    let mut groups = brace_groups(rest);
2531    if let (Some(name), Some(value)) = (groups.next(), groups.next()) {
2532      let name = name.trim();
2533      record_newlabel(name, value, None);
2534      // read as the `.aux` is, `@` a letter
2535      let value = mouth::tokenize_internal(TeXString::assembled(value.to_string()));
2536      DefMacro!(T_CS!(s!("\\r@{name}")), None, value, scope => Some(Scope::Global));
2537    }
2538  }
2539  Ok(())
2540}

2542/// The balanced `{…}` groups at the start of `text`, their contents without the outer braces; a `\{` or `\}` does not
2543/// count.
2544pub fn brace_groups(text: &str) -> impl Iterator<Item = &str> {
2545  let mut rest = text;
2546  std::iter::from_fn(move || {
2547    let start = rest.find(|c: char| !c.is_whitespace())?;
2548    if !rest[start..].starts_with('{') {
2549      return None;
2550    }
2551    let (mut depth, mut escaped) = (0usize, false);
2552    for (offset, c) in rest[start..].char_indices() {
2553      match c {
2554        _ if escaped => escaped = false,
2555        '\\' => escaped = true,
2556        '{' => depth += 1,
2557        '}' => {
2558          depth -= 1;
2559          if depth == 0 {
2560            let group = &rest[start + 1..start + offset];
2561            rest = &rest[start + offset + 1..];
2562            return Some(group);
2563          }
2564        },
2565        _ => {},
2566      }
2567    }
2568    None
2569  })
2570}

2572/// The type and the number of a cleveref record, `[<type>][<n>][<parents>]<number>` (cleveref.sty `\cref@label`).
2573fn cref_record(text: &str) -> Option<(&str, &str)> {
2574  let mut rest = text.trim_start();
2575  let mut groups = Vec::new();
2576  for _ in 0..3 {
2577    let inner = rest.strip_prefix('[')?;
2578    let close = inner.find(']')?;
2579    groups.push(&inner[..close]);
2580    rest = &inner[close + 1..];
2581  }
2582  Some((groups[0].trim(), rest.trim()))
2583}
