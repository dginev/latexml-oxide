114  // latex.ltx:17332 `\newcount\c@secnumdepth` leaves it 0, as Perl does (latex_base.pool.ltxml:312): the class sets
115  // it (article.cls 3, report/book 2), and letter.cls sets nothing, so a letter's `\section` is unnumbered and needs no
116  // `\thesubsection` (the rawclasses lever's 11 letter papers, 0907.4214; regression study R8). The kernel's own 3

118  NewCounter!("secnumdepth");
119  DefMacro!(
120    "\\@startsection{}{}{}{}{}{} OptionalMatch:*",
121    sub[(type_tokens, level_arg, _ignore3, _ignore4, _ignore5, _ignore6, flag)] {
122      // Aside: Guard mode
123      // Never start sections in math mode -- this is a good recovery point for broken documents
124      if lookup_bool_sym(pin!("IN_MATH")) {
125        let mode = lookup_string_from_sym(pin!("MODE"));
126        if mode.contains("math") { // double-check we're really in math
127          end_mode(&mode)?;
128        } else { // otherwise, just unset the flag?
129          assign_value("IN_MATH", false, Some(Scope::Global));
130        }
131      }
132      // Main logic. The level is a TeX <number> — latex.ltx `\@sect` compares
133      // it with `\ifnum #2>\c@secnumdepth` — so a non-literal level is READ as
134      // one: scrartcl.cls L3421/L3425 pass every heading's level as
135      // `{\numexpr #2\relax}` with `#2` = `\csname <name>numdepth\endcsname`.
136      // Perl's `$level > …` string-coerces that to 0, numbering every KOMA
137      // heading down to `\subparagraph` (witness tudaexercise / any raw KOMA
138      // class; guard `perfect_kernel_batch53::startsection_level_is_a_tex_number`).
139      // An empty or unreadable level still coerces to 0 as in Perl.
140      let level = level_arg.to_string();
141      let level_int = match level.trim().parse::<i64>() {
142        Ok(n) => n,
143        Err(_) if level.trim().is_empty() => 0,
144        Err(_) => {
145          let level_tokens = level_arg.clone();
146          let mouth = Mouth::new("", None)?;
147          reading_from_mouth(mouth, move || {
148            unread(level_tokens);
149            read_number()
150          })
151          .map(|n| n.value_of())
152          .unwrap_or(0)
153        },
154      };
155      // A section type the schema does not know (`\DeclareSectionCommand`
156      // headings) is bound to the element of its level here — the only place
157      // the level is visible; see `section_element_for_type`.
158      let stype = strip_trailing_cs(type_tokens.to_string().trim());
159      if !stype.is_empty() && !is_known_section_type(&stype) && stype != "app" {
160        assign_mapping("SECTION_ELEMENT", &stype, Some(pin(section_element_for_level(level_int))));
161      }
162      // An empty type names no unit and no counter — latex.ltx `\@sect` reaches `\refstepcounter{#1}` only at a level
163      // within secnumdepth, where pdflatex rejects an empty one (`\refstepcounter{}`: no counter '') — so the heading
164      // is an unnumbered `subparagraph`, the lowest unit, which closes nothing above it: amsart's `\@starttoc` heads its
165      // lists with `\@startsection{}\@M…`, exframe.sty:538 its problems with `\@startsection{}{}…*` (an empty level,
166      // read as 0, made every later section a chapter's). Kept empty, `RefStepID`'s `NewCounter` would define `\the`
167      // and `\p@` themselves, and every later `\the` printed 0 (an amsart-copy class, 1102.4889: 94 errors; Perl the
168      // same, KNOWN_PERL_ERRORS #483).
169      let empty_type = stype.is_empty();
170      let type_tokens = if empty_type {
171        Tokens::new(Explode!("subparagraph"))
172      } else {
173        type_tokens.clone()
174      };
175      // The heading ends the trivlists begun in its group (`\lx@trivlist@end@group`, sect06.rs; KNOWN_PERL_ERRORS
176      // #456): an unended `\trivlist\item[Proof.]` does not hold the section, nor the lists after it.
177      let mut tokens: Vec<Token>;
178      if flag.is_some() { // No number, not in TOC
179        tokens = vec![
180          T_CS!("\\par"), T_CS!("\\lx@trivlist@end@group"), T_CS!("\\@startsection@hook"), T_CS!("\\@@unnumbered@section"),
181        T_BEGIN!()];
182        tokens.extend(type_tokens.unlist());
183        tokens.extend(vec![T_END!(), T_BEGIN!(), T_END!()]);
184      } else if empty_type || level_int > CounterValue!("secnumdepth").value_of() ||
185        lookup_bool("no_number_sections") ||
186        // A class whose chapters are numbered in the main matter only (book.cls:355-366 `\@chapter`'s
187        // `\if@mainmatter`, after `\backmatter`; book_cls.rs): the chapter level (an appendix's too), the other
188        // levels numbered still.
189        (level_int == 0 && lookup_bool("no_number_chapters")) {
190        // No number, but in TOC
191        tokens = vec![
192          T_CS!("\\par"), T_CS!("\\lx@trivlist@end@group"), T_CS!("\\@startsection@hook"), T_CS!("\\@@unnumbered@section"),
193        T_BEGIN!()];
194        tokens.extend(type_tokens.unlist());
195        tokens.extend(vec![T_END!(), T_BEGIN!(), T_OTHER!("toc"), T_END!()]);
196      } else { // Number and in TOC
197        tokens = vec![T_CS!("\\par"), T_CS!("\\lx@trivlist@end@group"), T_CS!("\\@startsection@hook"), T_CS!("\\@@numbered@section"),
198        T_BEGIN!()];
199        tokens.extend(type_tokens.unlist());
200        tokens.extend(vec![T_END!(), T_BEGIN!(), T_OTHER!("toc"), T_END!()]);
201      };
202      // At section level or above, a heading inside an abstract or keywords ends it first
203      // (`\lx@heading@ends@frontmatter`, sect05.rs; user ruling 2026-10-18).
204      if level_int <= 1 {
205        tokens.insert(0, T_CS!("\\lx@heading@ends@frontmatter"));
206      }
207      Ok(Tokens::new(tokens))
208    },
209    locked => true
210  );
211  // The dispatcher under a name of its own, which the class-level workers a raw `\@startsection` calls back
212  // (`\@sect`, `\@part`, …; latex_constructs_rust_only.rs) route to: a class that re-lets `\@startsection` to a
213  // latex.ltx-style worker ending in `\@sect` (cup-journal.cls:1066-1076) otherwise ran `\@sect` → its worker →
214  // `\@sect` until the recursion limit, a Fatal (2112.11969).

218    "\\@@numbered@section{} Undigested OptionalUndigested Undigested",
219    sub[document, args, props] {
220      // args:=(stype,inlist,toctitle,title)
221      // Sanitize stype: under some upstream conditions (figure-block + section
222      // sequencing — see math0010095 / Cluster A) the {} parameter reader picks
223      // up a trailing \par token that pollutes the section type identifier.
224      // \par should never appear inside a section type; strip any trailing
225      // backslash-prefixed CS to recover the bare identifier.
226      let stype = type_name(args[0].as_ref().unwrap());
227      let inlist = args[1].as_ref().unwrap().to_string();
228      // TODO: This bizarre argument API interaction needs to be simplified down to Perl's
229      // intuitive level of:       let (x,y,z, ...) = @args;
230      // If backmatter, find insertion point as if inserting the backmatter element type
231      if let Some(asif) = props.get("backmatterelement") {
232        let target = backmatter_insertion_target(
233          document, &asif.to_string(),
234          &section_element_for_type_maybe(&stype).unwrap_or_else(|| "ltx:section".into()));
235        let point = document.find_insertion_point(&target, None)?;
236        document.set_node(&point);
237      }
238      let clean_id = prop_string!(props,"id"); // TODO: CleanID($id);
239      // Mirror Perl `latex_constructs.pool.ltxml:599-607`: sanitize the type
240      // name so we don't open arbitrary elements outside the schema. Perl
241      // checks `isKnownTag(ltx:$type)`; if unknown, special-cases `ltx:app
242      // → ltx:appendix`, otherwise falls back to `ltx:section` and warns.
243      // Without this guard, papers that do
244      // `\newcommand\Proof{\@startsection{Proof}{5}{...}}` (e.g.
245      // mst-stylefile.sty in 1608.04650) opened `<ltx:Proof>` and cascaded
246      // 1500+ malformed errors on every nested element.
247      let mut tagname = section_element_for_type(&stype, true);
248      let mut attrs = string_map!("xml:id" => clean_id, "inlist" => inlist);
249      // An appendix unit in a float body goes into its inline sectional block (document.rs `AUTO_OPEN_BRIDGES`),
250      // which holds sections, not appendices: a section of class `ltx_appendix`, its letter, id and labels kept
251      // (2609 witnesses, 41 of the 117 errors).
252      if tagname == "ltx:appendix" && props.get("in_float_body").is_some() {
253        tagname = "ltx:section".into();
254        attrs.insert("class".to_string(), "ltx_appendix".to_string());
255      }
256      let section = document.open_element(&tagname, Some(attrs), None)?;
257      // TODO: Another instance where the immutability of props causes endless cloning
258      //       which is slow and wasteful.
259      // The big problem is that for props to be mutable, the entire parent whatsit needs to
260      // be mutable, and Rust hits a mutability conflict between the parent, and the
261      // "args" and "props" children ... will come back here after performance becomes
262      // an issue again
263      //
264      // Part 2: I have now, with great attention and profiling, solidified the position that
265      //       Whatsits are immutable during the absorption phase -- and hence
266      // the args and props passed in here will remain immutable in latexml_oxide.
267      // Hence, for this absorb call to run correctly, it must either:
268      // 1) Accept a cloned value as currently, paying with performance
269      // 2) Accept immutable references to digested objects,
270      // which may lead to far-reaching borrowing constraints
271      //   e.g. unlist()-ing a digested List will have to produce box references,
272      //  rather than provide the owned boxes directly.
273      //   would have to experiment with this - as it is of course much lighter on performance
274      //
275
276      // Update 2022: The notes are generally still accurate,
277      // but cloning a Digested object is now cheap enough,
278      // as each enum variant is guarded by an Rc reference counter. Rc<Tbox>, Rc<List>, etc.
279      if let Some(Stored::Digested(tags)) = props.get("tags") {
280        document.absorb(tags, None)?;
281      }
282      let title = prop_digested!(props, "title");
283      document.insert_element("ltx:title", title, None)?;
284
285      let toctitle = prop_digested!(props, "toctitle");
286      if !toctitle.is_empty() {
287        document.insert_element("ltx:toctitle", toctitle, None)?;
288      }
289      precede_heading_with_its_pagebreaks(&section)?;
290    },
291    properties => sub[args] {
292      // The document body begins: the frontmatter digested so far is the paper's (`seal_digested_frontmatter`).
293      seal_digested_frontmatter();
294      let stype = args[0].as_ref().unwrap();
295      // let inlist = args[1].as_ref().unwrap();
296      let toctitle_arg = args[2].as_ref();
297      let title = args[3].as_ref().unwrap();
298
299      maybe_peek_label()?;
300      // See Cluster A note in the body closure above; sanitize identical here.
301      let stype_str = type_name(stype);
302      pin_float_id_before_its_sectioning()?;
303      let mut props = ref_step_counter_by_meaning(&stype_str)?;
304      // Digested in a float body (the constructor, which runs after the float, reads it for an appendix unit).
305      if lookup_bool("lx@in@float") {
306        props.insert("in_float_body", Stored::Bool(true));
307      }
308      // For appendix, look up the backmatter element mapping
309      if stype_str == "appendix"
310        && let Some(bme) = lookup_mapping("BACKMATTER_ELEMENT", &s!("ltx:{stype_str}")) {
311          props.insert("backmatterelement", bme);
312        }
313      let short = toctitle_arg.filter(|v| !v.to_string().is_empty());
314      let toctitle = short.unwrap_or(title);
315      // Cluster A: rebuild tokens from sanitized stype_str so the trailing
316      // \par token doesn't propagate into \lx@format@title@@'s body.
317      let stype_clean_tokens = Tokens::new(Explode!(stype_str));
318      // The title font is handed the title as a braced group, as latex.ltx:17256-17259 `\@sect` hands `#6` its
319      // `{\@hangfrom…#8\@@par}`: a style ending in a one-argument command (sectsty's `\sectionfont{\large\centering
320      // \uppercase}`, 1509.02957: "Expected opening '{'") takes the whole title; the group is transparent for a
321      // declaration. A caption is not braced: `\@makecaption` sets its text in one `\sbox` with no font argument
322      // (2206.05542, 2407.00688). Guard `perfect_kernel_batch73b::section_font_ending_in_a_command_takes_the_title`.
323      let title_tokens =
324        Tokens::new(std::iter::once(T_BEGIN!()).chain(title.revert()?.unlist()).chain(std::iter::once(T_END!())).collect());
325      let invoked_title =
326        Invocation!(T_CS!("\\lx@format@title@@"), vec![stype_clean_tokens.clone(), title_tokens]);
327      let xtitle    = digest(invoked_title)?;
328      // Without a `[short]` the toc copy re-reads the title, which the heading typesets: `\lx@reread@copy` (a Warning for
329      // one source occurrence is given there only: OXIDIZED_DESIGN_DIVERGENCES #721; guard
330      // `perfect_kernel_batch71a::undelimited_argument_warning_once_per_occurrence`). A `[short]` is its own source,
331      // digested here only.
332      let mut toctitle_tokens = toctitle.revert()?;
333      if short.is_none() {
334        toctitle_tokens = Tokens::new(
335          std::iter::once(T_CS!("\\lx@reread@copy")).chain(toctitle_tokens.unlist()).collect(),
336        );
337      }
338      let invoked_toctitle = Invocation!(T_CS!("\\lx@format@toctitle@@"),
339          vec![stype_clean_tokens, toctitle_tokens]);
340      let xtoctitle = digest(invoked_toctitle)?;
341
342      if xtoctitle.to_string() != xtitle.to_string() {
343        props.insert("toctitle", xtoctitle.into());
344      }
345      props.insert("title", xtitle.into());
346
347      Ok(props)
348    },
349    after_digest => sub[whatsit] { after_heading_hook(whatsit)?; }
350  );

353  DefConstructor!("\\@@unnumbered@section{} Undigested OptionalUndigested Undigested",
354  sub[document, args, props] {
355      let stype = args[0].as_ref().unwrap();
356      let inlist = args[1].as_ref().unwrap();
357      // If backmatter, find insertion point as if inserting the backmatter element type
358      if let Some(asif) = props.get("backmatterelement") {
359        let target = backmatter_insertion_target(
360          document, &asif.to_string(),
361          &section_element_for_type_maybe(&type_name(stype))
362            .unwrap_or_else(|| "ltx:section".into()));

371      let mut attrs = string_map!(
372          "xml:id" => clean_id(&id),
373          "inlist"  => inlist.to_string()
374        );
375      // An appendix unit in a float body, as in `\@@numbered@section` (a `\section*` after `\appendix` in a figure:
376      // 2605.02378, 2605.25534).
377      if tagname == "ltx:appendix" && props.get("in_float_body").is_some() {
378        tagname = "ltx:section".into();
379        attrs.insert("class".to_string(), "ltx_appendix".to_string());
380      }
381      let section = document.open_element(&tagname, Some(attrs), None)?;
382      let title = prop_digested!(props, "title");
383      document.insert_element("ltx:title", title, None)?;

391    properties => sub[args] {
392      use DigestedData::*;
393      // The document body begins (`seal_digested_frontmatter`), as for a numbered unit.
394      seal_digested_frontmatter();
395      let stype = args[0].as_ref().unwrap();
396      // let inlist = args[1].as_ref().unwrap();
397      let toctitle_arg = args[2].as_ref();
398      let title = args[3].as_ref().unwrap();
399      maybe_peek_label()?;
400      // Cluster A sanitization (see \@@numbered@section).
401      let stype_str = type_name(stype);
402      let mut props = RefStepID!(&stype_str)?;
403      // Digested in a float body, as for a numbered unit.
404      if lookup_bool("lx@in@float") {
405        props.insert("in_float_body", Stored::Bool(true));
406      }
407      // For appendix, look up the backmatter element mapping
408      if stype_str == "appendix"
409        && let Some(bme) = lookup_mapping("BACKMATTER_ELEMENT", &s!("ltx:{stype_str}")) {
410          props.insert("backmatterelement", bme);
411        }
412      let title_digested = if let Postponed(tokens) = title.data() {
413        // TODO: is .clone() on the tokens before they are unlisted a sign that
414        // the DigestedData::Postponed variant isn't ideal?
415        // should we be draining it? Or is there a better conceptual organization?
416        // In the class's title font, as a numbered unit's (`\lx@format@title@@`): Perl

627/// A page break digested INSIDE a section heading — titlesec's `\titleformat{…}
628/// {\clearpage…}` runs its format while the title is built — floats out of
629/// `<ltx:title>` and lands between the title and the `<ltx:toctitle>`, where the
630/// section model has no room for it (bmstu-iu8/bmstu-example,
631/// IU8-02-construction.sty:26-27: `element "toctitle" not allowed here`; SHARED,
632/// Perl places it identically). In TeX the break comes BEFORE the heading, so each
633/// `<ltx:pagination>` among the section's leading heading elements is moved in front
634/// of the section when its parent admits it, else after the headings. Content-free:
635/// nothing visible moves. OXIDIZED_DESIGN #264. Guard `perfect_kernel_batch56::pagebreak_in_a_title_format_precedes_the_section`.
636fn precede_heading_with_its_pagebreaks(section: &Node) -> Result<()> {
637  use latexml_core::document::{can_contain_qsym, get_node_qname};
638  let heading = |q: SymStr| {
639    q == pin_static("ltx:tags") || q == pin_static("ltx:title") || q == pin_static("ltx:toctitle")
640  };
641  let pagination = pin_static("ltx:pagination");
642  let kids: Vec<Node> = section
643    .get_child_nodes()
644    .into_iter()
645    .filter(|c| c.get_type() == Some(NodeType::ElementNode))
646    .collect();
647  // The leading run of headings and page breaks; only breaks BEFORE the last heading move.
648  let run = kids
649    .iter()
650    .take_while(|k| {
651      let q = get_node_qname(k);
652      heading(q) || q == pagination
653    })
654    .count();
655  let Some(last_heading) = kids[..run].iter().rposition(|k| heading(get_node_qname(k))) else {
656    return Ok(());
657  };
658  let breaks: Vec<Node> = kids[..last_heading]
659    .iter()
660    .filter(|k| get_node_qname(k) == pagination)
661    .cloned()
662    .collect();
663  if breaks.is_empty() {
664    return Ok(());
665  }
666  let parent_admits = section
667    .get_parent()
668    .is_some_and(|p| can_contain_qsym(get_node_qname(&p), pagination));
669  let mut anchor = kids[last_heading].clone();
670  for mut b in breaks {
671    b.unlink();
672    if parent_admits {
673      let mut s = section.clone();
674      s.add_prev_sibling(&mut b)?;
675    } else {
676      anchor.add_next_sibling(&mut b)?;
677      anchor = b;
678    }
679  }
680  Ok(())
681}

684/// `\processmotto`) set material of the unit after its title. A binding that has some defines
685/// `\lx@startsection@afterhook{<type>}{<inlist>}` — `<inlist>` empty for a starred heading, whose class head may set less
686/// (svmult's `\@makeschapterhead` only the motto, :649) — and the heading puts it back to be read right after itself,

689fn after_heading_hook(whatsit: &mut Whatsit) -> Result<()> {

691    && let Some(stype) = whatsit.get_arg(1)

708/// The `select` of a table of contents: the sectional units down to `tocdepth`, then the appendix, index and
709/// bibliography. Perl latex_constructs.pool.ltxml L727-733: `$td = tocdepth+1`, clamp to the last section-type index,
710/// then take `s[0 .. $td]`. Perl's `0 .. $td` is an EMPTY range when `$td < 0`, so `\setcounter{tocdepth}{-1}` (parts
711/// only) yields `[part]` and `{-2}` or lower yields `select=''` (an empty ToC). Computed in SIGNED space: `tocdepth` is a
712/// signed counter, so the old `value_of() as usize` wrapped on negatives — a debug overflow panic, and in release a
713/// silently over-full ToC (`{-2}` listed everything).
714fn toc_select() -> Result<String> {
715  let s = [
716    "ltx:part",
717    "ltx:chapter",
718    "ltx:section",
719    "ltx:subsection",
720    "ltx:subsubsection",
721    "ltx:paragraph",
722    "ltx:subparagraph",
723  ];
724  let td = (CounterValue!("tocdepth").value_of() + 1).min((s.len() - 1) as i64);
725  let take = usize::try_from(td + 1).unwrap_or(0);
726  let mut s_depth: Vec<&'static str> = s.into_iter().take(take).collect();
727  if !s_depth.is_empty() {
728    s_depth.push("ltx:appendix");
729    s_depth.push("ltx:index");
730    s_depth.push("ltx:bibliography");
731  }
732  Ok(s_depth.join(" | "))
733}

745pub fn kernel_chapter_with_warning(category: &str, message: &str) -> Result<Tokens> {
746  if !lookup_bool("\\chapter:fallback@warned") {

753  }
754  Ok(Tokens!(T_CS!("\\lx@kernel@chapter")))
755}

757/// The kernel's `\tableofcontents`/`\listoffigures`/`\listoftables` element, `<ltx:TOC lists=… scope='global'
758/// [select=…]><ltx:title>name</ltx:title></ltx:TOC>` (Perl latex_constructs.pool.ltxml:725-743). Inside a frontmatter
759/// container, whose model refuses it (`ltx:abstract`, Block.model), it goes to the nearest open ancestor that admits
760/// it, after the container, and what follows stays in the container (`Document::float_out_of_frontmatter`, Perl's
761/// `floatToElement`): pdflatex prints it inside the abstract's quotation, where the schema has no place for it — a
762/// LyX abstract that ends with `\tableofcontents{}` (1011.3976, 1310.0349, 1509.04736), one with it mid-abstract
763/// (1602.08755). OXIDIZED_DESIGN_DIVERGENCES #753.
764fn insert_toc(
765  document: &mut Document,
766  lists: &str,
767  select: Option<String>,
768  title: Option<Vec<&Digested>>,
769) -> Result<()> {
770  let save = document.float_out_of_frontmatter("ltx:TOC")?;
771  let mut attrs = string_map!("lists" => lists.to_string(), "scope" => "global".to_string());
772  if let Some(select) = select.filter(|s| !s.is_empty()) {
773    attrs.insert("select".to_string(), select);
774  }
775  document.open_element("ltx:TOC", Some(attrs), None)?;
776  if let Some(title) = title {
777    document.insert_element("ltx:title", title, None)?;
778  }
779  document.close_element("ltx:TOC")?;
780  if let Some(save) = save {
781    document.set_node(&save);
782  }
783  Ok(())
