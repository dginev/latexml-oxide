8#[rustfmt::skip]
9pub(crate) fn load() -> Result<()> {
10  // latex.ltx:14326 `\normalsize` placeholder ("font size command not defined"), before any class.
11  Let!("\\lx@kernel@normalsize", "\\normalsize");
12  // The text of a float starts at the body size (latex.ltx `\@floatboxreset`, `begin_float`).
13  DefMacro!("\\lx@floatbox@reset", "\\reset@font\\normalsize");
14  // The next `{table}`/`{figure}` a binding opens is not a LaTeX float: no reset (`reset_float_box`).
15  DefPrimitive!("\\lx@float@keepsize", {
16    assign_value("lx@float@keepsize", true, Some(Scope::Global));
17  });
18  // ======================================================================
19  // C.2 The Structure of the Document
20  // ======================================================================
21
22  //**********************************************************************
23  // C.2. The Structure of the Document
24  //**********************************************************************
25  //   prepended files (using filecontents environment)
26  //   preamble (starting with \documentclass)
27  //   \begin{document}
28  //    text
29  //   \end{document}
30
31  // Perl: PushValue('@at@begin@document', $_[1]->unlist)
32  // latex.ltx:18901 `\DeclareRobustCommand\AtBeginDocument{\AddToHook
33  // {begindocument}}` (and `\AtEndDocument` = `\AddToHook{enddocument}`):
34  // the TeX-level macro IS the L3 hook under the current default label, so
35  // `\RemoveFromHook{begindocument}[l3doc]` (source2edoc.cls:12) can cancel
36  // l3doc.cls:511's `\AtBeginDocument{\MakeShortVerb\"}`. Perl
37  // (latex_constructs.pool.ltxml:296-297, 93f875a6) keeps a private
38  // `@at@begin@document` store that ignores the label and that
39  // `\RemoveFromHook` never sees, so the shortverb stayed live and
40  // `"` scanned across ltoutenc.dtx's macrocode blocks (base/source2e, 30
41  // errors; SHARED, pdflatex clean). With the hook system loaded, route
42  // through `\AddToHook` (honouring an explicit `[label]`); the private
43  // store stays the fallback for the format-less path and for the Rust
44  // bindings' `AtBeginDocument()` helper (prelude.rs:28), which fires just
45  // before the L3 `begindocument` hook (below).
46  // Guard: `perfect_kernel_batch56::atbegindocument_joins_the_l3_begindocument_hook`.
47  DefMacro!("\\AtBeginDocument[]{}", sub[(label, rules)] {
48    at_document_hook("begindocument", "@at@begin@document", label, rules)
49  });
50  DefMacro!("\\AtEndDocument[]{}", sub[(label, rules)] {
51    at_document_hook("enddocument", "@at@end@document", label, rules)
52  });
53
54  // Like  "<ltx:document xml:id='#id'>#body</ltx:document>",
55  // But more complicated due to id, at begin/end document and so forth.
56  // AND, lower-level so that we can cope with common errors at document end.
57  DefConstructor!(T_CS!("\\begin{document}"), None, sub[document, _args, props] {
58    let id = prop_str!(props,"id");
59    // Already (auto) created?
60    match document.findnode("/ltx:document", None) { Some(mut docel) => {
61      if id != pin!("") {
62        let id_s = with(id, |s| s.to_string());
63        document.set_attribute(&mut docel, "xml:id", &id_s)?;
64      }
65      // The document was opened by the PREAMBLE: whatever it holds now was built
66      // before `\begin{document}` — an undefined command's `<ERROR>` and its
67      // argument text, a mis-emulated engine primitive's parameters. LaTeX forbids
68      // typesetting there (`\@nodocument`, latex.ltx `\everypar` in the preamble:
69      // "Missing \begin{document}"), so none of it is body. Mark it for the
70      // frontmatter placement pass, which moves it below the frontmatter and never
71      // reads it as a cover (OXIDIZED_DESIGN #262).
72      // Its open paragraph stays open: `\begin{document}` issues no `\par`, so
73      // the first body words continue it, as in TeX.
74      mark_preamble_residue(&docel);
75    } _ => {
76      let props = with(id, |id_str| string_map!("xml:id" => id_str));
77      document.open_element("ltx:document", Some(props), None)?;
78    }}
79  },
80  after_digest => sub[whatsit] {
81    // Perl: beginMode('internal_vertical', 1) — noframe=1
82    // Begin internal_vertical mode WITHOUT pushing a stack frame, keeping level=0
83    begin_mode_opt("internal_vertical", true)?;
84    // we need to re-bind in order to nest calls to the binding macro machinery
85    DefMacro!("\\@currenvir", "document");
86    assign_value("current_environment", "document", None);
87    let expanded_id = Expand!(T_CS!("\\thedocument@ID"));
88    whatsit.set_property("id", expanded_id);
89    Let!("\\@nodocument", "\\relax", Scope::Global);
90    // A misplaced alignment command in the preamble starts no cascade in the body (OXIDIZED_DESIGN_DIVERGENCES #540).
91    crate::tex_tables::end_misplaced_alignment_cascade();
92    // latex.ltx `\document` (:9491-9494): the `.aux` is opened under the job's name and starts with `\relax`, so a
93    // class `\enddocument` that `\input`s `\jobname.aux` (jpsj3.cls:90-96, bio.cls, lamuphys.sty:1240-1247,
94    // caosp.sty:916) finds it. Perl's `\begin{document}` opens none; its only open, `latex_base`'s
95    // `\openout\@mainaux\jobname.aux` (latex_base.pool.ltxml:722), runs when LaTeX.pool loads: at `\documentclass`
96    // without a preload, after `\jobname` (Core.pm:152), so plain Perl finds the `.aux`; with the format preloaded
97    // (`ar5iv.sty`) at state initialization (Core.pm:143), before `\jobname`. Ours (latex_base.rs:743) ran at that early
98    // point or, under a dump, at the dump's build, whose streams the dump does not keep, and the `.aux` was missing:
99    // `Error:missing_file:<jobname>.aux` (witnesses 2605.24927, 2605.04825, 2605.28673, 2605.20125, cond-mat9607109, astro-ph9805185).
100    // Repro loader/jobname_aux_exists_after_begin_document.
101    TeX!(r"\if@filesw\immediate\openout\@mainaux\jobname.aux \immediate\write\@mainaux{\relax}\fi");
102    // Clear \everypar at document start (Perl `AssignRegister('\everypar', Tokens(), 'global')`,
103    // latex_constructs.pool L319) through its current meaning, as latex.ltx:9498 `\everypar{}`
104    // does. A package that re-pointed `\everypar` in the preamble at a register chained from the
105    // kernel's (arabicore.sty:123-128, babel's rlbabel.def:121-125) has that register emptied and
106    // its chain kept; one that redefined it as a macro (rlbicig.sty:62-64, witness: montex manual)
107    // has the macro run, so its own prefix stays, as in pdflatex — Perl's by-name assignment warned
108    // "not a register" (KNOWN_PERL_ERRORS #415). The register the paragraph hook runs is
109    // `\lx@para@everypar` (stomach.rs `pending_everypar`).
110    // A `\let` to a register of another type takes the digest path, which reports what TeX reports.
111    // No corpus witness yet for the preamble chain; the macro shape is the montex manual's.
112    let mut boxes = Vec::new();
113    if lookup_register_definition(&T_CS!("\\everypar"))
114      .is_some_and(|register| matches!(register.register_type, RegisterType::Tokens))
115    {
116      assign_register("\\everypar", RegisterValue::Tokens(Tokens!()), Some(Scope::Global), Vec::new())?;
117    } else {
118      boxes.push(digest(Tokens!(T_CS!("\\everypar"), T_BEGIN!(), T_END!()))?);
119    }
120    // latex.ltx:9497 `\normalsize` (beside :9498's `\everypar{}`): the body starts at the class's
121    // body size and its leading — a preamble `\setlength{\baselineskip}{1.5cm}` does not reach the
122    // body (witness 2605.29990). Only a class's `\normalsize`: one that defines none (amsppt) leaves
123    // latex.ltx:14326's error placeholder, kept as `\lx@kernel@normalsize`. Perl's
124    // `\begin{document}` runs no size command. Guard
125    // `perfect_kernel_batch58::size_switches_set_the_leading`.
126    if lookup_bool("inPreamble")
127      && !x_equals(&T_CS!("\\normalsize"), &T_CS!("\\lx@kernel@normalsize"))
128    {
129      digest(Tokens!(T_CS!("\\normalsize")))?;
130      // The body size is whatever `\normalsize` gives here, by LaTeX's definition: the nominal size
131      // output sizes are measured against (as `\lx@nominal@fontsize` records it for a
132      // `\@setfontsize\normalsize`), so a `\renewcommand\normalsize{\fontsize{11}{13}\selectfont}` does
133      // not wrap the body in `fontsize="110%"`.
134      if let Ok(points) = Expand!(T_CS!("\\f@size")).to_string().trim().parse::<f64>()
135        && points > 0.0
136      {
137        AssignValue!("NOMINAL_FONT_SIZE", Float(points));
138      }
139    }
140    // Perl #2798: at \begin{document}, make the fill widths consistent —
141    //   \columnwidth = \hsize = \linewidth = \textwidth
142    // (\columnwidth/\linewidth otherwise keep their 6in=433.62pt DefRegister
143    // default, which is wrong for the default 345pt article \textwidth).
144    if let Some(textwidth) = lookup_register("\\textwidth", Vec::new())? {
145      assign_register("\\columnwidth", textwidth.clone(), None, Vec::new())?;
146      assign_register("\\hsize", textwidth.clone(), None, Vec::new())?;
147      assign_register("\\linewidth", textwidth, None, Vec::new())?;
148    }
149    // Rust-only divergence (OXIDIZED_DESIGN "Frontmatter / locked-macro protection
150    // at begin-document"; reproducer docs/reproducers/frontmatter_maketitle_double.tex):
151    // digest the begin-document hook lists (\AtBeginDocument via @at@begin@document,
152    // and @document@preamble@atend) with the state RE-LOCKED.
153    //
154    // These hooks carry RAW document/package TeX. A constructor's after-digest runs
155    // state-UNLOCKED (execute_after_digest, definition.rs) so bindings can rebind/load
156    // within their own before/after methods — but that unlock leaks into this nested
157    // raw-TeX digest, letting a raw `\def`/`\let` of a binding-LOCKED macro (e.g.
158    // `\AtBeginDocument{\def\maketitle{...}}`) silently win. LaTeXML deliberately owns
159    // `\maketitle` (locked) so that `\title`/`\author` produce SEMANTIC frontmatter and
160    // the class's visual `\@maketitle` reconstruction is suppressed; when a class's raw
161    // redefinition slips the lock, the title is emitted twice (once semantic, once
162    // visual). Perl shares this bug on an inline `\AtBeginDocument` (both engines double
163    // vs pdflatex's single); it only escapes on acl.sty because its lock incidentally
164    // holds for a raw-loaded `.sty`. Re-locking here enforces the intended rule — a
165    // binding-locked macro is never redefined from the raw TeX layer — generally, for
166    // every begin-document hook, so LaTeXML's own `\maketitle` runs and the frontmatter
167    // is emitted exactly once (matching Perl's acl output, incl. `ltx_authors_1line`,
168    // and surpassing Perl on the inline case). Scope is narrow: only these two nested
169    // hook digests, not the general before/after-digest unlock.
170    // Leave the preamble (`inPreamble=0`) only AFTER the begindocument hooks below —
171    // the pre-#2846 latex.ltx placement (`\@preamblecmds` disables the \@onlypreamble
172    // commands at L9522, AFTER firing the begindocument hook at L9512). So
173    // `\RequirePackage`/`\usepackage` deferred to \AtBeginDocument runs while still
174    // `inPreamble=1` and stays legal. Paragraph-breaking inside \AtBeginDocument
175    // (upstream #2754) does NOT depend on this timing: `\par` (tex_paragraph.rs) closes
176    // a paragraph whenever one is being built in the `document` environment — the hooks
177    // run with `current_environment=document` — so a blank line after some
178    // \AtBeginDocument text (horizontal mode) breaks a paragraph even while
179    // `inPreamble=1`. That makes upstream #2846's early clear — and the
180    // `inBeginDocumentHook` guard-decouple it forced (#2848) — unnecessary; one flag,
181    // one transition. Ground truth: pdflatex
182    // AND same-host Perl accept both a paragraph-splitting and a `\RequirePackage`-
183    // loading \AtBeginDocument (reproducers atbegindocument_requirepackage.tex + the
184    // #2754 book example; corpus witnesses arXiv:2605.00022 / 2605.00119).
185    // KNOWN_PERL_ERRORS #43.
186    // `\document` is `\@onlypreamble` and its hooks are `\UseOneTimeHook`s
187    // (latex.ltx:9512/9537): a SECOND `\begin{document}` — ltnews.tex:236/296
188    // and l3news.tex:109/177 `\renewenvironment{document}` then `\input` each
189    // issue file with its own `\begin{document}` — fires nothing. Re-firing
190    // ran csquotes' end-preamble block twice, whose hooks it `\undef`s after
191    // use (csquotes.sty:2434-2446 `\csq@hook@nomultilang`/`@hyperref`; Perl
192    // pool:304-335 re-fires too). Guard:
193    // `perfect_kernel_batch54::second_begin_document_fires_no_hooks`.
194    let first_begin = lookup_bool("inPreamble");
195    if first_begin
196      && let Some(ops) = lookup_tokens("@document@preamble@atend") {
197      boxes.push(digest_relocked(ops)?);
198    }
199    // Fire the L3 hook system for begindocument/before, then begindocument.
200    // Modern LaTeX (with expl3) fires these in order at \begin{document}:
201    //   1. \hook_use_once:n {begindocument/before}  — pre-init hook, STILL preamble
202    //   2. @at@begin@document + \hook_use_once:n {begindocument}  — \AtBeginDocument,
203    //      STILL preamble (latex.ltx disables \@onlypreamble only afterwards)
204    //   3. leave the preamble  (inPreamble=0, AFTER the hooks — see below)
205    // begindocument/before therefore fires while inPreamble=1: it carries
206    // last-minute preamble setup (deferred `\RequirePackage`, translations.sty's
207    // language initialiser) that must precede leaving the preamble — firing it
208    // after inPreamble=0 wrongly rejects a deferred `\RequirePackage`.
209    // Driver for begindocument/before: translations.sty L73-85 wraps its
210    //   `\def\@trnslt@current@language{\languagename}` initialiser in
211    //   `\AddToHook{begindocument/before}{…}`. Without this dispatch the
212    //   .trsl dictionary loads (queued via `\AtBeginDocument`) inside the
213    //   subsequent begindocument firing reference an undefined CS.
214    // Witnesses: stage-2/3 of 100k warning corpus (2603.25051, 2604.07448).
215    //
216    // NOTE: this is a Rust-only deviation from Perl (Perl does not fire a
217    // begindocument hook dispatch), but it's load-bearing because our raw
218    // expl3-code.tex load path *does* define `\hook_use_once:n` and enqueues
219    // real hook code against it. Keep until the kernel-parity direction
220    // either (a) stops loading raw expl3-code.tex, or (b) ports l3hooks
221    // natively with storage. See SYNC_STATUS.md "l3hooks parity".
222    // Hook code runs RE-LOCKED like the sibling lists above: Perl runs every
223    // begin-document hook under `$UNLOCKED=0` (State.pm:502-514 ignores a
224    // redefinition of a `:locked` cs) and never fires the L3 hook at all, so a
225    // binding-locked kernel macro must not be replaceable from here —
226    // polyglossia.sty:1442-1456 `\cs_set:Npn \@caption #1 [#2] #3` in the
227    // `begindocument` hook overrode our locked `\@caption` and its `[`-scan
228    // overshot every figure (beamerdarkthemes guide, 101 caption errors).
229    if first_begin && lookup_definition(&T_CS!("\\hook_use_once:n"))?.is_some() {
230      boxes.push(digest_relocked(hook_use_once_tokens("begindocument/before"))?);
231    }
232    // latex.ltx `\document` L9471: `\@kernel@after@begindocument@before`, whose
233    // `\reinstall@nfss@defs` (:12513-12514) makes the shape switches
234    // `\protected` for the body (latex_constructs_rust_only.rs).
235    if first_begin && lookup_definition(&T_CS!("\\reinstall@nfss@defs"))?.is_some() {
236      boxes.push(digest_relocked(Tokens!(T_CS!("\\reinstall@nfss@defs")))?);
237    }
238    // The hook's other half (latex.ltx:12513-12514), `\init@series@setup` (:13902-13920): `\reset@font`, then
239    // `\mdseries` and `\let\seriesdefault\f@series` while `\seriesdefault` is the kernel's. A font switch of the
240    // preamble or the class body (`\itshape\bfseries`, a class's `\usefont`) does not reach the body: pdflatex
241    // sets it upright medium in the document's `\familydefault` (`\reset@font` is `\normalfont`, sect13.rs; a
242    // `\renewcommand\familydefault{\sfdefault}` body is sans). Perl runs neither half (its body kept the preamble's
243    // font, `<text font="bold italic">`). OXIDIZED_DESIGN_DIVERGENCES #315 addendum, #779; guards
244    // `perfect_kernel_batch77u::{preamble_font_switch_does_not_reach_the_body,
245    // class_body_font_does_not_reach_the_body, family_default_survives_a_preamble_switch}`.
246    if first_begin && lookup_definition(&T_CS!("\\init@series@setup"))?.is_some() {
247      boxes.push(digest_relocked(Tokens!(T_CS!("\\init@series@setup")))?);
248    }
249    // latex.ltx `\document` L9472: right after begindocument/before, the
250    // kernel loads the expl3 BACKEND — `\@expl@sys@load@backend@@` →
251    // `\sys_load_backend:n {}` unless one was chosen — raw-inputting
252    // `l3backend-<driver>.def`, which defines the `\__color_backend_*` /
253    // `\__pdf_backend_*` function families that l3color / l3pdf / raw
254    // package code call at runtime. The embedded dump cannot carry these:
255    // backend selection is a job-start decision (exactly why the kernel
256    // defers it to `\document`), so without this firing every dump-mode
257    // conversion left them undefined (witnesses: prettytok / spath3 manuals,
258    // `Error:undefined:\__color_backend_reset:`; TL doc corpus 2026-08-31).
259    //
260    // The kernel's `\@expl@sys@load@backend@@` (expl3.ltx:130-134) is
261    // `\str_if_exist:NF \c_sys_backend_str { \sys_load_backend:n {} }`;
262    // `\lx@sys@load@backend` (latex_constructs_rust_only.rs) is that guard
263    // with the DVI case named: in our default `\pdfoutput=0` state a
264    // `\documentclass[pdftex]` option maps to `pdfmode` (expl3-code.tex:8352)
265    // and `\__sys_load_backend_check:N` (:7992) rejects it with a counted
266    // "inconsistent … using 'dvips'" (elpres, scidoc), so DVI names `dvips`
267    // outright; a document that set `\pdfoutput=1` (commath, isorot, thinsp,
268    // webguide, quantum-bibliographystyle-demo) gets the blank auto-select
269    // (pdftex), and one whose preamble already loaded a backend
270    // (quantum-template) is left alone. All RUST-ONLY: Perl never fires the
271    // loader. Guards: `perfect_kernel_batch56::{backend_load_names_the_dvi_backend,
272    // backend_load_follows_pdfoutput_and_prior_choice}`.
273    if first_begin && lookup_definition(&T_CS!("\\lx@sys@load@backend"))?.is_some() {
274      boxes.push(digest(Tokens!(T_CS!("\\lx@sys@load@backend")))?);
275    } else if first_begin && lookup_definition(&T_CS!("\\@expl@sys@load@backend@@"))?.is_some() {
276      boxes.push(digest(Tokens!(T_CS!("\\@expl@sys@load@backend@@")))?);
277    }
278    // latex.ltx:9511 `\@kernel@before@begindocument` (since 2020-10, :1076-1081), before the `begindocument` hook; its
279    // sibling after it, below.
280    if first_begin && lookup_definition(&T_CS!("\\@kernel@before@begindocument"))?.is_some() {
281      boxes.push(digest_relocked(Tokens!(T_CS!("\\@kernel@before@begindocument")))?);
282    }
283    // @at@begin@document (\AtBeginDocument) + the begindocument hook. `inPreamble`
284    // is STILL 1 here (we leave the preamble only after this block), so a deferred
285    // `\RequirePackage`/`\usepackage` remains legal (corpus witnesses
286    // arXiv:2605.00022 / 2605.00119: inconsolata.sty →
287    // `\AtBeginDocument{...\usepackage{upquote}}` → upquote.sty's `\RequirePackage
288    // {textcomp}`). A blank line here still splits paragraphs (fixes #2754): the hooks
289    // run inside the `document` environment, so `\par` is active (only the RAW preamble,
290    // where `document` is not yet on the env stack, no-ops it) regardless of
291    // `inPreamble` (tex_paragraph.rs).
292    // Raw `#`-bearing `\AtBeginDocument` chunks (see `at_document_hook`) run
293    // first — before the L3 hook that holds the later, `#`-free raw chunks.
294    // (what the hooks below define is a class default for the unclaimed-text net, frontmatter_stores.rs
295    // `note_document_hooks`)
296    if first_begin {
297      crate::frontmatter_stores::note_document_hooks(true);
298    }
299    if first_begin && let Some(ops) = lookup_tokens("@at@begin@document@rawparam") {
300      boxes.push(digest_relocked(ops)?);
301    }
302    // Order: the L3 `begindocument` hook (raw packages' `\AtBeginDocument`,
303    // routed there since batch 56i) fires FIRST, then the bindings' private
304    // `@at@begin@document` store — bindings outrank raw: cleveref_sty.rs
305    // defers `\let\label\lx@cleverref@label` here, and the raw cleveref.sty:66
306    // hook's `\def\label{\@ifnextchar[\label@optarg\label@noarg}` must not
307    // shadow it (its `\cref@override@label@type` `[#1][#2]` scan ran to EOF:
308    // crossreftools_driver, test-autonum fatal; RUST-ONLY, Perl has no raw
309    // cleveref). Guard: `perfect_kernel_batch56::binding_begin_document_code_outranks_raw_hook`.
310    if first_begin && lookup_definition(&T_CS!("\\hook_use_once:n"))?.is_some() {
311      boxes.push(digest_relocked(hook_use_once_tokens("begindocument"))?);
312    }
313    if first_begin && let Some(ops) = lookup_tokens("@at@begin@document") {
314      boxes.push(digest_relocked(ops)?);
315    }
316    // latex.ltx:9513 `\@kernel@after@begindocument`: its `\__hook_cmd_begindocument_code:` (:6865-6873) applies the
317    // preamble's `cmd/<name>/<hook>` hooks (a binding's are counted, `\__hook_patch_command:Nnn`) and patches later
318    // ones at once. Witnesses 2602.09577, 2510.06714, 2411.13565, 2511.21986, 2603.08400.
319    if first_begin && lookup_definition(&T_CS!("\\@kernel@after@begindocument"))?.is_some() {
320      boxes.push(digest_relocked(Tokens!(T_CS!("\\@kernel@after@begindocument")))?);
321    }
322    // The begin-document hooks have run (their tests of `\chapter` got TeX's answer): a `\chapter` the class left
323    // undefined reads LaTeXML's chapter from here, with one Warning (OXIDIZED_DESIGN #734).
324    if first_begin {
325      sect01::supply_chapter_fallback();
326    }
327    // Leave the preamble now — AFTER the begindocument hooks (the `\@preamblecmds`
328    // point, latex.ltx L9522). This single transition both (a) disables the
329    // \@onlypreamble commands, so from here the onlyPreamble guard rejects body-level
330    // `\RequirePackage`/`\usepackage` (matching pdflatex + Perl), and (b) restores the
331    // pre-#2846 placement — `\AtBeginDocument` above ran while still `inPreamble=1`.
332    // `\par` paragraph-breaking is governed by mode + the `document` env, not this flag.
333    assign_value("inPreamble", false, None);
334    // latex.ltx:9521-9522: the `\@preamblecmds` walk leaves `\@preamblecmds`
335    // itself `\let` to `\@notprerr` (it is `\@onlypreamble`, latex.ltx:1228).
336    // Raw code probes exactly that (lppl.tex:44) to detect a running document;
337    // see `\@notprerr` in sect05.rs.
338    assign_meaning(
339      &T_CS!("\\@preamblecmds"),
340      lookup_meaning(&T_CS!("\\@notprerr")).unwrap_or(Stored::None),
341      Some(Scope::Global),
342    );
343    // latex.ltx:9525 `\UseOneTimeHook{begindocument/end}` — AFTER the
344    // preamble is left. jwjournal.cls:643-650 wraps the whole body in a `+b`
345    // environment from this hook (and closes it from `enddocument`); without
346    // it the markdown body typeset raw (`##` reaching the stomach, 4 docs).
347    // Guard: `perfect_kernel_batch54::begindocument_end_and_enddocument_hooks_fire`.
348    // UNREAD onto the main stream rather than digest in a string mouth: a
349    // `+b` environment opened from this hook must read its body from the
350    // document file, not from the hook's own (empty) mouth.
351    // Collected with the other inline hook below and unread once, in order.
352    let mut inline_hooks: Vec<Token> = Vec::new();
353    if first_begin && lookup_definition(&T_CS!("\\hook_use_once:n"))?.is_some() {
354      inline_hooks.extend(hook_use_once_tokens("begindocument/end").unlist());
355    }
356    // Preamble cleanup: force `\ExplSyntaxOff` if `_` is still LETTER at
357    // document start. LaTeX restores expl3 syntax per file (the kernel's
358    // `\@expl@push@filename@@`/`\@expl@pop@filename@@`), so a package's
359    // unmatched `\ExplSyntaxOn` never reaches the document; packages like
360    // mhchem.sty end with one (its tail, "legacy" block), and this cleanup
361    // restores the catcodes before the document body where ours did not.
362    // Without this, `\sum_{...}` tokenizes as the CS `\sum_` (letter `_`)
363    // rather than `\sum` + `_` + `{...}`.
364    if lookup_catcode('_') == Some(Catcode::LETTER)
365      && lookup_definition(&T_CS!("\\ExplSyntaxOff"))?.is_some()
366    {
367      boxes.push(digest(Tokens!(T_CS!("\\ExplSyntaxOff")))?);
368    }
369    // Fire babel language activation AFTER all hooks (including babel's own
370    // \selectlanguage call). This runs even if babel's hook code has errors.
371    // Use T_CS! directly since @ is OTHER catcode at \begin{document} time.
372    if lookup_definition(&T_CS!("\\lx@babel@activate@mainlang"))?.is_some() {
373      boxes.push(digest(Tokens!(T_CS!("\\lx@babel@activate@mainlang")))?);
374    }
375    // @document@preamble@afterend (etoolbox `\AfterEndPreamble`) runs after
376    // the \@preamblecmds point (both inPreamble=0 and the hook window closed
377    // above), so onlyPreamble commands here are already disabled — matching
378    // latex.ltx's begindocument/end, which it IS under a 2020+ format
379    // (etoolbox.sty:1745) and which the legacy `\@afterendpreamblehook` runs
380    // INLINE from `\document` (:1774-1776). Unread with `begindocument/end`,
381    // not digested in a string mouth: modernposter.cls opens its
382    // document-spanning `tikzpicture[overlay]` here, and a box reader opened
383    // in an isolated mouth runs dry before the body ("`\hbox` Attempt to end
384    // mode restricted_horizontal", modernposter/demo). Guard:
385    // `perfect_kernel_batch56::atenddocument_closer_reaches_the_galley_box`.
386    if first_begin
387      && let Some(ops) = lookup_tokens("@document@preamble@afterend") {
388      inline_hooks.extend(ops.unlist());
389    }
390    // `begindocument/end` has run: a later `\AfterEndPreamble` runs its code at once (etoolbox_sty.rs).
391    if first_begin {
392      assign_value("@document@preamble@afterend@done", true, Some(Scope::Global));
393    }
394    // (what the hooks made, `begindocument/end`'s too, is a class default for the unclaimed-text net: read after them,
395    // frontmatter_stores.rs `note_document_hooks`)
396    // (and a `\maketitle` a raw class appended to `\document`, which latex.ltx runs at its very end, after
397    // `begindocument/end`: its print noted for the title-page check, `\lx@note@title@page`; cup-journal 2207.13252)
398    let title_page = first_begin && lookup_bool("lx_title_page_at_begin_document");
399    if !inline_hooks.is_empty() {
400      if first_begin {
401        inline_hooks.push(T_CS!("\\lx@net@document@hooks@done"));
402      }
403      if title_page {
404        inline_hooks.push(T_CS!("\\lx@note@title@page"));
405      }
406      unread(Tokens::new(inline_hooks));
407    } else {
408      if first_begin {
409        crate::frontmatter_stores::note_document_hooks(false);
410      }
411      if title_page {
412        unread(Tokens!(T_CS!("\\lx@note@title@page")));
413      }
414    }
415    whatsit.set_font(lookup_font().unwrap()); // Start w/ whatever font was last selected.
416    leave_horizontal_internal();
417    boxes
418  });
419
420  // \document is used directly in e.g. expl3.sty
421  Let!("\\document", "\\begin{document}", Scope::Global);
422  // The ORIGINAL, for `\begin{document}`'s user-redefinition test
423  // (sect01.rs `is_user_document_macro`): a binding may later redefine the
424  // `\begin{document}` CS itself (subfiles), so "still the original" cannot
425  // be read off that CS.
426  Let!("\\lx@orig@document", "\\begin{document}", Scope::Global);
427
428  /// `\hook_use_once:n {name}` as explicit tokens: `Tokenize!` runs under the
429  /// runtime catcodes, where `:` is OTHER and `/` must stay OTHER inside the
430  /// name. latex.ltx runs each document hook with `\UseOneTimeHook` (=
431  /// `\hook_use_once:n`, :6729): `begindocument/before` (:9470), `begindocument`
432  /// (:9512), `begindocument/end` (:9525), `enddocument` (:15257). Code added to
433  /// such a hook once it has run executes at once (lthooks), so an
434  /// `\AtBeginDocument` issued from the begin-document hook itself — a package
435  /// loaded there, `\AtBeginDocument{\usepackage{underscore}}` — still runs; under
436  /// `\hook_use:n` it was appended to a hook never used again, and dropped. Guard:
437  /// `perfect_kernel_batch77j::begin_document_code_added_by_the_hook_runs`.
438  fn hook_use_once_tokens(name: &str) -> Tokens {
439    let mut toks = vec![T_CS!("\\hook_use_once:n"), T_BEGIN!()];
440    for c in name.chars() {
441      toks.push(if c.is_ascii_alphabetic() {
442        T_LETTER!(c.to_string())
443      } else {
444        T_OTHER!(c.to_string())
445      });
446    }
447    toks.push(T_END!());
448    Tokens::new(toks)
449  }
450
451  // latex.ltx:15255-15259 `\enddocument`: the end-document hooks run INLINE
452  // in the main token stream (`\UseOneTimeHook{enddocument}`, preceded by the
453  // legacy `\AtEndDocument` list) and only then `\@checkend{document}` and
454  // the final cleanup. They were digested here in an isolated mouth, so a
455  // box reader suspended in the galley — modernposter.cls's document-spanning
456  // `tikzpicture[overlay]` opened from `\AfterEndPreamble` and closed by
457  // `\AtEndDocument{\end{tikzpicture}}`: the pgfpicture `\setbox\hbox\bgroup`
458  // and every node box — never received its `\egroup`; its loop ran dry and
459  // `end_mode` met pgf's `\begingroup` on top ("`\hbox` Attempt to end mode
460  // restricted_horizontal", modernposter/demo 14; Perl 15 — SHARED; pdflatex
461  // clean). The opener side already unreads `begindocument/end` (above, the
462  // jwjournal `+b` environment); this is the closer side. The hooks are
463  // unread once (`lx@enddocument@hooks@fired`), followed by the finalizer.
464  // Guard: `perfect_kernel_batch56::atenddocument_closer_reaches_the_galley_box`.
465  // …and the expansion ENDS with `\@@end` as latex.ltx:15278's does
466  // (`\deadcycles\z@\@@end`): morewrites.sty:550-557's `\AtEndDocument` hook
467  // `\__morewrites_close_all_at_end:nw #1#2 \@@end` grabs the rest of
468  // `\enddocument` up to that token and re-emits it after closing its
469  // streams — without the token the scan ran to the end of the file
470  // (cntperchap_example, runcode_troubleshoot: "Missing argument
471  // Until:\@@end", sweep 54). Our `\@@end` is the job-end primitive
472  // (tex_job.rs `\lx@end@document`: leave horizontal mode, flush).
473  // Never reaches the kernel's `enddocument/afterlastpage` hook or
474  // `\@kernel@after@enddocument@afterlastpage` (latex.ltx:20226-20260, in the
475  // dump): that one does `\tex_shipout:D\vbox to\textheight{…}` with the
476  // "Temporary page!" text (latex.ltx:20261-20269), which the `\shipout`
477  // primitive would now emit in place (DIVERGENCES #314).
478  DefMacro!(T_CS!("\\end{document}"), None, "\\lx@end@open@frontmatter\\lx@enddocument@hooks\\lx@finalize@document\\@@end");
479  // A frontmatter body still open at `\end{document}` — an unclosed `\begin{abstract}`, `\abstract{`, or a brace-less
480  // `\abstract` no sectioning command ended (`\lx@add@frontmatter@until`, base_utilities.rs) — ends first: its terminal
481  // is read before the end-document code, which then runs at the document's level, so the abstract keeps its text as an
482  // unclosed `{itemize}` keeps its list (65c; the open environment's Warning stays). Read inside the body, the finalizer
483  // closed the document within the abstract's box, which nothing placed: an empty `<document/>` with 0 errors.
484  // (none of the document's end runs in a speculative digestion: `gullet::speculative_digestion`)
485  DefPrimitive!("\\lx@end@open@frontmatter", {
486    if in_speculative_digestion() {
487      return Ok(Vec::new());
488    }
489    if let Some(terminal) = lookup_token("lx@until@terminal")
490      && (terminal == T_CS!("\\lx@end@abstract") || terminal == T_CS!("\\lx@end@keywords"))
491    {
492      // One body per pass, re-read after its terminal, so keywords left open inside an abstract end both (int26
493      // review p8); a body its terminal did not end (the same depth again) stops the passes.
494      let depth = match lookup_value("lx@until@depth") {
495        Some(Stored::Number(n)) => n.0 + 1,
496        _ => 0,
497      };
498      let last = lookup_int("lx_end_open_frontmatter_depth");
499      if last != 0 && depth >= last {
500        return Ok(Vec::new());
501      }
502      assign_value("lx_end_open_frontmatter_depth", Stored::Int(depth), Some(Scope::Global));
503      // An environment's body left open is LaTeX's error (latex.ltx `\@checkend`: "\begin{abstract} on input line …
504      // ended by \end{document}"; Perl errs too), raised once, for the innermost open environment (`\@currenvir`), on
505      // the first pass. A begin code called as a command opens no environment LaTeX checks (`begun_environment`): the
506      // brace-less `\abstract`, mn2e's `\keywords{…}` (1004.2791), spconf's `\keywords …` (2309.07195) — no
507      // diagnostic in pdflatex; an unbalanced `\abstract{` reports its open group (`until_terminal_inside_group`).
508      if last == 0
509        && let Some(name) = begun_environment()
510      {
511        Error!(
512          "unexpected",
513          "\\end{document}",
514          s!("\\begin{{{name}}} ended by \\end{{document}}: the {name} ends here")
515        );
516      }
517      unread(Tokens!(terminal, T_CS!("\\lx@end@open@frontmatter")));
518    } else if lookup_bool("lx_abstract_owes_end") {
519      // a `\begin{abstract}` a heading ended (`\lx@heading@ends@frontmatter`, sect05.rs) whose `\end{abstract}` never came
520      assign_value("lx_abstract_owes_end", false, Some(Scope::Global));
521      Error!("unexpected", "\\end{document}", "\\begin{abstract} ended by \\end{document}");
522    }
523  });
524  DefPrimitive!("\\lx@enddocument@hooks", sub[_args] {
525    if in_speculative_digestion() || lookup_bool("lx@enddocument@hooks@fired") {
526      return Ok(Vec::new());
527    }
528    AssignValue!("lx@enddocument@hooks@fired" => true, Some(Scope::Global));
529    let mut toks: Vec<Token> = Vec::new();
530    if let Some(ops) = lookup_tokens("@at@end@document") {
531      toks.extend(ops.unlist());
532    }
533    // latex.ltx:15257 `\UseOneTimeHook{enddocument}` (the lthooks slot; the
534    // legacy `\AtEndDocument` list above is `@at@end@document`).
535    if lookup_definition(&T_CS!("\\hook_use_once:n"))?.is_some() {
536      toks.extend(hook_use_once_tokens("enddocument").unlist());
537    }
538    // Rust-only: a binding's code that must see what every end-document hook did (prelude.rs
539    // `after_end_document_hooks`; biblatex's hidden bibliography, built only when no hook printed one).
540    if let Some(ops) = lookup_tokens("@after@enddocument@hooks") {
541      toks.extend(ops.unlist());
542    }
543    unread(Tokens::new(toks));
544    Ok(Vec::new())
545  });
546
547  // The `\@checkend{document}` + `final_cleanup` half of `\enddocument`:
548  // abandon open groups (tex.web §1335), end the document's mode, close
549  // `ltx:document`.
550  DefConstructor!(T_CS!("\\lx@finalize@document"), None, sub[document,_args,_props] {
551      // Idempotent: an `\end{document}` digested inside a deferred body (the
552      // unbalanced `\abstract{…` of OXIDIZED_DESIGN #207) is replayed when
553      // that body is inserted; the document is closed once.
554      if lookup_bool("lx@document@closed") {
555        return Ok(());
556      }
557      AssignValue!("lx@document@closed" => true, Some(Scope::Global));
558      // #246/#247 position markers never reach the output (the core finalize
559      // sweeps stragglers too).
560      remove_frontmatter_marks(document);
561      // A class's `\title` print on a cover stays only beside a top-level `ltx:titlepage` (OXIDIZED_DESIGN #629).
562      settle_cover_title_ink(document);
563      // A field print no placement settled keeps its ink (`\lx@field@mark`, OXIDIZED_DESIGN #715).
564      settle_field_marks(document);
565      // When groups or environments were abandoned (before_digest below),
566      // this whatsit may sit inside the innermost one's body, built before the
567      // constructors around it close their elements: the `ltx:document` close
568      // then waits until the whole document is absorbed
569      // (`close_abandoned_document`, core_interface.rs), so every element
570      // still open closes with it, close-time hooks included (repros
571      // boxes-groups/abandoned_document_close_*; tikz-optics,
572      // jpnedumathsymbols-doc).
573      if !lookup_bool("lx@end@document@abandoned") {
574        document.close_element("ltx:document")?;
575      }
576    },
577    before_digest => {
578      let mut boxes : Vec<Digested> = Vec::new();
579      if in_speculative_digestion() {
580        return Ok(boxes);
581      }
582      // Should we try to indent the last paragraph? If so, it goes like this:
583      boxes.push(digest(T_CS!("\\lx@normal@par"))?);
584      // Pop unclosed groups and environments back to the document frame
585      // so endMode's strict BOUND_MODE check sees the right frame at the
586      // top. Mirrors Perl latex_constructs.pool.ltxml L350-374. Without
587      // this loop, papers with a dangling `\begingroup` inside the body
588      // (e.g. `\providecommand{\href}[2]{#2}\begingroup\raggedright
589      // \begin{thebibliography}{99}`) trigger
590      // "Attempt to end mode `internal_vertical` in `internal_vertical`"
591      // because the top frame is the dangling group, not the document.
592      let top_is_document = is_value_bound("current_environment", Some(0))
593        && lookup_string("current_environment") == "document";
594      // The groups abandoned here end with no closer of their own (§1335): what a
595      // group's end checks (setspace_sty.rs `\lx@setspace@check`, run by
596      // `abandon_stack_frame`'s beforeAfterGroup hooks) does not apply. Set even when
597      // nothing is unwound, so it also covers the document frame's own hooks at its
598      // close (`end_mode_opt`).
599      AssignValue!("lx@ending@document" => true, Some(Scope::Global));
600      if !top_is_document || if_stack_depth() > 0 {
601        let mut popped_lines: Vec<String> = Vec::new();
602        let mut plain_groups = 0usize;
603        while !(is_value_bound("current_environment", Some(0))
604          && lookup_string("current_environment") == "document")
605          && get_frame_depth() > 0
606        {
607          // Perl: `ToString($tok) . ' ' . $loc` (latex_constructs.pool.ltxml:364-372).
608          let initiator = lookup_token("groupInitiator")
609            .map_or_else(|| "<unknown>".to_string(), |token| token.to_string());
610          let locator = match lookup_value("groupInitiatorLocator") {
611            Some(Stored::Locator(locator)) => locator.to_string(),
612            _ => "<unknown>".to_string(),
613          };
614          let env_bound = is_value_bound("current_environment", Some(0));
615          let env_name = if env_bound {
616            lookup_string("current_environment")
617          } else {
618            String::new()
619          };
620          if !env_name.is_empty() {
621            popped_lines.push(s!("Environment {env_name} opened by {initiator} {locator}"));
622          } else {
623            // tex.web §1335 `final_cleanup`: a plain group still open at `\end`
624            // is reported ("(\end occurred inside a group at level N)") and
625            // simply abandoned — its `\aftergroup` list never runs.
626            plain_groups += 1;
627            popped_lines.push(s!("Group opened by {initiator} {locator}"));
628          }
629          // Abandon the STACK frame (frame + its `boxing` entry), not only the
630          // undo frame: a `{` left open by an unbalanced `\abstract{…`
631          // (screenplay-pkg.tex:67) otherwise kept `boxing` one deeper than
632          // the frames, and the bounded primitive reading that body closed
633          // the document frame instead of its own. The abandoned group's
634          // `\aftergroup` tokens are discarded (§1335), and every body still
635          // digesting inside it ends as at end of input, keeping its boxes
636          // (`abandon_stack_frame`).
637          abandon_stack_frame()?;
638        }
639        let abandoned_frames = popped_lines.len();
640        // Perl latex_constructs.pool.ltxml:374-377: conditionals still open
641        // are closed and reported ("(\end occurred when \iftrue on line N was
642        // incomplete)", tex.web §1335).
643        while let Some(frame) = pop_if_frame() {
644          let frame = frame.borrow();
645          popped_lines.push(s!("Conditional {} started {}", frame.token, frame.start));
646        }
647        if abandoned_frames > 0 {
648          // Read by the constructor above: the `ltx:document` close is
649          // deferred to the end of the build.
650          AssignValue!("lx@end@document@abandoned" => true, Some(Scope::Global));
651          // Every construct still digesting never ended (OXIDIZED_DESIGN #542; 1611.05714).
652          end_inside_groups();
653        }
654        // tex.web §1335 reports ONE line with the total level count.
655        let level_note = if plain_groups > 0 {
656          s!(" (\\end occurred inside a group at level {plain_groups})")
657        } else {
658          String::new()
659        };
660        let message = s!(
661          "Attempt to end document with open groups, environments or conditionals{level_note}"
662        );
663        // Perl passes the lines as the warning's details, one per line after
664        // the locator.
665        if popped_lines.is_empty() {
666          Warn!("unexpected", "\\end{document}", message);
667        } else {
668          Warn!("unexpected", "\\end{document}", message, popped_lines.join("\n\t"));
669        }
670      }
671      // Perl: endMode('internal_vertical', 1) — noframe=1
672      // End mode without popping stack frame (executes beforeAfterGroup).
673      //
674      // Skip the end-mode call if the document frame was never opened (e.g.
675      // AmsTeX papers using `\input amstex` + `\documentstyle{amsppt}` +
676      // `\enddocument` — they never call `\begin{document}`, so no
677      // `internal_vertical` mode-frame exists to close). Recognized by:
678      // current_environment is bound-on-top with value "document". Without
679      // this guard, AmsTeX papers fail with `Attempt to end mode
680      // 'internal_vertical' in 'vertical'` at the very last token.
681      let in_document_env = is_value_bound("current_environment", Some(0))
682        && lookup_string("current_environment") == "document";
683      if in_document_env {
684        end_mode_opt("internal_vertical", true)?;
685      }
686      flush();
687      boxes
688    },
689    // Perl's constructor is `\end{document}` itself; this internal half reverts to nothing
690    // (a math left open to `\end{document}` reverts through it into its `tex=`).
691    reversion => "");
692
693  // \enddocument is used directly in e.g. standalone.cls
694  Let!("\\enddocument", "\\end{document}", Scope::Global);
695  // The ORIGINAL (see `\lx@orig@document` above): docmute and subfiles
696  // redefine the `\end{document}` CS to end the FILE while `\enddocument`
697  // keeps this definition.
698  Let!("\\lx@orig@enddocument", "\\end{document}", Scope::Global);
699
700  Ok(())
701}

703/// Digest begin-document code with the state RE-LOCKED (see `\begin{document}`): a binding-locked macro is never
704/// redefined from it.
705fn digest_relocked(tokens: Tokens) -> Result<Digested> {
706  local_state_unlocked(false);
707  let r = digest(tokens);
708  expire_state_unlocked();
709  r
710}

712/// `\AtBeginDocument[label]{code}` / `\AtEndDocument`: `\AddToHook{hook}
713/// [label]{code}` when the L3 hook system is loaded, else the private store.
714fn at_document_hook(
715  hook: &str,
716  store: &str,
717  label: Option<Tokens>,
718  rules: Tokens,
719) -> Result<Tokens> {
720  // (a stand-in's copy registers no document-time code: `state::copy_provides_defers`)
721  if copy_provides_defers(if hook == "begindocument" {
722    "\\AtBeginDocument"
723  } else {
724    "\\AtEndDocument"
725  }) {
726    return Ok(Tokens!());
727  }
728  // A standalone sub-file's preamble runs inside the main document only for its definitions (standalone_sty.rs
729  // `\@standalone@documentclass`; real standalone.sty:602-646 gobbles it, OXIDIZED_DESIGN_DIVERGENCES #63, #65): its
730  // begin-document code is not run, as before the document hooks were one-time hooks and as in Perl (its
731  // `\AtBeginDocument` list has already run). The main one-time `begindocument` hook would execute it at once, in the
732  // middle of the package that adds it — babel.sty:1140-1162 before babel's own internals are defined (2405.01743,
733  // six undefined `\bbl@…`); run later, at the sub-file's `\begin{document}`, its `\selectlanguage` leaves an
734  // `\aftergroup\bbl@pop@language` the binding does not define.
735  if hook == "begindocument"
736    && matches!(
737      lookup_value("lx@subpreamble@begindocument"),
738      Some(Stored::Bool(true))
739    )
740  {
741    return Ok(Tokens!());
742  }
743  // INTERIM (K3 correctness item, KERNEL_CAPABILITIES.md): a parameter-
744  // bearing chunk under a package label (pm-isomath.sty's
745  // `\AtBeginDocument{…\NewDocumentCommand\vectorsymbol{s m}{…#2…}}`)
746  // takes lthooks' labeled `\exp_args:Nx` cleanup path (latex.ltx:5375,
747  // 5401-5416), whose x-expansion our gullet does not yet reproduce: a
748  // `\noexpand`-family `\tl_if_empty:nTF` surfaced inside the
749  // `\csname g__hook_…` (euclideangeometry-man 100× + Fatal, sweep #41).
750  // Until the gullet is faithful there, such chunks stay in the private
751  // store (pre-56i behaviour); `#`-free chunks and explicitly labeled ones
752  // go through lthooks, which is what `\RemoveFromHook` and the top-level-
753  // last order need. Guard: `perfect_kernel_batch56::hashful_begin_document_chunk_under_a_package_label`.
754  let has_param = rules
755    .unlist_ref()
756    .iter()
757    .any(|t| t.get_catcode() == Catcode::PARAM);
758  if has_param && label.is_none() && hook == "begindocument" {
759    // K7 (KERNEL_CAPABILITIES.md): latex.ltx keeps ONE FIFO `\@begindocumenthook`
760    // (`\AtBeginDocument` = `\AddToHook{begindocument}`). A `#`-bearing chunk
761    // cannot go through lthooks' cleanup here (K3), so it is stored under a
762    // fresh `\lx@bdhook@N` macro — a no-parameter body keeps its `#` tokens
763    // verbatim, exactly what the private store replayed — and THAT `#`-free
764    // name is what lthooks receives, at the chunk's registration position:
765    // alphabeta.sty:103-699's `#`-chunk before hep-math-font.sty:150-202's
766    // `#`-free one (hep-paper-documentation; a separate first-fired store
767    // inverted the opposite order, italian.ldf:156's nested `##1` hook running
768    // before verifica.cls:65's earlier class hook — Gemini T5). Without
769    // lthooks (plain-derived formats) the private store keeps the old order.
770    // Guards: `perfect_kernel_batch56::{raw_hashful_begin_document_chunk_keeps_fifo_order,
771    // begin_document_hooks_run_in_registration_order}`.
772    if lookup_meaning(&T_CS!("\\hook_gput_code:nnn")).is_some() {
773      use std::sync::atomic::{AtomicUsize, Ordering};
774      static BDHOOK_SEQ: AtomicUsize = AtomicUsize::new(0);
775      let n = BDHOOK_SEQ.fetch_add(1, Ordering::Relaxed);
776      let cs = T_CS!(s!("\\lx@bdhook@{n}"));
777      DefMacro!(cs, None, rules, scope => Some(Scope::Global));
778      let mut out = vec![T_CS!("\\AddToHook"), T_BEGIN!()];
779      out.extend(ExplodeText!(hook));
780      out.push(T_END!());
781      out.push(T_BEGIN!());
782      out.push(cs);
783      out.push(T_END!());
784      return Ok(Tokens::new(out));
785    }
786    push_value("@at@begin@document@rawparam", rules)?;
787    return Ok(Tokens!());
788  }
789  if lookup_meaning(&T_CS!("\\hook_gput_code:nnn")).is_some() && (label.is_some() || !has_param) {
790    let mut out = vec![T_CS!("\\AddToHook"), T_BEGIN!()];
791    out.extend(ExplodeText!(hook));
792    out.push(T_END!());
793    if let Some(label) = label {
794      out.push(T_OTHER!("["));
795      out.extend(label.unlist());
796      out.push(T_OTHER!("]"));
797    }
798    out.push(T_BEGIN!());
799    out.extend(rules.unlist());
800    out.push(T_END!());
801    Ok(Tokens::new(out))
802  } else {
803    push_value(store, rules)?;
804    Ok(Tokens!())
805  }
