22  // \@tabacckludge body lives in `latex_constructs_rust_only.rs` (Perl
23  // has it in latex_base.pool.ltxml L357; the rust_only.rs copy is the
24  // dump-path override per its docstring section 7).
25  // latex.ltx L10007 — `\let\a=\@tabacckludge`. The dump carries
26  // `\a` as an E record (the serializer captured
27  // `\@tabacckludge`'s body with a `\@changed@cmd` wrapper, not a
28  // pure Let-alias), which the outer M-gate rejects as a
29  // public-CS Expandable (expl3-cascade safety). Neither the
30  // PA/MPA gate relaxation nor the deferred-alias retry pass in
31  // `dump_reader.rs` applies, so we keep this alias hand-written
32  // to match latex.ltx source. Inside a `tabbing` environment,
33  // tabbing_bindings() overrides this local to `\@tabbing@accent`.
34  // Found in arxiv 1611.05395.
35  //
36  // **Lazy-pool-load guard (2026-05-01)**: in Perl, the kernel
37  // `\let\a=\@tabacckludge` runs at engine init, BEFORE user TeX.
38  // If the user defines `\def\a{\alpha}` in their preamble (a
39  // common Greek-letter abbreviation), the user assignment runs
40  // LATER in TeX's normal "later wins" semantics and overrides the
41  // kernel Let cleanly. In Rust, `latex_constructs` runs at
42  // `\documentclass`-time (lazy-pool-load —
43  // `wisdom_lazy_pool_load.md`), AFTER the user preamble — so this
44  // bare `Let!` would clobber the user's `\def\a{\alpha}` and
45  // route subsequent `\a` invocations through the kernel
46  // `\@changed@cmd → \+ → \tabalign` chain that triggers a
47  // `\halign`/`\hbox` mode-mismatch runaway. Witness:
48  // hep-th0005268, see `wisdom_tabalign_math_runaway.md`.
49  //
50  // Guard: only Let `\a` if it's not already user-defined. The
51  // dump E record was rejected, so `\a` would be undefined here
52  // unless the user defined it.
53  if !has_meaning(&T_CS!("\\a")) {
54    Let!("\\a", "\\@tabacckludge");
55  }

57  // A starred declarator's macro is not `\long` (latex.ltx:1229-1233 `\@star@or@long`), so its arguments end at a
58  // `\par` (tex.web §392): `tex_declared` for the starred forms with arguments — here and in `\DeclareRobustCommand`,
59  // `\newenvironment`, `\renewenvironment` below. The unstarred ones stay unchecked (their `\long`, latex.ltx:1281, is
60  // not recorded, as in Perl). Witness 1001.1670; guards `scanner_status::a_starred_*`.
61  DefPrimitive!("\\newcommand OptionalMatch:* SkipSpaces LaTeXDefToken [Number][]{}",
62  sub[(star,cs_token,nargs,opt,body)] {
63    let nargs = nargs.value_of() as usize;
64    let (definable, plain_origin) = is_definable_latex(&cs_token)?;
65    // (a stand-in's copy declares a text default LaTeXML predefined, which the real class defines itself:
66    // `state::copy_provides_text_default_yields`)
67    let definable = definable || copy_provides_text_default_yields(&cs_token);
68    if !definable {
69      note_copy_provides_title_page(&cs_token, &body);
70      if has_value(&s!("{}:locked", cs_token.to_string())) && copy_provides_refuses_locked(&cs_token) {
71        // (a stand-in's copy takes nothing over: `state::copy_provides_refuses_locked`)
72      } else if copy_provides_text_default_clashes(&cs_token) {
73        // (a stand-in's copy declares a name its base class has: LaTeX's error, the definition kept —
74        // `state::copy_provides_text_default_clashes`)
75        let message = s!("Command {} already defined", cs_token.to_string());
76        Error!("latex", "already_defined", message);
77      } else if !has_value(&s!("{}:locked", cs_token.to_string())) { // not locked, inform.
78        let message = s!("Ignoring redefinition (\\newcommand) of {}", cs_token.stringify());
79        Info!("ignore", cs_token, message);
80      } else {
81        // The lock drops the class's definition but keeps its SIGNATURE
82        // (ryethesis.cls:344 `\newcommand{\abstract}[1]{…\gdef…{#1}}`: the
83        // braced text is one stored argument) — the same record a dropped
84        // raw `\def` leaves (state.rs `<cs>:redefined@nargs`).
85        let redefined = s!("{}:redefined", cs_token.to_string());
86        let redefined_nargs = s!("{}:redefined@nargs", cs_token.to_string());
87        AssignValue!(&redefined => true, Some(Scope::Global));
88        AssignValue!(&redefined_nargs => Number::new(nargs as i64), Some(Scope::Global));
89        // …and the dropped definition itself, as `\lx@dropped@<name>` (the same
90        // record `install_definition` keeps for a dropped `\def`/`\renewcommand`).
91        let dropped = T_CS!(s!("\\lx@dropped@{}", cs_token.to_string().trim_start_matches('\\')));
92        let macro_args = convert_latex_args(nargs, opt)?;
93        DefMacro!(dropped, macro_args, body, scope => Some(Scope::Global));
94      }
95      return Ok(vec![]);
96    }
97    if opt.is_some() {
98      define_optional_internal(&cs_token, nargs, &body, star.is_some())?;
99    }
100    let macro_args = convert_latex_args(nargs, opt)?;
101    // When the CS came from the plain pool, bypass any `<cs>:locked`
102    // guard so latex.ltx can layer over plain. RAII guard ensures the
103    // lock state is restored even if `DefMacro!` errors. No-op when
104    // the CS was previously undefined.
105    let _unlock = plain_origin.then(
106      || local_state_unlocked_guard(true));
107    DefMacro!(cs_token, macro_args, body, tex_declared => { star.is_some() && nargs > 0 });
108  });

111  sub[(star, cs, nargs_num, opt, body)] {
112    let nargs = nargs_num.value_of() as usize;

131  // Perl L2597-2602: \@yargdef checks if arg2 equals \tw@ (2) for optional arg type.
132  // `convert_latex_args` flags the optional parameter `testopt` (a futurelet
133  // peek): exact for the `\@xargdef` path; the `\\cs` latex.ltx:1254-1257
134  // builds here is a plain delimited `[#1]` macro that never peeks, but it is
135  // never met in a number scan before its outer macro runs, so the flag is inert.

137    let nargs_str = nargs_toks.to_string();

153    // `IsDefinable!`, for the same reason as `\newcommand`/`\newenvironment`: an
154    // autoload TRIGGER (e.g. `\align`→amsmath from `def_autoload`) appears defined

200  // The following commands define encoding-specific expansions
201  // or glyphs.  The control-sequence is defined to use the expansion for
202  // the current encoding, if any, or the default expansion (for encoding "?").
203  // We don't want to redefine control-sequence if it already has a definition:
204  // It may be that we've already defined it to expand into the above conditional.
205  // But more importantly, we don't want to override a hand-written definition (if any).
206  // Perl latex_constructs.pool.ltxml:2588-2591/2602-2605: the bare command
207  // is the CALL-TIME encoding dispatcher (`\<cf@encoding>\cs`, else `\?\cs`),
208  // exactly as `\DeclareTextSymbol` below installs it. The port froze the bare
209  // command to the FIRST encoding body instead, so lgrenc.def:278's
210  // `\ProvideTextCommand{\textbetasymbol}{LGR}{\LGR@TextSymbolUnavailable…}`
211  // stuck and textalpha's later `normalize-symbols` override of
212  // `\LGR\textbetasymbol` never reached it: "character beta symbol not
213  // available" ×~200 across the greek-fontenc manuals (char-list,
214  // hyperref-with-greek, alphabeta-doc). Guard:
215  // `perfect_kernel_batch54::provide_text_command_dispatches_on_encoding`.
216  // `\fi`-free: the chosen command must see its ARGUMENT next. With Perl's
217  // `…\else\csname…\endcsname\fi` shape an argument-taking text command
218  // (`\accperispomeni{a}`) read the `\fi` as its argument and the real
219  // `{a}` followed the mark ("͂α" — the mark BEFORE the letter).
220  // Perl keeps an existing bare definition (a hand-written macro must win,
221  // and the kernel's Unicode glyph primitives `\textdegree` … beat a fontmap
222  // slot lookup — the whole 30_encoding family depends on that); real
223  // `\DeclareTextCommand` always redefines. One carve-out: a CONTROL SYMBOL
224  // (`\<`, `\>`) that is a non-expandable primitive gives way —
225  // textalpha.sty:187-188 declares the breathings on `\<`/`\>`, and `\>` is
226  // otherwise the math `\mskip` primitive, which in text turned
227  // `\>'\textalpha` into a space + `'`.
228  fn text_command_may_define(cs: &Token) -> Result<bool> {
229    let is_control_symbol =
230      cs.with_str(|s| s.len() == 2 && !s.ends_with(|c: char| c.is_alphabetic()));
231    Ok(match lookup_definition(cs)? {
232      None => true,
233      Some(defn) => is_control_symbol && !defn.is_expandable(),
234    })
235  }

237  // `locked => true` on the `\Declare...`/`\Provide...` text primitives
238  // below: a raw-loaded package may `\def\DeclareTextSymbol{...}` to
239  // route through its own TeX-level dispatch (e.g. mathtext.sty's
240  // `\DeclareTextMathSymbol` chain). When the package's chain depends
241  // on LaTeX kernel internals we don't fully implement
242  // (`\@changed@tmcmd`, `\@tmchar@`, `\csname\cf@encoding\string<cs>
243  // \endcsname → \chardef` resolution), the override produces an
244  // unbounded macro-expansion loop. Perl LaTeXML has no binding for
245  // mathtext.sty and skips it entirely (`Warning:missing_file`); we
246  // raw-load instead (default `INCLUDE_STYLES=true` + ar5iv
247  // parity), so the override fires and breaks. Locking keeps our
248  // primitive in place even under raw-load; the package's `\def` is
249  // logged as Info:ignore and skipped. Witness: paper 2305.16331 +
250  // `\u\i` under `mathtext + T2A`.
251  DefPrimitive!("\\DeclareTextCommand DefToken {}[Number][]{}",
252  sub[(cs, encoding, nargs, opts, expansion)] {
253    let cs_str = cs.to_string();
254    let nargs = nargs.value_of() as usize;
255    let encoding_str = Expand!(encoding).to_string();
256    let ecs = T_CS!(s!("\\{encoding_str}{cs_str}"));
257    if let Some(glyph) = bound_text_symbol(&cs_str).filter(|_| reading_encoding_file(&encoding_str)) {
258      let_i(&ecs, &glyph, None);
259    } else {
260      let ecs_args = convert_latex_args(nargs, opts)?;
261      DefMacro!(ecs, ecs_args, expansion);
262    }
263    if text_command_may_define(&cs)? {
264      def_text_command_dispatcher(&cs, &cs_str, false)?;
265    }
266  }, locked => true);

299  // The body of a text symbol declared in an encoding that has no font map (below): the glyph is unknown, so the
300  // symbol prints nothing, and says so once per symbol, naming the encoding.
301  // An encoding with a map whose slot holds no character (tipa's T3 `\texthooktop`, slot 35: t3_fontmap.rs) says that.
302  DefMacro!("\\lx@unmapped@text@symbol{}{}", sub[(encoding, cs)] {
303    let encoding = encoding.to_string();
304    let cs = cs.to_string();
305    if !already_reported(&s!("unmapped_text_symbol_{encoding}{cs}")) {
306      let reason = if load_font_map(&encoding).is_none() { "has no font map" } else { "has no Unicode glyph for it" };

312  DefPrimitive!("\\DeclareTextSymbol DefToken {}{Number}", sub[(cs, encoding, code)] {
313    // Perl `latex_constructs.pool.ltxml:2671-2682`:
314    //   if (isDefinableLaTeX($cs)) {
315    //     DefMacroI($cs, undef,
316    //       '\expandafter\ifx\csname\cf@encoding\string'.$css.'\endcsname\relax
317    //          \csname?\string'.$css.'\endcsname
318    //        \else\csname\cf@encoding\string'.$css.'\endcsname\fi');
319    //   }
320    //   my $ecs = T_CS('\\'.$encoding.$css);
321    //   $STATE->installDefinition(CharDef->new($ecs, 'restricted_horizontal', $code, $encoding));
322    //
323    // The bare `\cs` is ALWAYS defined as a dispatch-chain macro that
324    // resolves to `\<encoding>\cs` via `\csname`. The encoding-specific
325    // `\<encoding>\cs` carries the actual glyph (as a CharDef in Perl;
326    // as a `PrimitiveBody::String` primitive here — both produce a
327    // single char on invocation).
328    //
329    // Witness for why this matters: paper 2305.16331 + mathtext.sty +
330    // `\u\i` under T2A. mathtext.sty raw-loads and `\def\DeclareTextSymbol`
331    // redefines our handler. T2A's `\DeclareTextSymbol{\i}{T2A}{25}` then
332    // runs mathtext's chain (`\DeclareTextMathSymbol`), which redefines
333    // `\i` to `\T2A-tmcmd \i \T2A\i \T2Amath\i`. That chain bottoms out
334    // at `\csname\cf@encoding\string\i\endcsname = \T2A\i` — but only if
335    // `\T2A\i` is a real definition. The prior happy-path bypassed both
336    // the macro chain AND assumed `\i` would never be re-routed; under
337    // mathtext's override `\i`'s chain looped, growing pushback past
338    // the 4 GiB OOM boundary. With this Perl-faithful chain in place,
339    // mathtext's override produces an equivalent chain that terminates
340    // at the same `\T2A\i` primitive.
341    // The slot decodes through [`decode_slot`].
342    let cs_str = cs.to_string();
343    let encoding_str = Expand!(encoding).to_string();
344    let ecs = T_CS!(s!("\\{encoding_str}{cs_str}"));
345    // `decode_str`, not `decode`: a fontmap slot may hold MORE than one
346    // character. Perl's `FontDecode` returns the whole string
347    // (`$$map[$code]`), and Rust splits that across a single-`char` array plus
348    // the `_fontmap_multichar` side table — which only `decode_str` consults,
349    // along with T1's `SS` and the NBSP-prefixed combining accents. Decoding
350    // with `decode` silently dropped the second character: T2B slot 128 came
351    // out as `Ӷ` (U+04F6) instead of `Ӷ̶` (U+04F6 U+0336), i.e. a DIFFERENT
352    // letter with its stroke removed, where Perl keeps the pair.
353    if let Some(replacement_value) = decode_slot(code.value_of(), &encoding_str)
354      .map(|decoded| cyrillic_letter_in_latin_slot(&cs_str, decoded))
355    {
356      // Encoding-specific carries the actual glyph.
357      def_primitive(ecs, None, Some(PrimitiveBody::String(replacement_value)),
358        PrimitiveOptions::default())?;
359    } else if let Some(glyph) = bound_text_symbol(&cs_str) {
360      // Can't decode a bound symbol's slot: the binding's glyph.
361      let_i(&ecs, &glyph, None);
362    } else if IsDefinable!(&ecs) {
363      // Can't decode: install no-op fallback so downstream chains find
364      // *something* to resolve to. Witness arXiv:1802.05444 / tipa T3.
365      // The symbol is lost where it is used — in an encoding with no font map
366      // at all, or at a slot its map has no character for (tipa's T3 hooktop,
367      // slot 35) — and says so, naming the encoding (`\lx@unmapped@text@symbol`;
368      // X2's `\cyrje` vanished silently before the X2 map, 77s). A Token body,
369      // not a closure, so a format dump keeps it.
370      let mut body = vec![T_CS!("\\lx@unmapped@text@symbol"), T_BEGIN!()];
371      body.extend(Explode!(&encoding_str));
372      body.extend([T_END!(), T_BEGIN!()]);
373      body.extend(Explode!(&cs_str));
374      body.push(T_END!());
375      DefMacro!(ecs, None, Tokens::new(body));
376    }
377    // Bare `\cs` always becomes a dispatch chain (matches Perl).
378    if IsDefinable!(&cs) {
379      DefMacro!(cs, None, Some(s!(
380        r"\expandafter\ifx\csname\cf@encoding\string{cs_str}\endcsname\relax\csname?\string{cs_str}\endcsname\else\csname\cf@encoding\string{cs_str}\endcsname\fi"
381      ).into()));
382    }
383  }, locked => true);

385  // Perl `latex_constructs.pool.ltxml:2684-2688`:
386  //   DefPrimitive('\DeclareTextSymbolDefault DefToken {}', sub {
387  //     my ($stomach, $cs, $encoding) = @_;
388  //     $encoding = ToString(Expand($encoding));
389  //     DefMacroI(T_CS('\?' . ToString($cs)), undef,
390  //               T_CS('\' . $encoding . ToString($cs))); });
391  //
392  // Registers the `\?<cs>` → `\<encoding><cs>` alias used by the
393  // `DeclareTextSymbol` fallback (line 5662 above). Without this side
394  // effect, tipa-style `\DeclareTextSymbolDefault\textrhookrevepsilon{T3}`
395  // leaves `\textrhookrevepsilon` undefined for any encoding other than
396  // T3, so a paper that loads tipa but is typeset in T1 errors.
397  DefPrimitive!("\\DeclareTextSymbolDefault DefToken {}", sub[(cs, encoding)] {
398    let cs_str = cs.to_string();
399    let encoding_str = Expand!(encoding).to_string();
400    // ltoutenc.dtx: `\DeclareTextSymbolDefault{\cs}{enc}` is
401    // `\DeclareTextCommandDefault{\cs}{\UseTextSymbol{enc}\cs}` — the symbol
402    // is typeset IN that encoding. Perl's bare `\<enc>\cs` alias only works
403    // for numeric-slot symbols; lgrenc.def:190 `\DeclareTextCommand{\textsigma}
404    // {LGR}{s\noboundary}` needs the LGR fontmap active or it prints a Latin
405    // `s` (textalpha under T1; witnesses arXiv:2603.02703, 2604.09141).
406    let alias_cs = T_CS!(s!("\\?{cs_str}"));
407    DefMacro!(alias_cs, None, Some(s!("\\UseTextSymbol{{{encoding_str}}}{{{cs_str}}}").into()));
408  }, locked => true);

2379/// The presentation of a math symbol or accent its symbol font cannot decode — no glyph at its slot in the
2380/// declared font's map, or no map at all, and no reference slot ([`reference_symbol`]) — and what it does when printed.
2381/// The presentation is U+FFFD, the unrepresentable character, never the raw slot as a character: that was the glyph of
2382/// no font, kpfonts' `\neq` (`U/jkpsyc` 44) as `,`, a slot "0A Ω as a line break, newtx's `\lbrace` at 215 as `×`, and
2383/// a control slot is no XML character (1501.05180). Printed, the symbol says so once per symbol font, under
2384/// `missing_font_encoding` as a text symbol does (`\lx@unmapped@text@symbol`), naming the font, its encoding and family
2385/// (`<encoding>/<family>`; an undeclared symbol font, which has neither, by LaTeX's own name for it, `\sym<font>`), and
2386/// the symbol or accent and its slot; the font's further undecodable symbols say nothing more. The Warning is the box's
2387/// ([`latexml_core::document::absorb_warning`]), raised when the document absorbs it: a symbol declared and never used,
2388/// or digested only into a box never printed — program.sty's `\powersetsymbol` in a savebox (program.sty:836-837,
2389/// sn-jnl.cls copies), a `\settowidth`, a counter's `\the…` digested under a discarded hold — says nothing, and leaves
2390/// the once-per-font key unset for the symbol the document prints. A character symbol (`\DeclareMathSymbol{-}…`)
2391/// prints its own character (`def_math`'s rewrite), so it loses nothing and says nothing; its `name` is kept, as before, unless it sits at its own code (2307.16667's `-`; newtxsf's digits keep
2392/// `name="2"`). Witnesses 1604.04618, 2008.13643 (kpfonts), 1502.03923 (esvect `\fldr`), 2402.09138 (cmll
2393/// `\invamp`), 2307.11186 (heppennames2). OXIDIZED_DESIGN_DIVERGENCES #27; guards `perfect_kernel_batch89b::*`.
2394fn undecodable_math_slot(
2395  font_name: &str,
2396  encoding: &str,
2397  decl: Option<&Rc<Font>>,
2398  cs: &Token,
2399  code: i64,
2400  kind: &str,
2401) -> (String, Option<Stored>) {
2402  // A character symbol prints itself; its presentation only decides whether `def_math` keeps a `name`, which it keeps
2403  // unless the character sits at its own code — as when the raw slot was the presentation.
2404  if let Some(character) = cs.with_str(|name| (name.len() == 1).then(|| name.to_string())) {
2405    let at_own_code = character
2406      .chars()
2407      .next()
2408      .is_some_and(|c| i64::from(u32::from(c)) == code);
2409    return (
2410      if at_own_code {
2411        character
2412      } else {
2413        "\u{FFFD}".to_string()
2414      },
2415      None,
2416    );
2417  }
2418  // The Warning's object: the font's layout, or for an undeclared symbol font (no encoding, no family), its name as
2419  // LaTeX keys it, `\sym<font>` (latex.ltx's `\DeclareMathSymbol` raises "Symbol font `<font>' is not defined").
2420  let layout = match decl.and_then(|font| font.get_family()) {
2421    Some(family) => s!("{encoding}/{family}"),
2422    None if decl.is_none() => s!("\\sym{font_name}"),
2423    None => encoding.to_string(),
2424  };
2425  let font = if decl.is_none() {
2426    s!("Symbol font {font_name} is not declared")
2427  } else if !symbol_font_has_map(encoding, decl) {
2428    s!("Symbol font {font_name} ({layout}) has no font map")
2429  } else if symbol_font_lacks_slot(encoding, decl, code) {
2430    s!("Symbol font {font_name} ({layout}) has no character at slot \"{code:02X}")
2431  } else {
2432    s!("Symbol font {font_name} ({layout}) has no Unicode glyph at slot \"{code:02X}")
2433  };
2434  let cs_name = cs.with_str(ToString::to_string);
2435  let message = s!(
2436    "{font}: the math {kind} {cs_name} (slot \"{code:02X}) prints U+FFFD, as will any other symbol of this font it \
2437     cannot decode"
2438  );
2439  let warning = document::absorb_warning(
2440    &s!("unmapped_math_symbol_font_{font_name}"),
2441    "missing_font_encoding",
2442    &layout,
2443    &message,
2444  );
2445  ("\u{FFFD}".to_string(), Some(warning))
2446}
