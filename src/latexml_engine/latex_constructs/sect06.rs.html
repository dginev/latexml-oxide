26/// The open `\trivlist`s, innermost last: the group level each began at, with the list state `begin_itemize` bound
27/// over (a global value stack; see `\trivlist`).
28const OPEN_TRIVLISTS: &str = "open_trivlists";
29/// What `begin_itemize` binds: the values that decide where the next list nests, the meanings it lets (`\item`,
30/// `\par`, `\@listctr`) and the `\itemsep` it sets.
31const TRIVLIST_SAVED_VALUES: [&str; 4] = [
32  "itemcounter",
33  "itemization_level",
34  "@itemlevel",
35  "itemization_items",
36];
37const TRIVLIST_SAVED_MEANINGS: [&str; 3] = ["\\item", "\\par", "\\@listctr"];
38const TRIVLIST_SAVED_REGISTERS: [&str; 1] = ["\\itemsep"];
39
40/// The group level the innermost open `\trivlist` began at.
41fn innermost_trivlist_group_level() -> Option<usize> {
42  with_value(OPEN_TRIVLISTS, |v| match v {
43    Some(Stored::VecDequeStored(open)) => match open.back() {
44      Some(Stored::HashStored(saved)) => match saved.get("group_level") {
45        Some(Stored::Int(level)) => usize::try_from(*level).ok(),
46        _ => None,
47      },
48      _ => None,
49    },
50    _ => None,
51  })
52}

54/// Records a `\trivlist` begun at group level `level`, with the list state it is about to bind over.
55fn save_list_state(level: usize) -> Result<()> {
56  drop_trivlists_deeper_than(level)?;
57  let mut values = SymHashMap::default();
58  for key in TRIVLIST_SAVED_VALUES {
59    values.insert(key, lookup_value(key).unwrap_or(Stored::None));
60  }
61  let mut meanings = SymHashMap::default();
62  for cs in TRIVLIST_SAVED_MEANINGS {
63    meanings.insert(cs, lookup_meaning(&T_CS!(cs)).unwrap_or(Stored::None));
64  }
65  let mut registers = SymHashMap::default();
66  for cs in TRIVLIST_SAVED_REGISTERS {
67    if let Some(value) = lookup_register(cs, Vec::new())? {
68      registers.insert(cs, Stored::from(value));
69    }
70  }
71  let mut saved = SymHashMap::default();
72  saved.insert("group_level", Stored::Int(level as i64));
73  saved.insert("values", Stored::HashStored(values));
74  saved.insert("meanings", Stored::HashStored(meanings));
75  saved.insert("registers", Stored::HashStored(registers));
76  push_value(OPEN_TRIVLISTS, Stored::HashStored(saved))
77}

79/// Puts back, in the current group, the list state a `\trivlist` recorded (an undefined meaning stays undefined).
80fn restore_list_state(saved: SymHashMap<Stored>) -> Result<()> {
81  for (part, entries) in saved {
82    let Stored::HashStored(entries) = entries else {
83      continue;
84    };
85    for (key, value) in entries {
86      let name = with(key, |k| k.to_string());
87      if part == pin!("values") {
88        assign_value(&name, value, None);
89      } else if part == pin!("meanings") {
90        assign_meaning(&T_CS!(name), value, None);
91      } else if part == pin!("registers") {
92        let register = match value {
93          Stored::Glue(glue) => RegisterValue::Glue(glue),
94          Stored::Dimension(dimension) => RegisterValue::Dimension(dimension),
95          Stored::Number(number) => RegisterValue::Number(number),
96          _ => continue,
97        };
98        assign_register(&name, register, None, Vec::new())?;
99      }
100    }
101  }
102  Ok(())
103}

105/// Forgets the trivlists recorded deeper than group level `level`: their groups have ended (and their state with
106/// them), whether or not their `\aftergroup` closer ran (an abandoned group drops it, stomach.rs
107/// `until_terminal_inside_group`).
108fn drop_trivlists_deeper_than(level: usize) -> Result<()> {
109  while innermost_trivlist_group_level().is_some_and(|d| d > level) {
110    pop_value(OPEN_TRIVLISTS)?;
111  }
112  Ok(())
113}
114
115/// Ends, at digestion, every trivlist begun at group level `level`: forgets them and puts back the list state the
116/// outermost of them bound over.
117fn end_trivlists_of_group(level: usize) -> Result<()> {
118  drop_trivlists_deeper_than(level)?;
119  let mut outermost = None;
120  while innermost_trivlist_group_level() == Some(level) {
121    outermost = pop_value(OPEN_TRIVLISTS)?;
122  }
123  if let Some(Stored::HashStored(saved)) = outermost {
124    restore_list_state(saved)?;
125  }
126  Ok(())
127}

129fn prop_group_level(props: &SymHashMap<Stored>) -> usize {
130  props
131    .get("group_level")
132    .map(|d| d.to_string())
133    .and_then(|d| d.parse().ok())
134    .unwrap_or(0)

1169/// no other quote (one around it stays open: probe b_qsq, a `{quote}` in a `{quotation}`), only the paragraph
1170/// the body's tail is in, as TeX's `\endlist` ends it (latex.ltx `\@endparenv`), so the text after the environment is
1171/// a paragraph of its own (probe b_quote_text). A run-in `\paragraph`/`\subparagraph` leaves the quote open (an
1172/// `ltx:inline-sectional-block` in it, closed here). Witnesses 2203.09193 1504.06728 (a bibliography in a quote, a
1173/// verse), 0712.1985 (`\section*` in a `{quotation}`), 2607.27159 (`\paragraph` in a quote). Guards
1174/// `perfect_kernel_batch69a::{bibliography_in_a_quote_keeps_the_tail_paragraph,
1175/// bibliography_in_a_nested_quote_keeps_the_outer_quote, paragraph_in_a_quote_stays_in_it}`.
1176fn quote_environment(
1177  document: &mut Document,
1178  props: &SymHashMap<Stored>,
1179  role: Option<&str>,
1180) -> Result<()> {
1181  // The template's open: no attributes, no font (`<ltx:quote>`); with one, the whatsit's font.
1182  let quote = match role {
1183    None => document.open_element("ltx:quote", None, None)?,
1184    Some(role) => {

1187        Some(Stored::Font(f)) => Some(Cow::Borrowed(&**f)),
1188        Some(Stored::FontDirective(FontDirective::Asset(fa))) => Some(Cow::Borrowed(&**fa)),
1189        Some(Stored::FontDirective(FontDirective::Closure(code))) => Some(Cow::Owned(code(None)?)),
1190        _ => None,
1191      };
1192      document.open_element("ltx:quote", Some(attrs), font.as_deref())?
1193    },
1194  };
1195  if let Some(Stored::Digested(body)) = props.get("body") {
1196    document.absorb(body, None)?;
1197  }
1198  // `close_quotes_blocking` may have removed the quote (closed and left empty, then freed): the freed handle is
1199  // blanked (libxml's `free_subtree` neutralizes every wrapper into the subtree), so it is neither the current node
1200  // nor has children, and `is_open` is false — the quote counts as closed, as it is.
1201  if document.is_open(&quote) {
1202    document.close_element_if_open("ltx:quote")?;
1203  } else {
1204    // The tail's paragraph text: the innermost `ltx:p` reached through elements that auto-close.

1207      if current.get_type() == Some(NodeType::ElementNode) {

1209          document.close_node_internal(&current)?;
1210          break;
1211        }
1212        if !document::can_auto_close(&current) {
1213          break;
1214        }
1215      }
1216      node = current.get_parent();
1217    }
1218  }
1219  Ok(())
1220}
