8/// The type TeX scans for an assignment to a register of `register` type
9/// (`\setlength#1#2{#1 #2\relax}`, latex.ltx:10253): its own, a `\chardef`'s
10/// being a number. A register of no numeric type falls back to a dimension,
11/// Perl's `{Dimension}`.
12fn length_register_type(register: Option<RegisterType>) -> RegisterType {
13  match register {
14    Some(
15      kind @ (RegisterType::Number
16      | RegisterType::Dimension
17      | RegisterType::Glue
18      | RegisterType::MuDimension
19      | RegisterType::MuGlue),
20    ) => kind,
21    Some(RegisterType::CharDef) => RegisterType::Number,
22    _ => RegisterType::Dimension,
23  }
24}

26/// `\setlength`/`\addtolength` as latex.ltx:10253-10254 define them — `\def\setlength#1#2{#1 #2\relax}`,
27/// `\def\addtolength#1#2{\advance#1 #2\relax}`: TeX scans the register from `#1` and its value from what follows it
28/// there, the rest of `#1`, a space, then `#2`; what the scan leaves is read after the assignment. So
29/// `\setlength{\oddsidemargin 0.5cm}\setlength{\evensidemargin 0.5cm}` (the second `\setlength` its `#2`) sets
30/// 0.5cm and the second runs next: read as two arguments, `#1`'s rest went back after `#2` was taken and the next
31/// `\setlength` read "0" as its register ("A <variable> was supposed to be here", 2605.02145, 2605.06074,
32/// 2605.08367, 2605.24444, 2605.29037, 2606.03162; Perl drops the rest, KNOWN_PERL_ERRORS #275). The register is
33/// read by the `Variable` reader (prefixes, a braced `\relax` and spaces skipped; a non-register is reported and
34/// skipped, as Perl), its value by its own type (OXIDIZED_DESIGN #317; guards `braced_quantity_tail`,
35/// `perfect_kernel_batch61::setlength_reads_its_register_and_value_as_one_stream`). A `\setlength` whose
36/// `#1` begins with no register is TeX's code `#1 #2\relax`, run as it stands (OXIDIZED_DESIGN #746; guard
37/// `perfect_kernel_batch73b::setlength_on_a_non_register_runs_its_code`).
38fn assign_braced_length(target: Tokens, length: Tokens, add: bool) -> Result<()> {
39  thread_local! {
40    /// The `Variable` reader, parsed once.
41    static VARIABLE: std::cell::OnceCell<Option<Parameters>> = const { std::cell::OnceCell::new() };
42  }
43  let Some(variable) = VARIABLE.with(|cell| {
44    cell
45      .get_or_init(|| {
46        parse_parameters("Variable", &T_CS!("\\setlength"), true)
47          .ok()
48          .flatten()
49      })
50      .clone()
51  }) else {
52    return Ok(());
53  };
54  let length_tokens = length.clone();
55  let mut stream = target.unlist();
56  stream.push(T_SPACE!());
57  stream.extend(length.unlist());
58  let code = stream.clone();
59  let mut executed = false;
60  let (assignment, tail) = read_braced(Tokens::new(stream), || {
61    // `\setlength`'s `#1 #2\relax` is code TeX executes: what `#1` begins with, past spaces and `\relax`, when it is
62    // neither a register nor a prefix, is simply run (`\setlength` before an `\item`, IEEEtran templates' empty
63    // `\setlength` line: 2201.08831, 2401.09168), where the `Variable` read reported "A <variable> was supposed to be
64    // here" and dropped it. `\addtolength`'s `\advance` reads a variable, and errs as TeX does. The preamble runs it
65    // too: what it typesets there starts a paragraph, whose `\everypar{\@nodocument}` (latex.ltx:9137) is TeX's
66    // "Missing \begin{document}" (fancyheadings' `\headrulewidth`, a macro since 2.0: `\setlength{\headrulewidth}{0pt}`
67    // prints "0.4pt 0pt"; `\lx@setlength@nodocument` below; guard
68    // `perfect_kernel_batch73b::setlength_on_a_macro_in_the_preamble_errs_as_tex`).
69    if !add {
70      loop {
71        match read_x_token(None, false, None)? {
72          Some(token) if token.get_catcode() == Catcode::SPACE || token == T_CS!("\\relax") => {},
73          Some(token) => {
74            let assignable = lookup_register_definition(&token).is_some()
75              || (token.get_catcode().is_active_or_cs()
76                && lookup_definition(&token)?.is_some_and(|defn| defn.is_prefix()));
77            unread_vec(vec![token]);
78            if !assignable {
79              executed = true;
80              return Ok(None);
81            }
82            break;
83          },
84          None => {
85            executed = true;
86            return Ok(None);
87          },
88        }
89      }
90    }
91    let Some(ArgWrap::RegisterDefinition(dbox)) = variable.read_arguments(None)?.into_iter().next()
92    else {
93      return Ok(None);
94    };
95    let (rtoken, params) = *dbox;
96    let Some(defn) = rtoken.to_register() else {
97      return Ok(None);
98    };
99    let value = read_length_value(length_register_type(defn.register_type()))?;
100    Ok(Some((defn, params, value)))
101  })?;
102  if executed {
103    let mut code = code;
104    code.push(T_CS!("\\relax"));
105    // In the preamble a paragraph the code starts runs `\@nodocument` (LaTeXML fires no preamble `\everypar`,
106    // stomach.rs `pending_everypar`): tested once the code has run, from the mode it began in.
107    if !x_equals(&T_CS!("\\@nodocument"), &T_CS!("\\relax"))
108      && lookup_string("MODE").ends_with("vertical")
109    {
110      code.push(T_CS!("\\lx@setlength@nodocument"));
111    }
112    unread_vec(code);
113    return Ok(());
114  }
115  if let Some((defn, params, value)) = assignment {
116    if add {
117      let oldlength = defn.value_of(params.clone()).unwrap_or_default();
118      defn.set_value(oldlength.add(value), None, params);
119    } else {
120      // What it states, for a float's width read through the register (the panel grid rule, `note_setlength`).
121      if params.is_empty() {
122        note_setlength(
123          &defn.get_cs_name(),
124          &length_tokens,
125          value.clone().value_of(),
126        );
127      }
128      defn.set_value(value, None, params);
129    }
130  }
131  unread_vec(tail);
132  Ok(())
133}
