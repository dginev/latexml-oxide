64  // Use fonts (w/ special flag) to propogate emphasis as a font change,
65  // but preserve it's "emph"-ness.
66  // Perl latex_constructs.pool.ltxml L401-408: mode => 'restricted_horizontal',
67  //   enterHorizontal => 1, font => { emph => 1 }, alias => '\emph', beforeDigest => {...}.
68  // Perl emits `<ltx:text>` inside math context, `<ltx:emph>` outside —
69  // dispatch via `findnodes('ancestor::ltx:Math')` in the body sub.
70  // Rust template `?#isMath(...)(...)` is the equivalent gate (same
71  // pattern used by `\newline`, `\thinspace`, etc.) — ported 2026-05-01.
72  // The earlier note here claimed a Rust-only `$$`-in-`\emph{}` math
73  // leak; verified 2026-05-03 it is SHARED-FAILURE with Perl, not a
74  // Rust-only divergence. Both engines treat `$$` inside
75  // `restricted_horizontal` as two inline-math toggles (per the
76  // `BOUND_MODE =~ /vertical$/` gate in `\lx@dollar@default`). See
77  // SYNC_STATUS.md Gate 2.A. mode "text" here is fine; flipping to
78  // "restricted_horizontal" is a stylistic difference but does not
79  // change the parity-relevant behavior.
80  // The `?#isMath` branch is Perl's reading of `\emph` in math; since 62b the argument is text
81  // there too (OXIDIZED_DESIGN_DIVERGENCES #440), so it no longer fires — kept as Perl's template.
82  DefConstructor!("\\emph{}",
83    "?#isMath(<ltx:text _force_font='1'>#1)(<ltx:emph _force_font='1'>#1)",
84    // NB: no `mode => "text"` — Perl's \emph (latex_constructs.pool.ltxml:411) uses
85    // only `enterHorizontal => 1, bounded => 1`. Adding `mode => "text"` digested
86    // the argument in restricted_horizontal mode, so BOUND_MODE was not "vertical"
87    // and the `$$` display-math probe in \lx@dollar@default was skipped — `$$…$$`
88    // inside \emph{…} (common in theorem bodies) degraded to two empty inline `$`
89    // and its sub/superscripts errored "Script _/^ can only appear in math mode"
90    // (witness 2203.05327: 34 spurious errors; Perl 0).
91    bounded        => true,
92    enter_horizontal => true,
93    font=> { emph => true },
94    alias => "\\emph",
95    // In math, `\DeclareTextFontCommand`'s `\emph` typesets its argument as text (latex.ltx
96    // `\text@command` → `\nfss@text`, an `\mbox`), as `\textit@math` does: in
97    // `$\emph{P$\bar{3}$m1}$` the inner `$` opens a formula, not closes the outer one (2502.18190,
98    // run 306 error, then TooManyErrors; Perl alike). Outside math no mode, as above.
99    before_digest => {
100      if lookup_bool_sym(pin!("IN_MATH")) {
101        begin_mode("restricted_horizontal")?;
102        assign_frame_value("lx@emph@text", Stored::Bool(true));
103        // Leaving math restores the text font from before the formula, over the emphasis the
104        // declared `font` merged just before: toggle it again, so `$\emph{G}$` in an italic
105        // theorem is upright, as `\nfss@text` gives.
106        merge_font(Font { emph: Some(true), ..Font::default() });
107      }
108    },
109    after_digest => sub[_whatsit] {
110      if is_value_bound("lx@emph@text", Some(0)) {
111        end_mode("restricted_horizontal")?;
112      }
113    },
114    // Perl's `\f@shape` it/n toggle (latex_constructs.pool.ltxml:414-416) is
115    // subsumed by the font merge's NFSS sync (content.rs `merge_font_ref`).
116    after_construct => sub[doc,_args] {
117      doc.maybe_close_element("ltx:emph")?; }
118  );
