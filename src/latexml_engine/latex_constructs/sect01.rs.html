8/// The counters of the kernel's float environments (sect09.rs), made once the document class has loaded when it made
9/// none. latex.ltx defines no float counter; a class makes each with its environment (article.cls:452-464), but a raw
10/// class that makes its counter only where the environment is new — sig-alternate.cls:699
11/// `\@ifundefined{figure}{\newcounter{figure}}` — sees the kernel's and made none, and every caption raised
12/// `\thefigure` undefined (1605.02827, 1607.07514, 1906.01122; Perl the same, KNOWN_PERL_ERRORS #485). Made here, the
13/// preamble and the packages' begin-document hooks can set them (`\setcounter{figure}{4}`), with article's id prefixes,
14/// and a `\thefigure` the class already defined is kept. Not with the environment: KOMA's tocbasic warns "using
15/// already defined counter" when its class makes its own.
16/// Guard: `perfect_kernel_batch61::a_raw_class_testing_the_figure_environment_has_its_counter`.
17pub(crate) fn make_missing_float_counters() -> Result<()> {
18  for (counter, prefix) in [("figure", "F"), ("table", "T")] {
19    if lookup_register(&s!("\\c@{counter}"), Vec::new())
20      .ok()
21      .flatten()
22      .is_some()
23    {
24      continue;
25    }
26    let the = T_CS!(s!("\\the{counter}"));
27    let kept = lookup_meaning(&the);
28    NewCounter!(counter, "document", idprefix => prefix, idwithin => "section");
29    if let Some(meaning) = kept {
30      assign_meaning(&the, meaning, Some(Scope::Global));
31    }
32  }
33  Ok(())
34}

36/// OXIDIZED_DESIGN #179, before the document's class loads: the kernel-level `\chapter` (Perl pool:557, locked) is
37/// cleared, as latex.ltx defines none, so the class's own code meets `\chapter` undefined, as in TeX: a raw class's
38/// `\@ifundefined{chapter}` takes the undefined branch and its `\providecommand\chapter` defines. The lock stays, with
39/// the kernel's binding kept aside as for a file's clearing `\let` (`\chapter:locked@cleared`, state.rs `let_i`,
40/// OXIDIZED_DESIGN #569): the class's own definition is refused, the kernel's chapter put back in its place and the
41/// refusal recorded (`\chapter:redefined`, (a) of the "has chapters" test), so a raw class's chapters stay LaTeXML's
42/// `<chapter>` (scrbook.cls, read raw by scrbook_cls.rs). Where the class has chapters by a binding's counter or
43/// OmniBus's guess, [`settle_chapter_after_class_load`] puts the kernel's back (a binding that needs it during its own
44/// load reads `\lx@kernel@chapter`: amsart_cls.rs `\specialsection`). Only the document's first `\documentclass`: a
45/// second one (a merged file's) loads no class in TeX — in the preamble latex.ltx:18618 of TL 2025 has made it
46/// `\@twoclasseserror`, in the body :18621 `\@onlypreamble\documentclass` stops it ("Can be used only in preamble",
47/// a diagnostic we do not raise yet, SYNC_STATUS) — and keeps the first class's chapters (memoir with a second
48/// `\documentclass{memoir}` in the body: both chapters).
49pub(crate) fn retract_chapter_before_class_load() {
50  if lookup_bool("\\chapter:settles") {
51    return;
52  }
53  let chapter = T_CS!("\\chapter");
54  // (kept aside: LaTeXML's chapter, with a counter of its own at the first call where the class made none)
55  if lookup_bool("\\chapter:locked")
56    && let Some(binding) = lookup_meaning(&T_CS!("\\lx@kernel@chapter@counted"))
57  {
58    assign_value("\\chapter:locked@cleared", binding, Some(Scope::Global));
59  }
60  assign_meaning(&chapter, Stored::None, Some(Scope::Global));
61  // (a refusal of the class's own `\chapter` by the lock is read after the load)
62  assign_value("\\chapter:redefined", false, Some(Scope::Global));
63  assign_value(CLASS_HAS_CHAPTERS, false, Some(Scope::Global));
64  // (the LaTeX 2.09 `\documentstyle` keeps the kernel's `\chapter` in every class: only a `\documentclass` settles)
65  assign_value("\\chapter:settles", true, Some(Scope::Global));
66}

83/// OXIDIZED_DESIGN #179, the "has chapters" step, after each class load of a `\documentclass` document: the class's
84/// own (`\documentclass`), a raw class's `\LoadClass`/`\LoadClassWithOptions` (sect05.rs), a binding's `LoadClass!`
85/// (setup_binding_language.rs) and a runtime script binding's `LoadClass` (latexml_contrib script_bindings/engine.rs;
86/// guard `perfect_kernel_batch72b::a_script_binding_class_load_settles_the_chapters`). It makes the one "has chapters" test (`state::class_has_chapters_now`: (a) the class's
87/// own code assigned `\chapter`, (b) a compiled binding made `\c@chapter`, (c) OmniBus guesses an unknown class) and
88/// stores its answer for `state::class_defines_chapter` — natbib's forks, appendix.sty, titlesec.sty and the kernel's
89/// `\appendix` read it. Where the class has chapters by (b) or (c) and `\chapter` is undefined, the kernel's is put
90/// back, locked (a report-based binding's later code and packages see it); where it has none, `\chapter` is left
91/// undefined and unlocked, so every `\@ifundefined{chapter}` a class or package makes gets TeX's answer: the raw natbib
92/// forks under bmvc2k (bmvc2k_natbib.sty:779) and WileyNJD (NJDnatbib.sty:1050) took the `\chapter*{\bibname}` branch
93/// (2412.14509, 2212.00057, 2210.06270, 1501.07873, 2311.17607, 2406.08486; 2101.06297, 2211.16474, 2411.11493,
94/// 2206.08798, 2602.03829, 2408.07575, 2609.08425, 2506.14995), blindtext's `\blinddocument` made a chapter
95/// (blindtext.sty:243, hvfloat's test documents, coseoul, xassoccnt), and a class body's probe after its
96/// `\LoadClass{article}` (aguplus.cls:524 "figcaps may only be used with article-like classes") saw article's answer.
97/// A class that builds its own chapters afterwards (source3body.tex:100-123 `\newcounter{chapter}` +
98/// `\newcommand\chapter`) is not "Ignoring redefinition of \chapter" (KNOWN_PERL_ERRORS #141). A document that calls
99/// `\chapter` all the same gets LaTeXML's chapter with one Warning (OXIDIZED_DESIGN #734, `\lx@chapterless@chapter`).
100/// Its own removal is no assignment of a stand-in's copy: a copy's `\LoadClass` (tlp) settles outside the copy's
101/// provides-mode.
102/// Guards: `perfect_kernel_batch56::loadclass_return_retracts_kernel_chapter`,
103/// `perfect_kernel_batch72b::chapter_exists_where_the_class_has_chapters`,
104/// `perfect_kernel_batch72b::a_class_has_chapters_when_its_code_defines_chapter`.
105pub fn settle_chapter_after_class_load() -> Result<()> {
106  if !lookup_bool("\\chapter:settles") {
107    return Ok(());
108  }
109  let chapter = T_CS!("\\chapter");
110  let (assigned, kernel_kept) = class_chapter_sources();
111  assign_value(CLASS_HAS_CHAPTERS, assigned || kernel_kept, Some(Scope::Global));
112  if assigned {
113    return Ok(());
114  }
115  let defined = lookup_meaning(&chapter).is_some_and(|m| !matches!(m, Stored::None));
116  let _outside_copy = (definition::origin::copy_provides() != 0)
117    .then(|| definition::origin::CopyProvidesGuard::new(0));
118  if kernel_kept {
119    if !defined {
120      // (with a chapter counter of its own at the first call where the class made none: OmniBus's guess (c) and
121      // subfiles make none, so 1906.09200's chapters were "Counter 'chapter' was not defined")
122      let_i(&chapter, &T_CS!("\\lx@kernel@chapter@counted"), Some(Scope::Global));
123      assign_value("\\chapter:locked", true, Some(Scope::Global));
124    }
125  } else {
126    if defined {
127      let_i(&chapter, &T_CS!("\\@undefined"), Some(Scope::Global));
128    }
129    assign_value("\\chapter:locked", false, Some(Scope::Global));
130  }
131  // (the kernel's binding is in place or the name is free: nothing kept aside for a refusal to put back)
132  assign_value("\\chapter:locked@cleared", Stored::None, Some(Scope::Global));
133  Ok(())
134}

136/// OXIDIZED_DESIGN #179, the LaTeX 2.09 `\documentstyle` document (tex_job.rs): its class keeps the kernel's `\chapter`
137/// (only a `\documentclass` retracts it, `\chapter:settles`), so nothing is put back or cleared, but the "has chapters"
138/// answer is stored where the class's load ends — at its end-of-class hook, after latex209.def's `\@documentclasshook`
139/// and before `\compat@loadpackages` loads the style options — and `\chapter:stored` makes
140/// `state::class_defines_chapter` read it from then on. A refusal the document's own `\def\chapter` meets later
141/// (`\chapter:redefined`) is no class's chapters: `\documentstyle{article}` + `\newcounter{chapter}` +
142/// `\def\chapter#1{…}` letters its appendix "A", as pdflatex, and `\documentstyle[appendix]{report}` still numbers its
143/// appendix chapters "Appendix A". `begin`: the refusals recorded before the class load are no class's either.
144/// Guard: `perfect_kernel_batch72b::appendices_follow_the_class_chapters`.
145pub fn store_chapter_after_latex_ix_class_load(begin: bool) {
146  if begin {
147    assign_value("\\chapter:redefined", false, Some(Scope::Global));
148    assign_value("\\chapter:stored", false, Some(Scope::Global));
149    return;
150  }
151  assign_value(
152    CLASS_HAS_CHAPTERS,
153    class_has_chapters_now(),
154    Some(Scope::Global),
155  );
156  assign_value("\\chapter:stored", true, Some(Scope::Global));
157}

159/// OXIDIZED_DESIGN #734: at `\begin{document}`, after the begin-document hooks have run (so a package's
160/// `\AtBeginDocument` test of `\chapter` still gets TeX's answer), a `\chapter` the class left undefined is given
161/// LaTeXML's chapter as its stand-in (`kernel_autoload::name_undefined_fallback`, read where the undefined token is
162/// met). The name stays undefined, so a test made in the body (blindtext.sty:243 `\blinddocument`, hvfloat's test
163/// documents) also gets TeX's answer, while a document that calls `\chapter` converts, with one Warning
164/// (`\lx@chapterless@chapter`). A call made in the preamble or a begin-document hook met the undefined name, as in TeX
165/// (one Error), and left its undefined-error stub: TeX's `\chapter` is still undefined after it, so the stub goes and the
166/// body's call reads the stand-in. Witnesses: 0802.3014, 0909.2812, 0907.3938, 1403.2269, 1507.01872, 1511.03874,
167/// 1606.03650, 2504.00704 (`Error:undefined:\chapter` before).
168/// Guards: `perfect_kernel_batch72b::chapter_in_a_chapterless_class_is_the_kernel_chapter_with_a_warning`,
169/// `perfect_kernel_batch72b::a_preamble_chapter_errs_and_the_body_chapter_reads_the_stand_in`.
170pub(crate) fn supply_chapter_fallback() {
171  let chapter = T_CS!("\\chapter");
172  if lookup_bool("\\chapter:error@stub") {
173    let_i(&chapter, &T_CS!("\\@undefined"), Some(Scope::Global));
174    assign_value("\\chapter:error@stub", false, Some(Scope::Global));
175  }
176  if lookup_meaning(&chapter).is_none_or(|m| matches!(m, Stored::None)) {
177    binding::kernel_autoload::name_undefined_fallback(&chapter, T_CS!("\\lx@chapterless@chapter"));
178  }
179}

181/// What `\end{<env>}` expands to, typesetting (latex.ltx:15382-15393 `\end `).
182fn end_environment_tokens(env: Tokens) -> Result<Tokens> {
183  let name = Expand!(env).to_string();
184  // (the document does not end in a speculative digestion, by an `\enddocument` a class wrapped neither)
185  if name == "document" && in_speculative_digestion() {
186    return Ok(Tokens::new(Vec::new()));
187  }
188  let before = lookup_tokens(&s!("@environment@{name}@atend"));
189  let after = lookup_tokens(&s!("@environment@{name}@afterend"));
190  let use_hook = |point: &str| env_hook_tokens(&name, point);
191  let t = T_CS!(s!("\\end{{{name}}}"));
192  let mut out_tokens = Vec::new();
193  // The `\end{document}` half of the indirection (see `\begin`): a
194  // user-redefined `\enddocument` runs in place of the constructor (no
195  // `\endgroup`: see `\begin` — the document environment opens no group);
196  // the binding's own definition keeps the constructor.
197  // … \endgroup`); the binding's alias keeps the constructor.
198  // Symmetric with `\begin`: the user's `\enddocument` runs when the
199  // user's `\document` ran, or when `\document` was left alone and only
200  // `\enddocument` was redefined (tools-overview.tex:65). standalone.cls
201  // wraps both; its begin wrapper is self-referential and skipped, so its
202  // end wrapper (`\endstandalone`, closing a minipage the skipped begin
203  // never opened) is skipped too.
204  let indirected = name == "document" && lookup_bool("document:indirected");
205  let user_enddocument = name == "document"
206    && is_user_document_macro(&T_CS!("\\enddocument"), &T_CS!("\\lx@orig@enddocument"))
207    && (indirected || x_equals(&T_CS!("\\document"), &T_CS!("\\lx@orig@document")));
208  let orig_end = x_equals(&T_CS!("\\enddocument"), &T_CS!("\\lx@orig@enddocument"));
209  if indirected && !user_enddocument && !orig_end {
210    // The begin opened the group and `\enddocument` is a closure, not a
211    // user macro — exam-n's `\let\enddocument\endinput`, whose `\endinput`
212    // stops the included file after this line while `\end`'s tokens are
213    // already expanded, as in TeX. LaTeX's `\end{document}` =
214    // `\enddocument\endgroup`: run it, then close the begin's group.
215    // With the ORIGINAL `\enddocument` (a package wrapped `\document`
216    // around the original — dblfnote, etoolbox's `\AtEndPreamble` — and
217    // left the end alone) the finalizer runs as on the constructor path:
218    // in TeX the original never returns, so `\end`'s `\endgroup` is never
219    // reached, and here an explicit one met the document constructor's
220    // mode frame above the begin's group ("\endgroup Attempt to close a
221    // group that switched to mode internal_vertical" at `\end{document}`,
222    // seven manuals in sweep 83: yafoot-man, guitartabs, zanabazr, …).
223    let mut tks = before.map(Tokens::unlist).unwrap_or_default();
224    tks.extend(use_hook("end"));
225    tks.push(T_CS!("\\enddocument"));
226    tks.push(T_CS!("\\par"));
227    tks.push(T_CS!("\\endgroup"));
228    tks.extend(use_hook("after"));
229    if let Some(afterend_toks) = after {
230      tks.extend(afterend_toks.unlist())
231    }
232    return Ok(Tokens::new(tks));
233  }
234  if user_enddocument {
235    let mut tks = before.map(Tokens::unlist).unwrap_or_default();
236    tks.extend(use_hook("end"));
237    tks.push(T_CS!("\\enddocument"));
238    // `\par` first: an issue's last paragraph is still open, and this
239    // engine's group bookkeeping refuses to close a group that switched
240    // mode ("Attempt to close a group that switched to mode horizontal");
241    // TeX ends that paragraph at the next page/vertical material anyway.
242    tks.push(T_CS!("\\par"));
243    tks.push(T_CS!("\\endgroup"));
244    tks.extend(use_hook("after"));
245    if let Some(afterend_toks) = after {
246      tks.extend(afterend_toks.unlist())
247    }
248    return Ok(Tokens::new(tks));
249  }
250  if name != "document" && lookup_bool(&s!("lx@env@indirected@{name}")) {
251    // the begin took LaTeX's indirection (see `\begin`): `\endX`, then the begin's group closes
252    out_tokens = end_environment_by_macro(&name);
253  } else if is_defined_token(&t) {
254    // Magic CS! (its constructor fires `env/NAME/end` inside the group)
255    out_tokens.push(t);
256    out_tokens.extend(use_hook("after"));
257    if let Some(afterend_toks) = after {
258      out_tokens.extend(afterend_toks.unlist())
259    }
260  } else {
261    out_tokens = end_environment_by_macro(&name);
262  }
263  out_tokens.extend([
264    T_CS!("\\if@ignore"),
265    T_CS!("\\@ignorefalse"),
266    T_CS!("\\ignorespaces"),
267    T_CS!("\\fi"),
268  ]);
269  Ok(Tokens::new(out_tokens))
270}

272pub(crate) fn load() -> Result<()> {
273  // ======================================================================
274  // C.1 Commands and Environments
275  // ======================================================================
276
277  // Perl latex_constructs.pool.ltxml L45-48 — page counter + early Lets.
278  // These are redundant with `NewCounter!("page")` later in the file
279  // (L3791) and the `Let!("\\nobreakspace", ...)` later (L4723), but
280  // Perl-faithful: Perl runs them here in the early "C.1 Commands and
281  // Environments" block so any subsequent dump-load (which writes via
282  // `assign_internal` and bypasses :locked) cannot leave these CSes
283  // pointing at raw plain.tex bodies.
284  Let!("\\nobreakspace", "\\lx@nobreakspace");
285
286  //======================================================================
287  // Just to pass test t/alignment/halignatt
288  // Perl latex_constructs.pool.ltxml L51 — redundant with plain_base.rs
289  // (Perl plain_base.pool.ltxml L147), but Perl-faithful.
290  def_macro_noop("\\hidewidth")?;
291
292  // Perl-faithful: LaTeX does NOT define `\magnification`, and babel
293  // uses this to detect whether it's running under LaTeX (vs plain
294  // TeX). The unconditional Let here would also kill plain-TeX papers
295  // whose first line is `\magnification=\magstep1` — fine in Perl
296  // because the Perl driver doesn't pre-load latex_constructs for plain
297  // TeX sources, but the Rust cortex_worker eagerly preloads
298  // `LaTeX.pool` for the `\UseRawInputEncoding`-before-`\documentclass`
299  // case (2407.00348). To keep both working, defer the Let until
300  // `\documentclass` actually fires (see `\documentclass`
301  // after_digest below). Witness: 15 wp4 papers like 2305.09030 ship
302  // `\magnification=\magstep1` and previously failed with
303  // `Error:undefined:\magnification` under the worker.
304  //
305  // `latex_kernel::autoload_latex_kernel` is a second route into the same
306  // state (LaTeX.pool loaded in a plain-TeX document), so the deferral is
307  // load-bearing there too. It cannot fire ON `\magnification` itself:
308  // latex.ltx does not define it, so it is absent from the kernel dump that
309  // the autoload's membership test consults — checked when that hook landed.
310  Let!("\\@empty", "\\lx@empty");
311  Let!("\\@ifundefined", "\\lx@ifundefined");
312  //**********************************************************************
313  // Basic \documentclass & \documentstyle
314
315  DefConditional!("\\if@compatibility", { lookup_bool("2.09_COMPATIBILITY") });
316  def_macro_noop("\\@compatibilitytrue")?;
317  def_macro_noop("\\@compatibilityfalse")?;
318
319  Let!("\\@currentlabel", "\\@empty");
320  DefMacro!("\\@currdir", "./");
321
322  // Defensive `\let \@halignto \@empty` for the kernel macro that's
323  // initialized inline inside \tabular / \tabular* / \array setup
324  // (`\let\@halignto\@empty` at the top of each), but referenced
325  // unprotected inside \edef expansions. If the inline init doesn't
326  // fire before the edef-referencing macro runs (e.g. `\@array`
327  // called outside a tabular context), the reference becomes
328  // undefined and trips `Error:undefined:\\@halignto` cascade.
329  // Witness: 2306.10481 (4 errors), 2307.05820 (22 errors), plus
330  // 2 other stage-2 papers using \begin{array} outside standard wrapping.
331  Let!("\\@halignto", "\\@empty");
332
333  // Defensive `\let \@arrayright \@empty` for array.sty's macro
334  // (array.sty L: `\let\@arrayright\@empty`). Referenced by revtex's
335  // ltxutil.sty (`\def\endarray{...\@arrayright...}`) BEFORE array.sty
336  // has loaded. Without this default, raw-loading ltxutil while in
337  // revtex's class init reads \@arrayright as undefined, defines it
338  // as <ltx:ERROR/>, and the resulting bad <ltx:ERROR/> token propagates
339  // through `\endarray` causing 60GB+ OOM cascades. Witness 2305.18141.
340  Let!("\\@arrayright", "\\@empty");
341  AssignValue!("inPreamble", true); // \begin{document} will clear this.
342
343  DefConstructor!("\\documentclass PackageOptions SkipSpaces ExpandedSemiverbatim []",
344                  "<?latexml class='#2' ?#options(options='#options')?>",
345    properties => sub[args] { Ok(sect05::options_pi_spelling(args[0].as_ref())) },
346    after_digest => sub[whatsit] {
347      // Now that we know we're a LaTeX document, undefine `\magnification`
348      // (babel's plain-TeX vs LaTeX discriminator). Deferred from the
349      // pool-load site above so plain-TeX papers that never run
350      // `\documentclass` keep `\magnification` as a usable register.
351      if let Some(undef_meaning) = lookup_meaning(&T_CS!("\\@undefined")) {
352        assign_meaning(&T_CS!("\\magnification"), undef_meaning, Some(Scope::Global));
353      }
354      // Revert the digested option list rather than `to_string` it: the
355      // digested form drops the braces of a braced key value, so
356      // `thesis={type=dr,dr=rernat}` (DEMO-TUDaPhD) came out as
357      // `thesis=type=dr,dr=rernat` — split at its inner comma and, once
358      // recorded in `\@raw@classoptionslist`, mis-parsed by l3keys.
359      let options: Option<&Digested> = whatsit.get_arg(1);
360      let class_opts = match options {
361        Some(opts) => split_trim_options(&sect05::protected_xdef_options(opts.revert()?)?.untex()),
362        None => Vec::new(),
363      };
364      // Perl LaTeX.pool.ltxml:57 — `$class =~ s/\s+//g;`. Strip ALL
365      // whitespace from the class name before LoadClass. A multi-line
366      // `\documentclass[...]{<newline>revtex4}` makes the Semiverbatim arg
367      // ` revtex4` (the newline right after `{` becomes a leading space
368      // token); unstripped, the name fails to match the `revtex4` binding
369      // registration and falls through to OmniBus — which lacks revtex4's
370      // `\email [] Semiverbatim`, so an `\email{a_b@…}` with `_` then errors
371      // "Script _ can only appear in math mode". Witness 1601.06734.
372      // A lone brace group goes too, as latex.ltx reads the name (`\documentclass{{article}}`;
373      // `sect05::requested_class_name`).
374      let requested = whatsit.get_arg(2).unwrap().to_string();
375      let class_name = sect05::requested_class_name(&requested);
376      sect05::respell_file_argument(whatsit, 2)?;
377      assign_value(CLASS_DECLARED, true, Some(Scope::Global));
378      retract_chapter_before_class_load();
379      let omnibus_before = load_recorded("OmniBus.cls");
380      note_document_class(&class_name);
381      load_class(&class_name,
382                class_opts,
383                Tokens!(T_CS!("\\AtBeginDocument"), T_CS!("\\warn@unusedclassoptions")))?;
384      // A class no binding and no file answered (the OmniBus fallback) is no declaration a later `\documentstyle`
385      // meets as the second (`tex_job.rs` `\lx@documentstyle@impl`; astro-ph9812142).
386      let base = class_name.rsplit('/').next().unwrap_or(&class_name);
387      if !omnibus_before && load_recorded("OmniBus.cls") && base != "OmniBus"
388        && [class_name.as_str(), base].iter().all(|name| {
389          !load_recorded(&s!("{name}.cls"))
390        })
391      {
392        assign_value(CLASS_DECLARED, false, Some(Scope::Global));
393      }

848  // `[SetlengthGlue]`: latex.ltx:9260-9261 `\@newline[#1]` → `\@vspace@calcify{#1}`
849  // = `\setlength\sp@ce@skip{#1}` (9254), so a package's `\setlength` reads it
850  // (bxcalc's `\\[1ex-1ls]`; OXIDIZED_DESIGN #325).
851  DefConstructor!("\\lx@newline OptionalMatch:* [SetlengthGlue]", sub[document, args] {
852    if lookup_bool_sym(pin!("IN_MATH")) {
853      document.insert_element("ltx:XMHint", Vec::new(), Some(map!("name" => s!("newline"))))?;
854    } else {
855      // OXIDIZED_DESIGN surpass-Perl (#722): the optional [Glue] of `\\[20pt]` is the
856      // extra vertical space LaTeX inserts at the break. Perl parses it and drops it
857      // (ltx:break has no spacing slot in the schema). We PRESERVE it as a themeable CSS
858      // custom property `--ltx-break-space` on the break; NO default rule consumes it, so
859      // default rendering is byte-identical to a plain break — a theme (ar5iv) may map it
860      // to margin/padding. Plain `\\` has no [Glue] (arg absent) so it stays attribute-free.
861      // args[1] is the [Glue] (args[0] is the OptionalMatch:* star).
862      let break_attrs = args
863        .get(1)
864        .and_then(|a| a.as_ref())
865        .map(|g| g.to_attribute())
866        .filter(|v| !v.is_empty() && v != "0.0pt" && v != "0pt")
867        .map(|v| map!("cssstyle" => s!("--ltx-break-space:{v}")));
868      if let Some(context) = document.get_element() {
869        let tag = document::get_node_qname(&context);
870        let capture_block = pin!("ltx:_CaptureBlock_");
871        if tag == capture_block {
872          // skip, if in insertBlock — except between a float's panels in a `{center}`, where it ends their row (the
873          // panel grid rule, `note_panel_separator`; 0912.4528 Fig 4, 1206.5015 Fig 3): `\@centercr`, whose `\par`
874          // ends the paragraph (`cr`)
875          note_panel_separator(document, "cr", None)?;
876        } else if tag == pin!("ltx:p") {
877          // Close <p> if parent is _CaptureBlock_
878          if let Some(parent) = context.get_parent() {
879            if document::get_node_qname(&parent) == capture_block {
880              document.maybe_close_element("ltx:p")?;
881            } else if document::can_contain(&context, "ltx:break") {
882              document.insert_element("ltx:break", Vec::new(), break_attrs)?;
883            }
884          }
885        } else if document::can_contain(&context, "ltx:break") {
886          document.insert_element("ltx:break", Vec::new(), break_attrs)?;
887        }
888      }
889      // else: no context => skip
890    }
891  },
892    reversion => Tokens!(T_CS!("\\\\"), T_CR!()),
893    properties => { stored_map!("isBreak" => true) },
894  );
895  Let!("\\\\", "\\lx@newline");
896
897  // latex.ltx:9256 `\DeclareRobustCommand\newline{\@normalcr\relax}`: `\newline` stays the justified break where
898  // `\centering`, `\raggedright` and `{center}` let `\\` to `\@centercr`, so an `\hspace` after it is a line's start,
899  // discarded at the break, never a paragraph's (the panel grid rule's `arrange_panels`, `PANEL_NORMALCR`; review
900  // rev78x4 finding 1: n3 F4, n7 F1). The internal attribute is gone from the markup at the document's end.
901  DefConstructor!("\\newline", "?#isMath(<ltx:XMHint name='newline'/>)(<ltx:break _normalcr='1'/>)",
902    reversion  => Tokens!(T_CS!("\\newline"), T_CR!()),
903    properties => { Ok(stored_map!("isBreak" => true)) },
904  );
905
906  Let!("\\@normalcr", "\\\\");
907  Let!("\\@normalnewline", "\\newline");
908  // latex.ltx:9247-9266: `\@normalcr` → `\@xnewline` → `\@newline[glue]` →
909  // `\@gnewline`, whose raw body (inherited from the dump) dereferences the
910  // scratch `\reserved@e{\reserved@f#1}` that only `\@normalcr` itself sets.
911  // `\\` is native here, but raw classes enter the chain directly —
912  // brief.cls:496 `\@nobreakcr` (`\stopbreaks`, :485) — and a parbox body
913  // holding it is re-expanded during sizing, where the scratch macro is
914  // undefined (ntgclass brief-sample, 2nd letter). Route the chain to the
915  // native newline, as `\\`/`\@normalcr` already are. Guard:
916  // `perfect_kernel_batch56::raw_newline_chain_is_the_native_newline`.
917  DefMacro!("\\@xnewline", "\\@ifnextchar[\\@newline{\\lx@newline}");
918  DefMacro!("\\@newline[]", "\\lx@newline[#1]");
919  DefMacro!("\\@gnewline{}", "\\lx@newline");
920  // NOTE: Activating this binding messes up an \afterassign test,
921  //       so it may be best left disabled.
922  // PushValue!("TEXT_MODE_BINDINGS" => Tokens!(T_CS!("\\\\"), T_CS!("\\@normalcr")));
923
924  def_macro_noop("\\@nolnerr")?;
925  DefMacro!(
926    "\\@centercr",
927    r"\ifhmode\unskip\else\@nolnerr\fi\par\@ifstar{\nobreak\@xcentercr}\@xcentercr"
928  );
929  DefMacro!(
930    "\\@xcentercr",
931    r"\addvspace{-\parskip}\@ifnextchar[\@icentercr\ignorespaces"
932  );
933  DefMacro!("\\@icentercr[]", "\\vskip #1\\ignorespaces");
934
935  Ok(())
936}

938/// Is `cs` (`\document` / `\enddocument`) a macro the DOCUMENT gave it, which
939/// `\begin{document}` / `\end{document}` should run (OXIDIZED_DESIGN_DIVERGENCES
940/// #232)? Three things are NOT that: the binding's own definition (`\ifx`-equal
941/// to the ORIGINAL `\begin{document}`/`\end{document}`, kept as
942/// `\lx@orig@document`/`\lx@orig@enddocument` in `sect02.rs` — docmute and
943/// subfiles redefine the `\end{document}` CS itself to end an `\input` file
944/// while `\enddocument` keeps the original, so the current CS is no yardstick);
945/// the kernel's re-let of `\document` to `\@notprerr` once the document has
946/// begun (latex.ltx:1228 `\@onlypreamble`, sect02.rs), which every nested
947/// `\begin{document}` of an `\input` complete document meets; and a wrapper
948/// whose body contains the control sequence ITSELF — standalone.cls:1073-1083
949/// captures the kernel macro's body with `\toks@\expandafter{\document …}`,
950/// which for this engine's unexpandable alias is the token `\document` (a
951/// self-loop), so such a wrapper keeps the constructor.
952fn is_user_document_macro(cs: &Token, alias: &Token) -> bool {
953  if x_equals(cs, alias) || x_equals(cs, &T_CS!("\\@notprerr")) {
954    return false;
955  }
956  match lookup_definition(cs) {
957    Ok(Some(defn)) if defn.is_expandable() => match defn.get_expansion() {
958      Some(ExpansionBody::Tokens(body)) => !body.unlist_ref().iter().any(|t| t == cs),
959      // A macro with no body at all — `\renewenvironment{document}{…}{}`'s
960      // empty end code — is a user macro too.
961      None => true,
962      Some(ExpansionBody::Closure(_)) => false,
963    },
964    _ => false,
965  }
