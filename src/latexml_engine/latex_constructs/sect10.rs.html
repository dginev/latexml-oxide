403            }

406          },
407          '-' => {
408            // `\cline{-3}` (no leading number) is malformed but appears in

411            num.clear();
412          }
413          c if c.is_ascii_digit() => num.push(c_next),
414          _ => break
415        }
416      }
417      if !num.is_empty() {

546  // full array setup (`\@array`, the internal behind LaTeXML's `\array`), so a
547  // package that builds its own array on `\@tabarray` (t-angles.sty:491
548  // `\def\array{…\@tabarray}`) gets `\@array@bindings` and

551  // but never started the alignment, so nested in an outer array cell under
552  // a `\begingroup` the outer cell's `egroup` met the non-boxing frame
553  // ("`\lx@begin@alignment` Attempt to close boxing group"; t-angles/t-manual
554  // 101 errors, pdflatex clean). Guard:
555  // `perfect_kernel_batch54::tabarray_is_the_full_array_setup`.
556  // `\@array` is the latex.ltx INTERNAL the kernel's own callers use (a
557  // package may redefine the user `\array` on top of `\@tabarray`, as
558  // t-angles does — routing through `\array` would recurse).
559  // The box the array lives in follows `\@classz` too: `\@tabclassz` cells
560  // are text, so the container is the tabular one (`\@@tabular`, mode
561  // restricted_horizontal — the `\vcenter` of latex.ltx:16565 leaves the
562  // enclosing math); `\@arrayclassz` keeps `\@@array` (math cells).
563  DefMacro!(
564    "\\@array[]{}",
565    r"\let\lx@array@text@close\relax\@array@bindings[#1]{#2}\ifx\@classz\@tabclassz\expandafter\@@tabular\else\expandafter\@@array\fi[#1]{#2}\lx@begin@alignment"
566  );
567  DefMacro!("\\@tabarray", r"\m@th\@ifnextchar[\@array{\@array[c]}");
568
569  Ok(())

580/// The math-cell alignment `\@array@bindings` opens (Perl latex_constructs.pool.ltxml:3792-3809) for a template its
581/// caller built: `\array`'s from its column letters, IEEEtrantools' `\IEEEeqnarraybox` from the IEEEtrantools column
582/// types (`ieeetrantools_sty.rs`).
583/// `omit_is_text`: the cells' math is their template's `$` (latex.ltx `\@arrayclassz` `$\relax\@sharp$`), so an
584/// `\omit`ted cell is text (tex.web §789; tex_tables.rs `omit_cell_is_text`).
585pub fn math_array_bindings(
586  pos: Option<&Tokens>,
587  template: Template,
588  omit_is_text: bool,
589) -> Result<()> {
590  let mut attrs = HashMap::default();
591  attrs.insert(String::from("vattach"), array_attachment(pos));
592  attrs.insert(String::from("role"), String::from("ARRAY"));
593  // Determine column and row separations, if non default
594  let colsep = lookup_dimension("\\arraycolsep");
595  if let Some(sep) = colsep
596    && sep.value_of()
597      != lookup_dimension("\\lx@default@arraycolsep")
598        .unwrap_or_default()
599        .value_of()
600  {
601    attrs.insert(String::from("colsep"), sep.to_attribute());
602  }
603  let astr = do_expand(T_CS!("\\arraystretch"))?.to_string();
604  if astr != "1"
605    && let Ok(astr_f) = astr.parse::<f64>()
606    && astr_f != 1.0
607  {
608    let rowsep = Dimension::from_str(&s!("{}em", astr_f - 1.0))?;
609    attrs.insert(String::from("rowsep"), rowsep.to_attribute());
610  }
611  let mut properties = SymHashMap::default();
612  if array_zeroes_interline() {
613    properties.insert("zero_interline", Stored::Bool(true));
614  }
615  if let Some(strut) = array_strut() {
616    properties.insert("strut", strut);
617  }
618  // latex.ltx `\@arrayclassz`: each cell is `$\relax\@sharp$`, its math the template's, so an `\omit`ted cell is
619  // text (tex.web §789; tex_tables.rs `omit_cell_is_text`). Rust-only key: LaTeXML's array cells are math by the
620  // alignment's mode.
621  if omit_is_text {
622    properties.insert("math_cells_template", Stored::Bool(true));
623  }
624  alignment_bindings(template, String::from("math"), properties, attrs);
625  // Perl: if display math, switch to text mathstyle
626  if lookup_string_from_sym(pin!("MODE")).ends_with("math") {
627    MergeFont!(mathstyle => "text");
628  }
629  Let!("\\\\", "\\lx@alignment@newline");
630  // latex.ltx:16576 `\let\tabularnewline\\` in `\@array` — for `array` as
631  // for `tabular` (the text binding does it). Without it a column template
632  // that re-lets `\\` after opening a box (tabvar.sty:118 `>{\begin{varwidth}
633  // …\let\\=\TVtabularnewline $}` with `\TVtabularnewline` → `\tabularnewline`)
634  // got latex.ltx's top-level `\relax`, the row break vanished inside the
635  // last cell's box and every later `&` was an "Extra alignment tab" (tabvar
636  // demo ×80; KPE #192). Guard:
637  // `perfect_kernel_batch54::math_array_lets_tabularnewline_to_the_row_break`.
638  Let!("\\tabularnewline", "\\\\");
639  Let!("\\lx@intercol", "\\lx@math@intercol");
640  Ok(())
641}
