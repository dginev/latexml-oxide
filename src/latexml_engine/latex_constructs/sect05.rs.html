15/// latex.ltx `\@parse@version`: `YYYY-MM-DD` / `YYYY/MM/DD` → `YYYYMMDD`, 0 if unparsable.
16fn parse_version_date(s: &str) -> i64 {
17  let parts: Vec<i64> = s
18    .split(['-', '/'])
19    .filter_map(|p| p.trim().parse::<i64>().ok())
20    .collect();
21  match parts.as_slice() {
22    [y, m, d, ..] => y * 10000 + m * 100 + d,
23    [n] if *n > 10000 => *n,
24    _ => 0,
25  }
26}

43/// latex.ltx:19133-19171 `\pkgcls@parse@date@arg`. A leading `=` is a
44/// rollback request: its suffix is either a release label or a date. A bare
45/// date is only a minimum-version check and does not select an old file.
46fn pkgcls_set_request(request: Option<&str>) {
47  pkgcls_clear_request();
48  let Some(target) = request.and_then(|s| s.trim().strip_prefix('=')) else {
49    return;
50  };
51  let target = target.trim();
52  let date = parse_version_date(target);
53  if date > 0 {
54    assign_value("pkgcls@targetdate", date, Some(Scope::Global));
55  } else if !target.is_empty() {
56    assign_value("pkgcls@targetdate", 1i64, Some(Scope::Global));
57    assign_value(
58      "pkgcls@targetlabel",
59      target.to_string(),
60      Some(Scope::Global),
61    );
62  }
63}
64fn require_package_with_rollback(
65  package: &str,
66  options: Vec<String>,
67  request: Option<&str>,
68) -> Result<()> {
69  pkgcls_set_request(request);
70  let result = require_package(package, RequireOptions {
71    options,
72    ..RequireOptions::default()
73  });
74  pkgcls_clear_request();
75  result
76}
77/// latex.ltx:19216-19222 `\pkgcls@use@this@release`: load the chosen release
78/// and abort the rest of the current file. `\@@input` reads that very file, so
79/// it is an INTERPRETABLE source (Perl Package.pm:2124): the loader must not
80/// version-strip `parskip-2001-04-09.sty` back to the parskip binding it is
81/// loading (parskip.sty:41 `\DeclareRelease{v1}{2001-04-09}{…}`), which warned
82/// `recursion:parskip.sty` twice and loaded neither release
83/// (biblatex-sbl-ibid, biblatex-sbl-examples `\usepackage{parskip}[=v1]`).
84/// Guard: `sweep125_roots::rollback_release_file_is_read`.
85fn pkgcls_use_this_release(file: &str) -> Result<()> {
86  assign_value("pkgcls@targetdate", 0i64, Some(Scope::Global));
87  AssignMapping!("INTERPRETABLE_SOURCES", file => 1);
88  let mut toks = vec![T_CS!("\\@addtofilelist"), T_BEGIN!()];
89  toks.extend(ExplodeText!(file));
90  toks.push(T_END!());
91  toks.push(T_CS!("\\@@input"));
92  toks.extend(ExplodeText!(file));
93  toks.push(T_CS!("\\relax"));
94  toks.push(T_CS!("\\endinput"));
95  unread(Tokens::new(toks));
96  Ok(())
97}

2396                level -= 1;
2397              }
2398              level == 0
2399            })
2400          })
2401          .flatten();
2402        if let Some(end) = end
2403          && !expanding
2404          && let Some(StoredDefinition { past, .. }) = stored_definition(body, t, end)
2405        {
2406          if !vocabulary_is_defined(&body[k + 1..end], what, definees, depth) {
2407            return false;
2408          }
2409          i = past;
2410        }
2411        continue;
2412      }
2413      // `\def\cs<parameter text>{<replacement text>}`: the parameter text is never run, and only `\edef`/`\xdef`
2414      // expand the replacement text as they define (tex.web §473, §1218); the others store it, to run where the
2415      // body calls `\cs` (article.cls:206's `\def\@makefnmark{…\@thefnmark}`, copied into sig-alternate-05-2015.cls:
2416      // 726-730's `\maketitle` and run only by its footnotes, where latex.ltx defines `\@thefnmark`: 1505.04771).
2417      // `\newcommand{\cs}[n][default]{…}` stores the default and the body alike.
2418      if let Some(StoredDefinition {
2419        replacement,
2420        params,
2421        past,
2422        open,
2423      }) = stored_definition(body, t, at)
2424      {
2425        let unexpanded = !t.with_str(|s| s == "\\edef" || s == "\\xdef");
2426        definees.push(BodyDefinee {
2427          cs: definee,
2428          replacement: unexpanded.then_some(replacement),
2429          params,
2430          checked: !unexpanded,
2431        });
2432        // an `\edef`'s replacement text is expanded as it is defined: checked here, past the parameter text
2433        i = if t.with_str(|s| s == "\\edef" || s == "\\xdef") {
2434          open
2435        } else {
2436          past
2437        };
2438        continue;
2439      }
2440      definees.push(BodyDefinee {
2441        cs:          definee,
2442        replacement: None,
2443        params:      None,
2444        checked:     true,
2445      });
2446    }
2447    if let Some(k) = definees.iter().rposition(|d| d.cs == *t) {
2448      if let Some(params) = definees[k].params.clone() {
2449        delimiters.extend(delimiter_positions(body, i, &params));
2450      }
2451      if !definees[k].checked {
2452        definees[k].checked = true;
2453        if let Some(replacement) = definees[k].replacement.clone() {
2454          // a body whose own definitions call one another past the bound is not read to the end: refused, not admitted
2455          // unread
2456          if depth >= GATE_DEFINITION_DEPTH {
2457            Info!(
2458              "ignore",
2459              what,
2460              s!(
2461                "The {what} body was not replayed: its definitions call one another more than {GATE_DEFINITION_DEPTH} deep"
2462              )
2463            );
2464            return false;
2465          }
2466          if !vocabulary_is_defined(&replacement, what, definees, depth + 1) {
2467            return false;
2468          }
2469        }
2470      }
2471      continue;
2472    }
2473    let Some(meaning) = lookup_meaning(t) else {
2474      Info!(
2475        "ignore",
2476        what,
2477        s!(
2478          "The {what} body was not replayed: it uses {}, which is undefined here",
2479          t.to_string()
2480        )
2481      );
2482      return false;
2483    };
2484    if let Stored::Expandable(ref d) = meaning {
2485      if let Some(params) = d.get_parameters() {
2486        delimiters.extend(delimiter_positions(body, i, &GateParam::of(params)));
2487      }
2488      // What a macro the body calls defines is defined after the call: nowfnt.cls:552-575's `\nowfnt@splitauthor`
2489      // `\def`s the `\nowfnt@lastname` its `\author` (:581-600) then reads (1912.00312). Only a control sequence
2490      // undefined here, so a real meaning is never shadowed ([`callee_definees`]).
2491      if let Some(ExpansionBody::Tokens(callee)) = d.get_expansion() {
2492        for n in callee_definees(callee.unlist_ref()) {
2493          if !definees.iter().any(|d| d.cs == n) {
2494            definees.push(BodyDefinee {
2495              cs:          n,
2496              replacement: None,
2497              params:      None,
2498              checked:     true,
2499            });
2500          }
2501        }
2502      }
2503    }
2504    let own_hold = lookup_bool(&s!("replay_gate_skips_arguments:{}", t.to_string()));
2505    // An empty expansion is stored as `None`.
2506    let params = match meaning {
2507      Stored::Expandable(ref d)
2508        if own_hold
2509          || d
2510            .get_expansion()
2511            .is_none_or(|e| matches!(e, ExpansionBody::Tokens(x) if x.unlist_ref().is_empty())) =>
2512      {
2513        d.get_parameters().cloned()
2514      },
2515      Stored::Primitive(ref d) if own_hold => d.get_parameters().cloned(),
2516      _ => None,
2517    };
2518    if let Some(params) = params {
2519      for p in params.get_parameters() {
2520        i = skip_no_op_argument(body, i, p);
2521      }
2522    }
2523  }
2524  true
2525}

2528/// primitive that reads them unexpanded: `\ifx`'s two comparands (tex.web §507, `get_next`), e-TeX's `\ifdefined`'s

2530/// sequence there is none TeX meets: zaa.cls:166's `\ifx\@dedication\undefined` (1902.07328) and cms-tdr.cls:786's
2531/// `\ifx\cmsMessage\undefined` (2307.11186, 2507.13469) refused their `\maketitle` bodies, and the addresses they
2532/// print were lost. A parameter (`#1`) is one comparand. After `\expandafter\ifx`, a `\csname…\endcsname` first
2533/// comparand is made as the condition runs (its name tokens run, checked as any others) and the token after its
2534/// `\endcsname` is the second; any other token there is expanded once (§367) and nothing is skipped. `\noexpand`'s
2535/// operand is checked: in an `\edef`/`\xdef` replacement text (its main use) it is stored and runs later.
2536fn unrun_operands(body: &[Token], at: usize) -> Vec<usize> {
2537  let t = &body[at];
2538  let count = if *t == T_CS!("\\ifx") {
2539    2
2540  } else if [T_CS!("\\ifdefined"), T_CS!("\\string"), T_CS!("\\meaning")].contains(t) {
2541    1
2542  } else {
2543    return Vec::new();
2544  };
2545  // the positions one operand from `k` takes: a parameter's `#`s and the token after them, or one token

2547    let mut end = k;
2548    while body
2549      .get(end)
2550      .is_some_and(|u| u.get_catcode() == Catcode::PARAM)
2551    {
2552      end += 1;
2553    }
2554    k..(end + 1).min(body.len())
2555  };
2556  let mut k = at + 1;
2557  if count == 2 && at > 0 && body[at - 1] == T_CS!("\\expandafter") {
2558    if body.get(k) != Some(&T_CS!("\\csname")) {
2559      return Vec::new();
2560    }
2561    let mut level = 0usize;
2562    let Some(end) = (k..body.len()).find(|&e| {

2565      } else if body[e] == T_CS!("\\endcsname") {
2566        level -= 1;
2567      }
2568      level == 0
2569    }) else {
2570      return Vec::new();
2571    };
2572    return operand(end + 1).collect();
2573  }
2574  let mut out = Vec::new();

2579  }

2595/// After a definer `t` whose definee is `body[at]`: the replacement text a `\def`-family or `\newcommand`-family
2596/// definer stores (a `\newcommand`'s default of its optional argument with it), the parameters a `\def`'s parameter
2597/// text declares, and the indices past the definition and of its replacement text's opening brace. `None` for any
2598/// other definer, or a definition this cannot read (no braced replacement text).
2599fn stored_definition(body: &[Token], t: &Token, at: usize) -> Option<StoredDefinition> {
2600  // (a name made, or a replacement text expanded, as the definition runs is read in place by the caller)
2601  if body[at] == T_CS!("\\csname") || body[at] == T_CS!("\\expandafter") {
2602    return None;
2603  }
2604  let group_past = |open: usize| -> Option<usize> {
2605    let mut level = 0usize;
2606    for (k, u) in body.iter().enumerate().skip(open) {
2607      match u.get_catcode() {
2608        Catcode::BEGIN => level += 1,
2609        Catcode::END => {
2610          level = level.checked_sub(1)?;
2611          if level == 0 {
2612            return Some(k + 1);
2613          }
2614        },
2615        _ => {},
2616      }
2617    }
2618    None
2619  };
2620  let name = t.to_string();
2621  if ["\\def", "\\gdef", "\\edef", "\\xdef"].contains(&name.as_str()) {
2622    // the parameter text runs up to the first `{` (tex.web §473-476)
2623    let open = (at + 1..body.len()).find(|&k| body[k].get_catcode() == Catcode::BEGIN)?;
2624    let past = group_past(open)?;
2625    let params = gate_parameters(&body[at + 1..open]);
2626    return Some(StoredDefinition {
2627      replacement: body[open + 1..past - 1].to_vec(),
2628      params,
2629      past,
2630      open,
2631    });
2632  }
2633  if ![
2634    "\\newcommand",
2635    "\\renewcommand",
2636    "\\providecommand",
2637    "\\DeclareRobustCommand",
2638  ]
2639  .contains(&name.as_str())
2640  {
2641    return None;
2642  }
2643  let skip_spaces = |mut k: usize| {
2644    while body
2645      .get(k)
2646      .is_some_and(|u| u.get_catcode() == Catcode::SPACE)
2647    {
2648      k += 1;
2649    }
2650    k
2651  };
2652  // past `\cs` or `{\cs}`
2653  let mut k = skip_spaces(at + 1);
2654  if body.get(k).is_some_and(|u| u.get_catcode() == Catcode::END) {
2655    k = skip_spaces(k + 1);
2656  }
2657  let mut replacement = Vec::new();
2658  // `[n]`, then `[default]`
2659  for n in 0..2 {
2660    if body.get(k) != Some(&T_OTHER!("[")) {
2661      break;
2662    }
2663    let close = (k + 1..body.len()).find(|&j| body[j] == T_OTHER!("]"))?;
2664    if n == 1 {
2665      replacement.extend_from_slice(&body[k + 1..close]);
2666    }
2667    k = skip_spaces(close + 1);
2668  }
2669  if !body
2670    .get(k)
2671    .is_some_and(|u| u.get_catcode() == Catcode::BEGIN)
2672  {
2673    return None;
2674  }
2675  let past = group_past(k)?;
2676  replacement.extend_from_slice(&body[k + 1..past - 1]);
2677  Some(StoredDefinition {
2678    replacement,
2679    params: None,
2680    past,
2681    open: k,
2682  })
2683}

2685/// The positions in `body` from `i` on of the delimiters a call of a macro with `params` matches: a `Match`
2686/// parameter's literal tokens and an `Until` parameter's delimiter, found as TeX finds them (tex.web §392-399: a
2687/// delimited argument runs to the first occurrence of its delimiter outside braces). TeX never runs these tokens —
2688/// the macro's parameter text consumes them (nowfnt.cls:552's `\nowfnt@splitauthor#1,#2,#3,#4\@nil`, called as
2689/// `\nowfnt@splitauthor#2,!,!,\@nil` by its `\author`: 1912.00312). The arguments themselves are checked as before.
2690/// Matching stops at the first parameter of another kind, or at one that does not match.
2691fn delimiter_positions(body: &[Token], i: usize, params: &[GateParam]) -> Vec<usize> {
2692  let mut out = Vec::new();
2693  if !params
2694    .iter()
2695    .any(|p| matches!(p, GateParam::Until(_) | GateParam::Match(_)))
2696  {
2697    return out;
2698  }
2699  let mut k = i;
2700  for p in params {
2701    if let GateParam::Match(lit) = p {
2702      if lit.is_empty() || !body[k.min(body.len())..].starts_with(lit) {
2703        break;
2704      }
2705      out.extend(k..k + lit.len());
2706      k += lit.len();
2707    } else if let GateParam::Until(delim) = p {
2708      let mut level = 0usize;
2709      let mut found = None;
2710      for j in k..body.len() {
2711        if level == 0 && !delim.is_empty() && body[j..].starts_with(delim) {
2712          found = Some(j);
2713          break;
2714        }
2715        match body[j].get_catcode() {
2716          Catcode::BEGIN => level += 1,
2717          Catcode::END => match level.checked_sub(1) {
2718            Some(l) => level = l,
2719            None => break,
2720          },
2721          _ => {},
2722        }
2723      }
2724      let Some(j) = found else { break };
2725      out.extend(j..j + delim.len());
2726      k = j + delim.len();
2727    } else if matches!(p, GateParam::Plain) {
2728      while body
2729        .get(k)
2730        .is_some_and(|u| u.get_catcode() == Catcode::SPACE)
2731      {
2732        k += 1;
2733      }
2734      match body.get(k) {
2735        Some(u) if u.get_catcode() == Catcode::BEGIN => {
2736          let mut level = 0usize;
2737          let mut past = None;
2738          for (j, v) in body.iter().enumerate().skip(k) {
2739            match v.get_catcode() {
2740              Catcode::BEGIN => level += 1,
2741              Catcode::END => {
2742                level -= 1;
2743                if level == 0 {
2744                  past = Some(j + 1);
2745                  break;
2746                }
2747              },
2748              _ => {},
2749            }
2750          }
2751          let Some(past) = past else { break };
2752          k = past;
2753        },
2754        Some(_) => k += 1,
2755        None => break,
2756      }
2757    } else {
2758      break;
2759    }
2760  }
2761  out
2762}

2768/// The parameters a `\def`'s parameter text `text` declares, read as tex.web §473-476 reads them (as
2769/// `parse_def_parameters` does, which reports an ill-formed text, so the gate never calls it): `None` for a text it
2770/// cannot know before the body runs — one holding an argument of the enclosing macro (an `ARG` token, `#1` of a
2771/// `\renewcommand{\author}[1]{\expandafter\def\csname…#1\endcsname…}`), a doubled `#`, or parameters out of order.
2772/// The definition is still stored; its call's arguments are then checked as tokens, which errs toward not replaying.
2773fn gate_parameters(text: &[Token]) -> Option<Vec<GateParam>> {
2774  let is_param = |t: &Token| {
2775    t.get_catcode() == Catcode::PARAM
2776      || (matches!(t.get_catcode(), Catcode::CS | Catcode::ACTIVE)
2777        && matches!(lookup_meaning(t), Some(Stored::Token(l)) if l.get_catcode() == Catcode::PARAM))
2778  };
2779  let ends_text = |t: &Token| t.get_catcode() == Catcode::ARG || is_param(t);
2780  let mut out = Vec::new();
2781  let mut n = 0u8;
2782  let mut k = 0;
2783  while k < text.len() {
2784    let t = &text[k];
2785    if t.get_catcode() == Catcode::ARG {
2786      return None;
2787    }
2788    if !is_param(t) {
2789      // the literal text a call must open with
2790      let start = k;
2791      while k < text.len() && !ends_text(&text[k]) {
2792        k += 1;
2793      }
2794      out.push(GateParam::Match(text[start..k].to_vec()));
2795      continue;
2796    }
2797    let Some(next) = text.get(k + 1) else {
2798      // `#{`: the last parameter is delimited by the replacement text's brace
2799      out.push(GateParam::Other);
2800      break;
2801    };
2802    if next.get_catcode() != Catcode::OTHER
2803      || next.with_str(|d| d.parse::<u8>().ok()) != Some(n + 1)
2804    {
2805      return None;
2806    }
2807    n += 1;
2808    k += 2;
2809    let start = k;
2810    while k < text.len() && !ends_text(&text[k]) {
2811      k += 1;
2812    }
2813    if k > start {
2814      out.push(GateParam::Until(text[start..k].to_vec()));
2815    } else if k + 1 == text.len() && is_param(&text[k]) {
2816      // `#n#`: delimited by the replacement text's brace
2817      out.push(GateParam::Other);
2818      k += 1;
2819    } else {
2820      out.push(GateParam::Plain);
2821    }
2822  }
2823  Some(out)
2824}

2922/// `body` without the dead branch of each `\newif` switch, keeping the live one (itself filtered); an
2923/// unbalanced switch is kept as written, so the gate still checks everything after it. `\unless` inverts
2924/// the switch it prefixes. A switch is checked on both branches when it is data, not a test — an
2925/// assignment's operand (`\let\ifX\iftrue`, `\newif\ifX`) — or when the body sets it before testing it
2926/// (`\Xtrue … \ifX`): its state at deposit time is not the one the body tests.
2927fn live_branches(body: &[Token]) -> Vec<Token> {
2928  let mut out = Vec::with_capacity(body.len());
2929  let mut i = 0;
2930  while i < body.len() {
2931    let t = body[i];
2932    i += 1;
2933    let Some(mut state) = switch_state(&t) else {
2934      out.push(t);
2935      continue;
2936    };
2937    if switch_is_operand(&out) || switch_set_before(&t, &out) {
2938      out.push(t);
2939      continue;
2940    }
2941    // the matching \else and \fi, through nested conditionals (`\unless` prefixes an `\if…`, which counts)
2942    let (mut depth, mut else_at, mut fi_at) = (0usize, None, None);
2943    for (k, u) in body.iter().enumerate().skip(i) {
2944      match lookup_conditional(u) {
2945        Some(ConditionalType::If) => depth += 1,
2946        Some(ConditionalType::Else) if depth == 0 && else_at.is_none() => else_at = Some(k),

4291  let author = author.to_string();
4292  let holds_author = |span: &[Token]| {
4293    span
4294      .iter()
4295      .any(|n| n.get_catcode() == Catcode::ARG && n.with_str(|s| s == author))
4296  };
4297  // Only furniture the body itself runs counts: at its top brace level (not inside another macro's argument,
4298  // `\@gobble{\lfoot{#1}}`) and outside any conditional (`\if@authfoot\lfoot{#1}\fi`, `\iffalse…\fi`; a
4299  // conditional known by its meaning, a `\newif` flag included). Fail-safe: an empty-check
4300  // `\ifx\relax#1\relax\else\lfoot{#1}\fi` is dropped too, as the base drops it.
4301  // A group `\begingroup…\endgroup`/`\bgroup…\egroup` (known by meaning, as a `\let` alias) counts as a brace level: a
4302  // local furniture setting there is discarded at its end (fancyhdr.sty:34, :50-51 `\lfoot` is local). `\unless` is a
4303  // prefix of the conditional it negates, not one of its own.
4304  let begingroup = lookup_meaning(&T_CS!("\\begingroup"));
4305  let endgroup = lookup_meaning(&T_CS!("\\endgroup"));
4306  let group_change = |t: &Token| -> i32 {
4307    match lookup_meaning(t) {
4308      Some(Stored::Token(ref x)) if x.get_catcode() == Catcode::BEGIN => 1,
4309      Some(Stored::Token(ref x)) if x.get_catcode() == Catcode::END => -1,
4310      m if m.is_some() && same_meaning(m.as_ref(), begingroup.as_ref()) => 1,
4311      m if m.is_some() && same_meaning(m.as_ref(), endgroup.as_ref()) => -1,
4312      _ => 0,
4313    }
4314  };
4315  let ends_conditional = |t: &Token| t.with_cs_name(|n| n == "\\fi");
4316  let inside_conditional = |t: &Token| t.with_cs_name(|n| n == "\\else" || n == "\\or");
4317  let opens_conditional = |t: &Token| {
4318    !ends_conditional(t)
4319      && !inside_conditional(t)
4320      && !t.with_cs_name(|n| n == "\\unless")
4321      && matches!(lookup_meaning(t), Some(Stored::Conditional(_)))
4322  };
4323  // A `\fi`, `\else` or `\or` with no conditional known to be open closes one opened by something not recognised
4324  // (`\def\ifauthfoot{\iffalse}` then `\ifauthfoot\lfoot{#1}\fi`): no unit author can be read from the body.
4325  let mut depth = 0usize;
4326  for t in body.iter().filter(|t| t.get_catcode() == Catcode::CS) {
4327    if ends_conditional(t) || inside_conditional(t) {
4328      if depth == 0 {
4329        return false;
4330      }
4331      if ends_conditional(t) {
4332        depth -= 1;
4333      }
4334    } else if opens_conditional(t) {
4335      depth += 1;
4336    }
4337  }
4338  let mut braces = 0usize;
4339  let mut conditionals = 0usize;
4340  for (i, t) in body.iter().enumerate() {
4341    match t.get_catcode() {
4342      Catcode::BEGIN => braces += 1,
4343      Catcode::END => braces = braces.saturating_sub(1),
4344      Catcode::CS => {
4345        if ends_conditional(t) {
4346          conditionals = conditionals.saturating_sub(1);
4347        } else if opens_conditional(t) {
4348          conditionals += 1;
4349        } else {
4350          match group_change(t) {
4351            1 => braces += 1,
4352            -1 => braces = braces.saturating_sub(1),
4353            _ => {},
4354          }
4355        }
4356      },
4357      _ => {},
4358    }
4359    if braces > 0 || conditionals > 0 {
4360      continue;
4361    }
4362    if named(t, PAGE_FURNITURE) {
4363      // a call: its optional `[…]` and up to two brace groups (`\markboth{…}{…}`)
4364      let mut at = i + 1;
4365      let mut groups = 0;
4366      while let Some(n) = body.get(at) {
4367        if n.get_catcode() == Catcode::SPACE {
4368          at += 1;
4369        } else if n.get_catcode() == Catcode::OTHER && n.with_str(|s| s == "[") {
4370          at += body[at..]
4371            .iter()
4372            .position(|c| c.with_str(|s| s == "]"))
4373            .unwrap_or(body.len() - at)
4374            + 1;
4375        } else if n.get_catcode() == Catcode::BEGIN && groups < 2 {
4376          let Some((open, close)) = group_end(at) else {
4377            break;
4378          };
4379          if holds_author(&body[open..=close]) {
4380            return true;
4381          }
4382          groups += 1;
4383          at = close + 1;
4384        } else {
4385          break;
4386        }
4387      }
4388    } else if named(t, BODY_DEFINERS_FOR_FURNITURE)
4389      && let Some(target) = body[i + 1..]
4390        .iter()
4391        .position(|n| is_cs(n) && !named(n, BODY_DEFINERS_FOR_FURNITURE))
4392      && named(&body[i + 1 + target], PAGE_FURNITURE)
4393      && let Some((open, close)) = group_end(i + 2 + target)
4394      && holds_author(&body[open..=close])
4395    {
4396      return true;
4397    }
4398  }
4399  false
4400}

4607      *i -= 1;

4615    return false;
4616  };

4621      while level > 0 {

4645    };

4698/// Whether code `tokens` calls a macro whose parameter text has a delimiter that does not occur later in the code, so
4699/// the call reads on past the code's end: wds10.sty:527-531's `\abstract` ends `\pickup\ignorespaces`, and `\pickup`

4702/// (`\edef\x{\expandafter\strip@prefix\meaning\relax}`). A definition's operands are no calls: the target and parameter
4703/// text of `\def`/`\gdef`/`\edef`/`\xdef` (`\def\pickup#1\end#2{…}` in the code), both of `\let`'s (`\let\saved\pickup`),
4704/// `\futurelet`'s target and the command a `\newcommand`/`\renewcommand`/`\providecommand`/`\DeclareRobustCommand`
4705/// defines. Nor is a call whose argument the code's expansion produces: one after `\expandafter`

4707/// expansion is text the code does not show (`\strip@prefix`'s `>` comes from `\meaning`).

4721    "\\the",

4739      .get(i)

4755      || catcode == Catcode::BEGIN
4756      || catcode == Catcode::END
4757      || catcode == Catcode::SPACE

4759      i += 1;

4791        .get(i)
4792        .is_some_and(|t| t.get_catcode() == Catcode::OTHER && t.to_string() == "*")

4794        i = next_operand(i + 1);

4798        .is_some_and(|t| t.get_catcode().is_active_or_cs())

5350pub(crate) type RunReader = (fn(&Digested, &mut String), Option<char>);

5352/// The speculative run ([`title_page_probe_run`]), its material read by each of `readers` in turn: their texts when it
5353/// printed, else how it ended. It reads `budget` tokens at most (`gullet::read_budget`): past them it is abandoned and
5354/// rolled back as a failed run is, its diagnostics discarded, and gives [`SpeculativeRun::OverBudget`] — never a Fatal,
5355/// even when its assignments could not be undone —, counted as having read what it read, its conditionals by no limit.
5356fn speculative_run_read(
5357  tokens: Tokens,
5358  isolated: bool,
5359  readers: &[RunReader],
5360  budget: usize,
5361) -> Result<std::result::Result<Vec<String>, SpeculativeRun>> {
5362  let Some(journal) = begin_assignment_journal() else {
5363    return Ok(Err(SpeculativeRun::Unavailable));
5364  };
5365  let _speculative = speculative_digestion();
5366  const FLAGS: [&str; 3] = [
5367    "lx_replaying_class_setter",
5368    "lx_depositing_frontmatter_fields",
5369    "lx_depositing_class_maketitle",
5370  ];
5371  let saved: Vec<bool> = FLAGS.iter().map(|flag| lookup_bool(flag)).collect();
5372  for flag in FLAGS {
5373    assign_value(flag, true, Some(Scope::Global));
5374  }
5375  let hold = util::logger::DiagnosticsHold::begin();
5376  let resource_fatal_before = resource_fatal_latched();
5377  // (and its terminal output — a class's `\typeout{Warning: Missing \institute}`, an.cls's `\ch@ckobl` — is not the
5378  // document's: the class's own run of it, the deposit or the replay, writes it once; put back however the run ends)
5379  let muted = SuppressLogOutputGuard::new(true);
5380  let depth = get_frame_depth();
5381  let boxes = box_lists_mark();
5382  // (the tokens being invoked and the conditionals as they are, put back however the run ends: `StomachMark`)
5383  let stomach_mark = StomachMark::take();
5384  // (what an `\insert` in it sets is stashed for after the title page, tex_inserts.rs — printed by it all the same, at
5385  // the page foot: its text follows the run's, so a store and the label the class prints before it there are printed,
5386  // as they are inline in Perl, and taken out of the deposit with it, `note_store_handed`)
5387  let outer_inserts = remove_value("lx_deposit_inserts");
5388  assign_value(
5389    "lx_deposit_inserts",
5390    Stored::VecDigested(Vec::new()),
5391    Some(Scope::Global),
5392  );
5393  let conditionals = if_count();
5394  let budget = read_budget(budget);
5395  let result = if isolated {
5396    reading_from_mouth(Mouth::default(), || {
5397      with_origin(DefinitionOrigin::File, || digest(tokens))
5398    })
5399  } else {
5400    with_origin(DefinitionOrigin::File, || digest(tokens))
5401  };
5402  // (off before anything else reads: closing what the run left open is no part of it)
5403  let end = budget.end();
5404  note_check_run_reads(end.reads);
5405  let over_budget = end.over.then_some(end.reads);
5406  if over_budget.is_some() {
5407    set_if_count(conditionals);
5408  }
5409  let inserts = take_deposit_inserts();
5410  let unwound = reading_from_mouth(Mouth::default(), || {
5411    new_local_box_list();
5412    while get_frame_depth() > depth {
5413      let before = get_frame_depth();
5414      let nobox = lookup_bool("groupNonBoxing");
5415      if pop_stack_frame(nobox).is_err() || get_frame_depth() >= before {
5416        break;
5417      }
5418    }
5419    expire_local_box_list();
5420    Ok(())
5421  });
5422  stomach_mark.restore();
5423  let text = match &result {
5424    Ok(digested) => Some(
5425      readers
5426        .iter()
5427        .map(|(read, insert_mark)| {
5428          let mut text = String::new();
5429          read(digested, &mut text);
5430          for insert in &inserts {
5431            text.extend(*insert_mark);
5432            read(insert, &mut text);
5433          }
5434          text
5435        })
5436        .collect::<Vec<_>>(),
5437    ),
5438    Err(_) => None,
5439  };
5440  drop(muted);
5441  let undone = journal.rollback();
5442  remove_value("lx_deposit_inserts");
5443  if let Some(outer) = outer_inserts {
5444    assign_value("lx_deposit_inserts", outer, Some(Scope::Global));
5445  }
5446  for (flag, value) in FLAGS.iter().zip(saved) {
5447    assign_value(flag, value, Some(Scope::Global));
5448  }
5449  if let Some(reads) = over_budget {
5450    if undone {
5451      hold.discard();
5452    } else {
5453      hold.commit();
5454      Info!(
5455        "frontmatter",
5456        "\\maketitle",
5457        "A speculative run the title-page check abandoned could not be undone (its groups changed): its diagnostics and its global assignments stay"
5458      );
5459    }
5460    unwound?;
5461    unwind_box_lists_to(boxes);
5462    return Ok(Err(SpeculativeRun::OverBudget(reads)));
5463  }
5464  if let Err(err) = result {
5465    // A resource Fatal (the conversion's deadline, its memory budget, a runaway limit) is the document's, and so is
5466    // one whose run could not be undone. Any other ends only this run, which the document never makes: a value only
5467    // the sentinels give may raise it (a letter where the class's code expects a number). It is discarded with the
5468    // run's diagnostics, and the run printed nothing — no store counts as printed by it. A runaway limit (the
5469    // pushback, a recursion) is the run's own, as the real `\maketitle` never runs this
5470    // code: VisionStyle.cls's title page loops on `\title{X \small Y}`, in pdflatex too (astro-ph0203021; user ruling
5471    // 2026-10-19).
5472    if !undone || (resource_fatal_latched() && !resource_fatal_before && !unlatch_runaway_fatal()) {
5473      hold.commit();
5474      return Err(err);
5475    }
5476    hold.discard();
5477    unwound?;
5478    // (and the box lists the failed digestion left open)
5479    unwind_box_lists_to(boxes);
5480    Info!(
5481      "frontmatter",
5482      "\\maketitle",
5483      "A speculative run of the class's title page ended in a Fatal error, discarded with it: no store counts as printed by it"
5484    );
5485    return Ok(Err(SpeculativeRun::Aborted));
5486  }
5487  if undone {
5488    hold.discard();
5489  } else {
5490    hold.commit();
5491    Info!(
5492      "frontmatter",
5493      "\\maketitle",
5494      "A speculative run of the class's title page could not be undone (its groups changed): its diagnostics and its global assignments stay"
5495    );
5496  }
5497  unwound?;
5498  Ok(match text {
5499    Some(texts) if undone => Ok(texts),
5500    _ => Err(SpeculativeRun::Unavailable),
5501  })
5502}
