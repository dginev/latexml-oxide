21/// The `EQUATION_NUMBERING` key of LaTeX's own `{equation}` (latex.ltx:15737-15739 `\def\endequation{\eqno
22/// \hbox{\@eqnnum}…}`): its number is printed whatever a `\nonumber` in it said — `\nonumber` (:15775
23/// `\global\@eqnswfalse`) is eqnarray's switch, which the next `\eqnarray` sets again. Perl lets `\nonumber` remove the
24/// number of any `{equation}` (latex_constructs.pool.ltxml:1915-1916, "even though stock LaTeX provides no means to
25/// override them"), which is amsmath's `\nonumber` (amsmath.sty:1268-1269) and so stays where amsmath is loaded
26/// ([`amsmath_nonumber`]). hep-ph9710521 (`\nl` in `\beq`) printed every number from (3.22) on one behind its PDF;
27/// nucl-th0404042's `j-sum1` and `l-sum1` (PDF 20, 23) were unnumbered. REVTeX 3's aps.sty:401-450 and iopart.cls:771-776
28/// print the number as LaTeX does. OXIDIZED_DESIGN_DIVERGENCES #865.
29pub(crate) const LATEX_EQNNUM: &str = "latex_eqnnum";

31/// Whether amsmath is loaded, as `\@ifpackageloaded{amsmath}` asks (`\ver@amsmath.sty`) — its binding or the real
32/// file; not a binding's load beyond its original (`amsmath_loaded_beyond_original`, the old aa.cls's), which counts as
33/// not loaded — so that its `\nonumber` (amsmath.sty:1268-1269) removes an `{equation}`'s number ([`LATEX_EQNNUM`]).
34/// (`state::package_loaded`: aa.cls's binding clears amsmath's load record after today's aa.cls loads it, `aa_cls.rs`,
35/// Perl aa.cls.ltxml:57; 1712.01010, 1812.04463, 2501.07980 ship that aa.cls.) Or its definitions read here
36/// (`load_recorded`): a document's `\let\ver@amsmath.sty\relax` takes back the mark, not amsmath's `\nonumber`.
37fn amsmath_nonumber() -> bool { package_loaded("amsmath") || load_recorded("amsmath.sty") }

39/// The `EQUATION_NUMBERING` key of amsmath's `{equation}` (amsmath.sty:2913-2932, where amsmath is loaded:
40/// [`amsmath_nonumber`]), whose `\nonumber` leaves its labels the number it stepped. `\incr@eqnum` is
41/// `\refstepcounter{equation}` at the start, which sets `\@currentlabel`; `\nonumber` (amsmath.sty:1269-1275) only
42/// steps the counter back, and the display's `\label` is made at its end (`\endmathdisplay@a`, :2821-2830), so it names
43/// the stepped number — the next equation's, which no equation of the two prints. Perl retracts the number and the
44/// label names the enclosing unit; so did this port (1306.2116's `eq:covariate`: PDF 17, ours "3.1"). The equation
45/// keeps the number's reference forms (`eqnarray_next_number_tags`, no printed tag, the eqnarray form of
46/// OXIDIZED_DESIGN_DIVERGENCES #847) when it holds a label ([`NEXT_NUMBER_EQUATION`]). OXIDIZED_DESIGN_DIVERGENCES #875,
47/// KNOWN_PERL_ERRORS #1128. Guard `perfect_kernel_batch78t::ams_equation_nonumber_label_names_the_stepped_number`.
48const AMS_EQUATION: &str = "ams_equation";

54/// The numbering of LaTeX's `{equation}`, for its own binding and for a class's `{equation}` that sets it as LaTeX

63  } else {

67  }

70/// Whether the current equation row will be left unnumbered at its end (`after_equation`'s retraction): its own
71/// `retract` (a deferred `\nonumber`, `\IEEEnonumber`) or the numbering's (`retract=1`: eqnarray*'s, IEEEeqnarray*'s)
72/// on a preset number, unless the row says `noretract` (`\IEEEyesnumber`).
73fn equation_row_retracts() -> bool {
74  with_value("EQUATION_NUMBERING", |v| {
75    let Some(Stored::HashStored(numbering)) = v else {
76      return false;
77    };
78    with_value("EQUATIONROW_TAGS", |t| {
79      let Some(Stored::HashStored(tags)) = t else {
80        return false;
81      };
82      let set = |m: &SymHashMap<Stored>, key: &str| matches!(m.get(key), Some(&Stored::Bool(true)));
83      !set(tags, "noretract")
84        && (set(tags, "retract")
85          || (set(numbering, "retract") && set(numbering, "preset") && set(tags, "preset")))
86    })
87  })
88}

90/// Whether a `\label` in a row its numbering leaves unnumbered names the next printed number (the numbering key
91/// `labelnext=1`; implied by `eqncr=1`): IEEEtrantools.sty's `\@IEEEeqnarray` and `\@@IEEEeqnarraycr` (:1986-1991,
92/// :2072-2117) pre-advance the equation counter and set `\@currentlabel` to it for the next row, and `\IEEEnonumber`
93/// (:1405, `\global\@eqnswfalse`) prints no number and steps nothing, so as in latex.ltx's eqnarray
94/// ([`eqnarray_eqncr_numbering`]) the label names the number the next numbered row prints, or the counter's next
95/// value (rolled back at the end, :2022-2023) when none does.
96fn eqnarray_label_names_next_number() -> bool {
97  eqnarray_eqncr_numbering()
98    || with_value(
99      "EQUATION_NUMBERING",
100      |v| matches!(v, Some(Stored::HashStored(n)) if matches!(n.get("labelnext"), Some(&Stored::Bool(true)))),
101    )
102}

113/// Whether the current equation numbering presets its rows with `counter` (`\IEEEyessubnumber*`'s `subequation`).

1003/// Whether `t` ends a first-column `\lefteqn`'s cell: a tab or row end, or by meaning `\\`, `\nonumber`, `\notag`,
1004/// `\label`, the environment's `\end`, `\tabularnewline`. A command that is undefined or `\relax` matches only itself
1005/// (`\notag` without amsmath; latex.ltx's top-level `\tabularnewline`, which `\protect` would equal).
1006fn lefteqn_ends(t: &Token) -> bool {
1007  const ENDS: [&str; 7] = [
1008    "\\\\",
1009    "\\nonumber",
1010    "\\notag",
1011    "\\label",
1012    "\\end",
1013    "\\tabularnewline",
1014    "\\cr",
1015  ];
1016  is_column_end(t).is_some()
1017    || ENDS.iter().any(|name| {
1018      let end = T_CS!(*name);
1019      *t == end
1020        || (lookup_meaning(&end).is_some()
1021          && !x_equals(&end, &T_CS!("\\relax"))
1022          && x_equals(t, &end))
1023    })
1024}

1026/// The look-ahead after a first-column `\lefteqn`: whether the rest of its cell holds nothing the cell's math would
1027/// (`ends_cell`), read as TeX reads it, through the macros it starts. The input it reads is kept in `seen`, to be put
1028/// back; the bodies of the macros it follows are on `bodies`; `back` holds tokens a keyword test read and put back.
1029/// Passed over: spaces, braces (`{}`, `{\nonumber}`), the spacing of `lefteqn_spacing` (`\relax`, `\protect`
1030/// before a robust command's inner macro, `\strut`, `\mathstrut`), `\hspace`/`\vphantom` with their argument,
1031/// `\hskip` with its glue, `\kern` with its dimension. A macro is followed through 8 expansions (OXIDIZED_DESIGN_
1032/// DIVERGENCES #556 lists the limits).
1033#[derive(Default)]
1034struct LefteqnLookAhead {
1035  seen:   Vec<Token>,
1036  bodies: Vec<(Vec<Token>, usize)>,
1037  back:   Vec<Token>,
1038}
1039
1040impl LefteqnLookAhead {
1041  fn next(&mut self) -> Result<Option<Token>> {
1042    if let Some(t) = self.back.pop() {
1043      return Ok(Some(t));
1044    }
1045    while let Some((body, at)) = self.bodies.last_mut() {
1046      if let Some(t) = body.get(*at).copied() {
1047        *at += 1;
1048        return Ok(Some(t));
1049      }
1050      self.bodies.pop();
1051    }
1052    let t = read_token()?;
1053    if let Some(t) = t {
1054      self.seen.push(t);
1055    }
1056    Ok(t)
1057  }
1058
1059  /// The next token that is not a space (read).
1060  fn next_non_space(&mut self) -> Result<Option<Token>> {
1061    loop {
1062      match self.next()? {
1063        Some(t) if t.get_catcode() == Catcode::SPACE => {},
1064        other => return Ok(other),
1065      }
1066    }
1067  }
1068
1069  /// Reads the letters of `keyword` (any case, after spaces), or reads nothing.
1070  fn keyword(&mut self, keyword: &str) -> Result<bool> {
1071    let mut read = Vec::new();
1072    let first = self.next_non_space()?;
1073    let mut t = first;
1074    for c in keyword.chars() {
1075      match t {
1076        Some(tok)
1077          if tok.get_catcode() == Catcode::LETTER
1078            && tok.with_str(|s| s.eq_ignore_ascii_case(&c.to_string())) =>
1079        {
1080          read.push(tok);
1081          t = if read.len() < keyword.len() {
1082            self.next()?
1083          } else {
1084            None
1085          };
1086        },
1087        _ => {
1088          if let Some(tok) = t {
1089            self.back.push(tok);
1090          }
1091          self.back.extend(read.into_iter().rev());
1092          return Ok(false);
1093        },
1094      }
1095    }
1096    Ok(true)
1097  }
1098
1099  /// A macro argument: a balanced group (after spaces and, for `\hspace`, a `*`), or one token.
1100  fn argument(&mut self, star: bool) -> Result<()> {
1101    let mut t = self.next_non_space()?;
1102    if star && t == Some(T_OTHER!("*")) {
1103      t = self.next_non_space()?;
1104    }
1105    if t.is_some_and(|t| t.get_catcode() == Catcode::BEGIN) {
1106      let mut depth = 1usize;
1107      while depth > 0 {
1108        match self.next()? {
1109          Some(t) if t.get_catcode() == Catcode::BEGIN => depth += 1,
1110          Some(t) if t.get_catcode() == Catcode::END => depth -= 1,
1111          Some(_) => {},
1112          None => break,
1113        }
1114      }
1115    }
1116    Ok(())
1117  }
1118
1119  /// A dimension (tex.web §448): signs, then a register, or a factor and a unit (`true`, and `fil`/`fill`/`filll` in
1120  /// a glue's stretch or shrink).
1121  fn dimension(&mut self, fil: bool) -> Result<()> {
1122    let mut t = self.next_non_space()?;
1123    while t.is_some_and(|t| t == T_OTHER!("+") || t == T_OTHER!("-")) {
1124      t = self.next_non_space()?;
1125    }
1126    let factor = |t: &Token| {
1127      t.get_catcode() == Catcode::OTHER
1128        && t.with_str(|s| {
1129          s.chars()
1130            .all(|c| c.is_ascii_digit() || c == '.' || c == ',')
1131        })
1132    };
1133    match t {
1134      Some(t) if matches!(t.get_catcode(), Catcode::CS | Catcode::ACTIVE) => return Ok(()),
1135      Some(t) if factor(&t) => {},
1136      Some(t) => {
1137        self.back.push(t);
1138        return Ok(());
1139      },
1140      None => return Ok(()),
1141    }
1142    loop {
1143      match self.next()? {
1144        Some(t) if factor(&t) => {},
1145        Some(t) => {
1146          self.back.push(t);
1147          break;
1148        },
1149        None => break,
1150      }
1151    }
1152    // a register after the factor (`2\arraycolsep`)
1153    match self.next_non_space()? {
1154      Some(t) if matches!(t.get_catcode(), Catcode::CS | Catcode::ACTIVE) => return Ok(()),
1155      Some(t) => self.back.push(t),
1156      None => return Ok(()),
1157    }
1158    if fil && self.keyword("fil")? {
1159      while self.keyword("l")? {}
1160      return Ok(());
1161    }
1162    self.keyword("true")?;
1163    for unit in [
1164      "pt", "pc", "in", "bp", "cm", "mm", "dd", "cc", "sp", "em", "ex", "mu", "px",
1165    ] {
1166      if self.keyword(unit)? {
1167        break;
1168      }
1169    }
1170    Ok(())
1171  }

1173  /// A glue (tex.web §461): a dimension, then `plus` and `minus` ones.
1174  fn glue(&mut self) -> Result<()> {
1175    self.dimension(false)?;
1176    if self.keyword("plus")? {
1177      self.dimension(true)?;
1178    }
1179    if self.keyword("minus")? {
1180      self.dimension(true)?;
1181    }
1182    Ok(())
1183  }

1185  fn ends_cell(&mut self) -> Result<bool> {
1186    let mut expansions = 0;
1187    for _ in 0..256 {
1188      let Some(t) = self.next()? else {
1189        return Ok(true);
1190      };
1191      if matches!(
1192        t.get_catcode(),
1193        Catcode::SPACE | Catcode::BEGIN | Catcode::END
1194      ) {
1195        continue;
1196      }
1197      if lefteqn_ends(&t) {
1198        return Ok(true);
1199      }
1200      if lefteqn_spacing(&t) {
1201        continue;
1202      }
1203      if t == T_CS!("\\hspace") || t == T_CS!("\\vphantom") {
1204        self.argument(t == T_CS!("\\hspace"))?;
1205        continue;
1206      }
1207      if t == T_CS!("\\hskip") {
1208        self.glue()?;
1209        continue;
1210      }
1211      if t == T_CS!("\\kern") {
1212        self.dimension(false)?;
1213        continue;
1214      }
1215      let body = match lookup_meaning(&t) {
1216        Some(Stored::Expandable(defn))
1217          if matches!(t.get_catcode(), Catcode::CS | Catcode::ACTIVE) =>
1218        {
1219          match defn.get_expansion() {
1220            Some(ExpansionBody::Tokens(body)) => Some(body.unlist_ref().clone()),
1221            _ => None,
1222          }
1223        },
1224        _ => None,
1225      };
1226      match body {
1227        Some(body) if expansions < 8 => {
1228          expansions += 1;
1229          self.bodies.push((body, 0));
1230        },
1231        _ => return Ok(false),
1232      }
1233    }
1234    Ok(false)
1235  }
