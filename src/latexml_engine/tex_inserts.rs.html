13  // Inserting material
14  //----------------------------------------------------------------------
15  // \insert           c  places material into an insertions class.
16  // \insert<8bit><filler>{<vertical mode material>}
17  // Inside the class's own title page (the deposit, `\lx@deposit@maketitle`) the material goes to the page foot, out of
18  // the title page's flow (AAAI's copyright slug, aaai21.sty:53 `\insert\footins{\noindent\footnotesize…}`; 2011.04405,
19  // 1909.05089): as TeX makes it an internal vertical list of its own (tex.web §1099, its `}` ends its paragraph,
20  // §1100), it is digested in vertical mode, ended by a `\par`, and set after the title page
21  // (`lx_deposit_inserts`, handed on by `\lx@deposit@maketitle`) — not inside the paragraph the `\insert` stands in,
22  // nor running into the title page's next line or what the body sets after it (2501.04820's `\footnotetext`,
23  // 2110.04708's teaser figure) — marked (`\lx@deposit@insert@begin` … `\lx@deposit@insert@end`), so a
24  // late abstract's cover passes it over (OXIDIZED_DESIGN_DIVERGENCES #903). Material opened by an implicit `\bgroup`
25  // (§403 `scan_left_brace` takes it too: revtex4*, aastex701, memoir) is read live to its group's end, as `\vadjust`'s
26  // (`predigest_insert_group_contents`), and set after the title page the same way. A speculative run of the title page
27  // reads it as printed (sect05.rs `title_page_probe_run`), so a store it prints and the label before it are
28  // the title page's, the label taken out with the store (no orphan "Keywords:" line). Elsewhere, as Perl, the
29  // material is set where it stands.
30  DefPrimitive!("\\insert Number", sub[_args] {
31    if !lookup_bool("lx_depositing_frontmatter_fields") {
32      return Ok(Vec::new());
33    }
34    let Some(next) = read_non_space()? else {
35      return Ok(Vec::new());
36    };
37    let set = if next.get_catcode() == Catcode::BEGIN {
38      unread(Tokens::new(vec![next]));
39      let material = read_arg(ExpansionLevel::Off)?;
40      let mut tokens = vec![T_CS!("\\lx@deposit@insert@begin")];
41      tokens.extend(material.unlist());
42      tokens.extend([T_CS!("\\par"), T_CS!("\\lx@deposit@insert@end")]);
43      begin_mode("inline_internal_vertical")?;
44      let set = digest(Tokens::new(tokens));
45      end_mode("inline_internal_vertical")?;
46      vec![set?]
47    } else if is_left_brace(&next) {
48      let begin = digest(Tokens!(T_CS!("\\lx@deposit@insert@begin")))?;
49      let material = predigest_insert_group_contents()?;
50      let end = digest(Tokens!(T_CS!("\\lx@deposit@insert@end")))?;
51      [Some(begin), material, Some(end)].into_iter().flatten().collect()
52    } else {
53      unread(Tokens::new(vec![next]));
54      return Ok(Vec::new());
55    };
56    let mut set = Some(set);
57    with_value_mut("lx_deposit_inserts", |value| {
58      if let Some(Stored::VecDigested(inserts)) = value {
59        inserts.extend(set.take().into_iter().flatten());
60      }
61    });
62    if let Some(set) = set {
63      assign_value("lx_deposit_inserts", Stored::VecDigested(set), Some(Scope::Global));
64    }
65    Ok(Vec::new())
66  });

68  //======================================================================
69  // Splitting a box
70  //----------------------------------------------------------------------
71  // \vsplit c removes a specified amount of material from a box register .
72  // \splitbotmark c is the mark text of the last mark in the most recent \vsplit operation .
73  // \splitfirstmark c is the mark text of the first mark in the most recent \vsplit operation .
74  // tex.web §977-979: `\vsplit N to D` removes the top D-worth of material
75  // from box N, RETURNS it, and leaves the REMAINDER in the register (void
76  // once exhausted). Perl returns the whole box and leaves the register
77  // untouched — so the classic drain idiom
78  //   \def\do{\setbox2=\vsplit0 to\dimen@ … \ifdim\ht0>\z@ \expandafter\do\fi}
79  // (short-math-guide's column splitter, 6 docs of the Stomach:Recursion
80  // family) both duplicates content every pass AND never terminates.
81  // `vert_break` (§970-974) breaks only at glue that follows a non-discardable item, at a kern
82  // that glue follows, or at a penalty, and takes the LAST such break whose piece fits the split
83  // height (none fitting: the first; §974 `awful_bad`). LaTeXML's vertical lists carry no
84  // interline glue, so a box — or any item with a size — after a non-discardable item other than
85  // a rule stands for the interline glue before it (and a kern before one is a break); glue at the top
86  // of the box or after other glue is no break, nor is a zero-size item that is no box (the `\par`
87  // of an `\endgraf`, an anchor), which stays with the piece before it — split off as its own last
88  // piece it gave reledmac's `\do@line` an empty numbered line after every wrapped `\pstart`
89  // paragraph. The remainder loses the glue and kerns at its top (§968 `prune_page_top`) and is
90  // stored back, void when empty (§977). A penalty below 10000 is a break, at the top of the box
91  // too, and one of -10000 or less ends the piece there (§974 `eject_penalty`: short-math-guide's
92  // `\null\penalty-\@M` gives the `\null` alone, its first symbol row was thrown away with it);
93  // a penalty is discardable, so the box after `\nobreak` is no break (§970). The heights follow
94  // §970-977 below: interline glue, `\splitmaxdepth`, the piece packed to the split height and
95  // `\splittopskip` atop the remainder. Residuals: a paragraph is one item, never split into its
96  // lines (reledmac numbers a wrapped `\pstart` once); the piece is a list, not a vbox (R2); no
97  // stretch, shrink or penalty costs (§974-975: the last fitting break wins), and `\vfil` leaves no
98  // item; `\nointerlineskip` (`\prevdepth`) leaves none either, so interline glue is assumed there;
99  // a paragraph joins the box above with its whole height, not its first line's.
100  // Perl's `\vsplit` returns the whole box and never empties the
101  // register (TeX_Inserts.pool.ltxml:36-40, KNOWN_PERL_ERRORS #402). Guards
102  // `perfect_kernel_batch58::vsplit_breaks_only_where_tex_can`, `box_primitives::vsplit`.
103  // `to` is a keyword (tex.web §1082 `scan_keyword("to")`, §407): spaces before it are skipped and its letters match in
104  // either case, so cutwin.sty:122's `\vsplit#1 to\baselineskip` with `#1` a `\newbox` register (a chardef, after which
105  // the number scan leaves the space) splits (witness 2505.21389). Perl's `Match:to` (TeX_Inserts.pool.ltxml:36) takes
106  // the exact tokens: "Missing argument Match:to" (KNOWN_PERL_ERRORS #926). Without the keyword TeX says "Missing `to'
107  // inserted" and reads the dimension. Guard `perfect_kernel_batch70b::vsplit_to_is_a_keyword`.
108  DefPrimitive!("\\vsplit Number", sub[(number)] {
109    if read_keyword(&["to"])?.is_none() {
110      Error!(
111        "expected",
112        "to",
113        "Missing `to' inserted",
114        "I'm working on `\\vsplit<box number> to <dimen>'; will look for the <dimen> next."
115      );
116    }
117    let dimension = read_dimension()?;
118    let box_key   = s!("box{}", number.value_of());
119    match lookup_value(&box_key) { Some(Stored::Digested(stuff)) => {
120      adjust_box_color(&stuff)?;
121      // tex.web §977: `box(n):=null` / `box(n):=vpack(q)` store at the
122      // register's EXISTING eq_level ("the eq_level of the box stays the
123      // same") — no save-stack entry, so the drain survives the caller's
124      // group. eledmac.sty:1363-1369 `\do@line` splits one line per pass
125      // inside `{…\global\setbox\one@line=\vsplit\raw@text to\baselineskip}`
126      // and its `\loop\ifvbox\raw@text` (L1304) ends only when the register
127      // goes void; a local store was undone by the `}` every pass (eledform
128      // example: box-list memory runaway). `Scope::InPlace` is the analog.
129      // Guard: `box_primitives::vsplit`.
130      if stuff.is_empty()? {
131        assign_value(&box_key, Stored::None, Some(Scope::InPlace));
132        Digested::from(List::default())
133      } else {
134        let target = dimension.value_of();
135        // A register filled by `\setbox0=\vbox{…}` holds the `\vbox` WHATSIT,
136        // whose vertical list is its `content_box` (tex_box.rs after_digest);
137        // splitting the whatsit itself as one opaque item swept the whole
138        // box into the first split-off and voided the register — the
139        // discard-top idiom `\setbox2=\vsplit0 to\baselineskip` then threw
140        // away every line (short-math-guide's amssymb tables, 361 words,
141        // zero errors). tex.web §977 splits the box's list; the remainder
142        // is stored back as the same kind of box (`vpack(q)`).
143        // The interline parameters the box was built with (base_utilities.rs `predigest_box_contents_in_mode`).
144        let recorded = |l: &List| -> [Option<i64>; 3] {
145          ["baseline", "lineskip", "lineskiplimit"].map(|key| match l.properties.get(key) {
146            Some(Stored::Dimension(d)) => Some(d.value_of()),
147            _ => None,
148          })
149        };
150        let (items, vbox, interline): (Vec<Digested>, Option<Whatsit>, [Option<i64>; 3]) =
151          match stuff.data() {
152            DigestedData::List(l) => match l.try_borrow() {
153              Ok(l) => (l.boxes.clone(), None, recorded(&l)),
154              Err(_) => (vec![stuff.clone()], None, [None; 3]),
155            },
156            DigestedData::Whatsit(w) => match w.try_borrow() {
157              Ok(w) if w.get_property_bool("is_vbox") => {
158                let (items, interline) = match w.get_arg(2).map(|c| (c.clone(), c.data())) {
159                  Some((_, DigestedData::List(l))) => match l.try_borrow() {
160                    Ok(l) => (l.boxes.clone(), recorded(&l)),
161                    Err(_) => (Vec::new(), [None; 3]),
162                  },
163                  Some((c, _)) => (vec![c], [None; 3]),
164                  None => (Vec::new(), [None; 3]),
165                };
166                (items, Some(w.clone()), interline)
167              },
168              _ => (vec![stuff.clone()], None, [None; 3]),
169            },
170            _ => (vec![stuff.clone()], None, [None; 3]),
171          };
172        // A vertical kern is vertical space too, but only glue is a breakpoint after material (§973).
173        let glue = |item: &Digested| {
174          item.get_property_bool("isVerticalSpace") && !item.get_property_bool("isKern")
175        };
176        let penalty = |item: &Digested| {
177          item.get_property_bool("isPenalty").then(|| match item.get_property("penalty").as_deref() {
178            Some(Stored::Int(value)) => *value,
179            _ => 0,
180          })
181        };
182        let discardable = |item: &Digested| {
183          glue(item) || item.get_property_bool("isKern") || item.get_property_bool("isPenalty")
184        };
185        let defined_by = |item: &Digested, names: &[&str]| match item.data() {
186          DigestedData::Whatsit(w) => w.try_borrow().is_ok_and(|w| {
187            let definition = w.get_definition();
188            names.contains(&definition.get_cs_name().as_ref())
189          }),
190          _ => false,
191        };
192        let boxlike = |item: &Digested| defined_by(item, &["\\hbox", "\\vbox", "\\vtop"]);
193        // A rule is no breakpoint, and no interline glue follows one (§1056: `prev_depth` becomes
194        // `ignore_depth`), so the box after it is none either.
195        let rule = |item: &Digested| defined_by(item, &["\\hrule"]);
196        // Each item's height and depth and whether it has any size at all, computed as the break
197        // search reaches it (one item ahead for a kern): a drain loop sizes each item about once.
198        let mut sizes = ItemSizes { items: &items, known: Vec::new() };
199        // `interline`: a box before, so interline glue precedes the next box (none at the top of the list or after a
200        // rule, §1056, §1083 `ignore_depth`).
201        let is_breakpoint = |i: usize, interline: bool, sizes: &mut ItemSizes| -> Result<bool> {
202          let after_material = i > 0 && !discardable(&items[i - 1]);
203          Ok(if let Some(value) = penalty(&items[i]) {
204            value < 10000
205          } else if glue(&items[i]) {
206            after_material
207          } else if items[i].get_property_bool("isKern") {
208            // a kern before glue — or before a box, which stands for its interline glue
209            i + 1 < items.len()
210              && (glue(&items[i + 1])
211                || (interline
212                  && !rule(&items[i + 1])
213                  && (boxlike(&items[i + 1]) || sizes.get(i + 1)?.2)))
214          } else {
215            after_material
216              && interline
217              && !rule(&items[i])
218              && (boxlike(&items[i]) || sizes.get(i)?.2)
219          })
220        };
221        // A box or rule of the list: what `cur_height` and the interline glue measure.
222        let is_box = |i: usize, sizes: &mut ItemSizes| -> Result<bool> {
223          Ok(rule(&items[i]) || boxlike(&items[i]) || (!discardable(&items[i]) && sizes.get(i)?.2))
224        };
225        // §970-976 `vert_break`: `cur_height` runs to the baseline of the last box and `prev_dp` is that box's depth,
226        // at most `\splitmaxdepth` (§976); glue and kerns add their size. LaTeXML's lists carry no interline glue, so
227        // a box after a box adds it as §679 `append_to_vlist` made it, with the box's own parameters: `\baselineskip`
228        // less the depth above and the height below, `\lineskip` when that is under `\lineskiplimit`; glue in between keeps the depth above, a rule
229        // cancels it (`ignore_depth`, §1056). The piece ends at the last breakpoint whose piece fits (§974-975: with
230        // no stretch an underfull break costs `deplorable`, and the later of equal costs wins); none fitting, at the
231        // first breakpoint (`awful_bad`), else the whole list. Summing the items alone put three lines in a piece of
232        // `2.5\baselineskip`, where TeX puts two.
233        let register = |name: &str| lookup_register_quiet(name).map_or(0, |value| value.value_of());
234        let [baselineskip, lineskip, lineskiplimit] = interline;
235        let baselineskip = baselineskip.unwrap_or_else(|| register("\\baselineskip"));
236        let lineskip = lineskip.unwrap_or_else(|| register("\\lineskip"));
237        // As the sizing (font.rs): a list that recorded no limit is stacked with LaTeX's 0pt.
238        let lineskiplimit = lineskiplimit.unwrap_or(0);
239        let max_depth = register("\\splitmaxdepth");
240        let mut end = items.len();
241        let mut last_fit: Option<usize> = None;
242        let (mut cur_height, mut prev_dp): (i64, i64) = (0, 0);
243        let mut depth_above: Option<i64> = None;
244        for (i, item) in items.iter().enumerate() {
245          let interline = depth_above.is_some();
246          if (i > 0 || penalty(item).is_some()) && is_breakpoint(i, interline, &mut sizes)? {
247            if cur_height > target {
248              end = last_fit.unwrap_or(i);
249              break;
250            }
251            last_fit = Some(i);
252            if penalty(item).is_some_and(|value| value <= -10000) {
253              end = i;
254              break;
255            }
256          }
257          let (height, depth, _) = sizes.get(i)?;
258          if glue(item) || item.get_property_bool("isKern") {
259            cur_height += prev_dp + height + depth;
260            prev_dp = 0;
261          } else if is_box(i, &mut sizes)? {
262            if let Some(above) = depth_above.filter(|_| !rule(item)) {
263              let gap = baselineskip - above - height;
264              cur_height += prev_dp + if gap >= lineskiplimit { gap } else { lineskip };
265              prev_dp = 0;
266            }
267            cur_height += prev_dp + height;
268            prev_dp = depth;
269            depth_above = (!rule(item)).then_some(depth);
270            if prev_dp > max_depth {
271              cur_height += prev_dp - max_depth;
272              prev_dp = max_depth;
273            }
274          }
275        }
276        if end == items.len()
277          && cur_height > target
278          && let Some(fit) = last_fit
279        {
280          end = fit;
281        }
282        // §977 `vpackage(p, h, exactly, split_max_depth)`: the piece is exactly the split height, as deep as its last
283        // box (0pt under glue or a kern after it), at most `\splitmaxdepth`.
284        let mut piece_depth = 0;
285        for j in (0..end).rev() {
286          if glue(&items[j]) || items[j].get_property_bool("isKern") {
287            break;
288          }
289          if is_box(j, &mut sizes)? {
290            piece_depth = sizes.get(j)?.1.min(max_depth);
291            break;
292          }
293        }
294        // §968 `prune_page_top`: the remainder loses the glue, kerns and penalties above its first box or rule (marks
295        // and whatsits stay), and `\splittopskip` glue goes before that box, less its height (§969).
296        let mut first_box = None;
297        let mut pruned = Vec::new();
298        for (j, item) in items.iter().enumerate().skip(end) {
299          if is_box(j, &mut sizes)? {
300            first_box = Some(j);
301            break;
302          }
303          if discardable(item) {
304            pruned.push(j);
305          }
306        }
307        let top_skip = match first_box {
308          Some(j) => {
309            let skip = lookup_register_quiet("\\splittopskip").map_or(0, |value| value.value_of());
310            Some((j, (skip - sizes.get(j)?.0).max(0)))
311          },
312          None => None,
313        };
314        let mut split_off = items;
315        let tail = split_off.split_off(end);
316        let mut rest = Vec::with_capacity(tail.len() + 1);
317        for (j, item) in (end..).zip(tail) {
318          if pruned.contains(&j) {
319            continue;
320          }
321          if let Some((first, skip)) = top_skip
322            && first == j
323          {
324            rest.push(split_top_skip(skip));
325          }
326          rest.push(item);
327        }
328        if rest.is_empty() {
329          assign_value(&box_key, Stored::None, Some(Scope::InPlace));
330        } else {
331          // The remaining list keeps what its list recorded (mode, interline parameters), not its size (§977
332          // `vpack(q, natural)`: a re-split piece's remainder is not the piece's height).
333          let remaining = |list: Option<&Digested>| {
334            let mut remaining = List::new(rest);
335            if let Some(DigestedData::List(old)) = list.map(Digested::data)
336              && let Ok(old) = old.try_borrow()
337            {
338              remaining.mode = old.mode.clone();
339              remaining.properties = old.properties.clone();
340              for key in ["height", "depth", "width", "cached_width", "cached_height", "cached_depth"] {
341                remaining.properties.remove(key);
342              }
343            }
344            // `vpack`: the remainder is a vbox — the remainder of a split piece (a list without a mode) too.
345            if !remaining.properties.contains_key("mode") {
346              remaining
347                .properties
348                .insert("mode", Stored::String(pin!("internal_vertical")));
349            }
350            remaining.properties.insert("vattach", Stored::String(pin!("bottom")));
351            Digested::from(remaining)
352          };
353          let remainder = match vbox {
354            Some(mut w) => {
355              // `vpack(q, natural)`: the same box with the remaining list, at its natural size — not the split
356              // box's `to`/`spread` height, a `\wd` set on it, nor its cached size. Residual: a `\vtop`'s remainder
357              // is sized as a vbox but still typeset top-attached (the `\vtop` constructor's own attachment).
358              let list = remaining(w.get_arg(2));
359              if let Some(arg) = w.get_arg_mut(2) {
360                *arg = list;
361              }
362              w.set_property("content_box", w.get_arg(2).cloned());
363              for key in ["height", "depth", "width", "cached_width", "cached_height", "cached_depth"] {
364                w.get_properties_mut().remove(key);
365              }
366              Digested::from(w)
367            },
368            None => remaining(Some(&stuff)),
369          };
370          assign_value(&box_key, Stored::Digested(remainder), Some(Scope::InPlace));
371        }
372        // §977 packs the piece as a vbox; here it stays a list, which `\unvbox` (Perl `unlist`, TeX_Box.pool.ltxml:
373        // 725-733) unpacks — a vbox whatsit it would not, and reledmac's `\do@line` and short-math-guide's
374        // `\vtop{\unvbox2}` columns rely on it. A piece of several lines typeset directly runs them together (RED
375        // boxes-groups/box_primitives_vsplit, R2).
376        let mut piece = List::new(split_off);
377        for (key, value) in ["baseline", "lineskip", "lineskiplimit"].into_iter().zip(interline) {
378          if let Some(value) = value {
379            piece.properties.insert(key, Stored::Dimension(Dimension::new(value)));
380          }
381        }
382        piece.properties.insert("height", Stored::Dimension(Dimension::new(target)));
383        piece.properties.insert("depth", Stored::Dimension(Dimension::new(piece_depth)));
384        Digested::from(piece)
385      }
386    } _ => {
387      Digested::from(List::default())
388    }}
389  });
390  DefMacro!(T_CS!("\\splitfirstmark"), None, Tokens!());
391  DefMacro!(T_CS!("\\splitbotmark"), None, Tokens!());

406/// The vertical sizes of a `\vsplit` list's items, computed on first use: each item's height and
407/// depth, and whether it has any size at all.
408struct ItemSizes<'a> {
409  items: &'a [Digested],
410  known: Vec<(i64, i64, bool)>,
411}
412
413impl ItemSizes<'_> {
414  fn get(&mut self, i: usize) -> Result<(i64, i64, bool)> {
415    while self.known.len() <= i {
416      let item = &self.items[self.known.len()];
417      let (w, h, d) = item.compute_size(Default::default())?;
418      let (h, d) = (h.value_of(), d.value_of());
419      // A paragraph's list is a line however small (tex.web §1096: a paragraph that is not null leaves at least one
420      // line box, `\mbox{}\par` and the empty line `end_graf` leaves included).
421      let sized = matches!(item.data(), DigestedData::List(_))
422        && item
423          .get_property("mode")
424          .is_some_and(|mode| mode.to_string() == "horizontal")
425        || h != 0
426        || d != 0
427        || w.value_of() != 0;
428      self.known.push((h, d, sized));
429    }
430    Ok(self.known[i])
431  }
432}

434/// §969: the `\splittopskip` glue above a `\vsplit` remainder's first box, `skip` (in sp) its size less the box's
435/// height. An empty box of that height, as the vertical penalty items are (tex_penalties.rs).
436fn split_top_skip(skip: i64) -> Digested {
437  let mut reversion = vec![T_CS!("\\vskip")];
438  reversion.extend(ExplodeText!(&Dimension::new(skip).to_string()));
439  let mut props = SymHashMap::default();
440  props.insert("isVerticalSpace", Stored::Bool(true));
441  props.insert("isSkip", Stored::Bool(true));
442  props.insert("width", Stored::Dimension(Dimension::default()));
443  props.insert("height", Stored::Dimension(Dimension::new(skip)));
444  props.insert("depth", Stored::Dimension(Dimension::default()));
445  Digested::from(Tbox::new(
446    pin!(""),
447    None,
448    None,
449    Tokens::new(reversion),
450    props,
451  ))
