33  // LaTeX has a very particular, but useful, notion of "Undefined",
34  //    so let's get that squared away at the outset; it's useful for TeX, too!
35  //
36  // Naturally, it uses \csname to check, which ends up DEFINING the possibly undefined macro as
37  // \relax
38  // Perl Base_Utility.pool.ltxml L23-31
39  DefMacro!("\\lx@ifundefined{}{}{}", sub[(name, if_token, else_token)] {
40    let cs = T_CS!(s!("\\{}", Expand!(name).to_string()));
41    // Autoload triggers (declared via `def_autoload` at tex.rs:238-247)
42    // install a closure under the trigger CS so the package auto-loads on
43    // first invocation. Perl scopes the equivalent `DefAutoload` entries
44    // to `OmniBus.cls.ltxml`, so for non-OmniBus papers Perl sees these
45    // CSes as truly undefined. Mirror Perl by treating an unfired
46    // autoload trigger as "undefined" in `\@ifundefined`. We must NOT
47    // overwrite the trigger CS with `\relax` in this case (the kernel
48    // `\csname X\endcsname \ifx \relax` idiom DOES overwrite, but doing
49    // so here would destroy the autoload — subsequent use of the trigger
50    // CS would no-op instead of loading its package). Driver:
51    // arXiv:2507.23241v1 (smfart.cls) — line 373's
52    // `\@ifundefined{numberwithin}` branches wrong when our preloaded
53    // `\numberwithin` autoload makes the CS "look" defined, then the
54    // `\@gobbletwo` branch eats `\ifx \relax` and orphans `\else` / `\fi`.
55    // An autoload TRIGGER counts as "undefined" only while UNFIRED — i.e. its
56    // target package has not yet loaded. `def_autoload` stores the package name
57    // (`.sty`) as a String; once `<pkg>.sty_loaded`/`_raw_loaded` is set the
58    // trigger CS holds the package's real definition and must read as DEFINED
59    // (fixes `\@ifundefined{align}` after `\usepackage{amsmath}`). `.pool`
60    // triggers keep the legacy Bool form and stay "undefined until used".
61    let is_autoload = cs.with_cs_name(|cs_name| {
62      match lookup_value(&s!("{cs_name}:autoload")) {
63        Some(Stored::String(pkg_sym)) => {
64          let pkg = with(pkg_sym, |s| s.to_string());
65          !lookup_bool(&s!("{pkg}.sty_loaded"))
66            && !lookup_bool(&s!("{pkg}.sty_raw_loaded"))
67        },
68        Some(Stored::Bool(b)) => b,
69        _ => false,
70      }
71    });
72    if IsDefined!(&cs) && !is_autoload {
73      Ok(else_token)
74    } else {
75      if !is_autoload {
76        assign_meaning(&cs, lookup_meaning(&TOKEN_RELAX), None);  // Let w/o AfterAssign
77      }
78      Ok(if_token)
79    }
80  }, locked=>true);

458  // to define "creators" (authors, editors,etc) and a variety of markup strategies to

506        for key in dkv.get_keyvals().keys() {
507          if let Some(v) = dkv.get_value_digested(key) {
508            options.insert(key.clone(), v.to_string());
509          }
510        }
511      }
512    // Perl: $role = $options{role} = ToString($options{role}) — digested value here
513    let role = options.get("role").cloned().unwrap_or_default();
514    options.insert("role".to_string(), role.clone());
515    let mut labels = clean_frontmatter_labels(
516      options.get("label").map(String::as_str).unwrap_or(""), "");
517    if !role.is_empty() {
518      let n = lookup_mapping_int(&s!("num_{tag}"), &role) + 1;
519      assign_mapping(&s!("num_{tag}"), &role, Some(Stored::Int(n)));
520      if let Some(labelseq) = options.get("labelseq")
521        && !labelseq.is_empty() {
522          labels.push(clean_label(&n.to_string(), Some(labelseq)).into_owned());
523        }
524    }
525    match get_frontmatter_name(options.get("name"), &tag, &role)? {
526      Some(name) => { options.insert("name".to_string(), name); },
527      None => { options.remove("name"); },
528    }
529    options.insert("_label".to_string(), labels.join(","));
530
531    // Snapshot the parent entries (those not role=pending), inherit labels
532    // from an enclosing pending entry, and push a tentative stub entry —
533    // in case digestion changes labels!
534    let (parent_indices, stub_idx) = with_value_mut("frontmatter", |val_opt| {
535      if let Some(&mut Stored::HashTagData(ref mut frnt)) = val_opt {
536        let list = frnt.entry(parenttag.clone()).or_insert_with(Vec::new);
537        let parent_indices: Vec<usize> = list.iter().enumerate()
538          .filter(|(_, e)| e.attr.get("role").map(String::as_str).unwrap_or("") != "pending")
539          .map(|(i, _)| i)
540          .collect();
541        // IF this item encountered WITHIN another frontmatter
542        // we inherit that frontmatter's labels (even override!)
543        if let Some(last) = list.last()
544          && last.attr.get("role").map(String::as_str) == Some("pending")
545            && matches!(last.content.first(), Some(TagContent::PlaceKeeper))
546          {
547            let inherited = last.attr.get("_annotations").cloned().unwrap_or_default();
548            options.insert("_label".to_string(), inherited);
549          }
550        let mut stub_attr = TagAttrs::default();
551        stub_attr.insert("role".to_string(), "pending".to_string());
552        stub_attr.insert("_annotations".to_string(),
553          options.get("_label").cloned().unwrap_or_default());
554        let stub = TagData {
555          tag: parenttag.clone(),
556          attr: stub_attr,
557          content: vec![TagContent::PlaceKeeper],
558        };
559        DebugFeature!("frontmatter", "FRONT Add stub {}\n  for annotation {} [{}]",
560          show_frontmatter(&stub), tag, options.get("_label").map(String::as_str).unwrap_or(""));
561        list.push(stub);
562        (parent_indices, list.len() - 1)
563      } else {
564        (Vec::new(), 0)
565      }
566    });
567    let nparents = parent_indices.len();
568    let xcontent = digest_frontmatter_item(&tag, content)?;
569    // Reset if changed (eg. by \lx@set@frontmatter@label during digestion)!
570    let stub_label = with_value("frontmatter", |v| {
571      if let Some(Stored::HashTagData(frnt)) = v {
572        frnt.get(&parenttag)
573          .and_then(|l| l.get(stub_idx))
574          .and_then(|e| e.attr.get("_annotations"))
575          .cloned()
576      } else {
577        None
578      }
579    }).unwrap_or_default();
580    options.insert("_label".to_string(), stub_label.clone());
581    let datum = if preformatted {
582      TagContent::Box(xcontent)
583    } else {
584      TagContent::Entry(TagData {
585        tag: tag.clone(),
586        attr: options.clone(),
587        content: vec![TagContent::Box(xcontent)],
588      })
589    };
590    if !preformatted && (!stub_label.is_empty() || nparents == 0) {
591      // deferred until we can compare labels
592      DebugFeature!("frontmatter", "... deferring to label={stub_label}");
593      frontmatter_set_first_content(&parenttag, stub_idx, datum);
594    } else {
595      let annotate = options.get("annotate").cloned().unwrap_or_default();
596      let (mut nprev, newonly): (i64, bool) = if annotate.is_empty() {
597        (1, false)
598      } else if annotate.chars().all(|c| c.is_ascii_digit()) {
599        (annotate.parse().unwrap_or(1), false)
600      } else if annotate == "all" {
601        (nparents as i64, false)
602      } else if annotate == "new" {
603        (nparents as i64, true)
604      } else {
605        Info!("unexpected", &tag, s!("Frontmatter annotate '{annotate}' unrecognized"));
606        (1, false)
607      };
608      DebugFeature!("frontmatter", "...adding to {nprev} previous{}",
609        if newonly { " new" } else { "" });
610      with_value_mut("frontmatter", |val_opt| {
611        if let Some(&mut Stored::HashTagData(ref mut frnt)) = val_opt
612          && let Some(list) = frnt.get_mut(&parenttag) {
613            // Remove unneeded (stub) entry.
614            if list.get(stub_idx)
615              .map(|e| e.attr.get("role").map(String::as_str) == Some("pending"))
616              .unwrap_or(false)
617            {
618              list.remove(stub_idx);
619            }
620            let has_role_key = s!("_has{role}");
621            let mut indices = parent_indices.clone();
622            while let Some(pi) = indices.pop() {
623              if nprev <= 0 {
624                break;
625              }
626              nprev -= 1;
627              if let Some(parent) = list.get_mut(pi) {
628                parent.content.push(datum.clone());
629                if !role.is_empty() {
630                  parent.attr.insert(has_role_key.clone(), "1".to_string());
631                  if newonly
632                    && let Some(&next_pi) = indices.last()
633                      && list.get(next_pi)
634                        .map(|p| p.attr.contains_key(&has_role_key))
635                        .unwrap_or(false)
636                      {
637                        break;
638                      }
639                }
640              }
641            }
642          }
643      });
644    }
645  });

825  // affiliations into an author \thanks{...}, linking them to authors by a
826  // leading superscript mark ($^{n}$ / \textsuperscript{n}). \thanks is
827  // semantically an acknowledgement, so we re-route ONLY when the content BEGINS
828  // with such a mark (the abuse signature — a genuine acknowledgement never
829  // starts with a bare superscript); every other \thanks stays the parity-
830  // faithful Semiverbatim role=thanks contact. When detected we re-tokenize (the
831  // Semiverbatim read froze $/^ to catcode-other) and feed each $^{n}$-delimited
832  // segment through \lx@affiliation@withsup, which sets the affiliation:N label
833  // that the authors' own marks already request (relocate_annotations then links
834  // author<->affiliation). Witness arXiv:2606.00313.
835  DefMacro!("\\lx@add@thanks [] Semiverbatim", sub[(attr, content)] {
836    if starts_with_affiliation_mark(&content) {
837      // `untex_string()`, NOT `to_string()`: flattening with `Display` welds a
838      // control word to whatever letter follows it (`\c c` → `\cc`), because the
839      // space that terminated the control word was eaten by the tokenizer and is
840      // not in the token list. `\thanks{...}` carries author names, so that is
841      // exactly the space-form-accent case that broke `\bib@@names` (PR #399) and
842      // the MathSciNet review path (issue 410). Witness arXiv:2606.00313.
843      let retok = mouth::tokenize_internal(content.clone().untex_string());

846        if seg.unlist_ref().iter().all(|t| *t == T_SPACE!()) {
847          continue;
848        }
849        let withsup = Invocation!(T_CS!("\\lx@affiliation@withsup"), vec![Some(seg)]);
850        calls.extend(
851          Invocation!(T_CS!("\\lx@add@affiliation"), vec![None, Some(withsup)]).unlist());
852      }
853      Ok(Tokens::new(calls))
854    } else {

856      // former template `\lx@annotate@frontmatter{ltx:creator}{ltx:contact}[role=thanks,#1]{#2}`.
857      let mut opts = mouth::tokenize_internal("role=thanks").unlist();

861      }
862      Ok(Invocation!(T_CS!("\\lx@annotate@frontmatter"),
863        vec![Some(mouth::tokenize_internal("ltx:creator")),
864             Some(mouth::tokenize_internal("ltx:contact")),
865             Some(Tokens::new(opts)), Some(content)]))
866    }

873  // This corresponds to standard LaTeX,
874  // The command replaces any previous authors/creators;
875  // It defining several creators separated by \and;
876  // and can use \\ to separate affiliation from each author.
877  // BUT ALSO, this markup is commonly abused by putting authors & affiliations in
878  // seemingly random orders, but adding superscript markers to connect them.
879  // Assumption: superscript near END is used for authors; near FRONT for affiliation.

883  DefMacro!("\\lx@add@authors{}", sub[(stuff)] {
884    // Beyond-Perl (surpasses Perl; KNOWN_PERL_ERRORS #100): IJCAI-style author
885    // blocks — ijcai97.sty and its derivatives (e.g. the ttm.sty in
886    // arXiv:2401.03955) — pack names, `\affiliations` and a comma-separated
887    // `\emails` list into ONE `\author{}`. The default splitter below does not
888    // recognise those section markers, so the comma-joined email list is shredded
889    // into phantom author creators and the affiliation is dropped (Perl 0.8.8 does
890    // the same, and both raise `\affiliations`/`\emails` as undefined). When the
891    // body carries either marker, delegate to the shared sectioned-author machinery
892    // (`\lx@ijcai@authorsplit`, also used by `ijcai_sty`): it consumes the markers
893    // as `Until:` delimiters (so they no longer error) and splits names /
894    // affiliations / emails, attaching the n-th email to the n-th author. This runs
895    // before any dequeue/normalization because the delegate re-enters
896    // `\lx@add@authors` on the (marker-free) name list.
897    // Witness html_feedback#1361 + #1362.
898    if position_of(&stuff, &[T_CS!("\\affiliations"), T_CS!("\\emails")]).is_some() {
899      let mut out = vec![T_CS!("\\lx@ijcai@authorsplit")];
900      out.extend(stuff.unlist());
901      out.push(T_CS!("\\affiliations"));
902      out.push(T_CS!("\\done"));
903      return Ok(Tokens::new(out));
904    }
905    let mut calls: Vec<Token> = Vec::new();
906    dequeue_front_matter("ltx:creator", &[("role", "author")]);
907    // Consume any `\\[len]` / `\\*[len]` row-break optionals up front so the line
908    // splits below see a bare `\\` (KNOWN_PERL_ERRORS #75, witness 2605.23553). This
909    // also runs ahead of the tabular/minipage fallback: that fallback is a
910    // last-resort escape hatch for author markup we cannot structure, and its
911    // `<tabular>`-in-`<personname>` output is a presentational artifact we do not
912    // want in frontmatter anyway, so there is nothing to preserve by skipping it.
913    let stuff = strip_linebreak_options(stuff);
914    // Beyond-Perl (OXIDIZED_DESIGN #52), two composable normalizations applied
915    // BEFORE branch selection so both branches benefit, and so a symbol mark can
916    // no longer spuriously trigger the affiliation-marker branch:
917    //   1. horizontal-space separators (`\hspace{len}` / `\hfill`) → `\quad`, the
918    //      "regular" poor-man's author separator LaTeXML's `\and`/`\quad` splitter
919    //      otherwise misses (witness arXiv:2506.06941, six authors bunched);
920    //   2. footnote-SYMBOL superscripts (`$^{*}$`, `\textsuperscript{\dagger}` —
921    //      equal-contribution / corresponding notes, NEVER affiliation numbers) →
922    //      a visible `\lx@frontmatter@keepsup` sup, so they render instead of
923    //      being consumed into an unmatched `affiliation:*` label and dropped.
924    // For the witness, (2) removes its ONLY `^`, dropping the whole block into the
925    // clean no-marker branch where (1)'s `\quad`s split all six authors.
926    let stuff = normalize_hspace_separators(stuff);
927    let stuff = rewrite_symbol_superscripts(stuff);
928    // If too much formatting, fall back to unstructured author content
929    let stuff_string = stuff.to_string();
930    if stuff_string.contains("{tabular}")
931      || stuff_string.contains("{minipage}")
932      || stuff_string.contains("\\halign")
933    {
934      calls.extend(Invocation!(T_CS!("\\lx@add@author"), vec![None, Some(stuff)]).unlist());
935    } else if position_of(&stuff, &authorsup_markers()).is_some() {
936      let mut entries: Vec<(AuthorLineKind, Tokens)> = Vec::new();
937      // Split on the `\and` family FIRST so an `\and` is a HARD author boundary:
938      // a marker-less line — an author whose only superscript is macro-delivered
939      // (`\handPointerZ`), or a continuation affiliation — never merges into an
940      // author from a PREVIOUS `\and` group. Within a group, `\quad`/`\\` still
941      // separate the name line from its affiliation lines, and a marker-less line
942      // still continues the group's own last entry. Groups carry no `\and`, so
943      // reusing `author_affil_splits` for the intra-group split is equivalent to
944      // splitting on `\quad`/`\qquad`/`\\`. html_feedback#1021 F2 residual:
945      // `Alice\mk$^*$ \and Bob\mk` was one merged creator; now two.
946      // OXIDIZED_DESIGN #52(g).
947      for group in split_tokens(stuff, author_and_splits()) {
948        if group.is_empty() {
949          continue;
950        }
951        // Marker-less lines may only continue an entry created WITHIN this
952        // `\and` group — never one from a previous group.
953        let group_start = entries.len();
954        for line in split_tokens(group, author_affil_splits()) {
955          if line.is_empty() {
956            continue;
957          }
958          match position_of(&line, &authorsup_markers()) {
959            None => {
960              // A marker-less line that is purely a list of email addresses is a
961              // SHARED email line (\texttt{a@x, b@y}) covering all the authors, not
962              // a continuation of the previous affiliation — give it its own email
963              // contact so it is shown once instead of being welded into an
964              // affiliation's text (KNOWN_PERL_ERRORS #75, witness 2605.23553).
965              if line_is_email_list(&line) {
966                entries.push((AuthorLineKind::Email, line));
967              } else if entries.len() > group_start {
968                // continues an entry from THIS `\and` group; Append
969                let last = entries.last_mut().unwrap();
970                let mut appended = last.1.clone().unlist();
971                appended.extend(line.unlist());
972                last.1 = Tokens::new(appended);
973              } else {
974                // First line of this `\and` group and it has no marker → a NEW
975                // author (never merge back into the previous group).
976                entries.push((AuthorLineKind::Author, line));
977              }
978            },
979            Some(p) => {
980              // "\textsuperscript{n}Affil" (the marker LEADS the line) → an
981              // affiliation; "Name\textsuperscript{n}" (a name precedes the
982              // marker) → an author line, split into the individual creators it
983              // names (see split_author_line). The old `p < 8` token-count proxy
984              // misread short author names like "Min Xu" (html_feedback#6614) —
985              // key on name-before-marker, which is length-independent.
986              if name_precedes_marker(&line, p) {
987                for author in split_author_line(line) {
988                  entries.push((AuthorLineKind::Author, author));
989                }
990              } else {
991                // A marker-led affiliation line may carry MULTIPLE
992                // `\textsuperscript{n}Affil` institutions on one space-separated line
993                // (html_feedback#6242, arXiv:2510.02340) — `\textsuperscript{1}Univ A
994                // \textsuperscript{2}Univ B`. Split at each whitespace-preceded mark
995                // (reusing the `\thanks`-abuse splitter, which never breaks a
996                // superscript glued INSIDE an institution name) so each numbered
997                // institution becomes its own affiliation and attaches to its authors
998                // by number, instead of merging into one.
999                for seg in split_before_affiliation_marks(line) {
1000                  if seg.unlist_ref().iter().all(|t| *t == T_SPACE!()) {
1001                    continue;
1002                  }
1003                  entries.push((AuthorLineKind::Affiliation, seg));
1004                }
1005              }
1006            },
1007          }
1008        }
1009      }
1010      let author_count = entries
1011        .iter()
1012        .filter(|(k, _)| *k == AuthorLineKind::Author)
1013        .count();
1014      for (kind, line) in entries {
1015        match kind {
1016          AuthorLineKind::Author => {
1017            let withsup = Invocation!(T_CS!("\\lx@author@withsup"), vec![Some(line)]);
1018            calls.extend(
1019              Invocation!(T_CS!("\\lx@add@author"), vec![None, Some(withsup)]).unlist());
1020          },
1021          AuthorLineKind::Affiliation => {
1022            let withsup = Invocation!(T_CS!("\\lx@affiliation@withsup"), vec![Some(line)]);
1023            calls.extend(
1024              Invocation!(T_CS!("\\lx@add@affiliation"), vec![None, Some(withsup)]).unlist());
1025          },
1026          AuthorLineKind::Email => {
1027            // A shared email line otherwise attaches to whatever creator is
1028            // "current" (the LAST author), bunching every address under one
1029            // person. Instead, resolve the individual addresses (`a@x, b@y, c@z`
1030            // distributed, or `{a,b,c}@dom` grouped -> expanded) and, when there
1031            // are no more than one per author, hand each to `\lx@add@email` with
1032            // labelseq=author: address i gets label author:i, which relocates to
1033            // the creator's own author:N sequence label. So N addresses distribute
1034            // one-per-author, and a single shared address lands on author:1 (the
1035            // lead), never a random trailing author. Preserve a whole-line
1036            // \texttt/\url wrapper by re-wrapping each address. If there are MORE
1037            // addresses than authors (cannot map cleanly), keep the original line
1038            // as one contact (the prior behavior). OXIDIZED_DESIGN #52(j).
1039            let wrapper = whole_line_cs_wrapper(&line).map(|(cmd, _)| cmd);
1040            let addrs = email_addresses(&line).unwrap_or_default();
1041            if !addrs.is_empty() && author_count >= 1 && addrs.len() <= author_count {
1042              for addr in addrs {
1043                let mut toks = mouth::tokenize(TeXString::assembled(addr)).unlist();
1044                if let Some(cmd) = wrapper {
1045                  let mut wrapped = vec![cmd, T_BEGIN!()];
1046                  wrapped.append(&mut toks);
1047                  wrapped.push(T_END!());
1048                  toks = wrapped;
1049                }
1050                let opts = mouth::tokenize_internal("labelseq=author");
1051                calls.extend(
1052                  Invocation!(T_CS!("\\lx@add@email"), vec![Some(opts), Some(Tokens::new(toks))])
1053                    .unlist(),
1054                );
1055              }
1056            } else {
1057              calls.extend(
1058                Invocation!(T_CS!("\\lx@add@email"), vec![None, Some(line)]).unlist());
1059            }
1060          },
1061        }
1062      }
1063    } else {
1064      // No superscript markers. Split into author GROUPS on the \and family /
1065      // \quad only — NOT comma (author_group_splits). Within a group, the first
1066      // \\-delimited line is the author-name list (comma / " and "-split via
1067      // split_author_line); each remaining \\-line is an affiliation, attached to
1068      // the group's LAST author (the "affiliation follows the name(s) before \\"
1069      // convention). Splitting groups on comma shredded multi-part addresses
1070      // ("…Laboratory, Laurel, MD 20723") into fake authors and mislabeled the
1071      // trailing \email line as an affiliation. OXIDIZED_DESIGN #52 (surpass-Perl,
1072      // Perl's @authorsplits shares the comma); witness arXiv:2606.00315.
1073      for block in split_tokens(stuff, author_group_splits()) {
1074        if block.is_empty() {
1075          continue;
1076        }
1077        // Drop empty `\\`-pieces up front. A group that BEGINS with `\\` — because
1078        // the previous author line ended with a trailing `\quad\\` (or `\and\\`) that
1079        // leaked the line-break into this group — would otherwise give an EMPTY
1080        // names_line, demoting the group's real first author to an affiliation. The
1081        // first NON-empty piece is the name list; the rest are affiliations. Witness
1082        // arXiv 2507.06670 (acl): `…Zhiyuan Zhu\quad \\ \textbf{Ruiqi Li}\quad…`
1083        // rendered "Ruiqi Li" as an empty `<personname/>` + a bold "Ruiqi Li"
1084        // affiliation; now "Ruiqi Li" is an author. (A `\\`-only block still yields a
1085        // single empty author below, so its affiliations are not dropped.)
1086        let mut pieces = split_tokens(block, vec![SplitDelim::Token(T_CS!("\\\\"))])
1087          .into_iter()
1088          .filter(|l| !l.is_empty());
1089        let names_line = pieces.next().unwrap_or_default();
1090        let affils: Vec<Tokens> = pieces.collect();
1091        let mut names = split_author_line(names_line);
1092        if names.is_empty() {
1093          names.push(Tokens::default());
1094        }
1095        let last = names.len() - 1;
1096        for (i, name) in names.into_iter().enumerate() {
1097          let mut body: Vec<Token> = name.unlist();
1098          if i == last {
1099            for line in &affils {
1100              // A bare email line (\texttt{user@host}) is an email, not an
1101              // affiliation (OXIDIZED_DESIGN #52; witness arXiv:2606.00315).
1102              let cs = if line_is_email(line) {
1103                T_CS!("\\lx@add@email")
1104              } else {
1105                T_CS!("\\lx@add@affiliation")
1106              };
1107              body.extend(Invocation!(cs, vec![None, Some(line.clone())]).unlist());
1108            }
1109          }
1110          calls.extend(
1111            Invocation!(T_CS!("\\lx@add@author"), vec![None, Some(Tokens::new(body))]).unlist());
1112        }
1113      }
1114    }
1115    Ok(Tokens::new(calls))
1116  });

1396    // so we don't clone the whole hashmap envelope; the per-role tokens

1408            .into_iter()

1411                Some(Stored::Token(t)) => Some(*t),
1412                Some(Stored::String(sym)) => {

1415                _ => None,

1929/// wrapper is stripped. Both Perl and Rust emit the wrapper (SHARED-FAILURE).
1930fn unwrap_whole_name_bold(document: &mut Document, node: &Node) -> Result<()> {
1931  let mut sole_bold: Option<Node> = None;
1932  for child in node.get_child_nodes() {
1933    match child.get_type() {
1934      Some(NodeType::TextNode) => {
1935        // Any non-whitespace text directly under the name means it is not a
1936        // single whole-name wrapper — leave it alone.
1937        if !child.get_content().chars().all(char::is_whitespace) {
1938          return Ok(());
1939        }
1940      },
1941      Some(NodeType::ElementNode) => {
1942        // A reference marker (`\footnotemark`/`\thanks` → `<ltx:note>`) or a trailing
1943        // `<ltx:break>` from a misused `\\` is not name content — skip it so it does
1944        // not block the unwrap of an otherwise whole-name bold. Witness: "Zhou Zhao"
1945        // in arXiv 2507.06670, `\textbf{Zhou Zhao} \footnotemark[2]`.
1946        if with(document::get_node_qname(&child), |qname| {
1947          qname == "ltx:note" || qname == "ltx:break"
1948        }) {
1949          continue;
1950        }
1951        // A `<ltx:text>` whose decoded font would serialize as EXACTLY `font="bold"` —
1952        // i.e. bold is its only departure from the default text font. Reuse the
1953        // serializer's own `font_attribute_string()` rather than hand-matching the
1954        // family/series/shape defaults, so this cannot drift from what `font=` emits.
1955        // Bold+italic / bold-sans yield "bold italic"/"… bold" ≠ "bold" and are left
1956        // untouched (extra intent).
1957        let is_pure_bold = with(document::get_node_qname(&child), |qname| {
1958          qname == "ltx:text"
1959        }) && child.get_attribute("_font").is_some_and(|fid| {
1960          document
1961            .decode_font(&fid)
1962            .is_some_and(|f| f.font_attribute_string() == "bold")
1963        });
1964        if is_pure_bold && sole_bold.is_none() {
1965          sole_bold = Some(child);
1966        } else {
1967          // A second element, or a non-pure-bold element → not a sole whole-name bold.
1968          return Ok(());
1969        }
1970      },
1971      _ => {},
1972    }
1973  }
1974  if let Some(bold) = sole_bold {
1975    document.unwrap_nodes(bold)?;
1976  }
1977  Ok(())
1978}

1980/// Read a full superscript operand as RAW (undigested) tokens for a frontmatter
1981/// author/affiliation marker (`^X` / `\textsuperscript{X}`).
1982///
1983/// A real math superscript digests its nucleus (`TeX_Math` `scriptHandler`
1984/// "invoke tokens until you get a box"), so `^\text{x}` binds `\text` to its
1985/// `{x}` and reads as one box — i.e. `^\text{x}` == `^{\text{x}}`. The hijacked
1986/// marker instead captures the operand as tokens for a matching label; the bare
1987/// `{}` read used to grab only the leading control sequence (`\text`), orphaning
1988/// its `{...}` argument. In inline math that stray `{...}` left a brace-group
1989/// frame on top, so the marker's closing `$` fired `\lx@end@inline@math` against
1990/// it — "Attempt to end mode math" — arbitrarily deep for nested
1991/// `$^\text{$...$}$` markers (html_feedback#1021, arXiv:2403.11905; both engines
1992/// erred, SHARED-FAILURE). Grabbing the control sequence together with its
1993/// following group keeps `\text{...}` — and any `$...$` nested inside it — whole
1994/// and undigested, so the surrounding math stays balanced. Numeric/char markers
1995/// (`^1`) and already-braced markers (`^{...}`, `\textsuperscript{...}`) read
1996/// exactly as before. See OXIDIZED_DESIGN #129.
1997fn read_frontmatter_sup_operand() -> Result<Tokens> {
1998  skip_spaces()?;
1999  // `^{...}` / `\textsuperscript{...}`: the whole braced group is the operand.
2000  if if_next(T_BEGIN!())? {
2001    return read_arg(ExpansionLevel::Off);
2002  }
2003  let Some(first) = read_token()? else {
2004    return Ok(Tokens::new(Vec::new()));
2005  };
2006  // `^\text{...}` (bare control-sequence operand): keep a following `{...}` group
2007  // so the CS is not severed from its argument.
2008  if first.get_catcode() == Catcode::CS && if_next(T_BEGIN!())? {
2009    let group = read_arg(ExpansionLevel::Off)?;
2010    let inner = group.unlist_ref();
2011    let mut operand = Vec::with_capacity(inner.len() + 3);
2012    operand.push(first);
2013    operand.push(T_BEGIN!());
2014    operand.extend(inner.iter().copied());
2015    operand.push(T_END!());
2016    return Ok(Tokens::new(operand));
2017  }
2018  Ok(Tokens::new(vec![first]))
2019}

2476/// Remove author creators whose `<ltx:personname>` is empty, moving any contacts they
2477/// carry to the preceding real creator (else the first following one).
2478///
2479/// A flat comma author list with interspersed `\IEEEmembership{…}`/`\thanks{…}` — e.g.
2480/// `\author{Alice, \IEEEmembership{…}, and Bob, …\thanks{…}}` (html_feedback#4539,
2481/// witness 2508.00603) — comma/" and "-splits into pieces where the membership pieces
2482/// digest to nothing, surfacing as empty `<ltx:personname/>` creators; a trailing
2483/// `\thanks` then strands its affil/email on a nameless creator. This coalesces those:
2484/// contactless empties are dropped; a contact-bearing empty's contacts move to the
2485/// preceding real author. `\footnotemark`-note markers keep a personname non-empty
2486/// (2507.06670 "Yu Zhang"), so real authors are untouched.
2487fn coalesce_empty_creators(document: &mut Document) -> Result<()> {
2488  let creators = document.findnodes("//ltx:creator[@role='author']", None);
2489  let mut to_remove: Vec<Node> = Vec::new();
2490  let mut last_real: Option<Node> = None;
2491  // Contacts of leading empties (before any real author) held until the first real one.
2492  let mut orphan_contacts: Vec<Node> = Vec::new();
2493  for creator in creators {
2494    if creator_personname_empty(document, &creator) {
2495      let contacts: Vec<Node> = creator
2496        .get_child_nodes()
2497        .into_iter()
2498        .filter(|c| {
2499          c.get_type() == Some(NodeType::ElementNode)
2500            && with(document::get_node_qname(c), |q| q == "ltx:contact")
2501        })
2502        .collect();
2503      if let Some(ref mut prev) = last_real {
2504        if !contacts.is_empty() {
2505          document.append_clone(prev, contacts)?;
2506        }
2507      } else {
2508        orphan_contacts.extend(contacts);
2509      }
2510      to_remove.push(creator);
2511    } else {
2512      if !orphan_contacts.is_empty() {
2513        let mut first = creator.clone();
2514        document.append_clone(&mut first, std::mem::take(&mut orphan_contacts))?;
2515      }
2516      last_real = Some(creator);
2517    }
2518  }
2519  for creator in to_remove {
2520    document.remove_node(creator);
2521  }
2522  Ok(())
2523}

2538/// Streaming: insert frontmatter that arrived AFTER the in-document insertion
2539/// point had already been absorbed.
2540///
2541/// Eager conversion runs every `insert_frontmatter` with digestion complete
2542/// (build starts after digestion ends), so the `\maketitle`-time insertion
2543/// sees the whole document's frontmatter. Interleaved pass 1 runs it
2544/// mid-digestion — with the REAL builder navigation, so paragraph splitting
2545/// and ids come out exactly as eager — but anything queued after that moment
2546/// (an abstract following `\maketitle` in an AMS-style document; sweep
2547/// witnesses tests/structure/amsarticle.tex, titlepage.tex) misses the
2548/// consumption. Those late arrivals re-populate the `frontmatter` state hash;
2549/// this inserts them at their canonical slot — after the last frontmatter
2550/// element already placed — by re-running `insert_frontmatter` there
2551/// (element order within it is canonical, not arrival order).
2552pub fn insert_late_frontmatter(document: &mut Document) -> Result<()> {
2553  debug_assert!(!document.root_after_open_deferred());
2554  let has_late = with_value(
2555    "frontmatter",
2556    |v| matches!(v, Some(Stored::HashTagData(h)) if !h.is_empty()),
2557  ) || with_value(
2558    "frontmatter_raw",
2559    |v| matches!(v, Some(Stored::FrontmatterRaw(q)) if !q.is_empty()),
2560  );
2561  if !has_late || !lookup_bool("frontmatter_done") {
2562    // Nothing late, or nothing was ever inserted early — in the latter case
2563    // the root's deferred late hook performs the one-and-only insertion.
2564    return Ok(());
2565  }
2566  // Anchor: the late elements' CANONICAL slot among the frontmatter already
2567  // placed — `insert_frontmatter` emits in `FRONTMATTER_ELEMENTS` order, so a
2568  // late abstract belongs between an existing date and existing keywords
2569  // (sweep witness tests/structure/amsarticle.tex), not after everything.
2570  // ROOT CHILDREN only: `//ltx:title` would also match every SECTION's
2571  // title, and the last of those anchors the insertion deep inside the body.
2572  // If several late kinds straddle an existing kind they clump at the first
2573  // late kind's slot — the byte-parity gate guards the day that case
2574  // materializes.
2575  let existing = document.findnodes(
2576    &FRONTMATTER_ELEMENTS
2577      .iter()
2578      .map(|q| format!("/ltx:document/{q}"))
2579      .collect::<Vec<_>>()
2580      .join(" | "),
2581    None,
2582  );
2583  let Some(last) = existing.last() else {
2584    return Ok(());
2585  };
2586  let rank = |qname: &str| {
2587    FRONTMATTER_ELEMENTS
2588      .iter()
2589      .position(|q| *q == qname)
2590      .unwrap_or(usize::MAX)
2591  };
2592  let min_late_rank = with_value("frontmatter", |v| match v {
2593    Some(Stored::HashTagData(h)) => h.keys().map(|k| rank(k)).min(),
2594    _ => None,
2595  })
2596  .unwrap_or(usize::MAX);
2597  let displaced = existing.iter().find(|node| {
2598    let qname = format!("ltx:{}", node.get_name());
2599    rank(&qname) > min_late_rank
2600  });
2601  let savenode = document.get_node().clone();
2602  let wrapper = match displaced {
2603    // Canonically later frontmatter exists: insert right before it.
2604    Some(next) => document.insert_element_before(next, "ltx:_Capture_", None)?,
2605    // Everything placed is canonically earlier: insert after the last.
2606    None => match last.get_next_sibling() {
2607      Some(next) => document.insert_element_before(&next, "ltx:_Capture_", None)?,
2608      None => {
2609        let mut parent = last
2610          .get_parent()
2611          .expect("a frontmatter element always has a parent");
2612        document.open_element_at(&mut parent, "ltx:_Capture_", None, None)?
2613      },
2614    },
2615  };
2616  document.set_node(&wrapper);
2617  assign_value("frontmatter_done", false, Some(Scope::Global));
2618  insert_frontmatter(document)?;
2619  document.unwrap_nodes(wrapper)?;
2620  document.set_node(&savenode);
2621  Ok(())
2622}

2624/// Insert one frontmatter entry `[tag, {attr}, @content]`.
2625/// Perl: insertFrontMatter_rec($document, $item) for the ARRAY case.
2626fn insert_frontmatter_entry(document: &mut Document, entry: &TagData) -> Result<()> {
2627  let TagData { tag, attr, content } = entry;
2628  DebugFeature!("frontmatter", "FRONT Inserting {}", show_frontmatter(entry));
2629  // token-locators: frontmatter elements (e.g. <ltx:title> from `\title{…}`)
2630  // are opened here, far from their source, around content that was digested
2631  // and stored back at `\lx@add@frontmatter` time. open_element would otherwise
2632  // stamp them with no/last locator (→ the whole-document fallback in clients).
2633  // Recover the deferred content's span and stamp the element with it.
2634  #[cfg(feature = "token-locators")]
2635  if let Some(TagContent::Box(stuff)) = content.iter().find(|c| matches!(c, TagContent::Box(_))) {
2636    document.set_current_box_locator(latexml_core::definition::constructor::child_span(stuff));
2637  }
2638  // Perl: font => $stuff[0]->getFont, _force_font => 'true' when the tag
2639  // can have a font attribute and there is content.
2640  let mut attributes: HashMap<String, String> = attr.clone();
2641  let mut font: Option<Font> = None;
2642  if !content.is_empty()
2643    && document::can_have_attribute(tag, "font")
2644    && let Some(TagContent::Box(first)) = content.first()
2645    && let Ok(Some(f)) = first.get_font()
2646  {
2647    font = Some((*f).clone());
2648    attributes.insert("_force_font".to_string(), "true".to_string());
2649  }
2650  document.open_element(tag, Some(attributes), font.as_ref())?;
2651  for item in content {
2652    insert_frontmatter_rec(document, item)?;
2653  }
2654  document.close_element(tag)?;
2655  // At this time, the frontmatter element should really carry the actual literal values intended.
2656  // (Perl PR #2767 disables the former empty-element pruning here.)
2657  Ok(())
2658}

2674/// Find all dummy frontmatter entries (role "pending") containing unattached annotations
2675/// and attempt to attach to the appropriate frontmatter, based on the identifying labels.
2676/// Perl: relocateAnnotations($document).
2677fn relocate_annotations(document: &mut Document) -> Result<()> {
2678  // Find dummy frontmatter elements containing not-yet-attached annotations
2679  let pending_nodes = document.findnodes(".//*[@role='pending']", None);
2680  if pending_nodes.is_empty() {
2681    return Ok(());
2682  }
2683  // Collect the frontmatter that have attachment labels
2684  let mut labeltable: HashMap<String, Vec<Node>> = HashMap::default();
2685  // fallback: Same, but without prefix
2686  let mut unlabeltable: HashMap<String, Vec<Node>> = HashMap::default();
2687  for target in document.findnodes(".//*[@_annotations]", None) {
2688    if target.get_attribute("role").unwrap_or_default() == "pending" {
2689      continue;
2690    }
2691    // Dedup labels PER TARGET: a creator that cites the same annotation label
2692    // from several of its authors (LLNCS `\author{A\inst{1} B\inst{2,1}}` — both
2693    // A and B cite institution 1) must receive that affiliation ONCE, not once
2694    // per citing author. Each duplicate label would otherwise push `target`
2695    // again, and the pending note gets cloned into it per push. Witness arXiv
2696    // 2603.23669: two-author creators rendered "Mila … Mila … McGill … McGill".
2697    let annotations = target.get_attribute("_annotations").unwrap_or_default();
2698    let mut seen: HashSet<&str> = HashSet::default();
2699    for label in annotations.split(',') {
2700      if label.is_empty() || !seen.insert(label) {
2701        continue;
2702      }
2703      labeltable
2704        .entry(label.to_string())
2705        .or_default()
2706        .push(target.clone());
2707      // Misuse of labelling macros can lead to prefix mismatch
2708      if let Some(pos) = label.find(':') {
2709        unlabeltable
2710          .entry(label[pos + 1..].to_string())
2711          .or_default()
2712          .push(target.clone());
2713      }
2714    }
2715  }
2716  for pending in pending_nodes {
2717    for note in element_nodes(&pending) {
2718      let label = note.get_attribute("_label").unwrap_or_default();
2719      if label.is_empty() {
2720        continue;
2721      }
2722      let noprefix = label
2723        .find(':')
2724        .map(|pos| &label[pos + 1..])
2725        .unwrap_or(label.as_str());
2726      let targets = labeltable
2727        .get(&label)
2728        .or_else(|| unlabeltable.get(&label))
2729        .or_else(|| labeltable.get(noprefix))
2730        .or_else(|| unlabeltable.get(noprefix));
2731      if let Some(targets) = targets {
2732        for target in targets.clone() {
2733          DebugFeature!("frontmatter", "FRONT Moving annotation for {label}");
2734          let mut target = target;
2735          document.append_clone(&mut target, vec![note.clone()])?;
2736        }
2737      } else {
2738        let mut known: Vec<&String> = labeltable.keys().collect();
2739        known.sort();
2740        Warn!(
2741          "unexpected",
2742          "annotation",
2743          s!("Orphaned frontmatter annotation couldn't find target for label={label}"),
2744          s!(
2745            "known labels={}",
2746            known
2747              .iter()
2748              .map(|k| k.as_str())
2749              .collect::<Vec<_>>()
2750              .join(",")
2751          )
2752        );
2753      }
2754    }
2755    document.remove_node(pending);
2756  }
2757  Ok(())
2758}

2809/// Today's date as LaTeX's `\today` renders it — `Month D, YYYY`.
2810///
2811/// Mirrors Perl `TeX_Job.pool.ltxml` L52-55, reading `\year`/`\month`/`\day`
2812/// from the **value** table where job startup put them — not from the register
2813/// meanings, which a class file is free to `\def` out from under us. A missing
2814/// value falls back to a default instead of aborting the conversion; the body
2815/// comment records that divergence and its witnesses.
2816pub fn today() -> Result<String> {
2817  let month_names = [
2818    "January",
2819    "February",
2820    "March",
2821    "April",
2822    "May",
2823    "June",
2824    "July",
2825    "August",
2826    "September",
2827    "October",
2828    "November",
2829    "December",
2830  ];
2831  // Mirror Perl TeX_Job.pool.ltxml L52-55:
2832  //   $MonthNames[LookupValue('\month')->valueOf - 1]
2833  //     . " " . LookupValue('\day')->valueOf
2834  //     . ', ' . LookupValue('\year')->valueOf
2835  // Read from the VALUE table (assigned in tex_job.rs:47-49 at job
2836  // startup), NOT the register-meaning table. `lookup_register` walks
2837  // meanings and returns None when a class file (e.g. iopart.cls)
2838  // `\def`s `\day` for its own purposes, panicking the unwrap on
2839  // 1208.0134/1702.02270/1705.08909. Intentional Perl divergence:
2840  // default to 1900-01-01 when a value is missing, where Perl would
2841  // die on `Can't call method "valueOf" on undef` — keeps the
2842  // conversion alive when something has clobbered the value table.
2843  let read = |key: &str, default: i32| -> i32 {
2844    match lookup_value(key) {
2845      Some(Stored::Int(n)) => n as i32,
2846      Some(Stored::Number(n)) => n.value_of() as i32,
2847      _ => default,
2848    }
2849  };
2850  let m = read("\\month", 1).clamp(1, 12) as usize;
2851  let month = month_names[m - 1];
2852  let day = read("\\day", 1);
2853  let year = read("\\year", 1900);
2854  Ok(s!("{} {}, {}", month, day, year))
2855}

2857pub fn parse_def_parameters(cs: &Token, params_in: Tokens) -> Result<Option<Parameters>> {
2858  let mut tokens: VecDeque<Token> = VecDeque::from(params_in.pack_parameters()?.unlist());
2859  // Now, recognize parameters and delimiters.
2860  let mut params = Vec::new();
2861  let mut n = 0;
2862  while let Some(mut t) = tokens.pop_front() {
2863    let cc = t.get_catcode();
2864    if cc == Catcode::PARAM || cc == Catcode::ARG {
2865      if cc == Catcode::PARAM {
2866        if tokens.is_empty() {
2867          // Special case: lone # NOT following a numbered parameter
2868          // Note that we require a { to appear next, but do NOT read it!
2869          params.push(Parameter::new(
2870            Cow::Borrowed("RequireBrace"),
2871            Cow::Borrowed("RequireBrace"),
2872            None,
2873          )?);
2874          break;
2875        } else {
2876          n += 1;
2877          if let Some(t_next) = tokens.pop_front() {
2878            t = t_next;
2879          } else {
2880            unreachable!("tokens.is_empty() was false, so pop_front must return Some");
2881          }
2882        }
2883      } else {
2884        // CC_ARG case, keep looking at this token
2885        n += 1;
2886      }
2887      if n > 0 {
2888        let t_num = t.with_str(|ts| ts.parse::<i8>()).unwrap_or(-1);
2889        if t_num != n {
2890          fatal!(
2891            ParamSpec,
2892            Expected,
2893            s!(
2894              "Parameters for {:?} not in order. Got {:?}, expected {:?}. in {:?}",
2895              cs,
2896              t,
2897              n,
2898              params
2899            )
2900          );
2901        }
2902      }
2903      // Check for delimiting text following the parameter #n
2904      let mut delim = Vec::new();
2905      let mut pc = Catcode::MARKER; // throwaway initial val
2906      while !tokens.is_empty() {
2907        let inner_cc = tokens.front().unwrap().get_catcode();
2908        if inner_cc == Catcode::PARAM || inner_cc == Catcode::ARG {
2909          break;
2910        }
2911        let d = tokens.pop_front().unwrap();
2912        if !(pc == Catcode::SPACE && inner_cc == Catcode::SPACE) {
2913          // BUT collapse whitespace!
2914          delim.push(d);
2915        }
2916        pc = inner_cc;
2917      }
2918      // Found text that marks the end of the parameter
2919      if !delim.is_empty() {
2920        let extra = Tokens::new(delim);
2921        params.push(
2922          Parameter {
2923            name: pin!("Until"),
2924            spec: pin(format!("Until:{extra}")),
2925            extra: vec![extra],
2926            ..Parameter::default()
2927          }
2928          .init()?,
2929        );
2930      } else if tokens.len() == 1 && tokens.front().unwrap().get_catcode() == Catcode::PARAM {
2931        // Special case: trailing sole # => delimited by next opening brace.
2932        tokens.pop_front();
2933        params.push(Parameter::new("UntilBrace", "UntilBrace", None)?);
2934      } else {
2935        // Nothing? Just a plain parameter.
2936        params.push(Parameter::new("Plain", "{}", None)?);
2937      }
2938    } else {
2939      // Initial delimiting text is required.
2940      let mut lit: Vec<Token> = vec![t];
2941      while !tokens.is_empty() {
2942        let lit_cc = tokens.front().unwrap().get_catcode();
2943        if lit_cc == Catcode::PARAM || lit_cc == Catcode::ARG {
2944          break;
2945        }
2946        lit.push(tokens.pop_front().unwrap());
2947      }
2948      let expected = Tokens::new(lit);
2949      params.push(
2950        Parameter {
2951          name: pin!("Match"),
2952          spec: pin(s!("Match:{expected}")),
2953          extra: vec![expected],
2954          novalue: true,
2955          ..Parameter::default()
2956        }
2957        .init()?,
2958      );
2959    }
2960  }
2961  // return (@params ? LaTeXML::Core::Parameters->new(@params) : undef);
2962  if params.is_empty() {
2963    Ok(None)
2964  } else {
2965    Ok(Some(Parameters::new(params)))
2966  }
2967}

3026/// Perl-faithful body-digest for VBoxContents / HBoxContents parameters.
3027/// Mirrors `readBoxContents($gullet, $everybox, $mode)` exactly:
3028/// pushes a fresh frame for `$mode`, digests tokens until matching T_END,
3029/// pops the frame.
3030///
3031/// The mode-aware variant matters when `\vtop` is invoked inside a `p{}`
3032/// column body (alignment is in horizontal mode, but the `\vtop`'s VBox
3033/// content must be digested in internal_vertical mode so `\@startpbox`'s
3034/// `\vtop\bgroup ...` doesn't try to close groups in the wrong mode).
3035/// Witness: 2210.13325 `\begin{tabular}{|p{1cm}|...}` — pre-fix Rust
3036/// inherited the surrounding horizontal mode and emitted `\vtop`
3037/// errors; Perl uses 'internal_vertical' here regardless of where the
3038/// `\vtop` was invoked.
3039pub fn predigest_box_contents_in_mode(_tokens: ArgWrap, mode: &str) -> Result<Option<Digested>> {
3040  // Perl: readBoxContents calls beginMode($mode) / endMode($mode) around the body reading.
3041  // This creates a scoped frame where enterHorizontal can change MODE inplace.
3042  // When endMode is called, leaveHorizontal_internal detects MODE='horizontal' with
3043  // BOUND_MODE ending in 'vertical', triggers repackHorizontal, then pops the frame.
3044  //
3045  // NOTE: read_box_contents already consumed the opening { or \bgroup via defined_as(T_BEGIN).
3046  // invoke_token(T_BEGIN) pushes a synthetic group frame. The matching } or \egroup
3047  // in the content will pop this frame, since \egroup is \let to T_END and
3048  // invoke_token handles it via the standard group-closing mechanism.
3049  // Perl: $stomach->beginMode($mode) — push a new frame for this box content scope
3050  // VERTICAL box modes (`\vbox`/`\vtop`/p{}-cell VBoxContents): faithful port of
3051  // Perl `readBoxContents` (TeX_Box.pool.ltxml L139-160). ONE mode frame as the
3052  // group; a loop that STOPS at the matching `}` WITHOUT processing it, so
3053  // `end_mode` runs `leave_horizontal_internal` (repack — capturing the *inner*
3054  // `\hsize`) BEFORE `pop_stack_frame` restores it. The `invoke_token(&T_BEGIN!())`
3055  // path used for other modes lets the body's `}` egroup-restore `\hsize` before
3056  // the post-hoc repack, so a `\vbox{\hsize=W …para…}` would wrap at the OUTER
3057  // `\hsize` (the Cluster G `\narrow`-hang; rotated p{}-cell mis-sizing —
3058  // SYNC_STATUS 1610.00974 step-3). hbox/math don't paragraph-wrap → the simpler
3059  // `invoke_token(T_BEGIN)` path below. (A tabular inside a `\vbox`/`\vtop` is
3060  // correctly SKIPPED by the repack because `\@@tabular`/`\halign` mark the result
3061  // box `internal_vertical`.)
3062  if mode.ends_with("vertical") {
3063    begin_mode(mode)?;
3064    let level = get_frame_depth(); // depth AFTER the mode frame (Perl: $level)
3065    new_local_box_list(); // Perl: local @LaTeXML::LIST = ()
3066    loop {
3067      let next = match get_pending_comment() {
3068        Some(comment) => Some(comment),
3069        None => read_x_token(Some(true), false, None)?,
3070      };
3071      let Some(token) = next else { break };
3072      // Perl: last if T_END && (level >= getFrameDepth). The box's own `}` (no
3073      // nested group open → depth == level) ends the loop and is closed by
3074      // `end_mode`, NOT egroup; a nested group's `}` (depth > level) is processed.
3075      if token.defined_as(&T_END!()) && level >= get_frame_depth() {
3076        break;
3077      }
3078      check_timeout()?; // runaway guard (mirrors the canonical group-digest loop)
3079      extend_box_list(invoke_token(&token)?);
3080    }
3081    // Perl: $stomach->endMode($mode) — leave_horizontal_internal (repack with the
3082    // still-in-scope inner `\hsize`) THEN pop_stack_frame (restores `\hsize`).
3083    end_mode(mode)?;
3084    let mode_tex = if lookup_bool_sym(pin!("IN_MATH")) {
3085      TexMode::Math
3086    } else {
3087      TexMode::Text
3088    };
3089    let mut digested_list = List::new(expire_local_box_list());
3090    digested_list.mode = Some(mode_tex);
3091    let mut item: Digested = digested_list.into();
3092    // Perl: List(@LaTeXML::LIST, mode => $mode)
3093    item.set_property("mode", Stored::String(pin(mode)));
3094    return Ok(Some(simplify_vertical_list(item)));
3095  }
3096  if mode.ends_with("vertical") || mode.ends_with("horizontal") {
3097    begin_mode(mode)?;
3098  }
3099  let mut contents = invoke_token(&T_BEGIN!())?;
3100  if contents.is_empty() {
3101    // Perl: $stomach->endMode($mode)
3102    if mode.ends_with("vertical") || mode.ends_with("horizontal") {
3103      end_mode(mode)?;
3104    }
3105    Ok(None)
3106  } else {
3107    let mut item = contents.remove(0);
3108    // Perl's endMode triggers leaveHorizontal_internal → repackHorizontal
3109    // when enterHorizontal changed MODE to 'horizontal' inplace within this frame.
3110    // Check the condition BEFORE endMode pops the frame.
3111    let post_mode = lookup_string_from_sym(pin!("MODE"));
3112    let bound_mode = lookup_string_from_sym(pin!("BOUND_MODE"));
3113    if post_mode == "horizontal"
3114      && bound_mode.ends_with("vertical")
3115      && has_only_simple_horizontal_content(&item)
3116    {
3117      repack_horizontal_in_list(&mut item);
3118      // Restore MODE like leave_horizontal_internal does
3119      assign_value_inplace_sym(pin!("MODE"), pin(&bound_mode));
3120    }
3121    // Perl: $stomach->endMode($mode) — pop the frame
3122    if mode.ends_with("vertical") || mode.ends_with("horizontal") {
3123      end_mode(mode)?;
3124    }
3125    // Set the mode property on the resulting item (matching Perl's List(@boxes, mode => $mode))
3126    if !mode.is_empty() {
3127      item.set_property("mode", Stored::String(pin(mode)));
3128    }
3129    // Apply Perl's List() single-item simplification for vertical modes.
3130    // In Perl, List(@boxes, mode=>'internal_vertical') returns the single box
3131    // directly when @boxes has 1 element and the box's mode is also vertical.
3132    // This is critical for nested \vbox/\vtop: the inner box's `is_vbox` property
3133    // must be visible to the outer box's constructor.
3134    Ok(Some(simplify_vertical_list(item)))
3135  }
3136}

3190///
3191/// Perl (Stomach.pm lines 442-456): In readBoxContents, after digesting box content
3192/// in vertical mode, repackHorizontal groups consecutive horizontal-mode items
3193/// from @LaTeXML::LIST into a single List(@para, mode => 'horizontal') with
3194/// width set to \hsize. This enables compute_boxes_size to do paragraph wrapping.
3195///
3196/// Without this, \vbox{hop} measures each character individually (width=5.55pt)
3197/// instead of wrapping as a paragraph at \hsize (width=469.75pt).

3200    let mut list = l.borrow_mut();

3202    let mut result: Vec<Digested> = Vec::new();
3203    let mut para: Vec<Digested> = Vec::new();

3205

3207      let child_mode = child
3208        .get_property("mode")
3209        .map(|v| v.to_string())
3210        .unwrap_or_else(|| "horizontal".to_string());
3211      // Empty-string mode means "implicit hbox" — produced by inline
3212      // brace-groups `{...}` in horizontal context. Treat as horizontal
3213      // so the surrounding running text doesn't get fragmented.
3214      let effective_mode = if child_mode.is_empty() {
3215        "horizontal"
3216      } else {
3217        child_mode.as_str()
3218      };
3219      if effective_mode == "horizontal"
3220        || effective_mode == "restricted_horizontal"
3221        || effective_mode == "math"
3222      {
3223        // Perl: $keep = 1 if ($mode ne 'horizontal') || !$item->getProperty('isSpace');
3224        if effective_mode != "horizontal" || !child.get_property_bool("isSpace") {

3227        para.push(child);
3228      } else {

3232          result.push(Digested::from(horiz_list));

3238      }
3239    }
3240    // Flush remaining horizontal items
3241    if keep {
3242      let horiz_list = make_horizontal_list(para);

3244    } else {

3246    }

3863/// A `SplitTokens` delimiter: Perl PR #2767 allows each delimiter to be a
3864/// single `Token`, OR a `Tokens` sequence to match in order.
3865#[derive(Debug, Clone)]
3866pub enum SplitDelim {
3867  /// A single Token delimiter (matched with the meaning-aware Equals below)
3868  Token(Token),
3869  /// A token sequence delimiter (matched literally, with the
3870  /// `T_SPACE` ~ `\ ` "HACK space" equivalence)
3871  Tokens(Tokens),
3872}
3873impl From<Token> for SplitDelim {
3874  fn from(t: Token) -> Self { SplitDelim::Token(t) }
3875}
3876impl From<Tokens> for SplitDelim {
3877  fn from(t: Tokens) -> Self { SplitDelim::Tokens(t) }
3878}

3880/// Does the remaining `stream` contain a `)` that balances the `(` just
3881/// popped? Used by `split_tokens` to protect delimiters inside *balanced*
3882/// parens only — an unbalanced `(` must NOT trigger paren-protection (else it
3883/// would greedily swallow the rest of the author block, including `\\`
3884/// name/affiliation separators).
3885fn paren_closes_ahead(stream: &VecDeque<Token>) -> bool {
3886  let mut level = 1usize;
3887  for t in stream {
3888    if *t == T_OTHER!("(") {
3889      level += 1;
3890    } else if *t == T_OTHER!(")") {
3891      level -= 1;
3892      if level == 0 {
3893        return true;
3894      }
3895    }
3896  }
3897  false
3898}

3900/// Perl: SplitTokens($tokens, @delims) — Base_Utility.pool.ltxml (PR #2767).
3901/// Each of `delims` is a Token, OR Tokens to match a sequence.
3902/// Returns a list of Tokens for the sub-sequences, with leading/trailing
3903/// spaces trimmed from each piece, and any empty trailing piece dropped.
3904///
3905/// Like Perl, delimiters are not matched inside `{…}` braces or `$…$` math.
3906/// As an INTENTIONAL DIVERGENCE FROM PERL, they are also not matched inside
3907/// balanced `(…)` parentheses (see the paren branch below + OXIDIZED_DESIGN).
3908pub fn split_tokens(tokens: Tokens, delims: Vec<SplitDelim>) -> Vec<Tokens> {
3909  let mut items: Vec<Tokens> = Vec::new();
3910  let mut toks: Vec<Token> = Vec::new();
3911  let trim_spaces = |toks: &mut Vec<Token>| {
3912    while toks.first().is_some_and(|x| *x == T_SPACE!()) {
3913      toks.remove(0);
3914    }
3915    while toks.last().is_some_and(|x| *x == T_SPACE!()) {
3916      toks.pop();
3917    }
3918  };
3919  if !tokens.is_empty() {
3920    let mut stream: VecDeque<Token> = VecDeque::from(tokens.unlist());
3921    while let Some(t) = stream.pop_front() {
3922      let mut matched = false;
3923      for delim in &delims {
3924        match delim {
3925          SplitDelim::Token(d) => {
3926            // Perl: Equals($t, $delim); the Rust port additionally matches by
3927            // meaning, so \AND (let to \and) matches \and as delimiter.
3928            if *d == t
3929              || (t.get_catcode() == Catcode::CS && d.get_catcode() == Catcode::CS && {
3930                let meaning_t = lookup_definition(&t).ok().flatten();
3931                let meaning_d = lookup_definition(d).ok().flatten();
3932                meaning_t.is_some() && meaning_t == meaning_d
3933              })
3934            {
3935              matched = true;
3936              break;
3937            }
3938          },
3939          SplitDelim::Tokens(seq) => {
3940            let mut tomatch: &[Token] = seq.unlist_ref();
3941            let mut peeked: Vec<Token> = Vec::new(); // tokens consumed beyond `t`
3942            let mut cur: Option<Token> = Some(t);
3943            while let (Some(c), Some(m)) = (cur, tomatch.first()) {
3944              if c == *m || (*m == T_SPACE!() && c == T_CS!("\\ ")) {
3945                // HACK space!
3946                tomatch = &tomatch[1..];
3947                if !tomatch.is_empty() {
3948                  cur = stream.pop_front();
3949                  if let Some(p) = cur {
3950                    peeked.push(p);
3951                  }
3952                }
3953              } else {
3954                break;
3955              }
3956            }
3957            if tomatch.is_empty() {
3958              matched = true;
3959              break;
3960            } else {
3961              // failed to match all: put back the peeked tokens
3962              for p in peeked.into_iter().rev() {
3963                stream.push_front(p);
3964              }
3965            }
3966          },
3967        }
3968      }
3969      if matched {
3970        trim_spaces(&mut toks);
3971        items.push(Tokens::new(std::mem::take(&mut toks)));
3972      } else if t.defined_as(&T_BEGIN!()) {
3973        toks.push(t);
3974        let mut level = 1;
3975        while let Some(t) = stream.pop_front() {
3976          match t.get_catcode() {
3977            Catcode::BEGIN => level += 1,
3978            Catcode::END => level -= 1,
3979            _ => {},
3980          }
3981          toks.push(t);
3982          if level < 1 {
3983            // done if balanced.
3984            break;
3985          }
3986        }
3987      } else if t.defined_as(&T_MATH!()) {
3988        toks.push(t);
3989        while let Some(t) = stream.pop_front() {
3990          let is_math = t.get_catcode() == Catcode::MATH;
3991          toks.push(t);
3992          if is_math {
3993            break;
3994          }
3995        }
3996      } else if t == T_OTHER!("(") && paren_closes_ahead(&stream) {
3997        // INTENTIONAL DIVERGENCE FROM PERL: also protect delimiters inside
3998        // BALANCED parentheses (mirroring the brace/math skipping above), so a
3999        // parenthesized affiliation like "(Scuola Normale Superiore, Pisa)" is
4000        // NOT split at its internal comma into a spurious second author. Perl's
4001        // SplitTokens has no paren-awareness and makes exactly this mistake
4002        // (witness arXiv 0804.0870 — "Pisa)" became a second <personname>).
4003        // The `paren_closes_ahead` guard means an UNBALANCED `(` (rare/
4004        // malformed) is treated as an ordinary token, so it never swallows a
4005        // later `\\` name/affiliation separator. See OXIDIZED_DESIGN
4006        // "Intentional divergences". NOTE: bare (unparenthesized) commas in an
4007        // affiliation ("MIT, Cambridge") remain genuinely ambiguous — the same
4008        // tokens read as either one comma-affiliation or two authors — so we
4009        // match Perl's recall-oriented over-split there rather than guess.
4010        toks.push(t);
4011        let mut level = 1;
4012        while let Some(t) = stream.pop_front() {
4013          if t == T_OTHER!("(") {
4014            level += 1;
4015          } else if t == T_OTHER!(")") {
4016            level -= 1;
4017          }
4018          toks.push(t);
4019          if level < 1 {
4020            break;
4021          }
4022        }
4023      } else {
4024        toks.push(t);
4025      }
4026    }
4027  }
4028  trim_spaces(&mut toks);
4029  if !toks.is_empty() {
4030    items.push(Tokens::new(toks));
4031  }
4032  items
4033}

4035/// If `tokens` — ignoring leading/trailing spaces — is exactly a single control
4036/// sequence applied to one brace group that spans to the end (`\cmd{ … }`),
4037/// return `(\cmd, inner)`. Used to see through a whole-line font wrapper when
4038/// splitting an author line.
4039fn whole_line_cs_wrapper(tokens: &Tokens) -> Option<(Token, Tokens)> {
4040  let v = tokens.unlist_ref();
4041  let mut start = 0;
4042  let mut end = v.len();
4043  while start < end && v[start] == T_SPACE!() {
4044    start += 1;
4045  }
4046  while end > start && v[end - 1] == T_SPACE!() {
4047    end -= 1;
4048  }
4049  let s = &v[start..end];
4050  // Need at least `\cmd { }` (the group may be empty, but then there is nothing
4051  // to split, so the caller's `len > 1` guard makes it a no-op anyway).
4052  if s.len() < 3 || s[0].get_catcode() != Catcode::CS || s[1].get_catcode() != Catcode::BEGIN {

4073/// Second-level split of the author heuristic: divide ONE line that has already
4074/// been classified as an *author* line (its superscript marker sits far enough
4075/// from the front — `Some(p)` with `p >= 8` in `\lx@add@authors`) into the
4076/// individual creators it names.
4077///
4078/// This is guarded for affiliations by that upstream classification: an
4079/// *affiliation* line (marker near the front, `p < 8`) or a continuation line
4080/// (no marker) never reaches here — they take the other arms — so name-level
4081/// separators are only ever applied to author text, never to institution names.
4082/// That is why the literal word " and " is a separator here (it splits "Alice
4083/// and Bob") yet is deliberately absent from the line-level `author_affil_splits`
4084/// (where it would shred "Language and Intelligence").
4085///
4086/// The split proceeds by hierarchy:
4087///  1. the name separators ("," and " and ") at top level; if that already
4088///     yields more than one name we are done;
4089///  2. otherwise, if the whole line is a single font wrapper `\cmd{ a, b, c }`
4090///     (Perl's `SplitTokens` can't see the brace-hidden separators, so the
4091///     wrapper collapses into ONE creator that then hoards every `$^n$` marker
4092///     as a duplicate affiliation — arXiv 2605.00347), descend through the
4093///     wrapper, split its inner name list, and re-apply `\cmd` to each name so
4094///     every author becomes its own creator with the correct affiliation.
4095///
4096/// Public so a class binding whose `\authors{...}` is a comma/"and"-separated
4097/// name list (e.g. AGU's `agujournal2019`) reuses this canonical splitter
4098/// instead of re-listing the separators — the same split `\lx@add@authors`
4099/// applies to a superscript-marked author line.
4100pub fn split_author_line(line: Tokens) -> Vec<Tokens> {
4101  // Name-level separators: comma and the literal word " and " ("Alice and Bob").
4102  let name_seps = || vec![SplitDelim::Token(T_OTHER!(",")), literal_and()];
4103  let top = split_tokens(line.clone(), name_seps());
4104  if top.len() > 1 {
4105    return top;
4106  }
4107  if let Some((cmd, inner)) = whole_line_cs_wrapper(&line) {
4108    let inner_split = split_tokens(inner, name_seps());
4109    if inner_split.len() > 1 {
4110      return inner_split
4111        .into_iter()
4112        .map(|piece| {
4113          let mut v = vec![cmd, T_BEGIN!()];
4114          v.extend(piece.unlist());
4115          v.push(T_END!());
4116          Tokens::new(v)
4117        })
4118        .collect();
4119    }
4120  }
4121  top
4122}

4247/// author/affiliation block, before the block is split into lines. The

4251/// following `\textsuperscript` past the author-vs-affiliation position

4256/// improvement (KNOWN_PERL_ERRORS #75, witness arXiv:2605.23553). The `[...]` is

4276        let mut k = j + 1;
4277        let mut depth = 1usize;

4398/// Rewrite footnote-SYMBOL author superscripts — `$^{*}$`, `${}^{\dagger}$`,
4399/// `\textsuperscript{\ddagger}`, a bare `^{*}` — onto the visible
4400/// `\lx@frontmatter@keepsup` sentinel, BEFORE author-block branch selection.
4401///
4402/// Two effects, both wanted (OXIDIZED_DESIGN #52; witness arXiv:2506.06941, where
4403/// Iman Mirzadeh's literal `$^{*}$` was silently dropped):
4404///   * the mark no longer counts as an affiliation superscript, so a block whose
4405///     ONLY superscript is such a note-mark takes the clean no-marker author
4406///     branch instead of the affiliation-linking one;
4407///   * the symbol renders as a real superscript instead of being consumed into an
4408///     `affiliation:*` label that matches no affiliation and is discarded.
4409///
4410/// NUMERIC/lettered affiliation marks (`$^{1}$`, `\textsuperscript{a}`) are left
4411/// untouched, so affiliation linking is unaffected. Only the specific
4412/// superscript-mark token shapes are matched; a superscript inside real math
4413/// (`$x^2$`) is not (its base is not empty), so math content is preserved.
4414fn rewrite_symbol_superscripts(tokens: Tokens) -> Tokens {
4415  let src = tokens.unlist();
4416  let n = src.len();
4417  let mut out: Vec<Token> = Vec::with_capacity(n);
4418  let mut i = 0;
4419  // Read a superscript operand at `src[k]`: a braced `{...}` group, or a single
4420  // token. Returns (operand tokens, index past the operand) or None.
4421  let read_operand = |k: usize| -> Option<(Vec<Token>, usize)> {
4422    if k >= n {
4423      return None;
4424    }
4425    if src[k] == T_BEGIN!() {
4426      let mut depth = 1usize;
4427      let mut m = k + 1;
4428      let mut inner = Vec::new();
4429      while m < n && depth > 0 {
4430        if src[m] == T_BEGIN!() {
4431          depth += 1;
4432        } else if src[m] == T_END!() {
4433          depth -= 1;
4434          if depth == 0 {
4435            break;
4436          }
4437        }
4438        inner.push(src[m]);
4439        m += 1;
4440      }
4441      if depth == 0 {
4442        Some((inner, m + 1))
4443      } else {
4444        None
4445      }
4446    } else {
4447      Some((vec![src[k]], k + 1))
4448    }
4449  };
4450  while i < n {
4451    let t = src[i];
4452    // `\textsuperscript{sym}` — always braced in practice.
4453    if t == T_CS!("\\textsuperscript")
4454      && src.get(i + 1) == Some(&T_BEGIN!())
4455      && let Some((sym, next)) = read_operand(i + 1)
4456      && is_footnote_symbol_operand(&sym)
4457    {
4458      out.extend(keepsup(sym));
4459      i = next;
4460      continue;
4461    }
4462    // `$ [ {} ] ^ sym $` — a math span whose whole content is a superscript on an
4463    // empty base. Rewrite the entire span (delimiters included) to a text sup.
4464    if t == T_MATH!()
4465      && let Some(close) = (i + 1..n).find(|&m| src[m] == T_MATH!())
4466    {
4467      // optional empty base `{}` before the superscript
4468      let p = if i + 2 < close && src[i + 1] == T_BEGIN!() && src[i + 2] == T_END!() {
4469        i + 3
4470      } else {
4471        i + 1
4472      };
4473      if p < close
4474        && src[p] == T_SUPER!()
4475        && let Some((sym, next)) = read_operand(p + 1)
4476        && next == close
4477        && is_footnote_symbol_operand(&sym)
4478      {
4479        out.extend(keepsup(sym));
4480        i = close + 1;
4481        continue;
4482      }
4483    }
4484    // Only the math-delimited (`$…$`) and `\textsuperscript{…}` spellings are
4485    // rewritten — the correct ways to write a text superscript. A bare `^` is NOT
4486    // matched here: outside math it is invalid LaTeX, and a `^` reached mid-scan is
4487    // the superscript operator of real inline math on a non-empty base (`$a^{*}$`),
4488    // which must be left intact.
4489    out.push(t);
4490    i += 1;
4491  }
4492  Tokens::new(out)
4493}

4536/// Split an affiliation blob at each embedded superscript mark (`$^{n}$` /
4537/// `\textsuperscript{n}`), keeping the mark at the head of the following
4538/// segment. The abuse idiom packs several affiliations onto ONE line delimited
4539/// only by their marks (no `\\`), so `affil_splits()` (which splits on `\\`)
4540/// cannot separate them. Each returned segment then flows through
4541/// `\lx@affiliation@withsup`, which turns its leading mark into the
4542/// `affiliation:N` label that the authors' own marks already request
4543/// (`relocate_annotations` links them).
4544fn split_before_affiliation_marks(tokens: Tokens) -> Vec<Tokens> {

4546  // affiliations are comma-joined (`\textsuperscript{1}Univ A, \textsuperscript{2}
4547  // Univ B`) rather than space-joined: without this the affiliation contact reads
4548  // "Univ A," with a stray trailing comma (html_feedback#6588, arXiv:2606.01317).
4549  // Only trailing whitespace + a SINGLE trailing comma are removed, so a comma
4550  // INSIDE an institution name ("Dept X, Univ Y, City") is untouched.
4551  fn trim_trailing_separator(mut toks: Vec<Token>) -> Tokens {
4552    while toks.last() == Some(&T_SPACE!()) {
4553      toks.pop();
4554    }
4555    if toks.last() == Some(&T_OTHER!(",")) {
4556      toks.pop();
4557      while toks.last() == Some(&T_SPACE!()) {
4558        toks.pop();
4559      }
4560    }
4561    Tokens::new(toks)
4562  }
4563  let toks = tokens.unlist();
4564  let mut segments: Vec<Tokens> = Vec::new();
4565  let mut current: Vec<Token> = Vec::new();
4566  for (i, t) in toks.iter().enumerate() {

4575    if is_mark_start && current.last() == Some(&T_SPACE!()) {
4576      segments.push(trim_trailing_separator(std::mem::take(&mut current)));

4582  }

4586/// A `\\`-delimited line in a no-marker author block is really an EMAIL, not an

4607/// A marker-less line that is *purely* a list of email addresses — a shared
4608/// `\texttt{a@x, b@y,}` / `\email{...}` line covering all authors. Like

4610/// comma authors often leave): every non-empty comma item must itself carry '@'.

4615/// Parse an email line into its individual address strings, or `None` if the line
4616/// is not an email list. Two forms are recognized (both after skipping wrapper
4617/// commands / braces — only visible letter/other/space chars are read):
4618/// - **Distributed** — `a@x, b@y, c@z`: every comma item carries its own `@`.
4619/// - **Grouped brace-expansion** — `{a,b,c}@dom` (flattens to `a,b,c@dom`): only
4620///   the LAST item carries `@`; the earlier bare local-parts are each expanded
4621///   against the shared domain → `a@dom, b@dom, c@dom`. This is the compact form
4622///   co-located authors use for one shared address per person.
4623///
4624/// A prose affiliation line ("Dept. of Foo, University of Pisa, Italy") has no `@`
4625/// and is rejected, so an address is never misclassified.
4626fn email_addresses(line: &Tokens) -> Option<Vec<String>> {
4627  let mut visible = String::new();
4628  for t in line.unlist_ref() {
4629    match t.code {
4630      Catcode::LETTER | Catcode::OTHER => t.with_str(|s| visible.push_str(s)),
4631      Catcode::SPACE => visible.push(' '),
4632      _ => {},
4633    }
4634  }
4635  let v = visible.trim();
4636  if !v.contains('@') {
4637    return None;
4638  }
4639  let parts: Vec<&str> = v
4640    .split(',')
4641    .map(str::trim)
4642    .filter(|p| !p.is_empty())
4643    .collect();
4644  if parts.is_empty() {
4645    return None;
4646  }
4647  let has_at = |p: &str| p.contains('@') && !p.chars().any(char::is_whitespace);
4648  // Distributed: every item is a full address.
4649  if parts.iter().all(|p| has_at(p)) {
4650    return Some(parts.iter().map(|s| (*s).to_string()).collect());
4651  }
4652  // Grouped: only the last item carries the domain; earlier items are bare
4653  // local-parts expanded against it (`{a,b,c}@dom` → a@dom, b@dom, c@dom).
4654  if let Some((last, heads)) = parts.split_last()
4655    && has_at(last)
4656    && heads
4657      .iter()
4658      .all(|p| !p.contains('@') && !p.chars().any(char::is_whitespace))
4659    && let Some(at) = last.find('@')
4660  {
4661    let domain = &last[at..];
4662    let mut out: Vec<String> = heads.iter().map(|p| format!("{p}{domain}")).collect();
4663    out.push((*last).to_string());
4664    return Some(out);
4665  }
4666  None
4667}

4720/// Support for Key / Value arguments.
4721// The very basic form is
4722//   RequiredKeyVals: $keyset
4723//   OptionalKeyVals: $keyset
4724// to parse Key-Value pairs from a given keyset (see the 'keyval' package
4725// documentation for more information). These types of KeyVal
4726// parameters will return a LaTeXML::Core::KeyVals object, which can then be
4727// used to access the values of the individual items.
4728// The difference between the two forms is that RequiredKeyVals expects a set of
4729// key-value pairs wrapped in T_BEGIN T_END, where as OptionalKeyVals optionally
4730// expects a set of KeyValue pairs wrapped in T_OTHER('[') T_OTHER(']')
4731//
4732// Several extension of the keyval package exist, the most common one we support
4733// is the xkeyval package. This introduces further variations on the keyval
4734// arguments parsing, in particular it allows to read keys from more than one
4735// keyset at once. These can be specified by giving comma-separated values in
4736// the keyset argument. By default, a key will only be set in the **first**
4737// keyset it occurs in. By using
4738//   RequiredKeyVals+: $keysets
4739//   OptionalKeyVals+: $keysets
4740// the key will be set in all keysets instead.
4741//
4742// All keys to be parsed with these arguments should be declared using
4743// DefKeyVal in LaTeXML::Package. By default, an error is thrown if an unknown
4744// key is encountered. To surpress this behaviour, and instead store all
4745// undefined keys, use
4746//   RequiredKeyVals*: $keysets
4747//   OptionalKeyVals*: $keysets
4748// instead. The '*' and '+' modifiers can be combined by using:
4749//   RequiredKeyVals*+: $keysets
4750//   OptionalKeyVals*+: $keysets
4751//
4752// Furthermore, the xkeyval package supports giving prefixes to keys,
4753//   RequiredKeyVals[*][+]: $prefix|$keysets
4754//   OptionalKeyVals[*][+]: $prefix|$keysets
4755//
4756// Finally, it is possible to specify specific keys to skip when digesting the
4757// object. This can be achieved using comma-separated key values in
4758//   RequiredKeyVals[*][+]: $prefix|$keysets|$skip
4759//   OptionalKeyVals[*][+]: $prefix|$keysets|$skip
4760
4761// function to handle all the
4762#[derive(Default)]
4763pub struct KVSpec {
4764  pub star:    bool,
4765  pub plus:    bool,
4766  pub prefix:  Option<String>,
4767  pub keysets: Vec<String>,
4768  pub skip:    Vec<String>,
4769}
4770pub fn keyvals_aux(until: Option<Token>, spec: KVSpec) -> Result<KeyVals> {
4771  let KVSpec {
4772    mut star,
4773    plus,
4774    mut prefix,
4775    mut keysets,
4776    skip,
4777  } = spec;
4778  // support both "keysets" and "prefix|keysets"
4779  if keysets.is_empty() {
4780    if let Some(pfx) = prefix.take() {
4781      keysets = vec![pfx];
4782    }
4783
4784    // to emulate old behaviour, throw no errors
4785    // when we have a single keyset and no prefix (or no keyset at all)
4786    if keysets.is_empty() {
4787      star = true;
4788    }
4789  }
4790
4791  // create a new set of Key-Value arguments
4792  let mut keyvals = KeyVals::new(KeyvalsConfig {

4794    keysets,

4796    set_internals: true,

4800    } else {

4811}

4915pub fn make_generic_message(cmd: &str, args: Vec<Tokens>, kind: &str) -> Result<()> {
4916  // Faithful port of Perl latex_constructs.pool.ltxml `make_message`
4917  // (L5569-5588). The first arg is the message PREFIX: its ToString,
4918  // stripped of `\@spaces`/`\space` runs, becomes the diagnostic `what`
4919  // ($type), falling back to the command name when the prefix is empty
4920  // (e.g. `\@latex@error`'s all-`\space` prefix). The REMAINING args form
4921  // the body — Perl `join(" ", map { ToString(Expand($_, \@spaces,
4922  // \@spaces)) } @args)`. Earlier the Rust port discarded the prefix and
4923  // pinned `what` to the command name, so package/class diagnostics lost
4924  // their `(pkgname)` $type and `\GenericWarning`/`\GenericInfo` even
4925  // leaked the prefix into the body. Witness: hep-th-style `\PackageError`
4926  // /`\GenericWarning` now match Perl byte-for-byte (`(mypkg)` $type, no
4927  // prefix in body).
4928  let mut args = args.into_iter();
4929  let lead = args.next().unwrap_or_default();
4930  // $type = ToString(lead) with `\@?spaces?` runs removed (Perl regex
4931  // `s/(?:\\\@?spaces?)+//g`); fall back to the command name when empty.
4932  let type_str = strip_message_spaces(&lead.to_string());
4933  let type_str: &str = if type_str.is_empty() { cmd } else { &type_str };
4934
4935  bgroup();
4936  let_i(&T_CS!("\\protect"), &T_CS!("\\string"), None);
4937  let_i(
4938    &T_CS!("\\MessageBreak"),
4939    &T_CS!("\\ltx@hard@MessageBreak"),
4940    None,
4941  ); // tricky, we need Expand() to execute it
4942  let mut parts: Vec<String> = Vec::new();
4943  for arg in args {
4944    let mut arg_toks = arg.unlist();
4945    // Perl `make_message` pads each Expand input with two `\@spaces` tokens:
4946    //   ToString(Expand($_, T_CS('\@spaces'), T_CS('\@spaces')))
4947    // Comment in Perl: "expand padded with two \@spaces, to avoid pointless
4948    // errors. e.g. a trailing `\csq@noline` from csquotes.sty (let to
4949    // `\@gobble`) will produce a `gobble has no argument` error on every
4950    // message." Same fix needed here for etoc.sty's
4951    // `\PackageWarning{etoc}{...!\@gobbletwo}` — `\@gobbletwo` would otherwise
4952    // consume the do_expand-inserted T_END closer, leaving readBalanced
4953    // unbalanced.
4954    arg_toks.push(T_CS!("\\@spaces"));
4955    arg_toks.push(T_CS!("\\@spaces"));
4956    parts.push(Expand!(arg_toks).to_string());
4957  }
4958  // Perl joins the body args with a single space (the `\MessageBreak`s
4959  // *within* an arg already became hard newlines via the let above).
4960  let message = parts.join(" ");
4961
4962  egroup()?;
4963  // Downgrade vendor-class typesetting-only errors to Info. Publisher
4964  // classes routinely guard line widths, header heights, and other
4965  // PDF-layout concerns with `\PackageError`/`\GenericError`. We
4966  // produce XML/HTML, not PDF — these guards have no semantic value
4967  // in our output. See WISDOM #50 and
4968  // memory/feedback_size_layout_errors_moot.md.
4969  let effective_kind = if kind == "error" && is_typesetting_only_message(&message) {
4970    "info"
4971  } else {
4972    kind
4973  };
4974  //   return ('latex', $type, $stomach, $message);
4975  match effective_kind {
4976    "error" => {
4977      Error!("latex", type_str, message);
4978    },
4979    "warn" => {
4980      Warn!("latex", type_str, message);
4981    },
4982    "info" => {
4983      Info!("latex", type_str, message);
4984    },
4985    _other => panic!("Only call make_generic_message with error|warn|info message kinds."),
4986  };
4987  Ok(())
4988}
