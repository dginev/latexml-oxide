11  //======================================================================
12  // Basics
13  //----------------------------------------------------------------------
14  // \begingroup       c  starts a group that must be ended by \endgroup.
15  // \endgroup         c  ends a group that was begun by \begingroup.
16  // \relax            c  is a control sequence which typesets nothing.
17  // \afterassignment  c  saves a token and inserts it after the next assignment.
18  // \aftergroup       c  saves a token and inserts it after the current group is complete.
19  DefPrimitive!("\\begingroup", {
20    begingroup();
21  });
22  DefPrimitive!("\\endgroup", {
23    endgroup()?;
24  });
25  // This makes \relax disappear completely after digestion
26  // (which seems most TeX like).
27  DefPrimitive!(T_CS!("\\relax"), None, {});
28  // \protect is used in LaTeX for robust commands; acts like \relax in the base engine.
29  Let!("\\protect", "\\relax");
30  //## However, this keeps a box, so it can appear in UnTeX
31  //## DefPrimitive('\relax',undef);
32  //# But if you do that, you've got to watch out since it usually
33  //## shouldn't be a box; See the isRelax code in handleScripts, below
34
35  // NON-STANDARD: Internal token produced by Gullet in response to \dont_expand;
36  // Acts like \relax, but isn't equal to it.
37  DefPrimitive!(T_CS!("\\special_relax"), None, {});
38
39  // \afterassignment saves ONE token (globally!) to execute after the next assignment
40  DefPrimitive!("\\afterassignment Token", sub[(t)] {
41    assign_value("afterAssignment", t, Some(Scope::Global));
42  });
43  // \aftergroup saves ALL tokens (from repeated calls) to be executed IN ORDER after the next
44  // egroup or }
45  DefPrimitive!("\\aftergroup Token", sub[(t)] { push_value("afterGroup", t) });

48  // CSName
49  //----------------------------------------------------------------------
50  // \csname           c  forms a control sequence name from the characters making up a collection
51  // of tokens. \endcsname        c  is used with \csname to make a control sequence name.
52  DefParameterType!(CSName, reader => reader!( _inner, _extra, { read_cs_name() }));
53  // Quiet version: used by \ifcsname — no errors for non-expandable CS tokens
54  DefParameterType!(CSNameQuiet, reader => reader!( _inner, _extra, { read_cs_name_quiet() }));
55  DefMacro!("\\csname CSName", sub[(token)] {
56    if !has_meaning(&token) {
57      let relax_meaning = lookup_meaning(&TOKEN_RELAX).unwrap();
58      assign_meaning(&token, relax_meaning, None);
59    }
60    token
61  });
62
63  DefPrimitive!("\\endcsname", {
64    if !lookup_bool("SUPPRESS_UNEXPECTED_ERRORS") {
65      Error!("unexpected", "\\endcsname", "Extra \\endcsname");
66    }
67  });

70  // Definition flags
71  //----------------------------------------------------------------------
72  // \global        c  is an assignment prefix which makes the assignment transcend its group.
73  // \long          c  is a prefix for definitions which require multi-paragraph arguments.
74  // \outer         c  is a prefix for a definition which restricts where the definition may be
75  // used. \globaldefs    pi if positive, all assignments are global; if negative, \global is
76  // ignored.
77
78  DefPrimitive!("\\global",{ set_prefix("global"); }, is_prefix => true);
79  DefPrimitive!("\\long",  { set_prefix("long");   }, is_prefix => true);
80  DefPrimitive!("\\outer", { set_prefix("outer");  }, is_prefix => true);
81
82  DefRegister!("\\globaldefs", Number!(0));
83  //======================================================================
84  // Definitions
85  //----------------------------------------------------------------------
86  // \def          c  defines a macro.
87  // \edef         c  is similar to \def, except control sequences in the replacement
88  //                  text are expanded when the definition is made.
89  // \gdef         d  is equivalent to `\global\def'.
90  // \xdef         d  is equivalent to `\global\edef'.
91  DefPrimitive!("\\def RedefinableToken UntilBrace DefPlain",
92    sub[(cs,params,body)] {
93      do_def(false,cs,params,body)?;
94    },
95    locked => true);
96  DefPrimitive!("\\gdef RedefinableToken UntilBrace DefPlain",
97    sub[(cs,params,body)] {
98      do_def(true,cs,params,body)?;
99    },
100    locked => true);
101  DefPrimitive!("\\edef RedefinableToken UntilBrace DefExpanded",
102    sub[(cs,params,body)] {
103      do_def(false,cs,params,body)?;
104    },
105    locked => true);
106  DefPrimitive!("\\xdef RedefinableToken UntilBrace DefExpanded",
107    sub[(cs,params,body)] {
108      do_def(true,cs,params,body)?;
109    },
110    locked => true);
111  //======================================================================
112  // Copying definitions
113  //----------------------------------------------------------------------
114  //  \let       c  gives a control sequence a token's current meaning.
115  // \futurelet  c  `<cs> <token1> <token2>' is equivalent to `\let <cs> = <token2> <token1>
116  // <token2>'.
117  // `\let` — TeXbook Ch 24:
118  //   `\let <cs> <equals> <one optional space> <token>`
119  // where `<equals> = <optional spaces> | <optional spaces> =`,
120  // and `<one optional space>` is exactly one space if present.
121  //
122  // Perl uses the parameter spec
123  //   `\let SkipSpaces Token SkipSpaces SkipMatch:= Skip1Space Token`
124  // which in our engine routed the optional-`=` consumption through
125  // the generic Match parameter type's unread-on-no-match recovery.
126  // The reads are tex.web §1221's instead: `repeat get_token until
127  // cur_cmd<>spacer`, and after an `=` one more `get_token`, repeated once
128  // if it is a space. A space there is explicit or implicit (`cur_cmd`), so
129  // `\let\c\@sptoken=\b` lets `\c` to `\b` (pdflatex; Rust let it to `=`).
130  // Each read is the primitive's `get_token`, which at a file's end goes on
131  // in the enclosing input (`read_primitive_token`): a file ending in
132  // `\let\y=` lets `\y` to the next token after it. The name is §1215
133  // `get_r_token` (`RedefinableToken`), which skips spaces, also after a file
134  // that ends in a bare `\let`.
135  DefPrimitive!("\\let RedefinableToken", sub[(token1)] {
136    let is_spacer = |t: &Option<Token>| t.as_ref().is_some_and(is_space_or_implicit_space);
137    let mut token2 = read_primitive_token()?;
138    while is_spacer(&token2) {
139      token2 = read_primitive_token()?;
140    }
141    if token2.is_some_and(|t| t.get_catcode() == Catcode::OTHER && t.text == pin!("=")) {
142      token2 = read_primitive_token()?;
143      if is_spacer(&token2) {
144        token2 = read_primitive_token()?;
145      }
146    }
147    // TeX's `\let`, never declined by a provide-only load (`let_definition`).
148    let_i(&token1, &token2.unwrap_or_else(|| T_CS!("\\relax")), None);
149  });
150  DefPrimitive!("\\futurelet RedefinableToken Token Token", sub[(cs, token1, token2)] {
151    // NOT expandable, but puts tokens back
152    unread(Tokens!(token1,token2));
153    let_i(&cs, &token2, None);
154  });
155  //======================================================================
156  // Expansion control
157  //----------------------------------------------------------------------
158  // \expandafter      c  `<token1><token2>' is equivalent to `<token1> expansion of <token2>'.
159  // \noexpand         c  prevents the expansion of the following token.
160  //
161  // The two tokens are read by the primitive itself (tex.web §368 `get_token;
162  // t:=cur_tok; get_token`), at the scanner status of the scan it expands in,
163  // not as a macro's arguments (`matching`): at `normal` status a file's end
164  // is crossed, so a file ending in `\expandafter\x` expands the first token
165  // after it (`read_primitive_token`).
166  DefMacro!(T_CS!("\\expandafter"), None, {
167    let tok = read_primitive_token()?.unwrap_or_else(|| T_CS!("\\relax"));
168    let mut xtok = read_primitive_token()?.unwrap_or_else(|| T_CS!("\\relax"));
169    let mut skipped: Vec<Token> = vec![tok];
170    while xtok.defined_as(&TOKEN_EXPANDAFTER) {
171      if let Some(ntok) = read_primitive_token()? {
172        skipped.push(ntok);
173        if let Some(nxtok) = read_primitive_token()? {
174          xtok = nxtok;
175        } else {
176          // Stream ended mid-chain (an isolated argument mouth ran out —
177          // csquotes' `\csq@fixkern` 7-chain firing on a quote char that
178          // was the LAST token of a constructor argument, biblatex.tex
179          // L6575). Leaving `xtok` = \expandafter here spun this while
180          // loop against the exhausted mouth, one Error per lap to the
181          // TooManyErrors cap (~96 phantom errors + Fatal). Emit ONE
182          // error and stop — the missing-token recovery mirrors the
183          // undefined-`xtok` stub arm below. (Perl instead crashes:
184          // "Can't call defined_as on undefined", TeX_Macro.pool L207.)
185          Error!(
186            "expected",
187            "expandafter",
188            "\\expandafter wrongly used without 2 arguments."
189          );
190          retract_scanned_braces(&skipped);
191          return Ok(Tokens::new(skipped));
192        }
193      } else {
194        Error!(
195          "expected",
196          "expandafter",
197          "\\expandafter wrongly used without 2 arguments."
198        );
199        retract_scanned_braces(&skipped);
200        return Ok(Tokens::new(skipped));
201      }
202    }
203    // The saved tokens were READ (and brace-counted); they re-enter the
204    // stream inside our expansion result, so retract them — tex.web §368
205    // `expand`s the second token FIRST and only then `back_input`s the saved
206    // one (§325 retracts a scanned brace). The retraction must therefore
207    // follow the one-step expansion: `\exp_after:wN { \use_none:nn & …}`
208    // (numerica.sty:1748 `\__nmc_delim_arg:`, l3fp; mhchem, tablists-rus)
209    // has `\use_none:nn` READ the `&` while the saved `{` still counts, so
210    // the tab is brace-nested; retracting before the expansion put the
211    // ledger at 0 and the `&` fired the column template mid-cell (the
212    // 6-error "close a group that switched to mode math" cascade, numerica
213    // 83, tablists-rus 101, mhchem 14; Perl identical). Without any
214    // retraction the idiom double-counts and the ledger drifts positive.
215    // Guard: `perfect_kernel_batch54::argument_scan_is_align_state_neutral`.
216    // A peeking macro met by `\expandafter` inside a number scan stays
217    // unexpanded (`gullet::scan_stops_at_futurelet_peek`): TeX's one-level
218    // expansion reaches latex.ltx's unexpandable `\let` head, which ends the
219    // scan (`\count@=1\expandafter\empty\@ifstar{7}{5}*` assigns 1, then "7").
220    // Guard: `ifnextchar_scans::expandafter_in_a_scan_stops_at_a_peek`.
221    match lookup_expandable(&xtok, None)? {
222      Some(defn) if !scan_stops_at_futurelet_peek(&defn) => {
223        local_current_token(xtok);
224        let invoked = defn.invoke(true)?;
225        expire_current_token();
226        retract_scanned_braces(&skipped);
227        if !invoked.is_empty() {
228          skipped.extend(invoked.unlist()); // Expand `xtok` ONCE ONLY!
229        }
230      },
231      _ => {
232        if !has_meaning(&xtok) {
233          retract_scanned_braces(&skipped);
234          // Undefined token is an error, as expansion is expected.
235          // BUT The unknown token is NOT consumed, (see TeX B book, item 367)
236          // since probably in a real TeX run it would have been defined.
237          generate_error_stub(&xtok)?;
238          retract_scanned_brace(&xtok);
239          skipped.push(xtok);
240        } else {
241          // Unexpandable `xtok` (e.g. `\expandafter A {`): it was read and
242          // counted, and re-enters via our expansion — retract like the prefix.
243          retract_scanned_braces(&skipped);
244          retract_scanned_brace(&xtok);
245          skipped.push(xtok);
246        }
247      },
248    };
249    Ok(Tokens::new(skipped))
250  });
251  // If next token is expandable, prefix it with the internal marker \dont_expand
252  // That token is never defined, explicitly handled in Gullet & should never escape the Gullet.
253  //
254  // Mirrors Perl `TeX_Macro.pool.ltxml:228-235`: uses `isDontExpandable`
255  // (not `lookupExpandable`). The predicate is true for CS/Active that are
256  // expandable OR undefined — so `\noexpand` smuggles `\dont_expand` for
257  // both, which is what makes `\noexpand\undef` ≠ `\undef` under `\ifx`
258  // (Gullet rewrites `\dont_expand X` into `\special_relax` with `X`
259  // smuggled in slot[2]; that asymmetry is the whole point).
260  DefMacro!(T_CS!("\\noexpand"), None, {
261    // tex.web §367: the token is read under normal scanner status, so it may
262    // be the first token after the end of the current input level
263    // (`\everyeof{\noexpand}`; `read_token_across_input_ends`).
264    if let Some(token) = read_token_across_input_ends()? {
265      let cc = token.get_catcode();
266      if matches!(cc, Catcode::CS | Catcode::ACTIVE) && is_dont_expandable(&token) {
267        vec![T_CS!("\\dont_expand"), token]
268      } else {
269        // `\noexpand{`: the brace was read (counted) and re-enters via our
270        // expansion — retract, tex.web §369 back_input flavor.
271        retract_scanned_brace(&token);
272        vec![token]
273      }
274    } else {
275      // The end of a `reading_from_mouth` context (e.g. "{\noexpand}" read as
276      // an argument), for which TeX would be unperturbed.
277      Vec::new()
278    }
279  });
280  // NON-STANDARD:
281  DefPrimitive!(T_CS!("\\dont_expand"), None, {
282    Error!(
283      "misdefined",
284      "\\dont_expand",
285      "The token \\dont_expand should never reach Stomach!"
286    );
287  });

289  //======================================================================
290  // \the
291  //----------------------------------------------------------------------
292  // \the              c  returns character tokens for an internal quantity's or parameter's current
293  // value. The argument to \the is a variety of "Internal Quantities", being parameters,
294  // registers, internal registers, codenames, etc. See TeX Book, pp.214--215.
295  // [Since \the is expandable, perhaps should just be built into \the's code? Never need to revert]
296  DefMacro!("\\the", {
297    if let Some(token) = read_x_token(None, false, None)? {
298      // Follow `\let`-alias chain so that `\the \tex_count:D` (= `\count`)
299      // and `\the \__int_eval:w` (= `\numexpr`) — and any other expl3
300      // primitive-alias to a register-style CS — find the underlying
301      // register definition. Without this, `lookup_definition` returns
302      // None for `Stored::Token` let-aliases and `\the` falls through
303      // to the "undefined" branch, emitting a 0 plus a stub-installation.
304      // Driver: 2406.14142 expl3 regex VM (`\the \__int_eval:w ... \__int_eval_end:`).
305      let mut effective_token = token;
306      for _ in 0..16 {
307        match with_meaning(&effective_token, |m| match m {
308          Some(Stored::Token(t)) => Some(*t),
309          _ => None,
310        }) {
311          Some(t) if t != effective_token => effective_token = t,
312          _ => break,
313        }
314      }
315      match lookup_definition(&effective_token)? {
316        Some(defn) => {
317          if defn.is_register() {
318            // SOME kind of register is acceptable
319            let args = if let Some(params) = defn.get_parameters() {
320              params.read_arguments(None)?
321            } else {
322              Vec::new()
323            };
324            return match defn.value_of(args) {
325              Some(RegisterValue::Token(t)) => Ok(Tokens!(t)),
326              Some(RegisterValue::Tokens(ts)) => Ok(ts),
327              Some(other) => Ok(Tokens!(Explode!(other.to_string()))),
328              None => Ok(Tokens!()),
329            };
330          } else if defn.get_cs_name() == "\\font" {
331            // Perl's HACK (TeX_Macro.pool.ltxml:272) answers the last `\font`
332            // identifier invoked, else `\lx@default@font`; TeX answers the
333            // current font's identifier (tex_fonts.rs `current_font_identifier`).
334            return Ok(Tokens!(crate::tex_fonts::current_font_identifier()?));
335          } else {
336            // Perl: elsif ($defn->isFontDef) { return $defn->getCS; }
337            // Check if this is a font CS defined by \font (has fontinfo in state)
338            let cs_str = token.to_string();
339            if let Some(Stored::Font(_)) = lookup_value(&s!("fontinfo_{cs_str}")) {
340              return Ok(Tokens!(token));
341            }
342            // A primitive that is an INTERNAL QUANTITY with its own assignment
343            // syntax (no `=`, no register type) publishes its `\the` value
344            // under `the_value_<cs>`: luatex's direction primitives
345            // (`\the\bodydir` → `TLT`; `latexml_sty` under the luatex
346            // profile), whose `scan_direction` argument the register
347            // machinery cannot read.
348            if let Some(Stored::String(v)) = lookup_value(&s!("the_value_{cs_str}")) {
349              return Ok(Tokens::new(Explode!(with(v, |s| s.to_string()))));
350            }
351          }
352        },
353        _ => {
354          // the token is Undefined
355          if token.get_catcode() == Catcode::CS {
356            // but IS a cs \something
357            Error!(
358              "expected",
359              "<register>",
360              "A <register> was supposed to be here",
361              s!("Got {} Defining it now.", token)
362            );
363            // Hackery: to avoid potential repeated errors, define it now as a number register
364            def_register(token, None, Number!(0), None)?; // Dimension, or what?
365            return Ok(Tokens!(T_OTHER!("0")));
366          }
367        },
368      }
369      // If we fall through to here, whatever $token is shouldn't have been used with \the
370      let (the_t, msg) =
371        token.with_str(|tstr| (s!("\\the{tstr}"), s!("You can't use {tstr} after \\the")));
372      Error!("unexpected", the_t, msg);
373      T_OTHER!("0")
374    } else {
375      Error!(
376        "expected",
377        "<register>",
378        "A <register> was supposed to be here. Got nothing."
379      );
380      T_OTHER!("0")
381    }
382  });
