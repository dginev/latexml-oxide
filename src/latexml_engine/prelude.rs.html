145// Functions callable from constructor templates via &GetKeyVal(#1,key) syntax.
146// Returns Option<Digested> to be compatible with both attribute (to_attribute/to_string)
147// and body absorption (Into<Option<Digested>>) contexts in constructor templates.
148#[allow(non_snake_case)]
149pub fn GetKeyVal(keyval_opt: &Option<Digested>, key: &str) -> Option<Digested> {
150  match keyval_opt {
151    Some(digested) => match digested.data() {
152      DigestedData::KeyVals(keyval) => keyval.get_value_digested(key).cloned(),
153      _ => None,
154    },
155    _ => None,
156  }
157}

161// (txfonts, mathabx, amssymb, math_common, biblatex, beamer, …).

170/// Whether the LaTeX counter `name` exists: latex.ltx's `\@ifundefined{c@#1}`
171/// test in `\setcounter`/`\addtocounter` (:10115-10122).
172pub fn counter_is_defined(name: &str) -> bool {
173  lookup_register_definition(&T_CS!(s!("\\c@{name}"))).is_some()
174}

176/// Empty-body `DefMacro!("\\cs[opt-spec]", "")` stub via runtime call.
177pub fn def_macro_noop(proto: &str) -> Result<()> {
178  let (cs_tok, params) = parse_prototype(proto, true)?;
179  let body = mouth::tokenize_internal("");
180  def_macro(cs_tok, params, ExpansionBody::Tokens(body), None)?;
181  Ok(())
182}

184/// Identity `DefMacro!("\\cs{}", "#1")` — CS takes one mandatory arg
185/// and expands to it unchanged.
186pub fn def_macro_identity(proto: &str) -> Result<()> {
187  let (cs_tok, params) = parse_prototype(proto, true)?;
188  let body = mouth::tokenize_internal("#1");
189  def_macro(cs_tok, params, ExpansionBody::Tokens(body), None)?;
190  Ok(())
191}

193/// Empty-body `DefPrimitive!("\\cs[opt-spec]", None);` stub — Perl's undef body: an empty box
194/// reverting to the call, its arguments included (Primitive.pm:74-76).
195pub fn def_primitive_noop(proto: &str) -> Result<()> {
196  let (cs_tok, params) = parse_prototype(proto, true)?;
197  def_primitive(cs_tok, params, None, PrimitiveOptions::default())?;
198  Ok(())
199}

201/// DEP-17 `DefMath!("\\cs", "char"[, role => "X"[, meaning => "Y"]])`
202/// shape. 2-arg form: prototype includes the CS; params come from
203/// `parse_prototype` (Some(empty)).
204pub fn def_math_sym(
205  cs: &str,
206  present: &str,
207  role: Option<&str>,
208  meaning: Option<&str>,
209) -> Result<()> {
210  let (cs_tok, params) = parse_prototype(cs, true)?;
211  let mut opts = MathPrimitiveOptions::default();
212  if let Some(r) = role {
213    opts.role = Some(r.to_string());
214  }
215  if let Some(m) = meaning {
216    opts.meaning = Some(m.to_string());
217  }
218  def_math(cs_tok, params, present.to_string(), opts)?;
219  Ok(())
220}

222/// DEP-17d `DefMath!("\\cs", None, "char"[, ...])` 3-arg form —
223/// Token built directly via `T_CS!`, params stays None.
224pub fn def_math_atom(
225  cs: &str,
226  present: &str,
227  role: Option<&str>,
228  meaning: Option<&str>,
229) -> Result<()> {
230  let cs_tok = T_CS!(cs);
231  let mut opts = MathPrimitiveOptions::default();
232  if let Some(r) = role {
233    opts.role = Some(r.to_string());
234  }
235  if let Some(m) = meaning {
236    opts.meaning = Some(m.to_string());
237  }
238  def_math(cs_tok, None, present.to_string(), opts)?;
239  Ok(())
