120// Functions callable from constructor templates via &GetKeyVal(#1,key) syntax.
121// Returns Option<Digested> to be compatible with both attribute (to_attribute/to_string)
122// and body absorption (Into<Option<Digested>>) contexts in constructor templates.
123#[allow(non_snake_case)]
124pub fn GetKeyVal(keyval_opt: &Option<Digested>, key: &str) -> Option<Digested> {
125  match keyval_opt {
126    Some(digested) => match digested.data() {
127      DigestedData::KeyVals(keyval) => keyval.get_value_digested(key).cloned(),
128      _ => None,
129    },
130    _ => None,
131  }
132}

136// (txfonts, mathabx, amssymb, math_common, biblatex, beamer, …).
137// Each replaces a compile-time-inlined macro arm with a runtime call.
138// LTO inlines the helper at each call site, so the binary impact is
139// the same as a per-file copy; defining once here removes ~200
140// source-level duplicates of the same body. See memory:
141//   - wisdom_data_drive_min_call_sites (≥5 sites per file threshold)
142//   - wisdom_helper_monomorphization_trap (no generic T: Into<X>)

145/// Empty-body `DefMacro!("\\cs[opt-spec]", "")` stub via runtime call.
146pub fn def_macro_noop(proto: &str) -> Result<()> {
147  let (cs_tok, params) = parse_prototype(proto, true)?;
148  let body = mouth::tokenize_internal("");
149  def_macro(cs_tok, params, ExpansionBody::Tokens(body), None)?;
150  Ok(())
151}
152
153/// Identity `DefMacro!("\\cs{}", "#1")` — CS takes one mandatory arg
154/// and expands to it unchanged.
155pub fn def_macro_identity(proto: &str) -> Result<()> {
156  let (cs_tok, params) = parse_prototype(proto, true)?;
157  let body = mouth::tokenize_internal("#1");
158  def_macro(cs_tok, params, ExpansionBody::Tokens(body), None)?;
159  Ok(())
160}

162/// Empty-body `DefPrimitive!("\\cs[opt-spec]", None);` stub —
163/// digestion-time no-op primitive (no Box emitted).
164pub fn def_primitive_noop(proto: &str) -> Result<()> {
165  let (cs_tok, params) = parse_prototype(proto, true)?;
166  def_primitive(cs_tok, params, None, PrimitiveOptions::default())?;
167  Ok(())
168}

170/// DEP-17 `DefMath!("\\cs", "char"[, role => "X"[, meaning => "Y"]])`
171/// shape. 2-arg form: prototype includes the CS; params come from
172/// `parse_prototype` (Some(empty)).
173pub fn def_math_sym(
174  cs: &str,
175  present: &str,
176  role: Option<&str>,
177  meaning: Option<&str>,
178) -> Result<()> {
179  let (cs_tok, params) = parse_prototype(cs, true)?;
180  let mut opts = MathPrimitiveOptions::default();
181  if let Some(r) = role {
182    opts.role = Some(r.to_string());
183  }
184  if let Some(m) = meaning {
185    opts.meaning = Some(m.to_string());
186  }
187  def_math(cs_tok, params, present.to_string(), opts)?;
188  Ok(())
189}

191/// DEP-17d `DefMath!("\\cs", None, "char"[, ...])` 3-arg form —
192/// Token built directly via `T_CS!`, params stays None.
193pub fn def_math_atom(
194  cs: &str,
195  present: &str,
196  role: Option<&str>,
197  meaning: Option<&str>,
198) -> Result<()> {
199  let cs_tok = T_CS!(cs);
200  let mut opts = MathPrimitiveOptions::default();
201  if let Some(r) = role {
202    opts.role = Some(r.to_string());
203  }
204  if let Some(m) = meaning {
205    opts.meaning = Some(m.to_string());
206  }
207  def_math(cs_tok, None, present.to_string(), opts)?;
208  Ok(())
