39  // Perl: latex_bootstrap.pool.ltxml L51-54 — counter/font stubs
40  DefPrimitive!("\\newcounter{}[]", sub[(cs, default_opt)] {
41    let default = if let Some(tks) = default_opt {
42      if !tks.is_empty() { Expand!(tks) } else { Tokens!() }
43    } else {
44      Tokens!()
45    };
46    let cs_expanded = &Expand!(cs).to_string();
47    // latex.ltx:10147-10155 `\@definecounter`: a `\the<ctr>` already defined (not `\relax`) is kept, "Command
48    // `\the<ctr>' already defined -- not changed" — llncs's `\let\c@case\relax\newcounter{case}` over its alias keeps
49    // `\thecase` = `\thetheorem` (2303.08726; pdflatex "Lemma 1, Case 1, Case 1, Lemma 2"). Perl's `NewCounter` always
50    // defines it (Package.pm:686). Guard `perfect_kernel_batch77v::llncs_alias_newcounter`.
51    // Not kept: a counter whose register exists already (LaTeX's "Command \c@<ctr> already defined" comes before
52    // `\@definecounter`; `NewCounter` redefines it as before), a fallback class's or stub's guess at `\the<ctr>`, which
53    // the real file's definitions replace (OmniBus's `\thechapter` under informs.cls's `\newcounter{chapter}`:
54    // 2310.17392), and a stand-in's copy read over its binding, whose counters are the binding's
55    // (`copy_provides_keeps_counter`; elsart_support's `\theaddress` under arxbj's: 1205.2947).
56    let the_cs = T_CS!(s!("\\the{cs_expanded}"));
57    let is_new = !lookup_definition(&T_CS!(s!("\\c@{cs_expanded}")))?.is_some_and(|defn| defn.is_register());
58    let kept = lookup_meaning(&the_cs).filter(|meaning| {
59      is_new
60        && Some(meaning) != lookup_meaning(&T_RELAX!()).as_ref()
61        && !is_fallback_meaning(&the_cs)
62        && definition::origin::copy_provides() == 0
63    });
64    NewCounter!(cs_expanded, &default.to_string());
65    if let Some(meaning) = kept {
66      assign_meaning(&the_cs, meaning, Some(Scope::Global));
67      if has_meaning(&T_CS!("\\@latex@warning")) {
68        Digest!(mouth::tokenize_internal(TeXString::assembled(s!(
69          "\\@latex@warning{{Command `\\string\\the{cs_expanded}' already defined -- not changed}}"
70        ))))?;
71      }
72    }
73  }, locked => true);
74  // Perl latex_bootstrap.pool.ltxml:54: `\@definecounter` is the LOCKED
75  // macro `\newcounter`, so latex.ltx:15736's raw `\def\@definecounter` is
76  // refused during the format build and every kernel counter it creates
77  // (`equation`, `footnote`, `enumi`…, and any package's `\@definecounter`
78  // later) goes through `NewCounter`, which always defines the counter's
79  // `\the<ctr>@ID` formatter (Package.pm:695-703, prefix = the counter
80  // name). An unlocked `Let!` snapshot here was overwritten by the raw
81  // definition: the dump then carried latex.ltx's `\@definecounter` and
82  // no `@ID` formatter at all, so under a class with no binding loaded raw
83  // (ptptex's manptp) `\theequation@ID` was undefined and `subequations`
84  // ids degenerated to `.1` (`X.1` after `clean_id`), colliding across
85  // groups; Perl gives `equation1`/`equation1.1`. No loop is possible:
86  // `\newcounter` is the locked primitive, never latex.ltx's macro.
87  DefMacro!("\\@definecounter", "\\newcounter", locked => true);
88  // Perl defines `\try@load@fontshape` empty (latex_bootstrap.pool.ltxml:62, "different font handling"). Ours is
89  // latex.ltx:10605-10620: when the family is not known (`\<enc>+<family>`), remember it and input
90  // `<enc><family>.fd`, lowercased, else as written — the .fd's `\DeclareFontFamily`/`\DeclareFontShape`
91  // (sect08.rs) declare `\<enc>/<family>/<series>/<shape>`, which is what amsart-derived classes test to decide
92  // whether to load amsfonts (amsart.cls:377-391 `\fontencoding{U}\fontfamily{msa}\try@load@fontshape`
93  // `\@ifundefined{U/msa/m/n}`; nyjm.cls 2401.06615, cedram.cls 2105.03188, myart.cls 1506.02470: with it empty
94  // the class skipped amsfonts and `\symAMSb` was undefined). OXIDIZED_DESIGN_DIVERGENCES #645. The kernel's
95  // `\@font@info` note and `\@input@`'s "No file" typeout are left out (neither is in the dumpless base).
96  // Locked as in Perl, so latex.ltx's raw copy does not replace it in the format build.
97  DefMacro!("\\try@load@fontshape",
98    r"\expandafter\ifx\csname\f@encoding+\f@family\endcsname\relax\global\expandafter\let\csname\f@encoding+\f@family\endcsname\@empty\nfss@catcodes\let\nfss@catcodes\relax\edef\reserved@a{\lowercase{\noexpand\InputIfFileExists{\f@encoding\f@family.fd}}}\reserved@a\relax{\InputIfFileExists{\f@encoding\f@family.fd}{}{}}\fi",
99    locked => true);
100  DefMacro!("\\define@newfont", "", locked => true);

107  // Dump-replay rollback shims — must be defined BEFORE the dump loads.
108  //
109  // The kernel dump replays latexrelease/IncludeInRelease blocks that
110  // reference TeX primitives the dump-time TeXLive didn't actually
111  // include. Predefine them so the dump replay's "Applying:" arm
112  // doesn't hit `Error:undefined:`. These need to live in
113  // latex_bootstrap (not latex_constructs_rust_only) because the dump
114  // load — which is what probes them — runs BEFORE constructs.

116  // \tracingstacklevels: TeX primitive added in TL 2021/06/01.
117  // \@expl@str@if@eq@@nnTF: expl3 internal predicate used in compat
118  //   rollback (4-arg gobble matches \str_if_eq:nnTF semantics).
119  // Witness 2408.00879, 2408.02823, 2406.00475.
120  // `\@nil` is NOT predefined: latex.ltx never defines it (a delimiter /
121  // `\ifx\@nil#1\@nil` sentinel only, e.g. latexrelease.sty:144), and an
122  // empty-macro stand-in makes `\ifx\@nil\<any empty macro>` true —
123  // polynom.sty:1695 `\pld@MeasureCells@` then stops at an empty
124  // `\pld@resultstyle` and leaks its `&\@nil&` (polynom/polydemo "Stray
125  // alignment"). The dump-replay `undefined:\@nil` the shim once silenced
126  // must be fixed where the `\@parse@version …\@nil` scan runs, never here.

128  def_macro_noop("\\@expl@str@if@eq@@nnTF{}{}{}{}")?;
