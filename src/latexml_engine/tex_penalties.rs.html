7LoadDefinitions!({
8  //%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
9  // Penalties Family of primitive control sequences
10  //%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
11
12  //======================================================================
13  // Adding/removing penalties
14  //----------------------------------------------------------------------
15  // \penalty          c  adds a penalty to the current list.
16  // \unpenalty        c  removes a penalty from the current list.
17  // \lastpenalty      iq is 0 or the last penalty on the current list.
18  // tex.web §1103: in vertical mode a penalty is a node of the list — `\vsplit` breaks at it (§974, a
19  // `\penalty-10000` forces the break; short-math-guide's `\null\penalty-\@M` before each symbol list),
20  // `\lastpenalty` reads it (§424) and `\unpenalty` removes it (the `\unkern` shape, tex_kern.rs). It is an
21  // empty box, as Perl's undef primitive makes (TeX_Penalties.pool.ltxml:29, Primitive.pm `invoke`): it absorbs to
22  // nothing. `\lastbox` is void at it (§1080): Perl's box was taken by `\lastbox`, so
23  // `\loop \unskip\unpenalty\unskip\unpenalty \setbox0\lastbox \ifvoid0…` (caesar_book.cls:106-115 counting
24  // title lines; sidenotes caesar_example) gained a box per iteration and never ended, in Perl too. Guard:
25  // `box_primitives::{lastbox, vsplit}`.
26  // In horizontal and math mode a penalty stays
27  // nothing, so `\lastpenalty` there is 0 where TeX gives the value (OXIDIZED_DESIGN_DIVERGENCES #419).
28  DefPrimitive!("\\penalty Number", sub[(n)] {
29    let value = n.value_of();
30    if !lookup_string_from_sym(pin!("MODE")).ends_with("vertical") {
31      return Ok(Vec::new());
32    }
33    let mut reversion = vec![T_CS!("\\penalty")];
34    reversion.extend(ExplodeText!(&value.to_string()));
35    let mut props = SymHashMap::default();
36    props.insert("isEmpty", Stored::Bool(true));
37    props.insert("isPenalty", Stored::Bool(true));
38    props.insert("penalty", Stored::Int(value));
39    Ok(vec![Digested::from(Tbox::new(pin!(""), None, None, Tokens::new(reversion), props))])
40  });
41  // Remove penalty, if last on LIST — never below the paragraph's start (tex.web §1105, `peek_own_box`).
42  DefPrimitive!("\\unpenalty", {
43    if peek_own_box().is_some_and(|item| item.get_property_bool("isPenalty")) {
44      pop_own_box();
45    }
46  });
47  // tex.web §424: the last item's penalty when it is one, else 0 (the `\lastkern` shape, tex_kern.rs).
48  DefRegister!("\\lastpenalty" => Number::new(0), readonly => true,
49  getter => {
50    with_own_box_list(|list| {
51      list
52        .iter()
53        .rev()
54        .find(|item| !is_list_tail_transparent(item))
55        .filter(|item| item.get_property_bool("isPenalty"))
56        .and_then(|item| match item.get_property("penalty").as_deref() {
57          Some(Stored::Int(value)) => Some(Number::new(*value)),
58          _ => None,
59        })
60        .unwrap_or_else(|| Number::new(0))
61    })
62  });
63
64  //======================================================================
65  // values for various penalties
66  //----------------------------------------------------------------------
67  // \brokenpenalty    pi is the penalty added after a line ending with an hyphenated word.
68  // \clubpenalty      pi is the penalty added after the first line in a paragraph.
69  // \exhyphenpenalty  pi is the penalty for a line break after an explicit hyphen.
70  // \floatingpenalty  pi is the penalty for insertions that are split between pages.
71  // \hyphenpenalty    pi is the penalty for a line break after a discretionary hyphen.
72  // \interlinepenalty pi is the penalty added between lines in a paragraph.
73  // \linepenalty      pi is an amount added to the \badness calculated for every line in a
74  // paragraph. \outputpenalty    pi holds the penalty from the current page break.
75  // \widowpenalty     pi is the penalty added after the penultimate line in a paragraph.
76  DefRegister!("\\brokenpenalty", Number!(100));
77  DefRegister!("\\clubpenalty", Number!(150));
78  DefRegister!("\\exhyphenpenalty", Number!(50));
79  DefRegister!("\\floatingpenalty", Number!(0));
80  DefRegister!("\\hyphenpenalty", Number!(50));
81  DefRegister!("\\interlinepenalty", Number!(0));
82  DefRegister!("\\linepenalty", Number!(10));
83  DefRegister!("\\outputpenalty", Number!(0));
84  DefRegister!("\\widowpenalty", Number!(150));
85});
