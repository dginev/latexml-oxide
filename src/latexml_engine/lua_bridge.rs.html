1//! LuaTeX-escape bridge: evaluate `\directlua` bodies in a persistent
2//! external `texlua` interpreter (user directive 2026-08-31: "we can assume a
3//! lua interpreter is available on devices with texlive installed").
4//!
5//! First-principles semantics (LuaTeX manual §2.1, §10.3): `\directlua
6//! {<general text>}` runs the (fully expanded) body as a Lua chunk in the
7//! engine's Lua state — one state for the whole job, so globals persist
8//! across calls — and everything the chunk emits through `tex.print` /
9//! `tex.sprint` / `tex.write` is inserted into the TeX input stream and
10//! read with CURRENT catcodes (`tex.print` as full lines, `tex.sprint` as
11//! a partial line).
12//!
13//! Strategy: **rebind-as-we-emulate** (docs/perfect_kernel/LUA_REBINDING.md).
14//! texlua is Lua + TeX's support libraries (kpse, lpeg, md5, pdfe, …) MINUS
15//! the engine — its built-in `tex` table holds only run/initialize/finish —
16//! so every `tex.*` touchpoint an author's chunk uses is necessarily OUR
17//! shim, and the design question per API is only how deep the shim goes:
18//! * **translate** — emissions re-enter our pipeline (`tex.print`/`sprint`
19//!   retokenized with current catcodes);
20//! * **mirror** — state our engine genuinely has is served LIVE over the
21//!   pipe (`tex.count`/`tex.dimen` reads AND writes round-trip to the Rust
22//!   State via framed `Q`/`A` messages, `service_query`);
23//! * **absorb** — intents with no XML-output meaning are accepted and
24//!   dropped (`pdf.set*`, `tex.primitives()`); the node/font/callback layers
25//!   (`node.*`, `font.*`, `callback.*` — LuaTeX typesetter internals) stay
26//!   out of scope and error visibly rather than pretend.
27//!
28//! `require` is rebound through texlua's built-in kpse (plus lualibs, whose
29//! `file`/`string` extensions lualatex gets via luaotfload), so texmf-shipped
30//! Lua modules (newpax, tkz-elements, …) resolve.
31//!
32//! Document Lua is untrusted input: the prelude sandboxes it to TeX's defaults
33//! (`shell_escape=f`, `openout_any=p`) before any chunk runs, each refusal an
34//! `Error:lua:sandbox` (OXIDIZED_DESIGN_DIVERGENCES #1081).
35//!
36//! Process shape: one `texlua` child per conversion thread, spawned on first
37//! use with a prelude that loops over length-framed requests on stdin and
38//! answers `OK`/`ERR` length-framed responses on stdout, each header carrying
39//! the nonce the bridge sent as the child's first line. A std
40//! `thread_local!` (destructor RUNS at thread exit, unlike `#[thread_local]`)
41//! drops the pipes so the child's `io.read` returns nil and it exits.
42

44  cell::RefCell,

46  process::{ChildStdin, ChildStdout, Command, Stdio},

74local function wire_len(s)
75  if CRLF then
76    local _, newlines = s_gsub(s, '\n', '')
77    return #s + newlines
78  end
79  return #s
80end
81-- One write per frame, so no frame lands inside another. Built by concatenation: '%s' would ask the string
82-- metatable's __tostring, which document code can set.
83local function frame(tag, payload)
84  f_write(stdout, tag .. ' ' .. NONCE .. ' ' .. s_format('%d', wire_len(payload)) .. '\n' .. payload)
85  f_flush(stdout)
86end

88local function emit(sep, ...)
89  local args = {...}
90  local start = 1
91  if type(args[1]) == 'number' and #args > 1 then start = 2 end
92  for i = start, #args do
93    local v = args[i]
94    if type(v) == 'table' then
95      for _, s in ipairs(v) do buf[#buf+1] = tostring(s) .. sep end
96    else
97      buf[#buf+1] = tostring(v) .. sep
98    end
99  end
100end

112  local header = f_read(stdin, '*l')

155-- Sandbox (OXIDIZED_DESIGN_DIVERGENCES #1081): document Lua is untrusted input,
156-- held to TeX's defaults for it, shell_escape=f (no process runs) and
157-- openout_any=p (kpathsea's kpse_out_name_ok: a write only to a relative path
158-- with no '..' part and no dotfile), checked on every path component and through
159-- no symbolic link; reads stay open (openin_any=a). texlua's
160-- --safer/--nosocket/--shell-escape do not restrict a script, so the prelude
161-- does it before any document chunk runs. A denied call reports a 'D' frame (an
162-- Error a pcall cannot swallow) and raises a Lua error.
163-- While the prelude loads lualibs (texmf code, trusted like a format), the
164-- originals still answer; the wrappers it captures deny once the loop starts.
165local booting = true
166local function deny(what)
167  if booting then error(what .. ' is not allowed', 3) end
168  local msg = what .. ' is not allowed (document Lua runs sandboxed: no programs, writes only below the job directory)'
169  frame('D', msg)
170  error(msg, 3)
171end
172local function denier(what, original)
173  return function(...)
174    if booting and original then return original(...) end
175    deny(what)
176  end
177end
178-- A name for a refusal message: the document's value is never asked to describe itself, and a long one is cut.
179local function shown(name)
180  if type(name) ~= 'string' then return '<' .. type(name) .. '>' end
181  if #name > 256 then return s_sub(name, 1, 256) .. '... (' .. s_format('%d', #name) .. ' bytes)' end
182  return name
183end
184local symlinkattributes = lfs and lfs.symlinkattributes
185-- kpse_out_name_ok under openout_any=p, decided here rather than by texlua's
186-- kpse.out_name_ok, which follows the host's texmf.cnf (openout_any=a there
187-- allows anything) and checks only the last component for a dotfile.
188local function out_name_ok(name)
189  if type(name) ~= 'string' or name == '' or s_find(name, '\0', 1, true) then return false end
190  local path = name
191  if CRLF then
192    -- Windows: '\' separates too; a ':' is a drive or an alternate data stream; '\\?\' and '\\.\' are device paths.
193    if s_find(name, '^[\\/][\\/][%?%.][\\/]') or s_find(name, ':', 1, true) then return false end
194    path = s_gsub(name, '\\', '/')
195  elseif s_find(name, '\\', 1, true) then
196    -- Unix: '\' is a filename character; the name checked must be the name the system opens.
197    return false
198  end
199  if s_sub(path, 1, 1) == '/' then return false end
200  local prefix = nil
201  for part in s_gmatch(path, '[^/]+') do
202    if part == '..' or (s_sub(part, 1, 1) == '.' and part ~= '.' and part ~= '.tex') then return false end
203    -- (Windows drops a component's trailing dots and spaces: 'a.' opens 'a')
204    if CRLF and part ~= '.' and s_find(part, '[%. ]$') then return false end
205    -- No component may be a symbolic link: one made by the source tree could point anywhere.
206    prefix = prefix and (prefix .. '/' .. part) or part
207    if symlinkattributes and symlinkattributes(prefix, 'mode') == 'link' then return false end
208  end
209  return true
210end

376struct LuaProc {
377  /// The child, registered so a timeout or memory exit (which runs no destructor) kills it too; dropping the guard
378  /// kills and reaps it.
379  child:   Option<latexml_core::watchdog::KillOnExit>,
380  stdin:   ChildStdin,
381  stdout:  BufReader<ChildStdout>,
382  /// A private prelude copy (`Prelude::Own`), removed with the process.
383  prelude: Option<std::path::PathBuf>,
384  /// This interpreter's frame nonce (`session_nonce`), sent as its first line.
385  nonce:   String,
386}

388/// This thread's interpreter for the current conversion (`reset_session` ends it: a persistent worker converts its
389/// papers on one thread, and LuaTeX starts each job with a fresh Lua state in the job's directory).
390#[derive(Default)]
391struct LuaSlot {
392  proc_:       Option<LuaProc>,
393  /// texlua could not be started (reported once per conversion as "texlua not available").
394  unavailable: bool,
395  /// Interpreters that ended mid-conversion (a crash, a broken frame); each is respawned up to `MAX_RESTARTS`.
396  ended:       u32,
397}
398
399/// Restarts per conversion after an interpreter ends; past them, later chunks are not evaluated.
400const MAX_RESTARTS: u32 = 3;
401/// The longest frame the bridge reads: a length past it is a broken protocol, not output.
402const MAX_FRAME: usize = 64 << 20;
403/// The refusals a chunk reports one by one, by count and by bytes; the rest are counted in one summary Error.
404const MAX_DENIALS: usize = 20;
405const MAX_DENIAL_BYTES: usize = 64 << 10;
406
407thread_local! {
408  static LUA: RefCell<LuaSlot> = RefCell::new(LuaSlot::default());
409}
410
411/// Ends this thread's interpreter and forgets its restarts: called at each conversion's start
412/// (`Converter::initialize_session`), so no Lua state or working directory carries over to the next paper.
413pub fn reset_session() { LUA.with(|slot| *slot.borrow_mut() = LuaSlot::default()); }

415impl Drop for LuaProc {
416  fn drop(&mut self) {
417    // The guard kills and reaps the child, so no zombie outlives the thread.
418    drop(self.child.take());
419    if let Some(path) = self.prelude.take() {
420      let _ = std::fs::remove_file(path);
421    }
422  }
423}

425fn content_hash(text: &str) -> u64 {
426  text.bytes().fold(0xcbf2_9ce4_8422_2325, |hash, byte| {
427    (hash ^ u64::from(byte)).wrapping_mul(0x0100_0000_01b3)
428  })
429}

431/// Where the prelude script is: the file shared by this user's processes, or a private copy the
432/// process removes when its interpreter ends.
433enum Prelude {
434  Shared(std::path::PathBuf),
435  Own(std::path::PathBuf),
436}

438/// The prelude script on disk: `latexml_lua_prelude_<uid>_<content hash>.lua` in the temp dir, shared
439/// by this user's processes and used only when it is a regular file (not a link) of this user holding
440/// exactly the prelude — the temp dir is world-writable, and a planted or truncated file would run as
441/// this user. Written aside under a name no other writer uses (a pid and a counter; a name left by an
442/// earlier process of the same pid is skipped) and renamed into place, so a concurrent reader never
443/// sees a partial file; where the shared name cannot be taken, the private copy is used and removed
444/// with the interpreter (`LuaProc`'s drop).
445fn prelude_file() -> Option<Prelude> {
446  use std::io::Write;
447  static WRITES: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
448  let dir = std::env::temp_dir();
449  let hash = content_hash(LUA_PRELUDE);
450  let (own, mut file) = (0..64).find_map(|_| {
451    let own = dir.join(format!(
452      "latexml_lua_prelude_{hash:016x}.{}.{}.lua",
453      std::process::id(),
454      WRITES.fetch_add(1, std::sync::atomic::Ordering::Relaxed)
455    ));
456    let mut options = std::fs::OpenOptions::new();
457    options.write(true).create_new(true);
458    #[cfg(unix)]
459    std::os::unix::fs::OpenOptionsExt::mode(&mut options, 0o644);
460    match options.open(&own) {
461      Ok(file) => Some(Some((own, file))),
462      Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => None,
463      Err(_) => Some(None),
464    }
465  })??;
466  if file.write_all(LUA_PRELUDE.as_bytes()).is_err() {
467    let _ = std::fs::remove_file(&own);
468    return None;
469  }
470  drop(file);
471  #[cfg(unix)]
472  let owner = {
473    use std::os::unix::fs::MetadataExt;
474    std::fs::symlink_metadata(&own).ok()?.uid()
475  };
476  #[cfg(not(unix))]
477  let owner = 0;
478  let shared = dir.join(format!("latexml_lua_prelude_{owner}_{hash:016x}.lua"));
479  let is_ours = |path: &std::path::Path| {
480    std::fs::symlink_metadata(path).is_ok_and(|meta| {
481      #[cfg(unix)]
482      let same_owner = std::os::unix::fs::MetadataExt::uid(&meta) == owner;
483      #[cfg(not(unix))]
484      let same_owner = true;
485      meta.is_file() && same_owner
486    })
487  };
488  if is_ours(&shared) && std::fs::read(&shared).is_ok_and(|bytes| bytes == LUA_PRELUDE.as_bytes()) {
489    let _ = std::fs::remove_file(&own);
490    return Some(Prelude::Shared(shared));
491  }
492  if std::fs::rename(&own, &shared).is_ok() {
493    return Some(Prelude::Shared(shared));
494  }
495  Some(Prelude::Own(own))
496}

498/// The environment variables the texlua child keeps: the program and library search (`PATH`, `LD_LIBRARY_PATH`,

544/// A per-interpreter frame nonce: 128 bits of SipHash keyed by the standard library's random `RandomState` keys,
545/// over the pid, the clock and a counter. The prelude reads it before any other code loads; a chunk never sees it.
546fn session_nonce() -> String {
547  use std::hash::{BuildHasher, Hasher};
548  static SPAWNS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
549  let state = std::collections::hash_map::RandomState::new();
550  let count = SPAWNS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
551  let clock = std::time::SystemTime::now()
552    .duration_since(std::time::UNIX_EPOCH)
553    .map_or(0, |since| since.as_nanos());
554  let word = |half: u64| {
555    let mut hasher = state.build_hasher();
556    hasher.write_u64(half);
557    hasher.write_u32(std::process::id());
558    hasher.write_u128(clock);
559    hasher.write_u64(count);
560    hasher.finish()
561  };
562  format!("{:016x}{:016x}", word(0), word(1))
563}

565fn spawn() -> Option<LuaProc> {
566  // `texlua` takes a script FILE (its `-e` is not lua's inline-chunk flag) — materialize the
567  // prelude in the temp dir, named by its owner and content: one file shared by this user's processes
568  // and version-safe (a per-process name was never removed, and one-shot runs and test processes left ~98,000
569  // of them, exhausting `/tmp`'s inodes).
570  let (prelude_path, prelude) = match prelude_file()? {
571    Prelude::Shared(path) => (path, None),
572    Prelude::Own(path) => (path.clone(), Some(path)),
573  };
574  let mut cmd = Command::new("texlua");
575  cmd
576    .arg(&prelude_path)
577    .stdin(Stdio::piped())
578    .stdout(Stdio::piped())
579    .stderr(Stdio::null())
580    // Untrusted Lua sees only what the texmf lookup and the locale need, not the worker's environment.
581    .env_clear()
582    .envs(std::env::vars_os().filter(|(name, _)| name.to_str().is_some_and(child_env_allowed)));
583  // LuaTeX job semantics: chunks resolve relative files against the
584  // document's directory (witness newpax/newpax — `newpax.writenewpax
585  // ("doc-input1")` reads the PDFs shipped next to the manual).
586  let source_dir = latexml_core::state::lookup_string("SOURCEDIRECTORY");
587  if !source_dir.is_empty() && std::path::Path::new(&source_dir).is_dir() {
588    cmd.current_dir(&source_dir);
589  }
590  let Ok(mut child) = cmd.spawn() else {
591    if let Some(path) = prelude {
592      let _ = std::fs::remove_file(path);
593    }
594    return None;
595  };
596  let (Some(stdin), Some(stdout)) = (child.stdin.take(), child.stdout.take()) else {
597    let _ = child.kill();
598    let _ = child.wait();
599    if let Some(path) = prelude {
600      let _ = std::fs::remove_file(path);
601    }
602    return None;
603  };
604  let mut proc_ = LuaProc {
605    child: Some(latexml_core::watchdog::kill_on_exit(child)),
606    stdin,
607    stdout: BufReader::new(stdout),
608    prelude,
609    nonce: session_nonce(),
610  };
611  let line = format!("{}\n", proc_.nonce);
612  if proc_.stdin.write_all(line.as_bytes()).is_err() {
613    return None;
614  }
615  Some(proc_)
616}

618/// Service one live tex-state query from the Lua side (the mirror half of
619/// rebind-as-we-emulate): `get{count,dimen} <key>` reads a register from the
620/// engine State, `set{count,dimen} [global] <key> <value>` writes one. Keys
621/// are numeric register indices (`tex.count[255]` → `\count255`) or
622/// `\countdef`'d names (`tex.count.scratchcounter` → `\scratchcounter`).
623/// Counts answer as integers, dimens in scaled points (LuaTeX convention).
624/// Unknown ops or lookup failures answer "0" — the pre-mirror stub value —
625/// so a chunk probing state we don't model degrades exactly as before.
626fn service_query(payload: &str) -> String {
627  use latexml_core::{
628    common::{dimension::Dimension, number::Number},
629    definition::{argument::ArgWrap, register::RegisterValue},
630    state,
631  };
632  let mut it = payload.split_whitespace();
633  let op = it.next().unwrap_or("");
634  let mut key = it.next().unwrap_or("");
635  let global = key == "global";
636  if global {
637    key = it.next().unwrap_or("");
638  }
639  let (cs, params) = match key.parse::<i64>() {
640    Ok(n) => {
641      let reg = if op.ends_with("dimen") {
642        "\\dimen"
643      } else {
644        "\\count"
645      };
646      (reg.to_string(), vec![ArgWrap::Number(Number(n))])
647    },
648    Err(_) => (format!("\\{key}"), Vec::new()),
649  };
650  match op {
651    "getcount" | "getdimen" => match state::lookup_register(&cs, params).ok().flatten() {
652      Some(RegisterValue::Number(n)) => n.0.to_string(),
653      Some(RegisterValue::Dimension(d)) => d.0.to_string(),
654      _ => "0".to_string(),
655    },
656    "setcount" | "setdimen" => {
657      let value: i64 = it.next().and_then(|v| v.parse().ok()).unwrap_or(0);
658      let rv = if op == "setdimen" {
659        RegisterValue::Dimension(Dimension(value))
660      } else {
661        RegisterValue::Number(Number(value))
662      };
663      let scope = global.then_some(state::Scope::Global);
664      let _ = state::assign_register(&cs, rv, scope, params);
665      String::new()
666    },
667    _ => "0".to_string(),
668  }
669}

671/// Execute one Lua chunk in this thread's persistent interpreter.
672///
673/// `Ok(text)` is everything the chunk `tex.print`ed (to be retokenized with
674/// current catcodes by the caller); `Err(msg)` is a load/runtime Lua error or
675/// bridge unavailability ("texlua not available" — callers degrade to a no-op
676/// expansion, matching what the document would get from a TeX engine without
677/// Lua). Each call the sandbox refused, and an interpreter that ended mid-chunk
678/// (its output lost), is raised as an `Error` (the outer `Err`, should that
679/// Error escalate to a Fatal).
680pub fn lua_exec(chunk: &str) -> latexml_core::common::error::Result<Result<String, String>> {
681  let framed = exec_framed(chunk);
682  use crate::prelude::*;
683  for denial in framed.denials.shown {
684    Error!("lua", "sandbox", denial);
685  }
686  if framed.denials.more > 0 {
687    Error!(
688      "lua",
689      "sandbox",
690      format!(
691        "{} more calls the sandbox refused in this chunk are not listed",
692        framed.denials.more
693      )
694    );
695  }
696  if let Some(ended) = framed.ended {
697    Error!("lua", "interpreter", ended);
698  }
699  Ok(framed.result)
700}

702/// One chunk's run: its result, the sandbox's refusals ('D' frames, OXIDIZED_DESIGN_DIVERGENCES #1081) — returned
703/// apart, as a pcall in the chunk can swallow the Lua error, never the record — and the report of an interpreter
704/// that ended during it.
705struct Framed {
706  result:  Result<String, String>,
707  denials: Denials,
708  ended:   Option<String>,
709}
710
711/// A chunk's refusals: the first `MAX_DENIALS` (within `MAX_DENIAL_BYTES`) kept, the rest counted.
712#[derive(Default)]
713struct Denials {
714  shown: Vec<String>,
715  bytes: usize,
716  more:  usize,
717}
718
719impl Denials {
720  fn push(&mut self, denial: String) {
721    if self.shown.len() < MAX_DENIALS && self.bytes + denial.len() <= MAX_DENIAL_BYTES {
722      self.bytes += denial.len();
723      self.shown.push(denial);
724    } else {
725      self.more += 1;
726    }
727  }
728}

730/// A frame header's tag and length: `<tag> <nonce> <length>`, with this interpreter's nonce (a header without it
731/// came from no prelude frame, OXIDIZED_DESIGN_DIVERGENCES #1081) and a length the bridge reads.
732fn parse_header<'h>(header: &'h str, nonce: &str) -> Result<(&'h str, usize), FrameError> {
733  let mut fields = header.trim_end().splitn(3, ' ');
734  let (tag, sent, len) = (fields.next(), fields.next(), fields.next());
735  match (tag, len.map(str::parse::<usize>)) {
736    (Some(tag), Some(Ok(n))) if sent == Some(nonce) && n <= MAX_FRAME => Ok((tag, n)),
737    _ if sent == Some(nonce) => Err(FrameError::Protocol(format!(
738      "a frame header '{}'",
739      header.trim_end()
740    ))),
741    _ => Err(FrameError::Protocol(
742      "output that is not a frame (no session nonce)".to_string(),
743    )),
744  }
745}

747/// Why a frame exchange failed: the pipe (the child ended) or a frame no prelude writes.
748enum FrameError {
749  Io(std::io::Error),
750  Protocol(String),
751}
752
753impl From<std::io::Error> for FrameError {
754  fn from(e: std::io::Error) -> Self { FrameError::Io(e) }
755}

757/// Sends `chunk` and reads frames up to its final `OK`/`ERR`, servicing `Q`ueries and collecting `D`enials.
758fn exchange(
759  p: &mut LuaProc,
760  chunk: &str,
761  denials: &mut Denials,
762) -> Result<(bool, String), FrameError> {
763  use std::io::Read;
764  p.stdin.write_all(format!("{}\n", chunk.len()).as_bytes())?;
765  p.stdin.write_all(chunk.as_bytes())?;
766  p.stdin.write_all(b"\n")?;
767  p.stdin.flush()?;
768  // The chunk may interleave 'Q'uery frames (live tex-state reads/writes)
769  // before its final OK/ERR frame — service each against engine State.
770  loop {
771    let mut header = String::new();
772    (&mut p.stdout).take(64).read_line(&mut header)?;
773    if header.is_empty() {
774      return Err(FrameError::Io(std::io::ErrorKind::UnexpectedEof.into()));
775    }
776    let (tag, n) = parse_header(&header, &p.nonce)?;
777    let mut body = vec![0u8; n];
778    p.stdout.read_exact(&mut body)?;
779    let mut body = String::from_utf8_lossy(&body).into_owned();
780    if cfg!(windows) {
781      // Undo the text-mode stdout: drop the `\r` written before each `\n` (a `\r\n` the chunk printed came
782      // through as `\r\r\n`, and stays `\r\n`).
783      body = body.replace("\r\n", "\n");
784    }
785    match tag {
786      "D" => denials.push(body),
787      "Q" => {
788        let answer = service_query(&body);
789        p.stdin
790          .write_all(format!("A {}\n", answer.len()).as_bytes())?;
791        p.stdin.write_all(answer.as_bytes())?;
792        p.stdin.flush()?;
793      },
794      "OK" => return Ok((true, body)),
795      "ERR" => return Ok((false, body)),
796      _ => return Err(FrameError::Protocol(format!("a frame tagged '{tag}'"))),
797    }
798  }
799}

801fn exec_framed(chunk: &str) -> Framed {
802  let mut framed = Framed {
803    result:  Ok(String::new()),
804    denials: Denials::default(),
805    ended:   None,
806  };
807  // (a speculative run is undone, the interpreter's globals could not be: `\\author{First\\luaexec{n=(n or 0)+1}}`, which a second
808  // `\\author` replaces, left its count for the body's `\\luaexec{tex.print(n or 0)}`, lualatex's 0)
809  if latexml_core::gullet::in_speculative_digestion() {
810    framed.result = Err("a speculative run evaluates no Lua".to_string());
811    return framed;
812  }
813  LUA.with(|slot| {
814    let mut slot = slot.borrow_mut();
815    if slot.unavailable {
816      framed.result = Err("texlua not available".to_string());
817      return;
818    }
819    if slot.ended > MAX_RESTARTS {
820      framed.result = Err(format!(
821        "texlua ended {} times in this conversion",
822        slot.ended
823      ));
824      return;
825    }
826    if slot.proc_.is_none() {
827      slot.proc_ = spawn();
828      if slot.proc_.is_none() {
829        slot.unavailable = true;
830        framed.result = Err("texlua not available".to_string());
831        return;
832      }
833    }
834    // Windows texlua reads stdin in text mode: a `\r\n` arrives as `\n`, one byte short of the frame's count, so
835    // it is sent as the `\n` it would read (the prelude's `wire_len` covers the other direction).
836    let chunk: std::borrow::Cow<str> = if cfg!(windows) {
837      chunk.replace("\r\n", "\n").into()
838    } else {
839      chunk.into()
840    };
841    let Some(proc_) = slot.proc_.as_mut() else {
842      return;
843    };
844    framed.result = match exchange(proc_, &chunk, &mut framed.denials) {
845      Ok((true, body)) => Ok(body),
846      Ok((false, body)) => Err(body),
847      Err(error) => {
848        // The interpreter ended (a crash, `os.exit`) or broke the protocol: its output is lost and its state gone.
849        // Dropping it kills and reaps the child; the next chunk starts a fresh one, up to MAX_RESTARTS times.
850        slot.proc_ = None;
851        slot.ended += 1;
852        let why = match error {
853          FrameError::Io(e) => format!("the texlua interpreter ended ({e})"),
854          FrameError::Protocol(what) => format!("the texlua interpreter sent {what}"),
855        };
856        let next = if slot.ended > MAX_RESTARTS {
857          "; later chunks are not evaluated"
858        } else {
859          "; the next chunk starts a fresh interpreter"
860        };
861        framed.ended = Some(format!(
862          "{why}: this chunk's output and the Lua state are lost{next}"
863        ));
864        Err(why)
865      },
866    };
867  });
868  framed
869}
