1//! LuaTeX-escape bridge: evaluate `\directlua` bodies in a persistent
2//! external `texlua` interpreter (user directive 2026-08-31: "we can assume a
3//! lua interpreter is available on devices with texlive installed").
4//!
5//! First-principles semantics (LuaTeX manual §2.1, §10.3): `\directlua
6//! {<general text>}` runs the (fully expanded) body as a Lua chunk in the
7//! engine's Lua state — one state for the whole job, so globals persist
8//! across calls — and everything the chunk emits through `tex.print` /
9//! `tex.sprint` / `tex.write` is inserted into the TeX input stream and
10//! read with CURRENT catcodes (`tex.print` as full lines, `tex.sprint` as
11//! a partial line).
12//!
13//! Strategy: **rebind-as-we-emulate** (docs/perfect_kernel/LUA_REBINDING.md).
14//! texlua is Lua + TeX's support libraries (kpse, lpeg, md5, pdfe, …) MINUS
15//! the engine — its built-in `tex` table holds only run/initialize/finish —
16//! so every `tex.*` touchpoint an author's chunk uses is necessarily OUR
17//! shim, and the design question per API is only how deep the shim goes:
18//! * **translate** — emissions re-enter our pipeline (`tex.print`/`sprint`
19//!   retokenized with current catcodes);
20//! * **mirror** — state our engine genuinely has is served LIVE over the
21//!   pipe (`tex.count`/`tex.dimen` reads AND writes round-trip to the Rust
22//!   State via framed `Q`/`A` messages, `service_query`);
23//! * **absorb** — intents with no XML-output meaning are accepted and
24//!   dropped (`pdf.set*`, `tex.primitives()`); the node/callback layers
25//!   (`node.*`, `callback.*` — LuaTeX typesetter internals) stay out of scope
26//!   and error visibly rather than pretend, as do `lang.*`, `vf.*` and the
27//!   `font.*` calls that need the engine's fonts; `font` mirrors one font,
28//!   the current one, whose characters answer `\iffontchar\font`.
29//!
30//! `require` is rebound through texlua's built-in kpse (plus lualibs, whose
31//! `file`/`string` extensions lualatex gets via luaotfload), so texmf-shipped
32//! Lua modules (newpax, tkz-elements, …) resolve.
33//!
34//! Document Lua is untrusted input: the prelude sandboxes it to TeX's defaults
35//! (`shell_escape=f`, `openout_any=p`) before any chunk runs. A call lualatex
36//! refuses too gets lualatex's values and a `Warning:lua:sandbox`; one only the
37//! sandbox refuses, an `Error:lua:sandbox` (OXIDIZED_DESIGN_DIVERGENCES #1081).
38//!
39//! Process shape: one `texlua` child per conversion thread, spawned on first
40//! use with a prelude that loops over length-framed requests on stdin and
41//! answers `OK`/`ERR` length-framed responses on stdout, each header carrying
42//! the nonce the bridge sent as the child's first line. A std
43//! `thread_local!` (destructor RUNS at thread exit, unlike `#[thread_local]`)
44//! drops the pipes so the child's `io.read` returns nil and it exits.
45

72local f_read, f_write, f_flush = stdin.read, stdout.write, stdout.flush
73-- The bridge's first line, read before any other code loads: every frame header carries it, and the bridge rejects
74-- a header without it, so bytes that reach the pipe by any other route are a broken protocol, never a frame.
75local NONCE = f_read(stdin, '*l') or ''
76-- A frame's length is the bytes that reach the pipe: Windows texlua writes
77-- stdout in text mode, each '\n' as '\r\n' (the bridge drops the added '\r').
78local CRLF = s_sub(package.config, 1, 1) == '\\'
79local function wire_len(s)
80  if CRLF then
81    local _, newlines = s_gsub(s, '\n', '')
82    return #s + newlines
83  end
84  return #s
85end
86-- One write per frame, so no frame lands inside another. Built by concatenation: '%s' would ask the string
87-- metatable's __tostring, which document code can set.
88local function frame(tag, payload)
89  f_write(stdout, tag .. ' ' .. NONCE .. ' ' .. s_format('%d', wire_len(payload)) .. '\n' .. payload)
90  f_flush(stdout)
91end
92local buf = {}
93local function emit(sep, ...)
94  local args = {...}
95  local start = 1
96  if type(args[1]) == 'number' and #args > 1 then start = 2 end
97  for i = start, #args do
98    local v = args[i]
99    if type(v) == 'table' then
100      for _, s in ipairs(v) do buf[#buf+1] = tostring(s) .. sep end
101    else
102      buf[#buf+1] = tostring(v) .. sep
103    end
104  end
105end

111-- Live register mirror (rebind-as-we-emulate): texlua has no engine, so
112-- tex.count/tex.dimen access round-trips to the Rust State over the pipe as
113-- a framed query ('Q <len>' out, 'A <len>' back). Values follow the LuaTeX
114-- convention: counts are integers, dimens are scaled points.
115local function query(payload)
116  frame('Q', payload)
117  local header = f_read(stdin, '*l')
118  local n = tonumber(header and s_match(header, '%d+')) or 0
119  return n > 0 and f_read(stdin, n) or ''
120end
121local function regtable(kind)
122  return setmetatable({}, {

124      return tonumber(query('get' .. kind .. ' ' .. tostring(k))) or 0
125    end,
126    __newindex = function(_, k, v)
127      query('set' .. kind .. ' ' .. tostring(k) .. ' ' .. tostring(m_floor(v)))
128    end})
129end

137  end

160-- Sandbox (OXIDIZED_DESIGN_DIVERGENCES #1081): document Lua is untrusted input,
161-- held to TeX's defaults for it, shell_escape=f (no process runs) and
162-- openout_any=p (kpathsea's kpse_out_name_ok: a write only to a relative path
163-- with no '..' part and no dotfile), checked on every path component and through
164-- no symbolic link; reads stay open (openin_any=a). texlua's
165-- --safer/--nosocket/--shell-escape do not restrict a script, so the prelude
166-- does it before any document chunk runs. A call lualatex refuses too returns
167-- lualatex's values and reports a 'W' frame (a Warning); one only the sandbox
168-- refuses reports a 'D' frame (an Error) and raises a Lua error. No record
169-- (nor an engine gap's 'G') can be swallowed by a pcall.
170-- While the prelude loads lualibs (texmf code, trusted like a format), the
171-- originals still answer; the wrappers it captures refuse once the loop starts.
172local booting = true
173local function deny(what)
174  if booting then error(what .. ' is not allowed', 3) end
175  local msg = what .. ' is not allowed (document Lua runs sandboxed: no programs, writes only below the job directory)'
176  frame('D', msg)
177  error(msg, 3)
178end
179-- An engine layer texlua lacks, which lualatex performs (a font, a language, a typesetter call): a 'G' frame, a
180-- Warning:lua:engine, never a refusal, so a chunk that then fails stays the caller's Info.
181local function gap(msg)
182  if not booting then frame('G', msg) end
183end
184local function denier(what, original)
185  return function(...)
186    if booting and original then return original(...) end
187    deny(what)
188  end
189end
190-- A name for a refusal message: the document's value is never asked to describe itself, and a long one is cut.
191local function shown(name)
192  if type(name) ~= 'string' then return '<' .. type(name) .. '>' end
193  if #name > 256 then return s_sub(name, 1, 256) .. '... (' .. s_format('%d', #name) .. ' bytes)' end
194  return name
195end
196local symlinkattributes = lfs and lfs.symlinkattributes
197-- kpse_out_name_ok under openout_any=p, decided here rather than by texlua's
198-- kpse.out_name_ok, which follows the host's texmf.cnf (openout_any=a there
199-- allows anything) and checks only the last component for a dotfile. A name
200-- refused only by the sandbox's stricter checks, which lualatex would write,
201-- returns false, true.
202local function out_name_ok(name)
203  if type(name) ~= 'string' or name == '' or s_find(name, '\0', 1, true) then return false end
204  local path, strict = name, false
205  if CRLF then
206    -- Windows: '\' separates too; a ':' is a drive or an alternate data stream; '\\?\' and '\\.\' are device paths.
207    if s_find(name, '^[\\/][\\/][%?%.][\\/]') or s_find(name, ':', 1, true) then return false end
208    path = s_gsub(name, '\\', '/')
209  elseif s_find(name, '\\', 1, true) then
210    -- Unix: '\' is a filename character; the name checked must be the name the system opens.
211    strict = true
212  end
213  if s_sub(path, 1, 1) == '/' then return false end
214  local prefix = nil
215  for part in s_gmatch(path, '[^/]+') do
216    if part == '..' or (s_sub(part, 1, 1) == '.' and part ~= '.' and part ~= '.tex') then return false end
217    -- (Windows drops a component's trailing dots and spaces: 'a.' opens 'a')
218    if CRLF and part ~= '.' and s_find(part, '[%. ]$') then strict = true end
219    -- No component may be a symbolic link: one made by the source tree could point anywhere.
220    prefix = prefix and (prefix .. '/' .. part) or part
221    if not strict and symlinkattributes and symlinkattributes(prefix, 'mode') == 'link' then strict = true end
222  end
223  if strict then return false, true end
224  return true
225end
226local function writes(mode) return type(mode) == 'string' and s_find(mode, '[wa+]') ~= nil end
227-- A program call answers as lualatex --no-shell-escape does (shell_escape=f, which \pdfshellescape and
228-- status.shell_escape also read; measured on TL 2025): nil and its message, and os.execute() asks whether a shell
229-- exists, 0 (pdftexcmds' \pdf@shellescape, imakeidx: TL manuals derivative, wheelchart, xassoccnt_doc).
230local function no_program(what, original, message)
231  return function(...)
232    if booting and original then return original(...) end
233    local command = ...
234    if what == 'os.execute' and command == nil then return 0 end
235    if type(command) == 'string' then command = '"' .. shown(command) .. '"' else command = shown(command) end
236    frame('W', what .. ' ' .. command .. ' was not run (document Lua runs no programs, as lualatex without shell escape)')
237    return nil, message
238  end
239end

249local io_open, io_output, io_input = io.open, io.output, io.input
250-- lualatex's io.open answers a write openout_any=p forbids with a lone nil; a name only the sandbox refuses (a
251-- symbolic link, a '\' name) and io.output and gzip.open, which lualatex does not check, are denied.
252io.open = function(name, mode)
253  if writes(mode) then
254    local ok, strict = out_name_ok(name)
255    if strict then deny('writing ' .. shown(name)) end
256    if not ok then
257      if not booting then
258        frame('W', 'writing ' .. shown(name) .. ' was refused (document Lua writes only below the job directory, as lualatex under openout_any=p)')
259      end
260      return nil
261    end
262  end
263  return io_open(name, mode)
264end
265io.output = function(file)
266  if type(file) == 'string' and not out_name_ok(file) then deny('writing ' .. shown(file)) end
267  return io_output(file)
268end

311local function needs_engine(what)

318local function partial(what, fields)

320    engine_warning(what .. ' has no field ' .. shown(key) .. ' here')
321    return nil
322  end})

324local function font_char(_, code)
325  if type(code) == 'number' and code >= 0 and code <= 0x10FFFF and code == m_floor(code)
326    and query('fontchar ' .. s_format('%d', code)) == '1' then
327    return partial('a character of the current font', {})
328  end
329  return nil
330end
331local function getfont(id)
332  if id == CURRENT_FONT then
333    return partial('the current font', {characters = setmetatable({}, {__index = font_char})})
334  end
335  if type(id) == 'number' and id >= 0 then engine_warning('font ' .. s_format('%.14g', id) .. ' is not known') end
336  return nil
337end

480  for _, var in ipairs({'MKTEXTFM', 'MKTEXPK', 'MKTEXMF', 'MKTEXTEX', 'MKTEXFMT', 'MKOCP', 'MKOFM'}) do
481    os.setenv(var, '0')
482  end
483  kpse_ready = pcall(kpse.set_program_name, 'luatex')
484  local find_file = kpse.find_file
485  t_insert(searchers, function(name)
486    -- LuaTeX's own kpse loader reads a dotted name as a path (pgf.gd.trees.library).
487    local f = find_file((s_gsub(name, '%.', '/')), 'lua') or find_file(name, 'lua')
488    if f then return assert(loadfile_text(f)), f end
489    return "\n\tno texmf module '" .. shown(name) .. "' (kpse)"
490  end)
491  pcall(require, 'lualibs')
492end
493-- debug reaches the upvalues the wrappers above hold (the raw io.open, ...).
494debug, package.loaded.debug = nil, nil
495-- An error value's text, asked of no metamethod (tostring would run the document's __tostring).
496local function error_text(value)
497  local kind = type(value)
498  if kind == 'string' then return value end
499  if kind == 'number' then return s_format('%.14g', value) end
500  return '(error object is a ' .. kind .. ' value)'
501end
502booting = false
503while true do
504  local header = f_read(stdin, '*l')
505  if not header then os_exit(0) end
506  local n = tonumber(header)
507  if not n then os_exit(1) end
508  local chunk = f_read(stdin, n)
509  f_read(stdin, '*l')
510  buf = {}
511  local out, tag
512  local f, err = load_text(chunk)
513  if f then
514    local ok, rterr = pcall(f)
515    if ok then
516      out, tag = t_concat(buf), 'OK'
517    else
518      out, tag = error_text(rterr), 'ERR'
519    end
520  else
521    out, tag = error_text(err), 'ERR'
522  end
523  frame(tag, out)
524end

528  /// The child, registered so a timeout or memory exit (which runs no destructor) kills it too; dropping the guard
529  /// kills and reaps it.

539/// This thread's interpreter for the current conversion (`reset_session` ends it: a persistent worker converts its
540/// papers on one thread, and LuaTeX starts each job with a fresh Lua state in the job's directory).
541#[derive(Default)]
542struct LuaSlot {
543  proc_:       Option<LuaProc>,
544  /// texlua could not be started (reported once per conversion as "texlua not available").
545  unavailable: bool,
546  /// Interpreters that ended mid-conversion (a crash, a broken frame); each is respawned up to `MAX_RESTARTS`.
547  ended:       u32,
548}
549
550/// Restarts per conversion after an interpreter ends; past them, later chunks are not evaluated.
551const MAX_RESTARTS: u32 = 3;
552/// The longest frame the bridge reads: a length past it is a broken protocol, not output.
553const MAX_FRAME: usize = 64 << 20;
554/// The refusals of each severity a chunk reports one by one, by count and by bytes; the rest are counted in a summary
555/// per severity.
556const MAX_DENIALS: usize = 20;
557const MAX_DENIAL_BYTES: usize = 64 << 10;
558
559thread_local! {
560  static LUA: RefCell<LuaSlot> = RefCell::new(LuaSlot::default());
561}

563/// Ends this thread's interpreter and forgets its restarts: called at each conversion's start
564/// (`Converter::initialize_session`), so no Lua state or working directory carries over to the next paper.
565pub fn reset_session() { LUA.with(|slot| *slot.borrow_mut() = LuaSlot::default()); }

567impl Drop for LuaProc {
568  fn drop(&mut self) {
569    // The guard kills and reaps the child, so no zombie outlives the thread.
570    drop(self.child.take());
571    if let Some(path) = self.prelude.take() {
572      let _ = std::fs::remove_file(path);
573    }
574  }

577fn content_hash(text: &str) -> u64 {
578  text.bytes().fold(0xcbf2_9ce4_8422_2325, |hash, byte| {
579    (hash ^ u64::from(byte)).wrapping_mul(0x0100_0000_01b3)
580  })

586  Shared(std::path::PathBuf),
587  Own(std::path::PathBuf),

590/// The prelude script on disk: `latexml_lua_prelude_<uid>_<content hash>.lua` in the temp dir, shared
591/// by this user's processes and used only when it is a regular file (not a link) of this user holding
592/// exactly the prelude — the temp dir is world-writable, and a planted or truncated file would run as
593/// this user. Written aside under a name no other writer uses (a pid and a counter; a name left by an
594/// earlier process of the same pid is skipped) and renamed into place, so a concurrent reader never
595/// sees a partial file; where the shared name cannot be taken, the private copy is used and removed
596/// with the interpreter (`LuaProc`'s drop).
597fn prelude_file() -> Option<Prelude> {
598  use std::io::Write;
599  static WRITES: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
600  let dir = std::env::temp_dir();
601  let hash = content_hash(LUA_PRELUDE);
602  let (own, mut file) = (0..64).find_map(|_| {
603    let own = dir.join(format!(
604      "latexml_lua_prelude_{hash:016x}.{}.{}.lua",
605      std::process::id(),
606      WRITES.fetch_add(1, std::sync::atomic::Ordering::Relaxed)
607    ));
608    let mut options = std::fs::OpenOptions::new();
609    options.write(true).create_new(true);
610    #[cfg(unix)]
611    std::os::unix::fs::OpenOptionsExt::mode(&mut options, 0o644);
612    match options.open(&own) {
613      Ok(file) => Some(Some((own, file))),
614      Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => None,
615      Err(_) => Some(None),
616    }
617  })??;
618  if file.write_all(LUA_PRELUDE.as_bytes()).is_err() {
619    let _ = std::fs::remove_file(&own);
620    return None;
621  }
622  drop(file);
623  #[cfg(unix)]
624  let owner = {
625    use std::os::unix::fs::MetadataExt;
626    std::fs::symlink_metadata(&own).ok()?.uid()
627  };
628  #[cfg(not(unix))]
629  let owner = 0;
630  let shared = dir.join(format!("latexml_lua_prelude_{owner}_{hash:016x}.lua"));
631  let is_ours = |path: &std::path::Path| {
632    std::fs::symlink_metadata(path).is_ok_and(|meta| {
633      #[cfg(unix)]
634      let same_owner = std::os::unix::fs::MetadataExt::uid(&meta) == owner;
635      #[cfg(not(unix))]
636      let same_owner = true;
637      meta.is_file() && same_owner
638    })
639  };
640  if is_ours(&shared) && std::fs::read(&shared).is_ok_and(|bytes| bytes == LUA_PRELUDE.as_bytes()) {
641    let _ = std::fs::remove_file(&own);
642    return Some(Prelude::Shared(shared));
643  }
644  if std::fs::rename(&own, &shared).is_ok() {
645    return Some(Prelude::Shared(shared));
646  }
647  Some(Prelude::Own(own))
648}

650/// The environment variables the texlua child keeps: the program and library search (`PATH`, `LD_LIBRARY_PATH`,
651/// `DYLD_*`, Windows' system ones), kpathsea's (`TEXMF*`, `KPATHSEA*`, the `*INPUTS`/`*FONTS` format paths), the user's home (`TEXMFHOME`), the
652/// locale, the reproducible-date clock and the temp dir. Windows' variable names are case-insensitive, and its usual
653/// spellings are mixed-case (`Path`, `SystemRoot`, `windir`), so there a name is matched in upper case.

657  } else {
658    std::borrow::Cow::Borrowed(name)
659  };
660  let name = name.as_ref();
661  const EXACT: &[&str] = &[
662    "PATH",
663    "HOME",
664    "LANG",
665    "LANGUAGE",
666    "SOURCE_DATE_EPOCH",
667    "FORCE_SOURCE_DATE",
668    "TMPDIR",
669    "TMP",
670    "TEMP",
671    "SYSTEMROOT",
672    "WINDIR",
673    "SYSTEMDRIVE",
674    "USERPROFILE",
675    "HOMEDRIVE",
676    "HOMEPATH",
677    "APPDATA",
678    "LOCALAPPDATA",
679    "PATHEXT",
680    "TEXCONFIG",
681    "TEXFORMATS",
682    "TEXPOOL",
683    "TEXPSHEADERS",
684    "TEXFONTMAPS",
685    "LD_LIBRARY_PATH",
686  ];
687  EXACT.contains(&name)
688    || name.starts_with("DYLD_")
689    || name.starts_with("LC_")

696/// A per-interpreter frame nonce: 128 bits of SipHash keyed by the standard library's random `RandomState` keys,
697/// over the pid, the clock and a counter. The prelude reads it before any other code loads; a chunk never sees it.
698fn session_nonce() -> String {
699  use std::hash::{BuildHasher, Hasher};
700  static SPAWNS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
701  let state = std::collections::hash_map::RandomState::new();
702  let count = SPAWNS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
703  let clock = std::time::SystemTime::now()
704    .duration_since(std::time::UNIX_EPOCH)
705    .map_or(0, |since| since.as_nanos());
706  let word = |half: u64| {
707    let mut hasher = state.build_hasher();
708    hasher.write_u64(half);
709    hasher.write_u32(std::process::id());
710    hasher.write_u128(clock);
711    hasher.write_u64(count);
712    hasher.finish()
713  };
714  format!("{:016x}{:016x}", word(0), word(1))
715}

717fn spawn() -> Option<LuaProc> {
718  // `texlua` takes a script FILE (its `-e` is not lua's inline-chunk flag) — materialize the
719  // prelude in the temp dir, named by its owner and content: one file shared by this user's processes
720  // and version-safe (a per-process name was never removed, and one-shot runs and test processes left ~98,000
721  // of them, exhausting `/tmp`'s inodes).
722  let (prelude_path, prelude) = match prelude_file()? {
723    Prelude::Shared(path) => (path, None),
724    Prelude::Own(path) => (path.clone(), Some(path)),
725  };
726  let mut cmd = Command::new("texlua");
727  cmd
728    .arg(&prelude_path)
729    .stdin(Stdio::piped())
730    .stdout(Stdio::piped())
731    .stderr(Stdio::null())
732    // Untrusted Lua sees only what the texmf lookup and the locale need, not the worker's environment.
733    .env_clear()
734    .envs(std::env::vars_os().filter(|(name, _)| name.to_str().is_some_and(child_env_allowed)))
735    // The TEXMF roots without a trailing separator, as every TeX tool is given them (issue #950).
736    .envs(
737      texmf_root_env()
738        .into_iter()
739        .filter(|(name, _)| child_env_allowed(name)),
740    );
741  // LuaTeX job semantics: chunks resolve relative files against the
742  // document's directory (witness newpax/newpax — `newpax.writenewpax
743  // ("doc-input1")` reads the PDFs shipped next to the manual).
744  let source_dir = latexml_core::state::lookup_string("SOURCEDIRECTORY");
745  if !source_dir.is_empty() && std::path::Path::new(&source_dir).is_dir() {
746    cmd.current_dir(&source_dir);
747  }
748  let Ok(mut child) = cmd.spawn() else {
749    if let Some(path) = prelude {
750      let _ = std::fs::remove_file(path);
751    }
752    return None;
753  };
754  let (Some(stdin), Some(stdout)) = (child.stdin.take(), child.stdout.take()) else {
755    let _ = child.kill();
756    let _ = child.wait();
757    if let Some(path) = prelude {
758      let _ = std::fs::remove_file(path);
759    }
760    return None;
761  };
762  let mut proc_ = LuaProc {
763    child: Some(latexml_core::watchdog::kill_on_exit(child)),
764    stdin,
765    stdout: BufReader::new(stdout),
766    prelude,
767    nonce: session_nonce(),
768  };
769  let line = format!("{}\n", proc_.nonce);
770  if proc_.stdin.write_all(line.as_bytes()).is_err() {
771    return None;
772  }
773  Some(proc_)
774}

776/// Service one live tex-state query from the Lua side (the mirror half of
777/// rebind-as-we-emulate): `get{count,dimen} <key>` reads a register from the
778/// engine State, `set{count,dimen} [global] <key> <value>` writes one. Keys
779/// are numeric register indices (`tex.count[255]` → `\count255`) or
780/// `\countdef`'d names (`tex.count.scratchcounter` → `\scratchcounter`).
781/// Counts answer as integers, dimens in scaled points (LuaTeX convention).
782/// Unknown ops or lookup failures answer "0" — the pre-mirror stub value —
783/// so a chunk probing state we don't model degrades exactly as before.
784fn service_query(payload: &str) -> String {
785  use latexml_core::{
786    common::{dimension::Dimension, number::Number},
787    definition::{argument::ArgWrap, register::RegisterValue},
788    state,
789  };
790  let mut it = payload.split_whitespace();
791  let op = it.next().unwrap_or("");
792  let mut key = it.next().unwrap_or("");
793  // `fontid <name>`: whether `\<name>` is a font identifier (`\font\tenln=line10`), read-only; the probe interns no
794  // name (a document's names would stay in the arena for good).
795  if let Some(name) = payload.strip_prefix("fontid ") {
796    let font = latexml_core::common::arena::get(format!("fontinfo_\\{name}")).is_some_and(|key| {
797      state::with_value_sym(key, |font| matches!(font, Some(state::Stored::Font(_))))
798    });
799    return if font { "1" } else { "0" }.to_string();
800  }
801  // `fontchar <code>`: the prelude's `font` mirror asks whether the current font has a character, read-only.
802  if op == "fontchar" {
803    let code = key.parse::<i64>().unwrap_or(-1);
804    let has = crate::etex::font_has_char(&latexml_core::T_CS!("\\font"), code);
805    return if has { "1" } else { "0" }.to_string();
806  }
807  let global = key == "global";
808  if global {
809    key = it.next().unwrap_or("");
810  }
811  let (cs, params) = match key.parse::<i64>() {
812    Ok(n) => {
813      let reg = if op.ends_with("dimen") {
814        "\\dimen"
815      } else {
816        "\\count"
817      };
818      (reg.to_string(), vec![ArgWrap::Number(Number(n))])
819    },
820    Err(_) => (format!("\\{key}"), Vec::new()),
821  };
822  match op {
823    "getcount" | "getdimen" => match state::lookup_register(&cs, params).ok().flatten() {
824      Some(RegisterValue::Number(n)) => n.0.to_string(),
825      Some(RegisterValue::Dimension(d)) => d.0.to_string(),
826      _ => "0".to_string(),
827    },
828    "setcount" | "setdimen" => {
829      let value: i64 = it.next().and_then(|v| v.parse().ok()).unwrap_or(0);
830      let rv = if op == "setdimen" {
831        RegisterValue::Dimension(Dimension(value))
832      } else {
833        RegisterValue::Number(Number(value))
834      };
835      let scope = global.then_some(state::Scope::Global);
836      let _ = state::assign_register(&cs, rv, scope, params);
837      String::new()
838    },
839    _ => "0".to_string(),
840  }
841}

843/// Execute one Lua chunk in this thread's persistent interpreter.
844///
845/// `Ok(text)` is everything the chunk `tex.print`ed (to be retokenized with
846/// current catcodes by the caller); `Err(msg)` is a load/runtime Lua error or
847/// bridge unavailability ("texlua not available" — callers degrade to a no-op
848/// expansion, matching what the document would get from a TeX engine without
849/// Lua). Each call the sandbox refused is a `Warning` where lualatex refuses it too and an `Error` otherwise; a
850/// chunk that fails after such a `Warning`, and an interpreter that ended mid-chunk (its output lost), are an `Error`
851/// (the outer `Err`, should one escalate to a Fatal).
852pub fn lua_exec(chunk: &str) -> latexml_core::common::error::Result<Result<String, String>> {
853  let framed = exec_framed(chunk);
854  use crate::prelude::*;
855  let failed_after_refusal = framed.failed_after_refusal();
856  for refusal in framed.denials.shown {
857    match refusal.kind {
858      Refused::Sandbox => Error!("lua", "sandbox", refusal.message),
859      Refused::AsLualatex => Warn!("lua", "sandbox", refusal.message),
860      Refused::EngineGap => Warn!("lua", "engine", refusal.message),
861    }
862  }
863  let unlisted =
864    |n: usize| format!("{n} more calls the sandbox refused in this chunk are not listed");
865  if framed.denials.errors.more > 0 {
866    Error!("lua", "sandbox", unlisted(framed.denials.errors.more));
867  }
868  if framed.denials.warnings.more > 0 {
869    Warn!("lua", "sandbox", unlisted(framed.denials.warnings.more));
870  }
871  if framed.denials.gaps.more > 0 {
872    Warn!(
873      "lua",
874      "engine",
875      format!(
876        "{} more calls texlua has no engine for in this chunk are not listed",
877        framed.denials.gaps.more
878      )
879    );
880  }
881  if let Some(ended) = framed.ended {
882    Error!("lua", "interpreter", ended);
883  } else if let Some(failure) = failed_after_refusal {
884    Error!("lua", "sandbox", failure);
885  }
886  Ok(framed.result)
887}

889/// One chunk's run: its result, the sandbox's refusals and engine gaps ('D', 'W', 'G' frames,
890/// OXIDIZED_DESIGN_DIVERGENCES #1081) — returned
891/// apart, as a pcall in the chunk can swallow the Lua error, never the record — and the report of an interpreter
892/// that ended during it.
893struct Framed {
894  result:  Result<String, String>,
895  denials: Denials,
896  ended:   Option<String>,
897}
898
899impl Framed {
900  /// The Error for a chunk whose Lua error followed a call refused with lualatex's values: it used that call's nil
901  /// (`io.popen(cmd):read()`) and is lost, as under lualatex (rc=1). An engine gap alone does not count (lualatex
902  /// performs that call), nor a chunk whose 'D' refusal is already its Error (a 'D' call raises the Lua error); any
903  /// other chunk's Lua error stays the caller's Info.
904  fn failed_after_refusal(&self) -> Option<String> {
905    let refused_as_error = self.denials.errors.listed + self.denials.errors.more > 0;
906    match (&self.result, &self.denials.first_warning, &self.ended) {
907      (Err(failure), Some(call), None) if !refused_as_error => Some(format!(
908        "the chunk failed after the sandbox refused {call}: {failure}"
909      )),
910      _ => None,
911    }
912  }
913}
914
915/// A chunk's refusals and engine gaps in order: the first `MAX_DENIALS` of each kind (within `MAX_DENIAL_BYTES` each)
916/// kept, the rest counted, so a flood of Warnings never hides an Error.
917#[derive(Default)]
918struct Denials {
919  shown:         Vec<Refusal>,
920  errors:        Budget,
921  warnings:      Budget,
922  gaps:          Budget,
923  /// The first call refused with lualatex's values, named should the chunk then fail.
924  first_warning: Option<String>,
925}
926
927/// One kind's share of a chunk's refusal record.
928#[derive(Default)]
929struct Budget {
930  listed: usize,
931  bytes:  usize,
932  more:   usize,
933}
934
935/// One refused call or engine gap.
936struct Refusal {
937  kind:    Refused,
938  message: String,
939}
940
941/// A frame's kind of refusal.
942#[derive(Clone, Copy, PartialEq, Eq, Debug)]
943enum Refused {
944  /// 'D': only the sandbox refuses the call (lualatex would perform it), an `Error:lua:sandbox`.
945  Sandbox,
946  /// 'W': lualatex refuses it too, and the call returned lualatex's values; a `Warning:lua:sandbox`.
947  AsLualatex,
948  /// 'G': an engine layer texlua lacks and lualatex performs; a `Warning:lua:engine`.
949  EngineGap,
950}
951
952impl Denials {
953  fn push(&mut self, kind: Refused, message: String) {
954    if kind == Refused::AsLualatex && self.first_warning.is_none() {
955      // The prelude's 'W' message is `<call> was not run (…)` or `writing <name> was refused (…)`.
956      let call = message
957        .rsplit_once(" was ")
958        .map_or(message.as_str(), |(call, _)| call);
959      self.first_warning = Some(call.to_string());
960    }
961    let budget = match kind {
962      Refused::Sandbox => &mut self.errors,
963      Refused::AsLualatex => &mut self.warnings,
964      Refused::EngineGap => &mut self.gaps,
965    };
966    if budget.listed < MAX_DENIALS && budget.bytes + message.len() <= MAX_DENIAL_BYTES {
967      budget.listed += 1;
968      budget.bytes += message.len();
969      self.shown.push(Refusal { kind, message });
970    } else {
971      budget.more += 1;
972    }
973  }
974}

976/// A frame header's tag and length: `<tag> <nonce> <length>`, with this interpreter's nonce (a header without it
977/// came from no prelude frame, OXIDIZED_DESIGN_DIVERGENCES #1081) and a length the bridge reads.
978fn parse_header<'h>(header: &'h str, nonce: &str) -> Result<(&'h str, usize), FrameError> {
979  let mut fields = header.trim_end().splitn(3, ' ');
980  let (tag, sent, len) = (fields.next(), fields.next(), fields.next());
981  match (tag, len.map(str::parse::<usize>)) {
982    (Some(tag), Some(Ok(n))) if sent == Some(nonce) && n <= MAX_FRAME => Ok((tag, n)),
983    _ if sent == Some(nonce) => Err(FrameError::Protocol(format!(
984      "a frame header '{}'",
985      header.trim_end()
986    ))),
987    _ => Err(FrameError::Protocol(
988      "output that is not a frame (no session nonce)".to_string(),
989    )),
990  }
991}

993/// Why a frame exchange failed: the pipe (the child ended) or a frame no prelude writes.
994enum FrameError {
995  Io(std::io::Error),
996  Protocol(String),
997}
998
999impl From<std::io::Error> for FrameError {
1000  fn from(e: std::io::Error) -> Self { FrameError::Io(e) }
1001}

1003/// Sends `chunk` and reads frames up to its final `OK`/`ERR`, servicing `Q`ueries and collecting refusals (`D`, `W`)
1004/// and engine gaps (`G`).
1005fn exchange(
1006  p: &mut LuaProc,
1007  chunk: &str,
1008  denials: &mut Denials,
1009) -> Result<(bool, String), FrameError> {
1010  use std::io::Read;
1011  p.stdin.write_all(format!("{}\n", chunk.len()).as_bytes())?;
1012  p.stdin.write_all(chunk.as_bytes())?;
1013  p.stdin.write_all(b"\n")?;
1014  p.stdin.flush()?;
1015  // The chunk may interleave 'Q'uery frames (live tex-state reads/writes)
1016  // before its final OK/ERR frame — service each against engine State.
1017  loop {
1018    let mut header = String::new();
1019    (&mut p.stdout).take(64).read_line(&mut header)?;
1020    if header.is_empty() {
1021      return Err(FrameError::Io(std::io::ErrorKind::UnexpectedEof.into()));
1022    }
1023    let (tag, n) = parse_header(&header, &p.nonce)?;
1024    let mut body = vec![0u8; n];
1025    p.stdout.read_exact(&mut body)?;
1026    let mut body = String::from_utf8_lossy(&body).into_owned();
1027    if cfg!(windows) {
1028      // Undo the text-mode stdout: drop the `\r` written before each `\n` (a `\r\n` the chunk printed came
1029      // through as `\r\r\n`, and stays `\r\n`).
1030      body = body.replace("\r\n", "\n");
1031    }
1032    match tag {
1033      "D" => denials.push(Refused::Sandbox, body),
1034      "W" => denials.push(Refused::AsLualatex, body),
1035      "G" => denials.push(Refused::EngineGap, body),
1036      "Q" => {
1037        let answer = service_query(&body);
1038        p.stdin
1039          .write_all(format!("A {}\n", answer.len()).as_bytes())?;
1040        p.stdin.write_all(answer.as_bytes())?;
1041        p.stdin.flush()?;
1042      },
1043      "OK" => return Ok((true, body)),
1044      "ERR" => return Ok((false, body)),
1045      _ => return Err(FrameError::Protocol(format!("a frame tagged '{tag}'"))),
1046    }
1047  }
1048}

1050fn exec_framed(chunk: &str) -> Framed {
1051  let mut framed = Framed {
1052    result:  Ok(String::new()),
1053    denials: Denials::default(),
1054    ended:   None,
1055  };
1056  // (a speculative run is undone, the interpreter's globals could not be: `\\author{First\\luaexec{n=(n or 0)+1}}`, which a second
1057  // `\\author` replaces, left its count for the body's `\\luaexec{tex.print(n or 0)}`, lualatex's 0)
1058  if latexml_core::gullet::in_speculative_digestion() {
1059    framed.result = Err("a speculative run evaluates no Lua".to_string());
1060    return framed;
1061  }
1062  LUA.with(|slot| {
1063    let mut slot = slot.borrow_mut();
1064    if slot.unavailable {
1065      framed.result = Err("texlua not available".to_string());
1066      return;
1067    }
1068    if slot.ended > MAX_RESTARTS {
1069      framed.result = Err(format!(
1070        "texlua ended {} times in this conversion",
1071        slot.ended
1072      ));
1073      return;
1074    }
1075    if slot.proc_.is_none() {
1076      slot.proc_ = spawn();
1077      if slot.proc_.is_none() {
1078        slot.unavailable = true;
1079        framed.result = Err("texlua not available".to_string());
1080        return;
1081      }
1082    }
1083    // Windows texlua reads stdin in text mode: a `\r\n` arrives as `\n`, one byte short of the frame's count, so
1084    // it is sent as the `\n` it would read (the prelude's `wire_len` covers the other direction).
1085    let chunk: std::borrow::Cow<str> = if cfg!(windows) {
1086      chunk.replace("\r\n", "\n").into()
1087    } else {
1088      chunk.into()
1089    };
1090    let Some(proc_) = slot.proc_.as_mut() else {
1091      return;
1092    };
1093    framed.result = match exchange(proc_, &chunk, &mut framed.denials) {
1094      Ok((true, body)) => Ok(body),
1095      Ok((false, body)) => Err(body),
1096      Err(error) => {
1097        // The interpreter ended (a crash, `os.exit`) or broke the protocol: its output is lost and its state gone.
1098        // Dropping it kills and reaps the child; the next chunk starts a fresh one, up to MAX_RESTARTS times.
1099        let ended = slot.proc_.take();
1100        slot.ended += 1;
1101        let why = match error {
1102          FrameError::Io(e) => {
1103            // A closed pipe is a child that is ending: say how (`signal: 11 (SIGSEGV)`, `exit status: 0`).
1104            let status = ended
1105              .and_then(|mut p| p.child.take())
1106              .and_then(|child| child.finish(std::time::Duration::from_millis(500)));
1107            match status {
1108              Some(status) => format!("the texlua interpreter ended ({status})"),
1109              None => format!("the texlua interpreter ended ({e})"),
1110            }
1111          },
1112          FrameError::Protocol(what) => format!("the texlua interpreter sent {what}"),
1113        };
1114        let next = if slot.ended > MAX_RESTARTS {
1115          "; later chunks are not evaluated"
1116        } else {
1117          "; the next chunk starts a fresh interpreter"
1118        };
1119        framed.ended = Some(format!(
1120          "{why}: this chunk's output and the Lua state are lost{next}"
1121        ));
1122        Err(why)
1123      },
1124    };
1125  });
1126  framed
1127}
