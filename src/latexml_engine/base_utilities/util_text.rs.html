6/// `\unitlength` in sp, exact (65536, 1pt, when it is undefined): the picture
7/// unit every `{picture}` and pict2e coordinate is multiplied by (Perl
8/// `picScale`, latex_constructs.pool.ltxml:4896-4921). A skip that calc's
9/// `\setlength` put there is its natural width (the register coerces it).
10pub fn unitlength_sp() -> Result<i64> {
11  Ok(match lookup_register("\\unitlength", Vec::new())? {
12    Some(value) => Dimension::from(value).value_of(),
13    None => 65536,
14  })
15}

17/// Has this diagnostic already been reported? Marks `key` reported either way.
18///
19/// Port of Perl's report-once diagnostic idiom, e.g. `\selectfont`
20/// (latex_constructs.pool.ltxml L5210-5211):
21///
22/// ```perl
23/// elsif (!LookupValue("reported_unrecognized_font_family_$family")) {
24///   AssignValue("reported_unrecognized_font_family_$family", 1, 'global');
25///   Info('unexpected', $family, $_[0], "Unrecognized font family '$family'."); }
26/// ```
27///
28/// So a caller reads `if !already_reported(&key) { Info!(…) }`, mirroring the
29/// Perl `elsif`. An unrecognized font family/series/shape — or a missing font
30/// encoding — is announced once per document, not once per occurrence: the
31/// font switch that provokes it typically sits inside a macro that a document
32/// expands in every table cell (witness 2503.04421, whose `\XSolidBrush`
33/// column produced 28 identical `Info:unexpected:ding` lines).
34///
35/// Global scope, as in Perl: the suppression has to outlive the group the
36/// diagnostic fired in, or a grouped `{\dingfamily …}` reports again each time.
37pub fn already_reported(key: &str) -> bool {
38  if lookup_bool(key) {
39    return true;
40  }
41  assign_value(key, true, Some(Scope::Global));
42  false
43}

45/// Join token groups with a conjunction token between them.
46///
47/// Port of Perl `JoinTokens($conjunction, @things)` (Base_Utility.pool.ltxml L142-148).
48#[allow(dead_code)]
49pub fn join_tokens(conjunction: &Tokens, things: Vec<Tokens>) -> Tokens {
50  if things.is_empty() {
51    return Tokens::new(vec![]);
52  }
53  let mut result: Vec<Token> = Vec::new();
54  let mut first = true;
55  for thing in things {
56    if !first {
57      result.extend_from_slice(conjunction.unlist_ref());
58    }
59    result.extend(thing.unlist());
60    first = false;
61  }
62  Tokens::new(result)
63}

69/// Perl: showFrontmatter($entry) — debug formatter for a frontmatter entry.
70pub(super) fn show_frontmatter(entry: &TagData) -> String {
71  let mut attrs: Vec<String> = entry.attr.iter().map(|(k, v)| s!("{k}={v}")).collect();
72  attrs.sort();
73  let content: String = entry
74    .content
75    .iter()
76    .map(|c| match c {
77      TagContent::PlaceKeeper => "place_keeper".to_string(),
78      TagContent::Box(d) => d.to_string(),
79      TagContent::Entry(e) => show_frontmatter(e),
80    })
81    .collect();
82  s!("{}  [{}] {}", entry.tag, attrs.join(","), content)
83}

85/// The tags `\lx@make@tags{type}` collects into an `ltx:tags` (Perl Base_Utility.pool.ltxml:975-983), one per declared
86/// role (`type_tag_formatter`), in role order, each with whether it is printed where it is built: the role-less tag is
87/// the target's own label (`\item[Espa~na]`, a bibliography entry's `\@biblabel`), the others are the forms only a
88/// reference prints (refnum, typerefnum, autoref, the cleveref forms). A caller that typesets those only where a
89/// reference prints them (sect11.rs `\lx@bibitem`, the `\b@key` a `\cite` prints) takes them apart here.
90pub fn tag_forms(ttype: &Tokens) -> Result<Vec<(bool, Tokens)>> {
91  // Pull the (role -> formatter) pairs out of HashStored via with_value
92  // so we don't clone the whole hashmap envelope; the per-role tokens
93  // are Copy and the per-role String arm just dereferences a SymStr.
94  let role_formatters: Vec<(String, Option<Token>)> =
95    with_value("type_tag_formatter", |v| match v {
96      Some(Stored::HashStored(formatters)) => {
97        let keys_sym: Vec<_> = formatters.keys().copied().collect();
98        let mut sorted_keys: Vec<String> = with_many(&keys_sym, |keys| {
99          keys.into_iter().map(str::to_owned).collect()
100        });
101        sorted_keys.sort();
102        sorted_keys
103          .into_iter()
104          .map(|role| {
105            let ft = match formatters.get(&role) {
106              Some(Stored::Token(t)) => Some(*t),
107              Some(Stored::String(sym)) => Some(Token!(sym * sym, Catcode::CS)),
108              _ => None,
109            };
110            (role, ft)
111          })
112          .collect()
113      },
114      _ => Vec::new(),
115    });
116  let mut forms = Vec::new();
117  for (role, formatter_opt) in role_formatters {
118    let Some(formatter_token) = formatter_opt else {
119      continue;
120    };
121    // A tag only a reference prints (`type_tag_deferred`) holds its diagnostics (`\lx@tag@intags@held`).
122    let constructor = if with_mapping("type_tag_deferred", &role, |held| held.is_some()) {
123      "\\lx@tag@intags@held"
124    } else {
125      "\\lx@tag@intags"
126    };
127    let invoked_tag = Invocation!(T_CS!(constructor), vec![
128      Tokens!(T_OTHER!(role.as_str())),
129      build_invocation(formatter_token, vec![Some(ttype.clone())])?
130    ]);
131    // A tag a reference prints (refnum, typerefnum, autoref, the cleveref forms) is built where its target is (an
132    // eqnarray row's in its math group, where a document may rebind `~`: vdm.sty:135-138, 202-205
133    // `\everymath{\let~\hook}`, a `\vbox{\ialign…}` that re-stepped the equation, an endless recursion on 1601.02132)
134    // and printed in text. Its `~` is the kernel's no-break space, LaTeX's `\nobreakspace`, which babel's `~` also is
135    // except in a language that makes it a shorthand; such a shorthand, or a document's own text `~`, no longer reaches
136    // these tags (OXIDIZED_DESIGN_DIVERGENCES #98 addendum). The role-less tag is the target's own label, printed where
137    // it is built (`\item[Espa~na]`), and keeps the document's `~`. Guard
138    // `perfect_kernel_batch63::autoref_tag_tilde_is_the_kernel_space`.
139    // A reference form re-reads the label's source, which the role-less tag typesets (`\lx@reread@copy`): an
140    // `\item[$\hat\mathbf{e}$]`'s typerefnum, a `\tag{$\hat\mathbf{t}$}`'s refnum give no second
141    // OXIDIZED_DESIGN_DIVERGENCES #721 Warning (guard `perfect_kernel_batch71a::undelimited_argument_warning_once_per_occurrence`).
142    if role.is_empty() {
143      forms.push((true, invoked_tag));
144    } else {
145      let mut wrapped = vec![
146        T_CS!("\\begingroup"),
147        T_CS!("\\lx@reread@copy"),
148        T_CS!("\\let"),
149        T_ACTIVE!('~'),
150        T_CS!("\\lx@tag@texttilde"),
151      ];
152      wrapped.extend(invoked_tag.unlist());
153      wrapped.push(T_CS!("\\endgroup"));
154      forms.push((false, Tokens::new(wrapped)));
155    }
156  }
157  Ok(forms)
158}

191/// Walk a `Digested` and concatenate its text content (for attribute use,
192/// matching Perl's `setAttribute(..., DigestText(...))` semantics). Tbox
193/// children contribute their text; nested Lists recurse; `\hskip`-style
194/// Whatsits (which are side-effect-only constructors with no text content)
195/// fall back to `dimension_to_spaces(width)` instead of reverting to the
196/// macro name. All other Whatsits use their normal `get_string` path.
197/// (Rust-only helper, previously in latex_constructs; moved here since
198/// digest_front_matter needs it for the creator `before` separators.)
199pub fn digested_to_text(d: &Digested) -> Result<String> {
200  let mut out = String::new();
201  match d.data() {
202    DigestedData::TBox(b) => out.push_str(&b.borrow().get_string()?),
203    DigestedData::List(l) => {
204      for child in l.borrow().boxes.iter() {
205        out.push_str(&digested_to_text(child)?);
206      }
207    },
208    DigestedData::Whatsit(w) => {
209      let w = w.borrow();
210      match w.get_property("width").as_deref() {
211        Some(Stored::Dimension(width)) => {
212          // The whatsit's OWN font — see tex_glue::dimension_to_spaces.
213          let font = w.get_font()?;
214          out.push_str(&crate::tex_glue::dimension_to_spaces(
215            *width,
216            font.as_deref(),
217          ));
218        },
219        _ => {
220          out.push_str(&w.get_string()?);
221        },
222      }
223    },
224    _ => out.push_str(&d.to_string()),
225  }
226  Ok(out)
227}

229/// Whether digested material typesets something a reader sees: a character that is not a space, an image (a whatsit
230/// carrying a `graphic` or image `candidates`), a cross-reference or citation, whose text the post-processor
231/// fills in (`fills_generated_text`), or a constructor whose template prints text of its own (`template_prints_text`,
232/// `\LaTeX`), anywhere in it — through lists, the whatsit arguments it shows
233/// (`shown_arguments`, as [`ink_text`] reads them) and bodies, and alignment cells. Paragraph and spacing commands,
234/// rules and empty boxes are not content, nor is an argument LaTeX only tests: the position letter of
235/// `\begin{tabular}[t]`, `\parbox[t]`, `{minipage}[t]`, `\makebox[w][l]` or `\shortstack[l]` (a `Position` parameter,
236/// base_parameter_types.rs) — counted, the deproc title code's `\begin{tabular}[t]{l}\bf\rule{\z@}{24pt}…` with an
237/// emptied `\@author` (iclr*_conference.sty:83-93, arxiv.sty, colm, corl, nips) kept its `\@maketitle` deposit, an empty
238/// strut table before the first section (HF14; 94 of 6,000 sampled papers, 2610.20409, 2310.10873).
239/// A whatsit's own string is its reversion (Whatsit.pm `toString`), so a digested `\noindent\par`
240/// stringifies as source; this walks the boxes instead. Used to keep a speculative typesetting (a
241/// class's `\maketitle` replay, a one-shot page overlay) only when it shows something.
242pub fn typesets_content(d: &Digested) -> bool {
243  match d.data() {
244    DigestedData::TBox(b) => b
245      .borrow()
246      .get_string()
247      .is_ok_and(|s| s.chars().any(|c| !c.is_whitespace())),
248    DigestedData::List(l) => l.borrow().boxes.iter().any(typesets_content),
249    DigestedData::Whatsit(w) => {
250      let w = w.borrow();
251      // (an image: graphics.sty's `\@includegraphics` records its `graphic`, graphicx's `\@includegraphicx` its
252      // `path`, both the image's `candidates`; its file name is no shown argument)
253      w.get_property("graphic").is_some()
254        || w.get_property("candidates").is_some()
255        || fills_generated_text(&w)
256        // A constructor whose template prints text of its own, without an argument or a body (`\LaTeX`, sect03.rs:
257        // the logo alone in a textpos block, the niepraschk-eso-pic manual).
258        || w.get_definition().replacement_template().is_some_and(template_prints_text)
259        // A tabular or `\halign` keeps its rows in its `alignment` property (tex_tables.rs): a class's `\maketitle`
260        // that is only a table typesets it (simplecv's header, brandeis-problemset's title table; dropped since 59p).
261        || printed_properties(&w).iter().any(typesets_content)
262        // (a math character, its `glyph`: `\mathchar`; a constant template's text is `template_prints_text`'s)
263        || printed_glyph(&w).is_some()
264        || w
265          .get_args()
266          .iter()
267          .zip(shown_arguments(&w))
268          .any(|(arg, shown)| shown && arg.as_ref().is_some_and(typesets_content))
269        || w
270          .get_body()
271          .ok()
272          .flatten()
273          .is_some_and(|body| typesets_content(&body))
274    },
275    DigestedData::Alignment(a) => a.borrow().rows().iter().any(|row| {
276      row
277        .get_columns()
278        .iter()
279        .any(|cell| cell.boxes.as_ref().is_some_and(typesets_content))
280    }),
281    _ => false,
282  }
283}

285/// Whether digested material prints nothing: every item in it is one known to leave no ink ([`first_ink`] finds
286/// none). The inverse of [`typesets_content`], failing safe: what is not known to print nothing is ink. A frontmatter
287/// note is dropped as one with no ink only on this test, the XML's (frontmatter_api.rs) and the frontmatter observer's
288/// (the rehearsal's note event, `Event::inkless`) alike (OXIDIZED_DESIGN_DIVERGENCES #916): a false "no ink" loses the
289/// note's content, a false "ink" keeps a blank note.
290pub fn leaves_no_ink(d: &Digested) -> bool { first_ink(d).is_none() }

292/// The first item of digested material not known to print nothing ([`leaves_no_ink`]), described: its command, or its
293/// text. Known to print nothing are
294/// - a box of white space (an interword space, a tie's no-break space) and a box Perl marks `isSpace` (glue, a kern,
295///   `\quad`, a phantom, `\indent`/`\noindent`: Box.pm:33-38 and the pool's `isSpace` properties), but a leader;
296/// - a rule with no width or no height and depth (a strut);
297/// - a line or page break, `\par`, a penalty, `\leavevmode` (`INKLESS_COMMANDS`);
298/// - a transparent wrapper whose arguments (but options and a position keyword) and body print nothing: a plain box or
299///   a formula (`INKLESS_WRAPPERS`: `\makebox[1em]{}`, `\raisebox{1ex}{}`, `$ $`), or a constructor whose whole
300///   construction is a template printing only its arguments or body (`prints_only_arguments`: `\mbox{}`, a font
301///   switch, `$\mathbb{}$`).
302///
303/// Anything else is ink: a constructor's own text, with or without a font (revsymb's `\openone`, `"1"` in the
304/// blackboard font; `\dots`, `\LaTeX`, `$\vdots$`), an element of its own (`\textsuperscript{}`'s `ltx:sup`, a framed
305/// box `\fbox{}`), a closure-built construction (`\hypertarget{a}{}`) or one with a construction hook (`\emph{}`), an
306/// empty template standing for what it does not show (the observer rehearsal's `\lx@fm@obs@url`, its footnote mark), a
307/// leader (`\hrulefill`), an image, a cross-reference, an alignment, an undefined command's error marker — and any
308/// item, blank or not, whose font carries a background colour: `\colorbox{c}{…}` is `\hbox{\pagecolor{c}…}`
309/// (color.sty, color_sty.rs), whose `\pagecolor` leaves an empty box in the coloured font, and pdflatex prints the
310/// coloured block (`\colorbox{red}{}`, `\colorbox{yellow}{\hspace{2em}}`, inside `\mbox{}` too).
311pub fn first_ink(d: &Digested) -> Option<String> {
312  if d
313    .get_font()
314    .ok()
315    .flatten()
316    .is_some_and(|font| font.get_background().is_some())
317  {
318    return Some(format!("a background colour: {d}"));
319  }
320  if is_spacing_item(d) || is_blank_item(d) {
321    return None;
322  }
323  match d.data() {
324    DigestedData::TBox(b) => {
325      let b = b.borrow();
326      match b.get_string() {
327        Ok(s) => Some(s.to_string()),
328        Err(_) => Some(b.to_string()),
329      }
330    },
331    DigestedData::List(l) => l.borrow().boxes.iter().find_map(first_ink),
332    DigestedData::Whatsit(w) => {
333      let w = w.borrow();
334      let definition = w.get_definition();
335      let cs = definition.get_cs_name();
336      let truthy = |key: &str| {
337        w.get_property(key)
338          .is_some_and(|v| !matches!(*v, Stored::Bool(false) | Stored::Int(0)))
339      };
340      let dimension = |key: &str| match w.get_property(key).as_deref() {
341        Some(Stored::Dimension(d)) => d.value_of(),
342        _ => 0,
343      };
344      if INKLESS_COMMANDS.contains(&cs.as_ref()) {
345        return None;
346      }
347      // (a rule shows with a width and a height or depth: LaTeX's `\rule` its `width`, `height` and `depth`
348      // (sect12.rs); a primitive rule its `cached_*` sizes, a `\vrule` of no stated height or depth the running
349      // one, of width 0 `invisible` (tex_box.rs, OXIDIZED_DESIGN #176))
350      let shows = match cs.as_ref() {
351        "\\rule" => Some(dimension("width") != 0 && dimension("height") + dimension("depth") != 0),
352        "\\vrule" => Some(
353          !truthy("invisible")
354            && (w.get_property("rheight").is_none()
355              || w.get_property("rdepth").is_none()
356              || dimension("cached_height") + dimension("cached_depth") != 0),
357        ),
358        "\\hrule" => Some(dimension("cached_height") + dimension("cached_depth") != 0),
359        _ => None,
360      };
361      if let Some(shows) = shows {
362        return shows.then(|| cs.to_string());
363      }
364      let body = w.get_body().ok().flatten();
365      let wrapper = INKLESS_WRAPPERS.contains(&cs.as_ref())
366        || definition
367          .replacement_template()
368          .is_some_and(prints_only_arguments);
369      if !wrapper {
370        return Some(cs.to_string());
371      }
372      // (every argument but its options and a position keyword: a verbatim one too, which a wrapper may print though
373      // `shown_arguments` does not count it — failing safe, as for jhep's `\href`, 0704.3719)
374      let kinds = argument_kinds(&w);
375      w.get_args()
376        .iter()
377        .enumerate()
378        .filter(|(i, _)| kinds.as_ref().is_none_or(|k| k.get(*i) != Some(&0)))
379        .find_map(|(_, arg)| arg.as_ref().and_then(first_ink))
380        .or_else(|| body.as_ref().and_then(first_ink))
381    },
382    DigestedData::Alignment(_) => Some("an alignment".to_string()),
383    DigestedData::Postponed(t) => Some(t.to_string()),
384    DigestedData::KeyVals(_) | DigestedData::RegisterValue(_) | DigestedData::Comment(_) => None,
385  }
386}

388/// Whether digested material is spacing, printing no ink but a gap between what it stands between ([`first_ink`]; the
389/// frontmatter label reader's separator, `queue_digest::label_ink`): a box of white space (an interword space, a tie's
390/// no-break space, math glue `\,` in a formula) and a box or whatsit Perl marks `isSpace` (Box.pm:33-38 and the pool's
391/// `isSpace` properties: `\!`'s zero-width-space box, glue, a kern, `\quad`, `\hspace`, a phantom), but a leader
392/// (`\hrulefill`, ink though Perl marks some `isSpace`) and what prints no gap ([`is_blank_item`]).
393pub fn is_spacing_item(d: &Digested) -> bool {
394  if is_blank_item(d) {
395    return false;
396  }
397  let truthy = is_space_marked(d);
398  match d.data() {
399    DigestedData::TBox(b) => {
400      truthy
401        || b
402          .borrow()
403          .get_string()
404          .is_ok_and(|s| s.chars().all(char::is_whitespace))
405    },
406    DigestedData::Whatsit(w) => {
407      truthy && !LEADERS.contains(&w.borrow().get_definition().get_cs_name().as_ref())
408    },
409    _ => false,
410  }
411}

413/// Whether digested material prints nothing and no gap — no ink ([`first_ink`]) and no spacing ([`is_spacing_item`]):
414/// a box of no text (`\-`'s), a kern Perl marks `isSpace` of zero width (the italic correction
415/// `\/`, `width => 0em`, TeX_Fonts.pool.ltxml:162, a kern tex.web §1113; `\kern0pt`), and glue of no natural width,
416/// stretch or shrink (`\hskip0pt`, `\hspace{0pt}`: `is_zero_glue`). `\textsuperscript{1\-2}`, `1\/2` and
417/// `1\hskip0pt 2` print "12", the label 12 (review rev92b_r3 P1 and rev92b_r4 S1: read as spacing, the labels 1 and
418/// 2, its `12` line on no name).
419pub fn is_blank_item(d: &Digested) -> bool {
420  if is_zero_glue(d) {
421    return true;
422  }
423  match d.data() {
424    DigestedData::TBox(b)
425      if b.borrow().get_string().is_ok_and(|s| s.is_empty()) && !is_space_marked(d) =>
426    {
427      true
428    },
429    DigestedData::TBox(_) | DigestedData::Whatsit(_) => {
430      is_space_marked(d)
431        && d
432          .get_property("isKern")
433          .is_some_and(|v| !matches!(*v, Stored::Bool(false) | Stored::Int(0)))
434        && matches!(d.get_property("width").as_deref(), Some(Stored::Dimension(w)) if w.value_of() == 0)
435    },
436    _ => false,
437  }
438}

440/// Whether digested material is glue whose natural width, stretch and shrink are all zero ([`is_blank_item`]): an
441/// `\hskip` whatsit by its glue argument (tex_glue.rs), an `\hspace` box by the glue it states (`glue`, sect12.rs).
442fn is_zero_glue(d: &Digested) -> bool {
443  let zero = |g: &Glue| g.skip == 0 && g.plus.unwrap_or(0) == 0 && g.minus.unwrap_or(0) == 0;
444  match d.data() {
445    DigestedData::Whatsit(w) => {
446      let w = w.borrow();
447      w.get_definition().get_cs_name().trim_end() == "\\hskip"
448        && matches!(
449          w.get_args().first().and_then(Option::as_ref).map(Digested::data),
450          Some(DigestedData::RegisterValue(RegisterValue::Glue(g))) if zero(g)
451        )
452    },
453    DigestedData::TBox(_) => {
454      matches!(d.get_property("glue").as_deref(), Some(Stored::Glue(g)) if zero(g))
455    },
456    _ => false,
457  }
458}

467/// `<svg:svg>`), `None` for any other whatsit: the boxes its text nodes insert — `\pgfsys@hbox`'s box (`thebox`), the
468/// one way pgf typesets text in a picture (a `\node{…}`, `\pgftext`, `\pgfsys@hboxsynced`; pgfsys.code.tex) —, in their
469/// order. Everything else a pgf picture holds draws: an SVG path's data and style (`\lxSVG@drawpath@unclipped{M 5.53 0
470/// C …}{fill:none}`), colours, transforms, scopes. A frontmatter mark's printed label reads only the text
471/// (`queue_digest::label_ink`, the observer's `events::script_ink`): 2605.02769's TikZ ORCID logo in
472/// `\inst{\ref{leiden}\orcidicon{…}}` holds none (its path data, parted at its spaces, was the marks M, C, 0, L, Z on
473/// every name — review rev92b_r1 B1), a circled letter `\tikz…\node[circle,draw](c){a};` prints its "a" (review
474/// rev92b_r2 B2: a picture read as no text lost it, its line a continuation of the one before). LaTeX's `picture`
475/// environment is read as any material is (`\put(0,0){a}`'s text is an argument).
476pub fn picture_text(w: &Whatsit) -> Option<Vec<Digested>> {
477  fn text_nodes(d: &Digested, nodes: &mut Vec<Digested>) {
478    match d.data() {
479      DigestedData::List(l) => {
480        for child in l.borrow().boxes.iter() {
481          text_nodes(child, nodes);
482        }
483      },
484      DigestedData::Whatsit(w) => whatsit_text_nodes(&w.borrow(), nodes),
485      DigestedData::Alignment(a) => {
486        for row in a.borrow().rows() {
487          for cell in row.get_columns() {
488            if let Some(boxes) = cell.boxes.as_ref() {
489              text_nodes(boxes, nodes);
490            }
491          }
492        }
493      },
494      _ => {},
495    }
496  }
497  fn whatsit_text_nodes(w: &Whatsit, nodes: &mut Vec<Digested>) {
498    if w.get_definition().get_cs_name().trim_end() == "\\pgfsys@hbox" {
499      if let Some(Stored::Digested(text)) = w.get_property("thebox").as_deref() {
500        nodes.push(text.clone());
501      }
502      return;
503    }
504    for a in &printed_properties(w) {
505      text_nodes(a, nodes);
506    }
507    for arg in w.get_args().iter().flatten() {
508      text_nodes(arg, nodes);
509    }
510    if let Ok(Some(body)) = w.get_body() {
511      text_nodes(&body, nodes);
512    }
513  }
514  (w.get_definition().get_cs_name().trim_end() == "\\lxSVG@insertpicture").then(|| {
515    let mut nodes = Vec::new();
516    for arg in w.get_args().iter().flatten() {
517      text_nodes(arg, &mut nodes);
518    }
519    nodes
520  })
521}

537/// The combining characters a math accent whatsit prints on its base, from the glyph its constructor carries
538/// (`accent`, set for an `OVERACCENT` or `UNDERACCENT` operator, latexml_core `transfer_common_constructor_options`):
539/// `\bar`'s U+00AF over is U+0304, `\hat`'s `^` U+0302, `\vec`'s → U+20D7, a combining glyph itself (`\mathring`'s
540/// U+030A), `\underline`'s U+00AF under U+0332; `None` for anything else, an accent whose glyph has no combining form
541/// (`\overbrace`'s ⏞) among them. The frontmatter label readers (`queue_digest::label_ink`, the observer's
542/// `events::script_ink`) write it after the base: `$^{\bar{\dag}}$` is the label `†̄`, no `\dag`'s `†`, and
543/// `$^{\hat a}$` `â`, no `a` (review rev92b_r4 S2; `\={x}` digests to `x̄` already).
544pub fn accent_combining_form(w: &Whatsit) -> Option<String> {
545  let accent = match w.get_property("accent").as_deref() {
546    Some(accent @ Stored::String(_)) => accent.to_string(),
547    _ => return None,
548  };
549  let under = w
550    .get_property("operator_role")
551    .is_some_and(|role| role.to_string() == "UNDERACCENT");
552  let form = |c: char| -> Option<char> {
553    if is_combining_mark(c) {
554      return Some(c);
555    }
556    Some(match (c, under) {
557      ('^' | '\u{02C6}', false) => '\u{0302}',
558      ('~' | '\u{02DC}', false) => '\u{0303}',
559      ('\u{00AF}', false) => '\u{0304}',
560      ('\u{02D8}', false) => '\u{0306}',
561      ('\u{02D9}', false) => '\u{0307}',
562      ('\u{00A8}', false) => '\u{0308}',
563      ('\u{02C7}', false) => '\u{030C}',
564      ('\u{00B4}', false) => '\u{0301}',
565      ('`', false) => '\u{0300}',
566      ('\u{2192}', false) => '\u{20D7}',
567      ('\u{2190}', false) => '\u{20D6}',
568      ('\u{2194}', false) => '\u{20E1}',
569      ('\u{00AF}', true) => '\u{0332}',
570      ('~' | '\u{02DC}', true) => '\u{0330}',
571      ('\u{2192}', true) => '\u{20EF}',
572      ('\u{2190}', true) => '\u{20EE}',
573      _ => return None,
574    })
575  };
576  accent.chars().map(form).collect()
577}

593/// Whether a constructor's replacement `template` ([`Definition::replacement_template`]) prints nothing of its own,
594/// only its arguments and body ([`first_ink`]): every piece of it is one of these
595/// - an argument (`#1`) or the body (`#body`), absorbed;
596/// - white space;
597/// - an `ltx:text` or `ltx:emph` element (opened or closed) whose attributes only set the font or how the element
598///   closes ([`FONT_ONLY_ATTRIBUTES`]: `<ltx:text _noautoclose='1'>#1</ltx:text>`, `\emph`'s `_force_font`);
599/// - a conditional both of whose branches are such (`\emph`'s `?#isMath(…)(…)`);
600///
601/// and one of them absorbs an argument or the body. Anything else is something of its own: literal text (revsymb's `\openone`,
602/// `"1"`), another element (`<ltx:sup>#1</ltx:sup>`, a rule, a framed or coloured `ltx:text`), an empty element, a
603/// processing instruction, an attribute set on the node around it, a property or function value, a template that
604/// does not parse — and an empty template, which stands for what it does not show.
605fn prints_only_arguments(template: &str) -> bool {
606  use latexml_core::binding::def::replacement::{
607    AttrPair, ReplacementOp, Value, parse_replacement,
608  };
609  fn font_only(attrs: &[AttrPair]) -> bool {
610    attrs.iter().all(|pair| match pair {
611      AttrPair::KeyValue { key, .. } => FONT_ONLY_ATTRIBUTES.contains(&key.as_str()),
612      AttrPair::Conditional { then_attrs, else_attrs, .. } => {
613        font_only(then_attrs) && font_only(else_attrs)
614      },
615    })
616  }
617  fn transparent(ops: &[ReplacementOp]) -> bool {
618    ops.iter().all(|op| match op {
619      ReplacementOp::Text { text } => text.chars().all(char::is_whitespace),
620      ReplacementOp::AbsorbValue { value: Value::Arg(_) } => true,
621      ReplacementOp::AbsorbValue { value: Value::Prop(name) } => name == "body",
622      ReplacementOp::OpenElement {
623        qname,
624        attrs,
625        self_closing: false,
626        ..
627      } => matches!(qname.as_str(), "ltx:text" | "ltx:emph") && font_only(attrs),
628      ReplacementOp::CloseElement { qname } => matches!(qname.as_str(), "ltx:text" | "ltx:emph"),
629      ReplacementOp::Conditional { then_ops, else_ops, .. } => {
630        transparent(then_ops) && transparent(else_ops)
631      },
632      _ => false,
633    })
634  }
635  // (and absorbs one: a template printing nothing at all stands for what it does not show — the frontmatter
636  // observer's rehearsal's `\lx@fm@obs@url`, `\lx@fm@obs@fnmark`, `""` in place of a URL or a footnote mark —, a doubt)
637  fn absorbs(ops: &[ReplacementOp]) -> bool {
638    ops.iter().any(|op| match op {
639      ReplacementOp::AbsorbValue { .. } => true,
640      ReplacementOp::Conditional { then_ops, else_ops, .. } => {
641        absorbs(then_ops) || absorbs(else_ops)
642      },
643      _ => false,
644    })
645  }
646  parse_replacement(template).is_ok_and(|ops| transparent(&ops) && absorbs(&ops))
647}

649/// Whether a constructor's template holds text that is not blank, in any branch of a conditional ([`typesets_content`]).
650/// A template that does not parse is read as printing none: an image, a reference and an alignment have their own tests.
651fn template_prints_text(template: &str) -> bool {
652  use latexml_core::binding::def::replacement::{ReplacementOp, parse_replacement};
653  fn prints(ops: &[ReplacementOp]) -> bool {
654    ops.iter().any(|op| match op {
655      ReplacementOp::Text { text } => text.chars().any(|c| !c.is_whitespace()),
656      ReplacementOp::Conditional { then_ops, else_ops, .. } => prints(then_ops) || prints(else_ops),
657      _ => false,
658    })
659  }
660  parse_replacement(template).is_ok_and(|ops| prints(&ops))
661}

663/// The text a constructor's template prints when it is constant (no value absorbed, no branch), its elements' too:
664/// `\LaTeX`'s `L<ltx:text …>a</ltx:text>T…X` is "LaTeX". None for white space alone or a template that does not parse.
665fn template_text(template: &str) -> Option<String> {
666  use latexml_core::binding::def::replacement::{ReplacementOp, parse_replacement};
667  let ops = parse_replacement(template).ok()?;
668  let mut text = String::new();
669  for op in &ops {
670    match op {
671      ReplacementOp::Text { text: piece } => text.push_str(piece),
672      ReplacementOp::OpenElement { .. } | ReplacementOp::CloseElement { .. } => {},
673      _ => return None,
674    }
675  }
676  Some(text).filter(|text| !text.trim().is_empty())
677}

750fn ink_text_with(d: &Digested, pictures: bool, out: &mut String) {
751  match d.data() {
752    DigestedData::TBox(b) => {
753      if let Ok(s) = b.borrow().get_string() {
754        out.push_str(&s);
755      }
756    },
757    DigestedData::List(l) => {
758      for child in l.borrow().boxes.iter() {
759        ink_text_with(child, pictures, out);
760      }
761    },
762    DigestedData::Whatsit(w) if pictures && let Some(nodes) = picture_text(&w.borrow()) => {

766        out.push(' ');
767      }

769    DigestedData::Whatsit(w) => {
770      let w = w.borrow();

772      for a in &printed_properties(&w) {
773        ink_text_with(a, pictures, out);
774      }
775      for (arg, shown) in w.get_args().iter().zip(shown_arguments(&w)) {
776        if shown && let Some(arg) = arg {
777          ink_text_with(arg, pictures, out);
778        }
779      }
780      if let Ok(Some(body)) = w.get_body() {
781        ink_text_with(&body, pictures, out);
782      }

787      }
788    },
789    DigestedData::Alignment(a) => {

794          }
795        }
796      }
797    },
798    _ => {},
799  }

832/// The text digested material typesets, as [`ink_text`] reads it, with where its words end: a space for a whatsit that
833/// prints nothing (a line break, `\par`, a skip), and [`INK_SEAM`] at the edges of a whatsit that prints (a font
834/// switch, a box, an accent) and of each alignment cell, where the text runs on in print but a line may end — an
835/// accent built as an alignment runs on (combelow's `\cb{s}`, `Babe\cb{s}-Bolyai` "Babes", 2007.07526), an author
836/// table's cells end their words. The title page's printed words (`base_utilities::frontmatter::title_page_check`).
837pub fn ink_word_text(d: &Digested, out: &mut String) {
838  // (a mark the title-page reading set in its run, where it stands, by its definition: [`INK_AUTHOR`])
839  if let DigestedData::Whatsit(w) = d.data()
840    && let Some(mark) = crate::latex_constructs::title_page_mark(&w.borrow().get_definition())
841  {
842    out.push(mark);
843    return;
844  }
845  // (spacing — a phantom's invisible ink, a kern, `\quad` — and a script print no word: a script is a mark)
846  if let DigestedData::Whatsit(w) = d.data()
847    && (is_spacing_item(d)
848      || WORDLESS.contains(&w.borrow().get_definition().get_cs_name().as_ref()))
849  {
850    out.push(' ');
851    return;
852  }
853  match d.data() {
854    DigestedData::TBox(b) => {
855      if let Ok(s) = b.borrow().get_string() {
856        out.push_str(&s);
857      }
858    },
859    DigestedData::List(l) => {
860      for child in l.borrow().boxes.iter() {
861        ink_word_text(child, out);
862      }
863    },
864    DigestedData::Whatsit(w) => {
865      let mut inner = String::new();
866      {
867        let w = w.borrow();
868        for a in &printed_properties(&w) {
869          ink_word_text(a, &mut inner);
870        }
871        // (each argument a field of its own: `\lx@note`'s type, "footnote", and its text are no one word; a quantity —
872        // `\scalebox{0.85}`'s factor, a skip — and an environment's required argument — a column specification,
873        // `{CJK}`'s `{UTF8}{gbsn}` — no text)
874        let mut hidden = settings_arguments(&w);
875        // (a note's type, the second argument of each of our `\lx@note…` constructors, `footnote`, prints nothing)
876        if w.get_definition().get_cs_name().starts_with("\\lx@note") {
877          hidden.resize(hidden.len().max(2), false);
878          hidden[1] = true;
879        }
880        for (i, (arg, shown)) in w.get_args().iter().zip(shown_arguments(&w)).enumerate() {
881          if shown
882            && !hidden.get(i).copied().unwrap_or(false)
883            && let Some(arg) = arg
884          {
885            inner.push(INK_SEAM);
886            ink_word_text(arg, &mut inner);
887            inner.push(INK_SEAM);
888          }
889        }
890        if let Ok(Some(body)) = w.get_body() {
891          ink_word_text(&body, &mut inner);
892        }
893        if inner.chars().all(|c| c.is_whitespace() || c == INK_SEAM)
894          && let Some(glyph) = printed_own_text(&w)
895        {
896          inner.push_str(&glyph);
897        }
898      }
899      if inner.chars().all(|c| c.is_whitespace() || c == INK_SEAM) {
900        out.push(' ');
901      } else {
902        out.push(INK_SEAM);
903        out.push_str(&inner);
904        out.push(INK_SEAM);
905      }
906    },
907    DigestedData::Alignment(a) => {
908      for row in a.borrow().rows() {
909        for cell in row.get_columns() {
910          if let Some(boxes) = cell.boxes.as_ref() {
911            out.push(INK_SEAM);
912            ink_word_text(boxes, out);
913            out.push(INK_SEAM);
914          }
915        }
916      }
917    },
918    _ => {},
919  }
920}

923/// arguments it shows (by position), for [`shown_arguments`]: llncs.cls's `\url` (llncs_cls.rs), hyperref's

925/// compiled, so what it puts in its element cannot be read off the whatsit; a verbatim argument is otherwise a key
926/// (`\ref`, `\pageref`, `\eqref`, `\cite`, `\nameref`, `\autoref`, `\label`), a link target or a file name, which the
927/// page does not show. url.sty's and hyperref's `\url`/`\path` and the bindings' `\email`s are macros: what they print
928/// reaches a constructor of its own, whose text argument is shown (`\lx@url@url@nolink`'s, `\lx@hyper@url@`'s `#5`), as
929/// aas's `\doi{…}`'s is. A constructor printing one of two verbatim arguments names it: `("\\href", &[false, true])` would show the
930/// text, not the URL (jhep's two-verbatim `\href`, a binding class's, which never meets the raw-class stores).
931const VERBATIM_PRINTERS: &[(&str, &[bool])] = &[
932  ("\\url", &[true]),
933  ("\\nolinkurl", &[true]),
934  ("\\@@email", &[true]),
935  ("\\lx@doi", &[true]),
936  ("\\harvardurl", &[true]),
937];

939/// What a whatsit prints of its own, which the text walkers print where it shows nothing else: a math character's
940/// `glyph` (`\mathchar`, `\delimiter`, `\mathaccent`: tex_math.rs; `\Udelimiter`), as Perl's `\mathchar` is a Box of it
941/// (TeX_Math.pool.ltxml:576-580; flexisym's digits, 2312.00656's `$^{2,3}$`), else a constant template's text
942/// ([`template_text`]: `\LaTeX`, `\TeX` — amsldoc's `\author{…, \LaTeX\ Project}` printed "Project").
943pub(crate) fn printed_own_text(w: &Whatsit) -> Option<String> {
944  printed_glyph(w).or_else(|| {
945    let definition = w.get_definition();
946    // (the kernel's `\and` is latex.ltx's column break, no text: its " and " is the XML's, Perl's
947    // `DefConstructor('\and', ' and ')` (LaTeX.pool), never the page's — fx25's title page print)
948    if is_kernel_separator(definition.as_ref()) {
949      return None;
950    }
951    definition.replacement_template().and_then(template_text)
952  })
953}

979/// Which arguments of the whatsit `w` it shows as text, for [`ink_text`], by their declared parameter types: none of an
980/// image (a whatsit carrying a `graphic` or image `candidates`: its arguments are the file and its options); else a
981/// verbatim argument (`Semiverbatim`, `Verbatim`, `HyperVerbatim`: a file name, a URL, a label) only when the whatsit
982/// is one that prints it ([`VERBATIM_PRINTERS`]: `\url{…}`, `\nolinkurl{…}`, the arguments its entry names) — never a
983/// reference's or a citation's key (`\ref{\@x}` prints the label's number, review probe `ctx3`), and
984/// `\href{<url>}{<text>}` and `\hypertarget{<name>}{<text>}` print the text —, and never keyvals (options) nor a
985/// position keyword (`Position`, `OptionalPosition`: the `[t]` of `\begin{tabular}[t]`, base_parameter_types.rs). Every
986/// argument when its parameters cannot be matched to them. Also what `\textcircled` circles (sect13.rs
987/// `typeset_text`).
988pub(crate) fn shown_arguments(w: &Whatsit) -> Vec<bool> {
989  let args = w.get_args();
990  // (graphics.sty's `\@includegraphics` records its `graphic`, graphicx's `\@includegraphicx` its `path`; both the
991  // image's `candidates`)
992  if w.get_property("graphic").is_some() || w.get_property("candidates").is_some() {
993    return vec![false; args.len()];
994  }
995  let Some(kinds) = argument_kinds(w) else {
996    return vec![true; args.len()];
997  };
998  let definition = w.get_definition();
999  let cs = definition.get_cs();
1000  let shown: &[bool] = VERBATIM_PRINTERS
1001    .iter()
1002    .find(|(name, _)| *cs == T_CS!(*name))
1003    .map_or(&[], |(_, shown)| shown);
1004  kinds
1005    .iter()
1006    .enumerate()
1007    .map(|(i, kind)| *kind == 2 || (*kind == 1 && shown.get(i).copied().unwrap_or(false)))
1008    .collect()
1009}

1011/// Which arguments of the whatsit `w` print no text of their own by their parameter's type, for [`ink_word_text`]: a
1012/// quantity (a number, a float, a dimension, a glue) and, when `w` is an environment's (it has a body), a required
1013/// argument, which sets the environment up (`{tabular}`'s columns, `{minipage}`'s width, `{CJK}`'s encoding and
1014/// font) where an optional one may name it (`{theorem}[Name]`); none when its parameters cannot be matched to them.
1015fn settings_arguments(w: &Whatsit) -> Vec<bool> {
1016  let types: Vec<SymStr> = w
1017    .get_definition()
1018    .get_parameters()
1019    .map(|params| {
1020      params
1021        .get_parameters()
1022        .into_iter()
1023        .filter(|p| !p.novalue)
1024        .map(|p| p.name)
1025        .collect()
1026    })
1027    .unwrap_or_default();
1028  if types.len() != w.get_args().len() {
1029    return Vec::new();
1030  }
1031  let environment = matches!(w.get_body(), Ok(Some(_)));
1032  types
1033    .iter()
1034    .map(|t| {
1035      with(*t, |name| {
1036        ["Number", "Float", "Dimension", "Glue", "Skip", "Integer"]
1037          .iter()
1038          .any(|kind| name.contains(kind))
1039          || (environment && !name.starts_with("Optional"))
1040      })
1041    })
1042    .collect()
1043}

1045/// The kind of each argument of the whatsit `w` by its parameter's type — 0: keyvals (options) or a position keyword
1046/// (`Position`, `OptionalPosition`), 1: verbatim (`Semiverbatim`, `Verbatim`, `HyperVerbatim`), 2: any other —; none
1047/// when its parameters cannot be matched to its arguments ([`shown_arguments`], [`first_ink`]).
1048fn argument_kinds(w: &Whatsit) -> Option<Vec<u8>> {
1049  let types: Vec<SymStr> = w
1050    .get_definition()
1051    .get_parameters()
1052    .map(|params| {
1053      params
1054        .get_parameters()
1055        .into_iter()
1056        .filter(|p| !p.novalue)
1057        .map(|p| p.name)
1058        .collect()
1059    })
1060    .unwrap_or_default();
1061  (types.len() == w.get_args().len()).then(|| {
1062    types
1063      .iter()
1064      .map(|t| {
1065        with(*t, |name| {
1066          if name.contains("KeyVals") || name.ends_with("Position") {
1067            0
1068          } else if name.contains("erbatim") {
1069            1
1070          } else {
1071            2
1072          }
1073        })
1074      })
1075      .collect()
1076  })
1077}
