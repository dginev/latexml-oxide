1//! `insert_block_as` (Perl `insertBlock`, TeX_Box.pool.ltxml), one function.
2
3use super::*;
4
5/// This attempts to be a generalize vbox construction;
6///
7/// The idea is to receeive block-like material, possibly wrapped in appropriate
8/// container which gets attributes.
9///
10/// The contents are constructed in an ltx:_CaptureBlock_ element,
11/// designed to accept all reasonable block material from several levels,
12/// and then determine which container element is most apprpriate for both the conent & context
13/// from block, logical-block or sectional-block, or the inline- variants.
14///
15/// `boxed`: the block is a TeX box (a `\vbox`, `\vtop`, `\parbox`, a minipage, a `p` column, a framed box), its
16/// capture marked for a frontmatter field's print (core `Document::mark_field_box`, OXIDIZED_DESIGN #715); not an
17/// aligning list environment's (`{center}`, `{flushright}`: [`aligning_environment`]).
18pub(super) fn insert_block_as(
19  document: &mut Document,
20  contents: &Digested,
21  block_attr: HashMap<String, String>,
22  in_paragraph: bool,
23  boxed: bool,
24) -> Result<Vec<Node>> {
25  // Perl #2829 (TeX_Box.pool insertBlock L449-456): callers may (sloppily)
26  // pass ALL Whatsit properties, but only SOME are intended as attributes.
27  // Filter out non-attributes, based on what ltx:figure accepts — a maximal
28  // set of the attributes for candidate containers.
29  let block_attr: HashMap<String, String> = block_attr
30    .into_iter()
31    .filter(|(k, _)| document::can_have_attribute("ltx:figure", k))
32    .collect();
33  // Create something like:
34  // "<ltx:inline-block vattach='$vattach' height='#height'>#2</ltx:inline-block>"
35  let context_opt = document.get_element(); // Where we originally start inserting.
36  if context_opt.is_none() {
37    // edge case: if we start the doc with a block, the context is empty
38    document.absorb(contents, None)?;
39    return Ok(Vec::new());
40  }
41  let mut context = context_opt.unwrap();
42  let mut context_tag = document::get_node_qname(&context);
43  // svg is slightly tricky
44  // An `svg:foreignObject` context is LaTeXML content again (it holds ltx
45  // elements), not SVG proper: the block keeps its attributes and its own
46  // wrapper there. With the foreignObject model widened to Flow
47  // (OXIDIZED_DESIGN #186) the SVG rule "attributes are ignorable" would
48  // unwrap a minipage straight into the foreignObject and hoist its
49  // `width="31.37em"` over the foreignObject's px width (fixture
50  // tikz/various_colors).
51  let (is_svg, is_xmath, is_xmtext) = with(context_tag, |tag| {
52    (
53      tag.starts_with("svg:") && tag != "svg:foreignObject",
54      tag.starts_with("ltx:XM"),
55      tag == "ltx:XMText",
56    )
57  });
58  // The width stays the Dimension (Perl's insertBlock), in SVG context too: a block a measured foreignObject holds
59  // is rewritten to ems of that foreignObject's anchor by its sizer (tex_box.rs), the one basis the browser reads.
60  let ignorable_attr = is_svg || block_attr.is_empty(); // if we do not REQUIRE the attributes
61  if is_xmath && !is_xmtext {
62    // but math always needs this
63    context = document.open_element("ltx:XMText", None, None)?;
64    context_tag = document::get_node_qname(&context);
65  }
66  // An inline-only container that holds `inline-block` (Misc.class) but neither
67  // `#PCDATA` nor `block` (Block.class): the picture family `ltx:g`/`ltx:picture`/
68  // `ltx:clippath`, plus `ltx:creator`/`ltx:equation`/`ltx:inline-item` (the full set
69  // this clause flips — verified against LaTeXML.model). `<block>` is schema-invalid in
70  // every one, and `<inline-block>` is valid there AND holds full Block.class content,
71  // so a `\put`-positioned `\parbox`/`\makebox` digested there — an LR-box, never a
72  // block — is renamed to a schema-valid `<inline-block>` IN PLACE: no move, so the
73  // `\put` position in the enclosing `<g transform="translate(x,y)">` is preserved
74  // (fidelity-safe). Emitting `<block>` here is invalid (e.g. `g_model`,
75  // LaTeXML-picture.rng) — a shared Perl bug (`TeX_Box.pool.ltxml:493`), surpassed on
76  // the schema-validity axis like #240. A `<para>`/`<inline-block>`/`<float>`/`<note>`
77  // holds BOTH block and inline-block, so this clause is false there and the #240
78  // block-climb (`context_tag == "ltx:para"`) stays untouched.
79  let mut is_inline = is_svg
80    || document::can_contain(&context, "#PCDATA")
81    || (document::can_contain_qsym(context_tag, pin_static("ltx:inline-block"))
82      && !document::can_contain_qsym(context_tag, pin_static("ltx:block")));
83  let container_attr = block_attr.clone();
84  let mut container = document.open_element("ltx:_CaptureBlock_", Some(container_attr), None)?;
85  // the box's own element: the mark rides the capture's rename, an attribute-taking lone child, a paragraph-move; an
86  // unwrapped capture leaves its material in the context, unmarked
87  if boxed {
88    document.mark_field_box(&mut container);
89  }
90  document.absorb(contents, None)?;
91
92  // The row separators the panel grid rule records in a float's `{center}`-like capture (`note_panel_separator`) are
93  // marks for the float's `arrange_panels`, not the box's content: they choose no container, are hoisted nowhere and
94  // stay where they are — reaching the float where the capture unwraps into it, and leaving with a capture that stays
95  // a container (`drop_panel_separators`), in which they separate none of the float's panels. As content, a leading
96  // `\hspace*{5pt}`'s (or `\hskip`'s) `ltx:break` chose the inline candidate that holds it: the `\parbox` panels
97  // after it were hoisted out as `logical-block`s and the caption kept in the capture's `ltx:block` (1710.06503 Fig 1,
98  // the ws-procs961x669 binding net; repro captions-floats/an_hspace_star_opens_the_row_paragraph).
99  let mut nodes = content_nodes(&container);
100  nodes.retain(|node| !crate::latex_constructs::is_panel_separator(node));
101  let mut node_tags = nodes
102    .iter()
103    .map(document::get_node_qname)
104    .collect::<Vec<_>>();
105  document.close_to_node(&container, true)?;
106  document.close_node(&container)?;
107  document.close_to_node(&context, true)?;
108
109  // A sectioning unit inside a block-mode box (`\\section` in a `minipage[t]`, a
110  // `framed`/`tcolorbox`) is a REAL section in LaTeX: `\\@startsection` ran, the
111  // counter advanced, and the text after the box belongs to the new unit. Perl's
112  // `insertBlock` (TeX_Box.pool.ltxml:512) instead wraps the box as
113  // `<sectional-block>` and leaves it inside the enclosing section, where the
114  // schema admits no `sectional-block` (only `document.body.class` does,
115  // LaTeXML-structure.rnc:99) — SHARED, and invalid. User ruling (2026-09-20): the
116  // box is presentation, the sectioning is semantics — never bend the schema to a
117  // layout container. So when the captured content holds a sectioning unit and the
118  // insertion context has an auto-closeable path to an ancestor that admits it, every
119  // built unit is MOVED to where a live `\\section` would open (closing the enclosing
120  // unit for a same-or-higher level, nesting a deeper one) and becomes the insertion
121  // node, so trailing box content and post-box document text nest inside it; the
122  // box's own annotation (`class="ltx_minipage"` and the frame attributes
123  // `Sectional.attributes` admit — `width`/`vattach` are not, as `sectional-block`
124  // already dropped them) rides the FIRST floated unit. A box inside a `<quote>`, a
125  // list `<item>` or a `<figure>` has no such path (none auto-closes to a section
126  // holder) and keeps today's Perl-parity shape. Witnesses: aguplus, fancyvrb-doc,
127  // latex-via-exemplos, phonenumbers-{de,en}, recorder-fingering, tasks-manual,
128  // unamth-template/tesis (8 s106 docs). OXIDIZED_DESIGN #250.
129  if !is_inline
130    && let Some(first_unit) = nodes.iter().find(|n| is_sectioning_unit(n))
131    && sectioning_floatable(&context, document::get_node_qname(first_unit))
132  {
133    document.set_node(&context);
134    // The floated units are what the caller gets back: `aligning_environment`
135    // ({center}, {flushleft}, {flushright}) stamps its alignment on them.
136    let mut floated: Vec<Node> = Vec::new();
137    for child in nodes.iter().cloned() {
138      if child.get_type() != Some(NodeType::ElementNode) {
139        continue;
140      }
141      let q = document::get_node_qname(&child);
142      let mut point = document.find_insertion_point_qsym(q, None)?;
143      let mut c = child;
144      c.unlink();
145      point.add_child(&mut c)?;
146      if is_sectioning_unit(&c) {
147        floated.push(c.clone());
148        document.set_node(&c);
149      }
150    }
151    if let Some(mut unit) = floated.first().cloned() {
152      let unit_q = document::get_node_qname(&unit);
153      for (k, v) in block_attr.iter() {
154        if k == "class" {
155          document.add_class(&mut unit, v)?;
156        } else if document::sym_can_have_attribute(unit_q, pin(k)) {
157          document.set_attribute(&mut unit, k, v)?;
158        }
159      }
160    }
161    // Everything of substance has been moved out; the capture is spent.
162    document.remove_node(container);
163    return Ok(floated);
164  }
165
166  // A box that is clearly a figure becomes its float, holding its image (`absorb_figure_material`,
167  // 57g); whether it then stands as that float or sits in a paragraph is settled when the
168  // paragraph closes (`settle_float_paragraph`, latex_constructs).
169  if let Some(float) = absorb_figure_material(document, &nodes)? {
170    node_tags = vec![document::get_node_qname(&float)];
171    nodes = vec![float];
172  }
173  // Paragraph material (`insert_block_in_paragraph`): where text would open an `ltx:p`, the box
174  // goes into that paragraph, which stays open for the text after it. Decided after the capture,
175  // so the sectioning float-out above (#250) still sees the original context. Not in a float,
176  // whose content is panels (`in_panel_float`: arXiv 2605.27134's `\parbox` at a figure
177  // minipage's top; captioned `\parbox` panels in a figure's `{center}` stay sub-figures); not
178  // inside a restricted box (`\rotatebox{90}{\parbox…}`, arXiv 2605.20645), where the flag is
179  // unset; not for an empty box, which leaves no material; not for content no inline box holds
180  // (a bibliography, an index, a caption), which the hoist below places after the box as before.
181  // A box whose whole content is one float goes into the paragraph too; a paragraph of nothing but
182  // such boxes becomes those floats when it closes (user ruling 2026-09-27: `<figure
183  // class="ltx_minipage">`, Perl's fold, for a lone one; a row of panels for several).
184  if in_paragraph
185    && !is_inline
186    && !nodes.is_empty()
187    && text_opens_paragraph(context_tag)
188    && !in_panel_float(&context)
189    && [
190      "ltx:inline-block",
191      "ltx:inline-logical-block",
192      "ltx:inline-sectional-block",
193    ]
194    .iter()
195    .any(|inline| {
196      let inline = pin_static(inline);
197      node_tags
198        .iter()
199        .all(|tag| document::sym_can_contain_somehow(inline, *tag).is_some())
200    })
201  {
202    let mut paragraph = document.open_element("ltx:p", None, None)?;
203    container.unlink();
204    paragraph.add_child(&mut container)?;
205    context = paragraph;
206    context_tag = document::get_node_qname(&context);
207    is_inline = true;
208  }
209
210  // Perl `insertBlock` (TeX_Box.pool.ltxml:406-472): the first candidate, in
211  // preference order, that can hold ALL of the box's content becomes the
212  // container (:457-458 give the candidate sets; :471 the hard `ltx:block`
213  // fallback that then reports every child the model rejects). One general
214  // rule extends it (Rust-only): when no candidate holds everything, the
215  // first candidate that holds the box's LEADING content is the box, and the
216  // content it cannot hold is placed after the box, in the nearest flow
217  // ancestor that can hold it — the outcome an autoclosing frame produces
218  // for the same content written live (`\printbibliography` in mdframed:
219  // `perfect_kernel_gemini::mdframed_block_bibliography_juradiss`; a
220  // `\caption` in a rotated `\parbox`: heria-proposal, rubik —
221  // `perfect_kernel_batch56::caption_in_inline_parbox_floats_to_figure`).
222  // The climb stays inside text containers (an ancestor holding paragraphs
223  // or text): a drawing or a math box is a different medium, and content
224  // hoisted past it would strand the rest of the box (Gemini J3,
225  // biblatex-ext) — there the model's own verdict stands, as in Perl.
226  // `LXML_TRACE_INSERT_BLOCK=1` prints each decision.
227  let all_candidates = [
228    "ltx:inline-block",
229    "ltx:inline-logical-block",
230    "ltx:inline-sectional-block",
231    "ltx:block",
232    "ltx:logical-block",
233    "ltx:sectional-block",
234    "ltx:figure",
235  ]
236  .map(pin_static);
237  let candidate_set: &[SymStr] = if is_inline {
238    &all_candidates[0..3]
239  } else {
240    &all_candidates
241  };
242  let holds = |c: SymStr, t: SymStr| document::sym_can_contain_somehow(c, t).is_some();
243  // A box no candidate holds whole only because its paragraphs are wrapped in `ltx:para` — a caption beside a
244  // `\noindent` paragraph (adjustbox's collectbox; `\caption{First} \noindent x` in a minipage, 2312.04535, 2209.02973,
245  // 2404.07198, 2404.18532, 2406.00259, 2502.04073) — holds them unwrapped, the `ltx:p`s a candidate takes beside the
246  // caption (the #244 demotion). Perl warns and falls back to an `ltx:block` the caption and the para are invalid in.
247  let para = pin_static("ltx:para");
248  if !candidate_set
249    .iter()
250    .any(|c| node_tags.iter().all(|t| holds(*c, *t)))
251    && node_tags.contains(&para)
252  {
253    let unwrapped: Vec<SymStr> = nodes
254      .iter()
255      .flat_map(|n| {
256        if document::get_node_qname(n) == para {
257          content_nodes(n)
258            .iter()
259            .map(document::get_node_qname)
260            .collect::<Vec<_>>()
261        } else {
262          vec![document::get_node_qname(n)]
263        }
264      })
265      .collect();
266    if candidate_set
267      .iter()
268      .any(|c| unwrapped.iter().all(|t| holds(*c, *t)))
269    {
270      for wrapper in nodes.iter().filter(|n| document::get_node_qname(n) == para) {
271        // (what the paragraph carried goes to the elements it held, its id going with it: a text paragraph (`ltx:p`)
272        // takes it as the para had it; a box (an adjustbox's `ltx:inline-block`, a picture) is centred as a box
273        // (`ltx_centering`: an `align` on it centres only its own text), and the paragraph's indentation is not its)
274        let attributes = wrapper.get_attributes();
275        for mut child in wrapper.get_child_elements() {
276          let tag = document::get_node_qname(&child);
277          let text_block = tag == pin_static("ltx:p");
278          for (key, value) in &attributes {
279            if key == "id" || key == "xml:id" || key.starts_with('_') {
280              continue;
281            }
282            if key == "class" {
283              for class in value.split_whitespace() {
284                if text_block || (class != "ltx_indent" && class != "ltx_noindent") {
285                  document.add_class(&mut child, class)?;
286                }
287              }
288            } else if key == "align" && !text_block {
289              // (a box's other alignments are not carried: no class places a box right; a residual)
290              if value == "center" {
291                document.add_class(&mut child, "ltx_centering")?;
292              }
293            } else if !child.has_attribute(key) && document::sym_can_have_attribute(tag, pin(key)) {
294              document.set_attribute(&mut child, key, value)?;
295            }
296          }
297        }
298        document.unwrap_nodes(wrapper.clone())?;
299      }
300      nodes = content_nodes(&container);
301      node_tags = nodes.iter().map(document::get_node_qname).collect();
302    }
303  }
304  let holds_all = |c: SymStr| node_tags.iter().all(|t| holds(c, *t));
305  // Perl's order (TeX_Box.pool.ltxml:503-509): the first candidate that holds all the content and that the context
306  // holds, else the first that holds all the content.
307  let allowed = |c: SymStr| document::can_contain_qsym(context_tag, c);
308  let fits_all = candidate_set
309    .iter()
310    .copied()
311    .find(|c| holds_all(*c) && allowed(*c))
312    .or_else(|| candidate_set.iter().copied().find(|c| holds_all(*c)));
313  let container_tag = match fits_all {
314    // One the context does not hold (a figure for a `\parbox` with a closing `\caption` in a `\rotatebox`'s
315    // `ltx:inline-block`, LaTeXML-misc.rnc:28: heria.cls:737-745 `SidewaysFigure`, schema-invalid) gives way to one it
316    // holds for the box's leading content, when what that one cannot hold is the caption closing the box and it has
317    // a home after it ([`tail_home`]: the caption goes to the figure the box is in, read in the same order; 1810.08269).
318    // A box that opens with its caption (`\rotatebox{90}{\begin{minipage}…\caption…\begin{tabular}…}`) keeps Perl's
319    // choice: hoisting the caption would put it after the content it precedes, and an empty box would be left in the
320    // transform; so does one whose closing part holds more than a caption, one beside other content in the
321    // transform or a frame around it (`\scalebox{…}{minipage \hfill minipage}`, 2605.06364: the hoist moves the
322    // insertion point, so the next panel would leave the transform), and one whose float holds another caption
323    // ([`caption_hoist_is_safe`]: two captions would become one float).
324    Some(c) if !allowed(c) => candidate_set
325      .iter()
326      .copied()
327      .filter(|alt| allowed(*alt))
328      .find(|alt| {
329        let kept = node_tags.iter().take_while(|t| holds(*alt, **t)).count();
330        let tail = &node_tags[kept..];
331        kept > 0
332          && !tail.is_empty()
333          && tail
334            .iter()
335            .all(|t| !holds(*alt, *t) && with(*t, |q| q == "ltx:caption" || q == "ltx:toccaption"))
336          && tail_home(
337            &context,
338            &container,
339            tail,
340            nodes[..kept]
341              .iter()
342              .all(|n| n.get_content().trim().is_empty()),
343          )
344          .is_some_and(|(_, home)| caption_hoist_is_safe(document, &context, &home))
345      })
346      .or(Some(c)),
347    Some(c) => Some(c),
348    None => node_tags
349      .first()
350      .and_then(|first| candidate_set.iter().copied().find(|c| holds(*c, *first))),
351  };
352  let hoisted: Vec<bool> = node_tags
353    .iter()
354    .map(|t| container_tag.is_none_or(|c| !holds(c, *t)))
355    .collect();
356  if std::env::var_os("LXML_TRACE_INSERT_BLOCK").is_some() {
357    eprintln!(
358      "insert_block: context={} inline={} tags={:?} container={:?} hoisted={:?}",
359      with(context_tag, |s| s.to_string()),
360      is_inline,
361      node_tags
362        .iter()
363        .map(|t| with(*t, |s| s.to_string()))
364        .collect::<Vec<_>>(),
365      container_tag.map(|c| with(c, |s| s.to_string())),
366      hoisted
367    );
368  }
369  if hoisted.iter().any(|h| *h) {
370    // The box's content in source order, for a figure panel that takes all of it (below).
371    let in_order = nodes.clone();
372    let order_tags = node_tags.clone();
373    let mut tail = Vec::new();
374    let mut kept = Vec::new();
375    let mut kept_tags = Vec::new();
376    for (n, (node, tag)) in nodes.drain(..).zip(node_tags.iter().copied()).enumerate() {
377      if hoisted[n] {
378        tail.push(node);
379      } else {
380        kept.push(node);
381        kept_tags.push(tag);
382      }
383    }
384    let tail_tags: Vec<SymStr> = tail.iter().map(document::get_node_qname).collect();
385    // A text container: holds paragraphs, or text itself (`ltx:p`, an
386    // inline block). A drawing (`ltx:picture`, `svg:g`) holds neither —
387    // backmatter trapped in a drawing (xebaposter's References `\headerbox`
388    // is a pgf node → `svg:foreignObject`) still floats to the enclosing flow
389    // exactly as from a plain minipage, but never at the cost of stranding
390    // TEXT: the climb crosses a drawing only when no kept node carries
391    // non-whitespace text (a graphic in the box is drawing content and stays
392    // where it is; Perl's `adjustBackmatterElement` cannot cross at all —
393    // `ltx:bibliography` isn't allowed in `ltx:block`, SHARED; batch 56bq).
394    let kept_empty = kept.iter().all(|n| n.get_content().trim().is_empty());
395    let home = tail_home(&context, &container, &tail_tags, kept_empty);
396    if let Some((anchor, ancestor)) = home {
397      // `anchor` is an element and a box's content nodes are already
398      // coalesced, so no two adjacent text nodes reach `add_next_sibling`
399      // (libxml2 merges those and frees the second — see `replace_node`).
400      let mut prev = anchor;
401      for mut n in tail {
402        n.unlink();
403        prev.add_next_sibling(&mut n)?;
404        prev = n;
405      }
406      document.set_node(&ancestor);
407      nodes = kept;
408      node_tags = kept_tags;
409      if nodes.is_empty() {
410        document.remove_node(container);
411        return Ok(Vec::new());
412      }
413    } else if is_inline
414      && order_tags
415        .iter()
416        .any(|t| with(*t, |s| s == "ltx:caption" || s == "ltx:toccaption"))
417      && order_tags
418        .iter()
419        .all(|t| holds(pin_static("ltx:figure"), *t))
420    {
421      // A box in running text holding a caption, all of whose content a figure may hold — a minipage in a table
422      // cell captioned as a figure, caption first or last: it becomes a figure panel in the box, its content in
423      // source order, as the same minipage between paragraphs
424      // becomes one (`figure class="ltx_figure_panel ltx_minipage"`), the box an inline logical block, whose model
425      // holds a figure where a cell holds none (captions_in_minipages_in_a_tabular; Perl reports the captions in
426      // an `ltx:block`).
427      drop((kept, tail));
428      if let Some(mut panel) = document.wrap_nodes("ltx:figure", in_order.clone())? {
429        document.add_class(&mut panel, "ltx_figure_panel")?;
430        nodes = vec![panel];
431        node_tags = vec![pin_static("ltx:figure")];
432      } else {
433        // The model's verdict, as below (a box's own nodes always wrap).
434        nodes = in_order;
435        node_tags = order_tags;
436      }
437    } else {
438      // No flow ancestor holds it: the content stays in the box and the
439      // model reports it (Perl's outcome).
440      nodes = kept;
441      nodes.extend(tail);
442      node_tags = kept_tags;
443      node_tags.extend(tail_tags);
444    }
445  }
446  let nnodes = nodes.len();
447
448  // Perl: Hack: apparently TeX doesn't shift (vattach) a single node in a vbox/vtop/...
449  #[allow(clippy::redundant_locals)]
450  let mut block_attr = block_attr;
451  let mut ignorable_attr = ignorable_attr;
452  if nnodes == 1 && block_attr.contains_key("vattach") && is_v_attached(&nodes[0]) {
453    container.remove_attribute("vattach")?;
454    block_attr.remove("vattach");
455    ignorable_attr = is_svg || block_attr.is_empty();
456  }
457
458  if nnodes < 1 {
459    // Insertion came up empty?
460    document.remove_node(container); // then remove the new block entirely
461    return Ok(nodes);
462  } else if ignorable_attr
463    && node_tags
464      .iter()
465      .all(|tag| document::can_contain_qsym(context_tag, *tag))
466  {
467    // No attributes, contents allowed in context?
468    document.unwrap_nodes(container)?; // No container needed, at all.
469    return Ok(nodes);
470  } else if nnodes == 1 {
471    if document::can_contain_qsym(context_tag, node_tags[0])
472      && (ignorable_attr
473        || block_attr
474          .keys()
475          .all(|key| document::sym_can_have_attribute(node_tags[0], pin(key))))
476    {
477      // IF: Single node, allowed in context & accepts attributes
478      // THEN: Add attributes and unwrap the single node
479      //
480      // SURPASS-PERL (OXIDIZED_DESIGN #125): `class` MERGES, it does not
481      // overwrite. Perl's insertBlock (`TeX_Box.pool.ltxml` L492) uses
482      // `setAttribute(class => …)`, so a wrapper's class (e.g. minipage's
483      // `ltx_minipage`) clobbered the single child's own class — a `lstlisting`
484      // that is the sole content of a `minipage`-in-`figure` became
485      // `class="ltx_minipage"`, losing `ltx_lstlisting` and thus the
486      // whitespace-preserving CSS, so its indentation collapsed
487      // (html_feedback#6632, arXiv:2605.03143). LaTeXML already distinguishes
488      // `addClass` from `setAttribute`; merging the child's class with the
489      // wrapper's keeps both (`ltx_lstlisting ltx_minipage`).
490      for (k, v) in block_attr.iter() {
491        if k == "class" {
492          document.add_class(&mut nodes[0], v)?;
493        } else {
494          document.set_attribute(&mut nodes[0], k, v)?;
495        }
496      }
497      // the lone child takes the box's attributes, and so its mark (OXIDIZED_DESIGN #715)
498      if container.has_attribute(document::FIELD_BOX_ATTRIBUTE) {
499        document.mark_field_box(&mut nodes[0]);
500      }
501      document.unwrap_nodes(container)?;
502      return Ok(nodes);
503    } else if let Some(newcontainer) = document::sym_can_contain_somehow(context_tag, node_tags[0])
504      && (ignorable_attr
505        || block_attr.keys().all(|key| {
506          newcontainer
507            .map(|nc| document::sym_can_have_attribute(nc, pin(key)))
508            .unwrap_or(false)
509        }))
510      && let Some(nc) = newcontainer
511      // never an SVG element: with the foreignObject model widened to Flow
512      // (OXIDIZED_DESIGN #186) a minipage in a TikZ node resolved here to
513      // `svg:foreignObject` and the capture — class, vattach, `width` —
514      // was renamed INTO a foreignObject, clobbering the real one's px
515      // width (fixture tikz/various_colors); it belongs in the inline-block
516      // the candidates below pick.
517      && !with(nc, |s| s.starts_with("svg:"))
518    {
519      // rename the capture to that container
520      drop_panel_separators(document, &container);
521      document.rename_node_qsym(container, nc, true)?;
522      return Ok(nodes);
523    }
524  }
525  // This jagged conditional is a "code smell", due to the difficulty of refactoring
526  // the in-conditional-assignments from Perl.
527
528  // Otherwise, rename the capture
529  drop_panel_separators(document, &container);
530  // MAY need foreignObject wrapper
531  if is_svg
532    && node_tags
533      .iter()
534      .any(|tag| with(*tag, |tag_str| tag_str.starts_with("ltx:")))
535  {
536    context = document
537      .wrap_nodes("svg:foreignObject", vec![container.clone()])?
538      .expect("foreign object wrap should always succeed in SVG");
539    context_tag = document::get_node_qname(&context);
540  }
541  let candidates = if is_inline {
542    [
543      "ltx:inline-block",
544      "ltx:inline-logical-block",
545      "ltx:inline-sectional-block",
546    ]
547    .map(pin_static)
548    .to_vec()
549  } else {
550    [
551      "ltx:block",
552      "ltx:logical-block",
553      "ltx:sectional-block",
554      "ltx:figure",
555    ]
556    .map(pin_static)
557    .to_vec()
558  };
559  let filtered_candidates = candidates
560    .into_iter()
561    .filter(|candidate| {
562      node_tags
563        .iter()
564        .all(|tag| document::sym_can_contain_somehow(*candidate, *tag).is_some())
565    })
566    .collect::<Vec<_>>();
567  // and are allowed in the context
568  let allowed_candidates = filtered_candidates
569    .iter()
570    .filter(|candidate| document::can_contain_qsym(context_tag, **candidate))
571    .copied()
572    .collect::<Vec<_>>();
573  // TODO: There is an arena code smell here. The `Model` interface needs to become lock-free
574  // where Symbol tickets and &str are equally intuitive to use without runtime panics from
575  // arena mutability exceptions.
576  if let Some(final_tag) = allowed_candidates.first() {
577    // The context can hold this block — rename the capture in place.
578    document.rename_node(container, &to_string(*final_tag), true)?;
579  } else if let Some(&final_tag) = filtered_candidates.first() {
580    // Surpass (OXIDIZED_DESIGN #240): the context (e.g. an already-open `<para>`)
581    // cannot hold this Para.class block, but an ancestor can. Climb to the
582    // nearest such ancestor and move the capture there — ending the paragraph —
583    // instead of renaming in place, which yields a schema-invalid
584    // `<para>/<logical-block>` (LaTeXML-para.rnc:16,56). Perl renames in place
585    // (TeX_Box.pool.ltxml:512-513, in insertBlock :449-519), a SHARED schema bug;
586    // we keep the block
587    // rendering (a `<div>`, LaTeXML-para-xhtml.xsl:53) — the inline variant would
588    // wrongly nest block content (a float/table) in a `<span>`. Witnesses:
589    // tikz-network, numerica (framed `{shaded}` / `{minipage}` after inline text).
590    // Only reached on the currently-invalid path (`allowed_candidates` empty in a
591    // non-inline context); the inline/`<p>` path always has a valid candidate.
592    // Restrict the climb to a `<para>` context — the "block box ends the open
593    // paragraph" case (tikz-network/numerica). Other block-holding contexts (an
594    // SVG `foreignObject` / box that wraps its block in place) keep the pre-#4
595    // in-place rename, so a `\parbox` inside a picture stays in its foreignObject.
596    // A `<para>` only ever lives inside a Para.model container (item, section
597    // body, logical-block, …), so the immediate parent holds a `logical-block`
598    // and the climb stops one level up. A `sectional-block` (Para.class content
599    // carrying a `\paragraph`/`\subsection`) instead ends any enclosing section
600    // and lands at body level — valid (document.body.class holds it), a net
601    // improvement over Perl's invalid in-place nesting.
602    if with(context_tag, |t| t == "ltx:para") {
603      let mut child = context.clone();
604      while let Some(parent) = child.get_parent() {
605        if !matches!(parent.get_type(), Some(NodeType::ElementNode)) {
606          break;
607        }
608        if document::can_contain_qsym(document::get_node_qname(&parent), final_tag) {
609          container.unlink();
610          child.add_next_sibling(&mut container)?;
611          document.set_node(&parent);
612          break;
613        }
614        child = parent;
615      }
616    } else if document::can_contain_qsym(context_tag, pin_static("ltx:item"))
617      && document::can_contain_indirect(context_tag, final_tag).is_some()
618    {
619      // A list container (`itemize`/`enumerate`/`description`, model `item*`) holds
620      // no block, so a box opened in a list before its first `\item`, or between
621      // items, lands where the model has no room for it. LaTeX typesets such a box
622      // as list material (framed.sty's `shaded` around an `\item`, colorframed-doc
623      // :176; a `minipage` of `\item`s beside a table minipage, tableaux/exemples
624      // :71). The only schema-valid home is an item, which is autoOpen
625      // (latex_constructs.pool.ltxml:1277): place the capture through
626      // `find_insertion_point`, which opens `item` → `para` from the list exactly
627      // as it does for text arriving before the first `\item` (Perl's
628      // `computeIndirectModel` route). Perl's `insertBlock` renames in place
629      // (TeX_Box.pool.ltxml:512), a SHARED schema error; the next `\item` closes the
630      // auto-opened item. Only where such a route exists: a `sectional-block` has
631      // none (no item holds it) and keeps Perl's in-place rename. OXIDIZED_DESIGN
632      // #261. Guard
633      // `perfect_kernel_batch56::block_in_a_list_before_an_item_gets_an_auto_item`.
634      container.unlink();
635      document.set_node(&context);
636      let mut point = document.find_insertion_point_qsym(final_tag, None)?;
637      point.add_child(&mut container)?;
638      document.set_node(&point);
639    }
640    // If no ancestor can hold it (rare — Para.class always fits the section/body
641    // model), the capture stays where it is. Perl renames in place to the invalid
642    // Para.class element (TeX_Box.pool.ltxml:512); instead, if the current context
643    // holds `<block>` but not `final_tag`, emit a schema-valid `<block>` and recursively
644    // demote the Para.model content (surpass #244) — a box-forming construct
645    // (`\begin{center}`/minipage/`\parbox`) whose body auto-opened a `<para>` when a
646    // genuine block sat mid-content, placed in a Block.model container (titlepage,
647    // quote, figure, abstract, inline-block). Demotion is IN PLACE (no reorder; render:
648    // ltx_para/block/logical-block all `display:block`, only a redundant `<para>`
649    // grouping + its auto-`xml:id` drop). The #240 `ltx:para` climb above already moved
650    // the capture to an ancestor that holds `final_tag`, so this branch is false there
651    // (parent holds `final_tag`) and that path is untouched. Witnesses: webquiz,
652    // tabularcalc, short-math-guide, heria (+ the `caption_in_inline_parbox` sibling).
653    let demote_para = !is_inline
654      && container.get_parent().is_some_and(|parent| {
655        let pq = document::get_node_qname(&parent);
656        !document::can_contain_qsym(pq, final_tag)
657          && document::can_contain_qsym(pq, pin_static("ltx:block"))
658      })
659      // Only demote when the result is provably valid: a container holding a
660      // Para.model-only element `<block>` can't take (a float, a TOC, a sectioning
661      // unit) stays at Perl-parity rather than acquiring a fresh `malformed` error.
662      && subtree_is_block_demotable(&container);
663    // A box (`ltx:inline-block`: `\resizebox`, `\scalebox`, `\rotatebox`) holds none of the
664    // block containers but does hold the inline ones: a `{minipage}` there whose content cannot be
665    // demoted (a float — tabularray's tall table, 57cp review —, a theorem, a `\section*`) is the
666    // first inline container its parent holds that holds all of it (`inline-logical-block` holds
667    // floats, theorems and paragraphs; `inline-sectional-block` sections). Perl renames it in place to
668    // its first block candidate, `ltx:logical-block` (TeX_Box.pool.ltxml:502-513), invalid in the box
669    // (divergence #385). Guard `perfect_kernel_batch57::minipage_blocks_in_a_scaled_box_are_inline`.
670    let inline_home = (!is_inline && !demote_para)
671      .then(|| container.get_parent())
672      .flatten()
673      .and_then(|parent| {
674        let pq = document::get_node_qname(&parent);
675        if document::can_contain_qsym(pq, final_tag) {
676          return None;
677        }
678        [
679          "ltx:inline-block",
680          "ltx:inline-logical-block",
681          "ltx:inline-sectional-block",
682        ]
683        .map(pin_static)
684        .into_iter()
685        .find(|c| {
686          document::can_contain_qsym(pq, *c)
687            && node_tags
688              .iter()
689              .all(|t| document::sym_can_contain_somehow(*c, *t).is_some())
690        })
691      });
692    if demote_para {
693      // Bottom-up: demote the Para.class content while the container still allows it,
694      // THEN rename the container to <block> — so nothing is ever moved into a <block>
695      // while still Para.class (which would emit a spurious `malformed` Error).
696      demote_para_class_content(document, &container)?;
697      document.rename_node(container, "ltx:block", true)?;
698    } else if let Some(inline) = inline_home {
699      document.rename_node_qsym(container, inline, true)?;
700    } else {
701      document.rename_node(container, &to_string(final_tag), true)?;
702    }
703  } else {
704    // we didn't know what to do?
705    let message = with(context_tag, |ctxt_str| {
706      s!(
707        "Did not find a block-like candidate in {} (with attributes ({})",
708        ctxt_str,
709        block_attr
710          .iter()
711          .map(|(k, v)| s!("{k}={v}"))
712          .collect::<Vec<_>>()
713          .join(";")
714      )
715    });
716    Warn!("malformed", "_CaptureBlock_", message);
717    document.rename_node(container, "ltx:block", true)?;
718  }
719  Ok(nodes)
720}

722/// The row separators a capture that stays a container holds (`insert_block_as`): inside it they separate none of the
723/// float's panels, and a container the model chose without them may not hold an `ltx:break`.
724fn drop_panel_separators(document: &mut Document, container: &Node) {
725  for child in container.get_child_elements() {
726    if crate::latex_constructs::is_panel_separator(&child) {
727      document.remove_node(child);
728    }
729  }
