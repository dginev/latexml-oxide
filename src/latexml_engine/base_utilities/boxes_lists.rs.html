5/// Stomach-level counterpart to `read_box_contents`.
6///
7/// Perl: readBoxContents calls $stomach->beginMode($mode), then reads/digests tokens
8/// Predigest box contents by invoking T_BEGIN, which triggers
9/// the stomach's bgroup/egroup mechanism to properly handle the box body.
10///
11/// Perl's `List()` simplification (List.pm line 41-44):
12/// When a vertical-mode List has exactly one non-empty item and that item's mode
13/// is also vertical, return the item directly instead of wrapping in a List.
14/// This enables `is_vbox` property propagation for nested \vbox/\vtop.
15pub fn predigest_box_contents(tokens: ArgWrap) -> Result<Option<Digested>> {
16  // Default: use the CURRENT mode (legacy behavior, e.g. for `{Body}` arguments
17  // that aren't \vbox/\vtop/\hbox-flavored). For VBoxContents / HBoxContents,
18  // call predigest_box_contents_in_mode("internal_vertical" / "restricted_horizontal")
19  // explicitly so we mirror Perl's `readBoxContents(..., $mode)` (TeX_Box.pool.ltxml L133).
20  let mode = lookup_string_from_sym(pin!("MODE"));
21  predigest_box_contents_in_mode(tokens, &mode)
22}

24/// Perl-faithful body-digest for VBoxContents / HBoxContents parameters.
25/// Mirrors `readBoxContents($gullet, $everybox, $mode)` exactly:
26/// pushes a fresh frame for `$mode`, digests tokens until matching T_END,
27/// pops the frame.
28///
29/// The mode-aware variant matters when `\vtop` is invoked inside a `p{}`
30/// column body (alignment is in horizontal mode, but the `\vtop`'s VBox
31/// content must be digested in internal_vertical mode so `\@startpbox`'s
32/// `\vtop\bgroup ...` doesn't try to close groups in the wrong mode).
33/// Witness: 2210.13325 `\begin{tabular}{|p{1cm}|...}` — pre-fix Rust
34/// inherited the surrounding horizontal mode and emitted `\vtop`
35/// errors; Perl uses 'internal_vertical' here regardless of where the
36/// `\vtop` was invoked.
37pub fn predigest_box_contents_in_mode(tokens: ArgWrap, mode: &str) -> Result<Option<Digested>> {
38  predigest_list_in_mode(tokens, mode, false)
39}

49}

51fn predigest_list_in_mode(
52  _tokens: ArgWrap,
53  mode: &str,
54  end_graf: bool,
55) -> Result<Option<Digested>> {
56  // Perl: readBoxContents calls beginMode($mode) / endMode($mode) around the body reading.
57  // This creates a scoped frame where enterHorizontal can change MODE inplace.
58  // When endMode is called, leaveHorizontal_internal detects MODE='horizontal' with
59  // BOUND_MODE ending in 'vertical', triggers repackHorizontal, then pops the frame.
60  //
61  // NOTE: read_box_contents already consumed the opening { or \bgroup (§403 `scan_left_brace`, `is_left_brace`).
62  // invoke_token(T_BEGIN) pushes a synthetic group frame. The matching } or \egroup
63  // in the content will pop this frame, since \egroup is \let to T_END and
64  // invoke_token handles it via the standard group-closing mechanism.
65  // Perl: $stomach->beginMode($mode) — push a new frame for this box content scope
66  // VERTICAL box modes (`\vbox`/`\vtop`/p{}-cell VBoxContents): faithful port of
67  // Perl `readBoxContents` (TeX_Box.pool.ltxml L139-160). ONE mode frame as the
68  // group; a loop that STOPS at the matching `}` WITHOUT processing it, so
69  // `end_mode` runs `leave_horizontal_internal` (repack — capturing the *inner*
70  // `\hsize`) BEFORE `pop_stack_frame` restores it. The `invoke_token(&T_BEGIN!())`
71  // path used for other modes lets the body's `}` egroup-restore `\hsize` before
72  // the post-hoc repack, so a `\vbox{\hsize=W …para…}` would wrap at the OUTER
73  // `\hsize` (the Cluster G `\narrow`-hang; rotated p{}-cell mis-sizing —
74  // SYNC_STATUS 1610.00974 step-3). hbox/math don't paragraph-wrap → the simpler
75  // `invoke_token(T_BEGIN)` path below. (A tabular inside a `\vbox`/`\vtop` is
76  // correctly SKIPPED by the repack because `\@@tabular`/`\halign` mark the result
77  // box `internal_vertical`.)
78  // RESTRICTED HORIZONTAL (`\\hbox`/`\\mbox`/`\\makebox`… contents) takes the
79  // same one-frame loop (batch 54n). tex.web §1083 `begin_box` pushes the
80  // nest and the save level TOGETHER for `\\hbox\\bgroup` and §1100 `package`
81  // pops both; Perl's one frame is that single push. The former two-frame
82  // shortcut (`begin_mode` + the `{` primitive's own `bgroup`) made the box
83  // depend on an `\\egroup` to close its synthetic group, and ulem's
84  // open-here/close-there word boxes (examdesign.cls:186-200 `\\UL@start` =
85  // `\\setbox\\UL@box\\hbox\\bgroup…\\bgroup`, `\\UL@stop` = `\\egroup\\egroup`) around a
86  // `\\makebox` whose argument digests in isolation met the wrong frame
87  // ("\\egroup Attempt to close a group that switched to mode
88  // restricted_horizontal"; examdesign examplea/b/c, Perl shares it). Guard:
89  // `perfect_kernel_batch54::hbox_reader_is_one_frame`.
90  if mode.ends_with("vertical") || mode == "restricted_horizontal" {
91    begin_mode(mode)?;
92    // A vertical box's own list (`VBoxContents`), where `\unvbox` puts a box's lines back (tex_box.rs `vlist_of`).
93    if mode == "internal_vertical" {
94      assign_frame_value("lx@vbox@list", true);
95    }
96    // The box's OWN frame (tex.web §1068 `handle_right_brace` dispatches on
97    // `cur_group`, the group itself): Perl keys the terminal on the save-stack
98    // DEPTH (`$level >= getFrameDepth`, TeX_Box.pool.ltxml:172), which a
99    // crossing — a `\begingroup` opened inside the box after a frame below
100    // it was popped, pgf's node text over our bound `\pgfsys@*` seam — can
101    // satisfy with a `\begingroup` frame on top, so `end_mode` then failed
102    // ("\hbox Attempt to end mode restricted_horizontal", modernposter; the
103    // msc/dsptricks cascades). A `}` that meets another frame is invoked
104    // instead: `egroup` reports and drops it (tex.web §1069
105    // `extra_right_brace`) and the box goes on to its own `}`.
106    let own_frame = current_frame_id();
107    new_local_box_list(); // Perl: local @LaTeXML::LIST = ()
108    // The box's contents are digested outside any number scan in progress
109    // (`gullet::NumberScan::suspend`, as `stomach::digest` does): this loop is
110    // a stomach entry of its own.
111    let _outside_scan = NumberScan::suspend();
112    loop {
113      let next = match get_pending_comment() {
114        Some(comment) => Some(comment),
115        None => read_x_token(Some(true), false, None)?,
116      };
117      let Some(token) = next else { break };
118      // The box's own `}` (its frame on top) ends the loop and is closed by
119      // `end_mode`, NOT egroup; any other `}` is processed.
120      if is_right_brace(&token) && current_frame_id() == own_frame {
121        break;
122      }
123      check_timeout()?; // runaway guard (mirrors the canonical group-digest loop)
124      extend_box_list(invoke_token(&token)?);
125    }
126    // tex.web §679 `append_to_vlist`: the interline glue between a vertical box's lines follows the
127    // `\baselineskip`, `\lineskip` and `\lineskiplimit` in force inside it (`\offinterlineskip` in the box), read
128    // before its group ends; Perl's `List` records the `\baselineskip` after it (List.pm:52-53). Its sizing and
129    // `\vsplit` (tex_inserts.rs) stack the lines with them (OXIDIZED_DESIGN_DIVERGENCES #421; short-math-guide's
130    // `symlist` columns split as its PDF).
131    let interline: Vec<(&str, Dimension)> = if mode.ends_with("vertical") {
132      [
133        ("baseline", "\\baselineskip"),
134        ("lineskip", "\\lineskip"),
135        ("lineskiplimit", "\\lineskiplimit"),
136      ]
137      .into_iter()
138      .filter_map(|(key, register)| {
139        lookup_register_quiet(register).map(|value| (key, Dimension::new(value.value_of())))
140      })
141      .collect()
142    } else {
143      Vec::new()
144    };
145    // §1100 `insert_group`: `end_graf` before the group's list is packaged. The primitive `\lx@normal@par` also when
146    // the list ends in vertical mode (where `end_graf` does nothing): it closes what the material opened, so a box
147    // set in it (`\vadjust{\box0}`) is no part of the paragraph after.
148    if end_graf {
149      extend_box_list(invoke_token(&T_CS!("\\lx@normal@par"))?);
150    }
151    // Perl: $stomach->endMode($mode) — leave_horizontal_internal (repack with the
152    // still-in-scope inner `\hsize`) THEN pop_stack_frame (restores `\hsize`).
153    end_mode(mode)?;
154    let mode_tex = if lookup_bool_sym(pin!("IN_MATH")) {
155      TexMode::Math
156    } else {
157      TexMode::Text
158    };
159    let mut digested_list = List::new(expire_local_box_list());
160    digested_list.mode = Some(mode_tex);
161    for (key, value) in interline {
162      digested_list
163        .properties
164        .insert(key, Stored::Dimension(value));
165    }
166    let mut item: Digested = digested_list.into();
167    // Perl: List(@LaTeXML::LIST, mode => $mode)
168    item.set_property("mode", Stored::String(pin(mode)));
169    // Perl's single-item `List()` simplification is a vertical-mode rule.
170    return Ok(Some(if mode.ends_with("vertical") {
171      simplify_vertical_list(item)
172    } else {
173      item
174    }));
175  }
176  if mode.ends_with("vertical") || mode.ends_with("horizontal") {
177    begin_mode(mode)?;
178  }
179  let mut contents = invoke_token(&T_BEGIN!())?;
180  if contents.is_empty() {
181    // Perl: $stomach->endMode($mode)
182    if mode.ends_with("vertical") || mode.ends_with("horizontal") {
183      end_mode(mode)?;
184    }
185    Ok(None)
186  } else {
187    let mut item = contents.remove(0);
188    // Perl's endMode triggers leaveHorizontal_internal → repackHorizontal
189    // when enterHorizontal changed MODE to 'horizontal' inplace within this frame.
190    // Check the condition BEFORE endMode pops the frame.
191    let post_mode = lookup_string_from_sym(pin!("MODE"));
192    let bound_mode = lookup_string_from_sym(pin!("BOUND_MODE"));
193    if post_mode == "horizontal"
194      && bound_mode.ends_with("vertical")
195      && has_only_simple_horizontal_content(&item)
196    {
197      repack_horizontal_in_list(&mut item);
198      // Restore MODE like leave_horizontal_internal does
199      assign_value_inplace_sym(pin!("MODE"), pin(&bound_mode));
200    }
201    // Perl: $stomach->endMode($mode) — pop the frame
202    if mode.ends_with("vertical") || mode.ends_with("horizontal") {
203      end_mode(mode)?;
204    }
205    // Set the mode property on the resulting item (matching Perl's List(@boxes, mode => $mode))
206    if !mode.is_empty() {
207      item.set_property("mode", Stored::String(pin(mode)));
208    }
209    // Apply Perl's List() single-item simplification for vertical modes.
210    // In Perl, List(@boxes, mode=>'internal_vertical') returns the single box
211    // directly when @boxes has 1 element and the box's mode is also vertical.
212    // This is critical for nested \vbox/\vtop: the inner box's `is_vbox` property
213    // must be visible to the outer box's constructor.
214    Ok(Some(simplify_vertical_list(item)))
215  }
216}

218/// Check if a List contains only simple horizontal content (TBoxes,
219/// Comments, or sub-Lists whose mode is horizontal/restricted_horizontal/
220/// math). This guards against repack being triggered for cases like
221/// `\vtop{\begin{tabular}...}` where the tabular processing leaks
222/// MODE='horizontal' but the content (an Alignment/Whatsit) should NOT
223/// be paragraph-wrapped.
224///
225/// Inline brace-groups `{...}` inside running text produce sub-Lists
226/// (one per group) whose mode is `restricted_horizontal`. Without
227/// accepting those, e.g. `\vbox{\small\bfseries hello {,} world}` would
228/// fail the repack gate and be measured as 3 separate vertical lines.
229/// Witness: aistats2026.sty's `\def\And{\unskip{,}\enspace}` in the
230/// `\@runningauthor` body — every author separator emits a `{,}` group
231/// that fragments the vbox into many short rows, making `\ht\autrun`
232/// far exceed 10pt and triggering the class's `\PackageError{Document}
233/// {Running heading author exceeds size limitations}` (driver paper:
234/// arXiv:2602.11863).
235fn has_only_simple_horizontal_content(item: &Digested) -> bool {
236  if let DigestedData::List(l) = item.data() {
237    let list = l.borrow();
238    let non_empty: Vec<_> = list
239      .boxes
240      .iter()
241      .filter(|b| !b.get_property_bool("isEmpty"))
242      .collect();
243    non_empty.iter().all(|b| match b.data() {
244      DigestedData::TBox(_) | DigestedData::Comment(_) => true,
245      DigestedData::List(sub) => {
246        // Inline brace-groups `{...}` in horizontal context digest to
247        // a sub-List with no `mode` property (implicit hbox). Accept
248        // those plus sub-Lists explicitly tagged as horizontal-flavour.
249        // Reject `mode=vertical|internal_vertical` (structural
250        // sub-vboxes) and any other tagged mode.
251        let sub_mode = sub
252          .borrow()
253          .properties
254          .get("mode")
255          .map(|v| v.to_string())
256          .unwrap_or_default();
257        matches!(
258          sub_mode.as_str(),
259          "" | "horizontal" | "restricted_horizontal" | "math"
260        )
261      },
262      _ => false,
263    })
264  } else {
265    false
266  }
267}

269/// Replicates Perl's repackHorizontal() within a List's children.
270///
271/// Perl (Stomach.pm lines 442-456): In readBoxContents, after digesting box content
272/// in vertical mode, repackHorizontal groups consecutive horizontal-mode items
273/// from @LaTeXML::LIST into a single List(@para, mode => 'horizontal') with
274/// width set to \hsize. This enables compute_boxes_size to do paragraph wrapping.
275///
276/// Without this, \vbox{hop} measures each character individually (width=5.55pt)
277/// instead of wrapping as a paragraph at \hsize (width=469.75pt).
278fn repack_horizontal_in_list(item: &mut Digested) {
279  if let DigestedData::List(l) = item.data() {
280    let mut list = l.borrow_mut();
281    let children = std::mem::take(&mut list.boxes);
282    let mut result: Vec<Digested> = Vec::new();
283    let mut para: Vec<Digested> = Vec::new();
284    let mut keep = false;
285
286    for child in children {
287      let child_mode = child
288        .get_property("mode")
289        .map(|v| v.to_string())
290        .unwrap_or_else(|| "horizontal".to_string());
291      // Empty-string mode means "implicit hbox" — produced by inline
292      // brace-groups `{...}` in horizontal context. Treat as horizontal
293      // so the surrounding running text doesn't get fragmented.
294      let effective_mode = if child_mode.is_empty() {
295        "horizontal"
296      } else {
297        child_mode.as_str()
298      };
299      if effective_mode == "horizontal"
300        || effective_mode == "restricted_horizontal"
301        || effective_mode == "math"
302      {
303        // Perl: $keep = 1 if ($mode ne 'horizontal') || !$item->getProperty('isSpace');
304        if effective_mode != "horizontal" || !child.get_property_bool("isSpace") {
305          keep = true;
306        }
307        para.push(child);
308      } else {
309        // Flush accumulated horizontal items as a horizontal List
310        if keep {
311          let horiz_list = make_horizontal_list(std::mem::take(&mut para));
312          result.push(Digested::from(horiz_list));
313        } else {
314          result.extend(std::mem::take(&mut para));
315        }
316        keep = false;
317        result.push(child);
318      }
319    }
320    // Flush remaining horizontal items
321    if keep {
322      let horiz_list = make_horizontal_list(para);
323      result.push(Digested::from(horiz_list));
324    } else {
325      result.extend(para);
326    }
327    list.boxes = result;
328  }
329}

331/// Create a horizontal List with mode='horizontal' and width=\hsize.
332/// Perl: push(@LaTeXML::LIST, List(@para, mode => 'horizontal')) if $keep;
333/// Perl: $list->setProperty(width => LookupRegister('\hsize')) if $mode eq 'horizontal';
334fn make_horizontal_list(para: Vec<Digested>) -> List {
335  let mut list = List::new(para);
336  list.mode = Some(TexMode::Text);
337  list.set_property("mode", Stored::String(pin!("horizontal")));
338  if let Some(hsize) = lookup_dimension("\\hsize") {
339    list.set_property("width", Stored::Dimension(hsize));
340  }
341  list
342}

344/// Perl's List() single-item simplification for vertical modes.
345///
346/// Perl (List.pm line 41-44):
347/// ```perl
348/// if ((scalar(@boxes) == 1)
349///     && (!$mode || ($mode !~ /vertical$/)
350///         || (($boxes[0]->getProperty('mode')||'') =~ /vertical$/))) {
351///     return $boxes[0]; }   # Simplify!
352/// ```
353///
354/// When a List in vertical mode contains a single non-empty item whose mode is also
355/// vertical, return that item directly. This is critical for nested \vbox/\vtop:
356/// the inner \vbox Whatsit has `is_vbox = true` set by after_digest, and the outer
357/// \vtop constructor needs to see this property to skip double insertBlock wrapping.
358fn simplify_vertical_list(item: Digested) -> Digested {
359  // Only simplify if the item is a List
360  let is_vertical_list = match item.data() {
361    DigestedData::List(l) => {
362      let list = l.borrow();
363      // Check if the List's mode property indicates vertical
364      list
365        .properties
366        .get("mode")
367        .map(|m| m.ends_with_text("vertical"))
368        .unwrap_or(false)
369    },
370    _ => false,
371  };
372  if !is_vertical_list {
373    return item;
374  }
375
376  // Extract the List's boxes, filtering out empty marker items (isEmpty property)
377  let non_empty: Vec<Digested> = match item.data() {
378    DigestedData::List(l) => {
379      let list = l.borrow();
380      list
381        .boxes
382        .iter()
383        .filter(|b| !b.get_property_bool("isEmpty"))
384        .cloned()
385        .collect()
386    },
387    _ => unreachable!(),
388  };
389
390  // Perl simplification: single non-empty item whose mode is also vertical
391  if non_empty.len() == 1 {
392    let single = &non_empty[0];
393    let child_is_vertical = match single.data() {
394      DigestedData::List(l) => l
395        .borrow()
396        .properties
397        .get("mode")
398        .map(|m| m.ends_with_text("vertical"))
399        .unwrap_or(false),
400      DigestedData::Whatsit(w) => {
401        // Check whatsit's mode property (set by DefConstructor mode => "internal_vertical")
402        w.borrow()
403          .get_property("mode")
404          .map(|m| m.ends_with_text("vertical"))
405          .unwrap_or(false)
406      },
407      _ => false,
408    };
409    if child_is_vertical {
410      return non_empty.into_iter().next().unwrap();
411    }
412  }
413  item
414}

416/// Perl: revertSpec($whatsit, $keyword)
417/// If whatsit has property $keyword, return Explode($keyword) ++ Revert($value)
418pub fn revert_spec(whatsit: &Whatsit, keyword: &str) -> Vec<Token> {
419  match whatsit.get_property(keyword) {
420    Some(value) => {
421      // Explode the keyword + value strings into T_OTHER tokens. `pin_char`
422      // uses a stack-buffer encode_utf8 and skips the per-char
423      // `c.to_string()` heap alloc the previous version did.
424      let mut tokens: Vec<Token> = keyword.chars().map(|c| CharToken!(c)).collect();
425      let val_str = value.to_attribute();
426      tokens.extend(val_str.chars().map(|c| CharToken!(c)));
427      tokens
428    },
429    _ => Vec::new(),
430  }
431}

433pub fn p_revert<T>(arg: T) -> Result<Tokens>
434where T: Sized + Object {
435  set_dual_branch("presentation");
436  let result = arg.revert();
437  expire_dual_branch();
438  result
439}
