11pub fn reenter_text_mode(vertical_mode: bool) {
12  let mode_key = if vertical_mode {
13    "VTEXT_MODE_BINDINGS"
14  } else {
15    "HTEXT_MODE_BINDINGS"
16  };
17  let text_key = "TEXT_MODE_BINDINGS";
18  let mode_bindings = checkout_value(mode_key);
19  let text_bindings = checkout_value(text_key);
20  let mut bindings: VecDeque<&Stored> = match mode_bindings {
21    Some(Stored::VecDequeStored(ref vdq)) => vdq.iter().collect::<VecDeque<&Stored>>(),
22    _ => VecDeque::new(),
23  };
24  if let Some(Stored::VecDequeStored(ref vdq)) = text_bindings {
25    bindings.extend(vdq.iter());
26  }
27  for binding in bindings {
28    if let Stored::Tokens(tks) = binding {
29      let vec = tks.unlist_ref();
30      let_i(&vec[0], &vec[1], None);
31    }
32  }
33  if let Some(value) = mode_bindings {
34    checkin_value(mode_key, value);
35  }
36  if let Some(value) = text_bindings {
37    checkin_value(text_key, value);
38  }
39}

41// Similarly, for metadata appearing within peculiar environments, fonts, etc
42// You'll typically want this within a group or bounded=>1.
43/// Reset the text and math fonts to their defaults, **locally**.
44///
45/// For metadata that must not inherit the font it happens to be written in —
46/// a title picked up inside an italic environment, an author in a small-caps
47/// box. The assignments are group-local, so use this inside a group (or a
48/// `bounded => 1` definition), otherwise the reset outlives the construct that
49/// wanted it.
50///
51/// The NFSS codes follow, as `\reset@font` (`\normalfont`, latex.ltx:14122)
52/// sets them where LaTeX resets a note's font (`\@footnotetext`,
53/// latex.ltx:17658-17659 `\reset@font\footnotesize`): else a size switch in
54/// the note, which ends in `\selectfont`, re-selected the surrounding family,
55/// series or shape (`\textit{A\footnote{\footnotesize note}}` came out
56/// italic).
57///
58/// The encoding is `\encodingdefault`, which `\reset@font` selects (`\fontencoding\encodingdefault`,
59/// latex.ltx:14113-14122): a document's text encoding — fontenc's last option, fontspec's TU, babel's `\latinencoding`
60/// (babel.sty:3936-3938), greek.ldf's `\greekfontencoding` (:151-153) — not the TeX-wide OT1, else a T1 document's
61/// footnotes (`\reset@font`, latex.ltx:17659) typeset `<b> x|y` through OT1 as `¡b¿ x—y` (Perl alike,
62/// TeX_Box.pool.ltxml:218-221). LaTeXML's tags neutralize too, where a LaTeX label inherits the body's encoding. Plain
63/// TeX has no `\encodingdefault`: OT1.
64///
65/// The family, series and shape are likewise the document's `\familydefault`, `\seriesdefault` and `\shapedefault`
66/// that `\reset@font` selects (`\edef\f@family{\familydefault}`, latex.ltx:14115-14117), where they name a font
67/// LaTeXML knows: a `\renewcommand\familydefault{\sfdefault}` document's footnotes and tags are sans, as its PDF
68/// (CMSS8 footnote text), not Perl's serif `textDefault`. OXIDIZED_DESIGN_DIVERGENCES #779; guard
69/// `perfect_kernel_batch77u::family_default_sets_the_body`.
70pub fn neutralize_font() {
71  let mut font = Font::text_default();
72  if let Some(encoding) = encoding_named("\\encodingdefault") {
73    font.encoding = Some(Cow::Owned(encoding));
74  }
75  let components = [
76    font_code_named("\\familydefault").and_then(|code| font::lookup_font_family(&code)),
77    font_code_named("\\seriesdefault").and_then(|code| font::lookup_font_series(&code)),
78    font_code_named("\\shapedefault").and_then(|code| font::lookup_font_shape(&code)),
79  ];
80  for component in components.into_iter().flatten() {
81    font = font.merge_ref(component);
82  }
83  sync_nfss_font_codes(&font);
84  assign_value("font", font, Some(Scope::Local));
85  assign_value("mathfont", Font::math_default(), Some(Scope::Local));
86}

88/// The NFSS code the macro `cs` (`\encodingdefault`, `\familydefault`, …) names, expanded as `\fontencoding` or
89/// `\edef\f@family{…}` expands it. The common cases are read without expanding, as they are on every tag: a literal
90/// name (`OT1`), or a chain of parameterless macros ending in one (`\familydefault` → `\rmdefault` → `cmr`). `None`
91/// when undefined (plain TeX) or empty.
92fn font_code_named(cs: &str) -> Option<String> {
93  let mut token = T_CS!(cs);
94  let mut literal = None;
95  for _ in 0..8 {
96    let defn = lookup_definition(&token).ok()??;
97    let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
98      break;
99    };
100    if defn
101      .get_parameters()
102      .is_some_and(|p| !p.get_parameters().is_empty())
103    {
104      break;
105    }
106    let tokens = body.unlist_ref();
107    if tokens
108      .iter()
109      .all(|t| matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER))
110    {
111      literal = Some(body.to_string());
112      break;
113    }
114    match tokens[..] {
115      [next] if matches!(next.get_catcode(), Catcode::CS) => token = next,
116      _ => break,
117    }
118  }
119  let code = match literal {
120    Some(code) => code,
121    None => do_expand(T_CS!(cs)).ok()?.to_string(),
122  };
123  let code = code.trim();
124  (!code.is_empty()).then(|| code.to_string())
125}

127/// The encoding the macro `cs` (`\encodingdefault`, `\latinencoding`) names ([`font_code_named`]). `None` when
128/// undefined (plain TeX) or not an encoding name.
129fn encoding_named(cs: &str) -> Option<String> {
130  font_code_named(cs).filter(|encoding| encoding != "ASCII" && !encoding.contains('\\'))
131}

133/// Today's date as LaTeX's `\today` renders it — `Month D, YYYY`.
134///
135/// Mirrors Perl `TeX_Job.pool.ltxml` L52-55, reading `\year`/`\month`/`\day`
136/// from the **value** table where job startup put them — not from the register
137/// meanings, which a class file is free to `\def` out from under us. A missing
138/// value falls back to a default instead of aborting the conversion; the body
139/// comment records that divergence and its witnesses.
140pub fn today() -> Result<String> {
141  let month_names = [
142    "January",
143    "February",
144    "March",
145    "April",
146    "May",
147    "June",
148    "July",
149    "August",
150    "September",
151    "October",
152    "November",
153    "December",
154  ];
155  // Mirror Perl TeX_Job.pool.ltxml L52-55:
156  //   $MonthNames[LookupValue('\month')->valueOf - 1]
157  //     . " " . LookupValue('\day')->valueOf
158  //     . ', ' . LookupValue('\year')->valueOf
159  // Read from the VALUE table (assigned in tex_job.rs:47-49 at job
160  // startup), NOT the register-meaning table. `lookup_register` walks
161  // meanings and returns None when a class file (e.g. iopart.cls)
162  // `\def`s `\day` for its own purposes, panicking the unwrap on
163  // 1208.0134/1702.02270/1705.08909. Intentional Perl divergence:
164  // default to 1900-01-01 when a value is missing, where Perl would
165  // die on `Can't call method "valueOf" on undef` — keeps the
166  // conversion alive when something has clobbered the value table.
167  let read = |key: &str, default: i32| -> i32 {
168    match lookup_value(key) {
169      Some(Stored::Int(n)) => n as i32,
170      Some(Stored::Number(n)) => n.value_of() as i32,
171      _ => default,
172    }
173  };
174  let m = read("\\month", 1).clamp(1, 12) as usize;
175  let month = month_names[m - 1];
176  let day = read("\\day", 1);
177  let year = read("\\year", 1900);
178  Ok(s!("{} {}, {}", month, day, year))
179}

181pub fn parse_def_parameters(cs: &Token, params_in: Tokens) -> Result<Option<Parameters>> {
182  let mut tokens: VecDeque<Token> = VecDeque::from(params_in.pack_parameters()?.unlist());
183  // Now, recognize parameters and delimiters.
184  let mut params = Vec::new();
185  let mut n = 0;
186  while let Some(mut t) = tokens.pop_front() {
187    let cc = t.get_catcode();
188    if cc == Catcode::PARAM || cc == Catcode::ARG {
189      if cc == Catcode::PARAM {
190        if tokens.is_empty() {
191          // Special case: lone # NOT following a numbered parameter
192          // Note that we require a { to appear next, but do NOT read it!
193          params.push(Parameter::new(
194            Cow::Borrowed("RequireBrace"),
195            Cow::Borrowed("RequireBrace"),
196            None,
197          )?);
198          break;
199        } else {
200          n += 1;
201          if let Some(t_next) = tokens.pop_front() {
202            t = t_next;
203          } else {
204            unreachable!("tokens.is_empty() was false, so pop_front must return Some");
205          }
206        }
207      } else {
208        // CC_ARG case, keep looking at this token
209        n += 1;
210      }
211      if n > 0 {
212        let t_num = t.with_str(|ts| ts.parse::<i8>()).unwrap_or(-1);
213        if t_num != n {
214          fatal!(
215            ParamSpec,
216            Expected,
217            s!(
218              "Parameters for {:?} not in order. Got {:?}, expected {:?}. in {:?}",
219              cs,
220              t,
221              n,
222              params
223            )
224          );
225        }
226      }
227      // Check for delimiting text following the parameter #n
228      // Every token of the delimiter is kept, consecutive SPACE tokens
229      // included (tex.web §473-476 `scan_toks` reads the parameter text with
230      // `get_token`, no collapsing; the tokenizer's `skip_blanks` state has
231      // already folded file-sourced runs before they get here). Perl
232      // TeX_Macro.pool.ltxml L127 drops the second of two adjacent spaces
233      // ("BUT collapse whitespace!"), which only ever bites an
234      // expansion-built parameter text: expkv.tex L709-712
235      // `\ekv@set@was@blank`'s `#1` delimiter carries TWO real spaces
236      // (`{ }` through `\ekv@strip@key` twice), so the collapsed `Until:`
237      // never matched, the marker dance derailed (`\ekv@stop`/`\ekv@nil`
238      // undefined) and the re-scanned tail ran to `Timeout:TokenLimit`
239      // (witness tutodoc-en/fr via clrstrip's `\ekvset{clrstrip}{}`;
240      // KNOWN_PERL_ERRORS #119; guard
241      // `perfect_kernel_batch53::def_delimiter_keeps_adjacent_spaces`).
242      let mut delim = Vec::new();
243      while !tokens.is_empty() {
244        let inner_cc = tokens.front().unwrap().get_catcode();
245        if inner_cc == Catcode::PARAM || inner_cc == Catcode::ARG {
246          break;
247        }
248        delim.push(tokens.pop_front().unwrap());
249      }
250      // Found text that marks the end of the parameter
251      if !delim.is_empty() {
252        // tex.web §476: a `#` that ends the parameter text (`#{`) appends the `{` to the last delimiter and to the
253        // replacement text, so `\def\pt(#1)#{…}` delimits #1 by `){` and gives the brace back after its expansion
254        // (the `Until` reader puts a matched delimiter's closing `{` back, `read_until` matches it as tex.web §392
255        // does). Perl (TeX_Macro.pool.ltxml:108-112) reads #1 to the first `)` and then requires a `{`: tkz-euclide's
256        // `\def\tkz@DefPoint[#1](#2)#{` (tkz-obj-eu-points.tex:82) split `\tkzDefPoint(2*cos(1),3){T}` at
257        // `cos(1` and raised "expected {" (witness 2610.26441, 9 Errors with what followed; KNOWN_PERL_ERRORS #1029).
258        // Guards `perfect_kernel_batch78d::hash_brace_extends_the_last_delimiter`, `kernel_def_hash_brace_delimiter`.
259        if tokens.len() == 1 && tokens.front().unwrap().get_catcode() == Catcode::PARAM {
260          tokens.pop_front();
261          delim.push(T_BEGIN!());
262        }
263        let extra = Tokens::new(delim);
264        params.push(
265          Parameter {
266            name: pin!("Until"),
267            spec: pin(format!("Until:{extra}")),
268            extra: vec![extra],
269            ..Parameter::default()
270          }
271          .init()?,
272        );
273      } else if tokens.len() == 1 && tokens.front().unwrap().get_catcode() == Catcode::PARAM {
274        // Special case: trailing sole # => delimited by next opening brace.
275        tokens.pop_front();
276        params.push(Parameter::new("UntilBrace", "UntilBrace", None)?);
277      } else {
278        // Nothing? Just a plain parameter.
279        params.push(Parameter::new("Plain", "{}", None)?);
280      }
281    } else {
282      // Initial delimiting text is required.
283      let mut lit: Vec<Token> = vec![t];
284      while !tokens.is_empty() {
285        let lit_cc = tokens.front().unwrap().get_catcode();
286        if lit_cc == Catcode::PARAM || lit_cc == Catcode::ARG {
287          break;
288        }
289        lit.push(tokens.pop_front().unwrap());
290      }
291      let expected = Tokens::new(lit);
292      params.push(
293        Parameter {
294          name: pin!("Match"),
295          spec: pin(s!("Match:{expected}")),
296          extra: vec![expected],
297          novalue: true,
298          ..Parameter::default()
299        }
300        .init()?,
301      );
302    }
303  }
304  // return (@params ? LaTeXML::Core::Parameters->new(@params) : undef);
305  if params.is_empty() {
306    Ok(None)
307  } else {
308    Ok(Some(Parameters::new(params)))
309  }
310}

312pub fn do_def(globally: bool, cs: Token, params: Tokens, body: Tokens) -> Result<()> {
313  let paramlist = parse_def_parameters(&cs, params)?;
314  let scope = if globally { Some(Scope::Global) } else { None };
315  install_definition(
316    Expandable::new(
317      cs,
318      paramlist,
319      Some(ExpansionBody::Tokens(body)),
320      Some(ExpandableOptions {
321        nopack_parameters: true,
322        tex_declared: true,
323        ..ExpandableOptions::default()
324      }),
325    )?,
326    scope,
327  );
328  after_assignment();
329  Ok(())
330}

332// Kinda rough: We don't really keep track of modes as carefully as TeX does.
333// We'll assume that a box is horizontal if there's anything at all,
334// but it's not a vbox (!?!?)
335pub fn classify_box(boxnum: Number) -> Result<&'static str> {
336  with_value(&s!("box{}", boxnum.value_of()), |val_opt| {
337    Ok(match val_opt {
338      Some(Stored::Digested(d)) => match d.data() {
339        DigestedData::Whatsit(w)
340          if w.borrow().definition == lookup_definition(&T_CS!("\\vbox"))?.unwrap() =>
341        {
342          "vbox"
343        },
344        _ => "hbox",
345      },
346      _ => "",
347    })
348  })
