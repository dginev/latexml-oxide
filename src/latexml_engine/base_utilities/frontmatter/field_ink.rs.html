5/// A `{titlepage}` built AFTER body content began cannot be the schema's leading
6/// `ltx:titlepage` (`document_model` admits it only in the front group): a leaked
7/// preamble argument, an undefined-command marker or genuine prose before it has
8/// already opened the body, so the element stands stranded — `element "titlepage"
9/// not allowed here` — in both engines (Perl's `{titlepage}` is the same
10/// `<ltx:titlepage>#body` constructor, latex_constructs.pool.ltxml:1167, with only an
11/// Info: "Frontmatter will not be well-structured"; verified on the minimal repro:
12/// identical output, one jing error). Its content is hand-typeset layout, so it is
13/// demoted in place to an `ltx:para` carrying `class="ltx_titlepage"` — the same
14/// text in the same order, no wrapper the schema forbids (OXIDIZED_DESIGN #257;
15/// chemexec_de/en, stanli, l2picfaq, pst-calendar-doc, classicthesis in sweep s110).
16/// A titlepage in its proper leading position, or one whose children `ltx:para`
17/// cannot hold (a `\maketitle` unwound inside it), is left alone.
18pub fn demote_stranded_titlepage(document: &mut Document) -> Result<()> {
19  let Some(root) = document.get_document().get_root_element() else {
20    return Ok(());
21  };
22  let kids: Vec<Node> = root
23    .get_child_nodes()
24    .into_iter()
25    .filter(|n| n.get_type() == Some(NodeType::ElementNode))
26    .collect();
27  let Some(pos) = kids
28    .iter()
29    .rposition(|n| document::get_node_qname(n) == pin_static("ltx:titlepage"))
30  else {
31    return Ok(());
32  };
33  let stranded = kids[..pos].iter().any(|k| {
34    document::get_node_qname(k) != pin_static("ltx:resource")
35      && !is_frontmatter_group_element(k)
36      && !node_is_content_free(k)
37  });
38  if !stranded {
39    return Ok(());
40  }
41  let titlepage = kids[pos].clone();
42  // The same child normalization `wrap_as_titlepage` applies: a `center`'s
43  // `logical-block` and a `para` become `block`s (#244's demotion), which
44  // `ltx:para` — like `ltx:titlepage` — admits. Only when every such child
45  // SURVIVES the demotion (`subtree_is_block_demotable`: no float/figure
46  // inside — stanli's cover holds `\begin{figure}`s in its `center`); else the
47  // titlepage is left as Perl leaves it, nothing touched.
48  let children: Vec<Node> = titlepage
49    .get_child_nodes()
50    .into_iter()
51    .filter(|c| c.get_type() == Some(NodeType::ElementNode))
52    .collect();
53  // Decide BEFORE mutating: every layout child must survive the demotion, and
54  // every child — as it will be after it (`logical-block`/`para` → `block`) —
55  // must be something `ltx:para` holds. Otherwise nothing is touched.
56  let layout = |q: SymStr| q == pin_static("ltx:logical-block") || q == pin_static("ltx:para");
57  let demotable = children
58    .iter()
59    .all(|c| !layout(document::get_node_qname(c)) || subtree_is_block_demotable(c));
60  let holdable = children.iter().all(|c| {
61    let q = document::get_node_qname(c);
62    layout(q) || document::can_contain_qsym(pin_static("ltx:para"), q)
63  });
64  if !(demotable && holdable) {
65    return Ok(());
66  }
67  // A demoted cover no longer stands in for the document's `<ltx:title>`, which the HTML then displays: the class's
68  // own printing `\title` run on it would be a second copy (OXIDIZED_DESIGN #629).
69  drop_cover_title_ink(document, &titlepage);
70  normalize_titlepage_layout_children(document, &titlepage)?;
71  let mut para = document.rename_node(titlepage, "ltx:para", true)?;
72  document.add_class(&mut para, "ltx_titlepage")?;
73  Ok(())
74}

76/// The marks around the class's own printing `\title` run on a cover (sect05.rs `\lx@cover@title@begin`/`@end`,
77/// `titlepage_class_run`): empty internal elements, admitted anywhere by the model (`_Capture_`-suffixed), never in the
78/// output (removed here; a stray one the core finalize drops).
79pub const COVER_TITLE_BEGIN: &str = "ltx:_CoverTitleBegin_Capture_";
80/// See [`COVER_TITLE_BEGIN`].
81pub const COVER_TITLE_END: &str = "ltx:_CoverTitleEnd_Capture_";

83/// The class's own printing `\title` run on a cover inside `scope` (sect05.rs `titlepage_class_run`): everything that
84/// lies between each pair of cover-title marks in document order — elements and bare text (`\def\title#1{#1}` in the
85/// cover's open paragraph) — removed with the marks, and the wrappers it leaves empty (a `center`'s paragraph). Called
86/// where a `{titlepage}` stops standing in for the document's `<ltx:title>`, so the HTML displays that title
87/// (LaTeXML-structure-xhtml.xsl:374-375 hides it only beside an `ltx:titlepage`): `\unwind@titlepage` (a `\maketitle`
88/// inside the cover), [`demote_stranded_titlepage`], and [`settle_cover_title_ink`]. An unmatched mark is removed
89/// alone. OXIDIZED_DESIGN #629.
90pub fn drop_cover_title_ink(document: &mut Document, scope: &Node) {
91  let begin_q = pin(COVER_TITLE_BEGIN);
92  let marks = cover_title_marks(document, scope);
93  if marks.is_empty() {
94    return;
95  }
96  let mut doomed: Vec<Node> = Vec::new();
97  let mut chains: Vec<Vec<Node>> = Vec::new();
98  let mut open: Option<Node> = None;
99  for mark in marks {
100    chains.push(ancestors_below(&mark, scope));
101    if document::get_node_qname(&mark) == begin_q {
102      if let Some(stray) = open.replace(mark) {
103        doomed.push(stray);
104      }
105    } else if let Some(begin) = open.take() {
106      doomed.extend(marked_range(scope, &begin, &mark));
107      doomed.push(begin);
108      doomed.push(mark);
109    } else {
110      doomed.push(mark);
111    }
112  }
113  doomed.extend(open);
114  remove_marked_ink(document, doomed, chains, |_| true);
115}

117/// Remove the `doomed` nodes (marks and the ink between them), then the marks' ancestors (`chains`) the removal leaves
118/// with no content (a `center`'s paragraph holding only the print), innermost first, until none more empties — each a
119/// wrapper `prunable` admits. Everything is decided before anything is removed: `remove_node` frees the subtree it
120/// takes out (`discard_subtree`), so no handle into it may be used afterwards.
121fn remove_marked_ink(
122  document: &mut Document,
123  doomed: Vec<Node>,
124  chains: Vec<Vec<Node>>,
125  prunable: impl Fn(&Node) -> bool,
126) {
127  // the nodes that go, by identity (`to_hashable`: a `Node` is no hash key, its handle is mutable)
128  let mut gone: HashSet<usize> = doomed.iter().map(Node::to_hashable).collect();
129  // is `node` or one of its ancestors going?
130  let goes_with = |node: &Node, gone: &HashSet<usize>| {
131    let mut n = Some(node.clone());
132    while let Some(cur) = n {
133      if gone.contains(&cur.to_hashable()) {
134        return true;
135      }
136      n = cur.get_parent();
137    }
138    false
139  };
140  // an ancestor of a mark inside another range goes with that range
141  let mut wrappers: Vec<Node> = Vec::new();
142  for node in chains.into_iter().flatten() {
143    if !goes_with(&node, &gone) && prunable(&node) && !wrappers.contains(&node) {
144      wrappers.push(node);
145    }
146  }
147  loop {
148    let emptied: Vec<Node> = wrappers
149      .iter()
150      .filter(|w| !gone.contains(&w.to_hashable()))
151      .filter(|w| {
152        w.get_child_nodes().iter().all(|c| {
153          gone.contains(&c.to_hashable())
154            || (c.get_type() != Some(NodeType::ElementNode) && c.get_content().trim().is_empty())
155        })
156      })
157      .cloned()
158      .collect();
159    if emptied.is_empty() {
160      break;
161    }
162    gone.extend(emptied.iter().map(Node::to_hashable));
163  }
164  // each removed whole, once: the outermost of the nodes that go (no ancestor of it goes), in a stable order
165  let mut victims: Vec<Node> = Vec::new();
166  for n in doomed.iter().chain(wrappers.iter()) {
167    if gone.contains(&n.to_hashable())
168      && !n.get_parent().is_some_and(|p| goes_with(&p, &gone))
169      && !victims.contains(n)
170    {
171      victims.push(n.clone());
172    }
173  }
174  for victim in victims {
175    document.remove_node(victim);
176  }
177}

179/// The cover-title marks inside `scope`, in document order.
180fn cover_title_marks(document: &mut Document, scope: &Node) -> Vec<Node> {
181  // (none made, none to find: `Document::made_mark`)
182  if !(document.made_mark(COVER_TITLE_BEGIN) || document.made_mark(COVER_TITLE_END)) {
183    return Vec::new();
184  }
185  document.findnodes(
186    "descendant::ltx:_CoverTitleBegin_Capture_ | descendant::ltx:_CoverTitleEnd_Capture_",
187    Some(scope),
188  )
189}

191/// `node`'s ancestors strictly below `scope`, innermost first.
192fn ancestors_below(node: &Node, scope: &Node) -> Vec<Node> {
193  let mut out = Vec::new();
194  let mut p = node.get_parent();
195  while let Some(n) = p {
196    if &n == scope {
197      break;
198    }
199    p = n.get_parent();
200    out.push(n);
201  }
202  out
203}

205/// The nodes after `begin` and before `end` in document order, each taken whole: the following siblings of `begin`
206/// and of each of its ancestors up to the one on the path to `end`, then the children before that path down to `end`.
207fn marked_range(scope: &Node, begin: &Node, end: &Node) -> Vec<Node> {
208  let mut end_path = ancestors_below(end, scope);
209  end_path.push(end.clone());
210  let mut range: Vec<Node> = Vec::new();
211  let mut cur = begin.clone();
212  let mut fork: Option<Node> = None;
213  'up: loop {
214    let mut sib = cur.get_next_sibling();
215    while let Some(s) = sib {
216      if end_path.contains(&s) {
217        fork = Some(s);
218        break 'up;
219      }
220      sib = s.get_next_sibling();
221      range.push(s);
222    }
223    match cur.get_parent() {
224      Some(p) if &p != scope => cur = p,
225      _ => break,
226    }
227  }
228  while let Some(node) = fork.take() {
229    if &node == end {
230      break;
231    }
232    for child in node.get_child_nodes() {
233      if end_path.contains(&child) {
234        fork = Some(child);
235        break;
236      }
237      range.push(child);
238    }
239  }
240  range
241}

243/// At `\end{document}`, the cover-title marks still in the tree (OXIDIZED_DESIGN #629), by the XSLT's own test
244/// (LaTeXML-structure-xhtml.xsl:374-375): while the document has a top-level `ltx:titlepage`, its `<ltx:title>` is not
245/// displayed and the class's print is the title the page shows — kept, its marks removed, wherever the cover's
246/// typesetting put it (a `\vbox` print the absorber set beside an emptied cover: hep-th0010274); without one (a
247/// titlepage nested in a section or a box, every cover unwound or demoted) the title is displayed and the print goes
248/// ([`drop_cover_title_ink`]).
249pub fn settle_cover_title_ink(document: &mut Document) {
250  // (no cover-title mark made: none to settle, either way below)
251  if !(document.made_mark(COVER_TITLE_BEGIN) || document.made_mark(COVER_TITLE_END)) {
252    return;
253  }
254  let Some(root) = document.get_document().get_root_element() else {
255    return;
256  };
257  // (a titlepage continuing the title block stands for no title: OXIDIZED_DESIGN_DIVERGENCES #903)
258  if document
259    .findnode(
260      &s!("/ltx:document/ltx:titlepage[not(contains(concat(' ', @class, ' '), ' {CONTINUED_TITLEPAGE} '))]"),
261      None,
262    )
263    .is_some()
264  {
265    for mark in cover_title_marks(document, &root) {
266      document.remove_node(mark);
267    }
268  } else {
269    drop_cover_title_ink(document, &root);
270  }
271}

273/// The field runs' marks (core `Document::absorb`, [`document::FIELD_MARK_TARGET`]): processing instructions
274/// `<?ltx-field-mark begin <field> <id>?>` … `<?ltx-field-mark end <field> <id> <min> <max>?>` around a run of ink
275/// digested from a frontmatter field's stored tokens (their out-of-band provenance, `\lx@field@mark`). Not elements: a
276/// processing instruction is no node of content (`content_nodes`, text content), so no block is built around it.
277/// Never in the output: removed by [`lift_marked_field_ink`] and [`settle_field_marks`], and by the core finalize.
278pub const FIELD_MARK_TARGET: &str = document::FIELD_MARK_TARGET;
279/// The fields the kernel setters stamp, by their provenance id (`Token::prov`'s field).
280const FIELD_MARK_FIELDS: &[(&str, u16)] = &[("title", 1), ("author", 2)];
281
282/// The provenance id of a stamped field (`title`, `author`).
283pub(in crate::base_utilities) fn field_id(field: &str) -> Option<u16> {
284  FIELD_MARK_FIELDS
285    .iter()
286    .find(|(name, _)| *name == field)
287    .map(|(_, id)| *id)
288}

290/// One run of a field's ink, from its two marks ([`FIELD_MARK_TARGET`]).
291pub(in crate::base_utilities) struct FieldInkRun {
292  field:       u16,
293  begin:       Node,
294  end:         Node,
295  /// the field-ledger serials of the boxes it absorbed, inclusive intervals (core `token::field_ledger_cover`), and
296  /// of those an alignment took between its parts (a joined run's cells' `\\`)
297  serials:     Vec<(u32, u32)>,
298  /// the positions of the current call's value (`lx_field_range_<id>`) the run consumed: sorted disjoint
299  /// `start..end` intervals
300  cover:       Vec<(u16, u16)>,
301  /// does the run consume positions of the current call's value only? A print of an earlier call's value (`\title{A}`
302  /// printed, then `\title{B}`; an `\edef` copy) is joined with no other run (re-review finding 2, probes r12, r13).
303  in_range:    bool,
304  /// is the run preceded, in the document's content order, by body text ([`field_runs_after_body_text`])? Then it
305  /// is not in the document's title block.
306  body_before: bool,
307  /// does the run hold a box of one character of the value twice? A print digests each character once, so it read
308  /// the value again from an earlier position the way a macro's expansion does (all its boxes at the macro's
309  /// position): `\StrLeft{\@title}{1}\@title`'s two `T`s. No whole print.
310  reread:      bool,
311}

315    self

317      .iter()
318      .map(|&(first, _)| first)
319      .min()
320      .unwrap_or(0)
321  }

323    self
324      .serials
325      .iter()

328      .unwrap_or(0)
329  }

332/// The `start:end` intervals of a field's extent (`lx_field_extent_<id>`), a call's range (`lx_field_range_<id>`) or a
333/// run's serials (its end mark).
334fn parse_intervals<T: FromStr>(text: &str) -> Vec<(T, T)> {
335  text
336    .split(',')
337    .filter_map(|iv| {
338      let (start, end) = iv.split_once(':')?;
339      Some((start.trim().parse().ok()?, end.trim().parse().ok()?))
340    })
341    .collect()
342}

344/// The positions of field `field`'s stored value a whole print covers (`\lx@field@mark`'s extent).
345fn field_extent(field: u16) -> Vec<(u16, u16)> {
346  parse_intervals(&lookup_string(&s!("lx_field_extent_{field}")))
347}

349/// The positions of the current call's value of field `field` (`\lx@field@mark`'s range): `start..end`.
350fn field_range(field: u16) -> Option<(u16, u16)> {
351  parse_intervals(&lookup_string(&s!("lx_field_range_{field}")))
352    .first()
353    .copied()
354}

356/// `cover` within `range` (sorted disjoint intervals stay so).
357fn clip_cover(cover: &[(u16, u16)], range: Option<(u16, u16)>) -> Vec<(u16, u16)> {
358  let Some((start, end)) = range else {
359    return Vec::new();
360  };
361  cover
362    .iter()
363    .filter_map(|&(s, e)| {
364      let (s, e) = (s.max(start), e.min(end));
365      (s < e).then_some((s, e))
366    })
367    .collect()
368}

370/// Does `cover` cover every position of `extent`, so the run is a whole print of the field (review finding 2: every
371/// position, a macro's too, not the first and the last character)? Both are sorted disjoint intervals, the cover's
372/// merged where they touch, so each extent interval lies in one cover interval.
373fn covers_extent(cover: &[(u16, u16)], extent: &[(u16, u16)]) -> bool {
374  !extent.is_empty()
375    && extent
376      .iter()
377      .all(|&(s, e)| cover.iter().any(|&(cs, ce)| cs <= s && e <= ce))
378}

380/// Is `node` inside invisible ink (`\phantom`, `\vphantom`, `\hphantom` in text: `<ltx:text class="ltx_phantom">`)?
381fn in_phantom(node: &Node) -> bool {
382  let mut p = node.get_parent();
383  while let Some(n) = p {
384    if n
385      .get_attribute("class")
386      .is_some_and(|c| c.split_whitespace().any(|c| c == "ltx_phantom"))
387    {
388      return true;
389    }
390    p = n.get_parent();
391  }
392  false
393}

395/// Do the boxes with serials `serials` (inclusive intervals) hold a box of one character of field `field`'s value
396/// (`lx_field_chars_<id>`) twice?
397fn rereads_a_char(field: u16, serials: &[(u32, u32)]) -> bool {
398  let chars: Vec<(u16, u16)> = parse_intervals(&lookup_string(&s!("lx_field_chars_{field}")));
399  let mut seen: HashSet<u16> = HashSet::default();
400  for &(first, last) in serials {
401    for serial in first..=last {
402      if let Some(entry) = field_ledger_entry(serial)
403        && entry.cover.field == field
404        && entry.cover.end == entry.cover.start + 1
405        && chars
406          .iter()
407          .any(|&(s, e)| s <= entry.cover.start && entry.cover.start < e)
408        && !seen.insert(entry.cover.start)
409      {
410        return true;
411      }
412    }
413  }
414  false
415}

417/// The serials of the field-ledger entries of field `field` an alignment took with serials `from..to` (exclusive), as
418/// inclusive intervals (core `token::field_ledger_take`).
419fn taken_serials(field: u16, from: u32, to: u32) -> Vec<(u32, u32)> {
420  (from..to)
421    .filter(|&serial| field_ledger_entry(serial).is_some_and(|e| e.taken && e.cover.field == field))
422    .map(|serial| (serial, serial))
423    .collect()
424}

426/// The field runs inside `scope`, each with its own two marks (paired by the run's id, so never with another print's),
427/// in document order of their begin marks; and every mark node. A run's cover is what the boxes it absorbed consumed
428/// of the current call's value (and what an alignment took between and right after them): never what a box absorbed
429/// elsewhere (in another run, in a formula) or never absorbed (an unused `\setbox`/`\sbox`/`{lrbox}`) consumed
430/// (re-review finding 1, probes r19, r20). A run set invisibly (`\phantom`) is no print.
431pub(in crate::base_utilities) fn field_ink_runs(
432  document: &mut Document,
433  scope: &Node,
434) -> (Vec<FieldInkRun>, Vec<Node>) {
435  let marks = field_marks(document, scope);
436  if marks.is_empty() {
437    return (Vec::new(), marks);
438  }
439  let body_before = field_runs_after_body_text(scope);
440  let mut begins: Vec<(u32, u16, usize)> = Vec::new();
441  let mut runs: Vec<(usize, FieldInkRun)> = Vec::new();
442  for (at, mark) in marks.iter().enumerate() {
443    let content = mark.get_content();
444    let parts: Vec<&str> = content.split_whitespace().collect();
445    match parts.as_slice() {
446      ["begin", field, id] => {
447        if let (Ok(field), Ok(id)) = (field.parse(), id.parse()) {
448          begins.push((id, field, at));
449        }
450      },
451      ["end", field, id, serials] => {
452        let (Ok(field), Ok(id)) = (field.parse::<u16>(), id.parse::<u32>()) else {
453          continue;
454        };
455        let serials: Vec<(u32, u32)> = parse_intervals(serials);
456        if let Some(i) = begins
457          .iter()
458          .position(|(bid, bfield, _)| *bid == id && *bfield == field)
459        {
460          let (_, _, begin_at) = begins.swap_remove(i);
461          if in_phantom(&marks[begin_at]) {
462            continue;
463          }
464          let range = field_range(field);
465          let cover = field_ledger_cover(field, &serials);
466          let in_range =
467            range.is_some_and(|(start, end)| cover.iter().all(|&(s, e)| start <= s && e <= end));
468          let reread = rereads_a_char(field, &serials);
469          runs.push((begin_at, FieldInkRun {
470            field,
471            begin: marks[begin_at].clone(),
472            end: mark.clone(),
473            reread,
474            serials,
475            cover: clip_cover(&cover, range),
476            in_range,
477            body_before: body_before.contains(&id),
478          }));
479        }
480      },
481      _ => {},
482    }
483  }
484  runs.sort_by_key(|(at, _)| *at);
485  // One print split by the structure it is set in (a cover `tabular`'s cells, `\and`'s `\end{tabular}…
486  // \begin{tabular}`): a run of a field that goes on with the field's later positions, with no ink between, is one run
487  // with the one before when, together with what the alignment took between them (the cells' `\\`, its row-end
488  // marker never absorbed, taken once the alignment is shown, core `document::field_ink_taken_serials`), they cover one contiguous stretch of the field's extent.
489  // Both must be prints of the current call's value. A second print starts over, and two slices of the value with a
490  // gap between (`\StrLeft…\par\StrRight…`, nothing of the gap consumed) stay two runs, neither whole (review finding
491  // 2).
492  let mut merged: Vec<FieldInkRun> = Vec::new();
493  for (_, run) in runs {
494    if let Some(last) = merged.last_mut()
495      && last.field == run.field
496      && last.in_range
497      && run.in_range
498      && !last.reread
499      && !run.reread
500      && run.min_serial() > last.max_serial()
501      && let (Some(&(_, last_end)), Some(&(run_start, _))) = (last.cover.last(), run.cover.first())
502      && run_start >= last_end
503      && marked_range(scope, &last.end, &run.begin)
504        .iter()
505        .all(is_inkless)
506    {
507      let mut serials = last.serials.clone();
508      serials.extend(taken_serials(
509        run.field,
510        last.max_serial() + 1,
511        run.min_serial(),
512      ));
513      serials.extend(run.serials.iter().copied());
514      let joined = clip_cover(
515        &field_ledger_cover(run.field, &serials),
516        field_range(run.field),
517      );
518      let extent = field_extent(run.field);
519      // every extent position from where the first stopped to where the second starts is consumed between them
520      let contiguous = extent
521        .iter()
522        .flat_map(|&(s, e)| s.max(last_end)..e.min(run_start))
523        .all(|p| joined.iter().any(|&(cs, ce)| cs <= p && p < ce));
524      if contiguous {
525        last.end = run.end;
526        last.serials = serials;
527        last.cover = joined;
528        continue;
529      }
530    }
531    merged.push(run);
532  }
533  // What the alignment took right after a run's last box (a value's closing `\\`, its row-end marker absorbed in no
534  // run, 1605.00182) is the run's too: the ledger's next entries of the field an alignment took, while they continue
535  // its cover (no position of the extent skipped), past entries that cover no position of the extent (the space the
536  // cell dropped before it). Nothing else: an entry absorbed elsewhere or never absorbed is no part of this print.
537  for run in &mut merged {
538    let Some(&(_, mut end)) = run.cover.last() else {
539      continue;
540    };
541    let extent = field_extent(run.field);
542    let limit = field_range(run.field).map_or(end, |(_, e)| e);
543    let mut serial = run.max_serial() + 1;
544    while let Some(next) = field_ledger_entry(serial)
545      && next.cover.field == run.field
546    {
547      let continues =
548        next.cover.start <= end || !extent.iter().any(|&(s, e)| s < next.cover.start && e > end);
549      if next.taken && continues {
550        end = end.max(next.cover.end).min(limit);
551      } else if next.taken
552        || extent
553          .iter()
554          .any(|&(s, e)| s < next.cover.end && next.cover.start < e)
555      {
556        break;
557      }
558      // (an entry of no extent position the alignment dropped, the space before a cell's end, is stepped over: it
559      // adds nothing to the cover)
560      serial += 1;
561    }
562    if let Some(last) = run.cover.last_mut() {
563      last.1 = last.1.max(end);
564    }
565  }
566  (merged, marks)
567}

569/// The ids of the field runs inside `scope` that body text precedes in the document's content order (re-review finding
570/// 3: a letter that ends `\noindent\@author` signs it; pdflatex prints the name there and only there). Body text is
571/// any text of the document before the run's begin mark, the text before it in the run's own block too, except:
572/// - a field's ink (between a run's two marks: a title before its author);
573/// - what is set apart from the page's flow: a frontmatter element ([`FRONTMATTER_ELEMENTS`]), a note, a float, an
574///   error's markup, invisible ink (`\phantom`), the internal `_Capture_` marks;
575/// - the text of a TeX box that does not hold the run: a cover's header box (a preprint-number `\parbox`, 0706.0979,
576///   1605.00182; `\rightline{LA-UR-…}`, `\makebox[\textwidth][r]{…}`, `\mbox{preprint}`, `\colorbox`,
577///   `\fcolorbox`, `\scalebox`) is no body. A box is decided by the TeX mechanism, not by the element it builds
578///   (review r4 finding 3), and marks only the element its own constructor opened, at the point it opens it (review r5
579///   finding 1; core `Document::mark_field_box`'s `_fieldbox`): a constructor declared `restricted_horizontal`
580///   outside math (`\hbox`, so `\rightline`/`\leftline`/`\centerline` and `\colorbox`; `\mbox`, `\makebox`,
581///   `\raisebox`, `\fcolorbox`, `\scalebox`/`\resizebox`/`\rotatebox`/`\reflectbox`), `\framebox`/`\fbox`, the
582///   capture of a `\vbox`, `\vtop`, `\parbox`, minipage, `p` column or framed box ([`insert_block`]), an
583///   environment whose begin code opens a box group, an alignment's container. Material a `\vbox` absorbs straight
584///   into its context marks nothing.
585///
586/// A list environment is no box: text in `{flushright}`, `{center}`, `quote` before the print is body text, and the
587/// print stays (the header cases OXIDIZED_DESIGN #715 lists).
588fn field_runs_after_body_text(scope: &Node) -> HashSet<u32> {
589  /// Is `node` set apart from the page's flow, its text no body of the document?
590  fn set_apart(node: &Node) -> bool {
591    node.get_name().ends_with("_Capture_")
592      || node
593        .get_attribute("class")
594        .is_some_and(|c| c.split_whitespace().any(|c| c == "ltx_phantom"))
595      || with(document::get_node_qname(node), |q| {
596        FRONTMATTER_ELEMENTS.contains(&q)
597          || matches!(
598            q,
599            "ltx:note" | "ltx:figure" | "ltx:table" | "ltx:float" | "ltx:ERROR" | "ltx:indexmark"
600          )
601      })
602  }
603  /// Was `node` opened by a TeX box's constructor (core `Document::mark_field_box`): its text is no body for a run
604  /// outside it?
605  fn is_box(node: &Node) -> bool { node.has_attribute(document::FIELD_BOX_ATTRIBUTE) }
606  #[derive(Default)]
607  struct Walk {
608    /// the ids of the runs open at this point of the walk (their marks seen, the end not yet)
609    open:  Vec<u32>,
610    /// body text seen outside every box
611    body:  bool,
612    /// the boxes around this point, innermost last, each with whether body text was seen in it
613    boxes: Vec<bool>,
614    /// the runs body text precedes
615    after: HashSet<u32>,
616  }
617  fn walk(node: &Node, w: &mut Walk) {
618    for child in node.get_child_nodes() {
619      match child.get_type() {
620        Some(NodeType::ElementNode) => {
621          if set_apart(&child) {
622            continue;
623          }
624          let boxed = is_box(&child);
625          if boxed {
626            w.boxes.push(false);
627          }
628          walk(&child, w);
629          if boxed {
630            w.boxes.pop();
631          }
632        },
633        Some(NodeType::TextNode) => {
634          if w.open.is_empty() && !child.get_content().trim().is_empty() {
635            match w.boxes.last_mut() {
636              Some(seen) => *seen = true,
637              None => w.body = true,
638            }
639          }
640        },
641        Some(NodeType::PiNode) if child.get_name() == FIELD_MARK_TARGET => {
642          let content = child.get_content();
643          match content.split_whitespace().collect::<Vec<_>>().as_slice() {
644            ["begin", _, id, ..] => {
645              if let Ok(id) = id.parse::<u32>() {
646                if w.body || w.boxes.iter().any(|seen| *seen) {
647                  w.after.insert(id);
648                }
649                w.open.push(id);
650              }
651            },
652            ["end", _, id, ..] => {
653              if let Ok(id) = id.parse::<u32>() {
654                w.open.retain(|open| *open != id);
655              }
656            },
657            _ => {},
658          }
659        },
660        _ => {},
661      }
662    }
663  }
664  let mut w = Walk::default();
665  walk(scope, &mut w);
666  w.after
667}

669/// Does `node` hold no ink: whitespace, a mark, a line break, or elements holding only those?

671  match node.get_type() {

675    },
676    Some(NodeType::TextNode) => node.get_content().trim().is_empty(),
677    _ => true,
678  }

694/// (`\begin{center}`, a cover minipage, a font group, `\emph`), and a cover's `tabular` (`\begin{tabular}[t]{c}\@author

705        | "ltx:block"
706        | "ltx:inline-block"

708        | "ltx:text"
709        | "ltx:emph"
710        | "ltx:tabular"
711        | "ltx:thead"
712        | "ltx:tbody"
713        | "ltx:tfoot"
714        | "ltx:tr"
715        | "ltx:td"
716    )
717  })

720/// Does `node` hold content besides the nodes in `gone`, empty internal marks (`_Capture_`) and line breaks? A
721/// processing instruction (a field mark) is none.
722fn holds_content_besides(node: &Node, gone: &[Node]) -> bool {
723  node.get_child_nodes().iter().any(|c| {
724    if gone.contains(c) {
725      return false;
726    }
727    match c.get_type() {
728      Some(NodeType::ElementNode) => {
729        if c.get_name().ends_with("_Capture_") && c.get_first_element_child().is_none() {
730          return false;
731        }
732        // a line's break (`\centerline`'s, a `\\` after the print) ends the line, it holds nothing
733        if document::get_node_qname(c) == pin_static("ltx:break") {
734          return false;
735        }
736        // an element with no children is content of its own (a `<break/>`, a graphic, a rule)
737        c.get_first_child().is_none() || holds_content_besides(c, gone)
738      },
739      Some(NodeType::TextNode) => !c.get_content().trim().is_empty(),
740      _ => false,
741    }
742  })
743}

745/// The printed ink of each field the frontmatter displays, removed — the field's print by construction: the ink
746/// digested from the field's stored tokens (their out-of-band provenance, `\lx@field@mark`; the title-print study's
747/// rule R3, frontmatter model M3 stage S1, OXIDIZED_DESIGN #715), one run of it (`field_ink_runs`) at a time, with the
748/// title-block wrappers it leaves empty (not a `\label` target, which is the author's own). Called once, where the
749/// frontmatter is placed (the `\maketitle` flush `\lx@frontmatterhere`, the fallback at the first sectional unit or
750/// `\end{document}`), so only what was printed before it is the field's ink; every mark goes, and later prints are not
751/// marked (`lx_field_ink_settled`, `Document::settle_field_ink`). A run is lifted only:
752/// - when it is the whole field: what its own boxes consumed of the current call's value covers every position of
753///   the field's extent (`lx_field_extent_<id>`, `covers_extent`), a macro's too, so a fragment the document takes of
754///   it (`\StrLeft{\@title}{5}`, also of `About \nm`; `\@car`, `\@gobble`) stays, also beside the rest of the value
755///   set in a box never shown, in a formula or in a `\phantom` (`field_ink_runs`);
756/// - where the frontmatter displays the field: a document-level `<ltx:title>`, or a document-level author
757///   `<ltx:creator>`, with no `<ltx:titlepage>` beside it (beside one the XSLT shows neither, LaTeXML-structure-xhtml.xsl
758///   :374-375 = Perl :358-359 and :687-692, and the print is the displayed copy: 1304.0681, 1503.03277);
759/// - in the document's title block: no body text before it in the document's content order
760///   (`field_runs_after_body_text`: a letter's closing `\noindent\@author` stays), and `is_field_ink_wrapper` for every ancestor of both marks — never inside a
761///   `{titlepage}`, a float, a note, a frontmatter element or a frame;
762/// - when the fields' prints fill their lines: the paragraph or table cell around each mark holds nothing but lifted
763///   ink, so a mention in prose (`Title: \thetitle`) keeps its words and a title and author set on one line go
764///   together.
765///
766/// Every whole print in the title block is lifted, a title the document prints twice too: it shows once, in the
767/// frontmatter element (user ruling 2026-10-17, "lift every whole print"; OXIDIZED_DESIGN #715).
768///
769/// Returns whether a title print was lifted: the text-matching dedup (`maybe_dedup_leading_title_ink`) then has
770/// nothing left to explain. P2 witnesses: 1605.00182, 2402.14614, 1603.09631 (lrec `\maketitleabstract`), 1503.06457.
771pub fn lift_marked_field_ink(document: &mut Document) -> Result<bool> {
772  if lookup_bool("lx_field_ink_settled") {
773    return Ok(false);
774  }
775  assign_value("lx_field_ink_settled", true, Some(Scope::Global));
776  document.settle_field_ink()?;
777  let Some(root) = document.get_document().get_root_element() else {
778    return Ok(false);
779  };
780  let (runs, marks) = field_ink_runs(document, &root);
781  if marks.is_empty() {
782    return Ok(false);
783  }
784  let no_titlepage = document
785    .findnode("/ltx:document/ltx:titlepage", None)
786    .is_none();
787  let title_shown = no_titlepage && document.findnode("/ltx:document/ltx:title", None).is_some();
788  let creators_shown = no_titlepage
789    && document
790      .findnode("/ltx:document/ltx:creator[@role='author']", None)
791      .is_some();
792  // a framed wrapper (`\fbox{\@title}`, a framed cover minipage) is the author's own ink around the print: emptied, it
793  // would leave a frame the PDF does not have (as the dedup, 68s review finding 4)
794  let in_title_block = |mark: &Node| {
795    ancestors_below(mark, &root)
796      .iter()
797      .all(|a| is_field_ink_wrapper(document::get_node_qname(a)) && !a.has_attribute("framed"))
798  };
799  // the line a mark sits in: its nearest paragraph or table cell
800  let line_of = |mark: &Node| {
801    ancestors_below(mark, &root).into_iter().find(|a| {
802      let q = document::get_node_qname(a);
803      q == pin_static("ltx:p") || q == pin_static("ltx:td")
804    })
805  };
806  // the whole prints of a displayed field in the title block, with their ink and their lines
807  let mut candidates: Vec<(&FieldInkRun, Vec<Node>, Vec<Node>)> = Vec::new();
808  for run in &runs {
809    let Some((name, _)) = FIELD_MARK_FIELDS.iter().find(|(_, id)| *id == run.field) else {
810      continue;
811    };
812    let shown = if *name == "title" {
813      title_shown
814    } else {
815      creators_shown
816    };
817    let whole = !run.reread && covers_extent(&run.cover, &field_extent(run.field));
818    if !shown
819      || !whole
820      || run.body_before
821      || !in_title_block(&run.begin)
822      || !in_title_block(&run.end)
823    {
824      continue;
825    }
826    let range = marked_range(&root, &run.begin, &run.end);
827    if range.is_empty() {
828      continue;
829    }
830    let lines: Vec<Node> = [&run.begin, &run.end]
831      .into_iter()
832      .filter_map(line_of)
833      .collect();
834    candidates.push((run, range, lines));
835  }
836  // the prints that fill their lines with lifted ink alone, all fields together (finding 5: a title and its author on
837  // one line), until no more drop out
838  loop {
839    let gone: Vec<Node> = candidates
840      .iter()
841      .flat_map(|(_, range, _)| range.iter())
842      .chain(marks.iter())
843      .cloned()
844      .collect();
845    let before = candidates.len();
846    candidates.retain(|(_, _, lines)| !lines.iter().any(|line| holds_content_besides(line, &gone)));
847    if candidates.len() == before {
848      break;
849    }
850  }
851  let mut doomed: Vec<Node> = marks.clone();
852  let mut chains: Vec<Vec<Node>> = Vec::new();
853  let mut lifted_title = false;
854  for (run, mut range, lines) in candidates {
855    let name = FIELD_MARK_FIELDS
856      .iter()
857      .find(|(_, id)| *id == run.field)
858      .map_or("", |(name, _)| *name);
859    // the lines' breaks go with the print they end
860    for line in &lines {
861      range.extend(document.findnodes("descendant::ltx:break", Some(line)));
862    }
863    let text = normalize_frontmatter_text(
864      &range
865        .iter()
866        .filter(|n| n.get_type() != Some(NodeType::PiNode))
867        .map(|n| n.get_content())
868        .collect::<Vec<_>>()
869        .join(" "),
870    );
871    Info!(
872      "frontmatter",
873      "field_ink",
874      s!("lifted the printed \\@{name} (\"{text}\"): the frontmatter displays it")
875    );
876    lifted_title |= name == "title";
877    chains.push(ancestors_below(&run.begin, &root));
878    chains.push(ancestors_below(&run.end, &root));
879    doomed.extend(range);
880  }
881  remove_marked_ink(document, doomed, chains, |w| {
882    is_field_ink_wrapper(document::get_node_qname(w)) && !w.has_attribute("labels")
883  });
884  Ok(lifted_title)
885}
