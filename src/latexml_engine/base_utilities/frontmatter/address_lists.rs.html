1//! A list of e-mail addresses one contact prints for several people, parted into each address's e-mail on its owner

6/// One address of a contact's list ([`listed_addresses`]): the address, the name printed beside it (`a@x (Ann Able)`,
7/// `Ann Able (a@x)`), if any, the bytes of the contact's text it takes with that name and its brackets, and those it
8/// takes alone.
9struct ListedAddress {
10  address: String,
11  name:    Option<String>,
12  span:    std::ops::Range<usize>,
13  bare:    std::ops::Range<usize>,
14}
15
16/// FIDELITY_CHARTER §4, ownership by evidence: a contact printing a list of addresses for several people — a shared
17/// corresponding-author note copied whole to each of its carriers (elsarticle's `\cortext[cor1]{Corresponding
18/// authors:\\ zjsong@…; Chenxizhang@…}`, 2302.03296; REVTeX's `\thanks{Corresponding Authors (chunfeng.lian@…;
19/// wzy218@…; cong6406@…; jhma@…)}` on four authors, 2508.07950), or an e-mail call naming the owner beside each address
20/// (`\email[Corresponding author]{: xiazh@… (Zhenhua Xia); lianghongstefanie@… (Hong Liang)}` on Hong Liang's call,
21/// 2009.14655; `\cortext{… Abrar Faiyaz (abrar\_faiyaz@…) and Md Nasir Uddin (nasir\_uddin@…)}`, 2507.07920) — gives each
22/// address as an e-mail to its owner, by the placement order: the author whose name is printed beside it, the strongest
23/// evidence; else the author it spells; else, among the contact's carriers, the consistent order or the elimination
24/// (`owners::list_owners`); else the block's own creator (OD #159), once. The names printed beside the addresses
25/// are no e-mail's text. The contact's other nodes stay its carriers', as printed — only the address spans and the
26/// separators beside them are cut out of it, in place ([`cut_ranges`], [`cut_printed`]): its words (`Corresponding
27/// authors:`; an "and"/"or" outside the list), links, breaks and math (2311.15476's DOI link, review rev107f B2); a
28/// contact left with no word is dropped. An e-mail keeps the call's name (`Corresponding author`, which `write_role_notes` then makes each
29/// owner's correspondence note); a note's addresses take the e-mail's own. Read once every route has placed the
30/// contacts, before the correspondence labels (census138 ADDR-b, brief 107f). A contact one creator holds is parted only
31/// where every address has an author's name beside it; an e-mail whose list prints other words, or a list none of whose
32/// addresses finds an author, is left as it is.
33pub(super) fn distribute_address_lists(document: &mut Document) -> Result<()> {
34  let authors = document.findnodes("//ltx:creator[@role='author'][ltx:personname]", None);
35  if authors.len() < 2 {
36    return Ok(());
37  }
38  let names: Vec<String> = authors
39    .iter()
40    .map(|author| {
41      document
42        .findnode("ltx:personname", Some(author))
43        .map(|person| person.get_content())
44        .unwrap_or_default()
45    })
46    .collect();
47  // (a name not yet written leaves the evidence incomplete: the lists stay as printed)
48  if names.iter().any(|name| name.trim().is_empty()) {
49    return Ok(());
50  }
51  let keys: Vec<NameKeys> = names
52    .iter()
53    .map(|name| NameKeys::of(&Tokens::new(ExplodeText!(name))))
54    .collect();
55  let printed_names: Vec<String> = names.iter().map(|name| name_letters(name)).collect();
56  // (each printed contact and its holders: one role, name and text)
57  type Group = (Option<String>, Option<String>, String, Vec<(usize, Node)>);
58  let mut groups: Vec<Group> = Vec::new();
59  for (at, author) in authors.iter().enumerate() {
60    for contact in document.findnodes(
61      "ltx:contact[@role='email' or @role='correspondent' or @role='thanks']",
62      Some(author),
63    ) {
64      let text = printed_text(&contact);
65      let (role, name) = (contact.get_attribute("role"), contact.get_attribute("name"));
66      match groups
67        .iter_mut()
68        .find(|(r, n, t, _)| *r == role && *n == name && *t == text)
69      {
70        Some((.., holders)) => holders.push((at, contact)),
71        None => groups.push((role, name, text, vec![(at, contact)])),
72      }
73    }
74  }
75  let default_name = get_frontmatter_name(None, "ltx:contact", "email")?;
76  for (role, name, text, holders) in groups {
77    let listed = listed_addresses(&text);
78    if listed.len() < 2 {
79      continue;
80    }
81    let is_email = role.as_deref() == Some("email");
82    let printed_owner = |item: &ListedAddress| {
83      let name = name_letters(item.name.as_deref()?);
84      let mut owners =
85        (0..printed_names.len()).filter(|&at| !name.is_empty() && printed_names[at] == name);
86      match (owners.next(), owners.next()) {
87        (Some(owner), None) => Some(owner),
88        _ => None,
89      }
90    };
91    let mut carriers: Vec<usize> = holders.iter().map(|(at, _)| *at).collect();
92    carriers.dedup();
93    if carriers.len() < 2 && !listed.iter().all(|item| printed_owner(item).is_some()) {
94      continue;
95    }
96    // (a name beside an address is cut with it only where it is an author's: a remark stays, `\{ann,bob\}@a.org
97    // (corresponding)` — perfect_kernel_batch92c)
98    let spans: Vec<std::ops::Range<usize>> = listed
99      .iter()
100      .map(|item| {
101        if printed_owner(item).is_some() {
102          item.span.clone()
103        } else {
104          item.bare.clone()
105        }
106      })
107      .collect();
108    let cuts = cut_ranges(&text, &spans, !is_email);
109    let rest = without_ranges(&text, &cuts);
110    if is_email && rest.chars().any(char::is_alphanumeric) {
111      continue;
112    }
113    // (the author printed beside it; else the one it spells; else, among the carriers, the order or the elimination)
114    let mut owners: Vec<Option<usize>> = listed
115      .iter()
116      .map(|item| printed_owner(item).or_else(|| spelled_owner(&keys, &item.address)))
117      .collect();
118    if owners.iter().any(Option::is_none)
119      && owners
120        .iter()
121        .flatten()
122        .all(|owner| carriers.contains(owner))
123    {
124      let carrier_keys: Vec<NameKeys> = carriers.iter().map(|&at| keys[at].clone()).collect();
125      let addresses: Vec<String> = listed.iter().map(|item| item.address.clone()).collect();
126      let spelled: Vec<Option<usize>> = owners
127        .iter()
128        .map(|owner| owner.and_then(|owner| carriers.iter().position(|&c| c == owner)))
129        .collect();
130      let taken = vec![false; carriers.len()];
131      for (at, (owner, _)) in list_owners(&carrier_keys, &addresses, &spelled, &taken, false, false)
132        .into_iter()
133        .enumerate()
134      {
135        if owners[at].is_none() {
136          owners[at] = owner.map(|owner| carriers[owner]);
137        }
138      }
139    }
140    DebugFeature!(
141      "frontmatter",
142      "FRONT address list of a {role:?} on {:?}: {:?} owners {owners:?}",
143      carriers
144        .iter()
145        .map(|&at| names[at].as_str())
146        .collect::<Vec<_>>(),
147      listed
148        .iter()
149        .map(|item| item.address.as_str())
150        .collect::<Vec<_>>()
151    );
152    // (a list none of whose addresses finds an author names no one: it stays as printed — PoS's `\speaker{Rob
153    // Fender$^{1,2*}$, Patrick Woudt$^2$}` and its `E-mail: \email{rob.fender@…, Patrick.Woudt@…}`, whose names are no
154    // creators yet here, 1711.04132)
155    if owners.iter().all(Option::is_none) {
156      continue;
157    }
158    let email_name = if is_email {
159      name.clone()
160    } else {
161      default_name.clone()
162    };
163    for (item, owner) in listed.iter().zip(&owners) {
164      let target = match owner {
165        Some(owner) => Some(authors[*owner].clone()),
166        None => block_creator(document, &authors)?,
167      };
168      let Some(mut target) = target else {
169        continue;
170      };
171      let held = document
172        .findnodes("ltx:contact[@role='email']", Some(&target))
173        .iter()
174        .any(|contact| contact.get_content().trim() == item.address);
175      if held {
176        continue;
177      }
178      let mut contact = Node::new("contact", None, document.get_document())?;
179      if let Some(namespace) = target.get_namespace() {
180        let _ = contact.set_namespace(&namespace);
181      }
182      contact.set_attribute("role", "email").ok();
183      if let Some(name) = &email_name {
184        contact.set_attribute("name", name).ok();
185      }
186      contact.append_text(&item.address)?;
187      // (after the list it printed in, where its owner holds it; else after the owner's contacts)
188      match holders.iter().find(|(at, _)| Some(*at) == *owner) {
189        Some((_, list)) => {
190          list.clone().add_next_sibling(&mut contact)?;
191        },
192        None => {
193          target.add_child(&mut contact)?;
194        },
195      }
196    }
197    // (the list's other nodes stay its carriers', as printed — its words, links, breaks and math (2311.15476's DOI
198    // link, review rev107f B2); nothing left, the contact goes)
199    for (_, list) in holders {
200      if rest.chars().any(char::is_alphanumeric) {
201        cut_printed(document, &list, &cuts);
202      } else {
203        document.remove_node(list);
204      }
205    }
206  }
207  Ok(())
208}

210/// A contact's text as printed, a line break (`ltx:break`) a line's end.
211fn printed_text(node: &Node) -> String {
212  let mut text = String::new();
213  for child in node.get_child_nodes() {
214    if child.get_type() == Some(NodeType::ElementNode) {
215      if child.get_name() == "break" {
216        text.push('\n');
217      } else {
218        text.push_str(&printed_text(&child));
219      }
220    } else {
221      text.push_str(&child.get_content());
222    }
223  }
224  text
225}

227/// A name's letters, lowercased, by their base letters: what a name printed beside an address is compared by.
228fn name_letters(name: &str) -> String {
229  resolver::base_letters(name)
230    .chars()
231    .filter(|c| c.is_alphabetic())
232    .flat_map(char::to_lowercase)
233    .collect()
234}

236/// The addresses a contact's text prints ([`ListedAddress`]): each plain address, a braced group of local parts
237/// written out (`{a, b}@x`, `resolver::addresses_in`), with the name printed in brackets after it (`a@x (Ann Able)`) or
238/// before its bracketed address (`Ann Able (a@x)`), back to a separator, a line's start or an "and".
239fn listed_addresses(text: &str) -> Vec<ListedAddress> {
240  let mut out: Vec<ListedAddress> = Vec::new();
241  let mut from = 0;
242  for (at, _) in text.match_indices('@') {
243    if at < from {
244      continue;
245    }
246    let domain_len = text[at + 1..]
247      .find(|c: char| !(c.is_ascii_alphanumeric() || ".-".contains(c)))
248      .unwrap_or(text.len() - at - 1);
249    let domain = text[at + 1..at + 1 + domain_len].trim_end_matches(['.', '-']);
250    if !domain.contains('.') {
251      continue;
252    }
253    let end = at + 1 + domain.len();
254    let (start, addresses) = if text[..at].ends_with('}') {
255      let Some(open) = text[from..at].rfind('{').map(|open| from + open) else {
256        continue;
257      };
258      (open, resolver::addresses_in(&text[open..end]))
259    } else {
260      let start = text[from..at]
261        .char_indices()
262        .rev()
263        .take_while(|(_, c)| is_local_char(*c))
264        .last()
265        .map_or(at, |(i, _)| from + i);
266      if start == at {
267        continue;
268      }
269      (start, vec![text[start..end].to_string()])
270    };
271    from = end;
272    if addresses.is_empty() {
273      continue;
274    }
275    let before = text[..start].trim_end();
276    let after = text[end..].trim_start();
277    let (mut span, mut name) = (start..end, None);
278    if before.ends_with('(') && after.starts_with(')') {
279      // (`Ann Able (a@x)`: the name before the bracket, back to a separator, a line's start or an "and")
280      let open = before.len() - 1;
281      let head = &text[..open];
282      let name_start = head
283        .rfind([':', ';', ',', '\n', '(', ')'])
284        .map_or(0, |p| p + 1);
285      let mut words = head[name_start..].trim();
286      let mut lead =
287        name_start + (head[name_start..].len() - head[name_start..].trim_start().len());
288      if let Some(rest) = words.strip_prefix("and ") {
289        lead += words.len() - rest.len();
290        words = rest.trim_start();
291      }
292      span = lead..text.len() - after.len() + 1;
293      name = (!words.is_empty()).then(|| words.to_string());
294    } else if let Some(tag) = after.strip_prefix('(')
295      && let Some(close) = tag.find(')')
296      && !tag[..close].contains('@')
297    {
298      // (`a@x (Ann Able)`)
299      let tag_start = text.len() - after.len();
300      span = start..tag_start + 1 + close + 1;
301      name = Some(tag[..close].trim().to_string());
302    }
303    from = from.max(span.end);
304    for address in addresses {
305      out.push(ListedAddress {
306        address,
307        name: name.clone(),
308        span: span.clone(),
309        bare: start..end,
310      });
311    }
312  }
313  out
314}

316/// The bytes of a contact's printed `text` ([`printed_text`]) its listed addresses take ([`listed_addresses`]), which
317/// [`cut_printed`] cuts out of the contact: their `spans`; between two of them, the separators — spaces, line breaks,
318/// `;` `,` `&` and an "and" or "or", only there; and around each run of them, the space before it, the brackets it
319/// leaves empty (`Corresponding Authors (a; b)` leaves "Corresponding Authors"), a `;` or `,` after it, a stop after it
320/// where a label or nothing opens it (`Corresponding authors:\\ a; b.` leaves "Corresponding authors:"). `strip_label`:
321/// an e-mail label opening the run after other words too, with that stop (`… addressed. Email: a or b.` leaves
322/// `… addressed.`, 2307.11827; a correspondent's own `e-mail: a, b, c` keeps its one word, 2409.06261; an e-mail's own
323/// label stays, so its list stays whole as printed: `Emails: \{ann,bob\}@a.org`, perfect_kernel_batch92c).
324fn cut_ranges(
325  text: &str,
326  spans: &[std::ops::Range<usize>],
327  strip_label: bool,
328) -> Vec<std::ops::Range<usize>> {
329  let mut spans = spans.to_vec();
330  spans.sort_by_key(|span| (span.start, span.end));
331  spans.dedup();
332  // (the runs of addresses only separators part)
333  let separators = |gap: &str| {
334    gap
335      .split(|c: char| c.is_whitespace() || ";,&".contains(c))
336      .all(|word| word.is_empty() || word == "and" || word == "or")
337  };
338  let mut runs: Vec<(usize, usize)> = Vec::new();
339  for span in spans {
340    match runs.last_mut() {
341      Some((_, end)) if span.start < *end => {},
342      Some((_, end)) if separators(&text[*end..span.start]) => *end = span.end,
343      _ => runs.push((span.start, span.end)),
344    }
345  }
346  let space_after = |at: usize| at + (text[at..].len() - text[at..].trim_start().len());
347  let space_before = |at: usize| text[..at].trim_end().len();
348  let mut cuts: Vec<std::ops::Range<usize>> = Vec::new();
349  for (start, end) in runs {
350    let (mut lo, mut hi) = (space_before(start), end);
351    let next = space_after(end);
352    let opener = text[..lo].trim_end();
353    if opener.ends_with('(') && text[next..].starts_with(')') {
354      lo = space_before(lo - 1);
355      hi = next + 1;
356    } else if text[next..].starts_with([';', ','])
357      || (text[next..].starts_with('.')
358        && (opener.ends_with(':') || opener.trim_matches([';', ',', ':']).is_empty()))
359    {
360      hi = next + 1;
361    }
362    if strip_label {
363      let head = text[..lo].trim_end();
364      // (past the last space, whatever its width: a no-break space is two bytes)
365      let word = head
366        .char_indices()
367        .rev()
368        .find(|(_, c)| c.is_whitespace())
369        .map_or(0, |(at, c)| at + c.len_utf8());
370      let label = &head[word..];
371      if label.ends_with(':')
372        && is_email_label(label.trim_end_matches(':'))
373        && text[..word].chars().any(char::is_alphanumeric)
374      {
375        lo = space_before(word);
376      }
377    }
378    // (a run opening the text, a stray `:` or `;` before it aside, takes the space after it)
379    if text[..lo]
380      .trim_matches(|c: char| c.is_whitespace() || ";,:".contains(c))
381      .is_empty()
382    {
383      lo = 0;
384      hi = space_after(hi);
385    }
386    cuts.push(lo..hi);
387  }
388  cuts
389}

391/// `text` without the bytes of `cuts` ([`cut_ranges`]).
392fn without_ranges(text: &str, cuts: &[std::ops::Range<usize>]) -> String {
393  let mut kept = String::new();
394  let mut at = 0;
395  for cut in cuts {
396    kept.push_str(&text[at..cut.start.max(at)]);
397    at = at.max(cut.end);
398  }
399  kept.push_str(&text[at.min(text.len())..]);
400  kept
401}

403/// Cuts the bytes `cuts` of a contact's printed text ([`printed_text`], [`cut_ranges`]) out of the contact, in place:
404/// each text node keeps its other characters, a line break or an element all of whose printed text is cut goes, and
405/// every other node stays as it is — the note's words, links, breaks and math (review rev107f B2: 2311.15476's
406/// `Corresponding authors: … DOI: \href{…}`).
407fn cut_printed(document: &mut Document, contact: &Node, cuts: &[std::ops::Range<usize>]) {
408  // (a snapshot of the printed pieces, each with its bytes of the printed text: the text nodes, the breaks, the
409  // elements around them)
410  enum Piece {
411    Text(Node, std::ops::Range<usize>),
412    Element(Node, std::ops::Range<usize>),
413  }
414  fn walk(node: &Node, at: &mut usize, pieces: &mut Vec<Piece>) {
415    for child in node.get_child_nodes() {
416      if child.get_type() == Some(NodeType::ElementNode) {
417        let start = *at;
418        if child.get_name() == "break" {
419          *at += 1;
420        } else {
421          walk(&child, at, pieces);
422        }
423        pieces.push(Piece::Element(child, start..*at));
424      } else {
425        let len = child.get_content().len();
426        if matches!(
427          child.get_type(),
428          Some(NodeType::TextNode | NodeType::CDataSectionNode)
429        ) {
430          pieces.push(Piece::Text(child, *at..*at + len));
431        }
432        *at += len;
433      }
434    }
435  }
436  let mut pieces: Vec<Piece> = Vec::new();
437  walk(contact, &mut 0, &mut pieces);
438  let covered = |range: &std::ops::Range<usize>| {
439    !range.is_empty()
440      && cuts
441        .iter()
442        .any(|cut| cut.start <= range.start && range.end <= cut.end)
443  };
444  // (the outermost elements wholly cut go whole; the text inside them with them)
445  let mut gone: Vec<std::ops::Range<usize>> = Vec::new();
446  let inside = |gone: &[std::ops::Range<usize>], range: &std::ops::Range<usize>| {
447    gone
448      .iter()
449      .any(|g| g.start <= range.start && range.end <= g.end && !range.is_empty())
450  };
451  // (outermost first: an element's piece follows its children's, so walk the snapshot from the end)
452  for piece in pieces.iter().rev() {
453    if let Piece::Element(node, range) = piece
454      && covered(range)
455      && !inside(&gone, range)
456    {
457      gone.push(range.clone());
458      document.remove_node(node.clone());
459    }
460  }
461  for piece in &pieces {
462    let Piece::Text(node, range) = piece else {
463      continue;
464    };
465    if inside(&gone, range) {
466      continue;
467    }
468    let content = node.get_content();
469    let local: Vec<std::ops::Range<usize>> = cuts
470      .iter()
471      .filter(|cut| cut.start < range.end && range.start < cut.end)
472      .map(|cut| cut.start.max(range.start) - range.start..cut.end.min(range.end) - range.start)
473      .collect();
474    if local.is_empty() {
475      continue;
476    }
477    let kept = without_ranges(&content, &local);
478    let mut node = node.clone();
479    if kept.is_empty() {
480      node.unlink_node();
481    } else {
482      let _ = node.set_content(&kept);
483    }
484  }
