15/// The attribute of a creator's contact that is part of a class's front-matter footnote, which `number_footnote_parts`
16/// numbers: `title`, `own`, `join` (the name's joined footnote), `affil:<n>` or [`GIVEN_PART`].
17pub const FOOTNOTE_PART: &str = "_fnpart";
18/// The [`FOOTNOTE_PART`] of a name's `\footnote` given its number (`\footnote[1]{…}`), which steps no counter.
19pub const GIVEN_PART: &str = "given";
20/// How a class that writes [`FOOTNOTE_PART`]s marks its front-matter footnotes: `fnsymbol` or `alph`.
21pub const FOOTNOTE_MARKS: &str = "lx_fm_footnote_marks";
22
23/// The marks of a class's front-matter footnotes (REVTeX's `\@AF@join`, revtex4-2.cls:2404-2495), numbered as printed:
24/// the title's, then each name's own and its joined one, an affiliation's after its group's names or after every name, a
25/// text printed before keeping its number (`\y@match`). A `\footnote` in a name steps the counter where the name prints
26/// and keeps its contact, unmarked; a `\thanks` there, which REVTeX prints nowhere, takes no number and no mark. A mark
27/// only where both affiliation orders agree; none when a footnote's step is unknown — a footnote contact with no part, a
28/// name's `\footnotemark`, a part outside the creators (OD #1054).
29pub(super) fn number_footnote_parts(document: &mut Document) -> Result<()> {
30  let style = lookup_string(FOOTNOTE_MARKS);
31  if style.is_empty() {
32    return Ok(());
33  }
34  let creators = document.findnodes("//ltx:creator", None);
35  // (each part: its node, its creator, its part, its text — an own part's as printed, a joined one's with its role)
36  let mut parts: Vec<(Node, usize, String, String)> = Vec::new();
37  // (a part a route placed outside the creators — a title's or a pubnote's — is a step not counted here)
38  let mut unknown = !document
39    .findnodes(
40      &s!(
41        "//*[@{FOOTNOTE_PART} and @{FOOTNOTE_PART}!='{GIVEN_PART}' and not(ancestor::ltx:creator)]"
42      ),
43      None,
44    )
45    .is_empty();
46  for (at, creator) in creators.iter().enumerate() {
47    let entries = document.findnodes(
48      "ltx:contact | ltx:note[@role='footnotemark']",
49      Some(creator),
50    );
51    for entry in entries {
52      let role = entry.get_attribute("role").unwrap_or_default();
53      let text = entry
54        .get_content()
55        .split_whitespace()
56        .collect::<Vec<_>>()
57        .join(" ");
58      match (entry.get_attribute(FOOTNOTE_PART), role.as_str()) {
59        // (a given number, `\footnote[1]{…}`, steps nothing: 1607.07280)
60        (Some(part), _) if part == GIVEN_PART => {},
61        (Some(part), _) if part == "own" => parts.push((entry, at, part, text)),
62        (Some(part), _) => {
63          let name = entry.get_attribute("name").unwrap_or_default();
64          parts.push((entry, at, part, s!("{role}\u{2}{name}\u{2}{text}")));
65        },
66        // (a name's `\footnote`, its own footnote where the name prints, never shared: "Ann Able∗ / Bob Baker† Cy Cole†"
67        // in REVTeX 4.0, "Ann Ablea" in 4.2's minipage letters)
68        (None, "note") => {
69          let own = s!("\u{3}{}", parts.len());
70          parts.push((entry, at, s!("own"), own));
71        },
72        // (a name's `\footnotemark` steps the counter unless given a number, which its note no longer tells)
73        (None, "footnotemark") => unknown = true,
74        (None, _) => {
75          unknown |= matches!(role.as_str(), "thanks" | "email" | "url" | "altaffiliation")
76        },
77      }
78    }
79  }
80  for (entry, ..) in &mut parts {
81    entry.remove_attribute(FOOTNOTE_PART).ok();
82  }
83  if unknown || parts.is_empty() {
84    return Ok(());
85  }
86  // (each part's footnote: its own text, or its parts' joined in order — the creator's, the title's, the affiliation's)
87  let group = |at: usize, part: &str| match part {
88    "title" => s!("title"),
89    _ if part.starts_with("affil:") => part.to_string(),
90    _ => s!("join:{at}"),
91  };
92  let mut joined: HashMap<String, String> = HashMap::default();
93  for (_, at, part, text) in parts.iter().filter(|(_, _, part, _)| part != "own") {
94    let footnote = joined.entry(group(*at, part)).or_default();
95    footnote.push_str(text);
96    footnote.push('\u{1}');
97  }
98  let footnotes: Vec<String> = parts
99    .iter()
100    .map(|(_, at, part, text)| match part.as_str() {
101      "own" => text.clone(),
102      _ => joined[&group(*at, part)].clone(),
103    })
104    .collect();
105  // (the printed order: the title's, each name's own then its joined one, an affiliation's after its group's names —
106  // its footnote on the group's last name, where `\email` after `\affiliation` puts it — or after every name)
107  let numbers = |affiliations_last: bool| -> HashMap<&String, usize> {
108    let mut order: Vec<&String> = Vec::new();
109    fn add<'a>(footnote: &'a String, order: &mut Vec<&'a String>) {
110      if !order.contains(&footnote) {
111        order.push(footnote);
112      }
113    }
114    let of = |which: &dyn Fn(usize, &str) -> bool| -> Vec<&String> {
115      parts
116        .iter()
117        .zip(&footnotes)
118        .filter(|((_, at, part, _), _)| which(*at, part))
119        .map(|(_, footnote)| footnote)
120        .collect()
121    };
122    for footnote in of(&|_, part| part == "title") {
123      add(footnote, &mut order);
124    }
125    let mut affiliations: Vec<&String> = Vec::new();
126    for holder in 0..creators.len() {
127      for footnote in of(&|at, part| at == holder && part == "own") {
128        add(footnote, &mut order);
129      }
130      for footnote in of(&|at, part| at == holder && part == "join") {
131        add(footnote, &mut order);
132      }
133      for footnote in of(&|at, part| at == holder && part.starts_with("affil:")) {
134        if affiliations_last {
135          add(footnote, &mut affiliations);
136        } else {
137          add(footnote, &mut order);
138        }
139      }
140    }
141    for footnote in affiliations {
142      add(footnote, &mut order);
143    }
144    order
145      .into_iter()
146      .enumerate()
147      .map(|(n, footnote)| (footnote, n + 1))
148      .collect()
149  };
150  let (grouped, last) = (numbers(false), numbers(true));
151  // (a document's or package's own `\@fnsymbol` prints glyphs not this table's, 2603.18989's †: none written, the mark
152  // unknown, `insert::count_unknown_marks`)
153  let own_fnsymbol = style != "alph"
154    && lookup_definition(&T_CS!("\\@fnsymbol"))
155      .ok()
156      .flatten()
157      .is_some_and(|d| !d.get_origin().is_latexml_owned());
158  for ((entry, ..), footnote) in parts.iter().zip(&footnotes) {
159    let mut entry = entry.clone();
160    let n = grouped[footnote];
161    if entry.get_attribute("role").as_deref() != Some("thanks") || n != last[footnote] {
162      continue;
163    }
164    let mark = match style.as_str() {
165      "alph" => u8::try_from(n)
166        .ok()
167        .filter(|n| (1..=26).contains(n))
168        .map(|n| char::from(b'a' + n - 1).to_string()),
169      _ => ["∗", "†", "‡", "§", "¶", "‖", "∗∗", "††", "‡‡"]
170        .get(n - 1)
171        .map(|mark| (*mark).to_string()),
172    };
173    if own_fnsymbol {
174      if entry.get_name() == "note" {
175        entry.set_attribute(MARK_UNKNOWN, "true")?;
176      }
177      continue;
178    }
179    // (a thanks note its mark, a contact the mark the Kind step reads)
180    if let Some(mark) = mark {
181      if entry.get_name() == "note" {
182        entry.set_attribute("mark", &mark)?;
183        entry.set_attribute(MARK_FROM_COUNTER, "true")?;
184      } else {
185        entry.set_attribute(PRINTED_MARK, &mark)?;
186      }
187    }
188  }
189  Ok(())
190}

192/// The Kind step for a class's footnote text that two or more creators carry under one mark ([`PRINTED_MARK`]): an
193/// elsarticle `\fntext[eq]{…}` its `\fnref{eq}` gives two names (each copy a contact `relocate_annotations` placed), a
194/// REVTeX `\thanks{…}` two names give alike — printed once, its mark on each name (elsarticle.cls:157-171, revtex4-2.cls:2472-2495
195/// `\frontmatter@footnote`): each copy the carriers' note with that `mark` (FIDELITY_CHARTER §4, the mark rule). One a
196/// single creator carries stays its contact, shown under its name.
197pub(super) fn kind_shared_marked_contacts(document: &mut Document) -> Result<()> {
198  let contacts = document.findnodes(&s!("//ltx:creator/ltx:contact[@{PRINTED_MARK}]"), None);
199  if contacts.is_empty() {
200    return Ok(());
201  }
202  // (each copy's mark and text; a copy is shared when two creators or more carry its mark and text)
203  let read: Vec<(String, String)> = contacts
204    .iter()
205    .map(|contact| {
206      (
207        contact.get_attribute(PRINTED_MARK).unwrap_or_default(),
208        contact
209          .get_content()
210          .split_whitespace()
211          .collect::<Vec<_>>()
212          .join(" "),
213      )
214    })
215    .collect();
216  let mut carriers: HashMap<&(String, String), Vec<Node>> = HashMap::default();
217  for (contact, copy) in contacts.iter().zip(&read) {
218    let held = carriers.entry(copy).or_default();
219    if let Some(creator) = contact.get_parent()
220      && !held.contains(&creator)
221    {
222      held.push(creator);
223    }
224  }
225  let shared: Vec<bool> = read
226    .iter()
227    .map(|copy| !copy.0.is_empty() && carriers[copy].len() >= 2)
228    .collect();
229  for (at, mut contact) in contacts.into_iter().enumerate() {
230    contact.remove_attribute(PRINTED_MARK).ok();
231    if !shared[at] {
232      continue;
233    }
234    let kind = classify_thanks(&read[at].1);
235    let mut note = document.rename_node(contact, "ltx:note", true)?;
236    note.remove_attribute("name").ok();
237    document.set_attribute(&mut note, "role", "thanks")?;
238    document.set_attribute(
239      &mut note,
240      "class",
241      &s!("ltx_note_frontmatter ltx_thanks_{kind}"),
242    )?;
243    document.set_attribute(&mut note, "mark", &read[at].0)?;
244    note.set_attribute(MARK_FROM_COUNTER, "true")?;
245  }
246  Ok(())
247}

249/// The superscript (or the formula of one) the text of `node` opens with, past the paragraphs and the font wrappers that
250/// open it, and what it prints: `<p align="center"><sup>‡</sup> These authors …</p>`.
251fn leading_mark(node: &Node) -> Option<(Node, String)> {
252  for child in node.get_child_nodes() {
253    match child.get_type() {
254      Some(NodeType::TextNode) if child.get_content().trim().is_empty() => {},
255      Some(NodeType::ElementNode) => {
256        return match child.get_name().as_str() {
257          "sup" => Some((child.clone(), child.get_content().trim().to_string())),
258          "Math" => match fm_read_formula(&child) {
259            FmFormulaMarks::Marks(scripts) if scripts.len() == 1 => {
260              Some((child.clone(), scripts[0].trim().to_string()))
261            },
262            // (a footnote symbol set in math, not raised: cnf.sty's `$\ast$~…`)
263            _ => Some(child.get_content().trim().to_string())
264              .filter(|text| !text.is_empty() && text.chars().all(events::is_fnsymbol))
265              .map(|text| (child.clone(), text)),
266          },
267          "p" | "para" | "text" | "inline-block" | "inline-para" => leading_mark(&child),
268          _ => None,
269        };
270      },
271      _ => return None,
272    }
273  }
274  None
275}

277/// A title-page list's note its leading mark gave to the names showing it ([`LIST_MARK`], `\lx@fm@list@note`: 2606.06621's
278/// `$^\ddagger$ These authors contributed equally` under `Windt$^\ddagger$`) shows that mark as its `mark`: the mark goes
279/// from its text, with the space after it (the mark rule, FIDELITY_CHARTER §4), and from its holder's name, as a name's
280/// mark a legend answers does (`settle_name_marks`). A name's superscript of other marks too (`$^{1,\ddagger}$`)
281/// stays.
282pub(super) fn show_list_note_marks(document: &mut Document) -> Result<()> {
283  let key = |mark: &str| mark.trim().replace('∗', "*");
284  for mut note in document.findnodes(&s!("//ltx:creator/ltx:note[@{LIST_MARK}]"), None) {
285    let mark = note.get_attribute(LIST_MARK).unwrap_or_default();
286    note.remove_attribute(LIST_MARK).ok();
287    if let Some((lead, printed)) = leading_mark(&note)
288      && key(&printed) == key(&mark)
289    {
290      // (the space after it: a text node, or the first of the font wrapper after it)
291      let next = lead
292        .get_next_sibling()
293        .and_then(|next| match next.get_type() {
294          Some(NodeType::TextNode) => Some(next),
295          _ if next.get_name() == "text" => next
296            .get_first_child()
297            .filter(|n| n.get_type() == Some(NodeType::TextNode)),
298          _ => None,
299        });
300      document.remove_node(lead);
301      if let Some(mut text) = next {
302        let content = text.get_content();
303        text.set_content(content.trim_start())?;
304      }
305    }
306    let Some(creator) = note.get_parent() else {
307      continue;
308    };
309    for sup in document.findnodes("ltx:personname//ltx:sup", Some(&creator)) {
310      if key(&sup.get_content()) == key(&mark) {
311        document.remove_node(sup);
312      }
313    }
314  }
315  Ok(())
