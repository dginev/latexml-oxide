1//! Deferred label references: `\lx@label@ref{<key>}`, the label a binding knows only once later input is read (cas's
2//! `\stmRef`, which LaTeX reads from the `.aux` of the run before), filled by the binding into one table
3//! ([`set_label_ref`]) and read where the frontmatter is flushed. The flush's readers of the queued keyvals read the
4//! label it stands for ([`resolve_label_refs`]), as the digestion's expansion of the macro does. Witness 2303.00352
5//! (cas's `\author[aff1]` before its `\affiliation[aff1]`).
6
7use super::*;
8
9/// The macro a binding writes for a label recorded later ([`set_label_ref`]); it expands to that label, else to
10/// nothing (`\lx@label@ref{<key>}`, defined in `defs/frontmatter_api.rs`).
11pub const LABEL_REF: &str = "\\lx@label@ref";
12
13fn label_ref_key(key: &str) -> String { s!("lx_label_ref:{}", key.trim()) }
14
15/// Record `text` as the label `key` stands for, globally; a later record replaces it (the binding's current label,
16/// `\@currentlabel`). A label a binding fills from an `.aux` round trip follows that binding's own redefinition rule
17/// as LaTeX reads the `.aux` (in a group, latex.ltx:9486-9490: a global definition stays, a local one goes): the
18/// first record wins under a `\newlabel`-like `\cs_new`-if-free (cas's `\NewLabel`, cas-common.sty:191-195,
19/// [`set_label_ref_once`]).
20pub fn set_label_ref(key: &str, text: &str) {
21  assign_value(
22    &label_ref_key(key),
23    Stored::String(pin(text)),
24    Some(Scope::Global),
25  );
26}
27
28/// Record `text` as the label `key` stands for unless one is recorded already: the first record wins, as a label
29/// LaTeX defines globally only when it is free and reads from the `.aux` in a group (cas's `\NewLabel`,
30/// cas-common.sty:191-195; [`set_label_ref`]).
31pub fn set_label_ref_once(key: &str, text: &str) {
32  if label_ref_text(key).is_none() {
33    set_label_ref(key, text);
34  }
35}
36
37/// The label `key` stands for, once recorded ([`set_label_ref`]).
38pub fn label_ref_text(key: &str) -> Option<String> {
39  lookup_value(&label_ref_key(key)).map(|_| lookup_string(&label_ref_key(key)))
40}
41
42/// `\lx@label@ref{key}`: the label `key` stands for, read where it is expanded.
43pub fn label_ref_tokens(key: &str) -> Vec<Token> {
44  let mut out = vec![T_CS!(LABEL_REF), T_BEGIN!()];
45  out.extend(mouth::tokenize_internal(TeXString::assembled(key.to_string())).unlist());
46  out.push(T_END!());
47  out
48}

50/// `text` (a queued entry's keyvals, `annotations={\lx@label@ref {a},b}`) with every `\lx@label@ref{key}` read as the
51/// label it stands for, nothing for one never recorded, as the macro expands: the text the flush's readers of the
52/// queue read, before the entries are digested.
53pub fn resolve_label_refs(text: &str) -> Cow<'_, str> {
54  if !text.contains(LABEL_REF) {
55    return Cow::Borrowed(text);
56  }
57  let mut out = String::with_capacity(text.len());
58  let mut rest = text;
59  while let Some(at) = rest.find(LABEL_REF) {
60    out.push_str(&rest[..at]);
61    let after = &rest[at + LABEL_REF.len()..];
62    let argument = after.trim_start();
63    // (a longer control word, or no braced argument: no reference)
64    let word_goes_on = after.starts_with(|c: char| c.is_ascii_alphabetic() || c == '@');
65    let end = (!word_goes_on && argument.starts_with('{'))
66      .then(|| {
67        let mut depth = 0i32;
68        argument.char_indices().find_map(|(i, c)| {
69          match c {
70            '{' => depth += 1,
71            '}' => depth -= 1,
72            _ => {},
73          }
74          (depth == 0).then_some(i)
75        })
76      })
77      .flatten();
78    match end {
79      Some(end) => {
80        out.push_str(&label_ref_text(&argument[1..end]).unwrap_or_default());
81        rest = &argument[end + 1..];
82      },
83      None => {
84        out.push_str(LABEL_REF);
85        rest = after;
86      },
87    }
88  }
89  out.push_str(rest);
90  Cow::Owned(out)
91}

93/// The queued entries' keyvals (`commands`, a flush's queue) with their label references read
94/// ([`resolve_label_refs`]), for the flush's readers of them (`place.rs`'s `author_labels` and legends,
95/// [`sequenced_contact_labels`]); the entries' tokens, which are digested, keep them.
96pub(in crate::base_utilities) fn resolve_queued_label_refs(commands: &mut [RawFrontmatter]) {
97  for entry in commands {
98    for value in entry.1.values_mut() {
99      if let Cow::Owned(resolved) = resolve_label_refs(value) {
100        *value = resolved;
101      }
102    }
103  }
