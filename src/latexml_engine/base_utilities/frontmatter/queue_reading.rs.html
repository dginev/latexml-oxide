1//! What the frontmatter queue's names ask of their marks, read once per entry and kept with the queue.
2//!
3//! Every contact of an author block asks what the queued names request, so reading the names again at each call made the
4//! block's cost the product of its names and its contacts: 28 % of 2103.04797's instructions (975 authors, the Cut,
5//! `frontmatter_api::cut::cut_marked_contact`) and 56 % of 1901.01540's (1,202, AASTeX's `\affiliation` and the Cut); 2005.08745's
6//! 592 are labelled, so the Cut leaves them at once. The reading is a value of the state beside the queue, so an
7//! assignment journal rolls both back together (a rehearsal) and it goes with the state; an entry is read once, when a
8//! reader first finds it queued. Perl has no counterpart, as it reads no names at a contact. Since Option C F0 an author's
9//! marks are its name's digested ones (`argument_marks`), keyed by printed text (OXIDIZED_DESIGN_DIVERGENCES #995).

17/// The state key of the queue's [`QueueReading`].
18const QUEUE_READING: &str = "frontmatter_raw_reading";

20/// The readings of the first `read` entries of the frontmatter queue: their name marks by tokens, and the marks of the
21/// authors among the first `marks_read`, one marks query per queued author ([`author_marks`]).
22#[derive(Clone, Default)]
23struct QueueReading {
24  read:            usize,
25  /// The entries read so far, a reading again counted anew (the block's count in the `--debug=frontmatter` log, which
26  /// `perfect_kernel_batchc1` pins).
27  entries_read:    usize,
28  name_marks:      Rc<(Vec<String>, Vec<String>)>,
29  marks_read:      usize,
30  author_marks:    Rc<Vec<Vec<String>>>,
31  /// The queue positions (from 1) of the authors requesting each label, ascending.
32  requests:        FxHashMap<String, Vec<usize>>,
33  /// each author's name text where its marks' rehearsal read it ([`argument_marks_and_name`])
34  name_texts:      Vec<Option<String>>,
35  /// the keys of the first `name_texts` authors' names as they print ([`queued_author_name_keys`])
36  name_keys:       Rc<Vec<NameKeys>>,
37  /// the `author:N` positions each author entry read holds, one a name the reading parts it into ([`entry_names`])
38  entry_positions: Rc<Vec<usize>>,
39  /// each position's name, in queue order
40  position_names:  Rc<Vec<Tokens>>,
41  /// the author entries whose marks are read
42  entries_marked:  usize,
43  /// the queued entries whose positions are read
44  positions_read:  usize,
45}
46
47impl QueueReading {
48  // (the names' marks by tokens, `name_marks_of`: token reading until P4c (F0 report))
49  fn read_names(&mut self, entries: &[RawFrontmatter]) {
50    self.entries_read += entries.len();
51    for (tag, _, command) in entries {
52      if tag == "ltx:creator" {
53        let (altaffil, sup) = name_marks_of(command);
54        let marks = Rc::make_mut(&mut self.name_marks);
55        marks.0.extend(altaffil);
56        marks.1.extend(sup);
57      }
58    }
59  }
60
61  // (`from`: the queue position of the first of `entries`)
62  fn read_positions(&mut self, entries: &[RawFrontmatter], from: usize) -> Result<()> {
63    for (at, (tag, attr, command)) in entries.iter().enumerate() {
64      if tag == "ltx:creator" && attr.get("role").map(String::as_str) == Some("author") {
65        let names = entry_pieces(&queued_name(command), from + at)?;
66        Rc::make_mut(&mut self.entry_positions).push(names.len());
67        Rc::make_mut(&mut self.position_names).extend(names);
68      }
69    }
70    Ok(())
71  }
72
73  fn read_marks(&mut self, entries: &[RawFrontmatter]) -> Result<()> {
74    for (tag, attr, command) in entries {
75      if tag != "ltx:creator" || attr.get("role").map(String::as_str) != Some("author") {
76        continue;
77      }
78      let count = self.entry_positions[self.entries_marked];
79      self.entries_marked += 1;
80      let from = self.author_marks.len();
81      for at in from..from + count {
82        let name = self.position_names[at].clone();
83        let (texts, keys, text) = if count == 1 {
84          author_marks(command, &name)?
85        } else {
86          author_marks(&name, &name)?
87        };
88        self.name_texts.push(text);
89        let position = at + 1;
90        for label in keys {
91          let requesters = self.requests.entry(label).or_default();
92          if requesters.last() != Some(&position) {
93            requesters.push(position);
94          }
95        }
96        Rc::make_mut(&mut self.author_marks).push(texts);
97      }
98    }
99    Ok(())
100  }
101}

103/// The marks a queued author `entry`'s `name` ([`queued_name`], or one name the reading parts it into) prints at its own
104/// level outside its notes, by their printed text (`argument_marks`: a document macro's mark too,
105/// `\newcommand\ua{\textsuperscript{1}}`), and the labels they request, as `\lx@sup@request@affiliation` keys them
106/// (`printed_label_keys`); a `\footnotemark` requests a note, none. Where the query cannot read the name, the entry's
107/// mark operands by spelling (`author_mark_operands`), counted. And the name's text where the query rehearsed it.
108fn author_marks(
109  entry: &Tokens,
110  name: &Tokens,
111) -> Result<(Vec<String>, Vec<String>, Option<String>)> {
112  let mut texts = Vec::new();
113  let mut keys = Vec::new();
114  if let Some((marks, name)) = argument_marks_and_name(name)? {
115    for mark in marks {
116      if mark.role.as_deref() == Some("footnotemark") {
117        continue;
118      }
119      // (a mark printing no label requests none: counted, as an unread name is)
120      let Some(text) = mark_request_text(&mark) else {
121        count_unread_marks("queued_author_marks, a mark without a label");
122        continue;
123      };
124      keys.extend(printed_label_keys(&text, "affiliation"));
125      texts.push(text);
126    }
127    return Ok((texts, keys, name));
128  }
129  count_unread_marks("queued_author_marks");
130  for operand in author_mark_operands(entry.unlist_ref()) {
131    let spellings = std::iter::once(operand.clone()).chain(label_pieces(&operand));
132    for spelling in spellings {
133      keys.extend(clean_frontmatter_labels(
134        &Tokens::new(spelling).to_string(),
135        "affiliation",
136      ));
137    }
138    texts.push(Tokens::new(operand).to_string());
139  }
140  Ok((texts, keys, None))
141}

143/// A mark's printed label as `printed_label` reads its operand, which keys its requests and the label it sets: a
144/// cross-reference's key as `\ref{key}` (`events::script_mark`'s `ref:key`); `None` for a mark printing no label.
145pub(in crate::base_utilities) fn mark_request_text(mark: &FmEvent) -> Option<String> {
146  let label = mark.label.as_deref()?;
147  if !mark.role.as_deref().is_some_and(|r| r.ends_with(" ref")) {
148    return Some(label.to_string());
149  }
150  Some(
151    label
152      .split(',')
153      .map(|piece| match piece.trim().strip_prefix("ref:") {
154        Some(key) => s!("\\ref{{{key}}}"),
155        None => piece.to_string(),
156      })
157      .collect::<Vec<_>>()
158      .join(","),
159  )
160}

178/// Keep `reading` in the state, beside a non-empty queue.
179fn keep(reading: QueueReading) -> Rc<QueueReading> {
180  let reading = Rc::new(reading);
181  if reading.read > 0 {
182    assign_value(
183      QUEUE_READING,
184      Stored::Opaque(Rc::clone(&reading) as Rc<dyn std::any::Any>),
185      Some(Scope::Global),
186    );
187  }
188  reading
189}

191/// The queue's reading, its new entries' name marks read.
192fn queue_reading() -> Rc<QueueReading> {
193  let queued = with_value("frontmatter_raw", |v| match v {
194    Some(Stored::FrontmatterRaw(queue)) => queue.len(),
195    _ => 0,
196  });
197  let kept = kept_queue_reading();
198  if let Some(kept) = kept.as_ref().filter(|kept| kept.read == queued) {
199    return Rc::clone(kept);
200  }
201  // (given up by the state, so that it is extended in place unless a journal holds it too)
202  remove_value(QUEUE_READING);
203  let mut reading = QueueReading::default();
204  if let Some(kept) = kept {
205    if kept.read <= queued {
206      reading = Rc::unwrap_or_clone(kept);
207    } else {
208      reading.entries_read = kept.entries_read;
209    }
210  }
211  reading.read_names(&queued_from(reading.read));
212  reading.read = queued;
213  keep(reading)
214}

216/// The queue's reading, its authors' `author:N` positions read too, one a name an entry holds ([`entry_pieces`]).
217fn queue_positions_reading() -> Result<Rc<QueueReading>> {
218  let reading = queue_reading();
219  if reading.positions_read == reading.read {
220    return Ok(reading);
221  }
222  remove_value(QUEUE_READING);
223  let mut reading = Rc::unwrap_or_clone(reading);
224  let from = reading.positions_read;
225  reading.read_positions(&queued_from(from)[..reading.read - from], from)?;
226  reading.positions_read = reading.read;
227  Ok(keep(reading))
228}

230/// The queue's reading, its authors' marks read too.
231fn queue_marks_reading() -> Result<Rc<QueueReading>> {
232  let reading = queue_positions_reading()?;
233  if reading.marks_read == reading.read {
234    return Ok(reading);
235  }
236  remove_value(QUEUE_READING);
237  let mut reading = Rc::unwrap_or_clone(reading);
238  // (the entries taken off the queue first: a marks query digests)
239  reading.read_marks(&queued_from(reading.marks_read)[..reading.read - reading.marks_read])?;
240  reading.marks_read = reading.read;
241  Ok(keep(reading))
242}

244/// The queue lost entries (a flush took it, `dequeue_front_matter` removed some): what was read of it is no more.
245pub(in crate::base_utilities) fn forget_queue_reading() {
246  if let Some(reading) = kept_queue_reading() {
247    DebugFeature!(
248      "frontmatter",
249      "FRONT the queue's reading: {} entries read for {} queued",
250      reading.entries_read,
251      reading.read
252    );
253    remove_value(QUEUE_READING);
254  }
255}

282/// The keys ([`NameKeys`]) of each queued author's name ([`queued_name`]) as it prints, one an `author:N` position
283/// ([`queued_author_marks`]): the words the rehearsal reads ([`argument_name_text`]: a macro's or a `\let` copy's
284/// expansion, an accent's letter), the marks' own rehearsal shared; an unread name spells no one, counted.
285pub(in crate::base_utilities) fn queued_author_name_keys() -> Result<Rc<Vec<NameKeys>>> {
286  let reading = queue_marks_reading()?;
287  if reading.name_keys.len() == reading.name_texts.len() {
288    return Ok(Rc::clone(&reading.name_keys));
289  }
290  remove_value(QUEUE_READING);
291  let mut reading = Rc::unwrap_or_clone(reading);
292  let from = reading.name_keys.len();
293  let keys: Vec<NameKeys> = reading
294    .position_names
295    .iter()
296    .zip(&reading.name_texts)
297    .skip(from)
298    .map(|(name, text)| printed_name_keys(name, text.clone()))
299    .collect::<Result<_>>()?;
300  Rc::make_mut(&mut reading.name_keys).extend(keys);
301  Ok(Rc::clone(&keep(reading).name_keys))
302}

367/// Whether the queued authors' names (`queued_name`) print every word `text` prints — each word one of a name's, or
368/// an initial of one, accents aside (`visible_name_text`), or the word "and" between names: `text` then prints no
369/// word the authors do not. True for a `text` printing no word; else false when no author is queued. svjour3's
370/// `\institute{F. Author \at …}` piece names its authors before the address (`sv_support_sty.rs`): a piece writing
371/// only their names adds nothing to the XML (`B. Baker, and C. Cole \at …`, batch 63's piece naming two), one writing
372/// more (`\textit{Contact author:} Erwan {Le Merrer}`, 1711.01894; `Xin-long Luo, Corresponding author`, 2107.13864;
373/// the whole name `Leonardo Anjoletto Ferreira` of `Leonardo A. Ferreira`, 1705.01399) prints words no name does.
374pub fn queued_author_name_prints(text: &Tokens) -> bool {
375  let written = printed_words(text);
376  if written.is_empty() {
377    return true;
378  }
379  let names: Vec<String> = queued_author_name_words().into_iter().flatten().collect();
380  !names.is_empty()
381    && written.iter().all(|w| {
382      w == "and"
383        || names
384          .iter()
385          .any(|o| o == w || (w.chars().count() == 1 && o.starts_with(w.as_str())))
386    })
387}

389/// How many queued authors' names (`queued_name`) `text` spells whole: each author whose name's words (as
390/// [`queued_author_name_prints`] reads them) stand together, in order, among the words `text` prints. svjour3's
391/// `\institute{\begin{itemize}\item[\Letter] Ann Able \\ \email{…} \item[] Bob Baker \\ \email{…} \at \item[1]
392/// Univ X …}` names its list's authors before the one `\at` (2312.14898): two or more names it no one by. A whole author
393/// list queued as one entry (sv_support's `\author{A \and B}`, one `\lx@add@author@split` call the reading cuts) is
394/// parted at its separators first (`entry_names`), so each of its names is counted as a name of its own.
395pub fn queued_author_names_spelled(text: &Tokens) -> Result<usize> {
396  let written = printed_words(text);
397  let mut spelled = 0;
398  for entry in queued_author_names() {
399    for name in entry_names(&entry)? {
400      let name = printed_words(&name);
401      if !name.is_empty()
402        && written
403          .windows(name.len())
404          .any(|run| run == name.as_slice())
405      {
406        spelled += 1;
407      }
408    }
409  }
410  Ok(spelled)
411}

413/// The names one queued author entry holds: the entry parted where the reading cuts its names, at each `\and`-family
414/// separator not inside a layout container (a brace group is none) the rehearsal reads ([`argument_separators`],
415/// OXIDIZED_DESIGN_DIVERGENCES #995), at its own tokens (`ArgSeparator::span`: a literal, a `\let` copy, an active
416/// character, a macro holding only the separator, the document's macro `\and`; never an `\end{tabular}` glued before it,
417/// which is the table's), whatever it means where the query is made: the rehearsal binds `\and` as the separator whatever
418/// it means (`rehearsal::ALWAYS_BOUND`), so the reading writes a creator per piece there, and svjour3, the one class whose
419/// binding asks, sets its own `\and` in its `\@maketitle` before the names print (svjour3.cls:608; svjour, svmult and
420/// svproc likewise), so a document's `\renewcommand\and` before it changes no part. A separator whose tokens print ink
421/// too (a macro holding names and the separator, `\newcommand\aut{Ann \and Bob}`) has no span: no part there. An entry
422/// the query could not read (`None`: no rehearsal made, over its allowance, a group left open) is one name, as it is with
423/// no separator: unread is not "no separator", and keeping the entry whole is the reading before any part was known.
424fn entry_names(name: &Tokens) -> Result<Vec<Tokens>> {
425  let Some(separators) = argument_separators(name)? else {
426    return Ok(vec![name.clone()]);
427  };
428  let content = name.unlist_ref();
429  let mut names = Vec::new();
430  let mut from = 0;
431  for separator in separators {
432    if let Some((start, end)) = separator.span
433      && let Some(name) = content.get(from..start)
434    {
435      names.push(Tokens::new(name.to_vec()));
436      from = end;
437    }
438  }
439  names.push(Tokens::new(
440    content.get(from..).unwrap_or_default().to_vec(),
441  ));
442  Ok(names)
443}

445/// The names a queued author entry's `name` holds ([`entry_names`]) where a whole-list call queued it, at
446/// `queue_position`: a binding's (`\lx@add@author@split`: sv_support's, aa's, OmniBus's, webofc's, ptephy's, llncs's
447/// `\author`) or the kernel `\author`'s (`\lx@add@authors@adaptive`); several are recorded for its block, whose positions
448/// the resolver counts so.
449fn entry_pieces(name: &Tokens, queue_position: usize) -> Result<Vec<Tokens>> {
450  // (one name: the entry's name whole, as it was read before its names were known)
451  let hook = ledger::queuing_hook(queue_position);
452  let whole_list =
453    hook == Some("\\lx@add@author@split") || hook == Some("\\lx@add@authors@adaptive");
454  let names = if whole_list {
455    entry_names(name)?
456  } else {
457    Vec::new()
458  };
459  if names.len() < 2 {
460    return Ok(vec![name.clone()]);
461  }
462  ledger::record_entry_names(queue_position, names.len());
463  Ok(names)
464}

471/// The words `tokens` print as a name-shape test reads them ([`visible_name_text`]), accents aside, lowercased.
472fn printed_words(tokens: &Tokens) -> Vec<String> {
473  resolver::base_letters(&visible_name_text(tokens.unlist_ref()))
474    .split(|c: char| !c.is_alphanumeric())
475    .filter(|w| !w.is_empty())
476    .map(str::to_lowercase)
477    .collect()
478}

485/// Each queued author entry's name ([`queued_name`]), in queue order.
486fn queued_author_names() -> Vec<Tokens> {
487  with_value("frontmatter_raw", |v| match v {
488    Some(Stored::FrontmatterRaw(queue)) => queue
489      .iter()
490      .filter(|entry| {
491        entry.0 == "ltx:creator" && entry.1.get("role").map(String::as_str) == Some("author")
492      })
493      .map(|entry| queued_name(&entry.2))
494      .collect(),
495    _ => Vec::new(),
496  })
