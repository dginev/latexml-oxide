11/// The footnote symbols a name's end may carry typed by hand (`Yuan Yuan*`, 2007.16032; `Shicheng Wang†`,
12/// math0304301; `D. Hernández Ruipérez¶`, alg-geom9405006). No `#`, no `\@fnsymbol` glyph: at a name's end it is its ink
13/// (`Team C\#`); no author block of the census's 35,942 papers types it there, its `#` marks are superscripts (1705.01500).
14const HAND_MARKS: [char; 6] = ['*', '\u{2217}', '\u{2020}', '\u{2021}', '\u{A7}', '\u{B6}'];

16/// Takes every mark off each creator's personname: its note's `mark`, the label of a contact it holds (consumed), or a
17/// mark-only note after the name; never dropped. OXIDIZED_DESIGN_DIVERGENCES #1055, guards `perfect_kernel_batchpure`.
18pub(in crate::base_utilities) fn keep_personnames_pure(document: &mut Document) -> Result<()> {
19  // (glued digits part into held labels only under single-digit numbering of the name's author block — the creators of
20  // one parent —, as the reader parts a request (`resolver::glued_request_parts`): 2409.11440's `$^{12}$` under entries
21  // 1 and 2; beside an entry 10, one number)
22  let mut blocks: Vec<(Option<Node>, bool)> = Vec::new();
23  for creator in document.findnodes("//ltx:creator[ltx:personname]", None) {
24    let block = creator.get_parent();
25    let single_digits = match blocks.iter().find(|(parent, _)| *parent == block) {
26      Some((_, single)) => *single,
27      None => {
28        let single = !block
29          .iter()
30          .flat_map(|parent| children_named(parent, "creator"))
31          .flat_map(|creator| children_named(&creator, "contact"))
32          .flat_map(|contact| contact_keys(&contact))
33          .any(|key| key.len() > 1 && key.chars().all(|c| c.is_ascii_digit()));
34        blocks.push((block, single));
35        single
36      },
37    };
38    for personname in children_named(&creator, "personname") {
39      let groups = mark_groups(&personname);
40      if groups.is_empty() {
41        continue;
42      }
43      // (each contact's keys, each the mark it prints; a note or a contact answers one mark printed on the name,
44      // [`Answerer`]: `\IEEEauthorrefmark{1}$^{*}$` over `∗Univ A` prints two ∗, one the contact's)
45      let held: Vec<Vec<(String, String)>> = children_named(&creator, "contact")
46        .iter()
47        .map(|contact| {
48          contact_keys(contact)
49            .into_iter()
50            .map(|key| (key.clone(), key))
51            .collect()
52        })
53        .collect();
54      let mut spent: HashSet<(Answerer, String)> = HashSet::default();
55      let notes: Vec<Node> = children_named(&creator, "note")
56        .into_iter()
57        .filter(|note| note.get_attribute("role").as_deref() != Some("footnotemark"))
58        .collect();
59      let mut after = personname.clone();
60      let mut parents: Vec<Node> = Vec::new();
61      // (the name's author reference marks, each with the glyph it printed — over every run: an ORCID icon, a graphic
62      // or a formula parts `Ann Able$^{*}$\orcidlink{…}\IEEEauthorrefmark{1}` in two,
63      // pure_name_orcid_between_typed_star_and_refmark)
64      let mut refmarks: Vec<(String, String)> = groups
65        .iter()
66        .flat_map(|group| &group.refmarks)
67        .filter_map(|(number, sole)| Some((number.clone(), mark_form(sole.as_ref()?))))
68        .collect();
69      for group in groups {
70        let printed = group.printed.trim().to_string();
71        let pieces = mark_pieces(&printed);
72        let mut left: Vec<String> = Vec::new();
73        let mut taken = false;
74        for piece in &pieces {
75          let key = mark_form(piece);
76          // (keyed by the number of the reference mark that printed it, each once — never by the name's other numbers:
77          // 2508.13181's `Dominik Loroch\IEEEauthorrefmark{2}\IEEEauthorrefmark{1}\textsuperscript{\textsection}`
78          // keys its § by none, and the § stays)
79          let printed_by = refmarks
80            .iter()
81            .position(|(_, glyph)| *glyph == key)
82            .map(|at| refmarks.remove(at).0);
83          let keys: Vec<String> = std::iter::once(key.clone()).chain(printed_by).collect();
84          if answer(document, &notes, &held, &mut spent, piece, &key, &keys)? {
85            taken = true;
86            continue;
87          }
88          // (a run of glued labels, each a held contact's, `$^{ab}$` over `$^a$Univ A`, `$^b$Univ B`, revtex3's — digits
89          // only under single-digit numbering; or a label glued to a note's number, `$^{a1}$` over `$^a$Department …`
90          // (hep-ph0011101), the number left)
91          let held_part = |part: &str| {
92            let part = mark_form(part);
93            held.iter().flatten().any(|(key, _)| *key == part)
94          };
95          let digits = piece.chars().all(|c| c.is_ascii_digit());
96          let glued: Vec<String> = if piece.chars().count() > 1
97            && piece.chars().all(char::is_alphanumeric)
98            && (single_digits || !digits)
99            && piece.chars().all(|c| held_part(&c.to_string()))
100          {
101            Vec::new()
102          } else {
103            let parts = letter_digit_runs(piece);
104            if parts.len() > 1 && parts.iter().any(|part| held_part(part)) {
105              parts.into_iter().filter(|part| !held_part(part)).collect()
106            } else {
107              left.push(piece.clone());
108              continue;
109            }
110          };
111          taken = true;
112          left.extend(glued);
113        }
114        let shown = left.join(",");
115        DebugFeature!(
116          "frontmatter",
117          "FRONT pure name: mark {printed}, {shown} left outside the name"
118        );
119        let label =
120          group.hand.is_none() && group.sups.iter().all(|sup| sup.has_attribute(LABEL_MARK));
121        group.cut(document, &mut parents);
122        if !left.is_empty() {
123          // (as printed where nothing was answered, `(a,b)`, or a range was printed, `1-3`; else the marks left, a list)
124          let ranged = printed.contains(['-', '\u{2013}', '\u{2212}']);
125          let mark = if !taken || ranged {
126            printed.clone()
127          } else {
128            left.join(",")
129          };
130          let mut note = mark_only_note(document, &personname, &mark, label)?;
131          after.add_next_sibling(&mut note)?;
132          after = note;
133        }
134      }
135      drop_emptied_wrappers(document, parents);
136      trim_name_edges(document, &personname);
137    }
138  }
139  Ok(())
140}

142/// Whether `piece` (keyed `key`; `keys` with the number of the author reference mark that printed it,
143/// `\IEEEauthorrefmark{4}` printing `§`) is answered on its creator: a note marked so; a contact keyed so (`held`, each
144/// contact's keys with the mark each prints, spent per printed mark); or an unmarked note whose text opens with a copy
145/// of it (a legend: `\thanks{*Equal Contribution}`, 1904.03977), which takes it as its `mark` ([`leading_mark`]).
146/// A symbol beside a `\thanks` is no `\thanks`'s mark: the `\thanks` prints its own (`Haoran Li$^{*}$\thanks{…}` prints
147/// "∗†", 2406.11149), the same glyph too (`Ann Able*\thanks{…}` prints "*∗": a note its command marks,
148/// [`MARK_FROM_COUNTER`], answers no typed glyph).
149fn answer(
150  document: &mut Document,
151  notes: &[Node],
152  held: &[Vec<(String, String)>],
153  spent: &mut HashSet<(Answerer, String)>,
154  piece: &str,
155  key: &str,
156  keys: &[String],
157) -> Result<bool> {
158  if let Some(at) = (0..notes.len()).find(|&at| {
159    notes[at].get_attribute(MARK_FROM_COUNTER).is_none()
160      && notes[at]
161        .get_attribute("mark")
162        .is_some_and(|mark| mark_form(&mark) == key)
163      && !spent.contains(&(Answerer::Note(at), key.to_string()))
164  }) {
165    spent.insert((Answerer::Note(at), key.to_string()));
166    return Ok(true);
167  }
168  // (a contact answers each mark it prints once, by any of its keys: under `∗Univ A` keyed `1` and `∗`, of
169  // `Ann Able$^{*}$\orcidlink{…}\IEEEauthorrefmark{1}`'s two ∗ one,
170  // pure_name_orcid_between_typed_star_and_refmark)
171  let answering = |at: usize| {
172    held[at]
173      .iter()
174      .filter(|(held_key, _)| keys.contains(held_key))
175      .map(|(_, printed)| printed.clone())
176      .collect::<Vec<_>>()
177  };
178  if let Some(at) = (0..held.len()).find(|&at| {
179    answering(at)
180      .iter()
181      .any(|printed| !spent.contains(&(Answerer::Contact(at), printed.clone())))
182  }) {
183    for printed in answering(at) {
184      spent.insert((Answerer::Contact(at), printed));
185    }
186    return Ok(true);
187  }
188  for (at, note) in notes.iter().enumerate() {
189    if note
190      .get_attribute("mark")
191      .is_some_and(|mark| !mark.trim().is_empty())
192    {
193      continue;
194    }
195    if let Some(lead) = leading_mark(note, piece) {
196      let mut note = note.clone();
197      document.set_attribute(&mut note, "mark", piece)?;
198      lead.cut(document);
199      spent.insert((Answerer::Note(at), key.to_string()));
200      return Ok(true);
201    }
202  }
203  Ok(false)
204}

215/// The runs of letters and of digits `piece` parts into (`a1`: `a`, `1`), each its own mark where one is a label.
216fn letter_digit_runs(piece: &str) -> Vec<String> {
217  let mut runs: Vec<String> = Vec::new();
218  let mut last_digit = None;
219  for c in piece.chars() {
220    let digit = c.is_ascii_digit();
221    match runs.last_mut() {
222      Some(run) if last_digit == Some(digit) => run.push(c),
223      _ => runs.push(c.to_string()),
224    }
225    last_digit = Some(digit);
226  }
227  runs
228}

239/// The mark keys a contact is placed by or prints ([`mark_form`]): its labels (`_label`, `_bymark`, the printed label
240/// of an emitted entry, [`PRINTED_LABEL`]) — an affiliation's, or one of no kind, a command's spelling as its glyph
241/// (`\dag`: †, [`unanswered_mark_text`]) — and the superscript its text opens with (`$^1$Raman Research Institute`,
242/// 0707.3527), inside its font wrappers too.
243fn contact_keys(contact: &Node) -> Vec<String> {
244  let mut keys = Vec::new();
245  for attribute in ["_label", "_bymark", PRINTED_LABEL] {
246    let Some(labels) = contact.get_attribute(attribute) else {
247      continue;
248    };
249    for label in clean_frontmatter_labels(&labels, "") {
250      let key = match label.split_once(':') {
251        Some(("affiliation", key)) => key,
252        Some(_) => continue,
253        None => label.as_str(),
254      };
255      let key = unanswered_mark_text(key).unwrap_or_else(|| key.to_string());
256      keys.push(mark_form(&key));
257    }
258  }
259  // (through the font or box wrappers it opens with: authblk's `\textsuperscript{\(*\)}`, `<text><sup>…`)
260  let first = |node: &Node| {
261    node.get_child_nodes().into_iter().find(|child| {
262      child.get_type() != Some(NodeType::TextNode) || !child.get_content().trim().is_empty()
263    })
264  };
265  let mut opening = first(contact);
266  while let Some(node) = opening
267    .as_ref()
268    .filter(|node| node.get_type() == Some(NodeType::ElementNode) && node.get_name() == "text")
269  {
270    opening = first(node);
271  }
272  let printed = opening
273    .filter(|node| node.get_type() == Some(NodeType::ElementNode))
274    .and_then(|node| match node.get_name().as_str() {
275      "sup" => Some(node.get_content().trim().to_string()),
276      "Math" => match fm_read_formula(&node) {
277        FmFormulaMarks::Marks(scripts) => Some(scripts.join(",")),
278        _ => None,
279      },
280      _ => None,
281    });
282  if let Some(printed) = printed {
283    keys.extend(mark_pieces(&printed).iter().map(|piece| mark_form(piece)));
284  }
285  keys.retain(|key| !key.is_empty());
286  keys.sort();
287  keys.dedup();
288  keys
289}

291/// The marks of a printed mark run, in order: its items at commas and spaces, out of their brackets (`(a,b,∗)`), each
292/// parted further where a letter or number run meets a symbol and between distinct symbols (`1∗`, `*‡`), a run of one
293/// symbol one mark (`**`); a range's dash parts its ends.
294fn mark_pieces(printed: &str) -> Vec<String> {
295  let mut pieces = Vec::new();
296  let mut current = String::new();
297  let flush = |current: &mut String, pieces: &mut Vec<String>| {
298    if !current.is_empty() {
299      pieces.push(std::mem::take(current));
300    }
301  };
302  for c in printed.chars() {
303    if c.is_whitespace()
304      || matches!(
305        c,
306        ',' | ';' | '(' | ')' | '[' | ']' | '-' | '\u{2013}' | '\u{2212}'
307      )
308    {
309      flush(&mut current, &mut pieces);
310      continue;
311    }
312    if is_combining_mark(c) {
313      current.push(c);
314      continue;
315    }
316    let last = current.chars().rev().find(|c| !is_combining_mark(*c));
317    let parts = last.is_some_and(|last| {
318      if last.is_alphanumeric() {
319        !c.is_alphanumeric()
320      } else {
321        c.is_alphanumeric() || c != last
322      }
323    });
324    if parts {
325      flush(&mut current, &mut pieces);
326    }
327    current.push(c);
328  }
329  flush(&mut current, &mut pieces);
330  pieces
331}

333/// One run of marks on a name, as printed: its superscripts and the blank text between them, or the footnote symbols
334/// typed at the end of a text (cut at `hand`).
335struct MarkGroup {
336  printed:  String,
337  /// its superscripts and formulas
338  sups:     Vec<Node>,
339  hand:     Option<(Node, usize)>,
340  /// the author reference marks' numbers its superscripts print (`_refmark`, `relocate::sup_marks`), each with the
341  /// glyph its superscript prints where it prints that one mark alone
342  refmarks: Vec<(String, Option<String>)>,
343}
344
345impl MarkGroup {
346  /// Takes the group out of the name, `parents` the wrappers it leaves.
347  fn cut(&self, document: &mut Document, parents: &mut Vec<Node>) {
348    if let Some((text, at)) = &self.hand {
349      let content = text.get_content();
350      let mut text = text.clone();
351      let _ = text.set_content(content[..*at].trim_end());
352    }
353    for sup in &self.sups {
354      parents.extend(sup.get_parent());
355      document.remove_node(sup.clone());
356    }
357  }
358}

360/// A piece of a name's ink, in print order ([`mark_groups`]).
361enum Ink {
362  Text(Node),
363  /// a superscript or a formula of scripts alone, and its printed text
364  Mark(Node, String),
365  Other,
366}

368/// The mark runs of `personname` ([`MarkGroup`]): its superscripts and formulas of scripts alone outside notes and
369/// links, but those that are
370/// the name's ink — a lowercase one glued in a word (`James M<sup>c</sup>Kernan`, math0610203), a number's suffix
371/// (`1<sup>st</sup> Siqiao Xue`, 2211.11758), a number glued before a word inside the name (`The <sup>3</sup>He
372/// Collaboration`) — and the footnote symbols ending its last text ([`HAND_MARKS`]).
373fn mark_groups(personname: &Node) -> Vec<MarkGroup> {
374  fn walk(node: &Node, out: &mut Vec<Ink>) {
375    for child in node.get_child_nodes() {
376      match child.get_type() {
377        Some(NodeType::TextNode) => out.push(Ink::Text(child)),
378        Some(NodeType::ElementNode) => match child.get_name().as_str() {
379          "sup" => {
380            let printed = child.get_content().trim().to_string();
381            // (a mark whose text a reference prints, resolved only after the conversion — cmpj's `\refaddr{la}`,
382            // `<sup><ref labelref="LABEL:la"/></sup>` — has no text to give a note's `mark` yet: it stays)
383            if name_ink(&child, &printed) || prints_later(&child) {
384              out.push(Ink::Other);
385            } else {
386              out.push(Ink::Mark(child, printed));
387            }
388          },
389          // (a formula of scripts alone is a mark, as the rewrite of math into text sets it after: `${}^{1}$`; any other
390          // formula, and the scripts on a base, `$\text{Leon Bello}^{1,3}$`, the name's)
391          "Math" => match fm_read_formula(&child) {
392            FmFormulaMarks::Marks(scripts) => {
393              let printed = scripts.join(",");
394              if name_ink(&child, &printed) || prints_later(&child) {
395                out.push(Ink::Other);
396              } else {
397                out.push(Ink::Mark(child, printed));
398              }
399            },
400            _ => out.push(Ink::Other),
401          },
402          // (a link around the name holds its marks too, `\href{…}{Gert Desmet$^2$\includegraphics{orcid}}`, 2208.09944;
403          // an ORCID iD's icon none)
404          "ref"
405            if !child
406              .get_attribute("class")
407              .is_some_and(|c| c.contains("ltx_orcid")) =>
408          {
409            walk(&child, out)
410          },
411          "note" | "ref" | "ERROR" | "graphics" | "picture" | "svg" => out.push(Ink::Other),
412          _ => walk(&child, out),
413        },
414        _ => {},
415      }
416    }
417  }
418  let mut ink = Vec::new();
419  walk(personname, &mut ink);
420  // (the footnote symbols ending the last text: past them, the name's letters)
421  let mut hand: Option<(usize, usize)> = None;
422  if let Some(index) = ink
423    .iter()
424    .rposition(|piece| matches!(piece, Ink::Text(text) if !text.get_content().trim().is_empty()))
425    && let Ink::Text(text) = &ink[index]
426  {
427    let content = text.get_content();
428    let end = content.trim_end();
429    let start = end
430      .char_indices()
431      .rev()
432      .take_while(|(_, c)| HAND_MARKS.contains(c) || *c == ',' || c.is_whitespace())
433      .last()
434      .map_or(end.len(), |(at, _)| at);
435    let run = &end[start..];
436    let letters_before = content[..start].chars().any(char::is_alphabetic)
437      || ink[..index].iter().any(
438        |piece| matches!(piece, Ink::Text(t) if t.get_content().chars().any(char::is_alphabetic)),
439      );
440    if run.chars().any(|c| HAND_MARKS.contains(&c)) && letters_before {
441      let start = start + (run.len() - run.trim_start_matches([',', ' ']).len());
442      hand = Some((index, start));
443    }
444  }
445  // (a mark with the name's ink on both sides parts a personname holding more than a name — an author block written whole,
446  // `Ann Able, $^1$Bob Baker, … $^1$Univ A $^2$Univ B`, or names merged, `Ann Able$^*$ Bob Baker`, each counted by its
447  // Warning or Error: its marks label what they stand between, and stay; ne2d, 108a)
448  let inked = |piece: &Ink| matches!(piece, Ink::Text(text) if text.get_content().chars().any(char::is_alphanumeric));
449  let mid = ink.iter().enumerate().any(|(index, piece)| {
450    matches!(piece, Ink::Mark(..))
451      && ink[..index].iter().any(inked)
452      && ink[index + 1..].iter().any(inked)
453  });
454  if mid {
455    return Vec::new();
456  }
457  let mut groups: Vec<MarkGroup> = Vec::new();
458  let mut open = false;
459  for (index, piece) in ink.iter().enumerate() {
460    match piece {
461      Ink::Mark(sup, content) => {
462        if !open {
463          groups.push(MarkGroup {
464            printed:  String::new(),
465            sups:     Vec::new(),
466            hand:     None,
467            refmarks: Vec::new(),
468          });
469          open = true;
470        }
471        let group = groups.last_mut().expect("an open group");
472        // (two superscripts are two marks, `$^1$ $^2$` "1,2", 1304.0778; a comma set as one of its own is theirs,
473        // `\authorrefmark{1}$^,$\authorrefmark{2}`, 2011.00594)
474        if !group.printed.is_empty() && !group.printed.ends_with(',') && !content.starts_with(',') {
475          group.printed.push(',');
476        }
477        group.printed.push_str(content);
478        group.sups.push(sup.clone());
479        // (its own and those of the superscripts inside it)
480        let mut numbers: Vec<String> = Vec::new();
481        refmark_numbers(sup, &mut numbers);
482        let sole = match (numbers.as_slice(), mark_pieces(content).as_slice()) {
483          ([_], [glyph]) => Some(glyph.clone()),
484          _ => None,
485        };
486        group
487          .refmarks
488          .extend(numbers.into_iter().map(|number| (number, sole.clone())));
489      },
490      Ink::Text(text) => match hand {
491        Some((at, cut)) if at == index => {
492          let content = text.get_content();
493          groups.push(MarkGroup {
494            printed:  content[cut..].trim_end().to_string(),
495            sups:     Vec::new(),
496            hand:     Some((text.clone(), cut)),
497            refmarks: Vec::new(),
498          });
499          open = true;
500        },
501        _ => open &= text.get_content().trim().is_empty(),
502      },
503      Ink::Other => open = false,
504    }
505  }
506  groups.retain(|group| !group.printed.trim().is_empty());
507  groups
508}

510/// The author reference marks' numbers `node` and the elements inside it print (`_refmark`), in order.
511fn refmark_numbers(node: &Node, out: &mut Vec<String>) {
512  if let Some(keys) = node.get_attribute("_refmark") {
513    out.extend(keys.split(',').map(mark_form));
514  }
515  for child in node.get_child_nodes() {
516    if child.get_type() == Some(NodeType::ElementNode) {
517      refmark_numbers(&child, out);
518    }
519  }
520}

522/// Whether `node` holds ink filled in after the conversion: a cross-reference or a citation of no text yet, whose text
523/// the post-processor writes.
524fn prints_later(node: &Node) -> bool {
525  node.get_child_nodes().iter().any(|child| {
526    child.get_type() == Some(NodeType::ElementNode)
527      && ((matches!(child.get_name().as_str(), "ref" | "cite" | "bibref")
528        && child.get_content().trim().is_empty())
529        || prints_later(child))
530  })
531}

533/// Whether a superscript (or a formula of scripts alone) printing `content` is the name's own ink, no mark
534/// ([`mark_groups`]).
535fn name_ink(sup: &Node, content: &str) -> bool {
536  let before = sibling_text(sup, false);
537  let after = sibling_text(sup, true);
538  if glued_in_a_word(content, before.as_deref(), after.as_deref()) {
539    return true;
540  }
541  let after_digit = before
542    .as_deref()
543    .and_then(|text| text.chars().next_back())
544    .is_some_and(|c| c.is_ascii_digit());
545  if after_digit && matches!(content, "st" | "nd" | "rd" | "th") {
546    return true;
547  }
548  // (a number glued before a word, after words of the name: an isotope's prescript)
549  !content.is_empty()
550    && content.chars().all(|c| c.is_ascii_digit())
551    && after
552      .as_deref()
553      .and_then(|text| text.chars().next())
554      .is_some_and(char::is_uppercase)
555    && before.as_deref().is_some_and(|text| {
556      text.ends_with(char::is_whitespace) && text.chars().any(char::is_alphabetic)
557    })
558}

560/// The text of the sibling just before `node` (`after`: just after it), when a text node.
561fn sibling_text(node: &Node, after: bool) -> Option<String> {
562  let side = if after {
563    node.get_next_sibling()
564  } else {
565    node.get_prev_sibling()
566  };
567  side
568    .filter(|n| n.get_type() == Some(NodeType::TextNode))
569    .map(|n| n.get_content())
570}

572/// The copy of a mark a note's text opens with: its leading superscript or formula, or the head of its first text.
573enum Lead {
574  Sup(Node),
575  Text(Node, usize),
576}
577
578impl Lead {
579  /// Takes the copy out, and the font wrappers it empties.
580  fn cut(self, document: &mut Document) {
581    let parent = match &self {
582      Lead::Sup(node) | Lead::Text(node, _) => node.get_parent(),
583    };
584    match self {
585      Lead::Sup(sup) => {
586        // (and the space it printed before the text, `$^\dagger$ Also with …`)
587        trim_space_after(document, &sup);
588        document.remove_node(sup);
589      },
590      Lead::Text(mut text, at) => {
591        let content = text.get_content();
592        let _ = text.set_content(content[at..].trim_start());
593      },
594    }
595    drop_emptied_wrappers(document, parent.into_iter().collect());
596  }
597}

599/// The copy of `piece` `note`'s printed text opens with ([`Lead`]): a superscript or a formula of scripts alone printing
600/// it, or, for a symbol, the head of its text where the mark ends there (`*`, not `**`) — a letter or a number in the
601/// text is the note's own word (`\thanks{a visitor …}` under `Ann Able$^{a}$`).
602fn leading_mark(note: &Node, piece: &str) -> Option<Lead> {
603  // (its first printed text node, or the formula it opens with)
604  fn first_ink(node: &Node) -> Option<Node> {
605    for child in node.get_child_nodes() {
606      match child.get_type() {
607        Some(NodeType::TextNode) if !child.get_content().trim().is_empty() => return Some(child),
608        Some(NodeType::ElementNode) if child.get_name() == "Math" => return Some(child),
609        Some(NodeType::ElementNode) if child.get_name() != "tags" => {
610          if let Some(ink) = first_ink(&child) {
611            return Some(ink);
612          }
613        },
614        _ => {},
615      }
616    }
617    None
618  }
619  let text = first_ink(note)?;
620  let key = mark_form(piece);
621  if text.get_type() == Some(NodeType::ElementNode) {
622    // (a formula of scripts alone printing the mark: `\thanks{$^{*}$Equal contribution}`)
623    return match fm_read_formula(&text) {
624      FmFormulaMarks::Marks(scripts) if mark_form(&scripts.join(",")) == key => {
625        Some(Lead::Sup(text))
626      },
627      _ => None,
628    };
629  }
630  // (inside a superscript of the note: that superscript, when it prints the mark alone)
631  let mut up = text.get_parent();
632  while let Some(parent) = up {
633    if parent == *note {
634      break;
635    }
636    if parent.get_name() == "sup" {
637      return (mark_form(&parent.get_content()) == key).then_some(Lead::Sup(parent));
638    }
639    up = parent.get_parent();
640  }
641  if piece.chars().any(char::is_alphanumeric) {
642    return None;
643  }
644  let content = text.get_content();
645  let head = content.trim_start();
646  let skipped = content.len() - head.len();
647  let glyph = head
648    .char_indices()
649    .find(|(at, _)| mark_form(&head[..*at]) == key && *at > 0)
650    .map(|(at, _)| at)
651    .or_else(|| (mark_form(head) == key).then_some(head.len()))?;
652  // (and not the head of a longer mark: `**`)
653  let next = head[glyph..].chars().next();
654  let continues = piece.chars().next_back().is_none_or(|last| {
655    next.is_some_and(|c| mark_form(&c.to_string()) == mark_form(&last.to_string()))
656  });
657  if continues {
658    return None;
659  }
660  Some(Lead::Text(text, skipped + glyph))
661}

663/// A mark-only note for `mark`, in `personname`'s namespace (Perl's `\footnotemark` note: `role="footnotemark"`),
664/// flagged `_namemark`: no `\footnotetext` of the body is its text (`relocate_footnote`). One of a binding's labels
665/// alone ([`LABEL_MARK`]) is classed `ltx_note_label`, which the page's ∗/* fold never links (OD #1052).
666fn mark_only_note(document: &Document, personname: &Node, mark: &str, label: bool) -> Result<Node> {
667  let mut note = Node::new("note", None, document.get_document())?;
668  if let Some(namespace) = personname.get_namespace() {
669    let _ = note.set_namespace(&namespace);
670  }
671  if label {
672    note.set_attribute("class", "ltx_note_label")?;
673  }
674  note.set_attribute("role", "footnotemark")?;
675  note.set_attribute("mark", mark)?;
676  note.set_attribute("_namemark", "true")?;
677  Ok(note)
678}

680/// The spaces a mark left at the name's edges, cut (`Conor Lawless <sup>*</sup>`, 2311.11099; `<sup>a,b</sup> Wolfgang
681/// Wieland`, 1912.09514; the line end between `Lorenzo Leal$^1$` and its `$^2$`, 1304.0778), and the zero-width spaces set
682/// between marks (`\up{\!1,2}`, 2209.14855).
683fn trim_name_edges(document: &mut Document, personname: &Node) {
684  fn blank(c: char) -> bool {
685    c.is_whitespace() || matches!(c, '\u{200B}' | '\u{2060}' | '\u{FEFF}')
686  }
687  fn edge_text(document: &mut Document, node: &Node, last: bool) -> Option<Node> {
688    let mut children = node.get_child_nodes();
689    if last {
690      children.reverse();
691    }
692    for child in children {
693      match child.get_type() {
694        Some(NodeType::TextNode) if child.get_content().trim_matches(blank).is_empty() => {
695          document.remove_node(child)
696        },
697        Some(NodeType::TextNode) => return Some(child),
698        Some(NodeType::ElementNode) if child.get_name() == "text" => {
699          return edge_text(document, &child, last);
700        },
701        _ => return None,
702      }
703    }
704    None
705  }
706  if let Some(mut text) = edge_text(document, personname, true) {
707    let content = text.get_content();
708    if content.trim_end_matches(blank).len() != content.len() {
709      let _ = text.set_content(content.trim_end_matches(blank));
710    }
711  }
712  if let Some(mut text) = edge_text(document, personname, false) {
713    let content = text.get_content();
714    if content.trim_start_matches(blank).len() != content.len() {
715      let _ = text.set_content(content.trim_start_matches(blank));
716    }
717  }
718}
