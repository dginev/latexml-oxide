1//! The observer's rehearsal: a block's stamped tokens digested as the article `\author` line prints them, with
2//! Layer-B event bindings (Perl's `@authorsplits`, Base_Utility.pool.ltxml:676-683: `\and` a column, `\\` a row; the
3//! notes and contacts as events), through the speculative-run engine (sect05.rs `title_page_probe_run`):
4//! assignments rolled back (`AssignmentJournalGuard`, a meaning's fallback standing with it: FM S2a.2 r9, 2409.10200),
5//! diagnostics held (`DiagnosticsHold`), terminal output muted, the frontmatter frozen (`lx_replaying_class_setter`:
6//! nothing is queued), the field ledger put back and the boxes it stamped unstamped (`FieldLedgerScope`: a register's
7//! box `\usebox` hands back, FM S2a.2 r9, 2405.14627, 2603.26792), progress notes silenced (an `\input` in the field:
8//! `SilenceProgressNotesGuard`). The
9//! digested list is handed to a callback before it is discarded. Not LaTeX's `\begin{tabular}[t]{c}` around the
10//! line: no alignment machinery runs, and a stray `&` is digested as a whatsit that errs only when built into the
11//! document (tex_tables.rs `&`), never here: nothing is held for it, and the lexer reads it as a column separator.
12//! What its boxes do not say is noted beside them (`field_notes::FieldNotes`): the definitions made, the errors held, by the
13//! field position of the invocation.

45/// Note, while a rehearsal digests, an entry `tag` its run would queue (`command`, the frozen frontmatter queues none:
46/// `queue::queue_front_matter`) that makes no creator nor annotates one (`resolver::creator_call`): a frontmatter setter run
47/// inside the block, a `\date` in `\author` (2503.02741), which `Rehearsal::Done::queued` says, as the queue would hold
48/// it (the emitter queues it at the flush: `emitter::emit::recover_queued`).
49pub(in crate::base_utilities) fn note_frozen_entry(
50  tag: &str,
51  attr: Option<&KeyVals>,
52  command: &Tokens,
53) {
54  if rehearsing() && !resolver::creator_call(tag, &annotated_parent(command)) {
55    let entry = (tag.to_string(), queued_attributes(attr), command.clone());
56    FROZEN_ENTRIES.with(|entries| entries.borrow_mut().push(entry));
57  }
58}

60/// Note, while a rehearsal digests, an entry `tag` its run digests where it stands (`\lx@add@frontmatter@until`: an
61/// abstract or keywords environment), undone with the run, which no queue holds: noted with no command, never queued again
62/// (`emitter::emit::recover_queued`). REVTeX 3's `\begin{abstract}` inside an `\address`'s minipage (cond-mat9801109).
63pub(in crate::base_utilities) fn note_frozen_digest(tag: &str) {
64  if rehearsing() {
65    let entry = (
66      tag.to_string(),
67      TagAttrs::default(),
68      Tokens::new(Vec::new()),
69    );
70    FROZEN_ENTRIES.with(|entries| entries.borrow_mut().push(entry));
71  }
72}

96/// The most tokens one rehearsal reads, as many as a run of the title-page check's (`title_page_check::CHECK_RUN_READS_DEFAULT`). By the
97/// data (the census, `FRONTMATTER_SINGLE_ENGINE_PLAN.md` "What a rollback leaves"): popZ01's 1,707 papers rehearsing read 166 at the median in their dearest rehearsal, 8,417 at the 99th percentile and
98/// 3,359,702 at most (2608.18287; 2011.00785 3,314,577); by what it costs, a looping block reads this in about two seconds.
99const REHEARSAL_READS_DEFAULT: usize = 8_000_000;
100
101/// The most tokens all the rehearsals of a conversion read, in [`REHEARSAL_READS_DEFAULT`]s: a paper rehearses per call. popZ01 reads 731 in a
102/// conversion at the median, 20,444 at the 99th percentile, and the most 3,360,403 (2608.18287; 2011.00785 3,314,871). The many-author witnesses are far under it:
103/// 1901.01540's 2,637 rehearsals read 118,996 (1,202 authors), 2103.04797's 2,136 read 92,093 (975), 2005.08745's 668 read 13,009 (592).
104const REHEARSALS_PER_CONVERSION: usize = 3;
105
106thread_local! {
107  /// The reads one rehearsal may make, this conversion's ([`rehearsal_budget`]).
108  static RUN_READS: Cell<usize> = const { Cell::new(REHEARSAL_READS_DEFAULT) };
109  /// The reads the conversion's rehearsals may still make.
110  static READS_LEFT: Cell<usize> =
111    const { Cell::new(REHEARSAL_READS_DEFAULT * REHEARSALS_PER_CONVERSION) };
112  /// How many rehearsals were abandoned or never made over budget.
113  static LEFT_OUT: Cell<usize> = const { Cell::new(0) };
114  /// The reads the content check's own rehearsals may still make ([`for_the_check`]): one rehearsal's, apart from the
115  /// conversion's.
116  static CHECK_READS_LEFT: Cell<usize> = const { Cell::new(REHEARSAL_READS_DEFAULT) };
117  /// The reads the argument queries (separators, marks) may still make, one shared allowance ([`for_a_query`]): one
118  /// rehearsal's, apart from the conversion's and the check's.
119  static QUERY_READS_LEFT: Cell<usize> = const { Cell::new(REHEARSAL_READS_DEFAULT) };
120}

122/// A new conversion's budget: [`REHEARSAL_READS_DEFAULT`], or `LATEXML_REHEARSAL_READ_BUDGET` (0: none).
123fn reset_rehearsal_budget() {
124  let reads = std::env::var("LATEXML_REHEARSAL_READ_BUDGET")
125    .ok()
126    .and_then(|reads| reads.trim().parse::<usize>().ok())
127    .unwrap_or(REHEARSAL_READS_DEFAULT);
128  let reads = if reads == 0 { usize::MAX >> 2 } else { reads };
129  RUN_READS.set(reads);
130  READS_LEFT.set(reads.saturating_mul(REHEARSALS_PER_CONVERSION));
131  CHECK_READS_LEFT.set(reads);
132  QUERY_READS_LEFT.set(reads);
133  LEFT_OUT.set(0);
134}

136/// Run `rehearsals` that read for the frontmatter content check alone (`reading::read_pending`'s calls a block's own
137/// expansion made, `ApiCall::inside_events`) from the check's own allowance, one rehearsal's reads ([`RUN_READS`]) for
138/// the conversion, never from the reads the output's rehearsals may make: a check never moves what is written (a later
139/// call's or a later flush's block over budget for it: revtex's `\collaboration{…}` after an `\author`, its block's own
140/// call, read before the next author's `\affiliation`, which lost its rehearsal). One it gives up on is in no budget
141/// Warning: nothing written depends on it, and the check counts that call by its content's text (`ApiCall::unread`).
142pub(in crate::base_utilities) fn for_the_check<R>(rehearsals: impl FnOnce() -> R) -> R {
143  apart_from_the_reading(&CHECK_READS_LEFT, rehearsals)
144}

146/// Run an argument query (`separators::read_argument`, which the separators and the marks queries share) from the
147/// queries' own allowance, one rehearsal's reads ([`RUN_READS`]) for the conversion, never from the reads the output's
148/// rehearsals may make nor in their count of calls left out: a query moves no reading (four costly queries before
149/// `\maketitle` once spent a small budget, and the block they preceded was written whole, its two authors one creator).
150/// One it gives up on answers `None`, which its caller reads by spelling as before the query (the marks query's
151/// fallback counted), in no budget Warning.
152pub(in crate::base_utilities) fn for_a_query<R>(rehearsals: impl FnOnce() -> R) -> R {
153  apart_from_the_reading(&QUERY_READS_LEFT, rehearsals)
154}

156/// Run `rehearsals` with `allowance` standing in for the conversion's reads ([`READS_LEFT`]), the count of calls left
157/// out ([`LEFT_OUT`]) put back after: [`for_the_check`], [`for_a_query`].
158fn apart_from_the_reading<R>(
159  allowance: &'static std::thread::LocalKey<Cell<usize>>,
160  rehearsals: impl FnOnce() -> R,
161) -> R {
162  let left_out = LEFT_OUT.get();
163  READS_LEFT.set(allowance.replace(READS_LEFT.get()));
164  let read = rehearsals();
165  allowance.set(READS_LEFT.replace(allowance.get()));
166  LEFT_OUT.set(left_out);
167  read
168}

170/// How many tokens the next rehearsal may read: a block's text may loop where nothing prints it (a first `\author` a second
171/// replaces, a class that drops it, `\author{First\loop}`), and a resource limit met here ends a conversion pdflatex finishes.
172/// None once the conversion's reads are spent. What is left out so is said once ([`rehearsal_budget_warning`]).
173pub(in crate::base_utilities) fn rehearsal_budget() -> Option<usize> {
174  let left = READS_LEFT.get();
175  (left != 0).then(|| left.min(RUN_READS.get()))
176}

178/// The Warning that says what no rehearsal read, when the document closes (`SkipReason::OverBudget`): an author block
179/// among them is written whole (Perl's whole-block creator, `writer::writer_stays_out` (deleted)) and any other call's words are not
180/// written — the content check counts them (`resolver::placed::content_check`, `ApiCall::unread`).
181pub(in crate::base_utilities) fn rehearsal_budget_warning() {
182  let left_out = LEFT_OUT.replace(0);
183  if left_out > 0 {
184    let read = "an author block among them is written whole, as Perl's whole-block creator, and any other call's words \
185                are not written, the frontmatter content check counting them";
186    Warn!(
187      "frontmatter",
188      "rehearsal_budget",
189      s!(
190        "The rehearsal gave up on {left_out} frontmatter call(s) that take more than {} reads to digest, or past the conversion's {}: {read}",
191        RUN_READS.get(),
192        RUN_READS.get().saturating_mul(REHEARSALS_PER_CONVERSION)
193      )
194    );
195  }
196}

198/// The event constructors of the Layer-B bindings ([`install_event_bindings`]): whatsits the lexer reads
199/// (`events::lex_events`), never absorbed (a rehearsal's list is discarded), so they construct nothing. The `\and`
200/// family keeps which command it was (`\AND` a new row in NeurIPS's and ICLR's `\author`); `\thanks`, `\footnote` and
201/// `\footnotemark` count the footnote steps they make in LaTeX (`FOOTNOTE_STEPS`), never stepping `\c@footnote`.
202pub(in crate::base_utilities) fn load() -> Result<()> {
203  // (each reverting as the command it is bound for, `alias`: a text made of an argument holding one — an argument read
204  // verbatim, svjour3's `\email{yjansson@kth.se \and tony@kth.se}`, 2106.06418; a link's target, lni's `\orcid`,
205  // 2106.11154, 1401.5951 — keeps its `\and`, `\\` or `\par`, never the rehearsal's name: review revN2)
206  DefConstructor!("\\lx@fm@obs@and", "", alias => "\\and");
207  DefConstructor!("\\lx@fm@obs@And", "", alias => "\\And");
208  DefConstructor!("\\lx@fm@obs@AND", "", alias => "\\AND");
209  DefConstructor!("\\lx@fm@obs@break OptionalMatch:* []", "", alias => "\\\\");
210  // (its own, for the Break to say so; a contact's raw argument holding it reads it as the break it is, `events::raw_lines`:
211  // 2106.11154's `\orcid`)
212  DefConstructor!("\\lx@fm@obs@par OptionalMatch:* []", "", alias => "\\par");
213  // `\thanks` and `\footnote` (inside `\maketitle`, `\footnote` is `\thanks`): latex.ltx `\thanks` = `\footnotemark`
214  // + the text, a step of the footnote counter unless an optional number is given
215  DefMacro!("\\lx@fm@obs@thanks []", sub[(number)] { note_step(T_CS!("\\lx@fm@obs@note"), number) });
216  // (the two tell apart what they make in an author's content, `digest_frontmatter_item`'s `Let`: `Event::footnote`)
217  DefMacro!("\\lx@fm@obs@footnote []", sub[(number)] { note_step(T_CS!("\\lx@fm@obs@fnnote"), number) });
218  // (its `printed` property the mark it shows on the name, `\thefootnote` at its number — the given one, or the counter's
219  // value the `\lx@fm@obs@thanks` step reached —, as `\lx@fm@obs@fnmark` prints its: a name's note mark is the request
220  // an entry led by the same `\footnotemark` answers, `Shun'ya Mizoguchi\footnote[1]{…}` under
221  // `\affiliation{\footnotemark[1]Theory Center, …}`, 1607.07280 — FM S2a.2 r7). Its text is text, read in text mode as
222  // the note's content is in the document (Perl's `\lx@annotate@frontmatter` digests it at the frontmatter, `DigestText`;
223  // LaTeX sets a footnote's text in a paragraph of its own): a note set in a formula, `Tianming Wei$^{2,3}\thanks{The
224  // first two authors contributed equally}$` (2407.15815; 2603.20851's `$^1\thanks{Corresponding Author}$`), kept its
225  // words' spaces, which math mode skipped ("Thefirsttwoauthorscontributedequally").
226  DefConstructor!("\\lx@fm@obs@note []{}", "", bounded => true, mode => "text", properties => sub[args] { note_properties(args.first().and_then(|a| a.as_ref())) });
227  DefConstructor!("\\lx@fm@obs@fnnote []{}", "", bounded => true, mode => "text", properties => sub[args] { note_properties(args.first().and_then(|a| a.as_ref())) });
228  // `\footnotetext`: the text of a mark printed elsewhere, no step
229  DefConstructor!("\\lx@fm@obs@notetext []{}", "", bounded => true, mode => "text");
230  // `\footnotemark[<n>]` prints `\thefootnote` at <n> (latex.ltx `\@xfootnotemark`), a bare `\footnotemark` at the
231  // counter's next value (`\stepcounter\@mpfn`): its `printed` property is that print, made with the counter set in a
232  // group of its own, to <n> or to its value at the rehearsal's start plus the steps the block made up to it, so the
233  // rehearsal steps nothing (the label is what the PDF prints when the counter at the flush is the one at
234  // `\maketitle`). A constructor, as the command it stands for: its box covers its optional argument's positions. A
235  // given `[<n>]` is read with `\c@footnote` at the rehearsal's start plus the steps the block made before it (local to
236  // the mark), as LaTeX reads it after the title's notes stepped the counter: `\footnotemark[\arabic{footnote}]`
237  // prints the mark of the last `\thanks` (2201.11183, 2111.13260; review rev78o2 finding 3), not of the unstepped
238  // counter.
239  DefMacro!("\\lx@fm@obs@footnotemark []", sub[(number)] {
240    // (the constructor token carries the field position of the `\footnotemark` it stands for, so the mark's box covers
241    // it as the command's own would)
242    let prov = get_current_token().map_or(0, |t| t.prov);
243    let mark = Token { prov, ..T_CS!("\\lx@fm@obs@fnmark") };
244    let Some(number) = number else {
245      return Ok(Tokens::new(vec![mark]));
246    };
247    let mut out = Vec::new();
248    let counted = FOOTNOTE_BASE_DEFINED.with(Cell::get);
249    if counted {
250      let value = FOOTNOTE_BASE
251        .with(Cell::get)
252        .saturating_add(FOOTNOTE_STEPS.with(Cell::get));
253      out.extend([T_CS!("\\begingroup"), T_CS!("\\c@footnote"), T_OTHER!("=")]);
254      out.extend(Explode!(value));
255      out.push(T_CS!("\\relax"));
256    }
257    out.extend([mark, T_OTHER!("["), T_BEGIN!()]);
258    out.extend(number.unlist());
259    out.extend([T_END!(), T_OTHER!("]")]);
260    if counted {
261      out.push(T_CS!("\\endgroup"));
262    }
263    Ok(Tokens::new(out))
264  });
265  DefConstructor!("\\lx@fm@obs@fnmark []", "", properties => sub[args] {
266    let given = args.first().and_then(|a| a.as_ref()).map(|number| {
267      let mut text = String::new();
268      ink_text(number, &mut text);
269      text.trim().to_string()
270    });
271    let value = match given {
272      Some(text) => match text.parse::<i64>() {
273        Ok(value) => value,
274        // (not a number: TeX's "Missing number"; the mark is what was given)
275        Err(_) => return Ok(stored_map!("printed" => Stored::String(pin(&text)))),
276      },
277      None => {
278        let steps = FOOTNOTE_STEPS.with(|steps| {
279          steps.set(steps.get().saturating_add(1));
280          steps.get()
281        });
282        FOOTNOTE_BASE.with(Cell::get).saturating_add(steps)
283      },
284    };
285    // (read quietly too: a bare `\footnotemark` under a class whose `\@fnsymbol` reads `\thefootnote` must not end the
286    // block's rehearsal — review revfms2a2_7 finding 4, jpconf.cls:153)
287    let printed = note_mark_quietly(value)?;
288    Ok(stored_map!("printed" => Stored::String(pin(&printed))))
289  });
290  // the marks the kernel `\maketitle` prints (latex.ltx/article.cls `\renewcommand\thefootnote{\@fnsymbol\c@footnote}`),
291  // unless a binding declares its title's own (`TITLE_FOOTNOTE_HOOK`)
292  DefMacro!("\\lx@fm@obs@thefootnote", "\\@fnsymbol\\c@footnote");
293  DefConstructor!("\\lx@fm@obs@mark {}", "");
294  // a binding's request of an annotation, printing nothing (elsarticle's `\corref{cor1}`, `\fnref{fn1}`:
295  // `\lx@request@frontmatter@annotation[cor]{cor1}`, elsart_support_core_sty.rs:58-60): a mark labelled
296  // `<prefix>:<label>` (S2a.2)
297  DefConstructor!("\\lx@fm@obs@request []{}", "");
298  DefConstructor!("\\lx@fm@obs@label Semiverbatim", "");
299  // (reverting as `\email`, as the `\and` family does: a note's text holding one, `\thanks{\email{…}}`, 1401.5951)
300  DefConstructor!("\\lx@fm@obs@email [] Semiverbatim", "", alias => "\\email");
301  // (an e-mail call by the class's binding printing in place, and by no binding's meaning: `ledger::email_meaning`)
302  DefConstructor!("\\lx@fm@obs@email@inplace [] Semiverbatim", "", alias => "\\email");
303  // (the label the class's own command prints before the address is read as it prints: `own_email_properties`; its
304  // address read as the class's macro reads its `#1`, unexpanded before it is digested — sig-alternate.cls:462 on
305  // `\email{\fontsize{11pt}{12pt}\selectfont …}`, whose `\@defaultunits` expanded ahead missed its `\@nnil` and `\@nil`
306  // ran undefined, 1802.01191: `UnexpandedSemiverbatim`)
307  DefConstructor!("\\lx@fm@obs@email@own [] UnexpandedSemiverbatim", "", alias => "\\email",
308    properties => sub[args] { own_email_properties(OWN_EMAIL[0].2, args.get(1).and_then(|a| a.as_ref())) });
309  DefConstructor!("\\lx@fm@obs@emailAdd@own [] UnexpandedSemiverbatim", "", alias => "\\email",
310    properties => sub[args] { own_email_properties(OWN_EMAIL[1].2, args.get(1).and_then(|a| a.as_ref())) });
311  DefConstructor!("\\lx@fm@obs@url Semiverbatim", "");
312  DefConstructor!("\\lx@fm@obs@href Semiverbatim {}", "");
313  DefConstructor!("\\lx@fm@obs@orcid [] Semiverbatim", "");
314  // (in a group of its own, as the API digests its entry, `digest_frontmatter_item`: a switch in one affiliation is
315  // no font of the next one's or of the names after it — `\affil{\itshape $^1$…}\affil{\textit{$^3$…}}`, whose
316  // second's italic read as no change of its own, `events::entry_font`; review rev94b r3 B1)
317  DefConstructor!("\\lx@fm@obs@affil []{}", "", bounded => true);
318  // (the affiliation of a raw file's in-place setter the API took, `annotate=run`: `frontmatter_stores`'
319  // `reroute_in_place_contact_setters`, whose rerouted setter keeps the file's own as `\lx@class@contact@<name>`)
320  DefConstructor!("\\lx@fm@obs@affil@run []{}", "", bounded => true);
321  // (its options undigested, as `\lx@annotate@frontmatter` reads keyvals: a private key's `_` digested as text is a Script
322  // error held on the block, the paper's writer kept out)
323  DefConstructor!("\\lx@fm@obs@annotate {}{} OptionalUndigested {}", "");
324  // (a binding's marked list, `\lx@add@affiliation@marked`, is an in-block affiliation: the marks and lines of its
325  // content read as digested, not cut by spelling and handed to the frozen API — `\IEEEauthorblockA{\IEEEauthorrefmark{6}
326  // Univ X\\ \IEEEauthorrefmark{2} Univ Y}` in a plain `\author`, 2603.28168, 1405.3507)
327  DefConstructor!("\\lx@fm@obs@affil@marked OptionalUndigested {}", "", bounded => true);
328  // (a binding's own creator inside a block prints its name there: PoS's `\speaker{R. Cassano}$^1$`, PoS.cls:681, 1412.5940,
329  // 1509.01480, 1211.4334 — the frozen API printed nothing, and the name's mark led the line. Not the `\lx@add@author` of an
330  // `\author` nested in a block, a call the class stores and prints no name of there: 2012.14750)
331  // (its keyvals, read as the API reads them, for the Name it prints, which says them nowhere: `\speaker`'s `role=speaker`,
332  // 0910.5686, 1007.2355, 1111.2428, 1111.6944 — written `role=author`; ams `\contrib`'s `name=`)
333  DefMacro!("\\lx@fm@obs@creator OptionalKeyVals:Frontmatter {}", sub[(keys, name)] {
334    if let Some(keys) = &keys {
335      note_creator_call(keys, &name)?;
336    }
337    Ok(name)
338  });
339  // (a creator call an API call's content makes, outside a block — amsart's `\thanks{… \author{Bob}\address{Univ B}
340  // \email{bob@b.org}}`, 0807.4720: Perl's digestion makes the creator there, and every `\address`/`\email` after it is
341  // that creator's, `\lx@annotate@frontmatter`, Base_Utility.pool.ltxml:520-531): the queueing sink, whichever meaning
342  // it has, called as it is, after the event it leaves, `EventKind::ApiCall` "ltx:creator" (`Resolver::api`'s
343  // `ltx:pubnote`)
344  DefConstructor!("\\lx@fm@obs@creatorcall", "");
345  DefMacro!("\\lx@fm@obs@add OptionalUndigested {}", sub[(keys, tag)] {
346    let mut out = Vec::new();
347    if tag.to_string().trim() == "ltx:creator" {
348      out.push(T_CS!(CREATOR_CALL));
349    }
350    out.push(T_CS!(SINK_MEANING));
351    if let Some(keys) = keys {
352      out.push(T_OTHER!("["));
353      out.extend(keys.unlist());
354      out.push(T_OTHER!("]"));
355    }
356    out.push(T_BEGIN!());
357    out.extend(tag.unlist());
358    out.push(T_END!());
359    Ok(Tokens::new(out))
360  });

425/// The text `print` reads from what it digests ([`note_mark_quietly`]'s reading, any print's): its diagnostics set
426/// aside, a print that fails unwound where it stopped and read as nothing, a resource Fatal that is no runaway kept.
427fn print_quietly(print: impl FnOnce() -> Result<String>) -> Result<String> {
428  let hold = util::logger::DiagnosticsHold::begin();
429  let latched_before = resource_fatal_latched();
430  let depth = get_frame_depth();
431  let boxes = box_lists_mark();
432  // (the tokens being invoked and the conditionals put back however the print ends, after a failed print's unwinding:
433  // `StomachMark`)
434  let stomach_mark = StomachMark::take();
435  let printed = print();
436  let failed = hold.errors_raised() > 0;
437  match printed {
438    Ok(printed) => {
439      stomach_mark.restore();
440      hold.discard();
441      Ok(if failed { String::new() } else { printed })
442    },
443    Err(err) => {
444      if resource_fatal_latched() && !latched_before && !unlatch_runaway_fatal() {
445        hold.commit();
446        return Err(err);
447      }
448      let unwound = reading_from_mouth(Mouth::default(), || {
449        new_local_box_list();
450        while get_frame_depth() > depth {
451          let before = get_frame_depth();
452          let nobox = lookup_bool("groupNonBoxing");
453          if pop_stack_frame(nobox).is_err() || get_frame_depth() >= before {
454            break;
455          }
456        }
457        expire_local_box_list();
458        Ok(())
459      });
460      unwind_box_lists_to(boxes);
461      stomach_mark.restore();
462      hold.discard();
463      unwound?;
464      Ok(String::new())
465    },
466  }
467}

527  }

548    .map(|(before, _)| before.trim().to_string())
549    .filter(|label| label.chars().any(char::is_alphabetic));
550  Ok(match label {
551    Some(label) => stored_map!("name" => Stored::String(pin(&label))),
552    None => SymHashMap::default(),
553  })

619/// The `\let` copies of an [`ALWAYS_BOUND`] command bound by its meaning that the block's tokens reach — spelled in it,
620/// or held in a macro of the document's own it spells (`\newcommand\aut{Ann \nx Bob}`, `\author{\aut}`) — each with the
621/// event constructor the command is bound to: a command whose meaning is a definition made for the command (`defined_as`
622/// it, so a `\relax` or an undefined name is no copy of a separator `\let` to `\relax` or left undefined), whatever the
623/// command means now — as the command itself is bound by its name, whatever its definition. A copy taken before the
624/// document renews the command keeps the command's separator or note in TeX (`\let\nx\and\renewcommand\and{…}`, two
625/// columns in pdflatex); a copy of a renewed `\and` is bound as the renewed `\and` is (read as its
626/// macro, a `\renewcommand\and{\\}`'s copy would make the second name the first's affiliation). Read before the
627/// bindings change any meaning; a name bound by its own name, or one the binding declares a block break, is left to
628/// those. The kernel's `\and` is a constructor printing " and " (`sect05.rs`): bound by name only, the copy kept it,
629/// the names merged into one creator (OC1's `\let` probe; article, eptcs, IEEEtran journal).
630fn let_copies_of_bound_commands(
631  tokens: &Tokens,
632  block_breaks: &[(&str, BreakKind)],
633) -> Vec<(Token, &'static str)> {
634  let mut found = Vec::new();
635  let commands: Vec<(Token, &'static str)> = ALWAYS_BOUND
636    .iter()
637    .filter(|(.., by_meaning)| *by_meaning)
638    .map(|(cs, target, _)| (T_CS!(*cs), *target))
639    .collect();
640  // (the distinct commands read: a block's and its macros', which may spell thousands)
641  let mut seen: HashSet<Token> = HashSet::default();
642  let mut pending: Vec<Token> = tokens.unlist_ref().iter().rev().copied().collect();
643  let mut macros = 0;
644  while let Some(t) = pending.pop() {
645    if !t.is_executable() || !seen.insert(t) {
646      continue;
647    }
648    let by_name =
649      t.with_cs_name(|name| bound_by_name(name) || block_breaks.iter().any(|(cs, _)| *cs == name));
650    if by_name {
651      continue;
652    }
653    if let Some((_, target)) = commands.iter().find(|(command, _)| t.defined_as(command)) {
654      found.push((t, *target));
655      continue;
656    }
657    if macros < COPY_SEARCH_MACROS
658      && let Ok(Some(defn)) = lookup_definition(&t)
659      && defn.is_expandable()
660      && defn.get_origin() == DefinitionOrigin::Document
661      && let Some(ExpansionBody::Tokens(body)) = defn.get_expansion()
662    {
663      macros += 1;
664      pending.extend(body.unlist_ref().iter().rev().copied());
665    }
666  }
667  found
668}

743/// `\\affiliation`'s `\\def\\ex{\\end{tabular}}` closing the tabular its `\\bx` opened, `perfect_kernel_batchz4`; an
744/// `\\author`'s `\\etab` closing the block's own) keeps its meaning, in any rehearsal; a command the binding declares a
745/// block break keeps its meaning. Before any of it, a `\let` copy of a separator or note command the block reaches is
746/// bound as the command is ([`let_copies_of_bound_commands`]: `\let\nx\and` parts the names).
747fn install_event_bindings(
748  tokens: &Tokens,
749  title_marks: TitleMarks,
750  block_breaks: &[(&str, BreakKind)],
751) -> Result<()> {
752  for (copy, target) in let_copies_of_bound_commands(tokens, block_breaks) {
753    let_i(&copy, &T_CS!(target), None);
754  }
755  for t in tokens.unlist_ref() {
756    let declared = || t.with_cs_name(|name| block_breaks.iter().any(|(cs, _)| *cs == name));
757    if let Some(between) = tabular_juggle_between(t)
758      && !declared()
759    {
760      let target = if prints_a_row_break(&between) {
761        ROW_BREAK
762      } else {
763        AUTHOR_AND
764      };
765      let_i(t, &T_CS!(target), None);
766    }
767  }
768  for (cs, target, _) in ALWAYS_BOUND {
769    let_i(&T_CS!(*cs), &T_CS!(*target), None);
770  }
771  if in_block_content() {
772    for (cs, target) in BOUND_IN_A_BLOCK {
773      let_i(&T_CS!(*cs), &T_CS!(*target), None);
774    }
775  } else {
776    let_i(&T_CS!(SINK_MEANING), &T_CS!(SINK), None);
777    let_i(&T_CS!(SINK), &T_CS!("\\lx@fm@obs@add"), None);
778  }
779  if lookup_definition(&T_CS!(TITLE_FOOTNOTE_HOOK))?.is_some() {
780    let_i(&T_CS!(THEFOOTNOTE), &T_CS!(TITLE_FOOTNOTE_HOOK), None);
781  } else if title_marks == TitleMarks::Fnsymbol
782    && lookup_definition(&T_CS!("\\@fnsymbol"))?.is_some()
783  {
784    let_i(&T_CS!(THEFOOTNOTE), &T_CS!("\\lx@fm@obs@thefootnote"), None);
785  }
786  for (cs, target, needs_argument) in BOUND_WHEN_DEFINED {
787    if !block_breaks.iter().any(|(declared, _)| declared == cs)
788      && lookup_definition(&T_CS!(*cs))?.is_some_and(|defn| {
789        (!needs_argument || defn.get_num_args() > 0)
790          && required_arguments(&*defn) <= binding_required_arguments(target)
791      })
792    {

874/// The juggle written inline: an `\end{tabular}` closing a tabular the block never opened, reopened after nothing but
875/// spaces and commands with their `[…]` and `{…}` arguments (what a class prints between, `\hfil`, `\newline`,
876/// `\\[2mm]`, `\vspace{2mm}`), the reopening `\begin{tabular}[…]{…}` its part. With no line break printed between it is the column `\alignauthor` is (sig-alternate.cls:274 `\end{tabular}%
877/// \begin{tabular}[t]{p{\auwidth}}\centering`, bound to an author separator, [`install_event_bindings`]): 1401.6981
878/// writes its second author's column so, `\end{tabular}\begin{tabular}[t]{p{1.3\auwidth}}\centering Gianmarco De
879/// Francisci Morales…`, three columns in the PDF. With one printed between (`\end{tabular}\newline\begin{tabular}{c}`,
880/// 1301.5612's macro written out; `\\[2mm]`, review revG3 finding 2, "Univ B" lost with no Warning before; a `\vskip`,
881/// [`prints_a_row_break`]) that break stands; a skip in the line is no break (`\hskip 1em`, `\vspace{2mm}`, two columns
882/// in the PDF). Digested as written it closed the title page's tabular and left
883/// one open, whose lines no box held — at `=2` authors 2 and 3 of 1401.6981 lost with no Warning. A close of a tabular
884/// the block opened (`\begin{tabular}…\end{tabular}`) or one no reopen follows keeps its meaning; with the emitter's
885/// switch on.
886fn inline_juggles_as_columns(tokens: Tokens) -> Tokens {
887  let (end, begin) = environment_commands();
888  let toks = tokens.unlist_ref();
889  if !toks.contains(&end) {
890    return tokens;
891  }
892  let optional = |k: usize| past_optional(toks, k);
893  let braced = |k: usize| past_braced(toks, k);
894  // The span past the reopen's `\begin{tabular}[…]{…}`, if one follows `at` after only spaces and commands, each with
895  // the `[…]` and `{…}` arguments right after it (a bare group, `\end{tabular} {Extra Words} \begin{tabular}`, is no
896  // command's: the juggle is read as written, review revH1 finding 2).
897  let argument = |k: usize| {
898    optional(k)
899      .or_else(|| braced(k))
900      .filter(|&(first, _)| first == k)
901      .map(|(_, end)| end)
902  };
903  let (space, command) = (Catcode::SPACE, Catcode::CS);
904  let (glue, dimen) = ([T_CS!("\\hskip"), T_CS!(VERTICAL_SKIP)], T_CS!("\\kern"));
905  let reopen = |at: usize| -> Option<(usize, usize)> {
906    let mut start = at;
907    loop {
908      let c = toks.get(start)?.get_catcode();
909      if c != space && (c != command || toks[start] == begin) {
910        break;
911      }
912      let cs = toks[start];
913      start += 1;
914      if c == command {
915        // What the command reads past spaces, as TeX reads it: `\\`'s `[…]` (its `*` right after it), by `\@ifstar` and
916        // `\@ifnextchar` (latex.ltx `\@normalcr`/`\@xnewline`, `\@tabularcr`/`\@xtabularcr`; `\\ [2mm]`, review
917        // revH4 finding 1, "Univ B" lost with no Warning before), and a glue or dimen spec written out
918        // (`\hskip 1em`, h3e). A `{…}` comes right after its command (`argument`).
919        if let Some(k) = past_line_break(toks, start - 1) {
920          start = k;
921        } else if glue.contains(&cs) {
922          start = past_glue(toks, start).unwrap_or(start);
923        } else if cs == dimen {
924          start = past_dimen(toks, start, false).unwrap_or(start);
925        }
926        while let Some(k) = argument(start) {
927          start = k;
928        }
929      }
930    }
931    Some((start, past_tabular_open(toks, start)?))
932  };
933  let mut out = Vec::with_capacity(toks.len());
934  let mut opened = 0usize;
935  let mut i = 0;
936  while i < toks.len() {
937    if let Some(len) = environment_call_len(&toks[i..], begin, "tabular") {
938      opened += 1;
939      out.extend_from_slice(&toks[i..i + len]);
940      i += len;
941    } else if let Some(len) = environment_call_len(&toks[i..], end, "tabular") {
942      match (opened, reopen(i + len)) {
943        (0, Some((start, past))) => {
944          // (what the commands between print goes with the column before, as the old driver's output orders it:
945          // `\mbox{Extra Words}` after Univ A, not the next column's first line, Bob Two lost with 2 Errors before —
946          // review revH1 finding 2; the writer still writes the column's source, `\end{tabular}` in it, so the words
947          // between are lost, counted by the content check: OD #980 (5))
948          let between = &toks[i + len..start];
949          out.extend_from_slice(between);
950          // (a bound break between is the break event itself; a vertical skip, which digests to none, is followed
951          // by one: `\vskip 2mm`, review revI1 finding 3, read as a column before)
952          if !prints_a_bound_break(between) {
953            let target = if prints_a_row_break(between) {
954              ROW_BREAK
955            } else {
956              AUTHOR_AND
957            };
958            out.push(Token {
959              prov: toks[i].prov,
960              ..T_CS!(target)
961            });
962          }
963          i = past;
964        },
965        _ => {
966          opened = opened.saturating_sub(1);
967          out.extend_from_slice(&toks[i..i + len]);
968          i += len;
969        },
970      }
971    } else {
972      out.push(toks[i]);
973      i += 1;
974    }
975  }
976  Tokens::new(out)
977}

979/// The position past a glue spec written out from `at` (TeX's `scan_glue`, tex.web §461: a dimen, then `plus` and
980/// `minus` ones that may be infinite), if one is; a register or anything else is not read.

983  for word in ["plus", "minus"] {
984    if let Some(after) = past_keyword(toks, k, word) {
985      k = past_dimen(toks, after, true)?;
986    }
987  }

991/// The position past a dimen written out from `at`: optional signs, digits with a `.` or `,`, and a unit (tex.web
992/// §448-458: `true`, a physical or font unit, `fil`…`filll` where `infinite`); the optional space after it is the
993/// caller's, which skips spaces.
994fn past_dimen(toks: &[Token], at: usize, infinite: bool) -> Option<usize> {
995  let other = |k: usize| {
996    let t = toks.get(k).filter(|t| t.get_catcode() == Catcode::OTHER);
997    t.and_then(|t| char::from_u32(t.get_charcode()))
998  };
999  let mut k = non_space(toks, at)?;
1000  while matches!(other(k), Some('+' | '-')) {
1001    k = non_space(toks, k + 1)?;
1002  }
1003  let digits = k;
1004  while matches!(other(k), Some('0'..='9' | '.' | ',')) {
1005    k += 1;
1006  }
1007  if k == digits {
1008    return None;
1009  }
1010  if infinite && let Some(mut k) = past_keyword(toks, k, "fil") {
1011    for _ in 0..2 {
1012      k = past_keyword(toks, k, "l").unwrap_or(k);
1013    }
1014    return Some(k);
1015  }
1016  let k = past_keyword(toks, k, "true").unwrap_or(k);
1017  [
1018    "em", "ex", "pt", "pc", "in", "bp", "cm", "mm", "dd", "cc", "sp", "px",
1019  ]
1020  .iter()
1021  .find_map(|unit| past_keyword(toks, k, unit))
1022}

1076/// The spans a column spec's tokens `spec` print (latex.ltx `\@mkpream`, array.sty `\@mkpream`): the text of `@{…}`
1077/// and `!{…}` between columns and of `>{…}` and `<{…}` in each cell (pdflatex: `{>{Prof }c<{ Dr}!{ or }c}` prints
1078/// "Prof Dept Dr or"), and those in a `*{n}{…}`'s repeated spec; a column letter, its argument (`p{3cm}`) and the
1079/// count `n` print nothing. `\@mkpream` compares the spec's characters by code (`\if`), so a letter `@` under
1080/// `\makeatletter` is read as an other one (review revI4 finding 2: `{@{Inst }c}` there printed "Inst" uncounted).
1081fn spec_printed(toks: &[Token], spec: Range<usize>, printed: &mut Vec<Range<usize>>) {
1082  let mut k = spec.start;
1083  while k < spec.end {
1084    let token = &toks[k];
1085    let group = |from: usize| past_braced(&toks[..spec.end], from);
1086    if matches!(token.get_catcode(), Catcode::OTHER | Catcode::LETTER)
1087      && let Some(c @ ('@' | '!' | '>' | '<' | '*')) = char::from_u32(token.get_charcode())
1088      && let Some((open, close)) = group(k + 1)
1089    {
1090      k = close;
1091      if c != '*' {
1092        printed.push(open + 1..close - 1);
1093      } else if let Some((open, close)) = group(k) {
1094        spec_printed(toks, open + 1..close - 1, printed);
1095        k = close;
1096      }
1097    } else if let Some((_, close)) = group(k).filter(|&(open, _)| open == k) {
1098      k = close;
1099    } else {
1100      k += 1;
1101    }
1102  }
1103}

1257/// [`rehearse`] of an API call's `content`, lexed by [`lex_events`]: the events of the same content rehearsed before in
1258/// this reading (`seen`) when that run left nothing behind — no diagnostic, conditional, footnote step, definition,
1259/// environment or queued entry, no group open — so the state each run's rollback restores decides them alike; its reads
1260/// counted again, from the same budget (1901.01540's 1,427 `\affiliation`s, 228 contents; 1710.09320's 1,257, 160;
1261/// 2103.04797's 1,161 REVTeX `\affiliation{\Amsterdam}`-style macros).
1262pub(in crate::base_utilities) fn rehearse_call(
1263  seen: &mut CallRehearsals,
1264  content: Tokens,
1265  title_marks: TitleMarks,
1266) -> Result<Rehearsal<Vec<Event>>> {
1267  let key: Vec<_> = content
1268    .unlist_ref()
1269    .iter()
1270    .map(|t| (t.text, t.code, t.prov))
1271    .collect();
1272  if let Some((events, reads)) = seen.0.get(&key)
1273    && rehearsal_budget().is_some_and(|budget| budget >= *reads)
1274  {
1275    let (footnotes, counter, carried) = footnote_start()?;
1276    set_footnote_base(footnotes, counter + carried);
1277    READS_LEFT.set(READS_LEFT.get() - reads);
1278    count_reads(*reads);
1279    if footnotes {
1280      set_footnotes_carried((counter, carried));
1281    }
1282    DebugFeature!("frontmatter", "FRONT a rehearsal read {reads} tokens");
1283    return Ok(Rehearsal::Done {
1284      value:        events.clone(),
1285      errors:       0,
1286      warnings:     0,
1287      declarations: Vec::new(),
1288      breaks:       Vec::new(),
1289      held:         Vec::new(),
1290      environments: Vec::new(),
1291      creators:     Vec::new(),
1292      left_open:    false,
1293      unsaid:       None,
1294      queued:       Vec::new(),
1295      expansions:   Vec::new(),
1296    });
1297  }
1298  let left = READS_LEFT.get();
1299  let progress = token_progress();
1300  let conditionals = (if_count(), if_stack_depth());
1301  let rehearsal = rehearse(content, title_marks, &[], lex_events)?;
1302  if let Rehearsal::Done {
1303    value,
1304    errors: 0,
1305    warnings: 0,
1306    declarations,
1307    breaks,
1308    held,
1309    environments,
1310    creators,
1311    left_open: false,
1312    unsaid: None,
1313    queued,
1314    expansions,
1315  } = &rehearsal
1316    && declarations.is_empty()
1317    && breaks.is_empty()
1318    && held.is_empty()
1319    && environments.is_empty()
1320    && creators.is_empty()
1321    && queued.is_empty()
1322    && expansions.is_empty()
1323    && (if_count(), if_stack_depth()) == conditionals
1324    && FOOTNOTE_STEPS.with(Cell::get) == 0
1325  {
1326    let reads = left - READS_LEFT.get();
1327    if token_progress() == progress + reads {
1328      seen.0.insert(key, (value.clone(), reads));
1329    }
1330  }
1331  Ok(rehearsal)
1332}

1334/// Digest `tokens` (a block's stamped tokens, `ledger::field_stamped`) under the event bindings and hand the
1335/// digested list to `inspect` before discarding it; nothing of it stays. A resource Fatal (the deadline, the memory
1336/// budget, a runaway limit) propagates, as does one whose run could not be undone (Fatal stays Fatal); any other Fatal
1337/// is the rehearsal's alone, discarded with its diagnostics.
1338pub(in crate::base_utilities) fn rehearse<R>(
1339  tokens: Tokens,
1340  title_marks: TitleMarks,
1341  block_breaks: &[(&str, BreakKind)],
1342  inspect: impl FnOnce(&Digested) -> R,
1343) -> Result<Rehearsal<R>> {
1344  let (footnotes, counter, carried) = footnote_start()?;
1345  let base = counter + carried;
1346  let Some(journal) = begin_assignment_journal() else {
1347    return Ok(Rehearsal::Skipped(SkipReason::JournalUnavailable));
1348  };
1349  let Some(budget) = rehearsal_budget() else {
1350    journal.rollback();
1351    LEFT_OUT.set(LEFT_OUT.get() + 1);
1352    return Ok(Rehearsal::Skipped(SkipReason::OverBudget));
1353  };
1354  let _speculative = speculative_digestion();
1355  set_footnote_base(footnotes, base);
1356  const FLAGS: [&str; 3] = [
1357    "lx_replaying_class_setter",
1358    "lx_depositing_frontmatter_fields",
1359    "lx_depositing_class_maketitle",
1360  ];
1361  let saved: Vec<bool> = FLAGS.iter().map(|flag| lookup_bool(flag)).collect();
1362  for flag in FLAGS {
1363    assign_value(flag, true, Some(Scope::Global));
1364  }
1365  let hold = util::logger::DiagnosticsHold::begin();
1366  begin_meaning_log();
1367  let resource_fatal_before = resource_fatal_latched();
1368  let muted = SuppressLogOutputGuard::new(true);
1369  // (an `\input` in the field opens its file again: its `(Processing content …` note is not the conversion's)
1370  let silent = SilenceProgressNotesGuard::new();
1371  let depth = get_frame_depth();
1372  let boxes = box_lists_mark();
1373  // (the tokens being invoked put back however the run ends; the conditionals are the sequence's, put back by the
1374  // reading after the last block — a block's run may open one the next block's closes, as TeX reads the list in one
1375  // pass, 2606.17718 — and only those of a failed run here: `StomachMark`)
1376  let tokens_depth = token_stack_depth();
1377  let conditionals = if_stack_depth();
1378  let ledger_scope = FieldLedgerScope::begin();
1379  FROZEN_ENTRIES.with(|entries| entries.borrow_mut().clear());
1380  REHEARSING.with(|r| r.set(true));
1381  FOOTNOTE_STEPS.with(|steps| steps.set(0));
1382  // The whole group in a reading context of its own: what its end unreads (an `\aftergroup` of the block's,
1383  // `pop_stack_frame_with`) lands in this mouth, closed with whatever is left (`reading_from_mouth`'s forced close),
1384  // never in the document's input; and the token the group's end and its diagnostics see is the rehearsal's own, not
1385  // the document's current one (a `}` closing an alignment template's brace, `egroup`'s `close_template_brace`).
1386  // Guards: the repro `sectioning-frontmatter/observer_rehearsal_aftergroup_stays_inside` (its `\aftergroup\leak`
1387  // printed "LEAKED" into the body before this wrap) and `perfect_kernel_batchfms2a1::
1388  // the_rehearsal_leaves_nothing_behind` (no `LEAKED` in any adversarial shape's output, each shape's body text and
1389  // counts pinned).
1390  local_current_token(T_CS!("\\relax"));
1391  let _hook = FieldNotesHookGuard;
1392  let if_count_before = if_count();
1393  let budget = read_budget(budget);
1394  let tokens = inline_juggles_as_columns(tokens);
1395  let result = with_origin(DefinitionOrigin::File, || {
1396    reading_from_mouth(Mouth::default(), || -> Result<Digested> {
1397      bgroup();
1398      install_event_bindings(&tokens, title_marks, block_breaks)?;
1399      begin_mode("text")?;
1400      field_notes_begin(&tokens, block_breaks);
1401      let digested = digest(tokens)?;
1402      end_mode("text")?;
1403      egroup()?;
1404      Ok(digested)
1405    })
1406  });
1407  // (off before anything else reads: closing what the run left open is no part of it)
1408  let budget = budget.end();
1409  let left_open = get_frame_depth() > depth;
1410  READS_LEFT.set(READS_LEFT.get().saturating_sub(budget.reads));
1411  if budget.over {
1412    set_if_count(if_count_before);
1413  }
1414  let unwound = reading_from_mouth(Mouth::default(), || {
1415    new_local_box_list();
1416    while get_frame_depth() > depth {
1417      let before = get_frame_depth();
1418      let nobox = lookup_bool("groupNonBoxing");
1419      if pop_stack_frame(nobox).is_err() || get_frame_depth() >= before {
1420        break;
1421      }
1422    }
1423    expire_local_box_list();
1424    Ok(())
1425  });
1426  truncate_token_stack(tokens_depth);
1427  expire_current_token();
1428  let notes = field_notes_end();
1429  let value = match &result {
1430    Ok(digested) => Some((
1431      inspect(digested),
1432      declaration_events(&notes.declarations),
1433      break_events(&notes.breaks),
1434    )),
1435    Err(_) => None,
1436  };
1437  REHEARSING.with(|r| r.set(false));
1438  drop(ledger_scope);
1439  drop(silent);
1440  drop(muted);
1441  let errors = hold.errors_raised();
1442  let warnings = hold.warnings_raised();
1443  // (each held message is stripped of its colors to be read: only for a block that held an error)
1444  let held = if errors > 0 {
1445    held_errors(&notes.errors, errors, hold.held_errors())
1446  } else {
1447    Vec::new()
1448  };
1449  let queued = FROZEN_ENTRIES.with(|entries| std::mem::take(&mut *entries.borrow_mut()));
1450  let undone = journal.rollback();
1451  // (a printing call's definitions that the block introduces, undefined again once undone: the class's own state such a call
1452  // sets — PoS's `\global\@speakertrue` in a paper's `\speaker`, 1009.6192; manyfoot's `\@thefnmark`, 2511.01971 — is the
1453  // class's to read, as `\maketitle` does)
1454  let printing: Vec<PrintingCall> = notes
1455    .printing
1456    .into_iter()
1457    .filter_map(|mut call| {
1458      call.definees.retain(|&key| !has_meaning_key(key));
1459      (!call.definees.is_empty()).then_some(call)
1460    })
1461    .collect();
1462  for (flag, value) in FLAGS.iter().zip(saved) {
1463    assign_value(flag, value, Some(Scope::Global));
1464  }
1465  // (past the hold's end: `hold` keeps it otherwise)
1466  let reads = budget.reads;
1467  if footnotes {
1468    set_footnotes_carried((counter, carried + FOOTNOTE_STEPS.with(Cell::get)));
1469  }
1470  // (a block's own, never a Fatal, even when its assignments could not be undone: its diagnostics and those stay, as a failed
1471  // run's: `sect05.rs` `speculative_run_read`)
1472  if budget.over {
1473    if undone {
1474      hold.discard();
1475    } else {
1476      hold.commit();
1477    }
1478    DebugFeature!("frontmatter", "FRONT a rehearsal read {reads} tokens");
1479    LEFT_OUT.set(LEFT_OUT.get() + 1);
1480    unwound?;
1481    unwind_box_lists_to(boxes);
1482    while if_stack_depth() > conditionals {
1483      pop_if_frame();
1484    }
1485    return Ok(Rehearsal::Skipped(SkipReason::OverBudget));
1486  }
1487  if let Err(err) = result {
1488    if !undone || (resource_fatal_latched() && !resource_fatal_before && !unlatch_runaway_fatal()) {
1489      hold.commit();
1490      return Err(err);
1491    }
1492    hold.discard();
1493    DebugFeature!("frontmatter", "FRONT a rehearsal read {reads} tokens");
1494    unwound?;
1495    unwind_box_lists_to(boxes);
1496    while if_stack_depth() > conditionals {
1497      pop_if_frame();
1498    }
1499    return Ok(Rehearsal::Skipped(SkipReason::Fatal));
1500  }
1501  if undone {
1502    hold.discard();
1503  } else {
1504    // (its diagnostics and its global assignments stay, as a speculative run's that could not be undone)
1505    hold.commit();
1506  }
1507  DebugFeature!("frontmatter", "FRONT a rehearsal read {reads} tokens");
1508  unwound?;
1509  Ok(match value {
1510    Some((value, declarations, breaks)) if undone => Rehearsal::Done {
1511      value,
1512      errors,
1513      warnings,
1514      declarations,
1515      breaks,
1516      held,
1517      environments: notes.environments,
1518      creators: notes.creators,
1519      left_open,
1520      unsaid: Unsaid {
1521        counter_moves: notes.counter_moves,
1522        printing,
1523      }
1524      .boxed(),
1525      queued,
1526      expansions: notes.expansions,
1527    },
1528    _ => Rehearsal::Skipped(SkipReason::NotUndone),
1529  })
1530}

1532/// The errors a rehearsal held: each of its `errors` raised, in order, at the position noted (`noted`: the invocations
1533/// that raised them, each with how many), the rest past the last field token, each with its log line.
1534fn held_errors(noted: &[(u16, u16, usize)], errors: usize, lines: Vec<String>) -> Vec<HeldError> {
1535  let spans = noted
1536    .iter()
1537    .flat_map(|&(start, end, n)| std::iter::repeat_n(Some((start, end)), n))
1538    .chain(std::iter::repeat(None));
1539  let mut lines = lines.into_iter();
1540  spans
1541    .take(errors)
1542    .map(|span| HeldError {
1543      span,
1544      message: lines.next().unwrap_or_default(),
1545    })
1546    .collect()
