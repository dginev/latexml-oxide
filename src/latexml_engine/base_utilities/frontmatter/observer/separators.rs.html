1//! A binding's query on an argument's author separators, by meaning (Option C step P2-0, OXIDIZED_DESIGN_DIVERGENCES #995):
2//! the `\and`-family separators an argument holds at its own level, as the rehearsal digests it to events
3//! (`EventKind::Sep(SepKind::And)` at nest 0: literal, held in a macro of the document's, a `\let` copy, a `\csname
4//! and\endcsname`), each with whether its meaning *is* the kernel separator's. The rehearsal binds `\and`, `\And` and
5//! `\AND` to its separator event whatever they mean here (`rehearsal::ALWAYS_BOUND`), so a `Sep` alone does not say the
6//! class parts names there (trap T8 of the P2 plan): a class's `\and` defined where the binding calls may print a word or
7//! a `&` (smfart's, 2508.07946), and a binding's own definition stands for whatever its class prints. The meaning is read
8//! outside the rehearsal, the one in force at the call site, where the binding calls the query (at `\author`,
9//! `\institute`): a class that defines or `\let`s `\and` only inside its title macro is read as the kernel's there, as
10//! `frontmatter_stores::kernel_and_list` reads it — jmlr.cls's word `\and` (jmlr.cls:336, inside `\jmlrpreauthor`,
11//! which its `\maketitle` runs) answers `kernel` before `\maketitle` (RED repro `separators_query_jmlr_title_time_and_red`;
12//! a reading of the title-time meaning is a mechanism of its own). The query rehearses from its own allowance (`rehearsal::for_a_query`),
13//! never the reading's budget nor its count of calls left out.
14//!
15//! Two front-ends of the one query: [`argument_separators`] for a binding's Rust code, and `\lx@fm@separators{<tokens>}`
16//! for a binding's macro text, expanding to the meanings' names in order, comma-separated (`kernel`, `binding`, `raw`),
17//! empty for none, `unread` where no rehearsal could be made or it left a group open.

30/// What a separator means where the query is made.
31#[derive(Debug, Clone, Copy, PartialEq, Eq)]
32pub enum SepMeaning {
33  /// the kernel's `\and` (`latex_constructs::sect05`'s constructor, which `\And` and `\AND` are `\let` to there, as are a
34  /// binding's `\let` copies: `inst_support_sty.rs:66` `\AND`, `ijcai_sty.rs:33-34` `\AND`/`\And`, `epl_cls.rs:109`
35  /// `\And` — their definition is `\and`'s own, so they read `Kernel`, not `Binding`): the
36  /// kernel's author-line column, which the blocks' readings part names at. The definition must be `\and`'s own (its
37  /// `get_cs()`), a kernel layer's: narrower than the pool-constructor test `frontmatter_stores::kernel_and_list` made
38  /// before P2-6, which now asks this (`texvc_sty.rs:132`'s `\let\and\land` is `raw`, no kernel separator; that test
39  /// is the gate's reading only where the query answers `None`)
40  Kernel,
41  /// a binding's own definition (its code, a fallback class's, a stand-in's or a borrowing's,
42  /// `DefinitionOrigin::is_binding_made`), whatever it prints: aas_support's real `\and` (`aas_support_sty.rs:358`), the
43  /// stand-ins for a class's column break printing a space or nothing (colm2025's `\And`/`\AND`, jmlr's and jmlr2e's
44  /// `\And`, jair's `\hskip`, neurips's no-ops), a word (ims_support's `\AND`), a `&` (elsart_support's `\AND`), and
45  /// amsmath's math `\And` (no separator at all). So `Binding` alone says neither "a separator" nor "none": the P2
46  /// callers' gates ask kernel or not (`kernel_and_list`'s swap, aa, l_aa, ecai, authblk), which needs no split; a caller
47  /// that must tell them apart reads its own binding's definition
48  Binding,
49  /// anything else: a raw class's, a package's or the document's own definition (a word, a `&`, a space), or no
50  /// separator's meaning at all (`\relax`, undefined)
51  Raw,
52}
53
54impl SepMeaning {
55  /// The name `\lx@fm@separators` gives it.
56  pub fn name(self) -> &'static str {
57    match self {
58      SepMeaning::Kernel => "kernel",
59      SepMeaning::Binding => "binding",
60      SepMeaning::Raw => "raw",
61    }
62  }
63}

65/// One separator of an argument at its own level ([`argument_separators`]).
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67pub struct ArgSeparator {
68  /// the command of the family the rehearsal read (`\and`, `\And`, `\AND`: `SepKind::And`)
69  pub command: &'static str,
70  /// the meaning of the token that printed it, where the query is made
71  pub meaning: SepMeaning,
72  /// its own tokens in the argument, `start..end` (a literal `\and`, a macro holding only the separator, a `\let` copy,
73  /// a `\csname and\endcsname`; a `\relax` before it, which prints nothing; never an environment closed right before
74  /// it — an `\end{tabular}`, `\end{minipage}`, a nested table, a user environment — which is the environment's,
75  /// `own_run`), where they print nothing else: no name, mark, contact or note's span meets them, the
76  /// criterion of the emitter's cut (`emit::cut_at_separators`, #993). `None` where the span is unread or the tokens
77  /// print ink too (a macro holding names and the separator together, `\newcommand\aut{Ann \and Bob}`): no cut by
78  /// tokens parts them, the reading's cut does. Its callers part an argument there: P2-4's eptcs cells
79  /// (OXIDIZED_DESIGN_DIVERGENCES #991 (8)) and the queue's name reader (`queue_reading::entry_names`, #993 P2-1b)
80  pub span:    Option<(usize, usize)>,
81}

83/// The kinds of meaning a definition origin says, for a meaning that is not the kernel separator's.
84fn made_by(origin: DefinitionOrigin) -> SepMeaning {
85  if origin.is_binding_made() {
86    SepMeaning::Binding
87  } else {
88    SepMeaning::Raw
89  }
90}

92/// The kernel layer's separator, `latex_constructs/sect05.rs`'s `\and` constructor: the definition a kernel meaning must be
93/// (`\And` and `\AND` are `\let` to it), the one `frontmatter_stores::kernel_and_list` asks for (P2-6).
94const KERNEL_SEPARATOR: &str = "\\and";
95
96/// Whether `definition` is [`KERNEL_SEPARATOR`]'s (a `\let` copy's too: its own cs) — a whatsit of it prints no text
97/// (`util_text::printed_own_text`). It reads the cs alone, so a binding's `\and` constructor counts too (`meaning_of`
98/// adds the origin).
99pub(in crate::base_utilities) fn is_kernel_separator(definition: &dyn Definition) -> bool {
100  *definition.get_cs() == T_CS!(KERNEL_SEPARATOR)
101}

103/// What `token` means now ([`SepMeaning`]): the kernel separator is the [`KERNEL_SEPARATOR`] constructor a kernel layer
104/// made (a pool, a dump, a format: `is_latexml_owned` and not `is_binding_made`).
105fn meaning_of(token: &Token) -> SepMeaning {
106  match lookup_meaning(token) {
107    Some(Stored::Constructor(defn)) => {
108      let kernel = defn.origin.is_latexml_owned() && !defn.origin.is_binding_made();
109      if kernel && is_kernel_separator(defn.as_ref()) {
110        SepMeaning::Kernel
111      } else {
112        made_by(defn.origin)
113      }
114    },
115    Some(Stored::Expandable(defn)) => made_by(defn.get_origin()),
116    Some(Stored::Primitive(defn)) => made_by(defn.origin),
117    Some(Stored::Conditional(defn)) => made_by(defn.origin),
118    _ => SepMeaning::Raw,
119  }
120}

122/// The token whose meaning a separator has, its own run `span` ([`own_run`]): the one token the span covers (a literal `\and`,
123/// a `\let` copy, an active character) where it is no macro; else the family's command the rehearsal read (`command`) —
124/// a macro holding the separator, whose call the span covers (`\newcommand\aut{Ann \and Bob}`), a class's macro `\and`
125/// itself or a copy of it (`rehearsal::let_copies_of_bound_commands` binds a copy only where it means what the command
126/// means), a `\csname and\endcsname` (which names the command), a span not read.
127fn printing_token(content: &[Token], span: Option<(u16, u16)>, command: &'static str) -> Token {
128  let run = span.and_then(|(s, t)| content.get(usize::from(s)..usize::from(t)));
129  match run {
130    Some([one]) if !matches!(lookup_meaning(one), Some(Stored::Expandable(_))) => *one,
131    _ => T_CS!(command),
132  }
133}

135/// A separator's span `span` past the environments closed right before it (`environments`, the rehearsal's steps): its
136/// own run of the argument. An invocation making no box leaves its cover for the next box (core
137/// `token::field_cover_pending_merge`, OD #715), so a separator's box covers an `\end{tabular}` glued to it
138/// (`\end{tabular}\and`, `\end{tabular}%⏎\and`; `\end{tabular}\relax\and` covers the `\relax` too, which is the
139/// separator's as a font switch before a name is the name's). Those tokens close the table and are the environment's,
140/// as the emitter's balanced runs take them (`emitter::Field::balanced`), not the separator's. A step reaching the span's
141/// end leaves it whole: the separator keeps a token.
142fn own_run(environments: &[EnvironmentStep], span: Option<(u16, u16)>) -> Option<(u16, u16)> {
143  let (mut s, t) = span?;
144  while let Some(step) = environments
145    .iter()
146    .find(|step| !step.opens && step.start <= s && s < step.end && step.end < t)
147  {
148    s = step.end;
149  }
150  Some((s, t))
151}

153/// A separator's own run `run` ([`own_run`]) as positions, where no ink event's span meets it — no name, mark, contact
154/// or note prints from its tokens ([`ArgSeparator::span`]).
155fn own_tokens(events: &[Event], run: Option<(u16, u16)>) -> Option<(usize, usize)> {
156  let (s, t) = run?;
157  let inks = |e: &&Event| {
158    matches!(
159      e.kind,
160      EventKind::Name | EventKind::Mark | EventKind::Contact | EventKind::Note
161    )
162  };
163  let met = events
164    .iter()
165    .filter(inks)
166    .any(|ink| ink.span.is_some_and(|(a, b)| a < t && b > s));
167  (!met).then_some((usize::from(s), usize::from(t)))
168}

170/// The `\and`-family separators `arg` holds at its own level (nest 0: not one a layout container of it holds, a
171/// `tabular`, a `\parbox`, nor a note's), in order, each with what it means where the query is made ([`SepMeaning`]).
172/// `None` where no rehearsal could be made — a call inside one (no journal opens), over the queries' allowance, an
173/// argument past the field's positions, a run whose assignments could not be undone (`Rehearsal::Skipped`) — or where
174/// the run left a group open (a `\begingroup` with no `\endgroup`: no kernel author line holds one, so its separators
175/// are unknown, not a list). What a rehearsal leaves for the reading, each put back (by reading `rehearse`'s writes to
176/// its thread-locals and its global assignments): the reads it spends and the calls it leaves out
177/// (`READS_LEFT`, `LEFT_OUT`: the queries' own allowance instead, `rehearsal::for_a_query`), the footnote steps it carries
178/// for the next block's rehearsal (`lx_fm_obs_counter_seen`, `lx_fm_obs_steps_carried`, restored here), its flags and its
179/// assignments (restored and rolled back by `rehearse`; a run whose assignments could not be undone keeps them, as every
180/// rehearsal's does, `SkipReason::NotUndone`, and answers `None`).
181/// OXIDIZED_DESIGN_DIVERGENCES #995.
182pub fn argument_separators(arg: &Tokens) -> Result<Option<Vec<ArgSeparator>>> {
183  separators_of(arg, false)
184}

194/// An argument as a query's rehearsal reads it ([`read_argument`]): its events and the environments it opens and closes.
195pub(in crate::base_utilities) struct ArgumentReading {
196  pub events:       Vec<Event>,
197  pub environments: Vec<EnvironmentStep>,
198}

200/// The one rehearsal of an argument the queries read (separators, marks; OXIDIZED_DESIGN_DIVERGENCES #995, Option C
201/// F0): `None` where no rehearsal could be made or the run left a group open ([`argument_separators`]).
202pub(in crate::base_utilities) fn read_argument(arg: &Tokens) -> Result<Option<ArgumentReading>> {
203  let content = arg.unlist_ref();
204  if content.len() > usize::from(AUTHOR_PROV_POS_MASK) {
205    return Ok(None);
206  }
207  const CARRIED: [&str; 2] = ["lx_fm_obs_counter_seen", "lx_fm_obs_steps_carried"];
208  let carried = CARRIED.map(lookup_int);
209  // (from the queries' own allowance, never the reading's budget nor its count of calls left out: `for_a_query`)
210  let read = for_a_query(|| {
211    rehearse(
212      field_stamped(2, 0, content),
213      TitleMarks::Fnsymbol,
214      &[],
215      lex_events,
216    )
217  })?;
218  for (key, value) in CARRIED.iter().zip(carried) {
219    if lookup_int(key) != value {
220      assign_value(key, Stored::from(value), Some(Scope::Global));
221    }
222  }
223  // (a run that left a group open read no event past where it opened — `\begingroup` and no `\endgroup`: unknown, not
224  // "no separator")
225  let Rehearsal::Done {
226    value: events,
227    left_open: false,
228    environments,
229    ..
230  } = read
231  else {
232    return Ok(None);
233  };
234  Ok(Some(ArgumentReading { events, environments }))
235}

237/// [`argument_separators`], at `arg`'s own level or (`every_level`) at all of them.
238fn separators_of(arg: &Tokens, every_level: bool) -> Result<Option<Vec<ArgSeparator>>> {
239  let Some(ArgumentReading { events, environments }) = read_argument(arg)? else {
240    return Ok(None);
241  };
242  let content = arg.unlist_ref();
243  let separators = events.iter().filter_map(|e| match e.kind {
244    EventKind::Sep(SepKind::And(command)) if every_level || e.nest == 0 => {
245      let run = own_run(&environments, e.span);
246      Some(ArgSeparator {
247        command,
248        meaning: meaning_of(&printing_token(content, run, command)),
249        span: own_tokens(&events, run),
250      })
251    },
252    _ => None,
253  });
254  Ok(Some(separators.collect()))
255}

257/// The TeX front-end, `\lx@fm@separators{<tokens>}`: [`argument_separators`]' meanings by name, comma-separated.
258pub(in crate::base_utilities) fn load() -> Result<()> {
259  DefMacro!("\\lx@fm@separators{}", sub[(arg)] {
260    let arg: Tokens = arg;
261    let names = match argument_separators(&arg)? {
262      Some(separators) => separators
263        .iter()
264        .map(|separator| separator.meaning.name())
265        .collect::<Vec<_>>()
266        .join(","),
267      None => String::from("unread"),
268    };
269    Ok(mouth::tokenize_internal(TeXString::assembled(names)))
270  });
271  Ok(())
272}

274/// The own runs ([`ArgSeparator::span`]) of `arg`'s separators that mean the kernel's `\and` ([`SepMeaning::Kernel`]):
275/// written, held in a macro of the document's that holds only the separator, or a `\let` copy, as the query reads them
276/// by meaning ([`argument_separators`]), each a run of whole units of `arg`'s top level (no group or environment holds
277/// it, [`tokens_unit_end`]), in order and apart; `None` where the query reads nothing (no rehearsal made, over its allowance, a
278/// group left open), for which a caller cuts by spelling. A separator of another meaning — a class's, a binding's or the
279/// document's renewed `\and`, amsmath's `\And` — is no run here (trap T8: what it prints is its caller's to say), nor
280/// one a macro holds together with ink (no span: the reading cuts that one at its `Sep(And)`, #993). The cut two bindings'
281/// routes share (Option C P2, OXIDIZED_DESIGN_DIVERGENCES #995): eptcs's cells (P2-4, `eptcs_cls.rs`), the
282/// `\institute` lists of inst_support, aa and llncs and llncs's marked `\author` list ([`split_at_kernel_separators`],
283/// P2-2, P2-3).
284pub fn kernel_separator_runs(arg: &Tokens) -> Result<Option<Vec<(usize, usize)>>> {
285  let Some(separators) = argument_separators(arg)? else {
286    return Ok(None);
287  };
288  let tokens = arg.unlist_ref();
289  let units = tokens_top_level_units(tokens);
290  let mut runs = Vec::new();
291  let mut from = 0;
292  for (start, end) in separators
293    .iter()
294    .filter(|found| found.meaning == SepMeaning::Kernel)
295    .filter_map(|found| found.span)
296  {
297    if start < from || !units.contains(&start) || !units.contains(&end) {
298      continue;
299    }
300    runs.push((start, end));
301    from = end;
302  }
303  Ok(Some(runs))
304}

306/// `arg` split as [`split_tokens_delimited`] splits it at `delims`, cut first at each [`kernel_separator_runs`] run:
307/// each segment between the runs split by spelling as it is, never a token rewritten (a run is cut where it stands,
308/// eptcs's cells' way), the segment a run ends keeping its last piece even when empty ([`split_tokens_segment`]: the
309/// split of the whole list keeps a piece a delimiter ends) and each segment's first piece the run before it as its
310/// delimiter; and the number of runs cut that hold no delimiter of `delims` by spelling (`split_tokens`' match,
311/// [`delimiter_token_matches`]): the cuts by meaning alone — a macro holding only `\and`, a `\let` copy of the kernel's
312/// `\and` where the document renewed `\and`. A run is cut inside balanced parentheses too, which the spelling split
313/// protects, as TeX's column break parts them (`Univ A (North \myand South)` after a document's `\let\myand\and`), and
314/// a run the spelling also cuts is cut once. The spelling's cuts in the segments stay, whatever their token means (a
315/// document's renewed `\and`, an undefined `\And`: trap T8, the cut the split made before, which aa.cls's own
316/// title-time `\and` keeps, aa.cls:866), as do its trimming and its protections inside a segment. Where the query reads
317/// nothing or finds no run, the spelling split of `arg` alone. Option C P2-2, P2-3 (OXIDIZED_DESIGN_DIVERGENCES #995):
318/// the `\institute` lists of inst_support (`\lx@splitting@by@meaning`), aa (`aa_cls.rs` `\lx@aa@institute@dispatch`)
319/// and llncs (`llncs_cls.rs` `\lx@llncs@institute@dispatch`), one contact call per piece, and llncs's marked `\author`
320/// list (`\lx@llncs@author@split`), one author entry per piece.
321pub fn split_at_kernel_separators(
322  arg: Tokens,
323  delims: Vec<SplitDelim>,
324) -> Result<(DelimitedPieces, usize)> {
325  let runs = kernel_separator_runs(&arg)?.unwrap_or_default();
326  Ok(split_at_runs(arg, delims, runs))
327}

329/// [`split_at_kernel_separators`] at the kernel separator runs `runs` a caller kept.
330pub(in crate::base_utilities) fn split_at_runs(
331  arg: Tokens,
332  delims: Vec<SplitDelim>,
333  runs: Vec<(usize, usize)>,
334) -> (DelimitedPieces, usize) {
335  if runs.is_empty() {
336    return (split_tokens_delimited(arg, delims), 0);
337  }
338  let spelled = |t: &Token| {
339    delims
340      .iter()
341      .any(|d| matches!(d, SplitDelim::Token(d) if delimiter_token_matches(d, t)))
342  };
343  let tokens = arg.unlist_ref();
344  let mut pieces = Vec::new();
345  let mut by_meaning = 0;
346  let mut before: Vec<Token> = Vec::new();
347  let mut from = 0;
348  // (each run, then the end of the list)
349  for run in runs.into_iter().map(Some).chain([None]) {
350    let (to, ended_by_cut) = run.map_or((tokens.len(), false), |(start, _)| (start, true));
351    let segment = Tokens::new(tokens[from..to].to_vec());
352    let mut split = split_tokens_segment(segment, delims.clone(), ended_by_cut);
353    if let Some((first, _)) = split.first_mut() {
354      *first = std::mem::take(&mut before);
355    }
356    pieces.extend(split);
357    if let Some((start, end)) = run {
358      if !tokens[start..end].iter().any(spelled) {
359        by_meaning += 1;
360      }
361      before = tokens[start..end].to_vec();
362      from = end;
363    }
364  }
365  (pieces, by_meaning)
366}

368/// Whether `arg` holds a token its spelled `delims` do not match that could mean the kernel's `\and` here: a copy of it
369/// ([`SepMeaning::Kernel`]) or a macro no binding made (the kernel's `\csname` too). Without one,
370/// [`kernel_separator_runs`] finds no run the spelling does not cut, and no rehearsal is needed.
371pub(in crate::base_utilities) fn may_hold_kernel_separator(
372  arg: &Tokens,
373  delims: &[SplitDelim],
374) -> bool {
375  arg.unlist_ref().iter().any(|t| {
376    matches!(t.get_catcode(), Catcode::CS | Catcode::ACTIVE)
377      && !delims
378        .iter()
379        .any(|d| matches!(d, SplitDelim::Token(d) if delimiter_token_matches(d, t)))
380      && match lookup_meaning(t) {
381        Some(Stored::Expandable(defn)) => !defn.get_origin().is_binding_made(),
382        _ => meaning_of(t) == SepMeaning::Kernel,
383      }
384  })
