39/// The flush's part (`digest_front_matter`, before the queue is digested): rehearse and lex each author block not
40/// rehearsed yet. A block without positions (out of positions) is skipped and says why; one no rehearsal could run for
41/// (a journal already open: the flush is itself inside a speculative run) waits for the next flush, deferred: left so
42/// at the end, the dump reports that reason (`JournalUnavailable`), not "no flush after the call".
43pub(in crate::base_utilities) fn read_at_flush() -> Result<HeldUndefined> {
44  // (a rehearsal's flush is undone with it; every real flush empties the queue, read or not — the epoch a clear
45  // compares, `ledger::record_clear`: an early `\maketitle` before any author block, in default mode, reads nothing
46  // but flushes the title it digests, which a later `\title` must not clear; review rev92a_r1 S2)
47  if rehearsal::rehearsing() {
48    return Ok(HeldUndefined::default());
49  }
50  rehearse_title_first()?;
51  let held = read_pending()?;
52  // (the flush empties the frontmatter queue: the calls after it are queued from position 0 again)
53  with_ledger(|ledger| ledger.flushes += 1);
54  Ok(held)
55}

57/// The last title set, rehearsed before the author blocks where the title marks its notes (`lx_title_counts_footnotes`;
58/// article.cls:203-214), so its footnote steps carry into theirs (`rehearsal::footnotes_carried`). OD #1053.
59fn rehearse_title_first() -> Result<()> {
60  if !lookup_bool("lx_title_counts_footnotes") {
61    return Ok(());
62  }
63  let title = with_ledger(|ledger| {
64    ledger
65      .entries()
66      .iter()
67      .enumerate()
68      .rev()
69      .find_map(|(i, entry)| match entry {
70        Entry::Api(call)
71          if call.tag == "ltx:title" && !call.rehearsed && !call.cleared && !call.inside =>
72        {
73          Some((i, call.content.clone()))
74        },
75        _ => None,
76      })
77  });
78  let Some((i, content)) = title.filter(|(_, content)| !content.is_empty()) else {
79    return Ok(());
80  };
81  if let Rehearsal::Done { value, errors, queued, .. } = rehearse_call(
82    &mut CallRehearsals::default(),
83    content,
84    TitleMarks::Fnsymbol,
85  )? {
86    with_ledger(|ledger| {
87      if let Some(Entry::Api(call)) = ledger.entries_mut().get_mut(i) {
88        call.events = value;
89        call.errors = errors;
90        call.queued = queued;
91        call.rehearsed = true;
92      }
93    });
94  }
95  Ok(())
96}

101/// `tokens` (positions from `base`) with each call of a document macro whose expansion printed several names, contacts or notes
102/// at its one position (`events`' shared span) replaced by that expansion (`expansions`, `field_notes::HeldExpansion`); `None`
103/// where none did. A read of the result gives each its own positions, as the inline text does.
104pub(in crate::base_utilities) fn unfold_held_names(
105  tokens: &[Token],
106  base: usize,
107  events: &[Event],
108  expansions: &[HeldExpansion],
109) -> Option<Vec<Token>> {
110  let mut printed_at: Vec<(u16, u16)> = events
111    .iter()
112    .filter(|e| match e.kind {
113      EventKind::Name => !e.text.trim().is_empty(),
114      EventKind::Contact | EventKind::Note => true,
115      _ => false,
116    })
117    .filter_map(|e| e.span)
118    .collect();
119  printed_at.sort_unstable();
120  let shared: Vec<(u16, u16)> = printed_at
121    .windows(2)
122    .filter(|pair| pair[0] == pair[1])
123    .map(|pair| pair[0])
124    .collect();
125  let mut out = Vec::with_capacity(tokens.len());
126  let mut next = 0;
127  for held in expansions
128    .iter()
129    .filter(|held| shared.iter().any(|&(s, t)| held.at <= s && t <= held.end))
130  {
131    let at = usize::from(held.at).checked_sub(base)?;
132    let end = usize::from(held.end).checked_sub(base)?;
133    if at < next || end > tokens.len() {
134      return None;
135    }
136    out.extend_from_slice(&tokens[next..at]);
137    out.extend_from_slice(held.tokens.unlist_ref());
138    next = end;
139  }
140  if next == 0 {
141    return None;
142  }
143  out.extend_from_slice(&tokens[next..]);
144  Some(out)
145}

147/// The undefined commands this flush's rehearsals held an Error for (`Error:undefined:\cs`, State.pm:547-551's
148/// `generateErrorStub`), each once, with how many times the conversion had raised it as the flush began (the tally of
149/// undefined names, `LogState::undefined`). A rehearsal holds each of its Errors and discards it with its stub, trusting
150/// the emitter's digestion of each piece to raise it again; a command whose token falls in no piece the emitter
151/// digests (it prints nothing, so no event covers it: `\author{… Lee\textsuperscript{1}\internship, …}`, 2607.29185),
152/// or that sits in a piece written as text, is raised by no one — Perl digests the block once and raises it in place
153/// (Base_Utility.pool.ltxml:691-693). [`HeldUndefined::raise_unraised`] is the flush's net for those: the Error the
154/// rest of the flush did not raise, once, as Perl's first encounter. Not an IJCAI sectioned block's: its Errors are the
155/// section markers its class defines only inside its own `\author` (ijcai19.sty:127, `\emails`, `\affiliations`;
156/// the resolver's `block_markers`, 2501.02356). OXIDIZED_DESIGN_DIVERGENCES #983; repro
157/// sectioning-frontmatter/emitter_undefined_command_in_author_errs_once; witnesses 1101.4098, 1402.2906, 1808.09392,
158/// 2205.12847, 2309.09045, 2405.06973, 2405.13308, 2506.10896, 2508.08170, 2508.21091, 2601.17318, 2602.18351,
159/// 2604.26688, 2607.29185, 2609.00161, hep-th9306034.
160#[derive(Debug, Default)]
161pub(in crate::base_utilities) struct HeldUndefined(Vec<(Token, usize)>);
162
163impl HeldUndefined {
164  /// Raise, through `generate_error_stub`, each held command the flush did not raise again — its tally of undefined
165  /// names unmoved since the flush began — that no code of the flush gave a meaning, even locally (`defined`, the
166  /// flush's meaning recorder: a class's title code defining its name parts only inside its own `\author` or
167  /// `\maketitle`, where the emitter's pieces are digested — vmsta's `\fnm`/`\snm`/`\inits`, the IOS classes'
168  /// `\fnms`, JAIR's row `\name`/`\email`/`\addr` —, as pdflatex and Perl digest them there, no error; the recorder
169  /// opens as the rehearsals' stubs are gone, so a command it saw no assignment of has no meaning): its Error once, its
170  /// stub standing for the rest of the document, as Perl's first encounter leaves it. Decided by the tally and the recorder, not by spelled names: a piece that raised it in place counted it there,
171  /// so nothing is raised twice.
172  pub(in crate::base_utilities) fn raise_unraised(
173    self,
174    defined: &[(SymStr, Option<Stored>)],
175  ) -> Result<()> {
176    for (token, before) in self.0 {
177      let (raised, assigned) = token.with_cs_name(|name| {
178        (
179          undefined_tally(name) > before,
180          defined.iter().any(|(key, _)| with(*key, |k| k == name)),
181        )
182      });
183      if !raised && !assigned {
184        generate_error_stub(&token)?;
185      }
186    }
187    Ok(())
188  }
189}

191/// Whether the author block at ledger index `i` was replaced before any flush digested it: cleared by a later clear, or
192/// before a later call of its queue that replaces the authors (`\def\@author`; Perl's `dequeueFrontMatter`,
193/// Base_Utility.pool.ltxml:691). Its commands are never digested, so they raise nothing (2605.25890, 2605.26243).
194fn superseded(i: usize) -> bool {
195  with_ledger(|ledger| {
196    let entries = ledger.entries();
197    let Some(Entry::Author(block)) = entries.get(i) else {
198      return false;
199    };
200    block.cleared
201      || entries[i + 1..].iter().any(|entry| {
202        matches!(entry, Entry::Author(later) if later.replace && later.epoch == block.epoch)
203      })
204  })
205}

213    .unwrap_or(0)

216/// The command a held `Error:undefined:\cs` record names (`generate_error_stub`'s message, its first line, which names a
217/// control sequence's token `T_CS[…]`): `None` for any other record.
218fn held_undefined(message: &str) -> Option<Token> {
219  let rest = message.strip_prefix("Error:undefined:")?;
220  let (name, tail) = rest.split_once(' ')?;
221  tail
222    .starts_with(&s!("The token T_CS[{name}] is not defined"))
223    .then_some(T_CS!(name))
224}

226/// [`read_at_flush`]'s reading, with the undefined commands its rehearsals held ([`HeldUndefined`]).
227fn read_pending() -> Result<HeldUndefined> {
228  // (the labels the calls' keyvals defer, read as this flush's digestion expands them: `\lx@label@ref`, cas's
229  // `\author[a]` under entries given after it — 2303.00352, 2603.22658, 2504.19452, 2509.10496)
230  with_ledger(|ledger| {
231    let resolved: Vec<(usize, String)> = ledger
232      .entries()
233      .iter()
234      .enumerate()
235      .filter_map(|(i, entry)| {
236        let text = match entry {
237          Entry::Author(block) => block.keyvals.as_deref()?,
238          Entry::Api(call) => call.attrs.as_str(),
239          _ => return None,
240        };
241        match resolve_label_refs(text) {
242          Cow::Owned(resolved) => Some((i, resolved)),
243          Cow::Borrowed(_) => None,
244        }
245      })
246      .collect();
247    for (i, text) in resolved {
248      match &mut ledger.entries_mut()[i] {
249        Entry::Author(block) => block.keyvals = Some(text),
250        Entry::Api(call) => call.attrs = text,
251        _ => {},
252      }
253    }
254  });
255  let pending: Vec<usize> = with_ledger(|ledger| {
256    ledger
257      .entries()
258      .iter()
259      .enumerate()
260      .filter_map(|(i, entry)| match entry {
261        Entry::Author(block)
262          if matches!(block.outcome, Outcome::Pending | Outcome::Deferred(_)) =>
263        {
264          Some(i)
265        },
266        _ => None,
267      })
268      .collect()
269  });
270  // (the blocks rehearsed in sequence share the conditionals, as TeX reads the author list in one pass — a block's run
271  // opens one the next block's closes, 2606.17718 — and the document gets them back as they were after the last:
272  // `StomachMark`)
273  let blocks_mark = StomachMark::take();
274  let mut undefined = HeldUndefined::default();
275  for i in pending {
276    let Some(block) = with_ledger(|ledger| match ledger.entries().get(i) {
277      Some(Entry::Author(block)) => Some(block.clone()),
278      _ => None,
279    }) else {
280      continue;
281    };
282    let mut queued_tags = Vec::new();
283    let mut unfolded = None;
284    let outcome = if block.stamp == Stamp::Overflow {
285      Outcome::Skipped(SkipReason::Overflow)
286    } else if block.tokens.is_empty() {
287      Outcome::Skipped(SkipReason::Empty)
288    } else {
289      let policy = policy_for(&block.binding, block.hook);
290      let title_marks = policy.title_marks;
291      let block_breaks = policy.block_breaks;
292      // (under the author markup its digestion reads it with, `author_markup_begin`: a class binding's (vmsta's
293      // `\inits` gobbles, 1508.02842, 1411.4499, 1506.02940), OmniBus's (IOS-Book-Article's `\orcid`, 2606.13132), the
294      // title code's (astlb.cls:117-118's `\address` a mark, 1202.2889, 1108.4222, 1206.0451; klmm.sty's `\instref`,
295      // 2606.12100) — OXIDIZED_DESIGN_DIVERGENCES #802, #941, #983)
296      let markup = author_markup_begin(true)?;
297      // (the block's tokens beside it: its in-block contacts' entries are read as the writer can cut them,
298      // `events::labelled_entries`)
299      let rehearse_block = |tokens: &Tokens| {
300        with_block_content(block.range.0, tokens.clone(), || {
301          rehearse(
302            field_stamped(2, block.range.0, tokens.unlist_ref()),
303            Some(&markup),
304            title_marks,
305            block_breaks,
306            lex_with_inks,
307          )
308        })
309      };
310      let carried = footnotes_carried();
311      let conditionals = if_stack_depth();
312      let mut rehearsal = rehearse_block(&block.tokens)?;
313      // (a document macro printing several names, `\author{\Authors}` (2103.13804): its events share its one position,
314      // so the block is read again with the macro's expansion in its place, each name its own run, as written inline)
315      for _ in 0..UNFOLD_LEVELS {
316        let Rehearsal::Done {
317          value: (events, ..),
318          expansions,
319          ..
320        } = &rehearsal
321        else {
322          break;
323        };
324        let read = unfolded.as_ref().unwrap_or(&block.tokens);
325        let Some(tokens) = unfold_held_names(read.unlist_ref(), block.range.0, events, expansions)
326          .filter(|tokens| {
327            block.range.0 + tokens.len() <= usize::from(AUTHOR_PROV_POS_MASK)
328              && if_stack_depth() == conditionals
329          })
330        else {
331          break;
332        };
333        let after = footnotes_carried();
334        set_footnotes_carried(carried);
335        let tokens = Tokens::new(tokens);
336        let again = rehearse_block(&tokens)?;
337        if !matches!(again, Rehearsal::Done { .. }) {
338          set_footnotes_carried(after);
339          break;
340        }
341        rehearsal = again;
342        unfolded = Some(tokens);
343      }
344      let read = unfolded.as_ref().unwrap_or(&block.tokens);
345      match rehearsal {
346        Rehearsal::Done {
347          value: (mut value, formulas, inks),
348          errors,
349          warnings,
350          declarations,
351          breaks,
352          held,
353          environments,
354          creators,
355          left_open,
356          unsaid,
357          queued,
358          expansions: _,
359        } => {
360          queued_tags = queued;
361          if !block.sectioned && !superseded(i) {
362            for token in held
363              .iter()
364              .filter_map(|error| held_undefined(&error.message))
365            {
366              if !undefined.0.iter().any(|(t, _)| *t == token) {
367                let before = token.with_cs_name(undefined_tally);
368                undefined.0.push((token, before));
369              }
370            }
371          }
372          mark_held(&mut value, &held, &inks);
373          creator_roles(&mut value, &creators);
374          Outcome::Events {
375            events: value,
376            formulas,
377            inks: Box::new(inks),
378            errors,
379            warnings,
380            declarations,
381            breaks,
382            held,
383            environments,
384            creators: creators.into_boxed_slice(),
385            unread: left_open.then(|| content_text(read).into()),
386            unsaid,
387          }
388        },
389        // (waits for the next flush; if none comes, the dump says why it was never rehearsed)
390        Rehearsal::Skipped(SkipReason::JournalUnavailable) => {
391          Outcome::Deferred(SkipReason::JournalUnavailable)
392        },
393        Rehearsal::Skipped(reason) => Outcome::Skipped(reason),
394      }
395    };
396    with_ledger(|ledger| {
397      if let Some(Entry::Author(block)) = ledger.entries_mut().get_mut(i) {
398        block.outcome = outcome;
399        block.queued = queued_tags;
400        block.unfolded = unfolded;
401      }
402    });
403  }
404  blocks_mark.restore();
405  // the API calls made outside the blocks' expansions: their contents rehearsed and lexed as a block's are (the
406  // resolver reads an affiliation's leading marks and text, an e-mail's address, a creator's name from them); and a
407  // call a block's own expansion made that annotates a creator, for the content check alone (`ApiCall::inside_events`:
408  // its words count unless its block's events print them), the block's own creator entry being the block
409  let calls: Vec<(usize, Tokens, bool)> = with_ledger(|ledger| {
410    ledger
411      .entries()
412      .iter()
413      .enumerate()
414      .filter_map(|(i, entry)| match entry {
415        Entry::Api(call) if !call.rehearsed && (!call.inside || call.parent == "ltx:creator") => {
416          Some((i, call.content.clone(), call.inside))
417        },
418        _ => None,
419      })
420      .collect()
421  });
422  let calls_mark = StomachMark::take();
423  // (a content the calls repeat is rehearsed once: AASTeX's 1,427 `\affiliation`s, 228 contents, 1901.01540)
424  let mut seen = CallRehearsals::default();
425  for (i, content, inside) in calls {
426    let read = if content.is_empty() {
427      Some((Vec::new(), 0, None, Vec::new()))
428    } else {
429      // (an inside call's, for the check alone, from the check's own allowance: `rehearsal::for_the_check`)
430      let mut rehearsal = || rehearse_call(&mut seen, content.clone(), TitleMarks::Fnsymbol);
431      let rehearsed = if inside {
432        for_the_check(rehearsal)
433      } else {
434        rehearsal()
435      };
436      match rehearsed? {
437        Rehearsal::Done { value, errors, queued, .. } => Some((value, errors, None, queued)),
438        // (a flush inside a speculative run: the next flush rehearses it)
439        Rehearsal::Skipped(SkipReason::JournalUnavailable) => None,
440        // (given up on: what it printed is its content's text, `ApiCall::unread`)
441        Rehearsal::Skipped(_) => Some((
442          Vec::new(),
443          0,
444          Some(content_text(&content).into()),
445          Vec::new(),
446        )),
447      }
448    };
449    if let Some((events, errors, unread, queued)) = read {
450      with_ledger(|ledger| {
451        if let Some(Entry::Api(call)) = ledger.entries_mut().get_mut(i) {
452          if call.inside {
453            call.inside_events = events.into();
454          } else {
455            call.events = events;
456            call.errors = errors;
457            call.queued = queued;
458          }
459          call.rehearsed = true;
460          call.unread = unread;
461        }
462      });
463    }
464  }
465  calls_mark.restore();
466  Ok(undefined)
467}

469/// A JSON string literal.
470fn json_str(out: &mut String, s: &str) {
471  out.push('"');
472  for c in s.chars() {
473    match c {
474      '"' => out.push_str("\\\""),
475      '\\' => out.push_str("\\\\"),
476      '\n' => out.push_str("\\n"),
477      '\r' => out.push_str("\\r"),
478      '\t' => out.push_str("\\t"),
479      c if (c as u32) < 0x20 => {
480        let _ = write!(out, "\\u{:04x}", c as u32);
481      },
482      c => out.push(c),
483    }
484  }
485  out.push('"');
486}

488fn json_opt_str(out: &mut String, s: Option<&str>) {

495fn json_span(out: &mut String, span: Option<(usize, usize)>) {
496  match span {

500    None => out.push_str("null"),

504fn json_event(out: &mut String, event: &Event) {
505  out.push_str("{\"kind\":");
506  json_str(out, event.kind.name());
507  if let EventKind::Sep(sep) = &event.kind {
508    out.push_str(",\"sep\":");
509    match sep {
510      SepKind::And(command) => {
511        out.push_str("\"and\",\"command\":");
512        json_str(out, command);
513      },
514      SepKind::Column(character) => {
515        out.push_str("\"column\",\"command\":");
516        json_str(out, character);
517      },
518      SepKind::Comma => json_str(out, "comma"),
519      SepKind::Word(word) => {
520        out.push_str("\"word\",\"word\":");
521        json_str(out, word);
522      },
523      SepKind::Gap { items, width, fill } => {
524        let _ = write!(
525          out,
526          "\"gap\",\"items\":{items},\"width\":{width},\"fill\":{fill}"
527        );
528      },
529    }
530  }
531  out.push_str(",\"span\":");
532  json_span(
533    out,
534    event.span.map(|(s, e)| (usize::from(s), usize::from(e))),
535  );
536  if !event.text.is_empty() {
537    out.push_str(",\"text\":");
538    json_str(out, &event.text);
539  }
540  if let Some(label) = &event.label {
541    out.push_str(",\"label\":");
542    json_str(out, label);
543  }
544  if event.kind == EventKind::Mark {
545    let _ = write!(
546      out,
547      ",\"symbol\":{},\"line_first\":{}",
548      event.symbol, event.line_first
549    );
550  }
551  if let Some(role) = &event.role {
552    out.push_str(",\"role\":");
553    json_str(out, role);
554  }
555  if let Some(target) = &event.target {
556    out.push_str(",\"target\":");
557    json_str(out, target);
558  }
559  if event.nest > 0 {
560    let _ = write!(out, ",\"nest\":{}", event.nest);
561  }
562  if let Some(formula) = &event.formula {
563    out.push_str(",\"formula_span\":");
564    json_span(
565      out,
566      formula.span.map(|(s, e)| (usize::from(s), usize::from(e))),
567    );
568    if let Some(base) = formula.base {
569      out.push_str(",\"base\":");
570      json_str(out, base.name());
571    }
572    let _ = write!(out, ",\"ends_formula\":{}", formula.ends);
573  }
574  if let Some(value) = &event.unprinted {
575    out.push_str(",\"unprinted\":");
576    json_str(out, value);
577  }
578  if let Some(kind) = event.break_kind {
579    out.push_str(",\"break\":");
580    json_str(out, kind.name());
581  }
582  if event.braces > 0 {
583    let _ = write!(out, ",\"braces\":{}", event.braces);
584  }
585  if event.ordinal {
586    out.push_str(",\"ordinal\":true");
587  }
588  if !event.printed_labels.is_empty() {
589    out.push_str(",\"printed_labels\":[");
590    for (i, label) in event.printed_labels.iter().enumerate() {
591      if i > 0 {
592        out.push(',');
593      }
594      out.push_str("{\"label\":");
595      json_str(out, &label.label);
596      let _ = write!(
597        out,
598        ",\"word\":{},\"line_first\":{},\"span\":",
599        label.word, label.line_first
600      );
601      json_span(
602        out,
603        label.span.map(|(s, e)| (usize::from(s), usize::from(e))),
604      );
605      out.push('}');
606    }
607    out.push(']');
608  }
609  if event.held {
610    out.push_str(",\"held\":true");
611  }
612  if event.meaning != EmailMeaning::Binding {
613    out.push_str(",\"meaning\":");
614    json_str(out, event.meaning.name());
615  }
616  if event.inkless {
617    out.push_str(",\"inkless\":true");
618  }
619  if !event.inner.is_empty() {
620    out.push_str(",\"inner\":[");
621    for (i, inner) in event.inner.iter().enumerate() {
622      if i > 0 {
623        out.push(',');
624      }
625      json_event(out, inner);
626    }
627    out.push(']');
628  }
629  out.push('}');
630}

663/// The ledger as JSON (read by the guards, in process): its entries in call order — author blocks with their events, title
664/// stamps, API calls — and a summary (blocks with events, skipped by reason, out of positions, rehearsals that held
665/// diagnostics). An author block's ApiCall events are the API calls after it its own expansion did not make.
666pub fn fm_reading_json() -> String {
667  with_ledger(|ledger| {
668    let mut out = String::from("{\"version\":1,\"entries\":[");
669    let mut with_events = 0usize;
670    let mut held = 0usize;
671    let mut skipped: Vec<(&str, usize)> = Vec::new();
672    let mut blocks = 0usize;
673    let mut overflow = 0usize;
674    for (n, entry) in ledger.entries().iter().enumerate() {
675      if n > 0 {
676        out.push(',');
677      }
678      match entry {
679        Entry::Title(title) => {
680          let _ = write!(
681            out,
682            "{{\"index\":{},\"kind\":\"title\",\"length\":{},\"stamp\":{},\"range\":",
683            title.index,
684            title.length,
685            if title.range.is_some() {
686              "\"kernel\""
687            } else {
688              "\"overflow\""
689            }
690          );
691          json_span(&mut out, title.range);
692          out.push('}');
693          if title.range.is_none() {
694            overflow += 1;
695          }
696        },
697        Entry::NoteText(text) => {
698          let _ = write!(
699            out,
700            "{{\"index\":{},\"kind\":\"notetext\",\"epoch\":{},\"mark\":",
701            text.index, text.epoch
702          );
703          json_str(&mut out, &text.mark);
704          out.push_str(",\"value\":");
705          json_opt_str(&mut out, text.value.as_deref());
706          out.push_str(",\"events\":[");
707          for (i, event) in text.events.iter().enumerate() {
708            if i > 0 {
709              out.push(',');
710            }
711            json_event(&mut out, event);
712          }
713          out.push_str("]}");
714        },
715        Entry::Api(call) => {
716          let _ = write!(out, "{{\"index\":{},\"kind\":\"api\",\"tag\":", call.index);
717          json_str(&mut out, &call.tag);
718          out.push_str(",\"role\":");
719          json_str(&mut out, &call.role);
720          let _ = write!(
721            out,
722            ",\"epoch\":{},\"queue_position\":{},\"after_block\":{},\"inside\":{}",
723            call.epoch,
724            call.queue_position,
725            call
726              .after_block
727              .map_or("null".to_string(), |b| b.to_string()),
728            call.inside
729          );
730          out.push_str(",\"attrs\":");
731          json_str(&mut out, &call.attrs);
732          if call.errors > 0 {
733            let _ = write!(out, ",\"errors\":{}", call.errors);
734          }
735          if !call.parent.is_empty() {
736            out.push_str(",\"parent\":");
737            json_str(&mut out, &call.parent);
738          }
739          if call.cleared {
740            out.push_str(",\"cleared\":true");
741          }
742          if let Some(text) = &call.unread {
743            out.push_str(",\"unread\":");
744            json_str(&mut out, text);
745          }
746          if call.rehearsed {
747            // (an inside call's, rehearsed for the content check alone: `ApiCall::inside_events`)
748            let events: &[Event] = if call.inside {
749              &call.inside_events
750            } else {
751              &call.events
752            };
753            out.push_str(",\"events\":[");
754            for (i, event) in events.iter().enumerate() {
755              if i > 0 {
756                out.push(',');
757              }
758              json_event(&mut out, event);
759            }
760            out.push(']');
761          }
762          out.push('}');
763        },
764        Entry::Author(block) => {
765          blocks += 1;
766          let _ = write!(
767            out,
768            "{{\"index\":{},\"kind\":\"author\",\"hook\":",
769            block.index
770          );
771          json_str(&mut out, block.hook);
772          out.push_str(",\"policy\":");
773          json_str(&mut out, &block.policy);
774          out.push_str(",\"binding\":");
775          json_str(&mut out, &block.binding);
776          out.push_str(",\"owner\":");
777          json_str(&mut out, &block.owner);
778          if let Some(counter) = block.counter {
779            let _ = write!(out, ",\"counter\":{counter}");
780          }
781          if block.cleared {
782            out.push_str(",\"cleared\":true");
783          }
784          let _ = write!(
785            out,
786            ",\"replace\":{},\"sectioned\":{},\"late\":{},\"cautious\":{},\"keyvals\":",
787            block.replace, block.sectioned, block.late, block.cautious
788          );
789          json_opt_str(&mut out, block.keyvals.as_deref());
790          let _ = write!(
791            out,
792            ",\"epoch\":{},\"queue_position\":{},\"stamp\":",
793            block.epoch, block.queue_position
794          );
795          json_str(&mut out, block.stamp.name());
796          out.push_str(",\"range\":");
797          json_span(
798            &mut out,
799            (block.stamp != Stamp::Overflow).then_some(block.range),
800          );
801          if block.stamp == Stamp::Overflow {
802            overflow += 1;
803          }
804          out.push_str(",\"source\":");
805          json_str(&mut out, &block.source);
806          out.push_str(",\"status\":");
807          let mut events: Vec<Event> = Vec::new();
808          match &block.outcome {
809            Outcome::Pending | Outcome::Skipped(_) | Outcome::Deferred(_) => {
810              let reason = skip_reason(&block.outcome).unwrap_or(SkipReason::NoFlush);
811              json_str(&mut out, "skipped");
812              out.push_str(",\"reason\":");
813              json_str(&mut out, reason.name());
814              bump(&mut skipped, reason.name());
815            },
816            Outcome::Events {
817              events: lexed,
818              errors,
819              warnings,
820              declarations,
821              breaks,
822              held: held_errors,
823              unread,
824              ..
825            } => {
826              json_str(&mut out, "events");
827              let _ = write!(out, ",\"errors\":{errors},\"warnings\":{warnings}");
828              if let Some(text) = unread {
829                out.push_str(",\"unread\":");
830                json_str(&mut out, text);
831              }
832              with_events += 1;
833              if errors + warnings > 0 {
834                held += 1;
835              }
836              events.clone_from(lexed);
837              if !declarations.is_empty() {
838                json_events(&mut out, "declarations", declarations);
839              }
840              if !breaks.is_empty() {
841                json_events(&mut out, "breaks", breaks);
842              }
843              if !held_errors.is_empty() {
844                json_held(&mut out, held_errors);
845              }
846            },
847          }
848          // the setters called outside the field after the block, up to the next block
849          for later in &ledger.entries()[n + 1..] {
850            match later {
851              Entry::Author(_) => break,
852              Entry::Api(call) if !call.inside && call.after_block == Some(block.index) => {
853                let mut event = Event::new(EventKind::ApiCall);
854                event.label = Some(s!("queue:{}:{}", call.epoch, call.queue_position));
855                event.role = Some(s!("{} {}", call.tag, call.role));
856                events.push(event);
857              },
858              _ => {},
859            }
860          }
861          if matches!(block.outcome, Outcome::Events { .. }) {
862            out.push_str(",\"events\":[");
863            for (i, event) in events.iter().enumerate() {
864              if i > 0 {
865                out.push(',');
866              }
867              json_event(&mut out, event);
868            }
869            out.push(']');
870          }
871          out.push('}');
872        },
873      }
874    }
875    out.push(']');
876    if !ledger.unprinted.is_empty() {
877      out.push_str(",\"unprinted\":[");
878      for (i, call) in ledger.unprinted.iter().enumerate() {
879        if i > 0 {
880          out.push(',');
881        }
882        let _ = write!(
883          out,
884          "{{\"at\":{},\"after_block\":{},\"inside\":{},\"attrs\":",
885          call.at,
886          call
887            .after_block
888            .map_or("null".to_string(), |b| b.to_string()),
889          call.inside
890        );
891        json_str(&mut out, &call.attrs);
892        out.push_str(",\"event\":");
893        json_event(&mut out, &call.event);
894        out.push('}');
895      }
896      out.push(']');
897    }
898    out.push_str(",\"resolved\":");
899    json_resolved(&mut out, &resolve(&ledger_inputs(ledger)));
900    let _ = write!(
901      out,
902      ",\"summary\":{{\"blocks\":{blocks},\"with_events\":{with_events},\"held\":{held},\"overflow\":{overflow},\"skipped\":{{"
903    );
904    for (i, (reason, count)) in skipped.iter().enumerate() {
905      if i > 0 {
906        out.push(',');
907      }
908      json_str(&mut out, reason);
909      let _ = write!(out, ":{count}");
910    }
911    // (whether a flush's emitter wrote the paper's frontmatter, `Ledger::emitted`)
912    let _ = writeln!(out, "}}}},\"emitted\":{}}}", ledger.emitted);
913    out
914  })
915}

917fn json_assoc(out: &mut String, assoc: &Assoc) {
918  out.push_str("{\"role\":");
919  json_str(out, &assoc.role);
920  out.push_str(",\"text\":");
921  json_str(out, &assoc.text);
922  if let Some(label) = &assoc.label {
923    out.push_str(",\"label\":");
924    json_str(out, label);
925  }
926  out.push_str(",\"rule\":");
927  json_str(out, &assoc.rule);
928  // (a link's target, where it points: FM S2a.2 r9)
929  if let Some(target) = &assoc.target {
930    out.push_str(",\"target\":");
931    json_str(out, target);
932  }
933  if let Some((index, span)) = assoc.source {
934    // (keyed by the block and the span: spans are unique only within a block)
935    let _ = write!(out, ",\"source\":[{index},");
936    json_span(out, span.map(|(s, e)| (usize::from(s), usize::from(e))));
937    out.push(']');
938  }
939  out.push('}');
940}

953/// The resolver's tuples: the creators (name without marks, requests, the rule that split it out, its block) with
954/// their affiliations, contacts and notes, and the block's own.
955fn json_resolved(out: &mut String, resolved: &Resolved) {
956  out.push_str("{\"creators\":[");
957  for (i, creator) in resolved.creators.iter().enumerate() {
958    if i > 0 {
959      out.push(',');
960    }
961    out.push_str("{\"name\":");
962    json_str(out, &creator.name);
963    out.push_str(",\"requests\":[");
964    for (j, request) in creator.requests().enumerate() {
965      if j > 0 {
966        out.push(',');
967      }
968      json_str(out, request);
969    }
970    out.push_str("],\"rule\":");
971    json_str(out, &creator.rule);
972    out.push_str(",\"joins\":[");
973    for (j, join) in creator.joins.iter().enumerate() {
974      if j > 0 {
975        out.push(',');
976      }
977      json_str(out, join.tag());
978    }
979    out.push(']');
980    let _ = write!(out, ",\"source\":{}", creator.source);
981    // (its creator call's role, where a call in its block printed it: `RCreator::role`)
982    if let Some(role) = &creator.role {
983      out.push_str(",\"role\":");
984      json_str(out, role);
985    }
986    json_assocs(out, "affiliations", &creator.affiliations);
987    json_assocs(out, "contacts", &creator.contacts);
988    json_assocs(out, "notes", &creator.notes);
989    out.push('}');
990  }
991  out.push(']');
992  json_assocs(out, "block", &resolved.block);
993  json_assocs(out, "dropped", &resolved.dropped);
994  out.push_str(",\"continued\":[");
995  for (i, lines) in resolved.continued.iter().flatten().enumerate() {
996    if i > 0 {
997      out.push(',');
998    }
999    out.push('[');
1000    for (j, line) in lines.iter().enumerate() {
1001      if j > 0 {
1002        out.push(',');
1003      }
1004      json_str(out, line);
1005    }
1006    out.push(']');
1007  }
1008  out.push(']');
1009  out.push_str(",\"policies\":[");
1010  for (i, (index, name)) in resolved.policies.iter().enumerate() {
1011    if i > 0 {
1012      out.push(',');
1013    }
1014    let _ = write!(out, "[{index},");
1015    json_str(out, name);
1016    out.push(']');
1017  }
1018  out.push_str("]}");
1019}

1021/// Why a block that has no events was not rehearsed: its skip's reason; one deferred by a flush no rehearsal could run
1022/// at (`JournalUnavailable`) and never rehearsed later, that reason; one no flush came after, `NoFlush`.
1023fn skip_reason(outcome: &Outcome) -> Option<SkipReason> {
1024  match outcome {
1025    Outcome::Pending => Some(SkipReason::NoFlush),
1026    Outcome::Skipped(reason) | Outcome::Deferred(reason) => Some(*reason),
1027    Outcome::Events { .. } => None,
1028  }
1029}
1030
1031fn bump<'a>(counts: &mut Vec<(&'a str, usize)>, reason: &'a str) {
1032  match counts.iter_mut().find(|(r, _)| *r == reason) {
1033    Some((_, count)) => *count += 1,
1034    None => counts.push((reason, 1)),
1035  }
1036}
