1//! The observer's flush and its dump (OXIDIZED_DESIGN_DIVERGENCES #837): at every real flush (`digest_front_matter`)
2//! the ledger's author blocks not rehearsed yet are rehearsed and lexed, and each API call's content
3//! ([`read_at_flush`]); [`fm_reading_json`] gives the ledger as JSON — its entries with their events and the resolver's
4//! tuples (`resolved`) — which the guards read the reading by, in process. The census that wrote it beside the output
5//! (`LATEXML_FM_SHADOW=1`, `<dest>.fmshadow.json`, `ab_audit fmshadow`) and its off state (`=0`) were deleted with the
6//! S2b writer (DEL4): the observer records and reads every paper.

38/// The flush's part (`digest_front_matter`, before the queue is digested): rehearse and lex each author block not
39/// rehearsed yet. A block without positions (out of positions) is skipped and says why; one no rehearsal could run for
40/// (a journal already open: the flush is itself inside a speculative run) waits for the next flush, deferred: left so
41/// at the end, the dump reports that reason (`JournalUnavailable`), not "no flush after the call".
42pub(in crate::base_utilities) fn read_at_flush() -> Result<HeldUndefined> {
43  // (a rehearsal's flush is undone with it; every real flush empties the queue, read or not — the epoch a clear
44  // compares, `ledger::record_clear`: an early `\maketitle` before any author block, in default mode, reads nothing
45  // but flushes the title it digests, which a later `\title` must not clear; review rev92a_r1 S2)
46  if rehearsal::rehearsing() {
47    return Ok(HeldUndefined::default());
48  }
49  let held = read_pending()?;
50  // (the flush empties the frontmatter queue: the calls after it are queued from position 0 again)
51  with_ledger(|ledger| ledger.flushes += 1);
52  Ok(held)
53}

58/// `tokens` (positions from `base`) with each call of a document macro whose expansion printed several names, contacts or notes
59/// at its one position (`events`' shared span) replaced by that expansion (`expansions`, `field_notes::HeldExpansion`); `None`
60/// where none did. A read of the result gives each its own positions, as the inline text does.
61pub(in crate::base_utilities) fn unfold_held_names(
62  tokens: &[Token],
63  base: usize,
64  events: &[Event],
65  expansions: &[HeldExpansion],
66) -> Option<Vec<Token>> {
67  let mut printed_at: Vec<(u16, u16)> = events
68    .iter()
69    .filter(|e| match e.kind {
70      EventKind::Name => !e.text.trim().is_empty(),
71      EventKind::Contact | EventKind::Note => true,
72      _ => false,
73    })
74    .filter_map(|e| e.span)
75    .collect();
76  printed_at.sort_unstable();
77  let shared: Vec<(u16, u16)> = printed_at
78    .windows(2)
79    .filter(|pair| pair[0] == pair[1])
80    .map(|pair| pair[0])
81    .collect();
82  let mut out = Vec::with_capacity(tokens.len());
83  let mut next = 0;
84  for held in expansions
85    .iter()
86    .filter(|held| shared.iter().any(|&(s, t)| held.at <= s && t <= held.end))
87  {
88    let at = usize::from(held.at).checked_sub(base)?;
89    let end = usize::from(held.end).checked_sub(base)?;
90    if at < next || end > tokens.len() {
91      return None;
92    }
93    out.extend_from_slice(&tokens[next..at]);
94    out.extend_from_slice(held.tokens.unlist_ref());
95    next = end;
96  }
97  if next == 0 {
98    return None;
99  }
100  out.extend_from_slice(&tokens[next..]);
101  Some(out)
102}

104/// The undefined commands this flush's rehearsals held an Error for (`Error:undefined:\cs`, State.pm:547-551's
105/// `generateErrorStub`), each once, with how many times the conversion had raised it as the flush began (the tally of
106/// undefined names, `LogState::undefined`). A rehearsal holds each of its Errors and discards it with its stub, trusting
107/// the emitter's digestion of each piece to raise it again; a command whose token falls in no piece the emitter
108/// digests (it prints nothing, so no event covers it: `\author{… Lee\textsuperscript{1}\internship, …}`, 2607.29185),
109/// or that sits in a piece written as text, is raised by no one — Perl digests the block once and raises it in place
110/// (Base_Utility.pool.ltxml:691-693). [`HeldUndefined::raise_unraised`] is the flush's net for those: the Error the
111/// rest of the flush did not raise, once, as Perl's first encounter. Not an IJCAI sectioned block's: its Errors are the
112/// section markers its class defines only inside its own `\author` (ijcai19.sty:127, `\emails`, `\affiliations`;
113/// the resolver's `block_markers`, 2501.02356). OXIDIZED_DESIGN_DIVERGENCES #983; repro
114/// sectioning-frontmatter/emitter_undefined_command_in_author_errs_once; witnesses 1101.4098, 1402.2906, 1808.09392,
115/// 2205.12847, 2309.09045, 2405.06973, 2405.13308, 2506.10896, 2508.08170, 2508.21091, 2601.17318, 2602.18351,
116/// 2604.26688, 2607.29185, 2609.00161, hep-th9306034.
117#[derive(Debug, Default)]
118pub(in crate::base_utilities) struct HeldUndefined(Vec<(Token, usize)>);
119
120impl HeldUndefined {
121  /// Raise, through `generate_error_stub`, each held command the flush did not raise again — its tally of undefined
122  /// names unmoved since the flush began — that no code of the flush gave a meaning, even locally (`defined`, the
123  /// flush's meaning recorder: a class's title code defining its name parts only inside its own `\author` or
124  /// `\maketitle`, where the emitter's pieces are digested — vmsta's `\fnm`/`\snm`/`\inits`, the IOS classes'
125  /// `\fnms`, JAIR's row `\name`/`\email`/`\addr` —, as pdflatex and Perl digest them there, no error; the recorder
126  /// opens as the rehearsals' stubs are gone, so a command it saw no assignment of has no meaning): its Error once, its
127  /// stub standing for the rest of the document, as Perl's first encounter leaves it. Decided by the tally and the recorder, not by spelled names: a piece that raised it in place counted it there,
128  /// so nothing is raised twice.
129  pub(in crate::base_utilities) fn raise_unraised(
130    self,
131    defined: &[(SymStr, Option<Stored>)],
132  ) -> Result<()> {
133    for (token, before) in self.0 {
134      let (raised, assigned) = token.with_cs_name(|name| {
135        (
136          undefined_tally(name) > before,
137          defined.iter().any(|(key, _)| with(*key, |k| k == name)),
138        )
139      });
140      if !raised && !assigned {
141        generate_error_stub(&token)?;
142      }
143    }
144    Ok(())
145  }
146}

148/// Whether the author block at ledger index `i` was replaced before any flush digested it: cleared by a later clear, or
149/// before a later call of its queue that replaces the authors (`\def\@author`; Perl's `dequeueFrontMatter`,
150/// Base_Utility.pool.ltxml:691). Its commands are never digested, so they raise nothing (2605.25890, 2605.26243).
151fn superseded(i: usize) -> bool {
152  with_ledger(|ledger| {
153    let entries = ledger.entries();
154    let Some(Entry::Author(block)) = entries.get(i) else {
155      return false;
156    };
157    block.cleared
158      || entries[i + 1..].iter().any(|entry| {
159        matches!(entry, Entry::Author(later) if later.replace && later.epoch == block.epoch)
160      })
161  })
162}

164/// How many times the conversion raised `name` undefined (the summary's tally, `LogState::undefined`).
165fn undefined_tally(name: &str) -> usize {
166  latexml_core::report!()
167    .undefined
168    .get(name)
169    .copied()
170    .unwrap_or(0)
171}

173/// The command a held `Error:undefined:\cs` record names (`generate_error_stub`'s message, its first line, which names a
174/// control sequence's token `T_CS[…]`): `None` for any other record.
175fn held_undefined(message: &str) -> Option<Token> {
176  let rest = message.strip_prefix("Error:undefined:")?;
177  let (name, tail) = rest.split_once(' ')?;
178  tail
179    .starts_with(&s!("The token T_CS[{name}] is not defined"))
180    .then_some(T_CS!(name))
181}

183/// [`read_at_flush`]'s reading, with the undefined commands its rehearsals held ([`HeldUndefined`]).
184fn read_pending() -> Result<HeldUndefined> {
185  // (the labels the calls' keyvals defer, read as this flush's digestion expands them: `\lx@label@ref`, cas's
186  // `\author[a]` under entries given after it — 2303.00352, 2603.22658, 2504.19452, 2509.10496)
187  with_ledger(|ledger| {
188    let resolved: Vec<(usize, String)> = ledger
189      .entries()
190      .iter()
191      .enumerate()
192      .filter_map(|(i, entry)| {
193        let text = match entry {
194          Entry::Author(block) => block.keyvals.as_deref()?,
195          Entry::Api(call) => call.attrs.as_str(),
196          _ => return None,
197        };
198        match resolve_label_refs(text) {
199          Cow::Owned(resolved) => Some((i, resolved)),
200          Cow::Borrowed(_) => None,
201        }
202      })
203      .collect();
204    for (i, text) in resolved {
205      match &mut ledger.entries_mut()[i] {
206        Entry::Author(block) => block.keyvals = Some(text),
207        Entry::Api(call) => call.attrs = text,
208        _ => {},
209      }
210    }
211  });
212  let pending: Vec<usize> = with_ledger(|ledger| {
213    ledger
214      .entries()
215      .iter()
216      .enumerate()
217      .filter_map(|(i, entry)| match entry {
218        Entry::Author(block)
219          if matches!(block.outcome, Outcome::Pending | Outcome::Deferred(_)) =>
220        {
221          Some(i)
222        },
223        _ => None,
224      })
225      .collect()
226  });
227  // (the blocks rehearsed in sequence share the conditionals, as TeX reads the author list in one pass — a block's run
228  // opens one the next block's closes, 2606.17718 — and the document gets them back as they were after the last:
229  // `StomachMark`)
230  let blocks_mark = StomachMark::take();
231  let mut undefined = HeldUndefined::default();
232  for i in pending {
233    let Some(block) = with_ledger(|ledger| match ledger.entries().get(i) {
234      Some(Entry::Author(block)) => Some(block.clone()),
235      _ => None,
236    }) else {
237      continue;
238    };
239    let mut queued_tags = Vec::new();
240    let mut unfolded = None;
241    let outcome = if block.stamp == Stamp::Overflow {
242      Outcome::Skipped(SkipReason::Overflow)
243    } else if block.tokens.is_empty() {
244      Outcome::Skipped(SkipReason::Empty)
245    } else {
246      let policy = policy_for(&block.binding, block.hook);
247      let title_marks = policy.title_marks;
248      let block_breaks = policy.block_breaks;
249      // (the block's tokens beside it: its in-block contacts' entries are read as the writer can cut them,
250      // `events::labelled_entries`)
251      let rehearse_block = |tokens: &Tokens| {
252        with_block_content(block.range.0, tokens.clone(), || {
253          rehearse(
254            field_stamped(2, block.range.0, tokens.unlist_ref()),
255            title_marks,
256            block_breaks,
257            lex_with_inks,
258          )
259        })
260      };
261      let carried = footnotes_carried();
262      let conditionals = if_stack_depth();
263      let mut rehearsal = rehearse_block(&block.tokens)?;
264      // (a document macro printing several names, `\author{\Authors}` (2103.13804): its events share its one position,
265      // so the block is read again with the macro's expansion in its place, each name its own run, as written inline)
266      for _ in 0..UNFOLD_LEVELS {
267        let Rehearsal::Done {
268          value: (events, ..),
269          expansions,
270          ..
271        } = &rehearsal
272        else {
273          break;
274        };
275        let read = unfolded.as_ref().unwrap_or(&block.tokens);
276        let Some(tokens) = unfold_held_names(read.unlist_ref(), block.range.0, events, expansions)
277          .filter(|tokens| {
278            block.range.0 + tokens.len() <= usize::from(AUTHOR_PROV_POS_MASK)
279              && if_stack_depth() == conditionals
280          })
281        else {
282          break;
283        };
284        let after = footnotes_carried();
285        set_footnotes_carried(carried);
286        let tokens = Tokens::new(tokens);
287        let again = rehearse_block(&tokens)?;
288        if !matches!(again, Rehearsal::Done { .. }) {
289          set_footnotes_carried(after);
290          break;
291        }
292        rehearsal = again;
293        unfolded = Some(tokens);
294      }
295      let read = unfolded.as_ref().unwrap_or(&block.tokens);
296      match rehearsal {
297        Rehearsal::Done {
298          value: (mut value, formulas, inks),
299          errors,
300          warnings,
301          declarations,
302          breaks,
303          held,
304          environments,
305          creators,
306          left_open,
307          unsaid,
308          queued,
309          expansions: _,
310        } => {
311          queued_tags = queued;
312          if !block.sectioned && !superseded(i) {
313            for token in held
314              .iter()
315              .filter_map(|error| held_undefined(&error.message))
316            {
317              if !undefined.0.iter().any(|(t, _)| *t == token) {
318                let before = token.with_cs_name(undefined_tally);
319                undefined.0.push((token, before));
320              }
321            }
322          }
323          mark_held(&mut value, &held, &inks);
324          creator_roles(&mut value, &creators);
325          Outcome::Events {
326            events: value,
327            formulas,
328            inks: Box::new(inks),
329            errors,
330            warnings,
331            declarations,
332            breaks,
333            held,
334            environments,
335            creators: creators.into_boxed_slice(),
336            unread: left_open.then(|| content_text(read).into()),
337            unsaid,
338          }
339        },
340        // (waits for the next flush; if none comes, the dump says why it was never rehearsed)
341        Rehearsal::Skipped(SkipReason::JournalUnavailable) => {
342          Outcome::Deferred(SkipReason::JournalUnavailable)
343        },
344        Rehearsal::Skipped(reason) => Outcome::Skipped(reason),
345      }
346    };
347    with_ledger(|ledger| {
348      if let Some(Entry::Author(block)) = ledger.entries_mut().get_mut(i) {
349        block.outcome = outcome;
350        block.queued = queued_tags;
351        block.unfolded = unfolded;
352      }
353    });
354  }
355  blocks_mark.restore();
356  // the API calls made outside the blocks' expansions: their contents rehearsed and lexed as a block's are (the
357  // resolver reads an affiliation's leading marks and text, an e-mail's address, a creator's name from them); and a
358  // call a block's own expansion made that annotates a creator, for the content check alone (`ApiCall::inside_events`:
359  // its words count unless its block's events print them), the block's own creator entry being the block
360  let calls: Vec<(usize, Tokens, bool)> = with_ledger(|ledger| {
361    ledger
362      .entries()
363      .iter()
364      .enumerate()
365      .filter_map(|(i, entry)| match entry {
366        Entry::Api(call) if !call.rehearsed && (!call.inside || call.parent == "ltx:creator") => {
367          Some((i, call.content.clone(), call.inside))
368        },
369        _ => None,
370      })
371      .collect()
372  });
373  let calls_mark = StomachMark::take();
374  // (a content the calls repeat is rehearsed once: AASTeX's 1,427 `\affiliation`s, 228 contents, 1901.01540)
375  let mut seen = CallRehearsals::default();
376  for (i, content, inside) in calls {
377    let read = if content.is_empty() {
378      Some((Vec::new(), 0, None, Vec::new()))
379    } else {
380      // (an inside call's, for the check alone, from the check's own allowance: `rehearsal::for_the_check`)
381      let mut rehearsal = || rehearse_call(&mut seen, content.clone(), TitleMarks::Fnsymbol);
382      let rehearsed = if inside {
383        for_the_check(rehearsal)
384      } else {
385        rehearsal()
386      };
387      match rehearsed? {
388        Rehearsal::Done { value, errors, queued, .. } => Some((value, errors, None, queued)),
389        // (a flush inside a speculative run: the next flush rehearses it)
390        Rehearsal::Skipped(SkipReason::JournalUnavailable) => None,
391        // (given up on: what it printed is its content's text, `ApiCall::unread`)
392        Rehearsal::Skipped(_) => Some((
393          Vec::new(),
394          0,
395          Some(content_text(&content).into()),
396          Vec::new(),
397        )),
398      }
399    };
400    if let Some((events, errors, unread, queued)) = read {
401      with_ledger(|ledger| {
402        if let Some(Entry::Api(call)) = ledger.entries_mut().get_mut(i) {
403          if call.inside {
404            call.inside_events = events.into();
405          } else {
406            call.events = events;
407            call.errors = errors;
408            call.queued = queued;
409          }
410          call.rehearsed = true;
411          call.unread = unread;
412        }
413      });
414    }
415  }
416  calls_mark.restore();
417  Ok(undefined)
418}

420/// A JSON string literal.
421fn json_str(out: &mut String, s: &str) {
422  out.push('"');
423  for c in s.chars() {
424    match c {
425      '"' => out.push_str("\\\""),
426      '\\' => out.push_str("\\\\"),
427      '\n' => out.push_str("\\n"),
428      '\r' => out.push_str("\\r"),
429      '\t' => out.push_str("\\t"),
430      c if (c as u32) < 0x20 => {
431        let _ = write!(out, "\\u{:04x}", c as u32);
432      },
433      c => out.push(c),
434    }
435  }
436  out.push('"');
437}

451    None => out.push_str("null"),

455fn json_event(out: &mut String, event: &Event) {
456  out.push_str("{\"kind\":");
457  json_str(out, event.kind.name());
458  if let EventKind::Sep(sep) = &event.kind {
459    out.push_str(",\"sep\":");
460    match sep {
461      SepKind::And(command) => {
462        out.push_str("\"and\",\"command\":");
463        json_str(out, command);
464      },
465      SepKind::Column(character) => {
466        out.push_str("\"column\",\"command\":");
467        json_str(out, character);
468      },
469      SepKind::Comma => json_str(out, "comma"),
470      SepKind::Word(word) => {
471        out.push_str("\"word\",\"word\":");
472        json_str(out, word);
473      },
474      SepKind::Gap { items, width, fill } => {
475        let _ = write!(
476          out,
477          "\"gap\",\"items\":{items},\"width\":{width},\"fill\":{fill}"
478        );
479      },
480    }
481  }
482  out.push_str(",\"span\":");
483  json_span(
484    out,
485    event.span.map(|(s, e)| (usize::from(s), usize::from(e))),
486  );
487  if !event.text.is_empty() {
488    out.push_str(",\"text\":");
489    json_str(out, &event.text);
490  }
491  if let Some(label) = &event.label {
492    out.push_str(",\"label\":");
493    json_str(out, label);
494  }
495  if event.kind == EventKind::Mark {
496    let _ = write!(
497      out,
498      ",\"symbol\":{},\"line_first\":{}",
499      event.symbol, event.line_first
500    );
501  }
502  if let Some(role) = &event.role {
503    out.push_str(",\"role\":");
504    json_str(out, role);
505  }
506  if let Some(target) = &event.target {
507    out.push_str(",\"target\":");
508    json_str(out, target);
509  }
510  if event.nest > 0 {
511    let _ = write!(out, ",\"nest\":{}", event.nest);
512  }
513  if let Some(formula) = &event.formula {
514    out.push_str(",\"formula_span\":");
515    json_span(
516      out,
517      formula.span.map(|(s, e)| (usize::from(s), usize::from(e))),
518    );
519    if let Some(base) = formula.base {
520      out.push_str(",\"base\":");
521      json_str(out, base.name());
522    }
523    let _ = write!(out, ",\"ends_formula\":{}", formula.ends);
524  }
525  if let Some(value) = &event.unprinted {
526    out.push_str(",\"unprinted\":");
527    json_str(out, value);
528  }
529  if let Some(kind) = event.break_kind {
530    out.push_str(",\"break\":");
531    json_str(out, kind.name());
532  }
533  if event.braces > 0 {
534    let _ = write!(out, ",\"braces\":{}", event.braces);
535  }
536  if event.ordinal {
537    out.push_str(",\"ordinal\":true");
538  }
539  if !event.printed_labels.is_empty() {
540    out.push_str(",\"printed_labels\":[");
541    for (i, label) in event.printed_labels.iter().enumerate() {
542      if i > 0 {
543        out.push(',');
544      }
545      out.push_str("{\"label\":");
546      json_str(out, &label.label);
547      let _ = write!(
548        out,
549        ",\"word\":{},\"line_first\":{},\"span\":",
550        label.word, label.line_first
551      );
552      json_span(
553        out,
554        label.span.map(|(s, e)| (usize::from(s), usize::from(e))),
555      );
556      out.push('}');
557    }
558    out.push(']');
559  }
560  if event.held {
561    out.push_str(",\"held\":true");
562  }
563  if event.meaning != EmailMeaning::Binding {
564    out.push_str(",\"meaning\":");
565    json_str(out, event.meaning.name());
566  }
567  if event.inkless {
568    out.push_str(",\"inkless\":true");
569  }
570  if !event.inner.is_empty() {
571    out.push_str(",\"inner\":[");
572    for (i, inner) in event.inner.iter().enumerate() {
573      if i > 0 {
574        out.push(',');
575      }
576      json_event(out, inner);
577    }
578    out.push(']');
579  }
580  out.push('}');
581}

583/// A list of events, under `key`.
584fn json_events(out: &mut String, key: &str, events: &[Event]) {
585  let _ = write!(out, ",\"{key}\":[");
586  for (i, event) in events.iter().enumerate() {
587    if i > 0 {
588      out.push(',');
589    }
590    json_event(out, event);
591  }
592  out.push(']');
593}

595/// The errors a block's rehearsal held, by position.
596fn json_held(out: &mut String, held: &[HeldError]) {
597  out.push_str(",\"held\":[");
598  for (i, error) in held.iter().enumerate() {
599    if i > 0 {
600      out.push(',');
601    }
602    out.push_str("{\"span\":");
603    json_span(
604      out,
605      error.span.map(|(s, e)| (usize::from(s), usize::from(e))),
606    );
607    out.push_str(",\"message\":");
608    json_str(out, &error.message);
609    out.push('}');
610  }
611  out.push(']');
612}

614/// The ledger as JSON (read by the guards, in process): its entries in call order — author blocks with their events, title
615/// stamps, API calls — and a summary (blocks with events, skipped by reason, out of positions, rehearsals that held
616/// diagnostics). An author block's ApiCall events are the API calls after it its own expansion did not make.
617pub fn fm_reading_json() -> String {
618  with_ledger(|ledger| {
619    let mut out = String::from("{\"version\":1,\"entries\":[");
620    let mut with_events = 0usize;
621    let mut held = 0usize;
622    let mut skipped: Vec<(&str, usize)> = Vec::new();
623    let mut blocks = 0usize;
624    let mut overflow = 0usize;
625    for (n, entry) in ledger.entries().iter().enumerate() {
626      if n > 0 {
627        out.push(',');
628      }
629      match entry {
630        Entry::Title(title) => {
631          let _ = write!(
632            out,
633            "{{\"index\":{},\"kind\":\"title\",\"length\":{},\"stamp\":{},\"range\":",
634            title.index,
635            title.length,
636            if title.range.is_some() {
637              "\"kernel\""
638            } else {
639              "\"overflow\""
640            }
641          );
642          json_span(&mut out, title.range);
643          out.push('}');
644          if title.range.is_none() {
645            overflow += 1;
646          }
647        },
648        Entry::NoteText(text) => {
649          let _ = write!(
650            out,
651            "{{\"index\":{},\"kind\":\"notetext\",\"epoch\":{},\"mark\":",
652            text.index, text.epoch
653          );
654          json_str(&mut out, &text.mark);
655          out.push_str(",\"value\":");
656          json_opt_str(&mut out, text.value.as_deref());
657          out.push_str(",\"events\":[");
658          for (i, event) in text.events.iter().enumerate() {
659            if i > 0 {
660              out.push(',');
661            }
662            json_event(&mut out, event);
663          }
664          out.push_str("]}");
665        },
666        Entry::Api(call) => {
667          let _ = write!(out, "{{\"index\":{},\"kind\":\"api\",\"tag\":", call.index);
668          json_str(&mut out, &call.tag);
669          out.push_str(",\"role\":");
670          json_str(&mut out, &call.role);
671          let _ = write!(
672            out,
673            ",\"epoch\":{},\"queue_position\":{},\"after_block\":{},\"inside\":{}",
674            call.epoch,
675            call.queue_position,
676            call
677              .after_block
678              .map_or("null".to_string(), |b| b.to_string()),
679            call.inside
680          );
681          out.push_str(",\"attrs\":");
682          json_str(&mut out, &call.attrs);
683          if call.errors > 0 {
684            let _ = write!(out, ",\"errors\":{}", call.errors);
685          }
686          if !call.parent.is_empty() {
687            out.push_str(",\"parent\":");
688            json_str(&mut out, &call.parent);
689          }
690          if call.cleared {
691            out.push_str(",\"cleared\":true");
692          }
693          if let Some(text) = &call.unread {
694            out.push_str(",\"unread\":");
695            json_str(&mut out, text);
696          }
697          if call.rehearsed {
698            // (an inside call's, rehearsed for the content check alone: `ApiCall::inside_events`)
699            let events: &[Event] = if call.inside {
700              &call.inside_events
701            } else {
702              &call.events
703            };
704            out.push_str(",\"events\":[");
705            for (i, event) in events.iter().enumerate() {
706              if i > 0 {
707                out.push(',');
708              }
709              json_event(&mut out, event);
710            }
711            out.push(']');
712          }
713          out.push('}');
714        },
715        Entry::Author(block) => {
716          blocks += 1;
717          let _ = write!(
718            out,
719            "{{\"index\":{},\"kind\":\"author\",\"hook\":",
720            block.index
721          );
722          json_str(&mut out, block.hook);
723          out.push_str(",\"policy\":");
724          json_str(&mut out, &block.policy);
725          out.push_str(",\"binding\":");
726          json_str(&mut out, &block.binding);
727          out.push_str(",\"owner\":");
728          json_str(&mut out, &block.owner);
729          if let Some(counter) = block.counter {
730            let _ = write!(out, ",\"counter\":{counter}");
731          }
732          if block.cleared {
733            out.push_str(",\"cleared\":true");
734          }
735          let _ = write!(
736            out,
737            ",\"replace\":{},\"sectioned\":{},\"late\":{},\"cautious\":{},\"keyvals\":",
738            block.replace, block.sectioned, block.late, block.cautious
739          );
740          json_opt_str(&mut out, block.keyvals.as_deref());
741          let _ = write!(
742            out,
743            ",\"epoch\":{},\"queue_position\":{},\"stamp\":",
744            block.epoch, block.queue_position
745          );
746          json_str(&mut out, block.stamp.name());
747          out.push_str(",\"range\":");
748          json_span(
749            &mut out,
750            (block.stamp != Stamp::Overflow).then_some(block.range),
751          );
752          if block.stamp == Stamp::Overflow {
753            overflow += 1;
754          }
755          out.push_str(",\"source\":");
756          json_str(&mut out, &block.source);
757          out.push_str(",\"status\":");
758          let mut events: Vec<Event> = Vec::new();
759          match &block.outcome {
760            Outcome::Pending | Outcome::Skipped(_) | Outcome::Deferred(_) => {
761              let reason = skip_reason(&block.outcome).unwrap_or(SkipReason::NoFlush);
762              json_str(&mut out, "skipped");
763              out.push_str(",\"reason\":");
764              json_str(&mut out, reason.name());
765              bump(&mut skipped, reason.name());
766            },
767            Outcome::Events {
768              events: lexed,
769              errors,
770              warnings,
771              declarations,
772              breaks,
773              held: held_errors,
774              unread,
775              ..
776            } => {
777              json_str(&mut out, "events");
778              let _ = write!(out, ",\"errors\":{errors},\"warnings\":{warnings}");
779              if let Some(text) = unread {
780                out.push_str(",\"unread\":");
781                json_str(&mut out, text);
782              }
783              with_events += 1;
784              if errors + warnings > 0 {
785                held += 1;
786              }
787              events.clone_from(lexed);
788              if !declarations.is_empty() {
789                json_events(&mut out, "declarations", declarations);
790              }
791              if !breaks.is_empty() {
792                json_events(&mut out, "breaks", breaks);
793              }
794              if !held_errors.is_empty() {
795                json_held(&mut out, held_errors);
796              }
797            },
798          }
799          // the setters called outside the field after the block, up to the next block
800          for later in &ledger.entries()[n + 1..] {
801            match later {
802              Entry::Author(_) => break,
803              Entry::Api(call) if !call.inside && call.after_block == Some(block.index) => {
804                let mut event = Event::new(EventKind::ApiCall);
805                event.label = Some(s!("queue:{}:{}", call.epoch, call.queue_position));
806                event.role = Some(s!("{} {}", call.tag, call.role));
807                events.push(event);
808              },
809              _ => {},
810            }
811          }
812          if matches!(block.outcome, Outcome::Events { .. }) {
813            out.push_str(",\"events\":[");
814            for (i, event) in events.iter().enumerate() {
815              if i > 0 {
816                out.push(',');
817              }
818              json_event(&mut out, event);
819            }
820            out.push(']');
821          }
822          out.push('}');
823        },
824      }
825    }
826    out.push(']');
827    if !ledger.unprinted.is_empty() {
828      out.push_str(",\"unprinted\":[");
829      for (i, call) in ledger.unprinted.iter().enumerate() {
830        if i > 0 {
831          out.push(',');
832        }
833        let _ = write!(
834          out,
835          "{{\"at\":{},\"after_block\":{},\"inside\":{},\"attrs\":",
836          call.at,
837          call
838            .after_block
839            .map_or("null".to_string(), |b| b.to_string()),
840          call.inside
841        );
842        json_str(&mut out, &call.attrs);
843        out.push_str(",\"event\":");
844        json_event(&mut out, &call.event);
845        out.push('}');
846      }
847      out.push(']');
848    }
849    out.push_str(",\"resolved\":");
850    json_resolved(&mut out, &resolve(&ledger_inputs(ledger)));
851    let _ = write!(
852      out,
853      ",\"summary\":{{\"blocks\":{blocks},\"with_events\":{with_events},\"held\":{held},\"overflow\":{overflow},\"skipped\":{{"
854    );
855    for (i, (reason, count)) in skipped.iter().enumerate() {
856      if i > 0 {
857        out.push(',');
858      }
859      json_str(&mut out, reason);
860      let _ = write!(out, ":{count}");
861    }
862    // (whether a flush's emitter wrote the paper's frontmatter, `Ledger::emitted`)
863    let _ = writeln!(out, "}}}},\"emitted\":{}}}", ledger.emitted);
864    out
865  })
866}

868fn json_assoc(out: &mut String, assoc: &Assoc) {
869  out.push_str("{\"role\":");
870  json_str(out, &assoc.role);
871  out.push_str(",\"text\":");
872  json_str(out, &assoc.text);
873  if let Some(label) = &assoc.label {
874    out.push_str(",\"label\":");
875    json_str(out, label);
876  }
877  out.push_str(",\"rule\":");
878  json_str(out, &assoc.rule);
879  // (a link's target, where it points: FM S2a.2 r9)
880  if let Some(target) = &assoc.target {
881    out.push_str(",\"target\":");
882    json_str(out, target);
883  }
884  if let Some((index, span)) = assoc.source {
885    // (keyed by the block and the span: spans are unique only within a block)
886    let _ = write!(out, ",\"source\":[{index},");
887    json_span(out, span.map(|(s, e)| (usize::from(s), usize::from(e))));
888    out.push(']');
889  }
890  out.push('}');
891}

893fn json_assocs(out: &mut String, key: &str, assocs: &[Assoc]) {

895  for (i, assoc) in assocs.iter().enumerate() {
896    if i > 0 {
897      out.push(',');
898    }
899    json_assoc(out, assoc);
900  }
901  out.push(']');

904/// The resolver's tuples: the creators (name without marks, requests, the rule that split it out, its block) with
905/// their affiliations, contacts and notes, and the block's own.
906fn json_resolved(out: &mut String, resolved: &Resolved) {
907  out.push_str("{\"creators\":[");
908  for (i, creator) in resolved.creators.iter().enumerate() {
909    if i > 0 {
910      out.push(',');
911    }
912    out.push_str("{\"name\":");
913    json_str(out, &creator.name);
914    out.push_str(",\"requests\":[");
915    for (j, request) in creator.requests.iter().enumerate() {
916      if j > 0 {
917        out.push(',');
918      }
919      json_str(out, request);
920    }
921    out.push_str("],\"rule\":");
922    json_str(out, &creator.rule);
923    out.push_str(",\"joins\":[");
924    for (j, join) in creator.joins.iter().enumerate() {
925      if j > 0 {
926        out.push(',');
927      }
928      json_str(out, join.tag());
929    }
930    out.push(']');
931    let _ = write!(out, ",\"source\":{}", creator.source);
932    // (its creator call's role, where a call in its block printed it: `RCreator::role`)
933    if let Some(role) = &creator.role {
934      out.push_str(",\"role\":");
935      json_str(out, role);
936    }
937    json_assocs(out, "affiliations", &creator.affiliations);
938    json_assocs(out, "contacts", &creator.contacts);
939    json_assocs(out, "notes", &creator.notes);
940    out.push('}');
941  }
942  out.push(']');
943  json_assocs(out, "block", &resolved.block);
944  json_assocs(out, "dropped", &resolved.dropped);
945  out.push_str(",\"continued\":[");
946  for (i, lines) in resolved.continued.iter().flatten().enumerate() {
947    if i > 0 {
948      out.push(',');
949    }
950    out.push('[');
951    for (j, line) in lines.iter().enumerate() {
952      if j > 0 {
953        out.push(',');
954      }
955      json_str(out, line);
956    }
957    out.push(']');
958  }
959  out.push(']');
960  out.push_str(",\"policies\":[");
961  for (i, (index, name)) in resolved.policies.iter().enumerate() {
962    if i > 0 {
963      out.push(',');
964    }
965    let _ = write!(out, "[{index},");
966    json_str(out, name);
967    out.push(']');
968  }
969  out.push_str("]}");
970}

972/// Why a block that has no events was not rehearsed: its skip's reason; one deferred by a flush no rehearsal could run
973/// at (`JournalUnavailable`) and never rehearsed later, that reason; one no flush came after, `NoFlush`.
974fn skip_reason(outcome: &Outcome) -> Option<SkipReason> {
975  match outcome {
976    Outcome::Pending => Some(SkipReason::NoFlush),
977    Outcome::Skipped(reason) | Outcome::Deferred(reason) => Some(*reason),
978    Outcome::Events { .. } => None,
979  }
980}
981
982fn bump<'a>(counts: &mut Vec<(&'a str, usize)>, reason: &'a str) {
983  match counts.iter_mut().find(|(r, _)| *r == reason) {
984    Some((_, count)) => *count += 1,
985    None => counts.push((reason, 1)),
986  }
987}
