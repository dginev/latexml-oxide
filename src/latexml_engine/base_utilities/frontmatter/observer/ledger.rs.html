29/// How a block's tokens got their positions in the author field. Positions are per call: a block's start where the
30/// field's own stamps end (`lx_field_base_2`, which the document's tokens carry: a block printing `\@author` reads them
31/// apart from its own), never after the blocks before it, which no token of the document carries — so a document of
32/// many calls never runs out, and one call has the author field's 15 bits (`token::AUTHOR_PROV_POS_MASK`; a block of
33/// 559 names, 2204.09614, 2404.06718).
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub(in crate::base_utilities) enum Stamp {
36  /// The kernel `\author`'s `\lx@field@mark` stamped the stored value these tokens are a copy of, at these positions
37  /// (S1, OXIDIZED_DESIGN #715).
38  Kernel,
39  /// No mark stamps them (a binding's own `\lx@add@authors*` call, a kernel value past the stamps' positions): the
40  /// positions an argument stamper (`\lx@field@stamp`) would give them, from where the field's stamps end.
41  Projected,
42  /// Out of positions ([`out_of_positions`]): the value went unstamped.
43  Overflow,
44}
45
46impl Stamp {
47  pub(in crate::base_utilities) fn name(self) -> &'static str {
48    match self {
49      Stamp::Kernel => "kernel",
50      Stamp::Projected => "projected",
51      Stamp::Overflow => "overflow",
52    }
53  }
54}

56/// What the flush made of a block.
57#[derive(Debug, Clone, PartialEq)]
58pub(in crate::base_utilities) enum Outcome {
59  /// Not yet rehearsed (no flush since the call).
60  Pending,
61  /// Not yet rehearsed, waiting for a later flush: at the latest no rehearsal could run, and why
62  /// ([`SkipReason::JournalUnavailable`]: that flush was itself inside a speculative run).
63  Deferred(SkipReason),
64  /// Rehearsed and lexed: its events, the errors and warnings the rehearsal held, and beside the events the block's
65  /// definitions, the breaks its binding declares and the errors by position (`rehearsal::Rehearsal::Done`).
66  Events {
67    events:       Vec<Event>,
68    /// the extents of its formulas, outermost, in order (`events::lex_with_formulas`): what a span's content reads
69    /// (`emitter::Field`). Not in a dump: nothing offline reads the content
70    formulas:     Vec<events::Formula>,
71    /// where its inks printing an `@` print, and in what font (`events::InkTable`): what the emitter cuts an entry's
72    /// addresses at (`emitter::address_cut`). Not in a dump
73    inks:         Box<events::InkTable>,
74    errors:       usize,
75    warnings:     usize,
76    declarations: Vec<Event>,
77    breaks:       Vec<Event>,
78    held:         Vec<HeldError>,
79    /// the environments its tokens open and close, what a span's content reads (`emitter::Field`). Not in a dump
80    environments: Vec<EnvironmentStep>,
81    /// the creator calls its own expansion made (`field_notes::FieldCreator`), the keyvals the emitter writes on the
82    /// creators their Names make (`Event::creator_call`), fixed once rehearsed (a slice: the variant stays under clippy's
83    /// `large_enum_variant` bound). Not in a dump
84    creators:     Box<[FieldCreator]>,
85    /// where the rehearsal left a group open (`rehearsal::Rehearsal::Done::left_open`), what it read past it no event
86    /// says: the block's content's text ([`content_text`]), whose words the events lack the content check counts
87    /// (`resolver::placed::content_check`), as a call's no rehearsal read (`ApiCall::unread`). In a dump only when set
88    unread:       Option<Box<ContentText>>,
89    /// what its own digestion did that no event says ([`Unsaid`]): what the emitter carries or writes whole. Not in a dump:
90    /// only the emitter reads it
91    unsaid:       Option<Box<Unsaid>>,
92  },
93  /// Not rehearsed, and why.
94  Skipped(SkipReason),
95}

365  /// span) and an earlier setter call of a block (the block's index and the call's span) — what the content check counts
366  /// no word of (`emitter::emit::replaced_by_setter`, `resolver::placed::content_check`; review revDEL3 S1)
367  pub replaced:   BTreeSet<(usize, Option<(u16, u16)>)>,
368  /// the entries the emitter wrote labelled by the mark leading them ([`MarkLedEntry`]), over its flushes
369  pub labelled:   Vec<MarkLedEntry>,

371  author_body:    (usize, usize),

373  held:           (usize, (BTreeSet<String>, BTreeSet<String>)),

378impl Ledger {
379  /// Its entries, in call order.
380  pub(in crate::base_utilities) fn entries(&self) -> &[Entry] { &self.entries }
381
382  /// Its entries, to change: a change of what the emitter reads, which a reading made before it no longer is
383  /// ([`Self::generation`]) — the one write path of the count, so taken only where an entry changes (a lookup reads
384  /// [`Self::entries`]: review revN1r3 4).
385  pub(in crate::base_utilities) fn entries_mut(&mut self) -> &mut Vec<Entry> {
386    self.generation += 1;
387    &mut self.entries
388  }
389
390  /// Its entries in `range` that `picks` picks, each changed by `change`, a change counted ([`Self::entries_mut`]) only
391  /// where one is picked, the emitter's reading being kept while the entries read the same ([`Self::analysis`]).
392  pub(in crate::base_utilities) fn change_each(
393    &mut self,
394    range: std::ops::Range<usize>,
395    picks: impl Fn(&Entry) -> bool,
396    mut change: impl FnMut(&mut Entry),
397  ) {
398    if !self
399      .entries
400      .iter()
401      .take(range.end)
402      .skip(range.start)
403      .any(&picks)
404    {
405      return;
406    }
407    for entry in self
408      .entries_mut()
409      .iter_mut()
410      .take(range.end)
411      .skip(range.start)
412    {
413      if picks(entry) {
414        change(entry);
415      }
416    }
417  }
418
419  /// A new entry, last.
420  pub(in crate::base_utilities) fn push(&mut self, entry: Entry) { self.entries_mut().push(entry); }
421
422  /// The changes to its entries so far.
423  pub(in crate::base_utilities) fn generation(&self) -> u64 { self.generation }
424}

503/// Run a compiled binding's loader (`name`.`extension`, the binding's file: `revtex4_support.sty`), noting it as the
504/// owner of each tracked command's meaning (`OWNED_COMMANDS`) when its load changed it and no binding it loaded did, or
505/// its own code redefined it after one did (revtex4-2.cls's binding loads revtex4_support.sty, whose `\author` it is;
506/// aa.cls's loads aa_support.sty, whose `\email` it is, and defines its own `\author` after inst_support.sty's): the binding half of the policy key (`author_binding`), and an e-mail call's binding
507/// (`email_meaning`).
508pub fn note_binding_load<R>(name: &str, extension: &str, load: impl FnOnce() -> R) -> R {
509  // (a speculative run's load is undone with it: the document's own, if any, is the one noted)
510  if in_speculative_digestion() {
511    return load();
512  }
513  let before: [Option<Rc<dyn Definition>>; OWNED] = std::array::from_fn(tracked_meaning);
514  let owners_before: [Option<Rc<dyn Definition>>; OWNED] = std::array::from_fn(owned_meaning);
515  CLASS_BINDINGS.with(|c| {
516    let (names, depth, in_class) = &mut *c.borrow_mut();
517    if *depth == 0 && extension == "cls" && names.is_empty() {
518      *in_class = true;
519    }
520    if *in_class && extension != "pool" {
521      names.push(s!("{name}.{extension}"));
522    }
523    *depth += 1;
524  });
525  let value = load();
526  CLASS_BINDINGS.with(|c| {
527    let (_, depth, in_class) = &mut *c.borrow_mut();
528    *depth = depth.saturating_sub(1);
529    if *depth == 0 {
530      *in_class = false;
531    }
532  });
533  for n in 0..OWNED_COMMANDS.len() {
534    // (a binding it loaded that defined the meaning first owns it unless this binding's own code redefined it after:
535    // aa.cls's `\author`, defined after its aa_support.sty's inst_support.sty defined one, is aa.cls's, OD #922; a raw
536    // file's redefinition stays no binding's)
537    let nested = !same_meaning(owned_meaning(n).as_ref(), owners_before[n].as_ref());
538    if let Some(after) = tracked_meaning(n)
539      && !same_meaning(before[n].as_ref(), Some(&after))
540      && (!nested
541        || (!same_meaning(owned_meaning(n).as_ref(), Some(&after))
542          && matches!(after.get_origin(), DefinitionOrigin::Binding)))
543    {
544      // (a pool is LaTeXML's own kernel: `kernel`, so a binding that loads it — latexml.sty — does not own its
545      // `\author`)
546      let owner = if extension == "pool" {
547        "kernel".to_string()
548      } else {
549        s!("{name}.{extension}")
550      };
551      OWNERS.with(|o| o.borrow_mut()[n] = Some((owner, after)));
552    }
553  }
554  value
555}

687/// Whose the meaning of the e-mail command `cs` (`\email`, `\emailAdd`) is now: the rehearsal's reading of a call to
688/// it, the meaning when the note is rehearsed (at the flush, `\maketitle`; `rehearsal::install_event_bindings`) —
689/// amsart's, which holds `\thanks` until `\maketitle` (amsart.cls:516), as TeX reads it; aa's `\offprints`

692/// failing safe (review rev92a_r4 B1, OD #913). The class's own, printing the address in place, when the
693/// binding owning it ([`meaning_owner`]) declares it ([`policy::declares_email_in_place`]: llncs.cls, also under a
694/// raw class that loads it; sv_support.sty, under svjour3, svjour and svmult), or is one of the document class's
695/// bindings and one of those declares it (aa.cls, whose aa_support.sty defines it — not kluwer's, which loads
696/// aa_support too); another binding's (amsart's ams_support.sty, REVTeX's, jheppub's `\emailAdd`, a package's), whose
697/// code routes the address to the current author; else no binding's — the document's own
698/// (`\renewcommand{\email}[1]{\texttt{#1}}`), a file read raw, the fallback class's guess for a class with no binding
699/// (OmniBus: the real class's print is unknown) —, which prints wherever its definition prints (review rev92a_r3 B1:
700/// 1412.0748, 1809.01965). A guess the fallback class locks is its binding's (OmniBus's `\email`, OD #408: 2511.14713).
701pub(in crate::base_utilities) fn email_meaning(cs: &str) -> EmailMeaning {
702  let Some(n) = OWNED_COMMANDS.iter().position(|c| *c == cs) else {
703    return EmailMeaning::Own;
704  };
705  // (the guess a fallback class makes is marked in the state's own record, whatever the definition carries:
706  // `fallback_meaning_origin`)
707  let guessed = matches!(
708    fallback_meaning_origin(&T_CS!(cs)),
709    Some(DefinitionOrigin::Fallback | DefinitionOrigin::Stub)
710  );
711  // (save the guess the fallback class locks against every later definition, which no document's print replaces: OmniBus's
712  // `\email` (`\email:locked`, OD #408), Perl's OmniBus.cls.ltxml:78 `\lx@add@email{#1}`, the binding's call routing the
713  // address to the current author — 2511.14713's siamart250211 `\thanks{…, \email{wning@vt.edu}, \email{amiedlar@vt.edu}.}`,
714  // read as the document's own, kept in the note's text, which the written note prints without them; main-session ruling
715  // 2026-11-08)
716  let by_binding = if guessed {
717    lookup_bool(&s!("{cs}:locked"))
718  } else {
719    tracked_meaning(n).is_some_and(|meaning| {
720      matches!(
721        meaning.get_origin(),
722        DefinitionOrigin::Binding | DefinitionOrigin::Borrowed
723      )
724    })
725  };
726  if !by_binding {
727    return EmailMeaning::Own;
728  }
729  let owner = meaning_owner(n);
730  let in_place = policy::declares_email_in_place(&owner)
731    || CLASS_BINDINGS.with(|c| {
732      let names = &c.borrow().0;
733      names.contains(&owner)
734        && names
735          .iter()
736          .any(|name| policy::declares_email_in_place(name))
737    });
738  if in_place {
739    EmailMeaning::InPlace
740  } else {
741    EmailMeaning::Binding
742  }
743}

792/// Record an `\author`-family call (`hook`, with `stuff` its argument) — whether it was
793/// recorded late ([`AuthorBlock::late`]).
794pub(in crate::base_utilities) fn record_author_call(
795  hook: &'static str,
796  stuff: &Tokens,
797  replace: bool,
798  sectioned: bool,
799  keyvals: Option<&Tokens>,
800  cautious: bool,
801) -> bool {
802  if !records_now() {
803    return false;
804  }
805  let late = lookup_bool("lx_frontmatter_flushing");
806  let len = stuff.unlist_ref().len();
807  let real_base = usize::try_from(lookup_int("lx_field_base_2")).unwrap_or(0);
808  let owner = author_binding();
809  let binding = policy_binding(&owner);
810  let policy = policy::policy_for(&binding, hook).name;
811  // (a policy's own state, read at the call: authblk's `\c@Maxaffil`, which decides whether its notes show)
812  let counter = policy::policy_for(&binding, hook)
813    .state_counter
814    .and_then(|name| lookup_register_quiet(&s!("\\c@{name}")).map(|value| value.value_of()));
815  let block = AuthorBlock {
816    index: 0,
817    hook,
818    policy: policy.to_string(),
819    binding,
820    owner,
821    counter,
822    replace,
823    cleared: false,
824    sectioned,
825    late,
826    cautious,
827    keyvals: keyvals.map(|kv| kv.to_string()),
828    keyval_tokens: keyvals.cloned(),
829    epoch: 0,
830    queue_position: queue_length(),
831    stamp: Stamp::Projected,
832    range: (real_base, real_base + len),
833    tokens: stuff.clone(),
834    source: stuff.to_string(),
835    unfolded: None,
836    mark: input_mark(),
837    window_open: true,
838    outcome: Outcome::Pending,
839    list_setter: list_setter_called(stuff),
840    queued: Vec::new(),
841    names: 0,
842  };
843  with_ledger(|ledger| {
844    let mut block = block;
845    block.index = ledger.entries().len();
846    block.epoch = ledger.flushes;
847    if out_of_positions(real_base, len) {
848      block.stamp = Stamp::Overflow;
849      block.range = (0, 0);
850    }
851    ledger
852      .queued_blocks
853      .insert((block.epoch, block.queue_position), block.index);
854    let entries = ledger.entries_mut();
855    for entry in entries.iter_mut() {
856      if let Entry::Author(earlier) = entry {
857        earlier.window_open = false;
858      }
859    }
860    entries.push(Entry::Author(Box::new(block)));
861  });
862  late
863}

904/// `\lx@field@mark` stamped field `field_id`'s value (`len` tokens) from `base` — `None`: out of positions, unstamped.
905/// A title records its stamp; the author field's confirms the kernel block it is the stored copy of (the kernel
906/// `\author` hands `\@author` to `\lx@add@authors@adaptive` just before it marks the store), whose positions are then
907/// the mark's own.
908pub(in crate::base_utilities) fn note_field_stamp(field_id: u16, base: Option<usize>, len: usize) {
909  if !records_now() {
910    return;
911  }
912  with_ledger(|ledger| {
913    if field_id == 1 {
914      let index = ledger.entries().len();
915      ledger.push(Entry::Title(TitleStamp {
916        index,
917        length: len,
918        range: base.map(|base| (base, base + len)),
919      }));
920      return;
921    }
922    if field_id != 2 {
923      return;
924    }
925    let Some(at) = ledger
926      .entries()
927      .iter()
928      .rposition(|entry| matches!(entry, Entry::Author(_)))
929    else {
930      return;
931    };
932    // (its entries and its positions change at once: the entries' change counted, `entries_mut`)
933    let Some(Entry::Author(block)) = ledger.entries().get(at) else {
934      return;
935    };
936    // (S2b's check, inactive in S2a: a binding call stamped for real takes the positions it was projected at)
937    if block.stamp == Stamp::Projected
938      && block.tokens.unlist_ref().len() == len
939      && let Some(base) = base
940    {
941      debug_assert!(projected_is_stamped(block, (base, base + len)));
942    }
943    if block.hook != "\\lx@add@authors@adaptive"
944      || block.stamp == Stamp::Kernel
945      || block.tokens.unlist_ref().len() != len
946    {
947      return;
948    }
949    let Some(Entry::Author(block)) = ledger.entries_mut().get_mut(at) else {
950      return;
951    };
952    // (past the stamps' positions, none: the call's own, projected, as a binding's)
953    if let Some(base) = base {
954      block.stamp = Stamp::Kernel;
955      block.range = (base, base + len);
956    }
957    // (the kernel block's own calls are all digested once the mark is reached)
958    block.window_open = false;
959  });

1085/// What a call no rehearsal read printed, as far as its tokens tell (`ApiCall::unread`, `Ledger::unrecorded`), read at
1086/// the API boundary that records its content, so the content check reads text alone, parted into words as an ink's is
1087/// (`resolver::words::printed_words`): its characters; a control symbol or a one-letter control word inside a word (an
1088/// accent, `Universit\"at`, a dotless i: a letter or digit, or a place read past, before it; a letter, a digit or a
1089/// group after it, past any group's end and one-letter control word, `F\'{\i}sica`, `Bras\'\i lia`; not past a control
1090/// symbol, `Smith\,\\Univ` two words), an accent's control symbol or a one-letter control word before a word's first
1091/// letter (`\'Ecole`, review revH3 2, [`accent_symbol`]; `\v{S}koda`, `\c{C}elik`, review revCK1b finding 4: the
1092/// accent and its argument one printed letter where the word begins) and a group there (`M{\"u}nchen`) part nothing,
1093/// the word counted whole, its accents and dotless letters unread (the check takes a word one side reads without them for the one the other prints with them,
1094/// `Fsica` for `física`, `resolver::placed`); a spacing control symbol (`Dept\\Muller`, `Dept,\\Muller`, review revCK1
1095/// 2-3; [`spacing_symbol`]; a `\\`'s `*` and `[…]` print nothing, `\\[2mm]`, review revCK1b finding 5,
1096/// `rehearsal::past_line_break`; a tabular's own call prints nothing, `\begin{tabular}{c}`, review revI2 finding 1,
1097/// but the text its column spec places, `@{Inst }`, review revI3 finding 1, `rehearsal::past_tabular_call`), another control sequence, its argument's group (`\IEEEauthorrefmark{1}Univ Alpha` →
1098/// ` 1 Univ Alpha`) or a math shift parts words. Nothing is spelled but a control sequence's length, its letter's
1099/// catcode and a control symbol's character: a letter command (`Gro{\ss}e`, a control word) parts its word.
1100pub(in crate::base_utilities) fn content_text(content: &Tokens) -> ContentText {
1101  let mut text = String::new();
1102  // (where it read past a control symbol or a one-letter control word inside a word, or an accent's control symbol or a
1103  // one-letter control word before its first letter, `\'Ecole Normale` (review revH3 2), `\v{S}koda`: its byte in `text`, and whether a one-letter control
1104  // word, which may print a letter)
1105  let mut past: Vec<(usize, bool)> = Vec::new();
1106  // (for each open group, whether it is a parting command's argument, which parts what is around it)
1107  let mut groups: Vec<bool> = Vec::new();
1108  let mut parted_by_command = false;
1109  let tokens = content.unlist_ref();
1110  // (past what prints nothing: a `\\`'s `*` and `[…]`, `\\[2mm]`, review revCK1b finding 5; a tabular's own call,
1111  // `\begin{tabular}{c}`, but for the spans its column spec prints, `{@{Inst }c}`, review revI3 finding 1)
1112  let mut unprinted_to = 0;
1113  let mut printed: Vec<std::ops::Range<usize>> = Vec::new();
1114  for (at, token) in tokens.iter().enumerate() {
1115    if at < unprinted_to && !printed.iter().any(|span| span.contains(&at)) {
1116      // (what prints nothing parts what is around it: `@{ and }` read apart from the spec around it)
1117      if !text.ends_with(' ') {
1118        text.push(' ');
1119      }
1120      parted_by_command = false;
1121      continue;
1122    }
1123    // (inside a word, a letter or digit or a place read past before it, or before its first letter, an accent's
1124    // control symbol or a one-letter control word there (`\v{S}koda`, review revCK1b finding 4; `\'Ecole`, never
1125    // `\\`, `\ `, `\,`, `\;`, `\:`, `\>`, the [`spacing_symbol`]s: review revCK1 2); and a letter, a digit or a group
1126    // after it, past any group's end and one-letter control word: `F\'{\i}sica`, `Bras\'\i lia`; a control
1127    // symbol before another parts, `Smith\,\\Univ` two words; a spacing one parts, `Dept\\Muller`, review revCK1 3)
1128    let in_word = || {
1129      !spacing_symbol(token)
1130        && (text.chars().next_back().is_some_and(char::is_alphanumeric)
1131          || past.last().is_some_and(|&(place, _)| place == text.len())
1132          || accent_symbol(token)
1133          || letter_cs(token))
1134        && tokens[at + 1..]
1135          .iter()
1136          .find(|next| next.get_catcode() != Catcode::END && !letter_cs(next))
1137          .is_some_and(starts_word)
1138    };
1139    let mut parting_command = false;
1140    match token.get_catcode() {
1141      Catcode::LETTER | Catcode::OTHER | Catcode::ACTIVE => text.push_str(&token.to_string()),
1142      Catcode::BEGIN => groups.push(parted_by_command),
1143      Catcode::END => {
1144        if groups.pop().unwrap_or(true) {
1145          text.push(' ');
1146        }
1147      },
1148      Catcode::CS => {
1149        if let Some((end, spans)) = past_tabular_call(tokens, at) {
1150          // (a tabular's own call prints nothing but its spec's printed spans: `\begin{tabular}{c}`, review revI2
1151          // finding 1; `{c@{ and }c}` prints "and", review revI3 finding 1)
1152          text.push(' ');
1153          unprinted_to = end;
1154          printed = spans;
1155        } else if let Some(end) = past_line_break(tokens, at) {
1156          // (a `\\` parts words and takes no group, its `*` and `[…]` print nothing: `\\[2mm]`, review revCK1b
1157          // finding 5)
1158          text.push(' ');
1159          unprinted_to = end;
1160        } else if short_cs(token) && in_word() {
1161          past.push((text.len(), letter_cs(token)));
1162        } else {
1163          text.push(' ');
1164          // (a spacing symbol takes no group: `\\{\'E}cole` one word)
1165          parting_command = !spacing_symbol(token);
1166        }
1167      },
1168      _ => text.push(' '),
1169    }
1170    parted_by_command = parting_command;
1171  }
1172  // (each word it read past one in, with the places in it, in characters)
1173  let mut read_past = Vec::new();
1174  let mut slots = past.into_iter().peekable();
1175  for word in printed_words(&text) {
1176    let start = word.as_ptr() as usize - text.as_ptr() as usize;
1177    let mut places = Vec::new();
1178    while let Some(&(at, letter)) = slots.peek()
1179      && at <= start + word.len()
1180    {
1181      slots.next();
1182      if at >= start {
1183        places.push((text[start..at].chars().count(), letter));
1184      }
1185    }
1186    if !places.is_empty() {
1187      read_past.push((word.to_lowercase(), places));
1188    }
1189  }
1190  ContentText { text: text.into(), read_past }
1191}

1531/// A call's label with its parameterless macros written out — `fuzzy:\lx@aas@correspondent` is "fuzzy:Bob Baker", aastex's
1532/// `\email` after a `\correspondingauthor` (aastex_cls.rs `\lx@aas@email@corresp`; 2506.07978, whose label spelled no
1533/// one and went to the block): the label today's `\lx@annotate@frontmatter@now` reads once the flush digests its
1534/// keyvals. A pure lookup of each macro's token body, nothing run through the gullet: the reading changes no state.
1535/// Bounded by the tokens visited (macros expanding to nothing included, [`LABEL_VISITS`]) and the depth (8).
1536fn label_text(value: &Tokens) -> String {
1537  fn written(tokens: &[Token], depth: usize, visited: &mut usize, out: &mut Vec<Token>) {
1538    for t in tokens {
1539      if *visited >= LABEL_VISITS {
1540        return;
1541      }
1542      *visited += 1;
1543      if depth < 8
1544        && t.get_catcode() == Catcode::CS
1545        && let Some(Stored::Expandable(defn)) = lookup_meaning(t)
1546        && defn
1547          .get_parameters()
1548          .is_none_or(|p| p.get_parameters().is_empty())
1549        && let Some(ExpansionBody::Tokens(body)) = defn.get_expansion()
1550      {
1551        written(body.unlist_ref(), depth + 1, visited, out);
1552      } else {
1553        out.push(*t);
1554      }
1555    }
1556  }
1557  let mut out = Vec::new();
1558  written(value.unlist_ref(), 0, &mut 0, &mut out);
1559  Tokens::new(out).to_string()
1560}

1562/// Whether `content` is wrapped in `\lx@affiliation@withsup` (inside the other affiliation wrapper or not), as
1563/// `queued_affiliation_text` unwraps it.
1564fn wraps_withsup(content: &Tokens) -> bool {
1565  let mut v = content.unlist_ref().as_slice();
1566  loop {
1567    let wrapped = v.len() >= 3
1568      && v[1].get_catcode() == Catcode::BEGIN
1569      && skip_group(v, 1) == v.len()
1570      && (v[0] == T_CS!("\\lx@affiliation@withinst") || v[0] == T_CS!("\\lx@affiliation@withsup"));
1571    if !wrapped {
1572      return false;
1573    }
1574    if v[0] == T_CS!("\\lx@affiliation@withsup") {
1575      return true;
1576    }
1577    v = &v[2..v.len() - 1];
1578  }
1579}

1584/// The content of the first top-level brace group of `tokens` (an annotation command's parent tag).
1585fn first_group(tokens: &[Token]) -> Tokens {
1586  let mut depth = 0usize;
1587  for (i, t) in tokens.iter().enumerate() {
1588    match t.get_catcode() {
1589      Catcode::BEGIN => depth += 1,
1590      Catcode::END if depth > 0 => {
1591        depth -= 1;
1592        if depth == 0 {
1593          let open = tokens[..i]
1594            .iter()
1595            .position(|t| t.get_catcode() == Catcode::BEGIN)
1596            .unwrap_or(0);
1597          return Tokens::new(tokens[open + 1..i].to_vec());
1598        }
1599      },
1600      _ => {},
1601    }
1602  }
1603  Tokens::new(Vec::new())
1604}
