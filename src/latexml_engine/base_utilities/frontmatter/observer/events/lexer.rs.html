1//! The lexer's second pass (OXIDIZED_DESIGN_DIVERGENCES #837): the walk's atoms — ink, spacing items, finished
2//! events — into the block's events: name runs split at the words "and" and `&` and at a separator glyph standing alone
3//! (`is_separator_word`), footnote-symbol characters and the other mark glyphs (`is_mark_glyph`) a Mark, commas a
4//! separator, a spacing run a Gap when it separates, a font change between runs a Layout event.

8/// The lexer's second pass: the atoms into events.
9#[derive(Default)]
10pub(super) struct Lexer {
11  pub(super) events:   Vec<Event>,
12  /// the name run being read: its characters with their spans, its font, its nest
13  run:                 Vec<(char, Option<(u16, u16)>)>,
14  /// each of its characters' own token position, beside `run`: a box's cover is wider than what it prints (`Event::marks`)
15  run_own:             Vec<Option<u16>>,
16  /// the font of the `@` it prints
17  address_font:        Option<Rc<Font>>,
18  run_font:            Option<FontKey>,
19  run_nest:            u8,
20  /// the spacing run since the last ink: items, width, fill, span
21  gap_items:           usize,
22  gap_width:           f64,
23  gap_fill:            bool,
24  gap_span:            Option<(u16, u16)>,
25  /// the spacing character the run opened with, when it is one printed character so far (a space or a tie)
26  gap_char:            Option<char>,
27  /// the line has shown ink (a name, a mark, a contact) since its last break
28  line_inked:          bool,
29  /// the font of the last name run
30  last_font:           Option<FontKey>,
31  /// the symbol mark being read (consecutive footnote-symbol characters and mark glyphs)
32  symbol:              String,
33  symbol_span:         Option<(u16, u16)>,
34  /// a printed space before the symbol mark being read
35  symbol_space:        bool,
36  /// what encloses the symbol mark being read: its first character's base, its last's end
37  symbol_formula:      Option<MarkFormula>,
38  /// the brackets `(`, `[` printed and not yet closed since the line's start, and how many were open where the name run
39  /// being read began: inside one a comma or the word "and" is the bracket's text, no separator (`Laurent Schoeffel (on
40  /// behalf of the H1 and ZEUS collaborations)`, 0811.3142; review rev94a B2)
41  open:                u8,
42  run_open:            u8,
43  /// the braces printed and not yet closed since the column's start ([`Event::braces`])
44  braces:              u8,
45  /// which lines (between Break events) print an `@`: their names' characters' positions are recorded (`inks`; a list
46  /// part before the address, `jhansen, msandeep,dudek@cim.mcgill.ca`, 1811.10103) — no other's, a thousand-author names line
47  /// none (1901.01540)
48  pub(super) at_lines: Vec<bool>,
49  line:                usize,
50  /// where the names on those lines print, and the font of each `@` they print ([`InkTable`]): the name run's `@`s, each at
51  /// its token's position, kept when the run is flushed on such a line
52  pub(super) inks:     InkTable,
53  run_at_fonts:        Vec<(u16, Rc<Font>)>,
54}

56impl Lexer {
57  fn gap_is_separator(&self) -> bool {
58    self.gap_fill || self.gap_items >= 2 || (self.gap_items == 1 && self.gap_width > INTERWORD)
59  }
60
61  /// Whether the pending gap is the word space of letter-spaced text: one spacing item narrower than a quad after a
62  /// run of letters each set apart (two or more single letters, the run's last word one letter, with or without its
63  /// period) — jfm's `F\ls I\ls R\ls S\ls T\ns A.\ns L\ls A\ls S\ls T` (jfm.cls:1024-1025, `\ls` `\kern.15em`, `\ns`
64  /// `\kern.5em`) spells one name, its `\ns` the space between its words (1510.04254, 1403.3574, astro-ph0406083: each
65  /// name read as two or three). Initials (`A. B. C.`) are no letter-spaced run: their letters carry periods.
66  fn letter_spaced_word_space(&self) -> bool {
67    if self.gap_fill || self.gap_items != 1 || self.gap_width >= 1.0 {
68      return false;
69    }
70    let text: String = self.run.iter().map(|(c, _)| *c).collect();
71    let words: Vec<&str> = text.split(' ').filter(|w| !w.is_empty()).collect();
72    let letter = |w: &str| w.chars().count() == 1 && w.chars().all(char::is_alphabetic);
73    let Some(last) = words.last() else {
74      return false;
75    };
76    let last_letter = letter(last) || letter(last.trim_end_matches('.'));
77    last_letter && words.iter().filter(|w| letter(w)).count() >= 2
78  }
79
80  fn clear_gap(&mut self) {
81    self.gap_items = 0;
82    self.gap_width = 0.0;
83    self.gap_fill = false;
84    self.gap_span = None;
85    self.gap_char = None;
86  }
87
88  /// Emit the pending gap when it separates (else it is dropped: an interword space at a boundary).
89  fn flush_gap(&mut self) {
90    if self.gap_is_separator() {
91      let mut gap = Event::new(EventKind::Sep(SepKind::Gap {
92        items: self.gap_items,
93        width: (self.gap_width * 1000.0).round() / 1000.0,
94        fill:  self.gap_fill,
95      }));
96      gap.span = self.gap_span;
97      self.events.push(gap);
98    }
99    self.clear_gap();
100  }
101
102  fn flush_symbol(&mut self) {
103    if self.symbol.is_empty() {
104      return;
105    }
106    let mut mark = Event::new(EventKind::Mark);
107    mark.text = std::mem::take(&mut self.symbol);
108    mark.label = Some(mark.text.clone());
109    // (a footnote symbol only when every character is one: a mark glyph, `⋄`, is a mark of no `\@fnsymbol` kind)
110    mark.symbol = mark.text.chars().all(is_fnsymbol);
111    mark.role = Some("text".to_string());
112    mark.span = self.symbol_span.take();
113    mark.space_before = std::mem::take(&mut self.symbol_space);
114    mark.formula = self.symbol_formula.take();
115    mark.line_first = !self.line_inked;
116    self.line_inked = true;
117    self.events.push(mark);
118  }
119
120  /// End the name run: a Name, split at the words "and" and `&` and at a separator glyph standing alone (Sep events:
121  /// `Chiara Simeoni -- \url{…}`'s `--`, 2511.07187), each piece with its own span. A dash inside a word stays as typed
122  /// (`F.~M\"uller--S\'anchez`, 2003.01448; the XML prints the ligature's `–`): the classify helpers today's split
123  /// shares read `--` as a hyphen and an en dash as no letter — `H. Strecker and H. V. Klapdor--Kleingrothaus`, an
124  /// elsarticle per-author line, read as one name with the en dash (hep-ex9811040, FM S2a.2 r7's census).
125  fn flush_run(&mut self) {
126    let run = std::mem::take(&mut self.run);
127    let run_own = std::mem::take(&mut self.run_own);
128    let font = self.run_font.take();
129    // the run's words, each with the span of its characters, and where each prints on a line printing an `@` (`InkTable`)
130    let record = self.at_lines.get(self.line).copied().unwrap_or(false);
131    let at_fonts = std::mem::take(&mut self.run_at_fonts);
132    if record {
133      self.inks.fonts.extend(at_fonts);
134    }
135    let mut words: Vec<(String, Option<(u16, u16)>)> = Vec::new();
136    let mut owns: Vec<Option<(u16, u16)>> = Vec::new();
137    let mut inks: Vec<Vec<Option<InkAt>>> = Vec::new();
138    let mut word = String::new();
139    let mut word_span = None;
140    let mut word_own = None;
141    let mut word_ink = Vec::new();
142    for ((c, span), own) in run
143      .into_iter()
144      .zip(run_own)
145      .chain(std::iter::once(((' ', None), None)))
146    {
147      if c == ' ' {
148        if !word.is_empty() {
149          words.push((std::mem::take(&mut word), word_span.take()));
150          owns.push(word_own.take());
151          if record {
152            inks.push(std::mem::take(&mut word_ink));
153          }
154        }
155      } else {
156        word.push(c);
157        word_span = join_span(word_span, span);
158        word_own = join_span(word_own, own.map(|own| (own, own + 1)));
159        if record {
160          word_ink.push(span.map(|cover| InkAt { cover, own }));
161        }
162      }
163    }
164    if words.is_empty() {
165      return;
166    }
167    if font.is_some() && self.last_font.is_some() && font != self.last_font {
168      let mut layout = Event::new(EventKind::Layout);
169      layout.role = Some("font".to_string());
170      layout.nest = self.run_nest;
171      self.events.push(layout);
172    }
173    // (the font of its first character, or of the address it prints behind a label word at most, `E-mail: \\texttt{a@x}`: the
174    // emitter's, 0802.3384)
175    let address_font = self.address_font.take().filter(|_| words.len() <= 3);
176    let ink_font = address_font.or(font.as_ref().map(|font| Rc::clone(&font.0)));
177    if font.is_some() {
178      self.last_font = font;
179    }
180    // (the brackets open at each word's start: a word inside one is the bracket's, no separator)
181    let mut open = self.run_open;
182    let mut opens = Vec::with_capacity(words.len());
183    for (word, _) in &words {
184      opens.push(open);
185      open = word.chars().fold(open, bracket_depth);
186    }
187    let mut name: Option<Event> = None;
188    // (where its characters print, on a line printing an `@`)
189    let mut name_ink: Vec<Option<InkAt>> = Vec::new();
190    // (a byline's "by", in any case, opening a line before more words is a Name of its own, its ink as printed: whether it
191    // is the byline the old reader's `without_leading_by` drops is the reading's to say, on a line it reads as names
192    // (`Resolver::names_line`): `\author{by David Harari and Tam\'as Szamuely}`, 0808.2556; an affiliation line's `By
193    // Lee Hall, \textit{Univ X}` keeps it, review revG1 1)
194    let byline = !self.line_inked && words.len() > 1 && words[0].0.eq_ignore_ascii_case("by");
195    for (at, (word, span)) in words.into_iter().enumerate() {
196      if at == 0 && byline {
197        let mut by = Event::new(EventKind::Name);
198        by.nest = self.run_nest;
199        by.font.clone_from(&ink_font);
200        by.text = word;
201        by.span = span;
202        by.marks = owns[at];
203        self.events.push(by);
204        continue;
205      }
206      // (a word "and", a printed `&` — `Zheng Zheng \& Jordi Miralda-Escud\'e`, astro-ph0201275; 0908.0757, a witness
207      // of the old reader's deleted names split — or a separator glyph parts two names)
208      if opens[at] == 0
209        && (word.eq_ignore_ascii_case("and") || word == "&" || is_separator_word(&word))
210      {
211        self.push_name(name.take(), std::mem::take(&mut name_ink));
212        let mut sep = Event::new(EventKind::Sep(SepKind::Word(word)));
213        sep.span = span;
214        sep.nest = self.run_nest;
215        self.events.push(sep);
216        continue;
217      }
218      let name = name.get_or_insert_with(|| {
219        let mut name = Event::new(EventKind::Name);
220        name.nest = self.run_nest;
221        name.font.clone_from(&ink_font);
222        name
223      });
224      if opens[at] == 0
225        && let Some(label) = printed_label(&word)
226      {
227        name.printed_labels.push(PrintedLabel {
228          label: label.to_string(),
229          word: name.text.split(' ').filter(|w| !w.is_empty()).count(),
230          line_first: at == 0 && !self.line_inked,
231          span,
232        });
233      }
234      if !name.text.is_empty() {
235        name.text.push(' ');
236        if record {
237          name_ink.push(None);
238        }
239      }
240      name.text.push_str(&word);
241      if let Some(ink) = inks.get_mut(at) {
242        name_ink.append(ink);
243      }
244      name.span = join_span(name.span, span);
245      name.marks = join_span(name.marks, owns[at]);
246    }
247    self.push_name(name, name_ink);
248    self.line_inked = true;
249  }

251  /// A name read, and where its characters print ([`InkTable`]: recorded on a line printing an `@` only).
252  fn push_name(&mut self, name: Option<Event>, ink: Vec<Option<InkAt>>) {
253    let Some(name) = name else {
254      return;
255    };
256    if !ink.is_empty()
257      && let Some(span) = name.span
258    {
259      self.inks.at.push((span, name.text.clone(), ink));
260    }
261    self.events.push(name);
262  }

264  pub(super) fn push_event(&mut self, mut event: Event) {
265    self.flush_symbol();
266    self.flush_run();
267    match event.kind {
268      EventKind::Break => {
269        self.clear_gap();
270        self.line += 1;
271        self.line_inked = false;
272        self.open = 0;
273        event.braces = self.braces;
274      },
275      EventKind::Mark => {
276        event.space_before = self.gap_items > 0;
277        self.flush_gap();
278        event.line_first = !self.line_inked;
279        self.line_inked = true;
280      },
281      EventKind::Contact | EventKind::Note => {
282        self.flush_gap();
283        self.line_inked = true;
284      },
285      EventKind::Sep(SepKind::And(_) | SepKind::Column(_)) => {
286        self.clear_gap();
287        self.open = 0;
288        self.braces = 0;
289      },
290      _ => self.flush_gap(),
291    }
292    self.events.push(event);
293  }

295  pub(super) fn ink(
296    &mut self,
297    text: &str,
298    span: Option<(u16, u16)>,
299    font: Option<FontKey>,
300    nest: u8,
301    formula: Option<MarkFormula>,
302    own: Option<u16>,
303  ) {
304    for (at, c) in text.char_indices() {
305      // (a zero-width character prints no ink: the zero-width space `{\,\!}` leaves in `${\,\!}^2$`, 2112.13115 —
306      // it was read into the name before the mark and into the entry after it)
307      if is_zero_width(c) {
308        continue;
309      }
310      // (a combining mark is part of the symbol it is on, `\={\dag}`'s `†̄`: no `†` mark and a mark-led name run —
311      // its line read as "̄Univ One" — review rev92b_r3 B1's sibling; `labels::mark_labels` reads a script so)
312      if !self.symbol.is_empty() && is_combining_mark(c) {
313        self.symbol.push(c);
314        self.symbol_span = join_span(self.symbol_span, span);
315        continue;
316      }
317      if is_mark_char(c) {
318        self.flush_run();
319        if self.symbol.is_empty() {
320          self.symbol_space = self.gap_items > 0;
321        }
322        self.flush_gap();
323        // (a letter or digit later in this box prints after it too)
324        let here = formula.map(|f| MarkFormula {
325          ends: f.ends && !has_word(&text[at + c.len_utf8()..]),
326          ..f
327        });
328        self.symbol_formula = match (self.symbol_formula, here) {
329          (Some(first), Some(here)) if !self.symbol.is_empty() => {
330            Some(MarkFormula { ends: here.ends, ..first })
331          },
332          (_, here) => here,
333        };
334        self.symbol.push(c);
335        self.symbol_span = join_span(self.symbol_span, span);
336        continue;
337      }
338      self.flush_symbol();
339      if (c == ',' || c == ';') && self.open == 0 {
340        self.flush_run();
341        self.clear_gap();
342        let mut comma = Event::new(EventKind::Sep(SepKind::Comma));
343        comma.span = span;
344        comma.nest = nest;
345        comma.braces = self.braces;
346        self.events.push(comma);
347        continue;
348      }
349      if c.is_whitespace() {
350        let word = match c {
351          ' ' | '\n' => Some(' '),
352          '\u{a0}' => Some(c),
353          _ => None,
354        };
355        if self.ties_the_gap(word) {
356          self.gap_span = join_span(self.gap_span, span);
357          continue;
358        }
359        self.gap_char = if self.gap_items == 0 { word } else { None };
360        self.gap_items += 1;
361        self.gap_width += space_width(c);
362        self.gap_span = join_span(self.gap_span, span);
363        continue;
364      }
365      if self.gap_items > 0 {
366        // (a printed space between a mark and the ink after it: `space_after`)
367        if self.run.is_empty()
368          && let Some(last) = self.events.last_mut()
369          && last.kind == EventKind::Mark
370        {
371          last.space_after = true;
372        }
373        if (self.gap_is_separator() && !self.letter_spaced_word_space()) || self.run.is_empty() {
374          self.flush_run();
375          self.flush_gap();
376        } else {
377          self.run.push((' ', None));
378          self.run_own.push(None);
379          self.clear_gap();
380        }
381      }
382      // (a font switched inside the run does not end it: the run is one Name whatever fonts its characters are in —
383      // gretsi's `first \textsc{last}`, 2508.03934 —, and the Layout event, `flush_run`, compares the runs' first
384      // fonts: OXIDIZED_DESIGN_DIVERGENCES #837's departures)
385      if self.run.is_empty() {
386        self.run_font = font.clone();
387        self.run_nest = nest;
388        self.run_open = self.open;
389      }
390      self.open = bracket_depth(self.open, c);
391      match c {
392        '{' => self.braces = self.braces.saturating_add(1),
393        '}' => self.braces = self.braces.saturating_sub(1),
394        _ => {},
395      }
396      if c == '@'
397        && let Some(font) = &font
398      {
399        self.address_font = Some(Rc::clone(&font.0));
400        if let Some(at) = own.or(span.map(|s| s.0)) {
401          self.run_at_fonts.push((at, Rc::clone(&font.0)));
402        }
403      }
404      self.run.push((c, span));
405      self.run_own.push(own);
406    }
407  }

409  /// Whether the run so far is no whole name: one word at most, or words ending in an initial (`N. B.`, `V.G.`, `J.-P.`,
410  /// `Yu.`) — the name's surname is still to come.
411  fn run_awaits_its_name(&self) -> bool {
412    let text: String = self.run.iter().map(|(c, _)| *c).collect();
413    let words: Vec<&str> = text.split(' ').filter(|w| !w.is_empty()).collect();
414    words.len() < 2 || words.last().is_some_and(|word| word.ends_with('.'))
415  }

417  /// Whether the word space or tie `word` and the one spacing character the pending gap holds are a space and a tie,
418  /// read as one word space of a name, the tie (the gap's character then spent): `N.~B. ~Li$^{22}$` in the BESIII list
419  /// of 340 (1202.2048) — read as a gap of two items, "N. B." was initials after `Lei~Li$^{1}$,` and both one name
420  /// written surname first, which declined the writer's reading of the whole block, its sub-lettered marks with it (the
421  /// rev107b review's B2); a tie and a space, `V.G.~ Kagramanova` (0802.4349), alike. Only inside one name
422  /// ([`Lexer::run_awaits_its_name`]): after a whole name they are a gap of two, the gap between two names
423  /// (`Gasnikov~\IEEEmembership{}⏎ Nikolay Malkovsky`, 1805.06045; `Ann Able ~Bob Baker`).
424  fn ties_the_gap(&mut self, word: Option<char>) -> bool {
425    let tie = self.gap_items == 1
426      && !self.gap_fill
427      && word.is_some_and(|c| self.gap_char.is_some_and(|first| first != c))
428      && self.run_awaits_its_name();
429    if tie {
430      self.gap_char = None;
431    }
432    tie
433  }

435  pub(super) fn space(
436    &mut self,
437    width: f64,
438    fill: bool,
439    span: Option<(u16, u16)>,
440    word: Option<char>,
441  ) {
442    self.flush_symbol();
443    if self.ties_the_gap(word) {
444      self.gap_span = join_span(self.gap_span, span);
445      return;
446    }
447    self.gap_char = if self.gap_items == 0 { word } else { None };
448    self.gap_items = self.gap_items.saturating_add(1);
449    self.gap_width += width;
450    self.gap_fill |= fill;
451    self.gap_span = join_span(self.gap_span, span);
452  }

454  pub(super) fn finish(mut self) -> (Vec<Event>, InkTable) {
455    self.flush_symbol();
456    self.flush_run();
457    self.clear_gap();
458    let inks = std::mem::take(&mut self.inks);
459    // a `\label` outside a note labels the note before it
460    let mut events: Vec<Event> = Vec::with_capacity(self.events.len());
461    let glued = glue_in_word_marks(self.events);
462    let glued = mark_ordinals(glued);
463    for event in glued {
464      if event.kind == EventKind::Layout && event.role.as_deref() == Some("label") {
465        if let Some(note) = events.iter_mut().rev().find(|e| e.kind == EventKind::Note)
466          && note.label.is_none()
467        {
468          note.label = event.label;
469          continue;
470        }
471        events.push(event);
472        continue;
473      }
474      events.push(event);
475    }
476    (events, inks)
477  }
478}

480/// The label a word of a name run prints by hand ([`PrintedLabel`]): `a)`, `12)`, one or two letters or digits and its `)`.
481fn printed_label(word: &str) -> Option<&str> {
482  let label = word.strip_suffix(')')?;
483  (matches!(label.chars().count(), 1 | 2) && label.chars().all(char::is_alphanumeric))
484    .then_some(label)
485}

487/// Mark each ordinal's suffix ([`Event::ordinal`]): a Mark printing `st`, `nd`, `rd` or `th` right after a Name ending in
488/// a digit.
489fn mark_ordinals(mut events: Vec<Event>) -> Vec<Event> {
490  for at in 1..events.len() {
491    let number = events[at - 1].kind == EventKind::Name
492      && events[at - 1]
493        .text
494        .chars()
495        .next_back()
496        .is_some_and(|c| c.is_ascii_digit());
497    let event = &mut events[at];
498    event.ordinal = number
499      && event.kind == EventKind::Mark
500      && matches!(event.label.as_deref(), Some("st" | "nd" | "rd" | "th"));
501  }
502  events
503}

505/// The brackets open after `c`, `open` open before it: a `(` or `[` opens one, a `)` or `]` closes one.
506fn bracket_depth(open: u8, c: char) -> u8 {
507  match c {
508    '(' | '[' => open.saturating_add(1),
509    ')' | ']' => open.saturating_sub(1),
510    _ => open,
511  }
512}

514/// A lowercase mark glued on both sides inside a word is the word's letters, not a mark between two names: `James
515/// M\textsuperscript{c}Kernan` is one name, "James McKernan" (math0610203). A Name, a Mark of one or two lowercase
516/// letters set with no space before or after it, and a Name are one Name. So is a mark of digits glued so, as today's
517/// reads it (`rewrite_in_word_superscripts`, 77n r4: no mark list, no "and" after it): `$^1$NLP$^2$CT Lab, Univ X` is
518/// the lab "NLP²CT Lab" (2603.20851, 2605.30274; `I$^2$R`, `DM$^3$L`, 2603.04869, 2604.00800), not entries "NLP" and
519/// "CT Lab, Univ X" — which the writer, cutting a block entry by its whole span (OD #945), would have written — when
520/// it stands glued in the source too ([`glued_in_source`]). So is a symbol on a line a mark leads, an entry, which names no
521/// one: `$^1$A$^*$STAR`, `$^1$IHPC A$^*$STAR` (2401.15006's institute, cut at its star), whatever `Bob Baker$^{*}$` shows elsewhere.
522/// On a line no mark leads it stays a mark: `Cher$^*$Bob Baker`.
523fn glue_in_word_marks(events: Vec<Event>) -> Vec<Event> {
524  let glued_script =
525    |e: &Event| e.kind == EventKind::Mark && e.formula.is_some_and(|f| f.base.is_some() && !f.ends);
526  let requested = events
527    .iter()
528    .any(glued_script)
529    .then(|| names_requested(&events));
530  let mut out: Vec<Event> = Vec::with_capacity(events.len());
531  let mut events = events.into_iter().peekable();
532  // (whether the line this event stands on is led by a mark; none printed yet)
533  let mut mark_led: Option<bool> = None;
534  while let Some(event) = events.next() {
535    match event.kind {
536      EventKind::Break | EventKind::Sep(SepKind::And(_) | SepKind::Column(_)) => mark_led = None,
537      EventKind::Layout => {},
538      _ => mark_led = mark_led.or(Some(event.kind == EventKind::Mark)),
539    }
540    let letters = event.label.as_deref().unwrap_or("");
541    let next = events.peek();
542    let number_or_letter = !event.symbol
543      && ((1..=2).contains(&letters.chars().count()) && letters.chars().all(char::is_lowercase)
544        || (1..=3).contains(&letters.chars().count())
545          && letters.chars().all(|c| c.is_ascii_digit())
546          && glued_in_source(out.last(), &event, next));
547    let in_word = event.kind == EventKind::Mark
548      && !event.space_before
549      && !event.space_after
550      && (number_or_letter
551        || (mark_led == Some(true)
552          && is_symbol_run(letters)
553          && glued_in_source(out.last(), &event, next)))
554      && out.last().is_some_and(|e| {
555        e.kind == EventKind::Name && e.text.chars().next_back().is_some_and(char::is_alphabetic)
556      })
557      && next.is_some_and(|e| {
558        e.kind == EventKind::Name && e.text.chars().next().is_some_and(char::is_alphabetic)
559      });
560    // (a script a letter or digit prints after in its own formula is the formula's text, a prescript or an exponent
561    // (`${\rm ^{3}He}$`, `${}^{87}{\rm Rb}$`; OD #911, 2105.07958's sibling), unless a name requests it (`${\rm
562    // ^2Lab}$` under `Bob Baker$^{2}$`, revISO1 B1) or, none requesting, it opens its line (`$ {\rm ^2Solid}$`, 1006.5052)
563    if glued_script(&event)
564      && match &requested {
565        Some(requested) if !requested.is_empty() => !is_requested(&event, requested),
566        _ => !event.line_first,
567      }
568      && let Some(next) = events.next_if(|e| e.kind == EventKind::Name)
569    {
570      out.push(Event {
571        text: format!("{}{}", event.text, next.text),
572        span: join_span(event.span, next.span),
573        ..next
574      });
575      continue;
576    }
577    if in_word {
578      let next = events.next().expect("peeked");
579      let name = out.last_mut().expect("checked");
580      name.text.push_str(letters);
581      name.text.push_str(&next.text);
582      name.span = join_span(join_span(name.span, event.span), next.span);
583      continue;
584    }
585    out.push(event);
586  }
587  out
588}

590/// Whether a mark's label prints footnote symbols or mark glyphs alone.
591fn is_symbol_run(label: &str) -> bool {
592  !label.is_empty() && label.chars().all(|c| is_fnsymbol(c) || is_mark_glyph(c))
593}

595/// Whether a mark stands between two names with no space printed or written on either side (`NLP$^2$CT`): the in-word
596/// test's frame, which a mark after a name and a space (`Able$^{1}$ Bob`) fails.
597pub(super) fn glued_inside_a_word(before: &Event, mark: &Event, after: Option<&Event>) -> bool {
598  !mark.space_before
599    && !mark.space_after
600    && after.is_some_and(|e| e.kind == EventKind::Name)
601    && glued_in_source(Some(before), mark, after)
602}

604/// Whether a digit mark stands in the source between the names around it with nothing but its own math shifts (or
605/// none) between them, as `rewrite_in_word_superscripts` reads a superscript inside a word: `NLP$^2$CT`, `I$^2$R` —
606/// not `Russia\break $^2$Nuclear` (hep-ph9809222's revtex `\address`, a line break before the next entry's mark,
607/// which TeX's spaces leave unseen).
608fn glued_in_source(before: Option<&Event>, mark: &Event, after: Option<&Event>) -> bool {
609  match (
610    before.and_then(|e| e.span),
611    mark.span,
612    after.and_then(|e| e.span),
613  ) {
614    (Some((_, end)), Some((start, stop)), Some((next, _))) => {
615      start <= end.saturating_add(1) && next <= stop.saturating_add(1)
616    },
617    _ => false,
618  }
619}

629/// LaTeX's `\@fnsymbol` characters (latex.ltx `\@fnsymbol`: `*`, `\dagger`, `\ddagger`, `\mathsection`,
630/// `\mathparagraph`, `\|`, and their doubles), with the math asterisk; `\|` as the kernel prints it in a formula, `∥`

636    '*' | '\u{2217}' | '†' | '‡' | '§' | '¶' | '‖' | '\u{2225}'
637  )
638}

689/// degree, standing alone: a bar, a colon, the math dot `\cdot` or a bullet, a dash (`Tom Y. Zhao$^{\dagger}$ | PhD`,
690/// 2311.00652; `MeerKLASS Collaboration\thanks{…}: Matilde Barberi-Squarotti`, 2407.21626; `Chiara Simeoni --

693/// prints between its parts as often (`Ann \textperiodcentered{} Able`, `perfect_kernel_batchfms2a2r6`).
694pub(in crate::base_utilities) fn is_separator_word(word: &str) -> bool {

696    && word
697      .chars()
698      .all(|c| matches!(c, '|' | ':' | '⋅' | '•' | '∙' | '-' | '–' | '—'))
