26/// Whether `tokens` can print a mark at all: a command or an active character that may ([`prints`]), read through
27/// parameterless macros' bodies, or a superscript; a character typed in the argument is its text. The marks queries ask
28/// it before their rehearsal (2103.04797's 1,161 `\affiliation{\Amsterdam}`, an institute's name; 1901.01540's 1,204
29/// names, `\author{M.~Soares-Santos}`: not rehearsed).
30fn may_mark(tokens: &[Token]) -> bool {
31  walk(tokens, MACRO_DEPTH, &mut |ink| match ink {
32    Ink::Char(text, typed) if typed || !text.chars().any(is_mark_char) => Continue(()),
33    Ink::Space | Ink::Command(Prints::Nothing | Prints::NoMark) | Ink::Other(true) => Continue(()),
34    _ => Break(true),
35  })
36  .is_break()
37}
38
39/// How many macros deep the gates read a body.
40const MACRO_DEPTH: u8 = 8;
41
42/// What the gates' walk ([`walk`]) meets, braces aside.
43enum Ink<'s> {
44  /// a character's text, `true` typed in the argument, `false` in a body
45  Char(&'s str, bool),
46  Space,
47  /// a command or an active character, a parameterless macro's body only past [`MACRO_DEPTH`]
48  Command(Prints),
49  Superscript,
50  /// any other token (a formula's shift, a `&`), `true` typed in the argument
51  Other(bool),
52}
53
54/// The marks gates' one walk: `tokens` in order, a parameterless macro's body read in its place `depth` macros deep, until
55/// `gate` breaks with its answer. A command whose reading went on is not read again.
56fn walk(
57  tokens: &[Token],
58  depth: u8,
59  gate: &mut impl FnMut(Ink<'_>) -> ControlFlow<bool>,
60) -> ControlFlow<bool> {
61  walk_in(tokens, true, depth, &mut Vec::new(), gate)
62}
63
64/// [`walk`] on the argument's own tokens (`typed`) or a body's; `read_on` the commands whose reading went on.
65fn walk_in(
66  tokens: &[Token],
67  typed: bool,
68  depth: u8,
69  read_on: &mut Vec<Token>,
70  gate: &mut impl FnMut(Ink<'_>) -> ControlFlow<bool>,
71) -> ControlFlow<bool> {
72  for t in tokens {
73    match t.get_catcode() {
74      Catcode::BEGIN | Catcode::END => {},
75      Catcode::SPACE => gate(Ink::Space)?,
76      Catcode::LETTER | Catcode::OTHER => t.with_str(|s| gate(Ink::Char(s, typed)))?,
77      Catcode::CS | Catcode::ACTIVE => {
78        if read_on.contains(t) {
79          continue;
80        }
81        match prints(t) {
82          Prints::Body(body) if depth > 0 => {
83            walk_in(body.unlist_ref(), false, depth - 1, read_on, gate)?
84          },
85          printed => gate(Ink::Command(printed))?,
86        }
87        read_on.push(*t);
88      },
89      Catcode::SUPER => gate(Ink::Superscript)?,
90      _ => gate(Ink::Other(typed))?,
91    }
92  }
93  Continue(())
94}

96/// What a command or an active character prints, as the marks gates read it.
97enum Prints {
98  /// a parameterless macro's body
99  Body(Tokens),
100  /// nothing: a macro of no expansion (`\newcommand\affilfont{}`)
101  Nothing,
102  /// no mark, the tokens after it read on: the kernel's `\relax` and tie (`~`'s `\lx@NBSP`), an accent
103  /// (`accent_inner_name`, on the argument after it), a primitive or a `\chardef` printing no mark character (`\ss`,
104  /// `\&`)
105  NoMark,
106  /// anything: a command the rehearsal binds ([`rehearsal::rebinds`]), a macro taking arguments, any other primitive
107  Any,
108}
109
110/// What `t` prints ([`Prints`]), by its meaning here.
111fn prints(t: &Token) -> Prints {
112  if rehearsal::rebinds(t) {
113    return Prints::Any;
114  }
115  match lookup_meaning(t) {
116    Some(Stored::Expandable(defn)) => {
117      if defn.get_origin().is_latexml_owned() && defn.get_cs().with_str(is_accent_inner) {
118        return Prints::NoMark;
119      }
120      if defn
121        .get_parameters()
122        .is_some_and(|p| !p.get_parameters().is_empty())
123      {
124        return Prints::Any;
125      }
126      match defn.get_expansion() {
127        Some(ExpansionBody::Tokens(body)) => Prints::Body(body.clone()),
128        None => Prints::Nothing,
129        _ => Prints::Any,
130      }
131    },
132    Some(Stored::Primitive(defn)) if defn.origin.is_latexml_owned() => {
133      // (`\relax`, a robust command's `\protect`; the tie, `~`, `\nobreakspace`)
134      let blank = [
135        T_CS!("\\relax"),
136        T_CS!("\\lx@NBSP"),
137        T_CS!("\\lx@nobreakspace"),
138      ]
139      .contains(&defn.cs);
140      match &defn.replacement {
141        _ if blank => Prints::NoMark,
142        Some(PrimitiveBody::String(text)) if !with(*text, |s| s.chars().any(is_mark_char)) => {
143          Prints::NoMark
144        },
145        _ => Prints::Any,
146      }
147    },
148    // (a `\chardef`, `\&`)
149    Some(Stored::Register(reg)) => match reg.chardef_glyph() {
150      Some(glyph) if !with(glyph, |g| g.chars().any(is_mark_char)) => Prints::NoMark,
151      _ => Prints::Any,
152    },
153    _ => Prints::Any,
154  }
155}

157/// The events of `arg`, or none read where it cannot print a mark.
158fn mark_events(arg: &Tokens) -> Result<Option<Vec<Event>>> {
159  if !may_mark(arg.unlist_ref()) {
160    return Ok(Some(Vec::new()));
161  }
162  Ok(read_argument(arg)?.map(|ArgumentReading { events, .. }| events))
163}

165/// The level `events` print their text at: the outermost a word of them stands at, inside the box when one box holds
166/// it all (IEEEtran's `\IEEEauthorblockN{\parbox{…}{Ann\IEEEauthorrefmark{4}, …}}`, 2508.17623; acmart's
167/// `\institution{$^1$Univ A …}`; apa6's `$^{1}$\mbox{Univ A}`, 2508.05776).
168fn own_level(events: &[Event]) -> u8 {
169  events
170    .iter()
171    .filter(|e| e.kind == EventKind::Name)
172    .map(|e| e.nest)
173    .min()
174    .unwrap_or(0)
175}
176
177/// Which of `events` print at the argument's own level: at or above its words' ([`own_level`]), or in a box of their own
178/// there, one holding no word (`\raisebox{0pt}[0pt][0pt]{\textsuperscript{1}}`, 2307.12121, 2311.01786); a mark in a box
179/// that also holds words is none.
180fn at_own_level(events: &[Event]) -> Vec<bool> {
181  let level = own_level(events);
182  let mut own = vec![true; events.len()];
183  let mut at = 0;
184  while at < events.len() {
185    if events[at].nest <= level {
186      at += 1;
187      continue;
188    }
189    // (the run of deeper events a box at the words' level holds)
190    let end = events[at..]
191      .iter()
192      .position(|e| e.nest <= level)
193      .map_or(events.len(), |k| at + k);
194    let wordless = events[at..end].iter().all(|e| e.kind != EventKind::Name);
195    own[at..end].fill(wordless);
196    at = end;
197  }
198  own
199}

201/// The marks `arg` prints at its own level (`at_own_level`), outside its notes (a note's marks are its `inner`), in
202/// order, each with its printed `label`; a number's suffix (`5\textsuperscript{th}`, `Event::ordinal`) is text, no mark.
203/// A mark a document macro prints is one (`\newcommand\ua{\textsuperscript{1}}`), as is a `\let` copy's.

208/// [`argument_marks`], and the name text of the same rehearsal ([`argument_name_text`]) where one was made.
209pub(in crate::base_utilities) fn argument_marks_and_name(
210  arg: &Tokens,
211) -> Result<Option<(Vec<Event>, Option<String>)>> {
212  if !may_mark(arg.unlist_ref()) {
213    return Ok(Some((Vec::new(), None)));
214  }
215  Ok(read_argument(arg)?.map(|ArgumentReading { events, .. }| {
216    let name = name_text(&events);
217    let own = at_own_level(&events);
218    let marks = events
219      .into_iter()
220      .zip(own)
221      .filter(|(e, own)| *own && e.kind == EventKind::Mark && !e.ordinal)
222      .map(|(e, _)| e)
223      .collect();
224    (marks, Some(name))
225  }))
226}

247/// What `tokens` print where they are characters and groups alone; else `None` (a tie too: the rehearsal reads it).
248fn typed_text(tokens: &[Token]) -> Option<String> {
249  let mut text = String::new();
250  let walked = walk(tokens, 0, &mut |ink| {
251    match ink {
252      Ink::Char(s, _) => text.push_str(s),
253      Ink::Space => text.push(' '),
254      _ => return Break(false),
255    }
256    Continue(())
257  });
258  walked.is_continue().then_some(text)
259}

261/// Whether `arg` prints a mark at any level (in a box beside its words too, `Ann\mbox{$^{1}$}`), and whether one is a
262/// label ([`is_label_mark`]), a `\footnotemark` and a number's suffix none; unread, by its tokens
263/// (`author_mark_position`, `leads_with_mark` from each mark), counted for `caller`.
264pub fn prints_marks_and_labels(arg: &Tokens, caller: &str) -> Result<(bool, bool)> {
265  if let Some(events) = mark_events(arg)? {
266    let marks: Vec<&Event> = events.iter().filter(|e| own_mark(e)).collect();
267    return Ok((!marks.is_empty(), marks.into_iter().any(is_label_mark)));
268  }
269  count_unread(caller);
270  let v = arg.unlist_ref();
271  let any = author_mark_position(arg).is_some();
272  let mut from = 0;
273  while let Some(at) = author_mark_position(&Tokens::new(v[from..].to_vec())) {
274    let at = from + at - 1;
275    if leads_with_mark(&Tokens::new(v[at..].to_vec())) {
276      return Ok((any, true));
277    }
278    from = at + 1;
279  }
280  Ok((any, false))
281}

311/// [`prints_list_marks`] of a list's events.
312fn list_prints_marks(events: &[Event]) -> bool {
313  let own = at_own_level(events);
314  events
315    .iter()
316    .zip(own)
317    .filter(|(_, own)| *own)
318    .any(|(e, _)| {
319      own_mark(e)
320        || (e.kind == EventKind::Note && e.inner.iter().any(|m| m.line_first && own_mark(m)))
321    })

324/// A list's marks as its cutter reads them ([`list_marks`]).
325pub struct ListMarks {
326  /// it prints a mark ([`prints_list_marks`])
327  pub prints:                           bool,
328  /// where its entries' opening marks stand in it, as the API Cut's decision cuts it (`events::entry_heads`); `None`
329  /// unread
330  pub starts:                           Option<Vec<usize>>,
331  /// where every mark it prints at its level stands, and a note opening with a label (as `prints_list_marks` reads them)
332  pub marked:                           Option<Vec<usize>>,
333  /// the reading's events, which say what each of its lines prints (`entry_starts::lines_starts`); none unread
334  pub(in crate::base_utilities) events: Vec<Event>,
335}

337/// Whether a list `arg` prints a mark ([`prints_list_marks`]) and where the marks opening its entries stand, the one
338/// reading of its entries for its cutter and the Cut's decision (`requested_entry_labels`): a mark a document macro prints
339/// (`\newcommand\ua{\textsuperscript{1}}`) or one in a group (`{\it \affmark{1} Univ A}`) opens one as `$^{1}$` does.
340/// Unread, the starts are none, counted for `caller`, and a mark printed is the tokens' (`author_mark_position`).
341pub fn list_marks(arg: &Tokens, caller: &str) -> Result<ListMarks> {
342  let Some(events) = mark_events(arg)? else {
343    count_unread(caller);
344    let prints = author_mark_position(arg).is_some();
345    return Ok(ListMarks {
346      prints,
347      starts: None,
348      marked: None,
349      events: Vec::new(),
350    });
351  };
352  let prints = list_prints_marks(&events);
353  if !prints {
354    return Ok(ListMarks {
355      prints,
356      starts: Some(Vec::new()),
357      marked: Some(Vec::new()),
358      events,
359    });
360  }
361  // (the queued names' marks read first: the entries part at the marks they request, `events::names_requested`)
362  queued_author_marks()?;
363  // (where a mark stands: in a formula on no base where the formula does, `$\rm ^{2}$`, `\( ^{4} \)`; on a base where it
364  // prints, `Lab $5^{2}$Univ Y`; a note opening with a label where the note does, 1504.01072)
365  let at = |mark: &Event| {
366    match &mark.formula {
367      Some(formula) if formula.base == Some(events::MarkBase::Floating) => {
368        formula.span.or(mark.span)
369      },
370      _ => mark.span,
371    }
372    .map(|(start, _)| usize::from(start))
373  };
374  let starts: Option<Vec<usize>> = events::entry_heads(&events)
375    .iter()
376    .filter(|head| own_mark(head))
377    .map(at)
378    .collect();
379  let own = at_own_level(&events);
380  let marked: Option<Vec<usize>> = events
381    .iter()
382    .zip(own)
383    .filter(|(_, own)| *own)
384    .filter_map(|(e, _)| {
385      if own_mark(e) {
386        Some(at(e))
387      } else if e.kind == EventKind::Note && e.inner.iter().any(|m| m.line_first && own_mark(m)) {
388        Some(e.span.map(|(start, _)| usize::from(start)))
389      } else {
390        None
391      }
392    })
393    .collect();
394  let starts = starts.zip(marked);
395  if starts.is_none() {
396    count_unread(caller);
397  }
398  let (starts, marked) = starts.unzip();
399  Ok(ListMarks { prints, starts, marked, events })
400}

459/// Whether a label — a mark printing a letter or a number first, not a footnote symbol — opens what `arg` prints
460/// (`$^1$Univ A`, `\ua Univ A` with `\newcommand\ua{$^{1}$}`, a `\let` copy's, one in a box of its own); a `\footnotemark`
461/// or a number's suffix none. Unread, its tokens (`notes::leads_with_mark`), counted for `caller`.
462pub fn leads_with_label(arg: &Tokens, caller: &str) -> Result<bool> {
463  if !may_open_with_mark(arg) {
464    return Ok(false);
465  }
466  let Some(events) = mark_events(arg)? else {
467    count_unread(caller);
468    return Ok(leads_with_mark(arg));
469  };
470  Ok(opening_mark(&events).is_some_and(is_label_mark))
471}

484/// Whether what `arg` prints may open with a mark: its first ink past spaces and braces, read through parameterless
485/// macros' bodies (8 deep) and on past a body that prints nothing (`\newcommand\affilfont{}`), is no letter or digit,
486/// which TeX prints first. The readers of an opening mark ask it before their rehearsal (2103.04797's 1,161
487/// `\affiliation{\Amsterdam}`, each a macro of an institute's name, not rehearsed).
488fn may_open_with_mark(arg: &Tokens) -> bool {
489  walk(arg.unlist_ref(), MACRO_DEPTH, &mut |ink| match ink {
490    Ink::Space | Ink::Command(Prints::Nothing) => Continue(()),
491    Ink::Char(text, _) => Break(!text.chars().all(char::is_alphanumeric)),
492    _ => Break(true),
493  }) == Break(true)
494}

499  /// the labels, distinct, each with the queue positions of the names requesting it

501  /// where the mark opening the list's first entry stands in it, when its entries are cut
502  pub first_start: Option<usize>,
503}

505/// The labels, distinct, set by the marks that open the entries of a list `arg`, its entries as the observer cuts an API
506/// call (`events::mark_led_pieces`), each with the queue positions of the names requesting it, a label no name
507/// requests left out: the API Cut's decision and its copy rule (`frontmatter_api::cut_list`). Unread, none — the list
508/// stays whole for the observer to read —, counted for `caller`.
509pub fn requested_entry_labels(arg: &Tokens, caller: &str) -> Result<RequestedEntries> {
510  if !may_open_with_mark(arg) {
511    return Ok(RequestedEntries::default());
512  }
513  let Some(events) = mark_events(arg)? else {
514    count_unread(caller);
515    return Ok(RequestedEntries::default());
516  };
517  if !events.iter().any(own_mark) {
518    return Ok(RequestedEntries::default());
519  }
520  // (the queued names' marks read first: the pieces part at the marks they request, `events::names_requested`)
521  queued_author_marks()?;
522  let mut labels: Vec<(String, Vec<usize>)> = Vec::new();
523  let pieces = events::mark_led_pieces(&events);
524  let first_start = pieces
525    .first()
526    .and_then(|piece| piece.iter().find(|e| e.kind == EventKind::Mark))
527    .and_then(|mark| mark.span)
528    .map(|(start, _)| usize::from(start));
529  for piece in pieces {
530    // (the label `\lx@sup@setlabel@affiliation` sets: the first of the mark's printed labels, `set_label`)
531    let Some(label) = piece
532      .iter()
533      .find(|e| e.kind == EventKind::Mark)
534      .filter(|mark| own_mark(mark))
535      .and_then(mark_request_text)
536      .and_then(|text| {
537        clean_frontmatter_labels(&text, "affiliation")
538          .into_iter()
539          .next()
540      })
541    else {
542      continue;
543    };
544    let requesters = queued_label_requesters(&label)?;
545    if !requesters.is_empty() && labels.iter().all(|(seen, _)| *seen != label) {
546      labels.push((label, requesters));
547    }
548  }
549  Ok(RequestedEntries { labels, first_start })
550}
