1//! The observer's conventions (Layer B of the design note's §2; OXIDIZED_DESIGN_DIVERGENCES #837): the association
2//! conventions a binding family applies on top of the class-independent resolver (`resolver/`, Layer C). Each binding
3//! declares its own at load ([`declare_frontmatter_conventions`], user ruling 2026-10-28 "in each binding", superseding
4//! the design note's decision 2, "no policy keyval in the binding files"): the declaration is kept in the state under
5//! the binding's file name, keyed on (binding, hook) as the ledger records them at the call — the binding owning the
6//! `\author` meaning (`ledger::author_binding`, or the class's binding with a declaration, `ledger::policy_binding`)
7//! and the `\lx@add@authors*` hook it called. Each declaration reproduces a convention the binding's family already
8//! had and cites the OXIDIZED_DESIGN_DIVERGENCES entry or the witness it reproduces. The kernel's own blocks (`\author`
9//! still LaTeXML's, under `article` or any class whose bindings declare none) take [`KERNEL`] (Perl's `@authorsplits`
10//! reading, Base_Utility.pool.ltxml:676-700, OD #925); a binding owning `\author` that declares none takes the same
11//! reading, `UNDECLARED` (OD #948). Every family is read and emitted: the `written` flag that gated the S2b writer per
12//! family went with the writer (DEL4).

41/// What a class's title page prints for its authors, as the binding owning `\author` declares it (a binding's page is
42/// not run: the title-page reading stands article.cls's in for it, `latex_constructs::title_page_reading`).
43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44pub enum AuthorPage {
45  /// `\@author` as LaTeX's page prints it, with what a document appends to it (article.cls:247), when LaTeX's `\author`
46  /// is in force; under a binding's `\author`, unknown
47  Kernel,
48  /// `\@author` and what a document appends to it, though this binding's `\author` stores nothing there: authblk's own
49  /// `\@author` is its block (authblk.sty:167-171), llncs's `\@maketitle` prints `\@author` (llncs.cls:1035). The tail
50  /// appended after the value `\@author` held as the authors were queued is the block's (`frontmatter_stores`
51  /// `seen_author_tail`, OXIDIZED_DESIGN_DIVERGENCES #996: 2511.15315)
52  Block,
53  /// the binding's own author list, never `\@author`, on the title page of these document classes (their bindings' file
54  /// names), so what a document appends to `\@author` prints nowhere: amsart's, amsproc's and amsbook's `\authors`
55  /// (amsart.cls:460-476, :638, :815-817; amsproc.cls :613, :752-754; amsbook.cls :1356; ams_support.sty), elsarticle's
56  /// `\elsauthors` (elsarticle.cls:247-288, :805, :848; elsart_support_core.sty), acmart's `\addresses`
57  /// (acmart.cls:1619-1643, :2240, :2631). The title-page check reports a lost word only such a tail prints as one the
58  /// page annotates, an Info (OXIDIZED_DESIGN_DIVERGENCES #953, FX27), where every other binding's stays a Warning. Keyed
59  /// to the document class, not the binding alone, which other classes load whose page may print the tail (OmniBus's
60  /// autoload of ams_support.sty, old elsart.cls's elsart_support_core.sty)
61  Own(&'static [&'static str]),
62}

293/// The declarations `binding` made, in declaration order, if any. The key probed, not formatted nor interned: a
294/// rehearsal's e-mail binding asks it of each class binding (`ledger::email_meaning`; 1901.01540's 2,637 rehearsals).
295fn declared(binding: &str) -> Option<Rc<Vec<&'static Policy>>> {
296  let key = get_suffixed(DECLARATIONS, binding)?;
297  with_value_sym(key, |v| match v {
298    Some(Stored::Opaque(payload)) => Rc::clone(payload).downcast::<Vec<&'static Policy>>().ok(),
299    _ => None,
300  })
301}

303/// The State value holding the labels a `labels_print` binding's names print that no line of its block explains, each its
304/// spelling in the call (`annotations=`) and the mark it prints, `\u{2}` between them and `\u{1}` after each:
305/// [`declare_unexplained_name_labels`].
306const UNEXPLAINED_NAME_LABELS: &str = "lx_fm_unexplained_name_labels";

308/// Declare the labels a binding whose labels print ([`Policy::labels_print`]) found its names print and no line of its own
309/// explains, as (spelling in the call, the mark it prints) — authblk's `count_lost_labels`, read at `\maketitle`, before
310/// the flush: the writer shows the mark on the names requesting it (2606.00446's `§`). Not a label a line explains that the
311/// reading places elsewhere (`\affil[$\dag$]`).
312pub fn declare_unexplained_name_labels(labels: &[(String, String)]) {
313  let value: String = labels
314    .iter()
315    .map(|(spelling, mark)| s!("{}\u{2}{mark}\u{1}", spelling.trim()))
316    .collect();
317  assign_value(
318    UNEXPLAINED_NAME_LABELS,
319    Stored::String(pin(value)),
320    Some(Scope::Global),
321  );
322}

324/// The mark a `labels_print` binding declared its names' label spelled `request` prints, where no line explains it
325/// ([`declare_unexplained_name_labels`]). Spellings compare without braces and spaces: the reading's request of
326/// `annotations={…,\textsuperscript{\textdagger}}` lost the brace its group's closed (2606.17809).
327pub(in crate::base_utilities) fn unexplained_name_label(request: &str) -> Option<String> {
328  let bare = |spelling: &str| -> String {
329    spelling
330      .chars()
331      .filter(|c| !matches!(c, '{' | '}') && !c.is_whitespace())
332      .collect()
333  };
334  let request = bare(request);
335  lookup_string(UNEXPLAINED_NAME_LABELS)
336    .split('\u{1}')
337    .filter_map(|entry| entry.split_once('\u{2}'))
338    .find(|(spelling, _)| !request.is_empty() && bare(spelling) == request)
339    .map(|(_, mark)| mark.to_string())
340}

342/// The marks a `labels_print` binding declared its names' unexplained labels print ([`declare_unexplained_name_labels`]):
343/// `insert::settle_name_marks` shows them as a binding's labels, no mark a name typed (`insert::LABEL_MARK`).
344pub(in crate::base_utilities) fn unexplained_name_label_marks() -> Vec<String> {

346    .split('\u{1}')
347    .filter_map(|entry| entry.split_once('\u{2}'))

352/// The labels a `labels_print` binding declared its names print that no line of its own explains
353/// ([`declare_unexplained_name_labels`]), each as (spelling in the call, printed mark).
354pub(in crate::base_utilities) fn unexplained_name_label_spellings() -> Vec<(String, String)> {

362/// The declared labels printing `marks` are explained by a line after all (a `\date` line under the names that the
363/// label leads, OD #1071): no longer shown on the names as unexplained, the line's contact answering them.
364pub(in crate::base_utilities) fn explain_name_labels(marks: &[String]) {
365  let left: Vec<(String, String)> = unexplained_name_label_spellings()
366    .into_iter()
367    .filter(|(_, mark)| !marks.contains(mark))
368    .collect();
369  declare_unexplained_name_labels(&left);
370}

372/// Declare a binding's frontmatter conventions (the binding's load, before anything else it defines): kept in the
373/// state, globally, under `conventions.binding` for the hook `conventions.hook` (`*` any); a binding may declare one
374/// per hook (the first matching, in declaration order, applies), and a later declaration for the same hook replaces the
375/// earlier in its place (a binding loaded again declares the same). What `policy_for` reads.
376pub fn declare_frontmatter_conventions(conventions: &'static Policy) {
377  debug_assert!(
378    !conventions.binding.is_empty() && conventions.binding != "*",
379    "{}: conventions are declared for the declaring binding's file",
380    conventions.name
381  );
382  debug_assert!(
383    conventions.cite.contains("OD #") || conventions.cite.contains(".ltxml"),
384    "{}: a declaration cites the entry or the Perl binding it reproduces",
385    conventions.name
386  );
387  let mut all: Vec<&'static Policy> =
388    declared(conventions.binding).map_or_else(Vec::new, |d| d.to_vec());
389  match all.iter_mut().find(|p| p.hook == conventions.hook) {
390    Some(earlier) => *earlier = conventions,
391    None => all.push(conventions),
392  }
393  assign_value(
394    &declarations_key(conventions.binding),
395    Stored::Opaque(Rc::new(all)),
396    Some(Scope::Global),
397  );
398}

458/// The conventions among `declared` for a block recorded under `binding` (the `\author` owner) and `hook`: the first
459/// declaration whose binding and hook match (`*` any), else the kernel's for the kernel's own blocks, `UNDECLARED`
460/// for any other — `policy_for`'s rule, for a caller holding
461/// the declarations itself (the deleted `ab_audit fmshadow --re-resolve`).
462pub fn conventions_among<'a>(
463  declared: impl IntoIterator<Item = &'a &'static Policy>,
464  binding: &str,
465  hook: &str,
466) -> &'static Policy {
467  declared
468    .into_iter()
469    .copied()
470    .find(|p| p.binding == binding && (p.hook == "*" || p.hook == hook))
471    .unwrap_or_else(|| undeclared(binding))
472}
473
474/// The conventions for a block recorded under `binding` (the `\author` owner) and `hook`: the first that binding
475/// declared whose hook matches (`*` any), else the kernel's for the kernel's own blocks, `UNDECLARED` for any other.
476pub fn policy_for(binding: &str, hook: &str) -> &'static Policy {
477  declared(binding).map_or_else(
478    || undeclared(binding),
479    |d| conventions_among(d.iter(), binding, hook),
480  )
481}
