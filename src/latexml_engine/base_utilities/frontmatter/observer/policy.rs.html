1//! The observer's conventions (Layer B of the design note's §2; OXIDIZED_DESIGN_DIVERGENCES #837): the association
2//! conventions a binding family applies on top of the class-independent resolver (`resolver/`, Layer C). Each binding
3//! declares its own at load ([`declare_frontmatter_conventions`], user ruling 2026-10-28 "in each binding", superseding
4//! the design note's decision 2, "no policy keyval in the binding files"): the declaration is kept in the state under
5//! the binding's file name, keyed on (binding, hook) as the ledger records them at the call — the binding owning the
6//! `\author` meaning (`ledger::author_binding`, or the class's binding with a declaration, `ledger::policy_binding`)
7//! and the `\lx@add@authors*` hook it called. Each declaration reproduces a convention the binding's family already
8//! had and cites the OXIDIZED_DESIGN_DIVERGENCES entry or the witness it reproduces. The kernel's own blocks (`\author`
9//! still LaTeXML's, under `article` or any class whose bindings declare none) take [`KERNEL`] (Perl's `@authorsplits`
10//! reading, Base_Utility.pool.ltxml:676-700, OD #925); a binding owning `\author` that declares none takes the same
11//! reading, `UNDECLARED` (OD #948). Every family is read and emitted: the `written` flag that gated the S2b writer per
12//! family went with the writer (DEL4).

41/// What a class's title page prints for its authors, as the binding owning `\author` declares it (a binding's page is
42/// not run: the title-page reading stands article.cls's in for it, `latex_constructs::title_page_reading`).
43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44pub enum AuthorPage {
45  /// `\@author` as LaTeX's page prints it, with what a document appends to it (article.cls:247), when LaTeX's `\author`
46  /// is in force; under a binding's `\author`, unknown
47  Kernel,
48  /// `\@author` and what a document appends to it, though this binding's `\author` stores nothing there: authblk's own
49  /// `\@author` is its block (authblk.sty:167-171), llncs's `\@maketitle` prints `\@author` (llncs.cls:1035). The tail
50  /// appended after the value `\@author` held as the authors were queued is the block's (`frontmatter_stores`
51  /// `seen_author_tail`, OXIDIZED_DESIGN_DIVERGENCES #996: 2511.15315)
52  Block,
53  /// the binding's own author list, never `\@author`, on the title page of these document classes (their bindings' file
54  /// names), so what a document appends to `\@author` prints nowhere: amsart's, amsproc's and amsbook's `\authors`
55  /// (amsart.cls:460-476, :638, :815-817; amsproc.cls :613, :752-754; amsbook.cls :1356; ams_support.sty), elsarticle's
56  /// `\elsauthors` (elsarticle.cls:247-288, :805, :848; elsart_support_core.sty), acmart's `\addresses`
57  /// (acmart.cls:1619-1643, :2240, :2631). The title-page check reports a lost word only such a tail prints as one the
58  /// page annotates, an Info (OXIDIZED_DESIGN_DIVERGENCES #953, FX27), where every other binding's stays a Warning. Keyed
59  /// to the document class, not the binding alone, which other classes load whose page may print the tail (OmniBus's
60  /// autoload of ams_support.sty, old elsart.cls's elsart_support_core.sty)
61  Own(&'static [&'static str]),
62}

94/// One family's association conventions, as its binding declares them ([`declare_frontmatter_conventions`]).
95#[derive(Debug, Clone, Copy, PartialEq, Eq)]
96pub struct Policy {
97  /// the declaration's name (reported with every decision it made)
98  pub name:              &'static str,
99  /// the declaring binding's file name, as the BINDINGS registry gives it (`<name>.<extension>`), and the hook
100  /// (`\lx@add@authors@append`) it applies to: `*` any
101  pub binding:           &'static str,
102  pub hook:              &'static str,
103  /// an `\inst` mark (the convention's own, the rehearsal's `\lx@fm@obs@mark`) after a comma is the name's before the
104  /// comma, glued or not (llncs.cls:894-896 and aa.cls:759-760 `\def\inst#1{\unskip$^{#1}$}`; OD #807, 2401.16957);
105  /// any other mark after a comma is the name's before only when glued to the comma
106  pub inst_after_comma:  bool,
107  /// `\and` is inline text joining the names of one row, not a column (llncs.cls:1013-1018 `\def\and{\unskip,…}`
108  /// in `\@maketitle`; OD #792: `Ann Able \and Bob Baker\\ University of Somewhere` is both names')
109  pub and_inline:        bool,
110  /// the `\\` lines of the class's `\author` paragraph are read as the binding's own split read them before the block
111  /// went to the reading whole (llncs's `\lx@llncs@author` until Z18e, OXIDIZED_DESIGN_DIVERGENCES #984: its
112  /// `reads_as_affiliation` per line): under a row's affiliation line, a line reading as no institution over one reading
113  /// as one starts a new row of names, and a line whose name carries an ORCID is names. Off for the kernel, whose `\\`
114  /// lines keep their affiliation reading (an article's "Einstein Drive" over an address line, hep-th0212274)
115  pub author_lines:      bool,
116  /// every `\\` line of the class's `\author` paragraph is names, as its title page sets them (webofc.cls:497-552:
117  /// `\@author` a paragraph whose lines are names, the institutes apart: the template's `\\The SND Collaboration` a
118  /// creator, 1609.01040), the binding's spelling split at `\\` until P2-1 (OD #993). Off for the kernel, whose
119  /// `\\` lines under names keep their affiliation reading
120  pub break_parts_names: bool,
121  /// a `labelseq=<role>` entry is labelled by its place among the entries of that role (llncs's and aa's
122  /// `\institute{A \and B}`, numbered 1, 2 as the class prints them: OD #791, #159)
123  pub label_sequence:    bool,
124  pub email_owner:       EmailOwner,
125  /// each call is an author group of its own (an e-mail nothing else places is the group's only author's: iopart's
126  /// `\author`, OD #808)
127  pub call_is_group:     bool,
128  pub title_marks:       TitleMarks,
129  /// authblk's notes (authblk.sty:114-118, 148-151): an unlabelled `\author` shows `\AB@thenote`, the `affil` counter
130  /// stepped at the first `\author` after an `\affil`, and an unlabelled `\affil` is labelled by it; the marks show
131  /// when a label is given or the counter passes `\c@Maxaffil` (`state_counter`), else the affiliations print under
132  /// the run of authors (block mode)
133  pub thenote:           bool,
134  /// a counter the policy reads at each block's call (the ledger's `counter`)
135  pub state_counter:     Option<&'static str>,
136  /// `\and` inside an annotation prints the word "and" (amsart.cls:599 `\def\and{\unskip{ }\@@and{ }\ignorespaces}`):
137  /// the entry's text holds it
138  pub and_word:          bool,
139  /// the title's notes print no mark (IEEEtran.cls:4756 `\def\thefootnote{}` in `\maketitle`, "the \thanks{} mark type
140  /// is empty"): a note stands beside a name only by its place, so its inner contacts are a name's only by spelling
141  /// (main-session decision 2 of 2026-10-22, user ruling 2026-10-20 #4)
142  pub unmarked_notes:    bool,
143  /// the class's `\email` prints its address where it is called (llncs.cls:897 `\def\email#1{{\tt#1}}`, aa.cls:824
144  /// `e-mail: {\aa@emailfont #1}`, svjour3.cls:303, svmult.cls:2304, svjour.cls:256): one inside a document's note
145  /// (`\offprints{O. Kochukhov, \email{…}}`, a `\thanks`) prints in the note, after the name the note gives, which
146  /// designates its owner — while the `\email` meaning is one of the class's bindings' (`ledger::email_meaning`, read at
147  /// the call; `Resolver::answer_document_notes`). Not amsart's (amsart.cls:538 `\def\email` stores the address for
148  /// the author's address block) nor REVTeX's (the author's footnote): there the address is the call's current
149  /// author's, as the binding routes it
150  pub email_in_place:    bool,
151  /// a call's `label=` is a key, never printed: the class prints the entry's place in its list instead
152  /// (elsarticle.cls:656-662 `\theaffn`, elsart.cls v2.18 :1135-1164 `\theaddress`), so no footnote's mark answers it,
153  /// a symbol's neither (FIDELITY_CHARTER §4 R1: an author whose keys request affiliations takes none by another route;
154  /// 1108.0201's `\address[*]{current address}` under `M.~Pesce-Rollins\footnotemark[1]`) — an author's own keys
155  /// (`annotations=` of `\author[a,b]`) and a mark typed by hand do, the places keys in order spell (0907.5014's
156  /// `Denes Molnar$^{a,b}$`)
157  pub keyed_labels:      bool,
158  /// the commands that part a block's lines or columns whatever they print here, as the class's title page does, each declared by
159  /// the binding owning the `\author` they stand in: OmniBus's no-op `\alignauthor` a column (inst_support_sty.rs; what sig-alternate.cls,
160  /// sigchi.cls print, whose `\alignauthor` read raw is the tabular it makes), ijcai's `\affiliations` and `\emails` a section
161  /// (ijcai19.sty:124-130). A rehearsal notes their calls beside the block's events (`Outcome::Events` `breaks`), which they do
162  /// not part; no reading reads them yet
163  pub block_breaks:      &'static [(&'static str, BreakKind)],
164  /// what the class's title page prints that the stand-in for article.cls's page (sect05.rs `title_page_print_tokens`)
165  /// cannot know: [`TitlePageDecl`]
166  pub title_page:        TitlePageDecl,
167  /// what the declaration reproduces
168  pub cite:              &'static str,
169}

249/// The conventions for a block of `binding` when it declared none: the kernel's for the kernel's own blocks, else
250/// `UNDECLARED`.
251fn undeclared(binding: &str) -> &'static Policy {

253    &KERNEL

255    &UNDECLARED

262/// The state key a binding's declarations are kept under ([`DECLARATIONS`]).

265/// The declarations `binding` made, in declaration order, if any. The key probed, not formatted nor interned: a
266/// rehearsal's e-mail binding asks it of each class binding (`ledger::email_meaning`; 1901.01540's 2,637 rehearsals).
267fn declared(binding: &str) -> Option<Rc<Vec<&'static Policy>>> {
268  let key = get_suffixed(DECLARATIONS, binding)?;
269  with_value_sym(key, |v| match v {
270    Some(Stored::Opaque(payload)) => Rc::clone(payload).downcast::<Vec<&'static Policy>>().ok(),
271    _ => None,
272  })
273}

275/// Declare a binding's frontmatter conventions (the binding's load, before anything else it defines): kept in the
276/// state, globally, under `conventions.binding` for the hook `conventions.hook` (`*` any); a binding may declare one
277/// per hook (the first matching, in declaration order, applies), and a later declaration for the same hook replaces the
278/// earlier in its place (a binding loaded again declares the same). What `policy_for` reads.
279pub fn declare_frontmatter_conventions(conventions: &'static Policy) {
280  debug_assert!(
281    !conventions.binding.is_empty() && conventions.binding != "*",
282    "{}: conventions are declared for the declaring binding's file",
283    conventions.name
284  );
285  debug_assert!(
286    conventions.cite.contains("OD #") || conventions.cite.contains(".ltxml"),
287    "{}: a declaration cites the entry or the Perl binding it reproduces",
288    conventions.name
289  );
290  let mut all: Vec<&'static Policy> =
291    declared(conventions.binding).map_or_else(Vec::new, |d| d.to_vec());
292  match all.iter_mut().find(|p| p.hook == conventions.hook) {
293    Some(earlier) => *earlier = conventions,
294    None => all.push(conventions),
295  }
296  assign_value(
297    &declarations_key(conventions.binding),
298    Stored::Opaque(Rc::new(all)),
299    Some(Scope::Global),
300  );
301}

311/// Whether `binding` declared that the `\maketitle` of `class` (its binding's file name) runs `\@maketitle`
312/// ([`TitlePageDecl::maketitle_runs`], any of its hooks), `None` when it declared nothing for it.
313pub fn declared_maketitle_runs(binding: &str, class: &str) -> Option<MaketitleRuns> {
314  declared(binding).and_then(|d| {
315    d.iter().find_map(|p| {
316      p.title_page
317        .maketitle_runs
318        .iter()
319        .find(|(name, ..)| *name == class)
320        .map(|(_, runs, _)| *runs)
321    })
322  })
323}

325/// What `binding` declared its class's title page prints for the authors ([`TitlePageDecl::author`]: the first of its
326/// hooks' declarations that says more than the kernel's), `None` when it declared nothing.
327pub fn declared_author_page(binding: &str) -> Option<AuthorPage> {
328  declared(binding).and_then(|d| {
329    d.iter()
330      .map(|p| p.title_page.author)
331      .find(|page| *page != AuthorPage::Kernel)
332  })

335/// The conventions among `declared` for a block recorded under `binding` (the `\author` owner) and `hook`: the first
336/// declaration whose binding and hook match (`*` any), else the kernel's for the kernel's own blocks, `UNDECLARED`
337/// for any other — `policy_for`'s rule, for a caller holding
338/// the declarations itself (the deleted `ab_audit fmshadow --re-resolve`).
339pub fn conventions_among<'a>(
340  declared: impl IntoIterator<Item = &'a &'static Policy>,
341  binding: &str,
342  hook: &str,
343) -> &'static Policy {
344  declared
345    .into_iter()
346    .copied()
347    .find(|p| p.binding == binding && (p.hook == "*" || p.hook == hook))
348    .unwrap_or_else(|| undeclared(binding))
349}
350
351/// The conventions for a block recorded under `binding` (the `\author` owner) and `hook`: the first that binding
352/// declared whose hook matches (`*` any), else the kernel's for the kernel's own blocks, `UNDECLARED` for any other.
353pub fn policy_for(binding: &str, hook: &str) -> &'static Policy {
354  declared(binding).map_or_else(
355    || undeclared(binding),
356    |d| conventions_among(d.iter(), binding, hook),
357  )
358}

360#[cfg(test)]
361mod tests {
362  use super::*;
363
364  /// A synthetic binding's declarations: one for every hook, one for a hook of its own (declared first, so it wins
365  /// for that hook, as the first matching declaration does).
366  const ANY_HOOK: Policy = Policy {
367    name: "synthetic",
368    binding: "synthetic_binding",
369    and_inline: true,
370    cite: "OD #837 (a synthetic binding: the declaration's lookup)",
371    ..KERNEL
372  };
373  const APPEND_HOOK: Policy = Policy {
374    name: "synthetic_append",
375    hook: "\\lx@add@authors@append",
376    and_word: true,
377    ..ANY_HOOK
378  };
379
380  #[test]
381  fn a_declaration_is_read_and_an_undeclared_family_takes_the_kernel() {
382    assert_eq!(
383      policy_for("synthetic_binding", "\\lx@add@author@split").name,
384      "undeclared"
385    );
386    assert!(!has_entry("synthetic_binding"));
387    declare_frontmatter_conventions(&APPEND_HOOK);
388    declare_frontmatter_conventions(&ANY_HOOK);
389    assert!(has_entry("synthetic_binding"));
390    assert_eq!(
391      policy_for("synthetic_binding", "\\lx@add@authors@append").name,
392      "synthetic_append"
393    );
394    assert_eq!(
395      policy_for("synthetic_binding", "\\lx@add@author@split").name,
396      "synthetic"
397    );
398    assert_eq!(
399      policy_for("kernel", "\\lx@add@authors@adaptive").name,
400      "kernel"
401    );
402    assert_eq!(
403      policy_for("undeclared_binding", "\\lx@add@authors@append").name,
404      "undeclared"
405    );
406    // (a declaration again for a hook replaces the earlier in its place: a binding loaded again)
407    declare_frontmatter_conventions(&APPEND_HOOK);
408    assert_eq!(
409      policy_for("synthetic_binding", "\\lx@add@authors@append").name,
410      "synthetic_append"
411    );
412  }
