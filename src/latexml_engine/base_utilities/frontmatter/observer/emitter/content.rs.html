1//! What a span of a field holds (the digested refactor's Stage 3, `docs/archive/FRONTMATTER_DIGESTED_REFACTOR_2026-11-07.md`): the tokens a
2//! piece, a line or an entry the reading cut is made from, read from the digested list and its spans. A box covers what
3//! it prints and no more, so a span stops short of what prints nothing around it, and that is said by the digested list
4//! too, not by walking tokens back from the span: the formula it starts or ends in, whole (`Field::formulas`, whatever
5//! opens it: `$\rm ^{3}$`, `${}^{*}$`, `\( ^{1} \)`, `\protect\(`, 2105.07958, hep-ph0110378, 2307.12984), and a mark's own
6//! print inside a wrapper whose cover its first box took ([`Field::own_print`], `\textbf{$^2$Univ B}`, 2605.25195). What
7//! is left of the tokens is brace balance, a property of the content itself, as the declarations' spans clamp at it

38/// A field's groups as its braces open and close them, read once as far as asked, not again from its start for each
39/// piece: before token `i`, the innermost group still open — `inner[i]`, its opening brace's index plus one, 0 for none —,
40/// a group's own enclosing one the innermost open before its brace (`inner[brace]`). A closing brace no group is open at
41/// closes none (1507.02068's 418 names, each piece's open groups read from the block's start: about 0.7 % of the run).
42#[derive(Debug, Default)]
43pub(in crate::base_utilities::frontmatter::observer) struct Groups {
44  inner: std::cell::RefCell<Vec<usize>>,
45}

47impl Groups {
48  /// The opening braces in `content[from..to]` of the groups still open at `to`, in order: the innermost open group's
49  /// enclosing ones down to `from`, each read once for the field.
50  fn open(&self, content: &[Token], from: usize, to: usize) -> Vec<usize> {
51    let to = to.min(content.len());
52    let mut inner = self.inner.borrow_mut();
53    if inner.is_empty() {
54      inner.push(0);
55    }
56    while inner.len() <= to {
57      let i = inner.len() - 1;
58      let here = inner[i];
59      let next = match content[i].group_step() {
60        1 => i + 1,
61        -1 => here.checked_sub(1).map_or(0, |brace| inner[brace]),
62        _ => here,
63      };
64      inner.push(next);
65    }
66    let mut open = Vec::new();
67    let mut at = inner[to];
68    while let Some(brace) = at.checked_sub(1).filter(|&brace| brace >= from) {
69      open.push(brace);
70      at = inner[brace];
71    }
72    open.reverse();
73    open
74  }
75}

77/// Whether a digested list's events hold an alignment's cell (`events::Walker`'s `Layout{role="cell"}`): where none is, no tab of
78/// it is an alignment's — a verbatim argument read it as a character (`\url{http://y.org/?c=1&d=2}`), or it was a stray one in
79/// the rehearsal as everywhere. A note's events too (`Event::inner`, `Event::body`): a tabular in a `\thanks`.
80pub(in crate::base_utilities::frontmatter::observer) fn aligned(events: &[Event]) -> bool {
81  events.iter().any(|e| {
82    (e.kind == EventKind::Layout && e.role.as_deref() == Some("cell"))
83      || aligned(&e.inner)
84      || aligned(&e.body)
85  })
86}

88/// Where a call in `tokens` reads past them: a macro at their top level whose parameters are each undelimited (TeX's
89/// `#1#2`, `\newcommand`'s `[2]`; a verbatim group, `\href`'s URL) or an optional one (`[…]`), by its meaning now, an
90/// undelimited argument of which the tokens don't hold — the next read at a closing brace or past the last token, there
91/// the wrapper's own tokens (`\lx@author@withinst`'s `\let\inst\lx@saved@inst…`, `frontmatter_fields.rs`) or the item's
92/// brace. Whether it is cut between its arguments (`\def\authord#1#2{…}`'s `\authord{Anh Thu Nguyen}`, 2209.12410) or
93/// the tokens end at it, none of its arguments theirs (a mark's span covering the call alone: aiaa-tc's
94/// `\thanksibid{2}`, 1505.07844, 1411.3642, 1602.00849; a preamble's `\footrecall{uva}`, 2503.21719, 2407.13730,
95/// 2606.19583; a URL cut out of `\href{…}{…}`, 2603.11001). A parameter of any other kind (a delimited one) is not
96/// judged, nor a command inside a group of theirs (an argument a primitive may never expand: `\detokenize{\begin}`). A
97/// call's arguments are its [`tail_call`]'s where `tail` (a printed call's); a declaration's are its own macro's.
98pub(super) fn reads_past(tokens: &[Token], tail: bool) -> Option<usize> {
99  let mut depth = 0i32;
100  tokens.iter().enumerate().position(|(i, t)| {
101    depth += t.group_step();
102    depth == 0 && arguments_end(tokens, i, tail) == Some(None)
103  })
104}

106/// Whether `tokens` read a meaning of `definees`: its control sequence, or a counter's name as an argument (`\c@uni`'s
107/// `{uni}`: `\footnoterecall{uni}`, `\value{uni}`), in them or one level into a document's macro they call.
108pub(super) fn reads_definition(tokens: &[Token], definees: &[SymStr]) -> bool {
109  let counters: Vec<String> = definees
110    .iter()
111    .filter_map(|&key| with(key, |cs| cs.strip_prefix("\\c@").map(str::to_string)))
112    .collect();
113  let reads = |tokens: &[Token]| {
114    tokens.iter().enumerate().any(|(i, t)| {
115      if t.get_catcode().is_active_or_cs() {
116        return definees.contains(&meaning_key(t));
117      }
118      if t.get_catcode() != Catcode::BEGIN || counters.is_empty() {
119        return false;
120      }
121      let name: Option<String> = tokens[i + 1..]
122        .iter()
123        .take_while(|u| u.get_catcode() != Catcode::END)
124        .map(|u| {
125          matches!(u.get_catcode(), Catcode::LETTER | Catcode::OTHER)
126            .then(|| u.with_str(str::to_string))
127        })
128        .collect();
129      name.is_some_and(|name| counters.contains(&name))
130    })
131  };
132  reads(tokens)
133    || tokens.iter().any(|t| match lookup_meaning(t) {
134      Some(Stored::Expandable(d))
135        if d.made_in_own_file || d.get_origin() == DefinitionOrigin::Document =>
136      {
137        matches!(d.get_expansion(), Some(ExpansionBody::Tokens(body)) if reads(body.unlist_ref()))
138      },
139      _ => false,
140    })
141}

143/// Where the call at `tokens[at]` ends, past the arguments its meaning now reads ([`reads_past`]'s parameters):
144/// `Some(None)` where an undelimited one is not in the tokens, `None` for no such call (a primitive, an unexpandable
145/// command, no parameters, a parameter of another kind).
146pub(super) fn arguments_end(tokens: &[Token], at: usize, tail: bool) -> Option<Option<usize>> {
147  let (plain, verbatim, hyper, optional) = (
148    pin!("Plain"),
149    pin!("Semiverbatim"),
150    pin!("HyperVerbatim"),
151    pin!("Optional"),
152  );
153  let meaning = if tail {
154    tail_call(&tokens[at])?
155  } else {
156    let Some(Stored::Expandable(meaning)) = lookup_meaning(&tokens[at]) else {
157      return None;
158    };
159    meaning
160  };
161  let parameters = meaning.get_parameters()?.get_parameters();
162  if parameters.is_empty()
163    || parameters
164      .iter()
165      .any(|p| !(p.name == plain || p.name == verbatim || p.name == hyper || p.name == optional))
166  {
167    return None;
168  }
169  let mut j = at + 1;
170  for parameter in parameters {
171    while tokens
172      .get(j)
173      .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
174    {
175      j += 1;
176    }
177    if parameter.name == optional {
178      // (an option present is read to its `]` at the top level; one absent reads nothing: the next token is peeked)
179      if tokens
180        .get(j)
181        .is_some_and(|t| t.get_catcode() == Catcode::OTHER && t.with_str(|s| s == "["))
182      {
183        let mut inner = 0i32;
184        loop {
185          let Some(t) = tokens.get(j) else {
186            return Some(None);
187          };
188          inner += t.group_step();
189          j += 1;
190          if inner == 0 && t.get_catcode() == Catcode::OTHER && t.with_str(|s| s == "]") {
191            break;
192          }
193        }
194      }
195      continue;
196    }
197    match tokens.get(j).map(Token::group_step) {
198      None | Some(-1) => return Some(None),
199      Some(1) => {
200        let mut inner = 0i32;
201        loop {
202          let Some(t) = tokens.get(j) else {
203            return Some(None);
204          };
205          inner += t.group_step();
206          j += 1;
207          if inner == 0 {
208            break;
209          }
210        }
211      },
212      Some(_) => j += 1,
213    }
214  }
215  Some(Some(j))
216}

218/// The macro whose parameters a call of `cs` reads, by its meaning now: `cs`'s own, or, for a parameterless macro whose
219/// expansion ends in a command, that command's — a robust command's `\protect \cs␣` (icrc2009.cls's
220/// `\DeclareRobustCommand*{\IEEEauthorrefmark}[1]`, 0912.0896), IEEEtran's `\def\authorrefmark{…\IEEEauthorrefmark}`
221/// (2609.25589): cut from its `{1}`, the mark read the wrapper's `\let` as its `\ifcase` number.
222fn tail_call(cs: &Token) -> Option<Rc<Expandable>> {
223  let mut called = *cs;
224  for _ in 0..8 {
225    let Some(Stored::Expandable(meaning)) = lookup_meaning(&called) else {
226      return None;
227    };
228    let parameterless = meaning
229      .get_parameters()
230      .is_none_or(|parameters| parameters.get_parameters().is_empty());
231    match meaning.get_expansion() {
232      Some(ExpansionBody::Tokens(body)) if parameterless => {
233        called = *body.unlist_ref().last()?;
234        if called.get_catcode() != Catcode::CS {
235          return None;
236        }
237      },
238      _ => return Some(meaning),
239    }
240  }
241  None
242}

252/// Where `tokens` hold a conditional cut apart, by their meanings now: an `\else`, `\or` or `\fi` whose `\if…` is not
253/// theirs, or an `\if…` whose `\fi` is not theirs — digested apart, the one errs ("not in a conditional"), the other
254/// stays open to the end of the document (`\iffull Bob Baker\fnm{2}\fi`'s name run `\iffull Bob Baker`, review revX4 2).
255/// A switch's name not executed where it stands is one as well (`\newif\iffull`, `\let\ifa\iftrue`, what `\ifx`
256/// compares), which no meaning tells apart from a test: such a run is refused too, its item written as its text, counted.
257fn conditional_cut(tokens: &[Token]) -> Option<usize> {
258  let mut open = Vec::new();
259  for (i, t) in tokens.iter().enumerate() {
260    match conditional_of(t) {
261      Some(ConditionalType::If) => open.push(i),
262      Some(ConditionalType::Else | ConditionalType::Or | ConditionalType::Fi)
263        if open.is_empty() =>
264      {
265        return Some(i);
266      },
267      Some(ConditionalType::Fi) => {
268        open.pop();
269      },
270      _ => {},
271    }
272  }
273  open.first().copied()
274}

276/// `end` grown over the `\fi`s right after `content[start..end]` that close a conditional it opens, as [`Field::balanced`]
277/// grows over closing braces: an entry printed by its last branch ends before the `\fi` that prints nothing (`\ifx\zz\end
278/// Q\else X\fi`, `perfect_kernel_batchz4::an_environment_by_its_frames`).
279fn past_closing_fis(content: &[Token], start: usize, end: usize) -> usize {
280  let mut open = 0usize;
281  for t in &content[start..end] {
282    match conditional_of(t) {
283      Some(ConditionalType::If) => open += 1,
284      Some(ConditionalType::Fi) => open = open.saturating_sub(1),
285      _ => {},
286    }
287  }
288  let mut end = end;
289  let mut next = end;
290  while open > 0 {
291    while content
292      .get(next)
293      .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
294    {
295      next += 1;
296    }
297    if !content
298      .get(next)
299      .is_some_and(|t| conditional_of(t) == Some(ConditionalType::Fi))
300    {
301      break;
302    }
303    next += 1;
304    end = next;
305    open -= 1;
306  }
307  end
308}

321  /// The formula holding `tokens[index]`: its indices, opening shift to past its closing shift.
322  fn formula_at(&self, index: usize) -> Option<(usize, usize)> {
323    let pos = u16::try_from(self.base + index).ok()?;
324    let at = self.formulas.partition_point(|formula| formula.end <= pos);
325    let formula = self.formulas.get(at)?;
326    (formula.start <= pos).then_some((
327      usize::from(formula.start).checked_sub(self.base)?,
328      usize::from(formula.end).checked_sub(self.base)?,
329    ))
330  }
331
332  /// `index`, or where the formula holding it starts.
333  pub(super) fn formula_start(&self, index: usize) -> usize {
334    self.formula_at(index).map_or(index, |(start, _)| start)
335  }
336
337  /// `index` (the end of a span), or past the formula holding the token before it.
338  pub(super) fn formula_end(&self, index: usize) -> usize {
339    index
340      .checked_sub(1)
341      .and_then(|last| self.formula_at(last))
342      .map_or(index, |(_, end)| end.max(index))
343  }

345  /// A Mark's own print as indices, where its span is wider or narrower: the whole formula it is in (the first box in a
346  /// wrapper takes the cover of the call around it: `\textbf{$^2$Univ B}`'s span is the wrapper's), or the
347  /// `\textsuperscript{…}` call to its closing brace. `None` for a mark of neither.
348  pub(super) fn own_print(&self, mark: &Event) -> Option<(usize, usize)> {
349    let index = |p: u16| usize::from(p).checked_sub(self.base);
350    if let Some(formula) = mark.formula.filter(|f| f.base.is_some()) {
351      let (start, end) = formula.span?;
352      return Some((index(start)?, index(end)?));
353    }
354    let call = index(mark.own_at?)?;
355    let mut depth = 0;
356    for (k, t) in self.tokens.iter().enumerate().skip(call + 1) {
357      if depth == 0 && t.group_step() != 1 {
358        return None;
359      }
360      depth += t.group_step();
361      if depth == 0 {
362        return Some((call, k + 1));
363      }
364    }
365    None
366  }

368  /// The content of the span `start..end` (indices), past `floor` (where the piece before ended): the balanced run of
369  /// tokens holding it, and where that ends — with the switches that open the group it sits in, when it sits inside one
370  /// (`{\bf $^1$Univ A, $^2$Univ B}`: each piece keeps the `\bf`, unless `first` is 0: a piece carrying the font it
371  /// starts in, `events::entry_font`) —, or `None` when no balanced run holds it.
372  ///
373  /// The run takes each formula the span starts or ends in whole, a mark's own font group from its opening brace
374  /// (`{\bf $^1$} Univ A`) and the closing braces right after a group it opens.
375  pub(super) fn content_of(
376    &self,
377    range: (usize, usize),
378    floor: usize,
379    first: usize,
380  ) -> Option<(Tokens, usize)> {
381    self.content_cut(range, floor, first, &[])
382  }
383
384  /// [`Self::content_of`] without the runs `cuts` (indices, each balanced: an entry's address runs, `address_cut`), `None`
385  /// where one of them is partly inside the content and partly outside it.
386  pub(super) fn content_cut(
387    &self,
388    (start, end): (usize, usize),
389    floor: usize,
390    first: usize,
391    cuts: &[(usize, usize)],
392  ) -> Option<(Tokens, usize)> {
393    let content = self.tokens;
394    if start > end || end > content.len() || start < floor {
395      return None;
396    }
397    let (mut start, mut end) = (start, end);
398    // (a formula the piece before ended in stays as it was, its other end past `floor`: the mark's closing shift is the
399    // mark's, `$^1 Univ$`)
400    if let Some((s, e)) = self.formula_at(start).filter(|&(s, _)| s >= floor) {
401      start = s;
402      end = end.max(e);
403    }
404    if let Some((s, e)) = end
405      .checked_sub(1)
406      .and_then(|last| self.formula_at(last))
407      .filter(|&(s, _)| s >= floor)
408    {
409      start = start.min(s);
410      end = end.max(e);
411    }
412    end = end.min(content.len());
413    let end = self.past_held_undefined(end);
414    let (start, end) = self.balanced(start, end, floor)?;
415    let end = past_closing_fis(content, start, end);
416    // (the group the pieces sit in: its opening brace, and the switches before the first piece in it)
417    let mut tokens = Vec::new();
418    if let Some(&group) = self.open_groups(0, start).last()
419      && group < first
420    {
421      let switches = &content[group + 1..first.min(start)];
422      if switches
423        .iter()
424        .all(|t| matches!(t.get_catcode(), Catcode::CS | Catcode::SPACE))
425      {
426        tokens.extend_from_slice(switches);
427      }
428    }
429    if cuts
430      .iter()
431      .any(|&(s, t)| s < end && t > start && (s < start || t > end))
432    {
433      return None;
434    }
435    if cuts.is_empty() && self.held.is_empty() {
436      tokens.extend_from_slice(&content[start..end]);
437    } else {
438      // (a script character its rehearsal read outside math, in no formula, an error held there, is the character that rehearsal printed:
439      // `\textbf{Ningkai Mo} ^1` is a text-mode `^`, which TeX errs at (Missing $) and the digestion's recovery prints;
440      // digested apart as a script it erred again, once per piece (2204.01080). `\lx@fm@script@char` prints it and
441      // counts the recovery, a Warning. An `_` a verbatim argument reads as a character holds no error: `\url{a_b}` stays)
442      let cut = |i: &usize| cuts.iter().any(|&(s, t)| (s..t).contains(i));
443      for i in (start..end).filter(|i| !cut(i)) {
444        let t = content[i];
445        if matches!(t.get_catcode(), Catcode::SUPER | Catcode::SUB)
446          && self.formula_at(i).is_none()
447          && self.held_at(i)
448        {
449          tokens.push(T_CS!("\\lx@fm@script@char"));
450          tokens.push(T_BEGIN!());
451          tokens.push(T_OTHER!(if t.get_catcode() == Catcode::SUPER {
452            "^"
453          } else {
454            "_"
455          }));
456          tokens.push(T_END!());
457        } else {
458          tokens.push(t);
459        }
460      }
461    }
462    // (never a run holding an alignment's tab or alignment-only command outside an environment or a formula it holds whole, as
463    // the digestion's frames say them (a macro pair's `\bx a&b\\ c&d\ex` is one): digested apart, it is a stray one, an Error
464    // — its caller writes the item's text, counted (the rule of `plan::cell_tokens`; 1304.3138, 1504.01048, 2607.22455). Only
465    // where the digestion holds an alignment ([`aligned`]): a `\url{…?c=1&d=2}` in a block of none keeps its link (review
466    // revNE1 1))
467    // (nor a run a macro of which reads past its end, by its meaning, which the run is digested under: `\def\authord#1#2{…}`'s
468    // `\authord{Anh Thu Nguyen}`, its second argument the affiliation's piece, read the item's closing brace (an undefined
469    // `\@undefined` and a group ended in the wrong mode, 2209.12410) — its caller writes the item's text, counted)
470    // (nor a run holding a conditional cut apart, an `\else` whose `\if` is not the run's: `\ifFull and \else \and \fi
471    // Joseph A. Simons`, whose name's run took the `\else \and \fi` the rehearsal skipped, an Error and a tabular's group
472    // ended in the wrong mode (1409.0597) — its caller writes the item's text, counted)
473    if let Some(at) = conditional_cut(&tokens) {
474      DebugFeature!(
475        "frontmatter",
476        "FRONT writer: the run {start}..{end} holds {:?}, a conditional cut apart",
477        tokens[at]
478      );
479      return None;
480    }
481    if let Some(call) = reads_past(&tokens, true) {
482      DebugFeature!(
483        "frontmatter",
484        "FRONT writer: the run {start}..{end} holds {:?}, whose arguments run past it",
485        tokens[call]
486      );
487      return None;
488    }
489    // (what it holds whole read only for a run holding such a token at all)
490    let mut held: Option<Vec<(usize, usize)>> = None;
491    if let Some(stray) = (start..end).find(|&i| {
492      self.aligned
493        && reads_as_alignment(&content[i])
494        && !cuts.iter().any(|&(s, t)| (s..t).contains(&i))
495        && !held
496          .get_or_insert_with(|| self.held_whole(start, end))
497          .iter()
498          .any(|&(s, t)| (s..t).contains(&i))
499    }) {
500      DebugFeature!(
501        "frontmatter",
502        "FRONT writer: the run {start}..{end} holds an alignment's {:?} at {stray}",
503        content[stray]
504      );
505      return None;
506    }
507    Some((Tokens::new(tokens), end))
508  }

510  /// `end` moved past the undefined commands right after it, each one its rehearsal held an Error at (one past a space or
511  /// a mark is a name's by [`super::emit`]'s `block_names`, else the flush's net raises it, `reading::HeldUndefined`): the
512  /// command prints nothing (TeX's "Undefined control sequence"), so no event covers it and the piece before it would
513  /// stop short of it, its Error — the rehearsal's, held and discarded with its stub — never raised again. Taken into the
514  /// piece, its digestion raises it there once, its `<ltx:ERROR>` in the piece, as Perl's whole-block creator raises it

559  /// The environments (their frames' steps, paired) and the formulas the run `start..end` holds whole: their indices.
560  fn held_whole(&self, start: usize, end: usize) -> Vec<(usize, usize)> {
561    let at = |p: u16| usize::from(p).checked_sub(self.base);
562    let mut held = Vec::new();
563    let mut open = Vec::new();
564    for step in self.environments {
565      let (Some(from), Some(to)) = (at(step.start), at(step.end)) else {
566        continue;
567      };
568      if from < start || from >= end {
569        continue;
570      }
571      if step.opens {
572        open.push(from);
573      } else if let Some(opened) = open.pop()
574        && to <= end
575      {
576        held.push((opened, to));
577      }
578    }
579    let formulas = self
580      .formulas
581      .iter()
582      .filter_map(|formula| Some((at(formula.start)?, at(formula.end)?)));
583    held.extend(formulas.filter(|&(s, t)| start <= s && t <= end));
584    held
585  }

587  /// Whether the `nth` group around the formula starting at `from`, from the innermost, closing at `close`, starts at its brace with that formula: it prints
588  /// the formula alone (`events::Formula::alone_in`: `{\bf $^{1}$} Univ X`, `{$^{1}$\,} Univ X`, a mark's own group), or the formula opens it where a line
589  /// starts (`Formula::opens_line_in`: `\\ \normalsize{$^{1}$ Univ X,} Moscow`, 2005.08290, the entry's own group). A group going on after a mark elsewhere is
590  /// no such thing, and no cut: `Univ A, \textit{$^{2}$H} Lab`, an isotope (`perfect_kernel_batch107e::a_wrapped_isotope_is_entry_text`).
591  fn starts_at_its_brace(&self, nth: u8, from: usize, close: usize) -> bool {
592    let Ok(pos) = u16::try_from(self.base + from) else {
593      return false;
594    };
595    let at = self.formulas.partition_point(|formula| formula.end <= pos);
596    self.formulas.get(at).is_some_and(|formula| {
597      formula.start == pos
598        && usize::from(formula.end) <= self.base + close
599        && (formula.alone_in >= nth || formula.opens_line_in >= nth)
600    })
601  }

603  /// `start..end` grown to a balanced run, its shifts in pairs: back to the opening brace of a group it closes that starts there
604  /// ([`Self::starts_at_its_brace`]; any other group it closes is no run), on over the closing braces right after a group it
605  /// opens.
606  pub(super) fn balanced(&self, start: usize, end: usize, floor: usize) -> Option<(usize, usize)> {
607    let content = self.tokens;
608    let mut start = start;
609    let mut before = self.open_groups(floor, start);
610    let from = start;
611    let mut depth = 0;
612    let mut closed = 0u8;
613    for (close, t) in content.iter().enumerate().take(end).skip(from) {
614      depth += t.group_step();
615      if depth < 0 {
616        let open = before.pop()?;
617        closed = closed.saturating_add(1);
618        self
619          .starts_at_its_brace(closed, from, close)
620          .then_some(())?;
621        start = open;
622        depth = 0;
623      }
624    }
625    let end = past_closers(content, end, depth)?;
626    let mut end = self.environments_closed(start, end)?;
627    // (a formula the digested list did not say, its shifts in pairs all the same)
628    let run = &content[start..end];
629    let spelled = |name| run.iter().filter(|t| is_cs(t, name)).count();
630    let shifts = run
631      .iter()
632      .filter(|t| t.get_catcode() == Catcode::MATH)
633      .count();
634    if spelled("\\(") != spelled("\\)") {
635      return None;
636    }
637    if shifts % 2 == 1 {
638      (content.get(end)?.get_catcode() == Catcode::MATH).then_some(())?;
639      end += 1;
640    }
641    Some((start, end))
642  }

644  /// `end`, grown past the `\end` of each environment the run `start..end` opens, as the digestion's frames pair them;
645  /// `None` where the run closes one it doesn't open.
646  fn environments_closed(&self, start: usize, end: usize) -> Option<usize> {
647    let content = self.tokens;
648    let mut end = end;
649    // (its environments in pairs, as the digestion's frames say them: what `\end{ee}` prints or closes holds a position no box stops at, its
650    // body's `E` written in `\newenvironment{ee}{B}{E}`, an alignment's last cell, and a run with no `\end` lost them, in an Error; a `\let` copy
651    // and a macro pair count, a `\begin` under `\detokenize` doesn't)
652    let at = |p: u16| usize::from(p).checked_sub(self.base);
653    let mut environments = 0i32;
654    for step in self.environments {
655      // (one a formula holds is the formula's, taken whole: `$\begin{array}{c}a\end{array}$`, review revZ4a)
656      let Some(from) = at(step.start).filter(|&from| self.formula_at(from).is_none()) else {
657        continue;
658      };
659      if from < start {
660        continue;
661      }
662      if from >= end && environments == 0 {
663        break;
664      }
665      environments += if step.opens { 1 } else { -1 };
666      (environments >= 0).then_some(())?;
667      if from >= end || environments == 0 {
668        end = end.max(at(step.end)?.min(content.len()));
669        // (an alignment's cell read its `\\end` and heard the name's brace next: the name is the `\\end`'s)
670        if !step.opens
671          && at(step.end)? == from + 1
672          && content
673            .get(end)
674            .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
675        {
676          let mut depth = 0;
677          while let Some(t) = content.get(end) {
678            depth += t.group_step();
679            end += 1;
680            if depth == 0 {
681              break;
682            }
683          }
684        }
685      }
686    }
687    (environments == 0).then_some(())?;
688    Some(end)
689  }

691  /// `start`, or past the environment a span opens and doesn't close: its first box takes the cover of what opened it
692  /// (1507.02068's `\\begin{center}M. Ablikim`; a `\thanks{\begin{tabular}{l}$^{1}$Univ A\\ $^{2}$Univ B\end{tabular}}`
693  /// line's piece), a column spec with it, as the digestion's frames say it; and past the spaces after each such opening,
694  /// which print nothing (1507.02068, 2609.17357).
695  pub(super) fn past_open_environment(&self, start: usize, end: usize) -> usize {
696    let at = |p: u16| usize::from(p).saturating_sub(self.base);
697    let environments = self.environments;
698    let mut from = start;
699    for (n, step) in environments.iter().enumerate() {
700      if step.opens && at(step.start) == from && at(step.end) < end {
701        let closed = environments[n + 1..]
702          .iter()
703          .any(|later| !later.opens && at(later.start) < end);
704        if !closed {
705          from = at(step.end);
706          from += self.tokens.get(from..end).map_or(0, leading_spaces);
707        }
708      }
709    }
710    from
711  }

713  /// `all[start..end]`, a piece cut from the field, made whole as TeX reads it: an `\end{…}` it opens with is the piece
714  /// before's (2606.27043's `\end{CJK}\yangchar`), an environment it opens and doesn't close is the pieces' around it
715  /// ([`Self::past_open_environment`]); a command it ends on takes its arguments (`notes::takes_arguments`; 2606.14188);
716  /// a group it closes but does not open is opened again (`notes::reopened`; 2605.27539, 2605.00634); one it opens is
717  /// closed by the closers after it. `None` where a group or an environment stays unpaired.
718  pub(super) fn whole(&self, start: usize, end: usize) -> Option<Vec<Token>> {
719    let all = self.tokens;
720    let past_spaces = |i: usize| i + all.get(i..).map_or(0, leading_spaces);
721    let (mut start, mut end) = (start.min(all.len()), end.min(all.len()));
722    let first = past_spaces(start);
723    if first < end && all[first] == T_CS!("\\end") {
724      start = past_spaces(skip_group(all, past_spaces(first + 1)));
725    }
726    start = self.past_open_environment(start.min(end), end);
727    if let Some(last) = all[start..end].iter().rposition(|t| *t != T_SPACE!())
728      && takes_arguments(&all[start + last])
729    {
730      loop {
731        let past = optional_arguments_end(all, end);
732        let open = past_spaces(past);
733        if !all
734          .get(open)
735          .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
736        {
737          end = past;
738          break;
739        }
740        end = skip_group(all, open);
741      }
742    }
743    // (the groups it closes but does not open, innermost first)
744    let mut reopen = Vec::new();
745    let mut depth = 0i32;
746    let mut before = (0..start).rev();
747    for t in &all[start..end] {
748      depth += t.group_step();
749      if depth < 0 {
750        let mut inner = 0i32;
751        let open = before.by_ref().find(|&k| {
752          inner += all[k].group_step();
753          inner > 0
754        })?;
755        reopen.push(reopened(all, open, start));
756        depth = 0;
757      }
758    }
759    // (a group it leaves open closes only where the field closes it: what lies between is no piece's to drop, a picture's
760    // `;` in `\tikz\draw (0,0)--(1,0);}`, perfect_kernel_batchcr8)
761    let end = past_closers(all, end, depth)?;
762    (self.environments_closed(start, end)? == end).then_some(())?;
763    let mut out: Vec<Token> = reopen.into_iter().rev().flatten().collect();
764    out.extend_from_slice(&all[start..end]);
765    Some(out)
766  }
767}

769/// `end`, past the closing braces right after it of the `depth` groups a run leaves open; `None` where another token comes
770/// first.
771fn past_closers(tokens: &[Token], mut end: usize, depth: i32) -> Option<usize> {
772  for _ in 0..depth.max(0) {
773    (tokens.get(end)?.group_step() == -1).then_some(())?;
774    end += 1;
775  }
776  Some(end)
