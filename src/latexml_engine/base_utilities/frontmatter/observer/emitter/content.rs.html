1//! What a span of a field holds (the digested refactor's Stage 3, `docs/archive/FRONTMATTER_DIGESTED_REFACTOR_2026-11-07.md`): the tokens a
2//! piece, a line or an entry the reading cut is made from, read from the digested list and its spans. A box covers what
3//! it prints and no more, so a span stops short of what prints nothing around it, and that is said by the digested list
4//! too, not by walking tokens back from the span: the formula it starts or ends in, whole (`Field::formulas`, whatever
5//! opens it: `$\rm ^{3}$`, `${}^{*}$`, `\( ^{1} \)`, `\protect\(`, 2105.07958, hep-ph0110378, 2307.12984), and a mark's own
6//! print inside a wrapper whose cover its first box took ([`Field::own_print`], `\textbf{$^2$Univ B}`, 2605.25195). What
7//! is left of the tokens is brace balance, a property of the content itself, as the declarations' spans clamp at it
8//! (`field_notes::settle_declarations`).

25/// A field's tokens as a rehearsal read them: `tokens[0]` at position `base`, and its formulas (`events::lex_with_formulas`:

29  pub base:         usize,

41/// A field's groups as its braces open and close them, read once as far as asked, not again from its start for each
42/// piece: before token `i`, the innermost group still open — `inner[i]`, its opening brace's index plus one, 0 for none —,
43/// a group's own enclosing one the innermost open before its brace (`inner[brace]`). A closing brace no group is open at
44/// closes none (1507.02068's 418 names, each piece's open groups read from the block's start: about 0.7 % of the run).
45#[derive(Debug, Default)]
46pub(in crate::base_utilities::frontmatter::observer) struct Groups {
47  inner: std::cell::RefCell<Vec<usize>>,
48}

50impl Groups {
51  /// The opening braces in `content[from..to]` of the groups still open at `to`, in order: the innermost open group's
52  /// enclosing ones down to `from`, each read once for the field.
53  fn open(&self, content: &[Token], from: usize, to: usize) -> Vec<usize> {
54    let to = to.min(content.len());
55    let mut inner = self.inner.borrow_mut();
56    if inner.is_empty() {
57      inner.push(0);
58    }
59    while inner.len() <= to {
60      let i = inner.len() - 1;
61      let here = inner[i];
62      let next = match content[i].group_step() {
63        1 => i + 1,
64        -1 => here.checked_sub(1).map_or(0, |brace| inner[brace]),
65        _ => here,
66      };
67      inner.push(next);
68    }
69    let mut open = Vec::new();
70    let mut at = inner[to];
71    while let Some(brace) = at.checked_sub(1).filter(|&brace| brace >= from) {
72      open.push(brace);
73      at = inner[brace];
74    }
75    open.reverse();
76    open
77  }
78}

80/// Whether a digested list's events hold an alignment's cell (`events::Walker`'s `Layout{role="cell"}`): where none is, no tab of
81/// it is an alignment's — a verbatim argument read it as a character (`\url{http://y.org/?c=1&d=2}`), or it was a stray one in
82/// the rehearsal as everywhere. A note's events too (`Event::inner`, `Event::body`): a tabular in a `\thanks`.
83pub(in crate::base_utilities::frontmatter::observer) fn aligned(events: &[Event]) -> bool {
84  events.iter().any(|e| {
85    (e.kind == EventKind::Layout && e.role.as_deref() == Some("cell"))
86      || aligned(&e.inner)
87      || aligned(&e.body)
88  })
89}

91/// Where a call in `tokens` reads past them: a macro at their top level whose parameters are each undelimited (TeX's
92/// `#1#2`, `\newcommand`'s `[2]`; a verbatim group, `\href`'s URL) or an optional one (`[…]`), by its meaning now, an
93/// undelimited argument of which the tokens don't hold — the next read at a closing brace or past the last token, there
94/// the wrapper's own tokens (`\lx@author@withinst`'s `\let\inst\lx@saved@inst…`, `frontmatter_fields.rs`) or the item's
95/// brace. Whether it is cut between its arguments (`\def\authord#1#2{…}`'s `\authord{Anh Thu Nguyen}`, 2209.12410) or
96/// the tokens end at it, none of its arguments theirs (a mark's span covering the call alone: aiaa-tc's
97/// `\thanksibid{2}`, 1505.07844, 1411.3642, 1602.00849; a preamble's `\footrecall{uva}`, 2503.21719, 2407.13730,
98/// 2606.19583; a URL cut out of `\href{…}{…}`, 2603.11001). A parameter of any other kind (a delimited one) is not
99/// judged, nor a command inside a group of theirs (an argument a primitive may never expand: `\detokenize{\begin}`). A
100/// call's arguments are its [`tail_call`]'s where `tail` (a printed call's); a declaration's are its own macro's.
101pub(super) fn reads_past(tokens: &[Token], tail: bool) -> Option<usize> {
102  let mut depth = 0i32;
103  tokens.iter().enumerate().position(|(i, t)| {
104    depth += t.group_step();
105    depth == 0 && arguments_end(tokens, i, tail) == Some(None)
106  })
107}

109/// Whether `tokens` read a meaning of `definees`: its control sequence, or a counter's name as an argument (`\c@uni`'s
110/// `{uni}`: `\footnoterecall{uni}`, `\value{uni}`), in them or one level into a document's macro they call.
111pub(super) fn reads_definition(tokens: &[Token], definees: &[SymStr]) -> bool {
112  let counters: Vec<String> = definees
113    .iter()
114    .filter_map(|&key| with(key, |cs| cs.strip_prefix("\\c@").map(str::to_string)))
115    .collect();
116  let reads = |tokens: &[Token]| {
117    tokens.iter().enumerate().any(|(i, t)| {
118      if t.get_catcode().is_active_or_cs() {
119        return definees.contains(&meaning_key(t));
120      }
121      if t.get_catcode() != Catcode::BEGIN || counters.is_empty() {
122        return false;
123      }
124      let name: Option<String> = tokens[i + 1..]
125        .iter()
126        .take_while(|u| u.get_catcode() != Catcode::END)
127        .map(|u| {
128          matches!(u.get_catcode(), Catcode::LETTER | Catcode::OTHER)
129            .then(|| u.with_str(str::to_string))
130        })
131        .collect();
132      name.is_some_and(|name| counters.contains(&name))
133    })
134  };
135  reads(tokens)
136    || tokens.iter().any(|t| match lookup_meaning(t) {
137      Some(Stored::Expandable(d))
138        if d.made_in_own_file || d.get_origin() == DefinitionOrigin::Document =>
139      {
140        matches!(d.get_expansion(), Some(ExpansionBody::Tokens(body)) if reads(body.unlist_ref()))
141      },
142      _ => false,
143    })
144}

146/// Where the call at `tokens[at]` ends, past the arguments its meaning now reads ([`reads_past`]'s parameters):
147/// `Some(None)` where an undelimited one is not in the tokens, `None` for no such call (a primitive, an unexpandable
148/// command, no parameters, a parameter of another kind).
149pub(super) fn arguments_end(tokens: &[Token], at: usize, tail: bool) -> Option<Option<usize>> {
150  let (plain, verbatim, hyper, optional) = (
151    pin!("Plain"),
152    pin!("Semiverbatim"),
153    pin!("HyperVerbatim"),
154    pin!("Optional"),
155  );
156  let meaning = if tail {
157    tail_call(&tokens[at])?
158  } else {
159    let Some(Stored::Expandable(meaning)) = lookup_meaning(&tokens[at]) else {
160      return None;
161    };
162    meaning
163  };
164  let parameters = meaning.get_parameters()?.get_parameters();
165  if parameters.is_empty()
166    || parameters
167      .iter()
168      .any(|p| !(p.name == plain || p.name == verbatim || p.name == hyper || p.name == optional))
169  {
170    return None;
171  }
172  let mut j = at + 1;
173  for parameter in parameters {
174    while tokens
175      .get(j)
176      .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
177    {
178      j += 1;
179    }
180    if parameter.name == optional {
181      // (an option present is read to its `]` at the top level; one absent reads nothing: the next token is peeked)
182      if tokens
183        .get(j)
184        .is_some_and(|t| t.get_catcode() == Catcode::OTHER && t.with_str(|s| s == "["))
185      {
186        let mut inner = 0i32;
187        loop {
188          let Some(t) = tokens.get(j) else {
189            return Some(None);
190          };
191          inner += t.group_step();
192          j += 1;
193          if inner == 0 && t.get_catcode() == Catcode::OTHER && t.with_str(|s| s == "]") {
194            break;
195          }
196        }
197      }
198      continue;
199    }
200    match tokens.get(j).map(Token::group_step) {
201      None | Some(-1) => return Some(None),
202      Some(1) => {
203        let mut inner = 0i32;
204        loop {
205          let Some(t) = tokens.get(j) else {
206            return Some(None);
207          };
208          inner += t.group_step();
209          j += 1;
210          if inner == 0 {
211            break;
212          }
213        }
214      },
215      Some(_) => j += 1,
216    }
217  }
218  Some(Some(j))
219}

221/// The macro whose parameters a call of `cs` reads, by its meaning now: `cs`'s own, or, for a parameterless macro whose
222/// expansion ends in a command, that command's — a robust command's `\protect \cs␣` (icrc2009.cls's
223/// `\DeclareRobustCommand*{\IEEEauthorrefmark}[1]`, 0912.0896), IEEEtran's `\def\authorrefmark{…\IEEEauthorrefmark}`
224/// (2609.25589): cut from its `{1}`, the mark read the wrapper's `\let` as its `\ifcase` number.
225fn tail_call(cs: &Token) -> Option<Rc<Expandable>> {
226  let mut called = *cs;
227  for _ in 0..8 {
228    let Some(Stored::Expandable(meaning)) = lookup_meaning(&called) else {
229      return None;
230    };
231    let parameterless = meaning
232      .get_parameters()
233      .is_none_or(|parameters| parameters.get_parameters().is_empty());
234    match meaning.get_expansion() {
235      Some(ExpansionBody::Tokens(body)) if parameterless => {
236        called = *body.unlist_ref().last()?;
237        if called.get_catcode() != Catcode::CS {
238          return None;
239        }
240      },
241      _ => return Some(meaning),
242    }
243  }
244  None
245}

255/// Where `tokens` hold a conditional cut apart, by their meanings now: an `\else`, `\or` or `\fi` whose `\if…` is not
256/// theirs, or an `\if…` whose `\fi` is not theirs — digested apart, the one errs ("not in a conditional"), the other
257/// stays open to the end of the document (`\iffull Bob Baker\fnm{2}\fi`'s name run `\iffull Bob Baker`, review revX4 2).
258/// A switch's name not executed where it stands is one as well (`\newif\iffull`, `\let\ifa\iftrue`, what `\ifx`
259/// compares), which no meaning tells apart from a test: such a run is refused too, its item written as its text, counted.
260fn conditional_cut(tokens: &[Token]) -> Option<usize> {
261  let mut open = Vec::new();
262  for (i, t) in tokens.iter().enumerate() {
263    match conditional_of(t) {
264      Some(ConditionalType::If) => open.push(i),
265      Some(ConditionalType::Else | ConditionalType::Or | ConditionalType::Fi)
266        if open.is_empty() =>
267      {
268        return Some(i);
269      },
270      Some(ConditionalType::Fi) => {
271        open.pop();
272      },
273      _ => {},
274    }
275  }
276  open.first().copied()
277}

279/// `end` grown over the `\fi`s right after `content[start..end]` that close a conditional it opens, as [`Field::balanced`]
280/// grows over closing braces: an entry printed by its last branch ends before the `\fi` that prints nothing (`\ifx\zz\end
281/// Q\else X\fi`, `perfect_kernel_batchz4::an_environment_by_its_frames`).
282fn past_closing_fis(content: &[Token], start: usize, end: usize) -> usize {
283  let mut open = 0usize;
284  for t in &content[start..end] {
285    match conditional_of(t) {
286      Some(ConditionalType::If) => open += 1,
287      Some(ConditionalType::Fi) => open = open.saturating_sub(1),
288      _ => {},
289    }
290  }
291  let mut end = end;
292  let mut next = end;
293  while open > 0 {
294    while content
295      .get(next)
296      .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
297    {
298      next += 1;
299    }
300    if !content
301      .get(next)
302      .is_some_and(|t| conditional_of(t) == Some(ConditionalType::Fi))
303    {
304      break;
305    }
306    next += 1;
307    end = next;
308    open -= 1;
309  }
310  end
311}

324  /// The formula holding `tokens[index]`: its indices, opening shift to past its closing shift.
325  fn formula_at(&self, index: usize) -> Option<(usize, usize)> {
326    let pos = u16::try_from(self.base + index).ok()?;
327    let at = self.formulas.partition_point(|formula| formula.end <= pos);
328    let formula = self.formulas.get(at)?;
329    (formula.start <= pos).then_some((
330      usize::from(formula.start).checked_sub(self.base)?,
331      usize::from(formula.end).checked_sub(self.base)?,
332    ))
333  }
334
335  /// `index`, or where the formula holding it starts.
336  pub(super) fn formula_start(&self, index: usize) -> usize {
337    self.formula_at(index).map_or(index, |(start, _)| start)
338  }
339
340  /// `index` (the end of a span), or past the formula holding the token before it.
341  pub(super) fn formula_end(&self, index: usize) -> usize {
342    index
343      .checked_sub(1)
344      .and_then(|last| self.formula_at(last))
345      .map_or(index, |(_, end)| end.max(index))
346  }

348  /// A Mark's own print as indices, where its span is wider or narrower: the whole formula it is in (the first box in a
349  /// wrapper takes the cover of the call around it: `\textbf{$^2$Univ B}`'s span is the wrapper's), or the
350  /// `\textsuperscript{…}` call to its closing brace. `None` for a mark of neither.
351  pub(super) fn own_print(&self, mark: &Event) -> Option<(usize, usize)> {
352    let index = |p: u16| usize::from(p).checked_sub(self.base);
353    if let Some(formula) = mark.formula.filter(|f| f.base.is_some()) {
354      let (start, end) = formula.span?;
355      return Some((index(start)?, index(end)?));
356    }
357    let call = index(mark.own_at?)?;
358    let mut depth = 0;
359    for (k, t) in self.tokens.iter().enumerate().skip(call + 1) {
360      if depth == 0 && t.group_step() != 1 {
361        return None;
362      }
363      depth += t.group_step();
364      if depth == 0 {
365        return Some((call, k + 1));
366      }
367    }
368    None
369  }

371  /// The content of the span `start..end` (indices), past `floor` (where the piece before ended): the balanced run of
372  /// tokens holding it, and where that ends — with the switches that open the group it sits in, when it sits inside one
373  /// (`{\bf $^1$Univ A, $^2$Univ B}`: each piece keeps the `\bf`, unless `first` is 0: a piece carrying the font it
374  /// starts in, `events::entry_font`) —, or `None` when no balanced run holds it.
375  ///
376  /// The run takes each formula the span starts or ends in whole, a mark's own font group from its opening brace
377  /// (`{\bf $^1$} Univ A`) and the closing braces right after a group it opens.
378  pub(super) fn content_of(
379    &self,
380    range: (usize, usize),
381    floor: usize,
382    first: usize,
383  ) -> Option<(Tokens, usize)> {
384    self.content_cut(range, floor, first, &[])
385  }
386
387  /// [`Self::content_of`] without the runs `cuts` (indices, each balanced: an entry's address runs, `address_cut`), `None`
388  /// where one of them is partly inside the content and partly outside it.
389  pub(super) fn content_cut(
390    &self,
391    (start, end): (usize, usize),
392    floor: usize,
393    first: usize,
394    cuts: &[(usize, usize)],
395  ) -> Option<(Tokens, usize)> {
396    let content = self.tokens;
397    if start > end || end > content.len() || start < floor {
398      return None;
399    }
400    let (mut start, mut end) = (start, end);
401    // (a formula the piece before ended in stays as it was, its other end past `floor`: the mark's closing shift is the
402    // mark's, `$^1 Univ$`)
403    if let Some((s, e)) = self.formula_at(start).filter(|&(s, _)| s >= floor) {
404      start = s;
405      end = end.max(e);
406    }
407    if let Some((s, e)) = end
408      .checked_sub(1)
409      .and_then(|last| self.formula_at(last))
410      .filter(|&(s, _)| s >= floor)
411    {
412      start = start.min(s);
413      end = end.max(e);
414    }
415    end = end.min(content.len());
416    let end = self.past_held_undefined(end);
417    let (start, end) = self.balanced(start, end, floor)?;
418    let end = past_closing_fis(content, start, end);
419    // (the group the pieces sit in: its opening brace, and the switches before the first piece in it)
420    let mut tokens = Vec::new();
421    if let Some(&group) = self.open_groups(0, start).last()
422      && group < first
423    {
424      let switches = &content[group + 1..first.min(start)];
425      if switches
426        .iter()
427        .all(|t| matches!(t.get_catcode(), Catcode::CS | Catcode::SPACE))
428      {
429        tokens.extend_from_slice(switches);
430      }
431    }
432    if cuts
433      .iter()
434      .any(|&(s, t)| s < end && t > start && (s < start || t > end))
435    {
436      return None;
437    }
438    if cuts.is_empty() && self.held.is_empty() {
439      tokens.extend_from_slice(&content[start..end]);
440    } else {
441      // (a script character its rehearsal read outside math, in no formula, an error held there, is the character that rehearsal printed:
442      // `\textbf{Ningkai Mo} ^1` is a text-mode `^`, which TeX errs at (Missing $) and the digestion's recovery prints;
443      // digested apart as a script it erred again, once per piece (2204.01080). `\lx@fm@script@char` prints it and
444      // counts the recovery, a Warning. An `_` a verbatim argument reads as a character holds no error: `\url{a_b}` stays)
445      let cut = |i: &usize| cuts.iter().any(|&(s, t)| (s..t).contains(i));
446      for i in (start..end).filter(|i| !cut(i)) {
447        let t = content[i];
448        if matches!(t.get_catcode(), Catcode::SUPER | Catcode::SUB)
449          && self.formula_at(i).is_none()
450          && self.held_at(i)
451        {
452          tokens.push(T_CS!("\\lx@fm@script@char"));
453          tokens.push(T_BEGIN!());
454          tokens.push(T_OTHER!(if t.get_catcode() == Catcode::SUPER {
455            "^"
456          } else {
457            "_"
458          }));
459          tokens.push(T_END!());
460        } else {
461          tokens.push(t);
462        }
463      }
464    }
465    // (never a run holding an alignment's tab or alignment-only command outside an environment or a formula it holds whole, as
466    // the digestion's frames say them (a macro pair's `\bx a&b\\ c&d\ex` is one): digested apart, it is a stray one, an Error
467    // — its caller writes the item's text, counted (the rule of `plan::cell_tokens`; 1304.3138, 1504.01048, 2607.22455). Only
468    // where the digestion holds an alignment ([`aligned`]): a `\url{…?c=1&d=2}` in a block of none keeps its link (review
469    // revNE1 1))
470    // (nor a run a macro of which reads past its end, by its meaning, which the run is digested under: `\def\authord#1#2{…}`'s
471    // `\authord{Anh Thu Nguyen}`, its second argument the affiliation's piece, read the item's closing brace (an undefined
472    // `\@undefined` and a group ended in the wrong mode, 2209.12410) — its caller writes the item's text, counted)
473    // (nor a run holding a conditional cut apart, an `\else` whose `\if` is not the run's: `\ifFull and \else \and \fi
474    // Joseph A. Simons`, whose name's run took the `\else \and \fi` the rehearsal skipped, an Error and a tabular's group
475    // ended in the wrong mode (1409.0597) — its caller writes the item's text, counted)
476    if let Some(at) = conditional_cut(&tokens) {
477      DebugFeature!(
478        "frontmatter",
479        "FRONT writer: the run {start}..{end} holds {:?}, a conditional cut apart",
480        tokens[at]
481      );
482      return None;
483    }
484    if let Some(call) = reads_past(&tokens, true) {
485      DebugFeature!(
486        "frontmatter",
487        "FRONT writer: the run {start}..{end} holds {:?}, whose arguments run past it",
488        tokens[call]
489      );
490      return None;
491    }
492    // (what it holds whole read only for a run holding such a token at all)
493    let mut held: Option<Vec<(usize, usize)>> = None;
494    if let Some(stray) = (start..end).find(|&i| {
495      self.aligned
496        && reads_as_alignment(&content[i])
497        && !cuts.iter().any(|&(s, t)| (s..t).contains(&i))
498        && !held
499          .get_or_insert_with(|| self.held_whole(start, end))
500          .iter()
501          .any(|&(s, t)| (s..t).contains(&i))
502    }) {
503      DebugFeature!(
504        "frontmatter",
505        "FRONT writer: the run {start}..{end} holds an alignment's {:?} at {stray}",
506        content[stray]
507      );
508      return None;
509    }
510    Some((Tokens::new(tokens), end))
511  }

513  /// `end` moved past the undefined commands right after it, each one its rehearsal held an Error at (one past a space or
514  /// a mark is a name's by [`super::emit`]'s `block_names`, else the flush's net raises it, `reading::HeldUndefined`): the
515  /// command prints nothing (TeX's "Undefined control sequence"), so no event covers it and the piece before it would
516  /// stop short of it, its Error — the rehearsal's, held and discarded with its stub — never raised again. Taken into the

569  /// The environments (their frames' steps, paired) and the formulas the run `start..end` holds whole: their indices.
570  fn held_whole(&self, start: usize, end: usize) -> Vec<(usize, usize)> {
571    let at = |p: u16| usize::from(p).checked_sub(self.base);
572    let mut held = Vec::new();
573    let mut open = Vec::new();
574    for step in self.environments {
575      let (Some(from), Some(to)) = (at(step.start), at(step.end)) else {
576        continue;
577      };
578      if from < start || from >= end {
579        continue;
580      }
581      if step.opens {
582        open.push(from);
583      } else if let Some(opened) = open.pop()
584        && to <= end
585      {
586        held.push((opened, to));
587      }
588    }
589    let formulas = self
590      .formulas
591      .iter()
592      .filter_map(|formula| Some((at(formula.start)?, at(formula.end)?)));
593    held.extend(formulas.filter(|&(s, t)| start <= s && t <= end));
594    held
595  }

597  /// Whether the `nth` group around the formula starting at `from`, from the innermost, closing at `close`, starts at its brace with that formula: it prints
598  /// the formula alone (`events::Formula::alone_in`: `{\bf $^{1}$} Univ X`, `{$^{1}$\,} Univ X`, a mark's own group), or the formula opens it where a line
599  /// starts (`Formula::opens_line_in`: `\\ \normalsize{$^{1}$ Univ X,} Moscow`, 2005.08290, the entry's own group). A group going on after a mark elsewhere is
600  /// no such thing, and no cut: `Univ A, \textit{$^{2}$H} Lab`, an isotope (`perfect_kernel_batch107e::a_wrapped_isotope_is_entry_text`).
601  fn starts_at_its_brace(&self, nth: u8, from: usize, close: usize) -> bool {
602    let Ok(pos) = u16::try_from(self.base + from) else {
603      return false;
604    };
605    let at = self.formulas.partition_point(|formula| formula.end <= pos);
606    self.formulas.get(at).is_some_and(|formula| {
607      formula.start == pos
608        && usize::from(formula.end) <= self.base + close
609        && (formula.alone_in >= nth || formula.opens_line_in >= nth)
610    })
611  }

613  /// `start..end` grown to a balanced run, its shifts in pairs: back to the opening brace of a group it closes that starts there
614  /// ([`Self::starts_at_its_brace`]; any other group it closes is no run), on over the closing braces right after a group it
615  /// opens.
616  pub(super) fn balanced(&self, start: usize, end: usize, floor: usize) -> Option<(usize, usize)> {
617    let content = self.tokens;
618    let mut start = start;
619    let mut before = self.open_groups(floor, start);
620    let from = start;
621    let mut depth = 0;
622    let mut closed = 0u8;
623    for (close, t) in content.iter().enumerate().take(end).skip(from) {
624      depth += t.group_step();
625      if depth < 0 {
626        let open = before.pop()?;
627        closed = closed.saturating_add(1);
628        self
629          .starts_at_its_brace(closed, from, close)
630          .then_some(())?;
631        start = open;
632        depth = 0;
633      }
634    }
635    let end = past_closers(content, end, depth)?;
636    let mut end = self.environments_closed(start, end)?;
637    // (a formula the digested list did not say, its shifts in pairs all the same)
638    let run = &content[start..end];
639    let spelled = |name| run.iter().filter(|t| is_cs(t, name)).count();
640    let shifts = run
641      .iter()
642      .filter(|t| t.get_catcode() == Catcode::MATH)
643      .count();
644    if spelled("\\(") != spelled("\\)") {
645      return None;
646    }
647    if shifts % 2 == 1 {
648      (content.get(end)?.get_catcode() == Catcode::MATH).then_some(())?;
649      end += 1;
650    }
651    Some((start, end))
652  }

654  /// `end`, grown past the `\end` of each environment the run `start..end` opens, as the digestion's frames pair them;
655  /// `None` where the run closes one it doesn't open.
656  fn environments_closed(&self, start: usize, end: usize) -> Option<usize> {
657    let content = self.tokens;
658    let mut end = end;
659    // (its environments in pairs, as the digestion's frames say them: what `\end{ee}` prints or closes holds a position no box stops at, its
660    // body's `E` written in `\newenvironment{ee}{B}{E}`, an alignment's last cell, and a run with no `\end` lost them, in an Error; a `\let` copy
661    // and a macro pair count, a `\begin` under `\detokenize` doesn't)
662    let at = |p: u16| usize::from(p).checked_sub(self.base);
663    let mut environments = 0i32;
664    for step in self.environments {
665      // (one a formula holds is the formula's, taken whole: `$\begin{array}{c}a\end{array}$`, review revZ4a)
666      let Some(from) = at(step.start).filter(|&from| self.formula_at(from).is_none()) else {
667        continue;
668      };
669      if from < start {
670        continue;
671      }
672      if from >= end && environments == 0 {
673        break;
674      }
675      environments += if step.opens { 1 } else { -1 };
676      (environments >= 0).then_some(())?;
677      if from >= end || environments == 0 {
678        end = end.max(at(step.end)?.min(content.len()));
679        // (an alignment's cell read its `\\end` and heard the name's brace next: the name is the `\\end`'s)
680        if !step.opens
681          && at(step.end)? == from + 1
682          && content
683            .get(end)
684            .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
685        {
686          let mut depth = 0;
687          while let Some(t) = content.get(end) {
688            depth += t.group_step();
689            end += 1;
690            if depth == 0 {
691              break;
692            }
693          }
694        }
695      }
696    }
697    (environments == 0).then_some(())?;
698    Some(end)
699  }

701  /// `start`, or past the environment a span opens and doesn't close: its first box takes the cover of what opened it
702  /// (1507.02068's `\\begin{center}M. Ablikim`; a `\thanks{\begin{tabular}{l}$^{1}$Univ A\\ $^{2}$Univ B\end{tabular}}`
703  /// line's piece), a column spec with it, as the digestion's frames say it; and past the spaces after each such opening,
704  /// which print nothing (1507.02068, 2609.17357).
705  pub(super) fn past_open_environment(&self, start: usize, end: usize) -> usize {
706    let at = |p: u16| usize::from(p).saturating_sub(self.base);
707    let environments = self.environments;
708    let mut from = start;
709    for (n, step) in environments.iter().enumerate() {
710      if step.opens && at(step.start) == from && at(step.end) < end {
711        let closed = environments[n + 1..]
712          .iter()
713          .any(|later| !later.opens && at(later.start) < end);
714        if !closed {
715          from = at(step.end);
716          from += self.tokens.get(from..end).map_or(0, leading_spaces);
717        }
718      }
719    }
720    from
721  }

723  /// `all[start..end]`, a piece cut from the field, made whole as TeX reads it: an `\end{…}` it opens with is the piece
724  /// before's (2606.27043's `\end{CJK}\yangchar`), an environment it opens and doesn't close is the pieces' around it
725  /// ([`Self::past_open_environment`]); a command it ends on takes its arguments (`notes::takes_arguments`; 2606.14188);
726  /// a group it closes but does not open is opened again (`notes::reopened`; 2605.27539, 2605.00634); one it opens is
727  /// closed by the closers after it. `None` where a group or an environment stays unpaired.
728  pub(super) fn whole(&self, start: usize, end: usize) -> Option<Vec<Token>> {
729    let all = self.tokens;
730    let past_spaces = |i: usize| i + all.get(i..).map_or(0, leading_spaces);
731    let (mut start, mut end) = (start.min(all.len()), end.min(all.len()));
732    let first = past_spaces(start);
733    if first < end && all[first] == T_CS!("\\end") {
734      start = past_spaces(skip_group(all, past_spaces(first + 1)));
735    }
736    start = self.past_open_environment(start.min(end), end);
737    if let Some(last) = all[start..end].iter().rposition(|t| *t != T_SPACE!())
738      && takes_arguments(&all[start + last])
739    {
740      loop {
741        let past = optional_arguments_end(all, end);
742        let open = past_spaces(past);
743        if !all
744          .get(open)
745          .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
746        {
747          end = past;
748          break;
749        }
750        end = skip_group(all, open);
751      }
752    }
753    // (the groups it closes but does not open, innermost first)
754    let mut reopen = Vec::new();
755    let mut depth = 0i32;
756    let mut before = (0..start).rev();
757    for t in &all[start..end] {
758      depth += t.group_step();
759      if depth < 0 {
760        let mut inner = 0i32;
761        let open = before.by_ref().find(|&k| {
762          inner += all[k].group_step();
763          inner > 0
764        })?;
765        reopen.push(reopened(all, open, start));
766        depth = 0;
767      }
768    }
769    // (a group it leaves open closes only where the field closes it: what lies between is no piece's to drop, a picture's
770    // `;` in `\tikz\draw (0,0)--(1,0);}`, perfect_kernel_batchcr8)
771    let end = past_closers(all, end, depth)?;
772    (self.environments_closed(start, end)? == end).then_some(())?;
773    let mut out: Vec<Token> = reopen.into_iter().rev().flatten().collect();
774    out.extend_from_slice(&all[start..end]);
775    Some(out)
776  }
777}

779/// `end`, past the closing braces right after it of the `depth` groups a run leaves open; `None` where another token comes
780/// first.
781fn past_closers(tokens: &[Token], mut end: usize, depth: i32) -> Option<usize> {
782  for _ in 0..depth.max(0) {
783    (tokens.get(end)?.group_step() == -1).then_some(())?;
784    end += 1;
785  }
786  Some(end)
