19/// The span of a block's entry the reading keys `key` ([`entry_span_in`] on the block's events at ledger index `index`).
20pub(super) fn entry_span(index: usize, span: (u16, u16), key: &str) -> Option<(u16, u16)> {
21  with_ledger(|ledger| {
22    let Some(Entry::Author(block)) = ledger.entries().get(index) else {
23      return None;
24    };
25    let Outcome::Events { events, .. } = &block.outcome else {
26      return None;
27    };
28    entry_span_in(events, span, key)
29  })
30}

32/// The tokens from `span`'s start of the block at ledger index `index` that print `key`, where `key` ends inside an ink
33/// ([`entry_span`] reads whole inks): an item the reading writes without the addresses its line prints last
34/// (`Resolver::block`'s row address: `Dresden, Saxony 01062, Germany (martin.waltz@tu-dresden.de)` the affiliation `Dresden,
35/// Saxony 01062, Germany`, 2203.10777). The cut ends at the first address's own first token — past every token printing
36/// `key`'s letters, the writer's own check of a span's letters ([`block_span_tokens`]) — less the spaces and separators
37/// before it, the commands opening it and the bracket it opens, as the reading's cut leaves them (`address_cuts`); a bracket it leaves open closed by
38/// its own closing token after the address: `Dresden [Germany] (bob.baker@a.org)` keeps `Dresden [Germany]`, `(Tel. 351
39/// 463, bob.baker@a.org)` `(Tel. 351 463)` (review revLO2 2: the shortest letter match dropped the ink after the last letter).
40/// A mark among the inks before it is its ink (`Dresden$^{\dagger}$, Germany`, review revLO2 base). `None` where the inks
41/// before it don't open `key`, or no run prints it.
42pub(super) fn prefix_tokens(
43  index: usize,
44  span: (u16, u16),
45  key: &str,
46) -> Option<((u16, u16), Tokens)> {
47  let (ends_in, tokens, base) = with_ledger(|ledger| {
48    let Some(Entry::Author(block)) = ledger.entries().get(index) else {
49      return None;
50    };
51    let Outcome::Events { events, .. } = &block.outcome else {
52      return None;
53    };
54    let first = events
55      .iter()
56      .position(|e| e.kind != EventKind::Layout && e.span.is_some_and(|s| s.0 == span.0))?;
57    let mut joined = String::new();
58    for event in &events[first..] {
59      let text = match &event.kind {
60        EventKind::Name | EventKind::Contact => norm(&event.text),
61        EventKind::Sep(SepKind::Word(word)) => norm(word),
62        EventKind::Sep(_) | EventKind::Break | EventKind::Layout | EventKind::Mark => continue,
63        _ => return None,
64      };
65      joined.push_str(&text);
66      if joined.len() > key.len() && joined.starts_with(key) {
67        return Some((event.span?, block.read_tokens().clone(), block.range.0));
68      }
69      if !key.starts_with(&joined) {
70        return None;
71      }
72    }
73    None
74  })?;
75  let prints_key = |end: u16| block_span_tokens(index, (span.0, end), Some(key), false);
76  // (the first address's own first token: the last end still printing `key` alone — or the address's start, where its
77  // first token is a call reading its arguments, `\textit{\{aable, bgh, ccole\}@uber.com, …}`: the end past the call
78  // alone is no run (`content::reads_past`, the call reading what follows the run), 2603.07345)
79  let mut address = (ends_in.0 + 1..ends_in.1)
80    .find(|&end| prints_key(end).is_some())
81    .or_else(|| prints_key(ends_in.0).map(|_| ends_in.0))?;
82  while address + 1 < ends_in.1 && prints_key(address + 1).is_some() {
83    address += 1;
84  }
85  let content = tokens.unlist_ref();
86  let at = |p: u16| {
87    usize::from(p)
88      .checked_sub(base)
89      .and_then(|i| content.get(i))
90  };
91  let char_of = |p: u16| {
92    at(p)
93      .filter(|t| t.get_catcode() == Catcode::OTHER)
94      .and_then(|t| t.with_str(|s| s.chars().next()))
95  };
96  let space = |p: u16| at(p).is_some_and(|t| t.get_catcode() == Catcode::SPACE);
97  // (and the commands and groups opening the address: `(\texttt{bob@a.org})`)
98  let opening = |p: u16| {
99    at(p).is_some_and(|t| {
100      matches!(
101        t.get_catcode(),
102        Catcode::SPACE | Catcode::CS | Catcode::BEGIN
103      )
104    })
105  };
106  const SEPARATORS: &[char] = &[',', ';', ':', '-', '\u{2013}', '\u{2014}', '/'];
107  let mut end = address;
108  while end > span.0 && opening(end - 1) {
109    end -= 1;
110  }
111  if end > span.0 && char_of(end - 1).is_some_and(|c| "([<".contains(c)) {
112    end -= 1;
113  }
114  while end > span.0
115    && (space(end - 1) || char_of(end - 1).is_some_and(|c| SEPARATORS.contains(&c)))
116  {
117    end -= 1;
118  }
119  let cut = (span.0, end);
120  let mut kept = prints_key(end)?.unlist();
121  // (a bracket the cut leaves open, closed by its own token after the address)
122  let mut open: Vec<char> = Vec::new();
123  for c in (span.0..end).filter_map(char_of) {
124    match c {
125      '(' | '[' => open.push(c),
126      ')' | ']' => {
127        open.pop();
128      },
129      _ => {},
130    }
131  }
132  let mut from = address;
133  for opener in open.into_iter().rev() {
134    let closer = if opener == '(' { ')' } else { ']' };
135    let Some(p) = (from..ends_in.1).find(|&p| char_of(p) == Some(closer)) else {
136      break;
137    };
138    kept.extend(at(p).copied());
139    from = p + 1;
140  }
141  Some((cut, Tokens::new(kept)))
142}

144/// The words of a text: its runs of letters and digits, lowercased.
145fn words(text: &str) -> Vec<String> {
146  text
147    .split(|c: char| !c.is_alphanumeric())
148    .filter(|w| !w.is_empty())
149    .map(str::to_lowercase)
150    .collect()
151}

153/// Whether the reading joined a creator's name of Name events with nothing but layout between them
154/// (`name:layout-joined`, the resolver's `joins_previous`) where an alignment's cell boundary is among that layout: the
155/// name's words run on from one cell into the next.
156pub(super) fn joined_across_cells(c: &RCreator) -> bool {
157  if !c.rule.contains("name:layout-joined") {
158    return false;
159  }
160  let name = words(&c.name);
161  with_ledger(|ledger| {
162    let Some(Entry::Author(block)) = ledger.entries().get(c.source) else {
163      return false;
164    };
165    let Outcome::Events { events, .. } = &block.outcome else {
166      return false;
167    };
168    let events = read_events(events);
169    let Some(at) = events.iter().position(|e| {
170      let text = words(&e.text);
171      e.kind == EventKind::Name && !text.is_empty() && name.starts_with(&text)
172    }) else {
173      return false;
174    };
175    let mut read = words(&events[at].text).len();
176    let mut cell = false;
177    for e in &events[at + 1..] {
178      if read >= name.len() {
179        break;
180      }
181      match e.kind {
182        EventKind::Layout => cell |= e.role.as_deref() == Some("cell"),
183        EventKind::Name => {
184          let text = words(&e.text);
185          if text.is_empty() || !name[read..].starts_with(&text) {
186            return false;
187          }
188          if cell {
189            return true;
190          }
191          read += text.len();
192        },
193        _ => return false,
194      }
195    }
196    false
197  })
198}

200/// Whether the ink the span `span` of the author block at ledger index `index` covers runs on from one column of an
201/// alignment into another ([`events_cross_columns`]): its tokens hold the alignment's `&`, or join a column's row with
202/// another column's — an item no cut of the block can make, which the emitter declines with its block (Perl's
203/// whole-block creator, Base_Utility.pool.ltxml:691-693): `$^\star$Univ X & $^\dagger$Univ Y\\ Dept A & Dept B`
204/// (1304.3138, 1711.10325, 1704.06967, 2511.14288, 2605.30319, 1802.06894).
205pub(super) fn crosses_columns(index: usize, span: (u16, u16)) -> bool {
206  with_ledger(|ledger| {
207    let Some(Entry::Author(block)) = ledger.entries().get(index) else {
208      return false;
209    };
210    let Outcome::Events { events, .. } = &block.outcome else {
211      return false;
212    };
213    events_cross_columns(events, span)
214  })
215}

217/// Whether, between the first and the last ink of `events` inside `start..end`, an alignment enclosing that ink (at a
218/// lower nest than any of it) starts a cell of a later column (its digested `Layout{role="cell"}`, labelled
219/// `row,column`) or ends a row of more than one cell (`Break{role="row"}` after a cell of a later column). The rows of
220/// one column are lines an item reads on over (`Univ X\\ City`, nucl-th0306055); an alignment the ink holds whole is
221/// its own.
222fn events_cross_columns(events: &[Event], (start, end): (u16, u16)) -> bool {
223  // (a separator is no ink of its own: a gap's is made at the block's level, `Canada~~G1V 0A6` in a cell, 1304.3138)
224  let ink = |e: &Event| {
225    !matches!(
226      e.kind,
227      EventKind::Layout | EventKind::Break | EventKind::Sep(_)
228    ) && e.span.is_some_and(|(s, t)| s >= start && t <= end)
229  };
230  let (Some(first), Some(last)) = (events.iter().position(ink), events.iter().rposition(ink))
231  else {
232    return false;
233  };
234  let Some(nest) = events[first..=last]
235    .iter()
236    .filter(|e| ink(e))
237    .map(|e| e.nest)
238    .min()
239  else {
240    return false;
241  };
242  let column = |e: &Event| {
243    (e.kind == EventKind::Layout && e.role.as_deref() == Some("cell"))
244      .then(|| e.label.as_deref()?.split_once(',')?.1.parse::<usize>().ok())
245      .flatten()
246  };
247  (first..last).any(|at| {
248    let e = &events[at];
249    if e.nest >= nest {
250      return false;
251    }
252    match e.kind {
253      EventKind::Layout => column(e).is_some_and(|c| c > 0),
254      // (the row's last cell: the event before the row end at its alignment's nest)
255      EventKind::Break if e.role.as_deref() == Some("row") => events[..at]
256        .iter()
257        .rev()
258        .find(|cell| cell.nest <= e.nest)
259        .and_then(column)
260        .is_some_and(|c| c > 0),
261      _ => false,
262    }
263  })
