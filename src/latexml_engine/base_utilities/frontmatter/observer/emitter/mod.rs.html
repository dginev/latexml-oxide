39/// The content invariant's check on the placed output (`resolver::placed`; the safety net's bar (2)), the last step of
40/// `insert_frontmatter`: the words the ledger's author blocks and creator calls printed
41/// against the placed creators' (their names, contacts and notes, as [`dom::read`] reads them; a publication note's, a title's
42/// note's and a contact's label holding a word too), each finding counted in one Warning per paper naming its words — a printed word or
43/// address the output lacks (1605.03019's `IDSIA`), one placed more often than printed (2607.00684's `mtablo@postech.ac.kr`
44/// twice, 2506.02663's footnote junk). It counts; it changes nothing.
45pub(in crate::base_utilities) fn check_placed_content(document: &mut Document) {
46  if rehearsal::rehearsing() {
47    return;
48  }
49  let front = dom::read(document);
50  let mut placed = Placed::default();
51  // (a link's target, which an ink may print: an ORCID's id behind its logo, 2607.00684; each target once, a shared
52  // entry's link under every holder one print)
53  let mut targets = BTreeSet::new();
54  for (holder, creator) in front.creators.iter().chain(&front.nameless).enumerate() {
55    let mut links = Vec::new();
56    dom::descendants(&creator.node, "ltx:ref", &mut links);
57    targets.extend(links.iter().filter_map(|link| link.get_attribute("href")));
58    if !creator.name.trim().is_empty() {
59      // (its words parted at every layout part, as the block's lines print them: a block written whole, 1908.02923; read again
60      // only where a layout part holds them)
61      let names = creator.node.get_child_nodes().into_iter();
62      let names: Vec<Node> = names
63        .filter(|child| dom::is(child, "ltx:personname"))
64        .collect();
65      let text = if names.iter().any(dom::holds_a_layout_part) {
66        let words: Vec<String> = names.iter().map(dom::words_without_marks).collect();
67        words.join(" ")
68      } else {
69        creator.name.clone()
70      };
71      placed.items.push(PlacedItem {
72        holder,
73        kind: PlacedKind::Name,
74        text,
75        label: None,
76      });
77    }
78    // (what a personname holds besides its notes, an ORCID link, its name's text holds)
79    for item in creator
80      .items
81      .iter()
82      .filter(|item| item.movable || item.kind == dom::Kind::Note)
83    {
84      // (one shared item is one kind's under its text: a note on one creator and an affiliation on another printing the same
85      // words are two, each printed)
86      let kind = match &item.kind {
87        dom::Kind::Contact(role) if role == "email" => PlacedKind::Email(item.key.clone()),
88        kind => PlacedKind::Item(s!("{kind:?}|{}", item.key)),
89      };
90      placed.items.push(PlacedItem {
91        holder,
92        kind,
93        text: item.text.clone(),
94        label: item.node.get_attribute("name"),
95      });
96    }
97  }
98  placed.beside.extend(targets);
99  // (a creator call's note relocated to the title, elsarticle's `\tnotetext[t1]{…}` labelled `tnote:t1`, 1611.08809)
100  for note in document.findnodes("//ltx:pubnote | //ltx:title//ltx:note", None) {
101    placed.beside.push(dom::text_without_marks(&note));
102  }
103  // (each publication note as written: one written more often than it printed is counted, review rck2)
104  placed.notes = document
105    .findnodes("//ltx:pubnote", None)
106    .iter()
107    .map(dom::text_without_marks)
108    .collect();
109  placed.dropped = emit::reading_drops();
110  // (the call labels of the placed items: the entries the writer cut that are placed, `retold_words`)
111  let placed_calls: BTreeSet<String> = front
112    .creators
113    .iter()
114    .chain(&front.nameless)
115    .flat_map(|creator| &creator.items)
116    .filter_map(|item| item.node.get_attribute("_label"))
117    .flat_map(|labels| {
118      labels
119        .split([',', ' '])
120        .map(str::to_string)
121        .collect::<Vec<_>>()
122    })
123    .collect();
124  placed.retold = with_ledger(|ledger| retold_words(&ledger.labelled, &placed_calls));
125  let check = with_ledger(|ledger| content_check(ledger, &placed, &emit::kept_whole));
126  for (words, what) in [
127    (
128      &check.lost,
129      "words the author blocks print are missing from the placed frontmatter",
130    ),
131    (
132      &check.excess,
133      "words or addresses appear in the placed frontmatter more often than the author blocks print them",
134    ),
135  ] {
136    if !words.is_empty() {
137      Warn!(
138        "unexpected",
139        "frontmatter",
140        s!(
141          "Frontmatter content check: {} {what}: {}.",
142          words.len(),
143          words.join(" ")
144        )
145      );
146    }
147  }
148  // (a note printing no word, which the word check can't see: written nowhere, `emit::unwritten_notes`)
149  let notes = emit::unwritten_notes();
150  if !notes.is_empty() {
151    Warn!(
152      "unexpected",
153      "frontmatter",
154      s!(
155        "Frontmatter content check: {} notes the author blocks print with no word are missing from the placed frontmatter: {}.",
156        notes.len(),
157        notes.join(" ")
158      )
159    );
160  }
161}

163/// The entry `taken` given back in the reading `out`: its addresses, taken out, no contacts of its holders.
164pub(super) fn give_back_entry(out: &mut Resolved, taken: &resolver::Taken) {
165  for creator in &mut out.creators {
166    creator
167      .contacts
168      .retain(|a| !(a.source == taken.source && a.rule.contains("owner:entry-address")));
169  }
170  for (holder, at, assoc) in &taken.holders {
171    let slots = match holder {
172      Some(c) => &mut out.creators[*c].affiliations,
173      None => &mut out.block,
174    };
175    match slots.iter().position(|a| {
176      a.source == taken.source
177        && a.text == taken.stripped
178        && a.rule.ends_with(" ruling3:address-out")
179    }) {
180      Some(i) => slots[i] = assoc.clone(),
181      None => slots.insert((*at).min(slots.len()), assoc.clone()),
182    }
183  }
