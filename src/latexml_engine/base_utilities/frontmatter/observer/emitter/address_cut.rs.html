1//! An entry the reading took printed addresses out of (`Resolved::taken`, `Resolver::take_out_entry_addresses`), as the
2//! emitter writes it (OXIDIZED_DESIGN_DIVERGENCES #980 P1b): the old driver's writer's cut of
3//! its XML (deleted with it) (`plan::cut_entry_addresses`, `dom::address_cut`, `Op::CutAddresses`, `Op::Address`), translated onto the
4//! entry's tokens. The reading's runs and cut are bytes of its text; the lexer recorded where each character of an ink
5//! printing an `@` prints (`events::InkTable`, from its box's stamp), so each byte the cut takes is a position of the field,
6//! and the entry is its own tokens without the runs those positions cover (`Field::content_cut`). As the DOM cut: what
7//! prints nothing between a separator the cut takes and the ink beside it — a line break, a mark (`$^*\,$a@x`, a whole line
8//! with its mark: 2506.02272) — goes with it; a cut that would divide a box or take a note declines; the cut is the entry's
9//! own, so it declines entry by entry, never the paper (the old driver's `declined`). Each address is its owner's e-mail,
10//! as printed: its own tokens, in their font or wrapper (`\texttt{a@x}`, `{\tt\small a@x; b@x}`), else its text in its
11//! ink's font, named by the run's label word (`E-mail: `). Witnesses: ieeeconf's `\thanks{… {\tt\small a@x; b@x}}`
12//! (2011.02560, 2104.05078, 2204.02371), svmult's `\institute{… \texttt{a@x}}` (astro-ph0610446, cond-mat0610708),
13//! authblk's `\affil[1]{\texttt{\{a,b\}@x}, Univ}` (2406.02913), IEEEtran's `… - Email: a@x / b@x` (1705.06453).
14

32/// One e-mail a cut made: its address, its tokens as printed, the label its run took.
33type Mail = (String, Vec<Token>, Option<String>);

35/// The cut of one taken entry ([`entry_cut`]).
36#[derive(Debug)]
37pub(super) struct EntryCut {
38  source:       Option<(usize, Option<(u16, u16)>)>,
39  pub original: String,
40  stripped:     String,
41  /// the runs the entry's tokens lose: indices of the block's tokens (a block's entry, a note's), or of the call's content
42  pub cuts:     Vec<(usize, usize)>,
43  /// an API call's command, its content cut
44  pub command:  Option<Tokens>,
45  /// each address: its tokens as printed, and the label word its run took
46  mails:        Vec<Mail>,
47  /// the entry was written cut (a path that writes it otherwise leaves it unset: written by its text, counted)
48  pub applied:  Cell<bool>,
49}

51impl EntryCut {
52  /// Whether `assoc` is this cut's entry as the reading keeps it, its addresses out.
53  pub(super) fn is_cut_of(&self, assoc: &Assoc) -> bool {
54    self.source == assoc.source && self.stripped == assoc.text
55  }
56
57  /// Whether `assoc` is this cut's entry as it printed, before the take-out.
58  pub(super) fn is_entry(&self, assoc: &Assoc) -> bool {
59    self.source == assoc.source && self.original == assoc.text
60  }
61
62  /// The e-mail `address` of `assoc`'s source this cut made: its tokens, and the `name=` its label gives (`E-mail: `, as the
63  /// old driver's `email_contact` names it).
64  pub(super) fn mail(&self, assoc: &Assoc) -> Option<(Vec<Token>, Option<String>)> {
65    if self.source != assoc.source {
66      return None;
67    }
68    let (_, tokens, label) = self.mails.iter().find(|(a, ..)| *a == assoc.text)?;
69    Some((tokens.clone(), label.as_ref().map(|l| s!("{l} "))))
70  }
71}

73/// The cut of the entry `taken`, `None` where its tokens can't be cut at the positions its address runs print at: given back
74/// whole, the Kind steps' after the flush, as before.
75fn entry_cut(taken: &Taken, names: &[&str]) -> Result<Option<EntryCut>> {
76  let spells = &spells_one_of(names);
77  let Some(source) = taken.source else {
78    return Ok(None);
79  };
80  let made = match source.1 {
81    Some(span) => with_ledger(|ledger| {
82      let Some(Entry::Author(block)) = ledger.entries().get(source.0) else {
83        return None;
84      };
85      let Outcome::Events {
86        events,
87        formulas,
88        environments,
89        inks,
90        held,
91        ..
92      } = &block.outcome
93      else {
94        return None;
95      };
96      let field = Field {
97        tokens: block.read_tokens().unlist_ref(),
98        base: block.range.0,
99        formulas,
100        environments,
101        held,
102        aligned: aligned(events),
103        groups: Default::default(),
104      };
105      // (a note's entry is its text; any other the text of the events its span covers, or of those from it on, the lines the
106      // reading read on with it: 2308.10621)
107      let note = events
108        .iter()
109        .find(|e| e.kind == EventKind::Note && e.span == Some(span));
110      match note {
111        Some(note) => {
112          let texts = [(note.text.clone(), inks.of(note)?.to_vec())];
113          // (the inks its body prints, whose fonts its addresses print in: the note's own is the text's around it)
114          let inner: Vec<&Event> = note.inner.iter().collect();
115          let around = Around {
116            inks:     &inner,
117            stops:    events
118              .iter()
119              .chain(&note.inner)
120              .filter(|e| stops(e) && e.span != Some(span))
121              .collect(),
122            apart:    Vec::new(),
123            marks:    note
124              .inner
125              .iter()
126              .filter(|e| e.kind == EventKind::Mark)
127              .collect(),
128            contacts: note.inner.iter().filter(|e| contact_command(e)).collect(),
129          };
130          cut_at(&field, &texts, &around, inks, (taken, spells))
131        },
132        None => {
133          let from: Vec<&Event> = events
134            .iter()
135            .filter(|e| e.span.is_some_and(|s| s.0 >= span.0))
136            .collect();
137          let within: Vec<&Event> = from
138            .iter()
139            .copied()
140            .filter(|e| e.span.is_some_and(|s| s.1 <= span.1))
141            .collect();
142          let [text, lined] = events_texts_at(&within, inks);
143          let [_, read_on] = events_texts_at(&from, inks);
144          let around = Around {
145            inks:     &from,
146            stops:    events.iter().filter(|e| stops(e)).collect(),
147            apart:    Vec::new(),
148            marks:    events
149              .iter()
150              .filter(|e| e.kind == EventKind::Mark)
151              .collect(),
152            contacts: events.iter().filter(|e| contact_command(e)).collect(),
153          };
154          cut_at(
155            &field,
156            &[text, lined, read_on],
157            &around,
158            inks,
159            (taken, spells),
160          )
161        },
162      }
163    })
164    .map(|(cuts, mails)| (cuts, None, mails)),
165    None => api_cut(source.0, (taken, spells))?,
166  };
167  if made.is_some() {
168    DebugFeature!(
169      "frontmatter",
170      "FRONT emit: an entry cut: {:?} -> {:?}",
171      taken.lined,
172      taken.stripped
173    );
174  }
175  Ok(made.map(|(cuts, command, mails)| EntryCut {
176    source: taken.source,
177    original: taken.original.clone(),
178    stripped: taken.stripped.clone(),
179    cuts,
180    command,
181    mails,
182    applied: Cell::new(false),
183  }))
184}

189/// An API call's entry: its command's last argument rehearsed with its positions (the call's own, from 0), its text without
190/// the contacts the reading reads apart (an `\email`, an ORCID: `Resolver::api`), cut, in the command.
191fn api_cut(index: usize, (taken, spells): (&Taken, &dyn Fn(&str) -> bool)) -> Result<Option<Made>> {
192  let Some(call) = api_call(index) else {
193    return Ok(None);
194  };
195  let Some(command) = call.command.clone() else {
196    return Ok(None);
197  };
198  let Some((before, content, after)) = last_argument(&command) else {
199    return Ok(None);
200  };
201  let Some((events, formulas, environments, inks)) = rehearsed(&content)? else {
202    return Ok(None);
203  };
204  let field = Field {
205    tokens:       &content,
206    base:         0,
207    formulas:     &formulas,
208    environments: &environments,
209    held:         &[],
210    aligned:      aligned(&events),
211    groups:       Default::default(),
212  };
213  let apart = |e: &Event| {
214    e.kind == EventKind::Contact && matches!(e.role.as_deref(), Some("email" | "orcid"))
215  };
216  let refs: Vec<&Event> = events.iter().filter(|e| !apart(e)).collect();
217  let around = Around {
218    inks:     &refs,
219    stops:    events.iter().filter(|e| stops(e)).collect(),
220    apart:    events.iter().filter(|e| apart(e)).collect(),
221    marks:    events
222      .iter()
223      .filter(|e| e.kind == EventKind::Mark)
224      .collect(),
225    contacts: events.iter().filter(|e| contact_command(e)).collect(),
226  };
227  let Some((cuts, mails)) = cut_at(
228    &field,
229    &events_texts_at(&refs, &inks),
230    &around,
231    &inks,
232    (taken, spells),
233  ) else {
234    return Ok(None);
235  };
236  let Some((cut, _)) = field.content_cut((0, content.len()), 0, 0, &cuts) else {
237    return Ok(None);
238  };
239  let mut tokens = before;
240  tokens.extend(cut.unlist());
241  tokens.extend(after);
242  Ok(Some((cuts, Some(Tokens::new(tokens)), mails)))
243}

245/// Whether `e` is a contact a command prints whose call an e-mail's item would mute (an `\\email`, an ORCID), no link: a
246/// `\\href{mailto:a@x}{a@x}` (a `url` contact) prints inline, the e-mail's own wrapper (2608.15371).
247fn contact_command(e: &Event) -> bool {
248  e.kind == EventKind::Contact && e.role.as_deref() != Some("url")
249}

251/// Whether a cut may not take `e`: a note (the DOM cut's `Seg::Note`), a line's break.

254/// `None`, the cut declined for `why` (`--debug=frontmatter`).

257  None

260/// The span a character prints at: its box's cover with its own token.
261fn reach(ink: &InkAt) -> (usize, usize) {
262  let (s, t) = (usize::from(ink.cover.0), usize::from(ink.cover.1));
263  match ink.own.map(usize::from) {
264    Some(own) => (s.min(own), t.max(own + 1)),
265    None => (s, t),
266  }
267}
268
269/// Whether a kept character prints inside `range`: its own token, or its whole cover where it has none.
270fn prints_in(ink: &InkAt, (s, t): (usize, usize)) -> bool {
271  match ink.own.map(usize::from) {
272    Some(own) => s <= own && own < t,
273    None => s <= usize::from(ink.cover.0) && usize::from(ink.cover.1) <= t,
274  }
275}

277/// The events of an entry's source a cut reads: those printing it (`inks`, for their fonts), those it may not take (`stops`: a
278/// note; a line's break, but the one between the lines a cut takes whole), and the contacts the text leaves out, which a cut is
279/// parted around (`apart`).
280struct Around<'a> {
281  inks:     &'a [&'a Event],
282  stops:    Vec<&'a Event>,
283  apart:    Vec<&'a Event>,
284  /// the marks the source prints, which no cut takes past the separator it takes ([`cut_at`])
285  marks:    Vec<&'a Event>,
286  /// the contacts it prints, whose command no e-mail is written as (its item would mute the call: 1109.3369's
287  /// `Email:~$^a$\email{…}`)
288  contacts: Vec<&'a Event>,
289}

291/// The cut of `taken` in the field, where one of `texts` (each with where its characters print) holds its text once: each
292/// cut of the reading (`Taken::cuts`, its bytes) the positions its characters print at, grown over what prints nothing between
293/// a separator it takes and the ink beside it, parted around a contact read apart, each part a balanced run; and each
294/// address's tokens. `None` where none holds it, a cut takes a kept character's token, a note or a line's break, or no
295/// balanced run holds it.
296fn cut_at(
297  field: &Field,
298  texts: &[TextAt],
299  around: &Around,
300  inks: &InkTable,
301  (taken, spells): (&Taken, &dyn Fn(&str) -> bool),
302) -> Option<CutAt> {
303  let lined = taken.lined.as_str();
304  let Some(((text, at), map)) = texts.iter().find_map(|t| {
305    (t.1.len() == t.0.chars().count())
306      .then(|| locate(&t.0, lined))
307      .flatten()
308      .map(|map| (t, map))
309  }) else {
310    DebugFeature!(
311      "frontmatter",
312      "FRONT emit: the reading's {lined:?}, its events' {:?}",
313      texts.iter().map(|t| &t.0).collect::<Vec<_>>()
314    );
315    return declined("no text of its events holds the reading's once");
316  };
317  // (each character of the entry's text where it prints, and whether the cut takes it; what the text holds besides, kept)
318  let chars: Vec<(usize, char)> = lined.char_indices().collect();
319  let ink: Vec<Option<InkAt>> = map.iter().map(|j| j.and_then(|j| at[j])).collect();
320  let taken_at = |byte: usize| taken.cuts.iter().any(|&(s, t)| s <= byte && byte < t);
321  let base = field.base;
322  let index = |p: usize| p.checked_sub(base);
323  let cut_chars: Vec<usize> = (0..chars.len())
324    .filter(|&i| taken_at(chars[i].0))
325    .filter_map(|i| map[i])
326    .collect();
327  let kept: Vec<InkAt> = (0..at.len())
328    .filter(|j| !cut_chars.contains(j))
329    .filter_map(|j| at[j])
330    .collect();
331  // (the runs the text prints, its spaces as printed — the old driver's `mailed_runs` on today's layout: each one the cut
332  // touches every address of which the reading takes, else no cut; `kwb,flynn,walter.scheirer@nd.edu` is one run of three, the
333  // reading's text, spaced after each comma, one address after prose: 1903.10019)
334  let letters: Vec<char> = text.chars().collect();
335  let mut printed = String::new();
336  let mut printed_at: Vec<usize> = Vec::new();
337  // (the entry's own stretch of the text: a later entry the lines read on hold is another item)
338  let first = map.iter().flatten().min().copied().unwrap_or(0);
339  let last = map.iter().flatten().max().copied().unwrap_or(0);
340  for (j, &c) in letters.iter().enumerate().take(last + 1).skip(first) {
341    let glued = || {
342      let (Some(Some(before)), Some(Some(after))) =
343        (j.checked_sub(1).map(|b| at[b]), at.get(j + 1))
344      else {
345        return false;
346      };
347      reach(&before).1 >= reach(after).0
348    };
349    if at[j].is_none() && c == ' ' && glued() {
350      continue;
351    }
352    printed_at.extend(std::iter::repeat_n(j, c.len_utf8()));
353    printed.push(c);
354  }
355  let mailed = |a: &String| {
356    taken
357      .runs
358      .iter()
359      .flat_map(|run| &run.addresses)
360      .any(|m| m.eq_ignore_ascii_case(a))
361  };
362  for run in address_runs(&printed, spells) {
363    let touched = printed_at
364      .get(run.start..run.end)
365      .is_some_and(|js| js.iter().any(|j| cut_chars.contains(j)));
366    if touched && !run.addresses.iter().all(mailed) {
367      return declined("a printed run not every address of which the reading takes");
368    }
369    // (or glued to a list's comma before it, a local part the reading leaves — `kwb,flynn,walter.scheirer@nd.edu`, a dotted
370    // part ending no list (`ends_a_listed_run`): the old driver's block reader reads the line as the list and its writer
371    // declines the paper, 1903.10019; here the entry alone, given back to the Kind steps. A word that is no local part
372    // before it is the entry's, `México, D.F.,octavio@astro.unam.mx`, 1310.7021)
373    let local = |c: char| c.is_ascii_lowercase() || c.is_ascii_digit() || "._-".contains(c);
374    if touched
375      && let Some(before) = printed[..run.start].strip_suffix(',')
376      && before
377        .rsplit(|c: char| c.is_whitespace() || c == ',')
378        .next()
379        .is_some_and(|part| !part.is_empty() && part.chars().all(local))
380    {
381      return declined("a run glued to a list's comma");
382    }
383  }
384  let span_of = |e: &Event| e.span.map(|(a, b)| (usize::from(a), usize::from(b)));
385  // (where a mark prints: its formula, else its span)
386  let mark_of = |e: &Event| {
387    e.formula
388      .and_then(|f| f.span)
389      .or(e.span)
390      .map(|(a, b)| (usize::from(a), usize::from(b)))
391  };
392  // (a position something kept prints at, a stop or a mark: what no cut grows over)
393  let prints = |p: usize| {
394    kept.iter().any(|ink| prints_in(ink, (p, p + 1)))
395      || around
396        .stops
397        .iter()
398        .chain(&around.apart)
399        .filter_map(|e| span_of(e))
400        .chain(around.marks.iter().filter_map(|e| mark_of(e)))
401        .any(|(a, b)| a <= p && p < b)
402  };
403  let mut cuts: Vec<(usize, usize)> = Vec::new();
404  for &(start, end) in &taken.cuts {
405    let inside: Vec<usize> = (0..chars.len())
406      .filter(|&i| start <= chars[i].0 && chars[i].0 < end)
407      .collect();
408    let (Some(&first), Some(&last)) = (inside.first(), inside.last()) else {
409      return declined("a cut of no character");
410    };
411    let reaches: Vec<(usize, usize)> = inside
412      .iter()
413      .filter_map(|&i| ink[i].as_ref().map(reach))
414      .collect();
415    let (Some(mut from), Some(mut to)) = (
416      reaches.iter().map(|r| r.0).min(),
417      reaches.iter().map(|r| r.1).max(),
418    ) else {
419      return declined("a cut printing at no position");
420    };
421    let own = (from, to);
422    // (a separator the cut takes at an end takes what prints nothing up to the kept ink there: the line's break, a mark)
423    if ink[first].is_none()
424      && let Some(before) = (0..first)
425        .rev()
426        .find_map(|i| ink[i].filter(|_| !taken_at(chars[i].0)))
427    {
428      from = from.min(reach(&before).1);
429    }
430    // (but never past a mark: one before the kept ink after it leads that ink, `Univ X, {\tt a@x}, $^1$Dept`, as the Kind
431    // step leaves it — a mark leading the address side, `\\$^*\,$a@x`, goes with the cut: 2506.02272; review revP1b S1)
432    if ink[last].is_none()
433      && let Some(after) =
434        (last + 1..chars.len()).find_map(|i| ink[i].filter(|_| !taken_at(chars[i].0)))
435    {
436      let ink_at = reach(&after).0;
437      let mark = around
438        .marks
439        .iter()
440        .filter_map(|e| mark_of(e))
441        .map(|(a, _)| a)
442        .filter(|&a| a >= to && a < ink_at)
443        .min();
444      to = to.max(mark.unwrap_or(ink_at));
445    }
446    // (parted around a contact the text leaves out, which stays where it prints: `(\email{a@x}; b@x)`, 1106.3045)
447    let mut parts = vec![(from, to)];
448    for (a, b) in around.apart.iter().filter_map(|e| span_of(e)) {
449      parts = parts
450        .into_iter()
451        .flat_map(|(s, t)| {
452          if a >= s && b <= t {
453            vec![(s, a), (b, t)]
454          } else {
455            vec![(s, t)]
456          }
457        })
458        .filter(|(s, t)| s < t)
459        .collect();
460    }
461    // (what prints nothing a separator took at an end, as few tokens of it as a balanced run needs: the brace closing the
462    // kept ink's group before the break a whole line's cut takes, `\textit{… China}\\ \textit{*Corresponding author: a@x,
463    // b@x}`, 2405.13751)
464    for (from, to) in parts {
465      let starts = from..=from.max(own.0.min(to));
466      let Some(run) = starts
467        .take(8)
468        .flat_map(|s| {
469          (s.max(own.1.min(to))..=to)
470            .rev()
471            .take(8)
472            .map(move |t| (s, t))
473        })
474        .find_map(|(from, to)| {
475          let (from, to) = (index(from)?, index(to)?);
476          grown(field, from, to, &|i| prints(i + base))
477        })
478      else {
479        return declined("no balanced run holds a cut");
480      };
481      cuts.push(run);
482    }
483  }
484  cuts.sort_unstable();
485  cuts.dedup();
486  let overlaps =
487    |e: &Event, (s, t): (usize, usize)| span_of(e).is_some_and(|(a, b)| a < t && b > s);
488  for (s, t) in cuts.iter().map(|&(s, t)| (s + base, t + base)) {
489    if kept.iter().any(|ink| prints_in(ink, (s, t))) {
490      return declined("a cut takes a kept character's token");
491    }
492    if around
493      .stops
494      .iter()
495      .any(|e| e.kind == EventKind::Note && overlaps(e, (s, t)))
496    {
497      return declined("a cut takes a note");
498    }
499  }
500  // (each address as printed — RT1 row 56, the old driver's wrapper (`dom::address_cut`): the call wrapping it alone, whole, its
501  // cover's tokens where nothing else prints there and the cut takes it — `\href{mailto:a@x}{a@x}` (2608.15371), `\textit{a@x}`
502  // —; else its characters' own tokens, with the switches of the group the run's first address opens (`{\tt\small a@x; b@x}`'s
503  // second), else in the font of the ink printing it — `{\it Contact e-mail: a@x}` (2008.08137), `\texttt{a@x, b@y}`'s each
504  // (2608.14026), a `\thanks` body's (its inner inks: 2405.06709); a list's written-out address, which prints no tokens of its
505  // own, its text in that font (RP1's `EntryCut::mail`: `\texttt{\{a,b\}@x}`; one a contact command prints, whose call the
506  // e-mail's item would mute, 1109.3369's `Email:~$^a$\email{…}`))
507  let mut mails = Vec::new();
508  let reach_of = |from: usize, to: usize| {
509    let reaches: Vec<(usize, usize)> = (0..chars.len())
510      .filter(|&i| from <= chars[i].0 && chars[i].0 < to)
511      .filter_map(|i| ink[i].as_ref().map(reach))
512      .collect();
513    Some((
514      reaches.iter().map(|r| r.0).min()?,
515      reaches.iter().map(|r| r.1).max()?,
516    ))
517  };
518  // (whether an event prints inside `range`: what a wrapper taken whole would take with the address — a contact command its
519  // item would mute among them)
520  let holds_event = |(s, t): (usize, usize)| {
521    around
522      .stops
523      .iter()
524      .chain(&around.apart)
525      .chain(&around.contacts)
526      .filter_map(|e| span_of(e))
527      .chain(around.marks.iter().filter_map(|e| mark_of(e)))
528      .any(|(a, b)| a < t && b > s)
529  };
530  // (whether a formula the field prints overlaps `range`)
531  let in_a_formula = |(s, t): (usize, usize)| {
532    field
533      .formulas
534      .iter()
535      .any(|f| usize::from(f.start) < t && usize::from(f.end) > s)
536  };
537  for run in &taken.runs {
538    let run_at = reach_of(run.bytes.0, run.bytes.1);
539    let bytes_of = |address: &str| {
540      lined
541        .get(run.bytes.0..run.bytes.1)
542        .and_then(|r| r.find(address))
543        .map(|at| (run.bytes.0 + at, run.bytes.0 + at + address.len()))
544    };
545    // (where the run's first address prints: the switches before it are its font's, a label or a bracket before it none)
546    let first_at = run
547      .addresses
548      .iter()
549      .filter_map(|a| bytes_of(a))
550      .min()
551      .and_then(|(s, t)| reach_of(s, t));
552    for address in &run.addresses {
553      let bytes = bytes_of(address);
554      // (the font its `@` prints in (`InkTable::fonts`: a font switched inside the ink, `\thanks{… \texttt{E-Mail: a@x}}`,
555      // 2405.06709), else the ink's printing it)
556      let at_font = bytes
557        .and_then(|(s, t)| {
558          (0..chars.len()).find(|&i| s <= chars[i].0 && chars[i].0 < t && chars[i].1 == '@')
559        })
560        .or_else(|| {
561          (0..chars.len())
562            .find(|&i| run.bytes.0 <= chars[i].0 && chars[i].0 < run.bytes.1 && chars[i].1 == '@')
563        })
564        .and_then(|i| ink[i])
565        .and_then(|ink| inks.font_at(ink.own.unwrap_or(ink.cover.0)))
566        .cloned();
567      let font = || {
568        at_font.clone().or_else(|| {
569          around
570            .inks
571            .iter()
572            .filter(|e| e.font.is_some() && e.text.contains('@'))
573            .find(|e| e.text.contains(address.as_str()) || run_at.is_some_and(|r| overlaps(e, r)))
574            .and_then(|e| e.font.clone())
575        })
576      };
577      let printed = bytes.zip(first_at.or(run_at)).and_then(|((s, t), run_at)| {
578        let mine: Vec<usize> = (0..chars.len())
579          .filter(|&i| s <= chars[i].0 && chars[i].0 < t)
580          .collect();
581        mine.iter().all(|&i| ink[i].is_some()).then_some(())?;
582        let owns: Option<Vec<usize>> = mine
583          .iter()
584          .map(|&i| ink[i].and_then(|ink| ink.own).map(usize::from))
585          .collect();
586        let own = owns.and_then(|owns| Some((*owns.iter().min()?, owns.iter().max()? + 1)));
587        let range = reach_of(s, t)?;
588        let others_print = |range: (usize, usize)| {
589          (0..chars.len())
590            .filter(|i| !mine.contains(i))
591            .filter_map(|i| ink[i])
592            .any(|ink| prints_in(&ink, range))
593        };
594        let (from, to, first) = (index(range.0)?, index(range.1)?, index(run_at.0)?);
595        let taken_whole = cuts.iter().any(|&(s, t)| s <= from && to <= t);
596        if Some(range) != own
597          && taken_whole
598          && !others_print(range)
599          && !holds_event(range)
600          && !in_a_formula(range)
601        {
602          let (tokens, _) = field.content_of((from, to), 0, from)?;
603          return Some((tokens.unlist(), "its wrapper"));
604        }
605        let own = own?;
606        // (never its printed formula, `dcook12$@$uwyo.edu` (1409.0847): its text, as the old driver wrote it)
607        if others_print(own) || in_a_formula(own) {
608          return None;
609        }
610        let (own_from, own_to) = (index(own.0)?, index(own.1)?);
611        let (tokens, _) = field.content_of((own_from, own_to), 0, first.min(own_from))?;
612        let tokens = tokens.unlist();
613        Some(if tokens.len() > own_to - own_from {
614          (tokens, "its group's switches")
615        } else {
616          (in_its_font(font(), tokens).unlist(), "its ink's font")
617        })
618      });
619      let (tokens, how) = printed.unwrap_or_else(|| {
620        (
621          in_its_font(font(), printed_text_tokens(address)).unlist(),
622          "its text",
623        )
624      });
625      DebugFeature!(
626        "frontmatter",
627        "FRONT emit: a cut address {address:?}: {how}"
628      );
629      mails.push((address.clone(), tokens, run.label.clone()));
630    }
631  }
632  Some((cuts, mails))
633}

635/// Where each character of the reading's `lined` text is in `text` (an index of its characters; `None` for a line's end):
636/// `lined` once in it, or each of its lines once in a line of it, in order — the reading's lines are an entry's, trimmed of
637/// the separator a line of the block ends in (`Dept. of X, Univ Y,\\ Town`, 1203.0253, 2405.03617).
638fn locate(text: &str, lined: &str) -> Option<Vec<Option<usize>>> {
639  let once = |hay: &str, needle: &str| {
640    hay
641      .find(needle)
642      .filter(|&at| hay.rfind(needle) == Some(at))
643      .map(|at| hay[..at].chars().count())
644  };
645  let mut map = Vec::new();
646  if !lined.contains('\n') {
647    let at = once(text, lined)?;
648    map.extend((0..lined.chars().count()).map(|i| Some(at + i)));
649    return Some(map);
650  }
651  let rows: Vec<(usize, &str)> = {
652    let mut from = 0;
653    text
654      .split('\n')
655      .map(|row| {
656        let start = from;
657        from += row.chars().count() + 1;
658        (start, row)
659      })
660      .collect()
661  };
662  let lines: Vec<&str> = lined.split('\n').collect();
663  let fits = |first: usize| {
664    lines.iter().enumerate().all(|(k, line)| {
665      rows
666        .get(first + k)
667        .is_some_and(|(_, row)| once(row, line).is_some())
668    })
669  };
670  let mut starts = (0..rows.len()).filter(|&first| fits(first));
671  let (Some(first), None) = (starts.next(), starts.next()) else {
672    return None;
673  };
674  for (k, line) in lines.iter().enumerate() {
675    if k > 0 {
676      map.push(None);
677    }
678    let (start, row) = rows[first + k];
679    let at = start + once(row, line)?;
680    map.extend((0..line.chars().count()).map(|i| Some(at + i)));
681  }
682  Some(map)
683}

688/// The run `from..to` (indices) grown to a balanced one ([`Field::balanced`]), over a few tokens before or after it that print
689/// nothing (`prints`): the switches of the group it starts in (`{\tt(sluys@astro.uu.nl)}`, astro-ph0610492), the space before
690/// a group's closing brace (`{\tt\small a@x }`).
691fn grown(
692  field: &Field,
693  from: usize,
694  to: usize,
695  prints: &dyn Fn(usize) -> bool,
696) -> Option<(usize, usize)> {
697  let back = (1..=4)
698    .take_while(|&k| from >= k && !prints(from - k))
699    .count();
700  let on = (0..4)
701    .take_while(|&k| to + k < field.tokens.len() && !prints(to + k))
702    .count();
703  (0..=back)
704    .flat_map(|b| (0..=on).map(move |f| (b, f)))
705    .find_map(|(b, f)| field.balanced(from - b, to + f, 0))
706}

709pub(super) fn span_tokens_cut(

712  cuts: &[(usize, usize)],
713) -> Option<Tokens> {
714  span_tokens_within(index, span, None, cuts)

717/// [`span_tokens_cut`] within the span `within` of the block (a note's, `\thanks{…}`): its tokens and the formulas inside it
718/// alone, `None` where no balanced run of them holds `span`. A `\thanks` argument is typeset in the footnote (latex.ltx
719/// `\thanks` → `\footnotetext`), never part of the formula around the command, so a piece of it read whole by that formula
720/// is the formula, the `\thanks` itself digested once more (`Ann Able$^{1}$ and Bob Baker$^{2\thanks{$^{1}$Univ A;
721/// $^{2}$Univ B}}$`, review rck2).
722pub(super) fn span_tokens_within(
723  index: usize,
724  span: (u16, u16),
725  within: Option<(u16, u16)>,
726  cuts: &[(usize, usize)],
727) -> Option<Tokens> {
728  with_ledger(|ledger| {
729    let Some(Entry::Author(block)) = ledger.entries().get(index) else {
730      return None;
731    };
732    let Outcome::Events {
733      events,
734      formulas,
735      environments,
736      held,
737      ..
738    } = &block.outcome
739    else {
740      return None;
741    };
742    let at = |p: u16| usize::from(p).checked_sub(block.range.0);
743    let all = block.read_tokens().unlist_ref();
744    let Some((s, e)) = within else {
745      let field = Field {
746        tokens: all,
747        base: block.range.0,
748        formulas,
749        environments,
750        held,
751        aligned: aligned(events),
752        groups: Default::default(),
753      };
754      let start = at(span.0)?;
755      return Some(field.content_cut((start, at(span.1)?), 0, start, cuts)?.0);
756    };
757    let (from, to) = (at(s)?, at(e)?.min(all.len()));
758    // (the formulas inside it: outermost and in order, a run of them)
759    let inside = |f: &Formula| s <= f.start && f.end <= e;
760    let first = formulas.iter().position(inside).unwrap_or(formulas.len());
761    let count = formulas[first..].iter().take_while(|f| inside(f)).count();
762    let field = Field {
763      tokens: all.get(from..to)?,
764      base: block.range.0 + from,
765      formulas: &formulas[first..first + count],
766      environments,
767      held,
768      aligned: aligned(events),
769      groups: Default::default(),
770    };
771    let cuts: Vec<(usize, usize)> = cuts
772      .iter()
773      .filter(|&&(s, t)| s < to && t > from)
774      .map(|&(s, t)| (s.max(from) - from, t.min(to) - from))
775      .collect();
776    let end = at(span.1)?.checked_sub(from)?;
777    // (a line of a table the note sets is that line's: `\thanks{\begin{tabular}{l}$^{1}$Univ A\\ $^{2}$Univ B\end{tabular}}`)
778    let start = field.past_open_environment(at(span.0)?.checked_sub(from)?, end);
779    Some(field.content_cut((start, end), 0, start, &cuts)?.0)
780  })
781}

783/// The call for `assoc`, an entry the reading took addresses out of (its rule ends `ruling3:address-out`), `None` for any other:
784/// the entry as it printed written by `whole` with its cut, where the path it takes cuts it (`EntryCut::applied`), else its text,
785/// counted (`by_text`).
786pub(super) fn cut_entry_call(
787  cuts: &[EntryCut],
788  assoc: &Assoc,
789  whole: impl FnOnce(&Assoc, &EntryCut) -> Result<Tokens>,
790  by_text: impl FnOnce() -> Tokens,
791) -> Option<Result<Tokens>> {
792  let rule = assoc.rule.strip_suffix(" ruling3:address-out")?;
793  let Some(cut) = cuts.iter().find(|cut| cut.is_cut_of(assoc)) else {
794    return Some(Ok(by_text()));
795  };
796  let entry = Assoc {
797    text: cut.original.clone(),
798    rule: rule.to_string(),
799    ..assoc.clone()
800  };
801  cut.applied.set(false);
802  Some(whole(&entry, cut).map(|full| if cut.applied.get() { full } else { by_text() }))
803}

827        );
828        give_back_entry(resolved, taken);
829      },
830    }
831  }
832  cuts
833}

835/// Whether the cut of `taken` would strand the phrase an address it takes completes: a run with no label word the take-out
836/// takes with it (`Taken::runs`, `resolver::labels::take_labels`) after a colon or a "to" the cut keeps, the words before it
837/// on its line, from the last separator, naming a contact (`names_a_contact`) — "Author to whom correspondence should be
838/// addressed: a@x" (2306.04726), "Correspondence to: a@x" (2004.11812), "Email for correspondence: a@x" (2110.05460), "…,
839/// *Corresponding author: \texttt{a@x}" (2602.09324) —: the phrase is the address's own label, which the cut would leave
840/// dangling in the entry, its colon lost. An address after a place's colon (`…, France: a@x`, 1103.3532) strands nothing, nor
841/// does a phrase the entry holds alone: a legend's, its carriers' note (`\affil[*]{Corresponding author: a@x}`, 1812.03816).
842fn strands_a_phrase(taken: &Taken) -> bool {
843  let kept = |at: usize| !taken.cuts.iter().any(|&(s, t)| s <= at && at < t);
844  taken
845    .runs
846    .iter()
847    .filter(|run| run.label.is_none())
848    .any(|run| {
849      let Some(before) = taken.lined.get(..run.bytes.0) else {
850        return false;
851      };
852      let before = before.trim_end_matches(|c: char| c.is_whitespace() || "([<".contains(c));
853      let word = before
854        .rsplit(char::is_whitespace)
855        .next()
856        .unwrap_or_default();
857      let phrase = before.ends_with(':') || word.eq_ignore_ascii_case("to");
858      let phrase_end = before.trim_end_matches(':').len();
859      let head = &before[..phrase_end];
860      let from = head.rfind(['\n', ',', ';', '.']).map_or(0, |p| p + 1);
861      let names = head[from..]
862        .split(|c: char| !c.is_alphabetic() && c != '-')
863        .any(names_a_contact);
864      // (a phrase the entry holds alone, `\affil[*]{Corresponding author: a@x}`, is the legend its carriers' note is made of)
865      let beside = head[..from].chars().any(char::is_alphanumeric);
866      phrase && names && beside && kept(phrase_end - 1)
867    })
868}

870/// Whether the holders of `taken` are exactly the owners of the addresses the reading took out of it: the creators (the block,
871/// `None`) holding an `owner:entry-address` contact from its source. An entry given back whole gives its addresses to its
872/// holders, so one whose address another author owns (2004.11812's last line, "Correspondence to: \texttt{pascal.klink@…}",
873/// Klink's, read onto the Aalto entry only Pajarinen holds) or that a holder shares without owning it (authblk's `\affil[1]`
874/// held by two authors, one address) keeps its cut (the author an address spells owns it, FIDELITY_CHARTER §4).
875fn held_by_its_owners(reading: &Resolved, taken: &Taken) -> bool {
876  let owns = |contacts: &[Assoc]| {
877    contacts
878      .iter()
879      .any(|a| a.source == taken.source && a.rule.contains("owner:entry-address"))
880  };
881  let mut owners: Vec<Option<usize>> = reading
882    .creators
883    .iter()
884    .enumerate()
885    .filter(|(_, c)| owns(&c.contacts))
886    .map(|(i, _)| Some(i))
887    .collect();
888  if owns(&reading.block) {
889    owners.insert(0, None);
890  }
891  let mut holders: Vec<Option<usize>> = taken.holders.iter().map(|(c, ..)| *c).collect();
892  holders.sort_unstable();
893  holders.dedup();
894  !owners.is_empty() && owners == holders
895}
