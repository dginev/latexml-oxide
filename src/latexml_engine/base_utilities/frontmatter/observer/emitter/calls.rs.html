12/// The name a queued author entry holds: the content of its `\lx@personname` (inside `\lx@author@withinst`), not the
13/// `\lx@add@frontmatter@now{ltx:creator}[role=author]` around it, whose words read as surnames ("author" spelled in
14/// `Corresponding author: …`; guard `perfect_kernel_batch64u::osa_style_address_email_line`).
15pub(in crate::base_utilities) fn queued_name(entry: &Tokens) -> Tokens {
16  let tokens = entry.unlist_ref();
17  for cs in ["\\lx@author@withinst", "\\lx@personname"] {
18    if let Some(at) = tokens.iter().position(|t| *t == T_CS!(cs))
19      && tokens
20        .get(at + 1)
21        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
22    {
23      let mut depth = 0usize;
24      for (j, t) in tokens.iter().enumerate().skip(at + 1) {
25        match t.get_catcode() {
26          Catcode::BEGIN => depth += 1,
27          Catcode::END => {
28            depth -= 1;
29            if depth == 0 {
30              return Tokens::new(tokens[at + 2..j].to_vec());
31            }
32          },
33          _ => {},
34        }
35      }
36    }
37  }
38  entry.clone()
39}

41/// A symbol mark's key, made of its glyph (`unanswered_mark_text`: `\dag` and `\dagger` alike, `\ast` and `*` alike)
42/// as hexadecimal code points (`sym-2020`): only letters, digits and `-`, so it passes through a keyval's tokenization
43/// as it is (a `#`, `\#`'s glyph, read back as a parameter token raised `Error:misdefined:#`).
44pub(in crate::base_utilities) fn symbol_mark_key(mark: &str) -> Option<String> {
45  let glyph = unanswered_mark_text(mark)?.replace('\u{2217}', "*");
46  (!glyph.chars().all(char::is_alphanumeric)).then(|| {
47    s!(
48      "sym-{}",
49      glyph
50        .chars()
51        .map(|c| format!("{:x}", c as u32))
52        .collect::<Vec<_>>()
53        .join("-")
54    )
55  })
56}

58/// A class's email command: a macro whose expansion begins with a frontmatter email call — `\lx@add@email`
59/// (JHEP.cls.ltxml's `\email`), or one that decides how to record the address (AASTeX 7's `\lx@add@aas@email`, the
60/// `\lx@add@email@metadata` of an address the PDF does not print).
61pub(in crate::base_utilities) fn is_email_command(cs: &Token) -> Result<bool> {
62  let Some(def) = lookup_definition(cs)? else {
63    return Ok(false);
64  };
65  Ok(
66    matches!(def.get_expansion(), Some(ExpansionBody::Tokens(body))
67    if body.unlist_ref().first().is_some_and(|first| first.with_str(|name| {
68      name.strip_prefix("\\lx@add@").is_some_and(|rest| rest.contains("email"))
69    }))),
70  )
71}

73/// The address of a call of the email command `cmd` (`\email{x}`) as that command sets it: in the font the class's
74/// command gives its address, `\lx@address@urlfont` (llncs's `\email`, `{\tt#1}` at llncs.cls:897; amsart's, JHEP's)
75/// or `\lx@address@textfont` (svjour3's, REVTeX 4.1's), else as written — so a contact the author block makes of the
76/// call (`\institute{… \\ \email{a@x}}`, `E-mail: \email{x}`) is set as the command's own is. Only the command's own
77/// expansion body is read, one level: a command reaching the address macro through another macro is not seen.
78pub(in crate::base_utilities) fn email_command_address(
79  cmd: &Token,
80  address: Tokens,
81) -> Result<Tokens> {
82  let fonts = [
83    T_CS!("\\lx@address@urlfont"),
84    T_CS!("\\lx@address@textfont"),
85  ];
86  let font = match lookup_definition(cmd)?
87    .as_ref()
88    .and_then(|def| def.get_expansion())
89  {
90    Some(ExpansionBody::Tokens(body)) => body
91      .unlist_ref()
92      .iter()
93      .find(|t| fonts.contains(t))
94      .copied(),
95    _ => None,
96  };
97  Ok(match font {
98    Some(font) => Invocation!(font, vec![Some(address)]),
99    None => address,
100  })
101}

103/// The addresses `addrs` of an email `line` as tokens, the line's whole-line wrapper (`\texttt`, `\small`, `\url`)
104/// repeated on each — but not an email command's, the contact itself.
105pub(in crate::base_utilities) fn address_tokens(
106  line: &Tokens,
107  addrs: Vec<String>,
108) -> Result<Vec<Tokens>> {
109  // (a row of addresses the `\quad`s of its join part, `dispatch`'s: each piece as it is written, its own font and link —
110  // 2609.05531's `\texttt{john3678@…} \quad \texttt{johnson.michael@…}`)
111  if let Some(pieces) = address_row_pieces(line) {
112    let mut tokens = Vec::with_capacity(addrs.len());
113    for piece in &pieces {
114      // (a piece whose one address is written in printed braces: `\{hwoo\}@skku.edu`, 2609.18328)
115      let written_out = email_addresses(piece).filter(|addresses| {
116        addresses.len() == 1
117          && piece.unlist_ref().iter().any(|t| {
118            *t == T_CS!("\\{") || *t == T_CS!("\\lbrace") || *t == T_CS!("\\textbraceleft")
119          })
120      });
121      match written_out {
122        Some(addresses) => tokens.extend(address_tokens(piece, addresses)?),
123        None => tokens.extend(bare_email_addresses(piece)?.unwrap_or_default()),
124      }
125    }
126    if tokens.len() == addrs.len() {
127      return Ok(tokens);
128    }
129  }
130  let line_wrapper = whole_line_cs_wrapper(line).map(|(cmd, _)| cmd);
131  let wrapper = line_wrapper.filter(|cmd| !is_email_wrapper(cmd));
132  // (an email command's address set as the command sets it, [`email_command_address`])
133  let email_command = line_wrapper.filter(is_email_wrapper);
134  // (the line's links to a mailbox, by the address they reach: each address keeps its link, `\texttt{\{}\href{mailto:
135  // Xinhao.Zhang@…}{…xinhao.zhang,} \href{mailto:maxime.peyrard@…}{…}\texttt{\}@…}`, 2604.19440)
136  let v = line.unlist_ref();
137  let links: Vec<(String, Vec<Token>)> = (0..v.len())
138    .filter(|at| v[*at] == T_CS!("\\href"))
139    .filter_map(|at| {
140      let open = at + 1 + v[at + 1..].iter().take_while(|t| **t == T_SPACE!()).count();
141      let target = v[open..skip_group(v, open).clamp(open, v.len())].to_vec();
142      let (text, _) = link_texts(&v[at..], 0);
143      let address = text.strip_prefix("mailto:")?.trim().to_lowercase();
144      Some((address, target))
145    })
146    .collect();
147  addrs
148    .into_iter()
149    .map(|addr| {
150      let mut toks = mouth::tokenize(TeXString::assembled(addr.replace('_', "\\_"))).unlist();
151      if let Some(cmd) = wrapper {
152        let mut wrapped = vec![cmd, T_BEGIN!()];
153        wrapped.append(&mut toks);
154        wrapped.push(T_END!());
155        toks = wrapped;
156      }
157      if email_command.is_none()
158        && let Some((_, target)) = links
159          .iter()
160          .find(|(address, _)| *address == addr.to_lowercase())
161      {
162        let mut linked = vec![T_CS!("\\href")];
163        linked.extend(target.iter().copied());
164        linked.push(T_BEGIN!());
165        linked.append(&mut toks);
166        linked.push(T_END!());
167        toks = linked;
168      }
169      match email_command {
170        Some(cmd) => email_command_address(&cmd, Tokens::new(toks)),
171        None => Ok(Tokens::new(toks)),
172      }
173    })
174    .collect()
175}

177/// The pieces of `line` a top-level `\quad` or `\qquad` parts ([`without_row_spacing`] each), `None` unless two or more: a row of
178/// addresses, `dispatch`'s join or a class's own (`\email{\quad a@x \quad\qquad b@y}`, 2601.03839, 2606.13152).
179fn address_row_pieces(line: &Tokens) -> Option<Vec<Tokens>> {
180  let v = line.unlist_ref();
181  let mut pieces = Vec::new();
182  let (mut depth, mut from) = (0usize, 0usize);
183  for (at, t) in v.iter().enumerate() {
184    match t.get_catcode() {
185      Catcode::BEGIN => depth += 1,
186      Catcode::END => depth = depth.saturating_sub(1),
187      _ if depth == 0 && (*t == T_CS!("\\quad") || *t == T_CS!("\\qquad")) => {
188        pieces.push(Tokens::new(without_row_spacing(v[from..at].to_vec())));
189        from = at + 1;
190      },
191      _ => {},
192    }
193  }
194  pieces.push(Tokens::new(without_row_spacing(v[from..].to_vec())));
195  pieces.retain(|piece| !piece.is_empty());
196  (pieces.len() > 1).then_some(pieces)
197}

199/// A piece of a row of addresses without what parts it from the next, no text of an address: the spaces and ties around it
200/// (`\texttt{a@x} ~\quad~~ \texttt{b@y}`, 2610.06195, whose ties read `˜`) and the comma or semicolon it ends with, inside a
201/// group that ends it too (`a@x,`, `\texttt{a@x,}`; 2505.22665, 2505.05785's line ending `\texttt{simon.zl@…},`).
202pub(in crate::base_utilities) fn without_row_spacing(mut tokens: Vec<Token>) -> Vec<Token> {
203  let spacing = |t: &Token| *t == T_SPACE!() || *t == T_ACTIVE!('~');
204  // (and the skip it opens with, `\vspace{0.1cm} \texttt{liuqing25@…} \quad \texttt{wuou@…}`, 2606.01635)
205  let lead = leading_spacing_end(&tokens);
206  tokens.drain(..lead);
207  let lead = tokens.iter().take_while(|t| spacing(t)).count();
208  tokens.drain(..lead);
209  while tokens.last().is_some_and(spacing) {
210    tokens.pop();
211  }
212  let mut end = tokens.len();
213  while end > 0 && tokens[end - 1].get_catcode() == Catcode::END {
214    end -= 1;
215  }
216  if end > 0 && (tokens[end - 1] == T_OTHER!(",") || tokens[end - 1] == T_OTHER!(";")) {
217    tokens.remove(end - 1);
218  }
219  while tokens.last().is_some_and(spacing) {
220    tokens.pop();
221  }
222  tokens
223}

225/// The addresses of an author-block line that is only email addresses (`\texttt{a@x, b@y}`, `{a,b}@dom`), each as
226/// tokens ([`address_tokens`]); a line of one address is itself (`\href{mailto:a@x}{a@x}` stays a link). `None` for
227/// any other line.
228pub(in crate::base_utilities) fn bare_email_addresses(
229  line: &Tokens,
230) -> Result<Option<Vec<Tokens>>> {
231  let Some(addrs) = email_addresses(line).filter(|addrs| !addrs.is_empty()) else {
232    return Ok(None);
233  };
234  if addrs.len() == 1 {
235    return Ok(line_is_email(line).then(|| vec![line.clone()]));
236  }
237  address_tokens(line, addrs).map(Some)
238}

253/// The body a creator's note (`\lx@add@thanks`, `\lx@add@note`) digests: its leading marks as the author line reads
254/// them when an author's mark requests them and the note is one legend — a `\thanks{\textsuperscript{$\dagger$} School
255/// ...}` is labelled by its mark and goes to the starred authors (2609.00885, 2609.19600) — and the rest, or all of it
256/// when no author requests the mark or more legends follow, with the plain superscripts (`\lx@frontmatter@plainsups`): its `10 m$^{2}$` is text, not a mark
257/// (KNOWN_PERL_ERRORS #515), and an unrequested leading mark stays shown, its note with its author.
258pub(in crate::base_utilities) fn note_body(content: Tokens) -> Tokens {
259  let toks = content.unlist();
260  let lead = toks.iter().take_while(|t| **t == T_SPACE!()).count();
261  let end = lead + leading_marks_end(&toks[lead..]);
262  let requested = flush_requested_marks();
263  // One legend: no other mark after a space in its text (`$^{\dag}$ These authors ... $^{*}$ This author ...` is several
264  // legends and a funding line in one `\thanks`, 2609.39576, which stays whole with its author), where `10 m$^{2}$`
265  // is glued to its base.
266  // A numeral's spaced suffix (`5 $^{th}$`, shown by `rewrite_ordinal_superscripts`) is no legend.
267  let after_numeral = |i: usize| {
268    i >= 2
269      && toks[i - 2].get_catcode() == Catcode::OTHER
270      && toks[i - 2].with_str(|s| !s.is_empty() && s.chars().all(|c| c.is_ascii_digit()))
271  };
272  let another_legend = (end + 1..toks.len())
273    .any(|i| toks[i - 1] == T_SPACE!() && !after_numeral(i) && leading_marks_end(&toks[i..]) > 0);
274  let marks_requested = end > lead
275    && !another_legend
276    && mark_operands(&toks[lead..end]).iter().any(|operand| {
277      clean_frontmatter_labels(&Tokens::new(operand.clone()).to_string(), "affiliation")
278        .iter()
279        .any(|label| requested.contains(label))
280    });
281  let split = if marks_requested { end } else { 0 };
282  let mut body = toks[..split].to_vec();
283  body.extend([T_CS!("\\bgroup"), T_CS!("\\lx@frontmatter@plainsups")]);
284  body.extend_from_slice(&toks[split..]);
285  body.push(T_CS!("\\egroup"));
286  Tokens::new(body)
287}

301/// An institution without the SEPARATOR that clings to its end when the affiliations are comma-joined
302/// (`\textsuperscript{1}Univ A, \textsuperscript{2}Univ B`) rather than space-joined: without this the affiliation
303/// contact reads "Univ A," with a stray trailing comma (html_feedback#6588, arXiv:2606.01317). Only trailing
304/// whitespace + a SINGLE trailing comma (or semicolon, `$^1$Univ A; $^2$Univ B`, 2609.22690) are removed, so a comma
305/// INSIDE an institution name ("Dept X, Univ Y, City") is untouched. Applied where a marked piece is followed by
306/// another: between two on a line ([`split_marked_pieces`]), and at a line's end when the next line opens a new one
307/// (`…Univ C,\\ \textsuperscript{4}Univ D`); before a line continuing it the comma is the institution's own, as
308/// pdflatex prints it ("Université Laval,\\ pavillon …", 1507.08653).
309pub(in crate::base_utilities) fn trim_trailing_separator(mut toks: Vec<Token>) -> Vec<Token> {
310  // (and the spacing that parts two entries on one line, `$^1$Univ A \quad $^2$Univ B`, `~~~`: [`is_spacing`]; and a
311  // line break left at a piece's end, `{\bfseries $^3$Lab C\\ $^4$Lab D}`'s first)
312  while toks
313    .last()
314    .is_some_and(|t| is_spacing(t) || *t == T_CS!("\\\\") || *t == T_CS!("\\newline"))
315  {
316    toks.pop();
317  }
318  // (and the conjunction that joins the last two entries of a run, `$^2$Univ B and $^3$Univ C`, `\&`: no word of the
319  // entry before it — a word of its own, after a space or a comma)
320  let n = toks.len();
321  let and_word = n >= 4
322    && toks[n - 3..]
323      .iter()
324      .map(ToString::to_string)
325      .eq(["a", "n", "d"])
326    && (is_spacing(&toks[n - 4]) || toks[n - 4] == T_OTHER!(","));
327  let ampersand = n >= 2 && toks[n - 1] == T_CS!("\\&") && is_spacing(&toks[n - 2]);
328  if and_word {
329    toks.truncate(n - 3);
330  } else if ampersand {
331    toks.truncate(n - 1);
332  }
333  while toks.last().is_some_and(is_spacing) {
334    toks.pop();
335  }
336  // (a separator glyph between two entries, `\textit{$^1$A} $|$ \textit{$^2$B}` — 2508.07190 —, `A \textbullet{} B`: no
337  // glyph of either entry, [`separator_glyph_len`])
338  let glyph = separator_glyph_len(&toks);
339  if glyph > 0 {
340    toks.truncate(toks.len() - glyph);
341    while toks.last().is_some_and(is_spacing) {
342      toks.pop();
343    }
344  }
345  if toks.last() == Some(&T_OTHER!(",")) || toks.last() == Some(&T_OTHER!(";")) {
346    toks.pop();
347    while toks.last().is_some_and(is_spacing) {
348      toks.pop();
349    }
350  }
351  toks
352}

354/// The glyphs a run of entries is parted by between their marks, set in math (`|`, `\vert`, `\mid`, `\cdot`,
355/// `\bullet`, `\diamond`) or as text (`|`, `\textbar`, `\textbullet`, `\textperiodcentered`).
356const SEPARATOR_GLYPHS: &[&str] = &["|", "\\vert", "\\mid", "\\cdot", "\\bullet", "\\diamond"];
357
358/// The length of the separator glyph closing `toks` after a space, when it does: a formula of one separator glyph alone
359/// (`$|$`, `$\cdot$`: [`SEPARATOR_GLYPHS`]), or a text one (`|`, `\textbar{}`, `\textbullet`), else 0. After a space
360/// only: a glyph glued to a word is the entry's (`A|B`).
361fn separator_glyph_len(toks: &[Token]) -> usize {
362  let n = toks.len();
363  let spaced = |at: usize| at > 0 && is_spacing(&toks[at - 1]);
364  let glyph = |t: &Token| SEPARATOR_GLYPHS.iter().any(|g| t.with_str(|s| s == *g));
365  // (`$|$`, `$ \cdot $`)
366  if n >= 3 && toks[n - 1].get_catcode() == Catcode::MATH {
367    let open = toks[..n - 1]
368      .iter()
369      .rposition(|t| t.get_catcode() == Catcode::MATH);
370    if let Some(open) = open {
371      let inner: Vec<&Token> = toks[open + 1..n - 1]
372        .iter()
373        .filter(|t| **t != T_SPACE!())
374        .collect();
375      if inner.len() == 1 && glyph(inner[0]) && spaced(open) {
376        return n - open;
377      }
378    }
379    return 0;
380  }
381  // (`\textbar{}`, `\textbullet`)
382  let mut end = n;
383  if end >= 2
384    && toks[end - 1].get_catcode() == Catcode::END
385    && toks[end - 2].get_catcode() == Catcode::BEGIN
386  {
387    end -= 2;
388  }
389  if end >= 1
390    && ["\\textbar", "\\textbullet", "\\textperiodcentered"]
391      .iter()
392      .any(|cs| toks[end - 1] == T_CS!(*cs))
393    && spaced(end - 1)
394  {
395    return n - (end - 1);
396  }
397  if n >= 1 && toks[n - 1] == T_OTHER!("|") && spaced(n - 1) {
398    return 1;
399  }
400  0
401}

403/// [`split_before_affiliation_marks`] of a line (`line_start`) or of what follows a wrapper in one (`{…} \vspace{1mm}
404/// $^3$Univ C`), where a lead that prints nothing is a separator, not the start of the first institution.
405fn split_marked_pieces(tokens: Tokens, line_start: bool) -> Result<Vec<Tokens>> {
406  let toks = tokens.unlist();
407  // (the marks the queued names request, when they carry any: a number or letter mark none of them requests, after a
408  // word and a plain space, is the entry's own text — `$^1$Laboratory for $^3$He Studies`, the isotope — not the next
409  // entry's label; after a separator or wider spacing it still parts the entries)
410  let names_marked = queued_authors_marked()?;
411  // (each mark outside every group: where it starts, the label it sets, whether it is glued to the word before it —
412  // a plain space between, no separator: no spacing command, line break, comma or semicolon — and where it ends)
413  let marks = top_level_marks(&toks);
414  let mut requested_marks = Vec::with_capacity(marks.len());
415  for mark in &marks {
416    requested_marks.push(match &mark.operand {
417      Some(operand) if names_marked => !queued_requesters(operand)?.is_empty(),
418      _ => false,
419    });
420  }
421  let text_mark = |at: usize, last_head: Option<&str>| {
422    if !names_marked {
423      return false;
424    }
425    let Some(k) = marks.iter().position(|m| m.at == at) else {
426      return false;
427    };
428    let mark = &marks[k];
429    let requested = requested_marks[k];
430    // (a mark no name requests that heads nothing — only spacing or a separator follows it before the next mark or the
431    // end: `$^1$Univ A $^\dagger$ \quad $^2$Univ B` — is its entry's text, a symbol too; review rev99a S5)
432    let next = marks.get(k + 1).map_or(toks.len(), |m| m.at);
433    let heads_nothing = toks[mark.end.min(next)..next]
434      .iter()
435      .all(|t| is_spacing(t) || *t == T_OTHER!(",") || *t == T_OTHER!(";"));
436    if !requested && heads_nothing {
437      return true;
438    }
439    if !mark.glued {
440      return false;
441    }
442    // (glued to a word: a number or letter mark no name requests is the entry's own text — the isotope; one list labels
443    // each entry once, so a label that also heads an entry after a separator is text where it is glued, `$^1$Inst.\ for
444    // $^{2}$H Research \quad $^2$Univ B`, and so is one that breaks the list's running sequence before the sequence
445    // goes on, `$^1$Lab for $^3$He Studies \quad $^2$Univ B`; review rev99a S1)
446    let symbol = mark
447      .operand
448      .as_ref()
449      .is_none_or(|operand| is_symbol_mark(operand));
450    if !symbol && !requested {
451      return true;
452    }
453    let Some(label) = mark.label.as_deref() else {
454      return false;
455    };
456    let heads_elsewhere = marks
457      .iter()
458      .any(|m| m.at != at && !m.glued && m.label.as_deref() == Some(label));
459    // (in a list whose entries a separator parts — a later mark after one — not a run-in list of plain spaces, where a
460    // list written out of order, `$^1$Univ A $^3$Univ C $^2$Univ B`, still parts at each mark)
461    let separated = marks.iter().skip(1).any(|m| m.at != at && !m.glued);
462    let breaks_sequence = separated
463      && last_head.and_then(successor).is_some_and(|expected| {
464        expected != label
465          && marks[k + 1..]
466            .iter()
467            .any(|m| m.label.as_deref() == Some(expected.as_str()))
468      });
469    heads_elsewhere || breaks_sequence
470  };
471  // (the label of the last mark that opened a piece)
472  let mut last_head: Option<String> = marks.first().and_then(|m| m.label.clone());
473  let mut segments: Vec<Tokens> = Vec::new();
474  let mut current: Vec<Token> = Vec::new();
475  // A boundary only outside every group: a cut inside `{…}` would leave the pieces unbalanced (a mark
476  // list inside a font group is split by `split_wrapped_affiliation_marks`, which repeats the group).
477  let mut depth = 0usize;
478  for (i, t) in toks.iter().enumerate() {
479    let dollar_super = *t == T_MATH!() && toks.get(i + 1).is_some_and(|n| *n == T_SUPER!());
480    // (and a mark heading its entry from a group of its own or from the entry's wrapper, `{$^2$} ETH Zurich`,
481    // `\textit{$^2$CSIRO Data61}`: [`wrapped_mark_at`], read at the top level only)
482    let is_mark_start =
483      dollar_super || is_mark_command(t) || (depth == 0 && marks.iter().any(|m| m.at == i));
484    // Only treat a mark as an affiliation boundary when it is preceded by
485    // whitespace (institutions are space-separated). A mark glued to the
486    // preceding text — e.g. a superscript INSIDE an institution name,
487    // "Center for R$^2$ Studies" — is not a boundary, so the name is not
488    // wrongly split (reviewer-flagged). The first mark (current empty) always
489    // opens segment 0.
490    // A lead that prints nothing before the first mark — declarations (`\color{blue} $^1$…`), spacing (`\small \vspace{1mm}
491    // $^{1}$Univ A`, 2407.15815) — is no segment: it opens the first one.
492    // (a bare space is no lead: `\textit{$^1$A, $^2$B} $^3$C` keeps its boundary before `$^3$`)
493    let prints_nothing = || {
494      line_start
495        && segments.is_empty()
496        && current.iter().any(|t| t.get_catcode() == Catcode::CS)
497        && leading_unprinted_end(&current) == current.len()
498    };
499    // (a space, or wider spacing: `$^1$Univ A \quad $^2$Univ B` inside a wrapper, `~~~`; 2506.18881)
500    // (or a separator with no space after it, `Lisbon,\IEEEauthorrefmark{2}Univ B` — 2411.03010's IEEE blockA run)
501    if is_mark_start
502      && depth == 0
503      && current
504        .last()
505        .is_some_and(|t| is_spacing(t) || *t == T_OTHER!(",") || *t == T_OTHER!(";"))
506      && !text_mark(i, last_head.as_deref())
507    {
508      if let Some(m) = marks.iter().find(|m| m.at == i) {
509        last_head.clone_from(&m.label);
510      }
511      if prints_nothing() {
512        // …nor is the space between it and the mark the institution's
513        while current.last() == Some(&T_SPACE!()) {
514          current.pop();
515        }
516      } else {
517        segments.push(Tokens::new(trim_trailing_separator(std::mem::take(
518          &mut current,
519        ))));
520      }
521    }
522    match t.get_catcode() {
523      Catcode::BEGIN => depth += 1,
524      Catcode::END => depth = depth.saturating_sub(1),
525      _ => {},
526    }
527    current.push(*t);
528  }
529  // (the last piece keeps a comma closing it until the next line shows whether it separates: [`trim_trailing_separator`])
530  while current.last() == Some(&T_SPACE!()) {
531    current.pop();
532  }
533  if !current.is_empty() {
534    segments.push(Tokens::new(current));
535  }
536  Ok(segments)
537}

540/// sets, and whether it is glued to the word before it (only a plain space between).

543  end:     usize,
544  operand: Option<Vec<Token>>,
545  label:   Option<String>,
546  glued:   bool,

549/// A mark heading an entry from inside a group at `toks[i]`, outside every other group: a group the mark alone fills
550/// (`{$^2$} ETH Zurich`, 2203.16586's cvpr line `\small {$^1$} Beijing Institute of Technology {$^2$} ETH Zurich …`), or
551/// one entry's own wrapper — a text font command's argument or a group, past the declarations opening it — that the
552/// mark opens (`\textit{$^2$CSIRO Data61}`, 2508.07190's `\textit{$^1$…} $|$ \textit{$^2$…}`; `{\it $^2$Univ B}`;
553/// 2605.25195's `\textbf{$^1$Fudan University}  \textbf{$^2$Tencent Hunyuan}`). The mark's operand, and where what it
554/// labels starts: past the group a mark alone fills, past the mark inside a wrapper. A wrapper is the entry's own only
555/// where it holds the whole entry — nothing but a separator after it ([`trim_trailing_separator`]) up to the next mark or
556/// line break: a font group whose text goes on after it is the entry's text, its mark an isotope's (`Univ A,
557/// \textit{$^{2}$H} Lab, Rome $^{2}$Univ B`, `Lab for \textit{$^{3}$He} Studies`; review rev107e finding 3). A mark
558/// command's own group (`\textsuperscript{2}`) is the mark itself, read as one; a note's (`\thanks{$^1$…}`, an argument
559/// of a name annotation, [`is_name_annotation`]) none.
560fn wrapped_mark_at(toks: &[Token], i: usize) -> Option<(Vec<Token>, usize)> {
561  let open = match toks[i].get_catcode() {
562    Catcode::BEGIN if i > 0 && is_name_annotation(&toks[i - 1]) => return None,
563    Catcode::BEGIN => i,
564    Catcode::CS
565      if is_text_font_command(&toks[i])
566        && toks
567          .get(i + 1)
568          .is_some_and(|t| t.get_catcode() == Catcode::BEGIN) =>
569    {
570      i + 1
571    },
572    _ => return None,
573  };
574  let close = skip_group(toks, open);
575  let inner = toks.get(open + 1..close.checked_sub(1)?)?;
576  let mut j = leading_unprinted_end(inner);
577  while inner.get(j) == Some(&T_SPACE!()) {
578    j += 1;
579  }
580  let dollar_super =
581    inner.get(j)?.get_catcode() == Catcode::MATH && inner.get(j + 1) == Some(&T_SUPER!());
582  let (operand, mut next) = if dollar_super {
583    let (operand, next) = sup_operand_at(inner, j + 2)?;
584    (inner.get(next)?.get_catcode() == Catcode::MATH).then_some((operand, next + 1))?
585  } else if is_mark_command(&inner[j]) {
586    sup_operand_at(inner, j + 1)?
587  } else {
588    return None;
589  };
590  while inner.get(next) == Some(&T_SPACE!()) {
591    next += 1;
592  }
593  if next == inner.len() {
594    return Some((operand, close));
595  }
596  // (what follows the wrapper up to the next mark or line break: a separator alone, else the wrapper is no entry's own)
597  let mut k = close;
598  while k < toks.len() && !opens_entry_or_line(toks, k) {
599    k += 1;
600  }
601  // (or a lead that prints nothing before it, `\hfill`, `\vspace{1mm}`)
602  let after = &toks[close.min(k)..k];
603  let separator = |piece: &[Token]| trim_trailing_separator(piece.to_vec()).is_empty();
604  (separator(after) || separator(&after[leading_unprinted_end(after)..]))
605    .then_some((operand, open + 1 + next))
606}

608/// Whether `toks[k]` opens a mark ([`wrapped_mark_at`]'s scan past a wrapper): a superscript formula, a mark command, a
609/// group or text font command whose content a mark leads — or a line break, a paragraph's end (2603.19086's last entry
610/// `\textit{$^5$CSIRO Data61}` before a blank line).
611fn opens_entry_or_line(toks: &[Token], k: usize) -> bool {
612  let t = &toks[k];
613  if *t == T_CS!("\\\\") || *t == T_CS!("\\newline") || *t == T_CS!("\\par") || is_mark_command(t) {
614    return true;
615  }
616  if t.get_catcode() == Catcode::MATH {
617    return toks.get(k + 1) == Some(&T_SUPER!());
618  }
619  let brace = match t.get_catcode() {
620    Catcode::BEGIN => k,
621    Catcode::CS if is_text_font_command(t) => k + 1,
622    _ => return false,
623  };
624  if toks
625    .get(brace)
626    .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
627  {
628    return false;
629  }
630  let close = skip_group(toks, brace).min(toks.len());
631  let inner = &toks[brace + 1..close.saturating_sub(1).max(brace + 1)];
632  let mut j = leading_unprinted_end(inner);
633  while inner.get(j) == Some(&T_SPACE!()) {
634    j += 1;
635  }
636  inner.get(j).is_some_and(|t| {
637    is_mark_command(t)
638      || (t.get_catcode() == Catcode::MATH && inner.get(j + 1) == Some(&T_SUPER!()))
639  })
640}

642/// The marks of `toks` outside every group, in order (`$^{…}$`, a mark command: [`is_mark_command`]; a mark heading an
643/// entry from a group of its own or from the entry's wrapper, [`wrapped_mark_at`]).
644fn top_level_marks(toks: &[Token]) -> Vec<ListMark> {
645  let mut marks = Vec::new();
646  let mut depth = 0usize;
647  for (i, t) in toks.iter().enumerate() {
648    let dollar_super = *t == T_MATH!() && toks.get(i + 1).is_some_and(|n| *n == T_SUPER!());
649    let wrapped = (depth == 0 && !dollar_super && !is_mark_command(t))
650      .then(|| wrapped_mark_at(toks, i))
651      .flatten();
652    if depth == 0 && (dollar_super || is_mark_command(t) || wrapped.is_some()) {
653      let operand_at = if dollar_super { i + 2 } else { i + 1 };
654      let read = match wrapped {
655        Some((operand, end)) => Some((operand, end)),
656        None => sup_operand_at(toks, operand_at),
657      };
658      let end = match &read {
659        Some((_, next)) if dollar_super && toks.get(*next) == Some(&T_MATH!()) => next + 1,
660        Some((_, next)) => *next,
661        None => i + 1,
662      };
663      let operand = read.map(|(operand, _)| operand);
664      let label = operand.as_deref().and_then(mark_label);
665      let before = toks[..i].iter().rev().find(|t| **t != T_SPACE!());
666      let glued = toks[..i].last() == Some(&T_SPACE!())
667        && before.is_some_and(|t| {
668          matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER)
669            && *t != T_OTHER!(",")
670            && *t != T_OTHER!(";")
671        });
672      marks.push(ListMark {
673        at: i,
674        end,
675        operand,
676        label,
677        glued,
678      });
679    }
680    match t.get_catcode() {
681      Catcode::BEGIN => depth += 1,
682      Catcode::END => depth = depth.saturating_sub(1),
683      _ => {},
684    }
685  }
686  marks
687}

689/// The label after `label` in a running sequence of marks: the next number (`1` → `2`), the next letter (`a` → `b`).
690fn successor(label: &str) -> Option<String> {
691  let bare = label.rsplit(':').next().unwrap_or(label);
692  let next = if let Ok(n) = bare.parse::<u32>() {
693    (n + 1).to_string()
694  } else {
695    let mut chars = bare.chars();
696    match (chars.next(), chars.next()) {
697      (Some(c), None) if c.is_ascii_alphabetic() && c != 'z' && c != 'Z' => {
698        char::from(c as u8 + 1).to_string()
699      },
700      _ => return None,
701    }
702  };
703  Some(match label.rsplit_once(':') {
704    Some((prefix, _)) => s!("{prefix}:{next}"),
705    None => next,
706  })
707}

709/// [`split_before_affiliation_marks`] for a marker-led affiliation line that opens with a font command or a group
710/// holding the marks (`\textit{$^1$Univ A, $^2$Univ B}`, `{\small $^1$A; $^2$B}`, `\small{$^1$A, $^2$B}. Email: …`):
711/// the inner list is split, the wrapper repeated on each piece — a group with the declarations that open it
712/// (`\small`) — and what follows the wrapper kept after the last piece; else the marks inside the group were no
713/// boundary, or every piece after the first left the font (62zk review; 2609.00995; repro
714/// sectioning-frontmatter/affiliation_marks_inside_a_font_group_keep_the_font). A `\thanks{…}` wrapper is not
715/// repeated: its mark-led content is the affiliation list (the `\thanks` idiom of `\lx@add@thanks`; 2609.24896).
716pub(in crate::base_utilities) fn split_wrapped_affiliation_marks(
717  line: Tokens,
718) -> Result<Vec<Tokens>> {
719  if let Some((cmd, inner, trailing)) = leading_wrapper(&line) {
720    // A bare group's opening declarations (`\small`, `\it`, `\color{blue}` with its argument, among the spacing that
721    // prints nothing: `{\small\vspace{1mm}\noindent\it $^{1}$…}`, 2407.15815), repeated in each later piece.
722    let opening = if cmd.is_none() {
723      lead_declarations(inner.unlist_ref())
724    } else {
725      Vec::new()
726    };
727    let mut pieces = split_wrapped_affiliation_marks(inner)?;
728    if pieces.len() > 1 {
729      // (a comma closing the wrapped list separates it from what follows the wrapper: `\textit{$^1$A, $^2$B,} $^3$C`)
730      if let Some(last) = pieces.last_mut() {
731        *last = Tokens::new(trim_trailing_separator(std::mem::take(last).unlist()));
732      }
733      let mut out: Vec<Tokens> = pieces
734        .drain(..)
735        .enumerate()
736        .map(|(i, piece)| {
737          let mut wrapped = Vec::new();
738          if cmd == Some(T_CS!("\\thanks")) {
739            wrapped.extend(piece.unlist());
740          } else {
741            wrapped.extend(cmd);
742            wrapped.push(T_BEGIN!());
743            if i > 0 {
744              wrapped.extend(opening.iter().copied());
745            }
746            wrapped.extend(piece.unlist());
747            wrapped.push(T_END!());
748          }
749          Tokens::new(wrapped)
750        })
751        .collect();
752      // What follows the wrapper is split at its own marks too (`\textit{$^1$A, $^2$B} $^3$C`): its first piece stays
753      // with the last wrapped one, the rest are pieces of their own.
754      let mut after = split_marked_pieces(Tokens::new(trailing), false)?.into_iter();
755      if let (Some(first), Some(last_piece)) = (after.next(), out.last_mut()) {
756        let mut joined = last_piece.clone().unlist();
757        joined.extend(first.unlist());
758        *last_piece = Tokens::new(joined);
759      }
760      out.extend(after);
761      return Ok(out);
762    }
763  }
764  split_before_affiliation_marks(line)
765}

767/// A note an author block takes out as the document's, as it digests in a contact: `\thanks{…}` the
768/// `\lx@add@pubnote@thanks{…}` and `\footnote{…}` the `\lx@add@pubnote{…}` `digestFrontmatterItem` binds them to there
769/// (Perl Base_Utility.pool.ltxml:331-332, queue_digest.rs) — `note` one note's span, its command through its closing
770/// brace. `None` for a note of another command, or with a `[…]` before its text, which no pubnote takes: such a note
771/// stays where it stands. The one conversion of a line of notes no name requests (`notes_the_names_request`) and of an
772/// address line's notes (`address_line_notes`).
773pub(in crate::base_utilities) fn document_note_call(note: &[Token]) -> Option<Vec<Token>> {
774  let plain = !note
775    .iter()
776    .take_while(|t| t.get_catcode() != Catcode::BEGIN)
777    .any(|t| *t == T_OTHER!("["));
778  let call = match note.first()? {
779    t if plain && *t == T_CS!("\\thanks") => T_CS!("\\lx@add@pubnote@thanks"),
780    t if plain && *t == T_CS!("\\footnote") => T_CS!("\\lx@add@pubnote"),
781    _ => return None,
782  };
783  let mut out = Vec::with_capacity(note.len());
784  out.push(call);
785  out.extend_from_slice(&note[1..]);
786  Some(out)
787}

789/// The end of the note opening at `line[i]`: a `\thanks`, `\footnote` or `\IEEEcompsocitemizethanks`, its `[…]` and
790/// its braced text; `None` for another token or a note not so written.
791pub(in crate::base_utilities) fn note_call_end(line: &[Token], i: usize) -> Option<usize> {
792  let past_spaces = |mut at: usize| {
793    while line.get(at) == Some(&T_SPACE!()) {
794      at += 1;
795    }
796    at
797  };
798  let t = line.get(i)?;
799  if *t != T_CS!("\\thanks")
800    && *t != T_CS!("\\footnote")
801    && *t != T_CS!("\\IEEEcompsocitemizethanks")
802  {
803    return None;
804  }
805  let mut at = past_spaces(i + 1);
806  if line.get(at) == Some(&T_OTHER!("[")) {
807    let close = line[at..].iter().position(|t| *t == T_OTHER!("]"))?;
808    at = past_spaces(at + close + 1);
809  }
810  if line
811    .get(at)
812    .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
813  {
814    return None;
815  }
816  let end = skip_group(line, at);
817  line
818    .get(end.wrapping_sub(1))
819    .is_some_and(|t| t.get_catcode() == Catcode::END)
820    .then_some(end)
821}

823/// Rewrite footnote-SYMBOL author superscripts — `$^{*}$`, `${}^{\dagger}$`,
824/// `\textsuperscript{\ddagger}`, a bare `^{*}` — onto the visible
825/// `\lx@frontmatter@keepsup` sentinel, BEFORE author-block branch selection.
826///
827/// Two effects, both wanted (OXIDIZED_DESIGN #52; witness arXiv:2506.06941, where
828/// Iman Mirzadeh's literal `$^{*}$` was silently dropped):
829///   * the mark no longer counts as an affiliation superscript, so a block whose
830///     ONLY superscript is such a note-mark takes the clean no-marker author
831///     branch instead of the affiliation-linking one;
832///   * the symbol renders as a real superscript instead of being consumed into an
833///     `affiliation:*` label that matches no affiliation and is discarded.
834///
835/// A number or letter glued to such a symbol (`$^{1*}$`, `\textsuperscript{a\dagger}`, the symbol first too:
836/// `\textsuperscript{*1}`, 2505.22662) is the affiliation mark it is and the symbol beside it (`split_glued_symbol`): the
837/// mark requests affiliation 1, the symbol is shown, as for `$^{1,*}$` (else the one request `affiliation:1*` matched no
838/// affiliation, and affiliation 1 went to no one; authblk's `\author{Ann$^{1*}$}`, 77n r3; the neurips fixture's Wenqi
839/// Shao, 2507.06670). Both are written in the order they print, for an unanswered mark to show there
840/// (`show_unanswered_marks`: 2610.07087's `$^{1*}$`). The symbol is shown as written: a math `*` as `∗`, as the unsplit
841/// `$^{1,*}$` keeps it (2210.14772, 2508.09827, 2505.11121; [`keepsup_in_math`]), a text one as `*`.
842///
843/// NUMERIC/lettered affiliation marks (`$^{1}$`, `\textsuperscript{a}`) are left
844/// untouched, so affiliation linking is unaffected. Only the specific
845/// superscript-mark token shapes are matched; a superscript inside real math
846/// (`$x^2$`) is not (its base is not empty), so math content is preserved.
847pub(in crate::base_utilities) fn rewrite_symbol_superscripts(tokens: Tokens) -> Tokens {
848  let src = tokens.unlist();
849  let n = src.len();
850  let mut out: Vec<Token> = Vec::with_capacity(n);
851  let mut i = 0;
852  // Read a superscript operand at `src[k]`: a braced `{...}` group, or a single
853  // token. Returns (operand tokens, index past the operand) or None.
854  let read_operand = |k: usize| -> Option<(Vec<Token>, usize)> {
855    if k >= n {
856      return None;
857    }
858    if src[k] == T_BEGIN!() {
859      let mut depth = 1usize;
860      let mut m = k + 1;
861      let mut inner = Vec::new();
862      while m < n && depth > 0 {
863        if src[m] == T_BEGIN!() {
864          depth += 1;
865        } else if src[m] == T_END!() {
866          depth -= 1;
867          if depth == 0 {
868            break;
869          }
870        }
871        inner.push(src[m]);
872        m += 1;
873      }
874      if depth == 0 {
875        Some((inner, m + 1))
876      } else {
877        None
878      }
879    } else {
880      Some((vec![src[k]], k + 1))
881    }
882  };
883  while i < n {
884    let t = src[i];
885    // `\textsuperscript{sym}` — always braced in practice.
886    if t == T_CS!("\\textsuperscript")
887      && src.get(i + 1) == Some(&T_BEGIN!())
888      && let Some((sym, next)) = read_operand(i + 1)
889      && is_footnote_symbol_operand(&sym)
890    {
891      out.extend(keepsup(sym));
892      i = next;
893      continue;
894    }
895    if t == T_CS!("\\textsuperscript")
896      && src.get(i + 1) == Some(&T_BEGIN!())
897      && let Some((operand, next)) = read_operand(i + 1)
898      && let Some(glued) = glued_marks(&operand)
899    {
900      for part in glued {
901        match part {
902          GluedPart::Symbol(symbol) => out.extend(keepsup(symbol)),
903          GluedPart::Marks(marks) => {
904            out.extend([t, T_BEGIN!()]);
905            out.extend(marks);
906            out.push(T_END!());
907          },
908        }
909      }
910      i = next;
911      continue;
912    }
913    // `$ [ {} ] ^ sym $` — a math span whose whole content is a superscript on an
914    // empty base. Rewrite the entire span (delimiters included) to a text sup.
915    if t == T_MATH!()
916      && let Some(close) = (i + 1..n).find(|&m| src[m] == T_MATH!())
917    {
918      // optional empty base `{}` before the superscript
919      let p = if i + 2 < close && src[i + 1] == T_BEGIN!() && src[i + 2] == T_END!() {
920        i + 3
921      } else {
922        i + 1
923      };
924      if p < close
925        && src[p] == T_SUPER!()
926        && let Some((sym, next)) = read_operand(p + 1)
927        && next == close
928        && is_footnote_symbol_operand(&sym)
929      {
930        out.extend(keepsup(sym));
931        i = close + 1;
932        continue;
933      }
934      if p < close
935        && src[p] == T_SUPER!()
936        && let Some((operand, next)) = read_operand(p + 1)
937        && next == close
938        && let Some(glued) = glued_marks(&operand)
939      {
940        for part in glued {
941          match part {
942            GluedPart::Symbol(symbol) => out.extend(keepsup_in_math(symbol)),
943            GluedPart::Marks(marks) => {
944              out.extend_from_slice(&src[i..=p]);
945              out.push(T_BEGIN!());
946              out.extend(marks);
947              out.extend([T_END!(), T_MATH!()]);
948            },
949          }
950        }
951        i = close + 1;
952        continue;
953      }
954    }
955    // Only the math-delimited (`$…$`) and `\textsuperscript{…}` spellings are
956    // rewritten — the correct ways to write a text superscript. A bare `^` is NOT
957    // matched here: outside math it is invalid LaTeX, and a `^` reached mid-scan is
958    // the superscript operator of real inline math on a non-empty base (`$a^{*}$`),
959    // which must be left intact.
960    out.push(t);
961    i += 1;
962  }
963  Tokens::new(out)
964}

966/// [`keepsup`] for a symbol glued to a mark in math (`$^{1*}$`, `$^{1,2*}$`, 2505.11121): a math `*` prints `∗`, shown
967/// so (FIDELITY_CHARTER §4, "stays as written"; `\textsuperscript{1*}` keeps its text `*`), its key the `*` the entries
968/// answer; beside a second symbol too (`$^{2,1*\dagger}$` prints `∗†`: 2404.16821, 2505.17017). A lone `$^{*}$` shows
969/// `*` as it always has, in [`rewrite_symbol_superscripts`]' own branch.
970fn keepsup_in_math(sym: Vec<Token>) -> Vec<Token> {
971  let shown = sym
972    .iter()
973    .map(|token| {
974      if *token == T_OTHER!("*") {
975        T_CS!("\\ast")
976      } else {
977        *token
978      }
979    })
980    .collect();
981  let mut v = visible_sup(shown);
982  v.extend([T_CS!("\\lx@frontmatter@symbolmark"), T_BEGIN!()]);
983  v.extend(sym);
984  v.push(T_END!());
985  v
986}

988/// `\lx@frontmatter@keepsup{<sym>}` as a token list: a superscript shown as it is, requesting nothing.
989pub(in crate::base_utilities) fn visible_sup(sym: Vec<Token>) -> Vec<Token> {
990  let mut v = vec![T_CS!("\\lx@frontmatter@keepsup"), T_BEGIN!()];
991  v.extend(sym);
992  v.push(T_END!());
993  v
994}
