1//! The one reading of a formula's mark-ness from its digested XMath (frontmatter refactor stage 1,
2//! `docs/archive/FRONTMATTER_DIGESTED_REFACTOR_2026-11-07.md`; OXIDIZED_DESIGN_DIVERGENCES #837, #911): whether a formula in a name
3//! or an affiliation is a mark, what its scripts print, and the base a script sits on. It reads structure and printed
4//! text, never TeX spelling: a font switch is a token's font, spacing an `XMHint`, `\{` and `\lbrace` one "{".
5//!
6//! It takes two vocabularies:
7//! - **as digested** (what every reader sees: the content check's reading of the placed frontmatter, `emitter::dom`,
8//!   `relocate::shown_marks`, `name_text::is_name_mark` — all inside `insert_frontmatter`, before `finish_document`'s
9//!   rewrite and math parse): `XMApp[@role]` of
10//!   `FLOATSUPERSCRIPT`/`FLOATSUBSCRIPT` (a script with nothing before it) or `POSTSUPERSCRIPT`/`POSTSUBSCRIPT` (a script
11//!   after a base), its `XMArg[@rule]` the script, beside flat `XMTok`/`XMText`/`XMHint` siblings (the constructors of
12//!   `tex_math.rs` `\lx@floating@superscript`, `\lx@post@superscript`, chosen at digestion by `script_handler`);
13//! - **as parsed** (the final XML; no reader since `ab_audit fmshadow` went, DEL4 — only `tests` here read it): a lone
14//!   floating script is `ltx:sup` (no formula: the rewrite
15//!   of `tex_math.rs`'s floating-script rule; Perl `TeX_Math.pool.ltxml:530`); a script the parser attached is
16//!   `XMApp(XMTok[@role='SUPERSCRIPTOP'][@scriptpos], base, script)`, `scriptpos` `preN` for a script before its base
17//!   (³He, `^{a})`) and `postN` after it; a surviving `FLOATSUPERSCRIPT` holds its script unwrapped; `XMDual` (its
18//!   presentation, the last child) and `XMWrap` around lists; any other application LaTeXML lays out as a row read in
19//!   print order ([`printed_children`]), so a script the parser nests under a word's invisible times (`$Ann^{1}$` as
20//!   `⁢(A, n, n¹)`, `$\mathrm{Ann~Able}^{1}$`) or a hyphenated word's minus reads as the digested shape's flat items.
21//!
22//! In both vocabularies a two-dimensional layout is one printed item (`row_layout`, by the operator's role and
23//! meaning as Perl `Post/MathML.pm` picks its presenter): a fraction, a root, an accent, an enclosure, a big operator,
24//! limits. A script under a fraction's bar or a root's sign is not a script the formula ends on (`\frac{a}{b^{1}}`).
25//!
26//! A formula **is a mark** when it opens with scripts nothing visible precedes (floating, or after a box that prints
27//! nothing, `$\mbox{ }^1$`: cond-mat0206265), at least one a superscript, and the first item printed after them is no
28//! letter or digit. That criterion is ours, not the parser's: the parser attaches a prescript (`scriptpos="pre"`) to
29//! whatever follows, a `)` as readily as a word. Spacing is no ink,
30//! punctuation and symbols are no base (`^{a})`, cond-mat9808038; `^{1}{}^{,}`; `^{1,2}\dagger`, 2501.04244; `^1,2`,
31//! 1912.02075's `A.P.~Orlov$^1,2$`). Its marks are its superscripts before the first letter or digit. Scripts after
32//! printed ink are on a **base** (`97403^{1}`, `\text{Leon Bello}^{1,3}`): the base, with its subscripts, is the item's
33//! text, the superscripts no mark. Scripts right before a letter or digit (an isotope, `{}^{3}He`, `{\rm ^{3}He}`) are
34//! a prescript on a printed word, no mark: the fail-safe, the formula's text kept as printed. What a reader has only as
35//! tokens (the label keys of `\lx@frontmatter@sup`, revtex's group rule, `notes::leads_with_mark`) is stages 3–5's.

39/// A node of a formula as [`read_formula`] walks it: the engine's libxml node (a test's own tree in the unit tests;
40/// `ab_audit fmshadow`'s roxmltree newtype went with the census, DEL4).
41pub trait FmMathNode: Sized + Clone {
42  /// The element's local name when it is a LaTeXML element (`XMApp`, `XMTok`, …); `None` for any other node.
43  fn ltx_name(&self) -> Option<String>;
44  /// The element's attribute `name`.
45  fn attribute(&self, name: &str) -> Option<String>;
46  /// The node's children, elements and text, in document order.
47  fn children(&self) -> Vec<Self>;
48  /// A text node's text; `None` for an element.
49  fn text(&self) -> Option<String>;
50}
51
52/// The engine's document node (the writer's view, `shown_marks`, `is_name_mark`).
53impl FmMathNode for Node {
54  fn ltx_name(&self) -> Option<String> {
55    (self.get_type() == Some(NodeType::ElementNode))
56      .then(|| document::with_node_qname(self, |q| q.strip_prefix("ltx:").map(str::to_string)))
57      .flatten()
58  }
59
60  fn attribute(&self, name: &str) -> Option<String> { self.get_attribute(name) }
61
62  fn children(&self) -> Vec<Self> { self.get_child_nodes() }
63
64  fn text(&self) -> Option<String> {
65    (self.get_type() == Some(NodeType::TextNode)).then(|| self.get_content())
66  }
67}

69/// What a formula is to the frontmatter ([`read_formula`]).
70#[derive(Debug, Clone, PartialEq, Eq)]
71pub enum FormulaMarks {
72  /// A mark: the printed text of each of its superscripts, in order (`$^{1,2}$` "1,2"; `${}^{a,\;b}$` "a,b").
73  Marks(Vec<String>),
74  /// Scripts on a printed base, nothing printed after them: the base's printed text with its subscripts (`$97403^{1}$`
75  /// "97403", `$\text{Leon Bello}^{1,3}$` "Leon Bello", `$G_2'$` "G2") and the superscripts' texts.
76  OnBase {
77    base:    String,
78    scripts: Vec<String>,
79  },
80  /// Any other formula: its text is its own.
81  Other,
82}

84/// One printed item of a formula, in print order.
85enum Item {
86  /// ink: its printed text, and whether it is a text box (`XMText`, `\text{…}`)
87  Ink { text: String, text_box: bool },
88  /// a script, in print order (a parsed `preN` before its base): a superscript or not, its printed text
89  Script {
90    superscript: bool,
91    text:        String,
92  },
93}

95/// The element children of `node`.
96fn elements<N: FmMathNode>(node: &N) -> Vec<N> {
97  node
98    .children()
99    .into_iter()
100    .filter(|n| n.ltx_name().is_some())
101    .collect()
102}

104/// What `node` prints: its text, without spacing hints, an `XMDual` by its presentation (its last child), an `XMRef`
105/// nothing (the presentation holds what it refers to), an `XMApp` in print order ([`printed_children`]).
106fn printed<N: FmMathNode>(node: &N) -> String {
107  if let Some(text) = node.text() {
108    return text;
109  }
110  match node.ltx_name().as_deref() {
111    None | Some("XMHint" | "XMRef") => String::new(),
112    Some("XMDual") => elements(node).last().map(printed).unwrap_or_default(),
113    // (an application in print order, its silent operator none: `⁢(S, m, i, t, h)` "Smith")
114    Some("XMApp") => printed_children(node).iter().map(printed).collect(),
115    Some(_) => node.children().iter().map(printed).collect(),
116  }
117}

119/// Whether an operator prints nothing: no text, or only the parser's invisible operators (U+2061 function
120/// application, U+2062 times, U+2063 separator, U+2064 plus).
121fn prints_nothing<N: FmMathNode>(op: &N) -> bool {
122  printed(op)
123    .chars()
124    .all(|c| c.is_whitespace() || ('\u{2061}'..='\u{2064}').contains(&c))
125}

127/// The children of `node` in the order they print. An `XMApp`'s operator and arguments are set as LaTeXML's own
128/// presentation sets them (Perl `Post/MathML.pm`): a script before or after its base by its `scriptpos`
129/// (`pmml_script`, L876), the operator of an infix role between its arguments, or before a lone one (`pmml_infix`,
130/// L626; the roles of `Apply:ADDOP:?` and its siblings, L1540, L1600-1622, L1699-1700, L1739), a `POSTFIX` operator
131/// after its argument (L1632), any other before its arguments (`Apply:?:?`, L1523). That is the print order of a row
132/// layout; a two-dimensional one's parts (a fraction's, a root's, an accent's, a limit's: `row_layout`) stay in
133/// document order, the order the formula's text has always been read in (`\sqrt[3]{x}` "3x"). An operator that prints nothing (no text, or only the
134/// invisible U+2061–U+2064) is left out, and so is the text between an application's children (the XML's
135/// indentation). Any other node's children are as they are, its text nodes too.
136pub fn printed_children<N: FmMathNode>(node: &N) -> Vec<N> {
137  // (as digested, and a floating script the parser left, an application's role names its script, no operator)
138  if node.ltx_name().as_deref() != Some("XMApp")
139    || node
140      .attribute("role")
141      .as_deref()
142      .and_then(digested_script)
143      .is_some()
144  {
145    return node.children();
146  }
147  let mut parts = elements(node).into_iter();
148  let Some(op) = parts.next() else {
149    return Vec::new();
150  };
151  let args: Vec<N> = parts.collect();
152  let role = op.attribute("role").unwrap_or_default();
153  let silent = prints_nothing(&op);
154  let mut out = Vec::new();
155  match role.as_str() {
156    "SUPERSCRIPTOP" | "SUBSCRIPTOP" => {
157      let pre = op
158        .attribute("scriptpos")
159        .is_some_and(|p| p.starts_with("pre"));
160      if !silent {
161        out.push(op);
162      }
163      out.extend(if pre {
164        args.into_iter().rev().collect()
165      } else {
166        args
167      });
168    },
169    "ADDOP" | "MULOP" | "BINOP" | "COMPOSEOP" | "MODIFIEROP" | "MIDDLE" | "RELOP" | "METARELOP"
170    | "ARROW"
171      if args.len() > 1 =>
172    {
173      for (i, arg) in args.into_iter().enumerate() {
174        if i > 0 && !silent {
175          out.push(op.clone());
176        }
177        out.push(arg);
178      }
179    },
180    "POSTFIX" => {
181      out.extend(args);
182      if !silent {
183        out.push(op);
184      }
185    },
186    _ => {
187      if !silent {
188        out.push(op);
189      }
190      out.extend(args);
191    },
192  }
193  out
194}

196/// The script role of an `XMApp` as digested (floating, `FLOAT*`, or after a base, `POST*`): whether a superscript.
197fn digested_script(role: &str) -> Option<bool> {
198  match role {
199    "FLOATSUPERSCRIPT" | "POSTSUPERSCRIPT" => Some(true),
200    "FLOATSUBSCRIPT" | "POSTSUBSCRIPT" => Some(false),
201    _ => None,
202  }
203}

205/// Whether `node` holds a script: a digested script application, or a parsed script operator.
206fn holds_a_script<N: FmMathNode>(node: &N) -> bool {
207  elements(node).iter().any(|c| {
208    let role = c.attribute("role");
209    match c.ltx_name().as_deref() {
210      Some("XMTok") => matches!(role.as_deref(), Some("SUPERSCRIPTOP" | "SUBSCRIPTOP")),
211      Some("XMApp") if role.as_deref().and_then(digested_script).is_some() => true,
212      _ => holds_a_script(c),
213    }
214  })
215}

217/// The role an operator presents by (Perl `Post/MathML.pm` `getOperatorRole`, L173): its own `role`, or, for an
218/// operator a script or an accent embellishes (`\sum_{i}` as `XMApp(SUBSCRIPTOP, ∑, i)`), the role of what it
219/// embellishes.
220fn operator_role<N: FmMathNode>(op: &N) -> Option<String> {
221  if let Some(role) = op.attribute("role") {
222    return Some(role);
223  }
224  if op.ltx_name().as_deref() != Some("XMApp") {
225    return None;
226  }
227  let parts = elements(op);
228  let (Some(inner), Some(base)) = (parts.first(), parts.get(1)) else {
229    return None;
230  };
231  matches!(
232    inner.attribute("role").as_deref(),
233    Some(
234      "SUPERSCRIPTOP" | "SUBSCRIPTOP" | "OVERACCENT" | "UNDERACCENT" | "MODIFIER" | "MODIFIEROP"
235    )
236  )
237  .then(|| operator_role(base))
238  .flatten()
239}

241/// Whether LaTeXML lays the application `node` out as a row (Perl `Post/MathML.pm`, which picks a presenter by the
242/// operator's role and meaning, `lookupPresenter` L205): the infix roles (`pmml_infix`, L1540, L1600-1622, L1699-1700,
243/// L1739), `POSTFIX` (L1632), any other operator (`Apply:?:?`, L1523), and a script after or
244/// before its base (`pmml_script`'s `msup`/`mmultiscripts`, L876). Not a row, and so one printed item ([`flatten`]):
245/// a fraction (`Apply:FRACOP:?` → `mfrac`, L1609; `\frac`, `\tfrac`, `\binom`'s), a root by its meaning
246/// (`square-root`/`nth-root` → `msqrt`/`mroot`, L1636-1645), a continued fraction (L1954's display `mfrac`), an accent
247/// over or under
248/// its base (`Apply:OVERACCENT:?`/`UNDERACCENT:?` → `mover`/`munder`, L1489-1505; `\hat`, `\overline`, `\overset`),
249/// an enclosure (`Apply:ENCLOSE:?` → `menclose`, L1507), a big operator with its body, a
250/// script set over or under its base (`scriptpos` `mid`, `pmml_script_mid_layout`'s `munder`/`mover`: limits,
251/// `\stackrel`), and anything carrying `enclose` (any node, MathML.pm L338-342: `\boxed`, amsmath.sty.ltxml:893-895).
252/// Two of these Perl lays out as a row, and are read as one item here on the safe side, with no output change on any
253/// probe (review of 90a r4): a big operator (`pmml_summation`, L1796-1798, an `mrow` of the operator and its body) and
254/// an inline continued fraction (`cfrac-inline`, `pmml_row`, L1954). The operator is read as written, not through
255/// `realize` as Perl reads it (L419): an `XMRef` operator would count as a row (none in the probes or the census).
256fn row_layout<N: FmMathNode>(node: &N) -> bool {
257  if node.attribute("enclose").is_some() {
258    return false;
259  }
260  let Some(op) = elements(node).into_iter().next() else {
261    return true;
262  };
263  // (a root's presenter is keyed by its meaning, which Perl looks up before the role's: `Apply:?:square-root`)
264  if matches!(
265    op.attribute("meaning").as_deref(),
266    Some("square-root" | "nth-root" | "continued-fraction")
267  ) {
268    return false;
269  }
270  match operator_role(&op).as_deref() {
271    Some("FRACOP" | "OVERACCENT" | "UNDERACCENT" | "ENCLOSE" | "BIGOP" | "INTOP" | "SUMOP") => {
272      false
273    },
274    Some("SUPERSCRIPTOP" | "SUBSCRIPTOP") => !op
275      .attribute("scriptpos")
276      .is_some_and(|p| p.starts_with("mid")),
277    _ => true,
278  }
279}

281/// `node`'s printed items, in print order, into `out`.
282fn flatten<N: FmMathNode>(node: &N, out: &mut Vec<Item>) {
283  let Some(name) = node.ltx_name() else {
284    if let Some(text) = node.text().filter(|t| !t.trim().is_empty()) {
285      out.push(Item::Ink { text, text_box: false });
286    }
287    return;
288  };
289  // (an enclosure prints its content inside one frame, MathML.pm L338-342: `\boxed{x^1}` as digested is an
290  // `XMArg[@enclose]`, amsmath.sty.ltxml:893-895, as parsed an `XMApp[@enclose]`)
291  if node.attribute("enclose").is_some() {
292    out.push(Item::Ink {
293      text:     printed(node),
294      text_box: false,
295    });
296    return;
297  }
298  match name.as_str() {
299    "XMHint" | "XMRef" => {},
300    "XMWrap" | "XMArg" => node.children().iter().for_each(|c| flatten(c, out)),
301    "XMDual" => {
302      if let Some(presentation) = elements(node).last() {
303        flatten(presentation, out);
304      }
305    },
306    "XMApp" => {
307      if let Some(superscript) = node.attribute("role").as_deref().and_then(digested_script) {
308        let text = printed(node);
309        // (a limit set over or under the item before it, `scriptpos` `mid`, prints with it: ink, as parsed)
310        out.push(
311          if node
312            .attribute("scriptpos")
313            .is_some_and(|p| p.starts_with("mid"))
314          {
315            Item::Ink { text, text_box: false }
316          } else {
317            Item::Script { superscript, text }
318          },
319        );
320        return;
321      }
322      // (a two-dimensional layout is one printed item: a script under a fraction's bar or a root's sign is no script
323      // the formula ends on, review 90a r3 F1)
324      if !row_layout(node) {
325        out.push(Item::Ink {
326          text:     printed(node),
327          text_box: false,
328        });
329        return;
330      }
331      // (as parsed: the script operator, its base, its script)
332      let parts = elements(node);
333      if let [op, base, script] = parts.as_slice()
334        && op.ltx_name().as_deref() == Some("XMTok")
335        && let Some(superscript) = match op.attribute("role").as_deref() {
336          Some("SUPERSCRIPTOP") => Some(true),
337          Some("SUBSCRIPTOP") => Some(false),
338          _ => None,
339        }
340      {
341        let item = Item::Script {
342          superscript,
343          text: printed(script),
344        };
345        // (a prescript prints before its base: `^{a})`, ³He)
346        if op
347          .attribute("scriptpos")
348          .is_some_and(|p| p.starts_with("pre"))
349        {
350          out.push(item);
351          flatten(base, out);
352        } else {
353          flatten(base, out);
354          out.push(item);
355        }
356        return;
357      }
358      // (any other row holding a script: its operator and arguments as they print — a script the parser nested under
359      // the invisible times of a word, `$Ann^{1}$` as `⁢(A, n, n¹)`, `$\mathrm{Ann~Able}^{1}$`, `$Smith-Jones^1$` as
360      // `−(⁢(S, …), ⁢(J, …, s¹))`, is read as the digested shape's flat items are; one holding none prints one item)
361      if holds_a_script(node) {
362        printed_children(node).iter().for_each(|c| flatten(c, out));
363      } else {
364        out.push(Item::Ink {
365          text:     printed(node),
366          text_box: false,
367        });
368      }
369    },
370    other => out.push(Item::Ink {
371      text:     printed(node),
372      text_box: other == "XMText",
373    }),
374  }
375}

389/// A formula `math` (an `ltx:Math`, or its `ltx:XMath`) as the frontmatter reads it ([`FormulaMarks`]; the module
390/// docs).
391pub fn read_formula<N: FmMathNode>(math: &N) -> FormulaMarks {
392  let xmath = match math.ltx_name().as_deref() {
393    Some("XMath") => Some(math.children()),
394    Some("Math") => elements(math)
395      .into_iter()
396      .find(|n| n.ltx_name().as_deref() == Some("XMath"))
397      .map(|x| x.children()),
398    _ => None,
399  };
400  let Some(children) = xmath else {
401    return FormulaMarks::Other;
402  };
403  let mut items = Vec::new();
404  for child in &children {
405    flatten(child, &mut items);
406  }
407  let Some(first) = items.iter().position(|i| matches!(i, Item::Script { .. })) else {
408    return FormulaMarks::Other;
409  };
410  let (lead, rest) = items.split_at(first);
411  let base: String = lead.iter().map(item_text).collect();
412  // (the superscripts' texts, and the subscripts', to the first letter or digit printed after them)
413  let mut scripts = Vec::new();
414  let mut subscripts = String::new();
415  let mut word_after = false;
416  for item in rest {
417    match item {
418      Item::Ink { text, .. } if has_word(text) => {
419        word_after = true;
420        break;
421      },
422      Item::Script { superscript: true, text, .. } => scripts.push(text.clone()),
423      Item::Script { text, .. } => subscripts.push_str(text),
424      _ => {},
425    }
426  }
427  if scripts.is_empty() {
428    return FormulaMarks::Other;
429  }
430  if base.trim().is_empty() {
431    // (floating scripts opening the formula: a mark, unless the item printed right after them is a letter or digit —
432    // their base, a prescript on a printed word: an isotope; punctuation or a symbol is none, `^{a})`, `^1,2`)
433    let base_after = rest
434      .iter()
435      .find(|i| matches!(i, Item::Ink { .. }))
436      .is_some_and(|i| has_word(item_text(i)));
437    // (a script after a box that prints nothing, `$\mbox{ }^1$` — cond-mat0206265 — floats as printed: its digested
438    // role is `POSTSUPERSCRIPT`, its base blank)
439    return if base_after {
440      FormulaMarks::Other
441    } else {
442      FormulaMarks::Marks(scripts)
443    };
444  }
445  if word_after {
446    return FormulaMarks::Other;
447  }
448  // (a base is a text box, or a word or a number: `97403`, `Ul. Ainy $299^A$`'s 299 — 2203.01892, 0707.4047)
449  let text_box = matches!(lead, [Item::Ink { text_box: true, .. }]);
450  let word = base
451    .chars()
452    .all(|c| c.is_alphanumeric() || c == '.' || c == '-' || c.is_whitespace());
453  if text_box || word {
454    // (a subscript is the base's: `$G_2'$`, 2208.14409's keyword "G₂′")
455    FormulaMarks::OnBase {
456      base: format!("{}{subscripts}", base.trim()),
457      scripts,
458    }
459  } else {
460    FormulaMarks::Other
461  }
462}

464/// A superscript of one or two lowercase letters glued on both sides to letters (`James M<sup>c</sup>Kernan`,
465/// math0610203; `I. M. M$^{\rm c}$Hardy`, 2003.01448): part of the word, no mark (the lexer's reading, `events::lexer`).
466/// `before`/`after`: the text printed just before and after it, when a text node.
467pub fn glued_in_a_word(letters: &str, before: Option<&str>, after: Option<&str>) -> bool {
468  (1..=2).contains(&letters.chars().count())
469    && letters.chars().all(char::is_lowercase)
470    && before
471      .and_then(|t| t.chars().next_back())
472      .is_some_and(char::is_alphabetic)
473    && after
474      .and_then(|t| t.chars().next())
475      .is_some_and(char::is_alphabetic)
476}

499        T::E(_, attrs, _) => attrs
500          .iter()
501          .find(|(k, _)| *k == name)
502          .map(|(_, v)| (*v).to_string()),
503        T::X(_) => None,
504      }
505    }
506
507    fn children(&self) -> Vec<Self> {
508      match self {
509        T::E(_, _, kids) => kids.clone(),
510        T::X(_) => Vec::new(),
511      }
512    }
513
514    fn text(&self) -> Option<String> {
515      match self {
516        T::X(t) => Some((*t).to_string()),
517        T::E(..) => None,
518      }
519    }
520  }

522  fn tok(t: &'static str) -> T { T::E("XMTok", vec![], vec![T::X(t)]) }
523  fn hint() -> T { T::E("XMHint", vec![("width", "2.8pt")], vec![]) }
524  fn script(role: &'static str, kids: Vec<T>) -> T {
525    T::E("XMApp", vec![("role", role)], vec![T::E(
526      "XMArg",
527      vec![("rule", "Superscript")],
528      kids,
529    )])
530  }
531  fn math(kids: Vec<T>) -> T { T::E("Math", vec![], vec![T::E("XMath", vec![], kids)]) }
532  fn op(pos: &'static str) -> T {
533    T::E(
534      "XMTok",
535      vec![("role", "SUPERSCRIPTOP"), ("scriptpos", pos)],
536      vec![],
537    )
538  }
539  fn marks(texts: &[&str]) -> FormulaMarks {
540    FormulaMarks::Marks(texts.iter().map(|s| (*s).to_string()).collect())

544  /// no letter or digit printed right after them, the formula's end, spacing, punctuation or a symbol.
545  #[test]
546  fn floating_scripts_before_no_letter_or_digit_are_a_mark() {
547    let float = |kids| script("FLOATSUPERSCRIPT", kids);
548    // as digested: `$^{1}\hspace{1mm}$`, `$\,^{3}$`, `${}^{a,\;b}$`, `$^{a})$`, `$^{1}{}^{,}$`, `$\rm ^a$`
549    assert_eq!(
550      read_formula(&math(vec![float(vec![tok("1")]), hint()])),
551      marks(&["1"])
552    );
553    assert_eq!(

555      marks(&["3"])
556    );
557    assert_eq!(
558      read_formula(&math(vec![float(vec![
559        tok("a"),
560        tok(","),
561        hint(),
562        tok("b")
563      ])])),
564      marks(&["a,b"])
565    );
566    assert_eq!(
567      read_formula(&math(vec![float(vec![tok("a")]), tok(")")])),
568      marks(&["a"])
569    );
570    assert_eq!(
571      read_formula(&math(vec![float(vec![tok("1")]), float(vec![tok(",")])])),
572      marks(&["1", ","])
573    );
574    // `$\mbox{ }^1$` (cond-mat0206265): a script after a blank box floats
575    let blank = T::E("XMText", vec![], vec![T::X(" ")]);
576    assert_eq!(
577      read_formula(&math(vec![
578        blank,
579        script("POSTSUPERSCRIPT", vec![tok("1")])
580      ])),
581      marks(&["1"])
582    );
583    // `^1,2` (1912.02075's `A.P.~Orlov$^1,2$`): the comma is the script's base, no word — a mark, its marks before the 2
584    assert_eq!(
585      read_formula(&math(vec![float(vec![tok("1")]), tok(","), tok("2")])),
586      marks(&["1"])
587    );
588    // as parsed: a surviving float under a dual's wrap; a prescript on ")"
589    let dual = T::E("XMDual", vec![], vec![
590      T::E("XMRef", vec![], vec![]),
591      T::E("XMWrap", vec![], vec![
592        T::E("XMApp", vec![("role", "FLOATSUPERSCRIPT")], vec![tok("1")]),
593        hint(),
594      ]),
595    ]);
596    assert_eq!(read_formula(&math(vec![dual])), marks(&["1"]));
597    let pre = T::E("XMApp", vec![], vec![op("pre1"), tok(")"), tok("a")]);
598    assert_eq!(read_formula(&math(vec![pre])), marks(&["a"]));
599  }

601  #[test]
602  fn a_script_before_or_on_printed_ink_is_no_mark() {
603    let float = |kids| script("FLOATSUPERSCRIPT", kids);
604    let post = |kids| script("POSTSUPERSCRIPT", kids);
605    // an isotope: `${\rm ^{3}He}$` as digested, and as parsed
606    assert_eq!(
607      read_formula(&math(vec![float(vec![tok("3")]), tok("H"), tok("e")])),
608      FormulaMarks::Other
609    );
610    let pre = T::E("XMApp", vec![], vec![op("pre2"), tok("He"), tok("3")]);
611    assert_eq!(read_formula(&math(vec![pre])), FormulaMarks::Other);
612    // `^{3}\alpha`: a letter right after the script, its base
613    assert_eq!(
614      read_formula(&math(vec![float(vec![tok("3")]), tok("\u{3b1}")])),
615      FormulaMarks::Other
616    );
617    // a base: `$97403^{1}$`, `$\text{Leon Bello}^{1,3}$`, both as digested and parsed
618    let digits: Vec<T> = ["9", "7", "4", "0", "3"].into_iter().map(tok).collect();
619    let mut kids = digits;
620    kids.push(post(vec![tok("1")]));
621    let on = |base: &str, s: &str| FormulaMarks::OnBase {
622      base:    base.to_string(),
623      scripts: vec![s.to_string()],
624    };
625    assert_eq!(read_formula(&math(kids)), on("97403", "1"));
626    // a script the parser nests under the invisible times of a word, as the digested shape's flat items read
627    // (`$Ann^{1}$`: `⁢(A, n, n¹)`; `$\mathrm{Ann~Able}^{1}$`: `⁢(Ann, Able¹)`; `$Smith-Jones^1$`:
628    // `−(⁢(Smith), ⁢(J, o, n, e, s¹))`; review 90a r2 F1)
629    let times = || {
630      T::E(
631        "XMTok",
632        vec![("meaning", "times"), ("role", "MULOP")],
633        vec![T::X("\u{2062}")],
634      )
635    };
636    let app = |kids| T::E("XMApp", vec![], kids);
637    let on_base = |base, s| app(vec![op("post1"), base, tok(s)]);
638    let ann = |n| app(vec![times(), tok("A"), tok("n"), on_base(tok("n"), n)]);
639    assert_eq!(read_formula(&math(vec![ann("1")])), on("Ann", "1"));
640    let ann_digested = vec![tok("A"), tok("n"), tok("n"), post(vec![tok("1")])];
641    assert_eq!(read_formula(&math(ann_digested)), on("Ann", "1"));
642    let able = app(vec![times(), tok("Ann"), on_base(tok("Able"), "1")]);
643    assert_eq!(read_formula(&math(vec![able])), on("AnnAble", "1"));
644    let able_digested = vec![
645      tok("A"),
646      tok("n"),
647      tok("n"),
648      hint(),
649      tok("A"),
650      tok("b"),
651      tok("l"),
652      tok("e"),
653      post(vec![tok("1")]),
654    ];
655    assert_eq!(read_formula(&math(able_digested)), on("AnnAble", "1"));
656    let minus = T::E(
657      "XMTok",
658      vec![("meaning", "minus"), ("role", "ADDOP")],
659      vec![T::X("-")],
660    );
661    let jones = app(vec![
662      times(),
663      tok("J"),
664      tok("o"),
665      tok("n"),
666      tok("e"),
667      on_base(tok("s"), "1"),
668    ]);
669    let smith = app(vec![times(), tok("Smith")]);
670    assert_eq!(
671      read_formula(&math(vec![app(vec![minus, smith, jones])])),
672      on("Smith-Jones", "1")
673    );
674    // a prescript on an accented letter, `$^{1}\mathrm{\acute{e}}$`: an application holding no script prints one item,
675    // a letter (review 90a r2's probes AFs20, ANs20), as digested and as parsed
676    let acute = || {
677      T::E(
678        "XMTok",
679        vec![("name", "acute"), ("role", "OVERACCENT")],
680        vec![T::X("\u{b4}")],
681      )
682    };
683    let accented = app(vec![acute(), T::E("XMArg", vec![], vec![tok("e")])]);
684    assert_eq!(
685      read_formula(&math(vec![float(vec![tok("1")]), accented])),
686      FormulaMarks::Other
687    );
688    let pre_accented = app(vec![op("pre1"), app(vec![acute(), tok("e")]), tok("1")]);
689    assert_eq!(read_formula(&math(vec![pre_accented])), FormulaMarks::Other);
690    // a two-dimensional layout is one printed item, as digested and as parsed (review 90a r3 F1; Perl `Post/MathML.pm`
691    // `mfrac`, `msqrt`, `mroot`, `menclose`, `mover`): a script under a fraction's bar, a root's sign, a box or an
692    // accent is no script the formula ends on — `\frac{a}{b^{1}}`, `\sqrt{x^{1}}`, `\sqrt[3]{x^{1}}`, `\boxed{x^{1}}`,
693    // `\hat{x^{1}}`, `\stackrel{a}{b^{1}}` (a script set over its base, `scriptpos` `mid`)
694    let arg = |kids| T::E("XMArg", vec![], kids);
695    let with =
696      |name: &'static str, attrs: Vec<(&'static str, &'static str)>, text: &'static str| {
697        T::E(
698          name,
699          attrs,
700          if text.is_empty() {
701            vec![]
702          } else {
703            vec![T::X(text)]
704          },
705        )
706      };
707    let frac = || with("XMTok", vec![("meaning", "divide"), ("role", "FRACOP")], "");
708    let sqrt = || with("XMTok", vec![("meaning", "square-root")], "");
709    let root = || with("XMTok", vec![("meaning", "nth-root")], "");
710    let hat = || with("XMTok", vec![("name", "hat"), ("role", "OVERACCENT")], "^");
711    let mid = T::E(
712      "XMTok",
713      vec![("role", "SUPERSCRIPTOP"), ("scriptpos", "mid1")],
714      vec![],
715    );
716    let x1 = || vec![tok("x"), post(vec![tok("1")])];
717    let boxed = |kids| T::E("XMApp", vec![("enclose", "box")], kids);
718    let layouts = [
719      // (as digested, as parsed)
720      (
721        app(vec![
722          frac(),
723          arg(vec![tok("a")]),
724          arg(vec![tok("b"), post(vec![tok("1")])]),
725        ]),
726        app(vec![frac(), tok("a"), on_base(tok("b"), "1")]),
727      ),
728      (
729        app(vec![sqrt(), arg(x1())]),
730        app(vec![sqrt(), on_base(tok("x"), "1")]),
731      ),
732      (
733        app(vec![root(), arg(vec![tok("3")]), arg(x1())]),
734        app(vec![root(), tok("3"), on_base(tok("x"), "1")]),
735      ),
736      (
737        T::E("XMArg", vec![("enclose", "box")], x1()),
738        boxed(vec![op("post1"), tok("x"), tok("1")]),
739      ),
740      (
741        app(vec![hat(), arg(x1())]),
742        app(vec![hat(), on_base(tok("x"), "1")]),
743      ),
744      (
745        T::E("XMApp", vec![("role", "RELOP")], vec![
746          mid.clone(),
747          arg(vec![tok("b"), post(vec![tok("1")])]),
748          arg(vec![tok("a")]),
749        ]),
750        T::E("XMApp", vec![("role", "RELOP")], vec![
751          mid,
752          on_base(tok("b"), "1"),
753          tok("a"),
754        ]),
755      ),
756    ];
757    for (digested, parsed) in layouts {
758      assert_eq!(read_formula(&math(vec![digested])), FormulaMarks::Other);
759      assert_eq!(read_formula(&math(vec![parsed])), FormulaMarks::Other);
760    }
761    // a script on the last item of a parsed list, `$a,b^{1}$`: its base "a,b" is no word, the formula's text (as digested)
762    let list = T::E("XMDual", vec![], vec![
763      T::E("XMRef", vec![], vec![]),
764      T::E("XMWrap", vec![], vec![
765        tok("a"),
766        tok(","),
767        on_base(tok("b"), "1"),
768      ]),
769    ]);
770    assert_eq!(read_formula(&math(vec![list])), FormulaMarks::Other);
771    let text = T::E("XMText", vec![], vec![T::X("Leon Bello")]);
772    assert_eq!(
773      read_formula(&math(vec![
774        text.clone(),
775        post(vec![tok("1"), tok(","), tok("3")])
776      ])),
777      on("Leon Bello", "1,3")
778    );
779    let parsed = T::E("XMApp", vec![], vec![op("post1"), text, tok("1,3")]);
780    assert_eq!(read_formula(&math(vec![parsed])), on("Leon Bello", "1,3"));
781    // a subscript is the base's (`$G_2'$`, 2208.14409)
782    assert_eq!(
783      read_formula(&math(vec![
784        tok("G"),
785        script("POSTSUBSCRIPT", vec![tok("2")]),
786        post(vec![tok("\u{2032}")])
787      ])),
788      on("G2", "\u{2032}")
789    );
790    // a base no word: `$a+b^{2}$`; no superscript: `$_{1}$`; no script: `$x$`
791    assert_eq!(
792      read_formula(&math(vec![
793        tok("a"),
794        tok("+"),
795        tok("b"),
796        post(vec![tok("2")])
797      ])),
798      FormulaMarks::Other
799    );
800    assert_eq!(
801      read_formula(&math(vec![script("FLOATSUBSCRIPT", vec![tok("1")])])),
802      FormulaMarks::Other
803    );
804    assert_eq!(read_formula(&math(vec![tok("x")])), FormulaMarks::Other);
805  }
