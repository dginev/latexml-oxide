33/// A definition made while a rehearsal digests a field's tokens ([`FieldNotes`]): what no box says, a definition printing
34/// nothing.
35#[derive(Debug, Clone, PartialEq, Eq)]
36pub struct FieldDeclaration {
37  /// where it starts: the position of the field's token that made it, a definer or the call of a macro whose expansion is
38  /// the definer, or before it what leads to it ([`FieldNotes::lead`]: `\global`, `\long`, `\expandafter`)
39  pub start:    u16,
40  /// one past the last position that invocation read
41  pub end:      u16,
42  /// the meanings it assigned, in order, each once
43  pub definees: Vec<SymStr>,
44  /// whether every one of them was assigned globally
45  pub global:   bool,
46  /// the groups open around it, in the field
47  pub depth:    usize,
48  /// its call may still be being read ([`field_notes_after`])
49  open:         bool,
50  /// the position of the invocation that made it, which a macro's expansion shares
51  call:         u16,
52  /// the field's call it ran at, its token being one of the body of a macro the field defined ([`in_a_definition`]): that
53  /// token's position says where it was written, the call where it runs ([`FieldNotes::last_call`])
54  run_at:       Option<(u16, u16)>,
55  /// one past the last position its call read, the call's inner macros reading on past `end` (babel's robust
56  /// `\foreignlanguage`, `\ifthenelse`'s branches: 2606.14878, 2605.24599); its span's end once settled;
57  /// 0 for one made after its call printed
58  read:         u16,
59}
60
61impl FieldDeclaration {
62  /// Whether what it defined outlives the call that made it: global, or in no group of the field. A group of the call's own
63  /// ends with it (`\begingroup\def\thefootnote{…}\footnote[0]{…}\endgroup`: 0903.4816, 2311.17684, 2602.15181).
64  fn outlives_its_call(&self) -> bool { self.global || self.depth == 0 }
65}

76/// A creator call of the frontmatter API a block's own expansion makes (`\lx@add@creator[keys]{name}`, bound to the
77/// rehearsal's `\lx@fm@obs@creator`, which prints the name: PoS's and JHEP's `\speaker{…}`, `[role=speaker]`; ams's
78/// `\contrib[…]{…}`, `[role=contributor, name={…}]`): its keyvals, Perl's API argument, which no box holds, and the
79/// positions its name's tokens take (`events::creator_roles`).
80#[derive(Debug, Clone, PartialEq)]
81pub struct FieldCreator {
82  /// the keyvals as tokens, as the call gave them ([`KeyVals::revert`]), but its `annotations`: what the emitter writes on
83  /// Perl's call, with `annotations` once
84  pub keys:        Tokens,
85  /// the role they give (the KeyVals' last `role`, as `\lx@add@frontmatter@now` reads it)
86  pub role:        Option<String>,
87  /// the labels they give (the last `annotations`), which the emitter's labels join
88  pub annotations: Option<Tokens>,
89  pub start:       u16,
90  /// one past the last position
91  pub end:         u16,
92}

94impl FieldCreator {
95  /// The labels its `annotations` give the creator its name makes, Perl's `\lx@add@creator[annotations={ed}]` (the reading's
96  /// requests of that creator, `resolver::Input::Block::calls`; the emitter's keys, `emitter::emit::creator_keys`), cleaned as
97  /// the API cleans them, Perl's `cleanFrontmatterLabels` (Base_Utility.pool.ltxml:360-372: `\rm` out, `\ref{x}` the label
98  /// `LABEL:x`, spaces `_`).
99  pub fn labels(&self) -> Vec<String> {
100    self
101      .annotations
102      .as_ref()
103      .map(|labels| clean_frontmatter_labels(&labels.to_string(), ""))
104      .unwrap_or_default()
105  }

170  frames_heard:      usize,

175  /// invocations that moved it, making no box: an assignment of the register, whatever spells it
176  /// (`\addtocounter{footnote}{-2}`, `\setcounter`, `\global\advance\c@footnote`), which no event says. The rehearsal's

225/// Note the field token at `pos` heard ([`FieldNotes::environments`]). One whose meaning is `\begin` or `\end` says it, a `\let` copy
226/// too, a raw environment's (`\newenvironment`, `{CJK*}`, 2609.05616) that no frame says, its extent as far as what runs at its
227/// position reads (the environment's arguments: `\begin{tabular}{c}`); else the frames: one what a macro's expansion opened (a pair
228/// of macros), or an alignment's cell read, says by the next token heard. A frame count unchanged since the last says no environment
229/// changed either, the read a letter of a name takes.
230fn note_environment(token: &Token, pos: u16, reach: u16, expanded: bool) {
231  let frames = state::get_frame_depth();
232  let by_meaning = if token.get_catcode() == Catcode::CS {
233    lookup_definition(token).ok().flatten().and_then(|meaning| {
234      meaning.get_cs().with_cs_name(|name| match name {
235        "\\begin" => Some(true),
236        "\\end" => Some(false),
237        _ => None,
238      })
239    })
240  } else {
241    None
242  };
243  FIELD_NOTES.with(|notes| {
244    let notes = &mut *notes.borrow_mut();
245    // (what runs at the position of a `\begin` or `\end` heard is read too, its frame count unchanged: `\begin{small}`'s
246    // `\begingroup` opens the frame, its `\lx@setcurrenvir` then the environment, which the `\begin` said — left unread, the
247    // next group's frame (`Universit{\`a}`) heard it as an environment opened there, which the `\end` closed: the line's run
248    // took that `\end` and not its `\begin`, 2512.19748)
249    let at_said = notes.meaning_heard.is_some_and(|(at, _)| at == pos);
250    if frames != notes.frames_heard || at_said {
251      notes.frames_heard = frames;
252      let open = environments_open();
253      if open != notes.open_heard {
254        // (a macro's expansion did it, by this token; or what an alignment's cell read before it, past the last token heard; else
255        // this token's own invocation. What a `\begin` or `\end` heard did, it said)
256        let (start, end) = match notes.token_heard {
257          Some((_, _, true, true)) => (pos, pos),
258          Some((heard, _, true, false)) => (heard, pos),
259          Some((_, read_to, false, _)) if read_to < pos => (read_to, pos),
260          _ => (pos, reach.max(pos + 1)),
261        };
262        let opens = open > notes.open_heard;
263        // (a position whose `\begin` or `\end` was heard said it: a macro's `\begin{tikzpicture}…\end{tikzpicture}`, its tokens
264        // all at its call's position, closes once, 2210.03748; and what its reading read, an `{array}`'s arguments in a formula)
265        let said = notes
266          .meaning_heard
267          .is_some_and(|(at, read)| at == pos || (at <= start && start < read));
268        if start != end && !said {
269          for _ in 0..open.abs_diff(notes.open_heard) {
270            notes
271              .environments
272              .push(EnvironmentStep { start, end, opens });
273          }
274        }
275        notes.open_heard = open;
276      }
277    }
278    // (what runs at a `\begin`'s position reads its environment's arguments: its extent)
279    if let Some((at, read)) = &mut notes.meaning_heard
280      && *at == pos
281    {
282      *read = (*read).max(reach);
283      for step in notes
284        .environments
285        .iter_mut()
286        .rev()
287        .take_while(|step| step.start == pos)
288      {
289        step.end = step.end.max(reach);
290      }
291    }
292    if let Some(opens) = by_meaning {
293      notes.meaning_heard = Some((pos, reach.max(pos + 1)));
294      notes.environments.push(EnvironmentStep {
295        start: pos,
296        end: reach.max(pos + 1),
297        opens,
298      });
299    }
300    notes.token_heard = Some((pos, reach.max(pos + 1), expanded, by_meaning.is_some()));
301  });
302}

304/// Note a creator call the notes' tokens make ([`FieldCreator`]): its keyvals and the positions of its name's tokens of the
305/// field. A name none of whose tokens is the field's is no name the block's events can say.
306pub fn note_creator_call(keys: &KeyVals, name: &Tokens) -> Result<()> {
307  let positions = name
308    .unlist_ref()
309    .iter()
310    .filter(|t| field_prov_field(t.prov) == 2)
311    .map(|t| field_prov_pos(t.prov));
312  let (Some(start), Some(end)) = (positions.clone().min(), positions.max()) else {
313    return Ok(());
314  };
315  // (its keys but its annotations, which the emitter writes once, joined with the reading's labels: a key given twice is a
316  // list to Perl's KeyVals, KeyVals.pm:294-300, not the last value)
317  let mut others = keys.clone();
318  others.set_value("annotations", ArgWrap::None, false)?;
319  let creator = FieldCreator {
320    keys: others.revert()?,
321    role: keys
322      .get_value("role")
323      .map(|role| role.to_string())
324      .filter(|role| !role.is_empty()),
325    annotations: keys
326      .get_value("annotations")
327      .and_then(|labels| labels.clone().owned_tokens()),
328    start,
329    end: end.saturating_add(1),
330  };
331  FIELD_NOTES.with(|notes| notes.borrow_mut().creators.push(creator));
332  Ok(())
333}

345/// Begin the notes of the tokens about to be digested, `tokens` the field's (a rehearsal's; rehearsals are not nested,
346/// their journals not being: `state::begin_assignment_journal`). Their positions' end bounds what the last of them read.
347pub fn field_notes_begin(tokens: &Tokens, block_breaks: &[(&str, BreakKind)]) {
348  let mut field: Vec<Option<Token>> = Vec::new();
349  for token in tokens.unlist_ref().iter().filter(|t| t.prov != 0) {
350    let pos = usize::from(field_prov_pos(token.prov));
351    if field.len() <= pos {
352      field.resize(pos + 1, None);
353    }
354    field[pos].get_or_insert(*token);
355  }
356  let end = u16::try_from(field.len()).unwrap_or(u16::MAX);
357  FIELD_NOTES.with(|notes| {
358    *notes.borrow_mut() = FieldNotes {
359      end,
360      field,
361      errors_seen: REPORT.borrow().error,
362      depth: state::get_frame_depth(),
363      frames_heard: state::get_frame_depth(),
364      open_heard: environments_open(),
365      stack: token_stack_depth(),
366      footnote: footnote_counter(),
367      block_breaks: block_breaks
368        .iter()
369        .map(|(command, kind)| (T_CS!(*command), *kind))
370        .collect(),
371      ..FieldNotes::default()
372    };
373  });
374  set_field_notes_hook(Some(field_notes_after));
375}

377/// End the notes ([`field_notes_begin`]), each declaration's span a run of the field a replay can read
378/// ([`settle_declarations`]).

381  let mut notes = FIELD_NOTES.with(RefCell::take);
382  if !notes.declarations.is_empty() {
383    settle_declarations(&mut notes);
384  }
385  notes
386}

388/// The spans of a block's declarations as runs the emitter can replay, in order, none reading what another reads. Where the
389/// invocation's reach is true of the reading but not of the content:
390/// - **A body's definition.** One in the body of a macro defined in the field runs at the macro's call, its tokens still at their
391///   own positions, inside the braces of what defined them: `\def\m{\newcount\qq}\m`'s `\qq` read [3,7] over `\m`'s [0,6]. Its span is
392///   that call, wherever the call is ([`FieldDeclaration::run_at`]): `\def\s{\def\t{1}} Ann Bob\s`'s `\t` holds `\s`, not " Ann Bob";
393///   `\def\m{\def\n{\def\o{1}}}\m\n`'s `\n` holds `\m` and its `\o` holds `\n`; what one call defines is one declaration
394///   (`\def\x{\def\y{1}\def\z{2}}\x`). With no call known (`\aftergroup`), it starts past that definition; with nothing left of its own
395///   its names go with the next, or the last.
396/// - **A peek.** A macro testing the next token read as far as it (`\mstar\mfile`, whose `\@ifstar` peeked at `\mfile`, itself a
397///   declaration's call): outside the braces of the earlier, the later one holds its own call, and the earlier ends there.
398/// - **Brace balance.** A definer whose next token is no later one of the field read as far as the call around it read: the end of an
399///   argument, `\textbf{\def\x{1}}`, `\AtBeginDocument{\def\x{1}\def\y{2}}`'s last, holds a closing brace of what opened before it. The
400///   span ends before it.
401#[cold]
402#[inline(never)]
403fn settle_declarations(notes: &mut FieldNotes) {
404  let field = std::mem::take(&mut notes.field);
405  let step = |pos: u16| {
406    field
407      .get(usize::from(pos))
408      .copied()
409      .flatten()
410      .map_or(0, |t| t.group_step())
411  };
412  let mut settled: Vec<FieldDeclaration> = Vec::with_capacity(notes.declarations.len());
413  // (the spans as noted: a definition holds its body's tokens wherever those run)
414  let mut written: Vec<(u16, u16)> = Vec::with_capacity(notes.declarations.len());
415  let mut waiting: Vec<SymStr> = Vec::new();
416  let add = |to: &mut Vec<SymStr>, names: Vec<SymStr>| {
417    for name in names {
418      if !to.contains(&name) {
419        to.push(name);
420      }
421    }
422  };
423  for mut declaration in notes.declarations.drain(..) {
424    declaration.end = declaration.end.max(declaration.read);
425    let at = declaration.start;
426    let in_a_body = written
427      .iter()
428      .any(|&(s, e)| s <= at && at < e && (s..at).map(step).sum::<i32>() > 0);
429    written.push((declaration.start, declaration.end));
430    if in_a_body && let Some((call, reach)) = declaration.run_at {
431      declaration.start = call;
432      declaration.end = declaration.end.max(reach);
433      if let Some(last) = settled.last_mut()
434        && last.start == call
435      {
436        last.end = last.end.max(declaration.end);
437        add(&mut last.definees, declaration.definees);
438        continue;
439      }
440    }
441    if let Some(last) = settled.last_mut()
442      && declaration.start < last.end
443    {
444      let inside = (last.start..declaration.start).map(step).sum::<i32>() > 0;
445      if inside {
446        declaration.start = last.end;
447      } else if declaration.start > last.start {
448        last.end = declaration.start;
449      } else {
450        add(&mut last.definees, declaration.definees);
451        continue;
452      }
453    }
454    if declaration.end <= declaration.start {
455      waiting.append(&mut declaration.definees);
456      continue;
457    }
458    if !waiting.is_empty() {
459      let mine = std::mem::replace(&mut declaration.definees, std::mem::take(&mut waiting));
460      add(&mut declaration.definees, mine);
461    }
462    settled.push(declaration);
463  }
464  for declaration in &mut settled {
465    let mut depth = 0i32;
466    for pos in declaration.start..declaration.end {
467      depth += step(pos);
468      if depth < 0 {
469        declaration.end = pos;
470        break;
471      }
472    }
473  }
474  if let Some(last) = settled.last_mut() {
475    add(&mut last.definees, waiting);
476  }
477  notes.declarations = settled;
478  notes.field = field;
479}

489/// Whether `token`'s meaning reads nothing of the input by its parameters: a primitive naming no definee, whatever it
490/// assigns on its own account (`\selectfont`, `tex_fonts::pickup_font`: `\font@name` and the font it loads). A definer reads
491/// the name it defines.
492fn names_nothing(token: &Token) -> bool {
493  state::lookup_meaning(token)
494    .and_then(|meaning| meaning.to_definition())
495    .is_none_or(|meaning| {
496      meaning
497        .get_parameters()
498        .is_none_or(|params| params.get_parameters().is_empty())
499    })
500}

502/// `token`'s meaning when a macro.
503fn macro_of(token: &Token) -> Option<Rc<Expandable>> {
504  match state::lookup_meaning(token) {
505    Some(Stored::Expandable(meaning)) => Some(meaning),
506    _ => None,
507  }

510/// `reach`, or past the group the call at `start` opened before it and had not closed: an opening brace a macro reads is the start
511/// of an undelimited argument, which TeX reads whole to its matching close, though the cover seen so far stops at the brace
512/// (icrc2009's raw `\IEEEauthorblockN{J.K. de Jong …}`, whose expansion's `\gdef`s are made after it read the argument, 1111.2718).
513/// A span ending at that brace would be replayed as the call cut from its argument, with the names inside it printed by the
514/// call's box. A reach that closes a group opened before `start` (`\textbf{\def\x{1}}`) is `settle_declarations`'s.
515fn balanced_reach(field: &[Option<Token>], start: u16, reach: u16) -> u16 {
516  let step = |pos: usize| {
517    field
518      .get(pos)
519      .copied()
520      .flatten()
521      .map_or(0, |t| t.group_step())
522  };
523  let mut depth = 0i32;
524  for pos in usize::from(start)..usize::from(reach).min(field.len()) {
525    depth += step(pos);
526    if depth < 0 {
527      return reach;
528    }
529  }
530  let mut end = usize::from(reach);
531  while depth > 0 && end < field.len() {
532    depth += step(end);
533    end += 1;
534  }
535  if depth > 0 {
536    reach
537  } else {
538    u16::try_from(end).unwrap_or(reach)
539  }
540}

542/// What the field hands a call at `pos` reading up to `reach`: its arguments' tokens.
543fn handed_to_a_call(field: &[Option<Token>], pos: u16, reach: u16) -> &[Option<Token>] {
544  field
545    .get(usize::from(pos) + 1..usize::from(reach).min(field.len()))
546    .unwrap_or(&[])
547}

549/// The names a macro the document defined holds in its body, and those a macro of the same kind in its body holds: what a kernel's or
550/// a binding's macro it calls may be defining for it (`\newcommand\m{\newcount\qq}`, `\newcommand\m{\@ifundefined{qq}{\def\qq{1}}{}}`).
551/// A name built at run time is in no body (`\@namedef{qq}{1}`). None for a macro of a `.sty` or `.cls` it loaded: a raw package's body names its own
552/// state, which a run of the kernel's assigns for it (`\makecell`'s `\tabg@pe`, `\t@bset`, 2203.03195, 2502.11387).
553#[cold]
554#[inline(never)]
555fn names_in_a_papers_macro(call: &Rc<Expandable>) -> Vec<SymStr> {
556  let mut names: Vec<SymStr> = Vec::new();
557  if call.get_origin() != DefinitionOrigin::Document {
558    return names;
559  }
560  let mut bodies = vec![Rc::clone(call)];
561  let mut read = 0usize;
562  while let Some(macro_) = bodies.pop() {
563    let Some(ExpansionBody::Tokens(body)) = macro_.expansion.as_ref() else {
564      continue;
565    };
566    for token in body.unlist_ref() {
567      let key = state::meaning_key(token);
568      if names.contains(&key) {
569        continue;
570      }
571      names.push(key);
572      if let Some(meaning) = macro_of(token)
573        && meaning.get_origin() == DefinitionOrigin::Document
574        && read < MACROS_READ_FOR_NAMES
575      {
576        read += 1;
577        bodies.push(meaning);
578      }
579    }
580  }
581  names
582}

584/// How many macros [`names_in_a_papers_macro`] reads, past which a name is too far from the call to be its.
585const MACROS_READ_FOR_NAMES: usize = 64;

587/// Keep of what `token`, invoked at the field position `pos` and reading up to `reach`, assigned what the paper declares. The
588/// field's own token there, a definer invoked directly (`\def`, `\let`, `\newcommand`), declares what it assigns. So does the
589/// expansion of a call there of a macro the paper or a file it read defined, unless it names nothing. Within the run of a kernel's or
590/// a binding's macro that call reached (`inside`, [`close_runs`]) the kernel assigns on its own account (`\size@update`,
591/// `\@let@token`), so only what the call's body or its arguments name is the paper's: 2608.17767's `\newcommand\lxaff{\small\itshape}`
592/// declares nothing, `\itshape` being made by `\reinstall@nfss@defs` at `\begin{document}`; `\newcommand\m{\newcount\qq}`
593/// and `\newcommand\m{\@ifundefined{qq}{\def\qq{1}}{}}` declare `\qq` ([`names_in_a_papers_macro`]). A kernel's or a binding's
594/// macro called there assigns on its own account likewise (a font switch's `\f@series`, `\centering`'s `\\`, an environment's
595/// `\@currenvir`): only the names the field handed it are the paper's (`\newcount\nn`, `\newsavebox\bb`, whose arguments the
596/// expansion reads).
597#[cold]
598#[inline(never)]
599fn keep_declared(
600  token: &Token,
601  pos: u16,
602  reach: u16,
603  inside: bool,
604  field: &[Option<Token>],
605  assigned: &mut Vec<MeaningAssigned>,
606) {
607  let Some(Some(call)) = field.get(usize::from(pos)) else {
608    assigned.clear();
609    return;
610  };
611  let direct = call == token;
612  match state::lookup_meaning(call) {
613    Some(Stored::Expandable(_)) if direct => assigned.clear(),
614    Some(Stored::Expandable(definition)) if definition.get_origin().is_latexml_owned() => {
615      let handed = handed_to_a_call(field, pos, reach);
616      assigned.retain(|assignment| {
617        handed
618          .iter()
619          .flatten()
620          .any(|t| state::meaning_key(t) == assignment.key)
621      });
622    },
623    Some(Stored::Expandable(definition)) => {
624      if inside {
625        let named = names_in_a_papers_macro(&definition);
626        let handed = handed_to_a_call(field, pos, reach);
627        assigned.retain(|assignment| {
628          named.contains(&assignment.key)
629            || handed
630              .iter()
631              .flatten()
632              .any(|t| state::meaning_key(t) == assignment.key)
633        });
634      } else if names_nothing(token) {
635        assigned.clear();
636      }
637    },
638    _ if direct && !names_nothing(token) => {},
639    _ => assigned.clear(),
640  }
641}

643/// Close the runs of a kernel's or a binding's macro read by now: an invocation that was the run's took a token above what the
644/// pushback held before the expansion, and left it at least that (`held`, past what it read; its tokens take the position of
645/// the call they came from, `token::inherit_field_prov`, so nothing else tells). One that left it below came after the run: a
646/// run's last token reading on past it read what the field handed it.
647fn close_runs(notes: &mut FieldNotes, held: usize) {
648  while notes
649    .kernel_runs
650    .last()
651    .is_some_and(|(_, before)| held < *before)
652  {
653    notes.kernel_runs.pop();
654  }
655}

657/// What `test` says of `token`'s meaning when a macro; false else.
658fn with_expandable(token: &Token, test: impl Fn(&Expandable) -> bool) -> bool {
659  macro_of(token).is_some_and(|meaning| test(&meaning))
660}

662/// Note the expansion of the field token `token`, its tokens about to go on the input: a kernel's or a binding's macro a paper's
663/// macro reached opens a run ([`keep_declared`]), its arguments of the field's own positions none of it (`\mbox{\def\x{1}}`,
664/// `\parbox`); a block break its binding declares is noted; it is kept for what leads to the next invocation ([`read_lead`]).
665/// One for every tie of a block of a thousand names (1507.02068): a meaning is looked up only for a token no field holds there.
666fn field_expansion(token: &Token, cover: FieldCover, expansion: &Tokens) {
667  let pos = field_prov_pos(token.prov);
668  note_environment(token, pos, cover.end, true);
669  FIELD_NOTES.with(|notes| {
670    let notes = &mut *notes.borrow_mut();
671    let own = notes
672      .field
673      .get(usize::from(pos))
674      .copied()
675      .flatten()
676      .as_ref()
677      == Some(token);
678    if own && !in_a_definition(&notes.declarations, pos) {
679      let call = (pos, cover.end.max(pos + 1));
680      notes.last_call = Some(call);
681      // (a macro of no parameters: a call's cover past its arguments is not known at the field's end)
682      if !notes.expansions.iter().any(|e| e.at == pos)
683        && with_expandable(token, |meaning| {
684          meaning.get_origin() == DefinitionOrigin::Document
685            && meaning
686              .paramlist
687              .as_ref()
688              .is_none_or(|params| params.get_parameters().is_empty())
689        })
690      {
691        notes.expansions.push(HeldExpansion {
692          at:     call.0,
693          end:    call.1,
694          tokens: expansion.clone(),
695        });
696      }
697      if notes.any_open {
698        for declaration in &mut notes.declarations {
699          declaration.open &= declaration.run_at.is_none_or(|run_at| run_at == call);
700        }
701      }
702    }
703    note_call_read(notes, token, pos, cover.end, false);
704    let owned = !own && with_expandable(token, |meaning| meaning.get_origin().is_latexml_owned());
705    if !notes.kernel_runs.is_empty() || owned {
706      let held = pushback_len();
707      close_runs(notes, held);
708      if owned && notes.kernel_runs.last().is_none_or(|(at, _)| *at != pos) {
709        notes.kernel_runs.push((pos, held));
710      }
711    }
712    if !notes.block_breaks.is_empty()
713      && let Some((_, kind)) = notes
714        .block_breaks
715        .iter()
716        .find(|(command, _)| command == token)
717    {
718      notes.breaks.push(FieldBreak {
719        at:      pos,
720        command: *token,
721        kind:    *kind,
722      });
723    }
724    notes.expanded = match notes.expanded {
725      Some((start, last)) if last == *token => Some((start.min(pos), last)),
726      _ => Some((pos, *token)),
727    };
728  });
729}

732fn in_a_definition(declarations: &[FieldDeclaration], pos: u16) -> bool {
733  declarations
734    .iter()
735    .any(|declaration| declaration.start < pos && pos < declaration.end)
736}

738/// Note that the field token `token` at `pos` read as far as `end`: a token of a call's expansion (its position) or of its
739/// arguments (inside its reach) reads on for that call, any other of the field's own tokens starts the next. A declaration still
740/// open of that call reads as far ([`FieldDeclaration::read`]).
741fn note_call_read(notes: &mut FieldNotes, token: &Token, pos: u16, end: u16, boxed: bool) {
742  notes.call_read = match notes.call_read {
743    Some(call) if call.at == pos => Some(CallRead {
744      read: call.read.max(end),
745      ..call
746    }),
747    Some(call) if call.at < pos && pos < call.read => Some(CallRead {
748      printed: call.printed || boxed,
749      ..call
750    }),
751    _ if notes
752      .field
753      .get(usize::from(pos))
754      .copied()
755      .flatten()
756      .as_ref()
757      == Some(token) =>
758    {
759      Some(CallRead {
760        at:      pos,
761        read:    end.max(pos + 1),
762        printed: false,
763      })
764    },
765    other => other,
766  };
767  if notes.any_open {
768    for declaration in &mut notes.declarations {
769      if declaration.open
770        && declaration.call == pos
771        && declaration.run_at.is_none()
772        && declaration.read > 0
773      {
774        declaration.read = declaration.read.max(end);
775      }
776    }
777  }
778}

780/// Where what leads to an invocation starts ([`FieldNotes::lead`]): the prefixes waiting for it, and the `\expandafter`s last expanded
781/// before it, the inner ones first (`\expandafter\def\expandafter\y…`); any other expansion ends those, a prefix waiting past it.
782fn read_lead(waiting: Option<u16>, expanded: Option<(u16, Token)>) -> Option<u16> {
783  match expanded {
784    Some((pos, token)) if with_expandable(&token, |meaning| meaning.cs == *TOKEN_EXPANDAFTER) => {
785      Some(waiting.map_or(pos, |lead| lead.min(pos)))
786    },
787    _ => waiting,
788  }

793  match invocation {
794    FieldInvocation::Digested(result) => field_invocation(token, cover, result),
795    FieldInvocation::Expanded(expansion) => field_expansion(token, cover, expansion),
796  }

799/// Note the invocation of the field token `token` that read as far as `cover` and made `result`: the errors raised since
800/// the last one, and, making no box, the meanings it declared ([`field_declares`]), as far as its call read — a macro's
801/// arguments, the pending cover (its expansion's tokens inherit its position). A box made inside the reach of a declaration
802/// still open (a macro printing ink, its internal `\let`s) takes it back: that box's span holds the call, and a
803/// declaration of it would print it twice. The first token read past its reach closes it.
804fn field_invocation(token: &Token, cover: FieldCover, result: &[Digested]) {
805  let pos = field_prov_pos(token.prov);
806  note_environment(token, pos, cover.end, false);
807  // (the opening shift's invocation holds the box its closing shift made: the formula starts here)
808  for digested in result {
809    if matches!(digested.data(), DigestedData::Whatsit(w) if w.borrow().is_math()) {
810      digested.try_set_property(FIELD_FORMULA_START_PROPERTY, i64::from(pos));
811    }
812  }
813  // (a letter of a name, nearly every invocation: it makes a box, in no run, led by nothing, no error, no declaration its reach
814  // could close, nothing assigned)
815  let errors = REPORT.borrow().error;
816  if !result.is_empty()
817    && state::meaning_log_is_empty()
818    && FIELD_NOTES.with(|notes| {
819      let notes = &mut *notes.borrow_mut();
820      note_call_read(notes, token, pos, cover.end, true);
821      notes.kernel_runs.is_empty()
822        && notes.expanded.is_none()
823        && notes.lead.is_none()
824        && !notes.any_open
825        && errors == notes.errors_seen
826    })
827  {
828    return;
829  }
830  // (what it is read inside of and led by, as the last left them)
831  let waiting = result.is_empty() && state::has_tex_prefixes();
832  let (inside, quiet) = FIELD_NOTES.with(|notes| {
833    let notes = &mut *notes.borrow_mut();
834    if !notes.kernel_runs.is_empty() {
835      close_runs(notes, pushback_len());
836    }
837    let inside = notes.kernel_runs.last().is_some_and(|(at, _)| *at == pos);
838    // (the expansions before it, read at a declaration or by a prefix; the prefixes waiting for it, then for the next)
839    notes.read_expanded = notes.expanded.take();
840    notes.lead_read = notes.lead.take();
841    if waiting {
842      notes.lead = read_lead(notes.lead_read, notes.read_expanded).or(Some(pos));
843    }
844    (inside, errors == notes.errors_seen && !notes.any_open)
845  });
846  // (an invocation printing nothing that moved the footnote counter: where, `FieldNotes::counter_moves`)
847  if result.is_empty() {
848    FIELD_NOTES.with(|notes| {
849      let notes = &mut *notes.borrow_mut();
850      if notes.footnote.is_some() {
851        let now = footnote_counter();
852        if now != notes.footnote {
853          // (one invocation's moves that come back where they started are none: a given `\footnotemark[<n>]`'s
854          // binding sets the counter in a group of its own, `rehearsal::load`, both at the mark's position)
855          if notes
856            .moved_from
857            .last()
858            .is_some_and(|&(at, from)| at == pos && from == now)
859          {
860            notes.moved_from.pop();
861            notes.counter_moves.pop();
862          } else {
863            notes.moved_from.push((pos, notes.footnote));
864            notes.counter_moves.push((pos, cover.end.max(pos + 1)));
865          }
866          notes.footnote = now;
867        }
868      }
869    });
870  }
871  if quiet && state::meaning_log_is_empty() {
872    return;
873  }
874  let mut assigned = FIELD_NOTES.with(|notes| std::mem::take(&mut notes.borrow_mut().assigned));
875  state::swap_meaning_log(&mut assigned);
876  // (the call's whole reach when its expansion invoked this: a macro's arguments. A field's own token read what its cover says,
877  // an `\expandafter` before it having read on past it: `\expandafter\def\expandafter\y\expandafter{\expandafter}Ann`'s `A`)
878  let pending = field_cover_pending();
879  let own = FIELD_NOTES.with(|notes| {
880    notes
881      .borrow()
882      .field
883      .get(usize::from(pos))
884      .copied()
885      .flatten()
886      .as_ref()
887      == Some(token)
888  });
889  // (a field's own token whose next token is no later one of the field read what the call around it read, which shows no more:
890  // `\newcommand\pass[1]{#1\relax}\pass{\def\x{1}}`, `\textbf{\def\x{1}}`, the last of `\AtBeginDocument{\def\x{1}\def\y{2}}`)
891  let pending_here = (!own || cover.end <= pos + 1)
892    && pending.field == cover.field
893    && pending.start <= pos
894    && pos <= pending.end;
895  let reach = if pending_here {
896    pending.end.max(cover.end)
897  } else {
898    cover.end
899  };
900  let drained = pushback_is_empty();
901  let depth = (drained || !assigned.is_empty()).then(state::get_frame_depth);
902  FIELD_NOTES.with(|notes| {
903    let notes = &mut *notes.borrow_mut();
904    // (the field's last token, outside any group: nothing left in the input shows how far it read, which is to the end)
905    let reach = if drained && depth == Some(notes.depth) {
906      reach.max(notes.end)
907    } else {
908      reach
909    };
910    if errors > notes.errors_seen {
911      notes
912        .errors
913        .push((pos, reach.max(pos + 1), errors - notes.errors_seen));
914    }
915    notes.errors_seen = errors;
916    note_call_read(notes, token, pos, reach, !result.is_empty());
917    if notes.any_open {
918      let in_a_body = in_a_definition(&notes.declarations, pos);
919      let stack = notes.stack;
920      let printing = &mut notes.printing;
921      notes.declarations.retain_mut(|declaration| {
922        if declaration.open
923          && let Some((call, reach)) = declaration.run_at
924        {
925          // (a box of its call or of the body it runs prints at that call, which a declaration of it would print twice; one of what
926          // encloses the call, `\textbf{\m}`, does not)
927          if pos >= reach {
928            declaration.open = false;
929          } else if !result.is_empty() && (pos >= call || in_a_body) {
930            if declaration.outlives_its_call() {
931              printing.push(PrintingCall {
932                span:     (call, reach),
933                definees: declaration.definees.clone(),
934                global:   declaration.global,
935              });
936            }
937            return false;
938          }
939        } else if declaration.open {
940          // (a box of what its call digests past its reach: `\begin{CJK*}`'s arguments and body, 2606.12938)
941          let end = declaration.end.max(declaration.read);
942          let past = pos >= end && !field_position_encloses(stack, declaration.call);
943          if past || (pending.field == cover.field && pending.end > end) {
944            declaration.open = false;
945          } else if !result.is_empty() && declaration.start <= pos {
946            if declaration.outlives_its_call() {
947              printing.push(PrintingCall {
948                span:     (declaration.start, end),
949                definees: declaration.definees.clone(),
950                global:   declaration.global,
951              });
952            }
953            return false;
954          }
955        }
956        true
957      });
958      notes.any_open = notes
959        .declarations
960        .iter()
961        .any(|declaration| declaration.open);
962    }
963    if !result.is_empty() || assigned.is_empty() || field_token_encloses(notes.stack) {
964      return;
965    }
966    // (an enclosing invocation still running assigned the others: `\begin{minipage}`'s, which digests its body)
967    assigned.retain(|assignment| assignment.by == *token);
968    keep_declared(token, pos, reach, inside, &notes.field, &mut assigned);
969    if assigned.is_empty() {
970      return;
971    }
972    let global = assigned.iter().all(|assignment| assignment.global);
973    let definees = assigned.drain(..).map(|assignment| assignment.key);
974    match notes.declarations.last_mut() {
975      // (a macro's expansion defining several, one after another: one call's)
976      Some(last) if last.open && last.call == pos => {
977        last.end = balanced_reach(&notes.field, last.start, last.end.max(reach));
978        last.global &= global;
979        for key in definees {
980          if !last.definees.contains(&key) {
981            last.definees.push(key);
982          }
983        }
984      },
985      _ => {
986        let mut once = Vec::new();
987        for key in definees {
988          if !once.contains(&key) {
989            once.push(key);
990          }
991        }
992        let start = read_lead(notes.lead_read, notes.read_expanded).unwrap_or(pos);
993        let end = balanced_reach(&notes.field, start, reach.max(pos + 1));
994        let run_at = notes
995          .last_call
996          .filter(|_| in_a_definition(&notes.declarations, pos));
997        let call = notes
998          .call_read
999          .filter(|call| run_at.is_none() && call.at == pos);
1000        notes.declarations.push(FieldDeclaration {
1001          start,
1002          end,
1003          definees: once,
1004          global,
1005          depth: depth.unwrap_or(0).saturating_sub(notes.depth),
1006          open: true,
1007          call: pos,
1008          run_at,
1009          // (made after its call printed: its span as noted, which `emit::declaration_reads_past` reads, NE5's `\footremember`)
1010          read: call.map_or(
1011            end,
1012            |call| if call.printed { 0 } else { call.read.max(end) },
1013          ),
1014        });
1015        notes.any_open = true;
1016      },
1017    }
1018  });
1019  assigned.clear();
1020  FIELD_NOTES.with(|notes| notes.borrow_mut().assigned = assigned);
1021}
