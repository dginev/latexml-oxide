2//! block's lines by type, labelled entries to the names requesting them, unlabelled lines to the row above, a
3//! line's contacts by their own mark, spelling, order, the group's only author or elimination, and its notes.

7/// The chain [`Resolver::place_line_contacts`] places a print's addresses by.
8#[derive(Debug, Clone, Copy, PartialEq, Eq)]
9pub(super) enum Chain<'a> {
10  /// A line of a block (or an entry's e-mail call): the spelled addresses first, then — the one address left, and
11  /// the one candidate left without an address — by elimination (R10), then the rest of the chain.
12  Line,
13  /// A document's own note (`ltx:pubnote`, `answer_document_notes`): FIDELITY_CHARTER §4 exactly — the author it
14  /// spells, the authors' order, the only author, the corresponding author, the block; no elimination, so an address
15  /// that spells no one is not guessed onto an author (review rev92a_r1 B2: aa's `\offprints{the editors,
16  /// \email{office@y.org}}`). `named` is the creators the name the note prints before the address names, when the
17  /// class prints the address in the note after it ([`Resolver::note_named`], `Policy::email_in_place`): A&A's "Send
18  /// offprint requests to: O. Kochukhov, e-mail: …" designates the corresponding author (main-session decision on 92a
19  /// r1 question a, 0903.3512), so one creator named is the corresponding step's owner, and none or several the
20  /// block's — prose naming no one ("Supported by NSF.") too (an open design point, OD #913); a note printing nothing
21  /// before the address (`named` `None`) leaves the step to the creators' own "Corresponding author" notes. (A call
22  /// the class prints elsewhere, routed, is one author's whole call and never reaches the chain: review rev92a_r4 S1.)
23  DocumentNote { named: Option<&'a [usize]> },
24}

27  /// Contacts of `text` (one per printed address, ruling 3; the text whole when it holds none), owned by `chain`; each
28  /// rule led by `api:document-note` for a document's note.
29  pub(super) fn place_line_contacts(
30    &mut self,
31    text: &str,
32    role: &str,
33    candidates: &[usize],
34    source: (usize, Option<(u16, u16)>),
35    anchor: Option<&Assoc>,
36    chain: Chain<'_>,
37  ) {
38    let led = |rule: String| match chain {
39      Chain::Line => rule,
40      Chain::DocumentNote { .. } => s!("api:document-note {rule}"),
41    };
42    // (by the address reader, which reads a list of local parts sharing one domain where it opens the line or a part
43    // spells an author: `{\it
44    // pzou94, ozel, suresh@gwu.edu}` under three names, 2003.14069 — two of them nameless notes; spelling assigns them,
45    // position where it fails, below)
46    let addresses = addresses_in(text);
47    let addresses = {
48      let spells = |local: &str| self.spelled(&s!("{local}@x.org"), candidates).is_some();
49      let read: Vec<String> = address_runs(text, &spells)
50        .into_iter()
51        .flat_map(|run| run.addresses)
52        .collect();
53      // (where it reads every address the plain reading does and more; a tagged address it leaves where it prints,
54      // `ann@a.org (Ann)`, is one all the same)
55      if read.len() > addresses.len() && addresses.iter().all(|a| read.contains(a)) {
56        read
57      } else {
58        addresses
59      }
60    };
61    if addresses.is_empty() {
62      let (owner, rule) = match candidates {
63        [only] => (Some(*only), "owner:only-author"),
64        _ => (None, "owner:block"),
65      };
66      self.attach(
67        owner,
68        Assoc {
69          role:   role.to_string(),
70          text:   text.to_string(),
71          label:  None,
72          rule:   led(rule.into()),
73          source: Some(source),
74          target: None,
75        },
76        false,
77      );
78      return;
79    }
80    let spelled: Vec<Option<usize>> = addresses
81      .iter()
82      .map(|a| self.spelled(a, candidates))
83      .collect();
84    let unspelled = spelled.iter().filter(|s| s.is_none()).count();
85    let left: Vec<usize> = candidates
86      .iter()
87      .copied()
88      .filter(|c| !spelled.contains(&Some(*c)))
89      .filter(|&c| {
90        !self.out.creators[c]
91          .contacts
92          .iter()
93          .any(|a| a.role == "email")
94      })
95      .collect();
96    // (a list with one address per author that every spelled address agrees with is in author order: FIDELITY_CHARTER
97    // §4, 2026-10-21, OD #805 — `{qkv, riven, jiyicheng.cs, lihuan.cs, cwang85}@zju.edu.cn` under five names, 2603.19610;
98    // 2512.22629, 2305.12998, 2105.11174, 2412.09666)
99    // (and no address the block prints elsewhere spells one of them: `\{xx,yy\}@x.edu` under two names, one of whom
100    // `ann.able@gmail.com` on the next line spells, is no one-per-author list — three addresses for two authors, the
101    // two unspelled the block's, as the old reader's 77p r9 held them)
102    let in_order = addresses.len() == candidates.len()
103      && candidates.len() >= 2
104      && spelled
105        .iter()
106        .zip(candidates)
107        .all(|(s, c)| s.is_none_or(|s| s == *c))
108      && self
109        .block_addresses
110        .iter()
111        .filter(|a| !addresses.contains(a))
112        .all(|a| self.spelled(a, candidates).is_none());
113    let address_count = addresses.len();
114    // (a link printing one address: its anchor the contact's text, its address the target — position 2, review
115    // revfms2a2_9 finding 5; 1607.00911's `\href{mailto: edremets@…}{* edremets@…}`)
116    let anchor = anchor.filter(|_| address_count == 1);
117    for (at, (address, spelled)) in addresses.into_iter().zip(spelled).enumerate() {
118      let (owner, rule) = match spelled {
119        // (a document's note: the chain's first step, a ruling's change of today's — `api:note-spelled`)
120        Some(owner) if matches!(chain, Chain::DocumentNote { .. }) => {
121          (Some(owner), "owner:spelled api:note-spelled")
122        },
123        Some(owner) => (Some(owner), "owner:spelled"),
124        None if in_order => (Some(candidates[at]), "owner:order"),
125        None
126          if chain == Chain::Line && unspelled == 1 && candidates.len() > 1 && left.len() == 1 =>
127        {
128          (Some(left[0]), "owner:elimination")
129        },
130        None => match candidates {
131          [only] => (Some(*only), "owner:only-author"),
132          // (an address pattern, `\{FirstName.LastName\}@SIT.Fraunhofer.de` under three names, is every author's
133          // address and no one's own: the block's, as today's — not the corresponding author's, 2006.12418; where
134          // order or elimination places it, as today's too, 2110.05663)
135          _ if is_pattern_address(&address) => (None, "owner:pattern"),
136          // (the corresponding author's address is one printed alone: an address left over in a print of several is
137          // another author's — `{hong3924.cs10, evan20010126.cs12, …}@nycu.edu.tw` under a corresponding Ching-Chun
138          // Huang, 2512.08223; `{congjing, ryanflin}@uw.edu chris20030909@gmail.com`, 2602.04785; 2604.14972)
139          _ if address_count > 1 => (None, "owner:block"),
140          // (an address two or more names spell is the corresponding author's, or the author a document's note names,
141          // only where they are one of them, else the block's, as the old reader held it for both
142          // (`author_block/email_block_calls.rs` `spelled`, `owners.marked`, deleted at Z18l): `wang@x.org` under X. Wang, Y. Wang and a
143          // corresponding Z. Li — `perfect_kernel_batch64::marked_block_email_line_owner`, 77p r4; A&A's
144          // `\offprints{Z. Li, \email{wang@x.org}}` —
145          // `perfect_kernel_batchnl1::an_address_others_spell_is_not_the_named_authors`)
146          _ => {
147            let found = match chain {
148              Chain::DocumentNote { named: Some(named) } => match named {
149                [one] if candidates.len() > 1 => (Some(*one), "owner:corresponding api:note-name"),
150                _ => (None, "owner:block"),
151              },
152              _ => self.corresponding(candidates),
153            };
154            let spellers = self.spellers(&address, candidates);
155            match found {
156              (Some(c), _) if spellers.len() >= 2 && !spellers.contains(&c) => {
157                (None, "owner:block")
158              },
159              found => found,
160            }
161          },
162        },
163      };
164      self.attach(
165        owner,
166        Assoc {
167          role:   "email".into(),
168          text:   anchor.map_or_else(|| address.clone(), |a| a.text.clone()),
169          label:  None,
170          rule:   led(s!("{rule} ruling3:one-per-print")),
171          source: Some(source),
172          target: anchor.and_then(|a| a.target.clone()),
173        },
174        false,
175      );
176    }
177  }

179  /// Resolve one block (its events under `policy`).
180  pub(super) fn block(
181    &mut self,
182    index: usize,
183    policy: &'static Policy,
184    cautious: bool,
185    keyvals: Option<&str>,
186    events: &[Event],
187  ) {
188    self.out.policies.push((index, policy.name));
189    self.notes_marked = !policy.unmarked_notes;
190    let and_inline = policy.and_inline;
191    // (an IJCAI sectioned block's address section, from the marker its rehearsal held an Error at before the addresses
192    // it prints (`Input::Block`'s `markers`: `\emails`, defined only inside the class's `\author`): the class's own split
193    // reads it, its addresses the API's as off's are, so the reading ends there and no entry goes on over it — it
194    // continued `$^2$Univ B` onto the e-mail line, and wrote "Univ B" twice (the entry by its text, the split's call)
195    // and the addresses as an affiliation, 2501.02356)
196    let events = match address_section(&self.block_markers, events) {
197      Some(end) => &events[..end],
198      None => events,
199    };
200    let events = &read_events(events);
201    let lines = lines_of(events, and_inline);
202    let block_requests: Vec<String> = keyvals
203      .and_then(|kv| attr_value(kv, "annotations"))
204      .map(|labels| {
205        labels
206          .split(',')
207          .map(|l| l.trim().to_string())
208          .filter(|l| !l.is_empty())
209          .collect()
210      })
211      .unwrap_or_default();
212    let group = index;
213    let block_creators_start = self.out.creators.len();
214    self.block_addresses = lines
215      .iter()
216      .flat_map(|line| addresses_in(&events_text(&line.events)))
217      .collect();
218    // (the lines whose leading marks are their names' requests, read as a names line, their notes the names' (revtex's
219    // `\author{$^{1}$Masashi Kimura and $^{2,3}$Takahiro Tanaka}`, 1805.08625; 1405.2400, 2001.00788; `$^{1}$José Edgar
220    // Madriz Aguilar\thanks{…}`, 0710.4115, gr-qc0606119), not labelled affiliations (`names_led_by_marks`)
221    let (led_names, name_shaped) = names_led_by_marks(&lines, &self.outside_labels);
222    // per column: the creators of its last names row
223    let mut row_above: Vec<(usize, Vec<usize>)> = Vec::new();
224    // per column: the creators of its run of names rows, each directly under the one before with no line answering it
225    // (an unmarked institution under them is every name's of the run, user ruling R12: `A$^{1}$, B$^{2}$\\ C$^{3}$\\
226    // Univ` gives Univ to all three, as today's splitter does — 1511.01102's four rows over `(The Nearby Supernova
227    // Factory)`)
228    let mut run_above: Vec<(usize, Vec<usize>)> = Vec::new();
229    // the entry the previous line ended with, which an unmarked line under it continues (an address's second line:
230    // `$^1$Dept of X \\ 1-1-1 Street, City`, 0803.3589; one affiliation, as today's `\\`-joined contact)
231    let mut continues: Option<Continued> = None;
232    let mut first_in_column: Vec<usize> = Vec::new();
233    // the columns whose last line is a names row ending in a separator (a comma, "and"), which the next line continues
234    let mut open_rows: Vec<usize> = Vec::new();
235    // the columns whose names row has a line under it already (its affiliation)
236    let mut answered_rows: Vec<usize> = Vec::new();
237    // the columns whose last line was labelled entries, and how many entries it held
238    let mut labelled_last: Vec<(usize, usize)> = Vec::new();
239    // an unmarked line's affiliation whose group runs on past the line (its places, where the group ends)
240    let mut open_group: Option<(u16, Vec<(usize, usize)>)> = None;
241    // the places of the unmarked affiliation the line before made, which a line after a break inside the same braced
242    // group continues
243    let mut unmarked_before: Option<Vec<(usize, usize)>> = None;
244    // the last line an address list read on to (`place_line_contacts` over the lines of one list)
245    let mut skip_to = 0usize;
246    // the URL lines under every name so far, each with those names: the block's if no name comes after them
247    let mut block_links: Vec<(&Event, Vec<usize>)> = Vec::new();
248    // the last column holding a line that reads as names: a column after it opening with an institution holds no name
249    // of its own, and is the shared affiliation of every name above it (read on first need: only a column's first line
250    // past column 0 asks, so a one-column block never reads every line as names — 1507.02068's 200-name `\author`)
251    let last_named_once = std::cell::OnceCell::new();
252    let last_named = || {
253      *last_named_once.get_or_init(|| {
254        lines
255          .iter()
256          .filter(|line| reads_as_names(&events_text(&line.events)))
257          .map(|line| line.column)
258          .max()
259      })
260    };
261    for (n, line) in lines.iter().enumerate() {
262      if n > 0 && n <= skip_to {
263        continue;
264      }
265      self.line_column = Some((index, line.column));
266      for span in line.events.iter().filter_map(|e| e.span) {
267        self.entry_column.insert((index, span.0), line.column);
268      }
269      let unmarked_above = unmarked_before.take();
270      let unmarked_line_above = unmarked_above.is_some();
271      let continued = open_rows.contains(&line.column);
272      open_rows.retain(|c| *c != line.column);
273      // (the line before in the column was labelled entries: an unmarked name under two or more of them is a new row's —
274      // SYNC row 46, `$^\dagger$Univ A \quad $^\ddagger$Univ B \\ Cat Cole \\ Physics`; a line under one entry goes on
275      // with it, the old reader's N5 (the deleted `add_authors_calls_sectioned`'s unmarked line appended to the entry above), whatever
276      // its words read as: `$^1$The Graduate Center, CUNY\\ Computer Science\\ New York, USA`, 2501.04046; `$^{1}$Institute
277      // of …, Ionosphere and\\ Radio Wave Propagation\\ RAS, …`, hep-ph0511130 — each line one affiliation's)
278      let entries_above = labelled_last
279        .iter()
280        .find(|(c, _)| *c == line.column)
281        .map(|&(_, entries)| entries);
282      let after_labelled = entries_above.is_some();
283      let after_labelled_list = entries_above.is_some_and(|entries| entries >= 2);
284      labelled_last.retain(|(c, _)| *c != line.column);
285      let under_affiliation = answered_rows.contains(&line.column);
286      // (a line that is no names row answers the row above; a names row below takes this back)
287      if !answered_rows.contains(&line.column) && row_above.iter().any(|(c, _)| *c == line.column) {
288        answered_rows.push(line.column);
289      }
290      // (punctuation before a mark does not lead the line: `($^1$) LSIIT …`, 1004.4809)
291      let first_ink = line_first_ink(&line.events);
292      let column_first = !first_in_column.contains(&line.column);
293      if column_first {
294        first_in_column.push(line.column);
295      }
296      // (a column holding no name of its own after the last name is the shared affiliation of every name above it, under
297      // every name as a link printing an institution is: sig-alternate's `\sharedaffiliation` (`\end{tabular}
298      // \begin{tabular}{c}`, a column, `observer::rehearsal`) under its `\alignauthor` columns, the PDF centring it under
299      // all of them — 0712.1016, 1704.03168, the last author's alone before; written inline, `A One \and Bob Two
300      // \end{tabular}\begin{tabular}{c} Univ Shared`, "Univ Shared" a third creator, review revG3 findings 1 and Q3)
301      // (a column naming a collaboration or a team, whole or as the remark on a name speaking for it, is none — the
302      // old reader's collaboration reading, `reads_as_collaboration`, keeps it as the line reads: `\and \\the Solar
303      // Orbiter EPD and Parker Solar Probe ISIS teams`, 2408.02330; `\and Shuang-Nan Zhang\inst{23} on behalf of the
304      // \textit{SVOM} collaboration\inst{24}.`, 2607.01090 — each every author's affiliation before)
305      // (read on the column's own ink, not the text of a contact a call in it adds — `\and Bob Baker\institute{Univ
306      // B}`, OmniBus's whole list one block since P2-1, OD #993: "Bob Baker" is the column's name, "Univ B" his contact;
307      // read with the contact the line was "Bob Baker Univ B", an institution, and Bob every name's affiliation —
308      // nucl-th0308018's `\and Bertrand BOURIQUET \address{…}`; a column of contacts alone, `\and \institute{Univ
309      // B}`, is read on them. The rule is the reading's, not OmniBus's: the kernel `article` route with a document's
310      // `\institute` reads the same way, revP21d)
311      let shared_column = column_first
312        && line.column > 0
313        && last_named().is_some_and(|named| line.column > named)
314        && self.out.creators.len() > block_creators_start
315        && {
316          let own = column_own_text(&line.events);
317          is_institution(&own) && !names_collaboration(&own)
318        };
319      if shared_column {
320        let every: Vec<usize> = (block_creators_start..self.out.creators.len()).collect();
321        run_above.push((line.column, every.clone()));
322        row_above.push((line.column, every));
323        // (the names row it answers: the lines under it go on with the affiliation)
324        answered_rows.push(line.column);
325      }
326      let above = row_above
327        .iter()
328        .rev()
329        .find(|(c, _)| *c == line.column)
330        .map(|(_, r)| r.clone());
331      let above = above
332        .or_else(|| row_above.last().map(|(_, r)| r.clone()))
333        .unwrap_or_default();
334      let run = run_above
335        .iter()
336        .rev()
337        .find(|(c, _)| *c == line.column)
338        .or_else(|| run_above.last())
339        .map(|(_, r)| r.clone())
340        .unwrap_or_default();
341      let carried = continues.take();
342      // (a line a glyph leads — no superscript, `(\S) Inst…` — whose mark no name requests is no labelled line: Perl's
343      // block reader splits at superscripts alone (`positionOf($line, @authorsup_markers)`, Base_Utility.pool.ltxml:
344      // 699-707), so such a line continues the entry above it, as an unmarked line does (`line:continued`), and a printed
345      // mark no name requests is no label (FIDELITY_CHARTER §4) — cond-mat0309053's translators' `(\S)`, `(\dag)`,
346      // `(\ddag)` institutes, no author's marks, stay in the authors' affiliation as before, not a nameless creator's)
347      // (only where an entry is open above it, which the line continues — Perl's `if (@entries)`; else, an e-mail
348      // entry above or none, it stays the symbol line it was: `*Corresponding Author: Ayodeji Olalekan Salau (…)` under
349      // the `\textsuperscript{7}\texttt{…}` addresses is no names row, 2510.01302)
350      let unrequested_glyph = carried.is_some()
351        && first_ink.is_some_and(|e| {
352          e.kind == EventKind::Mark
353            && e.role.as_deref() == Some("text")
354            && !entries_of(&line.events)
355              .iter()
356              .flat_map(|entry| &entry.labels)
357              .any(|label| {
358                block_requests.iter().any(|r| label_matches(r, label))
359                  || self
360                    .out
361                    .creators
362                    .iter()
363                    .any(|c| c.requests.iter().any(|r| label_matches(r, label)))
364              })
365        });
366      if unrequested_glyph && let Some(carried) = carried {
367        let notes = line
368          .events
369          .iter()
370          .copied()
371          .filter(|e| e.kind == EventKind::Note);
372        self.continue_line(
373          index,
374          &carried,
375          &events_text(&line.events),
376          notes,
377          above.last().copied(),
378        );
379        continues = Some(carried);
380        continue;
381      }
382      match first_ink.map(|e| &e.kind) {
383        Some(EventKind::Mark) if !led_names[n] => {
384          let entries = entries_of(&line.events);
385          labelled_last.push((line.column, entries.len()));
386          continues = self.labelled_entries(index, entries, "");
387        },
388        Some(_) => {
389          let text = events_text(&line.events);
390          let only_contacts = line
391            .events
392            .iter()
393            .filter(|e| inked(e))
394            .all(|e| e.kind == EventKind::Contact);
395          // (a line printing a membership grade and notes alone, right under the names, is the name's the grade follows,
396          // the last above, never a row's: `and~Cy Coe,\\~\IEEEmembership{Member,~IEEE,}\thanks{…}`, 2202.10869; under an
397          // affiliation line it follows no name, revP5a S1)
398          let grade_line = |e: &&Event| {
399            e.kind == EventKind::Note
400              || (e.kind == EventKind::Contact && e.role.as_deref() == Some("membership"))
401          };
402          if let Some(&last) = above.last()
403            && !under_affiliation
404            && line
405              .events
406              .iter()
407              .any(|e| e.role.as_deref() == Some("membership"))
408            && line.events.iter().filter(|e| inked(e)).all(grade_line)
409          {
410            for event in &line.events {
411              match event.kind {
412                EventKind::Note => self.note(index, event, Some(last), "note:adjacent"),
413                EventKind::Contact => {
414                  self.contact_event(index, event, &[last], "line:contact membership:name-above")
415                },
416                _ => {},
417              }
418            }
419            continue;
420          }
421          // (a column's first line that is one link printing a name is that name: `\href{https://orcid.org/…}
422          // {\includegraphics{orcid}\hspace{1mm}Amanda A. Howard}`, 2311.06483)
423          if column_first && only_contacts {
424            let links: Vec<&Event> = line
425              .events
426              .iter()
427              .copied()
428              .filter(|e| e.kind == EventKind::Contact)
429              .collect();
430            if let [link] = links.as_slice()
431              && link.role.as_deref() == Some("url")
432              && reads_as_names(&link.text)
433            {
434              let c = self.new_creator(
435                link.text.clone(),
436                "line:names link-text",
437                index,
438                block_requests.clone(),
439                group,
440              );
441              answered_rows.retain(|c| *c != line.column);
442              run_above.push((line.column, vec![c]));
443              row_above.push((line.column, vec![c]));
444              continue;
445            }
446          }
447          // (a later line is more names when each of its pieces reads as a name: `Triantafyllos Afouras\quad\quad
448          // Himel Chowdhury …` under the first row, 2111.03635; the classify helpers on each piece, not the run of them)
449          // (a line of e-mail addresses alone is never names, a column's first line too: `\And
450          // \{sagghosh, …, niranjan\}@cs.stonybrook.edu`, 2210.05991)
451          // (nor one whose address an undefined command glued a word's ink onto, before its `@`, which no word of the
452          // source spells: `\and Bob\email{b@x.org}` reads "Bobb@x.org" — a name, as Perl's creator holds it, never the author
453          // before's e-mail; an undefined command never changes the segmentation, `Event::glued_undef`)
454          let address_line =
455            prints_only_addresses(&text) && !line.events.iter().any(|e| e.glued_undef);
456          let names_line = led_names[n]
457            || !only_contacts

1463            source: Some((index, span)),
1464            target: None,
1465          }));
1466        }
1467        return;
1468      }
1469    }
1470    // (a note opening with a Mark some name requests is that label's legend: `\thanks{$^{1}$ Canizaro … Center}`
1471    // under `Newaz$^{1,2,*}$`, 2311.16145 — the mark it prints, its leading Mark event; its owners by ruling 1)
1472    // (one whose text holds a later mark of other requested labels, none of the first's, is the legends of several
1473    // entries, no one label's: IEEEtran's `\IEEEcompsocitemizethanks{\IEEEcompsocthanksitem \Mark{1}College …
1474    // \IEEEcompsocthanksitem \Mark{2}College …}` was given whole to the authors of mark 1, 1802.03043 — a note of the
1475    // name before, as today's; a later `\textsuperscript{1,2}` sharing the first's label keeps the first's reading, as
1476    // today's, 2507.07006)
1477    if let Some((labels, legend)) = leading_mark(note)
1478      && labels
1479        .iter()
1480        .any(|l| !label_owners(&self.out.creators, &live, l).is_empty())
1481      && !note
1482        .inner
1483        .iter()
1484        // (in a re-routed `\thanks`, its raised marks alone: a glyph is its text, `*Corresponding authors` closing
1485        // `$^{4}$ Department of Mechanical Engineering, …`, 2505.06451)
1486        .filter(|e| e.kind == EventKind::Mark && !e.line_first && (!numbered || raised_mark(e)))
1487        .map(|e| mark_labels(e.label.as_deref().unwrap_or("")))
1488        .any(|later| {
1489          !later.is_empty()
1490            && later.iter().all(|l| !labels.contains(l))
1491            && later
1492              .iter()
1493              .any(|l| !label_owners(&self.out.creators, &live, l).is_empty())
1494        })
1495    {
1496      let symbol = labels.iter().all(|l| is_symbol_label(l));
1497      // (the old driver's procedure's kind, in its order (P2 NT; ported strictly, P2r2): a
1498      // `\footnote` is a note (`\lx@add@note`, frontmatter_fields.rs); a `\thanks` is an affiliation only where
1499      // `\lx@add@thanks` re-routes it, its content opening with `$^` or `\textsuperscript` whose first character is an
1500      // ASCII digit (`starts_with_affiliation_mark`, OXIDIZED_DESIGN #52) — here the note's leading Mark, a superscript
1501      // whose printed text opens with a digit —, else a note; an affiliation whose whole text is a legend is then a note
1502      // (the Kind step for a legend, `insert::kind_legend_contacts`; here `is_legend_text`). The note goes to its mark's
1503      // carriers, not the name before it (FIDELITY_CHARTER §4: a symbol-led line goes to the authors carrying the
1504      // symbol, affiliation or legend alike): `\thanks{\IEEEauthorrefmark{2} Also with …}`, 1107.5730; `\thanks{$\diamond$
1505      // Work conducted during an internship …}`, 2205.02029. One departure, the reading's own entry kept: a note that
1506      // prints e-mail addresses and is no legend stays the entry the reading made it, whose addresses go each to the
1507      // carrier it spells (FIDELITY_CHARTER §4's symbol-line rule; the old driver's note keeps them in its text, every
1508      // address on the note's owner): IEEEtran journal's `\thanks{\IEEEauthorrefmark{1}The authors are with …, E-mails:
1509      // a@, b@, c@}`, 1610.05419, 2405.06709 — review revP2 finding 1)
1510      let role =
1511        if (!numbered && addresses_in(&legend).is_empty()) || is_legend_text(&legend) || symbol {
1512          "legend"
1513        } else {
1514          "affiliation"
1515        };
1516      self.pending.push((labels.clone(), symbol, Assoc {
1517        role:   role.into(),
1518        text:   legend.clone(),
1519        label:  labels.first().cloned(),
1520        rule:   s!("{rule} note:legend ruling1:request"),
1521        source: Some((index, note.span)),
1522        target: None,
1523      }));
1524      return;
1525    }
1526    let role = note
1527      .role
1528      .as_deref()
1529      .unwrap_or("thanks")
1530      .split(' ')
1531      .next()
1532      .unwrap_or("thanks")
1533      .to_string();
1534    let rule = rule.to_string();
1535    self.attach(
1536      owner,
1537      Assoc {
1538        role,
1539        text: note.text.trim().to_string(),
1540        label: note.label.clone(),
1541        rule,
1542        source: Some((index, note.span)),
1543        target: None,
1544      },
1545      true,
1546    );
1547  }
1548}

1584/// [`collaboration_line_framing`]): a collaboration word suffices, a team word only by more than a bare "the" or listed

1587fn names_collaboration(text: &str) -> bool {
1588  let t = text.trim_end_matches(|c: char| c.is_whitespace() || ".,;".contains(c));
1589  let listed = t.split_whitespace().any(|w| matches!(w, "and" | "&"));
1590  let reads = |s: &str| collaboration_framing(s, listed, true).is_some();
1591  reads(t) || remark_start(t).is_some_and(|at| reads(&t[at..]))
1592}

1608/// Where a sectioned block's address section starts in `events`: the first of the section `markers` (positions, ascending)
1609/// whose events, up to the next one, print an address (an `@` in their text) — its `\emails`, not its `\affiliations`.
1610fn address_section(markers: &[u16], events: &[Event]) -> Option<usize> {
1611  let at = |marker: u16| {
1612    events
1613      .iter()
1614      .position(|e| e.span.is_some_and(|(start, _)| start >= marker))
1615      .unwrap_or(events.len())
1616  };
1617  markers.iter().enumerate().find_map(|(n, &marker)| {
1618    let (from, to) = (
1619      at(marker),
1620      markers.get(n + 1).map_or(events.len(), |&next| at(next)),
1621    );
1622    let section: Vec<&Event> = events[from..to.max(from)].iter().collect();
1623    events_text(&section).contains('@').then_some(from)
1624  })
1625}
