1//! The reading's coordination and separator checks (brief N1; OXIDIZED_DESIGN_DIVERGENCES #980's safety net), beside the
2//! word check (`words.rs`, their model): a pure pass over an author block's digested events and the reading
3//! ([`Resolved`]), what TeX printed against what the reading makes of it, in the block's own terms. Perl has none: its
4//! `\lx@add@authors` (Base_Utility.pool.ltxml:687-694) cuts by the spelling and checks only for a `{tabular}`.
5//!
6//! - **Marks** ([`uncoordinated_marks`]): a mark printed on a name whose label leads one of the block's lines reaches an item
7//!   of that name's creator carrying it; and an item a line's leading mark labels, which a name prints too, is some
8//!   creator's.
9//! - **Separators** ([`merged_names`]): two Names a separator parts (`Sep`: `\and`, a comma, the word "and"; or a `Break`)
10//!   are no one creator's name.
11//!
12//! Each is linear in the block (a label set's lookups, a word index of the names), as the reading is.
13

22/// A block's events, by ledger index, the blocks the reading reads (not cleared).
23fn block_events(ledger: &Ledger, index: usize) -> Option<&[Event]> {
24  match ledger.entries().get(index) {
25    Some(Entry::Author(block)) if !block.cleared => match &block.outcome {
26      Outcome::Events { events, .. } => Some(events),
27      _ => None,
28    },
29    _ => None,
30  }
31}

33/// The reading's creators by the block or call they come from, and the block's own items.
34fn by_source(resolved: &Resolved) -> BTreeMap<usize, (Vec<&RCreator>, Vec<&Assoc>)> {
35  let mut out: BTreeMap<usize, (Vec<&RCreator>, Vec<&Assoc>)> = BTreeMap::new();
36  for c in &resolved.creators {
37    out.entry(c.source).or_default().0.push(c);
38  }
39  for a in &resolved.block {
40    if let Some((at, _)) = a.source
41      && let Some((_, own)) = out.get_mut(&at)
42    {
43      own.push(a);
44    }
45  }
46  out
47}

49/// A set of labels, matched as `labels::label_matches` matches a request and a label (equal, or equal past one prefix or
50/// both), by lookups.
51#[derive(Default)]
52struct Labels<'a> {
53  exact: BTreeSet<&'a str>,
54  bare:  BTreeSet<&'a str>,
55}
56
57impl<'a> Labels<'a> {
58  fn insert(&mut self, label: &'a str) {
59    self.exact.insert(label);
60    if let Some(bare) = unprefixed(label) {
61      self.bare.insert(bare);
62    }
63  }
64
65  fn matches(&self, label: &str) -> bool {
66    let bare = unprefixed(label);
67    self.exact.contains(label)
68      || self.bare.contains(label)
69      || bare.is_some_and(|bare| self.exact.contains(bare) || self.bare.contains(bare))
70  }
71}

73/// The labels the marks of `events` print, those leading their line (an entry's: its first ink's run of marks, `$^1$ $^2$
74/// Univ A`, 1703.06483) and the others (a name's or an entry's own after its text), as the resolver reads a mark
75/// ([`mark_requests`]).
76fn printed_labels(events: &[Event]) -> (BTreeSet<String>, BTreeSet<String>) {
77  let (mut leading, mut other) = (BTreeSet::new(), BTreeSet::new());
78  let mut leads = false;
79  for e in events {
80    match &e.kind {
81      EventKind::Mark => {
82        leads |= e.line_first;
83        let into = if leads { &mut leading } else { &mut other };
84        into.extend(mark_requests(e));
85      },
86      EventKind::Name | EventKind::Contact | EventKind::Note | EventKind::Break => leads = false,
87      _ => {},
88    }
89  }
90  (leading, other)
91}

93/// Per block (its ledger index) the reading of which fails the marks check, the labels that fail it: a creator's request
94/// (a mark shown on its name, ruling 1) that a line of the block leads with, which no item the creator holds carries by its
95/// own label; an item's label a line leads with and a name's mark prints, which no creator holds (the block's own); a mark a
96/// creator requests printed inside an item, not leading it.
97pub(in crate::base_utilities::frontmatter::observer) fn uncoordinated_marks(
98  ledger: &Ledger,
99  resolved: &Resolved,
100) -> BTreeMap<usize, Vec<String>> {
101  let mut out: BTreeMap<usize, Vec<String>> = BTreeMap::new();
102  for (index, (creators, own)) in by_source(resolved) {
103    let Some(events) = block_events(ledger, index) else {
104      continue;
105    };
106    let (leading, other) = printed_labels(events);
107    if leading.is_empty() {
108      continue;
109    }
110    let (mut leads, mut others) = (Labels::default(), Labels::default());
111    leading.iter().for_each(|l| leads.insert(l));
112    other.iter().for_each(|l| others.insert(l));
113    let (mut held, mut requested) = (Labels::default(), Labels::default());
114    let mut failed: BTreeSet<String> = BTreeSet::new();
115    for c in &creators {
116      // (an item's own label, what the emitter answers a request by: an entry led by `$^1$ $^2$` labelled `1` alone leaves
117      // the name marked 2 its mark printed, 1703.06483)
118      let mut labels = Labels::default();
119      let items = c.affiliations.iter().chain(&c.contacts).chain(&c.notes);
120      for label in items.filter_map(|a| a.label.as_deref()) {
121        labels.insert(label);
122        held.insert(label);
123      }
124      for request in &c.requests {
125        requested.insert(request);
126        if leads.matches(request) && !labels.matches(request) {
127          failed.insert(request.clone());
128        }
129      }
130    }
131    // (a mark a creator requests printed inside an item, not leading it: an entry the reading didn't part at its mark, an
132    // e-mail line's `$^1$ann@x.org, $^2$bob@y.org` read into an affiliation, 1703.06483; the items' spans by their start,
133    // the farthest end before each)
134    let mut spans: Vec<(u16, u16)> = creators
135      .iter()
136      .flat_map(|c| c.affiliations.iter().chain(&c.contacts))
137      .chain(own.iter().copied())
138      .filter_map(|a| match a.source {
139        Some((at, Some(span))) if at == index => Some(span),
140        _ => None,
141      })
142      .collect();
143    spans.sort_unstable();
144    let reach: Vec<u16> = spans
145      .iter()
146      .scan(0u16, |far, &(_, to)| {
147        *far = (*far).max(to);
148        Some(*far)
149      })
150      .collect();
151    for mark in events.iter().filter(|e| e.kind == EventKind::Mark) {
152      let Some((from, to)) = mark.span else {
153        continue;
154      };
155      let before = spans.partition_point(|&(start, _)| start < from);
156      if before > 0 && reach[before - 1] >= to {
157        failed.extend(
158          mark_requests(mark)
159            .into_iter()
160            .filter(|label| requested.matches(label)),
161        );
162      }
163    }
164    for label in own.iter().filter_map(|a| a.label.as_deref()) {
165      if leads.matches(label) && others.matches(label) && !held.matches(label) {
166        failed.insert(label.to_string());
167      }
168    }
169    if !failed.is_empty() {
170      out.insert(index, failed.into_iter().collect());
171    }
172  }
173  out
174}

176/// Per block (its ledger index) the reading of which fails the separators check, the pairs of names it joins: two Names
177/// of its events a separator parts (a `Sep` other than a gap, or a `Break`, between them, only marks, notes and layout
178/// beside it), whose words one creator's name holds both of, the second no qualifier or remark and the name joined by no
179/// rule of the reading's (`RCreator::joins`) — or, each carrying its own mark, joined by any rule (review revZ18k B1); and a
180/// name holding a label before a colon and a space.
181pub(in crate::base_utilities::frontmatter::observer) fn merged_names(
182  ledger: &Ledger,
183  resolved: &Resolved,
184) -> BTreeMap<usize, Vec<String>> {
185  let words = |text: &str| -> Vec<String> { printed_words(text).map(str::to_lowercase).collect() };
186  let mut out: BTreeMap<usize, Vec<String>> = BTreeMap::new();
187  for (index, (creators, _)) in by_source(resolved) {
188    let Some(events) = block_events(ledger, index) else {
189      continue;
190    };
191    // (a name the reading joined across a separator by a rule of its own is explained ([`NameJoin`]): written surname first,
192    // `Eckart, A.`, 1501.02171; a qualifier, a remark, a macro's, a layout's, a gap's or initials' join)
193    let names: Vec<Vec<String>> = creators
194      .iter()
195      .map(|c| {
196        if !c.joins.is_empty() {
197          Vec::new()
198        } else {
199          words(&c.name)
200        }
201      })
202      .collect();
203    // (each word's names: a pair's candidates are the names holding its first word)
204    let mut holding: BTreeMap<&str, BTreeSet<usize>> = BTreeMap::new();
205    for (n, name) in names.iter().enumerate() {
206      for word in name {
207        holding.entry(word).or_default().insert(n);
208      }
209    }
210    let holds = |name: &[String], part: &[String]| {
211      let mut left = name.to_vec();
212      part.iter().all(|w| match left.iter().position(|l| l == w) {
213        Some(at) => {
214          left.remove(at);
215          true
216        },
217        None => false,
218      })
219    };
220    let mut failed = Vec::new();
221    let mut last: Option<(&Event, bool)> = None;
222    for e in events {
223      match &e.kind {
224        EventKind::Name => {
225          // (a qualifier or a remark after a comma is the name's, the reading's rule: `Robert W. Heath, Jr.`, 1004.2377)
226          if let Some((before, true)) = last
227            && !is_name_qualifier(&e.text)
228            && !is_name_remark(&e.text)
229          {
230            let (a, b) = (words(&before.text), words(&e.text));
231            let joined = [a.clone(), b.clone()].concat();
232            let candidates = a.first().and_then(|w| holding.get(w.as_str()));
233            if !b.is_empty()
234              && candidates
235                .is_some_and(|names_of| names_of.iter().any(|&n| holds(&names[n], &joined)))
236            {
237              failed.push(s!("{} | {}", before.text, e.text));
238            }
239          }
240          last = Some((e, false));
241        },
242        // (a gap parts no names: `S. A. \,Levshakov`, 0911.3853; `Micha\l  \,\,Boczek`, 1902.08839)
243        EventKind::Sep(SepKind::Gap { .. }) => {},
244        EventKind::Sep(_) | EventKind::Break => {
245          if let Some((before, _)) = last {
246            last = Some((before, true));
247          }
248        },
249        EventKind::Mark | EventKind::Note | EventKind::Layout | EventKind::Declaration => {},
250        EventKind::Contact | EventKind::ApiCall => last = None,
251      }
252    }
253    // (two names a separator parts, each carrying its own mark, are two people whatever rule of the reading's joined them:
254    // a qualifier or a remark carries no mark of its own — `Y. Dong\affil{6}, \add[ver3]{S. Curry}\affil{1}`, 2308.02720,
255    // review revZ18k B1, which the names line now reads as two names; any such join left is counted here)
256    let all: Vec<Vec<String>> = creators.iter().map(|c| words(&c.name)).collect();
257    // (each word's creators, as `holding` is for the join check above: a name holding the joined words holds the first
258    // piece's first word, so its creators are the only candidates — the same answer for a fraction of the `holds` calls,
259    // which copy a name each; 1507.02068's 418 creators made the scan over every one, added by Z18k `27360530a4`, +1.76 %
260    // instructions)
261    let mut all_holding: BTreeMap<&str, BTreeSet<usize>> = BTreeMap::new();
262    for (n, name) in all.iter().enumerate() {
263      for word in name {
264        all_holding.entry(word).or_default().insert(n);
265      }
266    }
267    // (each Name with whether a separator parts it from the Name before and whether a mark follows it before the next
268    // separator; a contact or a call ends the run)
269    let mut pieces: Vec<Option<(&Event, bool, bool)>> = Vec::new();
270    let mut parted = false;
271    for e in events {
272      match &e.kind {
273        EventKind::Name => {
274          pieces.push(Some((e, parted, false)));
275          parted = false;
276        },
277        EventKind::Mark => {
278          if let Some(Some((_, _, marked))) = pieces.last_mut()
279            && !parted
280          {
281            *marked = true;
282          }
283        },
284        EventKind::Sep(SepKind::Gap { .. }) => {},
285        EventKind::Sep(_) | EventKind::Break => parted = true,
286        EventKind::Note | EventKind::Layout | EventKind::Declaration => {},
287        EventKind::Contact | EventKind::ApiCall => {
288          pieces.push(None);
289          parted = false;
290        },
291      }
292    }
293    for pair in pieces.windows(2) {
294      if let [Some((before, _, true)), Some((after, true, true))] = pair {
295        // (each a word of its own: a bracket's `(` before a line's `$^1$` names no one, `($^1$) Univ A`)
296        let (a, b) = (words(&before.text), words(&after.text));
297        let joined = [a.clone(), b.clone()].concat();
298        let pair = s!("{} | {}", before.text, after.text);
299        if !a.is_empty()
300          && !b.is_empty()
301          && all_holding
302            .get(a[0].as_str())
303            .is_some_and(|holders| holders.iter().any(|&n| holds(&all[n], &joined)))
304          && !failed.contains(&pair)
305        {
306          failed.push(pair);
307        }
308      }
309    }
310    // (a label printed before a colon inside a name: a role's heading read into it, `Editor: Andrei V. Gritsan`, 2205.07715;
311    // `{\bf Authors:} F.~Arleo`, hep-ph0311131)
312    for c in &creators {
313      // (a colon a space follows: not a link's, `http://www.ligo.org`, gr-qc0505029)
314      if let Some((head, tail)) = c.name.split_once(':')
315        && printed_words(head).next().is_some()
316        && tail.starts_with(char::is_whitespace)
317        && printed_words(tail).next().is_some()
318      {
319        failed.push(s!("{head}: | {}", tail.trim()));
320      }
321    }
322    if !failed.is_empty() {
323      out.insert(index, failed);
324    }
325  }
326  out
