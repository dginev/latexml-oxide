17/// `key=value` of an attribute string (`role=affiliation,label=a`), braces stripped: a repeated key's last value, as
18/// `KeyVals::get_value` reads it (the ledger keeps every pair in source order: `role=affiliation,role=address`).
19pub(super) fn attr_value<'a>(attrs: &'a str, key: &str) -> Option<&'a str> {
20  split_top_commas(attrs).into_iter().rev().find_map(|kv| {
21    let (k, v) = kv.split_once('=')?;
22    (k.trim() == key).then(|| {
23      v.trim()
24        .trim_start_matches('{')
25        .trim_end_matches('}')
26        .trim()
27    })
28  })
29}

31/// An attribute string without any `key=` pair (`role=affiliation,labelseq=affiliation` without `labelseq`).
32pub(super) fn without_key(attrs: &str, key: &str) -> String {
33  split_top_commas(attrs)
34    .into_iter()
35    .filter(|kv| kv.split_once('=').is_none_or(|(k, _)| k.trim() != key))
36    .collect::<Vec<_>>()
37    .join(",")
38}

40/// `text` split at its commas outside braces.
41pub(super) fn split_top_commas(text: &str) -> Vec<&str> {
42  let mut parts = Vec::new();
43  let (mut depth, mut start) = (0i32, 0usize);
44  for (i, c) in text.char_indices() {
45    match c {
46      '{' => depth += 1,
47      '}' => depth -= 1,
48      ',' if depth == 0 => {
49        parts.push(&text[start..i]);
50        start = i + 1;
51      },
52      _ => {},
53    }
54  }
55  parts.push(&text[start..]);
56  parts
57}

64/// One mark's labels (ruling 1's split): at commas and spaces; a run of one footnote symbol is one label (`**`), a
65/// symbol glued to a label another (`*1` → `*`, `1`); parentheses and brackets around a label dropped (`(a)`); the
66/// asterisk forms (`∗`, `✱`, …) read as `*`, the formula `\|`'s `∥` as `‖` (0806.2911).
67pub fn mark_labels(label: &str) -> Vec<String> {
68  // (a `fuzzy:<name>` label — svjour's `\institute` entry after `\at`, a name with its spaces — is one label, never
69  // split: review revfms2a2 finding 2, 0807.3898, 2403.09642)
70  if label.starts_with("fuzzy:") {
71    return vec![label.trim().to_string()];
72  }
73  // (a mark holding `\ref`s is a list at its commas — `events::script_ink` writes each reference as an item of its
74  // own —, each `\ref`'s key one label, never split further — its spaces and brackets are the key's —, the rest read
75  // as any mark, what prints between two references no label: `\inst{\ref{ia},\ref{ib}}` and `\inst{\ref{ia}\ref{ib}}`
76  // request `ref:ia` and `ref:ib`, 1707.06238, 1302.1442; `\inst{1,\ref{ib}}` `1` and `ref:ib`; `\ref{a}--\ref{c}`
77  // `ref:a` and `ref:c`, its dash none)
78  if label
79    .split(',')
80    .any(|piece| piece.trim().starts_with("ref:"))
81  {
82    return label
83      .split(',')
84      .map(str::trim)
85      .filter(|piece| !piece.is_empty())
86      .flat_map(|piece| {
87        if piece.starts_with("ref:") {
88          vec![piece.to_string()]
89        } else {
90          mark_labels(piece)
91            .into_iter()
92            .filter(|l| l.chars().any(|c| c.is_alphanumeric() || is_symbol_char(c)))
93            .collect()
94        }
95      })
96      .collect();
97  }
98  let mut labels = Vec::new();
99  for part in label.split(|c: char| c == ',' || c == ';' || c.is_whitespace()) {
100    let part = part.trim_matches(|c: char| "()[]{}".contains(c));
101    // (a binding's prefix — aastex's `altaffil:` — stays on each piece: `altaffil:*` is one label, `altaffil:**`
102    // another; split off, every starred label shared a bare `altaffil:` and 0911.4695's `*`/`**` notes swapped)
103    if let Some((prefix, rest)) = part.split_once(':')
104      && !prefix.is_empty()
105      && prefix.chars().all(|c| c.is_ascii_alphabetic() || c == '@')
106      && !rest.is_empty()
107    {
108      labels.extend(
109        mark_labels(rest)
110          .into_iter()
111          .map(|piece| s!("{prefix}:{piece}")),
112      );
113      continue;
114    }
115    // (each character with the combining marks on it, as `parted_label` reads one: `\={\dag}`'s `†̄` and
116    // `\textcircled{\dag}`'s `†⃝` are a symbol of their own, never a bare `†` and a lone mark — review rev92b_r3 B1,
117    // where both requested `†` and a `\dag` author took the `\={\dag}` line —; a run is one such character repeated)
118    let mut characters: Vec<String> = Vec::new();
119    for c in part.chars().map(|c| match c {
120      // (every asterisk form one key, user ruling 2026-11-02: `queue_digest::one_asterisk`)
121      c if ASTERISK_FORMS.contains(&c) => '*',
122      '\u{2225}' => '‖',
123      c => c,
124    }) {
125      match characters.last_mut() {
126        Some(character) if is_combining_mark(c) => character.push(c),
127        _ => characters.push(c.to_string()),
128      }
129    }
130    let mut run = String::new();
131    let mut unit = String::new();
132    let mut rest = String::new();
133    for character in characters {
134      if character.starts_with(is_symbol_char) {
135        if !rest.is_empty() {
136          labels.push(std::mem::take(&mut rest));
137        }
138        if !run.is_empty() && unit != character {
139          labels.push(std::mem::take(&mut run));
140        }
141        run.push_str(&character);
142        unit = character;
143      } else {
144        if !run.is_empty() {
145          labels.push(std::mem::take(&mut run));
146        }
147        rest.push_str(&character);
148      }
149    }
150    labels.extend([run, rest].into_iter().filter(|s| !s.is_empty()));
151  }
152  labels
153}

155/// The labels [`mark_labels`] parts out of one written piece at a footnote symbol glued to a label (`1*` → `1`, `*`;
156/// `*1`; not `1,*` or `1 *`, written apart, nor a run of one symbol `**`): ruling 1's reading of a mark printed so
157/// (2506.20360's `\inst{1*}`). None for a `\ref` list or a `fuzzy:` label, which are never parted.
158pub(super) fn glued_symbol_parts(label: &str) -> Vec<String> {
159  if label.starts_with("fuzzy:") || label.contains("ref:") {
160    return Vec::new();
161  }
162  label
163    .split(|c: char| c == ',' || c == ';' || c.is_whitespace())
164    .map(mark_labels)
165    .filter(|labels| labels.len() >= 2)
166    .flatten()
167    .collect()
168}

176/// The creators a pending entry's label names, as the kernel relocates an annotation (Perl Base_Utility.pool.ltxml:
177/// 885-900, `relocate.rs`): the creators requesting it exactly; else — misuse of labelling macros can lead to a prefix
178/// mismatch — those whose request without its prefix is it, those requesting it without its prefix, those whose
179/// request and it agree without both prefixes, the first that finds any (elsarticle's `\fnref{label1}`, `fn:label1`,
180/// answers `\address[label1]`: 1712.00945; `\inst{1}`'s `affiliation:1` the entry `1`). A creator's own role-sequence
181/// label `author:N` (the N-th author, `\lx@add@frontmatter@now`'s `_annotations`) is matched exactly and never by
182/// that fallback (OD #159, relocate.rs:42-50: a shared `affiliation:1` must not reach the first author by number) — the
183/// N-th of the `live` creators, as today's numbers the authors it holds (`place.rs`'s `author:{n}`: a replaced `\author`
184/// call's creators are none of them, so they shift no later label; review revfms2a2_6 finding 4, FM S2a.2 r7).
185pub(super) fn label_owners(creators: &[RCreator], live: &[usize], label: &str) -> Vec<usize> {
186  if let Some(n) = label.strip_prefix("author:") {
187    return n
188      .trim()
189      .parse::<usize>()
190      .ok()
191      .and_then(|n| n.checked_sub(1))
192      .and_then(|at| live.get(at).copied())
193      .into_iter()
194      .collect();
195  }
196  let find = |matches: &dyn Fn(&str) -> bool| -> Vec<usize> {
197    (0..creators.len())
198      .filter(|&c| creators[c].requests.iter().any(|r| matches(r)))
199      .collect()
200  };
201  let bare = unprefixed(label);
202  let tiers: [&dyn Fn(&str) -> bool; 4] = [
203    &|r| r == label,
204    &|r| unprefixed(r) == Some(label),
205    &|r| bare == Some(r),
206    &|r| bare.is_some() && unprefixed(r) == bare,
207  ];
208  tiers
209    .iter()
210    .map(|tier| find(*tier))
211    .find(|owners| !owners.is_empty())
212    .unwrap_or_default()
213}

254/// The letters a label written in TeX source spells (`Jean-Ren\'e Chazottes`): accent commands dropped, `\l`, `\o`
255/// their letters, braces and ties aside, any other command's name dropped (its argument kept: `\textbf{Ann}`); the marks
256/// a name's source carries dropped with their arguments — a mark or note command's (`\inst{1}`, `\thanks{…}`), a
257/// formula (`$^{1}$`) — so `Luca Passalacqua (\Envelope)` spells "Luca Passalacqua" (0807.3898).
258pub(super) fn tex_letters(source: &str) -> String {
259  let mut out = String::new();
260  let mut chars = source.chars().peekable();
261  // skip a balanced group whose `{` is next
262  fn skip_group(chars: &mut std::iter::Peekable<std::str::Chars<'_>>) {
263    if chars.peek() != Some(&'{') {
264      return;
265    }
266    let mut depth = 0i32;
267    for c in chars.by_ref() {
268      match c {
269        '{' => depth += 1,
270        '}' => {
271          depth -= 1;
272          if depth == 0 {
273            return;
274          }
275        },
276        _ => {},
277      }
278    }
279  }
280  while let Some(c) = chars.next() {
281    match c {
282      '$' => {
283        for m in chars.by_ref() {
284          if m == '$' {
285            break;
286          }
287        }
288        continue;
289      },
290      '\\' => {},
291      c => {
292        out.push(if c == '~' { ' ' } else { c });
293        continue;
294      },
295    }
296    let mut word = String::new();
297    while let Some(&n) = chars.peek() {
298      if n.is_ascii_alphabetic() || n == '@' {
299        word.push(n);
300        chars.next();
301      } else {
302        break;
303      }
304    }
305    if word.contains('@') {
306      // (an internal command, its argument with it: what a symbol macro writes out to, `\Envelope` as
307      // `\@chooseSymbol{'014}` once `ledger::label_text` expands the label, 0807.3898 — the digits were read as letters)
308      while chars.peek() == Some(&' ') {
309        chars.next();
310      }
311      skip_group(&mut chars);
312    } else if word.is_empty() {
313      // (a control symbol printing its character keeps it: `\correspondingauthor{Jiacheng Meng \& Cheng Li}` names two,
314      // 2008.13733; an accent command, `\'`, `\"`, `\~`: its character dropped, the letter after it kept)
315      if let Some(&n) = chars.peek()
316        && "&%#_".contains(n)
317      {
318        out.push(n);
319      }
320      chars.next();
321    } else if let Some(rest) = word.strip_prefix("lxaccent") {
322      // (an accent written out as the kernel's inner command, `\lxaccentumlaut`, `\lxaccentc` — `accent_inner_name`,
323      // setup_binding_language.rs —, the letters glued after its name the label's: `Gizem \lxaccentc{S}eng\lxaccentumlautor`
324      // spells "Gizem Sengor", 2205.11550)
325      let tail = ["acute", "dot", "macron", "umlaut", "hat", "grave", "tilde"]
326        .iter()
327        .find_map(|name| rest.strip_prefix(name))
328        .unwrap_or_else(|| rest.get(1..).unwrap_or(""));
329      out.push_str(tail);
330    } else if matches!(
331      word.as_str(),
332      "l" | "o" | "ss" | "ae" | "oe" | "aa" | "i" | "j" | "L" | "O"
333    ) {
334      out.push_str(&word);
335    } else if matches!(
336      word.as_str(),
337      "inst"
338        | "thanks"
339        | "footnote"
340        | "footnotemark"
341        | "textsuperscript"
342        | "email"
343        | "orcid"
344        | "orcidlink"
345        | "label"
346        | "ref"
347    ) {
348      // (a mark or a note of the name, with its argument)
349      while chars.peek() == Some(&' ') {
350        chars.next();
351      }
352      skip_group(&mut chars);
353    }
354  }
355  out
356}

358/// The labels of the Mark a note's text opens with (the lexer's leading Mark of its body, `events.rs`: `$^{1}$Unit…`,
359/// `\textsuperscript{a}`, a footnote symbol `*Corresponding author`), and its text without that mark — never read
360/// from the flattened text, where `$^{1}$Unit` is the word "1Unit" (review revfms2a2 finding 5; 1909.02288).
361pub(super) fn leading_mark(note: &Event) -> Option<(Vec<String>, String)> {
362  let mark = note
363    .inner
364    .iter()
365    .find(|e| e.kind == EventKind::Mark && e.line_first)?;
366  let labels = mark_labels(mark.label.as_deref().unwrap_or(""));

370  let text = note.text.trim_start();
371  let legend = text.strip_prefix(mark.text.as_str()).unwrap_or(text);
372  Some((labels, legend.trim().to_string()))
373}

375/// Whether a later line of a column whose first line is names is more names (review revfms2a2 finding 6: by positive
376/// evidence, the text test only beside it — principled before heuristics). The evidence: the line follows a blank line
377/// (`\\ \\`, a new author's lines: 0802.1860), or a names row it continues (the row before ended in a comma or "and":
378/// `Juan Sebastian~Gomez-Diaz, and \newline Alejandro~Alvarez-Melcon`, 1403.0130), or the names row above has its
379/// affiliation line already and this line is set in a font of its own against it (`Kei Harada \\ {\it Nagoya
380/// University} \\ Hayato Saigo \\ …`: 1009.1429, 1108.3744; the same font, "Aptex Software" under "Chief Scientist",
381/// 0803.3838, is the affiliation's) or comes after a line of two or more labelled entries (`after_labelled`: `$^\dagger$Univ
382/// A \quad $^\ddagger$Univ B \\ Cat Cole \\ Physics`, SYNC row 46 — under one entry it goes on with that entry, the old
383/// reader's N5: `Resolver::block`), or its names show marks or notes (`Ann Able$^{1}$ Bob Baker$^{2}$`, `Irina Gheorghiciuc\thanks{…}`,
384/// 1403.5246: a Mark or a Note right after a Name), or it is a row of two or more name pieces parted
385/// by separators (`Triantafyllos Afouras\quad\quad Himel Chowdhury …` under the first row, 2111.03635), none holding
386/// an acronym (`IBM India, IBM CIO`, 2609.10596: institutions); and then each
387/// of its pieces must read as a name (the classify helpers). One piece with no evidence — "Capital Fund Management",
388/// "Google DeepMind" (2606.07382, 2307.02318, 2305.16559, 2609.09054) — is the names' affiliation line, whatever its
389/// text reads as.
390pub(super) fn later_line_is_names(
391  line: &Line<'_>,
392  continued: bool,
393  under_affiliation: bool,
394  after_labelled: bool,
395  orcid_names: bool,
396  over_institution: bool,
397) -> bool {
398  let events = &line.events;
399  // (or, under a binding reading its `\\` lines as its own split did (`orcid_names`, `Policy::author_lines`), an
400  // ORCID, a person's: llncs's `Marcos Kalinowski\orcidID{…} \\ Lucas Romao\orcidID{…} \\ …`, 2501.11774, each
401  // line a name as the old split read it — six names had become the first one's affiliations)
402  let marked = events.windows(2).any(|pair| {
403    pair[0].kind == EventKind::Name
404      && (pair[1].kind == EventKind::Mark && !pair[1].line_first
405        || pair[1].kind == EventKind::Note
406        || orcid_names
407          && pair[1].kind == EventKind::Contact
408          && pair[1].role.as_deref() == Some("orcid"))
409  });
410  // (an address the run prints is the line's e-mail, no piece of its names: `\name Marvin Pförtner$^1$ \email marvin@…`,
411  // jmlr2e's `\email` a font switch, 2312.14886, `names_without_addresses`)
412  let names: Vec<&&Event> = events
413    .iter()
414    .filter(|e| {
415      e.kind == EventKind::Name
416        && e.text.chars().any(char::is_alphanumeric)
417        && addresses_in(&e.text).is_empty()
418    })
419    .collect();
420  let pieces = names.iter().filter(|e| !is_name_qualifier(&e.text)).count();
421  // (the line's first run in a font of its own, against the affiliation line above it: the print sets the name apart —
422  // but not a cell's: each cell of an alignment opens in the alignment's font, so the lexer's change of font between
423  // the runs of two cells is a cell's reset as often as a switch (`$^1$\it Univ X,\\ \bf Dept Y\\ City Z` in a
424  // `tabular`: "City Z" read as an author, review rev94a r5 t06))
425  let mut in_cell = false;
426  let own_font = events.iter().take_while(|e| !inked(e)).any(|e| {
427    in_cell |= e.kind == EventKind::Layout && e.role.as_deref() == Some("cell");
428    !in_cell && e.kind == EventKind::Layout && e.role.as_deref() == Some("font")
429  });
430  // (a row whose pieces each hold an acronym — a word of two or more capitals, no roman numeral — reads as
431  // institutions, and its pieces alone are no evidence: `Bhanwar Gupta \\ IBM India, IBM CIO \\ …`, 2609.10596)
432  let acronym = |word: &str| {
433    let word = word.trim_matches(|c: char| !c.is_alphanumeric());
434    word.chars().count() >= 2
435      && word.chars().all(|c| c.is_ascii_uppercase())
436      && !word.chars().all(|c| "IVXLC".contains(c))
437  };
438  // (in a piece not set in capitals: `MARTHA D. FERREIRA`'s words are no acronyms, 2202.13867)
439  let institutions = names
440    .iter()
441    .filter(|e| !is_name_qualifier(&e.text))
442    .any(|e| e.text.chars().any(char::is_lowercase) && e.text.split_whitespace().any(acronym));
443  // (a line a role opens is a row of names where each of its pieces shows a person, a name's cue or the mark it carries:
444  // `Editor: P. Aurenche`, hep-ph0311131; `Editor: Physical Review`, `Author: Corresponding Author` are words)
445  let shows_a_person = |name: &Event| {
446    carries_a_mark(name, events)
447      || has_person_cue(after_role_heading(&name.text).unwrap_or(&name.text))
448  };
449  let headed = names
450    .first()
451    .is_some_and(|e| after_role_heading(&e.text).is_some())
452    && names.iter().all(|e| shows_a_person(e));
453  // (a line under the names opening with "and" goes on with them, its piece reading as a name: `Anthony Berthelier \\ and
454  // Yongzhe Yan`, 2007.10022, `classify::names_list_continues`; "and Max--Planck Institut für Mathematik, Bonn" reads
455  // as no name, math0208081; the space before the word, which the old test trims off the line's text, `text.trim()
456  // .starts_with("and ")` (`classify::old_reader::name_text::names_list_continues`), opens nothing: `\\~~and~~{\scshape Donald Robertson}`,
457  // 2603.27258, read as "and Donald Robertson" an affiliation of every name above it)
458  let and_opens = !under_affiliation
459    && events
460      .iter()
461      .find(|e| {
462        e.kind != EventKind::Layout && !matches!(e.kind, EventKind::Sep(SepKind::Gap { .. }))
463      })
464      .is_some_and(|e| match &e.kind {
465        EventKind::Sep(SepKind::Word(word)) => word.eq_ignore_ascii_case("and") || word == "&",
466        _ => false,
467      });
468  // (under its row's affiliation line, a line of llncs's `\author` that reads as no institution, over a line that reads
469  // as one (`over_institution`, asked only under `Policy::author_lines`), starts a new row of names, its affiliation
470  // under it — the row
471  // repeated, as llncs's old split read its `\author` lines by `reads_as_affiliation` (`\lx@llncs@author` until Z18c):
472  // `Ann Able \and Bob Baker\\ University of Somewhere\\ Cat Cole\\ Other Institute`, Cat Cole an author with Other
473  // Institute, not an affiliation of the row above. The last line, or one over no institution, stays the row's: `Google
474  // Research \\ Mountain View`, 0803.3838's `Chief Scientist \\ Aptex Software`)
475  let after_affiliation = under_affiliation && over_institution;
476  let evidence = line.after_blank
477    || continued
478    || and_opens
479    || (under_affiliation && (own_font || after_labelled || after_affiliation))
480    || marked
481    || headed
482    || (pieces >= 2 && !institutions);
483  let reads = |text: &str| {
484    let name = after_role_heading(text).filter(|_| headed).unwrap_or(text);
485    reads_as_names(name) || is_name_qualifier(text)
486  };
487  // (a row of two or more pieces each carrying its mark, most of them names and none an institution, is a row of names, its
488  // odd piece one too: `Huseyin Inan$^{1}$, Aaron (Ari) Bornstein$^{1}$ and Robert Sim$^{1}$`, 2409.07809 — the name's brackets
489  // gave the whole row to the row above as "affiliations", each of its marks trailing one)
490  let carried = names.len() >= 2 && {
491    let read = names.iter().filter(|e| reads(&e.text)).count();
492    2 * read > names.len()
493      && names.iter().all(|e| carries_a_mark(e, events))
494      && !names.iter().any(|e| is_institution(&e.text))
495  };
496  evidence && !names.is_empty() && (carried || names.iter().all(|e| reads(&e.text)))
497}

587/// The e-mail addresses printed in `text`, one per print (ruling 3); a braced group of local parts before a shared
588/// domain written out, one each (`{takuma,ichise}@nii.ac.jp`, 1711.05435; the `\lx@split@addresses` reading, OD #805).
589pub fn addresses_in(text: &str) -> Vec<String> {
590  // (the braces of a group set in math, as verbatim holds them: `\email{$\{$fargyriou,mikebekos$\}$@math.ntua.gr}`, 1106.2694)
591  let unshifted;
592  let text = if text.contains('$') {
593    unshifted = text.replace('$', "");
594    unshifted.as_str()
595  } else {
596    text
597  };
598  let mut out = Vec::new();
599  let mut rest = text;
600  while let Some(open) = rest.find('{') {
601    let Some(close) = rest[open..].find('}').map(|c| open + c) else {
602      break;
603    };
604    let after = &rest[close + 1..];
605    if let Some(domain) = after.strip_prefix('@') {
606      // (the addresses printed before the group are the list's too: `wulijun3@mail2.sysu.edu.cn; stsljh@…;
607      // \{fetia,taoqin,tyliu\}@microsoft.com`, 1808.08866 — they were lost)
608      out.extend(plain_addresses(&rest[..open]));
609      let domain: String = domain
610        .chars()
611        .take_while(|c| !c.is_whitespace() && !",;()<>{}".contains(*c))
612        .collect();
613      // (the local parts parted as the author block's reader parts them: a comma, a semicolon, a `|` — llncs's
614      // `\email{\{ivan|carolin\}@…}`, 0902.3918, 1909.00276; `classify::ADDRESS_GROUP_SEPARATORS`)
615      for local in rest[open + 1..close].split(ADDRESS_GROUP_SEPARATORS) {
616        let local = local.trim();
617        if !local.is_empty() && domain.contains('.') {
618          out.push(format!("{local}@{}", domain.trim_end_matches('.')));
619        }
620      }
621      rest = &after[1 + domain.len()..];
622    } else if let Some((group, taken)) = domain_after_group(&rest[open + 1..close], after) {
623      // (the addresses of the group ending in the domain printed after its brace: `\{xxh22@mails,
624      // huichen@mail\}.tsinghua.edu.cn`, 2407.18899 — each read whole; they were lost, review rev92c r2 B-3)
625      out.extend(plain_addresses(&rest[..open]));
626      out.extend(group);
627      rest = &after[taken..];
628    } else if let Some(group) = domain_inside_group(&rest[open + 1..close]) {
629      // (the shared domain printed inside the group, after its last local part: `\{blaz.skrlj,senja.pollak@ijs.si\}`,
630      // 1907.06944 — each local part at it, as the reader reads the line)
631      out.extend(plain_addresses(&rest[..open]));
632      out.extend(group);
633      rest = after;
634    } else {
635      out.extend(plain_addresses(&rest[..=close]));
636      rest = after;
637    }
638  }
639  out.extend(plain_addresses(rest));
640  out
641}

643/// One run of e-mail addresses printed in a text (FIDELITY_CHARTER §4: an address is a lexical object, the author it
644/// spells its owner — brief 95c): its byte range — with the e-mail label word printed before it ("E-mail:", "Emails",
645/// "e-mail addresses:") and the parentheses round it —, its addresses written out, one each (a braced group, a list
646/// of local parts sharing the domain), and the label word as printed, when it took one (the e-mail contact's name, as
647/// an e-mail line's label names its contacts: the API's `named_email_calls`, brief 95c r2 B5).

650  pub start:     usize,
651  pub end:       usize,
652  pub addresses: Vec<String>,
653  pub label:     Option<String>,

656/// The e-mail address runs printed in `text`, in print order ([`AddressRun`]): a braced group (`{ann,bob}@x.org`,
657/// `{ann@mails, bob@mail}.x.edu`, `{ann,bob@x.org}`, `{ann@x.org, bob@y.org}` — the forms [`addresses_in`] reads), a
658/// list of local parts sharing the domain, each part lowercase (`saro,borgani,tornator@oats.inaf.it`, 0810.2791;
659/// `julioleite,tmariz@fis.ufal.br`, 1110.2127 — a list written without spaces, the main session's ruling; one written
660/// with spaces, `hyunjun.jung, p.ruhkamp, b.busam@tum.de` (2308.10621), only where each of its parts `spells` an author
661/// or the list opens the text, a parenthesis, a brace or a label — else prose before one address: `Univ X, uk, ann@x.org`),
662/// and a plain address. Runs separated only by commas, semicolons, spaces or "and" are one run.
663pub(in crate::base_utilities::frontmatter) fn address_runs(
664  text: &str,
665  spells: &dyn Fn(&str) -> bool,
666) -> Vec<AddressRun> {
667  let domain_at = |from: usize| -> Option<(usize, String)> {
668    let domain: String = text[from..]
669      .chars()
670      .take_while(|c| c.is_ascii_alphanumeric() || ".-".contains(*c))
671      .collect();
672    let domain = domain.trim_end_matches(['.', '-']);
673    (domain.contains('.') && !domain.starts_with('.') && !domain.starts_with('-'))
674      .then(|| (from + domain.len(), domain.to_string()))
675  };
676  let mut runs: Vec<AddressRun> = Vec::new();
677  // (braced groups; a bracketed list of local parts before its domain reads as one, `Email: [i.iudice,
678  // d.pascarella]@cira.it`, 2405.17925: 107c)
679  let mut from = 0;
680  while let Some(open) = text[from..].find(['{', '[']).map(|o| from + o) {
681    let (opening, closing) = if text[open..].starts_with('{') {
682      ('{', '}')
683    } else {
684      ('[', ']')
685    };
686    let Some(close) = text[open + 1..]
687      .find([opening, closing])
688      .map(|c| open + 1 + c)
689      .filter(|&c| text[c..].starts_with(closing))
690    else {
691      from = open + 1;
692      continue;
693    };
694    let inner = &text[open + 1..close];
695    let after = &text[close + 1..];
696    let local = |p: &str| !p.is_empty() && p.chars().all(is_local_char);
697    // (spaces round the `@` are the group's: `\{ ann, bob \} @x.org`, 2311.14729; `\{aremonda, skrebs\} @
698    // know-center.at`, 2104.11106, `{\tt \{ tchakra2, rao \} @ asu.edu}`, 1707.04775: 107c)
699    let gap = after.len() - after.trim_start().len();
700    let at_gap = after.trim_start().strip_prefix('@').map_or(0, |domain| {
701      domain.len() - domain.trim_start_matches([' ', '\t']).len()
702    });
703    let found = if let Some((end, domain)) = after
704      .trim_start()
705      .strip_prefix('@')
706      .and_then(|_| domain_at(close + 2 + gap + at_gap))
707    {
708      let locals: Vec<&str> = inner
709        .split(|c: char| c.is_whitespace() || ADDRESS_GROUP_SEPARATORS.contains(&c))
710        .filter(|p| !p.is_empty())
711        .collect();
712      (!locals.is_empty() && locals.iter().all(|p| local(p)))
713        .then(|| (end, locals.iter().map(|l| s!("{l}@{domain}")).collect()))
714    } else if opening == '[' {
715      None
716    } else if let Some((group, taken)) = domain_after_group(inner, after) {
717      Some((close + 1 + taken, group))
718    } else if let Some(group) = domain_inside_group(inner) {
719      Some((close + 1, group))
720    } else if inner.contains('@') && only_addresses(inner) {
721      Some((close + 1, plain_addresses(inner)))
722    } else {
723      None
724    };
725    match found {
726      Some((end, addresses)) if !addresses.is_empty() => {
727        runs.push(AddressRun {
728          start: open,
729          end,
730          addresses,
731          label: None,
732        });
733        from = end;
734      },
735      _ => from = close + 1,
736    }
737  }
738  // (plain addresses and lists of local parts, outside the groups)
739  let in_group = |at: usize, runs: &[AddressRun]| runs.iter().any(|r| r.start <= at && at < r.end);
740  let boundary_before = |at: usize| {
741    text[..at]
742      .chars()
743      .next_back()
744      .is_none_or(|c| c.is_whitespace() || "({[<:;,|".contains(c))
745  };
746  let mut plain: Vec<AddressRun> = Vec::new();
747  for (at, _) in text.match_indices('@') {
748    if in_group(at, &runs) || plain.iter().any(|r| r.start <= at && at < r.end) {
749      continue;
750    }
751    let Some((end, domain)) = domain_at(at + 1) else {
752      continue;
753    };
754    let mut start = at;
755    while let Some(c) = text[..start].chars().next_back()
756      && is_local_char(c)
757    {
758      start -= c.len_utf8();
759    }
760    while text[start..at].starts_with('.') {
761      start += 1;
762    }
763    if start == at || !boundary_before(start) {
764      continue;
765    }
766    // (the list of local parts before it: `ann,bob@x`, `ann, bob@x`)
767    let mut locals: Vec<(usize, &str)> = Vec::new();
768    let mut spaced = false;
769    let mut head = start;
770    loop {
771      let before = text[..head].trim_end();
772      let gap = head - before.len();
773      // (a comma parts them, or a `|`: `robert.holland15|e.chotzoglou16|b.kainz@imperial.ac.uk`, 1909.00276 — the
774      // author block's reader's parting too, `email_addresses`)
775      let Some(comma) = before
776        .strip_suffix(',')
777        .or_else(|| before.strip_suffix('|'))
778      else {
779        break;
780      };
781      let word_end = comma.len();
782      let mut word_start = word_end;
783      while let Some(c) = text[..word_start].chars().next_back()
784        && is_local_char(c)
785      {
786        word_start -= c.len_utf8();
787      }
788      let word = &text[word_start..word_end];
789      // (each part a local part of letters: a digit run before the address is no part of a list — `Tel:5551234,info@x.org`
790      // is no `5551234@x.org`, `ORCID 0000-0002-1825-0097,ann@x.org` no `0000-…@x.org`; brief 95c r2 B3. A part with
791      // no space after its comma takes the unspaced list's checks, the readers' one test
792      // `classify::is_listed_local_part`; a part with one, the spaced-list rule's word test alone, so a labelled
793      // `E-mail: ann, b@x.org` keeps `ann@` — review rev103a2 B1)
794      let part_ok = if gap > 0 {
795        !word.is_empty()
796          && !word.starts_with('.')
797          && !word.chars().any(|c| c.is_ascii_uppercase())
798          && word.chars().any(|c| c.is_ascii_alphabetic())
799      } else {
800        is_listed_local_part(word)
801      };
802      if !part_ok || !boundary_before(word_start) {
803        break;
804      }
805      spaced |= gap > 0;
806      locals.push((word_start, word));
807      head = word_start;
808    }
809    let local = &text[start..at];
810    let opens = |at: usize| {
811      let before = text[..at].trim_end();
812      before.is_empty()
813        || before.ends_with(['(', '{', '[', '<', ':'])
814        || label_word_before(text, at, false).is_some()
815    };
816    // (an initial ends no unspaced list: `smith,j@x.org` — the readers' one test, `classify::ends_a_listed_run`; a
817    // spaced one, `E-mail: ann, b@x.org`, the spaced-list rule's)
818    let list_ok = !locals.is_empty()
819      && (spaced || ends_a_listed_run(local))
820      && (!spaced
821        || locals.iter().all(|(_, w)| spells(w)) && spells(local)
822        || opens(locals.last().map_or(start, |(s, _)| *s)));
823    let (run_start, mut addresses) = if list_ok {
824      let first = locals.last().map_or(start, |(s, _)| *s);
825      (
826        first,
827        locals
828          .iter()
829          .rev()
830          .map(|(_, w)| s!("{w}@{domain}"))
831          .collect::<Vec<_>>(),
832      )
833    } else {
834      (start, Vec::new())
835    };
836    addresses.push(s!("{local}@{domain}"));
837    plain.push(AddressRun {
838      start: run_start,
839      end,
840      addresses,
841      label: None,
842    });
843  }
844  runs.extend(plain);
845  runs.sort_by_key(|r| r.start);
846  // (runs separated only by separators, one run)
847  let mut merged: Vec<AddressRun> = Vec::new();
848  for run in runs {
849    if let Some(last) = merged.last_mut()
850      && text[last.end..run.start]
851        .split(|c: char| c.is_whitespace() || ",;&".contains(c))
852        .all(|w| w.is_empty() || w.eq_ignore_ascii_case("and"))
853    {
854      last.end = run.end;
855      last.addresses.extend(run.addresses);
856      continue;
857    }
858    merged.push(run);
859  }
860  // (its label word)
861  for run in &mut merged {
862    if let Some(label) = label_word_before(text, run.start, ends_a_list_item(text, run.end)) {
863      run.label = Some(text[label..run.start].trim_end().to_string());
864      run.start = label;
865    }
866  }
867  // (a run out of a text only at a list boundary: brief 95c r2 B2)
868  merged.retain(|run| at_list_boundary(text, run));
869  // (the parentheses round it)
870  for run in &mut merged {
871    let before = text[..run.start].trim_end();
872    let after = text[run.end..].trim_start();
873    for (open, close) in [('(', ')'), ('[', ']'), ('<', '>')] {
874      if before.ends_with(open) && after.starts_with(close) {
875        run.start = before.len() - 1;
876        run.end = text.len() - after.len() + 1;
877        break;
878      }
879    }
880  }
881  merged
882}

884/// Whether the address run `run` of `text` stands at a list boundary, so a reading may take it out of the text (brief
885/// 95c r2 B2): on one side at least the text's or a line's end (a `\n`, a line break), a list's separator (`,` `;`), a
886/// bracket (an opening one or a closing one before it, a closing one after it: `(…)`, `[…]`, `<…>`) — before it, its
887/// label word too ([`label_word_before`]); after it, a period ending the text or the line ([`ends_a_list_item`]) —, on
888/// the other no lowercase word without a colon (r3 F5: `Univ X, see also data@x.org`, `Univ W, see sales@w.org, City
889/// W`, `write to a@z.org.` are prose), and no parenthesised tag after it. A run with a plain word on both sides is
890/// prose (`write to info@x.org or data@x.org for data`), and one a tag follows is tagged (`ann@y.org (Ann), bob@y.org
891/// (Bob)`, `zhanghuamaths@163.com(H. Zhang)`, 0811.1419): it stays where it prints, as today's prints it (the
892/// fail-safe). One side is enough when the other is no such word: `Correspondence: {…}@westlake.edu.cn` (2601.07312),
893/// `SPAIN. {argimiro.arratia, …}@upc.edu` (2509.12185), `KTH, Sweden sbroome,…@kth.se` (2112.12175), `…, Russia.
894/// …@…` (2112.04350).
895fn at_list_boundary(text: &str, run: &AddressRun) -> bool {
896  let blank = |c: char| c.is_whitespace() && c != '\n';
897  let before = text[..run.start].trim_end_matches(blank);
898  let after = text[run.end..].trim_start_matches(blank);
899  let before_ok = run.label.is_some()
900    || before.is_empty()
901    || before.ends_with(['\n', ',', ';', '(', '[', '<', ')', ']', '>']);
902  let after_ok = ends_a_list_item(text, run.end);
903  // (a lowercase word without a colon beside it: prose's)
904  let prose = |word: Option<&str>| {
905    word.is_some_and(|w| w.chars().next().is_some_and(|c| c.is_lowercase()) && !w.ends_with(':'))
906  };
907  let word_before = before
908    .rsplit(char::is_whitespace)
909    .next()
910    .filter(|w| !w.is_empty());
911  let word_after = after
912    .split(char::is_whitespace)
913    .next()
914    .filter(|w| !w.is_empty());
915  let tagged = after.starts_with(['(', '[']);
916  // (a contact's words and "to" before it are its label, not prose: `Corresponding\ to\ lingxiao.wang@…`
917  // (2606.31381), "Correspondence to"; "write to a@z.org." stays prose: 107c)
918  let label_to = word_before.is_some_and(|word| word.eq_ignore_ascii_case("to"))
919    && before
920      .trim_end()
921      .rsplit_once(char::is_whitespace)
922      .and_then(|(head, _)| head.trim_end().rsplit(char::is_whitespace).next())
923      .is_some_and(names_a_contact);
924  ((before_ok && (after_ok || !prose(word_after)))
925    || (after_ok && (label_to || !prose(word_before))))
926    && !tagged
927}

1015/// leaves stranded, each read within its line (a `\n`, a line break, ends one: brief 95c r2 N2): a run ending its line
1016/// takes the separators before it (`Trieste, Italy, (ann@x.org)` leaves `Trieste, Italy`; cond-mat0607219's `FAX:
1017/// 951-827-2425; email: rlake@…` its `FAX: 951-827-2425` before the break), a run opening it those after it, a run

1019/// contact's e-mail unit took them (the old split's `row_email_calls`, deleted at P27b: `CVLab, EPFL, \\ \texttt{\{…\}@epfl.ch}` leaves
1020/// `CVLab, EPFL`, 2407.18381) —, a run between two separators the one after it (`FAX: 1, email: a@x.org, Jan.
1021/// 4` leaves `FAX: 1, Jan. 4`), a run between a word and a separator the spaces before it (`Univ X (a@x.org), City`
1022/// leaves `Univ X, City`); and a cut between two spaces one of them (N1: `Univ Y, ann@x.org, Phone` leaves `Univ Y,
1023/// Phone`).
1024pub(in crate::base_utilities::frontmatter::observer) fn address_cuts(
1025  text: &str,
1026  runs: &[AddressRun],
1027) -> Vec<(usize, usize)> {
1028  const SEPARATORS: &[char] = &[',', ';', ':', '-', '\u{2013}', '\u{2014}', '/'];
1029  let blank = |c: char| c.is_whitespace() && c != '\n';
1030  let stray = |c: char| blank(c) || SEPARATORS.contains(&c);
1031  let mut cuts: Vec<(usize, usize)> = Vec::new();
1032  for run in runs {
1033    let (mut start, mut end) = (run.start, run.end);
1034    let line_start = text[..start].rfind('\n').map_or(0, |p| p + 1);
1035    let line_end = text[end..].find('\n').map_or(text.len(), |p| end + p);
1036    let before = text[line_start..start].trim_end_matches(blank);
1037    let after = text[end..line_end].trim_start_matches(blank);
1038    if after.chars().all(|c| stray(c) || c == '.') {
1039      end = line_end;
1040      start = line_start + text[line_start..start].trim_end_matches(stray).len();
1041      // (the whole line: its break too)
1042      if start == line_start {
1043        if line_end < text.len() {
1044          end = line_end + 1;
1045        } else if line_start > 0 {
1046          start = text[..line_start - 1].trim_end_matches(stray).len();
1047        }
1048      }
1049    } else if before.chars().all(stray) {
1050      start = line_start;
1051      end = line_end - text[end..line_end].trim_start_matches(stray).len();
1052    } else if before.ends_with([',', ';']) && after.starts_with([',', ';']) {
1053      end = line_end - after.len() + 1;
1054    } else if after.starts_with([',', ';']) {
1055      start = line_start + before.len();
1056    } else if before.ends_with([',', ';']) && after.starts_with([')', ']', '>']) {
1057      start = line_start + before.trim_end_matches(stray).len();
1058    }
1059    // (one space of two)
1060    if text[..start].ends_with(blank) && text[end..].starts_with(blank) {
1061      end = text.len() - text[end..].trim_start_matches(blank).len();
1062    }
1063    match cuts.last_mut() {
1064      Some(last) if last.1 >= start => last.1 = last.1.max(end),
1065      _ => cuts.push((start, end)),
1066    }
1067  }
1068  cuts
1069}

1558/// `label` as two or more labels of `known` written together (`12` → `1`, `2`); with `symbols`, a run of characters that
1559/// are no letter or digit may stand alone too (`1⋆` → `1`, `⋆`), and so may one letter (`1a` → `1`, `a`: the letter a
1560/// note's mark, `$^{1{\rm a}}$`, 1904.02582), and, with `note_numbers` (the emitter's switch on), so may a number when
1561/// no known label is one (`a1` → `a`, `1`: the number a note's mark, `$^{\a 1}$` under entries `a`, `b`,
1562/// hep-ph0011101), as long as one piece is known. The most known pieces win, then the fewest
1563/// pieces: a symbol run spelling known labels is parted into them before it stands alone as one piece (`†♮♯` under
1564/// entries `†`, `♮`, `♯`: `$^{\dagger\natural\sharp}$`, 1004.1402). A combining mark is part of the character before
1565/// it, never parted from it (`\~{x}`'s `x̃` is no `x` and a `̃`, review rev92b_r2 B1). A run of digits is one number,
1566/// parted into its digits only under single-digit numbering — every number `known` holds one digit, as the names' side
1567/// of `part_glued_labels` reads a number —: BESIII's sub-lettered `23A` under entries numbered 1 to 70 is `23` and `A`,
1568/// never `2`, `3` and `A` (2007.02542, 36 sub-lettered authors each the institutes 2 and 3; 1507.02068's `48A`,
1569/// 2411.07730, 2606.14105; review rev98e B1). A label holding a backslash is
1570/// a call's label as its source spells it, not as it prints: only what it spells before its first command is read, as
1571/// the labels that part (authblk's `\author[3\authfn{2}]{…}`, which prints `3†`, is `3` — 2405.16051: parted whole,
1572/// its command's letters were labels and its argument the `2` of `\affil[1,2,4]`; `\author[1\thanks{…}]{…}` is `1`,
1573/// 2202.05736, 2410.10922).
1574pub(super) fn parted_label(
1575  label: &str,
1576  known: &[String],
1577  symbols: bool,
1578  note_numbers: bool,
1579) -> Option<Vec<String>> {
1580  if let Some(at) = label.find('\\') {
1581    let spelled = label[..at].trim();
1582    return match parted_label(spelled, known, symbols, note_numbers) {
1583      Some(parts) => Some(parts),
1584      None => known
1585        .iter()
1586        .any(|k| k == spelled)
1587        .then(|| vec![spelled.to_string()]),
1588    };
1589  }
1590  // (each character with the combining marks on it)
1591  let chars: Vec<(usize, char)> = label
1592    .char_indices()
1593    .filter(|(at, c)| *at == 0 || !is_combining_mark(*c))
1594    .collect();
1595  let n = chars.len();
1596  if n < 2 {
1597    return None;
1598  }
1599  let at = |i: usize| chars.get(i).map_or(label.len(), |(b, _)| *b);
1600  // (a cut between two digits only under single-digit numbering)
1601  let single_digit = known
1602    .iter()
1603    .filter(|k| !k.is_empty() && k.chars().all(|c| c.is_ascii_digit()))
1604    .all(|k| k.len() == 1);
1605  let digit_at = |i: usize| chars.get(i).is_some_and(|(_, c)| c.is_ascii_digit());
1606  let cuts = |i: usize| single_digit || i == 0 || i == n || !(digit_at(i - 1) && digit_at(i));
1607  // (under entries no number labels, a number is no entry's label but a note's mark, as a letter is under numbered
1608  // entries: `$^{\a 1}$` printing "a1" under entries `a`, `b` is address a and the footnote "$^1$presenter at
1609  // conference", hep-ph0011101 — never one label no entry holds)
1610  let lettered = note_numbers
1611    && !known.is_empty()
1612    && known
1613      .iter()
1614      .all(|k| !k.is_empty() && !k.chars().all(|c| c.is_ascii_digit()));
1615  // best[i]: the fewest pieces covering label[..i], with how many are known, and the piece before
1616  let mut best: Vec<Option<(usize, usize, usize)>> = vec![None; n + 1];
1617  best[0] = Some((0, 0, 0));
1618  for i in 0..n {
1619    let Some((count, hits, _)) = best[i] else {
1620      continue;
1621    };
1622    for j in (i + 1..=n).filter(|&j| cuts(j)) {
1623      let piece = &label[at(i)..at(j)];
1624      let is_known = known.iter().any(|k| k == piece);
1625      let is_symbol = symbols && piece.chars().all(|c| !c.is_alphanumeric());
1626      let mut bases = piece.chars().filter(|c| !is_combining_mark(*c));
1627      let is_letter =
1628        symbols && bases.next().is_some_and(char::is_alphabetic) && bases.next().is_none();
1629      let is_note_number = symbols && lettered && piece.chars().all(|c| c.is_ascii_digit());
1630      if !is_known && !is_symbol && !is_letter && !is_note_number {
1631        continue;
1632      }
1633      let candidate = (count + 1, hits + usize::from(is_known), i);
1634      if best[j].is_none_or(|(c, h, _)| {
1635        (std::cmp::Reverse(candidate.1), candidate.0) < (std::cmp::Reverse(h), c)
1636      }) {
1637        best[j] = Some(candidate);
1638      }
1639    }
1640  }
1641  let (count, hits, _) = best[n]?;
1642  if count < 2 || hits == 0 {
1643    return None;
1644  }
1645  let mut parts = Vec::new();
1646  let mut j = n;
1647  while j > 0 {
1648    let (_, _, i) = best[j]?;
1649    parts.push(label[at(i)..at(j)].to_string());
1650    j = i;
1651  }
1652  parts.reverse();
1653  Some(parts)
1654}
