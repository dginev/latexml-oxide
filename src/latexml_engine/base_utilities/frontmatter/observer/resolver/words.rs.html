14/// `text`'s runs of letters and digits, as printed.
15pub(in crate::base_utilities::frontmatter::observer) fn printed_words(
16  text: &str,
17) -> impl Iterator<Item = &str> {
18  text
19    .split(|c: char| !c.is_alphanumeric())
20    .filter(|w| !w.is_empty())
21}

23/// Text compared by its letters and digits alone, lowercased (the deleted `ab_audit fmshadow`'s `norm`): how the emitter
24/// keys an item, and [`entry_span_in`] reads its entry.
25pub(in crate::base_utilities::frontmatter::observer) fn norm(text: &str) -> String {
26  text
27    .chars()
28    .filter(|c| c.is_alphanumeric())
29    .flat_map(char::to_lowercase)
30    .collect()
31}

33/// The span of a block's entry the reading keys `key` (its text by [`norm`]), its source `span` its first event's (an entry
34/// of several pieces, `$^1$Department of Physics, Univ X, Japan`, is sourced at its first Name: 2307.03384): through the
35/// `events` printed after it, until their ink joined is the entry's text — over the lines an unmarked line continued, past
36/// the layout between (an image: `Zebra University, \includegraphics{…} \url{…}` one affiliation, review revIMG1
37/// finding 1) — `None` when it never is (a note, a mark, or other ink first). The span the emitter writes the entry from
38/// (`emitter::repart::entry_span`), and what holds a block's pictures ([`Written::unwritten_pictures`]).
39pub(in crate::base_utilities::frontmatter::observer) fn entry_span_in(
40  events: &[Event],
41  span: (u16, u16),
42  key: &str,
43) -> Option<(u16, u16)> {
44  let first = events
45    .iter()
46    .position(|e| e.kind != EventKind::Layout && e.span.is_some_and(|s| s.0 == span.0))?;
47  let (mut joined, mut end) = (String::new(), span.1);
48  for event in &events[first..] {
49    match &event.kind {
50      EventKind::Name | EventKind::Contact => joined.push_str(&norm(&event.text)),
51      // (the word "and" inside an entry is the entry's: `Computer Science and Engineering`, 1304.1574)
52      EventKind::Sep(SepKind::Word(word)) => joined.push_str(&norm(word)),
53      EventKind::Sep(_) | EventKind::Break | EventKind::Layout => continue,
54      _ => return None,
55    }
56    if let Some((_, e)) = event.span {
57      end = end.max(e);
58    }
59    if joined == key {
60      return Some((span.0, end));
61    }
62    if !key.starts_with(&joined) {
63      return None;
64    }
65  }
66  None
67}

76/// Where the addresses of `run`, an address run of `text`, begin: after its label word (a label is the contact's name,
77/// written or not, `email:` before `Duncan.Lorimer@mail.wvu.edu`, 0811.0762), else at the run.
78fn address_start(text: &str, run: &AddressRun) -> usize {
79  run
80    .label
81    .as_deref()
82    .and_then(|label| {
83      text[run.start..run.end]
84        .find(label)
85        .map(|at| run.start + at + label.len())
86    })
87    .unwrap_or(run.start)
88}

90/// What `text` prints, as the content check counts it (`placed`): its words ([`printed_words`]) outside its address runs
91/// ([`address_runs`]), and the addresses those runs print, lowercased, a list's each (`{ann,bob}@x.org` two). An address's
92/// local-part and domain words are the address's, never a word of the text: a domain `idsia` writes no `IDSIA`
93/// (1605.03019, review revN1r3).
94pub(in crate::base_utilities::frontmatter::observer) fn ink_words(
95  text: &str,
96) -> (Vec<String>, Vec<String>) {
97  if !text.contains('@') {
98    return (
99      printed_words(text).map(str::to_string).collect(),
100      Vec::new(),
101    );
102  }
103  // (the braces of a group set in math, as [`addresses_in`] reads them: `$\{$fargyriou,mikebekos$\}$@math.ntua.gr`, 1106.2694;
104  // and a tie a raw address text keeps, the space it prints — llncs's `\email{$\{$stomforde$|$bsick$\}$@uni-kassel.de;~cms@…}`,
105  // whose `cms@sra.uni-hanover.de` the writer reads past the tie as the e-mail readers do (`email_block_calls`), was no run
106  // here and its domain words counted as printed words, missing: 1701.08125, `perfect_kernel_batch85b::llncs_listed_local_parts_keep_their_braces`)
107  let unshifted;
108  let text = if text.contains(['$', '~']) {
109    unshifted = text.replace('$', "").replace('~', " ");
110    unshifted.as_str()
111  } else {
112    text
113  };
114  let runs = address_runs(text, &|_| false);
115  let mut cut: Vec<(usize, usize)> = runs
116    .iter()
117    .map(|run| (address_start(text, run), run.end))
118    .collect();
119  let mut addresses: Vec<String> = runs
120    .into_iter()
121    .flat_map(|run| run.addresses)
122    .map(|a| a.to_lowercase())
123    .collect();
124  // (and an address no run reads, glued to what prints before or after it: `——meizeh@cs.technion.ac.il`, 1503.06438; `†caprini@…`,
125  // 1202.5401), from its first letter or digit to its last
126  for token in text.split_whitespace() {
127    let from = token.as_ptr() as usize - text.as_ptr() as usize;
128    let to = from + token.len();
129    let address = token.trim_matches(|c: char| !c.is_alphanumeric());
130    let read = cut.iter().any(|&(s, t)| s < to && from < t);
131    if !read
132      && address
133        .split_once('@')
134        .is_some_and(|(local, domain)| !local.is_empty() && domain.contains('.'))
135    {
136      cut.push((from, to));
137      addresses.push(address.to_lowercase());
138    }
139  }
140  let words = words_at(text)
141    .filter(|((from, to), _)| !cut.iter().any(|&(s, t)| s <= *from && *to <= t))
142    .map(|(_, word)| word.to_string())
143    .collect();
144  (words, addresses)
145}

147/// The e-mail label the paper prints before `address` at `span` of a block's `events`, as the reading reads an address run's
148/// label on its line ([`address_runs`]' `label_word_before`): the contact's `name`, written as printed. "E-mail:" before a
149/// class's `\email{…}` (PoS's, 1412.5940, 1612.08209, 1812.02252, 1907.01311, 1005.2532, 0802.3384), a list's on each of its
150/// addresses ("Emails:", 2511.22090), "email" without a colon (1209.0869), "e-mail:" in parentheses (1405.3793), "Email:" before a
151/// tabular's marked addresses (2607.12142), a label and an address in one ink (`Email: wangfy@bnu.edu.cn`, 1009.1965; 1807.01821,
152/// 1509.01480, 2205.04084, 0704.3639); in any language, by its place: one word or two and a colon, all the line prints before
153/// the address's run (`Correo electrónico:`, `Courriel :`).
154pub(in crate::base_utilities::frontmatter::observer) fn printed_email_label(
155  events: &[Event],
156  span: (u16, u16),
157  address: &str,
158) -> Option<String> {
159  if let Some(name) = own_email_label(events, span) {
160    return Some(name);
161  }
162  let at = events.iter().position(|e| {
163    matches!(e.kind, EventKind::Name | EventKind::Contact)
164      && e.span.is_some_and(|s| s.0 <= span.0 && span.1 <= s.1)
165  })?;
166  let start = events[..at]
167    .iter()
168    .rposition(|e| e.kind == EventKind::Break)
169    .map_or(0, |at| at + 1);
170  let end = events[at..]
171    .iter()
172    .position(|e| e.kind == EventKind::Break)
173    .map_or(events.len(), |n| at + n);
174  let line: Vec<&Event> = events[start..end].iter().collect();
175  let text = events_text(&line);
176  // (the run holding the address, from the ink the reading read it at on: its label's, `E-mail:` before `\email{…}`, or its
177  // own, the run's)
178  let ink = events_text(&line[..at - start]).len();
179  let runs = address_runs(&text, &|_| false);
180  let after = || runs.iter().filter(|run| run.end > ink);
181  let run = after()
182    .find(|run| run.addresses.iter().any(|a| a == address))
183    .or_else(|| after().next())?;
184  // (or, in any language, what its line prints before the run, one word or two before a colon: `Correo electrónico:`,
185  // `Courriel :`, review revZ4br2)
186  let before = text[..run.start].trim();
187  run
188    .label
189    .clone()
190    .or_else(|| reads_as_email_label(before).then(|| before.to_string()))
191}

193/// The label the class's own e-mail command prints with the address at `span` of a block's `events`, read as it prints —
194/// jfm/jpp's `\email` printing "Email address for correspondence:" inside a `\thanks`, 2603.18265: the contact's `name`,
195/// the rehearsal's `own_email_properties`. The command's call prints it with the address wherever it is written whole
196/// (`insert::NAME_PRINTED`).
197pub(in crate::base_utilities::frontmatter::observer) fn own_email_label(
198  events: &[Event],
199  span: (u16, u16),
200) -> Option<String> {
201  let at_span = |e: &&Event| e.kind == EventKind::Contact && e.span == Some(span);
202  events
203    .iter()
204    .flat_map(|e| std::iter::once(e).chain(&e.inner))
205    .find(at_span)
206    .and_then(|e| e.name.as_deref())
207    .map(|name| name.trim().to_string())
208}

210/// Whether `before`, what a line prints before an address's run, is the address's label by its place, in any language: one
211/// word or two before a colon (`Correo electrónico:`, `Courriel :`, review revZ4br2).
212pub(in crate::base_utilities::frontmatter::observer) fn reads_as_email_label(before: &str) -> bool {
213  before.ends_with(':')
214    && before.split_whitespace().count() <= 2
215    && before.chars().any(char::is_alphabetic)
216}

218/// An ink of a block or a call whose words the reading writes nowhere ([`Written::lost_inks`]): its span, its text, the
219/// words lost, as printed, where they print in it, the address runs it prints that the reading writes as its e-mail
220/// items (cut out of a note of it), its owner, and whether it fails the block's names.
221#[derive(Debug)]
222pub(in crate::base_utilities::frontmatter::observer) struct LostInk {
223  pub span:       Option<(u16, u16)>,
224  pub text:       String,
225  pub words:      Vec<String>,
226  /// the byte ranges of `text` its lost words print at, in print order
227  lost_at:        Vec<(usize, usize)>,
228  /// the byte ranges of `text` its written address runs print at ([`address_runs`], each address an e-mail item the
229  /// reading writes for the block)
230  cut:            Vec<(usize, usize)>,
231  /// the creator (its index in [`Resolved::creators`]) whose line it is, by the reading's own rule of an address
232  /// ([`spelled_name`]): the one author the addresses its ink prints spell — `POC: K. Hobbs (kerianne.hobbs@us.af.mil)`
233  /// Kerianne L. Hobbs's (2110.03506), `Germany (martin.waltz@tu-dresden.de)` Martin Waltz's (2203.10777),
234  /// `yiliu.li@outlook` Aiden Yiliu Li's (2605.16481); not by its line's, a list's local parts printed apart owning none
235  pub owner:      Option<usize>,
236  /// whether it fails the block's names: a Name ink of a block losing a word of letters outside its addresses (a postcode
237  /// is no name's, 2501.09580's `Hefei 230026`, its block of 168 names written whole for it; nor an address's word,
238  /// 2601.00680), no author's line by its addresses, that is a name: on a line of its block's names, or one an address
239  /// its ink, its line or its column prints spells by the reading's rule — a name the reading lost, JAIR's `\name Shaul Markovitch
240  /// \email shaulm@…` (1109.6618; 1312.0286, 1111.6082, 1905.12960, 1807.03929, 2003.01013, 2008.13618; a column's own,
241  /// 1805.08288, 2002.00084: review revN1); not a label (`Emails:- …`, 2605.21396) or an entry's line (1109.0220)
242  pub names_fail: bool,
243  /// its font as digested (a Name's first character's), which a note of it keeps
244  pub font:       Option<Rc<Font>>,
245}

247impl LostInk {
248  /// What a note of it holds: its text from its first lost word to its last, as printed, where no written address run
249  /// prints among them (`POC: K. Hobbs` of `POC: K. Hobbs (kerianne.hobbs@us.af.mil)`, the address its contact's, 2110.03506;
250  /// `Correo electrónico:` of `Correo electrónico: ann@a.org`, review revZ4br2), by position, not by word (a word the
251  /// address repeats, `Hobbs`, is the address's where it prints in it: review revN1); its whole text, as the base wrote
252  /// it, where an address it writes prints among them.
253  pub fn note_text(&self) -> &str {
254    match (self.lost_at.first(), self.lost_at.last()) {
255      (Some(&(first, _)), Some(&(_, last)))
256        if !self.cut.iter().any(|&(from, to)| first < to && from < last) =>
257      {
258        &self.text[first..last]
259      },
260      _ => &self.text,
261    }
262  }
263
264  /// Whether it loses every word it prints, no address of it written: its tokens are its note.
265  pub fn whole(&self) -> bool {
266    self.cut.is_empty() && printed_words(&self.text).count() == self.words.len()
267  }
268}

270/// `text`'s printed words ([`printed_words`]) with their byte ranges.
271fn words_at(text: &str) -> impl Iterator<Item = ((usize, usize), &str)> {
272  printed_words(text).map(move |word| {
273    // (the word's offset: `printed_words` yields subslices of the text)
274    let start = word.as_ptr() as usize - text.as_ptr() as usize;
275    ((start, start + word.len()), word)
276  })
277}

279/// What the reading writes, by source: the words of its names, of its items' texts (an item each owner holds a copy of
280/// once) and of an e-mail's printed label ([`printed_email_label`]), as multisets per block or call; every word it or a
281/// document's note writes (a call's note may go there: hep-ex0609057's `\thanks` in an `\affiliation`, a publication note);
282/// where its items are read, and its names.
283pub(in crate::base_utilities::frontmatter::observer) struct Written {
284  counts:    BTreeMap<usize, BTreeMap<String, usize>>,
285  anywhere:  BTreeSet<String>,
286  /// the spans items are read at, and those of a contact the emit writes from its tokens (not an address, written as its
287  /// text)
288  spans:     BTreeSet<(usize, (u16, u16))>,
289  contacts:  BTreeSet<(usize, (u16, u16))>,
290  /// the spans of the items it writes, none a ruling dropped: what holds a block's pictures ([`Self::unwritten_pictures`])
291  holding:   BTreeSet<(usize, (u16, u16))>,
292  /// per source, its creators (their index in [`Resolved::creators`]) and names
293  names:     BTreeMap<usize, Vec<(usize, String)>>,
294  /// per source, the e-mail addresses it writes as items, lowercased, as often as one holder holds each: what an ink's
295  /// address run is read as ([`Self::lost_inks`])
296  addresses: BTreeMap<usize, BTreeMap<String, usize>>,
297  /// per source, the words of its names and of the items it writes, as a multiset, and of its names alone: what
298  /// [`Self::excess`] reads (no label; no note a ruling dropped; of an address its local part, a list's domain printed once
299  /// for its addresses, `{a,b}@x`, and an address list written out not at all)
300  read:      BTreeMap<usize, BTreeMap<String, usize>>,
301  named:     BTreeMap<usize, BTreeMap<String, usize>>,
302}
303
304impl Written {
305  /// `entry_key` gives an item's key as the emit reads its entry by (`emitter::plan::written_keys`: an ORCID by its
306  /// identifier, a picture note by its place), so each item holds the span the emit writes it from.
307  pub(in crate::base_utilities::frontmatter::observer) fn new(
308    ledger: &Ledger,
309    resolved: &Resolved,
310    entry_key: impl Fn(&Assoc) -> String,
311  ) -> Self {
312    let mut counts: BTreeMap<usize, BTreeMap<String, usize>> = BTreeMap::new();
313    let mut write = |at: usize, text: &str| {
314      let counts = counts.entry(at).or_default();
315      for word in words_of(text) {
316        *counts.entry(word).or_default() += 1;
317      }
318    };
319    let mut names: BTreeMap<usize, Vec<(usize, String)>> = BTreeMap::new();
320    let (mut read, mut named): (BTreeMap<usize, BTreeMap<String, usize>>, BTreeMap<_, _>) =
321      Default::default();
322    let tally = |counts: &mut BTreeMap<String, usize>, text: &str| {
323      for word in words_of(text) {
324        *counts.entry(word).or_default() += 1;
325      }
326    };
327    for (i, c) in resolved.creators.iter().enumerate() {
328      tally(read.entry(c.source).or_default(), &c.name);
329      tally(named.entry(c.source).or_default(), &c.name);
330      write(c.source, &c.name);
331      names.entry(c.source).or_default().push((i, c.name.clone()));
332    }
333    for (at, _, word) in &resolved.drops {
334      write(*at, word);
335    }
336    // (an item as often as one holder holds it: a copy each owner holds counts once, two in one holder twice, 2607.00684's
337    // `mtablo@postech.ac.kr` written twice for the block)
338    let mut items: BTreeMap<(Source, &str, &str), usize> = BTreeMap::new();
339    let mut unread: BTreeSet<(Source, &str, &str)> = BTreeSet::new();
340    let mut holding = BTreeSet::new();
341    let holders = resolved
342      .creators
343      .iter()
344      .map(|c| {
345        c.affiliations
346          .iter()
347          .chain(&c.contacts)
348          .chain(&c.notes)
349          .collect::<Vec<_>>()
350      })
351      .chain([
352        resolved.block.iter().collect(),
353        resolved.dropped.iter().collect(),
354      ]);
355    for held in holders {
356      let mut here: BTreeMap<(Source, &str, &str), usize> = BTreeMap::new();
357      for a in held {
358        if let Some((at, span)) = a.source {
359          let item = ((at, span), a.role.as_str(), a.text.as_str());
360          *here.entry(item).or_default() += 1;
361          if a.rule.contains("ruling3:written-out") || resolved.dropped.contains(a) {
362            unread.insert(item);
363          }
364          // (what it writes: its entry's span, the emit's (`entry_span_in`, by the emit's key: `entry_key`), wider than
365          // its source span where its text runs on past layout — an affiliation holding the URL after an image, `Zebra
366          // University, \includegraphics{…} \url{…}`, review revIMG1 finding 1 — and its source span where the entry
367          // can't be read so)
368          if let Some(span) = span
369            && !resolved.dropped.contains(a)
370          {
371            let written = match ledger.entries().get(at) {
372              Some(Entry::Author(block)) => match &block.outcome {
373                Outcome::Events { events, .. } => entry_span_in(events, span, &entry_key(a)),
374                _ => None,
375              },
376              _ => None,
377            };
378            holding.insert((at, written.unwrap_or(span)));
379          }
380        }
381      }
382      for (item, n) in here {
383        let most = items.entry(item).or_default();
384        *most = (*most).max(n);
385      }
386    }
387    let (mut spans, mut contacts) = (BTreeSet::new(), BTreeSet::new());
388    let mut addresses: BTreeMap<usize, BTreeMap<String, usize>> = BTreeMap::new();
389    let each = items
390      .iter()
391      .flat_map(|(item, n)| std::iter::repeat_n(*item, *n));
392    for item @ ((at, span), role, text) in each {
393      write(at, text);
394      if role == "email" {
395        *addresses
396          .entry(at)
397          .or_default()
398          .entry(text.to_lowercase())
399          .or_default() += 1;
400      }
401      if !unread.contains(&item) {
402        let own = match role {
403          "email" => text.split('@').next().unwrap_or(text),
404          _ => text,
405        };
406        let counts = read.entry(at).or_default();
407        tally(counts, own);
408        // (a note's inner address the reading writes as its contact counts once, in the contact: llncs's `\thanks{…
409        // \email{…}}`, 2406.13066; a note's call's too, siamart's `\thanks{… \email{…}}`, 1804.09455, whose `\email`
410        // (OmniBus's, locked) the rehearsal reads as the binding's, `EmailMeaning::Binding` (2511.14713), and digestion routes
411        // out. So an.cls's raw `\email` a `\thanks` keeps, 0807.0895, written in the note and its contact, goes uncounted)
412        let inner: &[Event] = match (span, ledger.entries().get(at)) {
413          (Some(span), Some(Entry::Author(block))) => match &block.outcome {
414            Outcome::Events { events, .. } => events
415              .iter()
416              .find(|e| e.kind == EventKind::Note && e.span == Some(span))
417              .map_or(&[], |note| &note.inner),
418            _ => &[],
419          },
420          (None, Some(Entry::Api(call))) if call.tag == "ltx:note" => &call.events,
421          _ => &[],
422        };
423        let routed = inner.iter().filter(|e| e.kind == EventKind::Contact);
424        for word in routed.flat_map(|e| words_of(&e.text)) {
425          counts.entry(word).and_modify(|n| *n = n.saturating_sub(1));
426        }
427      }
428      let Some(span) = span else {
429        continue;
430      };
431      spans.insert((at, span));
432      if role != "email" {
433        contacts.insert((at, span));
434      }
435      if role == "email"
436        && let Some(Entry::Author(block)) = ledger.entries().get(at)
437        && let Outcome::Events { events, .. } = &block.outcome
438        && let Some(label) = printed_email_label(events, span, text)
439      {
440        write(at, &label);
441      }
442    }
443    let notes = ledger.entries().iter().filter_map(|entry| match entry {
444      Entry::Api(call) if call.tag == "ltx:pubnote" => Some(events::printed_text(&call.events)),
445      _ => None,
446    });
447    let notes: Vec<String> = notes.collect();
448    let anywhere = counts
449      .values()
450      .flat_map(|counts| counts.keys().cloned())
451      .chain(notes.iter().flat_map(|note| words_of(note)))
452      .collect();
453    Self {
454      counts,
455      anywhere,
456      spans,
457      contacts,
458      holding,
459      names,
460      addresses,
461      read,
462      named,
463    }
464  }

466  /// The pictures of the block at ledger index `index` (an image, the Layout event of an `\includegraphics`; a picture note,
467  /// `events::PICTURE_NOTE`) that no item it writes holds: each one's span inside no written item's span, or no span to
468  /// tell (fail toward flagging). An image has no word, so the word check ([`Self::lost_inks`]) cannot miss it: the image
469  /// leading a mark or an address on a later line, which `events::place_pictures` has no place for
470  /// (`\includegraphics{…} $^{1}$Zebra University`, `\includegraphics{…} \texttt{ann@x.org}`), was lost silently. One a URL
471  /// contact's span holds (2503.08161's `hf-logo.png`, 2505.21327's icons) or its own note's is written.
472  pub(in crate::base_utilities::frontmatter::observer) fn unwritten_pictures(
473    &self,
474    index: usize,
475    events: &[Event],
476  ) -> usize {
477    let of_index = || {
478      self
479        .holding
480        .range((index, (0, 0))..=(index, (u16::MAX, u16::MAX)))
481    };
482    events
483      .iter()
484      .filter(|e| is_image(e) || events::is_picture_note(e))
485      .filter(|e| {
486        e.span
487          .is_none_or(|(s, t)| !of_index().any(|&(_, (from, to))| from <= s && t <= to))
488      })
489      .count()
490  }

492  /// The words the reading writes of the block or call at `index` more times than its `events` print them (the other side
493  /// of [`Self::lost_inks`]: a duplicated address, 2607.00684's `mtablo@postech.ac.kr` twice in one creator; an invented
494  /// word), each with whether its names alone do: its names' and items' words against what its inks and its separators'
495  /// words print, a link's target counting as printed where its anchor prints no address of its own: 2607.28418's
496  /// `\href{mailto:email@domain}{ann@a.edu; bob@b.edu}` prints no `email@domain` (reviews revZ4br2, revN1). A word of two
497  /// letters at most aside, a fragment or a mark.
498  pub(in crate::base_utilities::frontmatter::observer) fn excess(
499    &self,
500    index: usize,
501    events: &[Event],
502  ) -> Vec<(String, bool)> {
503    let mut printed: BTreeMap<String, usize> = BTreeMap::new();
504    let mut count = |text: &str| {
505      for word in words_of(text) {
506        *printed.entry(word).or_default() += 1;
507      }
508    };
509    for e in events {
510      match &e.kind {
511        EventKind::Name | EventKind::Contact | EventKind::Note => count(&e.text),
512        EventKind::Sep(SepKind::Word(word)) => count(word),
513        _ => {},
514      }
515      // (a link's target is what it carries where its anchor prints no address of its own, an ORCID's id behind a logo or a
516      // name, 2210.03748, 2206.03602, 2102.04676; an anchor printing addresses prints only them, its target none:
517      // `\href{mailto:email@domain}{ann@a.edu; bob@b.edu}`, 2607.28418, reviews revZ4br2, revN1)
518      for contact in std::iter::once(e).chain(&e.inner) {
519        if contact.kind == EventKind::Contact
520          && let Some(target) = contact.target.as_deref()
521          && addresses_in(&contact.text).is_empty()
522        {
523          count(target);
524        }
525      }
526    }
527    let printed_of = |word: &str| printed.get(word).copied().unwrap_or_default();
528    let named = self.named.get(&index);
529    let read = self.read.get(&index).into_iter().flatten();
530    let over = read.filter(|(word, n)| word.chars().count() > 2 && printed_of(word) < **n);
531    over
532      .map(|(word, _)| {
533        let by_names = named
534          .and_then(|named| named.get(word))
535          .is_some_and(|&n| printed_of(word) < n);
536        (word.clone(), by_names)
537      })
538      .collect()
539  }

541  /// Whether the ink `e` of the block or call at `index` is no loss: one a container in an item's span holds (a picture's in an
542  /// entry), a contact read at its own span (an ORCID link's logo, 2409.13798; not an address, 2607.28418), one a mark's
543  /// formula holds (its label or its logo: an ORCID icon, 2210.03748), and an ink of two letters at most, a fragment the
544  /// reading joins or reads as a mark (an accent's letter, `Karlu$\rm{\check{s}i\acute{c}}$`, 0911.1045; a label letter).
545  fn exempt(&self, index: usize, e: &Event, marks: &[(u16, u16)]) -> bool {
546    let within = |&(from, to): &(u16, u16)| {
547      e.span
548        .is_some_and(|(s, t)| from <= s && t <= to && (s, t) != (from, to))
549    };
550    let mut of_index = self
551      .spans
552      .range((index, (0, 0))..=(index, (u16::MAX, u16::MAX)));
553    e.text.chars().filter(|c| c.is_alphanumeric()).count() <= 2
554      || (e.kind == EventKind::Contact
555        && e.role.as_deref() != Some("email")
556        && e
557          .span
558          .is_some_and(|span| self.contacts.contains(&(index, span))))
559      || (e.nest > 0 && of_index.any(|(_, span)| within(span)))
560      || marks.iter().any(within)
561  }

563  /// The inks of `events`, a block's or a call's at ledger index `index`, whose words it doesn't write: a block's as a
564  /// multiset against the names and items read from it, which the emit writes from its tokens (an ink the reading parts
565  /// loses nothing, aa's two addresses in one `\email`, 2306.01880; "Ann" isn't held by "Annette", review revZ4ar2), an
566  /// address run it prints that the reading writes as an e-mail item first the address's, by position (its words none of a
567  /// name's: `POC: K. Hobbs (kerianne.hobbs@us.af.mil)`, 2110.03506; `Germany (martin.waltz@tu-dresden.de)`, 2203.10777;
568  /// review revN1); a call's against every word written (a call printing an entry another prints too is read once:
569  /// revtex's `\affiliation{\Argonne}` listed before the authors, to number the institutes, and again under one,
570  /// 2103.04797), an ink [`Self::exempt`] none.
571  pub(in crate::base_utilities::frontmatter::observer) fn lost_inks(
572    &self,
573    index: usize,
574    events: &[Event],
575    a_block: bool,
576  ) -> Vec<LostInk> {
577    let marks: Vec<(u16, u16)> = events
578      .iter()
579      .filter(|m| m.kind == EventKind::Mark)
580      .filter_map(|m| m.formula.and_then(|f| f.span))
581      .collect();
582    // (each ink's line: a run between breaks, or a container's (a box, an environment, a cell: a column of names, one author
583    // each, 1805.08288's minipages, 2002.00084))
584    // (and its column: a container's events, its lines with them)
585    let (mut unit, mut column) = (0usize, 0usize);
586    let (units, columns): (Vec<usize>, Vec<usize>) = events
587      .iter()
588      .enumerate()
589      .map(|(k, e)| {
590        let container =
591          e.kind == EventKind::Layout && events.get(k + 1).is_some_and(|next| next.nest > e.nest);
592        unit += usize::from(e.kind == EventKind::Break || container);
593        column += usize::from(container);
594        (unit, column)
595      })
596      .unzip();
597    let column_of: BTreeMap<usize, usize> =
598      units.iter().copied().zip(columns.iter().copied()).collect();
599    // (and its line of the block's names, a run between breaks: a unit is one line's)
600    let mut line = 0usize;
601    let line_of: BTreeMap<usize, usize> = events
602      .iter()
603      .zip(&units)
604      .map(|(e, unit)| {
605        line += usize::from(e.kind == EventKind::Break);
606        (*unit, line)
607      })
608      .collect();
609    let inks: Vec<(&Event, usize)> = events
610      .iter()
611      .zip(units.iter().copied())
612      .filter(|(e, _)| {
613        matches!(
614          e.kind,
615          EventKind::Name | EventKind::Contact | EventKind::Note
616        ) && !self.exempt(index, e, &marks)
617      })
618      .collect();
619    let mut counts = self.counts.get(&index).cloned().unwrap_or_default();
620    let take = |counts: &mut BTreeMap<String, usize>, word: &str| match counts.get_mut(word) {
621      Some(n) if *n > 0 => {
622        *n -= 1;
623        true
624      },
625      _ => false,
626    };
627    // (each ink's address runs first: one the reading writes, every address of it an e-mail item of the block, is cut out
628    // of it, its words the address's; a word of it no item writes, its label's say, lost where it prints)
629    let mut written = if a_block {
630      self.addresses.get(&index).cloned().unwrap_or_default()
631    } else {
632      BTreeMap::new()
633    };
634    struct Read<'a> {
635      cut:       Vec<(usize, usize)>,
636      addresses: Vec<String>,
637      lost:      Vec<((usize, usize), &'a str)>,
638    }
639    let mut reads: Vec<Read> = Vec::with_capacity(inks.len());
640    for (e, _) in &inks {
641      let mut read = Read {
642        cut:       Vec::new(),
643        addresses: Vec::new(),
644        lost:      Vec::new(),
645      };
646      if e.text.contains('@') {
647        for run in address_runs(&e.text, &|_| false) {
648          let mut wanted: BTreeMap<String, usize> = BTreeMap::new();
649          for address in &run.addresses {
650            *wanted.entry(address.to_lowercase()).or_default() += 1;
651          }
652          if a_block
653            && wanted
654              .iter()
655              .all(|(address, n)| written.get(address).is_some_and(|have| have >= n))
656          {
657            for (address, n) in wanted {
658              written.entry(address).and_modify(|have| *have -= n);
659            }
660            // (its addresses, from after its label word: a label is the contact's name, written or not, `email:` before
661            // `Duncan.Lorimer@mail.wvu.edu`, 0811.0762)
662            let start = address_start(&e.text, &run);
663            for ((from, to), word) in words_at(&e.text[start..run.end]) {
664              if !take(&mut counts, &word.to_lowercase()) {
665                read.lost.push(((start + from, start + to), word));
666              }
667            }
668            read.cut.push((start, run.end));
669          }
670        }
671        // (the addresses it prints, a run or not: `ann@a.org (corresponding)`; or the ink one word holding an `@`, its domain
672        // no address's: `yiliu.li@outlook`, 2605.16481)
673        read.addresses = addresses_in(&e.text);
674        let word = e.text.trim();
675        if read.addresses.is_empty() && !word.contains(char::is_whitespace) {
676          read.addresses.push(word.to_string());
677        }
678      }
679      reads.push(read);
680    }
681    let outside = |k: usize| {
682      let (e, _) = inks[k];
683      let cut = &reads[k].cut;
684      words_at(&e.text)
685        .filter(move |((from, to), _)| !cut.iter().any(|&(s, t)| s <= *from && *to <= t))
686    };
687    // (a block's ink every word of which is written takes its words first: a name the reading holds, before an address
688    // spelling it printed above it, `\texttt{ann.able@outlook}\\ Ann Able`; an ink of a line an item other than an address
689    // is read from (its source the line's first event) before any other, a line the reading lost taking no word of the
690    // items it repeats: `Dresden, Saxony 01062, Germany (…)` under `Technische Universität Dresden,`, 2203.10777)
691    let read_units: BTreeSet<usize> = events
692      .iter()
693      .zip(&units)
694      .filter(|(e, _)| {
695        e.span
696          .is_some_and(|span| self.contacts.contains(&(index, span)))
697      })
698      .map(|(_, unit)| *unit)
699      .collect();
700    let read_from = |k: usize| read_units.contains(&inks[k].1);
701    let mut held = vec![false; inks.len()];
702    if a_block {
703      let order = (0..inks.len())
704        .filter(|&k| read_from(k))
705        .chain((0..inks.len()).filter(|&k| !read_from(k)));
706      for k in order {
707        let mut words: BTreeMap<String, usize> = BTreeMap::new();
708        for (_, word) in outside(k) {
709          *words.entry(word.to_lowercase()).or_default() += 1;
710        }
711        if reads[k].lost.is_empty()
712          && words
713            .iter()
714            .all(|(word, n)| counts.get(word).is_some_and(|have| have >= n))
715        {
716          for (word, n) in words {
717            counts.entry(word).and_modify(|have| *have -= n);
718          }
719          held[k] = true;
720        }
721      }
722    }
723    let names = self.names.get(&index).map_or(&[][..], Vec::as_slice);
724    let spelled_by: Vec<&str> = names.iter().map(|(_, name)| name.as_str()).collect();
725    // (whom an address spells by the reading's rule, among the written names and `own`, a lost name the address may be its
726    // own: `philips@` Andy Philips's beside Philip Laird, 1908.02923; review revN1r2)
727    let holder = |address: &str, own: Option<&str>| -> Option<Holder> {
728      let mut candidates = spelled_by.clone();
729      candidates.extend(own);
730      spelled_name(&candidates, address).map(|k| match names.get(k) {
731        Some(&(creator, _)) => Holder::Author(creator),
732        None => Holder::Own,
733      })
734    };
735    // (the one holder addresses spell, every one of them spelling that one)
736    let one_holder = |addresses: &mut dyn Iterator<Item = &String>, own: Option<&str>| {
737      let spelled: Option<BTreeSet<Holder>> =
738        addresses.map(|address| holder(address, own)).collect();
739      match spelled.unwrap_or_default().into_iter().collect::<Vec<_>>()[..] {
740        [one] => Some(one),
741        _ => None,
742      }
743    };
744    // (the lines its written names print every word of a name the reading reads from it on: its names' lines)
745    let mut printed: BTreeMap<usize, BTreeSet<String>> = BTreeMap::new();
746    for (k, (e, unit)) in inks.iter().enumerate() {
747      // (an address's ink prints no name: `ann.hobbs@a.org` beside `Contact: A. Hobbs,`)
748      if held[k] && e.kind == EventKind::Name && reads[k].addresses.is_empty() {
749        printed
750          .entry(line_of[unit])
751          .or_default()
752          .extend(words_of(&e.text));
753      }
754    }
755    let names_line = |unit: usize| {
756      printed.get(&line_of[&unit]).is_some_and(|words| {
757        names.iter().any(|(_, name)| {
758          let mut name = words_of(name).peekable();
759          name.peek().is_some() && name.all(|word| words.contains(&word))
760        })
761      })
762    };
763    let (mut by_unit, mut by_column): (BTreeMap<usize, Vec<usize>>, BTreeMap<usize, Vec<usize>>) =
764      Default::default();
765    for (k, (_, unit)) in inks.iter().enumerate() {
766      by_unit.entry(*unit).or_default().push(k);
767      by_column.entry(column_of[unit]).or_default().push(k);
768    }
769    let mut lost = Vec::new();
770    for k in (0..inks.len()).filter(|&k| !held[k]) {
771      let (e, unit) = inks[k];
772      let mut at: Vec<((usize, usize), &str)> = reads[k].lost.clone();
773      for (range, word) in outside(k) {
774        let word_lc = word.to_lowercase();
775        let written = if a_block {
776          take(&mut counts, &word_lc)
777        } else {
778          self.anywhere.contains(&word_lc)
779        };
780        if !written {
781          at.push((range, word));
782        }
783      }
784      if at.is_empty() {
785        continue;
786      }
787      at.sort_unstable_by_key(|(range, _)| *range);
788      // (its lost words of letters outside a word printing an address, as the address reader reads one: what may be a name's
789      // — `\texttt{\{firstname\}.\{lastname\}@kit.edu}`'s are an address's, 2601.00680)
790      let address_words: Vec<(usize, usize)> = e
791        .text
792        .split_whitespace()
793        .filter(|w| !addresses_in(w).is_empty() || reads[k].addresses.iter().any(|a| a == w))
794        .map(|w| {
795          let start = w.as_ptr() as usize - e.text.as_ptr() as usize;
796          (start, start + w.len())
797        })
798        .collect();
799      let named: Vec<&str> = at
800        .iter()
801        .filter(|((from, to), w)| {
802          w.chars().any(char::is_alphabetic)
803            && !address_words.iter().any(|&(s, t)| s <= *from && *to <= t)
804        })
805        .map(|(_, w)| *w)
806        .collect();
807      // (a Name's: what its addresses may spell as its own)
808      let own =
809        (a_block && e.kind == EventKind::Name && !named.is_empty()).then(|| named.join(" "));
810      // (its owner by its own addresses, else by the addresses its line prints, read from the line's text whole: the line goes
811      // with its other inks to the author its own address gives, `ext. 5` after `POC: A. Able (ann.able@a.org)`, 2203.10777's
812      // `Dresden`, `Saxony 01062` before `Germany (martin.waltz@tu-dresden.de)`, review revN1r2; a list's local parts printed
813      // apart are one run of addresses spelling its authors, owning none, `{pzou94, ozel, suresh}@gwu.edu`, 2003.14069;
814      // 2102.08981, 1901.01753, 1708.00818, 2307.03645)
815      let spelled = if !a_block {
816        None
817      } else if reads[k].addresses.is_empty() {
818        let line: Vec<&Event> = events
819          .iter()
820          .zip(&units)
821          .filter(|&(_, &u)| u == unit)
822          .map(|(e, _)| e)
823          .collect();
824        let runs = address_runs(&events_text(&line), &|_| false);
825        one_holder(
826          &mut runs.iter().flat_map(|run| run.addresses.iter()),
827          own.as_deref(),
828        )
829      } else {
830        one_holder(&mut reads[k].addresses.iter(), own.as_deref())
831      };
832      let owner = match spelled {
833        Some(Holder::Author(creator)) => Some(creator),
834        _ => None,
835      };
836      // (the addresses its ink prints, else its line's)
837      let addresses: Vec<&String> = if !a_block {
838        Vec::new()
839      } else if reads[k].addresses.is_empty() {
840        let line = by_unit.get(&unit).into_iter().flatten();
841        line.flat_map(|&j| reads[j].addresses.iter()).collect()
842      } else {
843        reads[k].addresses.iter().collect()
844      };
845      // (a name: on a line of its block's names, or one an address its line or its column prints spells as its own, by the
846      // reading's rule of an address, ranked against the written names: a column's name above its address, 1805.08288's
847      // minipages; an address of a written author's spells that author's line, whatever else it prints: `Contact: A. Hobbs,
848      // ann.hobbs@a.org` under Ann Hobbs)
849      let names_fail = own.as_deref().is_some_and(|own| {
850        owner.is_none()
851          && (names_line(unit) || {
852            let column = by_column.get(&column_of[&unit]).into_iter().flatten();
853            let mut all = addresses
854              .iter()
855              .copied()
856              .chain(column.flat_map(|&j| reads[j].addresses.iter()));
857            all.any(|address| holder(address, Some(own)) == Some(Holder::Own))
858          })
859      });
860      lost.push(LostInk {
861        span: e.span,
862        text: e.text.clone(),
863        words: at.iter().map(|(_, w)| (*w).to_string()).collect(),
864        lost_at: at.iter().map(|(range, _)| *range).collect(),
865        cut: reads[k].cut.clone(),
866        owner,
867        names_fail,
868        font: e.font.clone(),
869      });
870    }
871    lost
872  }
873}

883/// One block's word check ([`check_blocks`]): its inks the reading writes nowhere, and the words it writes more times than
884/// they print.
885#[derive(Debug)]
886pub(in crate::base_utilities::frontmatter::observer) struct BlockCheck {
887  pub lost:     Vec<LostInk>,
888  pub excess:   Vec<(String, bool)>,
889  /// how many of its pictures no written item holds ([`Written::unwritten_pictures`])
890  pub pictures: usize,

893/// The author blocks of the ledger whose authors an output holds, with their ledger index: what a later call replaced or
894/// cleared, as the kernel's `\lx@clear@frontmatter` does, prints in no output (a template's `\author{First Author}` before
895/// the paper's own, 2412.15554).
896pub(in crate::base_utilities::frontmatter::observer) fn live_blocks(
897  ledger: &Ledger,
898) -> Vec<(usize, &AuthorBlock)> {
899  // (per entry, whether a later block of its epoch replaces the authors: the epochs a replacing block opens, from the end)
900  let mut replacing: BTreeSet<usize> = BTreeSet::new();
901  let mut replaced = vec![false; ledger.entries().len()];
902  for (index, entry) in ledger.entries().iter().enumerate().rev() {
903    if let Entry::Author(block) = entry {
904      replaced[index] = replacing.contains(&block.epoch);
905      if block.replace {
906        replacing.insert(block.epoch);
907      }
908    }
909  }
910  let blocks = ledger.entries().iter().enumerate();
911  let live = blocks.filter_map(|(index, entry)| match entry {
912    Entry::Author(block) if !block.cleared && !replaced[index] => Some((index, &**block)),
913    _ => None,
914  });
915  live.collect()
916}

918/// The block half of the word check, both ways: per author block (its ledger index) that fails it, its [`BlockCheck`],
919/// against `written`, the reading's ([`Written::new`]), over the [`live_blocks`] (2412.15554's template block none).
920pub(in crate::base_utilities::frontmatter::observer) fn check_blocks(
921  ledger: &Ledger,
922  written: &Written,
923) -> BTreeMap<usize, BlockCheck> {
924  let mut checks = BTreeMap::new();
925  for (index, block) in live_blocks(ledger) {
926    if let Outcome::Events { events, .. } = &block.outcome {
927      let pictures = written.unwritten_pictures(index, events);
928      // (a command's ink inside a word its letters, as the reading reads them: `lines::words_through_one_command`)
929      let events = &words_through_one_command(events);
930      let check = BlockCheck {
931        lost: written.lost_inks(index, events, true),
932        excess: written.excess(index, events),
933        pictures,
934      };
935      if !check.lost.is_empty() || !check.excess.is_empty() || check.pictures > 0 {
936        checks.insert(index, check);
937      }
938    }
939  }
940  checks
