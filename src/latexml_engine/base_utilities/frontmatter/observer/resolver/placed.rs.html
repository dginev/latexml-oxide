1//! The content invariant, checked on the placed output (FRONTMATTER_SINGLE_ENGINE_PLAN.md, the safety net's bar (2): every
2//! word TeX printed in an author block is in the output exactly once, or counted in a Warning; OXIDIZED_DESIGN_DIVERGENCES
3//! #980). The reading's word check (`words`) sees what the reading writes before the emit; what goes wrong after it — a
4//! shared entry's address the Kind step keeps for every holder, digestion's junk in a name (2506.02663), a call written
5//! whole beside its piece (2607.00684's `mtablo@postech.ac.kr` twice), a binding's lost entry text — shows only in the
6//! frontmatter `insert_frontmatter` placed. So one pass after it (Perl's `insertFrontMatter` and `relocateAnnotations`,
7//! Base_Utility.pool.ltxml:736-740, :891-900): the words the ledger's author blocks and
8//! creator calls printed (their rehearsed events), against the words the placed creators hold (`emitter::dom`'s reading of
9//! the document). It counts, it never answers: placement is over.
10//!
11//! What counts, both sides by one definition ([`ink_words`]): every word, a mark excluded by its event, not by its
12//! spelling (an inked letter, an initial or a two-digit number counts; a word a Mark event prints is that mark's, as
13//! often as the events print it, wherever a text holds it, neither lost nor in excess); an address is a unit of its
14//! own, its words none of the text's (1605.03019's `IDSIA` beside `@idsia.ch`). *Lost*: a word or an address the inks
15//! print more often than the placed frontmatter holds it — a shared item (one printed `\affiliation`, `annotate=all`)
16//! as often as the inks print its text, at least the most copies one creator holds and at most its copies in all, as
17//! the excess counts it (a line printed under each author and kept as one shared item is no loss), a name every copy,
18//! so a shared entry's copies hide no word lost on a line of other words (review revN2: `IDSIA` printed twice, one copy
19//! lost, the shared affiliation printed once and held twice; a lost line printing the shared item's own words is still
20//! covered, review revN2r3's `q1e`, SYNC leftovers 13f) —, a publication note, a contact's label (its `name`, with its item,
21//! never its text's where the text opens with it: a class's `Email: `, which the rehearsal's `\email` doesn't print,
22//! 2307.11138) and a link's target holding it too. *Excess*: a word placed more often than printed — a shared
23//! item counted the same way; an e-mail contact every time, an address being one person's (one address on two creators,
24//! or twice on one). What may print beside the authors without being theirs only answers an excess: a separator's word
25//! (`and`), a mark's text, a link's target (an ORCID's id behind a logo), a footnote's text (`\footnotetext`), a publication
26//! note's, a call a block's own expansion made whose words its block's events print (a call's words, else, are required:
27//! its rehearsal's, or its content's text where none read it — [`content_check`]), and the words of the creator a block
28//! written whole became (TeX's own digestion of it, its addresses still counted). An excess needs every live block and
29//! creator call read: a block no rehearsal read prints words no event holds.

60/// The placed frontmatter as the check reads it: the creators' items, and the texts that hold a printed word apart from
61/// them (a publication note, a link's target, each target once); and the words the reading drops by its own rule
62/// (`Resolved::drops`, a block's Name event by its ledger index, span and word), each a separator's.
63#[derive(Default)]
64pub(in crate::base_utilities::frontmatter::observer) struct Placed {
65  pub items:   Vec<PlacedItem>,
66  pub beside:  Vec<String>,
67  pub dropped: BTreeSet<RuleDrop>,
68  /// the document's publication notes' texts, each as written (in `beside` too): a note written more often than it
69  /// printed is in excess ([`note_excess`])
70  pub notes:   Vec<String>,
71  /// the words of the marks that led a second entry of their label in one list, as the writer cut it ([`retold_words`]):
72  /// printed, and no mark's or name's copy holds them
73  pub retold:  Vec<String>,
74}

76/// The words of a mark the writer read as the label of an entry where it is glued to the text of another entry of that label
77/// in its list ([`MarkLedEntry`], of the contacts placed by their calls' labels): that entry's text, as one list labels each
78/// entry once (`emitter::calls::split_marked_pieces`) — 2204.02412's `(CAP$^3$), Univ Z` cut into `Center for Astro (CAP`
79/// and `, Univ Z`, both labelled 3 ("glued": no separator word between the text and the mark; a space qualifies). A label
80/// leading a line, or glued to a name, is no text.
81pub(in crate::base_utilities::frontmatter::observer) fn retold_words(
82  entries: &[MarkLedEntry],
83  placed: &BTreeSet<String>,
84) -> Vec<String> {
85  let placed: Vec<&MarkLedEntry> = entries
86    .iter()
87    .filter(|e| placed.contains(&e.item))
88    .collect();
89  let mut retold = Vec::new();
90  let mut told = BTreeSet::new();
91  for entry in &placed {
92    let Some((from, to)) = entry.glued else {
93      continue;
94    };
95    let text_of_another = placed.iter().any(|other| {
96      other.list == entry.list
97        && other.label == entry.label
98        && other.mark != entry.mark
99        && other.entry.0 <= from
100        && to <= other.entry.1
101    });
102    if text_of_another && told.insert((entry.list, entry.mark)) {
103      retold.extend(entry.words.iter().cloned());
104    }
105  }
106  retold
107}

116/// A multiset of the words and addresses texts print, each word with its first spelling.
117#[derive(Default, Clone)]
118struct Tally {
119  words:     BTreeMap<String, (usize, String)>,
120  addresses: BTreeMap<String, usize>,
121}
122
123impl Tally {
124  fn add(&mut self, text: &str, times: usize) { self.add_parts(text, times, times); }
125
126  /// `text`'s words `words` times, its addresses `addresses` times.
127  fn add_parts(&mut self, text: &str, words: usize, addresses: usize) {
128    let (printed, printed_addresses) = ink_words(text);
129    for word in printed {
130      let entry = self
131        .words
132        .entry(word.to_lowercase())
133        .or_insert_with(|| (0, word.clone()));
134      entry.0 += words;
135    }
136    for address in printed_addresses {
137      *self.addresses.entry(address).or_default() += addresses;
138    }
139  }
140
141  /// The words of `address`'s local part, `times` more.
142  fn add_local(&mut self, address: &str, times: usize) {
143    if times == 0 {
144      return;
145    }
146    for word in local_words(address) {
147      let entry = self.words.entry(word.clone()).or_insert((0, word));
148      entry.0 += times;
149    }
150  }
151
152  /// `word`, lowercased already, `times` more.
153  fn add_word(&mut self, word: &str, times: usize) {
154    let entry = self
155      .words
156      .entry(word.to_string())
157      .or_insert_with(|| (0, word.to_string()));
158    entry.0 += times;
159  }
160
161  fn word(&self, word: &str) -> usize { self.words.get(word).map_or(0, |(n, _)| *n) }
162
163  fn address(&self, address: &str) -> usize { self.addresses.get(address).copied().unwrap_or(0) }
164}

166/// The texts `events` print as inks: each line's Names read whole, as the reading reads a line's addresses (`Switzerland
167/// {adam`, `samuli`, `monaldo}@idsia.ch` one list, 1605.03019), a Contact or a Note on its own (a url beside an address,
168/// `https://members.loria.fr/AClement alexandre.clement@loria.fr`, is no list, 2206.10577); a contact a mark's formula
169/// holds beside them (an ORCID's logo in `$^{1\,\orcidicon{…}}$`, 2210.03748; the names a formula prints with their marks
170/// are names, `$\text{Li Cheng}^{1}, …$`, 2001.05492, 2303.12302). Into `beside`, what prints beside them: a separator's
171/// word, a mark's text (into `marks` too, its words), a contact's label (`Event::name`), a link's target where its anchor prints no address of its own
172/// (`Written::excess`'s). And the same texts with their separators' words, as an item's text holds them (`Japan Science and
173/// Technology Corporation`, cond-mat0305063), each a print a shared item counts by ([`Prints`]).
174fn ink_texts(
175  events: &[Event],
176  beside: &mut Tally,
177  marks: &mut BTreeMap<String, usize>,
178) -> (Vec<String>, Vec<String>) {
179  let formulas: Vec<(u16, u16)> = events
180    .iter()
181    .filter(|m| m.kind == EventKind::Mark)
182    .filter_map(|m| m.formula.and_then(|f| f.span))
183    .collect();
184  let in_a_mark = |e: &Event| {
185    e.span.is_some_and(|(s, t)| {
186      formulas
187        .iter()
188        .any(|&(from, to)| from <= s && t <= to && (s, t) != (from, to))
189    })
190  };
191  let (mut texts, mut prints) = (Vec::new(), Vec::new());
192  // (a line's Names, and with its separators' words)
193  let (mut line, mut printed): (Vec<&Event>, Vec<&Event>) = (Vec::new(), Vec::new());
194  fn end_line<'a>(
195    (texts, prints): (&mut Vec<String>, &mut Vec<String>),
196    line: &mut Vec<&'a Event>,
197    printed: &mut Vec<&'a Event>,
198  ) {
199    texts.push(events_text(line));
200    prints.push(events_text(printed));
201    line.clear();
202    printed.clear();
203  }
204  for e in events {
205    match &e.kind {
206      EventKind::Name | EventKind::Sep(SepKind::Comma) => {
207        line.push(e);
208        printed.push(e);
209      },
210      EventKind::Contact if in_a_mark(e) => beside.add(&e.text, 1),
211      EventKind::Contact | EventKind::Note if e.kind == EventKind::Contact || !e.inkless => {
212        texts.push(e.text.clone());
213        prints.push(e.text.clone());
214      },
215      EventKind::Sep(SepKind::Word(word)) => {
216        beside.add(word, 1);
217        printed.push(e);
218      },
219      // (a line break inside a printed group of local parts parts no list: `\{yogarshi, …,\\ vittorca, drot, ballemig\}@amazon.com`,
220      // 2305.13191, `Event::braces`; and a column ends one, `\and`'s or an alignment tab's: the next author's lines — `Ann
221      // Able\\ IDSIA Lab, Lugano \and Bob Baker\\ IDSIA Lab, Lugano` prints that line twice, each its own, `Prints`)
222      EventKind::Break if e.braces == 0 => {
223        end_line((&mut texts, &mut prints), &mut line, &mut printed)
224      },
225      EventKind::Sep(SepKind::And(_) | SepKind::Column(_)) => {
226        end_line((&mut texts, &mut prints), &mut line, &mut printed)
227      },
228      _ => {},
229    }
230    for inner in std::iter::once(e).chain(&e.inner) {
231      // (a contact's label, printed with it, beside its address as a call's is (`call_label`): authblk's `Corresponding
232      // author e-mail:` before the address of an affiliation's `\thanks`, cond-mat0202268)
233      if inner.kind == EventKind::Contact
234        && let Some(name) = inner.name.as_deref()
235      {
236        beside.add(name, 1);
237      }
238      if inner.kind == EventKind::Contact
239        && let Some(target) = inner.target.as_deref()
240        && addresses_in(&inner.text).is_empty()
241      {
242        beside.add(target, 1);
243      }
244      // (a mark by its event, whatever its spelling: an ordinal's suffix is the line's text, `events_text`)
245      if inner.kind == EventKind::Mark && !inner.ordinal {
246        beside.add(&inner.text, 1);
247        for word in words_of(&inner.text) {
248          *marks.entry(word).or_default() += 1;
249        }
250      }
251    }
252  }
253  end_line((&mut texts, &mut prints), &mut line, &mut printed);
254  (texts, prints)
255}

263/// The content check of `placed` against what the ledger's live blocks ([`live_blocks`]) and creator calls printed. A call
264/// printing what another call prints too counts once (revtex's `\affiliation{\Argonne}` listed before the authors to number
265/// the institutes and again under one, 2103.04797); the pieces a binding cut one printed list into (its calls' shared
266/// `_cutlist`, [`CUT_LIST`]) are read as one text, the list's addresses with them (`{zeba.khanam, sehsan,
267/// kdm}@essex.ac.uk`, 1808.00588).
268pub(in crate::base_utilities::frontmatter::observer) fn content_check(
269  ledger: &Ledger,
270  placed: &Placed,
271  kept_whole: &dyn Fn(usize) -> bool,
272) -> ContentCheck {
273  let (mut printed, mut beside) = (Tally::default(), Tally::default());
274  let mut marks: BTreeMap<String, usize> = BTreeMap::new();
275  let mut all_read = true;
276  // (the words a content's text read past an accent or a dotless letter in, with the places: `ContentText::read_past`)
277  let mut past: BTreeMap<String, Vec<(usize, bool)>> = BTreeMap::new();
278  // (the Name words of each block written whole: TeX's own digestion of it, printing what its separators print — a class's
279  // `\and` printing "and", which the rehearsal binds to an event printing nothing: JAIR's, 1908.02923 —, words no event
280  // holds; the creator it became is found by them below)
281  let mut whole: Vec<BTreeMap<String, usize>> = Vec::new();
282  // (the texts the inks print, each block's and call's in order, which a shared item's copies count by: `Prints`)
283  let mut prints = Prints::default();
284  // (each live block's printed words, which its own expansion's calls take theirs from: `own_print`)
285  let mut block_print: BTreeMap<usize, Tally> = BTreeMap::new();
286  let live = live_blocks(ledger);
287  let live_index: BTreeSet<usize> = live.iter().map(|(index, _)| *index).collect();
288  for (index, block) in live {
289    let print = block_print.entry(index).or_default();
290    match &block.outcome {
291      Outcome::Events { events, unread, .. } => {
292        // (a Name the reading drops by its own rule is the separator it reads it as, its word beside the names: a byline's
293        // "by", `By~~{\scshape Bryna Kra}`, 2603.27258; 2603.25922, 2604.07186, 2604.12864)
294        let separated: Vec<Event>;
295        // (an earlier list setter call of the block the written one replaces prints nothing: `Ledger::replaced`)
296        let unreplaced: Vec<Event>;
297        let events = if ledger
298          .replaced
299          .iter()
300          .any(|(at, run)| *at == index && run.is_some())
301        {
302          let within = |span: (u16, u16)| {
303            ledger
304              .replaced
305              .iter()
306              .any(|&(at, run)| at == index && run.is_some_and(|(s, t)| s <= span.0 && span.1 <= t))
307          };
308          unreplaced = events
309            .iter()
310            .filter(|e| !e.span.is_some_and(within))
311            .cloned()
312            .collect();
313          &unreplaced
314        } else {
315          events
316        };
317        let events = if placed.dropped.iter().any(|(at, ..)| *at == index) {
318          separated = events
319            .iter()
320            .map(|e| match e.span {
321              Some(_)
322                if e.kind == EventKind::Name
323                  && placed
324                    .dropped
325                    .contains(&(index, e.span, e.text.trim().to_lowercase())) =>
326              {
327                let mut sep = Event::new(EventKind::Sep(SepKind::Word(e.text.trim().to_string())));
328                sep.span = e.span;
329                sep
330              },
331              _ => e.clone(),
332            })
333            .collect();
334          &separated
335        } else {
336          events
337        };
338        // (a line of link pieces the reading joins into its addresses prints them, not the pieces' words:
339        // `joined_address_pieces`, 1710.01968)
340        let joined = joined_address_pieces(events);
341        let (texts, units) = ink_texts(&joined, &mut beside, &mut marks);
342        for text in texts {
343          printed.add(&text, 1);
344          print.add(&text, 1);
345        }
346        // (a rehearsal that left a group open — an `\end{tabular}\begin{tabular}{c}` juggle in the line, 1401.6981 — read
347        // what follows into no box, so no event says it: the events are not trusted for the block, its content's text
348        // (`Outcome::Events::unread`, `ledger::content_text`, as for a call no rehearsal read) counting the words they
349        // lack, a possible loss)
350        if let Some(text) = unread {
351          read_past(&mut past, text);
352          let mut content = Tally::default();
353          content.add(text, 1);
354          // (a content word the block prints as spelled; then a word it read past an accent or a dotless letter in, as the
355          // word the block prints with them — `muller` for `müller`, `fsica` for `física` ([`reads_as`]) —, each print taken
356          // once: a word lost beside a printed one it differs from by a letter is lost, and named, `Lan` beside `Lian`)
357          let mut taken: BTreeMap<String, usize> = BTreeMap::new();
358          let mut held: BTreeMap<String, usize> = BTreeMap::new();
359          for (word, (n, _)) in &content.words {
360            let take = (*n).min(print.word(word));
361            taken.insert(word.clone(), take);
362            held.insert(word.clone(), take);
363          }
364          for (word, (n, _)) in &content.words {
365            let Some(places) = past.get(word) else {
366              continue;
367            };
368            for (other, (m, _)) in &print.words {
369              let have = held.get(word).copied().unwrap_or(0);
370              if have == *n {
371                break;
372              }
373              if other == word || !reads_as(word, places, other) {
374                continue;
375              }
376              let take = m
377                .saturating_sub(taken.get(other).copied().unwrap_or(0))
378                .min(n - have);
379              *taken.entry(other.clone()).or_default() += take;
380              *held.entry(word.clone()).or_default() += take;
381            }
382          }
383          // (then a word the block prints glued that is two content words joined — `ledger::content_text` parts a word at
384          // every multi-letter control sequence, `Able\begingroup\bfseries Bob` read `able bob`, printed `ablebob` —: both
385          // parts held, the document-level rule ([`glued`]) applied before a word is counted lost, as `CanadaPerimeter`'s)
386          let mut have = Tally::default();
387          held.iter().for_each(|(word, n)| have.add_word(word, *n));
388          for (other, (m, _)) in &print.words {
389            let gap = m.saturating_sub(taken.get(other).copied().unwrap_or(0));
390            if let Some((n, parts)) = glued(other, gap, &content, &have, &marks) {
391              *taken.entry(other.clone()).or_default() += n;
392              for part in parts {
393                have.add_word(&part, n);
394                *held.entry(part).or_default() += n;
395              }
396            }
397          }
398          for (word, (n, spelled)) in content.words {
399            let more = n.saturating_sub(held.get(&word).copied().unwrap_or(0));
400            for tally in [&mut printed, &mut *print] {
401              tally
402                .words
403                .entry(word.clone())
404                .or_insert((0, spelled.clone()))
405                .0 += more;
406            }
407          }
408          for (address, n) in content.addresses {
409            let more = n.saturating_sub(print.address(&address));
410            for tally in [&mut printed, &mut *print] {
411              *tally.addresses.entry(address.clone()).or_default() += more;
412            }
413          }
414        }
415        units.iter().for_each(|text| prints.add(index, text));
416        if kept_whole(index) {
417          let mut words: BTreeMap<String, usize> = BTreeMap::new();
418          let names = events.iter().filter(|e| e.kind == EventKind::Name);
419          for word in names.flat_map(|e| words_of(&e.text)) {
420            *words.entry(word).or_default() += 1;
421          }
422          whole.push(words);
423        }
424      },
425      Outcome::Skipped(SkipReason::Empty) => {},
426      _ => all_read = false,
427    }
428  }
429  let mut calls: BTreeSet<Vec<String>> = BTreeSet::new();
430  // (the pieces of a cut list so far, and its key)
431  let mut list: (Option<&str>, Vec<String>) = (None, Vec::new());
432  let aside = |events: &[Event], beside: &mut Tally, marks: &mut BTreeMap<String, usize>| {
433    for text in ink_texts(events, beside, marks).0 {
434      beside.add(&text, 1);
435    }
436  };
437  // (what TeX printed of a creator call, one rule for them all: its words count unless its block's events print them — a call
438  // outside its block's expansion none of them. Its words are its rehearsal's, or its content's text where the rehearsal gave
439  // up on it (`ApiCall::unread`; a queued call no call recorded, `Ledger::unrecorded`: texts the ledger reads, the check
440  // reads no token); an inside call's, rehearsed for this check alone (`ApiCall::inside_events`), are its block's where
441  // that block's events print every one of them, a block no longer live printing none that TeX printed. The block's own
442  // creator entry is the block, and so is an inside call no flush could rehearse (a journal unavailable to the end). An inside call the ledger
443  // misread — an input mark refilled to an equal token, IEEEtran's affiliations read as the author block's own, 2401.11731,
444  // 1801.03314 — is no longer exempt on trust)
445  let own_print = |call: &ApiCall, block_print: &mut BTreeMap<usize, Tally>| -> bool {
446    let Some(block) = call.after_block.filter(|_| call.inside) else {
447      return false;
448    };
449    if !live_index.contains(&block) {
450      return true;
451    }
452    let mut words = Tally::default();
453    if let Some(text) = &call.unread {
454      words.add(text, 1);
455    } else {
456      let (mut beside, mut marks) = (Tally::default(), BTreeMap::new());
457      for text in ink_texts(&call.inside_events, &mut beside, &mut marks).0 {
458        words.add(&text, 1);
459      }
460    }
461    let print = block_print.entry(block).or_default();
462    let held = words
463      .words
464      .iter()
465      .all(|(word, (n, _))| print.word(word) >= *n)
466      && words
467        .addresses
468        .iter()
469        .all(|(address, n)| print.address(address) >= *n);
470    if held {
471      for (word, (n, _)) in &words.words {
472        if let Some(entry) = print.words.get_mut(word) {
473          entry.0 -= n;
474        }
475      }
476      for (address, n) in &words.addresses {
477        if let Some(entry) = print.addresses.get_mut(address) {
478          *entry -= n;
479        }
480      }
481    }
482    held
483  };
484  for (index, entry) in ledger.entries().iter().enumerate() {
485    let piece = match entry {
486      Entry::Api(call)
487        if !call.cleared
488          && !ledger.replaced.contains(&(call.index, None))
489          && creator_call(&call.tag, &call.parent) =>
490      {
491        if let Some(label) = call_label(&call.attrs) {
492          beside.add(label, 1);
493        }
494        let events: &[Event] = if call.inside {
495          &call.inside_events
496        } else {
497          &call.events
498        };
499        if call.inside && call.parent.is_empty() {
500          // (the block's own creator entry: the block's events are its print)
501          None
502        } else if own_print(call, &mut block_print) {
503          aside(events, &mut beside, &mut marks);
504          None
505        } else if let Some(text) = &call.unread {
506          read_past(&mut past, text);
507          calls
508            .insert(words_of(text).collect())
509            .then(|| (None, (vec![text.to_string()], vec![text.to_string()])))
510        } else if !call.rehearsed {
511          // (the writer stays out for it, `writer::writer_stays_out` (deleted): the queue is digested as Perl's)
512          all_read = false;
513          None
514        } else if calls.insert(words_of(&events::printed_text(events)).collect()) {
515          let cut = call
516            .attrs
517            .split(',')
518            .find_map(|key| key.strip_prefix(CUT_LIST)?.strip_prefix('='));
519          Some((cut, ink_texts(events, &mut beside, &mut marks)))
520        } else {
521          None
522        }
523      },
524      // (a document note's text is its own, beside; the contacts the reading takes as the creators' are placed or counted —
525      // a late author's `\address` in amsart's `\thanks`, 0807.4720 — `api::note_owned_contacts`)
526      Entry::Api(call) if !call.cleared && call.tag == "ltx:pubnote" => {
527        let owned = note_owned_contacts(&call.events);
528        let rest: Vec<Event> = call
529          .events
530          .iter()
531          .filter(|e| !owned.iter().any(|o| std::ptr::eq(*o, *e)))
532          .cloned()
533          .collect();
534        aside(&rest, &mut beside, &mut marks);
535        let texts: Vec<String> = owned
536          .iter()
537          .map(|e| e.text.clone())
538          .filter(|text| !text.trim().is_empty())
539          .collect();
540        (!texts.is_empty()).then(|| (None, (texts.clone(), texts)))
541      },
542      Entry::NoteText(text) => {
543        aside(&text.events, &mut beside, &mut marks);
544        None
545      },
546      _ => None,
547    };
548    let Some((cut, (texts, units))) = piece else {
549      continue;
550    };
551    units.iter().for_each(|text| prints.add(index, text));
552    if cut.is_none() || cut != list.0 {
553      printed.add(&list.1.join(", "), 1);
554      list = (cut, Vec::new());
555    }
556    list.1.extend(texts);
557    if cut.is_none() {
558      printed.add(&list.1.join(", "), 1);
559      list = (None, Vec::new());
560    }
561  }
562  printed.add(&list.1.join(", "), 1);
563  // (the queued creator calls the emitter took that no call recorded: their contents' texts)
564  for (n, text) in ledger.unrecorded.iter().enumerate() {
565    if calls.insert(words_of(text).collect()) {
566      read_past(&mut past, text);
567      prints.add(ledger.entries().len() + n, text);
568      printed.add(text, 1);
569    }
570  }
571  // (a mark the writer cut at as a label where it is another entry's text, `retold_words`: printed as text, so neither the
572  // marks nor a name's copy holds it)
573  let mut retold = Tally::default();
574  for word in &placed.retold {
575    printed.add(word, 1);
576    retold.add(word, 1);
577  }
578  // (what the placed frontmatter holds, both ways alike: a name every copy, and an e-mail's addresses; a shared item once,
579  // as often as one creator holds it — its label with it)
580  let (mut names, mut emails): (BTreeMap<&str, usize>, BTreeMap<&str, usize>) = Default::default();
581  let mut per_holder: BTreeMap<(&str, usize), (usize, &PlacedItem)> = BTreeMap::new();
582  // (each creator's name words: which one a block written whole became)
583  let mut named: BTreeMap<usize, BTreeMap<String, usize>> = BTreeMap::new();
584  for item in &placed.items {
585    let key = match &item.kind {
586      PlacedKind::Name => {
587        *names.entry(&item.text).or_default() += 1;
588        let words = named.entry(item.holder).or_default();
589        for word in words_of(&item.text) {
590          *words.entry(word).or_default() += 1;
591        }
592        continue;
593      },
594      PlacedKind::Email(key) => {
595        *emails.entry(&item.text).or_default() += 1;
596        key
597      },
598      PlacedKind::Item(key) => key,
599    };
600    per_holder
601      .entry((key.as_str(), item.holder))
602      .or_insert((0, item))
603      .0 += 1;
604  }
605  let (mut shared, mut named_words) = (Tally::default(), Tally::default());
606  for (text, n) in names {
607    shared.add(text, n);
608    named_words.add(text, n);
609  }
610  for (text, n) in emails {
611    shared.add_parts(text, 0, n);
612  }
613  // (per shared item, the most copies one creator holds and its copies in all)
614  let mut copies: BTreeMap<&str, (usize, usize, &PlacedItem)> = BTreeMap::new();
615  for ((key, _), (n, item)) in per_holder {
616    let at = copies.entry(key).or_insert((0, 0, item));
617    at.0 = at.0.max(n);
618    at.1 += n;
619  }
620  let mut held = Tally::default();
621  // (the longest texts first: an entry printing two lines takes them before an entry printing one of them — revtex's
622  // `\address{Dept, Univ\\ CREST, …}` and another author's `\address{Dept, Univ}`, cond-mat0305063)
623  let mut copies: Vec<(usize, usize, &PlacedItem)> = copies.into_values().collect();
624  copies.sort_by_cached_key(|&(_, _, item)| std::cmp::Reverse(Prints::key(labelled(item).1).len()));
625  for (most, total, item) in copies {
626    let email = matches!(item.kind, PlacedKind::Email(_));
627    let (label, text) = labelled(item);
628    let n = prints.take(text, most, total);
629    shared.add_parts(text, n, if email { 0 } else { n });
630    // (a contact's label, its `name`: the label's words, held with its item, never its text's where the text opens with it — a
631    // class printing `Email: ` the rehearsal's `\email` doesn't, 2307.11138, 1602.07570; a label the text doesn't print answers
632    // no word of it, `Email: ` beside an unprinted `email@domain`, 2607.28418)
633    if let Some(label) = label {
634      held.add(label, n);
635    }
636  }
637  for text in &placed.beside {
638    held.add(text, 1);
639  }
640  // (`held` is `shared` and the texts beside it)
641  for (word, (n, spelled)) in &shared.words {
642    let entry = held
643      .words
644      .entry(word.clone())
645      .or_insert((0, spelled.clone()));
646    entry.0 += n;
647  }
648  for (address, n) in &shared.addresses {
649    *held.addresses.entry(address.clone()).or_default() += n;
650  }
651  // (a block written whole: the words of the creator it became — the one whose name holds the most of its Name words,
652  // more than half — answer an excess, that creator's addresses still counted)
653  for words in &whole {
654    let total: usize = words.values().sum();
655    let overlap = |name: &BTreeMap<String, usize>| -> usize {
656      words
657        .iter()
658        .map(|(word, n)| (*n).min(name.get(word).copied().unwrap_or(0)))
659        .sum()
660    };
661    let best = named
662      .iter()
663      .map(|(holder, name)| (overlap(name), *holder))
664      .max_by_key(|&(n, holder)| (n, std::cmp::Reverse(holder)));
665    if let Some((n, holder)) = best
666      && 2 * n > total
667    {
668      for item in placed.items.iter().filter(|item| item.holder == holder) {
669        beside.add_parts(&item.text, 1, 0);
670      }
671    }
672  }
673  // (an address one side reads where the other reads its local part's words — a list's parts a note keeps, `{\it aable,
674  // bbaker, cat@a.org}`, 2003.14069 —: the same ink, so neither lost nor in excess; a lost one's from the words both sides
675  // read, the placed creators' (`shared`), never a label's, a link's or a note's beside them)
676  let lost: Vec<(String, usize)> = printed
677    .addresses
678    .iter()
679    .map(|(address, &n)| (address.clone(), n.saturating_sub(held.address(address))))
680    .collect();
681  for (address, gap) in lost {
682    let n = same_ink(&address, gap, &shared, &printed);
683    *held.addresses.entry(address.clone()).or_default() += n;
684    printed.add_local(&address, n);
685  }
686  let more: Vec<(String, usize)> = shared
687    .addresses
688    .iter()
689    .map(|(address, &n)| {
690      let gap = n.saturating_sub(printed.address(address) + beside.address(address));
691      (address.clone(), gap)
692    })
693    .collect();
694  for (address, gap) in more {
695    let n = same_ink(&address, gap, &printed, &held);
696    *printed.addresses.entry(address.clone()).or_default() += n;
697    held.add_local(&address, n);
698  }
699  // (a word one side reads glued to the next, the other parted — a line's words the lexer reads across a break, `CanadaPerimeter`
700  // (0803.3421), a name and its ORCID's id (`Kim0000`, 2307.03558); a block written whole parted at its layout's cells —: the
701  // same ink; a lost one's parts from the words both sides read, as an address's)
702  let gaps: Vec<(String, usize)> = printed
703    .words
704    .iter()
705    .map(|(word, (n, _))| (word.clone(), n.saturating_sub(held.word(word))))
706    .collect();
707  for (word, gap) in gaps {
708    if let Some((n, parts)) = glued(&word, gap, &shared, &printed, &marks) {
709      held.add_word(&word, n);
710      parts.iter().for_each(|part| printed.add_word(part, n));
711    }
712  }
713  let gaps: Vec<(String, usize)> = shared
714    .words
715    .iter()
716    .map(|(word, (n, _))| {
717      let gap = n.saturating_sub(printed.word(word) + beside.word(word));
718      (word.clone(), gap)
719    })
720    .collect();
721  for (word, gap) in gaps {
722    if let Some((n, parts)) = glued(&word, gap, &printed, &held, &marks) {
723      printed.add_word(&word, n);
724      parts.iter().for_each(|part| held.add_word(part, n));
725    }
726  }
727  // (a word one side reads without the accents or the dotless letters the other prints — the text of a call no rehearsal
728  // read, which reads none (`ledger::content_text`), against its placed copy, `universitat` and `universität`, `fsica` and
729  // `física` ([`reads_as`]) —: the same ink, as a glued word)
730  let gaps: Vec<(String, usize)> = printed
731    .words
732    .iter()
733    .map(|(word, (n, _))| (word.clone(), n.saturating_sub(held.word(word))))
734    .filter(|(word, gap)| *gap > 0 && past.contains_key(word))
735    .collect();
736  for (word, mut gap) in gaps {
737    let places = &past[&word];
738    let accented: Vec<String> = shared
739      .words
740      .keys()
741      .filter(|other| **other != word && reads_as(&word, places, other))
742      .cloned()
743      .collect();
744    for other in accented {
745      let spare = shared
746        .word(&other)
747        .saturating_sub(printed.word(&other) + beside.word(&other));
748      let n = spare.min(gap);
749      if n > 0 {
750        held.add_word(&word, n);
751        printed.add_word(&other, n);
752        gap -= n;
753      }
754    }
755  }
756  // (a word a Mark event prints is that mark's, as often as the events print it, wherever a text holds it: an entry's
757  // label in its contact's text, IEEEtran's nested `\IEEEauthorblockA{\IEEEauthorrefmark{2}Univ B, City}`, 1812.05961)
758  let mut check = ContentCheck::default();
759  for (word, (n, spelled)) in &printed.words {
760    let (mark, have) = match retold.word(word) {
761      0 => (marks.get(word).copied().unwrap_or(0), held.word(word)),
762      told => (
763        marks.get(word).map_or(0, |m| (*m).min(n - told)),
764        held.word(word).saturating_sub(named_words.word(word)),
765      ),
766    };
767    if *n > have + mark {
768      check.lost.push(spelled.clone());
769    }
770  }
771  for (address, n) in &printed.addresses {
772    if *n > held.address(address) {
773      check.lost.push(address.clone());
774    }
775  }
776  if all_read {
777    for (word, (n, spelled)) in &shared.words {
778      if *n > printed.word(word) + beside.word(word) {
779        check.excess.push(spelled.clone());
780      }
781    }
782    for (address, n) in &shared.addresses {
783      if *n > printed.address(address) + beside.address(address) {
784        check.excess.push(address.clone());
785      }
786    }
787  }
788  check
789    .excess
790    .extend(note_excess(&placed.notes, &note_prints(ledger)));
791  check
792}

829/// The publication notes `written` holds more often than the notes `printed` print them (each at least once: one whose
830/// print no event read is its own), each extra copy's text: a `\thanks` digested twice (review rck2). Compared by their
831/// letters' words, in any order (a mark's digits and symbols none of them: `$^{1}$Univ A` prints `1Univ A`, written
832/// `<sup>1</sup>Univ A`, the note's text without its marks); a note of no letter is counted by no one.
833fn note_excess(written: &[String], printed: &[String]) -> Vec<String> {
834  fn key(text: &str) -> Vec<String> {
835    let mut words: Vec<String> = words_of(text)
836      .map(|word| {
837        word
838          .chars()
839          .filter(|c| c.is_alphabetic())
840          .collect::<String>()
841      })
842      .filter(|word| !word.is_empty())
843      .collect();
844    words.sort();
845    words
846  }
847  let mut copies: BTreeMap<Vec<String>, (usize, &str)> = BTreeMap::new();
848  for text in written {
849    copies.entry(key(text)).or_insert((0, text)).0 += 1;
850  }
851  let mut excess = Vec::new();
852  for (key_of, (n, text)) in copies {
853    if key_of.is_empty() {
854      continue;
855    }
856    let prints = printed.iter().filter(|p| key(p) == key_of).count();
857    for _ in prints.max(1)..n {
858      excess.push(text.trim().to_string());
859    }
860  }
861  excess
862}

871/// Whether `word`, a content text's read past an accent or a one-letter control word at `places`
872/// ([`ContentText::read_past`]: `F\'{\i}sica` read `fsica`, a dotless letter after its `f`), reads the word `other` an
873/// ink prints (`física`): `other`'s base letters ([`base_letters`]) are `word`'s, with at most one `i` or `j` where a
874/// one-letter control word was, the letter a dotless `\i`/`\j` printed — no other letter, and none elsewhere (`lu` is
875/// no `liu`). The one test both content-text reconciles share.
876pub(in crate::base_utilities::frontmatter::observer) fn reads_as(
877  word: &str,
878  places: &[(usize, bool)],
879  other: &str,
880) -> bool {
881  fn fits(word: &[char], other: &[char], places: &[usize]) -> bool {
882    let Some((&at, rest)) = places.split_first() else {
883      return word == other;
884    };
885    if word.len() < at || other.len() < at || word[..at] != other[..at] {
886      return false;
887    }
888    let (word, other) = (&word[at..], &other[at..]);
889    let rest: Vec<usize> = rest.iter().map(|p| p - at).collect();
890    fits(word, other, &rest)
891      || other.first().is_some_and(|c| matches!(c, 'i' | 'j' | 'ȷ'))
892        && fits(word, &other[1..], &rest)
893  }
894  let word: Vec<char> = base_letters(word).chars().collect();
895  let other: Vec<char> = base_letters(other).chars().collect();
896  let dotless: Vec<usize> = places
897    .iter()
898    .filter(|(_, letter)| *letter)
899    .map(|(at, _)| *at)
900    .collect();
901  fits(&word, &other, &dotless)
902}

904/// The texts the inks print, each block's and call's (by its ledger index) in order, by their words ([`ink_words`], lowercased,
905/// in order; an address a line prints beside them is an e-mail's, which the reading takes out of the entry: IEEEtran's
906/// `\IEEEauthorblockA{…\\ Burwood 3125, Australia \\ ann@deakin.edu.au}`, 2311.03657): what a shared item's copies count
907/// up to, each print taken once.
908#[derive(Default)]
909struct Prints {
910  prints: Vec<(usize, Vec<String>, bool)>,
911  /// each print's position by its first word, in order: where a run of prints spelling a text can start
912  starts: BTreeMap<String, Vec<usize>>,
913}
914
915impl Prints {
916  /// `text`'s words, lowercased.
917  fn key(text: &str) -> Vec<String> {
918    let (words, _) = ink_words(text);
919    words.iter().map(|w| w.to_lowercase()).collect()
920  }
921
922  fn add(&mut self, source: usize, text: &str) {
923    let key = Self::key(text);
924    if let Some(first) = key.first() {
925      self
926        .starts
927        .entry(first.clone())
928        .or_default()
929        .push(self.prints.len());
930      self.prints.push((source, key, false));
931    }
932  }
933
934  /// How often a shared item printing `text` counts: as often as the inks print its text — one print, or a run of one block's
935  /// or call's prints in a row (an entry of two lines), none taken already —, at least `most` (the most copies one creator holds)
936  /// and at most `total` (its copies in all): a line printed under each author and kept as one shared item is no loss, and a
937  /// line printed once never counts more than one creator's copies (review revN2, the main session's ruling on N2r2). Each print
938  /// it counts is taken: an item of another kind, or a shorter entry inside it, counts what is left.
939  fn take(&mut self, text: &str, most: usize, total: usize) -> usize {
940    let key = Self::key(text);
941    let mut found = Vec::new();
942    let mut at = 0;
943    // (a run spelling `key` starts at a print its first word starts — a print's words are a prefix of what is left to spell —,
944    // so only those are tried, in order, not every print for every item: 2103.04797's ~1,000 creators)
945    let starts = key
946      .first()
947      .and_then(|first| self.starts.get(first))
948      .map_or(&[][..], Vec::as_slice);
949    for &start in starts {
950      if found.len() >= total {
951        break;
952      }
953      if start < at {
954        continue;
955      }
956      // (how much of `key` the prints from `start` spell, in a row)
957      let (mut spelled, mut end) = (0, start);
958      while end < self.prints.len()
959        && spelled < key.len()
960        && !self.prints[end].2
961        && self.prints[end].0 == self.prints[start].0
962        && key[spelled..].starts_with(&self.prints[end].1)
963      {
964        spelled += self.prints[end].1.len();
965        end += 1;
966      }
967      if spelled == key.len() {
968        found.push(start..end);
969        at = end;
970      }
971    }
972    let n = found.len().clamp(most, total);
973    for range in found.into_iter().take(n) {
974      self.prints[range]
975        .iter_mut()
976        .for_each(|print| print.2 = true);
977    }
978    n
979  }
980}

988  let text = label

991  (label, text)
992}

994/// How many of `gap` copies of `address` one side lacks are the same ink as its local part's words: as many as `words`
995/// holds each of those words more often than `other`, at most `gap`.
996fn same_ink(address: &str, gap: usize, words: &Tally, other: &Tally) -> usize {
997  let local = local_words(address);
998  let spare = local
999    .iter()
1000    .map(|word| words.word(word).saturating_sub(other.word(word)))
1001    .min();
1002  spare.unwrap_or(0).min(gap)
1003}

1011/// How many of `gap` copies of `word` one side lacks are two words of the other glued, and those two: the first cut of `word`
1012/// into two whose every part not a mark's `words` holds more often than `other` (twice for a word glued to itself), a part
1013/// a mark printed (`marks`, a Mark event's words: `IDSIA1`) taken as printed; none where both parts are marks'.
1014fn glued(
1015  word: &str,
1016  gap: usize,
1017  words: &Tally,
1018  other: &Tally,
1019  marks: &BTreeMap<String, usize>,
1020) -> Option<(usize, [String; 2])> {
1021  if gap == 0 {
1022    return None;
1023  }
1024  let spare = |part: &str| words.word(part).saturating_sub(other.word(part));
1025  for (at, _) in word.char_indices().skip(1) {
1026    let (a, b) = word.split_at(at);
1027    let n = match (marks.contains_key(a), marks.contains_key(b)) {
1028      (true, true) => continue,
1029      (false, false) if a == b => spare(a) / 2,
1030      (false, false) => spare(a).min(spare(b)),
1031      (false, true) => spare(a),
1032      (true, false) => spare(b),
1033    };
1034    if n > 0 {
1035      return Some((n.min(gap), [a, b].map(|part| part.to_string())));
1036    }
1037  }
1038  None
1039}
