1//! The observer's resolver (Layer C of the design note's §2; OXIDIZED_DESIGN_DIVERGENCES #837): a document's ledger —
2//! each author block's events and the frontmatter API calls after them — read into the tuples `ab_audit`'s `front.rs`
3//! extracts from the XML (each creator's name, affiliations, contacts and notes, and the author block's own), each
4//! association with the rule that decided it. Class-independent; a family's conventions are those its binding declared
5//! (`policy.rs`, Layer B). Stage S2a.2 ran it in shadow mode, nothing it decided reaching the output; since the flip the
6//! emitter writes what it decides (OXIDIZED_DESIGN_DIVERGENCES #980).
7//!
8//! The rules, in order (design §2 points 1-9, the FIDELITY_CHARTER §4 rulings of 2026-10-21):
9//! 1. **Geometry**: columns from `Sep(And)` (the `\and` family) and alignment tabs, unless the policy reads `\and` as
10//!    inline (llncs); rows from `Break`.
11//! 2. **Line type**: a line whose first ink is a mark is a labelled line, unless it is the block's first line and its
12//!    entries read as names with their marks answered below (the prefix convention); a column's first line, or one
13//!    reading as names, is a names line; a line of contacts only a contact line; any other an unmarked line, the
14//!    classify helpers (`name_count`, `reads_as_affiliation`) deciding.
15//! 3. **Names**: a names line split at its separators; each mark attached to the name beside it — the one before,
16//!    or the one after a comma the mark is not glued to (a prefix), or the one after when no name precedes; an `\inst`
17//!    after a comma the one before under the policy (OD #807). A font switch inside a name run keeps it whole (the
18//!    lexer's run: gretsi's `first \textsc{last}`, 2508.03934).
19//! 4. **Ruling 1**: the marks shown on a name are its requests (`1,2` → 1, 2; a symbol run glued to a label, `*1`, two
20//!    requests); the name is its text without them. A labelled entry goes to every name requesting its label.
21//! 5. **Ruling 2**: a line led by a footnote-symbol mark labels that line for every name carrying the same character;
22//!    a symbol no name carries leaves its line to the block.
23//! 6. **Unlabelled lines** go to the names row above them in their column (R12; llncs's `\\` line, OD #792).
24//! 7. **Contacts**: own mark → spelled author → (the class's list in author order, `SpelledElseOrder`) → the group's
25//!    only author → elimination (the one address no name spells, the one name no address spells: today's, OD #52 under
26//!    user ruling R10) → the one author marked as the corresponding author → the block.
27//! 8. **Ruling 3**: one contact per print event, nothing deduplicated (a line's addresses one each).
28//! 9. **Ruling 4**: a note whose content all became contacts is dropped, and so is a note nothing in which prints (an
29//!    empty `\thanks{}`: `Event::inkless`, the user's ruling 2026-10-30; OD #916).
30//! 10. **API calls** after a block (not made by its expansion), creators and creator annotations only
31//!     ([`creator_call`]): a creator; a labelled contact (its `label=`, its `labelseq=` place, its leading mark, or a
32//!     `\label` a `\ref` mark names) to the names requesting it — a creator's own `author:N` exactly, any other label by
33//!     Perl's prefix chain ([`label_owners`]); an unlabelled one by its `annotate=` (the last author, all, the run of new
34//!     ones), authblk's by its `\AB@thenote` when the marks show.
35//!
36//! S2a.2 r2 (review revfms2a2) adds: names lines by evidence (`later_line_is_names`), a note's whole text and its
37//! legend by its leading Mark, marks mid-line and trailing, today's joined unmarked lines and contact grid, a replacing
38//! `\author` call (OXIDIZED_DESIGN_DIVERGENCES #837's r2 paragraph).

106/// A rule of the reading joining runs a separator parts into one name (`lines.rs`, `blocks.rs`), each its tag in the
107/// creator's `rule`: a command's own gap (`\EES`), initials or a given name at a gap (`V.G.~ Kagramanova`), runs only a
108/// layout parts (`\firstname{…} \surname{…}`), a name written surname first (`Eckart, A.`, 1501.02171), a qualifier
109/// (`Robert W. Heath, Jr.`, 1004.2377), a remark under it.
110#[derive(Debug, Clone, Copy, PartialEq, Eq)]
111pub enum NameJoin {
112  Macro,
113  Initials,
114  Gap,
115  Layout,
116  Reversed,
117  Qualifier,
118  Remark,
119}
120
121impl NameJoin {
122  /// Its tag in a creator's `rule`.
123  pub fn tag(self) -> &'static str {
124    match self {
125      NameJoin::Macro => "name:macro-joined",
126      NameJoin::Initials => "name:initials-joined",
127      NameJoin::Gap => "name:gap-joined",
128      NameJoin::Layout => "name:layout-joined",
129      NameJoin::Reversed => "name:reversed",
130      NameJoin::Qualifier => "name:qualifier",
131      NameJoin::Remark => "name:remark",
132    }
133  }
134}

136impl RCreator {
137  /// Record the reading's `join` of this name: its rule tag, once, and the join.
138  pub(super) fn join(&mut self, join: NameJoin) {
139    if !self.rule.contains(join.tag()) {
140      self.rule.push(' ');
141      self.rule.push_str(join.tag());
142    }
143    if !self.joins.contains(&join) {
144      self.joins.push(join);
145    }
146  }
147}

325/// A block's events with each run of link pieces that reads as addresses only once their inks are joined one e-mail
326/// contact, the run's addresses — `\url{robin.andre@ira.uka.de}, \{\url{sebastian.schlag, christian.schulz}\}\url{@kit.edu}`,
327/// `\url{christian.schulz}\url{@univie.ac.at}` (1710.01968): each link read apart was a URL, `@univie.ac.at` on every
328/// author, `christian.schulz @univie.ac.at` too. The old split's raw address line (`row_email_calls`, deleted at P27b), read
329/// as the line prints: a line of links (`url`) with a piece printing no `@` among them, between them only separators and
330/// the braces of a list of local parts. Only the run of pieces is joined, through the ink that completes its addresses
331/// (its span the first ink's start to the last ink's end); a link that is a whole address on its own after such a run
332/// stays its own contact, with its markup — `\texttt{\href{mailto:…}{\{samdie2016}, \href{mailto:…}
333/// {yanyibo70\}@gmail.com}}, \texttt{\href{mailto:xuminghu@hkust-gz.edu.cn}{…}}` keeps xuminghu@'s typewriter `mailto`
334/// link (2505.15472, review revZ18g B1).
335fn joined_address_pieces(events: &[Event]) -> Vec<Event> {
336  let url = |e: &Event| e.kind == EventKind::Contact && e.role.as_deref() == Some("url");
337  let glue = |e: &Event| {
338    e.kind == EventKind::Name
339      && e
340        .text
341        .chars()
342        .all(|c| "{}".contains(c) || c.is_whitespace())
343  };
344  // (the run's text as it prints: its inks, a comma's `, `, any other separator a space)
345  let text_of = |run: &[Event]| {
346    let mut joined = String::new();
347    for e in run {
348      match &e.kind {
349        EventKind::Name | EventKind::Contact => joined.push_str(&e.text),
350        EventKind::Sep(SepKind::Comma) => joined.push_str(", "),
351        EventKind::Sep(_) => joined.push(' '),
352        _ => {},
353      }
354    }
355    joined
356  };
357  // (whether a text is all addresses, each whole, its list's braces closed: a run complete, or a link that is one or more
358  // on its own)
359  let complete = |text: &str| {
360    let addresses = addresses_in(text);
361    text.contains('@')
362      && text.matches('{').count() == text.matches('}').count()
363      && only_addresses(text)
364      && !addresses.is_empty()
365      && addresses.iter().all(|a| is_whole_address(a))
366  };
367  let mut out: Vec<Event> = Vec::with_capacity(events.len());
368  let mut start = 0usize;
369  while start <= events.len() {
370    let end = (start..events.len())
371      .find(|&at| events[at].kind == EventKind::Break)
372      .unwrap_or(events.len());
373    let line = &events[start..end];
374    let take = line.iter().all(|e| {
375      (inked(e) && (url(e) || glue(e)))
376        || matches!(e.kind, EventKind::Sep(SepKind::Comma) | EventKind::Layout)
377    }) && line.iter().any(|e| url(e) && !e.text.contains('@'));
378    // (the runs of pieces, each its first and last ink's places in the line: a run opens at the line's first ink, or at
379    // the first ink after a run that is no whole address on its own, and closes at the ink after which it holds a piece
380    // and its text is all whole addresses; a whole address after a run stands alone. One before the first piece is the
381    // run's: `\textit{\url{robin.andre@…}, \{\url{…}}\}\url{@kit.edu}` (1710.01968) prints it in the pieces' own group,
382    // which the writer would take whole for it)
383    let alone = |e: &Event| url(e) && complete(&e.text);
384    let mut runs: Vec<(usize, usize)> = Vec::new();
385    let mut open: Option<usize> = None;
386    if take {
387      for (at, e) in line.iter().enumerate().filter(|(_, e)| inked(e)) {
388        if open.is_none() {
389          if alone(e) && !runs.is_empty() {
390            continue;
391          }
392          open = Some(at);
393        }
394        if let Some(from) = open
395          && line[from..=at].iter().any(|e| inked(e) && !alone(e))
396          && complete(&text_of(&line[from..=at]))
397        {
398          runs.push((from, at));
399          open = None;
400        }
401      }
402    }
403    if take && open.is_none() && !runs.is_empty() {
404      let mut at = 0usize;
405      for &(from, to) in &runs {
406        out.extend(line[at..from].iter().cloned());
407        let run = &line[from..=to];
408        let span = run
409          .iter()
410          .filter(|e| inked(e))
411          .filter_map(|e| e.span)
412          .fold(None, |acc: Option<(u16, u16)>, (s, t)| {
413            Some(acc.map_or((s, t), |(a, b)| (a.min(s), b.max(t))))
414          });
415        out.push(Event {
416          kind: EventKind::Contact,
417          role: Some("email".to_string()),
418          text: text_of(run)
419            .split_whitespace()
420            .collect::<Vec<_>>()
421            .join(" "),
422          target: None,
423          span,
424          ..line[from].clone()
425        });
426        at = to + 1;
427      }
428      out.extend(line[at..].iter().cloned());
429    } else {
430      out.extend(line.iter().cloned());
431    }
432    if end < events.len() {
433      out.push(events[end].clone());
434    }
435    start = end + 1;
436  }
437  out
438}

443/// department centred under both names, every name's of the row, its "and" no separator (R12; 2609.17357, 2609.01563,
444/// 2609.33831). The deleted old driver's rule (`trailing_alignment_environment_as_lines`, which wrote `\\` and the
445/// environment's content) on the digested container: one at the top level (nest 0) with an event before it and no ink
446/// past its content; a Break already before it is the line's own. Its container goes with it, as the old rule drops the
447/// environment, so its lines are lines of their own, not one print unit (`Resolver`'s open group). One that opens the
448/// block is a wrapper the lines read through (their Breaks part them at any nest: 2609.01563).
449fn trailing_alignment_as_lines(events: &[Event]) -> Vec<Event> {
450  // (the environment a container is, by the command its role names: `\\begin{center}`)
451  let alignment = |e: &Event| {
452    let environment = e.role.as_deref().and_then(|role| role.split_once("begin{"));
453    e.kind == EventKind::Layout
454      && e.nest == 0
455      && matches!(
456        environment,
457        Some((_, "center}" | "flushleft}" | "flushright}"))
458      )
459  };
460  let content = |at: usize| {
461    events[at + 1..]
462      .iter()
463      .take_while(|inner| inner.nest > 0)
464      .count()
465  };
466  let Some(at) = (1..events.len())
467    .find(|&at| alignment(&events[at]) && !events[at + 1 + content(at)..].iter().any(inked))
468  else {
469    return events.to_vec();
470  };
471  let mut out = events[..at].to_vec();
472  if events[..at]
473    .iter()
474    .rev()
475    .find(|e| e.kind != EventKind::Layout)
476    .is_some_and(|e| e.kind != EventKind::Break)
477  {
478    let mut row = Event::new(EventKind::Break);
479    row.span = events[at].span.map(|(start, _)| (start, start));
480    out.push(row);
481  }
482  let lines = &events[at + 1..at + 1 + content(at)];
483  out.extend(
484    lines
485      .iter()
486      .map(|e| Event { nest: e.nest - 1, ..e.clone() }),
487  );
488  out.extend_from_slice(&events[at + 1 + content(at)..]);
489  out
490}

802  fn attach(&mut self, owner: Option<usize>, mut assoc: Assoc, note: bool) {
803    // (an unlabelled line of the block that names a collaboration is its owners' collaboration, as today's reader makes
804    // it — `(XYZ Collaboration)` under the names, 2410.11607; `collaboration_line_framing`, OXIDIZED_DESIGN_DIVERGENCES
805    // #946 —, not a class's affiliation command holding one: revtex's `\affiliation{(WA98 Collaboration)}`, 0708.2630;
806    // a team's only framed by more than "the", `The Robotics Group` an institution's)
807    let line_affiliation = !note
808      && owner.is_some()
809      && assoc.role == "affiliation"
810      && assoc.label.is_none()
811      && assoc.rule.starts_with("line:");
812    let framing = line_affiliation
813      .then(|| collaboration_line_framing(&assoc.text))
814      .flatten();
815    if framing.is_some() {
816      assoc.role = "collaboration".into();
817    }
818    match owner {
819      Some(at) => {
820        // (an address line under an unframed one naming a collaboration makes that one an institution's, its address
821        // going on: `\textit{WIDE Consortium}\\ \.Zabbar\\ Malta`, 2501.13770 — as today's reader,
822        // `collaboration_line_before`)
823        if line_affiliation && framing.is_none() {
824          self.institution_after_all(at);
825        }
826        let creator = &mut self.out.creators[at];
827        if note {
828          creator.notes.push(assoc);
829        } else {
830          self.roles[at].push(assoc.role.clone());
831          if assoc.role == "affiliation" || assoc.role == "address" {
832            self.order[at].push((true, creator.affiliations.len()));
833            creator.affiliations.push(assoc);
834          } else {
835            if framing == Some(Framing::Unframed) {
836              self.unframed.push((at, creator.contacts.len()));
837            }
838            self.order[at].push((false, creator.contacts.len()));
839            creator.contacts.push(assoc);
840          }
841        }
842      },
843      None => self.out.block.push(assoc),
844    }

887    };

889  }

1417    let mut at = 0usize;

1470  }
1471  in_source_order(&mut out.block);
