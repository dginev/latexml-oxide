1//! The mark ledger: the name each `\footnotemark` or note printed beside spacing goes to — the name before it, at a line's
2//! or cell's start the name after, between two names the name after where the block's edge proof holds or the name before
3//! carries its key already (DG 2026-11-17) —, read once per block for the reading and the writer
4//! (OXIDIZED_DESIGN_DIVERGENCES #1057; 2412.20504, 1911.13299, 1807.02791, 1703.08733, 1111.1979).

8/// The name a mark or note of the ledger goes to.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub(in crate::base_utilities::frontmatter::observer) enum Side {
11  /// the name before it on its line
12  Before,
13  /// the name after it on its line
14  After,
15}

17/// The glue on one side of a mark out to a name's ink: its natural width (em of the box's font), whether a fill
18/// stretches it, and whether there is any (a Gap or an interword space, whatever its width).
19#[derive(Debug, Clone, Copy, PartialEq)]
20pub(in crate::base_utilities::frontmatter::observer) struct Reach {
21  pub width: f64,
22  pub fill:  bool,
23  pub space: bool,
24}
25
26impl Reach {
27  const NONE: Reach = Reach {
28    width: 0.0,
29    fill:  false,
30    space: false,
31  };
32
33  fn widest(self, other: Reach) -> Reach {
34    Reach {
35      width: self.width.max(other.width),
36      fill:  self.fill || other.fill,
37      space: self.space || other.space,
38    }
39  }
40}

42/// One `\footnotemark` or note set beside a name's spacing, and the name it goes to.
43#[derive(Debug, Clone, PartialEq)]
44pub(in crate::base_utilities::frontmatter::observer) struct MarkRecord {
45  /// its place among the block's read events (`resolver::read_events`)
46  pub at:     usize,
47  pub span:   Option<(u16, u16)>,
48  /// the Name event before it on its line, and the widest glue out to that name's last ink; `None` past a cell's start
49  pub before: Option<(usize, Reach)>,
50  /// the Name event after it on its line, and the widest glue out to its first ink; `None` before a separator, `\\` or
51  /// the block's end
52  pub after:  Option<(usize, Reach)>,
53  /// `None`: undecided
54  pub side:   Option<Side>,
55}
56
57impl MarkRecord {
58  /// Whether its decided side gives it to the Name event at `name`.
59  fn goes_to(&self, name: usize) -> bool {
60    match self.side {
61      Some(Side::Before) => self.before.is_some_and(|(at, _)| at == name),
62      Some(Side::After) => self.after.is_some_and(|(at, _)| at == name),
63      None => false,
64    }
65  }
66}

68/// The ledger of a block's marks and notes set beside spacing ([`mark_ledger`]).
69#[derive(Debug, Clone, Default, PartialEq)]
70pub(in crate::base_utilities::frontmatter::observer) struct MarkLedger {
71  pub records:    Vec<MarkRecord>,
72  /// each mark printed on top of the title mark before it, of the same glyph ([`overprints`]): its span and the glyph
73  pub overprints: Vec<((u16, u16), String)>,
74}
75
76impl MarkLedger {
77  /// The glyph the mark or note with span `span` prints on top of the same glyph ([`overprints`]).
78  pub fn overprint_of(&self, span: Option<(u16, u16)>) -> Option<&str> {
79    let span = span?;
80    self
81      .overprints
82      .iter()
83      .find(|(at, _)| *at == span)
84      .map(|(_, glyph)| glyph.as_str())
85  }
86
87  /// The record of the mark or note with span `span`.
88  pub fn record_of(&self, span: Option<(u16, u16)>) -> Option<&MarkRecord> {
89    span.and_then(|span| self.records.iter().find(|r| r.span == Some(span)))
90  }
91
92  /// The decided side of the mark or note with span `span`.
93  pub fn side_of(&self, span: Option<(u16, u16)>) -> Option<Side> {
94    self.record_of(span).and_then(|r| r.side)
95  }
96
97  /// The records past the Name event at `name`, before the next name, each with its side.
98  pub fn past(&self, name: usize) -> impl Iterator<Item = &MarkRecord> {
99    self
100      .records
101      .iter()
102      .filter(move |r| r.before.is_some_and(|(at, _)| at == name))
103  }
104
105  /// The records the Name event at `name` takes as its prefix, in print order.
106  pub fn prefix_of(&self, name: usize) -> impl Iterator<Item = &MarkRecord> {
107    self
108      .records
109      .iter()
110      .filter(move |r| r.side == Some(Side::After) && r.after.is_some_and(|(at, _)| at == name))
111  }
112}

114/// What a stretch of a line between two inks opened at.
115#[derive(Clone, Copy, PartialEq)]
116enum Opening {
117  /// a name (the Name event's place)
118  Name(usize),
119  /// a cell (`\and`, `&`): no name before it on its line
120  Cell,
121  /// the block's start
122  Start,
123  /// a `\\`'s line start: no name before it on its line
124  Line,
125  /// anything else: a comma, the word "and", other ink
126  Other,
127}
128
129/// A movable piece: a text `\footnotemark` or a note (a mark the reading writes apart from the name).
130fn movable(e: &Event) -> bool {
131  match e.kind {
132    EventKind::Mark => {
133      e.role.as_deref() == Some("footnotemark") && !e.formula.is_some_and(|f| f.base.is_some())
134    },
135    EventKind::Note => !events::is_picture_note(e),
136    _ => false,
137  }
138}

140/// The ledger of `events`, a block's read events ([`MarkLedger`]).
141pub(in crate::base_utilities::frontmatter::observer) fn mark_ledger(
142  events: &[Event],
143) -> MarkLedger {
144  let mut ledger = MarkLedger::default();
145  let mut opening = Opening::Start;
146  // (the records between two names, which the block's edge proof gives the name after)
147  let mut between: Vec<usize> = Vec::new();
148  // (the stretch's pieces: (place, movable), the glue before each, the glue since the last ink)
149  let mut pieces: Vec<(usize, bool)> = Vec::new();
150  let mut glues: Vec<Reach> = Vec::new();
151  let mut glue = Reach::NONE;
152  let mut gap_seen = false;
153  let mut space_after = false;
154  // (a comma glued to a name's run, read as its ink: the name opens what follows only when its first piece is glued to
155  // the comma too — a mark set off from the comma by a space is between the comma and the next name, as before)
156  let mut past_comma = false;
157  // (a single interword space no Gap event holds, the lexer's `space_before`/`space_after`: its width, `\fontdimen2`
158  // of cmr10, 1/3 em)
159  let hidden = |glue: &mut Reach, gap_seen: bool, space: bool| {
160    glue.space |= space;
161    if !gap_seen && space {
162      glue.width += 1.0 / 3.0;
163    }
164  };
165  for (k, e) in events.iter().enumerate() {
166    match &e.kind {
167      EventKind::Sep(SepKind::Gap { width, fill, .. }) => {
168        glue.width += width;
169        glue.fill |= fill;
170        glue.space = true;
171        gap_seen = true;
172      },
173      // (an alignment's cell opens as `&` does: `\begin{tabular}{cc}Ann Able & \footnotemark[2]Bob Baker`)
174      EventKind::Layout if e.role.as_deref() == Some("cell") => {
175        glues.push(glue);
176        decide(
177          &mut ledger,
178          &mut between,
179          events,
180          opening,
181          None,
182          &pieces,
183          &glues,
184        );
185        opening = Opening::Cell;
186        past_comma = false;
187        (pieces, glues, glue, gap_seen, space_after) =
188          (Vec::new(), Vec::new(), Reach::NONE, false, false);
189      },
190      EventKind::Layout if !is_image(e) => {},
191      // (what prints nothing: a contact recorded only, AASTeX 7's `\email`)
192      EventKind::Declaration | EventKind::ApiCall => {},
193      EventKind::Contact if e.unprinted.is_some() => {},
194      EventKind::Name if !norm(&e.text).is_empty() => {
195        hidden(&mut glue, gap_seen, space_after);
196        glues.push(glue);
197        decide(
198          &mut ledger,
199          &mut between,
200          events,
201          opening,
202          Some(k),
203          &pieces,
204          &glues,
205        );
206        opening = Opening::Name(k);
207        past_comma = false;
208        (pieces, glues, glue, gap_seen, space_after) =
209          (Vec::new(), Vec::new(), Reach::NONE, false, false);
210      },
211      EventKind::Mark
212      | EventKind::Note
213      | EventKind::Contact
214      | EventKind::Layout
215      | EventKind::Name => {
216        hidden(&mut glue, gap_seen, space_after || e.space_before);
217        if std::mem::take(&mut past_comma) && (glue.width != 0.0 || glue.fill) {
218          opening = Opening::Other;
219        }
220        glues.push(glue);
221        pieces.push((k, movable(e)));
222        (glue, gap_seen, space_after) = (Reach::NONE, false, e.space_after);
223      },
224      EventKind::Sep(kind) => {
225        glues.push(glue);
226        // (a formula's separator between marks, `$^,$`, is ink of the run: 1703.08733)
227        if e.role.as_deref() == Some("math") {
228          pieces.push((k, false));
229          (glue, gap_seen, space_after) = (Reach::NONE, false, false);
230          continue;
231        }
232        // (a comma glued to a name's run is that name's ink: what follows it is that name's — `Igor Pikovski,$^{1, 2
233        // }$\footnotemark $\, \,$ Michael`, Igor's (COMMAMARK: 1111.1979, 1912.01737, 1811.08622);
234        // `Lenk\footnotemark[1]\textsuperscript{,}\footnotemark[2] \and`, Lenk's (2407.04435))
235        let name_ink = matches!(opening, Opening::Name(_))
236          && *kind == SepKind::Comma
237          && !space_after
238          && !e.space_before
239          && glues.iter().all(|g| g.width == 0.0 && !g.fill);
240        decide(
241          &mut ledger,
242          &mut between,
243          events,
244          opening,
245          None,
246          &pieces,
247          &glues,
248        );
249        past_comma = name_ink;
250        opening = match kind {
251          _ if name_ink => opening,
252          SepKind::And(_) | SepKind::Column(_) => Opening::Cell,
253          _ => Opening::Other,
254        };
255        (pieces, glues, glue, gap_seen, space_after) =
256          (Vec::new(), Vec::new(), Reach::NONE, false, false);
257      },
258      EventKind::Break => {
259        glues.push(glue);
260        decide(
261          &mut ledger,
262          &mut between,
263          events,
264          opening,
265          None,
266          &pieces,
267          &glues,
268        );
269        opening = Opening::Line;
270        past_comma = false;
271        (pieces, glues, glue, gap_seen, space_after) =
272          (Vec::new(), Vec::new(), Reach::NONE, false, false);
273      },
274    }
275  }
276  glues.push(glue);
277  decide(
278    &mut ledger,
279    &mut between,
280    events,
281    opening,
282    None,
283    &pieces,
284    &glues,
285  );
286  // (the edge proof: the block's first piece before its first name and none after its last, every piece between two
287  // names is the name's after — 2406.12338, math0606768; both edges, the default, counted: 1511.05393, 0906.1223. DG
288  // 2026-11-17: the first piece sits in the stretch of the first name of the names' line, not before a line above it,
289  // maybe an affiliation's (S2); a piece glued to the name before it, no space between, proves the other side (S1))
290  let is_name = |k: usize| events[k].kind == EventKind::Name && !norm(&events[k].text).is_empty();
291  let last_name = (0..events.len()).rev().find(|&k| is_name(k));
292  let last_piece = (0..events.len()).rev().find(|&k| movable(&events[k]));
293  if let Some(&first_between) = between.first() {
294    let at = ledger.records[first_between].at;
295    let line = events[..at]
296      .iter()
297      .rposition(|e| e.kind == EventKind::Break)
298      .map_or(0, |k| k + 1);
299    let line_name = (line..events.len()).find(|&k| is_name(k));
300    if ledger.records[0].after.map(|(name, _)| name) == line_name {
301      let glued = ledger
302        .records
303        .iter()
304        .any(|r| r.before.is_some_and(|(_, reach)| !reach.space));
305      if last_piece < last_name && !glued {
306        for &at in &between {
307          ledger.records[at].side = Some(Side::After);
308        }
309      } else {
310        Info!(
311          "frontmatter",
312          "mark_convention",
313          s!(
314            "An author block sets notes both before its names and after them: {} between two names given the name \
315             before",
316            between.len()
317          )
318        );
319      }
320    }
321  }
322  // (DG 2026-11-17 S3: a piece between two names whose key, its footnote number, the name before carries already is the
323  // next name's, with the pieces after it on its stretch — a name printing one key twice contradicts itself: `Ann
324  // Able\footnotemark[2] \quad \footnotemark[2]Bob Baker`, "Ann Able† †Bob Baker". One glued to the piece of its key
325  // prints over it, once: `\footnotemark[1]\thanks{…}`, 2103.01544)
326  let apart = |from: usize, to: usize| {
327    events[from].space_after
328      || events[to].space_before
329      || events[from + 1..to]
330        .iter()
331        .any(|e| matches!(e.kind, EventKind::Sep(_)) || e.space_before || e.space_after)
332  };
333  for &i in &between {
334    let r = &ledger.records[i];
335    let (Some(Side::Before), Some((name, _)), Some(key)) = (r.side, r.before, events[r.at].number)
336    else {
337      continue;
338    };
339    let carried = ledger.records[..i]
340      .iter()
341      .rfind(|o| o.goes_to(name) && events[o.at].number == Some(key));
342    if carried.is_some_and(|o| apart(o.at, r.at)) {
343      let next = r.after.map(|(at, _)| at);
344      for o in &mut ledger.records[i..] {
345        if o.before.map(|(at, _)| at) == Some(name) && o.after.map(|(at, _)| at) == next {
346          o.side = Some(Side::After);
347        }
348      }
349    }
350  }
351  // (the block's decided sides beside its undecided marks: what a block's convention would decide, measured — MARKKEY K8)
352  if ledger.records.iter().any(|r| r.side.is_none()) {
353    let count = |side: Option<Side>| ledger.records.iter().filter(|r| r.side == side).count();
354    DebugFeature!(
355      "frontmatter_ledger",
356      "FRONT mark ledger: {} before, {} after, {} undecided",
357      count(Some(Side::Before)),
358      count(Some(Side::After)),
359      count(None)
360    );
361  }
362  ledger
363}

365/// Record the stretch `pieces` between `opening` and the Name event `next` (`None`: a separator, `\\` or the block's
366/// end), `glues[i]` the glue before piece `i` and its last the glue out to `next`'s ink.
367fn decide(
368  ledger: &mut MarkLedger,
369  between: &mut Vec<usize>,
370  events: &[Event],
371  opening: Opening,
372  next: Option<usize>,
373  pieces: &[(usize, bool)],
374  glues: &[Reach],
375) {
376  if !pieces.iter().any(|&(_, movable)| movable) {
377    return;
378  }
379  let before = match opening {
380    Opening::Name(at) => Some(at),
381    _ => None,
382  };
383  // (DG 2026-11-17: "nearly all notes attach on the right side of an author name and on the left side of an affiliation
384  // name" — between two names the name's before, whatever the glue; at a line's or cell's start the name's after)
385  let side = match (before, next) {
386    (Some(_), _) => Some(Side::Before),
387    (None, Some(_)) if matches!(opening, Opening::Start | Opening::Line | Opening::Cell) => {
388      Some(Side::After)
389    },
390    _ => None,
391  };
392  for (i, &(at, movable)) in pieces.iter().enumerate() {
393    if !movable {
394      continue;
395    }
396    if next.is_some() && matches!(opening, Opening::Name(_) | Opening::Other) {
397      between.push(ledger.records.len());
398    }
399    let reach = |range: &[Reach]| range.iter().copied().fold(Reach::NONE, Reach::widest);
400    ledger.records.push(MarkRecord {
401      at,
402      span: events[at].span,
403      before: before.map(|b| (b, reach(&glues[..=i]))),
404      after: next.map(|n| (n, reach(&glues[i + 1..]))),
405      side,
406    });
407  }
408}

410/// The marks of `events` (a block's read events) TeX prints on top of the mark before them, of the same glyph: where the
411/// title's footnote mark takes no width (`zero_width`: article.cls:206, report.cls:212, book.cls:218 `\rlap` it), a mark
412/// or note set after it with no glue between them prints over it — `Ward Struyve\footnote{A}$^{*}$\footnote{B}` reads
413/// "Struyve∗†", its two ∗ at one place (2012.08663), `Aaditya Singh\footnotemark[1]\thanks{…}` one ∗ (2103.01544) —, so
414/// the page shows that glyph once. A mark set after an inked one stands beside it (`Wei Fan$^{*}$\thanks{…}` "∗∗"),
415/// and a mark of another glyph is ink of its own. Decided on the glue the ledger reads (a `Gap`, an interword space);
416/// the glyphs compared as one asterisk.
417#[cfg_attr(
418  not(test),
419  allow(dead_code, reason = "no reader until MARKKEY R2b's display (OD #1062)")
420)]
421pub(in crate::base_utilities::frontmatter::observer) fn overprints(
422  events: &[Event],
423  zero_width: bool,
424) -> Vec<((u16, u16), String)> {
425  let mut out = Vec::new();
426  if !zero_width {
427    return out;
428  }
429  // (the glyph a mark or note prints on the name, and whether it is the title's own footnote mark, of no width)
430  let printed = |e: &Event| -> Option<(String, bool)> {
431    match e.kind {
432      EventKind::Note if !events::is_picture_note(e) => {
433        note_shows(e).map(|glyph| (glyph.to_string(), true))
434      },
435      EventKind::Mark => {
436        let glyph = e.text.trim();
437        (!glyph.is_empty()).then(|| (glyph.to_string(), printed_by_footnotemark(e)))
438      },
439      _ => None,
440    }
441  };
442  let form = |glyph: &str| one_asterisk(glyph.trim());
443  // (the zero-width mark before, its glyph and whether a space follows it)
444  let mut under: Option<(String, bool)> = None;
445  for e in events {
446    match &e.kind {
447      // (what prints nothing between them: a font change, a recorded contact)
448      EventKind::Layout if !is_image(e) => continue,
449      EventKind::Declaration | EventKind::ApiCall => continue,
450      EventKind::Contact if e.unprinted.is_some() => continue,
451      _ => {},
452    }
453    let Some((glyph, zero)) = printed(e) else {
454      under = None;
455      continue;
456    };
457    if let Some((before, space_after)) = &under
458      && !space_after
459      && !e.space_before
460      && form(before) == form(&glyph)
461      && let Some(span) = e.span
462    {
463      out.push((span, glyph.clone()));
464    }
465    under = zero.then_some((glyph, e.space_after));
466  }
467  out
468}

622  /// At a line's or a cell's start, no name before it, a piece is the name's after; between two names, the name's after
623  /// only under the block's edge proof — its first piece before its first name and none after its last (2406.12338's
624  /// `\thanks{…} {Carla Schenker,} \thanks{…} {Xiulin Wang,}`, math0606768, 1511.05393, 0906.1223). A block with pieces
625  /// at both edges mixes the conventions: the default. A `\\` line opening with a mark, maybe an affiliation's, proves
626  /// nothing (revWRONGFIX S2).
627  #[test]
628  fn the_edge_proof_gives_the_pieces_between_two_names_to_the_name_after() {
629    use Side::*;
630    let (ann, bob) = (name("Ann Able"), name("Bob Baker"));
631    let comma = Event::new(EventKind::Sep(SepKind::Comma));
632    assert_eq!(sides(&[mark(), ann.clone()]), [Some(After)]);
633    let mut cell = vec![
634      ann.clone(),
635      Event::new(EventKind::Sep(SepKind::And("\\and"))),
636    ];
637    cell.extend([mark(), bob.clone()]);
638    assert_eq!(sides(&cell), [Some(After)]);
639    // (the proof: past a spaced comma and glued to a name's comma alike)
640    let prefix = [
641      note(false, true),
642      ann.clone(),
643      comma.clone(),
644      note(true, true),
645      bob.clone(),
646    ];
647    assert_eq!(sides(&prefix), [Some(After), Some(After)]);
648    // (re-pinned, DG 2026-11-17 S1: a note glued to the name before it, its comma's or its own, is the after-the-name
649    // convention's evidence — the block mixes the two, the default; After, After before)
650    let glued = [
651      note(false, false),
652      ann.clone(),
653      comma.clone(),
654      note(false, false),
655      bob.clone(),
656    ];
657    assert_eq!(sides(&glued), [Some(After), Some(Before)]);
658    let carl = name("Carl Cole");
659    let suffixes = [
660      note(false, true),
661      ann.clone(),
662      note(false, true),
663      bob.clone(),
664      note(false, true),
665      carl.clone(),
666    ];
667    assert_eq!(sides(&suffixes), [Some(After), Some(Before), Some(Before)]);
668    // (an affiliation line after the names, its own start: the proof holds)
669    let mut lines = prefix.to_vec();
670    lines.push(Event::new(EventKind::Break));
671    lines.extend([mark(), name("University of Somewhere")]);
672    assert_eq!(sides(&lines), [Some(After), Some(After), Some(After)]);
673    // (both edges: the default)
674    let mut mixed = prefix.to_vec();
675    mixed.push(note(true, false));
676    assert_eq!(sides(&mixed), [Some(After), None, Some(Before)]);
677    // (no piece before the first name: the default, a line start after it proving nothing — `Ann Able, \thanks{A} Bob
678    // Baker\\ \footnotemark[1]University`)
679    let mut affil = vec![ann.clone(), comma.clone(), note(true, true), bob.clone()];
680    affil.push(Event::new(EventKind::Break));
681    affil.extend([mark(), name("University of Somewhere")]);
682    assert_eq!(sides(&affil), [None, Some(After)]);
683    // (DG 2026-11-17 S2: the first piece sits in the stretch of the first name of the names' line, not before a line
684    // above it, maybe an affiliation's — `\footnotemark[1]University of X\\ Ann Able \thanks{…} Bob Baker`)
685    let mut above = vec![mark(), name("University of X")];
686    above.push(Event::new(EventKind::Break));
687    above.extend([ann.clone(), note(true, true), bob.clone()]);
688    assert_eq!(sides(&above), [Some(After), Some(Before)]);
689    // (the names' line its own start's: the proof holds — `University of X\\ \thanks{A}Ann, \thanks{B}Bob`)
690    let mut below = vec![name("University of X"), Event::new(EventKind::Break)];
691    below.extend(prefix.iter().cloned());
692    assert_eq!(sides(&below), [Some(After), Some(After)]);
693  }

696  /// — a name printing one key twice contradicts itself — but the next name's, with the pieces after it on the stretch
697  /// (DG 2026-11-17 S3: `Ann Able\footnotemark[2] \quad \footnotemark[2]Bob Baker`, "Ann Able† †Bob Baker"). Another

701    use Side::*;

703    let keyed = |number: i64, space_before: bool| {

706      e
