9/// A note an author's own name holds ([`author_notes`]).
10pub(in crate::base_utilities) struct AuthorNote {
11  /// the author (index among the names read) whose name holds it
12  pub(in crate::base_utilities) owner:                usize,
13  /// for a jheppub-family `\note`, which prints its text under a mark it shows beside no name (jheppub.sty:84-89), the
14  /// label of that mark, which the authors it is about write on their names: its optional argument's (`\note[*]{…}`),
15  /// else the n-th such note's number, the footnote counter it steps (`\author[a,1]{M.V. Smirnov,\note{Corresponding
16  /// author.}}` … `\author[a,1]{{\rm and} J.J. Ling}`, 1802.03677); `None` for a `\thanks` or `\footnote`, which
17  /// prints its own mark on its owner's name
18  pub(in crate::base_utilities) mark:                 Option<String>,
19  /// whether `mark` is the counter's number rather than the note's own optional argument: a number an affiliation's
20  /// label also is, the authors who write it request that affiliation (`\author[1]{Ann Able,\note{…}}`
21  /// `\author[1]{Bob Baker}` `\affiliation[1]{Univ A}`: Bob Baker's 1 is Univ A's), so it is no legend of the note's
22  pub(in crate::base_utilities) numbered:             bool,
23  /// whether its text names the corresponding author ([`names_correspondence`])
24  pub(in crate::base_utilities) names_correspondence: bool,
25}

27/// The notes of the commands `notes` that the names `authors` hold, in their order (a binding's own note command too,
28/// jheppub's `\note{Corresponding author.}`, `\note[*]{…}`).
29pub(in crate::base_utilities) fn author_notes<'a>(
30  authors: impl Iterator<Item = &'a Tokens>,
31  notes: &[Token],
32) -> Vec<AuthorNote> {
33  let mut found = Vec::new();
34  let mut numbered = 0;
35  for (owner, author) in authors.enumerate() {
36    let toks = author.unlist_ref();
37    let mut i = 0;
38    while i < toks.len() {
39      if !notes.contains(&toks[i]) {
40        i += 1;
41        continue;
42      }
43      // (past an optional argument: `\thanks[1]{…}`, `\note[*]{…}`)
44      let mut at = i + 1;
45      let mut mark = None;
46      if toks.get(at) == Some(&T_OTHER!("["))
47        && let Some(close) = toks[at..].iter().position(|t| *t == T_OTHER!("]"))
48      {
49        mark = clean_frontmatter_labels(
50          &Tokens::new(toks[at + 1..at + close].to_vec()).to_string(),
51          "affiliation",
52        )
53        .into_iter()
54        .next();
55        at += close + 1;
56      }
57      if !toks
58        .get(at)
59        .is_some_and(|b| b.get_catcode() == Catcode::BEGIN)
60      {
61        i += 1;
62        continue;
63      }
64      let end = skip_group(toks, at);
65      // (a `\thanks`/`\footnote` prints its own mark on the name, whatever its optional number: its owner alone, and
66      // its number no legend of marks the authors write — `$^1$` there is an affiliation's)
67      let mut counted = false;
68      if toks[i] != T_CS!("\\note") {
69        mark = None;
70      } else if mark.is_none() {
71        numbered += 1;
72        counted = true;
73        mark = clean_frontmatter_labels(&numbered.to_string(), "affiliation")
74          .into_iter()
75          .next();
76      }
77      found.push(AuthorNote {
78        owner,
79        mark,
80        numbered: counted,
81        names_correspondence: names_correspondence(&visible_name_text(&toks[at..end])),
82      });
83      i = end;
84    }
85  }
86  found
87}

89/// A list of local parts written without spaces before one domain, `text` whole (spaces round it aside), whose every
90/// part spells exactly one author, each a different one (`owner`, letter case ignored as the spelling reads it), each
91/// part at the domain, as the source writes it: `M.R.Pestalozzi,a.chrysostomou,j.l.collett@herts.ac.uk` under the three
92/// names, astro-ph0612088 — a capitalised part, which [`write_out_unspaced_lists`] leaves whole, is a local part when it
93/// spells its author. `None` for any other text: an address is never invented.
94pub(in crate::base_utilities) fn spelled_unspaced_list(
95  text: &str,
96  owner: &dyn Fn(&str) -> Option<usize>,
97) -> Option<Vec<String>> {
98  let text = text.trim();
99  if text.contains(|c: char| c.is_whitespace() || "{};|/".contains(c)) {
100    return None;
101  }
102  let (list, domain) = text.split_once('@')?;
103  let domain_ok = domain.contains('.')
104    && !domain.starts_with(['.', '-'])
105    && !domain.ends_with(['.', '-'])
106    && domain
107      .chars()
108      .all(|c| c.is_ascii_alphanumeric() || ".-".contains(c));
109  let parts: Vec<&str> = list.split(',').collect();
110  if !domain_ok || parts.len() < 2 {
111    return None;
112  }
113  let mut owners: Vec<usize> = Vec::with_capacity(parts.len());
114  for part in &parts {
115    if part.chars().nth(1).is_none()
116      || part.starts_with('.')
117      || !part.chars().all(is_local_char)
118      || !part.chars().any(|c| c.is_ascii_alphabetic())
119    {
120      return None;
121    }
122    let one = owner(part)?;
123    if owners.contains(&one) {
124      return None;
125    }
126    owners.push(one);
127  }
128  Some(parts.iter().map(|part| s!("{part}@{domain}")).collect())
129}

184  /// [`NameKeys::of`] a name's printed `text` (`queue_reading::queued_author_name_keys`, the digested reading's words).
185  pub(in crate::base_utilities) fn of_printed(text: &str) -> NameKeys {
186    let base = resolver::base_letters(text);
187    let transliterated = resolver::base_letters(&resolver::transliterated(text));
188    let base_words = name_words(&base);
189    let classes: Vec<usize> = base_words.iter().map(|word| word.chars().count()).collect();
190    let mut keys = NameKeys::of_words(&base_words, &classes);
191    if transliterated != base {
192      // (each transliterated word in its base spelling's length class, the words one for one; else its own)
193      let words = name_words(&transliterated);
194      let own: Vec<usize> = words.iter().map(|word| word.chars().count()).collect();
195      let classes = if words.len() == classes.len() {
196        &classes
197      } else {
198        &own
199      };
200      let mut spelled = NameKeys::of_words(&words, classes);
201      spelled.joined.extend(hyphenated_surname(&transliterated));
202      keys.transliterated = Some(Box::new(spelled));
203    }
204    keys.joined.extend(hyphenated_surname(&base));
205    keys
206  }

208  /// The keys of a name as it prints, its letters unfolded, and of its transliteration (`Müller` "mueller", the
209  /// letters [`resolver::transliteration`] spells, the others as printed): whom the frontmatter observer's
210  /// `Resolver::respell_emails` moves an author's e-mail to, an address spelling that author as
211  /// printed — a transliterated spelling is one (user ruling 2026-11-05: `joerg.mueller@` is Jörg Müller's, "also
212  /// written Joerg Mueller"; `\email{mueller-group@x.de}` on Anna Smith's own call moves to `\author{J\"org M\"uller}`
213  /// as its ASCII twin does, FIDELITY_CHARTER §4 #420), a base-letter fold is not (`muller-group@` stays). Each word in
214  /// its printed length class ([`SpelledWord`]); a tie between a literal and a transliterated spelling is a tie
215  /// (`mueller@` beside Hans Mueller and Jörg Müller moves to neither, user ruling 2026-11-05). It expects the name's
216  /// digested tokens, as both respelling sites pass them: an undigested `Sch\"affler` reads through
217  /// [`visible_name_text`]'s accent fold as "Schaffler", whose base fold it then spells (review rev107a2 N3).
218  pub(in crate::base_utilities) fn as_printed(name: &Tokens) -> NameKeys {
219    let printed = name_words(&visible_name_text(name.unlist_ref()));
220    let classes: Vec<usize> = printed.iter().map(|word| word.chars().count()).collect();
221    let mut keys = NameKeys::of_words(&printed, &classes);
222    let transliterated = name_words(&resolver::transliterated(&spelling_name_text(
223      name.unlist_ref(),
224    )));
225    if transliterated != printed {
226      let own: Vec<usize> = transliterated
227        .iter()
228        .map(|word| word.chars().count())
229        .collect();
230      let classes = if transliterated.len() == classes.len() {
231        &classes
232      } else {
233        &own
234      };
235      keys.transliterated = Some(Box::new(NameKeys::of_words(&transliterated, classes)));
236    }
237    keys
238  }

245  /// The keys of a name's `words`, each in its length class (`classes`, its letters in the base spelling: [`SpelledWord`]).
246  fn of_words(words: &[String], classes: &[usize]) -> NameKeys {
247    let Some((first, later)) = words.split_first() else {
248      return NameKeys::default();
249    };
250    let class = |at: usize| classes.get(at).copied().unwrap_or_default();
251    let initial: String = first.chars().take(1).collect();
252    let mut joined: Vec<String> = Vec::new();
253    for (_, word) in later
254      .iter()
255      .enumerate()
256      .filter(|(at, _)| class(at + 1) >= 2)
257    {
258      joined.push(format!("{initial}{word}"));
259      if class(0) >= 2 {
260        joined.push(format!("{word}{first}"));
261        joined.push(format!("{first}{word}"));
262      }
263    }
264    // (the initials of every word before the last, then the last: `Kai-Wei Chang` "kwchang", 2604.01354)
265    if let Some((last, before)) = words.split_last()
266      && before.len() >= 2
267      && class(words.len() - 1) >= 2
268    {
269      let initials: String = before
270        .iter()
271        .filter_map(|word| word.chars().next())
272        .collect();
273      joined.push(format!("{initials}{last}"));
274    }
275    let initials: String = words
276      .iter()
277      .filter_map(|word| word.chars().next())
278      .collect();
279    NameKeys {
280      surnames: later
281        .iter()
282        .enumerate()
283        .filter(|(at, _)| class(at + 1) >= 3)
284        .map(|(at, word)| SpelledWord {
285          text:    word.clone(),
286          letters: class(at + 1),
287        })
288        .collect(),
289      // (a name particle another word of the name follows is no surname of its own: `Ana de Souza` spells no `de@`,
290      // review rev100b S3; the last word is the surname whatever it reads, `Thanh Le`'s `le@`, review rev100b2 S3)
291      short_surnames: later
292        .iter()
293        .enumerate()
294        .filter(|(at, word)| {
295          class(at + 1) == 2
296            && (*at + 1 == later.len()
297              || !matches!(word.as_str(), "de" | "la" | "da" | "le" | "di" | "du"))
298        })
299        .map(|(_, word)| word.clone())
300        .collect(),
301      given: (class(0) >= 3).then(|| SpelledWord {
302        text:    first.clone(),
303        letters: class(0),
304      }),
305      joined,
306      initials: [
307        Some(initials.clone()),
308        (words.len() >= 3).then(|| {
309          initials
310            .chars()
311            .take(1)
312            .chain(initials.chars().last())
313            .collect()
314        }),
315      ]
316      .into_iter()
317      .flatten()
318      .filter(|initials: &String| initials.chars().count() >= 2)
319      .collect(),
320      transliterated: None,
321    }
322  }
323}

325/// A name's words, lowercased: its runs of letters, an apostrophe between two letters inside its word (`D'Amico`
326/// "damico", whose address `damicos@` holds it whole, 2606.17317; `Gol'dshtein` "goldshtein" — brief 107f, where a
327/// surname spells only a run of the local part's words).
328/// A hyphen-linked surname with its hyphen dropped, a spelling as printed (`L. Tal-Or` "talor": `levtalor@`, 2112.09029,
329/// review rev107g), as a hyphenated given name's already is (`Wang-Chiew` "wangchiew"). The name's last word only, its
330/// parts all letters: spaced words stay apart (`de Vries`). A joined key, so where it ends a word it does not open it is
331/// a weak spelling ([`LocalPart::joined_standing`]) and owns only an address no other name spells.
332fn hyphenated_surname(text: &str) -> Option<String> {
333  let last = text.split_whitespace().last()?;
334  let parts: Vec<&str> = last.split(['-', '\u{2010}', '\u{2011}']).collect();
335  (parts.len() >= 2
336    && parts
337      .iter()
338      .all(|part| !part.is_empty() && part.chars().all(char::is_alphabetic)))
339  .then(|| parts.concat().to_lowercase())
340}
341
342fn name_words(text: &str) -> Vec<String> {
343  let chars: Vec<char> = text.chars().collect();
344  let text: String = chars
345    .iter()
346    .enumerate()
347    .filter(|&(at, c)| {
348      !(matches!(c, '\'' | '\u{2019}' | '\u{02bc}')
349        && at > 0
350        && chars[at - 1].is_alphabetic()
351        && chars.get(at + 1).is_some_and(|next| next.is_alphabetic()))
352    })
353    .map(|(_, c)| *c)
354    .collect();
355  text
356    .split(|c: char| !c.is_alphabetic())
357    .filter(|word| !word.is_empty())
358    .map(str::to_lowercase)
359    .collect()
360}

362/// An address's letters and other characters, as [`spelled_owners`] reads them.
363pub(in crate::base_utilities) fn address_text(address: &Tokens) -> String {
364  // (a colour, a font, a scale is no letter of it, [`without_non_printing_arguments`]: 2604.19440, 2609.25669)
365  let printed = without_non_printing_arguments(address.unlist_ref());
366  let v = printed.as_slice();
367  // (an escaped underscore prints one, and parts the local part's words: `ricky\_ang@`, 2604.02636 — [`spelled_owners`])
368  let text = |v: &[Token]| -> String {
369    v.iter()
370      .filter_map(|t| match t.code {
371        Catcode::LETTER | Catcode::OTHER => Some(t.to_string()),
372        _ if *t == T_CS!("\\_") || *t == T_CS!("\\textunderscore") => Some("_".to_string()),
373        _ => None,
374      })
375      .collect()
376  };
377  // `\href{mailto:t}{p}`: the address it prints, not its target and the printed one run together; its target's, past
378  // `mailto:`, when it prints none (a placeholder target, `\href{mailto:email@domain}{yuxi.xia@univie.ac.at}`,
379  // 2601.07974)
380  if let Some(at) = v.iter().position(|t| *t == T_CS!("\\href")) {
381    let open = at + 1 + v[at + 1..].iter().take_while(|t| **t == T_SPACE!()).count();
382    let close = skip_group(v, open).min(v.len());
383    let target = text(&v[open..close]);
384    let printed = text(&v[..at]) + &text(&v[close..]);
385    return if printed.contains('@') {
386      printed
387    } else {
388      target.trim_start_matches("mailto:").to_string()
389    };
390  }
391  text(v)
392}

429/// An address's local part as [`spelled_owners`] reads it: its letters, lowercased, and where its words meet (`bounds`,
430/// byte offsets into `letters`): its separators — `.` `_` `-`, a digit, any other character —, its start and its end.
431struct LocalPart {
432  letters: String,
433  bounds:  Vec<usize>,
434}
435
436impl LocalPart {
437  fn of(address: &str) -> Option<LocalPart> {
438    let raw = address.split('@').next()?;
439    let mut letters = String::new();
440    let mut bounds: Vec<usize> = vec![0];
441    for c in raw.chars() {
442      if c.is_alphabetic() {
443        letters.extend(c.to_lowercase());
444      } else if bounds.last() != Some(&letters.len()) {
445        bounds.push(letters.len());
446      }
447    }
448    if bounds.last() != Some(&letters.len()) {
449      bounds.push(letters.len());
450    }
451    Some(LocalPart { letters, bounds })
452  }
453
454  /// Every place `key` stands in the letters, overlapping ones too.
455  fn places(&self, key: &str) -> Vec<usize> {
456    if key.is_empty() {
457      return Vec::new();
458    }
459    (0..self.letters.len())
460      .filter(|&at| self.letters.is_char_boundary(at) && self.letters[at..].starts_with(key))
461      .collect()
462  }
463
464  /// The word holding the letters at `start..end`, or `None` when they run across a separator.
465  fn word_around(&self, start: usize, end: usize) -> Option<(usize, usize)> {
466    if self.bounds.iter().any(|&b| start < b && b < end) {
467      return None;
468    }
469    let before = self.bounds.iter().copied().filter(|&b| b <= start).max()?;
470    let after = self.bounds.iter().copied().filter(|&b| b >= end).min()?;
471    Some((before, after))
472  }

474  /// THE boundary predicate (brief 107f r2, on review rev107f's Q1): how a name's key at `start..end` stands. At full
475  /// strength when it is a whole word, or a word's prefix or suffix (`ricky_ang@` Lay Kee Ang's "ang", `yang91@`,
476  /// `kee.ang@`), or strictly inside a word between the name's own initials (`own`), at most two letters on each side
477  /// (`rjosem@` José M. Rodríguez-Nogales', 1809.06911; `kmfonsecar@` K. M. Fonseca Romero's, 1202.4210); weakly when a
478  /// key of [`WEAK_KEY_LETTERS`] letters or more stands there after two letters at most, whatever follows it (`vemarinc@`
479  /// Víctor Marín's, 2311.18024; `tpengqy@` Q. Y. Peng's, 2004.06253; `eetanggz@` Guozhi Tang's, 2106.10681;
480  /// `ylwanguestc@` Yinglong Wang's, 2208.05114); else not at all — a shorter key between other letters is a word's
481  /// letters (`yvanwy@` no "van", 2202.02002; `byang@` no "yan", 2602.00582; `bangkok@` no "ang"), and no key runs across
482  /// a separator (`ricky_ang@` is no Shengyuan A. Yang's "yang", 2604.02636). The sides are counted in characters (review
483  /// rev107f finding 5).
484  fn standing(&self, start: usize, end: usize, own: &str) -> Option<Standing> {
485    let (before, after) = self.word_around(start, end)?;
486    if start == before || end == after {
487      return Some(Standing::Full);
488    }
489    let (lead, trail) = (&self.letters[before..start], &self.letters[end..after]);
490    if lead.chars().count() > 2 {
491      return None;
492    }
493    let initials = |letters: &str| letters.chars().all(|c| own.contains(c));
494    if trail.chars().count() <= 2 && initials(lead) && initials(trail) {
495      Some(Standing::Full)
496    } else if self.letters[start..end].chars().count() >= WEAK_KEY_LETTERS {
497      Some(Standing::Weak)
498    } else {
499      None
500    }
501  }
502
503  /// How a joined spelling (`NameKeys::joined`) at `start..end` stands: at full strength opening a word, its separators
504  /// read through (`jhuang@`, `j.huang@`, `nlp.wangyi@`); weakly ending a word it does not open, at
505  /// [`WEAK_JOINED_LETTERS`] letters or more (`sechenqi@` Qi Chen's "chenqi", 2003.00397; `xjhuang@`'s "jhuang", which is
506  /// no tie-break between Jixuan Huang and Xuanjing Huang, 2510.24320 — review rev107f B1; `pyli@` no Yanan Li's "yli",
507  /// 2105.07637); else not at all.
508  fn joined_standing(&self, start: usize, end: usize) -> Option<Standing> {
509    if self.bounds.contains(&start) {
510      Some(Standing::Full)
511    } else if self.bounds.contains(&end)
512      && self.letters[start..end].chars().count() >= WEAK_JOINED_LETTERS
513    {
514      Some(Standing::Weak)
515    } else {
516      None
517    }
518  }
519}

521/// A name's score for the local part `part`, and its surname part ([`spelled_owners`], which reads the scores): with its
522/// weak spellings or without (`weak`, [`Standing`]).
523fn spelling_score(keys: &NameKeys, part: &LocalPart, weak: bool) -> (usize, usize) {
524  let local = part.letters.as_str();
525  let counts = |standing: Option<Standing>| standing.is_some_and(|s| weak || s == Standing::Full);
526  // (a surname's first six letters only outside the letters its given name matched (`given`, their place in the local
527  // part): `andrea.rossi@` is Andrea Rossi's, not Andrea Andreassen's "andrea" counted twice, `soeren.hansen@` Søren
528  // Hansen's, not Søren Sørensen's — review rev107a finding 2). A truncated surname stands as a whole one does: the
529  // longest of the surname's first letters standing there, six or more (`mkaufman@` for Kaufmann, `leyaouan@` for Le
530  // Yaouanc)
531  let surname_score = |key: &SpelledWord, given: Option<(usize, usize)>, own: &str| {
532    let len = key.text.len();
533    if part
534      .places(&key.text)
535      .into_iter()
536      .any(|at| counts(part.standing(at, at + len, own)))
537    {
538      return key.text.chars().count();
539    }
540    if key.letters < 6 {
541      return 0;
542    }
543    let prefix: String = key.text.chars().take(6).collect();
544    let outside = |at: usize| match given {
545      Some((start, end)) => at + prefix.len() <= start || at >= end,
546      None => true,
547    };
548    let truncated = part.places(&prefix).into_iter().any(|at| {
549      let held = local[at..]
550        .chars()
551        .zip(key.text.chars())
552        .take_while(|(a, b)| a == b)
553        .map(|(a, _)| a.len_utf8())
554        .sum::<usize>();
555      outside(at) && counts(part.standing(at, at + held, own))
556    });
557    if truncated { 6 } else { 0 }
558  };
559  // (where the given name stands in the local part: opening it, or anywhere when it has four letters or more)
560  let given_place = |given: &SpelledWord, own: &str| {
561    let len = given.text.len();
562    let fits = |at: usize| counts(part.standing(at, at + len, own));
563    if local.starts_with(given.text.as_str()) && fits(0) {
564      Some((0, len))
565    } else if given.letters >= 4 {
566      part
567        .places(&given.text)
568        .into_iter()
569        .find(|&at| fits(at))
570        .map(|at| (at, at + len))
571    } else {
572      None
573    }
574  };
575  // (a joined spelling standing in the local part, its length; the initials, the local part's letters alone)
576  // (or its first six letters or more, as a surname's: a truncated family-first spelling, `zhuzhenh22@` for Zhenhao
577  // Zhu, `chenyuxi22@` for Yuxiang Chen, 2602.03328, 2609.11716; the r7 re-review's finding 9 — scored the letters it
578  // holds less one, below a spelling whole of as many: `wangyu@` is Yu Wang's "wangyu", not Yuxin Wang's "wangyuxin",
579  // `liyang@` Yang Li's, not Yangfan Li's; the r8 re-review's finding 3 — where the local part ends there; one going on
580  // past a shared start scores its six letters, as before)
581  let joined_score = |keys: &NameKeys| {
582    keys
583      .joined
584      .iter()
585      .filter_map(|key| {
586        if part
587          .places(key)
588          .into_iter()
589          .any(|at| counts(part.joined_standing(at, at + key.len())))
590        {
591          Some(key.chars().count())
592        } else {
593          let held = local
594            .chars()
595            .zip(key.chars())
596            .take_while(|(a, b)| a == b)
597            .count();
598          // (a truncation only when the local part's letters are used up, else its first six letters: `zhangweiwei@`
599          // parts from "zhangweiwen" and is Wei Zhang's "zhangwei", the 77y review's finding 5, probe n1)
600          if held == local.chars().count() {
601            (held >= 6).then(|| held - 1)
602          } else {
603            (held >= 6).then_some(6)
604          }
605        }
606      })
607      .max()
608      .unwrap_or(0)
609      + keys
610        .initials
611        .iter()
612        .find(|initials| **initials == local)
613        .map_or(0, |initials| initials.chars().count())
614  };
615  // (the score, then its surname part: a tie goes to the surname, `chen@…` Wei Chen's before Chen Li's). A name scores
616  // the better of its words spelled apart and a joined spelling, never both added: one spelling of the letters is
617  // counted once (`khan.muhammad@ieee.org` scores 12 for Khan Muhammad, "muhammad" and "khan" apart, and 12 for
618  // Muhammad Haris Khan, "khanmuhammad" joined, and goes to the surname's spelling, 2602.07835).
619  // (a name's score the better of its spellings', its base letters' and its transliterated one's: `mo.schaeffler@`
620  // Moritz Schäffler's, html_feedback #7233. A tie between a name spelled literally and one spelled through the
621  // transliteration is a tie — an address carries no umlaut, so `mueller@` spells Hans Mueller and Jörg Müller alike
622  // and is neither's by its spelling (user ruling 2026-11-05, on review rev107a finding 1); the rest of the placement
623  // order decides it)
624  keys
625    .spellings()
626    .map(|keys| {
627      // (the initials of the name's words, which a part of it inside a word may stand between: `standing`)
628      let own = keys.initials.first().map_or("", String::as_str);
629      let given = keys
630        .given
631        .as_ref()
632        .and_then(|given| Some((given, given_place(given, own)?)));
633      let place = given.map(|(_, place)| place);
634      let surname: usize = keys
635        .surnames
636        .iter()
637        .map(|key| surname_score(key, place, own))
638        .sum::<usize>()
639        + keys
640          .short_surnames
641          .iter()
642          .find(|short| **short == local)
643          .map_or(0, |short| short.chars().count());
644      let given = given.map_or(0, |(given, _)| given.text.chars().count());
645      ((surname + given).max(joined_score(keys)), surname)
646    })
647    .max()
648    .unwrap_or((0, 0))
649}

651/// The names (indices into `name_keys`) the local part of `address` spells best: each name scores the letters of its
652/// keys the local part holds — a surname whole, or its first six letters (`leyaouan@…` for "A.~Le~Yaouanc"); the given
653/// name it opens with, or holds when the given name has four letters or more (2603.24730's
654/// `\{yuqihu,~ahna,~vasha\}@berkeley.edu` for Yuqi Hu, Ahna R. Girshick, Vasha DuTell; 2603.20851's
655/// `nlp2ct.shanshan@…` for Shanshan Wang, `lidiasc@…` for Lidia S. Chao), or a joined spelling it opens with when
656/// that scores better (the better of the two, never their sum: 2602.07835) — and the highest scores own it: a longer
657/// spelling outranks a key it holds (`fucheng.fuc@…` is Cheng Fu's, not Jiawei Chen's, 2403.00801; `wangchiew@…`
658/// Wang-Chiew Tan's, not Xiaolan Wang's, 2104.01371; `yipengzhang97@…` Yipeng Zhang's, not Xiaoying Zhang's,
659/// 2406.06326), a tie going to the surname's spelling. No name when none scores.
660///
661/// Each key counts only where the one boundary predicate places it ([`LocalPart::standing`], [`LocalPart::joined_standing`];
662/// census138 EMAILSPELL, OXIDIZED_DESIGN_DIVERGENCES #970 "107f"). A weak spelling owns an address only no other name
663/// spells; where another does, the names are ranked by their full-strength spellings, and a weak one never decides
664/// against a full one: the two rankings agreeing, their owner; else every name either ranks first — a tie, which the rest
665/// of the placement order (the list's order, the elimination, the block) decides. All of them weak, the one ranking
666/// decides. Nothing is guessed (FIDELITY_CHARTER §4): `xjhuang@`
667/// beside Jixuan Huang and Xuanjing Huang is a tie (each "huang" ending the word; Jixuan's "jhuang" weak, review rev107f
668/// B1, 2510.24320), and so is `xywang@` beside Yi Wang and Xiaoyu Wang, which the elimination gives to Xiaoyu Wang once
669/// `yiwang@` is Yi Wang's.
670pub(in crate::base_utilities) fn spelled_owners(
671  name_keys: &[NameKeys],
672  address: &str,
673) -> Vec<usize> {
674  let Some(part) = LocalPart::of(address) else {
675    return Vec::new();
676  };
677  let scores = |weak: bool| -> Vec<(usize, usize)> {
678    name_keys
679      .iter()
680      .map(|keys| spelling_score(keys, &part, weak))
681      .collect()
682  };
683  let all = scores(true);
684  let spelling: Vec<usize> = (0..all.len()).filter(|&at| all[at].0 > 0).collect();
685  if spelling.len() < 2 {
686    return spelling;
687  }
688  // (several names spell it: ranked with their weak spellings and without; where a weak one would decide against a full
689  // one, a tie. All of them weak, the longer spelling ranks first, with no tie-break by the surname: `sechenqi@` Qi Chen's
690  // "chenqi" beside Wei Chen's "chen"; `kkwongar@` no one's beside Hiu Tung Wong and Wong Kam-Kwai, 2608.26715)
691  let first = |scores: &[(usize, usize)]| -> Vec<usize> {
692    let best = scores.iter().copied().max().unwrap_or((0, 0));
693    (0..scores.len()).filter(|&at| scores[at] == best).collect()
694  };
695  let full = scores(false);
696  if full.iter().all(|score| score.0 == 0) {
697    let best = all.iter().map(|score| score.0).max().unwrap_or(0);
698    return (0..all.len()).filter(|&at| all[at].0 == best).collect();
699  }
700  let mut owners = first(&all);
701  for at in first(&full) {
702    if !owners.contains(&at) {
703      owners.push(at);
704    }
705  }
706  owners.sort_unstable();
707  owners
708}

758/// 2602.23777's `dongsli@` on the next line). The run between two spelled owners is read by the order alone, every

766/// the order) — the rest stay `None`, as does a role mailbox (RFC 2142's `info@`, `support@`, …, [`is_role_mailbox`]),
767/// which is no person's: the block's, on its shared creator (user ruling R10 2026-10-20).
768pub(in crate::base_utilities) fn owners_by_elimination(
769  name_keys: &[NameKeys],
770  addresses: &[String],
771  taken: &[bool],
772  elsewhere_open: bool,
773) -> Vec<Option<usize>> {
774  let mut owners: Vec<Option<usize>> = addresses
775    .iter()
776    .map(|address| spelled_owner(name_keys, address))
777    .collect();
778  let free = |name: usize| !taken.get(name).copied().unwrap_or(false);
779  let left: Vec<usize> = (0..name_keys.len())
780    .filter(|name| !owners.contains(&Some(*name)) && free(*name))
781    .collect();
782  let open: Vec<usize> = (0..owners.len())
783    .filter(|at| owners[*at].is_none() && !is_role_mailbox(&addresses[*at]))
784    .collect();
785  if let ([at], [name]) = (open.as_slice(), left.as_slice())
786    && !elsewhere_open
787  {
788    owners[*at] = Some(*name);
789    return owners;
790  }
791  let spelled: Vec<usize> = owners.iter().flatten().copied().collect();
792  if open.is_empty() || spelled.is_empty() || !spelled.windows(2).all(|pair| pair[0] < pair[1]) {
793    return owners;
794  }
795  // (as many left as there are names left: in order, FIDELITY_CHARTER §4's condition, so only when the order holds with
796  // the spelled addresses. It fails on 2606.04646, whose position read swapped `dishkw@` and `e0556732@`, and on 2605.26032,
797  // `congyued@` before `billf@` with Deng after Freeman, where it placed `jzxchen@` and `billf@` right by luck; those and
798  // 2102.08716's list stay on the block, rev108a3 B3)
799  if open.len() == left.len() && !elsewhere_open {
800    let mut placed = owners.clone();
801    for (at, name) in open.iter().zip(&left) {
802      placed[*at] = Some(*name);
803    }
804    let in_order: Vec<usize> = placed.iter().flatten().copied().collect();
805    if in_order.windows(2).all(|pair| pair[0] < pair[1]) {
806      return placed;
807    }
808  }
809  // (each run of open addresses, with the owners of the spelled addresses around it, and every name between them)
810  let mut run: Vec<usize> = Vec::new();
811  let mut below: Option<usize> = None;
812  for at in 0..=owners.len() {
813    let bound = if at == owners.len() {
814      Some(name_keys.len())
815    } else {
816      owners[at]
817    };
818    match bound {
819      Some(above) => {
820        let between = below.map_or(0, |below| below + 1)..above;
821        // (a run before the first spelled owner or after the last is bounded by the list's end alone, which no spelled
822        // address confirms: no position placement at a list's end (user ruling R10 2026-10-20; 2509.10377's `12421181@`
823        // first in its list, the r5 re-review's q7 `z99@`); a run between two spelled owners is read by the order,
824        // 2303.11225)
825        let at_an_end = below.is_none() || at == owners.len();
826        if !run.is_empty() && between.len() == run.len() && between.clone().all(free) && !at_an_end
827        {
828          for (address, name) in run.iter().zip(between) {
829            owners[*address] = Some(name);
830          }
831        }
832        run.clear();
833        below = Some(above);
834      },
835      None if open.contains(&at) => run.push(at),
836      None => {},
837    }
838  }
839  owners
840}

842/// The owners FIDELITY_CHARTER §4's chain gives the e-mail addresses printed inside an affiliation-kind entry (an
843/// affiliation, alternate affiliation or address: its lines of addresses, its e-mail calls), the one Link step both
844/// engines run — the author block's at the flush (`insert::link_entry_addresses`) and the frontmatter observer's
845/// reading (`Resolver::take_out_entry_addresses`, `Resolver::answer_inner_emails`). `spelled` is each address of the
846/// entry's the author it spells (any live author: [`spelled_owner`]), `run` the addresses read here (one run of them,
847/// in print order), `carriers` the authors holding the entry, in order, and `symbol_line` whether a footnote symbol
848/// leads the entry, `addressed` the carriers holding an address of their own beside it, and `sole_author` whether the
849/// document has one author, whose every address is (the charter's "the group's only author"). Each address of `run`, in

857/// the charter's step 4 wins over the "stays in the entry" fail-safe of OD #938; astro-ph0512555's `Email:dzg@,
858/// xywang@, hyf@` under four carriers of entry 2). Surname-first initials (`dzg` for Z. G. Dai) are not read: a false
859/// match is worse than the block.
860pub fn entry_address_owners(
861  spelled: &[Option<usize>],
862  run: std::ops::Range<usize>,
863  carriers: &[usize],
864  symbol_line: bool,
865  addressed: &[usize],
866  sole_author: bool,
867  facts: &EntryOwnerFacts,
868) -> Vec<(Option<Vec<usize>>, &'static str)> {
869  // (the only carrier's only where the entry prints that one address no one spells and the carrier holds none of its own:
870  // several unspelled under one name are no one name's — 2403.17706's `1022041122@, 1223045440@, hhp@` under Haiping
871  // Huang's second institute, 2506.06908's welded entry under Chenzhi Xu —, nor is one under a name already holding its
872  // own — 2411.17073's `bull@` under Awais Naeem, `awais.naeem@` his)
873  let unspelled = spelled.iter().filter(|s| s.is_none()).count();
874  let in_order = run.len() == carriers.len()
875    && carriers.len() >= 2
876    && facts.spellers[run.clone()]
877      .iter()
878      .zip(carriers)
879      .all(|(names, c)| names.len() < 2 || names.contains(c))
880    && spelled[run.clone()]
881      .iter()
882      .zip(carriers)
883      .all(|(s, c)| s.is_none_or(|s| s == *c));
884  // (the symbol carriers no other address of the entry spells, as the only-carrier step excludes them: 1907.01388's
885  // `* chinarentao@163.com, zhutou@ustc.edu` under Tao Ren* and Tao Zhou* gives `zhutou@` to Zhou alone, review rev100b
886  // S1; none left, the order and the block decide)
887  let free: Vec<usize> = carriers
888    .iter()
889    .copied()
890    .filter(|c| !spelled.contains(&Some(*c)))
891    .collect();
892  let mut owners: Vec<(Option<Vec<usize>>, &'static str)> = run
893    .clone()
894    .map(|at| match (spelled[at], carriers) {
895      (Some(owner), _) => (Some(vec![owner]), "owner:spelled"),
896      (None, [only])
897        if sole_author
898          || (!spelled.contains(&Some(*only)) && unspelled == 1 && !addressed.contains(only)) =>
899      {
900        (Some(vec![*only]), "owner:only-carrier")
901      },
902      (None, [] | [_]) => (None, "owner:block"),
903      (None, _) if symbol_line && !free.is_empty() => (Some(free.clone()), "owner:carriers"),
904      (None, _) if in_order => (Some(vec![carriers[at - run.start]]), "owner:order"),
905      (None, _) => (None, "owner:block"),
906    })
907    .collect();
908  // (the one address left on the block of several carriers: the carrier marked the corresponding author's, when they own no
909  // other address of the run — the old split's row e-mail owners, `row_email_calls`, deleted at P27b)
910  let left: Vec<usize> = (0..owners.len())
911    .filter(|&at| owners[at].0.is_none())
912    .collect();
913  if let ([at], Some(author)) = (left.as_slice(), facts.corresponding)
914    && !facts.no_ones[run.start + *at]
915    && carriers.len() > 1
916    && carriers.contains(&author)
917    && owners
918      .iter()
919      .all(|(to, _)| to.as_ref().is_none_or(|to| !to.contains(&author)))
920  {
921    owners[*at] = (Some(vec![author]), "owner:corresponding");
922  }
923  owners
924}

944/// The owners of the addresses of a `{…}@domain` list set over several lines, which the Kind step for an address line
945/// took out of an affiliation-kind entry (`insert::link_entry_addresses`, the list's `LINE_LIST`), placed as the author
946/// block's reader places the list on one line (FIDELITY_CHARTER §4's order, review rev107a finding 4; html_feedback
947/// #7233): `name_keys` every author's, in order; `addresses` the list's, in print order; `spelled` each one's owner as the
948/// Link step reads it ([`spelled_owner`], an entry's word aside: [`is_entry_word_mailbox`]); `taken` the authors an
949/// address outside the list spells or who hold one; `spelled_elsewhere` whether an address outside the list spells an
950/// author; `elsewhere_open` whether one outside it spells no one and is placed on no author. Each address: the author it
951/// spells (`owner:spelled`); else, the list one address per author that every spelled one agrees with and no address
952/// outside it spells an author, the author at its place (`owner:order`, OXIDIZED_DESIGN_DIVERGENCES #805 — the
953/// synthetic `email_list_split_order_only`); else the only author's, where there is one (`owner:only-author`, the
954/// charter's "the group's only author", as [`entry_address_owners`]' `sole_author` — 2311.03226); else by elimination
955/// (`owner:elimination`, batch 77p r4, [`owners_by_elimination`] — 1808.00588's `kdm@`; 2305.13191's `wshui@`, `drot@`,
956/// `ballemig@`, 9 entries for 10 names); else `None`, the block's
957/// (`owner:block`, OD #159's block creator).
958pub(in crate::base_utilities) fn list_owners(
959  name_keys: &[NameKeys],
960  addresses: &[String],
961  spelled: &[Option<usize>],
962  taken: &[bool],
963  spelled_elsewhere: bool,
964  elsewhere_open: bool,
965) -> Vec<(Option<usize>, &'static str)> {
966  let in_order = addresses.len() == name_keys.len()
967    && name_keys.len() >= 2
968    && !spelled_elsewhere
969    && spelled
970      .iter()
971      .enumerate()
972      .all(|(at, owner)| owner.is_none_or(|owner| owner == at));
973  if in_order {
974    return spelled
975      .iter()
976      .enumerate()
977      .map(|(at, owner)| match owner {
978        Some(owner) => (Some(*owner), "owner:spelled"),
979        None => (Some(at), "owner:order"),
980      })
981      .collect();
982  }
983  if let [_] = name_keys {
984    return spelled
985      .iter()
986      .map(|owner| match owner {
987        Some(owner) => (Some(*owner), "owner:spelled"),
988        None => (Some(0), "owner:only-author"),
989      })
990      .collect();
991  }
992  owners_by_elimination(name_keys, addresses, taken, elsewhere_open)
993    .into_iter()
994    .zip(spelled)
995    .zip(addresses)
996    .map(|((owner, spelled), address)| match (spelled, owner) {
997      (Some(owner), _) => (Some(*owner), "owner:spelled"),
998      // (an address the elimination reads as spelled that the Link step does not — an entry's word — is the block's)
999      (None, Some(owner)) if spelled_owner(name_keys, address).is_none() => {
1000        (Some(owner), "owner:elimination")
1001      },
1002      _ => (None, "owner:block"),
1003    })
1004    .collect()
1005}

1022/// A mark's operand without its footnote symbols (`is_symbol_mark`), comma-separated pieces kept apart: `1,\dagger` is
1023/// `1`, a number or letter glued to a symbol (`1\ast`, `1{\dagger}`: `split_glued_symbol`) its label; `None` when only
1024/// symbols are left (`*`, `\dagger`).
1025pub(in crate::base_utilities) fn label_pieces(operand: &[Token]) -> Option<Vec<Token>> {
1026  let mut pieces: Vec<&[Token]> = Vec::new();
1027  let (mut start, mut i) = (0, 0);
1028  while i < operand.len() {
1029    if operand[i].get_catcode() == Catcode::BEGIN {
1030      i = skip_group(operand, i);
1031      continue;
1032    }
1033    if operand[i] == T_OTHER!(",") {
1034      pieces.push(&operand[start..i]);
1035      start = i + 1;
1036    }
1037    i += 1;
1038  }
1039  pieces.push(&operand[start.min(operand.len())..]);
1040  let mut kept: Vec<Token> = Vec::with_capacity(operand.len());
1041  for piece in pieces {
1042    let piece = {
1043      let lead = piece.iter().take_while(|t| **t == T_SPACE!()).count();
1044      let trail = piece[lead..]
1045        .iter()
1046        .rev()
1047        .take_while(|t| **t == T_SPACE!())
1048        .count();
1049      &piece[lead..piece.len() - trail]
1050    };
1051    let unbraced = match piece {
1052      [open, inner @ .., close]
1053        if open.get_catcode() == Catcode::BEGIN
1054          && close.get_catcode() == Catcode::END
1055          && skip_group(piece, 0) == piece.len() =>
1056      {
1057        inner
1058      },
1059      _ => piece,
1060    };
1061    let label = if unbraced.is_empty() || is_symbol_mark(unbraced) {
1062      continue;
1063    } else if let Some((label, _)) = split_glued_symbol(piece) {
1064      label
1065    } else {
1066      piece.to_vec()
1067    };
1068    if !kept.is_empty() {
1069      kept.push(T_OTHER!(","));
1070    }
1071    kept.extend(label);
1072  }
1073  (!kept.is_empty()).then_some(kept)
1074}
