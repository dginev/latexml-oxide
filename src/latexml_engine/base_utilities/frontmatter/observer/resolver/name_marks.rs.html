1//! The reading's record of each mark a name prints (ADVARCH's `NameMark`, MARKKEY R2a; OXIDIZED_DESIGN_DIVERGENCES
2//! #1062): one per label a mark requests, made where the names line reads the mark (`lines.rs`), with what keys it, what
3//! printed it and how it printed. A creator's requests, its class's requests, its footnote marks, its parted labels and
4//! the marks its notes print are views of it ([`RCreator::requests`] and the rest), never parallel stores reconciled by
5//! count.
6
7use super::*;
8
9/// What printed a name's mark ([`NameMark::by`]).
10#[derive(Debug, Clone, Copy, PartialEq, Eq)]
11pub enum MarkBy {
12  /// typed on the name: `$^{1}$`, `\textsuperscript{a}`, a paper's mark macro
13  Hand,
14  /// the class's own request command: llncs's and aa's `\inst`, a binding's `\lx@request@frontmatter@annotation`, a
15  /// block's `annotations=`
16  Class,
17  /// a `\footnotemark`, keyed by its number ([`NameMark::number`])
18  Footnote,
19  /// a note's own mark on the name it is beside (`\thanks`, `\footnote`): no request, its note answers it (an entry led
20  /// by the same `\footnotemark` is that note's print: `answer_pending`, 1607.07280)
21  Note,
22  /// no mark printed: a creator call's own label (`\lx@add@creator[annotations={ed}]`)
23  Call,
24  /// authblk's `\AB@thenote` number, printed by the class once every call is read (`Resolver::thenote_finish`)
25  TheNote,
26}
27
28/// Whether a mark's glyph is inked in the source or implied by a counter ([`NameMark::origin`]; DG 2026-11-16: an inked
29/// mark keeps its glyph, an implied one never reuses a glyph inked elsewhere).
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub enum MarkOrigin {
32  Inked,
33  Implied,
34}
35
36/// One mark printed on a name, as the reading reads it.
37#[derive(Debug, Clone, PartialEq, Eq)]
38pub struct NameMark {
39  /// the label it requests, its key (`1`, `affiliation:1`, `*`, `fn:c1`)
40  pub label:      String,
41  /// a footnote mark's number, the key TeX pairs it by (`Event::number`, OD #1058)
42  pub number:     Option<i64>,
43  /// the event that printed it
44  pub span:       Option<(u16, u16)>,
45  pub by:         MarkBy,
46  /// the glued mark of letters and digits it was parted from, as printed (`part_glued_labels`): a piece no entry
47  /// answers shows the whole — BESIII's `$^{50A,50C}$` "50A,50C" beside its entry 50, never "A,C" (1705.11109,
48  /// 1507.02068, 2204.09614)
49  pub whole:      Option<String>,
50  /// read apart from the mark it was glued to (ruling 1: `\inst{1, 2\footnotemark[1]}`, 2404.09857; `\inst{1*}`,
51  /// 2506.20360)
52  pub parted:     bool,
53  /// a comma printed between it and the mark before it in its run (`$^{1,*}$`; a formula's comma,
54  /// `\endnotemark[1]$^,$\endnotemark[2]`, 1703.08733), or its class prints a name's marks as one list
55  /// ([`MARKS_LISTED`]: elsarticle, 2605.13555)
56  pub sep_before: bool,
57  /// printed on top of the zero-width title mark before it, of its glyph (`mark_ledger::overprints`: 2012.08663,
58  /// 2103.01544): TeX shows the glyph once. Unset until MARKKEY R2b, whose display fills `MarkLedger::overprints`
59  /// (always false now)
60  pub overprint:  bool,
61}
62
63impl NameMark {
64  /// A mark of `by` requesting `label`, printed by the event at `span`.
65  pub(super) fn new(label: String, by: MarkBy, span: Option<(u16, u16)>) -> NameMark {
66    NameMark {
67      label,
68      number: None,
69      span,
70      by,
71      whole: None,
72      parted: false,
73      sep_before: false,
74      overprint: false,
75    }
76  }
77
78  /// Whether its glyph is inked in the source (typed, or a class's command printing its operand) or implied by a
79  /// counter (a `\footnotemark`, a note's mark) — the bit DG's inked/implied rule reads.
80  pub fn origin(&self) -> MarkOrigin {
81    match self.by {
82      MarkBy::Hand | MarkBy::Class => MarkOrigin::Inked,
83      MarkBy::Footnote | MarkBy::Note | MarkBy::Call | MarkBy::TheNote => MarkOrigin::Implied,
84    }
85  }
86
87  /// Whether it asks for an entry: every mark but a note's own.
88  pub fn requests(&self) -> bool { self.by != MarkBy::Note }
89}
90
91impl RCreator {
92  /// The labels its marks request, in print order (ruling 1, the marks shown on a name are its requests).
93  pub fn requests(&self) -> impl Iterator<Item = &String> + '_ {
94    self.marks.iter().filter(|m| m.requests()).map(|m| &m.label)
95  }
96
97  /// Whether it requests `label` exactly.
98  pub fn requests_label(&self, label: &str) -> bool { self.requests().any(|r| r == label) }
99
100  /// Whether it requests nothing.
101  pub fn requests_nothing(&self) -> bool { !self.marks.iter().any(NameMark::requests) }
102
103  /// The labels its class's own request command made ([`MarkBy::Class`]): a `_share=authors` entry only the class's
104  /// `\inst{n}` picks; the emitter's kept `\inst` marks.
105  pub fn class_requests(&self) -> impl Iterator<Item = &String> + '_ {
106    self
107      .marks
108      .iter()
109      .filter(|m| m.by == MarkBy::Class)
110      .map(|m| &m.label)
111  }
112
113  /// Whether a mark other than a `\footnotemark` requests what `label` names ([`label_matches`]): a numbered
114  /// affiliation answers no request a `\footnotemark` made, its number being its footnote's (2404.09857).
115  pub fn asks_not_by_footnote(&self, label: &str) -> bool {
116    self
117      .marks
118      .iter()
119      .any(|m| m.requests() && m.by != MarkBy::Footnote && label_matches(&m.label, label))
120  }
121
122  /// Whether authblk's `\AB@thenote` number it shows is one of `labels` (ruling 1 on that mark).
123  pub fn thenote_requests(&self, labels: &[String]) -> bool {
124    self
125      .marks
126      .iter()
127      .any(|m| m.by == MarkBy::TheNote && labels.contains(&m.label))
128  }
129
130  /// Whether a mark it was parted into requests a label `matches` accepts (ruling 1 on that mark).
131  pub fn parted(&self, matches: impl Fn(&str) -> bool) -> bool {
132    self.marks.iter().any(|m| m.parted && matches(&m.label))
133  }
134}

136impl Resolver {
137  /// The marks a `\footnotemark` printed on a name, each with its creator, in creator order.
138  pub(super) fn footnote_marks(&self) -> impl Iterator<Item = (usize, &NameMark)> + '_ {
139    self
140      .out
141      .creators
142      .iter()
143      .enumerate()
144      .flat_map(|(c, creator)| {
145        creator
146          .marks
147          .iter()
148          .filter(|m| m.by == MarkBy::Footnote)
149          .map(move |m| (c, m))
150      })
151  }
152}

154/// The state key a binding sets whose class prints all of a name's marks as one list, a comma between each two
155/// (elsarticle.cls:250-262 `\sep`, :159-163 `\comma`: `\fnref{fn2,fn3}` "1,2", 2605.13555): every mark after the first
156/// of a name is `sep_before` in the reading's mark record.
157pub const MARKS_LISTED: &str = "lx_frontmatter_marks_listed";
158
159/// Declares [`MARKS_LISTED`]: the binding's class prints a name's marks as one comma-separated list.
160pub fn declare_marks_listed() { assign_value(MARKS_LISTED, true, Some(Scope::Global)); }
