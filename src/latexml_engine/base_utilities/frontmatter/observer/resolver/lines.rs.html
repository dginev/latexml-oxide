26/// The block's lines (rule 1): a `Break` ends a row, a column separator (unless `\and` is inline) a column.
27pub(super) fn lines_of<'a>(events: &'a [Event], and_inline: bool) -> Vec<Line<'a>> {
28  let new = |column| Line {
29    column,
30    events: Vec::new(),
31    after_blank: false,
32    group_break: false,
33  };
34  let mut lines = vec![new(0)];
35  let mut column = 0usize;
36  for event in events {
37    match &event.kind {
38      EventKind::Break => lines.push(Line {
39        group_break: event.role.as_deref() == Some("group"),
40        ..new(column)
41      }),
42      EventKind::Sep(SepKind::And(_) | SepKind::Column(_)) if !and_inline => {
43        column += 1;
44        lines.push(new(column));
45      },
46      // (an inline `\and` after a line of contacts alone opens the next names of the row, as the binding's old split
47      // cut its `\and` pieces (`\lx@llncs@author` until Z18e): `Ann Able\\ \email{ann@x.org} \and Bob Baker\\ University
48      // of Y`, Bob Baker a name, not words of Ann Able's e-mail line — rank 110, review probe L3)
49      EventKind::Sep(SepKind::And(_))
50        if lines.last().is_some_and(|line| {
51          line.events.iter().any(|e| e.kind == EventKind::Contact)
52            && line.events.iter().all(|e| {
53              matches!(
54                e.kind,
55                EventKind::Contact | EventKind::Layout | EventKind::Sep(_)
56              )
57            })
58        }) =>
59      {
60        lines.push(new(column));
61      },
62      EventKind::ApiCall => {},
63      _ => lines.last_mut().expect("a line").events.push(event),
64    }
65  }
66  // (a row of no ink between two inked rows of a column is a blank line)
67  let mut inked_before: Vec<usize> = Vec::new();
68  let mut blank = false;
69  for line in &mut lines {
70    let has_ink = line.events.iter().any(|e| inked(e));
71    if !has_ink {
72      blank = inked_before.contains(&line.column);
73      continue;
74    }
75    line.after_blank = blank && inked_before.contains(&line.column);
76    blank = false;
77    inked_before.push(line.column);
78  }
79  lines.retain(|line| line.events.iter().any(|e| inked(e)));
80  lines
81}

88/// The block's events with the ink one command draws inside a word read as the word's letters (row A2): a box a command
89/// draws right against the letters before or after it (no space between them in the source), whose cover holds a comma
90/// or a row break it draws as ink inside its box, parts no names — combelow's `\cb{t}` stacks a `t` over a comma in an
91/// alignment (combelow.sty:16), so `Bogdan Rai\cb{t}\u{a}` is one name, "Bogdan Raiță" as pdflatex prints it
92/// (2507.13598), not "Bogdan Rai", "t", "ă" at a comma. The cover is the events the invocation printed, each span
93/// opening at the command's one token (its letters' the widest), a row break or a cell of its own alignment no span at
94/// all; one Name with the letters before and after it, their texts joined as the source joins them ([`printed_join`]).
95/// A row break or a comma the command prints at its own level, after its box, parts what it ends as any does
96/// (`\newcommand\nlx{\mbox{Jr.}\\}`, `Ann Able~\nlx Bob Baker`: "Ann Able Jr." and "Bob Baker", review revNL1 S2).
97pub(super) fn words_through_one_command(events: &[Event]) -> Cow<'_, [Event]> {
98  // (a comma drawn inside the command's box, or a row of an alignment it opens there: deeper than the command's own
99  // event — not one it prints after its box)
100  let drawn_inside = |command: &Event, e: &Event| {
101    e.nest > command.nest
102      && match e.kind {
103        EventKind::Break => e.role.as_deref() == Some("row"),
104        EventKind::Sep(SepKind::Comma) => true,
105        _ => false,
106      }
107  };
108  // (most blocks hold no such cover: read as they are, uncopied — the old writer reads a block's events per creator)
109  let parted_cover = events.iter().enumerate().any(|(i, e)| {
110    e.kind == EventKind::Layout
111      && e.span.is_some_and(|(s, t)| {
112        t == s + 1
113          && events[i + 1..]
114            .iter()
115            .take_while(|e| e.span.is_none_or(|(a, _)| a == s))
116            .any(|inner| drawn_inside(e, inner))
117      })
118  });
119  if !parted_cover {
120    return Cow::Borrowed(events);
121  }
122  let mut out: Vec<Event> = Vec::with_capacity(events.len());
123  let mut i = 0;
124  while i < events.len() {
125    // (the command's own token, one: its box's span)
126    let (Some((s, one)), EventKind::Layout) = (events[i].span, &events[i].kind) else {
127      out.push(events[i].clone());
128      i += 1;
129      continue;
130    };
131    if one != s + 1 {
132      out.push(events[i].clone());
133      i += 1;
134      continue;
135    }
136    let mut end = one;
137    let mut last = i;
138    let mut j = i + 1;
139    while j < events.len() {
140      let e = &events[j];
141      // (an event its invocation printed: its span opens at the command)
142      let inside = match e.span {
143        Some((a, b)) if a == s => {
144          end = end.max(b);
145          true
146        },
147        Some(_) => false,
148        None => true,
149      };
150      if !inside
151        || !matches!(
152          e.kind,
153          EventKind::Layout | EventKind::Name | EventKind::Break | EventKind::Sep(SepKind::Comma)
154        )
155      {
156        break;
157      }
158      if e.span.is_some() {
159        last = j;
160      }
161      j += 1;
162    }
163    let cover = &events[i..=last];
164    let parts = cover.iter().any(|e| drawn_inside(&events[i], e));
165    let named = |e: &Event| e.kind == EventKind::Name;
166    let before = out
167      .last()
168      .filter(|e| e.kind == EventKind::Name && e.span.is_some_and(|(_, b)| b == s));
169    let after = events
170      .get(last + 1)
171      .filter(|e| e.kind == EventKind::Name && e.span.is_some_and(|(a, _)| a == end));
172    if !parts || !cover.iter().any(named) || (before.is_none() && after.is_none()) {
173      out.push(events[i].clone());
174      i += 1;
175      continue;
176    }
177    let mut word = match before {
178      Some(_) => out.pop().expect("the letters before"),
179      None => Event {
180        text: String::new(),
181        span: Some((s, end)),
182        ..cover.iter().find(|e| named(e)).expect("a letter").clone()
183      },
184    };
185    let (through, to) = match after {
186      Some(after) => (last + 1, after.span.map_or(end, |(_, b)| b)),
187      None => (last, end),
188    };
189    // (the letters before end at the command's token, so nothing parts them from its own)
190    word
191      .text
192      .push_str(&printed_join(&events[i..=through], named));
193    i = through + 1;
194    // (its own tokens too, to the last letter: the emitter's run of the name, `emitter::emit::event_run_at`)
195    word.span = word.span.map(|(a, _)| (a, to));
196    word.marks = word.marks.map(|(a, _)| (a, to));
197    out.push(word);
198  }
199  Cow::Owned(out)
200}

202/// The text of a run of events' pieces (`piece`) joined as the source joins them: a space between two pieces only where
203/// the source has one, a token none of the run's events covers between them — `(for the` and `\babar\ collaboration)`
204/// parted by the space before `\babar` (0711.1544); the letters before a command's box, its own and those after it, no
205/// space (`Rai\cb{t}\u{a}`, 2507.13598; [`words_through_one_command`]); a note reference between two pieces covering
206/// the tokens it prints, the space after it the source's (`BNL-Bielefeld\fnref{col1} Collaboration)`, 1212.4278). An
207/// event of no span (a cell of a command's alignment) parts nothing. Each piece's text trimmed; the reading's one join
208/// of a name's pieces ([`remark_pieces_joined`]) and of a word's ink.
209pub(super) fn printed_join(run: &[Event], piece: impl Fn(&Event) -> bool) -> String {
210  let mut text = String::new();
211  // (where the source covered so far ends, and whether a token no event covers lies since the last piece)
212  let mut covered: Option<u16> = None;
213  let mut spaced = false;
214  for e in run {
215    let is_piece = piece(e);
216    if let Some((a, b)) = e.span {
217      if covered.is_some_and(|c| a > c) {
218        spaced = true;
219      }
220      covered = Some(covered.map_or(b, |c| c.max(b)));
221    }
222    if !is_piece {
223      continue;
224    }
225    let t = e.text.trim();
226    if t.is_empty() {
227      continue;
228    }
229    if spaced && !text.is_empty() {
230      text.push(' ');
231    }
232    text.push_str(t);
233    spaced = false;
234  }
235  text
236}

238/// The block's events with an IEEE template's ordinal before a name (`1\textsuperscript{st} Sertac Kilickaya`,
239/// 2412.10792, 2202.06596) one Name with it, as today's personname "1 Sertac Kilickaya": the ordinal's superscript is
240/// no request. On a column's first line and on no line a mark leads: anywhere else the ordinal is its line's text
241/// (`Event::ordinal`, `events_text`), `5\textsuperscript{th} Floor` under a name.
242pub(super) fn join_ordinals(events: &[Event]) -> Vec<Event> {
243  let mut out: Vec<Event> = Vec::with_capacity(events.len());
244  let mut i = 0;
245  let mut names_line = true;
246  let mut line_inked = false;
247  while i < events.len() {
248    let event = &events[i];
249    match event.kind {
250      EventKind::Break => {
251        names_line = false;
252        line_inked = false;
253      },
254      EventKind::Sep(SepKind::And(_) | SepKind::Column(_)) => {
255        names_line = true;
256        line_inked = false;
257      },
258      EventKind::Mark if !line_inked => {
259        names_line = false;
260        line_inked = true;
261      },
262      _ if inked(event) => line_inked = true,
263      _ => {},
264    }
265    if names_line
266      && event.kind == EventKind::Name
267      && !event.text.is_empty()
268      && event.text.chars().all(|c| c.is_ascii_digit())
269      && let Some(mark) = events.get(i + 1)
270      && mark.ordinal
271      && let Some(name) = events.get(i + 2).filter(|e| e.kind == EventKind::Name)
272    {
273      out.push(Event {
274        text: s!("{} {}", event.text, name.text),
275        span: match (event.span, name.span) {
276          (Some((a, _)), Some((_, b))) => Some((a, b)),
277          (a, b) => a.or(b),
278        },
279        ..name.clone()
280      });
281      i += 3;
282      continue;
283    }
284    out.push(event.clone());
285    i += 1;
286  }
287  out
288}

290/// The block's events with each link printing a name read as that name. A link to an ORCID record that prints a name is that
291/// name and its ORCID, the link after it (`contact_parts`): `\href{https://orcid.org/0000-0002-9970-3034}{E. L. Gagnon}$^{*,1,2}$,
292/// …`, 2312.16289; orcidlink's `\orcidlinki{José-María Martín-Olalla}{0000-0002-3750-9113}`, 2602.03244; 2301.02927. Any other link
293/// that prints other text than its own target and fills a name slot ([`name_slots`]) is that text, a Name carrying the link,
294/// its span the link's call (written whole, with it): `\href{http://inspirehep.net/record/1021757}{A.~Cerri}$^{1}$`, 304 of them in
295/// 1812.07638 (longtail3k), whose url contacts were one creator's affiliation under the emitter's switch; authblk's
296/// `\author[1,2]{\href{kaizheng.wang@ntu.edu.sg}{Kaizheng Wang}\thanks{…}}`, 2602.22747, and an unmarked `\href{…}{Bob Baker}` between
297/// two names' commas, each a nameless creator's url contact before. A link printing its own target is an address (`\url{…}`,
298/// `\href{mailto:ann@x.org}{ann@x.org}`), one sharing its slot with other ink or on a later line carrying nothing a contact
299/// (`Ann Able \\ \href{…}{Project Page}`).
300pub(super) fn link_names(events: &[Event]) -> Vec<Event> {
301  let slots = name_slots(events);
302  let mut out = Vec::with_capacity(events.len());
303  for (at, event) in events.iter().enumerate() {
304    let link = event.kind == EventKind::Contact && !event.text.trim().is_empty();
305    let orcid = link && contact_parts(event).0 == "orcid";
306    let named = (orcid && reads_as_names(&event.text))
307      || (link
308        && !orcid
309        && event.role.as_deref() == Some("url")
310        && slots[at]
311        && !event
312          .target
313          .as_deref()
314          .is_some_and(|target| prints_its_target(&event.text, target)));
315    if !named {
316      out.push(event.clone());
317      continue;
318    }
319    let mut name = Event::new(EventKind::Name);
320    name.text = event.text.trim().to_string();
321    name.span = event.span;
322    name.nest = event.nest;
323    out.push(name);
324    if orcid {
325      out.push(Event {
326        text: String::new(),
327        ..event.clone()
328      });
329    }
330  }
331  out
332}

334/// Per event, whether it is a link filling a name slot: the only ink of the piece between its line's separators (a comma, a
335/// word, a gap, `\and`, an alignment tab, a break, the block's ends) but for the marks and notes a name carries, an inkless
336/// link (a logo) and punctuation, on a line no mark leads (an entry, `$^{1}$\href{…}{Univ A}`, is no name), which is its
337/// column's first line or a names line by its marks: the slot carries a mark or a note after the link, as a later line's
338/// names do (`later_line_is_names`; 1812.07638's `Editors: \\ \href{…}{A.~Cerri}$^{1}$, …`).
339fn name_slots(events: &[Event]) -> Vec<bool> {
340  let mut out = vec![false; events.len()];
341  let line_end = |e: &Event| {
342    matches!(
343      e.kind,
344      EventKind::Break | EventKind::Sep(SepKind::And(_) | SepKind::Column(_))
345    )
346  };
347  let link = |e: &Event| e.kind == EventKind::Contact && !e.text.trim().is_empty();
348  let mut column_first = true;
349  let mut start = 0;
350  while start <= events.len() {
351    let end = events[start..]
352      .iter()
353      .position(line_end)
354      .map_or(events.len(), |p| start + p);
355    let line = &events[start..end];
356    let mark_led = line
357      .iter()
358      .find(|e| {
359        inked(e) && !(e.kind == EventKind::Name && !e.text.chars().any(char::is_alphanumeric))
360      })
361      .is_some_and(|e| e.kind == EventKind::Mark);
362    if !mark_led {
363      let mut from = 0;
364      while from <= line.len() {
365        let to = line[from..]
366          .iter()
367          .position(|e| matches!(e.kind, EventKind::Sep(_)))
368          .map_or(line.len(), |p| from + p);
369        let slot = &line[from..to];
370        let other_ink = |e: &&Event| match e.kind {
371          EventKind::Name => e.text.chars().any(char::is_alphanumeric),
372          EventKind::Contact => link(e),
373          _ => false,
374        };
375        if let [at] = slot
376          .iter()
377          .enumerate()
378          .filter(|(_, e)| other_ink(e))
379          .map(|(i, _)| i)
380          .collect::<Vec<_>>()
381          .as_slice()
382          && link(&slot[*at])
383        {
384          let carries = slot[at + 1..]
385            .iter()
386            .any(|e| (e.kind == EventKind::Mark && !e.line_first) || e.kind == EventKind::Note);
387          out[start + from + at] = column_first || carries;
388        }
389        from = to + 1;
390      }
391    }
392    if let Some(sep) = events.get(end) {
393      column_first = sep.kind != EventKind::Break;
394    }
395    start = end + 1;
396  }
397  out
398}

400/// Whether a link's printed text is its own target, its address: the target, its scheme and a closing slash aside, in any
401/// case (`\href{https://x.ai/}{X.ai}`, 2609.25082; `\href{mailto:ann@x.org}{ann@x.org}`).
402fn prints_its_target(text: &str, target: &str) -> bool {
403  let (text, target) = (
404    text.trim().trim_end_matches('/'),
405    target.trim().trim_end_matches('/'),
406  );
407  let past_scheme = target
408    .split_once(':')
409    .filter(|(scheme, _)| {
410      !scheme.is_empty()
411        && scheme
412          .chars()
413          .all(|c| c.is_ascii_alphanumeric() || "+.-".contains(c))
414    })
415    .map(|(_, rest)| rest.trim_start_matches('/'));
416  text.eq_ignore_ascii_case(target)
417    || past_scheme.is_some_and(|rest| text.eq_ignore_ascii_case(rest))
418}

420/// The block's events with a mark printed as text where a line opens: a Name whose first word is a symbol some name
421/// shows as its mark (`\small\em$\sharp$ - CBCL, …` under `Mroueh$^{\sharp}$`, 1209.1360 — `♯` is no `\@fnsymbol`
422/// character, so the lexer read it as ink) parted into that Mark and the rest.
423pub(super) fn split_text_marks(events: &[Event]) -> Vec<Event> {
424  let shown: Vec<String> = events
425    .iter()
426    .filter(|e| e.kind == EventKind::Mark && !e.line_first)
427    .flat_map(|e| mark_labels(e.label.as_deref().unwrap_or("")))
428    .filter(|l| !l.chars().any(char::is_alphanumeric))
429    .collect();
430  if shown.is_empty() {
431    return events.to_vec();
432  }
433  let mut out = Vec::with_capacity(events.len());
434  let mut line_start = true;
435  for event in events {
436    match event.kind {
437      EventKind::Break | EventKind::Sep(SepKind::And(_) | SepKind::Column(_)) => line_start = true,
438      EventKind::Name if line_start => {
439        line_start = false;
440        let (first, rest) = event
441          .text
442          .split_once(' ')
443          .unwrap_or((event.text.as_str(), ""));
444        if shown.iter().any(|l| l == first) {
445          let mut mark = Event::new(EventKind::Mark);
446          mark.label = Some(first.to_string());
447          mark.text = first.to_string();
448          // (a footnote symbol only when every character is one, as the lexer reads a mark: `#` and a `★` set in text
449          // are marks of no `\@fnsymbol` kind — main-session decision 3 on FM S2a.2 r7, review revfms2a2_7 finding 6)
450          mark.symbol = first.chars().all(is_symbol_char);
451          mark.line_first = true;
452          mark.role = Some("text".to_string());
453          mark.span = event.span;
454          mark.nest = event.nest;
455          out.push(mark);
456          let rest = rest.trim_start_matches(|c: char| c == '-' || c.is_whitespace());
457          if !rest.is_empty() {
458            out.push(Event {
459              text: rest.to_string(),
460              ..event.clone()
461            });
462          }
463          continue;
464        }
465      },
466      EventKind::Name | EventKind::Mark | EventKind::Contact | EventKind::Note => {
467        line_start = false
468      },
469      _ => {},
470    }
471    out.push(event.clone());
472  }
473  out
474}

476/// The block's events with each run of footnote symbols written together (`§†`) that a line leads with and a name
477/// shows read as one mark ([`Event::glued`]), as a run of one symbol (`††`) is: 0907.1243's `W.Y.~Ching$^{\S\dagger}$`
478/// under the line `$^{\S\dagger}$Department of Physics, University of Missouri …` is that line's alone, not every `§`
479/// and `†` carrier's, and the lines `$^{\S}$DuPont …`, `$^{\dagger}$Laboratory …` are not Ching's (main-session
480/// decision 2 on FM S2a.2 r7). A run no line leads with stays the marks it is written of (`R.~Podgornik$^{\dagger\star}$`).
481pub(super) fn glue_shown_runs(events: &[Event]) -> Vec<Event> {
482  let run_of = |e: &Event| -> Option<String> {
483    if e.kind != EventKind::Mark {
484      return None;
485    }
486    let run: String = mark_text_symbols(e.label.as_deref().unwrap_or(""));
487    let first = run.chars().next()?;
488    (run.chars().count() >= 2 && run.chars().all(is_symbol_char) && run.chars().any(|c| c != first))
489      .then_some(run)
490  };
491  let led: Vec<String> = events
492    .iter()
493    .filter(|e| e.line_first)
494    .filter_map(run_of)
495    .collect();
496  let glued: Vec<String> = events
497    .iter()
498    .filter(|e| !e.line_first)
499    .filter_map(run_of)
500    .filter(|run| led.contains(run))
501    .collect();
502  events
503    .iter()
504    .map(|e| {
505      let mut e = e.clone();
506      e.glued = run_of(&e).is_some_and(|run| glued.contains(&run));
507      e
508    })
509    .collect()
510}

512/// A mark's label as [`mark_labels`] reads its characters (trimmed, every asterisk form as `*`, the formula `\|`'s `∥`
513/// as `‖`).
514fn mark_text_symbols(label: &str) -> String {
515  label
516    .trim()
517    .chars()
518    .map(|c| match c {
519      // (every asterisk form one key, user ruling 2026-11-02: `queue_digest::one_asterisk`)
520      c if ASTERISK_FORMS.contains(&c) => '*',
521      '\u{2225}' => '‖',
522      c => c,
523    })
524    .collect()
525}

545/// The labelled entries of a line whose marks trail their text (`Nanyang Technological University$^1$, Amazon$^2$`,
546/// 2507.05750): each mark after a Name closes an entry of the ink since the entry before; the line must end in a mark.
547pub(super) fn trailing_entries<'a>(line: &[&'a Event]) -> Vec<Entry<'a>> {
548  let mut entries: Vec<Entry<'a>> = Vec::new();
549  let mut run: Vec<&'a Event> = Vec::new();
550  let mut closed = false;
551  for event in line {
552    match &event.kind {
553      EventKind::Mark => {
554        let labels = event_labels(event);
555        if run.iter().any(|e| e.kind == EventKind::Name) {
556          while run
557            .last()
558            .is_some_and(|e| matches!(e.kind, EventKind::Sep(SepKind::Comma)))
559          {
560            run.pop();
561          }
562          let symbol = event.symbol;
563          entries.push(Entry {
564            labels,
565            symbol,
566            compound: false,
567            events: std::mem::take(&mut run),
568          });
569          closed = true;
570        } else if let Some(entry) = entries.last_mut().filter(|_| closed) {
571          entry.labels.extend(labels);
572        } else {
573          return Vec::new();
574        }
575      },
576      EventKind::Layout | EventKind::Sep(SepKind::Gap { .. }) => {},
577      EventKind::Sep(SepKind::Comma) if run.is_empty() => {},
578      _ => {
579        closed = false;
580        run.push(event);
581      },
582    }
583  }
584  if run.iter().any(|e| inked(e)) {
585    return Vec::new();
586  }
587  entries
588}

590/// The labelled entries of a mark-led line: each opened by a mark at the line's start or after a separator (a gap, a
591/// comma), and the ink up to the next such mark; a mark right after an entry's ink is shown on it (no label of it).
592pub(super) fn entries_of<'a>(line: &[&'a Event]) -> Vec<Entry<'a>> {
593  let mut entries: Vec<Entry<'a>> = Vec::new();
594  let mut after_sep = true;
595  let mut last_ink: Option<&Event> = None;
596  for event in line {
597    match &event.kind {
598      // (a number's suffix is the entry's text, `Event::ordinal`)
599      EventKind::Mark if event.ordinal => {
600        if let Some(entry) = entries.last_mut() {
601          entry.events.push(event);
602        }
603        after_sep = false;
604      },
605      EventKind::Mark => {
606        // (a mark set off from the ink before it by a space — the source positions between them — opens an entry of
607        // its own: `$^1$Arizona State University $^2$Rider University`, 2601.20069; one glued to it is shown on it)
608        if let (Some(prev), Some((start, _))) = (last_ink.and_then(|e| e.span), event.span)
609          && start > prev.1
610        {
611          after_sep = true;
612        }
613        let labels = event_labels(event);
614        let symbol =
615          event.symbol || (!labels.is_empty() && labels.iter().all(|l| is_symbol_label(l)));
616        // (one mark printing distinct symbols together, no separator between them: `$^{\dagger\ddagger}$`)
617        let compound = labels.len() > 1
618          && event
619            .label
620            .as_deref()
621            .is_some_and(|l| l.trim() == labels.concat());
622        let open = entries.last().is_some_and(|e| e.events.is_empty());
623        if open {
624          if let Some(entry) = entries.last_mut() {
625            entry.labels.extend(labels);
626            entry.symbol &= symbol;
627            entry.compound = false;
628          }
629        } else if after_sep || entries.is_empty() {
630          entries.push(Entry {
631            labels,
632            symbol,
633            compound,
634            events: Vec::new(),
635          });
636        }
637        after_sep = false;
638      },
639      EventKind::Sep(SepKind::Gap { .. }) => after_sep = true,
640      EventKind::Sep(SepKind::Comma) => {
641        after_sep = true;
642        if let Some(entry) = entries.last_mut() {
643          entry.events.push(event);
644        }
645      },
646      EventKind::Layout => {},
647      // (punctuation opening an entry is no text of it: the `)` of `($^1$) Univ A`, 1004.4809)
648      EventKind::Name
649        if !event.text.chars().any(char::is_alphanumeric)
650          && entries.last().is_some_and(|e| e.events.is_empty()) => {},
651      _ => {
652        after_sep = false;
653        last_ink = Some(event);
654        if let Some(entry) = entries.last_mut() {
655          entry.events.push(event);
656        }
657      },
658    }
659    if event.kind == EventKind::Mark {
660      last_ink = Some(event);
661    }
662  }
663  let count = entries.len();
664  for (i, entry) in entries.iter_mut().enumerate() {
665    // (the separator before the next entry is no entry's: `Berkeley and $^2$National Bureau …`, 1810.07216; the line's
666    // last entry keeps a word the next line continues, `Biocomputación and \\ Física …`, 1107.1186)
667    while entry.events.last().is_some_and(|e| match e.kind {
668      EventKind::Sep(SepKind::Comma) => true,
669      EventKind::Sep(SepKind::Word(_)) => i + 1 < count,
670      _ => false,
671    }) {
672      entry.events.pop();
673    }
674  }
675  entries
676}

686/// Per line of a block, whether the marks leading it are its names' requests, not the labels of affiliation entries: a
687/// line whose first ink is a mark and whose every entry reads as names, above the block's first line setting labels, and
688/// whose marks are requests — answered by a later line, set by an affiliation call outside the block (`outside`), or,
689/// no line setting any, every line of the block such a line. TeX prints the mark before the name it labels: `$^1$Ann
690/// Able \and $^2$Bob Baker` "¹Ann Able ²Bob Baker" (2507.23021, 2605.06126, 2505.14984, 1105.2204, 2205.03535).
691/// The second vector: the mark-led lines whose entries read as names, whatever the reading.
692pub(super) fn names_led_by_marks(lines: &[Line<'_>], outside: &[String]) -> (Vec<bool>, Vec<bool>) {
693  let led: Vec<Option<Vec<Entry<'_>>>> = lines
694    .iter()
695    .map(|line| {
696      line_first_ink(&line.events)
697        .filter(|e| e.kind == EventKind::Mark)
698        .map(|_| {
699          // (an entry of no ink is a mark shown on the name before it: `Adarsh Singh$^{1}$ \quad`, 2505.14984)
700          let mut entries = entries_of(&line.events);
701          entries.retain(|e| e.events.iter().any(|e| inked(e)));
702          entries
703        })
704    })
705    .collect();
706  if led.iter().all(Option::is_none) {
707    return (vec![false; lines.len()], vec![false; lines.len()]);
708  }
709  // (the labels each mark-led line's entries set, were it affiliations; the marks each line shows, requests above a setter)
710  let sets: Vec<Vec<String>> = led
711    .iter()
712    .map(|entries| {
713      entries
714        .iter()
715        .flatten()
716        .flat_map(|e| e.labels.clone())
717        .collect()
718    })
719    .collect();
720  let shown: Vec<Vec<String>> = lines
721    .iter()
722    .map(|line| {
723      line
724        .events
725        .iter()
726        .filter(|e| e.kind == EventKind::Mark)
727        .flat_map(|e| event_labels(e))
728        .collect()
729    })
730    .collect();
731  let matched = |labels: &[String], label: &str| labels.iter().any(|l| label_matches(l, label));
732  // (the first setter: a mark-led line under the first of its column (an `\and` opens an author's box, its first line
733  // the names'), setting each label once, every one requested by a mark above it, the lines from it on setting no
734  // number no mark above requests, and every number requested above set from it on or outside — the affiliations under
735  // `$^1$Ann, $^1$Bob, $^2$Cat \\ $^1$Dan, $^2$Eve` start at `$^1$Univ A \quad $^2$Univ B`, not at the second names row,
736  // which sets label 1 twice, 2507.23021; nor at `$^3$Dan, $^1$Fay`, which sets a 3 no one requested, 2605.06126)
737  let mut requested: Vec<String> = Vec::new();
738  let setter = (1..lines.len()).find(|&k| {
739    requested.extend(shown[k - 1].iter().cloned());
740    let line_sets = &sets[k];
741    if led[k].is_none()
742      || line_sets.is_empty()
743      || lines[..k].iter().all(|l| l.column != lines[k].column)
744    {
745      return false;
746    }
747    line_sets.iter().enumerate().all(|(i, l)| !line_sets[..i].contains(l))
748      && line_sets.iter().all(|l| matched(&requested, l))
749      // (labels an affiliation call outside the block sets are requests: `\inst{1}Ann \and \inst{1}Bob` under aa's
750      // `\institute{Univ A}`)
751      && !line_sets.iter().all(|l| matched(outside, l))
752      && sets[k + 1..]
753        .iter()
754        .flatten()
755        .filter(|l| !is_symbol_label(l))
756        .all(|l| matched(&requested, l))
757      && requested
758        .iter()
759        .filter(|l| !is_symbol_label(l))
760        .all(|l| sets[k..].iter().any(|set| matched(set, l)) || matched(outside, l))
761  });
762  let names = |entries: &Option<Vec<Entry<'_>>>| {
763    entries.as_ref().is_some_and(|entries| {
764      !entries.is_empty()
765        && entries
766          .iter()
767          .all(|e| reads_as_names(&events_text(&e.events)))
768    })
769  };
770  let shaped: Vec<bool> = led.iter().map(names).collect();
771  let whole_block = setter.is_none() && shaped.iter().all(|s| *s);
772  let led_names = (0..lines.len())
773    .map(|n| {
774      shaped[n]
775        && setter.is_none_or(|k| n < k)
776        && (whole_block
777          || shown[n]
778            .iter()
779            .any(|l| sets[n + 1..].iter().any(|set| matched(set, l)) || matched(outside, l)))
780    })
781    .collect();
782  (led_names, shaped)
783}

882/// A names line's events with each group of addresses whose printed braces hold the line's commas one Name run again,
883/// as an address reader reads it: `\IEEEauthorblockN{Ann Able, … E-mails: \{ann, bob\}@a.org, cat@b.org}` parts at those
884/// commas into `E-mails: {ann` and `bob}@a.org`, no names (2003.02512). Commas in braces that print no address part names

889  while at < events.len() {

912    } else {
913      out.push(event.clone());

915    }

1045/// collaboration)`, the remark in a box, 0711.1544; `Ann Able\thanks{…} (for the \textsc{Atlas} Collaboration)`,
1046/// review revNL1 S1); a name whose own event opens the remark joins with it (`Christian Schmidt (for the
1047/// BNL-Bielefeld\fnref{col1} Collaboration)`, elsarticle, 1212.4278). A remark reads as a collaboration as
1048/// `reads_as_collaboration` reads it, or after the name's notes or marks as `speaks_for_after_notes` does (`for
1049/// CAPP/IBS`, 1702.03664).
1050pub(super) fn remark_pieces_joined(events: Vec<Event>) -> Vec<Event> {
1051  let refs: Vec<&Event> = events.iter().collect();
1052  let named = |e: &Event| e.kind == EventKind::Name && e.text.chars().any(char::is_alphanumeric);
1053  // (the remark a text holds after the name it reads as, or is after a name; `annotated` its name's notes or marks)
1054  let remark = |text: &str, after_name: bool, annotated: bool| {
1055    let reads = |rest: &str| {
1056      reads_as_collaboration(rest, false) || (annotated && speaks_for_after_notes(rest))
1057    };
1058    name_before_remark(text).is_some_and(|name| reads(text.trim()[name.len()..].trim()))
1059      || (after_name && is_name_remark(text) && reads(text))
1060      || (after_name && annotated && speaks_for_after_notes(text))
1061  };
1062  let mut out: Vec<Event> = Vec::with_capacity(events.len());
1063  let mut i = 0;
1064  while i < events.len() {
1065    // (a run starts where a remark opens: a piece no remark opens in is read as it is)
1066    if !named(&events[i]) || remark_start(&events[i].text).is_none() {
1067      out.push(events[i].clone());
1068      i += 1;
1069      continue;
1070    }
1071    // (a name before it in the line, and its notes or marks between)
1072    let before = refs[..i]
1073      .iter()
1074      .rposition(|e| named(e) || matches!(e.kind, EventKind::Sep(_) | EventKind::Break));
1075    let after_name = before.is_some_and(|b| named(refs[b]));
1076    let annotated = before.is_some_and(|b| {
1077      refs[b + 1..i]
1078        .iter()
1079        .any(|e| matches!(e.kind, EventKind::Note | EventKind::Mark))
1080    });
1081    let mut last = i;
1082    let mut j = i + 1;
1083    while j < events.len() {
1084      match &events[j].kind {
1085        EventKind::Layout | EventKind::Note => j += 1,
1086        EventKind::Mark if name_annotation(&events[j]) => j += 1,
1087        // (a mark separator, `\fnmsep`'s and `\fnsep`'s comma, moves with the marks it parts; a mark no annotation, a
1088        // number's `$^{1}$`, stays in the remark, which goes on past it: the old reader's `name_takes_annotations`, review
1089        // revG1 2)
1090        EventKind::Sep(SepKind::Comma) if mark_separator(&events[j]) => j += 1,
1091        EventKind::Mark => j += 1,
1092        EventKind::Sep(SepKind::Gap { .. })
1093          if j + 1 < events.len() && gap_inside_one_command(&refs, j + 1, refs[j]) =>
1094        {
1095          j += 1;
1096        },
1097        _ if named(&events[j]) => {
1098          last = j;
1099          j += 1;
1100        },
1101        _ => break,
1102      }
1103    }
1104    let run = &events[i..=last];
1105    let pieces = run.iter().filter(|e| named(e)).count();
1106    let alone = remark(&events[i].text, after_name, annotated);
1107    let text = printed_join(run, named);
1108    // (the notes inside it count as the name's when they part the name from its remark)
1109    let inside = run
1110      .iter()
1111      .any(|e| e.kind == EventKind::Note || name_annotation(e));
1112    if pieces < 2 || alone || !remark(&text, after_name, annotated || inside) {
1113      out.push(events[i].clone());
1114      i += 1;
1115      continue;
1116    }
1117    out.push(Event {
1118      text,
1119      span: match (events[i].span, events[last].span) {
1120        (Some((a, _)), Some((_, b))) => Some((a, b)),
1121        (a, b) => a.or(b),
1122      },
1123      ..events[i].clone()
1124    });
1125    out.extend(
1126      run
1127        .iter()
1128        .filter(|e| e.kind == EventKind::Note || name_annotation(e) || mark_separator(e))
1129        .cloned(),
1130    );
1131    i = last + 1;
1132  }
1133  out
1134}

1923            .role
1924            .as_deref()
1925            .is_some_and(|r| r.ends_with(" parted"))
1926          {
1927            labels.clone()
1928          } else if event.glued {
1929            Vec::new()
1930          } else {
1931            glued_symbol_parts(event.label.as_deref().unwrap_or(""))
1932          };
1933          if !parted.is_empty() {
1934            let owner = if to_next { None } else { made.last().copied() };
1935            if let Some(c) = owner {
1936              self.parted.extend(parted.into_iter().map(|l| (c, l)));
1937            } else {
1938              waiting_parted.extend(parted);
1939            }
1940          }
1941          let class = is_class_request(event);
1942          // (a mark on a collaboration is no name's: `… and LHD experimental group$^{2}$`, 1711.07097)
1943          if after_collaboration && !to_next {
1944            continue;
1945          }
1946          if to_next {
1947            if footnote {
1948              waiting_fn.extend(labels.iter().cloned());
1949            }
1950            if class {
1951              waiting_class.extend(labels.iter().cloned());
1952            }
1953            waiting.extend(labels);
1954          } else if let Some(&c) = made.last() {
1955            if footnote {
1956              self
1957                .fn_requests
1958                .extend(labels.iter().map(|l| (c, l.clone())));
1959            }
1960            if class {
1961              self.out.creators[c]
1962                .class_requests
1963                .extend(labels.iter().cloned());
1964            }
1965            self.out.creators[c].requests.extend(labels);
1966            if after_comma {
1967              let rule = if inst && policy.inst_after_comma {
1968                "mark:inst-after-comma"
1969              } else {
1970                "mark:after-comma"
1971              };
1972              let creator = &mut self.out.creators[c];
1973              if !creator.rule.contains(rule) {
1974                creator.rule.push(' ');
1975                creator.rule.push_str(rule);
1976              }
1977            }
1978          }
1979        },
1980        EventKind::Note => match made.last() {
1981          Some(&c) => self.note(index, event, Some(c), "note:adjacent"),
1982          None => waiting_notes.push(event),
1983        },
1984        EventKind::Contact => {
1985          // (an unlabelled annotation of the run of new authors, `annotate=run` or `new` — a class's `\affil` printed
1986          // in place inside `\author`, rerouted to `\lx@add@affiliation[annotate=run]`: `JORGE RAMIREZ and EVGENY
1987          // KRIVOSHEEV \affil{University of Trento} MARCOS BAEZ and …`, ci2018, 1805.12376 —, the run's names of the
1988          // line as today's API attaches it (`Resolver::record_run`), not the name beside it alone)
1989          // (the run ends at the block's contact of the role before it, labelled or not: a labelled one waits for its
1990          // label, its owner holding no role yet — `ANN ABLE$^1$ \affil{$^1$Univ A} BOB BAKER \affil{Univ B}` gives
1991          // Univ B to Bob alone, review rev94a r6 B1; `Resolver::ends_a_run`)
1992          let role = contact_parts(event).0;
1993          let run: Vec<usize> = match event.annotate.as_deref() {
1994            Some(annotate)
1995              if event.label.is_none() && (annotate == "new" || annotate.starts_with("run")) =>
1996            {
1997              let run = if annotate == "new" {
1998                self.new_run(&role)
1999              } else {
2000                self.record_run(annotate, &role, false)
2001              };
2002              run.into_iter().filter(|c| made.contains(c)).collect()
2003            },
2004            _ => Vec::new(),
2005          };
2006          self.run_from.insert(role, self.out.creators.len());
2007          if run.len() > 1 {
2008            self.contact_event(index, event, &run, "owner:annotate:run");
2009          } else {
2010            let owner: Vec<usize> = made.last().copied().into_iter().collect();
2011            self.contact_event(index, event, &owner, "owner:adjacent");
2012          }
2013        },
2014        _ => {},
2015      }
2016    }
2017    for note in waiting_notes {
2018      self.note(index, note, made.last().copied(), "note:adjacent");
2019    }
2020    // (a heading over the names after it: those of the line)
2021    if let Some((text, from)) = heading
2022      && let Some(first) = events.first()
2023    {
2024      let covered = made[from.min(made.len())..].to_vec();
2025      self.collaboration(index, first.span, &covered, &text);
2026    }
2027    if !waiting.is_empty()
2028      && let Some(&c) = made.last()
2029    {
2030      self.out.creators[c].requests.extend(waiting);
2031      self
2032        .fn_requests
2033        .extend(waiting_fn.into_iter().map(|l| (c, l)));
2034      self.out.creators[c].class_requests.extend(waiting_class);
2035      self
2036        .parted
2037        .extend(waiting_parted.into_iter().map(|l| (c, l)));
2038    }
2039    NamesLine { creators: made }
2040  }

2042  /// The live creators of the document no collaboration covers yet: what a call of its own naming one covers (a replaced
2043  /// call's creators print nothing: the class's `\author{Ann}\author{for the X Collaboration}` prints only the second).
2044  fn uncovered_creators(&self) -> Vec<usize> {
2045    (self.collaboration_from..self.out.creators.len())
2046      .filter(|c| !self.replaced.contains(c))
2047      .collect()
2048  }

2051  /// covers, as today's reader gives it (`Collaborations`, OXIDIZED_DESIGN_DIVERGENCES #946), sourced at `span`, the
