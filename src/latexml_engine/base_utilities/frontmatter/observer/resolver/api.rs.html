19/// [`split_pieces`] of `events`, cut at its items where the reading's own events of the same call, `read`, are
20/// ([`item_led_pieces`]): the writer's rehearsal of the content at the flush reads the meanings then, a list's labels
21/// perhaps gone (cmpj's `\ukrainianpart` makes `\addr` a bare `\item`, cmpj2.sty:649; 1212.6367).
22pub(in crate::base_utilities::frontmatter::observer) fn split_pieces_as(
23  read: &[Event],
24  events: &[Event],
25  role: &str,
26) -> Vec<(String, Vec<Event>)> {
27  let splits_contacts = matches!(role, "affiliation" | "address" | "altaffiliation");
28  let pieces = if item_led_pieces(read).is_empty() {
29    mark_led_pieces(events)
30  } else {
31    list_items(events)
32  };
33  pieces
34    .into_iter()
35    .map(|piece| {
36      let refs: Vec<&Event> = piece
37        .iter()
38        .filter(|e| {
39          !(splits_contacts
40            && e.kind == EventKind::Contact
41            && matches!(e.role.as_deref(), Some("email" | "orcid")))
42        })
43        .collect();
44      (events_text(&refs), piece)
45    })
46    .collect()
47}

49/// Whether a call is a class's one shared institute (`_share=authors`, not a piece its marks split) holding marks typed by
50/// hand after a line's first words, a paper's own (`\affmark{1} City University … \quad \affmark{2} ShanghaiTech University
51/// … \\ \affmark{4} …`, 2608.21425): no line or mark of it leads an institute of its own — the one entry, which no class mark requests, the
52/// block's whole, as the old split's `marks_pick` left it ("an institute holding marks of its own" is not shared). The
53/// class's own `\inst` marks in it (aa's `\institute{\inst{1} Univ X \\ \inst{2} Univ Y}`, 1003.3020) still lead its
54/// lines. The reading keeps it whole ([`Resolver::api`]) and the emitter writes it so (`emit::item_call`).
55pub(in crate::base_utilities::frontmatter::observer) fn hand_marked_share(
56  attrs: &str,
57  events: &[Event],
58) -> bool {
59  // (a mark typed by hand after a line's first words: the line no one institute its leading mark labels — a line each
60  // led by its mark, `$^1$Univ A \\ $^2$Univ B` under `\inst{1,2}`, is still read line by line, 1803.02579)
61  shares_the_row(attrs)
62    && attr_value(attrs, "_markpiece") != Some("true")
63    && events.split(|e| e.kind == EventKind::Break).any(|line| {
64      line
65        .iter()
66        .skip_while(|e| e.kind != EventKind::Name)
67        .any(|e| e.kind == EventKind::Mark && !is_class_request(e))
68    })
69}

71/// The nest of a creator call's own ink ([`creator_call_pieces`]): inside the `\lx@personname` its call wraps the name in

76  }))

79/// The pieces a creator call's own `\and`s cut it into, each holding a name: `\lx@add@creator[role=author]{Ann Able \and
80/// Bob Baker}`, a binding handing one whole argument to the API — acmart's `\author` with no comma, quantumarticle's,
81/// authblk's and ecai's where their spelling test finds no separator (a macro-held `\and`) — whose `\and`, literal, held
82/// in a macro, spelled `\csname and\endcsname`, printed by a wrapper or by an active character, pdflatex parts into two
83/// authors (acmart.cls:2726 `\let\and\@typeset@author@line`; 1909.10278, 2501.11185), as the block routes' names line
84/// cuts (OXIDIZED_DESIGN_DIVERGENCES #993). `None` for a call it doesn't cut. Only an `\and` at the call's own level, in
85/// the name its call wraps (`\lx@personname`, a Layout event holding the rest): one inside a layout container of the name
86/// (a `tabular`, a `\parbox`) or a note is that container's. And only where every piece holding ink holds a name: a piece
87/// of a note or a contact alone (`Ann Able \and \thanks{…}`) has no creator to hold it, so the call stays whole. The
88/// reading's (`Resolver::api`) and the emitter's (`emit::creator_call_cut`), on the same events.
89pub(in crate::base_utilities::frontmatter::observer) fn creator_call_pieces(
90  events: &[Event],
91) -> Option<Vec<&[Event]>> {
92  let own = creator_call_level(events);
93  let own_and = |e: &Event| e.nest == own && matches!(e.kind, EventKind::Sep(SepKind::And(_)));
94  let named = |piece: &[Event]| piece.iter().any(|e| e.kind == EventKind::Name);
95  let inked = |piece: &[Event]| {
96    piece.iter().any(|e| {
97      matches!(
98        e.kind,
99        EventKind::Name | EventKind::Mark | EventKind::Contact | EventKind::Note
100      )
101    })
102  };
103  if !events.iter().any(own_and) || !events.split(own_and).all(|p| named(p) || !inked(p)) {
104    return None;
105  }
106  let pieces: Vec<&[Event]> = events.split(own_and).filter(|piece| named(piece)).collect();
107  (pieces.len() > 1).then_some(pieces)
108}

110/// The pieces a numbered list's `\and`s cut a contact call into (`labelseq=<role>`: llncs's and aa's `\institute{A \and B}`,
111/// numbered 1, 2 as the class prints them, llncs.cls:870-885, OD #791), each holding a name or a contact: every `Sep(And)`
112/// of the call parts it, under conventions that number by place (`label_sequence`, [`Policy::label_sequence`]). `None`
113/// for a call it doesn't cut. The reading's ([`Resolver::api`]) and the emitter's (`emit::labelseq_call_cut`, which writes
114/// a call per piece: OXIDIZED_DESIGN_DIVERGENCES #995), on the same events.
115pub(in crate::base_utilities::frontmatter::observer) fn labelseq_pieces<'a>(
116  attrs: &str,
117  events: &'a [Event],
118  label_sequence: bool,
119) -> Option<Vec<&'a [Event]>> {
120  let and = |e: &Event| matches!(e.kind, EventKind::Sep(SepKind::And(_)));
121  let pieces = events.split(and).filter(|piece| {
122    piece
123      .iter()
124      .any(|e| e.kind == EventKind::Name || e.kind == EventKind::Contact)
125  });
126  (attr_value(attrs, "labelseq").is_some_and(|s| !s.is_empty())
127    && label_sequence
128    && events.iter().any(and))
129  .then(|| pieces.collect())
130}

149    }

152    }

155  /// The authors today's attaches an unlabelled entry of `role` to by its `annotate=` (Perl
156  /// Base_Utility.pool.ltxml:520-536): all, the run of new ones, the last N, else the last one; none before any.
157  pub(super) fn annotate_owners(&self, annotate: &str, role: &str) -> Vec<usize> {
158    let n = self.out.creators.len();
159    if n == 0 {
160      Vec::new()
161    } else if annotate == "all" {
162      (0..n).collect()
163    } else if annotate == "new" || annotate.starts_with("run") {
164      self.new_run(role)
165    } else if let Ok(k) = annotate.parse::<usize>() {
166      (n.saturating_sub(k)..n).collect()
167    } else {
168      vec![n - 1]
169    }
170  }

172  /// The authors an `annotate=run[:<cap>]` contact reaches, as `frontmatter_api.rs` records its run: the run of the
173  /// contact of the role before it while no author came since (the same count of authors; a `legend`'s the run's whole
174  /// extent), else the authors back from the last to one holding the role or made before the last contact of the role
175  /// that bounds a run (`run_from`), at most <cap> — and the run recorded for the next. Only a call today's attaches at
176  /// once records it: a mark-picked line (`_withsup`) waits for its label, and its run is not taken.
177  ///
178  /// A new run's walk stops at an author a mark answers once it has reached one no mark answers (`answered`): ANN
179  /// ABLE's 1 is answered by `\affil{$^1$Univ A}`, so `\author{ANN ABLE$^1$ and BOB BAKER \affil{$^1$Univ A} \affil{Univ
180  /// B}}` gives Univ B to Bob Baker alone, as today's API's walk (`reached_unanswered`) and the authblk rule for the same
181  /// print (review rev94a r7 SF2, n/n01). When every author of the run is answered the run is whole (n06, 77n's
182  /// authblk/ptephy/opex3 rows). Not a legend's walk, whose mark answers no one (`marker_leads`): the authors it passed
183  /// over are its extent, which a legend continuing the run reaches.
184  pub(super) fn record_run(&mut self, annotate: &str, role: &str, legend: bool) -> Vec<usize> {
185    let n = self.out.creators.len();
186    let cap = annotate
187      .split_once(':')
188      .and_then(|(_, c)| c.trim().parse::<usize>().ok())
189      .filter(|c| *c >= 1);
190    let (run, extent) = match &self.last_run {
191      Some((r, count, run, extent)) if r == role && *count == n => {
192        // (a legend within the run leaves its record as it was, as today's: `!(legend && continued)`)
193        if legend {
194          return extent.clone();
195        }
196        (run.clone(), extent.clone())
197      },
198      _ => {
199        let answering = self.answering(role);
200        let answered = |c: usize| {
201          !legend
202            && self.out.creators[c]
203              .requests
204              .iter()
205              .any(|r| answering.contains(r))
206        };
207        let (mut run, mut passed) = (Vec::new(), Vec::new());
208        let mut reached_unanswered = false;
209        // (the calls the run holds so far: the cap counts calls, `calls_from`)
210        let mut calls = 0usize;
211        for c in (0..n).rev() {
212          if !run.is_empty() && self.ends_a_run(c, role) {
213            break;
214          }
215          // (the cap, the authors queued when the contact was called, counts `\author` calls, and so ends the run only
216          // between two: a line of names the reading parts where the kernel's read welded them is one queued author,
217          // every name of the call the run's — `Ann Able$^*$ \\ Bob Baker$^*$ \\ Cat Cole$^\dagger$`, two queued over
218          // three names, whose first lost the `\affil`, 2609.33075 —, and so is a list the reading parts at its `\and`s
219          // or commas: `\author{Cy Cole}\author{Ann Able \and Bob Baker}\affil{Univ X}` is two queued over three names,
220          // all three under Univ X as pdflatex prints them, where a cap counting names gave Cy Cole none, silently)
221          let new_call = run.last().is_none_or(|last: &usize| {
222            self.out.creators[*last].source != self.out.creators[c].source
223          });
224          if new_call && cap.is_some_and(|cap| calls >= cap) {
225            break;
226          }
227          if reached_unanswered && answered(c) {
228            // (the authors Perl's `new` reaches past here, the run's extent: up to one holding the role, within the cap's
229            // calls left)
230            passed = self
231              .calls_from(
232                c,
233                run.last().copied(),
234                cap.map(|cap| cap.saturating_sub(calls)),
235              )
236              .take_while(|&c| !self.roles[c].iter().any(|r| r == role))
237              .collect();
238            break;
239          }
240          reached_unanswered |= !answered(c);
241          run.push(c);
242          calls += usize::from(new_call);
243        }
244        let mut extent: Vec<usize> = run.iter().chain(&passed).copied().collect();
245        run.reverse();
246        extent.sort_unstable();
247        (run, extent)
248      },
249    };
250    self.last_run = Some((role.to_string(), n, run.clone(), extent));
251    run
252  }

254  /// The creators back from `c`, within the first `left` calls (`None`: every call) past the creator `after` (the
255  /// one before them in the walk, whose call continues into them uncounted): a creator's call is its `source`, the
256  /// queued entry it is read from ([`Self::record_run`]'s cap).
257  fn calls_from(
258    &self,
259    c: usize,
260    after: Option<usize>,
261    left: Option<usize>,
262  ) -> impl Iterator<Item = usize> + '_ {
263    let mut source = after.map(|a| self.out.creators[a].source);
264    let mut left = left.unwrap_or(usize::MAX);
265    (0..=c).rev().take_while(move |&c| {
266      let at = self.out.creators[c].source;
267      if source != Some(at) {
268        if left == 0 {
269          return false;
270        }
271        left -= 1;
272        source = Some(at);
273      }
274      true
275    })
276  }

279  /// call's leading or `_withsup` mark (a legend so read of an affiliation call too: today's keeps it an affiliation) —,
280  /// not a call's `label=`: what today's API's run walk reads as `answering` (`frontmatter_api.rs`; authblk's
281  /// `\author[1]{Ann}\author{Bob}\affil[1]{Univ X}\affil{Univ Y}` prints Univ Y under both, 2605.25017).

284      .pending
285      .iter()
286      .filter(|(_, _, assoc)| {

297      .collect()

300  /// An API call made apart from the blocks.
301  pub(super) fn api(
302    &mut self,
303    index: usize,
304    tag: &str,
305    role: &str,
306    attrs: &str,
307    events: &[Event],
308    policy: &'static Policy,
309  ) {
310    // (a creator call its own `\and`s cut, each piece a creator of its own with the call's keyvals: `creator_call_pieces`)
311    if tag == "ltx:creator"
312      && let Some(pieces) = creator_call_pieces(events)
313    {
314      for piece in pieces {
315        self.api(index, tag, role, attrs, piece, policy);
316      }
317      return;
318    }
319    let led_line = self.led_line.take();
320    // (the title gives no creator or contact; its marks are requests: aastex's `\title{…\altaffilmark{1}}`, whose
321    // `\altaffiltext{1}{Based on observations …}` is the title's note, 0801.1670)
322    if tag == "ltx:title" {
323      for e in events.iter().filter(|e| e.kind == EventKind::Mark) {
324        // (a footnote mark, by the mark it prints: `\title{…\footnotemark[1]}` and its `\footnotetext[1]{…}`,
325        // 1111.4983)
326        if printed_by_footnotemark(e)
327          && let Some(mark) = e.label.as_deref()
328        {
329          self.title_fn_marks.push(mark.to_string());
330        }
331        self
332          .title_requests
333          .extend(mark_labels(e.label.as_deref().unwrap_or("")));
334      }
335      return;
336    }
337    // (a publication note gives no creator nor contact of its own, but the e-mail calls it makes are the authors' —
338    // A&A's `\offprints{B. Baker, \email{bob.baker@y.org}}`, "Send offprint requests to: B. Baker, e-mail: …",
339    // 0711.1152, 0802.0707; placed once every name is read, `answer_document_notes`, with the text the note prints
340    // before each, the name it may give. Only an e-mail call, the rehearsal's `\lx@fm@obs@email`: a link the note
341    // prints is the note's text, as today's output and TeX keep it — amsart's `\thanks{Correspondence to
342    // \href{mailto:carl@z.org}{…}}`, revtex's `\preprint` holding a link (review rev92a_r1 B1) —, and only a
343    // binding's: one the document's own `\email` makes (`\renewcommand{\email}[1]{\texttt{#1}}`, a class read raw)
344    // prints wherever its definition prints, so it is the note's text too (`EmailMeaning::Own`, the meaning when
345    // the note is rehearsed, at the flush, `\maketitle`; review rev92a_r3 B1: 1412.0748, 1809.01965; rev92a_r4 B1))
346    // (a note making a creator — amsart's `\thanks{… \author{Bob}\address{Univ B}\email{bob@b.org}}`, 0807.4720, its
347    // `EventKind::ApiCall` "ltx:creator" — is digested as Perl's: each contact in it goes to the creator newest when it is
348    // digested, `\lx@annotate@frontmatter` (Base_Utility.pool.ltxml:520-531), and no reading of a flush answers one —
349    // the first flush's, before that creator exists, gave Ann Bob's e-mail; `emitter::emit`'s pubnote wrapper keeps them)
350    if tag == "ltx:pubnote" {
351      if events.iter().any(is_creator_call) {
352        return;
353      }
354      let current = self.out.creators.len().checked_sub(1);
355      let mut before: Vec<&Event> = Vec::new();
356      for e in events {
357        if e.kind != EventKind::Contact {
358          before.push(e);
359          continue;
360        }
361        let printed = events_text(&std::mem::take(&mut before));
362        if e.role.as_deref() != Some("email") || e.meaning == EmailMeaning::Own {
363          continue;
364        }
365        let in_place = e.meaning == EmailMeaning::InPlace;
366        let (role, text) = contact_parts(e);
367        if role == "email" && !text.is_empty() {
368          self.document_emails.push(DocumentEmail {
369            assoc: Assoc {
370              role,
371              text,
372              label: None,
373              rule: String::new(),
374              source: Some((index, e.span)),
375              target: link_target(e),
376            },
377            printed,
378            in_place,
379            current,
380          });
381        }
382      }
383      return;
384    }
385    // (the role's record before this call, which this contact call updates: the group of names printed since the role's
386    // last call, `print_group`)
387    let reads = self.role_reads.get(role).copied();
388    if tag == "ltx:contact" {
389      self.read_role(role);
390    }
391    // (a class's sole institute under the names, `_share=authors` (llncs's, aa's): with several authors a later line of
392    // it printing addresses alone, outside the class's e-mail calls, is an e-mail line of the row, each address read
393    // apart and placed as a call's — the old split read the row so (`row_email_calls`, deleted at P27b; OD #901): aa
394    // 2506.20360's `\email{guolianglv@sina.com};{chunhuazhu@sina.cn}` gave the second address to the institute's
395    // text, which then differed from today's and held the writing back)
396    let row_lines;
397    let events = if shares_the_row(attrs) && self.out.creators.len() > 1 {
398      row_lines = row_address_lines(events);
399      &row_lines[..]
400    } else {
401      events
402    };
403    let split = self.splitting;
404    let withsup = attr_value(attrs, "_withsup") == Some("true");
405    // its leading marks (`\affil{$^1$Univ X}`), its `\label`s, its text (`events::leading_marks`)
406    let (start, lead_end) = events::leading_marks(events);
407    let between_marks = events::between_leading_marks;
408    let lead_marks: Vec<&Event> = events[start..lead_end]
409      .iter()
410      .filter(|e| e.kind == EventKind::Mark)
411      .collect();
412    let lead: Vec<String> = lead_marks
413      .iter()
414      .flat_map(|e| mark_labels(e.label.as_deref().unwrap_or("")))
415      .collect();
416    // (marks a `\footnotemark` printed: what a name's footnote mark answers — the same print, `mark:footnote`, 1607.07280)
417    let lead_footnote =
418      !lead_marks.is_empty() && lead_marks.iter().all(|e| printed_by_footnotemark(e));
419    let keys: Vec<String> = events
420      .iter()
421      .filter(|e| e.kind == EventKind::Layout && e.role.as_deref() == Some("label"))
422      .filter_map(|e| e.label.clone())
423      .map(|key| s!("ref:{key}"))
424      .collect();
425    // (its Contact events, an `\email` inside an `\institute` entry, are contacts of their own, with the entry's owners;
426    // an address it prints as text stays in its text here — `strip_addresses` split them off at this point and disagreed
427    // with 66 + 52 papers of the 5,000 —, taken out once every name is read, by the chain, where the writer can cut
428    // today's entry the same: `Resolver::take_out_entry_addresses`, brief 95c)
429    // (an entry of another role — an `\email`, a `\thanks` contact — is its declared role whatever it holds: a
430    // `mailto:` link inside an `\email` is the e-mail, review revfms2a2 finding 10, 2506.17875)
431    let splits_contacts =
432      tag != "ltx:creator" && matches!(role, "affiliation" | "address" | "altaffiliation");
433    // (only its e-mails and ORCIDs: a link inside an affiliation is its text, as today's — `\href{…}{Institut Langevin},
434    // \href{…}{ESPCI Paris}`, 2609.39586; `Stardust Labs, Ness Ziona, Israel; \url{https://www.stardust-initiative.com}`,
435    // 2605.14947; 2308.00495, 2111.07288)
436    // (the contacts its commands declare, `\email`, `\orcid`: a link is text there, an e-mail address or an ORCID
437    // record it points to too — amsart's `\address{… \\ \href{mailto:…}{…}}`, 2408.13873, 2305.04702)
438    // (and an ORCID inside an e-mail call, another kind of contact: `\email{giulio.pascale@unina.it \orcidlink{…}}`,
439    // 2402.04675 — the e-mail read "giulio.pascale@unina.it0000-0003-1680-3425")
