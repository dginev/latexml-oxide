5/// A root child that is title-page layout when it stands before the abstract: a
6/// pagebreak, a paragraph, or a `logical-block` whose content survives #244's demotion
7/// to `<block>`.
8pub(in crate::base_utilities) fn is_titlepage_layout(node: &Node) -> bool {
9  let q = document::get_node_qname(node);
10  q == pin_static("ltx:pagination")
11    || q == pin_static("ltx:para")
12    || (q == pin_static("ltx:logical-block") && subtree_is_block_demotable(node))
13}

15/// Move `run` (consecutive root children) into a new `<ltx:titlepage>` in place, in
16/// order; `para` → `block` as is (its content is already Block.model), `logical-block`
17/// demoted bottom-up then renamed. Returns the titlepage (its own handle stays live —
18/// only children are renamed).
19pub(in crate::base_utilities) fn wrap_as_titlepage(
20  document: &mut Document,
21  root: &Node,
22  run: &[Node],
23) -> Result<Node> {
24  // Bare tag + root namespace, as `maybe_promote_leading_title` does for <title>.
25  let mut titlepage = document.insert_element_before(&run[0], "titlepage", None)?;
26  if let Some(rns) = root.get_namespace() {
27    let _ = titlepage.set_namespace(&rns);
28  }
29  for mut n in run.iter().cloned() {
30    n.unlink();
31    titlepage.add_child(&mut n)?;
32  }
33  normalize_titlepage_layout_children(document, &titlepage)?;
34  Ok(titlepage)
35}

37/// The layout children of a title page as the schema's `titlepage_model`
38/// (`Block.class*`) — and `ltx:para` — hold them: a `center`'s `logical-block`
39/// and a `para` become `block`s (#244's demotion, content first, rename after).
40/// Shared by `wrap_as_titlepage` and `demote_stranded_titlepage`.
41pub(in crate::base_utilities) fn normalize_titlepage_layout_children(
42  document: &mut Document,
43  titlepage: &Node,
44) -> Result<()> {
45  for child in titlepage.get_child_nodes() {
46    if child.get_type() != Some(NodeType::ElementNode) {
47      continue;
48    }
49    let q = document::get_node_qname(&child);
50    if q == pin_static("ltx:logical-block") {
51      demote_para_class_content(document, &child)?;
52      document.rename_node(child, "ltx:block", true)?;
53    } else if q == pin_static("ltx:para") {
54      document.rename_node(child, "ltx:block", true)?;
55    }
56  }
57  Ok(())
58}

60/// Flush the queued frontmatter right after `after` (a root child), or — `None` — at
61/// the top, after `/ltx:document/ltx:resource[last()]` exactly as Perl's fallback
62/// (`Base_Utility.pool.ltxml:927-945`; Perl's `findnode` is document-relative, the Rust
63/// cached XPath context is root-relative, hence the absolute path). The `_Capture_`
64/// wrapper is filled by `insert_frontmatter` and unwrapped; the placed elements are
65/// returned (their handles survive the unwrap — `unwrap_nodes` splices children up).
66pub(in crate::base_utilities) fn insert_frontmatter_at(
67  document: &mut Document,
68  after: Option<&Node>,
69) -> Result<Vec<Node>> {
70  let savenode = document.get_node().clone();
71  let point = match after {
72    Some(a) => a.get_next_sibling(),
73    None => document
74      .findnode("/ltx:document/ltx:resource[last()]", None)
75      .and_then(|r| r.get_next_sibling()),
76  };
77  let wrapper = match point {
78    Some(point) => document.insert_element_before(&point, "ltx:_Capture_", None)?,
79    None => {
80      let parent = match after {
81        Some(a) => a.get_parent(),
82        None => document.get_document().get_root_element(),
83      };
84      match parent {
85        Some(mut parent) => document.open_element_at(&mut parent, "ltx:_Capture_", None, None)?,
86        None => return Ok(Vec::new()),
87      }
88    },
89  };
90  document.set_node(&wrapper);
91  insert_frontmatter(document)?;
92  let placed: Vec<Node> = wrapper
93    .get_child_nodes()
94    .into_iter()
95    .filter(|n| n.get_type() == Some(NodeType::ElementNode))
96    .collect();
97  document.unwrap_nodes(wrapper)?;
98  document.set_node(&savenode);
99  Ok(placed)
100}

102/// Re-attach, in order, every node of `movers` that precedes `tail` to right after it.
103/// Elements only, so `add_next_sibling` never meets libxml2's text-merge-and-free.
104pub(in crate::base_utilities) fn move_content_free_after(
105  movers: &[Node],
106  mut tail: Node,
107) -> Result<()> {
108  for mut m in movers.iter().cloned() {
109    if node_precedes(&m, &tail) {
110      m.unlink();
111      tail.add_next_sibling(&mut m)?;
112      tail = m;
113    }
114  }
115  Ok(())
116}

118/// True when `a` comes before `b` among the same parent's children.
119pub(in crate::base_utilities) fn node_precedes(a: &Node, b: &Node) -> bool {
120  let mut cur = a.get_next_sibling();
121  while let Some(n) = cur {
122    if n == *b {
123      return true;
124    }
125    cur = n.get_next_sibling();
126  }
127  false
128}

130/// A `<ltx:pagination>` (self-closing pagebreak) or a truly empty `<ltx:para>`/
131/// `<ltx:p>`/`<ltx:break>` wrapper — a node with no visible content. "Empty" means no
132/// text AND no *visible* element descendant: a `<graphics>`/`<rule>`/`<svg>`/
133/// `<tabular>`/`<text>`/image inside an otherwise text-empty para (a logo- or
134/// rule-only cover top) renders above the title in the PDF, so it is NOT content-free
135/// and must block the hoist — only nested empty `<para>`/`<p>`/`<break>` wrappers are
136/// transparent. See `place_frontmatter`.
137pub(in crate::base_utilities) fn node_is_content_free(node: &Node) -> bool {
138  let is_pagination = with(document::get_node_qname(node), |q| q == "ltx:pagination");
139  if is_pagination {
140    return true;
141  }
142  let is_wrapper = with(document::get_node_qname(node), |q| {
143    matches!(q, "ltx:para" | "ltx:p" | "ltx:break")
144  });
145  is_wrapper
146    && text_outside_error_markers(node).trim().is_empty()
147    && subtree_elements_all_invisible(node)
148}

150/// The text of `node`'s subtree with every `ltx:ERROR` subtree left out. An
151/// undefined-command marker (`<ERROR class="undefined">\foo</ERROR>`) carries the
152/// control-sequence NAME as its text — diagnostic ink, not document content (pdflatex
153/// had a definition and typeset nothing for it). A leading paragraph holding only such
154/// markers is therefore content-free for the frontmatter hoist (OXIDIZED_DESIGN #258;
155/// pst-calendar-doc's `\DeclareDocumentMetadata`, forest-doc's `\@escapeifif`,
156/// pmhanguljamo's `\fontid`, 8 docs): the marker stays in the output, relocated below
157/// the frontmatter like any other content-free mover. A marker whose ARGUMENTS were
158/// typeset as text (`\foo{Real words}` → `<ERROR>\foo</ERROR>Real words`) is not
159/// content-free — that text is visible and stays where it is.
160fn text_outside_error_markers(node: &Node) -> String {
161  fn walk(node: &Node, out: &mut String) {
162    for c in node.get_child_nodes() {
163      match c.get_type() {
164        Some(NodeType::TextNode) => out.push_str(&c.get_content()),
165        Some(NodeType::ElementNode) if !is_undefined_command_marker(&c) => {
166          walk(&c, out);
167        },
168        _ => {},
169      }
170    }
171  }
172  let mut out = String::new();
173  walk(node, &mut out);
174  out
175}

177/// `<ltx:ERROR class="undefined">\foo</ltx:ERROR>` — the marker `make_error` leaves for
178/// an undefined control sequence; its text is the CS NAME. Only this class is
179/// diagnostic ink: `\lx@ERROR{cls}{text}` markers and constructor failures carry
180/// arbitrary text and stay visible content.
181fn is_undefined_command_marker(node: &Node) -> bool {
182  document::get_node_qname(node) == pin_static("ltx:ERROR")
183    && node.get_attribute("class").as_deref() == Some("undefined")
184}

186/// Every element descendant of `node` is itself a structural wrapper
187/// (`ltx:para`/`ltx:p`/`ltx:break`, or an ink-free `ltx:text`) — i.e. the subtree
188/// renders nothing visible. Any other element (`ltx:graphics`, `ltx:rule`, `ltx:svg`,
189/// `ltx:tabular`, an image, …) is visible content even with no text. An `ltx:text` is
190/// transparent only when it can draw nothing on its own: the empty `<ltx:text>` an
191/// `\hbox{}`/`\null`/`\mbox{}` leaves in a paragraph (`\cleardoublepage`'s blank
192/// page; memoir `\frontmatter`) is folded away by `auto_collapse_children`/`finalize`
193/// later anyway, but a `\fbox{}`/`\colorbox{..}{}` — `framed`/`backgroundcolor`, or a
194/// `class`/`cssstyle` a stylesheet may paint — has ink and stays visible. Text nodes
195/// are ignored here: the caller has already required the subtree's text to be
196/// whitespace-only, and LaTeXML wraps loose document text in `<para><p>`, so a bare
197/// non-whitespace text child does not occur at this level.
198fn subtree_elements_all_invisible(node: &Node) -> bool {
199  node.get_child_nodes().iter().all(|c| {
200    c.get_type() != Some(NodeType::ElementNode)
201      // An undefined-command marker is diagnostic ink (`text_outside_error_markers`).
202      || is_undefined_command_marker(c)
203      || ((with(document::get_node_qname(c), |q| {
204        matches!(q, "ltx:para" | "ltx:p" | "ltx:break")
205      }) || (document::get_node_qname(c) == pin_static("ltx:text") && text_is_ink_free(c)))
206        && subtree_elements_all_invisible(c))
207  })
208}

210/// An `ltx:text` with none of the attributes that can paint without content.
211fn text_is_ink_free(text: &Node) -> bool {
212  ![
213    "framed",
214    "framecolor",
215    "backgroundcolor",
216    "class",
217    "cssstyle",
218  ]
219  .iter()
220  .any(|attr| text.has_attribute(attr))
221}

223/// Is `node` one of the document model's FIRST-group elements (`LaTeXML-structure.rnc:34`),
224/// read from the schema: admitted by `ltx:document` but not by body-only content
225/// (`ltx:sectional-block`, model `document.body.class*`), and — since `ltx:document` also
226/// admits BackMatter.class (bibliography/appendix/index/glossary) that the body group
227/// takes — either `titlepage` or something `ltx:bibliography` admits (its model LEADS
228/// with exactly `FrontMatter.class*, SectionalFrontMatter.class*`). A Meta.class element
229/// (allowed in both groups) is NOT one. Used by `place_frontmatter` (#247).
230pub(in crate::base_utilities) fn is_frontmatter_group_element(node: &Node) -> bool {
231  let q = document::get_node_qname(node);
232  document::can_contain_qsym(pin_static("ltx:document"), q)
233    && !document::can_contain_qsym(pin_static("ltx:sectional-block"), q)
234    && (q == pin_static("ltx:titlepage")
235      || document::can_contain_qsym(pin_static("ltx:bibliography"), q))
236}

239/// title block for `maybe_promote_leading_title` and `maybe_dedup_leading_title_ink`.
240fn is_sectional_qname(q: SymStr) -> bool {
241  with(q, |q| {
242    matches!(
243      q,
244      "ltx:section"
245        | "ltx:subsection"
246        | "ltx:subsubsection"
247        | "ltx:paragraph"

250        | "ltx:part"
251        | "ltx:appendix"
252        | "ltx:bibliography"
253        | "ltx:index"
254        | "ltx:glossary"
255    )
256  })

259/// Beyond-Perl heuristic: recover a document title from a hand-formatted leading
260/// block when the author never used `\title`/`\maketitle`.
261///
262/// Many papers set their title as a plain `\begin{center}{\Large ...}\end{center}`
263/// block instead of the frontmatter machinery, then declare only an abstract (e.g.
264/// arXiv 1609.07638). With no `\title` there is no `ltx:title` frontmatter to
265/// flush, so the document renders titleless. This promotes that first block — gated
266/// conservatively so it does not fire on epigraphs, dedications, centered figures or
267/// "draft" notices — into a real `<ltx:title>` at the document top.
268///
269/// Called only from the ABSTRACT-ONLY `\lx@frontmatter@fallback` branch: reaching it
270/// already means no `\title`/`\author`/`\maketitle` were used yet the paper carries
271/// frontmatter (an abstract) — the strongest signal a leading display block is the
272/// title. It fires only when no `<ltx:title>` exists yet, the first non-resource body
273/// element holds a leading centered `<ltx:p>` set in a larger-than-body font, and
274/// that paragraph has non-whitespace text. On a match the paragraph's inline children
275/// are MOVED into a fresh `<ltx:title>` after the resources, pruning empty wrappers.
276pub(in crate::base_utilities) fn maybe_promote_leading_title(
277  document: &mut Document,
278) -> Result<Option<Node>> {
279  // Never override an existing (real) title.
280  if document.findnode("/ltx:document/ltx:title", None).is_some() {
281    return Ok(None);
282  }
283  // The first non-resource body element (the candidate title block). Fetch it
284  // with an ABSOLUTE query: nodes returned from a RELATIVE-context findnode are
285  // detached for child traversal (a rust-libxml shared-node artifact —
286  // get_content works but get_child_nodes is empty), so every step below walks
287  // the live DOM by hand from this anchor.
288  let Some(first_body) = document.findnode("/ltx:document/*[not(self::ltx:resource)][1]", None)
289  else {
290    return Ok(None);
291  };
292  // Never DESCEND INTO SECTIONAL content: on the second fallback pass (fired
293  // at the next \section) the first body element can be the COMPLETED first
294  // section, and an unconstrained DFS would steal a centered display-font
295  // paragraph from inside it (e.g. a "Dedicated to ..." block) as the paper
296  // title. A hand-formatted title block is a plain para/quote at document
297  // level — reject sectional anchors outright (PR_READINESS must-fix 3).
298  let anchor_q = document::get_node_qname(&first_body);
299  if is_sectional_qname(anchor_q) {
300    return Ok(None);
301  }
302  // First centered <ltx:p> in document order within the block; it must be set in
303  // a larger-than-body font (the display-title signal) and hold real text.
304  // Conservative rule (user ruling 2026-09-20, OXIDIZED_DESIGN #246): promote ONLY the
305  // unambiguous hand-made-title shape — a plain `center`/paragraph block at document
306  // level whose FIRST paragraph is centered and set in a display font that NO other
307  // paragraph of the block shares. Corpus counter-examples that a looser rule mis-read
308  // as titles: tikz-mirror-lens (author-first cover, `\Large{#1}` leaking onto the
309  // title paragraph → two display paragraphs → ambiguous), isosigns/bootstrapicons
310  // (a `VERSION 2.1` badge inside a tcolorbox → nested box), tipfr-doc (its real
311  // `tipfr.sty` title also sits in a tcolorbox — declined rather than coin-flipped).
312  // Whatever is not promoted stays hand-typeset layout and is wrapped as
313  // `<ltx:titlepage>` by `place_frontmatter (the titlepage wrap)`.
314  let is_plain_block = with(anchor_q, |q| matches!(q, "ltx:logical-block" | "ltx:para"));
315  if !is_plain_block {
316    return Ok(None);
317  }
318  let Some(title_p) = first_plain_paragraph(&first_body) else {
319    return Ok(None);
320  };
321  if title_p.get_attribute("align").as_deref() != Some("center") {
322    return Ok(None);
323  }
324  let nominal = {
325    // #542: NOMINAL_FONT_SIZE is a float (11pt = 10.95), not an int — mirror
326    // `common::font::defsize`, which reads it via lookup_float, not lookup_int.
327    let v = lookup_float("NOMINAL_FONT_SIZE").map_or(0.0, |f| f.0);
328    if v > 0.0 { v } else { 10.0 }
329  };
330  if !title_p.get_content().chars().any(char::is_alphabetic)
331    || !descendant_has_display_font(document, &title_p, nominal)
332  {
333    return Ok(None);
334  }
335  // Uniqueness: any OTHER plain paragraph of the block in a display font makes the
336  // title ambiguous (author-first covers, multi-line banners).
337  let mut paragraphs = Vec::new();
338  collect_plain_paragraphs(&first_body, &mut paragraphs);
339  if paragraphs
340    .iter()
341    .any(|p| *p != title_p && descendant_has_display_font(document, p, nominal))
342  {
343    return Ok(None);
344  }
345  // Create <ltx:title> at the document top (after the last resource), mirroring
346  // the frontmatter fallback's placement.
347  let mut point = document.findnode("/ltx:document/ltx:resource[last()]", None);
348  if let Some(p) = point.take() {
349    point = p.get_next_sibling();
350  }
351  // No resources (or resource is the last child): insert before the first
352  // body element instead of silently declining (review corner-case).
353  let point = point.unwrap_or_else(|| first_body.clone());
354  // Create with a BARE tag + bind the root (default LaTeXML) namespace, so it
355  // serializes as <title> like a real frontmatter title — a prefixed "ltx:title"
356  // qname would emit a stray <ltx:title>. Mirrors open_element_internal's
357  // default-namespace path.
358  let mut title = document.insert_element_before(&point, "title", None)?;
359  if let Some(rns) = document
360    .get_document()
361    .get_root_element()
362    .and_then(|r| r.get_namespace())
363  {
364    let _ = title.set_namespace(&rns);
365  }
366  // MOVE the paragraph's inline children into the title (a true move preserves
367  // xml:ids, so any footnotes/marks carry over rather than being cloned+orphaned).
368  for mut child in title_p.get_child_nodes() {
369    child.unbind();
370    title.add_child(&mut child)?;
371  }
372  // Drop the now-empty paragraph, and any wrapper ancestor it leaves empty
373  // (para → logical-block), so we don't strand empty blocks above the abstract.
374  let mut victim = title_p;
375  loop {
376    let parent = victim.get_parent();
377    document.remove_node(victim);
378    match parent {
379      Some(p) if !has_element_child(&p) => {
380        let is_wrapper = with(document::get_node_qname(&p), |name| {
381          name == "ltx:para" || name == "ltx:logical-block"
382        });
383        if is_wrapper {
384          victim = p;
385          continue;
386        }
387      },
388      _ => {},
389    }
390    break;
391  }
392  // The promoted <title>: the node a deferred frontmatter flush lands AFTER when no
393  // titlepage layout follows it. A surviving rest of the hand-formatted block (an
394  // author paragraph) is wrapped as <ltx:titlepage> by the caller. `first_body` may
395  // have been freed above — never return it.
396  Ok(Some(title))
397}

399/// Whitespace-collapsed, lowercased text — so a `<break>` (`\\`) and any spacing
400/// differences between the structured title and its hand-typeset copy don't
401/// defeat the comparison in [`maybe_dedup_leading_title_ink`].
402pub(in crate::base_utilities) fn normalize_frontmatter_text(s: &str) -> String {
403  s.split_whitespace()
404    .collect::<Vec<_>>()
405    .join(" ")
406    .to_lowercase()
407}

409/// Companion to [`maybe_promote_leading_title`] for the mirror case
410/// (arXiv/html_feedback#6924, witness arXiv 2608.10928): a structured
411/// `<ltx:title>` DOES exist (from `\title`), but the author ALSO hand-typeset the
412/// title as a leading centered display-font block and never called `\maketitle`,
413/// so that block reproduces the structured title and the title renders twice.
414///
415/// Prioritize the structured metadata (LaTeXML's unified Frontmatter API stays
416/// authoritative): remove the redundant leading title *ink*, keeping the semantic
417/// `<ltx:title>` and any author/abstract ink (which has no structured counterpart,
418/// so is the only copy). Fires only on a FULL normalized-text match against the
419/// structured title, and only for a leading, non-sectional, display-font centered
420/// paragraph of the document-level title block (not inside a float, a titlepage or any
421/// other container) — so an unrelated centered display block is never removed — and at
422/// most once per document, at the first flush that sees the structured title (68s;
423/// guard `perfect_kernel_batch68s`). Reuses the same detection helpers as the promote path.
424///
425/// It is the LAST RESORT of the title placement rules (OXIDIZED_DESIGN #146): a text match
426/// where no print event explains the copy — a print of the marked `\@title` is lifted by its marks first
427/// ([`lift_marked_field_ink`], OXIDIZED_DESIGN #715), and then this does not run (`lx_title_ink_dedup_done`).
428/// Never in a document with an `<ltx:titlepage>`: there the XSLT does not display the structured
429/// title (LaTeXML-structure-xhtml.xsl, the `ltx:title` template's
430/// `not(parent::*/child::ltx:titlepage)`; Perl's copy :358-359) — the cover IS the displayed title,
431/// and deleting its copy left none (1304.0681, 1503.03277: no `<h1>`, Perl shows the title once).
432pub(in crate::base_utilities) fn maybe_dedup_leading_title_ink(
433  document: &mut Document,
434) -> Result<()> {
435  // Needs a structured document title to prioritize over.
436  let Some(title) = document.findnode("/ltx:document/ltx:title", None) else {
437    return Ok(());
438  };
439  let title_text = normalize_frontmatter_text(&title.get_content());
440  if title_text.is_empty() {
441    return Ok(());
442  }
443  // Once per document: the hand-typeset title block is judged at the first flush that sees the
444  // structured title — the fallback runs at EVERY sectioning command (`\@startsection@hook`) and
445  // at `\end{document}`, and a second look removed a second copy each time (one per section), down
446  // into the body. A copy still there after that first look is the author's ink, not the title.
447  if lookup_bool("lx_title_ink_dedup_done") {
448    return Ok(());
449  }
450  assign_value("lx_title_ink_dedup_done", true, Some(Scope::Global));
451  // A titlepage beside the title: the structured title is metadata and the cover the display.
452  if document
453    .findnode("/ltx:document/ltx:titlepage", None)
454    .is_some()
455  {
456    return Ok(());
457  }
458  let nominal = {
459    let v = lookup_float("NOMINAL_FONT_SIZE").map_or(0.0, |f| f.0);
460    if v > 0.0 { v } else { 10.0 }
461  };
462  // Leading centered paragraphs of the document-level title block (before the first sectional
463  // unit, reached only through title-block wrappers — a para, a titlepage, a cover minipage;
464  // never inside a figure or any other container, whose content is the author's).
465  // The nodes come from a walk of the live DOM from the absolutely queried root, so
466  // they support child traversal (shared-node caveat in `maybe_promote_leading_title`).
467  let Some(root) = document.findnode("/ltx:document", None) else {
468    return Ok(());
469  };
470  let candidates = leading_centered_paragraphs(&root);
471  // Remove the FIRST leading centered display-font block that EXACTLY reproduces
472  // the structured title. Only one — a paper may repeat the title text later
473  // (e.g. a running head); we drop just the redundant hand-typeset title.
474  let Some(victim) = candidates.into_iter().find(|p| {
475    descendant_has_display_font(document, p, nominal)
476      && normalize_frontmatter_text(&p.get_content()) == title_text
477  }) else {
478    return Ok(());
479  };
480  // Log what we drop (never remove content silently; #6924).
481  Info!(
482    "frontmatter",
483    "title_ink_dedup",
484    s!(
485      "dropped a hand-typeset title block duplicating the structured <ltx:title> (\"{title_text}\")"
486    )
487  );
488  // Remove the redundant title ink, pruning any title-block wrapper it leaves empty (para →
489  // logical-block, a minipage that held only the title and the paragraph around it) —
490  // mirroring the promote path's cleanup. Sibling `<p>`s (e.g. the author block) keep their
491  // wrapper non-empty and survive. A wrapper that carries `labels` (a `\label` target) or a
492  // frame (`\fbox{\begin{minipage}…}`, `framed`) is kept, empty: removing it would leave a
493  // dangling reference or drop ink of its own (68s review probes p7_fbox_cover, p8).
494  let mut victim = victim;
495  loop {
496    let parent = victim.get_parent();
497    document.remove_node(victim);
498    match parent {
499      Some(p)
500        if is_title_block_wrapper(document::get_node_qname(&p))
501          && !has_element_child(&p)
502          && p.get_content().trim().is_empty()
503          && !p.has_attribute("labels")
504          && !p.has_attribute("framed") =>
505      {
506        victim = p;
507      },
508      _ => break,
509    }
510  }
511  Ok(())
512}

514/// The containers a hand-typeset title block is set in at document level: paragraph wrappers
515/// (`ltx:para`, `ltx:logical-block`), a display block (`ltx:block`, a `{center}` holding a logo),
516/// a cover minipage (`ltx:inline-block`, inside the `ltx:p` of its paragraph — entered only
517/// through a cover paragraph, [`is_cover_box_paragraph`]). Not a float, a list, a theorem, a
518/// table or a sectional unit: what is set there is the author's content even when it repeats
519/// the title. Not a `{titlepage}` (`ltx:titlepage`) either: the cover is the displayed title
520/// there, and the structured title is metadata ([`maybe_dedup_leading_title_ink`]).
521fn is_title_block_wrapper(q: SymStr) -> bool {
522  with(q, |q| {
523    matches!(
524      q,
525      "ltx:para" | "ltx:logical-block" | "ltx:block" | "ltx:inline-block" | "ltx:p"
526    )
527  })
528}

530/// A non-centered `ltx:p` holding only boxes: its element children are all `ltx:inline-block`s
531/// and its text is whitespace — the `\noindent\begin{minipage}…\end{minipage}` cover paragraph
532/// (2010.07703). A paragraph of running prose with a `\parbox` in it is not: the box's words
533/// are the sentence's (68s review probe p2_parbox_only, pdflatex prints them mid-sentence).
534fn is_cover_box_paragraph(p: &Node) -> bool {
535  let mut has_box = false;
536  for c in p.get_child_nodes() {
537    match c.get_type() {
538      Some(NodeType::ElementNode) => {
539        if document::get_node_qname(&c) != pin_static("ltx:inline-block") {
540          return false;
541        }
542        has_box = true;
543      },
544      Some(NodeType::TextNode) if !c.get_content().trim().is_empty() => return false,
545      _ => {},
546    }
547  }
548  has_box
549}

551/// The `<ltx:p align="center">` paragraphs of `node` reached only through title-block wrappers
552/// ([`is_title_block_wrapper`]), in document order; a centered paragraph is taken whole (a
553/// minipage inside it is part of its text), any other `ltx:p` is entered only when it is a
554/// cover paragraph of boxes ([`is_cover_box_paragraph`]).
555fn collect_title_block_paragraphs(node: &Node, out: &mut Vec<Node>) {
556  // A framed wrapper (`\fbox{\begin{minipage}…}`, `framed`) is the author's own ink around the title: removing the copy
557  // inside left an empty frame the PDF does not have, so it is never searched (main-session ruling 2026-10-16, until
558  // the last-resort step promotes the author's block in place; OXIDIZED_DESIGN #146).
559  if node.has_attribute("framed") {
560    return;
561  }
562  let q = document::get_node_qname(node);
563  if q == pin_static("ltx:p") {
564    if node.get_attribute("align").as_deref() == Some("center") {
565      out.push(node.clone());
566      return;
567    }
568    if !is_cover_box_paragraph(node) {
569      return;
570    }
571  }
572  if !is_title_block_wrapper(q) {
573    return;
574  }
575  for c in node.get_child_nodes() {
576    if c.get_type() == Some(NodeType::ElementNode) {
577      collect_title_block_paragraphs(&c, out);
578    }
579  }
580}

582/// The `<ltx:p align="center">` paragraphs of the document-level title block: those reached from
583/// `root`'s children only through title-block wrappers ([`collect_title_block_paragraphs`]) and
584/// before its first sectional child (a back-matter list is passed over), in document order. A
585/// paragraph inside a float, a list, a theorem, a titlepage or any other container, or inside a
586/// chapter or part, is not the leading title block: the walk once took every centered paragraph
587/// before the first `<ltx:section>`, so a figure's centered line and a chapter's opening block were
588/// removed as the title (68p review probe `float.tex`; 2608.10928 the witness of the rule; a cover
589/// minipage 2010.07703; the titlepage covers 1304.0681, 1503.03277, 1601.07172, 2606.13892 are kept
590/// whole since 68s round 2). Only the document's children are visited and the walk stops at the
591/// first sectional unit, so it stays linear in the leading block (it runs once per document; the
592/// old XPath scanned `preceding::` for every centered paragraph at every sectioning command —
593/// 2208.14314: 121 of 133 s; 2108.10297: 113 of 146 s; PERFORMANCE.md Principle 6).
594fn leading_centered_paragraphs(root: &Node) -> Vec<Node> {
595  let mut out = Vec::new();
596  let mut cur = root.get_first_child();
597  while let Some(node) = cur {
598    if node.get_type() == Some(NodeType::ElementNode) {
599      let q = document::get_node_qname(&node);
600      // WORKAROUND, not a rule of the title block: a bibliography, index or glossary list placed
601      // ahead of the cover does not end it. 2010.07703's bibliography (`\bibliography` at
602      // thesis.tex:480, under `\part{Listings and Bibliography}`) is built at the head of the XML,
603      // before its cover minipage, by bibentry's `\nobibliography*` in the preamble
604      // (texpreamble.tex:820) — a separate defect (RED index-bib/nobibliography_star_prints_nothing,
605      // SYNC_STATUS); once that is fixed this skip has nothing to pass over.
606      let back_matter = with(q, |q| {
607        matches!(q, "ltx:bibliography" | "ltx:index" | "ltx:glossary")
608      });
609      if is_sectional_qname(q) && !back_matter {
610        break;
611      }
612      collect_title_block_paragraphs(&node, &mut out);
613    }
614    cur = node.get_next_sibling();
615  }
616  out
617}
