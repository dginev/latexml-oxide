7/// Where the contacts a class lists apart from the names, in the order the document gives them
8/// (`\lx@add@contact@placed[owner=sequence]`: jheppub's `\emailAdd`, iopart's `\ead`), queued in `commands`, go: each
9/// the label of its creator, or empty to stay where `\lx@add@contact@placed` put it (the group's only author when it was
10/// given, else the author block). Read against the authors of the same flush, all of them, wherever the list stands
11/// among them (2108.07972's `\emailAdd{kadota.kenji@…}` before the four `\author`s):
12/// 1. the author the address spells ([`address_owners`]);
13/// 2. else the n-th author, the address at the n-th place — Perl's jheppub.sty.ltxml rule (`labelseq=author`) — when the
14///    list is one place for each author and in their order: no address spells an author other than the one at its
15///    place, as [`addresses_in_order`](crate::base_utilities::defs::frontmatter_api::email_lines::addresses_in_order) reads an author block's line (1303.3040's `zhl61@` for Zhen Liu, 2011.02499's
16///    `tpsg4@`/`tpsr@`). The place is the address's, or else the call's (an `\emailAdd{a, b}` of one author's two
17///    addresses). 2407.07862's `eichhorn@` listed first for Astrid Eichhorn, author 2, and 1804.07528's three addresses
18///    each one place on are no list in author order. The user ruled the consistent list evidence (2026-10-21);
19///    [`author_order_step`] is this step, alone;
20/// 3. else where `\lx@add@contact@placed` put it: the group's only author ([`placed_contact_owner`]), or the author
21///    block;
22/// 4. else, for the one address of the list left on the block of a multi-author flush, the ONE author marked as the
23///    corresponding author — by a note naming them so, or a mark whose legend says so —, among the authors of its
24///    group and, where the document places its e-mails by address block, those requesting its address's mark
25///    ([`marked_corresponding_author`]; user ruling 2026-10-21: 1901.08773's `beril@` A. Kopylov's, 2510.11030's
26///    `hjhspace223@` J.-H. Ha's, 2211.07892's `yoomin@` Y. M. Oh's; 2308.06532's `zyxu@` Zhaoyi Xu's, its address
27///    saying "(Corresponding author)" and he alone carrying a `*`; 1802.03677's `gear8mike@` stays the block's, the
28///    number of its "Corresponding author." note on two authors, and 2111.00753's `lad@`, two legend-less `*`s);
29/// 5. a flush with no authors, its list given after `\maketitle` digested them (1903.05554), keeps Perl's rule.
30///
31/// Step 3, for iopart's `\ead`, places after the spelling with the one author requesting the mark of the address block
32/// it is printed under (`mark=address`, `\lx@address@mark@set`: 2407.14305's `rockfish3141@` Hidetomo Hoshino's as printed —
33/// today the block's, RED `iopart_ead_under_its_address_mark_red` —
34/// 2311.07087's `sgess@` Spencer Gessner's) — weaker than a list in author order, which the PDF's address blocks do not
35/// contradict (1902.07874's `\ead{…, b97202056@gmail.com, …}` of five authors after `\address{$^9$ …}`, Che-Yu Chen's
36/// at place 3; 1710.02251's three `\mailto`s in an `\eads` after `$^{(2)}$`), and only in a document that writes an
37/// e-mail between its addresses: one whose e-mails all follow its addresses writes them where IOP's guidelines ask,
38/// "after the postal address(es)", under the last address by position only (0910.0753's `wolfgang.muender@` for W
39/// Münder, 1111.3344's `knarf@` for Frank Löffler, 1911.06133's `ledou@` for Pierre Le Doussal, each after the last
40/// author's `\address`; `_unmarked`).
41///
42/// The PDF prints the list as one shared line (jheppub.sty:127-131) or each address as a block under the address before
43/// it (iopart.cls:363), so the spelling, the marks and the order are the only evidence (OXIDIZED_DESIGN_DIVERGENCES #805,
44/// #806, #808, #809).
45pub(in crate::base_utilities) fn sequenced_contact_labels(
46  commands: &[RawFrontmatter],
47  names: &[Tokens],
48) -> Result<Vec<String>> {
49  // (a document that writes an e-mail between its addresses places each under its address, iopart.cls:363; one whose
50  // addresses all come first writes its e-mails "after the postal address(es)", as IOP's guidelines ask, under the
51  // last address only by position: there an address mark places nothing, `_unmarked`)
52  let Some(first_listed) = commands
53    .iter()
54    .position(|entry| entry.0 == "ltx:contact" && entry.1.contains_key("_sequence"))
55  else {
56    return Ok(Vec::new());
57  };
58  // (an amsart-style class's `\address` is an address too: lmcs's `\email` under each lettered `\address`, 2306.08266)
59  let interleaved = commands[first_listed..].iter().any(|entry| {
60    entry.0 == "ltx:contact"
61      && matches!(
62        entry.1.get("role").map(String::as_str),
63        Some("affiliation" | "address")
64      )
65  });
66  // (the names as they print, one a position, as `queued_author_name_keys` reads the queue's: `queued_position_names`)
67  let mut keys: Vec<NameKeys> = Vec::new();
68  for name in names {
69    keys.push(printed_name_keys(name, None)?);
70  }
71  // Each address with the class command's call it came from (`\lx@sequence@call`: `\emailAdd{a, b}` is one call of two
72  // addresses), the author it spells, whether it is on the block where `\lx@add@contact@placed` put it, and where it
73  // goes when its address mark places nothing.
74  let listed: Vec<(i64, Option<usize>, bool, Option<&String>)> = commands
75    .iter()
76    .filter(|entry| entry.0 == "ltx:contact" && entry.1.contains_key("_sequence"))
77    .map(|entry| {
78      let call = entry
79        .1
80        .get("_sequence")
81        .and_then(|n| n.parse().ok())
82        .unwrap_or(0);
83      let spelled = match address_owners(&keys, &last_group(&entry.2)).as_slice() {
84        [at] => Some(*at),
85        _ => None,
86      };
87      let unmarked = entry.1.get("_unmarked").filter(|_| !interleaved);
88      let placed = unmarked.or_else(|| entry.1.get("label"));
89      let blocked = placed.map(String::as_str) == Some("addresses:block");
90      (call, spelled, blocked, unmarked)
91    })
92    .collect();
93  let owners: Vec<(i64, Option<usize>)> = listed
94    .iter()
95    .map(|(call, owner, ..)| (*call, *owner))
96    .collect();
97  let places = author_order_step(&owners, keys.len());
98  let mut labels: Vec<String> = owners
99    .iter()
100    .enumerate()
101    .map(|(at, (_, owner))| match (owner, &places) {
102      (Some(owner), _) if !keys.is_empty() => s!("author:{}", owner + 1),
103      (_, Some(places)) if keys.is_empty() || places[at] < keys.len() => {
104        s!("author:{}", places[at] + 1)
105      },
106      _ => listed[at].3.cloned().unwrap_or_default(),
107    })
108    .collect();
109  let left: Vec<usize> = (0..listed.len())
110    .filter(|at| match labels[*at].as_str() {
111      "" => listed[*at].2,
112      label => label == "addresses:block",
113    })
114    .collect();
115  if let ([at], true) = (left.as_slice(), keys.len() > 1) {
116    // (among the authors it may be of: its author group's, and of those the ones requesting its address's mark, when
117    // the document places its e-mails by address block — `_group`, `_marks`, `\lx@add@contact@placed`)
118    let entry = commands
119      .iter()
120      .filter(|entry| entry.0 == "ltx:contact" && entry.1.contains_key("_sequence"))
121      .nth(*at);
122    let positions = |key: &str| -> Option<Vec<usize>> {
123      let value = entry?.1.get(key)?;
124      Some(
125        value
126          .split(';')
127          .filter_map(|n| n.trim().parse::<usize>().ok())
128          .filter_map(|n| n.checked_sub(1))
129          .collect(),
130      )
131    };
132    let group = positions("_group");
133    let marks = positions("_marks").filter(|marks| interleaved && !marks.is_empty());
134    let within: Option<Vec<usize>> = match (group, marks) {
135      (Some(group), Some(marks)) => {
136        Some(group.into_iter().filter(|at| marks.contains(at)).collect())
137      },
138      (Some(only), None) | (None, Some(only)) => Some(only),
139      (None, None) => None,
140    };
141    let address = entry
142      .map(|entry| visible_name_text(last_group(&entry.2).unlist_ref()))
143      .unwrap_or_default();
144    if let Some(author) = marked_corresponding_author(commands, &address, within.as_deref()) {
145      labels[*at] = s!("author:{}", author + 1);
146    }
147  }
148  Ok(labels)
149}

151/// The flush's e-mails printed in a block several names share, for a flush the reading does not write (OD #970 G6): an
152/// e-mail entry, or a note holding one, after a run of ≥ 2 names goes to the name it spells — acmart (2310.06444,
153/// 2601.17008, 1907.07581, 2504.00797), Wiley `\corres` (2311.07985, 2407.08541, 2505.23112, 2503.06763); one after a
154/// single name stays its own (2208.05647, 2306.14314, 2609.11148), as does an address two own calls give (2205.01906,
155/// 2211.05700). Read per entry in `\lx@annotate@frontmatter@now` ([`ENTRY_RESPELL`]).
156pub(in crate::base_utilities) fn shared_block_email_labels(
157  commands: &[RawFrontmatter],
158  emitted: bool,
159) -> Result<Vec<Option<(String, String)>>> {
160  // (a creator the emitter wrote, its keys in its call alone, is one of the reading's authors)
161  let author = |entry: &RawFrontmatter| {
162    entry.0 == "ltx:creator" && entry.1.get("role").is_none_or(|role| role == "author")
163  };
164  // (a name's note: a note, or a contact of role `note`, acmart's `\authornote`)
165  let note = |entry: &RawFrontmatter| {
166    entry.0 == "ltx:note" || entry.1.get("role").map(String::as_str) == Some("note")
167  };
168  let mut out = vec![None; commands.len()];
169  let numbered = lookup_mapping_int("num_ltx:creator", "author");
170  for (at, entry) in commands.iter().enumerate() {
171    let positional = entry.0 == "ltx:contact"
172      && entry.1.get("role").map(String::as_str) == Some("email")
173      && entry
174        .1
175        .get("label")
176        .is_none_or(|label| label.trim().is_empty())
177      && !entry.1.contains_key("_sequence");
178    // (where the reading wrote the flush, only an entry it left as queued: a document's note, not a creator's call)
179    if (!positional && entry.0 != "ltx:note")
180      || (emitted && resolver::creator_call(&entry.0, &annotated_parent(&entry.2)))
181    {
182      continue;
183    }
184    // (the run of names before it, back over the entries between; a name's note, acmart's `\authornote`, ends no run:
185    // 1907.07581)
186    let end = (0..at)
187      .rev()
188      .find(|&k| author(&commands[k]))
189      .map_or(0, |k| k + 1);
190    let start = (0..end)
191      .rev()
192      .find(|&k| !author(&commands[k]) && !note(&commands[k]))
193      .map_or(0, |k| k + 1);
194    let run: Vec<usize> = (start..end).filter(|&k| author(&commands[k])).collect();
195    if run.len() < 2 {
196      continue;
197    }
198    // (its one address, an `\href{mailto:a@x}{a@x}` read once)
199    let text = content_text(&last_group(&entry.2));
200    let mut addresses = resolver::addresses_in(&text);
201    addresses.sort_unstable();
202    addresses.dedup();
203    let [address] = addresses.as_slice() else {
204      continue;
205    };
206    if lists_local_parts(address) {
207      continue;
208    }
209    let mut printed = Vec::with_capacity(run.len());
210    let mut based = Vec::with_capacity(run.len());
211    for &k in &run {
212      let text = argument_name_text(&queued_name(&commands[k].2))?.unwrap_or_default();
213      let name = Tokens::new(ExplodeText!(&text));
214      printed.push(NameKeys::as_printed(&name));
215      based.push(NameKeys::of(&name));
216    }
217    let holder = run.len() - 1;
218    if let [owner] = spelled_owners(&printed, address).as_slice()
219      && *owner != holder
220      && !own_call_keeps(&based[holder], &printed[*owner], address, true)
221    {
222      let place = commands[..run[*owner]].iter().filter(|e| author(e)).count();
223      let n = numbered + i64::try_from(place).unwrap_or(0) + 1;
224      out[at] = Some((address.clone(), s!("author:{n}")));
225    }
226  }
227  Ok(out)
228}

233/// The owners (indices into `keys`, the authors' names in order) of the addresses a printed list gives with no mark of
234/// its own — a `\date` line under the names (`date_legend`), a note the PDF prints with no mark (`address_note_contacts`;
235/// 2506.06941's `\footnotetext{\{p\_shojaee, imirzadeh, …\}@apple.com}`), OD #1071 — by the rule a class's address list
236/// is placed by ([`sequenced_contact_labels`], `EmailOwner::SpelledElseOrder`): each the author it spells, else, in a
237/// list of one address per author that no spelled address contradicts, the author at its place
238/// ([`author_order_step`]); else none, the author block's.
239pub(in crate::base_utilities) fn listed_address_owners(
240  addresses: &[String],
241  keys: &[NameKeys],
242) -> Vec<Option<usize>> {
243  let spelled: Vec<Option<usize>> = addresses
244    .iter()
245    .map(|address| match spelled_owners(keys, address).as_slice() {
246      [at] => Some(*at),
247      _ => None,
248    })
249    .collect();
250  // (each address a place of its own: one printed list is no call per author, so `{ann, bob}@x.org` under one name is
251  // no list of that author's two addresses)
252  let listed: Vec<(i64, Option<usize>)> = spelled
253    .iter()
254    .enumerate()
255    .map(|(at, owner)| (at as i64, *owner))
256    .collect();
257  let places = if keys.is_empty() {
258    None
259  } else {
260    author_order_step(&listed, keys.len())
261  };
262  spelled
263    .iter()
264    .enumerate()
265    .map(|(at, owner)| owner.or_else(|| places.as_ref().map(|places| places[at])))
266    .collect()
267}

269/// Step 2 of [`sequenced_contact_labels`], kept apart so a ruling can switch it off in one change: the place (from 0)
270/// of each address of `listed` (its call, its known author) in the authors' order, when the list is one place for
271/// each of `authors` authors and no known author is another than the one at its place; `None` when it is no such list.
272/// With no authors (the list given after `\maketitle` digested them, 1903.05554), the call's place, as Perl.
273fn author_order_step(listed: &[(i64, Option<usize>)], authors: usize) -> Option<Vec<usize>> {
274  // The place of each address in the list, read two ways: by the call (Perl's, a call of several addresses one place),
275  // and by the address (`\emailAdd{jdhb@umich.edu, elvang@umich.edu}` for two authors, 2406.03543).
276  let mut by_call: Vec<usize> = Vec::with_capacity(listed.len());
277  for (at, (call, _)) in listed.iter().enumerate() {
278    let place = match by_call.last() {
279      Some(last) if listed[at - 1].0 == *call => *last,
280      Some(last) => last + 1,
281      None => 0,
282    };
283    by_call.push(place);
284  }
285  let by_address: Vec<usize> = (0..listed.len()).collect();
286  let in_order = |places: &[usize]| {
287    let count = places.last().map_or(0, |last| last + 1);
288    listed
289      .iter()
290      .zip(places)
291      .all(|((_, owner), place)| owner.is_none_or(|owner| owner == *place))
292      && count == authors
293  };
294  // (by the address first: a call listing several authors' addresses, `\emailAdd{divyeshsolanki98@gmail.com,
295  // srijuster@gmail.com}` for Divyesh N. Solanki and Srijit Bhattacharjee, 2308.02445, is read so; by the call when an
296  // author's two addresses in one call would put the next one place on, as `\emailAdd{a, b}` for one author of two;
297  // by the call, as Perl, with no authors to read)
298  if authors == 0 {
299    Some(by_call)
300  } else if in_order(&by_address) {
301    Some(by_address)
302  } else if in_order(&by_call) {
303    Some(by_call)
304  } else {
305    None
306  }
307}

309/// The one author (index among the authors of `commands`, a frontmatter flush's queue) marked as the corresponding
310/// author, for the one address of a class's list left over (`address`, its text), among the authors it may be of
311/// (`within`, indices; all when `None`). An author is marked
312/// - by a note in their own name that names the corresponding author ([`names_correspondence`]: jheppub's
313///   `\note{Corresponding author.}`, `\note[*]{…}`; `\thanks`, `\footnote`), or
314/// - by a mark they carry whose legend names the corresponding author: an affiliation line's (`\author[a,*]{…}` under
315///   `\affiliation[*]{Corresponding author}`, `$^{1,*}$` under `\address{$^*$ Author to whom any correspondence should
316///   be addressed.}`) or a note's — its own mark (`\note[*]{…}`) or the number a jheppub-family `\note` prints its text
317///   under ([`AuthorNote::mark`]): the authors who write that number on their names are the ones it is about.
318///
319/// A note or a legend that says something else ("Equal contribution", "Since October 1, 2015 at …", 1510.03434) marks
320/// no one. Exactly one marked author among `within` gets the address; none or two and more, the author block keeps it
321/// (1802.03677's "¹ Corresponding author." note, whose 1 both M.V. Smirnov and J.J. Ling carry; 2606.26596's
322/// "Corresponding authors." under the `*` of two authors; user ruling 2026-10-21). A mark with no legend at all is no
323/// evidence of correspondence (a bare `*`, 2111.00753's two `$^{*}$`), save where the address itself says it is the
324/// corresponding author's (`\emailAdd{zyxu@gzu.edu.cn(Corresponding author)}`) and exactly one author among `within`
325/// carries a `*` with no legend (2308.06532's `\author[1,*]{Zhaoyi Xu}`, "E-mail: zyxu@gzu.edu.cn(Corresponding
326/// author)" in its PDF).
327pub(super) fn marked_corresponding_author(
328  commands: &[RawFrontmatter],
329  address: &str,
330  within: Option<&[usize]>,
331) -> Option<usize> {
332  let authors: Vec<&RawFrontmatter> = commands
333    .iter()
334    .filter(|entry| {
335      entry.0 == "ltx:creator" && entry.1.get("role").map(String::as_str) == Some("author")
336    })
337    .collect();
338  let names: Vec<Tokens> = authors.iter().map(|entry| queued_name(&entry.2)).collect();
339  let mut notes = author_notes(names.iter(), &[
340    T_CS!("\\thanks"),
341    T_CS!("\\footnote"),
342    T_CS!("\\note"),
343  ]);
344  // …and each note given apart from the names, the author's before it, as it is placed (1909.06107's
345  // `\author[a,1]{and V.~Sharyy}\note{Corresponding author}`; 2404.00867's `\note[*]{…}` after Paul J. Steinhardt's
346  // `\author`), its mark its own (`_mark`, jheppub's `\note[*]`)
347  let mut before = None;
348  for entry in commands {
349    match (entry.0.as_str(), entry.1.get("role").map(String::as_str)) {
350      ("ltx:creator", Some("author")) => before = Some(before.map_or(0, |at: usize| at + 1)),
351      ("ltx:contact", Some("note")) => {
352        if let Some(owner) = before {
353          notes.push(AuthorNote {
354            owner,
355            mark: entry.1.get("_mark").and_then(|mark| {
356              clean_frontmatter_labels(mark, "affiliation")
357                .into_iter()
358                .next()
359            }),
360            numbered: false,
361            names_correspondence: names_correspondence(&visible_name_text(
362              last_group(&entry.2).unlist_ref(),
363            )),
364          });
365        }
366      },
367      _ => {},
368    }
369  }
370  // Each author's labels: its class's optional ones (`\author[a,*]`, `annotations=`) and its marks'.
371  let author_labels: Vec<Vec<String>> = authors
372    .iter()
373    .zip(&names)
374    .map(|(entry, name)| {
375      let mut labels = entry
376        .1
377        .get("annotations")
378        .map(|given| clean_frontmatter_labels(given, "affiliation"))
379        .unwrap_or_default();
380      for operand in author_mark_operands(name.unlist_ref()) {
381        labels.extend(clean_frontmatter_labels(
382          &Tokens::new(operand).to_string(),
383          "affiliation",
384        ));
385      }
386      labels
387    })
388    .collect();
389  // The legends: each affiliation's label (`\affiliation[*]{…}`'s, or the line's leading mark) and whether it names the
390  // corresponding author; and each note's mark (`\note[*]{…}`, the n-th `\note`'s n).
391  let mut legends: Vec<(String, bool)> = commands
392    .iter()
393    .filter(|entry| {
394      entry.0 == "ltx:contact" && entry.1.get("role").map(String::as_str) == Some("affiliation")
395    })
396    .filter_map(|entry| {
397      let content = last_group(&entry.2);
398      let inner = queued_affiliation_text(&content);
399      let label = match entry.1.get("label").filter(|label| !label.is_empty()) {
400        Some(label) => clean_frontmatter_labels(label, "affiliation")
401          .into_iter()
402          .next(),
403        None if marker_leads(&inner) => {
404          entry_mark(inner.unlist_ref()).and_then(|mark| mark_label(&mark))
405        },
406        None => None,
407      }?;
408      Some((
409        label,
410        names_correspondence(&visible_name_text(inner.unlist_ref())),
411      ))
412    })
413    .collect();
414  // (a note's counter number is its legend only where no affiliation's label is that number: the authors writing an
415  // affiliation's number request the affiliation, 2005.05695, 2005.14143, 2405.09762, 1403.3797, 2101.00866; a note's
416  // own mark, `\note[*]{…}`, is its legend whatever the affiliations are labelled)
417  let affiliation_legends = legends.len();
418  let note_legends: Vec<(String, bool)> = notes
419    .iter()
420    .filter_map(|note| Some((note.mark.clone()?, note.numbered, note.names_correspondence)))
421    .filter(|(mark, numbered, _)| {
422      !*numbered
423        || !legends[..affiliation_legends]
424          .iter()
425          .any(|(label, _)| label == mark)
426    })
427    .map(|(mark, _, says)| (mark, says))
428    .collect();
429  legends.extend(note_legends);
430  let candidate = |at: &usize| within.is_none_or(|within| within.contains(at));
431  let mut marked: std::collections::BTreeSet<usize> = notes
432    .iter()
433    .filter(|note| note.names_correspondence)
434    .map(|note| note.owner)
435    .collect();
436  for (at, labels) in author_labels.iter().enumerate() {
437    if labels.iter().any(|label| {
438      legends
439        .iter()
440        .any(|(legend, says)| legend == label && *says)
441    }) {
442      marked.insert(at);
443    }
444  }
445  let marked: Vec<usize> = marked.into_iter().filter(candidate).collect();
446  match marked.as_slice() {
447    [author] => return Some(*author),
448    [] => {},
449    _ => return None,
450  }
451  // (no author marked: the address itself naming correspondence, the one author carrying a `*` no legend explains)
452  if !names_correspondence(address) {
453    return None;
454  }
455  let star = clean_frontmatter_labels("*", "affiliation");
456  let starred: Vec<usize> = author_labels
457    .iter()
458    .enumerate()
459    .filter(|(at, labels)| {
460      candidate(at)
461        && labels
462          .iter()
463          .any(|label| star.contains(label) && !legends.iter().any(|(legend, _)| legend == label))
464    })
465    .map(|(at, _)| at)
466    .collect();
467  match starred.as_slice() {
468    [author] => Some(*author),
469    _ => None,
470  }
471}

473/// The content of a queued affiliation's entry inside the wrappers that read its marks (`\lx@affiliation@withinst`,
474/// `\lx@affiliation@withsup`), as written.
475pub(in crate::base_utilities) fn queued_affiliation_text(content: &Tokens) -> Tokens {
476  let mut inner = content.clone();
477  loop {
478    let v = inner.unlist_ref();
479    let wrappers = [
480      T_CS!("\\lx@affiliation@withinst"),
481      T_CS!("\\lx@affiliation@withsup"),
482    ];
483    let wrapped = v.first().is_some_and(|t| wrappers.contains(t))
484      && v.get(1).is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
485      && skip_group(v, 1) == v.len();
486    if !wrapped {
487      return inner;
488    }
489    inner = Tokens::new(v[2..v.len() - 1].to_vec());
490  }
491}

493/// The content of the last top-level brace group of `tokens` (a queued frontmatter command's content argument); empty
494/// when there is none.
495pub(in crate::base_utilities) fn last_group(tokens: &Tokens) -> Tokens {
496  let v = tokens.unlist_ref();
497  let mut depth = 0usize;
498  let mut start = None;
499  let mut last = Tokens::new(Vec::new());
500  for (i, t) in v.iter().enumerate() {
501    match t.get_catcode() {
502      Catcode::BEGIN => {
503        if depth == 0 {
504          start = Some(i);
505        }
506        depth += 1;
507      },
508      Catcode::END if depth > 0 => {
509        depth -= 1;
510        if depth == 0
511          && let Some(open) = start.take()
512        {
513          last = Tokens::new(v[open + 1..i].to_vec());
514        }
515      },
516      _ => {},
517    }
518  }
519  last
520}

522/// The meaning recorder over a flush ([`begin_meaning_recorder`]), closed when it is ended or dropped: a flush leaving
523/// by an `Err` leaves none open, whose recording every later meaning assignment of the conversion would pay. Closed by
524/// its depth ([`end_meaning_recorder_at`]), so a displaced call's recorder an `Err` left open inside it is closed with
525/// it, never in its place.
526struct FlushMeanings {
527  depth: usize,
528  open:  bool,
529}
530
531impl FlushMeanings {
532  fn begin() -> Self {
533    FlushMeanings {
534      depth: begin_meaning_recorder(),
535      open:  true,
536    }
537  }
538
539  /// The meanings assigned since it began ([`end_meaning_recorder_at`]).
540  fn end(mut self) -> Vec<(SymStr, Option<Stored>)> {
541    self.open = false;
542    end_meaning_recorder_at(self.depth)
543  }
544}
545
546impl Drop for FlushMeanings {
547  fn drop(&mut self) {
548    if self.open {
549      end_meaning_recorder_at(self.depth);
550    }
551  }
552}

554/// Digest FrontMatter (if not already?)
555/// Perl: digestFrontMatter().
556pub fn digest_front_matter() -> Result<()> {
557  // (whether no flush digests its queue around this one: the outermost places what a late block deferred to its end)
558  let outermost = !lookup_bool("lx_frontmatter_flushing");
559  // (an `\author` body read again with the definitions in force here: `reread_author_body_at_flush`)
560  if outermost && let Some(call) = take_author_body_reread() {
561    digest(call)?;
562  }
563  // (what the abstract's command form added before this flush is no longer in Perl's queue: [`COMMAND_FORM`])
564  let flushes = lookup_int(FRONTMATTER_FLUSHES);
565  assign_value(FRONTMATTER_FLUSHES, flushes + 1, Some(Scope::Global));
566  // (the title marks of the `\maketitle` that runs, read where no binding declared them, before the rehearsal binds them)
567  crate::latex_constructs::read_title_marks_in_force();
568  // The frontmatter observer's rehearsal of the author blocks, whose reading the emitter writes from at this flush
569  // (OXIDIZED_DESIGN_DIVERGENCES #837, #980), and the undefined commands its rehearsals held an Error for.
570  let held_undefined = read_at_flush()?;
571  // (and the meanings the flush assigns, which the held commands it defines are told by: `HeldUndefined`; closed on
572  // every path out, an `Err` one too)
573  let recorder = FlushMeanings::begin();
574  bgroup();
575  // (the `\thefootnote` in force, which a title field's marks are told against: `enter_title_footnotes`)
576  crate::latex_constructs::begin_title_footnotes();
577  // INTENTIONAL DIVERGENCE from Perl PR #2767 (KNOWN_PERL_ERRORS #30,
578  // OXIDIZED_DESIGN): clear the queue BEFORE digesting. Perl digests
579  // from the live queue and wipes it after the loop — but when a queued
580  // entry's own content re-triggers digestFrontMatter (witness aa.cls
581  // 0907.0384: `\abstract{...}{}` dispatches the 5-arg \abstract@new,
582  // whose greedy params swallow the document's `\maketitle` into arg #5,
583  // so the queued abstract CONTAINS \maketitle → \lx@frontmatterhere →
584  // afterDigest → re-entry), Perl re-digests the same queue unboundedly:
585  // PR-head Perl dies `Fatal:perl:deep_recursion ... invokeToken`, zero
586  // output (verified 2026-06-04). Pre-clearing makes the nested
587  // invocation see an empty queue and terminate; the same paper then
588  // converts with zero errors (real LaTeX also compiles it). Everything
589  // else — deferred timing, digestion order, late re-let/\def fidelity —
590  // is exactly the PR's. Newly-queued entries during this digest are
591  // processed by the next invocation (or the end-of-document fallback).
592  // (the queued authors' names, a position each, where a sequenced contact may read them, `\lx@sequence@call` made:
593  // `sequenced_contact_labels`)
594  let author_names = if lookup_int("lx_sequence_call") > 0 {
595    queued_position_names()?
596  } else {
597    Rc::default()
598  };
599  forget_queue_reading();
600  let mut commands: Vec<RawFrontmatter> = match remove_value("frontmatter_raw") {
601    Some(Stored::FrontmatterRaw(commands)) => commands,
602    _ => Vec::new(),
603  };
604  // (the labels the queue defers, read for the readers of its keyvals here: `\lx@label@ref`, cas's `\author[a]`)
605  resolve_queued_label_refs(&mut commands);
606  // The labels the authors' marks request, for the notes digested here (`note_body`) and the
607  // emitter's readings of its calls (`queue_reading::requested_labels`) — this call's authors only: a
608  // note queued for a later call (an appending `\author` after `\maketitle`) sees that call's.
609  let requested: Vec<String> = commands
610    .iter()
611    .filter(|entry| {
612      entry.0 == "ltx:creator" && entry.1.get("role").map(String::as_str) == Some("author")
613    })
614    .flat_map(|entry| author_mark_operands(entry.2.unlist_ref()))
615    .flat_map(|operand| clean_frontmatter_labels(&Tokens::new(operand).to_string(), "affiliation"))
616    .collect();
617  set_flush_requested_marks(&requested);
618  // (the labels this flush's names show, which its binding lists' entries read — the outer flush's restored after it)
619  let outer_shown_labels = lookup_string(NAMES_SHOWN_LABELS);
620  assign_value(
621    NAMES_SHOWN_LABELS,
622    Stored::String(pin("")),
623    Some(Scope::Global),
624  );
625  // (the one added step, under its switch: the queue's creators and annotations written from the reading, OD #980)
626  // (a contact the class prints under the whole row of names, every author's, OXIDIZED_DESIGN_DIVERGENCES #901, is the
627  // reading's: the emitter writes a flush holding one with or without an author block, `stays_out`; under a block no
628  // rehearsal read it is digested as Perl's, as the rest of the queue)
629  let emitted = emit_commands(&mut commands)?;
630  // (what this flush digests is off the queue: no later flush's emitter writes it again, astro-ph9808358)
631  take_digested(&commands);
632  // The entries a seal drops in this flush, whose annotations go with them ([`seal_digested_frontmatter`]) — this
633  // flush's own, the outer flush's restored after it (a queued entry may flush again, `\maketitle` in an abstract).
634  let outer_sealed_off = lookup_string("lx_frontmatter_sealed_off");
635  assign_value(
636    "lx_frontmatter_sealed_off",
637    Stored::String(pin("")),
638    Some(Scope::Global),
639  );
640  // The authors digested here are the current ones: no later `\maketitle` supersedes
641  // them unless another `\author` replaces them first (`\lx@maketitle@supersede`).
642  if commands.iter().any(|entry| {
643    entry.0 == "ltx:creator" && entry.1.get("role").map(String::as_str) == Some("author")
644  }) {
645    assign_value(
646      "lx_authors_superseded",
647      Stored::Bool(false),
648      Some(Scope::Global),
649    );
650  }
651  // Where this flush's sequenced contacts go ([`sequenced_contact_labels`]), in the order
652  // `\lx@annotate@frontmatter@now` digests them — this flush's, the outer flush's restored after it, as the seal's.
653  let outer_sequence_labels = lookup_string("lx_frontmatter_sequence_labels");
654  let outer_sequence_at = lookup_int("lx_frontmatter_sequence_at");
655  assign_value(
656    "lx_frontmatter_sequence_labels",
657    Stored::String(pin(
658      sequenced_contact_labels(&commands, &author_names)?.join("\u{1}"),
659    )),
660    Some(Scope::Global),
661  );
662  assign_value(
663    "lx_frontmatter_sequence_at",
664    Stored::Int(0),
665    Some(Scope::Global),
666  );
667  assign_value(
668    "lx_frontmatter_flushing",
669    Stored::Bool(true),
670    Some(Scope::Local),
671  );
672  // The labels the mark-picked contacts digested here answer, `role` and label (a marked line's first mark,
673  // `\lx@affiliation@withsup`): an author requesting one has its contact of that role even before the contact is
674  // digested, which a run's `new` reads (`\lx@annotate@frontmatter@now`; `\author{Ann$^{1}$}\author{Bob$^{2}$}
675  // \affil{$^1$Univ X}\author{Cat \affil{Univ Z}}\affil{$^2$Univ Y}`: Cat's `\affil` is digested with her, before
676  // Univ Y, 77n r4 review probe e4). Not a `label=` (`\affil[1]`): authblk prints the unlabelled `\affil` after it
677  // under its `\AB@thenote`, both authors' (`\author[1]{Ann}\author{Bob}\affil[1]{Univ X}\affil{Univ Y}`, 2605.25017's
678  // e-mail line, 77n r5 review probes q10, r07), as Perl's `annotate=new` (authblk.sty.ltxml:57) gives it.
679  let offered: Vec<String> = commands
680    .iter()
681    .filter(|entry| entry.0 == "ltx:contact")
682    .flat_map(|entry| {
683      let role = entry.1.get("role").cloned().unwrap_or_default();
684      let tokens = entry.2.unlist_ref();
685      tokens
686        .iter()
687        .position(|t| *t == T_CS!("\\lx@affiliation@withsup"))
688        .and_then(|at| entry_mark(&tokens[at + 1..]))
689        .and_then(|mark| mark_label(&mark))
690        .map(move |label| s!("{role}\u{2}{label}"))
691    })
692    .collect();
693  assign_value(
694    "lx_frontmatter_offered_labels",
695    Stored::String(pin(offered.join("\u{1}"))),
696    Some(Scope::Local),
697  );
698  if !commands.is_empty() {
699    let_i(
700      &T_CS!("\\lx@add@frontmatter"),
701      &T_CS!("\\lx@add@frontmatter@now"),
702      None,
703    );
704    let_i(
705      &T_CS!("\\lx@annotate@frontmatter"),
706      &T_CS!("\\lx@annotate@frontmatter@now"),
707      None,
708    );
709    // (the shared blocks' e-mails the reading did not place: `shared_block_email_labels`)
710    let respelled = shared_block_email_labels(&commands, emitted)?;
711    for (entry, respell) in commands.into_iter().zip(respelled) {
712      let respell = respell.map(|(address, label)| s!("{address}\u{2}{label}"));
713      assign_value(
714        ENTRY_RESPELL,
715        Stored::String(pin(respell.as_deref().unwrap_or(""))),
716        Some(Scope::Local),
717      );
718      digest_queued_entry(entry)?;
719    }
720    assign_value(ENTRY_RESPELL, Stored::String(pin("")), Some(Scope::Local));
721  }
722  // (an entry a late block's flush set after this queue's own, `defer_to_flush_end`: a title, subtitle or date read
723  // in a block this outermost flush digests — `\title{…\author{…\date{One}}}` — placed here, at the end of the
724  // flush, as TeX typesets it after the queue's entries; pdflatex prints "One" over a `\date{Two}`, with or without a
725  // `\section` after `\maketitle`: review revCR4 finding 3)
726  if outermost {
727    let deferred = match remove_value(DEFERRED_TO_FLUSH_END) {
728      Some(Stored::FrontmatterRaw(deferred)) => deferred,
729      _ => Vec::new(),
730    };
731    if !deferred.is_empty() {
732      let_i(
733        &T_CS!("\\lx@add@frontmatter"),
734        &T_CS!("\\lx@add@frontmatter@now"),
735        None,
736      );
737      let_i(
738        &T_CS!("\\lx@annotate@frontmatter"),
739        &T_CS!("\\lx@annotate@frontmatter@now"),
740        None,
741      );
742      for entry in deferred {
743        digest_queued_entry(entry)?;
744      }
745    }
746  }
747  // The held Errors no piece raised again — a command printing nothing between pieces, or in a piece written as text
748  // —, each once, as Perl's one digestion of the block raises it (`HeldUndefined`, OXIDIZED_DESIGN_DIVERGENCES #983).
749  held_undefined.raise_unraised(&recorder.end())?;
750  // Add punctuation to all ltx:creators, now that we know how many of each role.
751  let mut updates: Vec<(usize, bool)> = Vec::new(); // (index, use_conjunction)
752  with_value("frontmatter", |v| {
753    if let Some(Stored::HashTagData(frnt)) = v
754      && let Some(list) = frnt.get("ltx:creator")
755    {
756      for (i, item) in list.iter().enumerate() {
757        let role = item.attr.get("role").map(String::as_str).unwrap_or("");
758        let num: i64 = item
759          .attr
760          .get("_num")
761          .and_then(|n| n.parse().ok())
762          .unwrap_or(0);
763        if !role.is_empty() && num > 1 {
764          let n = lookup_mapping_int("num_ltx:creator", role);
765          updates.push((i, num >= n));
766        }
767      }
768    }
769  });
770  for (index, use_conjunction) in updates {
771    // (the separator the names line printed before it, where a per-author class's line was split there:
772    // `\lx@author@printed@before`, brief 97b part D — else the binding's)
773    let printed = with_value("frontmatter", |v| match v {
774      Some(Stored::HashTagData(frnt)) => frnt
775        .get("ltx:creator")
776        .and_then(|l| l.get(index))
777        .and_then(|entry| entry.attr.get("_printedbefore").cloned()),
778      _ => None,
779    });
780    let text = match printed {
781      Some(text) => text,
782      None => {
783        let separator = DigestText!(Tokens!(if use_conjunction {
784          T_CS!("\\lx@author@conj")
785        } else {
786          T_CS!("\\lx@author@sep")
787        }))?;
788        // `\lx@author@conj` may digest to \hskip Whatsits (\qquad) — extract
789        // text-or-spaces (see \lx@author@prefix note in the previous port).
790        digested_to_text(&separator)?
791      },
792    };
793    with_value_mut("frontmatter", |val_opt| {
794      if let Some(&mut Stored::HashTagData(ref mut frnt)) = val_opt
795        && let Some(mut entry) = frnt.get_mut("ltx:creator").and_then(|l| l.get_mut(index))
796      {
797        entry.attr.insert("before".to_string(), text.clone());
798      }
799    });
800  }
801  assign_value(
802    "lx_frontmatter_sealed_off",
803    Stored::String(pin(&outer_sealed_off)),
804    Some(Scope::Global),
805  );
806  assign_value(
807    "lx_frontmatter_sequence_labels",
808    Stored::String(pin(&outer_sequence_labels)),
809    Some(Scope::Global),
810  );
811  assign_value(
812    "lx_frontmatter_sequence_at",
813    Stored::Int(outer_sequence_at),
814    Some(Scope::Global),
815  );
816  assign_value(
817    NAMES_SHOWN_LABELS,
818    Stored::String(pin(&outer_shown_labels)),
819    Some(Scope::Global),
820  );
821  egroup()?;
822  Ok(())
823}

976/// insertion point and the document root when every one of them auto-closes (a `p`
977/// in a `para`) and none is a frontmatter-group element; otherwise leave the point
978/// alone. An open `{titlepage}` is not a paragraph — `\begin{titlepage}\maketitle
979/// \end{titlepage}` relies on the `\unwind@titlepage` hook finding it open (closing
980/// it here left an empty `<titlepage/>` behind).
981pub(super) fn close_auto_closable_to_root(document: &mut Document, root: &Node) -> Result<()> {
982  let mut n = document.get_node().clone();
983  if n.get_type() != Some(NodeType::ElementNode) {
984    match n.get_parent() {
985      Some(p) => n = p,
986      None => return Ok(()),
987    }
988  }
989  let mut cur = n;
990  while &cur != root {
991    if cur.get_type() != Some(NodeType::ElementNode)
992      || !document::can_auto_close(&cur)
993      || is_frontmatter_group_element(&cur)
994    {
995      return Ok(());
996    }
997    match cur.get_parent() {
998      Some(p) => cur = p,
999      None => return Ok(()),
1000    }
1001  }
1002  document.close_to_node(root, true)
1003}

1008/// The ONE frontmatter placement pass (OXIDIZED_DESIGN #247). The schema wants every
1009/// frontmatter-group element before all `document.body.class` content
1010/// (`LaTeXML-structure.rnc:34`), and `Para.class` — hence the body group — includes
1011/// `pagination`, `rule` and `para` (`LaTeXML-para.rnc:18-20`). Perl only ever places at
1012/// the current point (`\maketitle`) or at the top (fallback) and leaves whatever that
1013/// yields. This pass looks at the LEADING REGION — the root children between the
1014/// resources and the bound (the abstract's own `ltx:_Frontmatter_Capture_` position
1015/// marker when it has one, else everything built so far) — and classifies each node:
1016///
1017/// * a first-group element (`is_frontmatter_group_element`) stays and becomes the
1018///   anchor candidate;
1019/// * a content-free node (`node_is_content_free`: a pagebreak, an empty paragraph, an
1020///   ink-free empty box) or PREAMBLE residue (`is_preamble_residue`: built before
1021///   `\begin{document}`, where LaTeX forbids typesetting) is left in place and, if it
1022///   preceded the anchor, moved after the placed frontmatter;
1023/// * hand-typeset title-page LAYOUT — `para` / demotable `logical-block` / `pagination`
1024///   with visible ink, `wrap_layout` only (the abstract-only fallback: a cover typeset
1025///   before a `\begin{abstract}` with no `\maketitle`) — is wrapped, in its exact order,
1026///   into `<ltx:titlepage>` (`titlepage_model = (FrontMatter.class |
1027///   SectionalFrontMatter.class | Block.class)*` is the schema's container for exactly
1028///   that; `para`→`block`, `logical-block` demoted bottom-up as in #244);
1029/// * anything else — a `<TOC>`, a section, a float, a root `<rule>`, a VISIBLE body
1030///   paragraph on the `\maketitle` path — ends the region: it stays where it is and the
1031///   frontmatter is placed ahead of it, so the document still opens with its front
1032///   group (Perl places at the current point and strands the title; user-approved
1033///   surpass, OXIDIZED_DESIGN #262).
1034///
1035/// The frontmatter is inserted after the wrapped titlepage, else after the leading
1036/// first-group run (`FrontmatterAnchor::LeadingFrontmatter`) or at the top, via the
1037/// same `_Capture_` insert→fill→unwrap machinery Perl's fallback uses. Run at every
1038/// placement point — `\lx@frontmatterhere`, both `\lx@frontmatter@fallback` branches,
1039/// the `{titlepage}` environment's `after_construct` — so a leading pagebreak before a
1040/// built titlepage is settled when that titlepage is built, not by a later pass. The
1041/// bound matters: without the marker a sectionless manual's whole body would read as
1042/// cover (jourcl's cover LETTER, derivative's after-abstract cover). Superseded by this
1043/// pass: #242's content-free gate, #245's `\end{document}` relocation, #246's anchor
1044/// helpers. Kept beside it: the conservative title promotion (`maybe_promote_leading_title`,
1045/// user-ruled), `maybe_dedup_leading_title_ink` (#6924), `insert_late_frontmatter`, the
1046/// marker sweep in `insert_frontmatter` and the core `finalize_rec`.
1047///
1048/// Witnesses: amsldoc/tkz-doc/tzplot (#242, 14 docs), gitinfo2/gitlog/FrontespizioScudo/
1049/// toptesi-example-* (#245, 6), toptesi-example-magistrale/tipfr-doc/isosigns-docs/
1050/// bootstrapicons-docs/derivative/jourcl (#246, 16), the faketitlepage /
1051/// promote_center_title / svabstract fixtures, `114_streaming_structure` (eager ≡ streaming).
1052pub fn place_frontmatter(
1053  document: &mut Document,
1054  wrap_layout: bool,
1055  anchor: FrontmatterAnchor,
1056) -> Result<Option<TextBeforeTitle>> {
1057  // Drain the raw frontmatter queue first, as `insert_frontmatter` does, so
1058  // `frontmatter_pending` and the flush agree: the `{titlepage}` environment's
1059  // `after_construct` is the one placement point without its own digest hook, and
1060  // under `--streaming` (digestion interleaved with construction) a `\title` given
1061  // after `\end{titlepage}` may still sit undigested there.
1062  digest_front_matter()?;
1063  let Some(root) = document.get_document().get_root_element() else {
1064    insert_frontmatter(document)?;
1065    return Ok(None);
1066  };
1067  let kids: Vec<Node> = root
1068    .get_child_nodes()
1069    .into_iter()
1070    .filter(|n| n.get_type() == Some(NodeType::ElementNode))
1071    .collect();
1072  let mut i = 0;
1073  while i < kids.len() && document::get_node_qname(&kids[i]) == pin_static("ltx:resource") {
1074    i += 1;
1075  }
1076  let bound = kids[i..]
1077    .iter()
1078    .position(|n| document::get_node_qname(n) == pin_static("ltx:_Frontmatter_Capture_"))
1079    .map_or(kids.len(), |k| i + k);
1080  // The leading prefix: first-group elements and content-free nodes, in any order.
1081  let mut last_first: Option<Node> = None;
1082  let mut movers: Vec<Node> = Vec::new();
1083  while i < bound {
1084    if is_frontmatter_group_element(&kids[i]) {
1085      last_first = Some(kids[i].clone());
1086    } else if node_is_content_free(&kids[i]) || is_preamble_residue(&kids[i]) {
1087      movers.push(kids[i].clone());
1088    } else {
1089      break;
1090    }
1091    i += 1;
1092  }
1093  // Hand-typeset cover before the abstract → <titlepage>, in order.
1094  let mut titlepage: Option<Node> = None;
1095  if wrap_layout && i < bound {
1096    let end = i
1097      + kids[i..bound]
1098        .iter()
1099        .take_while(|k| is_titlepage_layout(k))
1100        .count();
1101    if kids[i..end].iter().any(|k| !node_is_content_free(k)) {
1102      // The flush ends the paragraph it interrupts before the cover moves: text right
1103      // before an abstract is still the open paragraph, and wrapping it open left the
1104      // insertion point inside the renamed `block` (`Error:malformed:ltx:document` at
1105      // `\end{document}`; serbian-def-cyr/proba).
1106      close_auto_closable_to_root(document, &root)?;
1107      titlepage = Some(wrap_as_titlepage(document, &root, &kids[i..end])?);
1108    }
1109  }
1110  // Nothing queued (or the abstract-only deferral): `insert_frontmatter` keeps its
1111  // side effects; a leading pagebreak before a built titlepage still settles here.
1112  if !frontmatter_pending() {
1113    insert_frontmatter(document)?;
1114    if let Some(tail) = last_first {
1115      move_content_free_after(&movers, tail)?;
1116    }
1117    return Ok(None);
1118  }
1119  // Visible body content before the flush (prose or a class-drawn cover before
1120  // `\maketitle`, a section before a late `\title`) does not hold the frontmatter
1121  // back: the schema's document model puts the frontmatter group first
1122  // (`LaTeXML-structure.rnc:34`), so it goes to the head and that content follows
1123  // it in its own order. Perl places at the current point (`\maketitle`,
1124  // Base_Utility.pool.ltxml:918), stranding `<title>` after the body — schema-invalid
1125  // (user-approved surpass 2026-09-23, OXIDIZED_DESIGN #262); text among it the
1126  // `\maketitle` caller then keeps ahead of the title ([`wrap_text_before_title`]). The flush ends the
1127  // paragraph it interrupts, as every flush point does in LaTeX (`\maketitle`:
1128  // article.cls:203 opens with `\par`, report/book with `\begin{titlepage}`;
1129  // `\section`; `\end{document}`) and as the current-point insertion did by
1130  // auto-closing: text after `\maketitle` starts a new paragraph instead of
1131  // continuing one that now stands after the frontmatter.
1132  close_auto_closable_to_root(document, &root)?;
1133  let after = titlepage.or(match anchor {
1134    FrontmatterAnchor::Top => None,
1135    FrontmatterAnchor::LeadingFrontmatter => last_first.clone(),
1136  });
1137  // (the body text the flush found ahead of it, which the caller keeps ahead of the title block: QA91)
1138  let lead: Vec<Node> = if wrap_layout {
1139    Vec::new()
1140  } else {
1141    kids[i..bound]
1142      .iter()
1143      .take_while(|k| is_titlepage_layout(k) && !is_preamble_residue(k))
1144      .cloned()
1145      .collect()
1146  };
1147  let placed = insert_frontmatter_at(document, after.as_ref())?;
1148  // Content-free nodes that preceded the placed run now follow it, in their order.
1149  let tail = match (placed.last().cloned(), last_first) {
1150    (Some(p), Some(f)) => Some(if node_precedes(&p, &f) { f } else { p }),
1151    (p, f) => p.or(f),
1152  };
1153  if let Some(tail) = tail {
1154    move_content_free_after(&movers, tail)?;
1155  }
1156  Ok(match placed.first() {
1157    Some(title_block) if lead.iter().any(|k| !node_is_content_free(k)) => Some(TextBeforeTitle {
1158      run:         lead,
1159      title_block: title_block.clone(),
1160    }),
1161    _ => None,
1162  })
1163}

1184/// Body text set before `\maketitle` stays ahead of the title block, as the PDF prints it (QA91, DG 2026-11-14), wrapped
1185/// as a `<titlepage>` of [`BEFORE_TITLEPAGE`] since the schema puts the frontmatter group first (LaTeXML-structure.rnc:34;
1186/// OXIDIZED_DESIGN #262). Run after the field-ink lift, which takes a print of the title or the authors out of it.
1187pub fn wrap_text_before_title(document: &mut Document, lead: TextBeforeTitle) -> Result<()> {
1188  let Some(root) = document.get_document().get_root_element() else {
1189    return Ok(());
1190  };
1191  if lead.title_block.get_parent().as_ref() != Some(&root) {
1192    return Ok(());
1193  }
1194  // (what the lift left of it; never the element holding the insertion point, a `\maketitle` in a box)
1195  let point = document.get_node().clone();
1196  let run: Vec<Node> = lead
1197    .run
1198    .into_iter()
1199    .filter(|k| k.get_parent().as_ref() == Some(&root))
1200    .take_while(|k| !is_ancestor_or_self(k, &point))
1201    .collect();
1202  // (a run printing no word, a graphic or a rule keeps the base placement: a mark left of an author line, `]` `*1`;
1203  // 2209.03353, 0905.0312, 2305.09250, 2309.09081, revQAB4b)
1204  if !run.iter().any(prints_a_word_or_ink) {
1205    return Ok(());
1206  }
1207  let mut titlepage = wrap_as_titlepage(document, &root, &run)?;
1208  document.add_class(&mut titlepage, BEFORE_TITLEPAGE)?;
1209  titlepage.unlink();
1210  let mut title_block = lead.title_block;
1211  title_block.add_prev_sibling(&mut titlepage)?;
1212  Ok(())
1213}

1215/// Does `node` print a content word (the title-page check's, [`net_content_words`]) or ink of an element of its own (a
1216/// childless element besides an empty paragraph or a `<break/>`: a graphic, a rule)?
1217fn prints_a_word_or_ink(node: &Node) -> bool {
1218  fn ink(node: &Node) -> bool {
1219    node.get_child_nodes().iter().any(|c| {
1220      c.get_type() == Some(NodeType::ElementNode)
1221        && if c.get_first_child().is_none() {
1222          !with(document::get_node_qname(c), |q| {
1223            matches!(q, "ltx:break" | "ltx:p" | "ltx:para")
1224          })
1225        } else {
1226          ink(c)
1227        }
1228    })
1229  }
1230  !net_content_words(&node.get_content()).is_empty() || ink(node)
1231}
