5/// Where the contacts a class lists apart from the names, in the order the document gives them
6/// (`\lx@add@contact@placed[owner=sequence]`: jheppub's `\emailAdd`, iopart's `\ead`), queued in `commands`, go: each
7/// the label of its creator, or empty to stay where `\lx@add@contact@placed` put it (the group's only author when it was
8/// given, else the author block). Read against the authors of the same flush, all of them, wherever the list stands
9/// among them (2108.07972's `\emailAdd{kadota.kenji@…}` before the four `\author`s):
10/// 1. the author the address spells ([`address_owners`]);
11/// 2. else the n-th author, the address at the n-th place — Perl's jheppub.sty.ltxml rule (`labelseq=author`) — when the
12///    list is one place for each author and in their order: no address spells an author other than the one at its
13///    place, as [`addresses_in_order`](crate::base_utilities::defs::frontmatter_api::email_lines::addresses_in_order) reads an author block's line (1303.3040's `zhl61@` for Zhen Liu, 2011.02499's
14///    `tpsg4@`/`tpsr@`). The place is the address's, or else the call's (an `\emailAdd{a, b}` of one author's two
15///    addresses). 2407.07862's `eichhorn@` listed first for Astrid Eichhorn, author 2, and 1804.07528's three addresses
16///    each one place on are no list in author order. The user ruled the consistent list evidence (2026-10-21);
17///    [`author_order_step`] is this step, alone;
18/// 3. else where `\lx@add@contact@placed` put it: the group's only author ([`placed_contact_owner`]), or the author
19///    block;
20/// 4. else, for the one address of the list left on the block of a multi-author flush, the ONE author marked as the
21///    corresponding author — by a note naming them so, or a mark whose legend says so —, among the authors of its
22///    group and, where the document places its e-mails by address block, those requesting its address's mark
23///    ([`marked_corresponding_author`]; user ruling 2026-10-21: 1901.08773's `beril@` A. Kopylov's, 2510.11030's
24///    `hjhspace223@` J.-H. Ha's, 2211.07892's `yoomin@` Y. M. Oh's; 2308.06532's `zyxu@` Zhaoyi Xu's, its address
25///    saying "(Corresponding author)" and he alone carrying a `*`; 1802.03677's `gear8mike@` stays the block's, the
26///    number of its "Corresponding author." note on two authors, and 2111.00753's `lad@`, two legend-less `*`s);
27/// 5. a flush with no authors, its list given after `\maketitle` digested them (1903.05554), keeps Perl's rule.
28///
29/// Step 3, for iopart's `\ead`, places after the spelling with the one author requesting the mark of the address block
30/// it is printed under (`mark=address`, `\lx@address@mark@set`: 2407.14305's `rockfish3141@` Hidetomo Hoshino's as printed —
31/// today the block's, RED `iopart_ead_under_its_address_mark_red` —
32/// 2311.07087's `sgess@` Spencer Gessner's) — weaker than a list in author order, which the PDF's address blocks do not
33/// contradict (1902.07874's `\ead{…, b97202056@gmail.com, …}` of five authors after `\address{$^9$ …}`, Che-Yu Chen's
34/// at place 3; 1710.02251's three `\mailto`s in an `\eads` after `$^{(2)}$`), and only in a document that writes an
35/// e-mail between its addresses: one whose e-mails all follow its addresses writes them where IOP's guidelines ask,
36/// "after the postal address(es)", under the last address by position only (0910.0753's `wolfgang.muender@` for W
37/// Münder, 1111.3344's `knarf@` for Frank Löffler, 1911.06133's `ledou@` for Pierre Le Doussal, each after the last
38/// author's `\address`; `_unmarked`).
39///
40/// The PDF prints the list as one shared line (jheppub.sty:127-131) or each address as a block under the address before
41/// it (iopart.cls:363), so the spelling, the marks and the order are the only evidence (OXIDIZED_DESIGN_DIVERGENCES #805,
42/// #806, #808, #809).
43pub(in crate::base_utilities) fn sequenced_contact_labels(
44  commands: &[RawFrontmatter],
45  names: &[Tokens],
46) -> Result<Vec<String>> {
47  // (a document that writes an e-mail between its addresses places each under its address, iopart.cls:363; one whose
48  // addresses all come first writes its e-mails "after the postal address(es)", as IOP's guidelines ask, under the
49  // last address only by position: there an address mark places nothing, `_unmarked`)
50  let Some(first_listed) = commands
51    .iter()
52    .position(|entry| entry.0 == "ltx:contact" && entry.1.contains_key("_sequence"))
53  else {
54    return Ok(Vec::new());
55  };
56  // (an amsart-style class's `\address` is an address too: lmcs's `\email` under each lettered `\address`, 2306.08266)
57  let interleaved = commands[first_listed..].iter().any(|entry| {
58    entry.0 == "ltx:contact"
59      && matches!(
60        entry.1.get("role").map(String::as_str),
61        Some("affiliation" | "address")
62      )
63  });
64  // (the names as they print, one a position, as `queued_author_name_keys` reads the queue's: `queued_position_names`)
65  let mut keys: Vec<NameKeys> = Vec::new();
66  for name in names {
67    keys.push(printed_name_keys(name, None)?);
68  }
69  // Each address with the class command's call it came from (`\lx@sequence@call`: `\emailAdd{a, b}` is one call of two
70  // addresses), the author it spells, whether it is on the block where `\lx@add@contact@placed` put it, and where it
71  // goes when its address mark places nothing.
72  let listed: Vec<(i64, Option<usize>, bool, Option<&String>)> = commands
73    .iter()
74    .filter(|entry| entry.0 == "ltx:contact" && entry.1.contains_key("_sequence"))
75    .map(|entry| {
76      let call = entry
77        .1
78        .get("_sequence")
79        .and_then(|n| n.parse().ok())
80        .unwrap_or(0);
81      let spelled = match address_owners(&keys, &last_group(&entry.2)).as_slice() {
82        [at] => Some(*at),
83        _ => None,
84      };
85      let unmarked = entry.1.get("_unmarked").filter(|_| !interleaved);
86      let placed = unmarked.or_else(|| entry.1.get("label"));
87      let blocked = placed.map(String::as_str) == Some("addresses:block");
88      (call, spelled, blocked, unmarked)
89    })
90    .collect();
91  let owners: Vec<(i64, Option<usize>)> = listed
92    .iter()
93    .map(|(call, owner, ..)| (*call, *owner))
94    .collect();
95  let places = author_order_step(&owners, keys.len());
96  let mut labels: Vec<String> = owners
97    .iter()
98    .enumerate()
99    .map(|(at, (_, owner))| match (owner, &places) {
100      (Some(owner), _) if !keys.is_empty() => s!("author:{}", owner + 1),
101      (_, Some(places)) if keys.is_empty() || places[at] < keys.len() => {
102        s!("author:{}", places[at] + 1)
103      },
104      _ => listed[at].3.cloned().unwrap_or_default(),
105    })
106    .collect();
107  let left: Vec<usize> = (0..listed.len())
108    .filter(|at| match labels[*at].as_str() {
109      "" => listed[*at].2,
110      label => label == "addresses:block",
111    })
112    .collect();
113  if let ([at], true) = (left.as_slice(), keys.len() > 1) {
114    // (among the authors it may be of: its author group's, and of those the ones requesting its address's mark, when
115    // the document places its e-mails by address block — `_group`, `_marks`, `\lx@add@contact@placed`)
116    let entry = commands
117      .iter()
118      .filter(|entry| entry.0 == "ltx:contact" && entry.1.contains_key("_sequence"))
119      .nth(*at);
120    let positions = |key: &str| -> Option<Vec<usize>> {
121      let value = entry?.1.get(key)?;
122      Some(
123        value
124          .split(';')
125          .filter_map(|n| n.trim().parse::<usize>().ok())
126          .filter_map(|n| n.checked_sub(1))
127          .collect(),
128      )
129    };
130    let group = positions("_group");
131    let marks = positions("_marks").filter(|marks| interleaved && !marks.is_empty());
132    let within: Option<Vec<usize>> = match (group, marks) {
133      (Some(group), Some(marks)) => {
134        Some(group.into_iter().filter(|at| marks.contains(at)).collect())
135      },
136      (Some(only), None) | (None, Some(only)) => Some(only),
137      (None, None) => None,
138    };
139    let address = entry
140      .map(|entry| visible_name_text(last_group(&entry.2).unlist_ref()))
141      .unwrap_or_default();
142    if let Some(author) = marked_corresponding_author(commands, &address, within.as_deref()) {
143      labels[*at] = s!("author:{}", author + 1);
144    }
145  }
146  Ok(labels)
147}

149/// The flush's e-mails printed in a block several names share, for a flush the reading does not write (OD #970 G6): an
150/// e-mail entry, or a note holding one, after a run of ≥ 2 names goes to the name it spells — acmart (2310.06444,
151/// 2601.17008, 1907.07581, 2504.00797), Wiley `\corres` (2311.07985, 2407.08541, 2505.23112, 2503.06763); one after a
152/// single name stays its own (2208.05647, 2306.14314, 2609.11148), as does an address two own calls give (2205.01906,
153/// 2211.05700). Read per entry in `\lx@annotate@frontmatter@now` ([`ENTRY_RESPELL`]).
154pub(in crate::base_utilities) fn shared_block_email_labels(
155  commands: &[RawFrontmatter],
156  emitted: bool,
157) -> Result<Vec<Option<(String, String)>>> {
158  // (a creator the emitter wrote, its keys in its call alone, is one of the reading's authors)
159  let author = |entry: &RawFrontmatter| {
160    entry.0 == "ltx:creator" && entry.1.get("role").is_none_or(|role| role == "author")
161  };
162  // (a name's note: a note, or a contact of role `note`, acmart's `\authornote`)
163  let note = |entry: &RawFrontmatter| {
164    entry.0 == "ltx:note" || entry.1.get("role").map(String::as_str) == Some("note")
165  };
166  let mut out = vec![None; commands.len()];
167  let numbered = lookup_mapping_int("num_ltx:creator", "author");
168  for (at, entry) in commands.iter().enumerate() {
169    let positional = entry.0 == "ltx:contact"
170      && entry.1.get("role").map(String::as_str) == Some("email")
171      && entry
172        .1
173        .get("label")
174        .is_none_or(|label| label.trim().is_empty())
175      && !entry.1.contains_key("_sequence");
176    // (where the reading wrote the flush, only an entry it left as queued: a document's note, not a creator's call)
177    if (!positional && entry.0 != "ltx:note")
178      || (emitted && resolver::creator_call(&entry.0, &annotated_parent(&entry.2)))
179    {
180      continue;
181    }
182    // (the run of names before it, back over the entries between; a name's note, acmart's `\authornote`, ends no run:
183    // 1907.07581)
184    let end = (0..at)
185      .rev()
186      .find(|&k| author(&commands[k]))
187      .map_or(0, |k| k + 1);
188    let start = (0..end)
189      .rev()
190      .find(|&k| !author(&commands[k]) && !note(&commands[k]))
191      .map_or(0, |k| k + 1);
192    let run: Vec<usize> = (start..end).filter(|&k| author(&commands[k])).collect();
193    if run.len() < 2 {
194      continue;
195    }
196    // (its one address, an `\href{mailto:a@x}{a@x}` read once)
197    let text = content_text(&last_group(&entry.2));
198    let mut addresses = resolver::addresses_in(&text);
199    addresses.sort_unstable();
200    addresses.dedup();
201    let [address] = addresses.as_slice() else {
202      continue;
203    };
204    if lists_local_parts(address) {
205      continue;
206    }
207    let mut printed = Vec::with_capacity(run.len());
208    let mut based = Vec::with_capacity(run.len());
209    for &k in &run {
210      let text = argument_name_text(&queued_name(&commands[k].2))?.unwrap_or_default();
211      let name = Tokens::new(ExplodeText!(&text));
212      printed.push(NameKeys::as_printed(&name));
213      based.push(NameKeys::of(&name));
214    }
215    let holder = run.len() - 1;
216    if let [owner] = spelled_owners(&printed, address).as_slice()
217      && *owner != holder
218      && !own_call_keeps(&based[holder], &printed[*owner], address, true)
219    {
220      let place = commands[..run[*owner]].iter().filter(|e| author(e)).count();
221      let n = numbered + i64::try_from(place).unwrap_or(0) + 1;
222      out[at] = Some((address.clone(), s!("author:{n}")));
223    }
224  }
225  Ok(out)
226}

231/// Step 2 of [`sequenced_contact_labels`], kept apart so a ruling can switch it off in one change: the place (from 0)
232/// of each address of `listed` (its call, its known author) in the authors' order, when the list is one place for
233/// each of `authors` authors and no known author is another than the one at its place; `None` when it is no such list.
234/// With no authors (the list given after `\maketitle` digested them, 1903.05554), the call's place, as Perl.
235fn author_order_step(listed: &[(i64, Option<usize>)], authors: usize) -> Option<Vec<usize>> {
236  // The place of each address in the list, read two ways: by the call (Perl's, a call of several addresses one place),
237  // and by the address (`\emailAdd{jdhb@umich.edu, elvang@umich.edu}` for two authors, 2406.03543).
238  let mut by_call: Vec<usize> = Vec::with_capacity(listed.len());
239  for (at, (call, _)) in listed.iter().enumerate() {
240    let place = match by_call.last() {
241      Some(last) if listed[at - 1].0 == *call => *last,
242      Some(last) => last + 1,
243      None => 0,
244    };
245    by_call.push(place);
246  }
247  let by_address: Vec<usize> = (0..listed.len()).collect();
248  let in_order = |places: &[usize]| {
249    let count = places.last().map_or(0, |last| last + 1);
250    listed
251      .iter()
252      .zip(places)
253      .all(|((_, owner), place)| owner.is_none_or(|owner| owner == *place))
254      && count == authors
255  };
256  // (by the address first: a call listing several authors' addresses, `\emailAdd{divyeshsolanki98@gmail.com,
257  // srijuster@gmail.com}` for Divyesh N. Solanki and Srijit Bhattacharjee, 2308.02445, is read so; by the call when an
258  // author's two addresses in one call would put the next one place on, as `\emailAdd{a, b}` for one author of two;
259  // by the call, as Perl, with no authors to read)
260  if authors == 0 {
261    Some(by_call)
262  } else if in_order(&by_address) {
263    Some(by_address)
264  } else if in_order(&by_call) {
265    Some(by_call)
266  } else {
267    None
268  }
269}

271/// The one author (index among the authors of `commands`, a frontmatter flush's queue) marked as the corresponding
272/// author, for the one address of a class's list left over (`address`, its text), among the authors it may be of
273/// (`within`, indices; all when `None`). An author is marked
274/// - by a note in their own name that names the corresponding author ([`names_correspondence`]: jheppub's
275///   `\note{Corresponding author.}`, `\note[*]{…}`; `\thanks`, `\footnote`), or
276/// - by a mark they carry whose legend names the corresponding author: an affiliation line's (`\author[a,*]{…}` under
277///   `\affiliation[*]{Corresponding author}`, `$^{1,*}$` under `\address{$^*$ Author to whom any correspondence should
278///   be addressed.}`) or a note's — its own mark (`\note[*]{…}`) or the number a jheppub-family `\note` prints its text
279///   under ([`AuthorNote::mark`]): the authors who write that number on their names are the ones it is about.
280///
281/// A note or a legend that says something else ("Equal contribution", "Since October 1, 2015 at …", 1510.03434) marks
282/// no one. Exactly one marked author among `within` gets the address; none or two and more, the author block keeps it
283/// (1802.03677's "¹ Corresponding author." note, whose 1 both M.V. Smirnov and J.J. Ling carry; 2606.26596's
284/// "Corresponding authors." under the `*` of two authors; user ruling 2026-10-21). A mark with no legend at all is no
285/// evidence of correspondence (a bare `*`, 2111.00753's two `$^{*}$`), save where the address itself says it is the
286/// corresponding author's (`\emailAdd{zyxu@gzu.edu.cn(Corresponding author)}`) and exactly one author among `within`
287/// carries a `*` with no legend (2308.06532's `\author[1,*]{Zhaoyi Xu}`, "E-mail: zyxu@gzu.edu.cn(Corresponding
288/// author)" in its PDF).
289pub(super) fn marked_corresponding_author(
290  commands: &[RawFrontmatter],
291  address: &str,
292  within: Option<&[usize]>,
293) -> Option<usize> {
294  let authors: Vec<&RawFrontmatter> = commands
295    .iter()
296    .filter(|entry| {
297      entry.0 == "ltx:creator" && entry.1.get("role").map(String::as_str) == Some("author")
298    })
299    .collect();
300  let names: Vec<Tokens> = authors.iter().map(|entry| queued_name(&entry.2)).collect();
301  let mut notes = author_notes(names.iter(), &[
302    T_CS!("\\thanks"),
303    T_CS!("\\footnote"),
304    T_CS!("\\note"),
305  ]);
306  // …and each note given apart from the names, the author's before it, as it is placed (1909.06107's
307  // `\author[a,1]{and V.~Sharyy}\note{Corresponding author}`; 2404.00867's `\note[*]{…}` after Paul J. Steinhardt's
308  // `\author`), its mark its own (`_mark`, jheppub's `\note[*]`)
309  let mut before = None;
310  for entry in commands {
311    match (entry.0.as_str(), entry.1.get("role").map(String::as_str)) {
312      ("ltx:creator", Some("author")) => before = Some(before.map_or(0, |at: usize| at + 1)),
313      ("ltx:contact", Some("note")) => {
314        if let Some(owner) = before {
315          notes.push(AuthorNote {
316            owner,
317            mark: entry.1.get("_mark").and_then(|mark| {
318              clean_frontmatter_labels(mark, "affiliation")
319                .into_iter()
320                .next()
321            }),
322            numbered: false,
323            names_correspondence: names_correspondence(&visible_name_text(
324              last_group(&entry.2).unlist_ref(),
325            )),
326          });
327        }
328      },
329      _ => {},
330    }
331  }
332  // Each author's labels: its class's optional ones (`\author[a,*]`, `annotations=`) and its marks'.
333  let author_labels: Vec<Vec<String>> = authors
334    .iter()
335    .zip(&names)
336    .map(|(entry, name)| {
337      let mut labels = entry
338        .1
339        .get("annotations")
340        .map(|given| clean_frontmatter_labels(given, "affiliation"))
341        .unwrap_or_default();
342      for operand in author_mark_operands(name.unlist_ref()) {
343        labels.extend(clean_frontmatter_labels(
344          &Tokens::new(operand).to_string(),
345          "affiliation",
346        ));
347      }
348      labels
349    })
350    .collect();
351  // The legends: each affiliation's label (`\affiliation[*]{…}`'s, or the line's leading mark) and whether it names the
352  // corresponding author; and each note's mark (`\note[*]{…}`, the n-th `\note`'s n).
353  let mut legends: Vec<(String, bool)> = commands
354    .iter()
355    .filter(|entry| {
356      entry.0 == "ltx:contact" && entry.1.get("role").map(String::as_str) == Some("affiliation")
357    })
358    .filter_map(|entry| {
359      let content = last_group(&entry.2);
360      let inner = queued_affiliation_text(&content);
361      let label = match entry.1.get("label").filter(|label| !label.is_empty()) {
362        Some(label) => clean_frontmatter_labels(label, "affiliation")
363          .into_iter()
364          .next(),
365        None if marker_leads(&inner) => {
366          entry_mark(inner.unlist_ref()).and_then(|mark| mark_label(&mark))
367        },
368        None => None,
369      }?;
370      Some((
371        label,
372        names_correspondence(&visible_name_text(inner.unlist_ref())),
373      ))
374    })
375    .collect();
376  // (a note's counter number is its legend only where no affiliation's label is that number: the authors writing an
377  // affiliation's number request the affiliation, 2005.05695, 2005.14143, 2405.09762, 1403.3797, 2101.00866; a note's
378  // own mark, `\note[*]{…}`, is its legend whatever the affiliations are labelled)
379  let affiliation_legends = legends.len();
380  let note_legends: Vec<(String, bool)> = notes
381    .iter()
382    .filter_map(|note| Some((note.mark.clone()?, note.numbered, note.names_correspondence)))
383    .filter(|(mark, numbered, _)| {
384      !*numbered
385        || !legends[..affiliation_legends]
386          .iter()
387          .any(|(label, _)| label == mark)
388    })
389    .map(|(mark, _, says)| (mark, says))
390    .collect();
391  legends.extend(note_legends);
392  let candidate = |at: &usize| within.is_none_or(|within| within.contains(at));
393  let mut marked: std::collections::BTreeSet<usize> = notes
394    .iter()
395    .filter(|note| note.names_correspondence)
396    .map(|note| note.owner)
397    .collect();
398  for (at, labels) in author_labels.iter().enumerate() {
399    if labels.iter().any(|label| {
400      legends
401        .iter()
402        .any(|(legend, says)| legend == label && *says)
403    }) {
404      marked.insert(at);
405    }
406  }
407  let marked: Vec<usize> = marked.into_iter().filter(candidate).collect();
408  match marked.as_slice() {
409    [author] => return Some(*author),
410    [] => {},
411    _ => return None,
412  }
413  // (no author marked: the address itself naming correspondence, the one author carrying a `*` no legend explains)
414  if !names_correspondence(address) {
415    return None;
416  }
417  let star = clean_frontmatter_labels("*", "affiliation");
418  let starred: Vec<usize> = author_labels
419    .iter()
420    .enumerate()
421    .filter(|(at, labels)| {
422      candidate(at)
423        && labels
424          .iter()
425          .any(|label| star.contains(label) && !legends.iter().any(|(legend, _)| legend == label))
426    })
427    .map(|(at, _)| at)
428    .collect();
429  match starred.as_slice() {
430    [author] => Some(*author),
431    _ => None,
432  }
433}

435/// The content of a queued affiliation's entry inside the wrappers that read its marks (`\lx@affiliation@withinst`,
436/// `\lx@affiliation@withsup`), as written.
437pub(in crate::base_utilities) fn queued_affiliation_text(content: &Tokens) -> Tokens {
438  let mut inner = content.clone();
439  loop {
440    let v = inner.unlist_ref();
441    let wrappers = [
442      T_CS!("\\lx@affiliation@withinst"),
443      T_CS!("\\lx@affiliation@withsup"),
444    ];
445    let wrapped = v.first().is_some_and(|t| wrappers.contains(t))
446      && v.get(1).is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
447      && skip_group(v, 1) == v.len();
448    if !wrapped {
449      return inner;
450    }
451    inner = Tokens::new(v[2..v.len() - 1].to_vec());
452  }
453}

455/// The content of the last top-level brace group of `tokens` (a queued frontmatter command's content argument); empty
456/// when there is none.
457pub(in crate::base_utilities) fn last_group(tokens: &Tokens) -> Tokens {
458  let v = tokens.unlist_ref();
459  let mut depth = 0usize;
460  let mut start = None;
461  let mut last = Tokens::new(Vec::new());
462  for (i, t) in v.iter().enumerate() {
463    match t.get_catcode() {
464      Catcode::BEGIN => {
465        if depth == 0 {
466          start = Some(i);
467        }
468        depth += 1;
469      },
470      Catcode::END if depth > 0 => {
471        depth -= 1;
472        if depth == 0
473          && let Some(open) = start.take()
474        {
475          last = Tokens::new(v[open + 1..i].to_vec());
476        }
477      },
478      _ => {},
479    }
480  }
481  last
482}

484/// The meaning recorder over a flush ([`begin_meaning_recorder`]), closed when it is ended or dropped: a flush leaving
485/// by an `Err` leaves none open, whose recording every later meaning assignment of the conversion would pay. Closed by
486/// its depth ([`end_meaning_recorder_at`]), so a displaced call's recorder an `Err` left open inside it is closed with
487/// it, never in its place.
488struct FlushMeanings {
489  depth: usize,
490  open:  bool,
491}
492
493impl FlushMeanings {
494  fn begin() -> Self {
495    FlushMeanings {
496      depth: begin_meaning_recorder(),
497      open:  true,
498    }
499  }
500
501  /// The meanings assigned since it began ([`end_meaning_recorder_at`]).
502  fn end(mut self) -> Vec<(SymStr, Option<Stored>)> {
503    self.open = false;
504    end_meaning_recorder_at(self.depth)
505  }
506}
507
508impl Drop for FlushMeanings {
509  fn drop(&mut self) {
510    if self.open {
511      end_meaning_recorder_at(self.depth);
512    }
513  }
514}

581    Some(Scope::Global),

638    .iter()
639    .filter(|entry| entry.0 == "ltx:contact")

757    });
758  }

763  );

768  );

773  );
774  assign_value(
775    NAMES_SHOWN_LABELS,
776    Stored::String(pin(&outer_shown_labels)),
777    Some(Scope::Global),
778  );
779  egroup()?;
780  Ok(())
781}

943      Some(p) => n = p,

957      None => return Ok(()),
958    }
959  }
960  document.close_to_node(root, true)
961}

968/// (`LaTeXML-structure.rnc:34`), and `Para.class` — hence the body group — includes

971/// yields. This pass looks at the LEADING REGION — the root children between the
972/// resources and the bound (the abstract's own `ltx:_Frontmatter_Capture_` position
973/// marker when it has one, else everything built so far) — and classifies each node:
974///

976///   anchor candidate;
977/// * a content-free node (`node_is_content_free`: a pagebreak, an empty paragraph, an
978///   ink-free empty box) or PREAMBLE residue (`is_preamble_residue`: built before
979///   `\begin{document}`, where LaTeX forbids typesetting) is left in place and, if it
980///   preceded the anchor, moved after the placed frontmatter;
981/// * hand-typeset title-page LAYOUT — `para` / demotable `logical-block` / `pagination`
982///   with visible ink, `wrap_layout` only (the abstract-only fallback: a cover typeset
983///   before a `\begin{abstract}` with no `\maketitle`) — is wrapped, in its exact order,
984///   into `<ltx:titlepage>` (`titlepage_model = (FrontMatter.class |
985///   SectionalFrontMatter.class | Block.class)*` is the schema's container for exactly
986///   that; `para`→`block`, `logical-block` demoted bottom-up as in #244);
987/// * anything else — a `<TOC>`, a section, a float, a root `<rule>`, a VISIBLE body
988///   paragraph on the `\maketitle` path — ends the region: it stays where it is and the
989///   frontmatter is placed ahead of it, so the document still opens with its front
990///   group (Perl places at the current point and strands the title; user-approved
991///   surpass, OXIDIZED_DESIGN #262).
992///
993/// The frontmatter is inserted after the wrapped titlepage, else after the leading
994/// first-group run (`FrontmatterAnchor::LeadingFrontmatter`) or at the top, via the
995/// same `_Capture_` insert→fill→unwrap machinery Perl's fallback uses. Run at every
996/// placement point — `\lx@frontmatterhere`, both `\lx@frontmatter@fallback` branches,
997/// the `{titlepage}` environment's `after_construct` — so a leading pagebreak before a
998/// built titlepage is settled when that titlepage is built, not by a later pass. The
999/// bound matters: without the marker a sectionless manual's whole body would read as
1000/// cover (jourcl's cover LETTER, derivative's after-abstract cover). Superseded by this
1001/// pass: #242's content-free gate, #245's `\end{document}` relocation, #246's anchor
1002/// helpers. Kept beside it: the conservative title promotion (`maybe_promote_leading_title`,
1003/// user-ruled), `maybe_dedup_leading_title_ink` (#6924), `insert_late_frontmatter`, the
1004/// marker sweep in `insert_frontmatter` and the core `finalize_rec`.
1005///
1006/// Witnesses: amsldoc/tkz-doc/tzplot (#242, 14 docs), gitinfo2/gitlog/FrontespizioScudo/
1007/// toptesi-example-* (#245, 6), toptesi-example-magistrale/tipfr-doc/isosigns-docs/
1008/// bootstrapicons-docs/derivative/jourcl (#246, 16), the faketitlepage /
1009/// promote_center_title / svabstract fixtures, `114_streaming_structure` (eager ≡ streaming).
1010pub fn place_frontmatter(
1011  document: &mut Document,
1012  wrap_layout: bool,
1013  anchor: FrontmatterAnchor,
1014) -> Result<()> {
1015  // Drain the raw frontmatter queue first, as `insert_frontmatter` does, so
1016  // `frontmatter_pending` and the flush agree: the `{titlepage}` environment's
1017  // `after_construct` is the one placement point without its own digest hook, and
1018  // under `--streaming` (digestion interleaved with construction) a `\title` given
1019  // after `\end{titlepage}` may still sit undigested there.
1020  digest_front_matter()?;
1021  let Some(root) = document.get_document().get_root_element() else {
1022    return insert_frontmatter(document);
1023  };
1024  let kids: Vec<Node> = root
1025    .get_child_nodes()
1026    .into_iter()
1027    .filter(|n| n.get_type() == Some(NodeType::ElementNode))
1028    .collect();
1029  let mut i = 0;
1030  while i < kids.len() && document::get_node_qname(&kids[i]) == pin_static("ltx:resource") {
1031    i += 1;
1032  }
1033  let bound = kids[i..]
1034    .iter()
1035    .position(|n| document::get_node_qname(n) == pin_static("ltx:_Frontmatter_Capture_"))
1036    .map_or(kids.len(), |k| i + k);
1037  // The leading prefix: first-group elements and content-free nodes, in any order.
1038  let mut last_first: Option<Node> = None;
1039  let mut movers: Vec<Node> = Vec::new();
1040  while i < bound {
1041    if is_frontmatter_group_element(&kids[i]) {
1042      last_first = Some(kids[i].clone());
1043    } else if node_is_content_free(&kids[i]) || is_preamble_residue(&kids[i]) {
1044      movers.push(kids[i].clone());
1045    } else {
1046      break;
1047    }
1048    i += 1;
1049  }
1050  // Hand-typeset cover before the abstract → <titlepage>, in order.
1051  let mut titlepage: Option<Node> = None;
1052  if wrap_layout && i < bound {
1053    let end = i
1054      + kids[i..bound]
1055        .iter()
1056        .take_while(|k| is_titlepage_layout(k))
1057        .count();
1058    if kids[i..end].iter().any(|k| !node_is_content_free(k)) {
1059      // The flush ends the paragraph it interrupts before the cover moves: text right
1060      // before an abstract is still the open paragraph, and wrapping it open left the
1061      // insertion point inside the renamed `block` (`Error:malformed:ltx:document` at
1062      // `\end{document}`; serbian-def-cyr/proba).
1063      close_auto_closable_to_root(document, &root)?;
1064      titlepage = Some(wrap_as_titlepage(document, &root, &kids[i..end])?);
1065    }
1066  }
1067  // Nothing queued (or the abstract-only deferral): `insert_frontmatter` keeps its
1068  // side effects; a leading pagebreak before a built titlepage still settles here.
1069  if !frontmatter_pending() {
1070    insert_frontmatter(document)?;
1071    if let Some(tail) = last_first {
1072      move_content_free_after(&movers, tail)?;
1073    }
1074    return Ok(());
1075  }
1076  // Visible body content before the flush (prose or a class-drawn cover before
1077  // `\maketitle`, a section before a late `\title`) does not hold the frontmatter
1078  // back: the schema's document model puts the frontmatter group first
1079  // (`LaTeXML-structure.rnc:34`), so it goes to the head and that content follows
1080  // it in its own order. Perl places at the current point (`\maketitle`,
1081  // Base_Utility.pool.ltxml:918), stranding `<title>` after the body — schema-invalid
1082  // (user-approved surpass 2026-09-23, OXIDIZED_DESIGN #262). The flush ends the
1083  // paragraph it interrupts, as every flush point does in LaTeX (`\maketitle`:
1084  // article.cls:203 opens with `\par`, report/book with `\begin{titlepage}`;
1085  // `\section`; `\end{document}`) and as the current-point insertion did by
1086  // auto-closing: text after `\maketitle` starts a new paragraph instead of
1087  // continuing one that now stands after the frontmatter.
1088  close_auto_closable_to_root(document, &root)?;
1089  let after = titlepage.or(match anchor {
1090    FrontmatterAnchor::Top => None,
1091    FrontmatterAnchor::LeadingFrontmatter => last_first.clone(),
1092  });
1093  let placed = insert_frontmatter_at(document, after.as_ref())?;
1094  // Content-free nodes that preceded the placed run now follow it, in their order.
1095  let tail = match (placed.last().cloned(), last_first) {
1096    (Some(p), Some(f)) => Some(if node_precedes(&p, &f) { f } else { p }),
1097    (p, f) => p.or(f),
1098  };
1099  if let Some(tail) = tail {
1100    move_content_free_after(&movers, tail)?;
1101  }
1102  Ok(())
1103}
