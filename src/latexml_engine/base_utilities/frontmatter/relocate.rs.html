8/// Find all dummy frontmatter entries (role "pending") containing unattached annotations
9/// and attempt to attach to the appropriate frontmatter, based on the identifying labels.
10/// Perl: relocateAnnotations($document).
11pub(in crate::base_utilities) fn relocate_annotations(document: &mut Document) -> Result<()> {
12  // Find dummy frontmatter elements containing not-yet-attached annotations
13  let pending_nodes = document.findnodes(".//*[@role='pending']", None);
14  // The author block's own creator, which the frontmatter emitter writes (`observer::emitter::emit::block_creator`): the
15  // trailing nameless creator the orphan procedure below shares into, written by the emitter where it would promote a
16  // stub — no target by label (the block's items go through the orphan rules: a sole author's line, a copy an author
17  // holds, a named owner's), placed below the last named author, and gone where nothing was shared into it. Witnesses
18  // 1012.4847, 1106.3045, 2009.10916.
19  // (searched for only where a flush wrote it: the whole-document search ran for every paper, `block_creator_written`)
20  let block_creator = block_creator_written()
21    .then(|| document.findnode("//ltx:creator[@_blockcreator]", None))
22    .flatten();
23  if pending_nodes.is_empty() {
24    return settle_block_creator(document, block_creator);
25  }
26  // Collect the frontmatter that have attachment labels
27  let mut labeltable: HashMap<String, Vec<Node>> = HashMap::default();
28  // fallback: Same, but without prefix
29  let mut unlabeltable: HashMap<String, Vec<Node>> = HashMap::default();
30  for target in document.findnodes(".//*[@_annotations]", None) {
31    let target_role = target.get_attribute("role").unwrap_or_default();
32    if target_role == "pending" || block_creator.as_ref() == Some(&target) {
33      continue;
34    }
35    // Dedup labels PER TARGET: a creator that cites the same annotation label
36    // from several of its authors (LLNCS `\author{A\inst{1} B\inst{2,1}}` — both
37    // A and B cite institution 1) must receive that affiliation ONCE, not once
38    // per citing author. Each duplicate label would otherwise push `target`
39    // again, and the pending note gets cloned into it per push. Witness arXiv
40    // 2603.23669: two-author creators rendered "Mila … Mila … McGill … McGill".
41    let annotations = target.get_attribute("_annotations").unwrap_or_default();
42    let mut seen: HashSet<&str> = HashSet::default();
43    for label in annotations.split(',') {
44      if label.is_empty() || !seen.insert(label) {
45        continue;
46      }
47      labeltable
48        .entry(label.to_string())
49        .or_default()
50        .push(target.clone());
51      // Misuse of labelling macros can lead to a prefix mismatch, so fall back to
52      // a prefix-stripped ("noprefix") index — e.g. an affiliation labelled
53      // `institute:1` can still find an author who cited `affiliation:1`.
54      // BUT a creator's OWN role-sequence label (`author:N` / `editor:N` /
55      // `translator:N`, auto-assigned in `\lx@add@frontmatter@now`) must NEVER
56      // enter that fallback: it would let a SHARED affiliation's `affiliation:1`
57      // bind to the first author's `author:1` purely by number, stranding one
58      // shared `\institute` on author 1 (Perl does exactly this — arXiv:2402.19043
59      // / WDM, 5 authors + one `\institute`, no `\inst`; OXIDIZED_DESIGN #159). A
60      // genuine per-author affiliation instead matches EXACTLY via `labeltable`
61      // (the `affiliation:1` that `\inst{1}` requests), which is untouched here.
62      if let Some(pos) = label.find(':')
63        && label[..pos] != target_role
64      {
65        unlabeltable
66          .entry(label[pos + 1..].to_string())
67          .or_default()
68          .push(target.clone());
69      }
70    }
71  }
72  let authors = document.findnodes("//ltx:creator[@role='author'][ltx:personname]", None);
73  // the marks each author's name shows (`$^{a,b}$`: two), read where an institute labelled by a mark asks (below)
74  let mut author_marks: Option<Vec<Vec<String>>> = None;
75  // Only a label a mark set is answered by the authors showing that mark, as the mark's own keys (`_bymark`, read as the
76  // authors' are): a number `labelseq` gave is no mark (aa's one `\institute{{1} Univ A \\ {2} Univ B …}` is one
77  // institute, `affiliation:1`, under authors marked 1,2 / 2,3: astro-ph0305539).
78  let by_mark: HashMap<String, Vec<String>> = pending_nodes
79    .iter()
80    .filter_map(|pending| {
81      let keys = pending.get_attribute("_bymark")?;
82      let label = pending.get_attribute("_annotations")?;
83      Some((label, keys.split(',').map(str::to_string).collect()))
84    })
85    .collect();
86  // A label's targets: the exact index, then the prefix-stripped fallbacks for a misused prefix — creators only, the
87  // annotations a prefix is misused on: a title's `tnote:t1` must not take an author's orphaned `fn:t1` (63h review,
88  // `elsarticle_orphan_fntext_stays_off_the_title`).
89  let indexed_targets = |label: &str| -> Option<Vec<Node>> {
90    if let Some(targets) = labeltable.get(label) {
91      return Some(targets.clone());
92    }
93    let noprefix = label
94      .find(':')
95      .map(|pos| &label[pos + 1..])
96      .unwrap_or(label);
97    [
98      unlabeltable.get(label),
99      labeltable.get(noprefix),
100      unlabeltable.get(noprefix),
101    ]
102    .into_iter()
103    .flatten()
104    .map(|targets| {
105      targets
106        .iter()
107        .filter(|target| target.get_name() == "creator")
108        .cloned()
109        .collect::<Vec<_>>()
110    })
111    .find(|targets| !targets.is_empty())
112  };
113  // A superscript mark and a bracketed label printed as the same superscript are one label (authblk sets
114  // `\author[1]{Bob}` and `\affil[1]{Univ X}` as a `\textsuperscript{1}`, `\AB@authnote`/`\AB@affilnote`,
115  // authblk.sty:98-99, :142, :162): the label a mark set (`\affil{$^1$Univ X}`, `affiliation:1` for the authors
116  // showing that mark) also reaches the authors whose label is its mark (`\author[1]{Bob}`, `1`), and a bracketed
117  // label (`\affil[1]{Univ X}`, `1`) the authors requesting it by a mark (`\author{Bob$^{1}$}`, whose answered mark
118  // the name no longer shows; `\inst{1}`: `affiliation:1`): Bob lost Univ X to the author whose spelling matched
119  // (77n r6 review probes f2, f7). Only authblk's names and `\inst` request `affiliation:N` from a printed label, so
120  // a class that renumbers its labels (elsarticle prints `\author[1]` as "a") is untouched (r7 review: elsarticle,
121  // cas, llncs, revtex, aastex, IEEE probes and 12 elsarticle/cas papers unchanged).
122  let find_targets = |label: &str| -> Option<Vec<Node>> {
123    let mut targets = indexed_targets(label).unwrap_or_default();
124    let respelt: Vec<Node> = if let Some(keys) = by_mark.get(label) {
125      keys
126        .iter()
127        .filter_map(|key| labeltable.get(key.as_str()))
128        .flatten()
129        .filter(|target| target.get_name() == "creator")
130        .cloned()
131        .collect()
132    } else if !label.is_empty() && !label.contains(':') {
133      labeltable
134        .get(&s!("affiliation:{label}"))
135        .into_iter()
136        .flatten()
137        .filter(|target| target.get_name() == "creator")
138        .cloned()
139        .collect()
140    } else {
141      Vec::new()
142    };
143    for target in respelt {
144      if !targets.contains(&target) {
145        targets.push(target);
146      }
147    }
148    (!targets.is_empty()).then_some(targets)
149  };
150  // An orphan no one cites is kept all the same, as LaTeX prints it. An institute-level contact, or a thanks note
151  // (svjour3's `\thankstext` without its `\thanksref`; an EPJ institute's own note, 63g review, repro
152  // svjour3_epj_title_and_institute_notes), is shared as a matter of course. A person's own note (elsarticle's
153  // `\fntext` without its `\fnref`) lost the link to its person, which is warned about; one placed on the block (the
154  // `addresses:block` label, the shared creator below) lost none: the evidence gave it to the block (ijcai's `\emails
155  // {$^{*}$Corresponding author}` under two starred names, 2609.15094; OXIDIZED_DESIGN_DIVERGENCES #908).
156  let shared_orphan = |note: &Node| {
157    let role = note.get_attribute("role").unwrap_or_default();
158    is_shared_contact_role(&role)
159      || role == "thanks"
160      || note.get_attribute("_label").as_deref() == Some("addresses:block")
161  };
162  // Beyond-Perl (OXIDIZED_DESIGN #159): an orphan's label is one institute with what inherited its label (its
163  // `\email`/`\url`), and it goes to the author the evidence names, else is shared:
164  // - an institute labelled by a mark (`\affiliation{$^{a}$Univ A}`, an `affiliation:a` label) is the authors' whose
165  //   names show that mark (`Ann Able$^{a}$`, a visible `<ltx:sup>` the author line kept), as the reader pairs them
166  //   (revtex 2011.01984, 2301.08449; repro revtex_marked_affiliations_link_by_mark); the rest are then not paired by
167  //   position;
168  // - an institute's `\at` names (svjour3 `fuzzy:` label) that missed the exact name match are the authors with those
169  //   surnames, when each names one (`A.M.Bykov \at …` for "Andrei Bykov", 1205.2208; `Olivier Augereau, Koichi Kise,
170  //   and Motoi Iwata \at …`, 1811.03214, every one of them), in any order; unless an unlabelled
171  //   piece sits among them, svjour3's `\institute{A \and B \at X}` where a name-only piece shares the next institute
172  //   (1709.00485, 2003.07295; repro svjour3_name_only_piece_shares_the_next_institute);
173  // - numbered institutes (`affiliation:N`, no `\inst`), as many as the authors, are the i-th author's: Perl's numeric
174  //   fallback and the grid `distribute_upfront_contacts` pairs (`\author{A \and B}` + `\institute{X \and Y}`);
175  // - a sole author's orphans are that author's, as Perl attaches them; its institutes only where no institute is
176  //   linked otherwise (an author marked for one institute is not given the one no one marks,
177  //   `author_block_orphan_mark_is_kept_red`).
178  // Any other orphan is shared, gathered into ONE trailing name-LESS `<ltx:creator role="author">` below the whole
179  // author row, where Perl warns and drops it, or strands one shared `\institute` on author 1 (arXiv:2402.19043, WDM: 5
180  // authors + one `\institute`). The ar5iv theme breaks a last-position shared affiliation out to full width. We reuse
181  // the first such pending stub in place: it is already a `<creator>` right after the last real author.
182  let mut orphan_labels: Vec<(String, bool)> = Vec::new();
183  let mut linked_institute = false;
184  for pending in &pending_nodes {
185    for note in element_nodes(pending) {
186      let label = note.get_attribute("_label").unwrap_or_default();
187      let institute = is_institute_role(&note.get_attribute("role").unwrap_or_default());
188      if label.is_empty() {
189        continue;
190      } else if find_targets(&label).is_some() {
191        linked_institute |= institute;
192      } else if let Some(entry) = orphan_labels.iter_mut().find(|(known, _)| *known == label) {
193        entry.1 |= institute;
194      } else {
195        orphan_labels.push((label, institute));
196      }
197    }
198  }
199  let mut owners: HashMap<String, Vec<Node>> = HashMap::default();
200  for (label, institute) in orphan_labels.iter() {
201    let Some(keys) = by_mark.get(label) else {
202      continue;
203    };
204    if !*institute {
205      continue;
206    }
207    let author_marks = author_marks.get_or_insert_with(|| {
208      authors
209        .iter()
210        .map(|author| shown_marks(document, author))
211        .collect()
212    });
213    let marked: Vec<Node> = (0..authors.len())
214      .filter(|&i| keys.iter().any(|key| author_marks[i].contains(key)))
215      .map(|i| authors[i].clone())
216      .collect();
217    if !marked.is_empty() {
218      owners.insert(label.clone(), marked);
219    }
220  }
221  let linked_institute = linked_institute || !owners.is_empty();
222  let unaffiliated = !linked_institute
223    && authors.iter().all(|author| {
224      element_nodes(author).iter().all(|child| {
225        child.get_name() != "contact"
226          || !is_institute_role(&child.get_attribute("role").unwrap_or_default())
227      })
228    });
229  let institutes: Vec<&String> = orphan_labels
230    .iter()
231    .filter(|(label, institute)| *institute && !owners.contains_key(label))
232    .map(|(label, _)| label)
233    .collect();
234  let numbered = |label: &str| {
235    label
236      .strip_prefix("affiliation:")
237      .is_some_and(|n| !n.is_empty() && n.bytes().all(|b| b.is_ascii_digit()))
238  };
239  if !institutes.iter().any(|label| numbered(label)) {
240    let surnames: Vec<Option<String>> = authors
241      .iter()
242      .map(|author| {
243        document
244          .findnode("ltx:personname", Some(author))
245          .and_then(|person| surname_key(&person.get_content()))
246      })
247      .collect();
248    // (any fuzzy orphan, an institute's or a contact's: `\correspondingauthor{Weipeng Lin, Lin Tang}`'s correspondent is
249    // both authors', 2408.02230, 2404.14129, 2410.19543, 2008.13733 — it had been orphaned with a Warning)
250    let fuzzy_orphans: Vec<&String> = orphan_labels
251      .iter()
252      .filter(|(label, _)| label.starts_with("fuzzy:") && !owners.contains_key(label))
253      .map(|(label, _)| label)
254      .collect();
255    for label in &fuzzy_orphans {
256      let Some(names) = label.strip_prefix("fuzzy:") else {
257        continue;
258      };
259      // each name the piece gives before its `\at` (a list: commas, " and ", " \& "; spaces are `_` in a label)
260      let mut named: Vec<Node> = Vec::new();
261      let each_named_once = names
262        .split([',', '&'])
263        .flat_map(|part| part.split("_and_"))
264        .map(|name| name.trim_matches('_').trim_start_matches("and_"))
265        .filter(|name| !name.is_empty())
266        .all(|name| {
267          let key = surname_key(name);
268          let matches: Vec<usize> = (0..authors.len())
269            .filter(|&i| key.is_some() && surnames[i] == key)
270            .collect();
271          if let [i] = matches.as_slice() {
272            if !named.contains(&authors[*i]) {
273              named.push(authors[*i].clone());
274            }
275            true
276          } else {
277            false
278          }
279        });
280      if each_named_once && !named.is_empty() {
281        owners.insert((*label).clone(), named);
282      }
283    }
284  } else if unaffiliated
285    && authors.len() > 1
286    && institutes.len() == authors.len()
287    && institutes.iter().all(|label| numbered(label))
288  {
289    owners.extend(
290      institutes
291        .iter()
292        .map(|label| (*label).clone())
293        .zip(authors.iter().map(|author| vec![author.clone()])),
294    );
295  }
296  if let [sole] = authors.as_slice() {
297    for (label, institute) in &orphan_labels {
298      if !*institute || unaffiliated {
299        owners
300          .entry(label.clone())
301          .or_insert_with(|| vec![sole.clone()]);
302      }
303    }
304  }
305  let mut shared_creator: Option<Node> = block_creator.clone();
306  // An orphan's owners among the authors (below), by its label and role.
307  // (several owners for an institute, or for a person's note whose fuzzy label names them each — a correspondent
308  // naming two; not an institute's `\email`/`\url` that inherited its label, which stays shared, 1205.6171)
309  let owned_by = |label: &str, note: &Node| {
310    let role = note.get_attribute("role").unwrap_or_default();
311    owners.get(label).filter(|named| {
312      named.len() == 1
313        || is_institute_role(&role)
314        || (label.starts_with("fuzzy:") && !is_shared_contact_role(&role))
315    })
316  };
317  // The annotations an author takes, by kind and text, with their labels: a copy of one under another label that no
318  // author takes is no orphan to share (`\address{$^{1,5}$Lab …}` with no author marked 5, osa_journal_sty.rs
319  // `lines_per_label`, 1101.5841). Read before placing any, as the orphan copy may come first (`$^{5,1}$`).
320  let mut taken: Vec<(String, Option<String>, String, String)> = Vec::new();
321  for pending in &pending_nodes {
322    for note in element_nodes(pending) {
323      let label = note.get_attribute("_label").unwrap_or_default();
324      let content = copy_text(&note);
325      if !label.is_empty()
326        && !content.is_empty()
327        && (find_targets(&label).is_some() || owned_by(&label, &note).is_some())
328      {
329        taken.push((note.get_name(), note.get_attribute("role"), content, label));
330      }
331    }
332  }
333  let mut held = HeldAnnotations::new(&authors);
334  for mut pending in pending_nodes {
335    let mut promote_this = false;
336    // the stub's notes placed elsewhere, which a promoted stub must not keep a second time
337    let mut placed: Vec<Node> = Vec::new();
338    for note in element_nodes(&pending) {
339      let label = note.get_attribute("_label").unwrap_or_default();
340      // An annotation given before any author with no label to find one by (cas's `\affiliation{…}` above the first
341      // `\author`, which cas-common.sty:1252-1290 prints under the authors) is an orphan like the labelled ones below:
342      // a sole author's, else shared. Perl drops it unplaced. Guard:
343      // `perfect_kernel_batch63::cas_affiliation_before_author`.
344      let unlabelled = label.is_empty();
345      // (an empty one is nothing to keep: it made a nameless creator of its own; nor one an author already holds — a
346      // store set between two `\maketitle`s is handed on to the authors a later `\author` names, and queued too:
347      // `perfect_kernel_batch59::store_set_before_a_superseding_author`)
348      let empty = annotation_value(&note).trim().is_empty() && element_nodes(&note).is_empty();
349      // (metadata, an `ltx:rdf`, is held by an author who has its value in any annotation: 2608.17008's `\email{x}` ahead
350      // of the authors, then `\email[show]{x}` on the first, is that author's displayed address: `HeldAnnotations`)
351      if unlabelled && (pending.get_name() != "creator" || empty || held.holds(&authors, &note)) {
352        continue;
353      }
354      let targets = if unlabelled {
355        match authors.as_slice() {
356          [author] => Some(vec![author.clone()]),
357          _ => None,
358        }
359      } else {
360        find_targets(&label)
361      };
362      if let Some(targets) = targets {
363        placed.push(note.clone());
364        for target in targets {
365          // (nor twice under two labels: an entry marked with several labels is one copy a label — OSA's
366          // `\address{$^{1,2}$Univ X}`, osa_journal_sty.rs `lines_per_label`, 1406.0824 — which reaches an author
367          // marked with two of them once; two entries under one label are two, as the PDF prints them; an annotation
368          // without text, an icon's link, is compared by nothing; nor a picture's note — a plain `role="note"` holding a
369          // picture or an image, `emit::picture_note_call`, its creator's own label —, whose text is its picture's, two
370          // pictures printing the same text two notes (`Ann Able \tikz\node{Zebraword}; \tikz\node{Zebraword};`, review
371          // revCR8 finding 3; OD #989); a `\thanks` holding a logo (role `thanks`) is compared by its text as any other note
372          // (review revCR8b finding 3). Guards `perfect_kernel_batch64u::osa_several_labels_one_line_each`,
373          // `perfect_kernel_batch61::marked_affiliation_review_shapes`, `perfect_kernel_batchcr8::two_same_text_pictures_are_two_notes`)
374          let (role, content) = (note.get_attribute("role"), annotation_value(&note));
375          let pictured = note.get_name() == "note"
376            && role.as_deref() == Some("note")
377            && !document
378              .findnodes(".//ltx:picture | .//ltx:graphics", Some(&note))
379              .is_empty();
380          if !content.trim().is_empty()
381            && !pictured
382            && element_nodes(&target).iter().any(|c| {
383              c.get_name() == note.get_name()
384                && c.get_attribute("role") == role
385                && c.get_attribute("_label").is_some_and(|l| l != label)
386                && annotation_value(c).trim() == content.trim()
387            })
388          {
389            continue;
390          }
391          DebugFeature!("frontmatter", "FRONT Moving annotation for {label}");
392          let mut target = target;
393          document.append_clone(&mut target, vec![note.clone()])?;
394          contact_in_title_as_note(document, &target)?;
395          at_queue_position(&target);
396          held.touched(&target);
397        }
398      } else {
399        let content = copy_text(&note);
400        if !unlabelled
401          && owned_by(&label, &note).is_none()
402          && taken.iter().any(|(name, role, text, other)| {
403            *name == note.get_name()
404              && *role == note.get_attribute("role")
405              && *text == content
406              && *other != label
407          })
408        {
409          placed.push(note.clone());
410          continue;
411        }
412        // (none for an orphan whose fuzzy label names its owners: a correspondent naming two authors, placed below; a
413        // sole author's orphan still warns, 63h)
414        let named_by_label = label.starts_with("fuzzy:") && owned_by(&label, &note).is_some();
415        // (an emitted paper's shared orphans are the author block's, which the emitter's creator holds: any other
416        // orphan there is Perl's, warned and dropped — `block_creator` above; 1012.4847, 1106.3045, 2009.10916)
417        let shares = !emitted() || label == "addresses:block";
418        if (!unlabelled && !shared_orphan(&note) && !named_by_label)
419          || (!shares && owned_by(&label, &note).is_none())
420        {
421          let mut known: Vec<&String> = labeltable.keys().collect();
422          known.sort();
423          Warn!(
424            "unexpected",
425            "annotation",
426            s!("Orphaned frontmatter annotation couldn't find target for label={label}"),
427            s!(
428              "known labels={}",
429              known
430                .iter()
431                .map(|k| k.as_str())
432                .collect::<Vec<_>>()
433                .join(",")
434            )
435          );
436        }
437        // (an institute several authors share is each one's; its `\email`/`\url` names one of them, which, unknown,
438        // stays shared: `kono@rice.edu` under one `L. Ren, Q. Zhang, S. Nanot, and J. Kono \at …` piece, 1205.6171)
439        if let Some(named) = owned_by(&label, &note) {
440          placed.push(note.clone());
441          DebugFeature!(
442            "frontmatter",
443            "FRONT Giving orphaned annotation {label} to its authors"
444          );
445          for owner in named {
446            let mut owner = owner.clone();
447            document.append_clone(&mut owner, vec![note.clone()])?;
448            at_queue_position(&owner);
449            held.touched(&owner);
450          }
451        } else if shares {
452          DebugFeature!(
453            "frontmatter",
454            "FRONT Sharing orphaned annotation {label} below authors"
455          );
456          match &mut shared_creator {
457            // First orphan: promote THIS pending stub in place into the trailing shared creator (its note already
458            // lives inside it).
459            None => {
460              promote_this = true;
461            },
462            // Later orphans (e.g. the institute's `\email`) join it.
463            Some(sc) => {
464              let mut sc = sc.clone();
465              document.append_clone(&mut sc, vec![note.clone()])?;
466              held.touched(&sc);
467            },
468          }
469        }
470      }
471    }
472    // (what follows changes the stub, and so any author holding it)
473    held.touched(&pending);
474    if promote_this {
475      // pending → real trailing creator (role=pending would otherwise be dropped
476      // with the node; the `_annotations`/`_label` bookkeeping attrs are stripped
477      // at serialization). Name-LESS: it carries no `<ltx:personname>`.
478      document.set_attribute(&mut pending, "role", "author")?;
479      for note in placed {
480        document.remove_node(note);
481      }

483      // (cas's `\affiliation{…}` above its `\author`s; aastex's `\correspondingauthor`+`\email` first, 2408.07136)

488      shared_creator = Some(pending);
489    } else {

491    }
492  }
493  settle_block_creator(document, block_creator)
494}

514/// A kind of annotation [`HeldAnnotations`] reads: an element name and role, or (`None`) a contact or metadata of any role.
515type AnnotationKind = Option<(String, Option<String>)>;

517/// The annotations the authors hold, for [`relocate_annotations`]'s test of an unlabelled annotation: is one of the same
518/// kind (element name and role) and text on an author, or — for metadata, an `ltx:rdf` — a contact or metadata of its
519/// text? Each author's values of one kind are read the first time that kind is asked about (an author's element
520/// children, their text trimmed), and read again after anything changed the author or what it holds (`touched`): the
521/// answer is the one reading every author's children at the time of the question gives, and an author's children are
522/// read once per kind, not once per question.
523struct HeldAnnotations {
524  /// each author's position, by node (`Node::to_hashable`)
525  position: HashMap<usize, usize>,
526  /// per author: the trimmed texts of its children of each kind read so far (`None`: a contact or metadata, any role)
527  read:     Vec<HashMap<AnnotationKind, HashSet<String>>>,
528}
529
530impl HeldAnnotations {
531  fn new(authors: &[Node]) -> Self {
532    HeldAnnotations {
533      position: authors
534        .iter()
535        .enumerate()
536        .map(|(i, author)| (author.to_hashable(), i))
537        .collect(),
538      read:     vec![HashMap::default(); authors.len()],
539    }
540  }
541
542  /// Does an author hold `note`'s kind with `note`'s text (or, `note` being metadata, a contact or metadata of it)?
543  fn holds(&mut self, authors: &[Node], note: &Node) -> bool {
544    let name = note.get_name();
545    let kind = Some((name.clone(), note.get_attribute("role")));
546    let metadata = name == "rdf";
547    let value = annotation_value(note);
548    let value = value.trim();
549    authors.iter().enumerate().any(|(i, author)| {
550      self.values(i, author, &kind).contains(value)
551        || (metadata && self.values(i, author, &None).contains(value))
552    })
553  }
554
555  /// The trimmed texts of author `i`'s element children of `kind`, read once.
556  fn values(&mut self, i: usize, author: &Node, kind: &AnnotationKind) -> &HashSet<String> {
557    self.read[i].entry(kind.clone()).or_insert_with(|| {
558      element_nodes(author)
559        .iter()
560        .filter(|c| match kind {
561          Some((name, role)) => c.get_name() == *name && c.get_attribute("role") == *role,
562          None => matches!(c.get_name().as_str(), "contact" | "rdf"),
563        })
564        .map(|c| annotation_value(c).trim().to_string())
565        .collect()
566    })
567  }
568
569  /// `node` changed (an annotation placed on it, renamed in it, removed from it, or it moved): the authors it is or
570  /// sits in are read again.
571  fn touched(&mut self, node: &Node) {
572    let mut cur = Some(node.clone());
573    while let Some(n) = cur {
574      if let Some(&i) = self.position.get(&n.to_hashable()) {
575        self.read[i].clear();
576      }
577      cur = n.get_parent();
578    }
579  }
580}

636/// The annotation just appended to `target` (its last element child) moved to the place its call had in the queue:
637/// before the first of `target`'s annotations queued after it ([`QUEUE_ORDINAL`]). Rust-only (user ruling R8, print
638/// order: a creator's contacts follow the order the PDF prints them; OXIDIZED_DESIGN_DIVERGENCES #954): an annotation
639/// placed by its label is appended once every annotation is digested, after the ones attached when they were digested

645/// `\altaffiltext` — prints in the footer, after the block, where it is appended.
646pub(in crate::base_utilities) fn at_queue_position(target: &Node) {
647  let place = |node: &Node| {
648    node
649      .get_attribute(QUEUE_ORDINAL)
650      .and_then(|n| n.parse::<i64>().ok())
651  };
652  let children = element_nodes(target);
653  let Some(mut moved) = children.last().cloned() else {
654    return;
655  };
656  let Some(mut at) = place(&moved) else {
657    return;
658  };
659  // (an entry of a printed list the Cut cut, which an uncut copy of the list printed for this author earlier, keeps
660  // that copy's place: `frontmatter_api::cut::place_uncut_copies`, `\lx@cut@copy@place`)
661  if let Some(key) = moved.get_attribute(CUT_LIST)
662    && let Some(copy) = target
663      .get_attribute("_annotations")
664      .unwrap_or_default()
665      .split(',')
666      .filter(|label| label.starts_with("author:"))
667      .find_map(
668        |label| match lookup_value(&s!("{CUT_COPY_PLACE}{key}:{label}")) {
669          Some(Stored::Int(copy)) => Some(copy),
670          _ => None,
671        },
672      )
673  {
674    at = copy;
675    let _ = moved.set_attribute(QUEUE_ORDINAL, &at.to_string());
676  }
677  if !moved
678    .get_attribute("role")
679    .is_some_and(|role| BLOCK_ROLES.contains(&role.as_str()))
680  {
681    return;
682  }
683  // (a collaboration is the names row's, printed before every entry under it, whenever its call came: `\author{Ann
684  // Able$^{1}$ \and The XYZ Collaboration \\ $^{1}$Univ X}`, `perfect_kernel_batch97d`)
685  if let Some(mut later) = children
686    .iter()
687    .find(|c| {
688      **c != moved
689        && place(c).is_some_and(|p| p > at)
690        && c.get_attribute("role").as_deref() != Some("collaboration")
691    })
692    .cloned()
693  {
694    let _ = later.add_prev_sibling(&mut moved);
695  }
696}

762/// A script's marks ([`script_marks`]): `past` skips what opens the script and each item before its mark (the
763/// spelling's switches and spacing, [`past_mark_switches`]; the printed text's spaces).
764fn split_marks(script: &str, past: fn(&str) -> &str) -> Vec<String> {
765  // (the whole script under its switches, `\footnotesize\ensuremath{1,2}`: the 1 and the 2; 2411.14110)
766  let script = past(script);
767  let mut items = Vec::new();
768  let (mut depth, mut start) = (0usize, 0usize);
769  for (i, c) in script.char_indices() {
770    match c {
771      '{' => depth += 1,
772      '}' => depth = depth.saturating_sub(1),
773      ',' if depth == 0 => {
774        items.push(&script[start..i]);
775        start = i + 1;
776      },
777      _ => {},
778    }
779  }
780  items.push(&script[start..]);
781  // (a number range is each number of it, `1-4`, `1--4`, the printed `1–4`/`1−4`: `queue_digest::spelled_ranges`, the one
782  // reader of a range — Bauer's `$^{1-4}$`, 2012.15817)
783  let items: Vec<String> = items
784    .into_iter()
785    .flat_map(|item| {
786      let cleaned: String = past(item)
787        .chars()
788        .filter(|c| !c.is_whitespace() && *c != '{' && *c != '}')
789        .collect();
790      let spelled = spelled_ranges(vec![cleaned.clone()]);
791      if spelled.len() > 1 {
792        spelled
793      } else {
794        vec![item.to_string()]
795      }
796    })
797    .collect();
798  items
799    .iter()
800    .map(String::as_str)
801    .filter_map(|item| {
802      // (a number in brackets or parentheses is the number: `$^{[1],[2]}$` against the line's `$^1$`, 1803.06822; so is
803      // one after a footnote symbol glued to it, `\textsuperscript{*3}`, 2412.01183)
804      let item: String = past(item)
805        .trim_start_matches(|c: char| {
806          matches!(c, '[' | '(' | '*') || (!c.is_ascii() && !c.is_alphanumeric())
807        })
808        .chars()
809        .filter(|c| !matches!(c, '{' | '}') && !c.is_whitespace())
810        .collect();
811      let lead: String = item.chars().take_while(|c| c.is_alphanumeric()).collect();
812      let rest = &item[lead.len()..];
813      let symbol_after = rest.is_empty()
814        || rest.starts_with(['*', '\\', ')', ']'])
815        || rest.starts_with(|c: char| !c.is_ascii() && !c.is_alphanumeric());
816      (!lead.is_empty() && symbol_after).then_some(lead)
817    })
818    .collect()
819}

848/// The spacing a mark is set apart with (`{}^{a,\;b}`, `{}^{\rm b,\ c}`, `\,{}^{b,c}`: 1307.6166, hep-th0410034,
849/// 2311.14432), which no mark is: the control words; the control symbols `\,` `\;` `\:` `\!` `\ ` and the tie besides.
850/// For the token readers, as [`MARK_WRAPPERS`].
851const MARK_SPACING: [&str; 4] = ["\\quad", "\\qquad", "\\thinspace", "\\enspace"];

853/// `text` past the switches it opens with ([`MARK_WRAPPERS`]) and the spacing ([`MARK_SPACING`]), with the braces and
854/// spaces around them.
855fn past_mark_switches(text: &str) -> &str {
856  let mut rest = text;
857  loop {
858    rest = rest.trim_start_matches(|c: char| c == '{' || c == '~' || c.is_whitespace());
859    let Some(after) = rest.strip_prefix('\\') else {
860      return rest;
861    };
862    if let Some(symbol) = after.chars().next()
863      && matches!(symbol, ',' | ';' | ':' | '!' | ' ')
864    {
865      rest = &after[1..];
866      continue;
867    }
868    let len = after
869      .find(|c: char| !c.is_ascii_alphabetic())
870      .unwrap_or(after.len());
871    if len == 0
872      || !(crate::frontmatter_stores::FONT_DECLARATIONS.contains(&&rest[..=len])
873        || MARK_WRAPPERS.contains(&&rest[..=len])
874        || MARK_SPACING.contains(&&rest[..=len]))
875    {
876      return rest;
877    }
878    rest = &after[len..];
879  }
880}

882/// The marks of each superscript in a TeX source string, one list per superscript (`Ann Able$^{1,2}$ and Bob$^{1}$`
883/// gives [1, 2] and [1]; `^a` gives [a]; `\textsuperscript{b}` gives [b]).
884fn superscript_scripts(tex: &str) -> Vec<Vec<String>> {
885  const TEXTSUP: &str = "\\textsuperscript";
886  let mut scripts = Vec::new();
887  let mut rest = tex;
888  loop {
889    // (a `^` that is no `\^` accent: `Ren\^{e}`)
890    let caret = rest
891      .char_indices()
892      .find(|&(i, c)| c == '^' && !rest[..i].ends_with('\\'))
893      .map(|(i, _)| i);
894    let (at, skip) = match (caret, rest.find(TEXTSUP)) {
895      (Some(caret), Some(text)) if text < caret => (text, TEXTSUP.len()),
896      (Some(caret), _) => (caret, 1),
897      (None, Some(text)) => (text, TEXTSUP.len()),
898      (None, None) => break,
899    };
900    rest = rest[at + skip..].trim_start();
901    if let Some(inner) = rest.strip_prefix('{') {
902      let (script, after) = braced_group(inner);
903      scripts.push(script_marks(script));
904      rest = after;
905    } else if let Some(c) = rest.chars().next() {
906      scripts.push(script_marks(&c.to_string()));
907      rest = &rest[c.len_utf8()..];
908    }
909  }
910  scripts
911}

1132/// The label an affiliation line leads with, whether it was typed by hand, and the line without it: a superscript
1133/// mark (`$^{1}$`, `$^1$`, `\textsuperscript{1}` or another mark command, `\IEEEauthorrefmark{1}`, or spelled
1134/// otherwise, `spaced_mark_at`: `${}^{1}$`, `\( ^{1} \)`) or a
1135/// hand-printed number (`1 Univ A`, `2) Univ B`, `3. Univ C`), behind the spaces and the font and size switches it is set in (`\scriptsize 2) Osservatorio …`, kept). `None` for a
1136/// line leading with neither (user ruling R2, 2026-11-01: a hand-printed leading number is a label wherever the names
1137/// request numbers).
1138pub fn leading_line_label(content: &Tokens) -> Option<(String, bool, Tokens)> {
1139  let tokens = content.unlist_ref();
1140  let mut at = 0;
1141  while let Some(t) = tokens.get(at) {
1142    if *t == T_SPACE!() || mark_switch(t) {
1143      at += 1;
1144    } else {
1145      break;
1146    }
1147  }
1148  let prefix = &tokens[..at];
1149  let rebuilt = |label: String, typed: bool, from: usize| {
1150    let mut rest = prefix.to_vec();

1152      tokens[from..]
1153        .iter()
1154        .copied()
1155        .skip_while(|t| *t == T_SPACE!()),
1156    );

1158  };

1164      false,
1165      next,
1166    );
1167  }

1170    let (operand, next) = sup_operand_at(tokens, at + 2)?;
1171    if tokens
1172      .get(next)
1173      .is_none_or(|t| t.get_catcode() != Catcode::MATH)
1174    {
1175      return None;
1176    }
1177    return rebuilt(
1178      mark_key(Tokens::new(operand).untex_string().as_ref()),
1179      false,
1180      next + 1,
1181    );
1182  }
1183  // (a mark command or an alias of one, `is_mark_command`: IEEEtran's `\IEEEauthorrefmark{1}\{ms1801,
1184  // bob\}@x.edu` address line as `\textsuperscript{1}`'s, 2301.03018, 2510.06608; read literally, its addresses
1185  // stayed an affiliation of every mark-1 author)
1186  if is_mark_command(&first) {

1188    return rebuilt(
1189      mark_key(Tokens::new(operand).untex_string().as_ref()),
1190      false,
1191      next,
1192    );
1193  }
1194  // (a number typed by hand, followed by `)`, `.` or a space)
1195  let digit = |t: &Token| {
1196    t.get_catcode() == Catcode::OTHER && t.with_str(|s| s.chars().all(|c| c.is_ascii_digit()))
1197  };
1198  let digits = tokens[at..].iter().take_while(|t| digit(t)).count();
1199  if digits == 0 {
1200    return None;
1201  }
1202  let label: String = tokens[at..at + digits]
1203    .iter()
1204    .map(ToString::to_string)
1205    .collect();
1206  match tokens.get(at + digits) {
1207    Some(t) if *t == T_OTHER!(")") || *t == T_OTHER!(".") => rebuilt(label, true, at + digits + 1),
1208    Some(t) if *t == T_SPACE!() => rebuilt(label, true, at + digits),
1209    _ => None,
1210  }
1211}

1348/// The lines a line of an affiliation block sets with one-argument line wrappers ([`LINE_MACROS`]: one line each;
1349/// [`LINE_BOXES`]: one box that is the whole line, or boxes marks lead), each the wrapper's argument; `None` for a line
1350/// holding anything else beside them but spacing.
1351fn wrapped_line_pieces(line: &[Token]) -> Option<Vec<Vec<Token>>> {
1352  let mut pieces: Vec<&[Token]> = Vec::new();
1353  let mut boxed = false;
1354  let mut i = 0;
1355  loop {
1356    while i < line.len() && (line[i] == T_CS!("\\par") || line[i] == T_CS!("\\noindent")) {
1357      i += 1;
1358    }
1359    let past = i + leading_spacing_end(&line[i..]);
1360    if past > i {
1361      i = past;
1362      continue;
1363    }
1364    let Some(t) = line.get(i) else {
1365      break;
1366    };
1367    let line_macro = LINE_MACROS.iter().any(|cs| *t == T_CS!(*cs));
1368    let line_box = LINE_BOXES.iter().any(|cs| *t == T_CS!(*cs));
1369    if !line_macro && !line_box {
1370      return None;
1371    }
1372    let mut open = i + 1;
1373    while line.get(open) == Some(&T_SPACE!()) {
1374      open += 1;
1375    }
1376    if line
1377      .get(open)
1378      .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
1379    {
1380      return None;
1381    }
1382    let end = skip_group(line, open);
1383    if end > line.len() || line[end - 1].get_catcode() != Catcode::END {
1384      return None;
1385    }
1386    pieces.push(&line[open + 1..end - 1]);
1387    boxed |= line_box;
1388    i = end;
1389  }
1390  if pieces.is_empty() {
1391    return None;
1392  }
1393  if !boxed || pieces.len() == 1 {
1394    return Some(pieces.into_iter().map(<[Token]>::to_vec).collect());
1395  }
1396  // (boxes side by side: entries where marks lead them, a box no mark leads the entry before it, a space between)
1397  let leads: Vec<bool> = pieces
1398    .iter()
1399    .map(|piece| leads_with_mark(&Tokens::new(piece.to_vec())))
1400    .collect();
1401  if !leads[0] || leads.iter().filter(|lead| **lead).count() < 2 {
1402    return None;
1403  }
1404  let mut entries: Vec<Vec<Token>> = Vec::new();
1405  for (piece, lead) in pieces.into_iter().zip(leads) {
1406    match entries.last_mut() {
1407      Some(entry) if !lead => {
1408        entry.push(T_SPACE!());
1409        entry.extend_from_slice(piece);
1410      },
1411      _ => entries.push(piece.to_vec()),
1412    }
1413  }
1414  Some(entries)
1415}

1459/// The rows of a one-column `tabular` that is the whole of `line` (`\begin{tabular}{c} $^{1}$ Univ A \\ $^{2}$ Univ B

1465    while line
1466      .get(i)
1467      .is_some_and(|t| *t == T_SPACE!() || *t == T_CS!("\\par"))
1468    {
1469      i += 1;
1470    }

1472  };

1484  }

1519/// A tabular row without the rules set between the rows, which print no text and, a line's, are an alignment's only
1520/// (`\hline` outside one: "\noalign cannot be used here", review rev104c S3): `\hline`, `\cline{2-3}`, `\hhline{…}`, and

1522/// `\specialrule{w}{a}{b}` (booktabs.sty:62-80, 136).
1523fn without_tabular_rules(row: &[Token]) -> Vec<Token> {
1524  // (each rule's arguments: how many groups after its optional `[…]` and booktabs' `(…)` trim)
1525  const RULES: [(&str, usize); 9] = [
1526    ("\\hline", 0),
1527    ("\\cline", 1),
1528    ("\\hhline", 1),
1529    ("\\toprule", 0),
1530    ("\\midrule", 0),
1531    ("\\bottomrule", 0),
1532    ("\\cmidrule", 1),
1533    ("\\addlinespace", 0),
1534    ("\\specialrule", 3),
1535  ];
1536  let past_spaces = |mut i: usize| {
1537    while row.get(i) == Some(&T_SPACE!()) {
1538      i += 1;
1539    }
1540    i
1541  };
1542  // (a delimited optional, `[1pt]`, `(lr)`, at `i` past its spaces: where it ends)
1543  let optional = |i: usize, open: &str, close: &str| {
1544    let at = past_spaces(i);
1545    if row.get(at) != Some(&T_OTHER!(open)) {
1546      return None;
1547    }
1548    row[at..]
1549      .iter()
1550      .position(|t| *t == T_OTHER!(close))
1551      .map(|k| at + k + 1)
1552  };
1553  let mut out = Vec::with_capacity(row.len());
1554  let mut depth = 0usize;
1555  let mut i = 0;
1556  while i < row.len() {
1557    let t = row[i];
1558    match t.get_catcode() {
1559      Catcode::BEGIN => depth += 1,
1560      Catcode::END => depth = depth.saturating_sub(1),
1561      _ => {},
1562    }
1563    if depth == 0
1564      && let Some(&(_, groups)) = RULES.iter().find(|(cs, _)| t == T_CS!(*cs))
1565    {
1566      let mut next = i + 1;
1567      next = optional(next, "[", "]").unwrap_or(next);
1568      next = optional(next, "(", ")").unwrap_or(next);
1569      for _ in 0..groups {

1571        if row
1572          .get(open)
1573          .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
1574        {
1575          next = skip_group(row, open);
1576        }
1577      }
1578      i = next;
1579      continue;
1580    }
1581    out.push(t);
1582    i += 1;
1583  }
1584  out
1585}

1932/// Where the ordinal `tokens` opens with ends ([`leads_with_ordinal`]).
1933fn leading_ordinal_end(tokens: &[Token]) -> Option<usize> {
1934  let start = tokens.iter().take_while(|t| **t == T_SPACE!()).count();
1935  let digits = tokens[start..]
1936    .iter()
1937    .take_while(|t| {
1938      t.get_catcode() == Catcode::OTHER && t.with_str(|s| s.chars().all(|c| c.is_ascii_digit()))
1939    })
1940    .count();
1941  (digits > 0)
1942    .then(|| ordinal_suffix_end(tokens, start + digits))
1943    .flatten()
1944}

1970/// Where an ordinal suffix starting at `at` ends, glued to the number before it ([`is_ordinal_suffix`]): a run of letters that
1971/// is the whole word, `rd` before a space or a comma, not the `St` of `Stanford`, nor an uppercase `St` before a `.` or
1972/// a space and a capital, the word "Saint" (`$1$St. Petersburg`, `$3$St Andrews`; review rev104c2 F3 — a lowercase
1973/// `$1$st Floor` is the ordinal); a `\textsuperscript{th}`; a `$^{nd}$`?
1974fn ordinal_suffix_end(tokens: &[Token], at: usize) -> Option<usize> {
1975  let letters = tokens[at.min(tokens.len())..]
1976    .iter()
1977    .take_while(|t| t.get_catcode() == Catcode::LETTER)
1978    .count();
1979  if letters > 0 {
1980    let saint = letters == 2
1981      && tokens[at] == T_LETTER!("S")
1982      && (tokens.get(at + 2) == Some(&T_OTHER!("."))
1983        || (tokens.get(at + 2) == Some(&T_SPACE!())
1984          && tokens.get(at + 3).is_some_and(|t| {
1985            t.get_catcode() == Catcode::LETTER && t.with_str(|s| s.starts_with(char::is_uppercase))
1986          })));
1987    return (!saint && is_ordinal_suffix(&tokens[at..at + letters])).then_some(at + letters);
1988  }
1989  match tokens.get(at) {
1990    Some(t) if *t == T_CS!("\\textsuperscript") => sup_operand_at(tokens, at + 1)
1991      .filter(|(operand, _)| is_ordinal_suffix(operand))
1992      .map(|(_, next)| next),
1993    Some(t) if t.get_catcode() == Catcode::MATH && tokens.get(at + 1) == Some(&T_SUPER!()) => {
1994      sup_operand_at(tokens, at + 2)
1995        .filter(|(operand, next)| {
1996          is_ordinal_suffix(operand)
1997            && tokens
1998              .get(*next)
1999              .is_some_and(|t| t.get_catcode() == Catcode::MATH)
2000        })
2001        .map(|(_, next)| next + 1)
2002    },
2003    _ => None,
2004  }
2005}

2007/// Does a line of `tokens` start at `i` — only spaces between it and a `\\` before it, or that `\\`'s optional
2008/// length (`\\[5pt]`)?
2009fn at_line_start(tokens: &[Token], i: usize) -> bool {
2010  let mut j = i;
2011  while j > 0 && tokens[j - 1] == T_SPACE!() {
2012    j -= 1;
2013  }
2014  if j > 0
2015    && tokens[j - 1] == T_OTHER!("]")
2016    && let Some(open) = tokens[..j - 1].iter().rposition(|t| *t == T_OTHER!("["))
2017  {
2018    j = open;
2019    while j > 0 && tokens[j - 1] == T_SPACE!() {
2020      j -= 1;
2021    }
2022  }
2023  j > 0 && tokens[j - 1].defined_as(&T_CS!("\\\\"))
2024}

2028  let marks = queued_name_marks();
2029  let requested = &marks.1;
2030  let switch = mark_switch;
2031  let skippable = |t: &Token| {
2032    *t == T_SPACE!()
2033      || *t == T_BEGIN!()
2034      || *t == T_CS!("\\noindent")
2035      || *t == T_CS!("\\nonumber")
2036      || *t == T_CS!("\\par")
2037      || switch(t)
2038  };
2039  let mut starts = vec![0];
2040  starts.extend(
2041    tokens
2042      .iter()
2043      .enumerate()
2044      .filter(|(_, t)| t.defined_as(&T_CS!("\\\\")))
2045      .map(|(i, _)| i + 1),
2046  );
2047  starts
2048    .into_iter()
2049    .filter(|&start| {
2050      let mut i = start;
2051      loop {
2052        while tokens.get(i).is_some_and(skippable) {
2053          i += 1;
2054        }
2055        // (a box the line is set in: `\parbox[t]{\linewidth}{…}`, `\mbox{…}`, `\hbox{…}`, an environment
2056        // `\begin{tabular}{c}` — past its opener; 0911.3234's two labelled lines in a `tabular`)
2057        match tokens.get(i) {
2058          Some(t) if *t == T_CS!("\\begin") => {

2060            while tokens.get(i) == Some(&T_SPACE!()) {
2061              i += 1;
2062            }
2063            let name_end = skip_group(tokens, i);
2064            let name: String = tokens
2065              .get(i + 1..name_end.saturating_sub(1))
2066              .unwrap_or_default()
2067              .iter()
2068              .map(ToString::to_string)
2069              .collect();
2070            i = name_end;
2071            // (the environment's own arguments: latex.ltx `tabular`/`array` [pos]{cols}, `tabular*` {width}[pos]{cols},
2072            // `minipage` [pos]{width}; any other none, its first group the line's)
2073            let groups = match name.as_str() {
2074              "tabular" | "array" | "minipage" => 1,
2075              "tabular*" => 2,
2076              _ => 0,
2077            };
2078            let mut taken = 0;
2079            while taken < groups {
2080              while tokens.get(i) == Some(&T_SPACE!()) {
2081                i += 1;
2082              }
2083              match tokens.get(i) {
2084                Some(t) if t.get_catcode() == Catcode::BEGIN => {
2085                  i = skip_group(tokens, i);
2086                  taken += 1;
2087                },
2088                Some(t) if *t == T_OTHER!("[") => {
2089                  while tokens.get(i).is_some_and(|t| *t != T_OTHER!("]")) {
2090                    i += 1;
2091                  }
2092                  i += 1;
2093                },
2094                _ => break,
2095              }
2096            }
2097          },
2098          Some(t) if *t == T_CS!("\\parbox") || *t == T_CS!("\\minipage") => {
2099            i += 1;
2100            if tokens.get(i) == Some(&T_OTHER!("[")) {
2101              while tokens.get(i).is_some_and(|t| *t != T_OTHER!("]")) {
2102                i += 1;
2103              }
2104              i += 1;
2105            }
2106            while tokens.get(i) == Some(&T_SPACE!()) {
2107              i += 1;
2108            }
2109            i = skip_group(tokens, i);
2110          },
2111          Some(t) if LINE_BOXES.iter().any(|cs| *t == T_CS!(*cs)) => i += 1,
2112          _ => break,
2113        }
2114      }
2115      let Some(t) = tokens.get(i) else {
2116        return false;
2117      };
2118      let operand = if let Some((operand, _)) = spaced_mark_at(tokens, i) {
2119        Some(operand)
2120      } else if t.get_catcode() == Catcode::MATH
2121        && tokens
2122          .get(i + 1..)
2123          .and_then(|rest| rest.iter().find(|t| **t != T_SPACE!()))
2124          == Some(&T_SUPER!())
2125      {
2126        let sup = i
2127          + 1
2128          + tokens[i + 1..]
2129            .iter()
2130            .take_while(|t| **t == T_SPACE!())
2131            .count();
2132        sup_operand_at(tokens, sup + 1).map(|(operand, _)| operand)
2133      } else if *t == T_CS!("\\textsuperscript") {
2134        sup_operand_at(tokens, i + 1).map(|(operand, _)| operand)
2135      } else {
2136        None
2137      };
2138      operand.is_some_and(|operand| {
2139        let mut mark = vec![T_MATH!(), T_SUPER!(), T_BEGIN!()];
2140        mark.extend(operand);
2141        mark.extend([T_END!(), T_MATH!()]);
2142        content_marks(&Tokens::new(mark))
2143          .iter()
2144          .any(|m| requested.contains(&mark_key(m)))
2145      })
2146    })

2148}

2150fn group_shares_the_mark(marks: &[String], alternating: bool) -> Result<bool> {
2151  // (how many authors were queued since the last affiliation: the last ones `queued_author_marks` holds)
2152  let (interleaved, since) = with_value("frontmatter_raw", |v| match v {
2153    Some(Stored::FrontmatterRaw(queue)) => {
2154      let previous = queue.iter().rposition(is_affiliation_entry);
2155      let run = &queue[previous.map_or(0, |at| at + 1)..];
2156      let follows_an_author = |at: usize| queue[..at].iter().any(|entry| entry.0 == "ltx:creator");
2157      let interleaved = run.iter().any(|entry| entry.0 == "ltx:creator")
2158        && previous.is_some_and(|at| !alternating || follows_an_author(at));
2159      (
2160        interleaved,
2161        run.iter().filter(|e| is_author_entry(e)).count(),
2162      )
2163    },
2164    _ => (false, 0),
2165  });
2166  if interleaved {
2167    return Ok(true);
2168  }
2169  let [mark] = marks else {
2170    return Ok(false);
2171  };
2172  if since == 0 {
2173    return Ok(false);
2174  }

2176  let since = positions_of_last_entries(since)?;
2177  // (every piece, symbols too, not `requested_labels`: a name showing only `$^{*}$` shows no mark the group shares, so the
2178  // line goes by the marked rule to the names that print its label)
2179  let scripts: Vec<Vec<String>> = names[names.len().saturating_sub(since)..]

2187}
