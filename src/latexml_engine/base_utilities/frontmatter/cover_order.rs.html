88/// Mark (`attribute`) every root child after the one holding the last `marker` element, frontmatter and markers aside;
89/// remove the marker.
90fn mark_root_children_after(document: &mut Document, marker: &str, attribute: &str) {
91  for marker in document.findnodes(&s!("//{marker}"), None) {
92    if let Some(top) = root_child_of(&marker) {
93      let mut next = top.get_next_element_sibling();
94      while let Some(mut node) = next {
95        next = node.get_next_element_sibling();
96        if !node.get_name().ends_with("_Capture_") && !is_frontmatter_group_element(&node) {
97          let _ = node.set_attribute(attribute, "true");
98        }
99      }
100    }
101    document.remove_node(marker);
102  }
103}

105/// The bound of [`place_abstract_after_cover`] for an abstract a flush is about to place, read from its position marker
106/// (`\lx@frontmatter@mark`, a root child) before the flush removes it: the root child after the marker, or `None` for
107/// the end. `None` when there is no such marker.
108pub(in crate::base_utilities) fn abstract_mark_bound(
109  document: &mut Document,
110) -> Option<Option<Node>> {
111  let marker = document.findnode("/ltx:document/ltx:_Frontmatter_Capture_", None)?;
112  // (not another marker, which the flush removes too: an `{altabstract}` right after the abstract)
113  let mut next = marker.get_next_element_sibling();
114  while let Some(node) = next
115    .as_ref()
116    .filter(|n| n.get_name() == "_Frontmatter_Capture_")
117  {
118    next = node.get_next_element_sibling();
119  }
120  Some(next)
121}

123/// The child of the document root that holds `node` (`node` itself when it is one).
124fn root_child_of(node: &Node) -> Option<Node> {
125  let mut cur = node.clone();
126  loop {
127    let parent = cur.get_parent()?;
128    if parent
129      .get_parent()
130      .is_none_or(|p| p.get_type() == Some(NodeType::DocumentNode))
131    {
132      return Some(cur);
133    }
134    cur = parent;
135  }
136}

138/// A late abstract's position marker ([`LATE_ABSTRACT_MARK`]), where the abstract is already placed (the eager build,
139/// whose `\maketitle` flush digested the whole queue): the abstract goes behind the cover printed before it
140/// (`place_abstract_after_cover`), and the marker is removed — as is the marker of a command that added no abstract (a
141/// repeat `frontmatter_drop_repeats` dropped, an empty `{abstract}`, a sealed supplement's), once the others are placed.
142/// Before the abstract is placed (a streamed build, where only an abstract `\maketitle` placed may be there), the
143/// marker stays for `insert_late_frontmatter`.
144pub fn settle_late_abstract_mark(document: &mut Document, marker: Node) -> Result<()> {
145  let placed = linked_late_abstract(document, &marker).is_some()
146    || document
147      .findnode(&s!("/ltx:document/ltx:abstract[@{LATE_ENTRY}]"), None)
148      .is_some();
149  if placed {
150    place_abstract_after_late_mark(document, &marker)?;
151    document.remove_node(marker);
152  }
153  Ok(())
154}

156/// The ordinal of a late abstract's marker (`\lx@frontmatter@mark`, counted in [`LATE_MARKS`]), which the abstract its
157/// command adds carries too (`queue_front_matter`, `frontmatter_push`): the marker places its own abstract, and a
158/// command that added none (a repeat dropped, an empty body) places nothing. Internal, stripped from the output.
159pub const LATE_MARK_ORDINAL: &str = "_late_mark";

161/// The late abstracts' markers digested so far ([`LATE_MARK_ORDINAL`]).
162pub const LATE_MARKS: &str = "lx_late_abstract_marks";

164/// The ordinal of the last late marker digested, until its command adds its abstract ([`take_late_mark`]).
165const LATE_MARK_PENDING: &str = "lx_late_abstract_mark_pending";
166
167/// The ordinal ([`LATE_MARK_ORDINAL`]) of the late abstract an abstract-adding command is adding: the last late
168/// marker's, taken by the first abstract added after it — its own command's, whether kept or not (an empty body, a
169/// repeat, a sealed supplement's) — so an abstract a later command adds with no marker of its own (a binding's
170/// `\lx@add@frontmatter{ltx:abstract}`: rmaa's hand-off, oup's `\otherabstract`) carries none, and no marker
171/// places it (review probe m6: an empty `{abstract}`'s marker placed the next one). Empty when there is none.
172pub(in crate::base_utilities) fn take_late_mark() -> String {
173  let ordinal = lookup_int(LATE_MARK_PENDING);
174  if ordinal > 0 {
175    assign_value(LATE_MARK_PENDING, 0, Some(Scope::Global));
176    ordinal.to_string()
177  } else {
178    String::new()
179  }
180}

182/// An abstract-adding command that adds none (an empty argument, a sealed supplement's): its marker's ordinal goes
183/// with it ([`take_late_mark`]). Not while a flush digests the queue, whose entries carry their own.
184pub(in crate::base_utilities) fn drop_late_mark(tag: &str) {
185  if tag == "ltx:abstract" && !lookup_bool("lx_frontmatter_flushing") {
186    take_late_mark();
187  }

207/// [`place_abstract_after_cover`] up to `marker`, a late abstract's position: the region ends right before it when it
208/// is a root child; one inside an open paragraph (text runs into `\begin{abstract}`) ends the region with that
209/// paragraph, when nothing follows the marker there. The marker of the first late abstract places the run of them all
210/// behind the cover; each later one's places its own abstract ([`linked_late_abstract`]) behind the cover lines printed
211/// between the one before and it ([`place_later_abstract`]: a second `{abstract}` after `\maketitle`, printed where it
212/// stands, 1006.5288). A marker whose command added no abstract places nothing (review probes q09, q10). Late
213/// abstracts keep their source order ([`keep_late_abstracts_in_source_order`]).
214pub(in crate::base_utilities) fn place_abstract_after_late_mark(
215  document: &mut Document,
216  marker: &Node,
217) -> Result<bool> {
218  let moved = place_linked_late_abstract(document, marker)?;
219  if moved {
220    keep_late_abstracts_in_source_order(document)?;
221  }
222  Ok(moved)
223}

225/// [`place_abstract_after_late_mark`]'s placement, before its source-order check.
226fn place_linked_late_abstract(document: &mut Document, marker: &Node) -> Result<bool> {
227  let Some(own) = linked_late_abstract(document, marker) else {
228    return Ok(false);
229  };
230  let mut earlier = own.get_prev_element_sibling();
231  let mut first = true;
232  while let Some(node) = earlier {
233    if document::get_node_qname(&node) == pin_static("ltx:abstract")
234      && node.has_attribute(LATE_ENTRY)
235    {
236      first = false;
237      break;
238    }
239    earlier = node.get_prev_element_sibling();
240  }
241  let place = |document: &mut Document, bound: Option<&Node>| {
242    if first {
243      place_abstract_after_cover(document, bound, true)
244    } else {
245      place_later_abstract(document, bound, &own)
246    }
247  };
248  let Some(top) = root_child_of(marker) else {
249    return Ok(false);
250  };
251  if top == *marker {
252    return place(document, Some(&top));
253  }
254  // (the marker ends its root child: nothing after it at any level)
255  let mut cur = marker.clone();
256  while cur != top {
257    let mut next = cur.get_next_sibling();
258    while let Some(n) = next {
259      let blank = n.get_type() == Some(NodeType::TextNode) && n.get_content().trim().is_empty();
260      if !blank {
261        return Ok(false);
262      }
263      next = n.get_next_sibling();
264    }
265    let Some(parent) = cur.get_parent() else {
266      return Ok(false);
267    };
268    cur = parent;
269  }
270  match top.get_next_element_sibling() {
271    Some(next) => place(document, Some(&next)),
272    None => place(document, None),
273  }
274}

276/// A late abstract after the first (the first is [`place_abstract_after_cover`]'s), `own`, `bound` its own position
277/// marker's bound. Its run — it and the frontmatter after it: its keywords, and every later late abstract still
278/// waiting right after it, marked or markerless, with theirs — moves to right before `bound` when every root child
279/// between the run and `bound` is title-page layout that is no running body prose — the cover lines pdflatex prints
280/// between the abstract before it and it (review probe a02: cover line one, the first abstract, cover line two, the
281/// second) — or what such a cover passes over (a content-free node, preamble residue, a root note, page-foot
282/// material). The later abstracts are carried along so that late abstracts keep their source order: each then scans
283/// from its new place, right after the one before it (review probe o2d: three late abstracts, each behind its own
284/// cover line), and no late abstract stands in the region a later scan crosses — one that did would refuse the move.
285/// Anything else there — a sectional unit, a float, a list, a TOC, body prose — and the move is refused: the run stays
286/// right after the abstract before it (r3 review probes q1–q4, q10), and a markerless one, which no marker places,
287/// right after the one before it (q9). The cover lines become a `<titlepage>` of their own, continued like the first
288/// late abstract's ([`CONTINUED_TITLEPAGE`]): a `<para>` is no frontmatter, and the abstract after it would be invalid
289/// (`\maketitle`, `\noindent\hrulefill`, `{abstract}`, `\noindent\hrulefill`, `{abstract}`: the simples-matrices,
290/// annee-scolaire and paresse manuals, 1006.5288's shape). Returns whether it moved anything.
291fn place_later_abstract(document: &mut Document, bound: Option<&Node>, own: &Node) -> Result<bool> {
292  let Some(root) = document.get_document().get_root_element() else {
293    return Ok(false);
294  };
295  let kids: Vec<Node> = root
296    .get_child_nodes()
297    .into_iter()
298    .filter(|n| n.get_type() == Some(NodeType::ElementNode))
299    .collect();
300  let end = match bound {
301    Some(bound) => match kids.iter().position(|k| k == bound) {
302      Some(end) => end,
303      None => return Ok(false),
304    },
305    None => kids.len(),
306  };
307  let is_late_abstract = |k: &Node| {
308    document::get_node_qname(k) == pin_static("ltx:abstract") && k.has_attribute(LATE_ENTRY)
309  };
310  let Some(first) = kids[..end].iter().position(|k| k == own) else {
311    return Ok(false);
312  };
313  // (its run, carrying the later late abstracts waiting right after it: never a titlepage)
314  let mut run_end = first + 1;
315  while run_end < end
316    && is_frontmatter_group_element(&kids[run_end])
317    && document::get_node_qname(&kids[run_end]) != pin_static("ltx:titlepage")
318  {
319    run_end += 1;
320  }
321  if kids[first..run_end].iter().any(from_the_title_page) {
322    return Ok(false);
323  }
324  let mut cover: Vec<Node> = Vec::new();
325  let mut passed: Vec<Node> = Vec::new();
326  for kid in &kids[run_end..end] {
327    // (none stands there while late abstracts keep their source order: a move past one would put it ahead)
328    if is_late_abstract(kid) {
329      return Ok(false);
330    }
331    if kid.get_name().ends_with("_Capture_") {
332      continue;
333    }
334    if node_is_content_free(kid)
335      || is_preamble_residue(kid)
336      || document::get_node_qname(kid) == pin_static("ltx:note")
337      || kid.has_attribute(DEPOSIT_INSERT)
338    {
339      passed.push(kid.clone());
340    } else if is_titlepage_layout(kid)
341      && (kid.has_attribute(DEPOSIT_PROVENANCE) || !is_body_prose(kid))
342    {
343      cover.push(kid.clone());
344    } else {
345      return Ok(false);
346    }
347  }
348  if cover.is_empty() {
349    return Ok(false);
350  }
351  // (the build goes on at the root, not inside a cover line the abstract now follows)
352  let region = &kids[run_end..end];
353  let inside = |document: &Document| {
354    let mut cur = Some(document.get_node().clone());
355    while let Some(n) = cur {
356      if region.contains(&n) {
357        return true;
358      }
359      cur = n.get_parent();
360    }
361    false
362  };
363  if inside(document) {
364    close_auto_closable_to_root(document, &root)?;
365    if inside(document) {
366      return Ok(false);
367    }
368  }
369  let mut titlepage = wrap_as_titlepage(document, &root, &cover)?;
370  document.add_class(&mut titlepage, CONTINUED_TITLEPAGE)?;
371  let mut tail = titlepage;
372  for mut node in kids[first..run_end].iter().cloned() {
373    node.unlink();
374    tail.add_next_sibling(&mut node)?;
375    tail = node;
376  }
377  move_content_free_after(&passed, tail)?;
378  Ok(true)
379}

381/// The invariant every late abstract's placement keeps: the root's late abstracts ([`LATE_ENTRY`]) stand in source
382/// order, their marker ordinals ([`LATE_MARK_ORDINAL`]) ascending (a markerless one carries none and is not compared).
383/// No move puts a late abstract ahead of one given before it ([`place_later_abstract`] carries the later ones along);
384/// a violation is a placement bug, which a debug build fails on (`debug_assert!`) and a release build reports
385/// (`Warning:unexpected:late_abstract_order`) and repairs: every late abstract, each with its run (the frontmatter after
386/// it up to the next late abstract: its keywords, its classification, as [`place_later_abstract`] carries them), is set
387/// in source order where the first of them stands — each markerless one after the marked one it followed — the order
388/// of a placement that moves only the first (round 2 of batch 86i); a first one with nothing before it, at the root's
389/// head.
390fn keep_late_abstracts_in_source_order(document: &mut Document) -> Result<()> {
391  let Some(mut root) = document.get_document().get_root_element() else {
392    return Ok(());
393  };
394  let is_late_abstract = |k: &Node| {
395    k.get_type() == Some(NodeType::ElementNode)
396      && document::get_node_qname(k) == pin_static("ltx:abstract")
397      && k.has_attribute(LATE_ENTRY)
398  };
399  let late: Vec<Node> = root
400    .get_child_nodes()
401    .into_iter()
402    .filter(is_late_abstract)
403    .collect();
404  let ordinal = |k: &Node| {
405    k.get_attribute(LATE_MARK_ORDINAL)
406      .and_then(|o| o.parse::<i64>().ok())
407  };
408  let marked: Vec<i64> = late.iter().filter_map(ordinal).collect();
409  let in_order = marked.is_sorted();
410  if !in_order {
411    Warn!(
412      "unexpected",
413      "late_abstract_order",
414      s!(
415        "Late abstracts placed out of source order (marker ordinals {marked:?}); set back in source order"
416      )
417    );
418  }
419  debug_assert!(
420    in_order,
421    "late abstracts out of source order (marker ordinals {marked:?}): a placement bug"
422  );
423  if in_order {
424    return Ok(());
425  }
426  // (each with its run — never a titlepage — and each markerless one keeping the place after the marked one it
427  // followed)
428  let mut last = 0;
429  let mut keyed: Vec<(i64, Vec<Node>)> = late
430    .iter()
431    .map(|k| {
432      last = ordinal(k).unwrap_or(last);
433      let mut run = vec![k.clone()];
434      let mut next = k.get_next_element_sibling();
435      while let Some(n) = next {
436        if is_late_abstract(&n)
437          || !is_frontmatter_group_element(&n)
438          || document::get_node_qname(&n) == pin_static("ltx:titlepage")
439        {
440          break;
441        }
442        next = n.get_next_element_sibling();
443        run.push(n);
444      }
445      (last, run)
446    })
447    .collect();
448  keyed.sort_by_key(|(key, _)| *key);
449  let mut tail = late[0].get_prev_sibling();
450  for mut node in keyed.into_iter().flat_map(|(_, run)| run) {
451    node.unlink();
452    if let Some(tail) = tail.as_mut() {
453      tail.add_next_sibling(&mut node)?;
454    } else if let Some(mut head) = root.get_first_child() {
455      head.add_prev_sibling(&mut node)?;
456    } else {
457      root.add_child(&mut node)?;
458    }
459    tail = Some(node);
460  }
461  Ok(())
462}

464/// The readability bar's print order for an abstract the frontmatter placed AHEAD of title-page text pdflatex prints
465/// before it: a `late` abstract (queued after `\maketitle`'s flush, which placed it with the head;
466/// `\lx@frontmatter@mark`, `insert_late_frontmatter`) — the first root `ltx:abstract` that is [`LATE_ENTRY`], `bound`
467/// its own position marker — or one a document without `\maketitle` sets after its in-place setters (the fallback,
468/// `\lx@frontmatter@fallback`) — the first root `ltx:abstract`, `bound` the first abstract's marker, which precedes
469/// every abstract in print. The root children between the placed run of frontmatter that begins with that abstract and
470/// `bound` (a root child, exclusive; `None`: to the end) are read as OXIDIZED_DESIGN #246 reads a cover:
471/// - title-page layout ([`is_titlepage_layout`]: a paragraph, a display block) that the class's title page printed
472///   ([`DEPOSIT_PROVENANCE`]) or that is no running body prose ([`is_body_prose`]) is the cover, wrapped as
473///   `<ltx:titlepage>` in its order;
474/// - a content-free node (a pagebreak), preamble residue, a root note (a footnote's text, out of the flow) and what an
475///   `\insert` in the class's title page set ([`DEPOSIT_INSERT`]: page-foot material) are passed over and follow the
476///   abstract;
477/// - anything else — a sectional unit, a TOC, a float, body prose — keeps today's placement: the region is not the
478///   cover.
479///
480/// With a cover, the abstract and the frontmatter after it in its run (keywords, classification, a title `\thanks`
481/// pubnote) move behind the `<titlepage>`, save an abstract queued before `\maketitle`'s flush
482/// ([`printed_before_the_cover`]), which keeps its place ahead of it. The fail-safe: a late run holding an entry handed
483/// on from the raw class's store that its title page may print ([`from_the_title_page`]), where the frontmatter cannot
484/// tell, is not moved at all — the abstract and everything after it keep today's slot. Once per document
485/// (`lx_late_abstract_placed`): a second late abstract has nothing more to place. Returns whether it moved anything.
486pub(in crate::base_utilities) fn place_abstract_after_cover(
487  document: &mut Document,
488  bound: Option<&Node>,
489  late: bool,
490) -> Result<bool> {
491  if lookup_bool("lx_late_abstract_placed") {
492    return Ok(false);
493  }
494  let Some(root) = document.get_document().get_root_element() else {
495    return Ok(false);
496  };
497  let kids: Vec<Node> = root
498    .get_child_nodes()
499    .into_iter()
500    .filter(|n| n.get_type() == Some(NodeType::ElementNode))
501    .collect();
502  let end = match bound {
503    Some(bound) => match kids.iter().position(|k| k == bound) {
504      Some(end) => end,
505      None => return Ok(false),
506    },
507    None => kids.len(),
508  };
509  let Some(first) = kids[..end].iter().position(|k| {
510    document::get_node_qname(k) == pin_static("ltx:abstract")
511      && (!late || k.has_attribute(LATE_ENTRY))
512  }) else {
513    return Ok(false);
514  };
515  // The abstract's run: it and the frontmatter after it, as placed (never a titlepage).
516  let mut run_end = first;
517  while run_end < end
518    && is_frontmatter_group_element(&kids[run_end])
519    && document::get_node_qname(&kids[run_end]) != pin_static("ltx:titlepage")
520  {
521    run_end += 1;
522  }
523  // (the fail-safe: what the class's title page printed of its stores may stand above the cover, below it or at the
524  // page foot — `\footnotetext{Keywords: \@keywords}` — so a late run holding one stays where it is)
525  if late && kids[first..run_end].iter().any(from_the_title_page) {
526    return Ok(false);
527  }
528  // (a title page still being built — the abstract printed inside the class's own — is the class's from its start)
529  let open_cover = kids[..end]
530    .iter()
531    .rposition(|k| document::get_node_qname(k) == pin_static(DEPOSIT_BEGIN));
532  let mut cover: Vec<Node> = Vec::new();
533  let mut passed: Vec<Node> = Vec::new();
534  for (at, kid) in kids.iter().enumerate().take(end).skip(run_end) {
535    if kid.get_name().ends_with("_Capture_") {
536      continue;
537    }
538    let q = document::get_node_qname(kid);
539    if node_is_content_free(kid)
540      || is_preamble_residue(kid)
541      || q == pin_static("ltx:note")
542      || kid.has_attribute(DEPOSIT_INSERT)
543    {
544      passed.push(kid.clone());
545    } else if is_titlepage_layout(kid)
546      && (kid.has_attribute(DEPOSIT_PROVENANCE)
547        || open_cover.is_some_and(|begin| begin < at)
548        || !is_body_prose(kid))
549    {
550      cover.push(kid.clone());
551    } else {
552      return Ok(false);
553    }
554  }
555  if cover.is_empty() {
556    return Ok(false);
557  }
558  // The build goes on at the root, not inside a paragraph that becomes a block of the titlepage.
559  let inside = |document: &Document| {
560    let mut cur = Some(document.get_node().clone());
561    while let Some(n) = cur {
562      if cover.contains(&n) {
563        return true;
564      }
565      cur = n.get_parent();
566    }
567    false
568  };
569  if inside(document) {
570    close_auto_closable_to_root(document, &root)?;
571    if inside(document) {
572      return Ok(false);
573    }
574  }
575  assign_value("lx_late_abstract_placed", true, Some(Scope::Global));
576  let mut titlepage = wrap_as_titlepage(document, &root, &cover)?;
577  // (it follows the title block, which printed the title: the structured title stays displayed beside it)
578  document.add_class(&mut titlepage, CONTINUED_TITLEPAGE)?;
579  let mut tail = titlepage;
580  // (what the title page printed before the cover keeps its place ahead of it: of a late run, an abstract `\maketitle`
581  // placed — set before it)
582  for mut node in kids[first..run_end]
583    .iter()
584    .filter(|k| !printed_before_the_cover(k, late))
585    .cloned()
586  {
587    node.unlink();
588    tail.add_next_sibling(&mut node)?;
589    tail = node;
590  }
591  move_content_free_after(&passed, tail)?;
592  Ok(true)
593}

595/// Of the run that follows a late abstract behind its cover, an entry printed before the cover, which keeps its place
596/// ahead of it: an abstract queued before `\maketitle`'s flush (not [`LATE_ENTRY`]: smfart's `{abstract}` before
597/// `\maketitle`). Everything else in the run — keywords and notes a binding prints after the abstract (dmtcs,
598/// 0905.1102; aaspp4, astro-ph9911463), a title `\thanks` pubnote (2102.01904) — follows the abstract. (A run holding
599/// an entry from a raw class's store its title page may print is not moved at all: [`from_the_title_page`].)
600fn printed_before_the_cover(node: &Node, late: bool) -> bool {
601  late
602    && !node.has_attribute(LATE_ENTRY)
603    && document::get_node_qname(node) == pin_static("ltx:abstract")
604}

606/// A paragraph of running body prose: an indented paragraph (no `\noindent`, no alignment of its own) holding text
607/// outside any box, or a list whose items hold one (a body list). A cover line is set apart — `\noindent`ed, centred or
608/// flushed, a box alone, a list of such lines (nature's `{affiliations}`, 2304.00893; ae100prg's `\address`, 1303.3004)
609/// — so the text set between `\maketitle` and a late abstract is cover only when none of it is prose (`First body
610/// paragraph.` before a `\DocInput`-ed abstract stays body,
611/// `perfect_kernel_batch56::late_abstract_after_maketitle_leaves_no_marker_and_wraps_nothing`).
612fn is_body_prose(node: &Node) -> bool {
613  if document::get_node_qname(node) != pin_static("ltx:para") || is_set_apart(node) {
614    return false;
615  }
616  element_nodes(node)
617    .iter()
618    .any(|child| match child.get_name().as_str() {
619      "p" => !is_set_apart(child) && !running_text(child).trim().is_empty(),
620      "itemize" | "enumerate" | "description" => element_nodes(child)
621        .iter()
622        .filter(|item| item.get_name() == "item")
623        .any(|item| element_nodes(item).iter().any(is_body_prose)),
624      _ => false,
625    })
626}

628/// A paragraph set apart from the running text: `\noindent`ed, or aligned (`center`, `flushleft`, `\raggedright`).
629fn is_set_apart(node: &Node) -> bool {
630  node.get_attribute("align").is_some()
631    || node.get_attribute("class").is_some_and(|class| {
632      class
633        .split_whitespace()
634        .any(|c| c == "ltx_noindent" || c.starts_with("ltx_align_"))
635    })
636}

638/// The text of a paragraph outside the boxes set in it (a minipage, a `\parbox`, a tabular, a picture) and its notes.
639fn running_text(node: &Node) -> String {
640  fn walk(node: &Node, out: &mut String) {
641    for c in node.get_child_nodes() {
642      match c.get_type() {
643        Some(NodeType::TextNode) => out.push_str(&c.get_content()),
644        Some(NodeType::ElementNode) => {
645          let name = c.get_name();
646          let boxed = name.starts_with("inline-")
647            || matches!(
648              name.as_str(),
649              "tabular" | "picture" | "svg" | "graphics" | "note" | "block" | "logical-block"
650            );
651          if !boxed {
652            walk(&c, out);
653          }
654        },
655        _ => {},
656      }
657    }
658  }
659  let mut out = String::new();
660  walk(node, &mut out);
661  out
662}
