5/// While a raw class's own frontmatter setter replays for its side effects (`replay_class_setter`, OXIDIZED_DESIGN
6/// #482), the frontmatter is the kernel's alone: nothing is queued (an entry, an annotation) or removed
7/// (`\lx@clear@frontmatter`, which the kernel's `\lx@add@title`, `\lx@add@date`, `\lx@add@abstract` and
8/// `\lx@add@keywords` run first), whatever path a body takes to the queue.
9fn frontmatter_frozen_for_replay() -> bool { lookup_bool("lx_replaying_class_setter") }
10
11/// frontmatter_raw contains the undigested commands to create frontmatter,
12/// along with the tag & attributes that would be created.
13/// Digestion is deferred until \maketitle, or something similar,
14/// to avoid extra side-effects, particularly when entries
15/// (eg. \title, \author) get redefined & replaced.
16/// Use dequeue_front_matter or \lx@clear@frontmatter if there should be only 1 entry of $tag
17/// Perl: queueFrontMatter($stomach, $tag, $attr, $command).
18pub fn queue_front_matter(tag: &str, attr: Option<&KeyVals>, command: Tokens) {
19  if frontmatter_frozen_for_replay() {
20    // (an abstract it queues no more: its late marker's ordinal goes with it, not to the next abstract added,
21    // [`take_late_mark`])
22    drop_late_mark(tag);
23    // (a block's rehearsal notes what it would have queued: the emitter counts an entry a block queues of its own,
24    // `rehearsal::note_frozen_entry`, 2503.02741)
25    rehearsal::note_frozen_entry(tag, attr, &command);
26    return;
27  }
28  let queue = frontmatter_queue_name();
29  // (the document's frontmatter commands this entry is made in, for the title-page check: OXIDIZED_DESIGN_DIVERGENCES
30  // #953 "106t")
31  note_frontmatter_entry(
32    tag,
33    attr
34      .and_then(|kv| kv.get_value("role"))
35      .map(|role| role.to_string())
36      .as_deref()
37      .unwrap_or(""),
38    Some(&command),
39  );
40  // (the frontmatter observer's ledger: OXIDIZED_DESIGN_DIVERGENCES #837)
41  record_api_call(tag, attr, &command);
42  // (what `\@author` holds as a creator is queued: the prefix of a tail appended to it after,
43  // `frontmatter_stores::author_tail_calls`)
44  if tag == "ltx:creator" {
45    crate::frontmatter_stores::note_author_seen();
46  }
47  note_document_sets_front_matter(
48    tag,
49    &attr
50      .and_then(|kv| kv.get_value("role"))
51      .map(|role| role.to_string())
52      .unwrap_or_default(),
53  );
54  let mut attr_hash = queued_attributes(attr);
55  // (queued after `\maketitle`'s flush: digested by a later one, still late, [`LATE_ENTRY`])
56  if lookup_bool("lx_frontmatter_here_digested") {
57    attr_hash.insert(LATE_ENTRY.to_string(), "true".to_string());
58    // (and the late marker before it, of its command: [`LATE_MARK_ORDINAL`])
59    if tag == "ltx:abstract" {
60      let ordinal = take_late_mark();
61      if !ordinal.is_empty() {
62        attr_hash.insert(LATE_MARK_ORDINAL.to_string(), ordinal);
63      }
64    }
65  }
66  // (handed on from a raw class's store: its title-page provenance, [`TITLE_PAGE_ENTRY`])
67  let from = lookup_string(crate::frontmatter_stores::STORE_ENTRY_FROM);
68  if !from.is_empty() {
69    attr_hash.insert(TITLE_PAGE_ENTRY.to_string(), from);
70  }
71  DebugFeature!(
72    "frontmatter",
73    "FRONT Queuing {tag} [{:?}] {command}",
74    attr_hash
75  );
76  let missing = with_value(queue, |v| !matches!(v, Some(Stored::FrontmatterRaw(_))));
77  if missing {
78    assign_value(
79      queue,
80      Stored::FrontmatterRaw(Vec::new()),
81      Some(Scope::Global),
82    );
83  }
84  with_value_mut(queue, |val_opt| {
85    if let Some(&mut Stored::FrontmatterRaw(ref mut queue)) = val_opt {
86      queue.push((tag.to_string(), attr_hash, command));
87    }
88  });
89}

91/// A queued entry's attributes, as `queue_front_matter` stores them.
92/// Convert KeyVals to a hash, but be concerned about multiple values!?!?
93/// (Perl: ToString($attr->getValue($_)) — the last value for multi-valued keys)
94pub(in crate::base_utilities) fn queued_attributes(attr: Option<&KeyVals>) -> TagAttrs {
95  let mut attr_hash = TagAttrs::default();
96  if let Some(kv) = attr {
97    for key in kv.get_keyvals().keys() {
98      if let Some(value) = kv.get_value(key) {
99        attr_hash.insert(key.clone(), value.to_string());
100      }
101    }
102  }
103  attr_hash
104}

106/// The queue a frontmatter entry goes to: the document's (`frontmatter_raw`), or, while a stand-in binding's copy of
107/// its author file is read (`binding::content::read_stand_in_copy`, OXIDIZED_DESIGN_DIVERGENCES #731), the copy's
108/// defaults (`frontmatter_copy_defaults`): in TeX a class's own setter call (atlasdoc.cls:189 `\author{The ATLAS
109/// Collaboration}`, :190 `\date{\today}`) only stores the default the document's own call replaces, which its title
110/// page prints if the document makes none ([`hand_copy_frontmatter_defaults`]). A removal (`\lx@clear@frontmatter`,
111/// which `\date` runs first) applies to the same queue: the copy's later `\date{}` (atlasdoc.cls:824 under
112/// `cernpreprint`) clears its `\today`.
113fn frontmatter_queue_name() -> &'static str {
114  if definition::origin::copy_provides() != 0 {
115    "frontmatter_copy_defaults"
116  } else {
117    "frontmatter_raw"
118  }
119}

121/// A frontmatter entry `tag` of role `role` (`""`: every role) is set — queued or removed — outside a stand-in copy's
122/// read ([`frontmatter_queue_name`]): the document's (or a binding's) setter call, whatever its value, replaces the
123/// class default the copy stored, as `\def\@date{#1}` does in TeX ([`hand_copy_frontmatter_defaults`]): `\date{}`
124/// removes the class's `\date{\today}`, and `\author{}`/`\title{}` theirs, though they queue nothing.
125fn note_document_sets_front_matter(tag: &str, role: &str) {
126  if definition::origin::copy_provides() != 0 {
127    // (the copy stored a default: from now on its kernel field's setter is observed, as a binding's `\author` may
128    // queue nothing for an empty call — inst_support's accumulates)
129    if !definition::origin::setter_watch() {
130      definition::origin::watch_setter_calls(&COPY_DEFAULT_SETTERS.map(|(setter, ..)| setter));
131    }
132  } else {
133    assign_value(
134      &s!("frontmatter_set:{tag}:{role}"),
135      true,
136      Some(Scope::Global),
137    );
138  }
139}

141/// The kernel's fields a stand-in copy's setter stores a default of: setter, tag, role, store.
142const COPY_DEFAULT_SETTERS: [(&str, &str, &str, &str); 3] = [
143  ("\\author", "ltx:creator", "author", "\\@author"),
144  ("\\date", "ltx:date", "creation", "\\@date"),
145  ("\\title", "ltx:title", "", "\\@title"),
146];

148/// Did the document set a frontmatter entry `tag` of role `role` — call its setter (`state::setter_called`, the
149/// kernel's fields: whatever binding defines `\author`), or queue or remove one ([`note_document_sets_front_matter`];
150/// a role-less removal sets every role)?
151fn document_set_front_matter(tag: &str, role: &str) -> bool {
152  lookup_bool(&s!("frontmatter_set:{tag}:{role}"))
153    || (!role.is_empty() && lookup_bool(&s!("frontmatter_set:{tag}:")))
154    || COPY_DEFAULT_SETTERS
155      .iter()
156      .any(|(setter, t, r, _)| *t == tag && *r == role && setter_called(&T_CS!(setter)))
157}

159/// At `\maketitle` (`\lx@store@defaults`): hand a stand-in copy's frontmatter defaults (`frontmatter_queue_name`) to
160/// the document's frontmatter, each only when the document's setter made no call of its kind (tag and role: its own
161/// `\author` replaces the class's, an empty `\date{}` too, `document_set_front_matter`), and, for the kernel's fields,
162/// only when the copy's own title page prints the store (`frontmatter_stores::copy_title_page_prints`; the binding's
163/// title page prints them when the copy defines none). Five ATLAS papers had the class's `<creator>` beside the
164/// document's (1805.01649, 1806.07355, 1807.02012, 1902.05892, 1904.03536) and a `<date>` the class does not print
165/// (1902.05892, 1904.03536).
166pub fn hand_copy_frontmatter_defaults() {
167  let defaults: Vec<RawFrontmatter> = match remove_value("frontmatter_copy_defaults") {
168    Some(Stored::FrontmatterRaw(defaults)) => defaults,
169    _ => return,
170  };
171  for (tag, attr, command) in defaults {
172    let role = attr.get("role").cloned().unwrap_or_default();
173    if document_set_front_matter(&tag, &role) || has_front_matter_with_role(&tag, &role) {
174      continue;
175    }
176    let store = COPY_DEFAULT_SETTERS
177      .iter()
178      .find(|(_, t, r, _)| *t == tag && (*r == role || r.is_empty()))
179      .map(|(.., store)| *store);
180    if let Some(store) = store
181      && crate::frontmatter_stores::copy_title_page_prints(&T_CS!(store)) == Some(false)
182    {
183      Info!(
184        "frontmatter",
185        tag,
186        s!("The class's default {tag} is not printed by its title page: not shown")
187      );
188      continue;
189    }
190    let missing = with_value("frontmatter_raw", |v| {
191      !matches!(v, Some(Stored::FrontmatterRaw(_)))
192    });
193    if missing {
194      assign_value(
195        "frontmatter_raw",
196        Stored::FrontmatterRaw(Vec::new()),
197        Some(Scope::Global),
198      );
199    }
200    with_value_mut("frontmatter_raw", |val_opt| {
201      if let Some(&mut Stored::FrontmatterRaw(ref mut queue)) = val_opt {
202        queue.push((tag, attr, command));
203      }
204    });
205  }
206}

208/// Is a frontmatter entry `tag` of role `role` (`""`: any) queued or digested ([`has_front_matter`])?
209fn has_front_matter_with_role(tag: &str, role: &str) -> bool {
210  let matches =
211    |attr: &TagAttrs| role.is_empty() || attr.get("role").map(String::as_str) == Some(role);
212  let queued = with_value("frontmatter_raw", |v| match v {
213    Some(Stored::FrontmatterRaw(queue)) => queue
214      .iter()
215      .any(|entry| entry.0 == tag && matches(&entry.1)),
216    _ => false,
217  });
218  queued
219    || with_value("frontmatter", |v| match v {
220      Some(Stored::HashTagData(frnt)) => frnt
221        .get(tag)
222        .is_some_and(|list| list.iter().any(|e| matches(&e.attr))),
223      _ => false,
224    })
225}

227/// Build and queue the deferred `\lx@add@frontmatter@now` invocation for a
228/// frontmatter entry `<tag attrs>content</tag>`. Shared by `\lx@add@frontmatter`
229/// (content present) and `\lx@add@frontmatter@container` (content = None — a
230/// deliberate empty container element that anchors later annotations).
231/// Perl: queueFrontMatter($stomach, $tag, $attr,
232///   Invocation(T_CS('\lx@add@frontmatter@now'), $keys, $tag, $attr, $tokens)).
233pub(in crate::base_utilities) fn queue_add_frontmatter_now(
234  keys_opt: Option<&KeyVals>,
235  tag_tks: &Tokens,
236  attrs_opt: Option<&KeyVals>,
237  content: Option<&Tokens>,
238) -> Result<()> {
239  let command = now_command(keys_opt, tag_tks, attrs_opt, content)?;
240  queue_front_matter(&tag_tks.to_string(), attrs_opt, command);
241  Ok(())
242}

244/// The command a queued entry is digested by: `\lx@add@frontmatter@now[keys]{tag}[attributes]{content}`.
245pub(in crate::base_utilities) fn now_command(
246  keys_opt: Option<&KeyVals>,
247  tag_tks: &Tokens,
248  attrs_opt: Option<&KeyVals>,
249  content: Option<&Tokens>,
250) -> Result<Tokens> {
251  let mut inv_tokens: Vec<Token> = vec![T_CS!("\\lx@add@frontmatter@now")];
252  if let Some(keys) = keys_opt {
253    inv_tokens.push(T_OTHER!("["));
254    inv_tokens.extend(keys.revert()?.unlist());
255    inv_tokens.push(T_OTHER!("]"));
256  }
257  inv_tokens.push(T_BEGIN!());
258  inv_tokens.extend(tag_tks.unlist_ref().iter().copied());
259  inv_tokens.push(T_END!());
260  if let Some(attrs) = attrs_opt {
261    inv_tokens.push(T_OTHER!("["));
262    inv_tokens.extend(attrs.revert()?.unlist());
263    inv_tokens.push(T_OTHER!("]"));
264  }
265  inv_tokens.push(T_BEGIN!());
266  if let Some(content) = content {
267    inv_tokens.extend(content.unlist_ref().iter().copied());
268  }
269  inv_tokens.push(T_END!());
270  Ok(Tokens::new(inv_tokens))
271}

284/// Is `affiliation`'s first mark (its label, `entry_mark`) one a queued author carries — an affiliation number, not a
285/// footnote symbol (`is_symbol_mark`: `$^*$Equal contribution` is the legend of the authors' stars)? Only then do the
286/// authors showing that mark answer the affiliation: a mark no author carries (a typo, or authors without marks after
287/// marked ones: `\author{Ann$^1$}\affil{$^1$X}\author{Bob}\affil{$^2$Y}`) would label an affiliation no one requests,
288/// which the frontmatter places on an author of its own without a name. Nor a combined mark
289/// (`\author{Ann$^{1}$}\author{Bob$^{2}$}\affil{$^{1,2}$Univ X}`, 77n r3 review probe w1): the affiliation is labelled
290/// by its first label only (`mark_label`), so Bob lost it; it is the run's.
291pub fn queued_authors_carry_entry_mark(affiliation: &Tokens) -> Result<bool> {
292  Ok(match entry_mark(affiliation.unlist_ref()) {
293    Some(mark) => {
294      !is_symbol_mark(&mark)
295        && clean_frontmatter_labels(&Tokens::new(mark.clone()).to_string(), "affiliation").len()
296          == 1
297        && !queued_requesters(&mark)?.is_empty()
298    },
299    None => false,
300  })
301}

303/// Is a frontmatter item with this tag queued (`frontmatter_raw`) or already digested into the
304/// frontmatter (`frontmatter`, which `digest_front_matter` fills at `\maketitle`; a pending
305/// annotation stub is not an item)?
306pub fn has_front_matter(tag: &str) -> bool {
307  let queued = with_value("frontmatter_raw", |v| match v {
308    Some(Stored::FrontmatterRaw(queue)) => queue.iter().any(|entry| entry.0 == tag),
309    _ => false,
310  });
311  queued
312    || with_value("frontmatter", |v| match v {
313      Some(Stored::HashTagData(frnt)) => frnt.get(tag).is_some_and(|list| {
314        list
315          .iter()
316          .any(|e| e.attr.get("role").map(String::as_str) != Some("pending"))
317      }),
318      _ => false,
319    })
320}

322/// This removes previously stored (but deferred) frontmatter that is being overridden.
323/// It matches the tag and any stored attributes in `attr`.
324/// Perl: dequeueFrontMatter($tag, %attr).
325pub fn dequeue_front_matter(tag: &str, attr: &[(&str, &str)]) {
326  if frontmatter_frozen_for_replay() {
327    return;
328  }
329  // (the observer's ledger: the calls it removes, OXIDIZED_DESIGN_DIVERGENCES #837)
330  record_clear(tag, attr);
331  // (and the title-page check's: what the calls it removes were given is no longer wanted, #953 "106t")
332  note_frontmatter_clear(
333    tag,
334    attr
335      .iter()
336      .find(|(key, _)| *key == "role")
337      .map(|(_, role)| *role),
338  );
339  note_document_sets_front_matter(
340    tag,
341    attr
342      .iter()
343      .find(|(key, _)| *key == "role")
344      .map_or("", |(_, role)| role),
345  );
346  forget_queue_reading();
347  with_value_mut(frontmatter_queue_name(), |val_opt| {
348    if let Some(&mut Stored::FrontmatterRaw(ref mut queue)) = val_opt {
349      queue.retain(|entry| {
350        let keep = entry.0 != tag
351          || attr
352            .iter()
353            .any(|(k, v)| entry.1.get(*k).map(String::as_str).unwrap_or("") != *v);
354        if !keep {
355          DebugFeature!("frontmatter", "FRONT DEQueuing {} [{:?}]", entry.0, entry.1);
356        }
357        keep
358      });
359    }
360  });
361}

363/// What LaTeX's `\author` does to the authors before it (`\def\@author{#2}`; Perl's `dequeueFrontMatter('ltx:creator', role =>
364/// 'author')`, Base_Utility.pool.ltxml:691): the last call wins. Also a binding's own `\author`, whose body adds its authors one
365/// at a time ([`begin_author_body`]: IEEEtran's blocks; 2606.16069's two `\author` calls).
366pub(in crate::base_utilities) fn replace_queued_authors() {
367  dequeue_front_matter("ltx:creator", &[("role", "author")]);
368  // LaTeX's `\def\@author{#2}` replaces the authors a `\maketitle` that left
369  // `\author` live already typeset (KOMA ≥ 3.12, scrbook.cls:3263-3290; a class's
370  // title page from `\AfterEndPreamble`, udesoftec.cls:1230-1241) for the NEXT
371  // `\maketitle`: they are superseded there (`\lx@maketitle@supersede`), not here —
372  // without one, the authors a fallback placed stay (Perl the same).
373  assign_value(
374    "lx_authors_superseded",
375    Stored::Bool(true),
376    Some(Scope::Global),
377  );
378  // A store handed on since that flush went to the authors replaced here: the next
379  // harvest hands it again (`harvest_stores`, `lx_stores_late`).
380  assign_value(
381    "lx_stores_late",
382    Stored::String(pin("")),
383    Some(Scope::Global),
384  );
385  // The replaced authors take their handed tail with them (a binding's own
386  // `\lx@add@authors` never runs `\lx@author@flush`), and the creators this
387  // `\author` makes are counted from here (`\lx@author@handed`).
388  assign_value("lx_author_handed", Stored::Bool(false), Some(Scope::Global));
389  assign_value(
390    "lx_author_creators_before",
391    Stored::Int(queued_creator_count() as i64),
392    Some(Scope::Global),
393  );
394}

396/// Begin a binding's `\author` body, whose calls queue its authors and their contacts one by one where Perl's `\author` queues
397/// them all inside one deferred creator (IEEEtran's `\IEEEauthorblockN`, `\IEEEauthorblockA`): the last call wins as in LaTeX
398/// (`\def\@author{#1}`, Perl's `dequeueFrontMatter`), so this removes the authors before it and whatever the last such body
399/// queued. 2606.16069: two `\author` calls, the PDF and Perl four creators; every call kept, eight. A replay of a raw class's setter
400/// that reaches it changes nothing (`frontmatter_frozen_for_replay`): it queues nothing, and the last body stays the last.
401pub fn begin_author_body() {
402  if frontmatter_frozen_for_replay() {
403    return;
404  }
405  replace_queued_authors();
406  forget_queue_reading();
407  let queue = frontmatter_queue_name();
408  let earlier = with_value(AUTHOR_BODY_ENTRIES, |value| match value {
409    Some(Stored::FrontmatterRaw(entries)) => entries.clone(),
410    _ => Vec::new(),
411  });
412  let start = with_value_mut(queue, |value| match value {
413    Some(&mut Stored::FrontmatterRaw(ref mut queued)) => {
414      for entry in &earlier {
415        if let Some(at) = queued.iter().position(|queued| queued == entry) {
416          queued.remove(at);
417        }
418      }
419      queued.len()
420    },
421    _ => 0,
422  });
423  assign_value(
424    AUTHOR_BODY_START,
425    Stored::Int(start as i64),
426    Some(Scope::Global),
427  );
428  record_author_body_clear();
429}

431/// End a binding's `\author` body ([`begin_author_body`]): what it queued is what a later `\author` takes back.
432pub fn end_author_body() {
433  if frontmatter_frozen_for_replay() {
434    return;
435  }
436  let start = usize::try_from(lookup_int(AUTHOR_BODY_START)).unwrap_or(0);
437  let entries = with_value(frontmatter_queue_name(), |value| match value {
438    Some(Stored::FrontmatterRaw(queued)) => queued.get(start..).unwrap_or(&[]).to_vec(),
439    _ => Vec::new(),
440  });
441  assign_value(
442    AUTHOR_BODY_ENTRIES,
443    Stored::FrontmatterRaw(entries),
444    Some(Scope::Global),
445  );
446  record_author_body_end();
447}

475/// A `\par` digested in the frontmatter item `item` (its whatsit's `frontmatter_item`), constructed where the schema is
476/// known: when the item's element holds only inline content (`ltx:acknowledgements`, `ltx:keywords`, `ltx:date`, …: no
477/// `ltx:p`, but `ltx:break`) and is reached from the insertion point without passing an element that holds paragraphs
478/// of its own, the paragraph break is a line break of the item, where text mode would merge the paragraphs (user ruling
479/// 2026-10-18; apa.cls's two-paragraph `\acknowledgements`, OXIDIZED_DESIGN_DIVERGENCES #766). A break only parts
480/// lines: none before the item's first content, none after another (a run of `\par`s), and those left at its end are
481/// dropped when it closes ([`drop_trailing_paragraph_breaks`]). Returns whether the `\par` was the item's (it then
482/// closes nothing); otherwise it is the ordinary `\par`.
483pub(crate) fn frontmatter_paragraph_break(document: &mut Document, item: &str) -> Result<bool> {
484  if document::can_contain_qname(item, "ltx:p") || !document::can_contain_qname(item, "ltx:break") {
485    return Ok(false);
486  }
487  let Some(current) = document.get_element() else {
488    return Ok(false);
489  };
490  let item_sym = pin(item);
491  let mut open = Vec::new();
492  let mut node = Some(current.clone());
493  let item_node = loop {
494    let Some(n) = node else {
495      return Ok(false);
496    };
497    if document::get_node_qname(&n) == item_sym {
498      break n;
499    }
500    if document::can_contain(&n, "ltx:p") || document::get_node_qname(&n) == pin!("ltx:p") {
501      return Ok(false);
502    }
503    node = n.get_parent();
504    open.push(n);
505  };
506  open.push(item_node.clone());
507  if last_line_end(&item_node, &open) == Some(false) && document::can_contain(&current, "ltx:break")
508  {
509    document.insert_element(
510      "ltx:break",
511      Vec::new(),
512      Some(map!(PARAGRAPH_BREAK_MARK.to_string() => "true".to_string())),
513    )?;
514  }
515  Ok(true)
516}

518/// Note that the `\par`s digested from here on, at this mode frame, are paragraph breaks of the frontmatter field `tag`
519/// ([`frontmatter_paragraph_break`]): a box, an alignment cell or math inside it begins a deeper mode frame, where a
520/// `\par` is the ordinary one (in a box or cell it ends nothing, tex.web §1096 `end_graf`), while a group (`{\bfseries …}`)
521/// begins none.
522pub(in crate::base_utilities) fn note_frontmatter_paragraphs(tag: &str) {
523  assign_value(
524    "lx_frontmatter_par_item",
525    tag.to_string(),
526    Some(Scope::Local),
527  );
528  assign_value(
529    "lx_frontmatter_par_depth",
530    Stored::Int(mode_frame_depth()),
531    Some(Scope::Local),
532  );
533}
534
535/// How many frames bind the mode (`MODE`): each mode switch — a box, a cell, math, a frontmatter item's text — binds it
536/// in a frame of its own.
537pub(crate) fn mode_frame_depth() -> i64 { with_stacked_values("MODE", |modes| modes.len() as i64) }
538
539/// The frontmatter field whose paragraph break a `\par` digested now is (what [`note_frontmatter_paragraphs`] noted, at
540/// the same mode frame), if any.
541pub(crate) fn frontmatter_paragraph_field() -> Option<String> {
542  let item = lookup_string("lx_frontmatter_par_item");
543  (!item.is_empty() && lookup_int("lx_frontmatter_par_depth") == mode_frame_depth()).then_some(item)
544}

546/// The transient mark on the `ltx:break`s [`frontmatter_paragraph_break`] makes (dropped with the other `_`
547/// attributes when the document is finalized).
548const PARAGRAPH_BREAK_MARK: &str = "_paragraph_break";
549
550/// Does `node`'s content end with a paragraph break the frontmatter made (`Some(true)`), with other content
551/// (`Some(false)`), or is it empty (`None`)? Whitespace counts for nothing, and so does an element still `open` (where
552/// the content goes on) that holds nothing yet.
553fn last_line_end(node: &Node, open: &[Node]) -> Option<bool> {
554  let mut child = node.get_last_child();
555  while let Some(n) = child {
556    match n.get_type() {
557      Some(NodeType::TextNode) if n.get_content().trim().is_empty() => {},
558      Some(NodeType::ElementNode) if n.has_attribute(PARAGRAPH_BREAK_MARK) => return Some(true),
559      Some(NodeType::ElementNode) => match last_line_end(&n, open) {
560        Some(found) => return Some(found),
561        None if open.iter().any(|o| o.to_hashable() == n.to_hashable()) => {},
562        None => return Some(false),
563      },
564      _ => return Some(false),
565    }
566    child = n.get_prev_sibling();
567  }
568  None
569}
570
571/// Drop the paragraph breaks [`frontmatter_paragraph_break`] left at the end of the frontmatter element `node` (in its
572/// last group too): a `\par` that ends an item ends no line of it.
573pub(in crate::base_utilities) fn drop_trailing_paragraph_breaks(
574  document: &mut Document,
575  node: &Node,
576) {
577  let mut child = node.get_last_child();
578  while let Some(n) = child {
579    let previous = n.get_prev_sibling();
580    match n.get_type() {
581      Some(NodeType::TextNode) if n.get_content().trim().is_empty() => {},
582      Some(NodeType::ElementNode) if n.has_attribute(PARAGRAPH_BREAK_MARK) => {
583        document.remove_node(n)
584      },
585      Some(NodeType::ElementNode) => return drop_trailing_paragraph_breaks(document, &n),
586      _ => return,
587    }
588    child = previous;
589  }
590}
