8pub fn insert_frontmatter(document: &mut Document) -> Result<()> {
9  if lookup_bool("frontmatter_done") {
10    return Ok(());
11  }
12  // A raw class's title-page stores in a document without `\maketitle` (frontmatter_stores.rs).
13  crate::frontmatter_stores::harvest_stores(false)?;
14  digest_front_matter()?; // If needed
15  let frontmatter_elements_set: HashSet<String> = FRONTMATTER_ELEMENTS
16    .iter()
17    .map(ToString::to_string)
18    .collect();
19
20  // Collect the frontmatter hash keys via with_value — we only need the
21  // key set here; the full HashTagData clone previously happened just
22  // to call .keys().cloned().collect(). The hash itself is consumed a
23  // few lines below via remove_value, so no iteration on the borrow
24  // survives past this closure.
25  let set_keys: Vec<String> = with_value("frontmatter", |v| match v {
26    Some(Stored::HashTagData(frnt)) => frnt.keys().cloned().collect(),
27    _ => Vec::new(),
28  });
29  if set_keys.is_empty() {
30    return Ok(());
31  }
32
33  // If doc ONLY has abstract as frontmatter, defer until abstract's document location
34  if set_keys.len() == 1 && set_keys[0] == "ltx:abstract" && !lookup_bool("frontmatter_deferred") {
35    assign_value("frontmatter_deferred", true, Some(Scope::Global));
36    return Ok(());
37  }
38
39  // OK, we're placing FrontMatter here, now.
40  assign_value("frontmatter_done", true, Some(Scope::Global));
41  remove_frontmatter_marks(document);
42
43  // Remove frontmatter and replace with empty
44  let mut frontmatter = match remove_value("frontmatter") {
45    Some(Stored::HashTagData(frnt)) => frnt,
46    _ => return Ok(()),
47  };
48  assign_value(
49    "frontmatter",
50    Stored::HashTagData(HashMap::default()),
51    Some(Scope::Global),
52  );
53
54  // Order: first go through frontmatter_elements order, then any custom keys
55  let custom_keys: Vec<String> = frontmatter
56    .keys()
57    .filter(|key| !frontmatter_elements_set.contains(key.as_str()))
58    .map(ToString::to_string)
59    .collect();
60  let mut all_keys: Vec<String> = FRONTMATTER_ELEMENTS
61    .iter()
62    .map(ToString::to_string)
63    .collect();
64  all_keys.extend(custom_keys);
65
66  // A `\maketitle` run inside a box capture (ltx-talk.cls:515 frames, unifront,
67  // `\parbox{…}{\maketitle}`) no longer reaches here trapped: `place_frontmatter`
68  // always inserts through a root-level `_Capture_` (#262), so the frontmatter lands
69  // at the head as real elements. (Before 56gf a current-point flush degraded each
70  // entry to `ltx:text class="ltx_<name>"` inside the box.) Guard
71  // `perfect_kernel_batch54::maketitle_inside_a_box_goes_to_the_head`.
72  for key in &all_keys {
73    if let Some(list) = frontmatter.remove(key) {
74      // Dubious, but assures that frontmatter appears in text mode...
75      document.set_box_to_absorb(
76        Tbox::new(
77          pin!(""),
78          lookup_font(),
79          None,
80          Tokens!(T_SPACE!()),
81          SymHashMap::default(),
82        )
83        .into(),
84      );
85      for item in list.into_entries() {
86        insert_frontmatter_entry(document, &item)?;
87      }
88      document.expire_box_to_absorb();
89    }
90  }
91  coalesce_empty_creators(document)?;
92  distribute_upfront_contacts(document)?;
93  let answered = answered_labels(document);
94  relocate_annotations(document)?;
95  link_entry_addresses(document, None)?;
96  let unshown_marks = show_unanswered_marks(document, &answered)?;
97  drop_empty_contacts(document)?;
98  drop_repeated_orcids(document);
99  count_lost_marks(document, &unshown_marks);
100  // (an entry's line of addresses, read once every route has placed the entry and the observer's writing is done: its
101  // addresses the entry's e-mails, placed by the Link step; OD #966)
102  join_split_address_contacts(document)?;
103  let kinded = kind_address_lines(document)?;
104  if !kinded.is_empty() {
105    link_entry_addresses(document, Some(&kinded))?;
106    drop_empty_contacts(document)?;
107  }
108  // (a list of addresses one contact prints for several people, each its owner's e-mail: census138 ADDR-b, OD #972)
109  distribute_address_lists(document)?;
110  // (an e-mail's correspondence label, read once every address is placed: its owner's note, review rev107c N4)
111  write_role_notes(document)?;
112  if !kinded.is_empty() {
113    show_line_marks(document)?;
114  }
115  // (an affiliation's dedication lines, read once every route has placed it: the paper's dedication, 107c)
116  kind_dedication_lines(document)?;
117  // (an affiliation that is a legend, read once every route has placed it: its carriers' note, OD #961)
118  kind_legend_contacts(document)?;
119  // (an affiliation that is a link or a membership grade, read where the legend's is: the block's link, its carriers'
120  // grade, OD #969)
121  kind_link_and_grade_contacts(document)?;
122  // (a contact a creator holds twice, once every route and Kind step has placed it: its first, OD #969)
123  drop_repeated_contacts(document);
124  // (the authors as written: a name the writer re-parted holds one person, OD #928)
125  flag_merged_creators(document)?;
126  // (the unclaimed-text net's notes: their lines no frontmatter field and no text already set holds)
127  keep_unclaimed_net_notes(document)?;
128  // (the content invariant on the placed output, under the emitter's `=2`: every printed word placed once or counted, OD #980)
129  check_placed_content(document);
130  Ok(())
131}

133/// The words a note line counts for [`keep_unclaimed_net_notes`] (the frontmatter scorer's, fm_scorer/score_tiers.py
134/// `content_words`, and its stop words): its letter runs of two letters or more outside the stop words and its digit
135/// runs of three or more, folded — accents dropped ([`resolver::base_letters`]), lower-cased — in printed order. Beyond
136/// the scorer's ASCII runs, a run of any script's letters is a word (`Москва`), and each Han ideograph or kana, of
137/// scripts written without spaces, is a word of its own (2206.01007's `北京师范大学天文系`, 2601.14329's
138/// `北京航空航天大学`; 105a r2).
139pub(super) fn net_content_words(text: &str) -> Vec<String> {
140  const STOPWORDS: &[&str] = &[
141    "a", "an", "the", "of", "and", "or", "in", "on", "at", "for", "to", "from", "by", "with",
142    "via", "de", "la", "le", "les", "du", "des", "del", "della", "di", "da", "do", "dos", "das",
143    "e", "y", "et", "und", "von", "van", "der", "den", "zu", "im", "am", "al", "el", "en", "is",
144    "are", "be", "as", "this", "that", "its", "it", "we", "our", "their", "his", "her", "no",
145    "not",
146  ];
147  let folded = resolver::base_letters(text).to_lowercase();
148  let mut words = Vec::new();
149  let mut run = String::new();
150  let mut letters = false;
151  let flush = |run: &mut String, letters: bool, words: &mut Vec<String>| {
152    let keep = if letters {
153      run.chars().count() >= 2 && !STOPWORDS.contains(&run.as_str())
154    } else {
155      run.len() >= 3
156    };
157    if keep {
158      words.push(std::mem::take(run));
159    } else {
160      run.clear();
161    }
162  };
163  for c in folded.chars() {
164    if unspaced_script(c) {
165      if !run.is_empty() {
166        flush(&mut run, letters, &mut words);
167      }
168      words.push(c.to_string());
169      continue;
170    }
171    let kind = if c.is_alphabetic() {
172      Some(true)
173    } else if c.is_ascii_digit() {
174      Some(false)
175    } else {
176      None
177    };
178    if kind != Some(letters) && !run.is_empty() {
179      flush(&mut run, letters, &mut words);
180    }
181    if let Some(kind) = kind {
182      letters = kind;
183      run.push(c);
184    }
185  }
186  if !run.is_empty() {
187    flush(&mut run, letters, &mut words);
188  }
189  words
190}

192/// Is `c` a Han ideograph or a kana — a letter of a script written without spaces between its words, each counted as a
193/// word of its own ([`net_content_words`])?
194fn unspaced_script(c: char) -> bool {
195  matches!(c,
196    '\u{3040}'..='\u{30FF}' // Hiragana, Katakana
197    | '\u{31F0}'..='\u{31FF}' // Katakana phonetic extensions
198    | '\u{3400}'..='\u{4DBF}' // CJK Unified Ideographs Extension A
199    | '\u{4E00}'..='\u{9FFF}' // CJK Unified Ideographs
200    | '\u{F900}'..='\u{FAFF}' // CJK Compatibility Ideographs
201    | '\u{20000}'..='\u{3134F}' // CJK Unified Ideographs Extensions B-G
202  )
203}

205/// The text of `node` without what the nodes `skip` hold.
206fn text_outside(node: &Node, skip: &[Node], out: &mut String) {
207  for child in node.get_child_nodes() {
208    if child.get_type() == Some(NodeType::ElementNode) {
209      if !skip.contains(&child) {
210        text_outside(&child, skip, out);
211        out.push(' ');
212      }
213    } else {
214      out.push_str(&child.get_content());
215    }
216  }
217}

219/// The text of `node` without the text of an `ltx:ERROR` at any depth in it (an error's text is no printed word: an
220/// undefined `\songti`'s name in the group `{\songti\zihao{-4}#2}`, physics_report.sty, 2601.14329).
221fn text_without_errors(node: &Node, out: &mut String) {
222  if node.get_type() == Some(NodeType::ElementNode) {
223    if node.get_name() != "ERROR" {
224      for child in node.get_child_nodes() {
225        text_without_errors(&child, out);
226      }
227    }
228  } else {
229    out.push_str(&node.get_content());
230  }
231}

233/// A note's lines ([`keep_unclaimed_net_notes`]): what stands between its `ltx:break`s, each `ltx:p` apart, with the
234/// break that ends it; the text of an `ltx:ERROR` in it, at any depth, left out ([`text_without_errors`]).
235fn net_note_lines(parent: &Node, lines: &mut Vec<(Vec<Node>, String)>) {
236  let mut line: (Vec<Node>, String) = (Vec::new(), String::new());
237  for child in parent.get_child_nodes() {
238    let element = child.get_type() == Some(NodeType::ElementNode);
239    if element && child.get_name() == "break" {
240      line.0.push(child);
241      lines.push(std::mem::take(&mut line));
242    } else if element && child.get_name() == "p" {
243      if !line.0.is_empty() {
244        lines.push(std::mem::take(&mut line));
245      }
246      net_note_lines(&child, lines);
247    } else {
248      text_without_errors(&child, &mut line.1);
249      line.1.push(' ');
250      line.0.push(child);
251    }
252  }
253  if !line.0.is_empty() {
254    lines.push(line);
255  }
256}

258/// Remove the `ltx:break`s a note starts or ends with, blank text aside (the lines [`keep_unclaimed_net_notes`] took out
259/// leave them: `\\AB@affillist` starts with its `\\\\`).
260fn trim_edge_breaks(document: &mut Document, note: &Node) {
261  let blank =
262    |n: &Node| n.get_type() != Some(NodeType::ElementNode) && n.get_content().trim().is_empty();
263  for from_end in [false, true] {
264    loop {
265      let mut children = note.get_child_nodes();
266      if from_end {
267        children.reverse();
268      }
269      match children.into_iter().find(|n| !blank(n)) {
270        Some(edge)
271          if edge.get_type() == Some(NodeType::ElementNode) && edge.get_name() == "break" =>
272        {
273          document.remove_node(edge);
274        },
275        _ => break,
276      }
277    }
278  }
279}

281/// The classification codes `text` prints, folded ([`resolver::base_letters`], lower-cased), which the net keeps below
282/// its floor ([`keep_unclaimed_net_notes`]): a piece of digits and letters, a digit in it, its letters at most 3 to a
283/// run, parted by dots or dashes or run together with the digits — a PACS code `98.80.Cq` (europhys `\Pacs`,
284/// cond-mat9812194), an MSC code `20N05` (ourlematema `\MSC`, 1909.06690; basm `\MSC`, 1601.01922), an ACM class
285/// `H.2.4`; not a bare number (OD #953, 105a).
286fn code_pieces(text: &str) -> Vec<String> {
287  resolver::base_letters(text)
288    .to_lowercase()
289    .split(|c: char| c.is_whitespace() || ",;:()[]".contains(c))
290    .map(|piece| piece.trim_matches(|c: char| c == '.' || c == '-'))
291    .filter(|piece| {
292      let runs: Vec<&str> = piece
293        .split(['.', '-'])
294        .filter(|run| !run.is_empty())
295        .collect();
296      let letter_runs_short = runs.iter().all(|run| {
297        run
298          .split(|c: char| c.is_ascii_digit())
299          .all(|letters| letters.len() <= 3)
300      });
301      piece.chars().any(|c| c.is_ascii_digit())
302        && piece
303          .chars()
304          .all(|c| c.is_ascii_alphanumeric() || c == '.' || c == '-')
305        && letter_runs_short
306        && (runs.len() > 1 || piece.chars().any(|c| c.is_ascii_alphabetic()))
307    })
308    .map(ToString::to_string)
309    .collect()
310}

312/// Is the content word `word` ([`net_content_words`], folded) a date's — a month's name, a year — or part of a code
313/// `codes` holds ([`code_pieces`])? The words the net keeps below its floor ([`keep_unclaimed_net_notes`]).
314fn code_or_date_word(word: &str, codes: &[String]) -> bool {
315  date_word(word) || codes.iter().any(|code| code.contains(word))
316}

318/// The label a code's or a date's line leads with, of its content words `words` ([`net_content_words`], in printed
319/// order): a dateline's (received, accepted, revised, published, submitted; any case, a colon after it or not) before a
320/// date's word ([`date_word`]: "Received: 2021", "Accepted 2021"), or a classification's (MSC, PACS; a year after it or
321/// not) on a line that prints a code `codes` holds ("MSC 2020: 05C12"). Kept below the floor with the line's code or
322/// date words ([`keep_unclaimed_net_notes`]); a leading word of no such list is none ("ICML 2021", "Room 101"; 105a r2).
323fn leading_label<'w>(words: &'w [String], codes: &[String]) -> Option<&'w str> {
324  const DATELINE: &[&str] = &["received", "accepted", "revised", "published", "submitted"];
325  let (first, rest) = words.split_first()?;
326  let label = if DATELINE.contains(&first.as_str()) {
327    rest.iter().any(|word| date_word(word))
328  } else {
329    matches!(first.as_str(), "msc" | "pacs") && !codes.is_empty()
330  };
331  label.then_some(first.as_str())
332}

335/// a year (1800-2100)? A month of another language is none (`mars 2021`, counted; the RED repro
336/// `net_non_english_month_line_red`).
337fn date_word(word: &str) -> bool {
338  const MONTHS: &[&str] = &[
339    "january",
340    "february",
341    "march",
342    "april",
343    "may",
344    "june",
345    "july",
346    "august",
347    "september",
348    "october",
349    "november",
350    "december",
351    "jan",
352    "feb",
353    "mar",
354    "apr",
355    "jun",
356    "jul",
357    "aug",
358    "sep",
359    "sept",
360    "oct",
361    "nov",
362    "dec",
363  ];
364  MONTHS.contains(&word) || (word.len() == 4 && matches!(word.parse::<u32>(), Ok(1800..=2100)))
365}

367/// The unclaimed-text net's notes (frontmatter_stores.rs `hand_net_stores`: what a raw class's title page prints of a
368/// store no frontmatter owner claims, OXIDIZED_DESIGN_DIVERGENCES #953), once the frontmatter is placed: each note line
369/// (split at `ltx:break` and `ltx:p`, [`net_note_lines`]) is read by its distinct content words
370/// ([`net_content_words`]) that neither the frontmatter nor the text set before it holds — the whole document's text but
371/// the net's notes: the deposit's paragraphs, the creators (`\AB@names` repeats the names), the abstract — nor the
372/// class's own code in the store (`net_class_text`). A line of e-mail addresses alone ([`resolver::only_addresses`]) is
373/// the e-mail contacts of the note's creator — the block's nameless one or the sole author (OD #159), no invented owner
374/// — each address no text kept holds, whether or not its words are used elsewhere (aaskaiid.sty's `\email@toks`,
375/// 2606.27706, 2607.25721). Else a line with no such word goes (it repeats what is kept); a line is kept whole with 3 or
376/// more, or with fewer when they are all a code's or a date's ([`code_or_date_word`]) or the label such a line leads
377/// with ([`leading_label`]); otherwise it goes, its words counted. A note left without a line goes, with a nameless
378/// creator it leaves empty. A store's kept lines raise one Warning, and its lost words one more, in printed order
379/// (FIDELITY_CHARTER §4: printed text no owner claims, kept beside the frontmatter or lost, counted — 2103.14934's
380/// "corresponding authors" below the floor; 105a).
381fn keep_unclaimed_net_notes(document: &mut Document) -> Result<()> {
382  let notes = document.findnodes("//ltx:note[@_net]", None);
383  if notes.is_empty() {
384    return Ok(());
385  }
386  let Some(root) = document.get_document().get_root_element() else {
387    return Ok(());
388  };
389  let mut claimed_text = String::new();
390  text_outside(&root, &notes, &mut claimed_text);
391  let claimed: HashSet<String> = net_content_words(&claimed_text).into_iter().collect();
392  let mut written: Vec<String> = resolver::addresses_in(&claimed_text);
393  let claimed_folded = resolver::base_letters(&claimed_text).to_lowercase();
394  let email_name = get_frontmatter_name(None, "ltx:contact", "email")?;
395  for mut note in notes {
396    let store = note.get_attribute("_net").unwrap_or_default();
397    // (and the words of the class's own code in the store, frontmatter_stores.rs `net_class_text`, as a multiset: each
398    // occurrence in the class's text accounts for one in the line, so a label's `单位` does not take the document's
399    // own `单位` with it — each Han ideograph is a word, `作者单位：#1` around `单位甲`; review rev105a3 finding 1)
400    let mut class: HashMap<String, usize> = HashMap::default();
401    for word in net_content_words(&lookup_string(&s!("lx_net_class_text_{store}"))) {
402      *class.entry(word).or_default() += 1;
403    }
404    let mut lines = Vec::new();
405    net_note_lines(&note, &mut lines);
406    let mut kept = 0usize;
407    let mut lost: Vec<String> = Vec::new();
408    let mut addresses: Vec<String> = Vec::new();
409    for (nodes, text) in lines {
410      let words = net_content_words(&text);
411      // (distinct, in printed order)
412      let mut unmatched: Vec<String> = Vec::new();
413      let mut class_left = class.clone();
414      for word in &words {
415        if claimed.contains(word) {
416          continue;
417        }
418        if let Some(left) = class_left.get_mut(word)
419          && *left > 0
420        {
421          *left -= 1;
422          continue;
423        }
424        if !unmatched.contains(word) {
425          unmatched.push(word.clone());
426        }
427      }
428      let listed = resolver::addresses_in(&text);
429      // (its codes no text kept prints: an MSC code's letters and digits are no content words, `20N05`)
430      let codes: Vec<String> = code_pieces(&text)
431        .into_iter()
432        .filter(|code| !claimed_folded.contains(code.as_str()))
433        .collect();
434      let label = leading_label(&words, &codes);
435      // (an address line: its addresses no kept text holds, whether or not their words are used elsewhere — a title's
436      // word in the domain; one every address of which is written repeats what is kept)
437      if !listed.is_empty() && resolver::only_addresses(&text) {
438        for address in listed {
439          if !written.contains(&address) {
440            written.push(address.clone());
441            addresses.push(address);
442          }
443        }
444      } else if unmatched.len() >= 3
445        || ((!unmatched.is_empty() || !codes.is_empty())
446          && unmatched
447            .iter()
448            .all(|word| Some(word.as_str()) == label || code_or_date_word(word, &codes)))
449      {
450        kept += 1;
451        continue;
452      } else {
453        for word in unmatched {
454          if !lost.contains(&word) {
455            lost.push(word);
456          }
457        }
458      }
459      for node in nodes {
460        document.remove_node(node);
461      }
462    }
463    // (the breaks left at either end of a kept line's run, and the paragraphs left empty)
464    for p in document.findnodes(".//ltx:p", Some(&note)) {
465      if element_nodes(&p).is_empty() && p.get_content().trim().is_empty() {
466        document.remove_node(p);
467      }
468    }
469    let creator = note.get_parent();
470    // (the addresses, the note's creator's e-mail contacts where the note stands)
471    if let Some(creator) = creator.as_ref() {
472      let saved = document.get_node().clone();
473      document.set_node(creator);
474      for address in &addresses {
475        let mut attributes = HashMap::default();
476        attributes.insert("role".to_string(), "email".to_string());
477        if let Some(name) = email_name.as_ref() {
478          attributes.insert("name".to_string(), name.clone());
479        }
480        let mut contact = document.open_element("ltx:contact", Some(attributes), None)?;
481        contact.append_text(address).ok();
482        document.close_element("ltx:contact")?;
483        contact.unlink_node();
484        note.add_prev_sibling(&mut contact).ok();
485      }
486      document.set_node(&saved);
487    }
488    if !lost.is_empty() {
489      // (named once: the title-page check counts no word named here again, `title_page_check`)
490      note_net_counted(&lost);
491      Warn!(
492        "frontmatter",
493        "unclaimed",
494        s!(
495          "The title page prints \\{store} with {} word(s) no frontmatter field holds, too few to keep as a note: {}",
496          lost.len(),
497          lost.join(" ")
498        )
499      );
500    }
501    if kept == 0 {
502      document.remove_node(note);
503      if addresses.is_empty()
504        && let Some(creator) = creator
505        && creator.get_name() == "creator"
506        && element_nodes(&creator).is_empty()
507      {
508        document.remove_node(creator);
509      }
510      continue;
511    }
512    trim_edge_breaks(document, &note);
513    note.remove_attribute("_net").ok();
514    Warn!(
515      "frontmatter",
516      "unclaimed",
517      s!(
518        "The title page prints {kept} line(s) of \\{store} that no frontmatter field holds: kept as a note of the author block"
519      )
520    );
521  }
522  Ok(())
523}

625/// The mark a legend contact prints ([`kind_legend_contacts`]; empty for none): the superscript its text opens with,
626/// else a key of a label it was placed by that a printed mark gave, that a carrier's name shows or requests, or that its
627/// binding prints ([`LABEL_PRINTS`]) — `named_creators` the author creators, a nameless creator's carriers. Read by the
628/// Kind step for a dedication too ([`kind_dedication_lines`]: a line a name's mark points to is no dedication).
629fn legend_mark(document: &mut Document, contact: &Node, named_creators: &[Node]) -> String {
630  // (the superscript its text opens with, as printed)
631  let printed = element_nodes(contact)
632    .first()
633    .filter(|first| {
634      first.get_name() == "sup"
635        && contact
636          .get_first_child()
637          .is_some_and(|child| child == **first || child.get_content().trim().is_empty())
638    })
639    .map(|sup| sup.get_content().trim().to_string())
640    .filter(|mark| !mark.is_empty());
641  // (else a key of a label it was placed by that a printed mark gave or that prints, whole — `*†` of 2602.19190's
642  // `$^{*\dag}$Project lead: Yi Yang (…)`; review rev103c2 B-A): an `affiliation:K` label read off a printed
643  // superscript (the marks path, the Cut, the observer's writer: 2609.03342, 2503.10710) but not one of the entry's
644  // place (`SEQUENCE_LABEL`: svjour's `\institute{… \and …}`, 1507.04774), a key a carrier's name shows or requests
645  // by its superscript (IEEEtran's `\IEEEauthorrefmark{4}` as `Tayal<sup>4</sup>`, 2410.12184), or a label its
646  // binding prints before the line (`LABEL_PRINTS`: authblk's `\affil[*]`, jheppub's `\affiliation[*]`, 2204.11919;
647  // cas's letter, 2303.06763) — the carrier's own among several (`\affil[1,2]`: Ann's "1", Bob's "2", 2301.13111).
648  // A binding's key prints no mark (icml's `\icmlaffiliation{intern}{…}`, whose number icml prints: 2106.09776): no
649  // mark, so it stays an affiliation.
650  printed
651    .or_else(|| {
652      // (each key, and whether a printed mark gave it: an `affiliation:K` label not of the entry's place)
653      let place = contact.get_attribute(SEQUENCE_LABEL).unwrap_or_default();
654      let mut keys: Vec<(String, bool)> =
655        clean_frontmatter_labels(&contact.get_attribute("_label")?, "")
656          .into_iter()
657          .filter_map(|label| match label.split_once(':') {
658            Some(("affiliation", key)) => Some((label_key(key), label != place)),
659            None => Some((label_key(&label), false)),
660            _ => None,
661          })
662          .filter(|(key, _)| !key.is_empty())
663          .collect();
664      let creator = contact.get_parent()?;
665      let printed_label = contact.get_attribute(PRINTED_LABEL);
666      let prints = printed_label.is_some() || contact.get_attribute(LABEL_PRINTS).is_some();
667      // (an entry the frontmatter emitter wrote is placed by the reading's label, so the label that prints is its own
668      // key, `PRINTED_LABEL`: the call's `label=` its binding prints (authblk's `\affil[*]`, 2605.31330), or the first
669      // mark the entry's content printed (`\thanks{\textsuperscript{1}These authors contributed equally …}`); OD #980,
670      // P2 NT)
671      if let Some(printed) = printed_label {
672        keys.splice(
673          0..0,
674          clean_frontmatter_labels(&printed, "")
675            .into_iter()
676            .map(|label| label_key(label.split_once(':').map_or(label.as_str(), |(_, key)| key)))
677            .filter(|key| !key.is_empty())
678            .map(|key| (key, true)),
679        );
680      }
681      let named = element_nodes(&creator)
682        .iter()
683        .any(|child| child.get_name() == "personname");
684      let holders = if named {
685        std::slice::from_ref(&creator)
686      } else {
687        named_creators
688      };
689      // (the marks the carriers' names show, and those they request by a printed superscript — `affiliation:K`, read
690      // as the name prints it, `\lx@sup@request@affiliation`; a binding's `annotations={…}` may be keys)
691      let mut shown: Vec<String> = Vec::new();
692      // (an author reference mark's number, as its label keys it, and the symbol it prints: `relocate::sup_marks`;
693      // IEEEtran's `\IEEEauthorrefmark{4}` keys 4 and prints `§`, the note's mark, 2410.12184)
694      let mut refmarks: Vec<(String, String)> = Vec::new();
695      for holder in holders {
696        // (each of a superscript's marks: `Bridgeford<sup>1,†</sup>`, 2503.10710)
697        for sup in document.findnodes("ltx:personname//ltx:sup", Some(holder)) {
698          shown.extend(sup.get_content().split(',').map(mark_form));
699          if let Some(key) = sup.get_attribute("_refmark") {
700            shown.push(mark_form(&key));
701            refmarks.push((mark_form(&key), sup.get_content().trim().to_string()));
702          }
703        }
704        let requests = holder.get_attribute("_annotations").unwrap_or_default();
705        shown.extend(
706          requests
707            .split(',')
708            .filter_map(|label| label.strip_prefix("affiliation:"))
709            .map(mark_form),
710        );
711      }
712      // (the creator's own requests, which pick its mark among several)
713      let requested: Vec<String> = creator
714        .get_attribute("_annotations")
715        .unwrap_or_default()
716        .split(',')
717        .map(|label| label_key(label.split_once(':').map_or(label, |(_, key)| key)))
718        .collect();
719      let marks: Vec<&String> = keys
720        .iter()
721        .filter(|(key, marked)| *marked || prints || shown.contains(&mark_form(key)))
722        .map(|(key, _)| key)
723        .collect();
724      marks
725        .iter()
726        .find(|key| requested.contains(key))
727        .or(marks.first())
728        .map(|key| {
729          refmarks
730            .iter()
731            .find(|(number, printed)| *number == mark_form(key) && !printed.is_empty())
732            .map_or_else(|| (*key).clone(), |(_, printed)| printed.clone())
733        })
734    })
735    .unwrap_or_default()
736}

761  let opening: Vec<(Node, Vec<EntryLine>)> = contacts
762    .into_iter()
763    .map(|contact| {
764      let lines = entry_lines(&contact);

766    })
767    .filter(|(_, lines)| {
768      lines
769        .iter()
770        .any(|(nodes, _)| opens_a_dedication(&line_text(nodes)))

793      (0..lines.len()).find(|&at| opens_a_dedication(&line_text(&lines[at].0)) && !requested(at))
794    else {
795      continue;
796    };

799      .unwrap_or(lines.len());

803      if at > from {
804        dedication.extend(before.clone());
805      }
806      dedication.extend(nodes.iter().cloned());
807    }
808    // (its lines for the note: one break between two, none at its edges — `\centerline{…}\\ \centerline{…}` printed

811      node.get_type() != Some(NodeType::ElementNode) && node.get_content().trim().is_empty()

813    let is_break =
814      |node: &Node| node.get_type() == Some(NodeType::ElementNode) && node.get_name() == "break";
815    let mut inner: Vec<Node> = Vec::new();
816    let mut after_break = true;
817    for node in &dedication {
818      if is_break(node) {
819        if after_break {
820          continue;
821        }
822        after_break = true;
823      } else if !blank(node) {
824        after_break = false;
825      }
826      inner.push(node.clone());
827    }
828    let text = line_text(&inner)
829      .split_whitespace()
830      .collect::<Vec<_>>()
831      .join(" ");
832    DebugFeature!(
833      "frontmatter",
834      "FRONT kind affiliation: a dedication: {text}"
835    );

839        continue;
840      };

843      };

849      document.close_element("ltx:pubnote")?;

852      note.unlink_node();
853      let mut last = last;
854      last.add_next_sibling(&mut note)?;

856    }
857    let mut parents: Vec<Node> = Vec::new();

860      document.remove_node(before.clone());
861    }
862    for node in dedication {

865    }

869      && (last.get_name() == "break"
870        || (last.get_content().trim().is_empty() && element_nodes(&last).is_empty()))
871    {
872      document.remove_node(last);
873    }
874    trim_edge_breaks(document, &contact);
875  }
876  drop_empty_contacts(document)
877}

1138/// Removes from `parent` everything printed before `link` (the label [`link_label`] read): each node before it, and
1139/// before it inside each wrapper holding it.

1146    }

1180/// An affiliation contact's link lines (each of its children between two line breaks one link printing as a URL does,
1181/// [`prints_as_url`], behind a link's label or none, [`is_link_label`], and nothing else), parted off: those ending it
1182/// after a line of other text (`\IEEEauthorblockA{\textsuperscript{2}Univ B\\ Project website: {\href{…}{…}}}`: 107c),
1183/// or, where every line is one and there are two or more, each but the first (an author's own block of links,
1184/// `\url{https://orcid.org/…}\\ \url{https://members.loria.fr/AClement}`, 2206.10577: 107c) — a copy of the contact
1185/// per line, holding it, after the contact in print order. Returns the copies made.
1186fn split_link_lines(document: &mut Document, contact: &Node) -> Result<Vec<Node>> {
1187  let mut lines: Vec<Vec<Node>> = vec![Vec::new()];
1188  for child in contact.get_child_nodes() {
1189    if child.get_type() == Some(NodeType::ElementNode) && child.get_name() == "break" {
1190      lines.push(Vec::new());
1191    } else if (child.get_type() != Some(NodeType::TextNode)
1192      || !child.get_content().trim().is_empty())
1193      && let Some(line) = lines.last_mut()
1194    {
1195      line.push(child);
1196    }
1197  }
1198  lines.retain(|line| !line.is_empty());
1199  let link_line = |document: &mut Document, line: &[Node]| -> bool {
1200    let mut refs: Vec<Node> = Vec::new();
1201    for node in line {
1202      if node.get_type() == Some(NodeType::ElementNode) && node.get_name() == "ref" {
1203        refs.push(node.clone());
1204      } else if node.get_type() == Some(NodeType::ElementNode) {
1205        refs.extend(document.findnodes(".//ltx:ref[@href]", Some(node)));
1206      }
1207    }
1208    let [link] = refs.as_slice() else {
1209      return false;
1210    };
1211    let text: String = line.iter().map(Node::get_content).collect();
1212    printed_link_label(&text, link).is_some_and(|label| label.is_empty() || is_link_label(&label))
1213  };
1214  let links: Vec<bool> = lines.iter().map(|line| link_line(document, line)).collect();
1215  let first = links.iter().rposition(|link| !link).map_or(1, |at| at + 1);
1216  if lines.len() < 2 || first >= lines.len() {
1217    return Ok(Vec::new());
1218  }
1219  let Some(mut creator) = contact.get_parent() else {
1220    return Ok(Vec::new());
1221  };
1222  let mut copies: Vec<Node> = Vec::new();
1223  let mut previous = contact.clone();
1224  for keep in first..lines.len() {
1225    document.append_clone(&mut creator, vec![contact.clone()])?;
1226    let Some(mut copy) = element_nodes(&creator).last().cloned() else {
1227      continue;
1228    };
1229    copy.unlink_node();
1230    previous.add_next_sibling(&mut copy)?;
1231    keep_only_lines(document, &copy, keep..keep + 1);
1232    previous = copy.clone();
1233    copies.push(copy);
1234  }
1235  keep_only_lines(document, contact, 0..first);
1236  Ok(copies)
1237}

1300/// Whether a contact's text ([`legend_text`]) is a membership grade in a society, its marks aside: "Senior Member,
1301/// IEEE", "Graduate Student Member, IEEE", "Life Fellow, IEEE", "Member, ACM" — grade words, then "Member" or "Fellow",
1302/// a comma and a society whose grades follow a name (`NAME_GRADE_SOCIETIES`, the names reader's: IEEE, ACM, OSA, ASCE,
1303/// …) — one per line ("Graduate Student Member, IEEE | Senior Member, IEEE"). Not "Member of Technical Staff, Bell
1304/// Labs", "Fellow, Trinity College", nor a position at an institute ("Senior Fellow, IAS": review rev106b S3).
1305fn reads_as_membership_grade(text: &str) -> bool {
1306  const GRADES: &[&str] = &[
1307    "life",
1308    "senior",
1309    "graduate",
1310    "student",
1311    "associate",
1312    "honorary",
1313    "distinguished",
1314    "affiliate",
1315  ];
1316  let grade = |line: &str| {
1317    let Some((grade, society)) = line.split_once(',') else {
1318      return false;
1319    };
1320    let words: Vec<&str> = grade
1321      .split(|c: char| !c.is_alphanumeric())
1322      .filter(|w| !w.is_empty())
1323      .collect();
1324    let society: Vec<&str> = society
1325      .split(|c: char| !c.is_alphanumeric())
1326      .filter(|w| !w.is_empty())
1327      .collect();
1328    matches!(words.split_last(), Some((last, before))
1329      if matches!(last.to_lowercase().as_str(), "member" | "fellow")
1330        && before.iter().all(|w| GRADES.contains(&w.to_lowercase().as_str())))
1331      && matches!(society.as_slice(), [society] if NAME_GRADE_SOCIETIES.contains(society))
1332  };
1333  let lines: Vec<&str> = text
1334    .split('|')
1335    .map(str::trim)
1336    .filter(|line| line.chars().any(char::is_alphanumeric))
1337    .collect();

1339}

2863/// Whether an entry's line `nodes` is a list of local parts set over several lines: a line break, nested or not, between
2864/// a printed brace and the `@` after the brace that closes it (`{a,b,` / `c}@x`; `{a,b}` / `@x`).
2865fn split_address_list(nodes: &[Node]) -> bool {
2866  // (the line's text, each line break in it a line separator)
2867  fn printed(node: &Node, text: &mut String) {
2868    match node.get_type() {
2869      Some(NodeType::ElementNode) if node.get_name() == "break" => text.push('\u{2028}'),
2870      Some(NodeType::ElementNode) => {
2871        for child in node.get_child_nodes() {
2872          printed(&child, text);
2873        }
2874      },
2875      _ => text.push_str(&node.get_content()),
2876    }
2877  }
2878  let mut text = String::new();
2879  for node in nodes {
2880    printed(node, &mut text);
2881  }
2882  let Some(open) = text.find('{') else {
2883    return false;
2884  };
2885  let mut depth = 0isize;
2886  for (at, c) in text[open..].char_indices() {
2887    match c {
2888      '{' => depth += 1,
2889      '}' => {
2890        depth -= 1;
2891        if depth == 0 {
2892          let rest = &text[open + at + 1..];
2893          let after = rest.trim_start_matches(|c: char| c.is_whitespace());
2894          return opens_with_domain(after)
2895            && text[open..open + at + 1 + (rest.len() - after.len())].contains('\u{2028}');
2896        }
2897      },
2898      _ => {},
2899    }
2900  }
2901  false
2902}

3589/// those after the address and before the next. Each address's place is its local part's in `text`, read on from the

3605    .first()

3609    .filter(|&i| Some(marks[i].0) == opening || marks[i].0 >= cut.start)

3613    .zip(places.first())
3614    .is_some_and(|(&i, &first)| marks[i].0 < first);

3927/// Remove author creators whose `<ltx:personname>` is empty, moving any contacts they
3928/// carry to the preceding real creator (else the first following one).
3929///
3930/// A flat comma author list with interspersed `\IEEEmembership{…}`/`\thanks{…}` — e.g.
3931/// `\author{Alice, \IEEEmembership{…}, and Bob, …\thanks{…}}` (html_feedback#4539,
3932/// witness 2508.00603) — comma/" and "-splits into pieces where the membership pieces
3933/// digest to nothing, surfacing as empty `<ltx:personname/>` creators; a trailing
3934/// `\thanks` then strands its affil/email on a nameless creator. This coalesces those:
3935/// contactless empties are dropped; a contact-bearing empty's contacts move to the
3936/// preceding real author. `\footnotemark`-note markers keep a personname non-empty
3937/// (2507.06670 "Yu Zhang"), so real authors are untouched.
3938///
3939/// Moves every annotation, `<ltx:contact>`, `<ltx:note>` and `<ltx:rdf>`: an author `\thanks` is a
3940/// marked `<ltx:note role="thanks">` (OXIDIZED_DESIGN #156), so a trailing `\thanks` on
3941/// a nameless comma-split creator would otherwise be dropped with the empty creator —
3942/// witness 1510.02728 (`\author{Sani,~\IEEEmembership{…} Vosoughi,~\IEEEmembership{…}%
3943/// \thanks{…NSF…}}`), where the note must land on the last real author, as a contact did;
3944/// and an author's metadata (OXIDIZED_DESIGN #479: an AASTeX 7 `\email` after an `\author` whose name prints nothing).
3945///
3946/// A coalesce never drops a printed word: when no named author is left to take a nameless creator's annotations (a
3947/// block of affiliation lines alone, IEEEtran's `\authorblockA{…}` with no `\authorblockN`, 1804.00021; a name
3948/// written inside `\IEEEmembership{…}`, 2203.14581), the first nameless creator that carries any stays with every one
3949/// of them, the block's own creator — Perl's `<ltx:personname/>` over the lines (OXIDIZED_DESIGN_DIVERGENCES #159's
3950/// block owner, #949). Any other printing child of a removed creator is reported (`Warning:unexpected:frontmatter`), never
3951/// silently lost.
3952fn coalesce_empty_creators(document: &mut Document) -> Result<()> {
3953  let creators = document.findnodes(NAMED_ROW_AUTHORS, None);
3954  let mut to_remove: Vec<Node> = Vec::new();
3955  let mut last_real: Option<Node> = None;
3956  // Annotations (contacts + notes) of leading empties (before any real author), held
3957  // until the first real one; and the first of those empties that carries any, which keeps them all when no real
3958  // one follows.
3959  let mut orphan_annotations: Vec<Node> = Vec::new();
3960  let mut orphan_holder: Option<Node> = None;
3961  for creator in creators {
3962    if creator_personname_empty(document, &creator) {
3963      let annotations = creator_annotations(&creator);
3964      if let Some(ref mut prev) = last_real {
3965        if !annotations.is_empty() {
3966          document.append_clone(prev, annotations)?;
3967        }
3968        to_remove.push(creator);
3969      } else if orphan_holder.is_none() && !annotations.is_empty() {
3970        // (kept for now: the holder of the leading empties' annotations, if no real author takes them)
3971        orphan_holder = Some(creator);
3972      } else {
3973        orphan_annotations.extend(annotations);
3974        to_remove.push(creator);
3975      }
3976    } else {
3977      if let Some(holder) = orphan_holder.take() {
3978        // (the holder's own annotations first, in the order they stood)
3979        let held = creator_annotations(&holder);
3980        orphan_annotations.splice(0..0, held);
3981        to_remove.push(holder);
3982      }
3983      if !orphan_annotations.is_empty() {
3984        let mut first = creator.clone();
3985        document.append_clone(&mut first, std::mem::take(&mut orphan_annotations))?;
3986      }
3987      last_real = Some(creator);
3988    }
3989  }
3990  // (no named author took the leading empties' annotations: the holder keeps them all, the block's creator)
3991  if let Some(mut holder) = orphan_holder
3992    && !orphan_annotations.is_empty()
3993  {
3994    document.append_clone(&mut holder, orphan_annotations)?;
3995  }
3996  for creator in to_remove {
3997    // (what else a removed creator prints, no annotation moved with it, is reported: a coalesce never drops a word
3998    // silently)
3999    for child in creator.get_child_nodes() {
4000      if child.get_type() != Some(NodeType::ElementNode) {
4001        continue;
4002      }
4003      let qname = document::get_node_qname(&child);
4004      if with(qname, |q| {
4005        !matches!(q, "ltx:personname" | "ltx:contact" | "ltx:note" | "ltx:rdf")
4006      }) && !child.get_content().trim().is_empty()
4007      {
4008        Warn!(
4009          "unexpected",
4010          "frontmatter",
4011          s!(
4012            "A nameless author's {} was dropped with it: {}",
4013            with(qname, |q| q.to_string()),
4014            child.get_content().trim()
4015          )
4016        );
4017      }
4018    }
4019    document.remove_node(creator);
4020  }

4038/// Redistribute a mis-piled contact block back to its authors (surpass-Perl).
4039///
4040/// The amsart idiom `\author{A}\author{B}\author{C}` followed by paired
4041/// `\address{}\email{}` declares every author up front, then all the contacts.
4042/// LaTeXML's default "attach a contact to the preceding creator" then bunches
4043/// EVERY address+email under the LAST author (arXiv:2308.06214v1,
4044/// arXiv/html_feedback#46); Perl 0.8.8 does the same (SHARED limitation).
4045///
4046/// Fix ONLY the clean, unambiguous pile — an `N × m` grid: the other `N-1`
4047/// authors carry no contact, the last author's `K` contacts split evenly
4048/// (`K = N·m`) into a role-periodic sequence (`role[i] == role[i+m]`), so group
4049/// `j` is handed to author `j`. Any irregular pile — a heterogeneous role
4050/// sequence, per-author counts that differ, or contacts already spread across
4051/// authors (the interleaved idiom, which is correct as-is) — fails the gate and
4052/// is left EXACTLY as Perl attached it. Mirrors the "distribute-when-clean, else
4053/// keep prior" rule of the shared-email splitter (OXIDIZED_DESIGN #52(j)).
4054fn distribute_upfront_contacts(document: &mut Document) -> Result<()> {
4055  let creators = document.findnodes(NAMED_ROW_AUTHORS, None);
4056  let n = creators.len();
4057  if n < 2 {
4058    return Ok(());
4059  }
4060  // (a call made after `\maketitle` placed the frontmatter ([`CONTACT_LATE`]) restating a contact the creator holds
4061  // from an earlier call — a supplement's restatement — is no contact of the pile: llncs's restated `\institute` put a
4062  // second `\email{libraboli@…}` beside the first on Bo Li, the last author, and the two read as a grid of one each,
4063  // handing the first to Lv Tang, who prints none (2009.10916: census138 EMAILSPELL, brief 107f);
4064  // `drop_repeated_contacts` drops the copy. Late calls of their own still are: 1303.3079's `\email`s in a footnote
4065  // after `\maketitle`)
4066  let printed = |x: &Node| {
4067    (
4068      x.get_attribute("role"),
4069      x.get_content()
4070        .split_whitespace()
4071        .collect::<Vec<_>>()
4072        .join(" "),
4073    )
4074  };
4075  let contacts_of = |c: &Node| -> Vec<Node> {
4076    let all: Vec<Node> = c
4077      .get_child_nodes()
4078      .into_iter()
4079      .filter(|x| {
4080        x.get_type() == Some(NodeType::ElementNode)
4081          && with(document::get_node_qname(x), |q| q == "ltx:contact")
4082      })
4083      .collect();
4084    all
4085      .iter()
4086      .filter(|x| {
4087        !x.has_attribute(CONTACT_LATE)
4088          || !all
4089            .iter()
4090            .any(|y| !y.has_attribute(CONTACT_LATE) && printed(y) == printed(x))
4091      })
4092      .cloned()
4093      .collect()
4094  };
4095  // Signature: only the LAST author carries any contacts.
4096  for creator in &creators[..n - 1] {
4097    if !contacts_of(creator).is_empty() {
4098      return Ok(());
4099    }
4100  }
4101  // …and no other author is to receive one of their kind: an author whose mark requests a label a pending contact of a
4102  // role the last author's contacts hold carries gets it from `relocate_annotations`, after this — the last author's
4103  // contacts are then the run's, not a pile (`\author{Ann Able$^{1}$}\author{Bob Baker}\affil{$^1$Univ X}\affil{Univ Y}
4104  // \affil{Univ W}` handed Univ Y to Ann and kept Univ W alone on Bob; 77n r5 review probes g01-g06, m11). A pending
4105  // contact of another role leaves the pile one (2605.30274's e-mail lines on the last of two authors whose affiliations
4106  // are mark-picked).
4107  let last_roles: HashSet<String> = contacts_of(&creators[n - 1])
4108    .iter()
4109    .map(|contact| contact.get_attribute("role").unwrap_or_default())
4110    .collect();

4112  for pending in document.findnodes(".//*[@role='pending']", None) {
4113    for note in element_nodes(&pending) {
4114      if !last_roles.contains(&note.get_attribute("role").unwrap_or_default()) {
4115        continue;
4116      }
4117      let labels = [
4118        note.get_attribute("_label").unwrap_or_default(),
4119        pending.get_attribute("_annotations").unwrap_or_default(),
4120      ];
4121      for label in labels.iter().flat_map(|labels| labels.split(',')) {
4122        if !label.is_empty() {
4123          pending_labels.insert(label.to_string());
4124        }
4125      }
4126    }
4127  }
4128  let answered_later = |creator: &Node| {
4129    creator
4130      .get_attribute("_annotations")
4131      .unwrap_or_default()
4132      .split(',')
4133      .any(|label| pending_labels.contains(label))
4134  };
4135  if creators[..n - 1].iter().any(answered_later) {
4136    return Ok(());
4137  }

4165  }

4356/// Takes `contact`'s `name` out of the front of its text, once, with the spaces after it ([`NAME_PRINTED`]): the text
4357/// nodes it spans, each emptied one and each wrapper it leaves empty removed. Nothing where the text does not open with
4358/// the name, its spaces aside, or where an element printing no text (a break, a formula) comes before its end.
4359fn drop_printed_name(document: &mut Document, contact: &Node) {
4360  let Some(name) = contact.get_attribute("name") else {
4361    return;
4362  };
4363  let label: Vec<char> = name.chars().filter(|c| !c.is_whitespace()).collect();
4364  if label.is_empty() {
4365    return;
4366  }
4367  fn texts(node: &Node, out: &mut Vec<Option<Node>>) {
4368    for child in node.get_child_nodes() {
4369      match child.get_type() {
4370        Some(NodeType::TextNode) => out.push(Some(child)),
4371        Some(NodeType::ElementNode) if child.get_name() != "Math" => texts(&child, out),
4372        Some(NodeType::ElementNode) => out.push(None),
4373        _ => {},
4374      }
4375    }
4376  }
4377  let mut nodes = Vec::new();
4378  texts(contact, &mut nodes);
4379  let mut matched = 0;
4380  let mut cuts: Vec<(Node, usize)> = Vec::new();
4381  'nodes: for node in nodes {
4382    // (an element printing no text before the name's end: no name to take)
4383    let Some(node) = node else {
4384      if matched < label.len() {
4385        return;
4386      }
4387      break;
4388    };
4389    let content = node.get_content();
4390    let mut cut = 0;
4391    for (at, c) in content.char_indices() {
4392      if c.is_whitespace() {
4393        cut = at + c.len_utf8();
4394      } else if matched == label.len() {
4395        cuts.push((node, cut));
4396        break 'nodes;
4397      } else if c == label[matched] {
4398        matched += 1;
4399        cut = at + c.len_utf8();
4400      } else {
4401        return;
4402      }
4403    }
4404    cuts.push((node, cut));
4405  }
4406  if matched < label.len() {
4407    return;
4408  }
4409  let mut parents = Vec::new();
4410  for (mut node, cut) in cuts {
4411    let content = node.get_content();
4412    if cut >= content.len() {
4413      if let Some(parent) = node.get_parent() {
4414        parents.push(parent);
4415      }
4416      document.remove_node(node);
4417    } else if cut > 0 {
4418      let _ = node.set_content(&content[cut..]);
4419    }
4420  }
4421  drop_emptied_wrappers(document, parents);
4422}

4795  // (its characters, each combining mark part of the one before it: `\={x}`'s `x̄` is a letter — review rev92b_r2 B1)
4796  let bases: Vec<char> = mark.chars().filter(|c| !is_combining_mark(*c)).collect();
4797  let marked = !mark.starts_with(is_combining_mark);
4798  if marked && !bases.is_empty() && bases.iter().all(|c| c.is_alphanumeric()) {
4799    return Some(mark.to_string());
4800  }
4801  let glyph = match mark {
4802    "\\dag" | "\\dagger" => "\u{2020}",
4803    "\\ddag" | "\\ddagger" => "\u{2021}",
4804    // (pdflatex sets a math `*` as `∗`; one requested in text stays `*`, `show_unanswered_marks`)
4805    "\\ast" | "*" => "\u{2217}",
4806    "\\star" => "\u{22C6}",
4807    "\\S" => "\u{A7}",
4808    "\\P" => "\u{B6}",
4809    "\\bullet" => "\u{2219}",
4810    "\\sharp" => "\u{266F}",
4811    "\\|" => "\u{2016}",
4812    "\\#" | "#" => "#",
4813    "\u{2020}" | "\u{2021}" | "\u{2217}" | "\u{A7}" | "\u{B6}" => mark,
4814    // (one printed symbol, `queue_digest::is_printed_symbol` — no letter, digit, space, ASCII, format character or
4815    // combining mark: `\up{\!1,2}`'s zero-width space is none, 2209.14855 —, with the marks on it; a run of several is
4816    // a whole-run label, its symbols requested apart, `parted_printed_label`)
4817    _ if marked && matches!(bases.as_slice(), [c] if is_printed_symbol(*c)) => mark,
4818    _ => return None,
4819  };
4820  Some(glyph.to_string())
4821}
