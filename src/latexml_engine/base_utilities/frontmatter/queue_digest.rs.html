10/// Digest the content for a frontmatter item, disabling or masking certain commands.
11/// Note that we shouldn't be digesting any frontmatter until within document, so inPreamble unnec.
12/// See clean_trailing_break for cleanup of misused \\.
13/// Perl: digestFrontmatterItem($stomach, $tag, $item).
14pub(in crate::base_utilities) fn digest_frontmatter_item(
15  tag: &str,
16  item: Tokens,
17) -> Result<Digested> {
18  bgroup();
19  // the field being made, for the commands a class's own setter of it defines for its argument
20  // (frontmatter_stores.rs `refused_environment_definition`: vmsta.cls's `\author` and its `\fnm`)
21  assign_value("lx_frontmatter_item", tag.to_string(), Some(Scope::Local));
22  // the counter step in scope as the item starts, which a `\label` naming a step of its own does not name
23  // (`\lx@set@frontmatter@label`)
24  assign_value(
25    ITEM_REFSTEP,
26    Stored::Int(refstep_in_scope().unwrap_or(0)),
27    Some(Scope::Local),
28  );
29  let_i(
30    &T_CS!("\\label"),
31    &T_CS!("\\lx@set@frontmatter@label"),
32    None,
33  );
34  let_i(
35    &T_CS!("\\footnote"),
36    &(if tag == "ltx:creator" {
37      T_CS!("\\lx@add@note")
38    } else {
39      T_CS!("\\lx@add@pubnote")
40    }),
41    None,
42  );
43  let_i(
44    &T_CS!("\\thanks"),
45    &(if tag == "ltx:creator" {
46      T_CS!("\\lx@add@thanks")
47    } else {
48      T_CS!("\\lx@add@pubnote@thanks")
49    }),
50    None,
51  );
52  // `\fnmsep` sets the comma between two footnote marks (`U.~Hopp\inst{1}\fnmsep\thanks{…}`, aa/llncs; webofc's
53  // `\fnsep`, webofc.cls:217); before a mark that becomes a link or a note here it would stand alone in the name
54  // ("U. Hopp,": astro-ph0001054, astro-ph0611016; "Andrea Shindler,": 2301.07438), so it sets nothing there; between
55  // two marks that stay visible it is still their comma.
56  // (once: an item digested within another — a `\thanks` in an author — would save the wrapper as itself, and a
57  // `\fnmsep` there not before a mark would call itself without end)
58  if !lookup_bool("lx_frontmatter_fnmsep") {
59    for (sep, saved, wrapper) in [
60      ("\\fnmsep", "\\lx@saved@fnmsep", "\\lx@frontmatter@fnmsep"),
61      ("\\fnsep", "\\lx@saved@fnsep", "\\lx@frontmatter@fnsep"),
62    ] {
63      if lookup_definition(&T_CS!(sep))?.is_some() {
64        let_i(&T_CS!(saved), &T_CS!(sep), None);
65        let_i(&T_CS!(sep), &T_CS!(wrapper), None);
66      }
67    }
68    assign_value("lx_frontmatter_fnmsep", Stored::Bool(true), None);
69  }
70  // `DigestText`, with the item's own paragraph breaks noted in its text mode's frame (`frontmatter_paragraph_break`)
71  begin_mode("text")?;
72  note_frontmatter_paragraphs(tag);
73  // What the item leaves open closes with it (`digest_in_own_group`), as Perl's `egroup` after the item closes its group
74  // whatever the item's own `Error`s: the ACM SIG template's `\def\sharedaffiliation{\end{tabular}\begin{tabular}{c}}`
75  // in `\author` — which the class sets inside its own tabular — opens a tabular here that nothing closes, so the
76  // text mode could not end, this group was never popped, and its `\label` (`\lx@set@frontmatter@label`) took every
77  // later `\label` of the document: each reference dangled (1704.03168, 1301.5612, 1401.6981, 0712.1016, sig-alternate).
78  // Guard: `perfect_kernel_batch79l::an_author_left_open_keeps_the_documents_labels`.
79  let (digested, _) = digest_in_own_group(|| digest(item))?;
80  end_mode("text")?;
81  egroup()?;
82  digested
83}

85/// Perl: cleanTrailingBreak($document, $node) — remove trailing whitespace
86/// text nodes and ltx:break elements.
87pub(in crate::base_utilities) fn clean_trailing_break(
88  document: &mut Document,
89  node: &mut Node,
90) -> Result<()> {
91  while let Some(last) = node.get_last_child() {
92    let is_ws_text = last.get_type() == Some(NodeType::TextNode)
93      && last.get_content().chars().all(char::is_whitespace);
94    let is_break = with(document::get_node_qname(&last), |qname| {
95      qname == "ltx:break"
96    });
97    if !(is_ws_text || is_break) {
98      break;
99    }
100    document.remove_node(last);
101  }
102  Ok(())
103}

105/// A note an author's name or contact holds is the creator's, beside it (Rust-only, OXIDIZED_DESIGN_DIVERGENCES #1001):
106/// the `ltx:note` a `\footnotemark[<n>]` makes where it stands — and `relocate_footnote` fills with its `\footnotetext`
107/// (latex_constructs.pool.ltxml:498) — leaves the `ltx:personname` (or the creator's `ltx:contact`) for the creator,
108/// right after it, its `mark` and tags kept, as a `\thanks` note sits beside the name; a wrapper the move leaves
109/// empty (`Bob Baker{\small\footnotemark[2]}`) goes with it. Perl keeps the note in the name, so a reader of the name
110/// got "Bob Baker * * footnote * Note B." (`Bob Baker\footnotemark[1]` and a `\footnotetext[1]{Note B.}` after
111/// `\maketitle`, on every class route: 2104.12292; RT1's F17). A note nested in another note stays in it.
112pub(in crate::base_utilities) fn lift_notes_beside(
113  document: &mut Document,
114  node: &Node,
115) -> Result<()> {
116  let Some(parent) = node.get_parent() else {
117    return Ok(());
118  };
119  // (a creator, not the `role="pending"` holder of annotations still to be placed by label: a note lifted there has no
120  // label and was orphaned; the clone placed in the creator closes again and lifts it there — revtex's
121  // `\affiliation{\footnotemark[1]Univ A}`, 1607.07280)
122  if !with(document::get_node_qname(&parent), |qname| {
123    qname == "ltx:creator"
124  }) || parent.get_attribute("role").as_deref() == Some("pending")
125  {
126    return Ok(());
127  }
128  let mut anchor = node.clone();
129  for mut note in document.findnodes(".//ltx:note[not(ancestor::ltx:note)]", Some(node)) {
130    let mut holder = note.get_parent();
131    let mut before = note.get_prev_sibling();
132    note.unlink();
133    join_texts_after(before)?;
134    anchor.add_next_sibling(&mut note)?;
135    anchor = note;
136    // (a wrapper of the mark alone, left empty, goes too)
137    while let Some(wrapper) = holder.take() {
138      if wrapper == *node || !wrapper.get_child_nodes().is_empty() {
139        break;
140      }
141      holder = wrapper.get_parent();
142      before = wrapper.get_prev_sibling();
143      document.remove_node(wrapper);
144      join_texts_after(before)?;
145    }
146  }
147  Ok(())
148}

150/// The text a lifted note or wrapper stood between is one text again (`Kent-Andre Mardal\footnotemark[1]
151/// \footnotemark[2]`, 1612.03574), so the name's end is cleaned whole, as with one note.
152fn join_texts_after(before: Option<Node>) -> Result<()> {
153  if let Some(mut text) = before
154    && text.get_type() == Some(NodeType::TextNode)
155    && let Some(mut next) = text.get_next_sibling()
156    && next.get_type() == Some(NodeType::TextNode)
157  {
158    text.set_content(&(text.get_content() + &next.get_content()))?;
159    next.unlink();
160  }
161  Ok(())
162}

164/// Unwrap a `font="bold"` that wraps an ENTIRE personname.
165///
166/// SURPASS-Perl (html_feedback#61, OXIDIZED_DESIGN_DIVERGENCES #122). Some classes
167/// (neurips_2023) bold the whole author block with a block-level `\bf` in their
168/// `\@maketitle`, which LaTeXML does not emulate — it captures semantic creators.
169/// A paper that then `\textbf`s only *some* name lines (relying on the class `\bf`
170/// for the rest) renders incoherently: bold on the `\textbf` lines, plain on the
171/// others. Bolding a whole author name is presentational author-block styling, not
172/// semantic, so a personname whose sole meaningful content is one `<ltx:text
173/// font="bold">` is normalized to plain. Mixed-content names (bold on part of the
174/// name, or bold+other styles) are left untouched — only the whole-name pure-bold
175/// wrapper is stripped. Both Perl and Rust emit the wrapper (SHARED-FAILURE).
176pub(in crate::base_utilities) fn unwrap_whole_name_bold(
177  document: &mut Document,
178  node: &Node,
179) -> Result<()> {
180  let mut sole_bold: Option<Node> = None;
181  for child in node.get_child_nodes() {
182    match child.get_type() {
183      Some(NodeType::TextNode) => {
184        // Any non-whitespace text directly under the name means it is not a
185        // single whole-name wrapper — leave it alone.
186        if !child.get_content().chars().all(char::is_whitespace) {
187          return Ok(());
188        }
189      },
190      Some(NodeType::ElementNode) => {
191        // A reference marker (`\footnotemark`/`\thanks` → `<ltx:note>`) or a trailing
192        // `<ltx:break>` from a misused `\\` is not name content — skip it so it does
193        // not block the unwrap of an otherwise whole-name bold. Witness: "Zhou Zhao"
194        // in arXiv 2507.06670, `\textbf{Zhou Zhao} \footnotemark[2]`.
195        if with(document::get_node_qname(&child), |qname| {
196          qname == "ltx:note" || qname == "ltx:break"
197        }) {
198          continue;
199        }
200        // A `<ltx:text>` whose decoded font would serialize as EXACTLY `font="bold"` —
201        // i.e. bold is its only departure from the default text font. Reuse the
202        // serializer's own `font_attribute_string()` rather than hand-matching the
203        // family/series/shape defaults, so this cannot drift from what `font=` emits.
204        // Bold+italic / bold-sans yield "bold italic"/"… bold" ≠ "bold" and are left
205        // untouched (extra intent).
206        let is_pure_bold = with(document::get_node_qname(&child), |qname| {
207          qname == "ltx:text"
208        }) && child.get_attribute("_font").is_some_and(|fid| {
209          document
210            .decode_font(&fid)
211            .is_some_and(|f| f.font_attribute_string() == "bold")
212        });
213        if is_pure_bold && sole_bold.is_none() {
214          sole_bold = Some(child);
215        } else {
216          // A second element, or a non-pure-bold element → not a sole whole-name bold.
217          return Ok(());
218        }
219      },
220      _ => {},
221    }
222  }
223  if let Some(bold) = sole_bold {
224    document.unwrap_nodes(bold)?;
225  }
226  Ok(())
227}

229/// Read a full superscript operand as RAW (undigested) tokens for a frontmatter
230/// author/affiliation marker (`^X` / `\textsuperscript{X}`).
231///
232/// A real math superscript digests its nucleus (`TeX_Math` `scriptHandler`
233/// "invoke tokens until you get a box"), so `^\text{x}` binds `\text` to its
234/// `{x}` and reads as one box — i.e. `^\text{x}` == `^{\text{x}}`. The hijacked
235/// marker instead captures the operand as tokens for a matching label; the bare
236/// `{}` read used to grab only the leading control sequence (`\text`), orphaning
237/// its `{...}` argument. In inline math that stray `{...}` left a brace-group
238/// frame on top, so the marker's closing `$` fired `\lx@end@inline@math` against
239/// it — "Attempt to end mode math" — arbitrarily deep for nested
240/// `$^\text{$...$}$` markers (html_feedback#1021, arXiv:2403.11905; both engines
241/// erred, SHARED-FAILURE). Grabbing the control sequence together with its
242/// following group keeps `\text{...}` — and any `$...$` nested inside it — whole
243/// and undigested, so the surrounding math stays balanced. Numeric/char markers
244/// (`^1`) and already-braced markers (`^{...}`, `\textsuperscript{...}`) read
245/// exactly as before. See OXIDIZED_DESIGN #129.
246pub(in crate::base_utilities) fn read_frontmatter_sup_operand() -> Result<Tokens> {
247  skip_spaces()?;
248  // `^{...}` / `\textsuperscript{...}`: the whole braced group is the operand.
249  if if_next(T_BEGIN!())? {
250    return read_arg(ExpansionLevel::Off);
251  }
252  let Some(first) = read_token()? else {
253    return Ok(Tokens::new(Vec::new()));
254  };
255  // `^\text{...}` (bare control-sequence operand): keep a following `{...}` group
256  // so the CS is not severed from its argument.
257  if first.get_catcode() == Catcode::CS && if_next(T_BEGIN!())? {
258    let group = read_arg(ExpansionLevel::Off)?;
259    let inner = group.unlist_ref();
260    let mut operand = Vec::with_capacity(inner.len() + 3);
261    operand.push(first);
262    operand.push(T_BEGIN!());
263    operand.extend(inner.iter().copied());
264    operand.push(T_END!());
265    return Ok(Tokens::new(operand));
266  }
267  Ok(Tokens::new(vec![first]))
268}

346      (all, pictured)

348    None => (
349      clean_frontmatter_labels(&operand.to_string(), prefix),
350      Vec::new(),
351    ),
352  })

364/// The affiliation keys a name's mark requests by its printed text: of each label, its parts ([`parted_printed_label`],
365/// a one-run label whole) that open with a letter, a digit or a cross-reference's `\` — `1∗` and `†1` request 1, `1a` 1a,
366/// `∇` none (2402.10256).
367pub(in crate::base_utilities) fn requested_label_keys(printed: &str) -> Vec<String> {
368  let opens_a_label = |key: &String| {
369    let label = key.strip_prefix("affiliation:").unwrap_or(key);
370    label.starts_with('\\') || label.chars().next().is_some_and(char::is_alphanumeric)
371  };
372  clean_frontmatter_labels(printed, "affiliation")
373    .into_iter()
374    .flat_map(|label| {
375      let mut parts = parted_printed_label(label);
376      // (a parted label's first key is the whole, which the page shows as its parts)
377      if parts.len() > 1 {
378        parts.remove(0);
379      }
380      parts.retain(opens_a_label);
381      parts
382    })
383    .collect()
384}

400/// A mark operand's printed text ([`printed_label`]), and the same with each pgf picture's text left out ([`label_ink`]:
401/// a picture prints its text nodes, a label apart).
402#[derive(Debug, Default)]
403pub(in crate::base_utilities) struct PrintedLabel {
404  /// What the operand prints.
405  pub text:             String,
406  /// What it prints outside its pictures, each picture a label separator.
407  pub outside_pictures: String,
408  /// What it prints where it stands, when asked for ([`printed_ink`]): outside its notes (`\lx@note…`, set at the
409  /// page's foot).
410  pub outside_notes:    Option<String>,
411  /// How many notes the character being written stands in.
412  notes:                usize,
413}
414
415impl PrintedLabel {
416  /// Write `c`, inside a picture or not.
417  fn push(&mut self, c: char, in_picture: bool) {
418    self.text.push(c);
419    if !in_picture {
420      self.outside_pictures.push(c);
421    }
422    if self.notes == 0
423      && let Some(outside) = self.outside_notes.as_mut()
424    {
425      outside.push(c);
426    }
427  }
428}

430/// A printed label read as the marks the page shows: a number or letters a symbol is glued to (`$^{\ICQD,\Phys\ast}$`
431/// prints "1,4∗", 2412.09857) the whole, then each run apart (`4∗`, `4`, `∗`) — the number an affiliation's and the
432/// symbol a note's —; a run of distinct symbols (`$^{\clubsuit\heartsuit}$` prints "♣♡", 1811.10830's spelled shape)
433/// each symbol too (a doubled `∗∗` stays one), as the spelled run of symbol commands is (`SYMBOL_COMMAND_RUN`,
434/// [`clean_frontmatter_labels`]); any other label as it is. A combining mark is part of the character before it
435/// (`x̄`, `\={x}`'s, is one letter: no `x` and no mark of its own, review rev92b_r2 B1). Printed text alone (review
436/// rev92b_r1 S2): a mark's spelling is parted by [`clean_frontmatter_labels`] only.
437fn parted_printed_label(label: String) -> Vec<String> {
438  let (prefix, bare) = match label.split_once(':') {
439    Some((prefix, bare)) => (s!("{prefix}:"), bare.to_string()),
440    None => (String::new(), label.clone()),
441  };
442  // (each run a kind — symbol or not — and its characters, a combining mark kept with the one before it)
443  let mut runs: Vec<(bool, Vec<String>)> = Vec::new();
444  for c in bare.chars() {
445    if is_combining_mark(c) {
446      match runs.last_mut().and_then(|(_, run)| run.last_mut()) {
447        Some(character) => {
448          character.push(c);
449          continue;
450        },
451        None => return vec![label],
452      }
453    }
454    let symbol = if c.is_alphanumeric() {
455      false
456    } else if is_printed_symbol(c) {
457      true
458    } else {
459      return vec![label];
460    };
461    match runs.last_mut() {
462      Some((kind, run)) if *kind == symbol => run.push(c.to_string()),
463      _ => runs.push((symbol, vec![c.to_string()])),
464    }
465  }
466  let mut parts: Vec<String> = Vec::new();
467  for (symbol, run) in &runs {
468    if runs.len() > 1 {
469      parts.push(run.concat());
470    }
471    if *symbol && run.iter().any(|c| *c != run[0]) {
472      parts.extend(run.iter().cloned());
473    }
474  }
475  if parts.is_empty() {

477  }
478  std::iter::once(label)

481}

587/// The printed text of `tokens` digested in a speculative run of their own (in a group, in a reading context of their
588/// own, as sect05.rs `title_page_probe_run` runs a title page): nothing of it stays — its assignments undone, its
589/// diagnostics discarded, an argument read past its end (`\altaffilmark` handed `\let`) reading nothing of the
590/// document's. `None` when no journal opens, the run is not undone, or it raised an error (a resource Fatal, or one
591/// that could not be undone, propagates: Fatal stays Fatal). `ink` is where it is written (what it asks for).
592fn speculative_label_ink(tokens: Tokens, ink: PrintedLabel) -> Result<Option<PrintedLabel>> {
593  let Some(journal) = begin_assignment_journal() else {
594    return Ok(None);
595  };
596  let _speculative = speculative_digestion();
597  let hold = util::logger::DiagnosticsHold::begin();
598  let resource_fatal_before = resource_fatal_latched();
599  let muted = SuppressLogOutputGuard::new(true);
600  let depth = get_frame_depth();
601  let boxes = box_lists_mark();
602  // (the tokens being invoked and the conditionals put back however the run ends: `StomachMark`)
603  let stomach_mark = StomachMark::take();
604  // (the field ledger put back and the boxes it stamped unstamped, as the observer's rehearsal does: a `\usebox` in
605  // the operand hands back a register's box, which the run's serial would otherwise stay on)
606  let ledger_scope = FieldLedgerScope::begin();
607  let result = reading_from_mouth(Mouth::default(), || -> Result<Digested> {
608    bgroup();
609    let digested = digest(tokens)?;
610    egroup()?;
611    Ok(digested)
612  });
613  let unwound = reading_from_mouth(Mouth::default(), || {
614    new_local_box_list();
615    while get_frame_depth() > depth {
616      let before = get_frame_depth();
617      let nobox = lookup_bool("groupNonBoxing");
618      if pop_stack_frame(nobox).is_err() || get_frame_depth() >= before {
619        break;
620      }
621    }
622    expire_local_box_list();
623    Ok(())
624  });
625  stomach_mark.restore();
626  let printed = result.as_ref().ok().map(|digested| {
627    let mut printed = ink;
628    label_ink(digested, &mut printed, false);
629    printed
630  });
631  drop(ledger_scope);
632  drop(muted);
633  let errors = hold.errors_raised();
634  let undone = journal.rollback();
635  if let Err(err) = result {
636    if !undone || (resource_fatal_latched() && !resource_fatal_before && !unlatch_runaway_fatal()) {
637      hold.commit();
638      return Err(err);
639    }
640    hold.discard();
641    unwound?;
642    unwind_box_lists_to(boxes);
643    return Ok(None);
644  }
645  if undone {
646    hold.discard();
647  } else {
648    hold.commit();
649  }
650  unwound?;
651  Ok(printed.filter(|_| undone && errors == 0))
652}

654/// The text a digested label operand prints ([`printed_label`]): its boxes' characters, through lists and the arguments
655/// a whatsit shows ([`shown_arguments`]); a cross-reference (`\ref`, its key resolved in post: it prints nothing yet)
656/// written `\ref{key}`, as [`clean_frontmatter_labels`] reads a key. Spacing parts two labels as a comma does — any
657/// white space in a box's text, and every item [`is_spacing_item`] reads as spacing, the reader 92e's [`first_ink`] uses
658/// (`$^{1\,2}$`'s math glue, a box of a space; `\!`'s zero-width-space box; `\:`, `\>`, `\ `, `\hspace`, `\kern`,
659/// `\mskip`): `$^{1\,2}$` is the labels 1 and 2 (2403.08629, 2601.02539, 1704.04058's `$^{* \, \dagger}$`; review
660/// rev92b_r1 B1: one label `1_2`, nothing answered it, and the mark was lost). A format character prints no label, a
661/// combining mark is part of the character before it ([`keeps_label_char`]: `\={x}`'s `x̄`), and a pgf/TikZ picture
662/// prints only its text nodes ([`picture_text`]: a circled `a`'s `a`, never an SVG path's data), a label apart.
663fn label_ink(d: &Digested, out: &mut PrintedLabel, in_picture: bool) {
664  if is_spacing_item(d) {
665    out.push(',', in_picture);
666    return;
667  }
668  match d.data() {
669    DigestedData::TBox(b) => {
670      if let Ok(text) = b.borrow().get_string() {
671        for c in text.chars() {
672          if c.is_whitespace() {
673            out.push(',', in_picture);
674          } else if keeps_label_char(c, out.text.chars().next_back()) {
675            out.push(c, in_picture);
676          }
677        }
678      }
679    },
680    DigestedData::List(l) => {
681      for child in l.borrow().boxes.iter() {
682        label_ink(child, out, in_picture);
683      }
684    },
685    DigestedData::Whatsit(w) => {
686      let w = w.borrow();
687      let cs = w.get_definition().get_cs_name().into_owned();
688      if matches!(
689        cs.trim_end(),
690        "\\ref" | "\\autoref" | "\\eqref" | "\\lx@cref"
691      ) {
692        if let Some(key) = w.get_args().iter().rev().flatten().next() {
693          for c in s!(",\\ref{{{}}},", key.to_string().trim()).chars() {
694            out.push(c, in_picture);
695          }
696        }
697        return;
698      }
699      // (a pgf picture prints only its text nodes, a label apart: 2605.02769's TikZ ORCID logo none, a circled `a` its
700      // `a`; what it prints is kept apart, `PrintedLabel::outside_pictures`)
701      if let Some(text) = picture_text(&w) {
702        out.push(',', in_picture);
703        for node in &text {
704          label_ink(node, out, true);
705        }
706        out.push(',', in_picture);
707        return;
708      }
709      let start = out.text.len();
710      // (the note constructors by name: a prefix also matched `\lx@notextattachfile`; revP23e)
711      let note = matches!(
712        cs.as_str(),
713        "\\lx@note" | "\\lx@note@live" | "\\lx@notemark" | "\\lx@notetext" | "\\lx@notetext@live"
714      );
715      out.notes += usize::from(note);
716      for a in &printed_properties(&w) {
717        label_ink(a, out, in_picture);
718      }
719      for (arg, shown) in w.get_args().iter().zip(shown_arguments(&w)) {
720        if shown && let Some(arg) = arg {
721          label_ink(arg, out, in_picture);
722        }
723      }
724      if let Ok(Some(body)) = w.get_body() {
725        label_ink(&body, out, in_picture);
726      }
727      if out.text.len() == start
728        && let Some(glyph) = printed_own_text(&w)
729      {
730        for c in glyph.chars() {
731          if keeps_label_char(c, out.text.chars().next_back()) {
732            out.push(c, in_picture);
733          }
734        }
735      }
736      // (a math accent prints its glyph on its base: `$^{\bar{\dag}}$` is `†̄`, no `\dag` — review rev92b_r4 S2)
737      for c in accent_combining_form(&w).unwrap_or_default().chars() {
738        if keeps_label_char(c, out.text.chars().next_back()) {
739          out.push(c, in_picture);
740        }
741      }
742      out.notes -= usize::from(note);
743    },
744    _ => {},
745  }
746}

748/// A name mark's labels with a number range spelled out: `1-4` requests 1, 2, 3 and 4 (`Bauer$^{1-4}$`, 2012.15817 —
749/// read as its ends, the 2 and 3 entries were not hers), and so does the en-dash `--` spells before its ligature
750/// (`$^{1\text{--}5}$`, 2507.06232). A falling range, or one of more than 50 numbers, stays as written. The one reading
751/// of a range, the frontmatter observer's (`resolver::labels`) and the kernel's labels' ([`clean_frontmatter_labels`]:
752/// the names' requests, the frontmatter API's Cut, `relocate_annotations`; batch 99a, OXIDIZED_DESIGN_DIVERGENCES #954).
753pub(in crate::base_utilities) fn spelled_ranges(labels: Vec<String>) -> Vec<String> {
754  const DASHES: [char; 3] = ['-', '\u{2013}', '\u{2212}'];
755  labels
756    .into_iter()
757    .flat_map(|label| {

762        ))

764      match range {
765        Some((a, b)) if a < b && b - a <= 50 => (a..=b).map(|n| n.to_string()).collect(),
766        _ => vec![label],
767      }
768    })

772/// The asterisk forms a mark may print or be written in: ASCII `*`, the math asterisk `∗` (U+2217, what `$^*$` and
773/// `$^\ast$` print), the low asterisk `⁎` (U+204E), the heavy asterisk `✱` (U+2731) and the full-width `＊` (U+FF0A).
774pub(in crate::base_utilities) const ASTERISK_FORMS: [char; 5] =
775  ['*', '\u{2217}', '\u{204E}', '\u{2731}', '\u{FF0A}'];
776
777/// The one key of an asterisk mark, however it is written or printed: every form of [`ASTERISK_FORMS`], and a label
778/// spelled `\ast` or `\textasteriskcentered`, all read `*` — "all asterisks are the same key, they are visually
779/// identical to readers" (user ruling 2026-11-02, FIDELITY_CHARTER §4). The key [`clean_frontmatter_labels`] gives both
780/// sides of a match, the mark a name requests and the label a line sets (review rev100a B1: a legend `$^\ast$Deceased`
781/// under `Ann Able$^{1,*}$` keyed `∗` against the name's `*`, so no name seemed to request it and the line was in doubt,
782/// the class's text). The readers of printed glyphs (the frontmatter observer writer's shown marks) read them by it too.
783pub(in crate::base_utilities) fn one_asterisk(text: &str) -> String {
784  match text {
785    "\\ast" | "\\textasteriskcentered" => "*".to_string(),
786    _ => text
787      .chars()
788      .map(|c| if ASTERISK_FORMS.contains(&c) { '*' } else { c })
789      .collect(),
790  }
791}

804/// Perl: cleanFrontmatterLabels($labels, $prefix). Rust also parts the labels at the spacing ([`SPACING_IN_LABEL`]), on
805/// both sides of the match: the mark an author requests and the one an affiliation sets (OXIDIZED_DESIGN #52(o)); and
806/// keys an asterisk one way, written or printed ([`one_asterisk`]).
807pub(in crate::base_utilities) fn clean_frontmatter_labels(
808  labels: &str,
809  prefix: &str,
810) -> Vec<String> {
811  let labels = FONT_SWITCH.replace_all(labels, "$1");
812  // (Perl's `s/\\rm//g`, Base_Utility.pool.ltxml:365, which strips `\rm` glued to the label it switches, too: the
813  // operand `{\rm a,b}` reads "\rma,b", where the tokenizer skipped the space after `\rm` — `$^{\rm a,b}$` requested
814  // "\rma" and "b", so no `${\rm ^a}$` line answered the first, 0905.3658)
815  let labels = labels.replace("\\rm", "");
816  let labels = SPACING_IN_LABEL.replace_all(&labels, ",$1");
817  let mut cleaned = Vec::new();
818  let mut pieces: Vec<&str> = labels.split(',').collect();
819  // Perl split drops trailing empty fields
820  while pieces.last().is_some_and(|p| p.is_empty()) {
821    pieces.pop();
822  }
823  // (a run of distinct symbol commands is a label each, as `$^{1,2}$` is two: `$^{\spadesuit\heartsuit}$` under
824  // `$^\spadesuit$Univ A \\ $^\heartsuit$Univ B`, 1811.10830)
825  let pieces: Vec<&str> = pieces
826    .into_iter()
827    .flat_map(|piece| match SYMBOL_COMMAND_RUN.is_match(piece.trim()) {
828      true => {
829        let symbols: Vec<&str> = SYMBOL_COMMAND
830          .find_iter(piece)
831          .map(|m| m.as_str())
832          .collect();
833        // (the run whole first, then each symbol: a line labelled by the run — `$^{*\dag}$Project lead`, which
834        // `\lx@sup@setlabel@affiliation` labels by its first label — is answered by a name carrying the whole run,
835        // not every name carrying one of its symbols, 2602.19190; a name's run requests each symbol too,
836        // 2606.15609's `$^{*\spadesuit}$`)
837        if symbols.iter().skip(1).any(|symbol| *symbol != symbols[0]) {
838          std::iter::once(piece).chain(symbols).collect()
839        } else {
840          vec![piece]
841        }
842      },
843      false => vec![piece],
844    })
845    .collect();
846  // (a number range is each number of it: `$^{1-4}$` requests 1, 2, 3 and 4, [`spelled_ranges`])
847  let pieces = spelled_ranges(
848    pieces
849      .into_iter()
850      .map(|piece| piece.trim().to_string())
851      .collect(),
852  );
853  for label in pieces {
854    let label = label.trim();
855    // INTENTIONAL DIVERGENCE (OXIDIZED_DESIGN #34, KNOWN_PERL_ERRORS #31,
856    // plan decisions log #5): Perl prefixes empty fields too, so a doubled
857    // comma or empty keyval yields a contentless "prefix:" label that can
858    // spuriously match another in relocate_annotations. Drop them.
859    if label.is_empty() {
860      continue;
861    }
862    let mut label = if let Some(inner) = label
863      .strip_prefix("\\ref{")
864      .and_then(|rest| rest.strip_suffix('}'))
865      .filter(|inner| !inner.contains('}'))
866    {
867      let inner = inner.trim();
868      if inner.is_empty() {
869        continue; // \ref{} ⇒ contentless "LABEL:" — drop (same divergence)
870      }
871      s!("LABEL:{inner}")
872    } else if !prefix.is_empty() {
873      s!("{prefix}:{label}")
874    } else {
875      label.to_string()
876    };
877    label = SPACES_RE.replace_all(&label, "_").into_owned();
878    label.retain(|c| !matches!(c, '{' | '}' | '(' | ')'));
879    // (the asterisk keyed one way however it is written or printed: [`one_asterisk`])
880    let label = match label.split_once(':') {
881      Some((head, bare)) if head == prefix => s!("{head}:{}", one_asterisk(bare)),
882      _ => one_asterisk(&label),
883    };
884    cleaned.push(label);
885  }
886  cleaned
887}

890/// Perl: getFrontmatterName($name, $tag, $role).
891///
892/// Rust-only (OXIDIZED_DESIGN_DIVERGENCES #921; FIDELITY_CHARTER §4, user ruling 2026-10-30): a `name=` the call gives

898  name: Option<&String>,
899  tag: &str,
900  role: &str,
901) -> Result<Option<String>> {
902  let stag = tag.strip_prefix("ltx:").unwrap_or(tag);
903  if let Some(name) = name {
904    return Ok((!name.is_empty()).then(|| name.clone()));
905  }
906  if !role.is_empty() {

911      // emptied because the iD glyph self-identifies) means NO label — return
912      // None rather than an empty `ltx:contact_name`, and do not fall back to
913      // the generic default. General, not role-special.
914      return Ok(if n.is_empty() { None } else { Some(n) });
915    }
916  }
917  let cs = T_CS!(s!("\\lx@{stag}@name"));
918  if lookup_definition(&cs)?.is_some() {
919    let n = digest_text(Tokens!(cs))?.to_string();
920    if !n.is_empty() {
921      return Ok(Some(n));
922    }
923  }
924  Ok(None)
925}

1537/// Drop the abstract entries before (tag, index) of the same `name`, language and store as it, and its text (whitespace
1538/// aside): a repeat, not a second block — what a class or binding adds twice, which the clearing of a later same-named
1539/// entry used to absorb. The later copy stays, as the clearing kept it: a `\maketitle` placed the first, the second
1540/// is where the paper prints it (JAC2001's `\abstract{…}` before `\maketitle` and the same `{abstract}` after it,
1541/// physics0206049). A block with other text is a second one, kept. Not while an entry of the tag is still being
1542/// digested, whose index its digestion holds. OXIDIZED_DESIGN #154.
1543pub(in crate::base_utilities) fn frontmatter_drop_repeats(tag: &str, index: usize) {
1544  type Key = [Option<String>; 3];
1545  // (the boxes' text is read outside the borrow of the frontmatter: reading it consults the state)
1546  let entries: Vec<(Key, Option<Digested>)> = with_value("frontmatter", |v| match v {
1547    Some(Stored::HashTagData(frnt)) => frnt
1548      .get(tag)
1549      .map(|list| {
1550        list
1551          .iter()
1552          .map(|e| {
1553            let key = ["name", "xml:lang", "_store"].map(|k| e.attr.get(k).cloned());
1554            let content = match e.content.as_slice() {
1555              [TagContent::PlaceKeeper] => None,
1556              [TagContent::Box(d), ..] => Some(d.clone()),
1557              _ => None,
1558            };
1559            (key, content)
1560          })
1561          .collect()
1562      })
1563      .unwrap_or_default(),
1564    _ => Vec::new(),
1565  });
1566  if index >= entries.len() || frontmatter_has_open_placekeeper(tag) {
1567    return;
1568  }
1569  let text = |d: &Option<Digested>| {
1570    d.as_ref()
1571      .map(|d| {
1572        d.to_string()
1573          .split_whitespace()
1574          .collect::<Vec<_>>()
1575          .join(" ")
1576      })
1577      .filter(|t| !t.is_empty())
1578  };
1579  let (new_key, new_content) = &entries[index];
1580  let Some(new_text) = text(new_content) else {
1581    return;
1582  };
1583  let repeats: Vec<usize> = entries[..index]
1584    .iter()
1585    .enumerate()
1586    .filter(|(_, (key, content))| {
1587      key == new_key && text(content).as_deref() == Some(new_text.as_str())
1588    })
1589    .map(|(at, _)| at)
1590    .collect();
1591  for at in repeats.into_iter().rev() {
1592    frontmatter_remove(tag, at);
1593  }
1594}

1596/// The abstract and keywords entries of a raw class's `store` (`_store`, frontmatter_stores.rs `labelled_api`) whose
1597/// title page does not print it (`\hypersetup{pdfsubject={\@abst}}`, review probe `md`) become ordinary entries, which
1598/// the document's own of the same name and language replaces (`frontmatter_clear_same_name`, [`RELEASED_STORE`]): one
1599/// already there drops the store's.
1600pub(crate) fn frontmatter_release_store(store: &str) {
1601  with_value_mut("frontmatter", |val_opt| {
1602    if let Some(&mut Stored::HashTagData(ref mut frnt)) = val_opt {
1603      for tag in ["ltx:abstract", "ltx:keywords"] {
1604        let Some(list) = frnt.get_mut(tag) else {
1605          continue;
1606        };
1607        let released: Vec<usize> = (0..list.len())
1608          .filter(|&i| list[i].attr.get("_store").map(String::as_str) == Some(store))
1609          .collect();
1610        let mut dropped: Vec<usize> = Vec::new();
1611        for &i in &released {
1612          if let Some(mut entry) = list.get_mut(i) {
1613            entry.attr.remove("_store");
1614            entry
1615              .attr
1616              .insert(RELEASED_STORE.to_string(), "true".to_string());
1617          }
1618          let (name, lang) = (list[i].attr.get("name"), list[i].attr.get("xml:lang"));
1619          let replaced = list.iter().enumerate().any(|(j, e)| {
1620            j != i
1621              && !released.contains(&j)
1622              && !e.attr.contains_key("_store")
1623              && e.attr.get("name") == name
1624              && e.attr.get("xml:lang") == lang
1625          });
1626          if replaced {
1627            dropped.push(i);
1628          }
1629        }
1630        for i in dropped.into_iter().rev() {
1631          list.remove(i);
1632        }
1633      }
1634    }
1635  });
1636}
