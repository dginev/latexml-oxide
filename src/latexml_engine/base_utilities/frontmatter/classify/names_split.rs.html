6/// The index in `body` past the glue specification starting at `i` (after `\hskip` or `\kern`): a dimension — a
7/// number and a unit, two letters or `fil`, `fill`, `filll` — then `plus` and `minus` dimensions (`1em plus 1fil`).
8pub(in crate::base_utilities) fn past_glue(body: &[Token], i: usize) -> usize {
9  let letter_at = |j: usize, c: char| {
10    body.get(j).is_some_and(|t| {
11      t.get_catcode() == Catcode::LETTER && t.with_str(|s| s.len() == 1 && s.starts_with(c))
12    })
13  };
14  let past_spaces = |mut j: usize| {
15    while body.get(j) == Some(&T_SPACE!()) {
16      j += 1;
17    }
18    j
19  };
20  let past_dimension = |j: usize| {
21    let mut j = past_spaces(j);
22    while body
23      .get(j)
24      .is_some_and(|u| u.with_str(|s| s.chars().all(|c| c.is_ascii_digit() || ".+-".contains(c))))
25    {
26      j += 1;
27    }
28    j = past_spaces(j);
29    if letter_at(j, 'f') && letter_at(j + 1, 'i') && letter_at(j + 2, 'l') {
30      j += 3;
31      while letter_at(j, 'l') {
32        j += 1;
33      }
34    } else {
35      for _ in 0..2 {
36        if body
37          .get(j)
38          .is_some_and(|u| u.get_catcode() == Catcode::LETTER)
39        {
40          j += 1;
41        }
42      }
43    }
44    j
45  };
46  let mut i = past_dimension(i);
47  for keyword in ["plus", "minus"] {
48    let j = past_spaces(i);
49    if keyword
50      .chars()
51      .enumerate()
52      .all(|(k, c)| letter_at(j + k, c))
53    {
54      i = past_dimension(j + keyword.len());
55    }
56  }
57  i
58}

60/// Where a piece's leading spacing ends: spaces, ties, space commands (`\ `, `\quad`) and `\vspace`/`\hspace` with
61/// their length.
62pub(in crate::base_utilities) fn leading_spacing_end(v: &[Token]) -> usize {
63  let mut i = 0;
64  while i < v.len() {
65    if is_spacing(&v[i]) || PARAMETERLESS_SPACING.iter().any(|cs| v[i] == T_CS!(*cs)) {
66      i += 1;
67    } else if v[i] == T_CS!("\\penalty") {
68      // an integer follows (tex.web §1103 `\penalty` scans an int: `\penalty10000`, `\penalty -50`)
69      let mut j = i + 1;
70      while v
71        .get(j)
72        .is_some_and(|t| *t == T_SPACE!() || *t == T_OTHER!("-") || *t == T_OTHER!("+"))
73      {
74        j += 1;
75      }
76      while v.get(j).is_some_and(|t| {
77        t.get_catcode() == Catcode::OTHER && t.with_str(|s| s.chars().all(|c| c.is_ascii_digit()))
78      }) {
79        j += 1;
80      }
81      i = j;
82    } else if ["\\vskip", "\\hskip", "\\kern"]
83      .iter()
84      .any(|cs| v[i] == T_CS!(*cs))
85    {
86      // a glue or dimension follows (tex.web §1057, §1061: `\kern2pt`, `\vskip 1em plus 1fil`)
87      i = past_glue(v, i + 1);
88    } else if v[i] == T_CS!("\\rule") {
89      // latex.ltx:16359 `\rule[raise]{width}{height}`, a rule box, no glyph
90      let mut j = i + 1;
91      if v.get(j) == Some(&T_OTHER!("[")) {
92        while j < v.len() && v[j] != T_OTHER!("]") {
93          j += 1;
94        }
95        j += 1;
96      }
97      for _ in 0..2 {
98        if v.get(j).is_none_or(|t| t.get_catcode() != Catcode::BEGIN) {
99          break;
100        }
101        j = skip_group(v, j).min(v.len());
102      }
103      i = j;
104    } else if v[i] == T_CS!("\\vspace") || v[i] == T_CS!("\\hspace") || v[i] == T_CS!("\\addvspace")
105    {
106      let mut j = i + 1;
107      if v.get(j) == Some(&T_OTHER!("*")) {
108        j += 1;
109      }
110      while v.get(j) == Some(&T_SPACE!()) {
111        j += 1;
112      }
113      // (only with its braced length: `\vspace\baselineskip` stops here, its length no wrapper)
114      if v.get(j).is_none_or(|t| t.get_catcode() != Catcode::BEGIN) {
115        break;
116      }
117      i = skip_group(v, j).min(v.len());
118    } else {
119      break;
120    }
121  }
122  i
123}

125/// Whether `text` reads as a person's name whose last word, an institution's ([`non_name_word`]: `Centers`,
126/// `College`, `Fellows`), follows the initial of a given name: two to five capitalised words, the one before the last
127/// an initial (`Gary P. Centers`; not `W. M. Keck Observatory`, whose institution word follows a name), no other
128/// institution word, no digit or address. In a list by its marks (the reading's `marked_name`, `resolver/lines.rs`) such
129/// a piece is a marked name (rows N2-N3).
130pub(in crate::base_utilities) fn surname_after_initial(text: &str) -> bool {
131  let text = text
132    .trim()
133    .trim_end_matches(|c: char| ",;*".contains(c))
134    .trim_end();
135  if text
136    .chars()
137    .any(|c| c.is_ascii_digit() || "@:/()".contains(c))
138  {
139    return false;
140  }
141  let words: Vec<&str> = text.split_whitespace().collect();
142  // (a place's abbreviation is no initial: `St.`, `Mt.`, `Ft.`)
143  let initial = |w: &str| {
144    let mut chars = w.chars();
145    if ["St.", "Mt.", "Ft."].contains(&w) {
146      return false;
147    }
148    chars.next().is_some_and(char::is_uppercase)
149      && chars
150        .as_str()
151        .strip_suffix('.')
152        .is_some_and(|rest| rest.chars().all(char::is_lowercase) && rest.len() <= 1)
153  };
154  let capitalised = |w: &&str| w.chars().next().is_some_and(char::is_uppercase);
155  (3..=5).contains(&words.len())
156    && words.iter().all(capitalised)
157    && initial(words[words.len() - 2])
158    && non_name_word(words[words.len() - 1])
159    && !words[..words.len() - 1].iter().any(|w| non_name_word(w))
160}
