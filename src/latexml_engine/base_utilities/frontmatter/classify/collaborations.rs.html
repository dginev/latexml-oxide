6/// What frames a collaboration's name on the line ([`collaboration_core`]): nothing, a bare "the" before it, or more —
7/// "for", "on behalf of", "representing", brackets, dashes.
8#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Debug)]
9pub(in crate::base_utilities) enum Framing {
10  /// the name alone: `\textit{WIDE Consortium}` (2501.13770), `\center BESIII Collaboration` (2212.06489)
11  Unframed,
12  /// a bare "the": `the ATLAS Collaboration`, `The Robotics Group`
13  The,
14  /// "for", "on behalf of", "representing", brackets or dashes: `On Behalf Of The ATLAS Collaboration` (0910.2149),
15  /// `Representing the BES Collaboration` (hep-ex0011032), `(XYZ Collaboration)`, `-- QCDSF-UKQCD Collaboration --`
16  Framed,
17}
18
19/// `text` without what frames a collaboration's name — a list's "and", brackets, dashes (`-- QCDSF-UKQCD Collaboration
20/// --`, hep-ph0502212), "for", "on behalf of", "representing", "the" — and what framed it ([`Framing`]); and whether
21/// brackets held a name opening with "The" (`(The Nearby Supernova Factory)`, 1511.01102).
22pub(in crate::base_utilities) fn collaboration_core(text: &str) -> (String, Framing, bool) {
23  let dash = |c: char| matches!(c, '-' | '–' | '—');
24  let mut t = text.trim().to_string();
25  let (mut framing, mut the_in_brackets) = (Framing::Unframed, false);
26  loop {
27    let before = t.clone();
28    for p in ["and ", "And ", "& "] {
29      if let Some(rest) = t.strip_prefix(p) {
30        t = rest.trim_start().to_string();
31      }
32    }
33    if t.len() >= 2
34      && ((t.starts_with('(') && t.ends_with(')')) || (t.starts_with('[') && t.ends_with(']')))
35    {
36      t = t[1..t.len() - 1].trim().to_string();
37      framing = Framing::Framed;
38      the_in_brackets |= t.get(..4).is_some_and(|p| p.eq_ignore_ascii_case("the "));
39    }
40    if t.starts_with(dash) && t.ends_with(dash) {
41      t = t.trim_matches(dash).trim().to_string();
42      framing = Framing::Framed;
43    }
44    for p in SPEAKS_FOR.iter().chain(&["the "]) {
45      if t.get(..p.len()).is_some_and(|q| q.eq_ignore_ascii_case(p)) {
46        t = t[p.len()..].trim_start().to_string();
47        framing = framing.max(if *p == "the " {
48          Framing::The
49        } else {
50          Framing::Framed
51        });
52        break;
53      }
54    }
55    if t == before {
56      break;
57    }
58  }
59  (t, framing, the_in_brackets)
60}

62/// A word without the punctuation around it, lowercased (as [`word_forms`] reads it).
63pub(in crate::base_utilities) fn bare_word(word: &str) -> String {
64  word
65    .trim_matches(|c: char| ".,;:()[]".contains(c))
66    .to_lowercase()
67}

69/// Whether a word names a collaboration, a team or a group ([`COLLABORATION_WORDS`], [`TEAM_WORDS`]).
70pub(in crate::base_utilities) fn collaboration_word(word: &str) -> bool {
71  let bare = bare_word(word);
72  COLLABORATION_WORDS.contains(&bare.as_str()) || TEAM_WORDS.contains(&bare.as_str())
73}

75/// Whether the visible text of an author-line piece names a collaboration, a team or a consortium and no person: its
76/// name ends with a collaboration word ([`COLLABORATION_WORDS`]: `XYZ Collaboration`, `for the ATLAS, CDF, CMS and D0
77/// Collaborations`, 1311.7305) or with a team word ([`TEAM_WORDS`]) the line frames — "the", "for", "on behalf of",
78/// brackets: `and the GAPS Team`, 1708.04166 — or a names line lists with "and" (`listed`: `and LHD experimental group`,
79/// 1711.07097), and no word before it names an institution (`The ATLAS Collaboration Office`, `Theory Group, Department
80/// of Physics` stay affiliations); or brackets hold a name opening with "The" that no word names an institution in
81/// (`(The Nearby Supernova Factory)`, 1511.01102). A name the collaboration follows is a name and its remark
82/// (`collaboration_split`), no collaboration: `Bob Baker for the ATLAS Collaboration`; nor is a surname holding such a
83/// word (`M.~Huertas-Company`, 1212.4143: "Company" no collaboration's).
84pub fn reads_as_collaboration(text: &str, listed: bool) -> bool {
85  collaboration_framing(text, listed, false).is_some()
86}

88/// Whether a line of its own under or over the names names a collaboration ([`reads_as_collaboration`]), and how its
89/// name is framed: a team word's only by more than a bare "the" (`for the GAPS Team`, `(The Robotics Group)`) — `The
90/// Robotics Group`, `the Applied Mathematics Group` under a name are its institution's, an affiliation.
91pub fn collaboration_line_framing(text: &str) -> Option<Framing> {
92  collaboration_framing(text, false, true)
93}

95/// [`reads_as_collaboration`] (`line` false) and [`collaboration_line_framing`] (`line` true): how the collaboration a
96/// piece or line names is framed, `None` when it names none.
97pub(in crate::base_utilities) fn collaboration_framing(
98  text: &str,
99  listed: bool,
100  line: bool,
101) -> Option<Framing> {
102  if text.contains('@') {
103    return None;
104  }
105  let (core, framing, the_in_brackets) = collaboration_core(text);
106  let lower = core.to_lowercase();
107  if SPEAKS_FOR.iter().any(|p| lower.contains(&s!(" {p}"))) || core.contains(['(', '[']) {
108    return None;
109  }
110  let words: Vec<&str> = core.split_whitespace().collect();
111  let last = words.last()?;
112  if words.len() > 12 {
113    return None;
114  }
115  let before = &words[..words.len() - 1];
116  if before
117    .iter()
118    .any(|w| non_name_word(w) && !collaboration_word(w))
119  {
120    return None;
121  }
122  // (a name, not a sentence or a person's: a lowercase word only right before the collaboration word, `LHD experimental
123  // group` — not `Monica Brigida\\on behalf of the \emph{Fermi}-LAT Collaboration`, whose line break the text does
124  // not part, 0907.0548)
125  if before.iter().enumerate().any(|(at, w)| {
126    !["and", "of", "the", "&"].contains(&bare_word(w).as_str())
127      && w.chars().next().is_some_and(char::is_lowercase)
128      && at + 1 != before.len()
129  }) {
130    return None;
131  }
132  let head = bare_word(last);
133  if COLLABORATION_WORDS.contains(&head.as_str()) {
134    return Some(framing);
135  }
136  if TEAM_WORDS.contains(&head.as_str()) {
137    let framed = if line {
138      framing == Framing::Framed
139    } else {
140      framing != Framing::Unframed
141    };
142    return (framed || listed).then_some(framing);
143  }
144  (the_in_brackets && !before.is_empty() && !words.iter().any(|w| non_name_word(w)))
145    .then_some(framing)
146}

148/// Whether a piece of a names line names a collaboration beside the names ([`reads_as_collaboration`]) the way a line
149/// lists one: framed — "the", "for", "on behalf of", brackets — or listed with "and" (`listed`). A piece the commas alone
150/// part from the names is no collaboration of theirs: revtex's `\collaboration{Fermilab Lattice, HPQCD, and MILC
151/// Collaborations}` (2412.18491, its `\author{#1}`) one collaboration, no name "Fermilab Lattice".
152pub fn reads_as_listed_collaboration(text: &str, listed: bool) -> bool {
153  reads_as_collaboration(text, listed)
154    && (listed || collaboration_core(text).1 != Framing::Unframed)
155}

157/// Whether a collaboration's text ends with a colon, a heading over the names after it (`Tam\'as Cs\"org\H{o}$^{4}$,
158/// for the TOTEM Collaboration:\\ G.~Antchev …`, 1204.5689; `JLQCD Collaboration: S.~Aoki$^1$, …`, hep-lat0212039).
159pub fn collaboration_heads(text: &str) -> bool { text.trim_end().ends_with(':') }

161/// The words a remark on a name opens with when it names whom the author speaks for (`for the ATLAS Collaboration`, `on
162/// behalf of the CMS Team`, `representing the ATLAS Pixel Collaboration` — a call of its own under the names,
163/// hep-ex9903035, as [`collaboration_core`]'s framing reads it), lowercased with the space after them:
164/// the reading's list.
165pub(in crate::base_utilities) const SPEAKS_FOR: [&str; 3] =
166  ["for ", "on behalf of ", "representing "];

168/// The [`NON_NAME_WORDS`] that name a collaboration as the last word of its name (`the ATLAS Collaboration`, `the GAPS
169/// Consortium`): read so in a names line or on a line of its own ([`reads_as_collaboration`]).
170pub(in crate::base_utilities) const COLLABORATION_WORDS: &[&str] =
171  &["collaboration", "collaborations", "consortium", "consortia"];

173/// The [`NON_NAME_WORDS`] that name a team, a group or an experiment as the last word of its name, which may as well be
174/// an institution's (`Speech Group`, `Google Brain Team`): a collaboration only where the line frames it so — after
175/// "the", "for", "on behalf of", in brackets — or a names line lists it with "and" (`… and LHD experimental group`,
176/// 1711.07097; [`reads_as_collaboration`]).
177pub(in crate::base_utilities) const TEAM_WORDS: &[&str] = &[
178  "team",
179  "teams",
180  "group",
181  "groups",
182  "experiment",
183  "experiments",
184];

186/// How a remark frames the group it names: "for", "on behalf of", "representing" ([`SPEAKS_FOR`], [`Framing::Framed`];
187/// `for the Hall A DVCS collaboration`, 1502.03246) or a bare "the" after a list's "and" ([`Framing::The`]; `and the MAST
188/// Team`, 1609.09534), its last word a collaboration's or a team's ([`collaboration_word`]); `None` when it names none.
189pub(in crate::base_utilities) fn collaboration_remark(text: &str) -> Option<Framing> {
190  let lower = text.trim().to_lowercase();
191  let t = lower.strip_prefix("and ").unwrap_or(&lower).trim_start();
192  let (framing, rest) = match SPEAKS_FOR.iter().find_map(|p| t.strip_prefix(p)) {
193    Some(rest) => (Framing::Framed, rest),
194    None => (Framing::The, t.strip_prefix("the ")?),
195  };
196  // (a heading's colon is no part of the word: `for the TOTEM Collaboration:`, 1204.5689, is `collaboration_heads`'s)
197  let last = rest.split_whitespace().last()?;
198  (!last.ends_with(':') && collaboration_word(last)).then_some(framing)
199}

201/// Whether the text after a name's notes is a remark naming the group the author speaks for: "for" or "on behalf of",
202/// then a few words that name a group (`Eleni Petrakou\inst{1}\fnsep\thanks{…} for CAPP/IBS`, webofc, 1702.03664) —
203/// the name ended at its notes, so the remark is no part of it, a collaboration contact as a collaboration's remark is
204/// (the reading's remark, row C2; moved from `author_block/collaborations.rs`). A
205/// group's words are acronyms or slashed names (`CAPP/IBS`, `CMS`) or a team's word, with "the", "and", "of" between
206/// them; a remark naming anything else stays the name's, as before — a funding line or a
207/// number (`for NSF Grant 123`, `for Section Three`), a company (`for ACME Corp`), a person (`on behalf of Ed Eve`), a
208/// phrase going on in lowercase (`for discussions`), more than six words or an address (review rev106c).
209pub(in crate::base_utilities) fn speaks_for_after_notes(text: &str) -> bool {
210  let lower = text.trim().to_lowercase();
211  let Some(rest) = SPEAKS_FOR
212    .iter()
213    .find_map(|p| lower.starts_with(p).then(|| text.trim()[p.len()..].trim()))
214  else {
215    return false;
216  };
217  let words: Vec<&str> = rest.split_whitespace().collect();
218  let joining = |w: &str| ["the", "and", "of", "&"].contains(&w.to_lowercase().as_str());
219  // (an acronym, `CMS`, or names a slash joins, `CAPP/IBS`: two capitals or a slash between letters, no lowercase
220  // letter nor digit)
221  let acronym = |w: &str| {
222    let w = w.trim_matches(|c: char| !c.is_alphanumeric());
223    !w.chars().any(|c| c.is_lowercase() || c.is_numeric())
224      && (w.chars().filter(|c| c.is_uppercase()).count() >= 2 || w.contains('/'))
225  };
226  words.len() <= 6
227    && !rest.contains('@')
228    && words.iter().any(|w| !joining(w))
229    && words
230      .iter()
231      .all(|w| joining(w) || collaboration_word(w) || acronym(w))
232}
