15/// [`link_prints_an_institution`] on a link as it printed, `\href{target}{printed}`: the reading's link contact, its
16/// target and text (`resolver/blocks.rs`'s block links, review revFB2a finding 3).
17pub(in crate::base_utilities) fn href_prints_an_institution(target: &str, printed: &str) -> bool {
18  let mut line = vec![T_CS!("\\href"), T_BEGIN!()];
19  line.extend(ExplodeText!(target));
20  line.extend([T_END!(), T_BEGIN!()]);
21  line.extend(ExplodeText!(printed));
22  line.push(T_END!());
23  link_prints_an_institution(&Tokens::new(line))

53/// Whether `text` holds a word of [`LINK_WORDS`], in any case, singular or plural.
54pub(in crate::base_utilities) fn link_words(text: &str) -> bool {
55  text
56    .split(|c: char| !c.is_alphanumeric())
57    .map(str::to_lowercase)
58    .any(|word| {
59      let word = word
60        .strip_suffix('s')
61        .filter(|_| word.len() > 3)
62        .unwrap_or(&word);
63      LINK_WORDS.contains(&word)
64    })
65}

67/// Whether the web address `target` reaches a page below its site's root: a path after its host (`https://x.org/proj`,
68/// `github.com/a/b`), not the bare domain an institution's link is (`https://www.google.com`, `https://x.ai/`).
69fn target_has_path(target: &str) -> bool {
70  let rest = target.split_once("://").map_or(target, |(_, rest)| rest);
71  rest
72    .split_once('/')
73    .is_some_and(|(_, path)| !path.trim_matches(['/', '#', '?']).is_empty())
74}

76/// Whether `line` is a link line ([`is_link_line`]) to a web page, a URL contact of the author block
77/// (`url_contact_call`), by what the link says and where it goes, not by the shape of its words: not to a mailbox
78/// (`\textbf{Correspondence:} \href{mailto:…}{…}`, 2601.00240, 2605.18563, an address and no URL: an e-mail line,
79/// [`links_a_mailbox`]); a `\url`, or an `\href` printing a web address or a link word before or in its text
80/// ([`link_words`]: `\href{…}{Project Page}`, `Demo Code:~\href{…}{here}`, 2602.09509); not one printing an institution,
81/// that institution linked, an affiliation (`\href{https://launchsafe.com}{\textbf{LaunchSafe Research}}`, 2606.26479,
82/// the r5 re-review's u32, u33, u35), nor a person's name, that name (`\textit{Author:} \href{…}{Masashi Sekine}`,
83/// 2603.22058); else a link printing no word (an icon) or to a page below a site's root is one, and a link to a bare
84/// domain is the institution it prints (`\href{https://www.google.com}{Google}`, `\href{https://x.ai}{xAI}`, the r6 re-review's x10, w8).
85pub(in crate::base_utilities) fn is_url_line(line: &Tokens) -> bool {
86  let Some((at, end)) = link_line_link(line) else {
87    return false;
88  };
89  let v = &line.unlist_ref()[..end];
90  let (target, printed) = link_texts(v, at);
91  if links_a_mailbox(&target, &printed) {
92    return false;
93  }
94  if v[at] == T_CS!("\\url") || reads_as_web_address(&printed) {
95    return true;
96  }
97  let label = visible_name_text(&printed_words(&v[..at]));
98  let (open, close) = link_printed_argument(v, at);
99  let printed_tokens = Tokens::new(printed_words(&v[open..close]));
100  // (an institution holding a link word is that institution linked: `\href{…}{Data Science Institute}`, the r7
101  // re-review's finding 4, probe n6)
102  if reads_as_affiliation(&printed_tokens) {
103    return false;
104  }
105  if link_words(&printed) || link_words(&label) {
106    return true;
107  }
108  if name_count(&visible_name_text(printed_tokens.unlist_ref())) > 0 {
109    return false;
110  }
111  // (an icon, a link printing no word, is one: 2604.10930; and one printing a word that names no organization, `here`,
112  // the r7 re-review's finding 10)
113  !printed.chars().any(char::is_alphanumeric)
114    || target_has_path(&target)
115    || !names_an_organization(&visible_name_text(printed_tokens.unlist_ref()), &target)
116}

118/// Whether `line` is a link line printing an institution, no URL line ([`is_url_line`]): its printed text reads as an
119/// affiliation (`\href{https://launchsafe.com}{\textbf{LaunchSafe Research}}`, 2606.26479), or it links a bare domain
120/// and prints no name (`\href{https://www.google.com}{Google}`, the r6 re-review's x9, x10) — not a linked name
121/// (`Advisor: \href{…}{Carl Doe}`, the r6 re-review's w5).
122pub(in crate::base_utilities) fn link_prints_an_institution(line: &Tokens) -> bool {
123  let Some((at, end)) = link_line_link(line) else {
124    return false;
125  };
126  let v = &line.unlist_ref()[..end];
127  let (target, printed) = link_texts(v, at);
128  if is_url_line(line) || links_a_mailbox(&target, &printed) {
129    return false;
130  }
131  let (open, close) = link_printed_argument(v, at);
132  let printed = Tokens::new(printed_words(&v[open..close]));
133  let text = visible_name_text(printed.unlist_ref());
134  reads_as_affiliation(&printed)
135    || (name_count(&text) == 0
136      && names_an_organization(&text, &target)
137      && !target_has_path(&target))
138}

140/// Whether a bare domain's link text names its organization, as such a name is written — each word with a capital
141/// (`Google`, `xAI`) — or as the domain `target` it links spells it in a handle
142/// (`\href{https://www.deepseek.com}{deepseek-ai}`, the r8 re-review's finding 9, probe s14; not a homepage's bare
143/// `{annable}`, the 77y review's finding 4), not a word of the sentence around it (`\href{https://ann.org}{here}`, a
144/// link: the r7 re-review's finding 10, probes n13, p9).
145fn names_an_organization(text: &str, target: &str) -> bool {
146  let capitalized = !text.trim().is_empty()
147    && text
148      .split_whitespace()
149      .all(|word| word.chars().any(char::is_uppercase));
150  // (the domain's name, the label before its top-level one: "deepseek" of `https://www.deepseek.com`)
151  let host = target
152    .split_once("://")
153    .map_or(target, |(_, rest)| rest)
154    .split(['/', '?', '#'])
155    .next()
156    .unwrap_or_default()
157    .to_lowercase();
158  let labels: Vec<&str> = host.split('.').filter(|label| !label.is_empty()).collect();
159  let spelled: String = text
160    .chars()
161    .filter(|c| c.is_alphanumeric())
162    .collect::<String>()
163    .to_lowercase();
164  // (an organization's handle, written with a mark no word holds — `deepseek-ai` —, not a person's: a homepage's link
165  // printing the name its domain spells, `\href{https://www.annable.com}{annable}`, is a URL contact; the 77y review's
166  // finding 4, probes d1, d2)
167  // (the label's letters and digits, as the text's are read: `ml-group.org`, `{ml-group}`)
168  let label: String = labels
169    .len()
170    .checked_sub(2)
171    .map(|at| labels[at].chars().filter(|c| c.is_alphanumeric()).collect())
172    .unwrap_or_default();
173  let domain_named = label.chars().count() >= 3
174    && spelled.contains(label.as_str())
175    && text
176      .chars()
177      .any(|c| !c.is_alphabetic() && !c.is_whitespace());
178  capitalized || domain_named
179}

181/// Whether `text` is a bare e-mail address, a mailbox and no web page: a local part, an `@`, a domain with a dot, and
182/// no `/` (`binzhu@smu.edu.sg` as the ACL template's `\href{binzhu@smu.edu.sg}{binzhu@smu.edu.sg}` writes it, without
183/// `mailto:`; 2604.04847, 2605.29064, 2605.29893, 2606.01600) — not a handle (`@aable`, the r6 re-review's w21), nor a
184/// list of them (`{a,b}@x.edu` is no one address, [`email_addresses`] reads it: 2609.00588).
185pub(in crate::base_utilities) fn is_bare_address(text: &str) -> bool {
186  let Some((local, domain)) = text.trim().split_once('@') else {
187    return false;
188  };
189  !local.is_empty()
190    && !local.contains([',', ';', '{', '}', ' '])
191    && !text.contains('/')
192    && domain.split('.').filter(|part| !part.is_empty()).count() >= 2
193    && !domain.contains(['@', ' ', ','])
194}

196/// Whether a link to `target` printing `printed` reaches a mailbox: to `mailto:`, or a bare address
197/// ([`is_bare_address`]) either way, its target no web page (`\href{https://x.com/aable}{@aable}` is a profile's, the r6
198/// re-review's w21).
199pub(in crate::base_utilities) fn links_a_mailbox(target: &str, printed: &str) -> bool {
200  target.starts_with("mailto:")
201    || (!target.contains("://") && (is_bare_address(target) || is_bare_address(printed)))
202}

204/// Whether `text` prints a web address: a word with `://`, opening with `www.`, with a `/` between two letters or digits
205/// (`github.io/traj_evo_search`), or with a `.` between two whose last label is a top-level domain of two letters or
206/// more (`x.org`), and no `@` — not a run of dotted initials (`Edward A.G. Schuur`, 2601.06085; `T.J.C. Bezerra`,
207/// 2606.03745; the r7 re-review's finding 2), nor a version number (`v1.2`).
208pub(in crate::base_utilities) fn reads_as_web_address(text: &str) -> bool {
209  text.split_whitespace().any(|word| {
210    let between = |c: u8| {
211      word
212        .as_bytes()
213        .windows(3)
214        .any(|w| w[0].is_ascii_alphanumeric() && w[1] == c && w[2].is_ascii_alphanumeric())
215    };
216    let initials = word
217      .as_bytes()
218      .windows(3)
219      .any(|w| w[0].is_ascii_uppercase() && w[1] == b'.' && w[2].is_ascii_uppercase());
220    let domain = word
221      .trim_end_matches(|c: char| !c.is_alphanumeric())
222      .rsplit_once('.')
223      .is_some_and(|(_, last)| last.len() >= 2 && last.chars().all(|c| c.is_ascii_alphabetic()));
224    !word.contains('@')
225      && (word.contains("://")
226        || word.starts_with("www.")
227        || between(b'/')
228        || (between(b'.') && domain && !initials))
229  })
230}

232/// Where the one link of a link line ([`is_link_line`]) stands among its tokens, from its command to the end of its
233/// arguments; `None` for any other line.
234pub(in crate::base_utilities) fn link_line_link(line: &Tokens) -> Option<(usize, usize)> {
235  let v = line.unlist_ref();
236  let (start, end) = (leading_spacing_end(v), spacing_end(v));
237  // (a line of spacing alone is no link: 2510.07961, 2607.21993)
238  if start >= end {
239    return None;
240  }
241  let v = &v[start..end];
242  let mut links = 0;
243  let mut link = (0, 0);
244  // (the last visible character of a label before the link)
245  let mut label: Option<char> = None;
246  let mut i = 0;
247  while i < v.len() {
248    let t = v[i];
249    if matches!(
250      t.get_catcode(),
251      Catcode::SPACE | Catcode::BEGIN | Catcode::END
252    ) || is_spacing_command(&t)
253      || is_spacing(&t)
254      || name_text_command(&t)
255      || t == T_CS!("\\texttt")
256    {
257      i += 1;
258    } else if let Some((end, _)) = non_printing_end(v, i) {
259      // (a colour, a font, a scale, an icon the link is set in or beside prints no word: `{\color{orange}\textbf{\url{…}}}`,
260      // `\textcolor{blue}{\url{…}}`, [`NON_PRINTING_ARGUMENTS`])
261      i = end.max(i + 1);
262    } else if t == T_CS!("\\href") || t == T_CS!("\\url") {
263      if label.is_some_and(|last| last != ':') {
264        return None;
265      }
266      // (a link that prints nothing is no link of the line: 2601.02680's `\href{…}{…} \url{}`)
267      let (open, close) = link_printed_argument(v, i);
268      if v[open..close].iter().all(|t| {
269        matches!(
270          t.get_catcode(),
271          Catcode::BEGIN | Catcode::END | Catcode::SPACE
272        )
273      }) {
274        i = close.max(i + 1);
275        continue;
276      }
277      links += 1;
278      let at = i;
279      // (its arguments, the spaces before each skipped as TeX skips them: `\href{…}⏎{…}`, 2605.06223; none after a
280      // bare `\href` that ends the line)
281      let argument = |at: usize| {
282        let at = at
283          + v.get(at..).map_or(0, |rest| {
284            rest.iter().take_while(|t| **t == T_SPACE!()).count()
285          });
286        skip_group(v, at)
287      };
288      i = argument(i + 1);
289      if t == T_CS!("\\href") {
290        i = argument(i);
291      }
292      link = (start + at, start + i.min(v.len()));
293    } else if links == 0 && matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER) {
294      label = t.with_str(|s| s.chars().last());
295      i += 1;
296    } else {
297      return None;
298    }
299  }
300  (links == 1).then_some(link)
301}

319/// Where the first two arguments after the link command at `at` in `v` stand, the spaces before each skipped as TeX
320/// skips them; empty past the end.
321pub(in crate::base_utilities) fn link_arguments(
322  v: &[Token],
323  at: usize,
324) -> ((usize, usize), (usize, usize)) {
325  let end = v.len();
326  let argument = |from: usize| -> (usize, usize) {
327    let from = from.min(end);
328    let open = from + v[from..].iter().take_while(|t| **t == T_SPACE!()).count();
329    (open, skip_group(v, open).clamp(open, end))
330  };
331  let first = argument(at + 1);
332  (first, argument(first.1))
333}

335/// The target and the printed text of the link at `at` in `v` (`\href{target}{printed}`; a `\url`'s argument is both),
336/// as their letters, other characters and spaces read, what prints no word aside ([`printed_words`]: a colour, an ORCID
337/// icon's `\includegraphics[scale=0.06]{orcid.pdf}`, a skip's length).
338pub(in crate::base_utilities) fn link_texts(v: &[Token], at: usize) -> (String, String) {
339  let text = |v: &[Token]| -> String {
340    let mut text = String::new();
341    for t in printed_words(v) {
342      match t.get_catcode() {
343        Catcode::LETTER | Catcode::OTHER => t.with_str(|s| text.push_str(s)),
344        Catcode::SPACE => text.push(' '),
345        Catcode::ACTIVE if t == T_ACTIVE!('~') => text.push(' '),
346        _ => {},
347      }
348    }
349    text.trim().to_string()
350  };
351  let ((open, close), _) = link_arguments(v, at);
352  let target = text(&v[open..close]);
353  let (open, close) = link_printed_argument(v, at);
354  (target, text(&v[open..close]))
355}

400  })

404/// `Contact: \href{mailto:…}{here}`; a target without `mailto:` that is a bare address, `\href{zz@x.org}{…}`, the ACL

406/// r8: 2601.00240, 2601.07974, 2604.23002, 2606.03014, 2605.18563; 2604.04847, 2605.29064, 2605.29893, 2606.01600) —
407/// that text as it prints without its font (`Correspondence:`), if any, and the link, kept a link, the address, whatever
408/// it prints (`\textbf{Correspondence:} \href{zz@x.org}{zz@x.org}`, 2604.04847: "Correspondence" is no e-mail label,
409/// so [`line_addresses`] reads no address of that line; a line it reads — addresses alone, or behind an e-mail label,
410/// `E-mail: \href{mailto:zz@x.org}{zz@x.org}` — is the Kind step's address line, read so before this; both named by
411/// the one label rule, [`entry_line_label`]). `None` for any other line. Perl reads no e-mail
412/// inside an affiliation: the link stays an affiliation's text (aas_support.sty.ltxml `\affil`). Review revZ18l F2,
413/// the home of the deleted `author_block/links.rs::mailto_line`; the Kind step's (`kind_entry_addresses`).
414pub(in crate::base_utilities) fn mailbox_link_line(
415  line: &Tokens,
416) -> Option<(Option<Tokens>, Tokens)> {
417  let (at, end) = link_line_link(line)?;
418  let v = line.unlist_ref();
419  let (target, printed) = link_texts(&v[..end], at);
420  if !links_a_mailbox(&target, &printed) {
421    return None;
422  }
423  // (the label as it prints, as `is_url_line` reads a link line's label)
424  let label = visible_name_text(&printed_words(&v[..at]));
425  let label = label.split_whitespace().collect::<Vec<_>>().join(" ");
426  Some((
427    (!label.is_empty()).then(|| Tokens::new(ExplodeText!(label.as_str()))),
428    Tokens::new(v[at..end].to_vec()),
429  ))
