7/// Best-effort content-kind classifier for a creator-scope `\thanks`, used ONLY to
8/// attach a semantic `ltx_thanks_<kind>` CSS hook (OXIDIZED_DESIGN #156) — it never
9/// affects core semantics. Keyword-matched over the flattened, lowercased note text.
10/// Order matters: correspondence and equal-contribution are checked before funding
11/// (a "supported by …" note is funding; "contributed equally" is contribution).
12/// The funding words are read as words (`funding_word`), not inside one: "erc" in "Perception" or "Commerce", "nsf"
13/// in "transfer", "fund" in "Fundamental" read an institution as a funder, so its line was no affiliation (2509.05368
14/// "Machine Perception", OXIDIZED_DESIGN_DIVERGENCES #880).
15/// "Deceased", as a word, is an annotation on the author of the same kind as "Corresponding author" (FIDELITY_CHARTER §4,
16/// user ruling 2026-11-02): a legend line `$^\dagger$Deceased` is its carriers' note, `ltx_thanks_deceased` (0708.2630's
17/// `$^*$Deceased`, 1003.2628's `\address{{\addrdagger}Deceased on …}`, 1510.03948's `$^\dagger$deceased`).
18/// Witnesses: arXiv 2512.24601 (correspondence), 1510.02728 (funding),
19/// 2506.06941 "Equal contribution", 2609.32870 "Equal advising." (contribution). Explicitly best-effort.
20/// An author's part in the work is a contribution too (batch 103c, census129 cluster 6): a project role, read as a
21/// phrase of whole words ([`project_role`]: "†Project lead", 2609.18514, 2609.19719; "Joint first authors.",
22/// 2608.22772; "Core Contribution", 2408.15240), and the work's own circumstances where they open the text ([`opens_with_work_done`]:
23/// "Work done during an internship", 2106.09776, 2502.02407) — not "Facebook (work done while at JHU/APL)", an
24/// institution the author is at (1702.02684).
25///
26/// The role readings are the ones an institution's name can hold ("Project Lead Institute, Univ D", "Core Contributor
27/// Network, Paris", "Senior Author Program, Acme Corp"), so a text reads by them only where every piece of it is a
28/// legend, as the Kind step reads an entry with no carrier's name in it ([`entry_legend_kind`]); else it reads as it
29/// would without them — one gate for every reader of the kind (the kernel's line route `queue_digest::legend_note_kind`,
30/// the frontmatter observer `resolver::is_legend_text`, the line readers' `== "note"` tests, a `\thanks`'s CSS kind),
31/// review rev103c S1. "Project lead: Yi Yang" (2602.19190) is left to the Kind step, which knows its carriers.
32pub(in crate::base_utilities) fn classify_thanks(text: &str) -> &'static str {
33  match thanks_kind(text, true) {
34    "contribution"
35      if role_contribution(text) && entry_legend_kind(&printed_words(text), &[]).is_none() =>
36    {
37      thanks_kind(text, false)
38    },
39    kind => kind,
40  }
41}

43/// A note's text as it prints, for [`classify_thanks`]'s gate: a caller's TeX source (`$^{\dagger}$Project leader.`, a
44/// `\thanks`'s untexed tokens) without its control words and the math, group and tie characters around them — the
45/// pieces then read are words, not `^{\dagger`.
46fn printed_words(text: &str) -> String {
47  let mut out = String::with_capacity(text.len());
48  let mut chars = text.chars().peekable();
49  while let Some(c) = chars.next() {
50    match c {
51      '\\' => {
52        // (a control word and the letters of its name; a control symbol, `\,`, its one character)
53        if chars.peek().is_some_and(|n| n.is_ascii_alphabetic()) {
54          while chars.peek().is_some_and(|n| n.is_ascii_alphabetic()) {
55            chars.next();
56          }
57        } else {
58          chars.next();
59        }
60        out.push(' ');
61      },
62      '$' | '^' | '_' | '{' | '}' | '~' => out.push(' '),
63      _ => out.push(c),
64    }
65  }
66  out
67}

69/// [`classify_thanks`]'s reading, with an author's part in the work read as a contribution when `roles`
70/// ([`project_role`], [`opens_with_work_done`]) whatever else the text names — the Kind step's reading of its pieces
71/// ([`entry_legend_kind`]).
72fn thanks_kind(text: &str, roles: bool) -> &'static str {
73  let t = text.to_lowercase();
74  if t.contains("correspond") {
75    "correspondence"
76  // (an equal share of the work, contributed or advised: "Equal advising.", "equal supervision", "advised equally" —
77  // 2609.32870's legend line, a note as "Equal contribution" is, review rev94a S4; the share a phrase, the two words
78  // side by side: not "I thank my advisor for equally generous support", nor the "Equal Opportunity Advisory Board",
79  // review rev94a r2 S1)
80  } else if (t.contains("equal") && t.contains("contribut"))
81    || equal_share(&t)
82    || (roles && {
83      // (the role read in the words the note prints, its caller's control words and math out: the frontmatter
84      // observer's writer reads a legend's untexed tokens, `\textsuperscript{$\star$}Work performed during …`,
85      // 2503.04130: 107c)
86      let printed = printed_words(&t);
87      project_role(&printed) || opens_with_work_done(&printed)
88    })
89  {
90    "contribution"
91  } else if t
92    .split(|c: char| !c.is_alphanumeric())
93    .any(|w| w == "deceased")
94  {
95    "deceased"
96  } else if t.contains("now at") || t.contains("present address") || t.contains("current address") {
97    "address"
98  } else if t.split(|c: char| !c.is_alphanumeric()).any(funding_word) {
99    "funding"
100  } else {
101    "note"
102  }
103}

105/// Whether a note's visible text opens with the word "deceased" (`†Deceased 3 March 2021; formerly at Univ C`): an
106/// annotation on the author even where an institution follows it, while a text naming an institution with the word
107/// (`Department of Deceased Estates, Univ C`) is an affiliation — the reading of the kernel's line route
108/// (`queue_digest::legend_note_kind`) and of the frontmatter observer (`is_legend_text`) alike (review rev100a S2, probes
109/// e1, e2, d9).
110pub(in crate::base_utilities) fn opens_with_deceased(text: &str) -> bool {
111  text.split_whitespace().next().is_some_and(|word| {
112    word
113      .trim_matches(|c: char| !c.is_alphanumeric())
114      .eq_ignore_ascii_case("deceased")
115  })
116}

118/// Whether a line's text opens a dedication, the marks before it aside (`insert::kind_dedication_lines`, 107c): "Dedicated
119/// to …", "This work (paper, article, thesis, volume) is dedicated to …", "To the memory of …", "In memory of …". Read
120/// by the frontmatter observer too: a dedication behind a name's symbol is its carriers' legend
121/// (`resolver::is_legend_text`, review rev107c S1).
122pub(in crate::base_utilities) fn opens_a_dedication(text: &str) -> bool {
123  let words: Vec<String> = text
124    .trim_start_matches(|c: char| !c.is_alphabetic())
125    .split(|c: char| !c.is_alphanumeric())
126    .filter(|w| !w.is_empty())
127    .map(str::to_lowercase)
128    .collect();
129  let words: Vec<&str> = words.iter().map(String::as_str).collect();
130  matches!(
131    words.as_slice(),
132    ["dedicated", "to", _, ..]
133      | [
134        "this",
135        "work" | "paper" | "article" | "thesis" | "volume" | "note",
136        "is",
137        "dedicated",
138        "to",
139        _,
140        ..
141      ]
142      | ["to", "the", "memory", "of", _, ..]
143      | ["in", "memory", "of", _, ..]
144  )
145}

147/// Whether a lowercased note says an equal share of advising: "equal" or "equally" next to a word of advising,
148/// supervising or mentoring, either side ("equal advising", "supervised equally") — [`classify_thanks`].
149fn equal_share(text: &str) -> bool {
150  let words: Vec<&str> = text
151    .split(|c: char| !c.is_alphanumeric())
152    .filter(|w| !w.is_empty())
153    .collect();
154  let equal = |w: &str| w == "equal" || w == "equally";
155  let shared = |w: &str| {
156    ["advis", "supervis", "mentor"]
157      .iter()
158      .any(|p| w.starts_with(p))
159  };
160  words
161    .windows(2)
162    .any(|pair| (equal(pair[0]) && shared(pair[1])) || (shared(pair[0]) && equal(pair[1])))
163}

167  text
168    .split(|c: char| !c.is_alphanumeric())
169    .filter(|w| !w.is_empty())

173/// Whether a lowercased note names an author's role in the project, as a phrase of whole words ([`classify_thanks`]):
174/// "project lead" (or "leader", "leads"), "joint first author(s)", "co-first author(s)", "joint co-author(s)" and "joint
175/// senior co-author(s)" (2403.05680: 107c), "core contribution" or "core contributor(s)", "senior author(s)" or "senior
176/// authorship" ("these authors share senior authorship", 2503.10710) — not "Project Leadership Institute", nor "Core
177/// Facility" — and a note that is "first author(s)" and nothing else (`\textsuperscript{*}First author`, 2609.03429:
178/// 107c; "The first author is supported by …" is funding).
179fn project_role(text: &str) -> bool {
180  let words = note_words(text);
181  let author = |w: &str| w == "author" || w == "authors";
182  words.windows(2).any(|pair| {
183    (pair[0] == "project" && matches!(pair[1], "lead" | "leader" | "leads" | "leaders"))
184      || (pair[0] == "core" && matches!(pair[1], "contribution" | "contributor" | "contributors"))
185      || (pair[0] == "senior" && matches!(pair[1], "author" | "authors" | "authorship"))
186  }) || words
187    .windows(3)
188    .any(|three| matches!(three[0], "joint" | "co") && three[1] == "first" && author(three[2]))
189    || words
190      .windows(3)
191      .any(|three| three[0] == "joint" && three[1] == "co" && author(three[2]))
192    || words
193      .windows(4)
194      .any(|four| four[0] == "joint" && four[1] == "senior" && four[2] == "co" && author(four[3]))
195    || matches!(words.as_slice(), [first, who] if *first == "first" && author(who))
196}

204/// Whether a lowercased note opens with the circumstances of the author's work, behind any mark: "work done during …",
205/// "work done while …", "work was done during …", "work done as an intern", "work performed during …" (2503.04130: 107c)
206/// ([`classify_thanks`]) — the opening words only, as [`opens_with_deceased`] reads its word: "Facebook (work done while
207/// at JHU/APL)" names the institution first (1702.02684).
208fn opens_with_work_done(text: &str) -> bool {
209  let words = note_words(text);
210  let rest = match words.as_slice() {
211    ["work", "was", "done" | "performed", rest @ ..]
212    | ["work", "done" | "performed", rest @ ..] => rest,
213    _ => return false,
214  };
215  matches!(rest.first(), Some(&("during" | "while" | "as")))
216}

218/// The kind of an affiliation-kind entry whose whole text is a legend — annotations on its authors, with no institution
219/// (FIDELITY_CHARTER §4): correspondence, a contribution (an equal share, a project role, the work's circumstances) or an
220/// author deceased ([`classify_thanks`]). The text is read without its addresses and their e-mail labels (`ann@x.org`,
221/// "Email:", [`is_email_label`]: its e-mails stay the entry's), in pieces parted at its marks, brackets, commas, colons,
222/// semicolons and lines ([`LEGEND_SEPARATORS`]; `|`, a line break as the caller writes it): "Equal contribution.|*
223/// Correspondence:" (2609.03342) is two legends, "Corresponding author: Ann Able" a legend and the name it gives. Every
224/// piece is a legend that names no institution by its words ([`reads_as_affiliation`]: "Department of Equal Contribution
225/// Studies, Univ A" is one; "Deceased" opening it, [`opens_with_deceased`], is a legend whatever follows, as the line
226/// route and the frontmatter observer read it), a name of one of its `carriers` (the personnames holding the entry, as
227/// printed: [`spells_a_carrier`]) or initials ("(Y.W.)"), or prints one letter at most (the "(s)" of "corresponding
228/// author(s): Xiaofan Xu, Hua Zhu", 2609.08556), and one piece at least is a legend. `None` for any other text: a mixed
229/// line is the affiliation it names ("DeepAuto.ai († : Equal advising)", 2609.34327; "ByteDance *equal contribution",
230/// 2404.02132; "Facebook (work done while at JHU/APL)", 1702.02684; "Equal contribution, Google DeepMind", a company's
231/// name shaped as a person's, review rev103c S2), and an address change and funding stay as the call declares them (an
232/// institution is named for its funder as often). The kind is the whole text's ([`classify_thanks`]: correspondence
233/// first). The frontmatter API's Kind step for a legend (`insert::kind_legend_contacts`, OXIDIZED_DESIGN_DIVERGENCES
234/// #961).
235pub(in crate::base_utilities) fn entry_legend_kind(
236  text: &str,
237  carriers: &[String],
238) -> Option<&'static str> {
239  if opens_with_deceased(text) {
240    return Some("deceased");
241  }
242  let text = text
243    .split_whitespace()
244    .filter(|word| {
245      !word.contains('@')
246        && !is_email_label(word.trim_matches(|c: char| !c.is_alphanumeric() && c != '-'))
247    })
248    .collect::<Vec<_>>()
249    .join(" ");
250  let legend = |piece: &str| {
251    let affiliation = || reads_as_affiliation(&Tokens::new(ExplodeText!(piece)));
252    match thanks_kind(piece, true) {
253      "correspondence" | "contribution" => !affiliation(),
254      "deceased" => opens_with_deceased(piece) || !affiliation(),
255      _ => false,
256    }
257  };
258  let mut legends = 0;
259  for piece in text.split(|c: char| LEGEND_SEPARATORS.contains(c)) {
260    let piece = piece.trim();
261    // (a piece of one letter at most is none — the plural's "(s)" of "corresponding author(s):", 2609.08556)
262    if piece.chars().filter(|c| c.is_alphabetic()).count() < 2 {
263      continue;
264    }
265    // (a name's initials, "Y.W.", "A.-M. K.", are a name: each capital followed by its period — not "GEELY.")
266    let initials = {
267      let letters: Vec<char> = piece.chars().filter(|c| !matches!(c, '-' | ' ')).collect();
268      letters.len() >= 2
269        && letters
270          .chunks(2)
271          .all(|pair| pair.len() == 2 && pair[0].is_uppercase() && pair[1] == '.')
272    };
273    if legend(piece) {
274      legends += 1;
275    } else if !initials
276      && (!spells_a_carrier(piece, carriers)
277        || reads_as_affiliation(&Tokens::new(ExplodeText!(piece))))
278    {
279      return None;
280    }
281  }
282  match thanks_kind(&text, true) {
283    kind @ ("correspondence" | "contribution" | "deceased") if legends > 0 => Some(kind),
284    _ => None,
285  }
286}

288/// Whether a piece of a legend's text spells one of its `carriers` by name ([`entry_legend_kind`]): each of its words of
289/// two letters or more is a word of one carrier's name, as printed (case aside; its marks, digits and initials not
290/// read), and it holds one at least — "Ann Able", "A. King" for Alice King; not "Google DeepMind", "Mountain View".
291fn spells_a_carrier(piece: &str, carriers: &[String]) -> bool {
292  let words = |text: &str| -> Vec<String> {
293    text
294      .split(|c: char| !c.is_alphabetic())
295      .filter(|w| w.chars().count() >= 2)
296      .map(str::to_lowercase)
297      .collect()
298  };
299  let spelled = words(piece);
300  !spelled.is_empty()
301    && carriers.iter().any(|carrier| {
302      let name = words(carrier);
303      spelled.iter().all(|word| name.contains(word))
304    })
305}

311/// Whether a lowercased word of a note names funding ([`classify_thanks`]): "support" and its forms ("supported",
312/// "supporting"), "grant" and "grants", "fund" and its forms ("funded", "funding", "funds"; not "fundamental"), a
313/// funder's acronym on its own (NSF and China's NSFC, NIH, ONR, DARPA, ERC, and the funders the in-word reading also
314/// caught: Switzerland's SNSF, Belgium's FNRS, Canada's NSERC, the UK's SERC — "SNSF Postdoctoral Fellow",
315/// 2507.18869; "Chercheur qualifié FNRS", cond-mat0504758) or its grant number glued to it ("nsf1234").
316fn funding_word(word: &str) -> bool {
317  const ACRONYMS: [&str; 10] = [
318    "nsf", "nsfc", "nih", "onr", "darpa", "erc", "snsf", "fnrs", "nserc", "serc",
319  ];
320  word.starts_with("support")
321    || matches!(word, "grant" | "grants" | "granted")
322    || (word.starts_with("fund") && !word.starts_with("fundament"))
323    || ACRONYMS.iter().any(|acronym| {
324      word
325        .strip_prefix(acronym)
326        .is_some_and(|rest| rest.chars().all(|c| c.is_ascii_digit()))
327    })
328}

330/// Does `content` start with a letter or number superscript mark (`$^{1}$Univ A`, `$^a$…`, `\textsuperscript{b}…`):
331/// a marked affiliation list, its lines labelled by their marks? A symbol (`$^{*}$Corresponding author`) is a note's
332/// mark, and a superscript later in the line (`Laboratory for $^{3}$He`) is the affiliation's own text.
333pub fn leads_with_mark(content: &Tokens) -> bool {
334  // (past a leading font switch or group: `\affiliation{\it $^1$ Key Laboratory …}`, 0808.2763)
335  let switch = mark_switch;
336  // (a mark spelled otherwise — `${\,\!}^1$`, `\( ^{1} \)`, `\mbox{${\ }^{1}$}` — as the shared reader reads it,
337  // `relocate::spaced_mark_at`: 2112.13115, 0807.4407)
338  let all = content.unlist_ref();
339  if let Some(lead) = all
340    .iter()
341    .position(|t| *t != T_SPACE!() && *t != T_BEGIN!() && *t != T_CS!("\\noindent") && !switch(t))
342    && let Some((operand, _)) = spaced_mark_at(all, lead)
343  {
344    return operand
345      .iter()
346      .find(|t| **t != T_BEGIN!() && **t != T_SPACE!() && !switch(t))
347      .is_some_and(|t| {
348        t.to_string()
349          .chars()
350          .next()
351          .is_some_and(char::is_alphanumeric)
352      });
353  }
354  let tokens: Vec<&Token> = content
355    .unlist_ref()
356    .iter()
357    .filter(|t| **t != T_SPACE!())
358    .skip_while(|t| **t == T_BEGIN!() || **t == T_CS!("\\noindent") || switch(t))
359    .collect();
360  let script = match tokens.as_slice() {
361    [t, rest @ ..] if is_mark_command(t) || **t == T_SUPER!() => rest,
362    [t, s, rest @ ..] if t.get_catcode() == Catcode::MATH && **s == T_SUPER!() => rest,
363    _ => return false,
364  };
365  script
366    .iter()
367    .find(|t| ***t != T_BEGIN!() && !switch(t))
368    .is_some_and(|t| {
369      t.to_string()
370        .chars()
371        .next()
372        .is_some_and(char::is_alphanumeric)
373    })
374}

376/// Does this token list *begin* with a NUMERIC affiliation superscript mark
377/// (`$^{1}…` / `$^1…` / `\textsuperscript{1}…`)? This is the signature of the
378/// arXiv "`\thanks` abuse" idiom (affiliations smuggled into an author
379/// `\thanks{...}` and linked to authors by a leading mark).
380///
381/// The mark must be a **digit**: affiliation linking is by number, so a numeric
382/// mark is the reliable abuse signal. Crucially, footnote-SYMBOL marks
383/// (`$^*$`, `$^\dagger$`, `$^\ddagger$`, `$^\S$`) and lettered marks head
384/// *legitimate* acknowledgements — corresponding-author / equal-contribution /
385/// present-address notes — which must stay `role=thanks`. Requiring a digit
386/// excludes them, so those notes are never re-routed into an affiliation that
387/// fails to link and is then discarded (i.e. no silent note loss). Operates on
388/// the string form so it is agnostic to the Semiverbatim catcode-freeze on
389/// `$`/`^`. (user-directed, 2026-07-07; reviewer-hardened.)
390pub(in crate::base_utilities) fn starts_with_affiliation_mark(content: &Tokens) -> bool {
391  let s = content.to_string();
392  let st = s.trim_start();
393  let after_opener = match st
394    .strip_prefix("$^")
395    .or_else(|| st.strip_prefix("\\textsuperscript"))
396  {
397    Some(rest) => rest,
398    None => return false,
399  };
400  // The superscript content may be braced (`{1}`) or bare (`1`); either way the
401  // first content character must be an ASCII digit.
402  let content_start = after_opener.strip_prefix('{').unwrap_or(after_opener);
403  content_start
404    .chars()
405    .next()
406    .is_some_and(|c| c.is_ascii_digit())
407}

409/// Does a mark lead `line` — no text before its first mark, looking inside a leading font command or group past the
410/// declarations that open it (`{\color{blue} $^1$Univ A, $^2$Univ B}`: the letters of `blue` are no name)?
411pub(in crate::base_utilities) fn marker_leads(line: &Tokens) -> bool {
412  // What the line prints from its first glyph on: the spacing and declarations before it print nothing, nor do the unit
413  // letters of their lengths, which read as a name before the mark (`\\ \vspace{0.1cm} $^{1}$ Univ A`: Univ A … read as
414  // authors, 2407.15815; `\kern2pt$^{1}$Univ A`).
415  let printed = |tokens: &[Token]| Tokens::new(tokens[leading_unprinted_end(tokens)..].to_vec());
416  let line = &printed(line.unlist_ref());
417  // (through each wrapper that opens the line and holds a mark, the wrappers a line keeps of the block's —
418  // `{\small{\it \vspace{0.2cm} $^{a}$ also at …}}`, a legend line of 1202.2048's `\begin{small} … {\it …}` once
419  // `wrapper_over_lines_as_lines` gives each line its wrappers: read past one wrapper only, the spacing's letters
420  // before the mark made the legend an author)
421  let mut probe = line.clone();
422  while let Some((_, inner, _)) = leading_wrapper(&probe) {
423    let inner = printed(inner.unlist_ref());
424    if author_mark_position(&inner).is_none() {
425      break;
426    }
427    probe = inner;
428  }
429  match author_mark_position(&probe) {
430    Some(p) => !name_precedes_marker(&probe, p),
431    // No mark inside the wrapper (`{}$^1$Univ A`, `\noindent{}$^1$…`): the whole line decides, as before.
432    None => author_mark_position(line).is_some_and(|p| !name_precedes_marker(line, p)),
433  }
434}

448/// The declarations among a line's unprinted lead that a cut piece carries, its spacing left out
449/// (`\small\vspace{1mm}\noindent\it` → `\small`, `\it`; [`leading_unprinted_end`], [`font_settings`]; 2407.15815).
450pub(in crate::base_utilities) fn lead_declarations(v: &[Token]) -> Vec<Token> {
451  let mut declarations = Vec::new();
452  let mut i = 0;
453  loop {
454    let start = i;
455    i += leading_spacing_end(&v[i..]);
456    let found = opening_declarations(&Tokens::new(v[i..].to_vec()));
457    i += found.len();
458    declarations.extend(font_settings(&found));
459    if i == start {
460      // (the space after them is no institution's: `{\color{blue} $^1$A, $^2$B}`)
461      while declarations.last() == Some(&T_SPACE!()) {
462        declarations.pop();
463      }
464      return declarations;
465    }
466  }
467}

469/// Of `found` ([`opening_declarations`]), the declarations that print nothing and set the font, size or colour
470/// ([`sets_font`]), with their arguments: what a cut piece carries, never a macro printing text (`\mydept`, `\ldots`)
471/// nor an alignment (revP4b4a2 B1, B2).
472fn font_settings(found: &[Token]) -> Vec<Token> {
473  let mut out = Vec::new();
474  let (mut keep, mut depth) = (false, 0usize);
475  for t in found {
476    match t.get_catcode() {
477      Catcode::BEGIN => depth += 1,
478      Catcode::END => depth = depth.saturating_sub(1),
479      Catcode::CS if depth == 0 => keep = sets_font(t),
480      Catcode::SPACE if depth == 0 => {
481        if !out.is_empty() {
482          out.push(*t);
483        }
484        continue;
485      },
486      _ => {},
487    }
488    if keep {
489      out.push(*t);
490    }
491  }
492  out
493}

540/// Where a line's unprinted lead ends: its spacing (`leading_spacing_end`) and its declarations
541/// (`opening_declarations`), as often as they interleave (`\vspace*{2mm}\noindent`, `\small\vspace{1mm}`). Public
542/// for authblk's `\affil` (latexml_package), which reads whether a line opens with a mark past it.
543pub fn leading_unprinted_end(v: &[Token]) -> usize {

545  loop {
546    let start = i;
547    i += leading_spacing_end(&v[i..]);
548    i += opening_declarations(&Tokens::new(v[i..].to_vec())).len();
549    if i == start {
550      return i;
551    }

555/// The declarations that open a group's content (`\small`, `\bfseries`, `\color{blue}` with its argument), up to its
556/// first mark, text, or a command over a braced argument (`\textbf{Ann Able}` is content, not a declaration: read
557/// as one, `\large\textbf{Ann Able}$^1$` became an affiliation line and lost its author).
558pub(in crate::base_utilities) fn opening_declarations(inner: &Tokens) -> Vec<Token> {
559  let v = inner.unlist_ref();
560  let mut out = Vec::new();
561  let mut i = 0;
562  while i < v.len() {
563    let t = v[i];
564    let braced = v
565      .get(i + 1)
566      .is_some_and(|n| n.get_catcode() == Catcode::BEGIN);
567    let with_arguments = FONT_DECLARATIONS_WITH_ARGUMENTS
568      .iter()
569      .any(|cs| t == T_CS!(*cs));
570    // (spacing that takes a length or a rule's dimensions is no declaration: [`leading_spacing_end`] reads it whole)
571    if [
572      "\\vspace",
573      "\\hspace",
574      "\\addvspace",
575      "\\vskip",
576      "\\hskip",
577      "\\kern",
578      "\\penalty",
579      "\\rule",
580    ]
581    .iter()
582    .any(|cs| t == T_CS!(*cs))
583    {
584      break;
585    }
586    if t == T_SPACE!() {
587      out.push(t);
588      i += 1;
589    } else if t.get_catcode() == Catcode::CS
590      && !is_mark_command(&t)
591      && (with_arguments || !braced)
592      // (a command over an optional argument is content too: `{\parbox[c]{\textwidth}{…}}`, 2504.03146)
593      && v.get(i + 1) != Some(&T_OTHER!("["))
594    {
595      out.push(t);
596      i += 1;
597      // its brace-group arguments
598      while i < v.len() && v[i].get_catcode() == Catcode::BEGIN {
599        let mut depth = 0usize;
600        while i < v.len() {
601          let a = v[i];
602          out.push(a);
603          i += 1;
604          match a.get_catcode() {
605            Catcode::BEGIN => depth += 1,
606            Catcode::END => {
607              depth -= 1;
608              if depth == 0 {
609                break;
610              }
611            },
612            _ => {},
613          }
614        }
615      }
616    } else {
617      break;
618    }
619  }
620  out
621}

623/// A line that opens (spaces aside) with `\cmd{…}` or a bare `{…}`: the command (none for a bare group), the group's
624/// inside, and the tokens after it. A leading `\textsuperscript{n}` is a mark, not a wrapper.
625pub(in crate::base_utilities) fn leading_wrapper(
626  line: &Tokens,
627) -> Option<(Option<Token>, Tokens, Vec<Token>)> {
628  let v = line.unlist_ref();
629  let start = v.iter().position(|t| *t != T_SPACE!())?;
630  let (cmd, open) = match v[start].get_catcode() {
631    Catcode::CS
632      if v
633        .get(start + 1)
634        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN) =>
635    {
636      if is_mark_command(&v[start]) {
637        return None;
638      }
639      (Some(v[start]), start + 1)
640    },
641    Catcode::BEGIN => (None, start),
642    _ => return None,
643  };
644  let mut depth = 0usize;
645  for (i, t) in v.iter().enumerate().skip(open) {
646    match t.get_catcode() {
647      Catcode::BEGIN => depth += 1,
648      Catcode::END => {
649        depth -= 1;
650        if depth == 0 {
651          return Some((
652            cmd,
653            Tokens::new(v[open + 1..i].to_vec()),
654            v[i + 1..].to_vec(),
655          ));
656        }
657      },
658      _ => {},
659    }
660  }
661  None
662}

712/// A macro that is a note command: it takes arguments and every one of them goes into a `\thanks` or `\footnote` in its
713/// body, so what it prints beside a name is that note's mark (`\newcommand{\blackthanks}[1]{\begingroup
714/// \renewcommand{\thefootnote}{…}\unskip \thanks{#1}\unskip\endgroup}`: `Yuke Peng\textsuperscript{1,}\blackthanks{Co-first
715/// authors.}\ ,`, 2506.03876) — or into one of the frontmatter's author annotations ([`NAME_ANNOTATORS`]: a binding's
716/// contact command, jheppub's and jinstpub's `\note{…}` = `\lx@add@contact[role=note]{…}`, whose "Corresponding
717/// author." read as the name's last word left the corresponding author's e-mail spelling no one, 2406.08712,
718/// 2603.01867). A macro that prints an argument itself (`\newcommand{\auth}[2]{#1\thanks{#2}}`) is none.
719pub(in crate::base_utilities) fn is_note_macro(cs: &Token) -> bool {
720  if cs.get_catcode() != Catcode::CS || lookup_conditional(cs).is_some() {
721    return false;
722  }
723  let Ok(Some(defn)) = lookup_definition(cs) else {
724    return false;
725  };
726  if !defn.is_expandable() || defn.get_num_args() == 0 {
727    return false;
728  }
729  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
730    return false;
731  };
732  let body = body.unlist_ref();
733  let mut notes = 0;
734  let mut i = 0;
735  while i < body.len() {
736    let t = body[i];
737    let annotator = t.with_str(|s| NAME_ANNOTATORS.contains(&s));
738    // (a class's address list, `\lx@add@contact@list{\op}{#1}` — jheppub's `\emailAdd`, iopart's `\ead` — makes a contact
739    // of each address, as an annotator does: its operation, then its list, the note's argument; `\emailAdd` inside
740    // `\author[a]{Ann Able \emailAdd{…}}` is Ann Able's e-mail, not an e-mail line of the name, rev78f probe j13)
741    if t == T_CS!("\\lx@add@contact@list")
742      && body
743        .get(i + 1)
744        .is_some_and(|n| n.get_catcode() == Catcode::BEGIN)
745    {
746      let list = skip_group(body, i + 1);
747      if body
748        .get(list)
749        .is_some_and(|n| n.get_catcode() == Catcode::BEGIN)
750      {
751        notes += 1;
752        i = skip_group(body, list);
753        continue;
754      }
755    }
756    if t == T_CS!("\\thanks") || t == T_CS!("\\footnote") || annotator {
757      let mut j = i + 1;
758      while body.get(j) == Some(&T_SPACE!()) {
759        j += 1;
760      }
761      // (an annotator's keyvals, `[role=note]`, are no argument of the name)
762      if annotator
763        && body.get(j) == Some(&T_OTHER!("["))
764        && let Some(close) = body[j..].iter().position(|u| *u == T_OTHER!("]"))
765      {
766        j += close + 1;
767      }
768      if body
769        .get(j)
770        .is_some_and(|n| n.get_catcode() == Catcode::BEGIN)
771      {
772        notes += 1;
773        i = skip_group(body, j);
774        continue;
775      }
776    }
777    if t.get_catcode() == Catcode::ARG {
778      return false;
779    }
780    i += 1;
781  }
782  notes > 0
783}

830/// Whether `cs` reads arguments by its definition: parameters of its own, or a parameterless macro whose expansion ends
831/// on a command that does or on an optional argument it hands on (tcolorbox's `\newtcbox{\linkpill}`, `\tcbox[…]`,
832/// which reads the box's text next: 2606.14188).
833pub(in crate::base_utilities) fn takes_arguments(cs: &Token) -> bool {
834  if cs.get_catcode() != Catcode::CS {
835    return false;
836  }
837  let Ok(Some(defn)) = lookup_definition(cs) else {
838    return false;
839  };
840  if defn.get_num_args() > 0 {
841    return true;
842  }
843  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
844    return false;
845  };
846  match body.unlist_ref().iter().rev().find(|t| **t != T_SPACE!()) {
847    Some(last) if *last == T_OTHER!("]") => true,
848    Some(last) if last.get_catcode() == Catcode::CS && last != cs => lookup_definition(last)
849      .ok()
850      .flatten()
851      .is_some_and(|d| d.get_num_args() > 0),
852    _ => false,
853  }

856/// Whether `cs` before a brace is the group's wrapper, which a piece cut inside the group takes again: the command reading
857/// it as its argument ([`takes_arguments`], a text font command) or a declaration setting its font ([`sets_font`]) —
858/// never a note, made twice, nor a parameterless macro printing text (`\univ{…}` with `\newcommand\univ{Univ}`); a macro
859/// with parameters that prints text around its argument is kept (revSBX2F: no duplication found).
860pub(in crate::base_utilities) fn wraps_group(cs: &Token) -> bool {
861  cs.get_catcode() == Catcode::CS
862    && (is_text_font_command(cs)
863      || sets_font(cs)
864      || (takes_arguments(cs)
865        && !is_note_macro(cs)
866        && !["\\thanks", "\\footnote", "\\footnotetext"]
867          .iter()
868          .any(|n| *cs == T_CS!(*n))))
869}

871/// The tokens opening the group `v[open]` again for a piece cut inside it: its wrapper ([`wraps_group`]), the brace and
872/// the declarations it opens with up to `upto` ([`lead_declarations`]; 2605.27539, 2605.00634, 2407.15815).
873pub(in crate::base_utilities) fn reopened(v: &[Token], open: usize, upto: usize) -> Vec<Token> {
874  let mut tokens: Vec<Token> = open
875    .checked_sub(1)
876    .map(|c| v[c])
877    .filter(wraps_group)
878    .into_iter()
879    .collect();
880  tokens.push(v[open]);
881  tokens.extend(lead_declarations(&v[open + 1..upto.max(open + 1)]));
882  tokens
