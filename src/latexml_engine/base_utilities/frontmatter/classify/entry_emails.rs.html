1//! The cut of an entry's class e-mail calls, which the frontmatter emitter writes each by its owner's label
2//! ([`entry_email_cut`], [`shared_call_item`]), and the text rules it reads a call by ([`is_call_separator`],
3//! [`is_whole_address`]); moved unchanged from `shared_row/` when the old row split (`share_row_contacts`, P27b) was
4//! deleted, the emitter writing a flush that holds a class's row contact (OXIDIZED_DESIGN_DIVERGENCES #901).

8/// The notes whose argument is no line of the contact: an e-mail in a `\thanks` stays in the note ([`email_calls_in`]).
9const NOTE_COMMANDS: [&str; 3] = ["\\thanks", "\\footnote", "\\footnotetext"];
10
11/// The separators that print nothing of their own before an e-mail call, which go with it: spaces, commas, semicolons,
12/// a line break, spacing, a space factor's `\@` (`USA;\@ Email: \email{…}`, 2602.18622).
13pub(in crate::base_utilities) fn is_call_separator(t: &Token) -> bool {
14  matches!(t.get_catcode(), Catcode::SPACE)
15    || *t == T_OTHER!(",")
16    || *t == T_OTHER!(";")
17    || *t == T_ACTIVE!('~')
18    || (t.get_catcode() == Catcode::CS
19      && t.with_str(|name| {
20        matches!(
21          name,
22          "\\\\" | "\\newline" | "\\quad" | "\\qquad" | "\\ " | "\\enspace" | "\\hfill" | "\\@"
23        )
24      }))
25}

27/// The end of the spacing at `tokens[from..to]`'s start that prints nothing of its own: separators
28/// ([`is_call_separator`]) and `\hspace{…}`, `\hspace*{…}`.
29fn past_spacing(tokens: &[Token], mut from: usize, to: usize) -> usize {
30  loop {
31    if from < to && is_call_separator(&tokens[from]) {
32      from += 1;
33    } else if from < to && tokens[from] == T_CS!("\\hspace") {
34      let mut arg = from + 1;
35      if tokens[arg..to].first() == Some(&T_OTHER!("*")) {
36        arg += 1;
37      }
38      if arg >= to || tokens[arg].get_catcode() != Catcode::BEGIN {
39        return from;
40      }
41      from = skip_group(tokens, arg).min(to);
42    } else {
43      return from;
44    }
45  }
46}

48/// The argument of the note command at `tokens[at]` ([`NOTE_COMMANDS`]), within `..to`: where its group opens (past
49/// the spaces and an optional `[…]`), and where it ends; `None` when `tokens[at]` is no note command.
50fn note_argument(tokens: &[Token], at: usize, to: usize) -> Option<(usize, usize)> {
51  let t = tokens.get(at)?;
52  if t.get_catcode() != Catcode::CS || !t.with_str(|name| NOTE_COMMANDS.contains(&name)) {
53    return None;
54  }
55  let mut next = at + 1;
56  while tokens.get(next) == Some(&T_SPACE!()) {
57    next += 1;
58  }
59  if tokens.get(next) == Some(&T_OTHER!("[")) {
60    next = tokens[next..to]
61      .iter()
62      .position(|t| *t == T_OTHER!("]"))
63      .map_or(to, |end| next + end + 1);
64  }
65  Some((next, skip_group(tokens, next).min(to)))
66}

68/// One e-mail call of an entry ([`email_calls_in`]): its span, the separators before it with it; where its
69/// command stands; the command; its argument; whether it is written inside a note of the contact (`\thanks{Mail me at
70/// \email{…}}`); the e-mail label printed before it, which its span takes ([`email_label_before`]).
71struct EmailCall {
72  start:   usize,
73  at:      usize,
74  end:     usize,
75  cs:      Token,
76  arg:     Vec<Token>,
77  in_note: bool,
78  label:   Option<Vec<Token>>,
79}
80
81/// The e-mail calls in `tokens[from..to]`: each class e-mail command ([`is_email_command`]) and its argument group, with
82/// the separators before it ([`is_call_separator`]); a note's are taken apart (`in_note`) when `notes` is set, else
83/// the note is passed over.
84fn email_calls_in(
85  tokens: &[Token],
86  from: usize,
87  to: usize,
88  notes: bool,
89  in_note: bool,
90  calls: &mut Vec<EmailCall>,
91) -> Result<()> {
92  let mut at = from;
93  while at < to {
94    let t = tokens[at];
95    if let Some((next, end)) = note_argument(tokens, at, to) {
96      if notes && next < end && tokens[next].get_catcode() == Catcode::BEGIN {
97        email_calls_in(tokens, next + 1, end - 1, notes, true, calls)?;
98      }
99      at = end;
100      continue;
101    }
102    if t.get_catcode() == Catcode::CS && is_email_command(&t)? {
103      let mut arg = at + 1;
104      while tokens.get(arg) == Some(&T_SPACE!()) {
105        arg += 1;
106      }
107      if arg < to && tokens[arg].get_catcode() == Catcode::BEGIN {
108        let end = skip_group(tokens, arg).min(to);
109        let mut start = at;
110        while !in_note && start > from && is_call_separator(&tokens[start - 1]) {
111          start -= 1;
112        }
113        // (and the label printed before it, which names the call's contacts: `Email: \email{…}`, 2102.12877)
114        let mut label_tokens = None;
115        if !in_note && let Some(label) = email_label_before(tokens, from, start) {
116          label_tokens = Some(tokens[label..start].to_vec());
117          start = label;
118          while start > from && is_call_separator(&tokens[start - 1]) {
119            start -= 1;
120          }
121        }
122        calls.push(EmailCall {
123          start,
124          at,
125          end,
126          cs: t,
127          arg: tokens[arg + 1..end - 1].to_vec(),
128          in_note,
129          label: label_tokens,
130        });
131        at = end;
132        continue;
133      }
134    }
135    at += 1;
136  }
137  Ok(())
138}

140/// An affiliation-kind entry whose class e-mail calls the frontmatter emitter places itself (the reading's inner
141/// contacts, `Resolver::entry_emails`, OXIDIZED_DESIGN_DIVERGENCES #980), as the emitter writes it: its `command` with
142/// each call's unit cut by the cut the old row split (`row_email_calls`, deleted at P27b) cut a row contact's by
143/// ([`unit_cut`], [`kept_without`], each call read by [`read_call`]) — the separators before it back to the text or unit
144/// before it (one at the content's start takes those after it: `\email{a}, Univ X`), the e-mail label it is named by
145/// (`Email: \email{…}`, [`email_label_before`]), the label of its own the rule recognises ([`call_label`]: `Corresponding
146/// author: \email{…}`, `\email{…} (Corresponding author)`) — so the entry keeps only its own text: `\institute{Univ X\\
147/// \email{p1@x.edu}, \email{p2@x.edu}}` under `\inst{1}` marks is "Univ X", it had kept "Univ X ," (`…; Email: ;`,
148/// `Unity Technologies Corresponding author:`). Each call stays where it was written, so it digests inside the entry and
149/// the Link step places its addresses among the entry's carriers (`insert::link_entry_addresses`); a labelled unit, and
150/// an unlabelled one after the first that reads as whole addresses, is written as the row's e-mail is, a label of its
151/// own in its content or its name, by the entry's `label` (`\lx@row@email`, `\lx@row@email@named`) — a later call in printed
152/// braces too (`\email{\{b2@x.edu\}}`, [`single_braced`]: it had stayed on the last author, review revZ18c S2); any other
153/// as written.
154/// With `only` (the entry's inner contacts written as items of their own, `\lx@frontmatter@itemonly`) the units are
155/// cut out whole, a label of the call's aside, which stays the entry's text — with `shared` (a `_share=authors` entry,
156/// llncs's sole `\institute`) the label goes with the call, its item carrying it ([`shared_call_item`]). A call inside
157/// a note stays in it. `None` when the content holds no class e-mail call outside a note.
158/// Witnesses (Z18b's llncs reroute, the units left in the entry): 1709.03776, 1709.08027, 2006.03363, 2008.02173,
159/// 2009.11534, 2102.12877, 2207.10888, 2406.08136, 2407.18381, 2410.06985, 2501.02196, 2509.05139, 2510.06692,
160/// 2609.00838, cs0606057.
161pub(in crate::base_utilities) fn entry_email_cut(
162  command: &Tokens,
163  label: &str,
164  only: bool,
165  shared: bool,
166) -> Result<Option<Tokens>> {
167  let tokens = command.unlist_ref();
168  let Some((open, close)) = content_group(tokens) else {
169    return Ok(None);
170  };
171  let mut calls: Vec<EmailCall> = Vec::new();
172  email_calls_in(tokens, open + 1, close, false, false, &mut calls)?;
173  if calls.iter().all(|call| call.in_note) {
174    return Ok(None);
175  }
176  let (inner_open, inner_close) = wrapped_span(tokens, open + 1, close);
177  let lines = content_lines(tokens, inner_open, inner_close);
178  let first_text = first_text_line(tokens, &lines);
179  let label_tokens = || mouth::tokenize_internal(TeXString::assembled(label.to_string()));
180  // (each unit: its span, and what it is written as)
181  let mut units: Vec<(usize, usize, Vec<Token>)> = Vec::new();
182  for (place, &(_, start, end)) in lines.iter().enumerate() {
183    for call in calls
184      .iter()
185      .filter(|call| !call.in_note && (start..end).contains(&call.at))
186    {
187      let floor = units.last().map_or(inner_open, |unit| unit.1);
188      let cut = unit_cut(tokens, call, (floor, start, end), Some(place) == first_text);
189      // (an address the reader reads whole, no template's: the entry's to place, its call after the first written by the
190      // entry's label, which it would not inherit — a call digested after another inside an entry takes no label
191      // (Base_Utility.pool.ltxml:497-501 inherits from the last stub only), so it had stayed on the last author as their
192      // own call: `\institute{Univ X\\ \email{a1@x.edu}, \email{b2@x.edu}, \email{c3@x.edu}}` gave Cid Cole b2@ and c3@)
193      let reads_whole = read_call(call)?
194        .is_some_and(|list| list.iter().all(|address| !is_template_address(address)));
195      let written = if only {
196        Vec::new()
197      } else if !cut.labelled() && call.label.is_none() && (units.is_empty() || !reads_whole) {
198        tokens[call.at..call.end].to_vec()
199      } else {
200        let whole = cut.labelled_address(tokens, call)?;
201        match cut.name(call) {
202          Some(name) => Invocation!(T_CS!("\\lx@row@email@named"), vec![
203            Some(label_tokens()),
204            Some(Tokens::new(name.clone())),
205            Some(Tokens::new(whole))
206          ])
207          .unlist(),
208          None => Invocation!(T_CS!("\\lx@row@email"), vec![
209            Some(label_tokens()),
210            Some(Tokens::new(whole))
211          ])
212          .unlist(),
213        }
214      };
215      // (with `only`, the address written apart, its item named by no label of the entry's: a label stays in the entry,
216      // the call alone leaving it — 2606.02634's `Norway\\ Correspondence: \email{elias.stenhede@ahus.no}`, whose address
217      // is Elias Stenhede's in another entry; the label had gone with the call, a word lost)
218      // — not a shared row entry's, whose labels go with the call to its item (`shared_call_item`), as the row's split
219      // gave its unit to the owner or the block, never every author's (85b's `llncs_email_label_goes_with_the_call`, 104d)
220      if only && !shared && (cut.labelled() || call.label.is_some()) {
221        units.push((call.at, call.end, written));
222        continue;
223      }
224      units.push((cut.start, cut.end, written));
225    }
226  }
227  if units.is_empty() {
228    return Ok(None);
229  }
230  // (a line break among the separators a unit takes stays where text the entry keeps comes after the unit: that text
231  // stays on a line of its own — `Shanghai AI Laboratory \\ \email{…} \url{…}`, 2310.01324, whose link line the
232  // entry's Kind step reads apart)
233  let taken = |at: usize| {
234    units
235      .iter()
236      .any(|(start, end, _)| (*start..*end).contains(&at))
237  };
238  let kept_after =
239    |end: usize| (end..inner_close).any(|at| !taken(at) && !is_call_separator(&tokens[at]));
240  let breaks: Vec<Option<Token>> = units
241    .iter()
242    .map(|(start, end, _)| {
243      tokens[*start..*end]
244        .iter()
245        .find(|t| is_line_break(t))
246        .copied()
247        .filter(|_| kept_after(*end))
248    })
249    .collect();
250  let edits: Vec<CutEdit> = units
251    .into_iter()
252    .zip(breaks)
253    .map(|((start, end, written), line_break)| {
254      (
255        start,
256        end,
257        line_break.into_iter().chain(written).collect(),
258        true,
259      )
260    })
261    .collect();
262  Ok(Some(Tokens::new(kept_without(
263    tokens,
264    edits,
265    (inner_open, inner_close),
266  ))))
267}

269/// A shared row entry's e-mail item as the row's split wrote its unit (`row_email_calls`' `call_unit`, deleted at
270/// P27b; [`unit_cut`]): the e-mail label printed before the call, which names it, and the address as the class sets it with the label of its
271/// own before and after it ([`UnitCut::labelled_address`]).
272type SharedItem = (Option<String>, Vec<Token>);
273
274/// The item of a shared row entry's e-mail call (llncs's sole `\institute`, `_share=authors`, its inner contacts written
275/// as items of their own, [`entry_email_cut`]'s `only` with `shared`) carrying the labels [`entry_email_cut`] cuts from the
276/// entry with the call, as the row's split gave its unit: the e-mail label printed before the call, its name
277/// (`USA;\@ Email: \email{\{aable,bbaker,ccole\}@a.edu}`, each part named "Email: ", 104d; 2102.12877, 2602.18622), and
278/// the label of its own the rule recognises ([`call_label`]: `Corresponding author email: \email{zz@x.edu}`, the 85b
279/// re-review's s23; `Corresponding author:`, 2410.06985), in its content — to the address's owner or the block, never
280/// every author's. `written` is the item's e-mail call as the emitter writes it (the call, or one address of its list);
281/// the call is the one of `content` whose addresses hold `address` ([`read_call`]), else its only call. `None` when no
282/// call is found, or it carries no label.
283pub(in crate::base_utilities) fn shared_call_item(
284  content: &Tokens,
285  address: &str,
286  written: &Tokens,
287) -> Result<Option<SharedItem>> {
288  let tokens = content.unlist_ref();
289  let mut calls: Vec<EmailCall> = Vec::new();
290  email_calls_in(tokens, 0, tokens.len(), false, false, &mut calls)?;
291  calls.retain(|call| !call.in_note);
292  let (inner_open, inner_close) = wrapped_span(tokens, 0, tokens.len());
293  let lines = content_lines(tokens, inner_open, inner_close);
294  let first_text = first_text_line(tokens, &lines);
295  let wanted = address.trim().to_lowercase();
296  let only_call = calls.len() == 1;
297  let mut floor = inner_open;
298  for (place, &(_, start, end)) in lines.iter().enumerate() {
299    for call in calls.iter().filter(|call| (start..end).contains(&call.at)) {
300      let cut = unit_cut(tokens, call, (floor, start, end), Some(place) == first_text);
301      floor = cut.end;
302      let holds =
303        read_call(call)?.is_some_and(|list| list.iter().any(|a| a.to_lowercase() == wanted));
304      if !holds && !only_call {
305        continue;
306      }
307      let name = cut.name(call).map(|name| {
308        let printed = s!("{}", Tokens::new(name.clone()).untex_string());
309        printed.trim().to_string()
310      });
311      if name.is_none() && !cut.labelled() {
312        return Ok(None);
313      }
314      // (the written call's own address, the one of its list the item is)
315      let item_tokens = written.unlist_ref();
316      let mut own: Vec<EmailCall> = Vec::new();
317      email_calls_in(item_tokens, 0, item_tokens.len(), false, false, &mut own)?;
318      let Some(item) = own.into_iter().next() else {
319        return Ok(None);
320      };
321      return Ok(Some((name, cut.labelled_address(tokens, &item)?)));
322    }
323  }
324  Ok(None)
325}

327/// The addresses a class e-mail call's reader reads from it, each whole ([`email_addresses`], [`is_whole_address`]); one
328/// address in printed braces is that address, set as written (`\{fatima.iribu@gmail.com\}`, 2410.04091,
329/// [`single_braced`]). `None` when the call is not read so. The entry's cut ([`entry_email_cut`]) and its item
330/// ([`shared_call_item`]) read a call by it, as the row's split did (`row_email_calls`, deleted at P27b).
331fn read_call(call: &EmailCall) -> Result<Option<Vec<String>>> {
332  let read_tokens = |tokens: &[Token]| -> Result<Option<Vec<String>>> {
333    Ok(
334      email_addresses(&Invocation!(call.cs, vec![Some(Tokens::new(
335        tokens.to_vec()
336      ))]))
337      .filter(|list| list.iter().all(|address| is_whole_address(address))),
338    )
339  };
340  if let Some(list) = read_tokens(&call.arg)? {
341    return Ok(Some(list));
342  }
343  Ok(match single_braced(&call.arg) {
344    Some(inner) => read_tokens(inner)?.filter(|list| list.len() == 1),
345    None => None,
346  })
347}

358/// One e-mail call's unit cut ([`unit_cut`]): its span, the separators before it back to `floor` with it, and the
359/// labels of its own the rule recognises ([`call_label`]).
360struct UnitCut {
361  start:  usize,
362  end:    usize,
363  before: Option<(usize, usize, usize)>,
364  after:  Option<(usize, usize, usize)>,
365}
366
367/// The unit of the call `call` on the line `tokens[start..end]`, no earlier than `floor` (the unit before it): the call
368/// with a label of its own before it (`Corresponding author: \email{…}`) or a parenthetical after it (`\email{…}
369/// (Corresponding author),`, [`call_label`]), and the separators before it back to the text or unit before it (`Univ A
370/// \\ \email{…}` leaves "Univ A"). The cut of the entry's ([`entry_email_cut`]) and
371/// of its item ([`shared_call_item`]), as of the row's split (`row_email_calls`, deleted at P27b).
372fn unit_cut(
373  tokens: &[Token],
374  call: &EmailCall,
375  (floor, start, end): (usize, usize, usize),
376  first: bool,
377) -> UnitCut {
378  let (before, after) = call_label(tokens, call, (floor, start, end), first);
379  let mut first_taken = before.map_or(call.start, |(taken, ..)| taken.min(call.start));
380  while first_taken > floor && is_call_separator(&tokens[first_taken - 1]) {
381    first_taken -= 1;
382  }
383  UnitCut {
384    start: first_taken.max(floor),
385    end: after.map_or(call.end, |(_, _, end)| end),
386    before,
387    after,
388  }
389}
390
391impl UnitCut {
392  /// Does the unit hold a label of the call's own?
393  fn labelled(&self) -> bool { self.before.is_some() || self.after.is_some() }
394
395  /// The e-mail label printed before the call that names it (`Email:`), unless a label of its own before it is in its
396  /// text already (`Corresponding author, Email:`).
397  fn name<'c>(&self, call: &'c EmailCall) -> Option<&'c Vec<Token>> {
398    call.label.as_ref().filter(|_| self.before.is_none())
399  }
400
401  /// The call's address as the class sets it ([`email_command_address`]), with its label of its own before it (a tie a
402  /// space, as printed: `Corresponding~author:`) and its parenthetical after it.
403  fn labelled_address(&self, tokens: &[Token], call: &EmailCall) -> Result<Vec<Token>> {
404    let mut whole: Vec<Token> = Vec::new();
405    if let Some((_, from, to)) = self.before {
406      whole.extend(
407        tokens[from..to]
408          .iter()
409          .map(|t| if *t == T_ACTIVE!('~') { T_SPACE!() } else { *t }),
410      );
411      whole.push(T_SPACE!());
412    }
413    whole.extend(email_command_address(&call.cs, Tokens::new(call.arg.clone()))?.unlist());
414    if let Some((from, to, _)) = self.after {
415      whole.push(T_SPACE!());
416      whole.extend_from_slice(&tokens[from..to]);
417    }
418    Ok(whole)
419  }
420}

422/// A span cut out of a contact ([`kept_without`]): its start and end, what is written in its place, and whether it is a
423/// unit, which at the content's start takes the separators after it too.
424type CutEdit = (usize, usize, Vec<Token>, bool);
425
426/// The contact `tokens` with each of `edits` (in order) cut out and its tokens written in its place. A unit at the start
427/// of the content `inner` (or right after a group's open) takes the separators after it as well: they may be the next
428/// unit's own (`\email{a}, \email{b}`). The reassembly of the entry's cut ([`entry_email_cut`]), as of the
429/// row's split (`row_email_calls`, deleted at P27b).
430fn kept_without(
431  tokens: &[Token],
432  edits: Vec<CutEdit>,
433  (inner_open, inner_close): (usize, usize),
434) -> Vec<Token> {
435  let mut kept: Vec<Token> = Vec::with_capacity(tokens.len());
436  let mut from = 0;
437  for (start, end, written, unit) in edits {
438    kept.extend_from_slice(&tokens[from.min(start)..start]);
439    from = end.max(from);
440    let at_start = unit
441      && (start <= inner_open
442        || kept
443          .last()
444          .is_some_and(|t| t.get_catcode() == Catcode::BEGIN));
445    kept.extend(written);
446    if at_start {
447      while from < inner_close && is_call_separator(&tokens[from]) {
448        from += 1;
449      }
450    }
451  }
452  kept.extend_from_slice(&tokens[from..]);
453  kept
454}

456/// The span of `tokens[from..to]` its lines are in: within the group of the one command that wraps the whole of it
457/// (`\lx@affiliation@withinst{…}`, as `\lx@add@affiliation` gives an institute), spaces aside; else the span itself.
458fn wrapped_span(tokens: &[Token], from: usize, to: usize) -> (usize, usize) {
459  let (mut first, mut last) = (from, to);
460  while first < last && tokens[first].get_catcode() == Catcode::SPACE {
461    first += 1;
462  }
463  while last > first && tokens[last - 1].get_catcode() == Catcode::SPACE {
464    last -= 1;
465  }
466  if first + 1 < last
467    && tokens[first].get_catcode() == Catcode::CS
468    && tokens[first + 1].get_catcode() == Catcode::BEGIN
469    && skip_group(tokens, first + 1) == last
470  {
471    return wrapped_span(tokens, first + 2, last - 1);
472  }
473  (from, to)
474}

476/// Is `t` a line break of a contact's lines ([`content_lines`])?
477fn is_line_break(t: &Token) -> bool {
478  t.get_catcode() == Catcode::CS
479    && t.with_str(|name| matches!(name, "\\\\" | "\\newline" | "\\par"))
480}

482/// The lines of `tokens[from..to]`, parted by a line break outside any group ([`is_line_break`], with `\\`'s `*` and
483/// `[…]`): each line's start with the break before it, its own start, and its end. The lines cover the span.
484fn content_lines(tokens: &[Token], from: usize, to: usize) -> Vec<(usize, usize, usize)> {
485  let mut lines = Vec::new();
486  let (mut sep, mut start, mut depth) = (from, from, 0usize);
487  let mut at = from;
488  while at < to {
489    match tokens[at].get_catcode() {
490      Catcode::BEGIN => depth += 1,
491      Catcode::END => depth = depth.saturating_sub(1),
492      _ if depth == 0 && is_line_break(&tokens[at]) => {
493        lines.push((sep, start, at));
494        sep = at;
495        at += 1;
496        if tokens[at..to].first() == Some(&T_OTHER!("*")) {
497          at += 1;
498        }
499        if tokens[at..to].first() == Some(&T_OTHER!("[")) {
500          at = tokens[at..to]
501            .iter()
502            .position(|t| *t == T_OTHER!("]"))
503            .map_or(to, |end| at + end + 1);
504        }
505        start = at;
506        continue;
507      },
508      _ => {},
509    }
510    at += 1;
511  }
512  lines.push((sep, start, to));
513  lines
514}

516/// A call's label before it (where the spacing before it starts, and its span) and parenthetical after it (its span,
517/// and where the punctuation after it ends), [`call_label`].
518type CallLabel = (Option<(usize, usize, usize)>, Option<(usize, usize, usize)>);

520/// The label of a line's own that goes with an e-mail call on the line `tokens[start..end]` ([`unit_cut`]), no
521/// earlier than `floor` (the unit before it): a short label before it ending in a colon (`Corresponding author:`,
522/// 2410.06985, 2609.00838; `Contact:`, 2509.05139), where the spacing before it starts and its span; and a
523/// parenthetical of words after it (`(Corresponding author)`, 2608.24697; `(Univ Y)`, the r3 re-review's n23), its
524/// span and where it ends with the punctuation after it. The label before it, read as printed (a font's group, a `~`,
525/// a spacing command: `\textit{Corresponding author:}`, `Corresponding~author:`, `\hspace{1cm}Contact:`, the 85b r3
526/// re-review's n14-n16), is one to four words of letters and hyphens, the first a contact's ([`names_a_contact`]:
527/// corresponding, correspondence, contact, author, e-mail, mail, address), from the line's start — not the first
528/// line's, the institute's own — or from a `,` or `;`, the spacing before it aside ([`past_spacing`]: the label is
529/// set from its first word). Any other text beside a call is the line's, the institute's as before: `… F-76000 Rouen,
530/// France. \email{…}` (2609.24668), `\hspace{0.5cm}` (2310.06847), and text that only ends in a colon (`Department of
531/// Computer Science:`, `Dept. of CS:`, `University of X, Robotics Lab:`, `Lab of A. B. Smith:`: the r3 re-review's n09,
532/// n10, n12, n35). So only the label forms the rule recognises go with the call; a label of another form (prose
533/// without a colon, `Corresponding author email \email{…}`; a label word the rule does not know, `Reach me at:
534/// \email{…}`; one after a `.`, or after other text on its line) stays in the shared institute, every author's — a
535/// residual.
536fn call_label(
537  tokens: &[Token],
538  call: &EmailCall,
539  (floor, start, end): (usize, usize, usize),
540  first: bool,
541) -> CallLabel {
542  let floor = floor.max(start);
543  let is_word = |t: &Token| t.get_catcode() == Catcode::LETTER || *t == T_OTHER!("-");
544  // (words of a label, as printed: letters, hyphens and spaces, the first naming a contact)
545  let is_label = |printed: &str| {
546    let words: Vec<&str> = printed.split_whitespace().collect();
547    (1..=4).contains(&words.len())
548      && names_a_contact(words[0])
549      && printed
550        .chars()
551        .all(|c| c.is_alphabetic() || c == '-' || c.is_whitespace())
552  };
553  let mut before = None;
554  // (the piece before the call from the line's start — not the first line's, the institute's own — or from the last
555  // `,` or `;` outside any group, the other calls of the line aside, read as printed: through a font's group, a `~`, a
556  // spacing command, `\textit{Corresponding author:}`, `Corresponding~author:`, `\hspace{1cm}Contact:`)
557  let mut colon = call.at;
558  while colon > floor && tokens[colon - 1] == T_SPACE!() {
559    colon -= 1;
560  }
561  let (mut from, mut depth) = (None, 0usize);
562  for at in (floor..colon).rev() {
563    match tokens[at].get_catcode() {
564      Catcode::END => depth += 1,
565      Catcode::BEGIN => depth = depth.saturating_sub(1),
566      _ if depth == 0 && [T_OTHER!(","), T_OTHER!(";")].contains(&tokens[at]) => {
567        from = Some(at + 1);
568        break;
569      },
570      _ => {},
571    }
572  }
573  let from = from.or((floor == start && !first).then_some(start));
574  if let Some(taken) = from {
575    // (the label without the spacing before it, which prints nothing of its own: `\hspace{1cm}Contact:`)
576    let from = past_spacing(tokens, taken, colon);
577    let printed = visible_name_text(&tokens[from..colon]);
578    // (an `Email:` label alone is the contact's own name, which the call's start already takes, [`email_label_before`])
579    if let Some(words) = printed.trim_end().strip_suffix(':')
580      && is_label(words)
581      && call.start > from
582    {
583      before = Some((taken, from, colon));
584    }
585  }
586  let mut open = call.end;
587  while open < end && tokens[open] == T_SPACE!() {
588    open += 1;
589  }
590  let mut after = None;
591  if open < end && tokens[open] == T_OTHER!("(") {
592    let close = (open + 1..end).find(|at| tokens[*at] == T_OTHER!(")"));
593    if let Some(close) = close
594      && tokens[open + 1..close]
595        .iter()
596        .all(|t| is_word(t) || *t == T_SPACE!())
597      && tokens[open + 1..close].iter().any(is_word)
598    {
599      let mut past = close + 1;
600      while past < end && [T_OTHER!(","), T_OTHER!(";"), T_OTHER!(".")].contains(&tokens[past]) {
601        past += 1;
602      }
603      after = Some((open, close + 1, past));
604    }
605  }
606  (before, after)
607}

609/// Is `address` a template's, the address form a paper gives no author (`{firstname.lastname}@x.edu`,
610/// `first.last@x.edu`, `name.surname@x.edu`; the 85b re-review's s08, s09; 2102.12877, 2109.05583)?
611fn is_template_address(address: &str) -> bool {
612  let local = address
613    .split_once('@')
614    .map_or(address, |(local, _)| local)
615    .to_lowercase();
616  let words: Vec<&str> = local
617    .split(|c: char| !c.is_alphanumeric())
618    .filter(|word| !word.is_empty())
619    .collect();
620  words.iter().any(|word| {
621    [
622      "firstname",
623      "lastname",
624      "forename",
625      "surname",
626      "givenname",
627      "familyname",
628    ]
629    .contains(word)
630  }) || words == ["first", "last"]
631}

633/// Whether `address`, read from an e-mail call ([`email_addresses`]), is an address whole: a local part and a domain of
634/// labels parted by dots, nothing of the line around it — not a piece of a list the reader cut at its commas
635/// (`[simon.barbarit--gaboriau@insa-, …, samia.ainouz@insa-]rouen.fr`, 2609.24668), which the call keeps as written.
636pub(in crate::base_utilities) fn is_whole_address(address: &str) -> bool {
637  let Some((local, domain)) = address.rsplit_once('@') else {
638    return false;
639  };
640  !local.is_empty()
641    && !local.contains(['[', ']', '(', ')', '<', '>', '@'])
642    && domain.contains('.')
643    && domain.split('.').all(|label| {
644      !label.is_empty()
645        && !label.starts_with('-')
646        && !label.ends_with('-')
647        && label.chars().all(|c| c.is_alphanumeric() || c == '-')
648    })
649}

651/// The tokens inside the printed braces of `arg` when they are the whole of it (`\{fatima.iribu@gmail.com\}`, spaces
652/// aside), and no other printed brace stands within.
653fn single_braced(arg: &[Token]) -> Option<&[Token]> {
654  let is = |t: &Token, names: [&str; 2]| {
655    t.get_catcode() == Catcode::CS && t.with_str(|name| names.contains(&name))
656  };
657  let open = ["\\{", "\\lbrace"];
658  let close = ["\\}", "\\rbrace"];
659  let first = arg.iter().position(|t| t.get_catcode() != Catcode::SPACE)?;
660  let last = arg
661    .iter()
662    .rposition(|t| t.get_catcode() != Catcode::SPACE)?;
663  if first >= last || !is(&arg[first], open) || !is(&arg[last], close) {
664    return None;
665  }
666  let inner = &arg[first + 1..last];
667  (!inner.iter().any(|t| is(t, open) || is(t, close))).then_some(inner)
668}

670/// The start of an e-mail label ending at `tokens[..end]` (`Email:`, `E-mail:`, `Emails:`, the colon after the word),
671/// no earlier than `from`, printed before an e-mail call it names, which the call's own contact name says.
672fn email_label_before(tokens: &[Token], from: usize, end: usize) -> Option<usize> {
673  if end <= from || tokens[end - 1] != T_OTHER!(":") {
674    return None;
675  }
676  let mut start = end - 1;
677  while start > from
678    && (tokens[start - 1].get_catcode() == Catcode::LETTER || tokens[start - 1] == T_OTHER!("-"))
679  {
680    start -= 1;
681  }
682  let word: String = tokens[start..end - 1]
683    .iter()
684    .map(Token::to_string)
685    .collect();
686  (start < end - 1 && is_email_label(&word)).then_some(start)
687}

689/// The span of a queued command's content, its last group (`\lx@add@frontmatter@now{ltx:creator}[role=author]{…}`,
690/// `\lx@annotate@frontmatter@now{ltx:creator}{ltx:contact}[…]{…}`): the places of its braces.
691fn content_group(tokens: &[Token]) -> Option<(usize, usize)> {
692  let close = tokens.len().checked_sub(1)?;
693  if tokens[close].get_catcode() != Catcode::END {
694    return None;
695  }
696  let mut depth = 0usize;
697  for at in (0..=close).rev() {
698    match tokens[at].get_catcode() {
699      Catcode::END => depth += 1,
700      Catcode::BEGIN => {
701        depth -= 1;
702        if depth == 0 {
703          return Some((at, close));
704        }
705      },
706      _ => {},
707    }
708  }
709  None
710}
