9/// The line breaks an author line's printed separator may stand beside, with their optional `*` and `[…]`: a break
10/// before the word or after it is no name's (`Ann Able\orcidID{…} and\\ Bob Baker`, svmult; `Badr~Kaoui$^{3,4}$\break
11/// and Jens~Harting`, jfm 1311.3650; `\linebreak and`, mdpi 2509.22118), nor is a control return — a line ending in `\`,
12/// which latex.ltx:558 `\def\^^M{\ }` prints as a space (2004.02831, 2404.12084).
13pub(in crate::base_utilities) const SEPARATOR_LINE_BREAKS: [&str; 5] =
14  ["\\\\", "\\break", "\\linebreak", "\\allowbreak", "\\\r"];
15
16/// The index past the spacing that starts at `v[i]` (`i` when none): spaces, ties, space commands and lengths
17/// ([`leading_spacing_end`]), the line breaks of [`SEPARATOR_LINE_BREAKS`], and — when `macros` — a parameterless
18/// macro whose body is only spacing (jfm.cls:1025 `\newcommand\ns{\kern.5em\relax}`: `Chang Liu \ns and Dennice F.
19/// Gayme`, 1912.02168).
20pub(in crate::base_utilities) fn separator_gap_end(
21  v: &[Token],
22  mut i: usize,
23  macros: bool,
24) -> usize {
25  while i < v.len() {
26    let past = i + leading_spacing_end(&v[i..]);
27    if past > i {
28      i = past;
29      continue;
30    }
31    let t = v[i];
32    if SEPARATOR_LINE_BREAKS.iter().any(|cs| t == T_CS!(*cs)) {
33      i += 1;
34      if t == T_CS!("\\\\") && v.get(i) == Some(&T_OTHER!("*")) {
35        i += 1;
36      }
37      if (t == T_CS!("\\\\") || t == T_CS!("\\linebreak"))
38        && v.get(i) == Some(&T_OTHER!("["))
39        && let Some(close) = v[i..].iter().position(|u| *u == T_OTHER!("]"))
40      {
41        i += close + 1;
42      }
43    } else if macros && prints_only_spacing(&t) {
44      i += 1;
45    } else {
46      break;
47    }
48  }
49  i
50}

52/// Whether `cs` is a parameterless macro whose body prints only spacing (`\kern.5em\relax`, `\hskip1em`, `\quad`).
53fn prints_only_spacing(cs: &Token) -> bool {
54  if cs.get_catcode() != Catcode::CS || lookup_conditional(cs).is_some() {
55    return false;
56  }
57  let Ok(Some(defn)) = lookup_definition(cs) else {
58    return false;
59  };
60  if !defn.is_expandable()
61    || defn
62      .get_parameters()
63      .is_some_and(|p| !p.get_parameters().is_empty())
64  {
65    return false;
66  }
67  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
68    return false;
69  };
70  let body = body.unlist_ref();
71  body.iter().any(|t| {
72    !matches!(
73      t.get_catcode(),
74      Catcode::BEGIN | Catcode::END | Catcode::SPACE
75    ) && *t != T_CS!("\\relax")
76  }) && only_spacing(body)
77}

79/// Whether `v` holds only spacing ([`separator_gap_end`]), `\relax` and groups of the same (`{\kern.5em}`); a spacing
80/// command's braced argument is its own (`\hspace{1em}`, `\rule{0pt}{1ex}`).
81pub(in crate::base_utilities) fn only_spacing(v: &[Token]) -> bool {
82  let mut i = 0;
83  while i < v.len() {
84    let past = separator_gap_end(v, i, false);
85    if past > i {
86      i = past;
87    } else if v[i] == T_CS!("\\relax") {
88      i += 1;
89    } else if v[i].get_catcode() == Catcode::BEGIN {
90      let end = skip_group(v, i);
91      if end > v.len()
92        || end < i + 2
93        || v[end - 1].get_catcode() != Catcode::END
94        || !only_spacing(&v[i + 1..end - 1])
95      {
96        return false;
97      }
98      i = end;
99    } else {
100      return false;
101    }
102  }
103  true
104}

106/// Whether `t` is a letter of a word: a letter, or a letter command (`\L`, `\ss`, `\dj`) or accent written in it.
107pub(in crate::base_utilities) fn letterlike(t: &Token) -> bool {
108  t.get_catcode() == Catcode::LETTER
109    || (t.get_catcode() == Catcode::CS
110      && t.with_str(|s| letter_command(s).is_some() || ACCENT_COMMANDS.contains(&s)))
111}

113/// Whether `t` is a control word that is no letter of the word beside it: a name of two characters or more, or of one
114/// letter (tex.web §354-356 — a control symbol's name is one non-letter), after which TeX skips the spaces, so the
115/// command itself parts the words (`\rm`, `\break`, `\P`). A control symbol skips none: `\-` (a discretionary
116/// hyphen) and `\/` (an italic correction) print inside a word (`Rol\-and Smith`, `Bertr\/and Smith` print "Roland
117/// Smith", "Bertrand Smith"); a spacing one (`\,`, `\ `) is a gap of its own ([`separator_gap_end`]).
118pub(in crate::base_utilities) fn parts_words(t: &Token) -> bool {
119  t.get_catcode() == Catcode::CS
120    && !letterlike(t)
121    && t.with_str(|s| {
122      let mut name = s.strip_prefix('\\').unwrap_or(s).chars();
123      match (name.next(), name.next()) {
124        (Some(_), Some(_)) => true,
125        (Some(c), None) => c.is_alphabetic(),
126        _ => false,
127      }
128    })
129}

131/// The end of the marks a name's tokens open with, with no space before them: `\textsuperscript{a}`, `$^{a}$`, the
132/// shown symbol mark `\lx@frontmatter@keepsup{*}\lx@frontmatter@symbolmark{*}`, and a `\thanks{...}` or
133/// `\footnote{...}`, whose mark prints with them (0 when none).
134pub(in crate::base_utilities) fn leading_marks_end(tokens: &[Token]) -> usize {
135  let mut i = 0;
136  while let Some(&first) = tokens.get(i) {
137    let next = if first == T_CS!("\\textsuperscript")
138      || first == T_CS!("\\lx@frontmatter@keepsup")
139      || first == T_CS!("\\lx@frontmatter@symbolmark")
140      // aastex's mark (aas_support: an affiliation request), written after the comma as often as before it
141      // (`Todd M. Tripp,\altaffilmark{2,3} Bart P. Wakker,\altaffilmark{4}`, astro-ph0302534)
142      || first == T_CS!("\\altaffilmark")
143    {
144      sup_operand_at(tokens, i + 1).map(|(_, next)| next)
145    } else if (first == T_CS!("\\thanks") || first == T_CS!("\\footnote"))
146      && tokens
147        .get(i + 1)
148        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
149    {
150      sup_operand_at(tokens, i + 1).map(|(_, next)| next)
151    } else if first == T_MATH!() && tokens.get(i + 1) == Some(&T_SUPER!()) {
152      sup_operand_at(tokens, i + 2)
153        .filter(|(_, next)| tokens.get(*next) == Some(&T_MATH!()))
154        .map(|(_, next)| next + 1)
155    } else if first == T_OTHER!("*") {
156      // a run of bare asterisks, a footnote mark written after the comma (`S. F. Pereira,* Z. Y. Ou,** and …`,
157      // quant-ph0003094)
158      Some(
159        i + tokens[i..]
160          .iter()
161          .take_while(|t| **t == T_OTHER!("*"))
162          .count(),
163      )
164    } else {
165      None
166    };
167    match next {
168      Some(next) => i = next,
169      None => break,
170    }
171  }
172  i
173}

181/// Perl: positionOf($tokens, @delims) — Base_Utility.pool.ltxml (PR #2767).
182/// Find the position of a Token from `delims` within `tokens`.
183/// 1 based, so None == token not present.
184pub fn position_of(tokens: &Tokens, delims: &[Token]) -> Option<usize> {
185  for (i, t) in tokens.unlist_ref().iter().enumerate() {
186    if delims.contains(t) {
187      return Some(i + 1);
188    }
189  }
190  None
191}

202/// The `\and` family only — the HARD author boundary the superscript-marker
203/// branch groups on FIRST, so a marker-less line never merges into an author
204/// from a previous `\and` group (html_feedback#1021 F2; OXIDIZED_DESIGN #52(g)).
205/// With it the ACM SIG classes' `\alignauthor`, which opens each author's column (sig-alternate.cls, sigchi.cls:
206/// `\author{\alignauthor Name\\ \affaddr{…}\\ \email{…} \alignauthor …}`): split here, its raw definition — a
207/// tabular juggle for the class's title page — never runs inside a name. Perl's OmniBus makes it empty
208/// (OmniBus.cls.ltxml:76) and its `@authorsplits` lacks it; since the arXiv profile runs these shipped classes raw
209/// (DIVERGENCES #444) their authors were lost (1605.02827, 2003.09061, 1906.01122).
210pub(in crate::base_utilities) fn author_and_splits() -> Vec<SplitDelim> {
211  vec![
212    T_CS!("\\and").into(),
213    T_CS!("\\And").into(),
214    T_CS!("\\AND").into(),
215    T_CS!("\\alignauthor").into(),
216  ]
217}

219// Things to split author & affiliation mix; NO comma in affiliations!!! An affiliation list's lines too (the frontmatter
220// API's entry cutter and Kind step, `entry_starts`).
221pub(in crate::base_utilities) fn author_affil_splits() -> Vec<SplitDelim> {
222  vec![
223    T_CS!("\\and").into(),
224    T_CS!("\\And").into(),
225    T_CS!("\\AND").into(),
226    // Beyond-Perl: the literal word " and " is deliberately NOT a line-level
227    // delimiter here (Perl's @authoraffilsplits includes it). This split runs
228    // BEFORE author/affiliation classification, so splitting on " and " shreds
229    // institution names that contain it — "Princeton Language and Intelligence"
230    // → "Princeton Language" + "Intelligence, …" (re-joined without a space).
231    // Authors written "Alice and Bob" still split: literal " and " is applied in
232    // the author arm (comma_split_author_line), AFTER a line is known to be an
233    // author line, so affiliations are never fragmented. Witness 2605.00347.
234    T_CS!("\\quad").into(),
235    T_CS!("\\qquad").into(),
236    T_CS!("\\\\").into(),
237    // latex.ltx:9256 `\DeclareRobustCommand\newline{\@normalcr\relax}`: the line break `\\` is (2401.14196's
238    // `Ann Able$^{1}$, Bob Baker$^{1,2}$ \newline Cat Cole$^{1}$`)
239    T_CS!("\\newline").into(),
240  ]
241}

243/// An affiliation list's lines, the frontmatter API's cutter's and Kind step's: cut first at the kernel's `\and` by
244/// meaning (a document's macro holding it, a `\csname and\endcsname`, as the spelled one; `kernel_separator_runs`),
245/// outside balanced parentheses as the spelling's cut (OD #36), then at [`author_affil_splits`] by spelling. Beamer's
246/// `\institute{\inst{1}Univ A \sep Univ B}`, `\def\sep{\and}`; `\inst{1}Univ A (North \and South)` one line.
247pub(in crate::base_utilities) fn affiliation_lines(stuff: &Tokens) -> Result<DelimitedPieces> {
248  let delims = author_affil_splits();
249  if !may_hold_kernel_separator(stuff, &delims) {
250    return Ok(split_tokens_delimited(stuff.clone(), delims));
251  }
252  let v = stuff.unlist_ref();
253  let runs = kernel_separator_runs(stuff)?
254    .unwrap_or_default()
255    .into_iter()
256    .filter(|&(start, _)| !inside_balanced_parens(v, start))
257    .collect();
258  Ok(split_at_runs(stuff.clone(), delims, runs).0)
259}

261/// Whether a cut of [`affiliation_lines`] parts two entries: the `\and` family spelled (a `\let` copy of the current
262/// `\and` too, [`delimiter_token_matches`]), or a cut by meaning (no [`author_affil_splits`] token spelled in it).
263pub(in crate::base_utilities) fn parts_entries(delimiter: &[Token]) -> bool {
264  let spelled = |delims: &[SplitDelim]| {
265    delimiter.iter().any(|t| {
266      delims
267        .iter()
268        .any(|d| matches!(d, SplitDelim::Token(d) if delimiter_token_matches(d, t)))
269    })
270  };
271  !delimiter.is_empty() && (spelled(&author_and_splits()) || !spelled(&author_affil_splits()))
272}
