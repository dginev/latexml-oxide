10/// If `tokens` (spaces around it aside) is one brace group, its opening declarations ([`opening_declarations`]:
11/// `\bf`, `\small`) and the rest of its content.
12pub(in crate::base_utilities) fn whole_line_group(tokens: &Tokens) -> Option<(Vec<Token>, Tokens)> {
13  let v = tokens.unlist_ref();
14  let start = v.iter().position(|t| *t != T_SPACE!())?;
15  let end = v.iter().rposition(|t| *t != T_SPACE!())?;
16  if end <= start || v[start].get_catcode() != Catcode::BEGIN || skip_group(v, start) != end + 1 {
17    return None;
18  }
19  let inner = Tokens::new(v[start + 1..end].to_vec());
20  let declarations = opening_declarations(&inner);
21  let rest = Tokens::new(inner.unlist_ref()[declarations.len()..].to_vec());
22  Some((declarations, rest))
23}

25/// What an author line's top level is made of, for [`spaced_name_separators`]: the spacing or line breaks between two
26/// words (a gap), the word "and" or `\&`, a brace group, a math span, a span the split keeps whole (balanced
27/// parentheses, an environment), or one token.
28#[derive(Clone, Copy, PartialEq)]
29enum LinePart {
30  Gap(usize, usize),
31  And(usize),
32  Ampersand(usize),
33  Group(usize, usize),
34  Math(usize, usize),
35  Span(usize, usize),
36  Token(usize),
37}
38
39/// The parts of an author line's top level ([`LinePart`]).
40fn line_parts(v: &[Token]) -> Vec<LinePart> {
41  let and = [T_LETTER!("a"), T_LETTER!("n"), T_LETTER!("d")];
42  let mut parts = Vec::with_capacity(v.len());
43  let mut i = 0;
44  while i < v.len() {
45    let t = v[i];
46    let gap = separator_gap_end(v, i, true);
47    let (part, end) = if gap > i {
48      (LinePart::Gap(i, gap), gap)
49    } else if t.get_catcode() == Catcode::BEGIN {
50      let end = skip_group(v, i).min(v.len());
51      (LinePart::Group(i, end), end)
52    } else if t.defined_as(&T_MATH!()) {
53      let end = (i + 1..v.len())
54        .find(|&j| v[j].defined_as(&T_MATH!()))
55        .map_or(v.len(), |j| j + 1);
56      (LinePart::Math(i, end), end)
57    } else if v[i..].starts_with(&and) {
58      (LinePart::And(i), i + 3)
59    } else if t == T_CS!("\\&") {
60      (LinePart::Ampersand(i), i + 1)
61    } else if t == T_OTHER!("(")
62      && let Some(close) = paren_close_offset(&v[i + 1..])
63    {
64      (LinePart::Span(i, i + close + 2), i + close + 2)
65    } else if t == T_CS!("\\begin")
66      && let Some(span) =
67        environment_closes_ahead(&v[i + 1..].iter().copied().collect::<VecDeque<Token>>())
68    {
69      (LinePart::Span(i, i + 1 + span), i + 1 + span)
70    } else {
71      (LinePart::Token(i), i + 1)
72    };
73    parts.push(part);
74    i = end;
75  }
76  parts
77}

79/// Whether the part before a separator word ends the word before it: a gap; a control word that is no letter (`\L`,
80/// `\ss`) or accent ([`parts_words`]); a note's or mark's argument (`Dan Dorn\thanks{…}and Eve Ess`, jfm 2005.00317),
81/// or a group whose text does not end in a letter or a control symbol (`{\bf Ann Able}and` prints "Ann Ableand",
82/// `{Rol\-}and` "Roland"); a math span holding only a superscript, a mark (`Bao-zhi Sun$^\ast$\&Shao-Ming Fei`,
83/// 1707.03934); a span the split keeps whole.
84fn ends_a_word(v: &[Token], parts: &[LinePart], at: usize) -> bool {
85  match parts[at] {
86    LinePart::Gap(..) | LinePart::Span(..) => true,
87    LinePart::Token(i) => parts_words(&v[i]),
88    LinePart::Group(start, end) => {
89      let owner = at.checked_sub(1).and_then(|before| match parts[before] {
90        LinePart::Token(k) if v[k].get_catcode() == Catcode::CS => Some(v[k]),
91        _ => None,
92      });
93      if let Some(owner) = owner {
94        if is_name_annotation(&owner) || is_footnote_mark(&owner) {
95          return true;
96        }
97        if letterlike(&owner) {
98          return false;
99        }
100      }
101      // (the group's last token: a letter, a letter command, a control symbol printing no space, or an inner group's
102      // end continue the word)
103      let last = v[start + 1..end.saturating_sub(1)]
104        .iter()
105        .rev()
106        .find(|t| **t != T_SPACE!());
107      !last.is_some_and(|t| {
108        letterlike(t)
109          || t.get_catcode() == Catcode::END
110          || (t.get_catcode() == Catcode::CS
111            && !parts_words(t)
112            && separator_gap_end(std::slice::from_ref(t), 0, false) == 0)
113      })
114    },
115    LinePart::Math(start, end) => {
116      v.get(start + 1) == Some(&T_SUPER!())
117        && sup_operand_at(v, start + 2).is_some_and(|(_, next)| {
118          let mut next = next;
119          while v.get(next) == Some(&T_SPACE!()) {
120            next += 1;
121          }
122          next + 1 == end
123        })
124    },
125    LinePart::And(_) | LinePart::Ampersand(_) => false,
126  }
127}

129/// Whether the part after a separator word begins the next one: a gap (`Eve Ess and\;Fay Fox`, 2502.11348), or a
130/// control word that is no letter or accent ([`parts_words`]: `and\rm`, not `and\-`).
131fn begins_a_word(v: &[Token], part: LinePart) -> bool {
132  match part {
133    LinePart::Gap(..) => true,
134    LinePart::Token(i) => parts_words(&v[i]),
135    _ => false,
136  }
137}

139/// An author line's name separators as the split reads them, at its top level (the split's re-reads of a wrapper's
140/// content see the content's own):
141///  - the word "and" between two word boundaries, and `\&` after one, written ` and `, ` \& `, the line
142///    breaks and spacing commands beside them dropped. TeX skips the spaces after a control word, so `\space and`,
143///    `\break and`, `\ns and` hold no space token; nor do `\thanks{…}and`, `$^\ast$\&`, `and\;`, svmult's `and\\` —
144///    each "and" the author wrote between two names, whether pdflatex prints it apart or, after a control word, glued to
145///    it ("ATLAS∗and", OXIDIZED_DESIGN_DIVERGENCES #765: a reading of the author's intent) (1311.3650, 1503.04190, 1505.06278, 2005.00317, 1912.02168,
146///    2509.22118, 1504.03105, 1607.04619, 2004.02831, 2404.12084, 2502.11348, astro-ph9704233, 1707.03934, 2401.17237,
147///    2506.15327, 2509.11693, 2403.07875, 2512.18788, 2603.27055). Perl's only such separator is " and " between
148///    spaces (Base_Utility.pool.ltxml:678), so it keeps these names one author (OXIDIZED_DESIGN_DIVERGENCES #765). A
149///    letter or a letter or accent command beside the word keeps it in its name ("Rand", "Anders", `Zo\"{e}and`), and
150///    "AT\&T" stays one word;
151///  - a comma at the edge of a math span that holds only a superscript mark, printed between the names, is written
152///    outside it, where the split sees it: `Wittor$^{1}, $Brüggen` (1610.00129), `Iacobuta$,^{5,6,7}$ Mabel`
153///    (2108.04461), `Manoj$^{2},$ Chandra` (2402.11182) — a mark after the comma then goes to the name before it
154///    (the reading's `resolver/lines.rs` `mark_goes_next`).
155pub(in crate::base_utilities) fn spaced_name_separators(line: Tokens) -> Tokens {
156  let v = line.unlist_ref();
157  let parts = line_parts(v);
158  let single_space = |part: Option<&LinePart>| match part {
159    Some(LinePart::Gap(a, b)) => b - a == 1 && (v[*a] == T_SPACE!() || v[*a] == T_ACTIVE!('~')),
160    _ => false,
161  };
162  let mut out: Vec<Token> = Vec::with_capacity(v.len() + 4);
163  let mut changed = false;
164  // the gap a separator just written took in (dropped), and the length of the gap last written, at the end of `out`
165  let mut skip_next_gap = false;
166  let mut written_gap: Option<usize> = None;
167  // the place and length in `out` of the gap last written when only declarations that typeset nothing follow it
168  // (`\sc Ann Able \rm and \sc Bob Baker`: the gap before `\rm` is the separator's too, no name's; not before
169  // `\LaTeX`, which prints a word)
170  let mut gap_before_layout: Option<(usize, usize)> = None;
171  for (at, part) in parts.iter().enumerate() {
172    let (word, start, len) = match *part {
173      LinePart::And(i) => (true, i, 3),
174      LinePart::Ampersand(i) => (false, i, 1),
175      LinePart::Gap(start, end) => {
176        if std::mem::take(&mut skip_next_gap) {
177          (written_gap, gap_before_layout) = (None, None);
178        } else {
179          gap_before_layout = Some((out.len(), end - start));
180          out.extend_from_slice(&v[start..end]);
181          written_gap = Some(end - start);
182        }
183        continue;
184      },
185      LinePart::Math(start, end) => {
186        (skip_next_gap, written_gap, gap_before_layout) = (false, None, None);
187        if let Some(moved) = comma_out_of_mark_math(&v[start..end]) {
188          out.extend(moved);
189          changed = true;
190        } else {
191          out.extend_from_slice(&v[start..end]);
192        }
193        continue;
194      },
195      LinePart::Group(start, end) | LinePart::Span(start, end) => {
196        (skip_next_gap, written_gap, gap_before_layout) = (false, None, None);
197        out.extend_from_slice(&v[start..end]);
198        continue;
199      },
200      LinePart::Token(i) => {
201        (skip_next_gap, written_gap) = (false, None);
202        if !layout(&v[i]) {
203          gap_before_layout = None;
204        }
205        out.push(v[i]);
206        continue;
207      },
208    };
209    // ("and" needs both boundaries; `\&`, no letter, only the one before it: `$^\ast$\&Shao-Ming Fei`, but `AT\&T`)
210    let before = at.checked_sub(1).is_some_and(|b| ends_a_word(v, &parts, b));
211    let after = parts
212      .get(at + 1)
213      .is_some_and(|next| begins_a_word(v, *next));
214    let separates = before && (after || !word);
215    // (between two plain spaces or ties the split's own separators match as written)
216    let plain = at
217      .checked_sub(1)
218      .is_some_and(|b| single_space(parts.get(b)))
219      && single_space(parts.get(at + 1));
220    if !separates || plain {
221      (skip_next_gap, written_gap, gap_before_layout) = (false, None, None);
222      out.extend_from_slice(&v[start..start + len]);
223      continue;
224    }
225    changed = true;
226    if let Some(gap) = written_gap.take() {
227      out.truncate(out.len() - gap);
228    } else if let Some((place, gap)) = gap_before_layout {
229      out.drain(place..place + gap);
230    }
231    gap_before_layout = None;
232    out.push(T_SPACE!());
233    out.extend_from_slice(&v[start..start + len]);
234    out.push(T_SPACE!());
235    skip_next_gap = true;
236  }
237  if changed { Tokens::new(out) } else { line }
238}

240/// A math span holding only a superscript mark and a comma at its edge (`$^{1}, $`, `$,^{5}$`), with the comma written
241/// outside it: `$^{1}$,`, `,$^{5}$`. `None` for any other span.
242fn comma_out_of_mark_math(span: &[Token]) -> Option<Vec<Token>> {
243  let (open, close) = (*span.first()?, *span.last()?);
244  if span.len() < 4 || !close.defined_as(&T_MATH!()) {
245    return None;
246  }
247  let inner: Vec<Token> = span[1..span.len() - 1]
248    .iter()
249    .copied()
250    .filter(|t| *t != T_SPACE!())
251    .collect();
252  let comma = T_OTHER!(",");
253  let mark_only = |mark: &[Token]| {
254    mark.first() == Some(&T_SUPER!())
255      && sup_operand_at(mark, 1).is_some_and(|(_, next)| next == mark.len())
256  };
257  if inner.last() == Some(&comma) && mark_only(&inner[..inner.len() - 1]) {
258    let mut out = vec![open];
259    out.extend_from_slice(&inner[..inner.len() - 1]);
260    out.extend([close, comma]);
261    Some(out)
262  } else if inner.first() == Some(&comma) && mark_only(&inner[1..]) {
263    let mut out = vec![comma, open];
264    out.extend_from_slice(&inner[1..]);
265    out.push(close);
266    Some(out)
267  } else {
268    None
269  }
270}

272// GROUP-level separators for the no-marker author heuristic (OXIDIZED_DESIGN
273// #52): the \and family plus \quad/\qquad, but NOT the comma and NOT the literal
274// " and ". A comma separates NAMES within one author line ("Alice, Bob") and the
275// tokens WITHIN a multi-part address ("Laurel, MD 20723") — so splitting author
276// *groups* on it shreds addresses into fake authors and turns the following
277// \email line into a bogus affiliation (witness arXiv:2606.00315, NeurIPS
278// idiom). Perl's @authorsplits (Base_Utility.pool.ltxml L679) includes the comma
279// and shares this mis-split; excluding it here is a deliberate surpass-Perl
280// divergence. Name-level comma/" and " splitting still happens, but only AFTER a
281// line is known to be author names — in the reading's names line (`resolver/lines.rs`
282// `names_line`) — never across address text. (Compare author_affil_splits(): "NO comma in affiliations!!!".)
283pub(in crate::base_utilities) fn author_group_splits() -> Vec<SplitDelim> {
284  let mut splits = author_and_splits();
285  splits.extend([T_CS!("\\quad").into(), T_CS!("\\qquad").into()]);
286  splits
287}

289/// Whether an author argument holds a separator the kernel's author read parts it at: as that read sees it
290/// (`normalize_hspace_separators`: `\hspace`, `\hfill` and a spacing run between two names as `\quad`; the space a name
291/// sets before its own note or mark none) it splits at `author_group_splits` or at the line breaks of
292/// `author_affil_splits` into two pieces or more that are not empty. [`split_tokens`] leaves a note's argument, a group
293/// and math alone. For a binding that reads an `\author` with none as one name: authblk's.
294pub fn has_author_block_separator(stuff: &Tokens) -> bool {
295  let separators = author_group_splits()
296    .into_iter()
297    .chain(author_affil_splits())
298    .collect();
299  split_tokens(normalize_hspace_separators(stuff.clone()), separators)
300    .iter()
301    .filter(|piece| !piece.is_empty())
302    .count()
303    > 1
304}
