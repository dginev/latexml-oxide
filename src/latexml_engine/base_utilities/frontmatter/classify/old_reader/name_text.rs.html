5/// The markup an author line opens with, before its first letter: its control sequences and the groups they open
6/// (`{\small\em Bartol …}` → `{`, `\small`, `\em`; `T. Stelzer` → none). Lines styled alike share it.
7pub(in crate::base_utilities) fn leading_markup(line: &Tokens) -> Vec<Token> {
8  line
9    .unlist_ref()
10    .iter()
11    .take_while(|t| t.get_catcode() != Catcode::LETTER)
12    .filter(|t| matches!(t.get_catcode(), Catcode::CS | Catcode::BEGIN))
13    .copied()
14    .collect()
15}
