8/// Whether `stuff` holds a piece, parted as the kernel's author read parts its names and lines, that is one macro whose
9/// expansion prints a link alone (`\newcommand{\githubrepo}{\href{https://github.com/NVIDIA/kvpress}{{\faGithub}
10/// NVIDIA/kvpress}\xspace}`, `… \\ \begin{center} \hflink{…}{NVIDIA/KVzap}\quad \githubrepo \end{center}`, 2601.07891): no
11/// person, which the read cannot tell, a macro there reading as a name (`\AuthorOne \quad \AuthorTwo`). For a binding that
12/// reads an `\author` as one name otherwise, and counts the merge: authblk's.
13pub fn holds_a_macro_printing_a_link(stuff: &Tokens) -> bool {
14  let separators = author_group_splits()
15    .into_iter()
16    .chain(author_affil_splits())
17    .collect();
18  // (without an environment's `\begin{…}`/`\end{…}`, which no part of a piece's text is and which `split_tokens` leaves whole)
19  let v = stuff.unlist_ref();
20  let mut bare = Vec::with_capacity(v.len());
21  let mut at = 0;
22  while at < v.len() {
23    if (v[at] == T_CS!("\\begin") || v[at] == T_CS!("\\end")) && v.get(at + 1) == Some(&T_BEGIN!())
24    {
25      at = skip_group(v, at + 1);
26    } else {
27      bare.push(v[at]);
28      at += 1;
29    }
30  }
31  split_tokens(normalize_hspace_separators(Tokens::new(bare)), separators)
32    .iter()
33    .any(|piece| {
34      let mut printed = piece.unlist_ref().iter().filter(|t| **t != T_SPACE!());
35      let (Some(only), None) = (printed.next(), printed.next()) else {
36        return false;
37      };
38      only.get_catcode() == Catcode::CS
39        && matches!(lookup_definition_stored(only), Ok(Some(Stored::Expandable(definition)))
40        if definition.get_expansion().is_some_and(|body| match body {
41          ExpansionBody::Tokens(body) => {
42            let body: Vec<Token> = body
43              .unlist_ref()
44              .iter()
45              .filter(|t| **t != T_CS!("\\xspace"))
46              .copied()
47              .collect();
48            is_link_line(&Tokens::new(body))
49          },
50          _ => false,
51        }))
52    })
53}

55/// Whether the lines under an unmarked author line's names read as more names and nothing tells them from an
56/// institution: each a name by its words, styled as the names line is, with no address, link, note or mark on it
57/// (`Ann Able\\ Bob Baker`, `Ann Able\quad Bob Baker\\ Cat Cole`; 2610.38214's `\\ Stefano Soatto` under two rows of
58/// names; not `Ann Able\\ Meta AI`, an acronym beside a word in small letters, as `flag_merged_creators` reads a line,
59/// nor `\\ \textit{Univ X}`, nor `\\ University of X`). The kernel's read gives such a line to the names above it as
60/// their affiliation (Perl's first-line rule), a name made an affiliation where it is one; a binding that reads an
61/// `\author` as one name otherwise must not route it there unread: authblk's `\author` (2610.38214). A row of names
62/// parted by horizontal space goes on with the names first (`space_parted_rows_joined`), as the kernel reads it.
63pub fn lines_under_names_read_as_names(stuff: &Tokens) -> bool {
64  let prepared = rewrite_symbol_superscripts(normalize_hspace_separators(strip_linebreak_options(
65    drop_empty_marks(stuff.clone()),
66  )));
67  if has_author_marks(&prepared) {
68    return false;
69  }
70  let joined = space_parted_rows_joined(prepared);
71  let lines: Vec<Tokens> = split_tokens(joined, author_line_breaks())
72    .into_iter()
73    .filter(|line| spacing_end(line.unlist_ref()) > 0 && !prints_nothing(line))
74    .collect();
75  let Some((names, under)) = lines.split_first() else {
76    return false;
77  };
78  let reads_as_a_name = |line: &Tokens| {
79    let text = visible_name_text(line.unlist_ref());
80    let bare = outside_notes(line.unlist_ref()).len() == line.unlist_ref().len();
81    bare
82      && name_count(text.trim()) >= 1
83      && !text.contains('@')
84      && !holds_mixed_case_acronym(&text)
85      && !is_link_line(line)
86      && !collaboration_line(line)
87      && classify_thanks(&text) == "note"
88      && leading_markup(line) == leading_markup(names)
89  };
90  !under.is_empty() && under.iter().all(reads_as_a_name)
91}

93/// `stuff` with its top-level line breaks (a `\\`, its options read with it; `\newline`) written as `\quad`: lines
94/// under the names read as more names ([`lines_under_names_read_as_names`]), the row they print in.
95pub fn line_breaks_as_name_separators(stuff: &Tokens) -> Tokens {
96  let mut out: Vec<Token> = Vec::new();
97  let mut depth = 0usize;
98  for t in strip_linebreak_options(stuff.clone()).unlist() {
99    match t.get_catcode() {
100      Catcode::BEGIN => depth += 1,
101      Catcode::END => depth = depth.saturating_sub(1),
102      _ => {},
103    }
104    if depth == 0 && (t == T_CS!("\\\\") || t == T_CS!("\\newline")) {
105      out.push(T_CS!("\\quad"));
106    } else {
107      out.push(t);
108    }
109  }
110  Tokens::new(out)
111}
