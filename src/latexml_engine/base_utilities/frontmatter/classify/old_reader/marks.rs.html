3//! nothing (`drop_empty_marks`, which the API's entry cutter and the IJCAI chain call too). Moved unchanged by Z18m, which deleted `author_block/`: the first two, with

11/// Does a superscript's operand read as mark labels (`1`, `1,2`, `1, 3`, `a`, `\dagger`, `\mathrm{c}`, `*`), not as
12/// text shown raised (`\textsuperscript{equal contrib.}`, `\textsuperscript{(*)}`, `$^{\rm (a)}$`,
13/// `\textsuperscript{ (equal)}`)? Labels hold no word of three or more letters, and [`clean_frontmatter_labels`] gives
14/// each back as it is (its font switches and braces aside): it drops parentheses and writes a space between the
15/// letters of one as `_`, so a mark read from such text loses what it shows (77n r4 review probes d1, d3, d4, c4, d6).
16fn reads_as_labels(operand: &[Token]) -> bool {
17  let mut word = 0;
18  for t in operand {
19    word = if t.get_catcode() == Catcode::LETTER {
20      word + 1
21    } else {
22      0
23    };
24    if word >= 3 {
25      return false;
26    }
27  }
28  let text = Tokens::new(
29    operand
30      .iter()
31      .filter(|t| {
32        !matches!(t.get_catcode(), Catcode::BEGIN | Catcode::END)
33          && !(t.get_catcode() == Catcode::CS && t.with_str(|name| FONT_SWITCH.is_match(name)))
34      })
35      .copied()
36      .collect(),
37  )
38  .to_string();
39  let pieces: Vec<&str> = text
40    .split(',')
41    .map(str::trim)
42    .filter(|piece| !piece.is_empty())
43    .collect();
44  clean_frontmatter_labels(&text, "") == pieces
45}

47/// Do an author's marks end what it prints (`Ann Able$^{1}$`, `Ann$^{1}$\thanks{…}`): after the first of them nothing
48/// but further marks, spacing, braces and math shifts (`${}^{1}$`), and notes with their arguments — `\thanks`,
49/// `\footnote`, the note commands of `is_name_note_command`, a document's note macro, and the binding's `contacts`,
50/// whose argument goes to the frontmatter (authblk's `\affil` inside `\author`, 2605.03796)? Not when a line or
51/// anything visible follows them: `Ann Able$^{1}$ \\ Univ X`, `Ann Able$^{a}$ (ann@x.org)`, `Ann Able$^{1}$
52/// \orcidlink{…}` (77n r3 review probes m2, m9, m8). A superscript is a mark only when its operand reads as labels
53/// (`reads_as_labels`): `Ann Able$^{1}$\textsuperscript{equal contrib.}`, `\textsuperscript{(*)}`, `$^{\rm (a)}$` show
54/// text, which the list path would have cut to a label or dropped (77n r4 review probes d1, d3, d4, c4, d6). Public for
55/// authblk's `\author` (latexml_package), which reads a single marked name as a list only then: the list path shows a
56/// mark nothing answers after the name's last text. (A line break reaches that path by its own,
57/// `has_author_block_separator`, where a mark before it stays, 108a.)
58pub fn author_marks_end_name(stuff: &Tokens, contacts: &[Token]) -> bool {
59  let tokens = outside_notes(stuff.unlist_ref());
60  // (the index past a mark at `i`: `$^{1}$`, `^{1}`, `\textsuperscript{1}` and the mark commands)
61  let mark_end = |i: usize| -> Option<usize> {
62    let t = tokens[i];
63    if t == T_MATH!() && tokens.get(i + 1) == Some(&T_SUPER!()) {
64      sup_operand_at(&tokens, i + 2)
65        .filter(|(operand, next)| tokens.get(*next) == Some(&T_MATH!()) && reads_as_labels(operand))
66        .map(|(_, next)| next + 1)
67    } else if t == T_SUPER!() || is_mark_command(&t) {
68      sup_operand_at(&tokens, i + 1)
69        .filter(|(operand, _)| reads_as_labels(operand))
70        .map(|(_, next)| next)
71    } else {
72      None
73    }
74  };
75  let mut marked = false;
76  let mut i = 0;
77  while i < tokens.len() {
78    let t = tokens[i];
79    if let Some(next) = mark_end(i) {
80      marked = true;
81      i = next;
82    } else if !marked
83      || is_spacing(&t)
84      || matches!(
85        t.get_catcode(),
86        Catcode::BEGIN | Catcode::END | Catcode::MATH
87      )
88    {
89      // (the name before its first mark; after it, spacing, braces and math shifts)
90      i += 1;
91    } else if t.with_str(is_name_note_command) || is_note_macro(&t) || contacts.contains(&t) {
92      // (its optional argument and its arguments)
93      i += 1;
94      if tokens.get(i) == Some(&T_OTHER!("["))
95        && let Some(close) = tokens[i..].iter().position(|t| *t == T_OTHER!("]"))
96      {
97        i += close + 1;
98      }
99      while let Some(t) = tokens.get(i) {
100        match t.get_catcode() {
101          Catcode::SPACE => i += 1,
102          Catcode::BEGIN => i = skip_group(&tokens, i),
103          _ => break,
104        }
105      }
106    } else {
107      return false;
108    }
109  }
110  marked
111}

113/// The marks an author's name shows that read as labels (`reads_as_labels`), outside its notes and not inside a word
114/// (`rewrite_in_word_superscripts`: `Craig M$^{\mathrm{c}}$Rae` spells the name): what the name requests, kept as it
115/// prints. For authblk's `\author` kept whole on `\lx@add@creator` because text follows its marks (`Bob
116/// Baker$^{1}$\textsuperscript{equal contrib.}`, `Bob Baker$^{1}$ (bob@x.org)`), which still requests the affiliation
117/// its mark picks, as a marked name read as a list does — it requested nothing, and lost `$^1$Univ X` to the author
118/// before it (77n r5 review probes k1, k2, k5). The binding's `contacts` (authblk's `\affil` inside `\author`) are left
119/// out with their arguments: their marks label them. Only the labels are requested (`label_pieces`): a footnote symbol
120/// (`$^{*}$`, `$^{1,\dagger}$`'s dagger, the `\ast` glued to `1\ast`) is the name's own text, which a request nothing
121/// answered showed a second time after it (`Bob Baker$^{*}$ (bob@x.org)` printed "Bob Baker∗ (bob@x.org)∗"; 2605.10201's
122/// `$^{1\ast}$`, `$^{1\dagger}$` gained "∗,†" in its name; 77n r6 review probes s1, s3, s5, z1).
123pub fn author_label_marks(stuff: &Tokens, contacts: &[Token]) -> Vec<Tokens> {
124  let tokens = rewrite_in_word_superscripts(stuff.clone());
125  let v = tokens.unlist_ref();
126  let mut name = Vec::with_capacity(v.len());
127  let mut i = 0;
128  while i < v.len() {
129    if contacts.contains(&v[i]) {
130      i += 1;
131      if v.get(i) == Some(&T_OTHER!("["))
132        && let Some(close) = v[i..].iter().position(|t| *t == T_OTHER!("]"))
133      {
134        i += close + 1;
135      }
136      while let Some(t) = v.get(i) {
137        match t.get_catcode() {
138          Catcode::SPACE => i += 1,
139          Catcode::BEGIN => i = skip_group(v, i),
140          _ => break,
141        }
142      }
143    } else {
144      name.push(v[i]);
145      i += 1;
146    }
147  }
148  author_mark_operands(&name)
149    .into_iter()
150    .filter(|operand| reads_as_labels(operand))
151    .filter_map(|operand| label_pieces(&operand))
152    .map(Tokens::new)
153    .collect()
154}

156/// `tokens` without the marks that print nothing: a mark command's empty argument (`\textsuperscript{}`, `mark_commands`)
157/// and an empty math superscript (`$^{}$`), spaces and `\relax` aside, when the block holds no other mark. An empty mark is no mark
158/// (user ruling R11 2026-10-20): it makes no block a marked one (2604.19440's `\textsuperscript{}Univ. Grenoble Alpes,
159/// …` under four unmarked names, which pdflatex prints with no mark, an affiliation of all four). In a block other marks
160/// make a marked one it stays, as before: it labels nothing there, and each name it ends is an author's piece of its
161/// own (2604.22129's `Robert Maier\textsuperscript{}~~~Aljaz Bozic\textsuperscript{}` beside `David
162/// Recasens\textsuperscript{\rm 1}`, which a marked block's marker-less pieces would weld into one creator). A note's
163/// text is the note's, not the block's marks: what a `\thanks`, `\footnote` or `\footnotetext` (or a document's note
164/// command, `is_note_macro`) holds is kept as written, so the note is read on what the author wrote — by the XML's
165/// no-ink test as by the frontmatter observer's rehearsal of the same block (`\thanks{\textsuperscript{}}`: taken out,
166/// the XML's note was empty and dropped where the observer's kept it; OXIDIZED_DESIGN_DIVERGENCES #916).
167pub fn drop_empty_marks(tokens: Tokens) -> Tokens {
168  let original = tokens.clone();
169  let v = tokens.unlist();
170  // (a `\relax` in it prints nothing either: `Ann Able\textsuperscript{\relax}, Bob Baker`, the r7 re-review's n10)
171  let blank = |range: &[Token]| {
172    range.iter().all(|t| {
173      matches!(
174        t.get_catcode(),
175        Catcode::BEGIN | Catcode::END | Catcode::SPACE
176      ) || *t == T_CS!("\\relax")
177    })
178  };
179  let mut out = Vec::with_capacity(v.len());
180  let mut i = 0;
181  while i < v.len() {
182    if cs_in(&v[i], NOTE_TEXT_COMMANDS) || is_note_macro(&v[i]) {
183      // (the command, an optional `[…]`, and its braced text, as written)
184      let mut end = i + 1 + v[i + 1..].iter().take_while(|t| **t == T_SPACE!()).count();
185      if v.get(end) == Some(&T_OTHER!("[")) {
186        end = v[end..]
187          .iter()
188          .position(|t| *t == T_OTHER!("]"))
189          .map_or(v.len(), |at| end + at + 1);
190        end += v[end..].iter().take_while(|t| **t == T_SPACE!()).count();
191      }
192      let close = skip_group(&v, end).max(i + 1).min(v.len());
193      out.extend_from_slice(&v[i..close]);
194      i = close;
195      continue;
196    }
197    let (operand, after) = if mark_commands().contains(&v[i]) {
198      let open = i + 1 + v[i + 1..].iter().take_while(|t| **t == T_SPACE!()).count();
199      (open, 0)
200    } else if v[i].get_catcode() == Catcode::MATH && v.get(i + 1) == Some(&T_SUPER!()) {
201      (i + 2, 1)
202    } else {
203      out.push(v[i]);
204      i += 1;
205      continue;
206    };
207    let close = skip_group(&v, operand);
208    if v
209      .get(operand)
210      .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
211      && blank(&v[operand..close.min(v.len())])
212      && (after == 0
213        || v
214          .get(close)
215          .is_some_and(|t| t.get_catcode() == Catcode::MATH))
216    {
217      i = close + after;
218    } else {
219      out.push(v[i]);
220      i += 1;
221    }
222  }
223  let out = Tokens::new(out);
224  if has_author_marks(&rewrite_symbol_superscripts(out.clone())) {
225    original
226  } else {
227    out
228  }
229}
