9/// An author line's names parted by horizontal space alone, each such space written as the comma the split reads
10/// (OXIDIZED_DESIGN_DIVERGENCES #897): pdflatex sets the names apart, where the split saw one name. At the line's top
11/// level, outside math, after a name letter, two kinds of space part two names:
12///  - a wide horizontal space ([`wide_space_end`]: `\quad`, `\qquad`, a group of them, `\hspace{…}`, `\hskip`/`\kern`
13///    a wide length, `\enspace`, `\enskip`, `\hfill`), marked names or not: `Sergio Antoy\inst{1} \kern1em Michael
14///    Hanus\inst{2}` (llncs 1909.09562), `{\normalsize Georgios Milis\textsuperscript{1} \quad Panagiotis P.
15///    Filntisis…}` (2312.06613), `Jiahao Su\thanks{…}$^{\;\;12}$\thanks{…} \hskip2em Wonmin Byeon` (2002.09131);
16///  - a name's marks or notes ([`name_marks_end`]) and any space after them (` `, `\ `, `\:`, `\,`, `\;`, `~`, a line
17///    ending in `\`): `Sijia Chen$^{1,2,3,*}$ \: Yibo Wang$^{1,2,3,}$\thanks{…} \: Yi-Feng Wu$^{3}$` (2406.07115),
18///    `Junxian Li$^{1}$\thanks{…} \⏎Ruixuan Yang$^{2*}$` (2609.32353), `Li Lucy\textsuperscript{2} \, Su Lin Blodgett`
19///    (2310.15398), `J. Wang\altaffilmark{1} Y. F. Mao \altaffilmark{1} and J. Y. Wei` (aastex 0811.3255),
20///    `\textsc{Francesco Tombesi}\altaffilmark{3,4}⏎\textsc{Demosthenes Kazanas}` (1311.0077), `Edward
21///    Liverts\inst{1}\fnmsep\thanks{…}⏎Michael Mond\inst{1}` (0806.3855), `Hao Wang$^{1,2,*}$⏎Defu Lian$^{1,2}$`
22///    (2308.08563).
23///
24/// A narrow space with no mark before it parts nothing (`Florian P.\ Busch`, 2212.12575; ecai's `Ann Able \ Bob Baker`,
25/// 77n r2). In a line whose marks or notes lead its names a mark is the name's after it, so there a space parts two
26/// names before the next name's marks (`$^{1}$Ann Able \: $^{2}$Bob Baker`; `\thanks{…} {Carla Schenker,} ⏎\thanks{…}
27/// {Xiulin Wang,}`, 2406.12338, each note its own name's) or a wide space before the name. Only where every piece then
28/// reads as a name or qualifies one and two or more read as names ([`name_count`]): `John Smith\quad Bell Labs` and
29/// `Mc\kern1pt Donald` stay as written.
30pub(in crate::base_utilities) fn space_parted_names(line: Tokens) -> Tokens {
31  let v = line.unlist_ref();
32  if !v.iter().any(|t| {
33    matches!(
34      t.get_catcode(),
35      Catcode::CS | Catcode::MATH | Catcode::SUPER | Catcode::BEGIN
36    )
37  }) {
38    return line;
39  }
40  // (the names led by their marks or notes: `$^{1}$Ann Able`, `\thanks{…} {Carla Schenker,}`, 2406.12338)
41  let lead = leading_spacing_end(v);
42  let prefix_marked = marker_leads(&line) || name_marks_end(v, lead) > lead;
43  let mut cuts: Vec<(usize, usize)> = Vec::new();
44  let mut piece_start = 0;
45  let mut depth = 0usize;
46  let mut in_math = false;
47  let mut i = 0;
48  while i < v.len() {
49    if depth == 0 && !in_math {
50      if prefix_marked {
51        // a space, then the next name's marks or notes and the name
52        if is_narrow_space(&v[i]) || wide_space_end(v, i).is_some() {
53          let gap = space_run_end(v, i);
54          let wide = (i..gap).any(|k| wide_space_end(v, k).is_some());
55          let marks = name_marks_end(v, gap);
56          if names_before(&v[piece_start..i])
57            && ((marks > gap && begins_a_name(v, space_run_end(v, marks), false))
58              || (wide && begins_a_name(v, gap, true)))
59          {
60            cuts.push((i, gap));
61            piece_start = gap;
62          }
63          i = gap;
64          continue;
65        }
66      } else {
67        // (a wrapper whose name ends in its marks ends where they do: `\textbf{Youngho Cho\textsuperscript{1}} \⏎
68        // \textbf{Junho Kim\textsuperscript{1}}`, 2608.29120 — the wrapper is transparent, [`wrapped_name_marks_end`])
69        if let Some(end) = wrapped_name_marks_end(v, i) {
70          let marks = name_marks_end(v, end);
71          let gap = space_run_end(v, marks);
72          if gap > marks && names_before(&v[piece_start..end]) && begins_a_name(v, gap, false) {
73            cuts.push((marks, gap));
74            piece_start = gap;
75            i = gap;
76          } else {
77            i = marks;
78          }
79          continue;
80        }
81        let marks = name_marks_end(v, i);
82        if marks > i {
83          let gap = space_run_end(v, marks);
84          if gap > marks && names_before(&v[piece_start..i]) && begins_a_name(v, gap, false) {
85            cuts.push((marks, gap));
86            piece_start = gap;
87            i = gap;
88          } else {
89            i = marks;
90          }
91          continue;
92        }
93        if wide_space_end(v, i).is_some() {
94          let gap = space_run_end(v, i);
95          if names_before(&v[piece_start..i]) && begins_a_name(v, gap, false) {
96            // (the spaces before it go with it)
97            let mut start = i;
98            while start > piece_start && is_narrow_space(&v[start - 1]) {
99              start -= 1;
100            }
101            cuts.push((start, gap));
102            piece_start = gap;
103          }
104          i = gap;
105          continue;
106        }
107      }
108    }
109    // (`\(`…`\)` is math as `$`…`$` is: a mark's space inside `Ann Able\( ^{1} \)` parts nothing)
110    match v[i].get_catcode() {
111      Catcode::BEGIN => depth += 1,
112      Catcode::END => depth = depth.saturating_sub(1),
113      Catcode::MATH if depth == 0 => in_math = !in_math,
114      Catcode::CS if depth == 0 && v[i] == T_CS!("\\(") => in_math = true,
115      Catcode::CS if depth == 0 && v[i] == T_CS!("\\)") => in_math = false,
116      _ => {},
117    }
118    i += 1;
119  }
120  if cuts.is_empty() {
121    return line;
122  }
123  let mut out: Vec<Token> = Vec::with_capacity(v.len() + cuts.len());
124  let mut from = 0;
125  for (start, end) in cuts {
126    out.extend_from_slice(&v[from..start]);
127    out.extend([T_OTHER!(","), T_SPACE!()]);
128    from = end;
129  }
130  out.extend_from_slice(&v[from..]);
131  // (a comma set in no width, `\rlap,`/`\llap,`, is the comma it prints: kept, the marks the split moves before a
132  // glued comma become `\rlap`'s argument — `Tatsuo~{\sc Kobayashi}\rlap,${}^{1,}$\footnote{…}⏎Hiroaki…`,
133  // hep-ph0411288's `\rlap${}^{1,}$`, its math broken)
134  let lapped = |k: usize| {
135    (out[k] == T_CS!("\\rlap") || out[k] == T_CS!("\\llap"))
136      && out.get(k + 1) == Some(&T_OTHER!(","))
137  };
138  let out: Vec<Token> = (0..out.len())
139    .filter(|&k| !lapped(k))
140    .map(|k| out[k])
141    .collect();
142  if name_count(visible_name_text(&out).trim()) >= 2 {
143    Tokens::new(out)
144  } else {
145    line
146  }
147}

149/// Whether `line` is a line of two or more names parted by horizontal space alone ([`space_parted_names`] parts it),
150/// at its top level or inside one wrapper over the whole line, a font command or a group of declarations
151/// (`\textbf{Ludwig Schmidt \quad Serena Yeung-Levy}` under a row of names, the NeurIPS block of 2507.19054). Such a
152/// line is a names line wherever it stands, never an affiliation of the name above it: `Ann Able \quad Bob Baker\\ Cat
153/// Cole \quad Dan Dee` is four authors, where Cat Cole had been Bob Baker's affiliation (the rev107b review's B1, probes
154/// r01, q09, r04). A line holding an institution word beside a name parts nothing (`John Smith\quad Bell Labs`: every
155/// piece must read as a name), so such an unmarked line under one name stays its affiliation (user ruling R12). A row
156/// of name-shaped institutions with no institution word (`Carnegie Mellon \quad Johns Hopkins`) still parts — a RED
157/// residual (review rev107b2 F1, `author_row_of_name_shaped_institutions_red`), synthetic only.
158pub(in crate::base_utilities) fn space_parted_name_line(line: &Tokens) -> bool {
159  let parts = |tokens: Tokens| {
160    let spaced = spaced_name_separators(tokens);
161    space_parted_names(spaced.clone()).unlist_ref() != spaced.unlist_ref()
162  };
163  parts(line.clone())
164    || whole_line_cs_wrapper(line).is_some_and(|(cs, inner)| name_text_command(&cs) && parts(inner))
165    || whole_line_group(line).is_some_and(|(_, inner)| parts(inner))
166}

168/// An unmarked author block with each line of names parted by horizontal space alone ([`space_parted_name_line`]) under
169/// another line written as its own row of the names: after the block's names line — the first line, one an `\and`
170/// opens, or such a line — the line break is the space between two names (`\quad`), the line more names of the row
171/// above; after a line under the names (an affiliation, a gap), an `\and`, the line the next author group's names
172/// (1203.0321's `F. Inti Pelupessy\hspace{0.5cm}Simon Portegies Zwart` under the first row's institution, welded into
173/// it).
174/// The block's groups are parted at `\and` and `\quad` before its lines (`author_group_splits`), so read as written
175/// `Ann Able \quad Bob Baker\\ Cat Cole \quad Dan Dee` gave Bob Baker the affiliation "Cat Cole" (in a `\parbox`
176/// too, once `wrapper_over_lines_as_lines` gives its lines to the block: the rev107b review's B1, probes q09, r01, r04;
177/// `\\ \textbf{Ludwig Schmidt \quad Serena Yeung-Levy} \\` under a row of names, 2507.19054). A line holding an
178/// institution word parts nothing (a name-shaped institution row still does: rev107b2 F1), so an unmarked line under one name stays its affiliation (user ruling R12), and a row of
179/// addresses set by position under the names (`A \quad B \\ Univ A \quad Univ B`) stays one. A line of names a
180/// comma parts is the comma rules' (`names_list_continues`), as before.
181pub(in crate::base_utilities) fn space_parted_rows_joined(stuff: Tokens) -> Tokens {
182  let pieces = split_tokens_delimited(stuff.clone(), author_line_breaks());
183  if pieces.len() < 2 {
184    return stuff;
185  }
186  let opens_a_group = |piece: &Tokens| split_tokens(piece.clone(), author_and_splits()).len() > 1;
187  let mut out: Vec<Token> = Vec::with_capacity(stuff.unlist_ref().len() + pieces.len());
188  let mut changed = false;
189  // (whether the line before is a names line by where it stands)
190  let mut names_line = true;
191  for (at, (delimiter, piece)) in pieces.into_iter().enumerate() {
192    let opens = opens_a_group(&piece);
193    // (a comma list is the comma's, read as before — `names_list_continues`: `… Antoine Garcon,$^{1}$\\ Peter W.
194    // Graham,$^{2}$ Derek F. Jackson Kimball,$^{3}$ …`, 1901.10843, its separators printed as the commas)
195    let row = at > 0
196      && !opens
197      && !visible_name_text(piece.unlist_ref()).contains(',')
198      && space_parted_name_line(&piece);
199    if at > 0 {
200      if row {
201        out.push(if names_line {
202          T_CS!("\\quad")
203        } else {
204          T_CS!("\\and")
205        });
206        changed = true;
207      } else {
208        out.extend(delimiter);
209      }
210      out.push(T_SPACE!());
211    }
212    names_line = at == 0 || row || opens;
213    out.extend(piece.unlist());
214    out.push(T_SPACE!());
215  }
216  if changed { Tokens::new(out) } else { stuff }
217}

219/// Where the wrapper at `v[i]` ends when the name it holds ends in the name's marks or notes ([`name_marks_end`]): a
220/// font command over a group ([`name_text_command`]: `\textbf{Ann Able\textsuperscript{1}}`) or a group
221/// (`{\bf Ann Able$^{1}$}`), a name letter before its marks. A wrapper prints its marks where its name ends, so a
222/// space after it parts the next name as after the marks themselves (2608.29120).
223fn wrapped_name_marks_end(v: &[Token], i: usize) -> Option<usize> {
224  let t = v.get(i)?;
225  let open = if t.get_catcode() == Catcode::BEGIN {
226    i
227  } else if name_text_command(t)
228    && v
229      .get(i + 1)
230      .is_some_and(|g| g.get_catcode() == Catcode::BEGIN)
231  {
232    i + 1
233  } else {
234    return None;
235  };
236  let end = skip_group(v, open);
237  if end > v.len() || end < open + 3 || v[end - 1].get_catcode() != Catcode::END {
238    return None;
239  }
240  let inner = &v[open + 1..end - 1];
241  (0..inner.len())
242    .find(|&k| k > 0 && name_marks_end(inner, k) == inner.len() && names_before(&inner[..k]))
243    .map(|_| end)
244}

246/// Whether a piece of an author line prints a letter of name text ([`visible_name_text`]: not a graphic's file name,
247/// `\includegraphics[scale=0.06]{orcid.pdf}\hspace{1mm}Hossein Bahak`, 2312.07592).
248fn names_before(tokens: &[Token]) -> bool {
249  visible_name_text(tokens).chars().any(char::is_alphabetic)
250}

252/// Whether `t` is a narrow space: a space, a tie, a control space or return, `\,`, `\:`, `\>`, `\;`, `\thinspace`,
253/// `\space` ([`is_spacing`], [`is_spacing_command`]).
254fn is_narrow_space(t: &Token) -> bool { is_spacing(t) || is_spacing_command(t) }

256/// The end of the run of horizontal space that starts at `v[i]` (`i` when none): narrow spaces ([`is_narrow_space`]) and
257/// wide ones ([`wide_space_end`]).
258fn space_run_end(v: &[Token], mut i: usize) -> usize {
259  while i < v.len() {
260    if let Some(end) = wide_space_end(v, i) {
261      i = end;
262    } else if is_narrow_space(&v[i]) {
263      i += 1;
264    } else {
265      break;
266    }
267  }
268  i
269}

271/// The end of the wide horizontal space at `v[i]`, if one stands there: `\quad`, `\qquad`, `\enspace`, `\enskip`
272/// (latex.ltx `\kern.5em`, `\hskip.5em`), `\hfill`, `\hfil`, `\hspace` with its `*` and length, `\hskip` or `\kern` with
273/// a length of 4pt or more ([`past_glue`], [`wide_length`]: not a kern inside a word; `\hspace`'s length the same), or a
274/// group holding only spacing
275/// and a wide space (`{\quad}`, 2609.03563).
276pub(in crate::base_utilities) fn wide_space_end(v: &[Token], i: usize) -> Option<usize> {
277  let t = v.get(i)?;
278  if t.get_catcode() == Catcode::BEGIN {
279    let end = skip_group(v, i);
280    if end > v.len() || end < i + 3 || v[end - 1].get_catcode() != Catcode::END {
281      return None;
282    }
283    let inner = &v[i + 1..end - 1];
284    let mut j = 0;
285    let mut wide = false;
286    while j < inner.len() {
287      if let Some(next) = wide_space_end(inner, j) {
288        (wide, j) = (true, next);
289      } else if is_narrow_space(&inner[j]) {
290        j += 1;
291      } else {
292        return None;
293      }
294    }
295    return wide.then_some(end);
296  }
297  if t.get_catcode() != Catcode::CS {
298    return None;
299  }
300  t.with_str(|s| match s {
301    "\\quad" | "\\qquad" | "\\enspace" | "\\enskip" | "\\hfill" | "\\hfil" => Some(i + 1),
302    "\\hspace" => {
303      let mut j = i + 1;
304      if v.get(j) == Some(&T_OTHER!("*")) {
305        j += 1;
306      }
307      while v.get(j) == Some(&T_SPACE!()) {
308        j += 1;
309      }
310      if v.get(j).is_none_or(|g| g.get_catcode() != Catcode::BEGIN) {
311        return None;
312      }
313      let end = skip_group(v, j);
314      wide_length(&v[j + 1..end.saturating_sub(1).max(j + 1)]).then_some(end)
315    },
316    "\\hskip" | "\\kern" => {
317      let end = past_glue(v, i + 1);
318      wide_length(&v[i + 1..end]).then_some(end)
319    },
320    _ => None,
321  })
322}

324/// Whether a length as written (`1em`, `2em plus 1fil`, ` 10pt`) is wide enough to part two words: 4pt or more, an em
325/// read as 10pt, an ex as 4.3pt (cmr10), or `\fill`, `\stretch{…}` (`\hspace{\fill}`). Negative and unreadable lengths
326/// (`\hspace{1mm}` beside a graphic, a register) are not.
327fn wide_length(tokens: &[Token]) -> bool {
328  if tokens
329    .iter()
330    .find(|t| t.get_catcode() != Catcode::SPACE)
331    .is_some_and(|t| *t == T_CS!("\\fill") || *t == T_CS!("\\stretch"))
332  {
333    return true;
334  }
335  let text: String = tokens
336    .iter()
337    .filter(|t| t.get_catcode() != Catcode::SPACE)
338    .map(|t| t.to_string())
339    .collect();
340  let number_end = text
341    .find(|c: char| !(c.is_ascii_digit() || c == '.' || c == '+'))
342    .unwrap_or(text.len());
343  let Ok(value) = text[..number_end].trim_start_matches('+').parse::<f64>() else {
344    return false;
345  };
346  let unit = text[number_end..].get(..2).unwrap_or("");
347  let points = match unit {
348    "pt" => 1.0,
349    "em" => 10.0,
350    "ex" => 4.3,
351    "mm" => 2.845,
352    "cm" => 28.45,
353    "in" => 72.27,
354    "bp" => 1.004,
355    "pc" => 12.0,
356    "dd" => 1.07,
357    "cc" => 12.84,
358    "sp" => 1.0 / 65536.0,
359    _ => return false,
360  };
361  value * points >= 4.0
362}

364/// The end of the run of a name's marks and notes that starts at `v[i]` (`i` when none): `$^{…}$` and other math holding
365/// only marks ([`only_marks_and_notes`]), a superscript, a mark or note command with its `[…]` and braced argument (the
366/// mark commands of [`mark_commands`], [`NAME_ANNOTATIONS`]: `\textsuperscript`, `\thanks`, `\inst`, `\altaffilmark`,
367/// a document's note macro), `\footnotemark[…]`, and llncs's `\fnmsep`/`\fnsep` between two of them.
368pub(in crate::base_utilities) fn name_marks_end(v: &[Token], mut i: usize) -> usize {
369  let start = i;
370  loop {
371    let Some(t) = v.get(i) else {
372      return i;
373    };
374    match t.get_catcode() {
375      Catcode::MATH => {
376        let Some(close) = v[i + 1..]
377          .iter()
378          .position(|u| u.get_catcode() == Catcode::MATH)
379        else {
380          return i;
381        };
382        let inner = &v[i + 1..i + 1 + close];
383        if inner.is_empty() || !only_marks_and_notes(inner) {
384          return i;
385        }
386        i += close + 2;
387      },
388      Catcode::SUPER => {
389        let end = skip_group(v, i + 1);
390        if end > v.len() {
391          return i;
392        }
393        i = end;
394      },
395      // (a group holding only marks: `Jia-Bin Huang{$^{2}$} ~ David W. Jacobs{$^{2}$}`, 2406.09417)
396      Catcode::BEGIN => {
397        let end = skip_group(v, i);
398        if end > v.len() || end < i + 3 || v[end - 1].get_catcode() != Catcode::END {
399          return i;
400        }
401        let inner = &v[i + 1..end - 1];
402        if name_marks_end(inner, 0) != inner.len() {
403          return i;
404        }
405        i = end;
406      },
407      Catcode::CS if is_footnote_mark(t) || is_name_annotation(t) || is_mark_command(t) => {
408        let mut j = i + 1;
409        if v.get(j) == Some(&T_OTHER!("["))
410          && let Some(close) = v[j..].iter().position(|u| *u == T_OTHER!("]"))
411        {
412          j += close + 1;
413        }
414        if !is_footnote_mark(t) {
415          if v.get(j).is_none_or(|g| g.get_catcode() != Catcode::BEGIN) {
416            return i;
417          }
418          let end = skip_group(v, j);
419          if end > v.len() {
420            return i;
421          }
422          j = end;
423        }
424        i = j;
425      },
426      Catcode::CS
427        if i > start
428          && (*t == T_CS!("\\fnmsep") || *t == T_CS!("\\fnsep"))
429          && name_marks_end(v, i + 1) > i + 1 =>
430      {
431        i += 1;
432      },
433      _ => return i,
434    }
435  }
436}

438/// Whether a name begins at `v[i]`: a letter (not the word "and"), a group, a command that is no separator, mark or
439/// note, or — when `mark_leads`, the names led by their marks — a mark. Not a comma, `;`, a line's end or a separator.
440fn begins_a_name(v: &[Token], i: usize, mark_leads: bool) -> bool {
441  let Some(t) = v.get(i) else {
442    return false;
443  };
444  match t.get_catcode() {
445    Catcode::LETTER => {
446      let and = [T_LETTER!("a"), T_LETTER!("n"), T_LETTER!("d")];
447      !(v[i..].starts_with(&and)
448        && v
449          .get(i + 3)
450          .is_none_or(|u| u.get_catcode() != Catcode::LETTER))
451    },
452    Catcode::OTHER => t.with_str(|s| s.chars().next().is_some_and(char::is_alphabetic)),
453    Catcode::BEGIN => true,
454    Catcode::MATH | Catcode::SUPER => mark_leads && name_marks_end(v, i) > i,
455    Catcode::CS => {
456      if is_footnote_mark(t) || is_name_annotation(t) || is_mark_command(t) {
457        return mark_leads;
458      }
459      !(is_line_separator(t)
460        || *t == T_CS!("\\&")
461        || *t == T_CS!("\\par")
462        || *t == T_CS!("\\fnmsep")
463        || *t == T_CS!("\\fnsep")
464        || author_and_splits()
465          .into_iter()
466          .any(|d| matches!(d, SplitDelim::Token(d) if d == *t)))
467    },
468    _ => false,
469  }
