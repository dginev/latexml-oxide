6/// The spacing commands a run of which parts two author names (user ruling 2026-10-19): the control space `\ ` — also
7/// as the control return a line ending in `\ ` leaves, TeX dropping the line's trailing space (latex.ltx's `\^^M` is
8/// `\ `; cs0207008's `K.V. Hindriks \affila\ \ ⏎ W. van der Hoek`) —, the medium and thick spaces `\:`, `\>`, `\;`,
9/// the tie `~`, `\enspace` and `\enskip` (latex.ltx `\hskip.5em\relax`, a separator by itself only after a mark or note,
10/// [`normalize_hspace_separators`]: `A.\enskip B. Smith` is one name).
11pub(in crate::base_utilities) fn is_spacing_command(t: &Token) -> bool {
12  *t == T_ACTIVE!('~')
13    || (t.get_catcode() == Catcode::CS
14      && t.with_str(|s| {
15        matches!(
16          s,
17          "\\ " | "\\\r" | "\\:" | "\\>" | "\\;" | "\\enspace" | "\\enskip"
18        )
19      }))
20}

22/// `tokens` after the markup a line or an entry may open with before its mark: spaces, group openings and the font and
23/// size commands of [`name_text_command`] (`\normalsize{$^1$I3A - Universidad de Zaragoza} ~~~~~~
24/// \normalsize{$^2$École Polytechnique Fédérale de Lausanne}`, 2308.10525).
25pub(in crate::base_utilities) fn after_leading_markup(tokens: &[Token]) -> &[Token] {
26  let lead = tokens
27    .iter()
28    .take_while(|t| {
29      matches!(t.get_catcode(), Catcode::SPACE | Catcode::BEGIN) || name_text_command(t)
30    })
31    .count();
32  &tokens[lead..]
33}

35/// A token that ends an author line or a name, before any spacing run reads it so: the `\and` family, `\quad`,
36/// `\qquad`, a line break, the horizontal spaces [`normalize_hspace_separators`] makes `\quad`. Not `\par`: an author
37/// block is set in a `c` column (latex.ltx's `\@maketitle`, `\begin{tabular}[t]{c}\@author\end{tabular}`; ACL's the
38/// same), restricted horizontal mode, where a blank line's `\par` ends no paragraph and prints nothing (tex.web §1094
39/// `hmode+par_end` calls `end_graf`, §1096 acts only when `mode=hmode`): 2009.12421's `Ehsan
40/// Shareghi$^\spadesuit$$^\clubsuit$ <blank line> \ \ Nigel Collier$^\clubsuit$` prints the two names on one row.
41pub(in crate::base_utilities) fn is_line_separator(t: &Token) -> bool {
42  t.get_catcode() == Catcode::CS
43    && (author_affil_splits()
44      .iter()
45      .chain(author_and_splits().iter())
46      .chain(author_line_breaks().iter())
47      .any(|d| matches!(d, SplitDelim::Token(d) if d == t))
48      || t.with_str(|s| matches!(s, "\\qquad" | "\\hfill" | "\\hfil" | "\\hspace")))
49}

51/// The font commands an author line styles a name or an entry with besides the font declarations: their argument
52/// prints as the name it styles ([`shows_name_text`]).
53pub(in crate::base_utilities) const NAME_TEXT_ARGUMENT_COMMANDS: &[&str] = &[
54  "\\textbf",
55  "\\textit",
56  "\\textsc",
57  "\\textsl",
58  "\\textup",
59  "\\textmd",
60  "\\textrm",

66/// Whether `t` is a font or size command an author line styles a name or an entry with: their argument, or the group
67/// after the declaration (`frontmatter_stores::font_declaration`), prints as the name it styles
68/// ([`shows_name_text`]), and an entry may open with them before its mark ([`after_leading_markup`]).
69pub(in crate::base_utilities) fn name_text_command(t: &Token) -> bool {
70  crate::frontmatter_stores::font_declaration(t)
71    || crate::frontmatter_stores::cs_in(t, NAME_TEXT_ARGUMENT_COMMANDS)
