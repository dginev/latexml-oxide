23/// [`visible_name_text`]; `spelling`: the letters with a transliteration kept whole ([`spelling_name_text`]).
24fn name_text(tokens: &[Token], spelling: bool) -> String {
25  let mut text = String::new();
26  let mut i = 0;
27  while i < tokens.len() {
28    let t = tokens[i];
29    match t.get_catcode() {
30      Catcode::CS => {
31        // a control space or thin space between initials (`Joshua N.\ Winn`, `J.\,R. Smith`) is a space
32        if t == T_CS!("\\ ") || t == T_CS!("\\,") {
33          text.push(' ');
34        } else if t == T_CS!("\\&") {
35          text.push('&');
36        } else if t == T_CS!("\\and") || t == T_CS!("\\And") || t == T_CS!("\\AND") {
37          // (an author separator inside a wrapper reads as the word it prints: `{\bf {\large A \and B}}`, 2401.03555)
38          text.push_str(" and ");
39        } else if let Some(letter) = t.with_str(letter_command) {
40          // a letter written as a command (`{\L}ukasz`, `\O stergaard`) is that letter, the spaces ending the
41          // command's name aside (the letter it prints, where an address may spell it by its transliteration)
42          match t.with_str(printed_letter_command) {
43            Some(printed) if spelling && resolver::transliteration(printed).is_some() => {
44              text.push(printed)
45            },
46            _ => text.push(letter),
47          }
48          while tokens.get(i + 1) == Some(&T_SPACE!()) {
49            i += 1;
50          }
51        }
52        // (and the kernel's own contact call a creator's content holds, a heading's collaboration after the name or a
53        // link line's URL, OXIDIZED_DESIGN_DIVERGENCES #946: its keyvals and text are no name's — read, they ran into
54        // the name's last word, "WestXYZ", and the address spelling A~D~West went to no one, 1612.09318)
55        let annotation = is_name_annotation(&t) || t == T_CS!("\\lx@add@contact");
56        i += 1;
57        // an environment's `\begin{…}`/`\end{…}` prints no word, its name no more than an item's label does
58        // (`\begin{itemize}\item[\Letter] Bob Baker`, svjour3's `\institute` before `\at`, 2305.05902: "itemize" read
59        // as a word of the line, the rev98b review's pe1)
60        if t == T_CS!("\\begin") || t == T_CS!("\\end") {
61          while tokens.get(i) == Some(&T_SPACE!()) {
62            i += 1;
63          }
64          i = skip_group(tokens, i);
65          continue;
66        }
67        // an accent's spaces before its letter are not text (`Plech\' a\v{c}`, cond-mat9705101; tex.web §1123
68        // `\accent` skips them)
69        if t.with_str(|s| ACCENT_COMMANDS.contains(&s)) {
70          while tokens.get(i) == Some(&T_SPACE!()) {
71            i += 1;
72          }
73          // (the accented letter, where it has a transliteration and an address may spell it so: `\"a`, `\"{u}`, `\r a`)
74          if spelling && let Some((letter, end)) = accented_letter(&t, tokens, i) {
75            text.push(letter);
76            i = end;
77            continue;
78          }
79        }
80        // a control word's optional argument is no text (`\parbox[c]{\textwidth}{…}`, 2504.03146), nor is glue
81        // (`and Yael Shadmi$^{d}$ \vskip 0.3in`, hep-ph0107182)
82        if !annotation
83          && parts_words(&t)
84          && tokens.get(i) == Some(&T_OTHER!("["))
85          && let Some(close) = tokens[i..].iter().position(|u| *u == T_OTHER!("]"))
86        {
87          i += close + 1;
88        }
89        if ["\\vskip", "\\hskip", "\\kern"]
90          .iter()
91          .any(|cs| t == T_CS!(*cs))
92        {
93          i = past_glue(tokens, i);
94          if t == T_CS!("\\hskip") {
95            text.push(' ');
96          }
97          continue;
98        }
99        // an optional `[...]` (`\footnotemark[2]` has nothing more), then a note's or mark's braced argument
100        if annotation || is_footnote_mark(&t) {
101          if tokens.get(i) == Some(&T_OTHER!("[")) {
102            while i < tokens.len() && tokens[i] != T_OTHER!("]") {
103              i += 1;
104            }
105            i += 1;
106          }
107          if annotation {
108            i = skip_group(tokens, i);
109          }
110        }
111        // a space with its length is no text (`Yu.S.Velikzhanin \vspace{1mm}\\`, hep-ex0105093), a horizontal one a
112        // space
113        if t == T_CS!("\\vspace") || t == T_CS!("\\hspace") {
114          if tokens.get(i) == Some(&T_OTHER!("*")) {
115            i += 1;
116          }
117          i = skip_group(tokens, i);
118          if t == T_CS!("\\hspace") {
119            text.push(' ');
120          }
121          continue;
122        }
123        // another command's argument with no letter in it is an identifier or a mark, no name text
124        // (`\orcidicon{0000-0001-…}`, `\authorrefmark{1, 3}`: 2104.02493, 2407.03625); a letter is one of any script
125        // (`\textbf{Μαρία}`, its letters OTHER under pdfTeX's catcodes)
126        if !annotation
127          && tokens
128            .get(i)
129            .is_some_and(|n| n.get_catcode() == Catcode::BEGIN)
130        {
131          let end = skip_group(tokens, i).min(tokens.len());
132          let lettered = tokens[i..end].iter().any(|u| match u.get_catcode() {
133            Catcode::LETTER | Catcode::OTHER => u.with_str(|s| s.chars().any(char::is_alphabetic)),
134            Catcode::CS => u.with_str(letter_command).is_some(),
135            _ => false,
136          });
137          // (nor is a graphic's file name or a link's address: `\href{https://…}{\includegraphics[scale=0.06]{orcid.pdf}
138          // \hspace{1mm}Hossein Bahak, Farzaneh Taheri}`, 2312.07592)
139          if !lettered || t == T_CS!("\\includegraphics") || t == T_CS!("\\href") {
140            i = end;
141          }
142        }
143        continue;
144      },
145      Catcode::MATH => {
146        // a word set in math is name text — a surname given in math to carry its mark (`M. ${\rm Schreiber}^{*}$`,
147        // cond-mat9402093): a run of two or more letters, broken by anything but a command or a brace; a script and its
148        // operand (`$^{a}$`, `$^\mathrm{ab}$`) and a lone letter (`$e^+e^-$`) are not
149        i += 1;
150        let mut word = String::new();
151        let end_word = |word: &mut String, text: &mut String| {
152          if word.chars().count() >= 2 {
153            text.push_str(word);
154          }
155          word.clear();
156        };
157        while i < tokens.len() && tokens[i].get_catcode() != Catcode::MATH {
158          match tokens[i].get_catcode() {
159            Catcode::SUPER | Catcode::SUB => {
160              end_word(&mut word, &mut text);
161              i = skip_script_operand(tokens, i + 1);
162              continue;
163            },
164            Catcode::LETTER => tokens[i].with_str(|s| word.push_str(s)),
165            Catcode::CS | Catcode::BEGIN | Catcode::END => {},
166            _ => end_word(&mut word, &mut text),
167          }
168          i += 1;
169        }
170        end_word(&mut word, &mut text);
171      },
172      Catcode::SUPER | Catcode::SUB => {
173        i = skip_script_operand(tokens, i + 1);
174        continue;
175      },
176      // a run of bare asterisks after a name is its mark (`S. F. Pereira,* Z. Y. Ou,** and H. J. Kimble`,
177      // quant-ph0003094)
178      Catcode::OTHER if t == T_OTHER!("*") => {},
179      Catcode::LETTER | Catcode::OTHER => t.with_str(|s| text.push_str(s)),
180      Catcode::SPACE => text.push(' '),
181      // `~` is a space; another active character (babel's `"`) is no text
182      Catcode::ACTIVE if t.with_str(|s| s == "~") => text.push(' '),
183      _ => {},
184    }
185    i += 1;
186  }
187  text
188}

190/// The letter the accent command `accent` makes of its operand at `i` (a letter, or a group of one letter) when that
191/// letter has a transliteration ([`resolver::transliteration`]: `\"a` ä, `\"{U}` Ü, `\r a` å), with the index after
192/// the operand; `None` for any other.
193fn accented_letter(accent: &Token, tokens: &[Token], i: usize) -> Option<(char, usize)> {
194  use unicode_normalization::UnicodeNormalization;
195  let mark = accent.with_str(|s| match s {
196    "\\\"" => Some('\u{308}'),
197    "\\r" => Some('\u{30A}'),
198    _ => None,
199  })?;
200  let (letter, end) = match tokens.get(i)?.get_catcode() {
201    Catcode::BEGIN
202      if tokens
203        .get(i + 2)
204        .is_some_and(|t| t.get_catcode() == Catcode::END) =>
205    {
206      (tokens.get(i + 1)?, i + 3)
207    },
208    _ => (tokens.get(i)?, i + 1),
209  };
210  if !matches!(letter.get_catcode(), Catcode::LETTER | Catcode::OTHER) {
211    return None;
212  }
213  let base = letter.with_str(|s| {
214    let mut chars = s.chars();
215    chars.next().filter(|_| chars.next().is_none())
216  })?;
217  let composed: String = [base, mark].iter().collect::<String>().nfc().collect();
218  let mut chars = composed.chars();
219  let one = chars.next().filter(|_| chars.next().is_none())?;
220  resolver::transliteration(one).map(|_| (one, end))
221}

223/// The index after a script's operand at `i`: a brace group, a command with its brace group (`^\mathrm{ab}`), or one
224/// token.
225fn skip_script_operand(tokens: &[Token], i: usize) -> usize {
226  if tokens
227    .get(i)
228    .is_some_and(|t| t.get_catcode() == Catcode::CS)
229    && tokens
230      .get(i + 1)
231      .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
232  {
233    skip_group(tokens, i + 1)
234  } else {
235    skip_group(tokens, i)
236  }
237}

239/// The index after the brace group (or single token) at `i`; the end of `tokens` for a group that never closes.
240pub(in crate::base_utilities) fn skip_group(tokens: &[Token], i: usize) -> usize {
241  match tokens_group_end(tokens, i) {
242    Some(end) => end,
243    None
244      if tokens
245        .get(i)
246        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN) =>
247    {
248      tokens.len()
249    },
250    None => i + 1,
251  }
252}

425/// The lowercased whole words of `word`: itself, the punctuation around it (`.,;:()[]`) dropped, and the parts of a
426/// hyphenated compound (`Max-Planck-Institut`).
427pub(in crate::base_utilities) fn word_forms(word: &str) -> impl Iterator<Item = String> + '_ {
428  let bare = word.trim_matches(|c: char| ".,;:()[]".contains(c));
429  std::iter::once(bare)
430    .chain(bare.split('-'))
431    .map(str::to_lowercase)
432}

434/// Whether a word names an institution, a place or a role ([`NON_NAME_WORDS`], [`NON_NAME_ACRONYMS`]), never a person.
435pub(in crate::base_utilities) fn non_name_word(word: &str) -> bool {
436  let bare = word.trim_matches(|c: char| ".,;:()[]".contains(c));
437  word_forms(word).any(|form| NON_NAME_WORDS.contains(&form.as_str()))
438    || bare
439      .split('-')
440      .any(|part| NON_NAME_ACRONYMS.contains(&part))
441}

443/// `text` without the marks glued to its end (`Ann Able*`, `Bob Baker1,`): digits and symbols right after a letter.
444/// Digits after a space are text — a postal code (`CA 93106`), not a mark.
445pub(in crate::base_utilities) fn strip_glued_marks(text: &str) -> &str {
446  let text = text.trim();
447  // (a list's closing `;` too: sn-jnl's `\author[1]{A.~Able, B.~Baker, C.~Cole;}`, 2504.09158)
448  let stripped = text.trim_end_matches(|c: char| c.is_ascii_digit() || "*∗†‡§¶♯#,;".contains(c));
449  if stripped.len() < text.len() && stripped.ends_with(char::is_whitespace) {
450    text
451  } else {
452    stripped.trim_end()
453  }
454}

456/// Whether `text` — one part of an author line, as [`visible_name_text`] reads it — is shaped like a personal name:
457/// two to five words, each capitalised or an initial (`J.`, `D.A.`), name particles inside but not at either end, no
458/// digit, `@`, `:`, `/` or parenthesis and no institution word, once the marks glued to its end are dropped.
459fn name_shaped(text: &str) -> bool {
460  let text = strip_glued_marks(text);
461  if text
462    .chars()
463    .any(|c| c.is_ascii_digit() || "@:/()".contains(c))
464  {
465    return false;
466  }
467  let words: Vec<&str> = text.split_whitespace().flat_map(unglued_initials).collect();
468  if !(2..=5).contains(&words.len()) {
469    return false;
470  }
471  let particle = |w: &str| NAME_PARTICLES.contains(&w);
472  if particle(words[0]) || particle(words[words.len() - 1]) {
473    return false;
474  }
475  words.iter().enumerate().all(|(at, word)| {
476    if non_name_word(word) {
477      return false;
478    }
479    // (a particle elided onto the surname is the surname's right after an initial: `D.~d'Enterria`, hep-ph0311131 —
480    // and only there, this helper deciding today's reader too: a place or an institution elides as well, `C. Vignat,
481    // Université d'Orsay`, 1009.2396; `Villeneuve d'Ascq`, `Saint Martin d'Heres`)
482    let after_initial = at > 0 && is_initial(words[at - 1]);
483    let word = ["d'", "d’", "l'", "l’", "dell'", "dell’"]
484      .iter()
485      .filter(|_| after_initial)
486      .find_map(|elided| word.strip_prefix(elided))
487      .unwrap_or(word);
488    particle(word)
489      || (word.chars().next().is_some_and(char::is_uppercase)
490        && word
491          .chars()
492          .all(|c| c.is_alphabetic() || "'’.-".contains(c)))
493  })
494}

496/// A word of initials glued to the surname they precede, as its words: `A.G.Bogdanchikov` → `A.`, `G.`,
497/// `Bogdanchikov`, `Yu.M.Shatunov` → `Yu.`, `M.`, `Shatunov` (hep-ex0105093's collaboration list; an initial is a
498/// capital and at most one small letter, not a place's abbreviation: `St.Petersburg`, `Mt.Stromlo`); a word without a
499/// surname after its initials (`U.S.A.`, `D.A.`, `Ph.D.`) or with none glued stays whole.
500pub(in crate::base_utilities) fn unglued_initials(word: &str) -> Vec<&str> {
501  let mut parts = Vec::new();
502  let mut rest = word;
503  loop {
504    let mut chars = rest.char_indices().peekable();
505    let Some((_, capital)) = chars.next() else {
506      break;
507    };
508    chars.next_if(|(_, c)| c.is_lowercase());
509    match (chars.next(), chars.next()) {
510      (Some((dot, '.')), Some((next, after)))
511        if capital.is_uppercase()
512          && after.is_uppercase()
513          && !["St", "Mt", "Ft"].contains(&&rest[..dot]) =>
514      {
515        parts.push(&rest[..next]);
516        rest = &rest[next..];
517      },
518      _ => break,
519    }
520  }
521  // (a surname in capitals too, of three letters or more: `M.CACCIA`, hep-ex9910019; not an abbreviation's last letter,
522  // `I.N.F.N`)
523  let capitals_surname =
524    rest.chars().count() >= 3 && rest.chars().all(|c| c.is_uppercase() || "'’-".contains(c));
525  if parts.is_empty() || !(rest.chars().skip(1).any(char::is_lowercase) || capitals_surname) {
526    return vec![word];
527  }
528  parts.push(rest);
529  parts
530}

556/// Whether an author-list part qualifies the name before it instead of naming a person: a generational suffix or a
557/// degree ([`NAME_SUFFIXES`]), or a professional membership grade (`Member, IEEE`, `Senior Member`, or the society
558/// alone once the list is split at the grade's comma), which LaTeX prints after the name, comma-separated.
559pub(in crate::base_utilities) fn is_name_qualifier(text: &str) -> bool {
560  let text = strip_glued_marks(text);
561  if NAME_SUFFIXES.contains(&text) {
562    return true;
563  }
564  // (the rest of a list left unnamed is no one: `Ann Able and others`, `\emph{and others}`, `Ann Able, et al.`)
565  let words: String = text.split_whitespace().collect::<Vec<_>>().join(" ");
566  if ["others", "et al.", "et al", "et. al.", "et.al.", "et.al"]
567    .contains(&words.to_lowercase().as_str())
568  {
569    return true;
570  }
571  // an abbreviated grade before its society, glued or spaced (`M.ASCE`, `M. ASCE`, `A. M. ASCE`)
572  let compact: String = text.chars().filter(|c| !c.is_whitespace()).collect();
573  if NAME_GRADE_ABBREVIATIONS.iter().any(|grade| {
574    compact
575      .strip_prefix(grade)
576      .is_some_and(|society| NAME_GRADE_SOCIETIES.contains(&society))
577  }) {
578    return true;
579  }
580  grade_words(text, false)
581}

583/// Whether `text`'s words are all a membership grade's or its society's (`Senior Member, IEEE`); with `ranked`, one of
584/// them the grade's rank (`Member`, `Fellow`), so a society alone is none.
585pub fn grade_words(text: &str, ranked: bool) -> bool {
586  let words: Vec<&str> = text
587    .split(|c: char| c.is_whitespace() || c == ',' || c == '~')
588    .filter(|w| !w.is_empty())
589    .collect();
590  let grade = |w: &&str| {
591    [
592      "Life",
593      "Senior",
594      "Student",
595      "Graduate",
596      "Associate",
597      "Member",
598      "Fellow",
599    ]
600    .contains(w)
601  };
602  !words.is_empty()
603    && words
604      .iter()
605      .all(|w| grade(w) || NAME_GRADE_SOCIETIES.contains(w))
606    && (!ranked || words.iter().any(|w| ["Member", "Fellow"].contains(w)))
607}

609/// Whether `text` is a membership grade or a part of one the names line's commas cut it into (`Senior Member`, `IEEE`
610/// of `\textit{Senior Member}, \textit{IEEE}`, 2009.02373): [`is_name_qualifier`]'s grades, no suffix, degree or "et
611/// al.". The frontmatter observer's reading leaves it out of the name before it, as `\lx@add@author` takes the grade out
612/// of the name ([`written_membership_out`]).
613pub(in crate::base_utilities) fn is_membership_part(text: &str) -> bool {
614  let text = strip_glued_marks(text);
615  let compact: String = text.chars().filter(|c| !c.is_whitespace()).collect();
616  grade_words(text, false)
617    || NAME_GRADE_ABBREVIATIONS.iter().any(|grade| {
618      compact
619        .strip_prefix(grade)
620        .is_some_and(|society| NAME_GRADE_SOCIETIES.contains(&society))
621    })
622}

624/// The parts of an author-line text that may each be one name: split at commas, " and " and " & ", a leading "and"
625/// dropped.
626fn name_parts(text: &str) -> Vec<String> {
627  text
628    .split(',')
629    .flat_map(|part| {
630      part
631        .split(" and ")
632        .flat_map(|p| p.split(" & "))
633        .map(str::to_string)
634        .collect::<Vec<_>>()
635    })
636    .map(|part| {
637      let part = part.trim();
638      part
639        .strip_prefix("and ")
640        .or_else(|| part.strip_prefix("& "))
641        .unwrap_or(part)
642        .trim()
643        .to_string()
644    })
645    // (a separator alone, of a doubled one: `Barrientos, \& and Omar López-Cruz`, astro-ph9612009)
646    .filter(|part| !part.is_empty() && part != "&" && part != "and")
647    .collect()
648}

650/// How many parts of `text` are names when every part is a name or qualifies one ([`is_name_qualifier`]); 0 when a
651/// part is neither (an institution, a place).
652pub(in crate::base_utilities) fn name_count(text: &str) -> usize {
653  let parts = name_parts(text);
654  if parts.iter().all(|p| name_shaped(p) || is_name_qualifier(p)) {
655    // (a qualifier is no name, whatever its shape: `M.ASCE` reads as a glued initial and surname)
656    parts
657      .iter()
658      .filter(|p| name_shaped(p) && !is_name_qualifier(p))
659      .count()
660  } else {
661    0
662  }
663}

665/// Whether a word is an initial: capitals, each with at most one small letter and its period, hyphens between (`B.`,
666/// `V.I.`, `J.-P.`, `Yu.`), not a place's abbreviation (`St.`, `Mt.`, `Ft.`).
667fn is_initial(word: &str) -> bool {
668  let word = word.trim_end_matches([',', ';']);
669  !word.is_empty()
670    && !["St.", "Mt.", "Ft."].contains(&word)
671    && word.split('-').all(|part| {
672      let mut chars = part.chars().peekable();
673      let mut any = false;
674      while let Some(capital) = chars.next() {
675        if !capital.is_uppercase() {
676          return false;
677        }
678        chars.next_if(|c| c.is_lowercase());
679        if chars.next() != Some('.') {
680          return false;
681        }
682        any = true;
683      }
684      any
685    })
686}

688/// Whether a name's text carries what only a person's name does: an initial ([`is_initial`]: `Palash B. Pal`, `James
689/// M. Robins`) or a particle inside it ([`NAME_PARTICLES`]: `Ludwig van Beethoven`). Two capitalized words alone are as
690/// often a place, an institution's head or a role (`Johns Hopkins`, `Max Planck`, `Santa Barbara`, `Associate Dean`)
691/// as a name.
692pub(in crate::base_utilities) fn has_person_cue(text: &str) -> bool {
693  let words: Vec<&str> = text.split_whitespace().collect();
694  words.iter().any(|word| is_initial(word))
695    || (words.len() > 2
696      && words[1..words.len() - 1]
697        .iter()
698        .any(|word| NAME_PARTICLES.contains(word)))
699}

701/// Whether a personname's child is a mark: an `<ltx:sup>`, or a formula that is one (`$^{1}$`, `$\rm ^{a}$`, read from
702/// its structure as digested, before the math is rewritten into an `<ltx:sup>`: [`fm_read_formula`]; [`shown_marks`]).
703fn is_name_mark(node: &Node, qname: SymStr) -> bool {
704  with(qname, |q| q == "ltx:sup")
705    || (with(qname, |q| q == "ltx:Math")
706      && matches!(fm_read_formula(node), FmFormulaMarks::Marks(_)))
707}

709/// Where a `<ltx:personname>`'s printed lines part in [`personname_text`]: at a `<ltx:break/>` and at the edges of a
710/// paragraph or a box's block (`\parbox{…}{A, B \begin{center}(X Collaboration)\end{center}}`'s `<ltx:p>`s). A
711/// whitespace character, so the whole text reads with it as with a space.
712const PERSONNAME_LINE_BREAK: char = '\u{2028}';
713
714/// The elements whose edges part a personname's printed lines ([`PERSONNAME_LINE_BREAK`]).
715const PERSONNAME_LINE_ELEMENTS: [&str; 6] = [
716  "ltx:break",
717  "ltx:p",
718  "ltx:para",
719  "ltx:inline-para",
720  "ltx:inline-block",
721  "ltx:block",
722];
723
724/// The text a `<ltx:personname>` shows, without its marks and notes (`ltx:sup`, `ltx:note`, math); an element
725/// (a `<ltx:break/>`, a cell) parts the words around it, a line's edge ([`PERSONNAME_LINE_ELEMENTS`]) with a
726/// [`PERSONNAME_LINE_BREAK`]. With `marks_part`, each mark ([`is_name_mark`]) reads as a comma, as an author's mark
727/// after a name ends it.
728fn personname_text(node: &Node, marks_part: bool) -> String {
729  let mut text = String::new();
730  for child in node.get_child_nodes() {
731    match child.get_type() {
732      Some(NodeType::TextNode) => text.push_str(&child.get_content()),
733      Some(NodeType::ElementNode) => {
734        let qname = document::get_node_qname(&child);
735        let edge = if with(qname, |q| PERSONNAME_LINE_ELEMENTS.contains(&q)) {
736          PERSONNAME_LINE_BREAK
737        } else {
738          ' '
739        };
740        text.push(edge);
741        if marks_part && is_name_mark(&child, qname) {
742          text.push(',');
743        } else if !with(qname, |q| {
744          q == "ltx:sup" || q == "ltx:note" || q == "ltx:Math" || q == "ltx:contact"
745        }) {
746          text.push_str(&personname_text(&child, marks_part));
747        }
748        text.push(edge);
749      },
750      _ => {},
751    }
752  }
753  text
754}

756/// Whether `line` holds a word of two or more capitals and no small letter or dot (`IST`, `NVIDIA`; not the initials
757/// `J.-F.`) beside a word that has small letters: an acronym in a mixed-case line, an institution's (`IST Austria`);
758/// not a line set all in capitals (`KHIEM HONG PHAN`).
759pub(in crate::base_utilities) fn holds_mixed_case_acronym(line: &str) -> bool {
760  let words: Vec<&str> = line
761    .split_whitespace()
762    .map(|w| w.trim_matches(|c: char| !c.is_alphabetic()))
763    .filter(|w| w.chars().any(char::is_alphabetic))
764    .collect();
765  let acronym = |w: &&str| {
766    w.chars().filter(|c| c.is_uppercase()).count() >= 2
767      && !w.chars().any(|c| c.is_lowercase() || c == '.')
768  };
769  words.iter().any(acronym) && words.iter().any(|w| w.chars().any(char::is_lowercase))
770}

772/// How many names a personname's text reads as ([`name_count`]), its marks as written or each read as a comma
773/// ([`personname_text`]'s two readings, `plain` and `marked`): the plain reading when it finds two or more, else the
774/// larger.
775fn personname_name_count(plain: &str, marked: &str) -> usize {
776  match name_count(plain) {
777    n if n >= 2 => n,
778    n => n.max(name_count(marked)),
779  }
780}

782/// A frontmatter author whose name reads as two or more people ("Ann Able and Bob Baker") lost the boundaries between
783/// them: LaTeX typesets several authors where the document has one, and nothing else tells. Each such author is an
784/// error, once (the creator is marked, for the frontmatter a streaming conversion places late). Every part must be a
785/// name or qualify one (`Jane Doe, PhD`): a name beside its institution ("John Smith, Bell Labs") is not two people.
786pub(in crate::base_utilities) fn flag_merged_creators(document: &mut Document) -> Result<()> {
787  for mut creator in
788    document.findnodes("//ltx:creator[@role='author'][not(@_merged_checked)]", None)
789  {
790    document.set_attribute(&mut creator, "_merged_checked", "1")?;
791    // An author block laid out as a tabular or a minipage is kept whole on purpose (the old driver's `add_authors_calls`, deleted in DEL2; Perl
792    // Base_Utility.pool.ltxml:693): its names are there, in their layout.
793    let Some(person) = document.findnode(
794      "ltx:personname[not(.//ltx:tabular) and not(.//ltx:inline-block[contains(@class, 'ltx_minipage')])]",
795      Some(&creator),
796    ) else {
797      continue;
798    };
799    let plain = personname_text(&person, false);
800    let marked = personname_text(&person, true);
801    let text = plain.split_whitespace().collect::<Vec<_>>().join(" ");
802    // (a mark after a name ends it, as an author's mark stands: names parted only by marks and spacing,
803    // `FLORIAN REETZ$^1$ \ns PRIYA SUBRAMANIAN$^{2,3}$`, 1911.02873, read as one name without them)
804    let names = personname_name_count(&text, &marked.replace(PERSONNAME_LINE_BREAK, " "));
805    // (and each printed line read alone, so that a line that names no one beside the names does not hide them: a
806    // names paragraph above its collaboration's, `\parbox{\textwidth}{\Large A,$^{1,2}$ B,$^{3}$ … \begin{center} (DES
807    // Collaboration) \end{center}}`, 2311.03613 (55 names); names lines above their affiliations' line,
808    // `\textbf{A$^{1}$\thanks{…}, B$^{1}$, …} \\ \textbf{…} \\ $^{1}$MIT \qquad $^{2}$NUS`, 2609.38746: one line of two
809    // or more names is a merge; so are two or more lines each of names and nothing else, `Botian Shi <break/>Chiyu
810    // Wang`, 2306.04988 — names a `\\` parted. OXIDIZED_DESIGN_DIVERGENCES #897)
811    let lines: Vec<(usize, bool)> = plain
812      .split(PERSONNAME_LINE_BREAK)
813      .zip(marked.split(PERSONNAME_LINE_BREAK))
814      .filter(|(line, _)| line.chars().any(char::is_alphabetic))
815      .map(|(line, marked_line)| {
816        (
817          personname_name_count(line, marked_line),
818          holds_mixed_case_acronym(line),
819        )
820      })
821      .collect();
822    let names = names.max(lines.iter().map(|(n, _)| *n).max().unwrap_or(0));
823    // (a line set in mixed case holding an acronym names an institution, not a person: `Vage Egiazarian <break/>IST
824    // Austria`, 2504.20039, is one name over its affiliation)
825    let by_lines = lines.len() >= 2 && lines.iter().all(|&(n, acronym)| n > 0 && !acronym);
826    let (names, line_names) = if by_lines {
827      let sum = lines.iter().map(|(n, _)| *n).sum();
828      (names.max(sum), names < sum)
829    } else {
830      (names, false)
831    };
832    // (and a names line kept whole two or more of whose pieces are marked names, `A.~Able$^{1}$, B.~in der Baker,
833    // C.~Cole$^{1}$`: the merge its marks contradict, whatever its names' shape — the count of marked names the cautious
834    // split left on it, `\lx@author@merged@names`, 1903.10014, 1212.4143; a creator the writer made again from the
835    // reading carries none. OXIDIZED_DESIGN_DIVERGENCES #943)
836    let names = names.max(
837      creator
838        .get_attribute("_mergednames")
839        .and_then(|n| n.parse().ok())
840        .unwrap_or(0),
841    );
842    // (counted by its lines, each one name, the line under a name may be the next author's or the name's own
843    // affiliation welded to it — `Bob Baker <break/>Carnegie Mellon`, a line no institution word marks as one: the
844    // wording says which reading, the rev107b review's N2)
845    if names >= 2 && line_names && lines.iter().all(|&(n, _)| n == 1) {
846      let shown = plain
847        .split(PERSONNAME_LINE_BREAK)
848        .map(|line| line.split_whitespace().collect::<Vec<_>>().join(" "))
849        .filter(|line| line.chars().any(char::is_alphabetic))
850        .collect::<Vec<_>>()
851        .join(" / ");
852      Error!(
853        "frontmatter",
854        "merged_creators",
855        s!(
856          "One author holds {names} lines that each read as a name: \"{shown}\"; the names a line break parts, or a name and the line under it (its affiliation), were not told apart"
857        )
858      );
859    } else if names >= 2 {
860      Error!(
861        "frontmatter",
862        "merged_creators",
863        s!(
864          "One author holds {names} names: \"{text}\"; the separators between them were not recognized"
865        )
866      );
867    } else if names == 0 && reads_as_a_line_under_names(&text) {
868      // (a creator whose name is a line under the names — a legend's or an affiliation's text, `also at the Moscow
869      // Institute of Physics and Technology, Moscow, Russia` of 1202.2048's `$^{a}$` legend: the rev107b review's B2 —
870      // is never silent)
871      Error!(
872        "frontmatter",
873        "creator_not_a_name",
874        s!(
875          "An author's name reads as no name: \"{text}\"; a line under the names (an affiliation or a legend) was read as an author"
876        )
877      );
878    }
879  }
880  Ok(())
881}

883/// Whether a personname's text reads as a line under the names, no name: it opens with a word in lower case that is no
884/// name particle ([`NAME_PARTICLES`]: `van der Berg` opens a name) and no article, holds a word naming an institution
885/// ([`non_name_word`]), no address and no name ([`name_count`]) — a legend's text (`also at the Moscow Institute of
886/// Physics and Technology, Moscow, Russia`, `on leave from …`, `now at …`; 1309.4007's `e-mail:` line) or an
887/// affiliation's (`at the University of X`). A collaboration or a team named as an author opens with a capital or an
888/// article (`The ATLAS Collaboration`, `the ATLAS Collaboration`, `the PE11 team of Kepler WG#1`) and is not read; nor
889/// is a group named as an author in lower case (`sPHENIX Collaboration`, 2606.17184; `openJiuwen Team`, 2605.10052: a
890/// creator, [`reads_as_collaboration`]) or whom the authors speak for, a group's name read so after [`SPEAKS_FOR`]
891/// (`for the PAROS Clinical Research Network`, 2605.24212; not `for correspondence: Physics Department`).
892fn reads_as_a_line_under_names(text: &str) -> bool {
893  let Some(first) = text.split_whitespace().next() else {
894    return false;
895  };
896  let speaks_for_a_group = || {
897    SPEAKS_FOR.iter().any(|p| {
898      text
899        .get(..p.len())
900        .is_some_and(|q| q.eq_ignore_ascii_case(p))
901        && {
902          let rest = text[p.len()..].trim_start();
903          !rest.is_empty() && !reads_as_a_line_under_names(rest)
904        }
905    })
906  };
907  first.chars().next().is_some_and(char::is_lowercase)
908    && !NAME_PARTICLES.contains(&first)
909    && !["the", "a", "an", "and"].contains(&first)
910    && !text.contains('@')
911    && name_count(text) == 0
912    && text.split_whitespace().any(non_name_word)
913    && !reads_as_collaboration(text, true)
914    && !speaks_for_a_group()
915}
