16/// Perl's `\lx@add@authors` splits it ([`perl_author_calls`], this only for its tabular branch; OD #988). No raw-token preparation runs but the emitter's tabular juggles
17/// ([`tabular_juggles_in_a_whole_block`], which digestion can't answer: 1511.05911): digestion does each rewrite's job in a
18/// block nothing splits — `\\[len]` is `\lx@newline`'s own optional, a `\begin{footnote}` runs `\footnote`, a

24  keyvals: Option<Tokens>,

26) -> Result<Tokens> {
27  Ok(Invocation!(T_CS!("\\lx@add@author"), vec![
28    keyvals,
29    Some(written)

33/// Which of Perl's `\lx@add@authors` branches (Base_Utility.pool.ltxml:686-720) a block takes ([`perl_author_branch`]).
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub(in crate::base_utilities) enum PerlAuthorBranch {
36  /// `{tabular}`, `{minipage}` or `\halign` in the block: one `\lx@add@author` of it whole (:691-693)
37  Whole,
38  /// a superscript mark in the block: its lines cut at `@authoraffilsplits`, each an author line cut at its commas (and
39  /// " and ") or a marked affiliation, a line of no mark continuing the entry before it (:694-712) — but across the `\and`
40  /// family an author of its own (OD #52(g))
41  Marks,
42  /// neither: cut at `@authorsplits`, each piece an author whose `\\` lines are its affiliations (:713-719), the comma and
43  /// " and " cutting only the piece's names line
44  Split,
45}

48  /// What the decline's counted Warning says the block is written as.

50    match self {

58        "the block is written as Perl's \\lx@add@authors splits it at its separators, its pieces digested in order"
59      },
60    }
61  }

70  let text = stuff.to_string();

80/// Perl's `\lx@add@authors` (Base_Utility.pool.ltxml:676-720) of a block the frontmatter emitter declines to read
81/// (`observer::emitter::emit`'s footnote-counter and definition declines): its calls, in order, the `n`-th `\lx@add@author`
82/// (from 0) given `keys(n)` — and how many creators they make —, its `dequeueFrontMatter` the emitter's own (it writes the queue's creators). The tabular branch is
83/// [`whole_block_author`] of the block, its tabular juggles the line breaks they print
84/// ([`tabular_juggles_in_a_whole_block`]); the two split branches cut it as Perl does (`@authorsplits`,
85/// `@authoraffilsplits`, :678-682), each piece digested in its place, so the footnote counter and a definition step as
86/// TeX steps them. Perl's comma and " and " cut only a names line, never a `\\` line under it (the kernel's own lists,
87/// OXIDIZED_DESIGN_DIVERGENCES #52, #988: `author_affil_splits`, witness 2605.00347): Perl's cut PoS's `\\ Department of
88/// Physics, College of Science, Swansea University, …` into four authors (1302.3028) and 1110.6375's `D. Schmidiger,\\
89/// E. Wulf, S. Zhao and A. Zheludev$^\dag$` glued "D. SchmidigerE. Wulf, S. Zhao" (cr5_fncounter_and: Ann Able∗ and Bob Baker‡; 1003.2281, cs0404029, 2410.22720; the `\footremember`
90/// of 1404.5602, 2503.21719, 0809.5152, 1110.6375). There a juggle — a macro closing the class's tabular,
91/// `\alignauthor` — parts two authors, as `\and` does (`author_and_splits`; Perl's OmniBus makes it empty,
92/// OmniBus.cls.ltxml:76, 1003.2281's sig-alternate). A `\\[len]` or `\\*` cut leaves no "[len]" or "*" at the head of
93/// the next piece ([`strip_linebreak_options`], the kernel's own beyond-Perl fix of Perl's `\\[1em]` gap,
94/// Base_Utility.pool.ltxml:670, KNOWN_PERL_ERRORS #75: 2007.01340's creator "[1em] Julia Fukuyama"). With the calls,
95/// each creator's names as written, in order (the names line of its piece, not the affiliations under it): what the
96/// emitter matches the reading's names against (`emit`'s owner labels). Beyond Perl, where the kernel reads a line
97/// otherwise (OD #52, #988): a line of no mark that starts an author at the `\and` family is a names line, cut at its
98/// commas as one is. As Perl's `next unless $block->unlist` (:714), an empty name between two cuts (`\corrauth, and
99/// Zhengru Zhang`, 2004.03371) makes no creator. And each call's tokens as cut, its wrapping aside, with the creator
100/// it makes (`None` for an affiliation): what a call digests of the block, in order ([`PerlCalls::cut`]).
101pub(in crate::base_utilities) fn perl_author_calls(
102  keys: impl Fn(usize) -> Option<Tokens>,
103  stuff: Tokens,
104) -> Result<PerlCalls> {
105  let branch = perl_author_branch(&stuff);
106  if branch == PerlAuthorBranch::Whole {
107    return Ok(PerlCalls {
108      calls: vec![whole_block_author(
109        keys(0),
110        tabular_juggles_in_a_whole_block(stuff.clone()),
111      )?],
112      cut:   vec![(Some(0), stuff.unlist_ref().to_vec())],
113      names: vec![stuff],
114    });
115  }
116  let mut cut: Vec<(Option<usize>, Vec<Token>)> = Vec::new();
117  // (the creators the calls cut so far make: `cut`'s numbers, as `author` numbers its keys)
118  let mut made = 0usize;
119  let mut next = || {
120    made += 1;
121    made - 1
122  };
123  let stuff = strip_linebreak_options(tabular_juggles_as(stuff, T_CS!("\\and")));
124  let mut pieces: Vec<Tokens> = Vec::new();
125  let mut author = |names: Tokens, content: Tokens| {
126    let n = pieces.len();
127    pieces.push(names);
128    whole_block_author(keys(n), content)
129  };
130  let affiliation = |content: Tokens| -> Result<Tokens> {
131    Ok(Invocation!(T_CS!("\\lx@add@affiliation"), vec![
132      None,
133      Some(content)
134    ]))
135  };
136  let markers = perl_authorsup_markers();
137  // (the `\and` family parts two authors: OD #52(g)'s hard boundary, `author_and_splits`)
138  let and_family = author_and_splits();
139  let parts_authors = |cut: &[Token]| {
140    cut.first().is_some_and(|t| {
141      and_family
142        .iter()
143        .any(|d| matches!(d, SplitDelim::Token(u) if u == t))
144    })
145  };
146  // (a names line's own cuts, Perl's comma and " and " — at the top level, as `SplitTokens` cuts: a note's text keeps its
147  // commas)
148  let names = |line: Tokens| split_tokens(line, vec![T_OTHER!(",").into(), literal_and()]);
149  // (a names line's names, none empty: Perl's `next unless $block->unlist`, :714)
150  let people_of = |line: Tokens| -> Vec<Vec<Token>> {
151    names(line)
152      .into_iter()
153      .filter(|name| !name.is_empty())
154      .map(Tokens::unlist)
155      .collect()
156  };
157  let mut calls = Vec::new();
158  if branch == PerlAuthorBranch::Marks {
159    // (each entry: whether an author, and its tokens)
160    let mut entries: Vec<(bool, Vec<Token>)> = Vec::new();
161    for (cut, line) in split_tokens_delimited(stuff, author_affil_splits()) {
162      if line.is_empty() {
163        continue;
164      }
165      match position_of(&line, &markers) {
166        // (a line of no mark continues the entry before it with the cut it was cut at — a line break, `\quad` the space
167        // it prints —, which Perl's `Tokens(…)` drops: 1609.02993's "Soumith ChintalaFacebook AI Research", 1110.6375's
168        // "G. EhlersOak Ridge" glued, their words lost; but one cut at the `\and` family starts an author, the kernel's
169        // boundary, OD #52(g): `Ann Able$^{1}$… \and Bob Baker\\ $^{1}$Univ X` is two authors, as pdflatex prints them,
170        // where Perl glues "Ann AbleBob Baker" — a names line, cut at its commas as one is: `\and Bob Baker, Carl Cole\\`
171        // two authors, review revS5CR7b S3)
172        None => match entries.last_mut() {
173          Some(last) if !parts_authors(&cut) => {
174            last.1.extend(cut);
175            last.1.extend(line.unlist());
176          },
177          Some(_) => entries.extend(people_of(line).into_iter().map(|name| (true, name))),
178          None => entries.push((true, line.unlist())),
179        },
180        Some(p) if p < 8 => entries.push((false, line.unlist())),
181        Some(_) => entries.extend(names(line).into_iter().map(|name| (true, name.unlist()))),
182      }
183    }
184    for (is_author, line) in entries {
185      cut.push((is_author.then(&mut next), line.clone()));
186      let line = Tokens::new(line);
187      calls.push(if is_author {
188        author(
189          line.clone(),
190          Invocation!(T_CS!("\\lx@author@withsup"), vec![Some(line)]),
191        )?
192      } else {
193        affiliation(Invocation!(T_CS!("\\lx@affiliation@withsup"), vec![Some(
194          line
195        )]))?
196      });
197    }
198  } else {
199    let splits: Vec<SplitDelim> = vec![
200      T_CS!("\\and").into(),
201      T_CS!("\\And").into(),
202      T_CS!("\\AND").into(),
203      T_CS!("\\quad").into(),
204      T_CS!("\\qquad").into(),
205    ];
206    for block in split_tokens(stuff, splits) {
207      if block.is_empty() {
208        continue;
209      }
210      let mut lines = split_tokens(block, vec![T_CS!("\\\\").into()]).into_iter();
211      // (each person: their names as written, and their call's content)
212      let mut people: Vec<(Vec<Token>, Vec<Token>)> = lines
213        .next()
214        .map(people_of)
215        .unwrap_or_default()
216        .into_iter()
217        .map(|name| (name.clone(), name))
218        .collect();
219      if people.is_empty() {
220        people.push((Vec::new(), Vec::new()));
221      }
222      // (the block's `\\` lines its last name's, as Perl's one author of the piece has them)
223      let mut raw: Vec<Vec<Token>> = people.iter().map(|(names, _)| names.clone()).collect();
224      if let Some(last) = people.last_mut() {
225        for line in lines.filter(|line| !line.is_empty()) {
226          if let Some(raw) = raw.last_mut() {
227            raw.extend_from_slice(line.unlist_ref());
228          }
229          last.1.extend(affiliation(line)?.unlist());
230        }
231      }
232      for ((names, person), raw) in people.into_iter().zip(raw) {
233        cut.push((Some(next()), raw));
234        calls.push(author(Tokens::new(names), Tokens::new(person))?);
235      }
236    }
237  }
238  Ok(PerlCalls { calls, names: pieces, cut })
239}

252/// Does an author block mark its names (`$^{a}$`, `\textsuperscript{1}`)? A superscript inside a `\footnote` or
253/// `\thanks` is the note's own text, not a mark: 0911.0568's `LPT Orsay (CNRS)~\footnote{$^a$Laboratoire …}` line
254/// made its PoS author block a marked list, its affiliation and email line read as a person (63k).
255pub fn has_author_marks(stuff: &Tokens) -> bool {
256  let tokens = stuff.unlist_ref();
257  let mut i = 0;
258  while i < tokens.len() {
259    let t = &tokens[i];
260    if *t == T_CS!("\\footnote") || *t == T_CS!("\\thanks") {
261      i += 1;
262      while i < tokens.len() && tokens[i] == T_SPACE!() {
263        i += 1;
264      }
265      if i < tokens.len() && tokens[i] == T_OTHER!("[") {
266        while i < tokens.len() && tokens[i] != T_OTHER!("]") {
267          i += 1;
268        }
269        i += 1;
270      }
271      if i < tokens.len() && tokens[i].code == Catcode::BEGIN {
272        let mut depth = 0usize;
273        while i < tokens.len() {
274          match tokens[i].code {
275            Catcode::BEGIN => depth += 1,
276            Catcode::END => {
277              depth -= 1;
278              if depth == 0 {
279                break;
280              }
281            },
282            _ => {},
283          }
284          i += 1;
285        }
286      }
287    } else if is_author_mark(t) {
288      return true;
289    }
290    i += 1;
291  }
292  false
293}

295/// Drop the optional `*` and `[<len>]` that follow a `\\` line-break token in an
296/// author/affiliation block, before the block is split into lines. The
297/// author/affiliation splitters key on the bare `\\` control sequence, so a
298/// `\\[1em]` (or `\\*[1em]`) would otherwise orphan `[1em]` at the head of the
299/// next segment — where it leaks as literal `[1em]` text AND, by pushing a
300/// following `\textsuperscript` past the author-vs-affiliation position
301/// threshold, flips a comma-bearing affiliation line into phantom author
302/// creators. Perl's own `\lx@add@authors` flags this exact gap ("matching `\\`
303/// this way fails to catch `\\[1em]`, so really should Let it") but never fixes
304/// it, so both engines garble it identically; consuming it here is a beyond-Perl
305/// improvement (KNOWN_PERL_ERRORS #75, witness arXiv:2605.23553). The `[...]` is
306/// matched only as a balanced OTHER-catcode bracket group following the `\\` and
307/// the spaces after it, which `\@ifnextchar` skips (`\\ [0.5em]`, 2604.20841), so
308/// ordinary bracketed content elsewhere in a name/affiliation is untouched.
309pub fn strip_linebreak_options(tokens: Tokens) -> Tokens {
310  linebreak_options_stripped(tokens, false)
311}

313/// Whether the tokens of a `\\[…]` option, between its brackets, are a positive length: not led by a minus sign, and
314/// a register (`\bigskipamount`, `\baselineskip`) or a number with a non-zero digit (`12pt`, `.5cm`; not `0pt`).
315fn positive_length(option: &[Token]) -> bool {
316  let mut tokens = option.iter().filter(|t| **t != T_SPACE!());
317  let Some(first) = tokens.next() else {
318    return false;
319  };
320  *first != T_OTHER!("-")
321    && std::iter::once(first).chain(tokens).any(|t| {
322      t.get_catcode() == Catcode::CS || t.with_str(|s| s.chars().any(|c| ('1'..='9').contains(&c)))
323    })
324}

326pub(in crate::base_utilities) fn linebreak_options_stripped(
327  tokens: Tokens,
328  keep_gaps: bool,
329) -> Tokens {
330  let src = tokens.unlist();
331  let n = src.len();
332  let mut out: Vec<Token> = Vec::with_capacity(n);
333  let brk = T_CS!("\\\\");
334  // (the brace depth and the environment depth outside braces; whether the line since the last line break holds a wide
335  // space or a run of spacing at its top level, a row)
336  let (mut depth, mut environments) = (0usize, 0usize);
337  let mut row = false;
338  let mut i = 0;
339  while i < n {
340    out.push(src[i]);
341    if depth == 0
342      && (is_separating_space(&src[i])
343        || (is_spacing_command(&src[i])
344          && src[i + 1..]
345            .iter()
346            .find(|t| **t != T_SPACE!())
347            .is_some_and(is_spacing_command)))
348    {
349      row = true;
350    }
351    match src[i].get_catcode() {
352      Catcode::BEGIN => depth += 1,
353      Catcode::END => depth = depth.saturating_sub(1),
354      _ if depth == 0 && src[i] == T_CS!("\\begin") => environments += 1,
355      _ if depth == 0 && src[i] == T_CS!("\\end") => environments = environments.saturating_sub(1),
356      _ => {},
357    }
358    if src[i] == brk {
359      let top = depth == 0 && environments == 0;
360      let after_row = std::mem::take(&mut row);
361      let mut j = i + 1;
362      let mut consumed = false;
363      // (the spaces before the optional argument skipped, as `\@ifnextchar` skips them: latex.ltx's `\@xnewline`,
364      // `\@xarraycr`; `\\ [0.5em]`, 2604.20841, the r7 re-review's finding 8. Not before a star: `\\ * Equal
365      // contribution` stays a legend line, though `\@ifstar` skips them too and pdflatex prints no star there —
366      // 2608.23637, a SYNC_STATUS residual)
367      let after_spaces = |from: usize| {
368        from
369          + src[from.min(n)..]
370            .iter()
371            .take_while(|t| **t == T_SPACE!())
372            .count()
373      };
374      if j < n && src[j] == T_OTHER!("*") {
375        j += 1;
376        consumed = true;
377      }
378      if after_spaces(j) < n && src[after_spaces(j)] == T_OTHER!("[") {
379        j = after_spaces(j);
380        let mut k = j + 1;
381        let mut depth = 1usize;
382        while k < n && depth > 0 {
383          if src[k] == T_OTHER!("[") {
384            depth += 1;
385          } else if src[k] == T_OTHER!("]") {
386            depth -= 1;
387          }
388          k += 1;
389        }
390        if depth == 0 {
391          if keep_gaps && top && !after_row && positive_length(&src[j + 1..k - 1]) {
392            out.extend([T_SPACE!(), brk]);
393          }
394          j = k;
395          consumed = true;
396        }
397      }
398      if consumed {
399        i = j;
400        continue;
401      }
402    }
403    i += 1;
404  }
405  Tokens::new(out)
406}

408/// Rewrite horizontal-space macros into `\quad`, so the author/affiliation
409/// splitter treats them as separators. `\author{A \hspace{1cm} B \hspace{1cm} C}`
410/// (and `\hfill`-separated variants) is a "regular" way to lay out co-authors
411/// that LaTeXML's `\and`/`\quad` splitter otherwise misses, collapsing every name
412/// into one `<personname>`. `\quad` is already a hard separator in every author /
413/// affiliation split set, so rewriting to it needs no new delimiter plumbing.
414/// `\hspace`'s optional `*` and mandatory `{len}` argument are consumed together
415/// so the length cannot leak as literal text. OXIDIZED_DESIGN #52; witness
416/// arXiv:2506.06941 (six authors separated by `\hspace{0.5cm}`).
417///
418/// A run of two or more spacing commands ([`is_spacing_command`]) between two names at the block's top level is such a
419/// separator too (user ruling 2026-10-19, batch 77p): `\textbf{Ann Able}\textsuperscript{1}\;\;\textbf{Bob Baker}`
420/// (2610.20409), `Ann Able$^{1}$ \ \ \ \ Bob Baker` (2310.10873), ACL's `Shanshan Wang$^1$~~~~ Derek F. Wong$^1$`
421/// (2603.20851; `~~`, 2605.30274), 1702.05821, 1508.00093, 2009.12421 (OXIDIZED_DESIGN #52(o)). One spacing command
422/// stays inside a name (`J.~Smith`, `Ann\ Able`, `van~der~Berg`); see [`spacing_run_separator`] for what counts as the
423/// names on either side. A run or `\quad` inside a group parts nothing here: neurips_2024's 2406.09417 (one tie between
424/// spaces) and 2603.20775 (`{\large \bf A \quad B}`), named with the ruling, are that shape.
425pub(in crate::base_utilities) fn normalize_hspace_separators(tokens: Tokens) -> Tokens {
426  /// A length written as zero, `0pt`, `0.0mm`, `0`, `\z@`: an `\hspace` of it is no separator.
427  static ZERO_LENGTH: std::sync::LazyLock<Regex> = std::sync::LazyLock::new(|| {
428    Regex::new(r"^\s*(?:[+-]?(?:0+(?:\.0*)?|\.0+)\s*(?:pt|mm|cm|in|em|ex|bp|sp|pc|dd|cc|mu)?|\\z@(?:skip)?)\s*$")
429      .unwrap()
430  });
431  // (marked as the branch choice reads it, a footnote-symbol mark no mark: `\textbf{Vaibhav Singh}$^{*}$`, 2603.09835)
432  let marked_block = has_author_marks(&rewrite_symbol_superscripts(tokens.clone()));
433  let src = spacing_math_unwrapped(tokens.unlist());
434  let n = src.len();
435  let mut out: Vec<Token> = Vec::with_capacity(n);
436  // (the group depth and math mode at `src[i]`: a run is a separator only at the top level, outside math; whether a
437  // name letter — outside math, outside a mark command's argument, at the top level or inside a group that shows its
438  // letters as name text ([`shows_name_text`]) — came since the last top-level separator, so a run after a
439  // line-leading mark is the mark's space, `$\ast$~~Department of Physics`, hep-ph9708397, and one after an icon is
440  // the icon's, `\raisebox{-0.3ex}{\includegraphics[height=1.1em]{daedalus.jpg}}\;\;Daedalus Research Team`,
441  // 2604.10005; the depth of the mark argument being read)
442  let mut depth = 0usize;
443  let mut in_math = false;
444  let mut named = false;
445  let mut mark_argument: Option<usize> = None;
446  // (for each open group, whether its letters are name text; and the depth of an optional argument being read, whose
447  // letters are not, `\includegraphics[height=0.7em]`)
448  let mut shown: Vec<bool> = Vec::new();
449  let mut closed_shown = true;
450  let mut optional_argument = false;
451  // (whether the line opens with a mark — an affiliation line, `$^1$The University of Hong Kong~~~~~~$^2$Zhejiang Lab`,
452  // 2310.15160 —, where a run before the next mark parts two entries; `None` before its first token)
453  let mut mark_led: Option<bool> = None;
454  // (whether this is a line of names — the block's first, or the first after an `\and`; a line break ends it — and
455  // where in `out` the piece since the last top-level separator starts)
456  let mut names_line = true;
457  let mut piece_start = 0usize;
458  // (in an unmarked block, the line break that began this line, where in `out` it stands, after a line of names: a
459  // run that parts two names on this line makes it a row of names going on with the one before, the break a `\quad`
460  // — `\textbf{Pratham Grover}\\ \textbf{Ramakrishna B.~Bairi}\ \ \textbf{Ritabrata Maiti}`, 2603.09835, where
461  // "Ramakrishna B. Bairi" had read as an address line of Pratham Grover; a marked block reads its rows by their marks)
462  let mut row_break: Option<usize> = None;
463  // (where in `out` the names line starts, and where in `src` the line after it does: a row of names goes on with the
464  // names line only when it lists another number of names than it, or a line that is no row of as many entries
465  // follows it — [`row_of_names`])
466  let mut names_start = 0usize;
467  let mut row_start = 0usize;
468  // (where in `src` the last top-level mark or note ended, for an `\enskip` right after it)
469  let mut mark_end: Option<usize> = None;
470  let mut i = 0;
471  while i < n {
472    let t = src[i];
473    if depth == 0 && !in_math && mark_end.is_none_or(|at| at <= i) {
474      let marks = leading_marks_end(&src[i..]);
475      if marks > 0 {
476        mark_end = Some(i + marks);
477      }
478    }
479    if mark_led.is_none() && t.get_catcode() != Catcode::SPACE && t != T_CS!("\\par") {
480      mark_led = Some(starts_with_mark(after_leading_markup(&src[i..])));
481    }
482    // (the horizontal space a name sets before its own mark or note is the name's, no separator: `Oleg
483    // Lychkovskiy\thanks{…}\hspace*{2mm}$^{\rm a,b}$, Sergei Blinnikov\hspace*{2mm}$^{\rm a,c}$`, 0905.3658, where the
484    // `\quad` it made parted ", Sergei Blinnikov" off as an affiliation line; `\hspace*{1.3mm}$^{,\ddag}$`, 1312.1414)
485    if depth == 0
486      && !in_math
487      && named
488      && let Some(end) = space_before_own_mark(&src, i)
489    {
490      i = end;
491      continue;
492    }
493    match t.get_catcode() {
494      Catcode::BEGIN => {
495        depth += 1;
496        if mark_argument.is_none() && out.last().is_some_and(is_mark_command) {
497          mark_argument = Some(depth);
498        }
499        let outer = shown.last().copied().unwrap_or(true);
500        shown.push(outer && !optional_argument && shows_name_text(out.last(), closed_shown));
501      },
502      Catcode::END => {
503        if mark_argument == Some(depth) {
504          mark_argument = None;
505        }
506        depth = depth.saturating_sub(1);
507        closed_shown = shown.pop().unwrap_or(true);
508      },
509      Catcode::MATH => in_math = !in_math,
510      Catcode::LETTER
511        if !in_math
512          && mark_argument.is_none()
513          && !optional_argument
514          && shown.last().copied().unwrap_or(true) =>
515      {
516        named = true
517      },
518      Catcode::OTHER
519        if t == T_OTHER!("[") && out.last().is_some_and(|p| p.get_catcode() == Catcode::CS) =>
520      {
521        optional_argument = true
522      },
523      Catcode::OTHER if t == T_OTHER!("]") => optional_argument = false,
524      _ if depth == 0 && is_line_separator(&t) => {
525        (named, mark_led) = (false, None);
526        piece_start = out.len() + 1;
527        if t == T_CS!("\\\\") || t == T_CS!("\\newline") {
528          row_break = (names_line && !marked_block).then_some(out.len());
529          row_start = i + 1;
530          names_line = false;
531        } else if t == T_CS!("\\and") || t == T_CS!("\\And") || t == T_CS!("\\AND") {
532          (names_line, row_break) = (true, None);
533          names_start = out.len() + 1;
534        }
535      },
536      _ => {},
537    }
538    // (a line opens with a mark only in a marked block, a mark being what the branch choice reads as one: in a block
539    // marked only with footnote symbols — no mark there, [`rewrite_symbol_superscripts`] — a symbol-led line is an
540    // affiliation line read as any other line off the names line, `$^{*}$\daplablogo\ Columbia University\ \ \
541    // $^{\dagger}$University at Buffalo` (2609.35807; was a creator "†University at Buffalo", 77p r4 re-review))
542    let entry_mark_led = marked_block && mark_led == Some(true);
543    if depth == 0
544      && !in_math
545      && named
546      && is_spacing_command(&t)
547      && let Some(run) = spacing_run_separator(&src, &out, i, RunContext {
548        mark_led: entry_mark_led,
549        names_line,
550        piece_start,
551      })
552      // (in an unmarked block, off the names line, a run between two names parts them only on a row of names going on
553      // with the names line — the line right after it, [`row_of_names`]: not institutions under the names, `Ann
554      // Able~~~~Bob Baker \\ Carnegie Mellon~~~~Johns Hopkins`, nor names under an affiliation, `Ann Able \\ Univ A
555      // \\ Bob Baker~~~~Cat Cole`, which read as the affiliation lines they stand as, nor on a line a footnote symbol
556      // leads, `$^{*}$Univ A~~~~$^{\dagger}$Univ B` (2609.35807); a marked block reads its lines by their marks)
557      && (names_line
558        || marked_block
559        || row_break
560          .is_some_and(|at| row_of_names(&out[names_start.min(at)..at], &src[row_start..])))
561    {
562      if !names_line
563        && !entry_mark_led
564        && let Some(at) = row_break.take()
565      {
566        out[at] = T_CS!("\\quad");
567        names_line = true;
568      }
569      if run.and_word {
570        out.extend([
571          T_SPACE!(),
572          T_LETTER!("a"),
573          T_LETTER!("n"),
574          T_LETTER!("d"),
575          T_SPACE!(),
576        ]);
577      } else {
578        out.push(T_CS!("\\quad"));
579      }
580      (named, mark_led) = (false, None);
581      piece_start = out.len();
582      i = run.end;
583      continue;
584    }
585    if t == T_CS!("\\hspace") {
586      let mut j = i + 1;
587      // optional `*` (\hspace*)
588      if j < n && src[j] == T_OTHER!("*") {
589        j += 1;
590      }
591      // mandatory `{len}` group
592      let length_start = j + 1;
593      if j < n && src[j] == T_BEGIN!() {
594        let mut depth = 1usize;
595        j += 1;
596        while j < n && depth > 0 {
597          if src[j] == T_BEGIN!() {
598            depth += 1;
599          } else if src[j] == T_END!() {
600            depth -= 1;
601          }
602          j += 1;
603        }
604      }
605      // (a zero length inside a group, a mark's argument or math prints no gap and parts nothing:
606      // `\textsuperscript{1\hspace{0pt}2}` prints the label 12, as `\hskip0pt` is no spacing,
607      // `util_text::is_blank_item`; at the top level of the author line it stays the `\quad` that parts two names,
608      // `Ann Able\hspace{0pt}Bob Baker` two creators)
609      let zero = (depth > 0 || in_math || mark_argument.is_some())
610        && length_start < j
611        && ZERO_LENGTH.is_match(&Tokens::new(src[length_start..j - 1].to_vec()).to_string());
612      if !zero {
613        out.push(T_CS!("\\quad"));
614      }
615      i = j;
616      continue;
617    }
618    if t == T_CS!("\\hfill") || t == T_CS!("\\hfil") {
619      out.push(T_CS!("\\quad"));
620      i += 1;
621      continue;
622    }
623    // (latex.ltx's `\enskip`, `\hskip.5em\relax`, is such a space too, at the top level right after a mark or note:
624    // `\normalsize Dominic W.\ Berry\thanks{…} \enskip \normalsize Andrew M.\ Childs\thanks{…}…`, 1312.1414, five
625    // names read as one; elsewhere it is a spacing command, one of which stays inside a name, `J.\enskip Smith$^{1}$`)
626    if t == T_CS!("\\enskip")
627      && depth == 0
628      && !in_math
629      && mark_end.is_some_and(|at| src[at..i].iter().all(|u| u.get_catcode() == Catcode::SPACE))
630    {
631      out.push(T_CS!("\\quad"));
632      i += 1;
633      continue;
634    }
635    out.push(t);
636    i += 1;
637  }
638  Tokens::new(out)
639}

641/// The horizontal spaces an author line parts its entries with ([`normalize_hspace_separators`]), which are separators
642/// at its top level: `\hspace` (its `*` and length read with it), `\hfill`, `\hfil`, `\quad`, `\qquad`. (Not `\enskip`,
643/// a separator only after a mark or note, else a spacing command, [`is_spacing_command`].)
644fn is_separating_space(t: &Token) -> bool {
645  t.get_catcode() == Catcode::CS
646    && t.with_str(|s| matches!(s, "\\hspace" | "\\hfill" | "\\hfil" | "\\quad" | "\\qquad"))
647}

649/// The run of horizontal space that opens `tokens` — spaces, separating spaces ([`is_separating_space`], `\hspace`'s `*`
650/// and length read with it) and spacing commands ([`is_spacing_command`]) —: where it ends, how many spacing commands
651/// it holds, and whether it holds a separating space.
652fn horizontal_space_run(tokens: &[Token]) -> (usize, usize, bool) {
653  let (mut end, mut commands, mut separating) = (0, 0, false);
654  while let Some(t) = tokens.get(end) {
655    if *t == T_CS!("\\hspace") {
656      separating = true;
657      end += 1;
658      if tokens.get(end) == Some(&T_OTHER!("*")) {
659        end += 1;
660      }
661      end = skip_group(tokens, end);
662    } else if is_separating_space(t) {
663      separating = true;
664      end += 1;
665    } else if is_spacing_command(t) {
666      commands += 1;
667      end += 1;
668    } else if t.get_catcode() == Catcode::SPACE {
669      end += 1;
670    } else {
671      break;
672    }
673  }
674  (end, commands, separating)
675}

677/// Where a run of horizontal space that starts at `src[start]` with a separating space ([`is_separating_space`]) ends,
678/// when what follows it is a mark or a note of the name before it ([`leading_marks_end`]: `$^{a,b}$`,
679/// `\textsuperscript{1}`, `\thanks{…}`) that no entry text follows — a comma, `;`, the word "and", a separator, a line
680/// break, a separating run of spacing (a separating space, or two or more spacing commands, OXIDIZED_DESIGN #52(o)) or
681/// the end come after it —, so the space is the name's: `Oleg L.\hspace*{2mm}$^{\rm a,b}$, Sergei B.` (0905.3658). A
682/// `\footnotemark[n]` is such a mark here, `\protect` before a mark no text ([`own_marks_end`]).
683/// `None` before a mark that leads an entry (`$^1$Univ A\quad$^2$Univ B`, an affiliation line's two entries; with one
684/// tie or control space after the mark, the entry's own, `$^1$~Univ A\quad$^2$~Univ B` and `\textsuperscript{2}\ Univ
685/// B`, whose two universities read as one) and before anything else.
686fn space_before_own_mark(src: &[Token], start: usize) -> Option<usize> {
687  if !src.get(start).is_some_and(is_separating_space) {
688    return None;
689  }
690  let end = start + horizontal_space_run(&src[start..]).0;
691  let marks = own_marks_end(&src[end..]);
692  if marks == 0 {
693    return None;
694  }
695  let rest = &src[end + marks..];
696  // (the spacing after the marks: a separating run ends the entry there — an `\enskip` too, which right after a mark or
697  // note is a separator, `Childs\thanks{…}\hspace*{1.1mm}$^{,}$\thanks{…} \enskip \normalsize Richard Cleve`,
698  // 1312.1414 —; one other spacing command, and the spaces around it, is the next entry's, which the token after it
699  // tells)
700  let (after, commands, separating) = horizontal_space_run(rest);
701  if separating || commands >= 2 || rest[..after].contains(&T_CS!("\\enskip")) {
702    return Some(end);
703  }
704  let rest = &rest[after..];
705  let own = match rest.first() {
706    None => true,
707    Some(t) if *t == T_OTHER!(",") || *t == T_OTHER!(";") || *t == T_END!() => true,
708    Some(t) if is_line_separator(t) || *t == T_CS!("\\par") => true,
709    Some(_) => and_word_end(rest, false).is_some(),
710  };
711  own.then_some(end)
712}

714/// [`leading_marks_end`], with what a name also sets as its own mark after a space: a `\footnotemark[n]`, and `\protect`
715/// before a mark (`Zhengyi Han\hspace{0.1em}\protect\footnotemark[1]`, 2609.40220, a creator of the mark alone once
716/// authblk's gate read the space, `has_author_block_separator`).
717fn own_marks_end(tokens: &[Token]) -> usize {
718  let mut end = 0;
719  loop {
720    let at = end + usize::from(tokens.get(end) == Some(&T_CS!("\\protect")));
721    let next = if tokens.get(at).is_some_and(is_footnote_mark) {
722      let optional = (tokens.get(at + 1) == Some(&T_OTHER!("[")))
723        .then(|| tokens[at + 1..].iter().position(|t| *t == T_OTHER!("]")))
724        .flatten();
725      at + 1 + optional.map_or(0, |close| close + 1)
726    } else {
727      at + leading_marks_end(&tokens[at..])
728    };
729    if next == at {
730      return end;
731    }
732    end = next;
733  }
734}

736/// `tokens` with each math group of its top level that holds only spacing — `$\qquad$` — written as that spacing,
737/// outside math, which the separators read as written so (`Michael J. Wilber$^{1,2}$ $\qquad$ Vitaly
738/// Shmatikov$^{1,2}$ $\qquad$ Serge Belongie$^{1,2}$`, 1602.04504, whose three names read as one); and so each brace
739/// group of its top level that holds only spacing with a `\quad` or `\qquad` in it, no command's argument (`Byeongjun
740/// Park$^{1}${\quad} Byung-Hoon Kim$^{1,2,*}${\quad} …\\ $^{1}$EverEx{\quad}$^{2}$Yonsei University`, 2609.03563:
741/// one creator, its institutions one; a glue's group changes nothing, OXIDIZED_DESIGN_DIVERGENCES #897).
742fn spacing_math_unwrapped(tokens: Vec<Token>) -> Vec<Token> {
743  let spacing = |t: &Token| {
744    t.get_catcode() == Catcode::SPACE
745      || is_spacing_command(t)
746      || (t.get_catcode() == Catcode::CS && t.with_str(|s| matches!(s, "\\quad" | "\\qquad")))
747  };
748  let mut out = Vec::with_capacity(tokens.len());
749  let mut depth = 0usize;
750  let mut in_math = false;
751  let mut i = 0;
752  while i < tokens.len() {
753    let t = tokens[i];
754    match t.get_catcode() {
755      Catcode::BEGIN
756        if depth == 0
757          && !in_math
758          && !out.last().is_some_and(|p: &Token| {
759            p.get_catcode() == Catcode::CS || *p == T_OTHER!("*") || *p == T_OTHER!("]")
760          }) =>
761      {
762        let end = skip_group(&tokens, i);
763        let inner = &tokens[i + 1..end.saturating_sub(1).max(i + 1)];
764        if end <= tokens.len()
765          && tokens[end - 1].get_catcode() == Catcode::END
766          && !inner.is_empty()
767          && inner.iter().all(spacing)
768          && inner
769            .iter()
770            .any(|u| u.with_str(|s| matches!(s, "\\quad" | "\\qquad")))
771        {
772          out.extend(
773            inner
774              .iter()
775              .copied()
776              .filter(|u| u.get_catcode() != Catcode::SPACE),
777          );
778          i = end;
779          continue;
780        }
781        depth += 1;
782      },
783      Catcode::BEGIN => depth += 1,
784      Catcode::END => depth = depth.saturating_sub(1),
785      Catcode::MATH if depth == 0 && !in_math => {
786        let close = tokens[i + 1..]
787          .iter()
788          .position(|u| u.get_catcode() == Catcode::MATH);
789        if let Some(close) = close {
790          let inner = &tokens[i + 1..i + 1 + close];
791          if inner.iter().all(spacing) && inner.iter().any(|u| u.get_catcode() != Catcode::SPACE) {
792            out.extend(
793              inner
794                .iter()
795                .copied()
796                .filter(|u| u.get_catcode() != Catcode::SPACE),
797            );
798            i += close + 2;
799            continue;
800          }
801        }
802        in_math = true;
803      },
804      Catcode::MATH => in_math = !in_math,
805      _ => {},
806    }
807    out.push(t);
808    i += 1;
809  }
810  out
811}

813/// Whether the line after a line of names `names` (as rewritten so far, the rows it already goes on with included) —
814/// `rest`, from just after the line break —, which a spacing run parts into names, is a row of more names going on with
815/// them rather than the institutions under them, one under each name (`Ann Able~~~~Bob Baker \\ Carnegie
816/// Mellon~~~~Johns Hopkins`, a column per author; `Ann Able \\ Carnegie Mellon~~~~Johns Hopkins`). The names line holds
817/// two or more names, and the row lists another number of names (2603.09835, 2607.07318, 2603.07624, 2603.16856: rows
818/// of five over three, two over four), or a line follows it that is no row of as many entries — the address under all
819/// the rows (2605.24414's six names over six over three, then `\textit{JIUTIAN Research}`; 2608.21409's three over
820/// three over its e-mail line; 2607.09029's five over five over `LiAuto Inc.`). A row of institutions under the names
821/// ends the block, or has a row of as many entries under it (their e-mails). Markup does not tell the two apart:
822/// 2607.07318's and 2603.16856's second rows open with `{\bf`/`\bf` under plain first rows.
823///
824/// A line of addresses under the row says whose they are ([`row_spelled_by_addresses`]): the row is names only when an
825/// address spells one of its pieces, or the addresses are as many as the names of both rows together — not institutions
826/// over the authors' addresses, `Ann Able~~~~Bob Baker \\ Carnegie Mellon~~~~Johns Hopkins \\
827/// \texttt{\{aa,bb\}@x.edu}`, nor three over three over one address (probes v1, v3: were creators "Carnegie Mellon",
828/// "Johns Hopkins", 77p r4 re-review).
829fn row_of_names(names: &[Token], rest: &[Token]) -> bool {
830  let names_count = row_name_count(names);
831  if names_count < 2 {
832    return false;
833  }
834  let rest = after_line_break_options(rest);
835  let mut at = rest.iter().position(is_row_break).unwrap_or(rest.len());
836  let row = &rest[..at];
837  let row_count = row_name_count(row);
838  // (the next line that prints something, past the lines of spacing alone: `\vspace{2mm}\\`)
839  let mut under: Option<&[Token]> = None;
840  while rest
841    .get(at)
842    .is_some_and(|t| *t == T_CS!("\\\\") || *t == T_CS!("\\newline"))
843  {
844    let start = rest.len() - after_line_break_options(&rest[at + 1..]).len();
845    let end = start
846      + rest[start..]
847        .iter()
848        .position(is_row_break)
849        .unwrap_or(rest.len() - start);
850    let line = &rest[start..end];
851    if spacing_end(line) > 0 {
852      under = Some(line);
853      break;
854    }
855    at = end;
856  }
857  if let Some(addresses) = under.and_then(|line| email_addresses(&Tokens::new(line.to_vec()))) {
858    return addresses.len() == names_count + row_count
859      || row_spelled_by_addresses(names, row, &addresses);
860  }
861  if row_count != names_count {
862    return true;
863  }
864  under.is_some_and(|line| row_entry_count(line) != names_count)
865}

867/// Whether one of `addresses` spells a piece of `row` ([`spelled_owners`], the names line's pieces `names` spelled
868/// against too, so an address spells the row's piece only when it spells it best).
869fn row_spelled_by_addresses(names: &[Token], row: &[Token], addresses: &[String]) -> bool {
870  let pieces = |tokens: &[Token]| -> Vec<NameKeys> {
871    let mut keys = Vec::new();
872    let mut i = 0;
873    while i < tokens.len() && !is_row_break(&tokens[i]) {
874      let piece = next_name_piece(&tokens[i..]);
875      if piece.is_empty() {
876        i += 1;
877        continue;
878      }
879      if has_name_letter(piece) {
880        keys.push(NameKeys::of(&Tokens::new(outside_notes(piece))));
881      }
882      i += piece.len();
883    }
884    keys
885  };
886  let mut keys = pieces(names);
887  let row_from = keys.len();
888  keys.extend(pieces(row));
889  addresses.iter().any(|address| {
890    spelled_owners(&keys, address)
891      .iter()
892      .any(|owner| *owner >= row_from)
893  })
894}

896/// `tokens` after the star and the optional `[…]` argument of the line break before them, and the spaces before each
897/// (`\\[0.2em]`, `\\*`).
898fn after_line_break_options(tokens: &[Token]) -> &[Token] {
899  let mut i = tokens
900    .iter()
901    .take_while(|t| t.get_catcode() == Catcode::SPACE)
902    .count();
903  if tokens.get(i) == Some(&T_OTHER!("*")) {
904    i += 1;
905  }
906  if tokens.get(i) == Some(&T_OTHER!("["))
907    && let Some(close) = tokens[i..].iter().position(|t| *t == T_OTHER!("]"))
908  {
909    i += close + 1;
910  }
911  &tokens[i..]
912}

914/// How many entries an author line lists up to its first line break: the pieces between its separators
915/// ([`next_name_piece`]) that print text (`\texttt{a@x}~~~~\texttt{b@y}` two).
916fn row_entry_count(tokens: &[Token]) -> usize {

918  let mut i = 0;
919  while i < tokens.len() && !is_row_break(&tokens[i]) {
920    let piece = next_name_piece(&tokens[i..]);
921    if piece.is_empty() {
922      i += 1;
923      continue;
924    }
925    if !visible_name_text(piece).trim().is_empty() {
926      count += 1;
927    }
928    i += piece.len();
929  }
930  count
931}

938/// How many names an author line lists up to its first line break ([`is_row_break`]): each piece between its
939/// separators ([`next_name_piece`]) with a name letter counts the names its text lists ([`name_count`], "Ann Able and
940/// Bob Baker" two), one at least.

1009/// Where a run of spacing commands that starts at `src[start]` ends, when it parts two names: two or more spacing
1010/// commands ([`is_spacing_command`], spaces and a blank line's `\par` between them), after a name — `out` holds what
1011/// precedes it, a name letter is on its line (the caller's test), and it does not end with a separator already (`\\`,
1012/// `\and`, `\quad`, a comma or `;`), nor with an initial (`R.~~Beyer`, 2603.17646; `L.~~Caballero-Ontanaya`,
1013/// 2604.25437; `Rafa\l ~R.\ ~Suszek`, 2603.20914: one person) — and before the next one, which starts with a letter, a
1014/// group or a command that is no name annotation (a run before `\thanks{…}`, `\textsuperscript{1}` or a `$^1$` mark
1015/// leaves the mark on its name) and has a name letter before the next separator (not `~~\JMorcid,`, 2605.12398;
1016/// `~~\mbox{\orcidlink{…}}`, 2605.06250); or, on a line that opens with a mark (`ctx.mark_led`), with the next mark.
1017/// Off the names line (`ctx.names_line`: after a line break, before the next `\and`) both sides must read as names, not
1018/// as an address, an e-mail or a link (`Ann Able \\ Department of Physics~~~~University of X`; `\url{…}~~~ \url{…}`,
1019/// 2605.28131). A run before the word "and" — in any case, alone or in one wrapper ([`and_word_end`]: `~~AND~~`,
1020/// `~~{and}~~`, `\;\;\textnormal{AND}\;\;`, 2606.27261, 2607.03677) — parts with it, the word the separator
1021/// ([`RunSeparator`]). `None` for a single spacing command, inside one name, and for a run after the word "and", or
1022/// after a byline's "By" (`By~~{\scshape Vitaly~Bergelson}~~and~~…`, 2603.25922, 2604.12864, 2603.27258, 2604.07186),
1023/// which separate the names themselves.
1024fn spacing_run_separator(
1025  src: &[Token],
1026  out: &[Token],
1027  start: usize,
1028  ctx: RunContext,
1029) -> Option<RunSeparator> {
1030  let mut end = start;
1031  let mut commands = 0;
1032  while let Some(t) = src.get(end) {
1033    if is_spacing_command(t) {
1034      commands += 1;
1035    } else if t.get_catcode() != Catcode::SPACE && *t != T_CS!("\\par") {
1036      break;
1037    }
1038    end += 1;
1039  }
1040  if commands < 2 {
1041    return None;
1042  }
1043  let unprinted = |t: &&Token| t.get_catcode() == Catcode::SPACE || **t == T_CS!("\\par");
1044  let mut before_run = out.iter().rev().skip_while(unprinted);
1045  let before = *before_run.next()?;
1046  if is_line_separator(&before)
1047    || before == T_CS!("\\&")
1048    || matches!(before.get_catcode(), Catcode::OTHER if before.with_str(|s| s == "," || s == ";"))
1049  {
1050    return None;
1051  }
1052  // (an initial: one letter and its period, after no letter)
1053  if before == T_OTHER!(".")
1054    && before_run
1055      .next()
1056      .is_some_and(|t| t.get_catcode() == Catcode::LETTER)
1057    && before_run
1058      .next()
1059      .is_none_or(|t| t.get_catcode() != Catcode::LETTER)
1060  {
1061    return None;
1062  }
1063  // (a run after the word "and" — in any case, alone or in one wrapper — leaves the separating to it: `Cat
1064  // Cole and~~Dan Dorn`; so does one after a byline's "by")
1065  let word_before: String = word_letters(out.iter().rev()).chars().rev().collect();
1066  if and_word_end(&out.iter().rev().copied().collect::<Vec<_>>(), true).is_some()
1067    || word_before.eq_ignore_ascii_case("by")
1068  {
1069    return None;
1070  }
1071  // (a run before the word "and" — `~~and~~`, `~~AND~~`, `~~{and}~~`, `\;\;\textnormal{AND}\;\;`, 2606.27261,
1072  // 2607.03677 — is one separator with it and the space after it: what follows must open a name as after a run)
1073  let (end, and_word) = match and_word_end(&src[end..], false) {
1074    Some(word_end) => (
1075      end
1076        + word_end
1077        + src[end + word_end..]
1078          .iter()
1079          .take_while(|t| {
1080            t.get_catcode() == Catcode::SPACE || is_spacing_command(t) || **t == T_CS!("\\par")
1081          })
1082          .count(),
1083      true,
1084    ),
1085    None => (end, false),
1086  };
1087  let after = src.get(end)?;
1088  if *after == T_CS!("\\&") {
1089    return None;
1090  }
1091  // (on a mark-led line, a run before the next mark parts its entries: `$^1$Univ A~~~~$^2$Univ B`; the word "and" in
1092  // it too, off the names line as the `\quad` that parts affiliation entries, `$^1$Univ A~~and~~$^2$Univ B` (probe
1093  // w10, was "Univ A and" an entry))
1094  if ctx.mark_led && starts_with_mark(after_leading_markup(&src[end..])) {
1095    return Some(RunSeparator {
1096      end,
1097      and_word: and_word && ctx.names_line,
1098    });
1099  }
1100  let starts_name = match after.get_catcode() {
1101    Catcode::LETTER | Catcode::BEGIN => true,
1102    Catcode::OTHER => after.with_str(|s| s.chars().next().is_some_and(char::is_alphabetic)),
1103    Catcode::CS => {
1104      !is_line_separator(after)
1105        && !is_name_annotation(after)
1106        && !is_footnote_mark(after)
1107        && !is_mark_command(after)
1108    },
1109    _ => false,
1110  };
1111  if !starts_name {
1112    return None;
1113  }
1114  let next_piece = next_name_piece(&src[end..]);
1115  if !has_name_letter(next_piece) {
1116    return None;
1117  }
1118  if !ctx.names_line
1119    && !(reads_as_name_piece(&out[ctx.piece_start.min(out.len())..])
1120      && reads_as_name_piece(next_piece))
1121  {
1122    return None;
1123  }
1124  Some(RunSeparator { end, and_word })
1125}

1136/// Where the word "and" that `tokens` open with ends (after their spaces), in any case (`and`, `And`, `AND`), written
1137/// alone or as the only word of one wrapper — a group, or a font command's argument (`{and}`, `\textrm{and}`,
1138/// `\textnormal{AND}`: 2606.27261, 2607.03677, 2608.24427). `reversed`: `tokens` run backwards (the tokens before a
1139/// run), the wrapper's closing brace first. `None` when they open with anything else (`Andrew`, `Anders`).
1140fn and_word_end(tokens: &[Token], reversed: bool) -> Option<usize> {
1141  let (open, close) = if reversed {
1142    (T_END!(), T_BEGIN!())
1143  } else {
1144    (T_BEGIN!(), T_END!())
1145  };
1146  let mut i = tokens
1147    .iter()
1148    .take_while(|t| t.get_catcode() == Catcode::SPACE)
1149    .count();
1150  // (a font command before its argument: read forwards, it opens the wrapper; read backwards, it stands after it)
1151  if !reversed && tokens.get(i).is_some_and(name_text_command) && tokens.get(i + 1) == Some(&open) {
1152    i += 1;
1153  }
1154  let wrapped = tokens.get(i) == Some(&open);
1155  if wrapped {
1156    i += 1;
1157  }
1158  let word_start = i
1159    + tokens[i..]
1160      .iter()
1161      .take_while(|t| wrapped && t.get_catcode() == Catcode::SPACE)
1162      .count();
1163  let word: String = word_letters(tokens[word_start..].iter());
1164  let word: String = if reversed {
1165    word.chars().rev().collect()
1166  } else {
1167    word
1168  };
1169  if !word.eq_ignore_ascii_case("and") {
1170    return None;
1171  }
1172  i = word_start + 3;
1173  if wrapped {
1174    i += tokens[i..]
1175      .iter()
1176      .take_while(|t| t.get_catcode() == Catcode::SPACE)
1177      .count();
1178    if tokens.get(i) != Some(&close) {
1179      return None;
1180    }
1181    i += 1;
1182  }
1183  // (a letter or a letter command after it is a longer word's: `and\o`; the closing word of a reversed read must not
1184  // be glued to one before it either, `Rand`)
1185  tokens
1186    .get(i)
1187    .is_none_or(|t| !parts_words_after(t))
1188    .then_some(i)
1189}

1411  }
1412  let Some(defn) = lookup_definition(t).ok()? else {
1413    return title_code_juggle(t);
1414  };
1415  if !defn.is_expandable()
1416    || defn
1417      .get_parameters()
1418      .is_some_and(|p| !p.get_parameters().is_empty())
1419  {
1420    return None;
1421  }
1422  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
1423    return None;
1424  };
1425  tabular_close_rest_of_body(body.unlist_ref())
1426}

1428/// The rest (spaces out) of a parameterless macro's expansion `body` past the `\end{tabular}` it starts with, if it does.

1434  }

1459/// `tokens` with a mark that leads a line of labelled entries written without the symbols glued to it (`$^{*1}$Kyoto
1460/// University`, `$^{1*}$Univ X`, 2606.23258): the mark labels the line, the symbol with it, and a line prints no label in
1461/// its text — it read "* Univ X", or matched no name once the names' `*1` asks for 1 and `*` apart (`glued_marks`).
1462/// From the first line if `first_line` (an affiliation list), else from after the first line break (an author block,
1463/// whose first line is names, its marks after them).
1464pub fn glued_line_marks_as_marks(tokens: Tokens, first_line: bool) -> Tokens {
1465  let v = tokens.unlist_ref();
1466  if !v
1467    .iter()
1468    .any(|t| *t == T_SUPER!() || *t == T_CS!("\\textsuperscript"))
1469  {
1470    return tokens;
1471  }
1472  let mut out: Vec<Token> = Vec::with_capacity(v.len());
1473  // (the group depth; whether a letter came since the last separator; whether the names line is over)
1474  let (mut depth, mut lettered, mut broke) = (0usize, !first_line, first_line);
1475  let mut i = 0;
1476  while i < v.len() {
1477    let t = v[i];
1478    if depth == 0 && !lettered {
1479      // (`\textsuperscript{…}`; `$^{…}$`, `${}^{…}$`)
1480      let at = match t.get_catcode() {
1481        Catcode::MATH => {
1482          let p = i
1483            + 1
1484            + usize::from(v.get(i + 1) == Some(&T_BEGIN!()) && v.get(i + 2) == Some(&T_END!())) * 2;
1485          (v.get(p) == Some(&T_SUPER!()) && v.get(p + 1) == Some(&T_BEGIN!())).then_some(p + 1)
1486        },
1487        _ if t == T_CS!("\\textsuperscript") && v.get(i + 1) == Some(&T_BEGIN!()) => Some(i + 1),
1488        _ => None,
1489      };
1490      if let Some(open) = at {
1491        let close = skip_group(v, open);
1492        let math = t.get_catcode() == Catcode::MATH;
1493        let end = if math { close + 1 } else { close };
1494        if close <= v.len()
1495          && (!math || v.get(close) == Some(&T_MATH!()))
1496          && let Some(glued) = glued_marks(&v[open + 1..close - 1])
1497        {
1498          out.extend_from_slice(&v[i..=open]);
1499          for marks in glued.into_iter().filter_map(|part| match part {
1500            GluedPart::Marks(marks) => Some(marks),
1501            GluedPart::Symbol(_) => None,
1502          }) {
1503            out.extend(marks);
1504          }
1505          out.push(T_END!());
1506          if math {
1507            out.push(T_MATH!());
1508          }
1509          i = end;
1510          continue;
1511        }
1512      }
1513    }
1514    match t.get_catcode() {
1515      Catcode::BEGIN => depth += 1,
1516      Catcode::END => depth = depth.saturating_sub(1),
1517      Catcode::LETTER => lettered = true,
1518      _ if depth == 0 && is_line_separator(&t) => {
1519        broke |= first_line || t == T_CS!("\\\\") || t == T_CS!("\\newline");
1520        lettered &= !broke;
1521      },
1522      _ => {},
1523    }
1524    out.push(t);
1525    i += 1;
1526  }
1527  Tokens::new(out)
1528}

1530/// Rewrite an ordinal suffix superscript after a numeral (`5\textsuperscript{th}`, `21$^{st}$`) onto a visible
1531/// superscript: it is the number's, not an affiliation mark, so a marked list is not split at it nor labelled by it
1532/// (repro sectioning-frontmatter/affiliation_line_superscript_ordinal_is_not_a_mark; iopart's
1533/// `\address{$^1$Univ A, 5 \textsuperscript{th} floor}`, 2609.01831).
1534pub(in crate::base_utilities) fn rewrite_ordinal_superscripts(tokens: Tokens) -> Tokens {
1535  let is_ordinal = is_ordinal_suffix;
1536  let is_digit = |t: &Token| {
1537    t.get_catcode() == Catcode::OTHER
1538      && t.with_str(|s| !s.is_empty() && s.chars().all(|c| c.is_ascii_digit()))
1539  };
1540  let after_numeral = |out: &[Token]| {
1541    out
1542      .iter()
1543      .rev()
1544      .find(|t| **t != T_SPACE!())
1545      .is_some_and(is_digit)
1546  };
1547  // The suffix is shown in text: a math-only wrapper takes its text form (`$^{\mathrm{th}}$`).
1548  let text_mode = |operand: Vec<Token>| -> Vec<Token> {
1549    operand
1550      .into_iter()
1551      .map(|t| {
1552        if t == T_CS!("\\mathrm") {
1553          T_CS!("\\textrm")
1554        } else if t == T_CS!("\\mathup") {
1555          T_CS!("\\textup")
1556        } else if t == T_CS!("\\mathit") {
1557          T_CS!("\\textit")
1558        } else if t == T_CS!("\\mathbf") {
1559          T_CS!("\\textbf")
1560        } else if t == T_CS!("\\mathsf") {
1561          T_CS!("\\textsf")
1562        } else {
1563          t
1564        }
1565      })
1566      .collect()
1567  };
1568  let src = tokens.unlist();
1569  let mut out: Vec<Token> = Vec::with_capacity(src.len());
1570  let mut i = 0;
1571  while i < src.len() {
1572    let t = src[i];
1573    if t == T_CS!("\\textsuperscript")
1574      && let Some((operand, next)) = sup_operand_at(&src, i + 1)
1575      && is_ordinal(&operand)
1576      && after_numeral(&out)
1577    {
1578      out.extend(visible_sup(text_mode(operand)));
1579      i = next;
1580      continue;
1581    }
1582    if t == T_MATH!() {
1583      // `$^{th}$` after a numeral, or `$5^{th}$` with the numeral inside: the digits stay, the suffix is shown.
1584      let digits_end = (i + 1..src.len())
1585        .find(|&k| !is_digit(&src[k]))
1586        .unwrap_or(src.len());
1587      if src.get(digits_end) == Some(&T_SUPER!())
1588        && let Some((operand, next)) = sup_operand_at(&src, digits_end + 1)
1589        && src.get(next) == Some(&T_MATH!())
1590        && is_ordinal(&operand)
1591        && (digits_end > i + 1 || after_numeral(&out))
1592      {
1593        out.extend_from_slice(&src[i + 1..digits_end]);
1594        out.extend(visible_sup(text_mode(operand)));
1595        i = next + 1;
1596        continue;
1597      }
1598    }
1599    out.push(t);
1600    i += 1;
1601  }
1602  Tokens::new(out)
1603}

1605/// Rewrite a superscript inside a word, a letter on each side of it (`Craig M$^{\mathrm{c}}$Rae`, 2505.09041;
1606/// `M\textsuperscript{c}Donald`; `I$^2$R`, `NLP$^2$C`), onto a visible superscript: it is the name's spelling, not an
1607/// affiliation mark, which follows a name — read as one, its letter was a request and left the name ("Craig MRae", the
1608/// "c" shown after it). Not a mark list (`$^{1,3}$`: a list of marks is never spelling) nor a mark glued to the word
1609/// "and" after it (`Hiroshi Ito$^{1,3}$and Tetsuya Ogata`, 2605.00471; `Bipasha Kundu$^{1}$and Cristian Linte`,
1610/// 2604.27101): those are the name's marks, which kept as its text lost their affiliations (77n r3 review).
1611pub(in crate::base_utilities) fn rewrite_in_word_superscripts(tokens: Tokens) -> Tokens {
1612  let src = tokens.unlist();
1613  let letter = |t: Option<&Token>| t.is_some_and(|t| t.get_catcode() == Catcode::LETTER);
1614  // (the operand a mark's: a list of marks, or the word after it "and", the separator before the next name; `Andrew`
1615  // is a word, 77n r4 review probe a8. Only "and" itself: `Ann Able$^{1}$And Bob Baker$^{2}$`, `AND`, stays one name,
1616  // which `flag_merged_creators` reports — read as a mark, its "And" merged the names silently, 77n r5 review probes
1617  // a1, a6)
1618  let marks = |operand: &[Token], after: usize| {
1619    operand.contains(&T_OTHER!(","))
1620      || (src.get(after..after + 3) == Some(&[T_LETTER!("a"), T_LETTER!("n"), T_LETTER!("d")][..])
1621        && !letter(src.get(after + 3)))
1622  };
1623  // (the letters before it a word, not a length's unit: `\kern2pt$^{1}$Univ A` is a spacing, then a mark; 2407.15815)
1624  let after_a_word = |out: &[Token]| {
1625    let word = out
1626      .iter()
1627      .rev()
1628      .take_while(|t| t.get_catcode() == Catcode::LETTER)
1629      .count();
1630    word > 0
1631      && out[..out.len() - word].last().is_none_or(|t| {
1632        !(t.get_catcode() == Catcode::OTHER
1633          && t.with_str(|s| {
1634            s.chars()
1635              .all(|c| c.is_ascii_digit() || c == '.' || c == ',')
1636          }))
1637      })
1638  };
1639  let mut out: Vec<Token> = Vec::with_capacity(src.len());
1640  let mut i = 0;
1641  while i < src.len() {
1642    let t = src[i];
1643    if after_a_word(&out) {
1644      if t == T_CS!("\\textsuperscript")
1645        && let Some((operand, next)) = sup_operand_at(&src, i + 1)
1646        && letter(src.get(next))
1647        && !marks(&operand, next)
1648      {
1649        out.extend(visible_sup(operand));
1650        i = next;
1651        continue;
1652      }
1653      if t == T_MATH!()
1654        && src.get(i + 1) == Some(&T_SUPER!())
1655        && let Some((operand, next)) = sup_operand_at(&src, i + 2)
1656        && src.get(next) == Some(&T_MATH!())
1657        && letter(src.get(next + 1))
1658        && !marks(&operand, next + 1)
1659      {
1660        let mut math = vec![T_MATH!()];
1661        math.extend(operand);
1662        math.push(T_MATH!());
1663        out.extend(visible_sup(math));
1664        i = next + 1;
1665        continue;
1666      }
1667    }
1668    out.push(t);
1669    i += 1;
1670  }
1671  Tokens::new(out)
