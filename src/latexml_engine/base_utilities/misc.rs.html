5/// Use tex_glue::dimension_to_spaces for the precise Perl-matching algorithm.
6/// This wrapper delegates to avoid breaking callers that import from here.
7///
8/// `Dimension::new` takes raw scaled-points (sp). `value_of()` returns the
9/// canonical i64 (sp for Dimension/Glue, units for Number). The previous
10/// implementation routed through `pt_value(None)` (which divides by UNITY)
11/// and then `new_f64` (which does NOT multiply back by UNITY), losing a
12/// factor of 65536 — so 2em (1310720 sp) became Dimension(20 sp) ≈ 0pt and
13/// `tex_glue::dimension_to_spaces` produced an empty string. That made
14/// `\hspace`'s `isSpace` Tbox skip the `if !s.is_empty()` gate at the
15/// caller, dropping the math-mode space marker between `^{...}` and a
16/// following `'` — surfacing as the false `unexpected:double-superscript`
17/// in hep-th9601176 (`\Si^{\mu\nu}\hs{0.25}'(p)`).
18///
19/// Both callers (`\hglue`, `\hspace`) are `DefPrimitive`s, i.e. DIGEST time,
20/// where `lookup_font()` already IS the font in effect — so the ambient read
21/// is the right one and no font is threaded through.
22pub fn dimension_to_spaces<T: NumericOps>(dimen: T) -> Cow<'static, str> {
23  let dim = Dimension::new(dimen.value_of());
24  Cow::Owned(crate::tex_glue::dimension_to_spaces(dim, None))
25}

27pub fn aligning_environment(
28  align: &str,
29  class: &str,
30  document: &mut Document,
31  props: &SymHashMap<Stored>,
32) -> Result<()> {
33  if let Some(Stored::Digested(body)) = props.get("body") {
34    // In a float, the environment's lines are rows of their own (the panel grid rule, `note_panel_separator`).
35    crate::latex_constructs::note_panel_separator(document, "env", None)?;
36    // Add class attribute to new nodes.
37    // (a list environment, no TeX box: its text is body text for a field's print, OXIDIZED_DESIGN #715)
38    let nodes = crate::latex_constructs::in_aligning_capture(|| {
39      insert_block_as(document, body, HashMap::default(), false, false)
40    })?;
41    for mut node in nodes.into_iter() {
42      set_align_or_class(document, &mut node, align, class)?;
43    }
44    crate::latex_constructs::note_panel_separator(document, "env", None)?;
45  }
46  Ok(())
47}

67/// Remove `\@spaces`/`\space` padding control sequences from a message
68/// prefix, mirroring Perl `make_message`'s `s/(?:\\\@?spaces?)+//g`
69/// (latex_constructs.pool.ltxml L5572). Longest-first so `\@spaces` is
70/// never left as a stray `s`.
71fn strip_message_spaces(s: &str) -> String {
72  s.replace("\\@spaces", "")
73    .replace("\\spaces", "")
74    .replace("\\@space", "")
75    .replace("\\space", "")
76}

78pub fn make_generic_message(cmd: &str, args: Vec<Tokens>, kind: &str) -> Result<()> {
79  // Faithful port of Perl latex_constructs.pool.ltxml `make_message`
80  // (L5569-5588). The first arg is the message PREFIX: its ToString,
81  // stripped of `\@spaces`/`\space` runs, becomes the diagnostic `what`
82  // ($type), falling back to the command name when the prefix is empty
83  // (e.g. `\@latex@error`'s all-`\space` prefix). The REMAINING args form
84  // the body — Perl `join(" ", map { ToString(Expand($_, \@spaces,
85  // \@spaces)) } @args)`. Earlier the Rust port discarded the prefix and
86  // pinned `what` to the command name, so package/class diagnostics lost
87  // their `(pkgname)` $type and `\GenericWarning`/`\GenericInfo` even
88  // leaked the prefix into the body. Witness: hep-th-style `\PackageError`
89  // /`\GenericWarning` now match Perl byte-for-byte (`(mypkg)` $type, no
90  // prefix in body).
91  let mut args = args.into_iter();
92  let lead = args.next().unwrap_or_default();
93  // $type = ToString(lead) with `\@?spaces?` runs removed (Perl regex
94  // `s/(?:\\\@?spaces?)+//g`); fall back to the command name when empty.
95  let type_str = strip_message_spaces(&lead.to_string());
96  let type_str: &str = if type_str.is_empty() { cmd } else { &type_str };
97
98  bgroup();
99  let_i(&T_CS!("\\protect"), &T_CS!("\\string"), None);
100  let_i(
101    &T_CS!("\\MessageBreak"),
102    &T_CS!("\\ltx@hard@MessageBreak"),
103    None,
104  ); // tricky, we need Expand() to execute it
105  let mut parts: Vec<String> = Vec::new();
106  for arg in args {
107    let mut arg_toks = arg.unlist();
108    // Perl `make_message` pads each Expand input with two `\@spaces` tokens:
109    //   ToString(Expand($_, T_CS('\@spaces'), T_CS('\@spaces')))
110    // Comment in Perl: "expand padded with two \@spaces, to avoid pointless
111    // errors. e.g. a trailing `\csq@noline` from csquotes.sty (let to
112    // `\@gobble`) will produce a `gobble has no argument` error on every
113    // message." Same fix needed here for etoc.sty's
114    // `\PackageWarning{etoc}{...!\@gobbletwo}` — `\@gobbletwo` would otherwise
115    // consume the do_expand-inserted T_END closer, leaving readBalanced
116    // unbalanced.
117    arg_toks.push(T_CS!("\\@spaces"));
118    arg_toks.push(T_CS!("\\@spaces"));
119    // Written as `\write`/`\message` write it (tex.web §262 print_cs): a control word is
120    // followed by a space, so `\PackageWarning{p}{B \noexpand\foo c}` reads "B \foo c". Perl's
121    // `ToString` (latex_constructs.pool.ltxml:5586-5587) runs it together ("B \fooc"; KPE #357).
122    // Byte-mouth runs are decoded here (as `Tokens`' Display does), not in `writable_tokens`,
123    // which also feeds `\scantokens`/`\detokenize` that need the byte spellings.
124    // Expanded as `\immediate\write` expands (latex.ltx:8773-8799, `\GenericInfo`/`\GenericWarning`/
125    // `\GenericError` print through it), i.e. as `\edef` and `\message`: what `\the`/`\unexpanded`
126    // yield is not expanded again, so `\PackageWarning{p}{\unexpanded{\foo}}` prints "\foo" and
127    // runs nothing. Perl `Expand`s fully (latex_constructs.pool.ltxml:5586-5587), executing an
128    // undefined `\foo` there. Guard `perfect_kernel_gemini::package_warning_keeps_unexpanded_text`.
129    let expanded = do_expand_partially(arg_toks)?;
130    parts.push(mouth::decode_byte_mouth_runs(&writable_tokens(&expanded)).into_owned());
131  }
132  // Perl joins the body args with a single space (the `\MessageBreak`s
133  // *within* an arg already became hard newlines via the let above).
134  let message = parts.join(" ");
135
136  egroup()?;
137  // Downgrade vendor-class typesetting-only errors to Info. Publisher
138  // classes routinely guard line widths, header heights, and other
139  // PDF-layout concerns with `\PackageError`/`\GenericError`. We
140  // produce XML/HTML, not PDF — these guards have no semantic value
141  // in our output. See WISDOM #50 and
142  // memory/feedback_size_layout_errors_moot.md.
143  let effective_kind = if kind == "error" && is_typesetting_only_message(&message) {
144    "info"
145  } else {
146    kind
147  };
148  //   return ('latex', $type, $stomach, $message);
149  match effective_kind {
150    "error" => {
151      Error!("latex", type_str, message);
152    },
153    "warn" => {
154      Warn!("latex", type_str, message);
155    },
156    "info" => {
157      Info!("latex", type_str, message);
158    },
159    _other => panic!("Only call make_generic_message with error|warn|info message kinds."),
160  };
161  Ok(())
162}

164/// Heuristic classifier for vendor `\PackageError`/`\GenericError`
165/// messages whose only concern is PDF typesetting (size, layout,
166/// position, page-fit). These have no signal in XML/HTML output and
167/// are downgraded to `Info:` per WISDOM #50.
168fn is_typesetting_only_message(message: &str) -> bool {
169  let lower = message.to_ascii_lowercase();
170  // Phrase set tuned against the stage-1 sweep of the 100k warning
171  // corpus. Conservative — every phrase here is purely about visual
172  // layout or vendor-deprecation chatter, never about semantic
173  // correctness. Examples:
174  //   "Running heading author exceeds size limitations" (AISTATS)
175  //   "Running heading title exceeds size limitations" (AISTATS)
176  //   "Caption too wide for page" (various)
177  //   "Heading breaks the line" (revtex, IEEEtran)
178  //   "You are loading directly a language style" (babel: czech.sty,
179  //     francais.sty, etc. unconditionally fire `\PackageError` to nag
180  //     the user toward `\usepackage[<lang>]{babel}`; pdflatex shows
181  //     the message but continues, and the document typesets normally
182  //     — the message is informational, not a real failure)
183  const PHRASES: &[&str] = &[
184    "exceeds size limitations",
185    "exceeds size limitation",
186    "running heading",
187    "running title",
188    "running author",
189    "breaks the line",
190    "too wide for",
191    "too tall for",
192    "too long for",
193    "too narrow for",
194    "doesn't fit",
195    "does not fit",
196    "page overflow",
197    "column overflow",
198    "exceeds the page",
199    "exceeds the column",
200    "exceeds the line",
201    "exceeds the textwidth",
202    "exceeds \\textwidth",
203    "exceeds \\columnwidth",
204    "exceeds \\linewidth",
205    "loading directly a language style",
206    "syntax is deprecated",
207    // babel TL2025: legacy `<lang>.ldf` files have been retired in
208    // favour of `locale/<iso>/babel-<lang>.tex` (the ini-file
209    // system). For papers that load `\usepackage[<lang>]{babel}`
210    // without `provide=*`, babel fires:
211    //   Package babel Error: Unknown option '<lang>'.
212    //   Either you misspelled it or the language definition file
213    //   <lang>.ldf was not found
214    // The .ldf-missing case is benign: pdflatex shows the message
215    // and proceeds — the document still typesets, just without
216    // the language's captions/shorthands loaded. Same effective
217    // outcome on our side: downgrade to Info, conversion continues.
218    // Surpass-Perl: Perl raw-loads babel.sty and errors identically
219    // on the same 58 papers in Round-27 Cluster D. This downgrade
220    // closes the cluster (cannot reach 0-error AND load the proper
221    // ini file without redesigning babel option processing).
222    "either you misspelled it",
223    // catoptions.sty (loaded transitively by many class/sty bundles)
224    // calls `\@latex@error{Command \protect\\special_relax already
225    // defined...}` because it `\def\special_relax{...}` and our engine
226    // pre-registers `\special_relax` as an internal Gullet helper.
227    // pdflatex shows the message and proceeds (cat's def takes over);
228    // surpass-Perl: Perl also raw-loads catoptions and errors. The
229    // resulting "Command ... already defined" cascade currently
230    // produces 100+ errors per paper on 14 wp5 papers. Downgrade so
231    // conversion continues. Same applies to "Command \end... illegal,
232    // see p.192 of the manual" tail that LaTeX appends to the same
233    // message ("\@latex@error" generic-error template).
234    "already defined. or name",
235    "command \\end... illegal",
236    // amsfonts-not-installed: aims-class L: `\@latex@error{Package
237    // `amsfonts' not installed, or version too old?}`. The class ships
238    // its own font tables and only loads amsfonts as an enhancement;
239    // pdflatex/Perl LaTeXML both proceed to typeset successfully when
240    // amsfonts is missing. Our raw-load reports the message verbatim
241    // but conversion is fine without the AMS msam/msbm fonts. Witness
242    // 2202.13120.
243    "amsfonts package will not be loaded",
244    "package `amsfonts'",
245    "package 'amsfonts'",
246    // hrefhide-format-too-old: hrefhide.sty requires LaTeX format
247    // 2022-11-01+; older formats trigger an error. The package is
248    // strictly visual (hides hyperref colors on print), so the error
249    // is moot for XML/HTML output. Witness 2202.03936.
250    "newer latex format needed",
251    "older hrefhide package",
252  ];
253  PHRASES.iter().any(|p| lower.contains(p))
254}

318/// Is this definition a NAME NOUN (`\figurename`): an expandable macro whose arguments are
319/// all optional? ltcmd's `\NewDocumentCommand` defines `\foo` as the protected,
320/// parameterless dispatcher `\__cmd_start_optimized: \foo code` (latex.ltx:1963-1972),
321/// so follow that one hop to `\foo code` (zero-arg xparse nouns qualify, `m`-taking
322/// drawing commands like argumentation.sty:403 `\afname` do not); the general
323/// `\__cmd_start:nNNnnn` / `\__cmd_start_expandable:nNNNNn` dispatchers carry the
324/// signature as written, which is read; a `\DeclareRobustCommand` wrapper is followed to
325/// its inner macro, and a `\protect\<cs>` body to `\<cs>` (`\protect` is `\relax` in a typeset tag,
326/// latex.ltx `\@typeset@protect`): `\renewcommand\figurename{\protect\rb}` with a robust `\rb` is two hops,
327/// the body to the wrapper and the wrapper to its inner macro, and was cut at the first, so the tag lost its name
328/// (pdflatex "Fx 1", ours "1"; `[figurename=\rb]{caption}` stores the same body). `visited` holds the macros
329/// followed so far: a hop back to one of them is a loop (`\def\x{\protect\x}`, on which pdflatex never ends) and
330/// is not a name; a chain of any length otherwise is followed (pdflatex "Ex 1" through five `\protect` hops), with a
331/// fuse of `MAX_HOPS` distinct macros. Guard `perfect_kernel_batch68h::protected_robust_name_is_a_name_noun`.
332pub(super) fn is_name_noun(def: &Rc<dyn Definition>, visited: &mut Vec<Token>) -> bool {
333  const MAX_HOPS: usize = 32;
334  // A name is set before `~<number>\null` (hyperref's autoref) or in a tag's composition: optional arguments find
335  // none and take their defaults, a required one would swallow what follows.
336  if !def.is_expandable()
337    || def
338      .get_parameters()
339      .is_some_and(|p| p.get_parameters().iter().any(|param| !param.optional))
340  {
341    return false;
342  }
343  let Some(ExpansionBody::Tokens(body)) = def.get_expansion() else {
344    return true;
345  };
346  let toks = body.clone().unlist();
347  let Some(first) = toks.first() else {
348    return true;
349  };
350  if first.get_catcode() != Catcode::CS {
351    return true;
352  }
353  let head = first.to_string();
354  // `\DeclareRobustCommand`'s wrapper `\protect \<cs>␣`: the name is the inner macro.
355  let mut follow = |next: &Token| {
356    if visited.len() >= MAX_HOPS || visited.contains(next) {
357      return false;
358    }
359    visited.push(*next);
360    matches!(lookup_definition(next), Ok(Some(next_def)) if is_name_noun(&next_def, visited))
361  };
362  if head == "\\protect" && toks.len() == 2 {
363    return follow(&toks[1]);
364  }
365  if head == "\\__cmd_start_optimized:" {
366    return toks.get(1).is_some_and(follow);
367  }
368  if head == "\\__cmd_start:nNNnnn" || head == "\\__cmd_start_expandable:nNNNNn" {
369    // ltcmd's general dispatcher carries the signature first: a name takes only optional arguments (`o`, `O{…}`,
370    // `d`, `s`, `t`, `e`); `m`, `r`, `R`, `v`, `b`, `l`, `u` are required.
371    return xparse_signature_is_optional(&toks[1..]);
372  }
373  true
374}

376/// Whether an ltcmd signature — the braced group at the start of `toks`, as written — has only optional argument
377/// types: `m r R v b l u` are required; `o s g` optional, `t` with its token, `d` with two, `D` two and a default,
378/// `O G` a default, `e` its tokens, `E` its tokens and defaults; `+ !` modify, `>` and `=` take a group. A type's
379/// delimiters and defaults are skipped, so `t m` (an optional `m` token) or `O{with m}` stay optional; a braced type
380/// is taken as required.
381fn xparse_signature_is_optional(toks: &[Token]) -> bool {
382  if toks
383    .first()
384    .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
385  {
386    return false;
387  }
388  // The signature's top-level items, a braced group counted as one.
389  let mut items: Vec<Option<String>> = Vec::new();
390  let mut depth = 0;
391  for token in &toks[1..] {
392    match token.get_catcode() {
393      Catcode::BEGIN => {
394        if depth == 0 {
395          items.push(None);
396        }
397        depth += 1;
398      },
399      Catcode::END if depth == 0 => break,
400      Catcode::END => depth -= 1,
401      Catcode::SPACE if depth == 0 => {},
402      _ if depth == 0 => items.push(Some(token.to_string())),
403      _ => {},
404    }
405  }
406  let mut items = items.into_iter();
407  while let Some(item) = items.next() {
408    let skip = match item.as_deref() {
409      Some("m" | "r" | "R" | "v" | "b" | "l" | "u") => return false,
410      Some("o" | "s" | "g" | "+" | "!") => 0,
411      // A braced type (`{m}`): ltcmd reads its contents as the type (latex.ltx:2291); taken as required.
412      None => return false,
413      Some("t" | "O" | "G" | "e" | ">" | "=") => 1,
414      Some("d" | "E") => 2,
415      Some("D") => 3,
416      Some(_) => return false,
417    };
418    for _ in 0..skip {
419      items.next();
420    }
421  }
422  true
423}

425/// An ORCID iD written as its orcid.org address — `https://orcid.org/<iD>`, `http://`, `www.`, or the bare
426/// `orcid.org/<iD>`, a trailing `/` — is the iD; anything else, and an argument holding a macro, is kept as written.
427pub(super) fn orcid_id(id: Tokens) -> Tokens {
428  if id
429    .unlist_ref()
430    .iter()
431    .any(|t| t.get_catcode() == Catcode::CS)
432  {
433    return id;
434  }
435  let text = id.to_string();
436  let trimmed = text.trim();
437  let lower = trimmed.to_ascii_lowercase();
438  let mut rest = lower.as_str();
439  for scheme in ["https://", "http://"] {
440    if let Some(r) = rest.strip_prefix(scheme) {
441      rest = r;
442      break;
443    }
444  }
445  rest = rest.strip_prefix("www.").unwrap_or(rest);
446  match rest.strip_prefix("orcid.org/") {
447    Some(r) => {
448      let start = trimmed.len() - r.len();
449      let bare = trimmed[start..].trim_end_matches('/');
450      Tokens::new(
451        bare
452          .chars()
453          .map(|c| {
454            if c == ' ' {
455              T_SPACE!()
456            } else {
457              T_OTHER!(c.to_string())
458            }
459          })
460          .collect(),
461      )
462    },
463    None => id,
464  }
