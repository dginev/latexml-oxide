5/// Perl `isVAttached` (TeX_Box.pool.ltxml:433-440): the node, or its only child all the way
6/// down, has `vattach`. Every child node counts, text included: a `<p>` holding a v-attached
7/// box and the text after it is not v-attached (`\parbox{5cm}{\parbox{2cm}{Inner} after
8/// inner.}`, which since 57d builds that `<p>`; counting elements alone dropped the outer box's
9/// `vattach`). Guard `perfect_kernel_batch56::a_nested_parbox_keeps_its_vattach`. The function
10/// matches Perl's; the DOM it reads may not: a paragraph's trimmed trailing whitespace node is
11/// freed here and kept empty by Perl (KNOWN_PERL_ERRORS #311).
12pub(super) fn is_v_attached(node: &Node) -> bool {
13  let mut current = node.clone();
14  loop {
15    if current.get_attribute("vattach").is_some() {
16      return true;
17    }
18    match current.get_child_nodes().as_slice() {
19      [only] if only.get_type() == Some(NodeType::ElementNode) => current = only.clone(),
20      _ => return false,
21    }
22  }
23}

25/// Make `node`'s content Block.class-valid so `node` can safely become a `<ltx:block>`,
26/// demoting the Para.class descendants in place. Every direct `<ltx:logical-block>`/
27/// `<ltx:sectional-block>` child is renamed to `<ltx:block>` — KEEPING its attributes (a
28/// minipage's `width`, alignment, …) — and every direct `<ltx:para>` child is unwrapped
29/// (its Block.model children float up, already valid in a block; a `<para>` never
30/// directly holds Para.class, so no pre-demotion is needed). Other children (a `<p>`,
31/// `<tabular>`, `<itemize>`, a nested `<block>`, …) are valid there and left untouched,
32/// so paragraphs inside a still-valid Para.model container (an `<item>`) are never
33/// over-flattened.
34///
35/// BOTTOM-UP: a logical-block child is demoted (its own Para.class content fixed) BEFORE
36/// it is renamed to `<block>`, and the caller renames `node` itself only AFTER calling
37/// this — so no Para.class node is ever moved into a `<block>` while still Para.class,
38/// which would fire the constructor's `malformed`-content check (a spurious `Error:`).
39///
40/// Memory-safe snapshot iteration: `rename_node` recreates the child (its handle dies —
41/// we recurse into the child BEFORE renaming, via the live handle) and `unwrap_nodes`
42/// splices the child's kids up; neither moves the SIBLING handles held in the
43/// `get_child_nodes` snapshot, and floated/renamed nodes are never revisited through it.
44/// True when `node`'s content can be losslessly demoted into a `<ltx:block>`: every
45/// element that would end up as a direct child of the block is valid there. Descends
46/// only through the transformable Para.class wrappers (`ltx:para`/`ltx:logical-block`/
47/// `ltx:sectional-block` — para is unwrapped, the others renamed to `<block>`), whose
48/// own content must likewise be demotable; any OTHER element is a leaf kept as-is, so it
49/// must itself be block-valid. A container holding a Para.model-only element that
50/// `<block>` cannot hold — a `<float>`, `<TOC>`, a sectioning unit, a `<figure>`/`<table>`
51/// — is NOT demotable (renaming to block would strand it, e.g. stanli/webquiz), so the
52/// caller leaves it at Perl-parity rather than emitting a fresh `malformed` error.
53pub(super) fn subtree_is_block_demotable(node: &Node) -> bool {
54  node.get_child_nodes().iter().all(|c| {
55    if c.get_type() != Some(NodeType::ElementNode) {
56      return true;
57    }
58    let transformable = with(document::get_node_qname(c), |q| {
59      matches!(q, "ltx:para" | "ltx:logical-block" | "ltx:sectional-block")
60    });
61    if transformable {
62      subtree_is_block_demotable(c)
63    } else {
64      document::can_contain_qsym(pin_static("ltx:block"), document::get_node_qname(c))
65    }
66  })
67}

69pub(super) fn demote_para_class_content(document: &mut Document, node: &Node) -> Result<()> {
70  for child in node.get_child_nodes() {
71    if child.get_type() != Some(NodeType::ElementNode) {
72      continue;
73    }
74    let kind = with(document::get_node_qname(&child), |q| match q {
75      "ltx:logical-block" | "ltx:sectional-block" => 1u8,
76      "ltx:para" => 2,
77      _ => 0,
78    });
79    match kind {
80      1 => {
81        demote_para_class_content(document, &child)?; // fix content while child still allows Para.class
82        document.rename_node(child, "ltx:block", true)?; // THEN rename (content now Block.class-valid)
83      },
84      2 => document.unwrap_nodes(child)?,
85      _ => {},
86    }
87  }
88  Ok(())
89}

91/// A sectioning unit, read from the schema: admitted by `ltx:sectional-block` (model
92/// `document.body.class*`) but not by `ltx:logical-block` (`Para.model`) — part,
93/// chapter, section, subsection, subsubsection, paragraph, slide, slidesequence,
94/// sidebar, sectional-block; never a figure/table/theorem/para (those are
95/// Para.model). `ltx:subparagraph` is not in `document.body` (LaTeXML.model), so a
96/// lone top-level `\subparagraph` in a box keeps Perl's shape; one under a floated
97/// `\paragraph` rides along inside it.
98pub(super) fn is_sectioning_unit(node: &Node) -> bool {
99  let q = document::get_node_qname(node);
100  document::can_contain_qsym(pin_static("ltx:sectional-block"), q)
101    && !document::can_contain_qsym(pin_static("ltx:logical-block"), q)
102}

104/// Can `unit` be floated out of `context`? Walk up while each node either admits
105/// the unit (then a live `\\section` would open there) or auto-closes (a section,
106/// a paragraph, the box's own wrappers). A `<quote>`, a list, a `<figure>` does
107/// neither, so a box inside one keeps Perl's shape (#250).
108pub(super) fn sectioning_floatable(context: &Node, unit: SymStr) -> bool {
109  let mut n = context.clone();
110  loop {
111    if document::can_contain_qsym(document::get_node_qname(&n), unit) {
112      return true;
113    }
114    if !document::can_auto_close(&n) {
115      return false;
116    }
117    match n.get_parent() {
118      Some(p) if p.get_type() == Some(NodeType::ElementNode) => n = p,
119      _ => return false,
120    }
121  }
122}

128/// Would text arriving at `tag` open an `ltx:p`? Follows the model's auto-open route for
129/// `#PCDATA` (a section opens `ltx:para`, a para `ltx:p`); a route that ends in another text
130/// holder (`svg:foreignObject` in a drawing, `ltx:bibblock` in a bibitem) opens none.
131pub(super) fn text_opens_paragraph(tag: SymStr) -> bool {
132  let (text, p) = (pin_static("#PCDATA"), pin_static("ltx:p"));
133  let mut step = tag;
134  // The route is a handful of steps in LaTeXML.model; the bound guards a model cycle.
135  for _ in 0..8 {
136    match document::sym_can_contain_somehow(step, text) {
137      Some(Some(open)) if open == p => return true,
138      Some(Some(open)) => step = open,
139      _ => return false,
140    }
141  }
142  false
143}

145/// Figure material: what a figure shows beside its caption — graphics, pictures, tabulars — and the
146/// wrappers a box body puts around them (`para`/`p`, a `\scalebox` inline block, a font `text`),
147/// with no prose: text nodes are whitespace only.
148fn is_figure_material(node: &Node) -> bool {
149  match node.get_type() {
150    Some(NodeType::TextNode) => node.get_content().trim().is_empty(),
151    Some(NodeType::ElementNode) => with(document::get_node_qname(node), |q| match q {
152      "ltx:graphics" | "ltx:picture" | "svg:svg" | "ltx:tabular" | "ltx:rule" | "ltx:break" => true,
153      "ltx:para" | "ltx:p" | "ltx:block" | "ltx:inline-block" | "ltx:text" => {
154        node.get_child_nodes().iter().all(is_figure_material)
155      },
156      _ => false,
157    }),
158    _ => true,
159  }
160}

162/// A `para`/`p` left holding nothing but whitespace and such shells once its material moved.
163fn is_empty_shell(node: &Node) -> bool {
164  with(document::get_node_qname(node), |q| {
165    q == "ltx:para" || q == "ltx:p"
166  }) && node
167    .get_child_nodes()
168    .iter()
169    .all(|child| match child.get_type() {
170      Some(NodeType::TextNode) => child.get_content().trim().is_empty(),
171      Some(NodeType::ElementNode) => is_empty_shell(child),
172      _ => true,
173    })
174}

176/// The figure material inside `node`, its `para`/`p` wrappers peeled off: the leaves a figure holds.
177fn figure_material_leaves(node: &Node, leaves: &mut Vec<Node>) {
178  match node.get_type() {
179    Some(NodeType::ElementNode)
180      if with(document::get_node_qname(node), |q| {
181        q == "ltx:para" || q == "ltx:p"
182      }) =>
183    {
184      for child in node.get_child_nodes() {
185        figure_material_leaves(&child, leaves);
186      }
187    },
188    Some(NodeType::ElementNode) => leaves.push(node.clone()),
189    _ => {},
190  }
191}

193/// A box that is clearly a figure (user ruling 2026-09-27, 57g): its content is one float holding
194/// only its caption — the shape `\captionof` makes, wrapping only the caption in the float
195/// (`\@captionof@`, caption_sty.rs; Perl the same, KNOWN_PERL_ERRORS #312) and leaving the image
196/// beside it — and figure material. The float takes the material in, in source order (what
197/// precedes the caption goes before it), so the box becomes the float with its image:
198/// `<figure class="ltx_minipage"><graphics/><caption/></figure>` (arXiv 2605.27134's parbox
199/// panels, 2605.06890's `\captionof{table}` minipages). A float that already holds content of its
200/// own (`\begin{figure}[H]…`) takes nothing: its order with the material is the author's.
201/// Returns the float.
202pub(super) fn absorb_figure_material(
203  document: &mut Document,
204  nodes: &[Node],
205) -> Result<Option<Node>> {
206  let is_float = |n: &Node| {
207    n.get_type() == Some(NodeType::ElementNode)
208      && with(document::get_node_qname(n), |q| PANEL_FLOATS.contains(&q))
209  };
210  let floats: Vec<usize> = (0..nodes.len()).filter(|&i| is_float(&nodes[i])).collect();
211  let [at] = floats.as_slice() else {
212    return Ok(None);
213  };
214  let float = nodes[*at].clone();
215  let float_children = float.get_child_elements();
216  let caption_only = float_children.iter().all(|c| {
217    with(document::get_node_qname(c), |q| {
218      q == "ltx:tags" || q == "ltx:toccaption" || q == "ltx:caption"
219    })
220  });
221  let caption = float_children.into_iter().find(|c| {
222    with(document::get_node_qname(c), |q| {
223      q == "ltx:toccaption" || q == "ltx:caption"
224    })
225  });
226  let (true, Some(mut caption)) = (caption_only, caption) else {
227    return Ok(None);
228  };
229  let others: Vec<&Node> = nodes
230    .iter()
231    .enumerate()
232    .filter(|(i, _)| *i != *at)
233    .map(|(_, n)| n)
234    .collect();
235  if !others
236    .iter()
237    .any(|n| n.get_type() == Some(NodeType::ElementNode))
238    || !others.iter().all(|n| is_figure_material(n))
239  {
240    return Ok(None);
241  }
242  let mut float = float;
243  for (i, node) in nodes.iter().enumerate() {
244    if i == *at || node.get_type() != Some(NodeType::ElementNode) {
245      continue;
246    }
247    let mut leaves = Vec::new();
248    figure_material_leaves(node, &mut leaves);
249    for mut leaf in leaves {
250      leaf.unlink();
251      if i < *at {
252        caption.add_prev_sibling(&mut leaf)?;
253      } else {
254        float.add_child(&mut leaf)?;
255      }
256    }
257    // The emptied `para`/`p` shells, nested ones included (a leaf that was its own node has
258    // moved, not emptied).
259    if is_empty_shell(node) {
260      document.remove_node(node.clone());
261    }
262  }
263  Ok(Some(float))
264}

266/// Is `context` a float's own content, seen through the captures of the boxes and environments
267/// (`{center}`, `{minipage}`) built in it? A float's content is arranged into panels
268/// (`arrange_panels`, Perl's `arrange_panels_and_breaks`, latex_constructs.pool.ltxml:3229-3349).
269pub(super) fn in_panel_float(context: &Node) -> bool {
270  let mut node = context.clone();
271  while document::is_capture_block(&node) {
272    match node.get_parent() {
273      Some(parent) if parent.get_type() == Some(NodeType::ElementNode) => node = parent,
274      _ => return false,
275    }
276  }
277  with(document::get_node_qname(&node), |q| {
278    PANEL_FLOATS.contains(&q)
279  })
280}

290/// [`insert_block`] for a box that is paragraph material: its definition's `\leavevmode`
291/// (`enter_horizontal`) left TeX in a paragraph, the whatsit's `in_paragraph` property
292/// (`Constructor::digest_to_whatsit`). latex.ltx's `\parbox` ends in `\@iiiparbox` and `{minipage}`
293/// in `\@iiiminipage`, which begin with `\leavevmode` (16249-16250, 16305-16306): between
294/// paragraphs the box starts one, and the text after it continues that paragraph (KNOWN_PERL_ERRORS
295/// #309, DIVERGENCES #338).
296pub fn insert_block_in_paragraph(
297  document: &mut Document,
298  contents: &Digested,
299  block_attr: HashMap<String, String>,
300  props: &SymHashMap<Stored>,
301) -> Result<Vec<Node>> {
302  let in_paragraph = matches!(props.get("in_paragraph"), Some(Stored::Bool(true)));
303  let Some(id) = props.get("caption_id").map(|id| id.to_string()) else {
304    return insert_block_as(document, contents, block_attr, in_paragraph, true);
305  };
306  let captype = props
307    .get("caption_type")
308    .map(|t| t.to_string())
309    .unwrap_or_default();
310  // The float the caption is of — the nearest enclosing float element of its type — numbered by it for now (when no
311  // caption of its own did), so that what the box's content builds forms under that number as before: its ids, and
312  // the `\label`s `float_to_label` gives the nearest numbered ancestor (an `\fbox`, a `\colorbox` or an `lrbox` around
313  // the box makes no panel of it, and the float keeps the number).
314  let float = caption_float(document, &captype);
315  // A float an earlier box's panel took the number from holds a placeholder id meanwhile (`adopt_box_caption`).
316  let float_id = float.as_ref().and_then(node_xml_id);
317  let placeholder = float_id
318    .clone()
319    .filter(|held| lookup_bool(&s!("{FLOAT_PLACEHOLDER_ID}{held}")));
320  let provisional = match float.clone() {
321    Some(mut float) if float_id.is_none() || placeholder.is_some() => {
322      if let Some(held) = &placeholder {
323        document.unrecord_id(held);
324        remove_xml_id(&mut float);
325        remove_value(&s!("{FLOAT_PLACEHOLDER_ID}{held}"));
326      }
327      document.set_attribute(&mut float, "xml:id", &id)?;
328      Some((float, placeholder))
329    },
330    _ => None,
331  };
332  let open = ancestor_labels(document);
333  let mut nodes = insert_block_as(document, contents, block_attr, in_paragraph, true)?;
334  adopt_box_caption(document, &mut nodes, props, &id, provisional, open)?;
335  Ok(nodes)
336}

346/// The nearest enclosing float element of a caption's type: `ltx:figure`, `ltx:table`, or `ltx:float` of class
347/// `ltx_float_<type>` (sect09.rs).
348fn caption_float(document: &Document, captype: &str) -> Option<Node> {
349  let mut node = document.get_element();
350  while let Some(n) = node {
351    if with(document::get_node_qname(&n), |q| PANEL_FLOATS.contains(&q)) {
352      return is_float_of_type(&n, captype).then_some(n);
353    }
354    node = n.get_parent();
355  }
356  None
357}

359fn is_float_of_type(node: &Node, captype: &str) -> bool {
360  with(document::get_node_qname(node), |q| match q {
361    "ltx:figure" => captype == "figure",
362    "ltx:table" => captype == "table",
363    "ltx:float" => node.get_attribute("class").is_some_and(|classes| {
364      let class = s!("ltx_float_{captype}");
365      classes.split_whitespace().any(|c| c == class)
366    }),
367    _ => false,
368  })
369}

371/// Each element from the insertion point up — the nodes open while a box is built — with its `labels` as they stand.
372fn ancestor_labels(document: &Document) -> Vec<(Node, String)> {
373  let mut ancestors = Vec::new();
374  let mut node = document.get_element();
375  while let Some(n) = node {
376    if !matches!(n.get_type(), Some(NodeType::ElementNode)) {
377      break;
378    }
379    ancestors.push((n.clone(), n.get_attribute("labels").unwrap_or_default()));
380    node = n.get_parent();
381  }
382  ancestors
383}

385/// A box that holds its own caption in a float (`end_caption_box`) is that caption's float: the panel it became —
386/// itself, or the figure panel it holds, never a node that was open around it — takes the caption's id (from the
387/// float that held it for now, `provisional_float`), tags and list, becomes the element of the float it is in when
388/// that float is of the caption's type (a captioned minipage in a `table` is a `table`, which `insert_block` never
389/// picks; a `\newfloat` one keeps its `ltx_float_<type>` class), and takes the `\label`s its content left on the nodes
390/// open around it. With no panel, a float numbered for now keeps the number, and takes its tags and list too.
391fn adopt_box_caption(
392  document: &mut Document,
393  nodes: &mut [Node],
394  props: &SymHashMap<Stored>,
395  id: &str,
396  provisional_float: Option<(Node, Option<String>)>,
397  open: Vec<(Node, String)>,
398) -> Result<()> {
399  let caption_qname = pin_static("ltx:caption");
400  let is_open = |n: &Node| open.iter().any(|(o, _)| o == n);
401  let in_math = |n: &Node| {
402    let mut node = n.get_parent();
403    while let Some(p) = node {
404      if with(document::get_node_qname(&p), |q| {
405        q.starts_with("ltx:XM") || q == "ltx:Math"
406      }) {
407        return true;
408      }
409      node = p.get_parent();
410    }
411    false
412  };
413  // A panel built here: a float element holding the caption, outside math (whose ids are the math's own).
414  let built_panel = |n: &Node| {
415    with(document::get_node_qname(n), |q| PANEL_FLOATS.contains(&q))
416      && !is_open(n)
417      && !in_math(n)
418      && n
419        .get_child_elements()
420        .iter()
421        .any(|c| document::get_node_qname(c) == caption_qname)
422  };
423  let panel = nodes.iter().find(|n| built_panel(n)).cloned().or_else(|| {
424    nodes
425      .first()
426      .and_then(|n| n.get_parent())
427      .filter(|p| built_panel(p))
428  });
429  let Some(mut panel) = panel else {
430    if let Some((float, _)) = provisional_float {
431      number_caption_float(document, float, props)?;
432    }
433    return Ok(());
434  };
435  // The id: the float's, held for it, or the caption's own. The float keeps an id for what is built in it later, a
436  // placeholder the next box holds its caption's id in: without one a `\label` there would reach the document.
437  if let Some((mut float, placeholder)) = provisional_float {
438    if let Some(held) = node_xml_id(&float) {
439      document.unrecord_id(&held);
440    }
441    remove_xml_id(&mut float);
442    match placeholder {
443      Some(held) => document.set_attribute(&mut float, "xml:id", &held)?,
444      None => {
445        let prefix = if with(document::get_node_qname(&float), |q| q == "ltx:figure") {
446          "fig"
447        } else {
448          "tab"
449        };
450        document.generate_id(&mut float, prefix)?;
451      },
452    }
453    if let Some(held) = node_xml_id(&float) {
454      assign_value(
455        &s!("{FLOAT_PLACEHOLDER_ID}{held}"),
456        true,
457        Some(Scope::Global),
458      );
459    }
460  }
461  if let Some(old) = node_xml_id(&panel) {
462    document.unrecord_id(&old);
463    remove_xml_id(&mut panel);
464  }
465  document.set_attribute(&mut panel, "xml:id", id)?;
466  // The element of the float it is in, when that float is of the caption's type.
467  let captype = props
468    .get("caption_type")
469    .map(|t| t.to_string())
470    .unwrap_or_default();
471  // A sub-caption's panel (`sub<type>`, subcaption's `\subcaption` in a minipage) is the float's element too, as
472  // `{subtable}` is an `ltx:table` (2502.10377's minipage sub-tables).
473  if let Some(float) = open
474    .iter()
475    .map(|(o, _)| o)
476    .find(|o| with(document::get_node_qname(o), |q| PANEL_FLOATS.contains(&q)))
477    && (is_float_of_type(float, &captype)
478      || captype
479        .strip_prefix("sub")
480        .is_some_and(|parent| is_float_of_type(float, parent)))
481  {
482    let float_qname = document::get_node_qname(float);
483    if document::get_node_qname(&panel) != float_qname
484      && panel
485        .get_parent()
486        .is_some_and(|p| with(float_qname, |q| document::can_contain(&p, q)))
487    {
488      let old = panel.clone();
489      panel = document.rename_node_qsym(panel, float_qname, true)?;
490      for node in nodes.iter_mut() {
491        if *node == old {
492          *node = panel.clone();
493        }
494      }
495    }
496    if with(float_qname, |q| q == "ltx:float") {
497      document.add_class(&mut panel, &s!("ltx_float_{captype}"))?;
498    }
499  }
500  let panel = number_caption_float(document, panel, props)?;
501  // The labels its content left on the nodes open around it.
502  let mut moved: Vec<String> = Vec::new();
503  for (mut ancestor, before) in open {
504    let Some(after) = ancestor.get_attribute("labels") else {
505      continue;
506    };
507    let kept: Vec<&str> = before.split_whitespace().collect();
508    let added: Vec<&str> = after
509      .split_whitespace()
510      .filter(|l| !kept.contains(l))
511      .collect();
512    if added.is_empty() {
513      continue;
514    }
515    moved.extend(added.iter().map(|l| l.to_string()));
516    if kept.is_empty() {
517      ancestor.remove_attribute("labels")?;
518    } else {
519      document.set_attribute(&mut ancestor, "labels", &kept.join(" "))?;
520    }
521  }
522  if !moved.is_empty() {
523    let mut panel = panel;
524    let mut labels: Vec<String> = panel
525      .get_attribute("labels")
526      .map(|l| l.split_whitespace().map(str::to_string).collect())
527      .unwrap_or_default();
528    for label in moved {
529      if !labels.contains(&label) {
530        labels.push(label);
531      }
532    }
533    document.set_attribute(&mut panel, "labels", &labels.join(" "))?;
534  }
535  Ok(())
536}

538/// `float` takes a caption's tags (first, where a float's own template places them) and list (`end_caption_box`'s
539/// `caption_tags` and `caption_inlist`).
540fn number_caption_float(
541  document: &mut Document,
542  mut float: Node,
543  props: &SymHashMap<Stored>,
544) -> Result<Node> {
545  if let Some(inlist) = props.get("caption_inlist").map(|l| l.to_string())
546    && !inlist.trim().is_empty()
547  {
548    document.set_attribute(&mut float, "inlist", inlist.trim())?;
549  }
550  if let Some(Stored::Digested(tags)) = props.get("caption_tags") {
551    let save = document.get_node().clone();
552    document.set_node(&float);
553    let before = float.get_last_element_child();
554    document.absorb(tags, None)?;
555    document.set_node(&save);
556    let mut added = float.get_last_element_child();
557    if added != before
558      && let Some(tags) = added.as_mut()
559      && let Some(mut first) = float.get_first_element_child()
560      && &first != tags
561    {
562      tags.unlink();
563      first.add_prev_sibling(tags)?;
564    }
565  }
566  Ok(float)
567}

569/// Whether the element `node` was built from one thing that shows: its box is a whatsit whose last argument — a
570/// constructor's body (`\rotatebox`, `\scalebox`, `\resizebox`, `\fbox`, `\colorbox`) — has exactly one item,
571/// nested groups flattened, that shows something ([`typesets_content`]). Any other box refuses. An alignment
572/// environment around the transform is read by [`environment_is_sole_content`] instead.
573fn box_is_sole_content(document: &Document, node: &Node) -> bool {
574  fn shown(d: &Digested) -> usize {
575    match d.data() {
576      DigestedData::List(l) => l.borrow().boxes.iter().map(shown).sum(),
577      _ => usize::from(typesets_content(d)),
578    }
579  }
580  let Some(whatsit) = document.get_node_box(node) else {
581    return false;
582  };
583  let DigestedData::Whatsit(w) = whatsit.data() else {
584    return false;
585  };
586  let w = w.borrow();
587  matches!(w.get_args().last(), Some(Some(body)) if shown(body) == 1)
588}

590/// Whether the environment an `ltx:_CaptureBlock_` is being built for is an alignment wrapper (`{center}`,
591/// `{flushleft}`, `{flushright}`: its box is the environment's whatsit, whose body is what it holds between `\begin`
592/// and `\end`) holding one thing that shows: the transform alone, not a transform and a tabular after it. Any other
593/// environment refuses (`{minipage}`, `{framed}`).
594fn environment_is_sole_content(document: &Document, capture: &Node) -> bool {
595  fn shown(d: &Digested) -> usize {
596    match d.data() {
597      DigestedData::List(l) => l.borrow().boxes.iter().map(shown).sum(),
598      _ => usize::from(typesets_content(d)),
599    }
600  }
601  let Some(whatsit) = document.get_node_box(capture) else {
602    return false;
603  };
604  let DigestedData::Whatsit(w) = whatsit.data() else {
605    return false;
606  };
607  let w = w.borrow();
608  // only the alignment wrappers, which make no box of their own (a `{minipage}` keeps its width and class as a panel,
609  // which the caption would leave: two minipages side by side shrank to the rotated box; a `{framed}` frame would move
610  // onto the float)
611  [
612    "\\begin{center}",
613    "\\begin{flushleft}",
614    "\\begin{flushright}",
615  ]
616  .contains(&w.get_definition().get_cs_name().as_ref())
617    && matches!(w.get_body(), Ok(Some(body)) if shown(&body) == 1)
618}

620/// Whether a box's closing caption may be moved from inside the transform `context` to the float `home` its climb
621/// found ([`insert_block_as`]): every element from `context` up to `home` was built from this box alone
622/// ([`box_is_sole_content`]: no content beside or after it would follow the moved insertion point out of a frame
623/// or a transform) and none is a paragraph, and `home` is a float whose own content holds no other caption (one
624/// `\lx@@caption`: a second would make the two one float, with one number for two `\label`s — two `\resizebox`ed
625/// captioned panels, a float's own `\caption` after the box, `\subcaption` panels). Otherwise Perl's container
626/// stands. Witnesses heria/heria-proposal (heria.cls:737-745 `SidewaysFigure`), 1810.08269; guards
627/// `perfect_kernel_batch65a::caption_in_a_rotated_parbox_stays_with_its_figure`,
628/// `caption_hoist_never_merges_two_captions`, `caption_hoist_never_moves_content_beside_the_box`.
629/// OXIDIZED_DESIGN #529.
630pub(super) fn caption_hoist_is_safe(document: &Document, context: &Node, home: &Node) -> bool {
631  fn captions(d: &Digested) -> usize {
632    match d.data() {
633      DigestedData::List(l) => l.borrow().boxes.iter().map(captions).sum(),
634      DigestedData::Whatsit(w) => {
635        let w = w.borrow();
636        usize::from(w.get_definition().get_cs_name() == "\\lx@@caption")
637          + w.get_args().iter().flatten().map(captions).sum::<usize>()
638          + w
639            .get_body()
640            .ok()
641            .flatten()
642            .map_or(0, |body| captions(&body))
643      },
644      _ => 0,
645    }
646  }
647  let is_float = |n: &Node| with(document::get_node_qname(n), |q| PANEL_FLOATS.contains(&q));
648  // The home an alignment environment's capture block (`{center}`, `{flushleft}`, `{flushright}` around the
649  // transform: the caption lands in what the environment builds, which its float then holds): the environment holds
650  // the transform alone, and the float right above it is the one whose captions count.
651  let float = if is_float(home) {
652    home.clone()
653  } else if with(document::get_node_qname(home), |q| {
654    q == "ltx:_CaptureBlock_"
655  }) && environment_is_sole_content(document, home)
656    && let Some(parent) = home.get_parent().filter(is_float)
657  {
658    parent
659  } else {
660    return false;
661  };
662  let mut node = Some(context.clone());
663  while let Some(n) = node {
664    if &n == home {
665      return document
666        .get_node_box(&float)
667        .is_some_and(|float| captions(&float) == 1);
668    }
669    // A paragraph on the way is open to what follows the box in it (`\fbox{\rotatebox…} trailing text`): its
670    // node box is only what opened it, so nothing tells the rest of it is not there.
671    if with(document::get_node_qname(&n), |q| {
672      q == "ltx:p" || q == "ltx:para"
673    }) {
674      return false;
675    }
676    if !box_is_sole_content(document, &n) {
677      return false;
678    }
679    node = n.get_parent();
680  }
681  false
682}

684/// Where the content a box cannot hold is placed after it ([`insert_block_as`]'s hoist): the nearest ancestor of
685/// `context` that holds all of `tail_tags`, climbing only through text containers, never out of a list, an alignment
686/// or a float, and the node it follows there (the box itself, or the ancestor's child the climb came from).
687pub(super) fn tail_home(
688  context: &Node,
689  container: &Node,
690  tail_tags: &[SymStr],
691  kept_empty: bool,
692) -> Option<(Node, Node)> {
693  let flow = |n: &Node| {
694    document::can_contain(n, "#PCDATA")
695      || with(document::get_node_qname(n), |t| {
696        document::can_contain_somehow(t, "ltx:p")
697      })
698  };
699  // LIST: the climb never leaves a list. Perl floats nothing out of a list
700  // (`adjustBackmatterElement` autocloses only `canAutoClose` nodes and a
701  // drawing's `svg:g` is not one, latex_constructs.pool.ltxml:3936 /
702  // Document.pm find_insertion_point): a bibliography arriving ALONE from a
703  // tcolorbox inside an `\item` (`kept` empty, so the TEXT guard above did
704  // not hold it) climbed through the `ltx:item`/`ltx:itemize` to the
705  // section, which closed the list under the remaining items — they then
706  // nested inside one another (biblatex-ext.tex:1301/1326 `optionlist` +
707  // `bibexample`, 4 schema errors; RUST-ONLY). The block stays in the box
708  // and the model reports it, as Perl does (SHARED `ltx:bibliography` in
709  // `ltx:block`). The list-free hoists keep their surpass (xebaposter,
710  // juradiss, the parbox caption). Guard
711  // `bibliography_in_a_list_item_box_keeps_the_list`.
712  let list_boundary = |n: &Node| {
713    with(document::get_node_qname(n), |t| {
714      matches!(
715        t,
716        "ltx:item"
717          | "ltx:inline-item"
718          | "ltx:itemize"
719          | "ltx:enumerate"
720          | "ltx:description"
721          | "ltx:inline-itemize"
722          | "ltx:inline-enumerate"
723          | "ltx:inline-description"
724      )
725    })
726  };
727  let mut anchor = container.clone();
728  let mut ancestor = context.clone();
729  loop {
730    if tail_tags
731      .iter()
732      .all(|t| with(*t, |s| document::can_contain(&ancestor, s)))
733    {
734      return Some((anchor, ancestor));
735    }
736    if list_boundary(&ancestor) || (!flow(&ancestor) && !kept_empty) {
737      break;
738    }
739    // Never out of an alignment: the hoist moves the insertion point too, past a cell whose `</td>` (and row, and
740    // table) the alignment still owes at its `&`, `\\` and `\end` — a caption in a minipage in a tabular cell left
741    // `<td>` open in the figure (1601.03744, 5 malformed errors); a math `array` likewise. Perl's `floatToElement`
742    // (Document.pm:1061-1072) floats past such nodes without closing them and restores the insertion point after;
743    // a box the climb leaves here (the rotated `\parbox`) is closed already.
744    let alignment = |n: &Node| {
745      with(document::get_node_qname(n), |t| {
746        matches!(
747          t,
748          "ltx:td"
749            | "ltx:tr"
750            | "ltx:thead"
751            | "ltx:tbody"
752            | "ltx:tfoot"
753            | "ltx:tabular"
754            | "ltx:XMCell"
755            | "ltx:XMRow"
756            | "ltx:XMArray"
757        )
758      })
759    };
760    // Nor out of a float: an appendix section (or a bibliography) in a minipage in a `table` was lifted past it,
761    // leaving the table empty and its caption at the document's top level, with no diagnostic; it now stays in the
762    // float and errors there, as Perl's does (a sectioning unit in a float errors, OXIDIZED_DESIGN_DIVERGENCES #189
763    // ruling). Witness 2609.05763 (a nomenclature glossary in a framed minipage in a table); guard
764    // `perfect_kernel_batch61::appendix_in_a_minipage_stays_in_its_float`.
765    // The climb now has three per-tag stops (list, alignment, float), and a quote still leaks
766    // (repros/boxes-groups/bibliography_in_minipage_in_quote_keeps_the_quote.tex): SYNC_STATUS 62zd.
767    let float = |n: &Node| {
768      with(document::get_node_qname(n), |t| {
769        matches!(t, "ltx:table" | "ltx:figure" | "ltx:float")
770      })
771    };
772    if alignment(&ancestor) || float(&ancestor) {
773      break;
774    }
775    match ancestor.get_parent() {
776      Some(p) if matches!(p.get_type(), Some(NodeType::ElementNode)) => {
777        anchor = ancestor;
778        ancestor = p;
779      },
780      _ => break,
781    }
782  }
783  None
784}
