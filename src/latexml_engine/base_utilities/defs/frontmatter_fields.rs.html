183  // Field provenance (frontmatter model M3, stage S1): `\lx@field@mark<prefix><store>{<field>}` — the store, a macro
184  // without parameters, redefined (`<prefix>` `\def` or `\gdef`, the store's own scope at the setter) with the same
185  // tokens, each now carrying out-of-band provenance "token <i> of field <field>" (`Token::prov`). Nothing is added to
186  // the store: every token operation sees exactly the tokens pdflatex sees (`\ifx\@title\same`, `\equal`, xstring's
187  // `\StrLen`/`\StrLeft`, `\pdfstrcmp`, `\@gobble`, `\@car`, `\csname`, `\write`), since token equality and every
188  // character form ignore the provenance. What expands from a stamped token inherits it and what is digested from
189  // one carries it (`fieldProv`), so the document's own print of the store is known for the field's ink by
190  // construction (core `Document::absorb`'s field runs; [`lift_marked_field_ink`]). The kernel `\title` and `\author`
191  // (sect05.rs) call it after the frontmatter took its copy. An empty or blank value is left as it is; anything but
192  // a parameterless macro is left alone. OXIDIZED_DESIGN #715.
193  DefMacro!("\\lx@field@mark Token DefToken {}", sub[(prefix, store, field)] {
194    let field = field.to_string();
195    let Some(field_id) = field_id(field.trim()) else {
196      return Ok(Tokens::default());
197    };
198    let Some(defn) = lookup_definition(&store)? else {
199      return Ok(Tokens::default());
200    };
201    if defn.get_parameters().is_some_and(|p| !p.get_parameters().is_empty()) {
202      return Ok(Tokens::default());
203    }
204    let Some(ExpansionBody::Tokens(value)) = defn.get_expansion() else {
205      return Ok(Tokens::default());
206    };
207    if value.unlist_ref().iter().all(|t| t.get_catcode() == Catcode::SPACE) {
208      return Ok(Tokens::default());
209    }
210    // Each call's positions follow the previous call's (`lx_field_base_<id>`), so a copy of an earlier value
211    // (`\let\old\@title`, then `\title{Short}`) is never taken for a print of this one (review finding 7).
212    let base = usize::try_from(lookup_int(&s!("lx_field_base_{field_id}"))).unwrap_or(0);
213    let len = value.unlist_ref().len();
214    if base + len > usize::from(FIELD_PROV_POS_MASK) {
215      // out of positions: this value is not stamped, and no earlier copy can be whole of it
216      assign_value(&s!("lx_field_extent_{field_id}"), String::new(), Some(Scope::Global));
217      assign_value(&s!("lx_field_range_{field_id}"), String::new(), Some(Scope::Global));
218      // (the frontmatter observer's ledger: OXIDIZED_DESIGN_DIVERGENCES #837)
219      note_field_stamp(field_id, None, len);
220      return Ok(Tokens::default());
221    }
222    assign_value(&s!("lx_field_base_{field_id}"), (base + len) as i64, Some(Scope::Global));
223    // this call's positions: a print is of the value it sets only within them (an earlier call's print or copy, r12,
224    // r13, is not joined with it, [`field_ink_runs`])
225    assign_value(
226      &s!("lx_field_range_{field_id}"),
227      s!("{base}:{}", base + len),
228      Some(Scope::Global),
229    );
230    // The field's extent: the position of every token of the value that bears ink or makes it (a character, a
231    // control sequence, an active character; a macro's expansion and a primitive's arguments are covered from its
232    // position, core `gullet::field_cover_of`), which a whole print covers, all of them: a fragment the document
233    // takes of the value (`\StrLeft{\@title}{5}` of `About \nm`, `\@car`, `\@gobble`) leaves some out (review
234    // finding 2). Not a space, a brace, `$`, `^`, `_`, `&`, `#`: what they make is covered by the tokens around them.
235    // (as `start:end` intervals, end exclusive, joined by `,`: the form of a run's cover, [`field_ink_runs`])
236    // The value's characters (a letter, an other character) are kept apart too: a print digests each once, so a run
237    // that holds one twice read the value again ([`field_ink_runs`]).
238    let mut extent: Vec<(usize, usize)> = Vec::new();
239    let mut chars: Vec<(usize, usize)> = Vec::new();
240    let push = |intervals: &mut Vec<(usize, usize)>, at: usize| match intervals.last_mut() {
241      Some((_, end)) if *end == at => *end += 1,
242      _ => intervals.push((at, at + 1)),
243    };
244    for (i, t) in value.unlist_ref().iter().enumerate() {
245      let code = t.get_catcode();
246      if matches!(
247        code,
248        Catcode::LETTER | Catcode::OTHER | Catcode::CS | Catcode::ACTIVE
249      ) {
250        push(&mut extent, base + i);
251      }
252      if matches!(code, Catcode::LETTER | Catcode::OTHER) {
253        push(&mut chars, base + i);
254      }
255    }
256    if extent.is_empty() {
257      return Ok(Tokens::default());
258    }
259    let joined = |intervals: &[(usize, usize)]| {
260      intervals
261        .iter()
262        .map(|(start, end)| s!("{start}:{end}"))
263        .collect::<Vec<_>>()
264        .join(",")
265    };
266    assign_value(&s!("lx_field_extent_{field_id}"), joined(&extent), Some(Scope::Global));
267    assign_value(&s!("lx_field_chars_{field_id}"), joined(&chars), Some(Scope::Global));
268    document::note_field_prov_stamped();
269    note_field_stamp(field_id, Some(base), len);
270    let mut out = vec![prefix, store, T_BEGIN!()];
271    for (i, t) in value.unlist_ref().iter().enumerate() {
272      let stamped = Token { prov: field_prov(field_id, base + i), ..*t };
273      // A `#` in the value is text here, not a parameter of the definition (as `\lx@deposit@field`).
274      if t.get_catcode() == Catcode::PARAM {
275        out.push(stamped);
276      }
277      out.push(stamped);
278    }
279    out.push(T_END!());
280    Ok(Tokens::new(out))
281  });
282  // The frontmatter observer (FM S2, OXIDIZED_DESIGN_DIVERGENCES #837): `\lx@field@stamp{<field>}{<tokens>}` — the
283  // tokens of an argument, each stamped with its position in field <field> as `\lx@field@mark` stamps a store's
284  // value, the positions following the field's earlier stamps (`lx_field_base_<id>`): M3's "marked field argument",
285  // for a binding's `\lx@add@authors*` call, whose argument no store holds. Nothing else changes (the extent and the
286  // range of the field's current call are a store's, `\lx@field@mark`'s). An unknown field, a blank argument or one
287  // past the field's positions comes back unstamped. No call site yet: the observer's ledger projects these
288  // positions for the bindings' calls (`record_author_call`), and stage S2b stamps them.

300    }

368  // A note the title block of a class whose `\thanks` is `\footnotemark` and its text makes (article, report, book,
369  // mnras, lni: `lx_title_counts_footnotes`, latex_constructs `declare_title_footnote_counting`) steps the footnote counter, as the
370  // `\maketitle` digests it (`lx_frontmatter_explicit`): `\thanks` and `\footnote` in the title, an author or the date
371  // (`lx_frontmatter_item`), so a `\footnotemark[\arabic{footnote}]` after one prints its mark (2201.11183, 2111.13260;
372  // KNOWN_PERL_ERRORS #1060, OXIDIZED_DESIGN_DIVERGENCES #851). A flush without `\maketitle` typesets no title, and
373  // steps nothing.
374  DefPrimitive!("\\lx@title@footnote@step", sub[_args] {
375    let titles = matches!(lookup_string("lx_frontmatter_item").as_str(), "ltx:title" | "ltx:date");
376    // (a title's or date's note: its number, which a name's `\footnotemark[n]` may name, `insert::link_marked_notes`;
377    // `?` for one TeX counts and the reading does not, which leaves the authors' numbers off by it, MARKKEY1c)
378    let value = if title_note_steps() {
379      step_counter("footnote", false)?;
380      titles.then(|| counter_value("footnote").map(|count| count.value_of().to_string())).transpose()?
381    } else {
382      titles.then(|| "?".to_string())
383    };
384    if let Some(value) = value {
385      let values = lookup_string(TITLE_NOTE_VALUES);
386      let values = if values.is_empty() { value } else { s!("{values},{value}") };
387      assign_value(TITLE_NOTE_VALUES, values, Some(Scope::Global));

1142pub(in crate::base_utilities) fn author_markup_begin(rehearsal: bool) -> Result<AuthorMarkup> {
1143  // A binding's author markup for a class that defines its own only where its title code prints the author list
1144  // (`\lx@author@markup@class`, `\lx@author@markup@provide` calls: vmsta.cls:278-291 `\name@m@front`, which prints
1145  // `\fnm`/`\snm` as "Yuliya~Mishura" and drops `\inits`; latexml_contrib vmsta_cls.rs) comes first, before
1146  // OmniBus's identity meanings would take the names it gives.
1147  let mut out = TokenizeInternal!(r"\def\lx@author@markup@unprovide{}").unlist();
1148  if IsDefined!(&T_CS!("\\lx@author@markup@class")) {
1149    out.push(T_CS!("\\lx@author@markup@class"));
1150  }
1151  out.extend(if lookup_bool("\\author:redefined") {
1152    TokenizeInternal!(
1153      r"\lx@author@markup@provide\prefix\@firstofone\lx@author@markup@provide\suffix\@firstofone\lx@author@markup@provide\particle\@firstofone\lx@author@markup@provide\fnms\@firstofone\lx@author@markup@provide\snm\@firstofone\lx@author@markup@provide\inits\@firstofone\lx@author@markup@provide\degs\@firstofone\lx@author@markup@provide\roles\@firstofone\lx@author@markup@provide\orcid\lx@author@orcid"
1154    )
1155    .unlist()
1156  } else {
1157    Vec::new()
1158  });
1159  // The author markup a class defines only where its title code reads `\@author` is in force for the author content
1160  // too: melba.cls:279-283 `{\def\aff{…}\def\name{…}\@author …}` (2405.09787, 2407.21368, 2105.14711), dmlr2e.sty:
1161  // 249-251 (2404.08403), amta2024.sty:77-78 (2405.16969), ssArxiv.sty:109-111 (2503.22625), a preamble's own
1162  // `\@maketitle` (2601.05137). In pdflatex `\@author` expands inside that group, while here the author content is
1163  // read at the frontmatter, before the title code runs. Each definition of a control sequence still undefined runs,
1164  // in source order, for the author content, which it is undefined again after, as the OmniBus author markup is
1165  // (#444); a definition already in force stands. Perl errs the same (`undefined:\name`).
1166  let definitions = title_code_author_definitions();
1167  let applies: Vec<bool> = definitions
1168    .iter()
1169    .map(|(definee, statement)| {
1170      lookup_meaning(definee).is_none() && (!rehearsal || reads_arguments(definee, statement))
1171    })
1172    .collect();

1174  for ((definee, statement), applies) in definitions.into_iter().zip(applies) {
1175    if !applies {
1176      continue;
1177    }
1178    title_code.push(definee);
1179    out.extend(statement);
1180    out.extend(
1181      TokenizeInternal!(r"\expandafter\def\expandafter\lx@author@markup@unprovide\expandafter{\lx@author@markup@unprovide\let")
1182        .unlist(),
1183    );
1184    out.extend([definee, T_CS!("\\@undefined"), T_END!()]);
1185  }
1186  Ok(AuthorMarkup {
1187    begin: Tokens::new(out),
1188    title_code,
1189  })
1190}

1215/// An address's formula as TeX prints it where the formula is glyphs: a pair of math shifts — read as characters (a
1216/// Semiverbatim argument made `$` other: the API's `\lx@add@url`/`\lx@add@email`, Perl's `neutralize`) or as shifts —
1217/// round a run of characters, of character commands ([`text_glyph`]: `\_`, `\{`) and of math symbols that each print
1218/// one glyph ([`math_symbol_glyph`]: `\sim`, `\cdot`) is the run, each command its glyph, the shifts and spaces dropped as math drops spaces:
1219/// `http://www.zebra.edu/$\sim$ann` prints "http://www.zebra.edu/∼ann", as pdflatex, where the dollars printed
1220/// (1301.1466's `\ead[label=u2,url]{http://www.math.duke.edu/$\sim$omer}`, math0608476's
1221/// `www.math.wisc.edu/$\sim$swanson`). The glyph is a character of the address, in its font (TeX takes it from the
1222/// math font), so the address stays one text and its link whole. The shifts pair left to right, as TeX pairs them: a
1223/// pair round a script (`^`, `_`: `$^*$x@y`'s mark), a command of another kind or nothing (`$$`) stays as it is,
1224/// whole, and the next pair opens after its closing shift — `$^a$a@x, $^b$b@y` keeps both marks and the text between
1225/// them (2508.20055's `ej$_$dresselhaus@berkeley.edu, s$_$g@berkeley.edu`). A pair round spacing alone
1226/// ([`spacing_formulas_as_space`]: `$~\quad$`, `$\,$`) is glue, no glyph: a space between printed text, none at an
1227/// edge (2205.14113's `$~\quad~~$`).
1228pub(in crate::base_utilities) fn math_glyphs_as_printed(address: &Tokens) -> Tokens {
1229  let address = spacing_formulas_as_space(address.clone());
1230  let v = address.unlist_ref();
1231  let shift = |t: &Token| *t == T_OTHER!("$") || t.get_catcode() == Catcode::MATH;
1232  let script = |t: &Token| [T_OTHER!("^"), T_OTHER!("_")].contains(t);
1233  let glyphs = |t: &Token| -> Option<Vec<Token>> {
1234    match t.get_catcode() {
1235      Catcode::SPACE => Some(Vec::new()),
1236      Catcode::LETTER | Catcode::OTHER if !shift(t) && !script(t) => Some(vec![*t]),
1237      Catcode::CS => text_glyph(t)
1238        .map(|glyph| vec![glyph])
1239        .or_else(|| math_symbol_glyph(t).map(|glyph| text_tokens(&glyph))),
1240      _ => None,
1241    }
1242  };
1243  let mut out: Vec<Token> = Vec::with_capacity(v.len());
1244  let mut at = 0;
1245  while at < v.len() {
1246    // the shifts pair left to right, as TeX pairs them: a pair that is not glyphs (a script, the empty `$$`) stays as
1247    // read, whole, and the next pair starts after its closing shift (2508.20055's `ej$_$d@x, s$_$g@y`)
1248    if shift(&v[at])
1249      && let Some(len) = v[at + 1..].iter().position(shift)
1250    {
1251      let pair = &v[at..at + len + 2];
1252      match pair[1..=len]
1253        .iter()
1254        .map(glyphs)
1255        .collect::<Option<Vec<Vec<Token>>>>()
1256      {
1257        Some(printed) if printed.iter().any(|glyph| !glyph.is_empty()) => {
1258          out.extend(printed.into_iter().flatten())
1259        },
1260        _ => out.extend_from_slice(pair),
1261      }
1262      at += len + 2;
1263      continue;
1264    }
1265    out.push(v[at]);
1266    at += 1;
1267  }
1268  Tokens::new(out)
1269}
