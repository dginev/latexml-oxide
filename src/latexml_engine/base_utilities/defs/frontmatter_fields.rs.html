43  // superscript inside its title box; beamerbasetitle.sty:262

45  // content ONLY — the shape of Perl's `\lx@author@withsup`

166  // Where the abstract SITS in the source: an empty internal marker constructed at
167  // the current point when the abstract is queued (both the `{abstract}` env and a
168  // class's `\abstract{…}` store route through the two macros below). The
169  // abstract-only fallback reads it to tell title-page LAYOUT (what precedes the
170  // abstract) from BODY (what follows it) — the flush point (first `\section`, or
171  // `\end{document}`) cannot: a sectionless manual would otherwise see its whole
172  // body as cover. `_Capture_`-suffixed names are admitted anywhere by the model
173  // (model.rs) so no paragraph is auto-opened; every marker is removed by
174  // `insert_frontmatter`/`remove_frontmatter_marks` before output. OXIDIZED_DESIGN #246.
175  //
176  // A LATE abstract — queued after `\maketitle`'s flush (`\lx@frontmatterhere`'s `after_digest`), which placed it
177  // with the head of the frontmatter (the eager build digests the whole queue there) or leaves it to
178  // `insert_late_frontmatter` (a streamed one) — gets a marker of its own (`LATE_ABSTRACT_MARK`): the title-page text
179  // printed between `\maketitle` and it is its cover, and the abstract goes behind that (`place_abstract_after_cover`,
180  // OXIDIZED_DESIGN_DIVERGENCES #903; AAAI's `\affiliations` in its `\@maketitle` deposit, 2011.04405, 2102.01904;
181  // hand-typeset lines, 1302.5352, 1702.07128). Settled at once where the abstract is placed, so no marker outlives
182  // its reader (s107: 8 docs leaked `<_Frontmatter_Capture_>`; lips' `\DocInput`-ed `\begin{abstract}` after
183  // `\maketitle`).
184  DefConstructor!("\\lx@frontmatter@mark{}", sub[document, _args, props] {
185    // Only once there is an open element to sit in: a preamble `\abstract{…}`
186    // (svjour's setter, fixture structure/svabstract) is constructed BEFORE the
187    // whatsit that opens <ltx:document>, and inserting anything there auto-creates
188    // the root early (a duplicated RelaxNGSchema PI). Construction runs after
189    // digestion, so the digest-time `inPreamble` flag is useless here — ask the
190    // tree via the current element (which, unlike the root element, also holds
191    // inside a streaming fragment; 114_streaming_structure faketitlepage).
192    if document.get_element().is_none() {
193      return Ok(());
194    }
195    if !lookup_bool("frontmatter_done") {
196      document.insert_mark(FRONTMATTER_MARK, None)?;
197    } else if prop_bool!(props, "late") {
198      let ordinal = match props.get("late_mark") {
199        Some(&Stored::Int(n)) => n.to_string(),
200        _ => String::new(),
201      };
202      let attrib: HashMap<String, String> =
203        [(LATE_MARK_ORDINAL.to_string(), ordinal)].into_iter().collect();
204      let marker = document.insert_element(LATE_ABSTRACT_MARK, Vec::new(), Some(attrib))?;
205      settle_late_abstract_mark(document, marker)?;
206    }
207  },
208  // (each late marker's ordinal, which the abstract its command adds carries: [`LATE_MARK_ORDINAL`])
209  properties => {
210    Ok(if lookup_bool("lx_frontmatter_here_digested") {
211      let ordinal = lookup_int(LATE_MARKS) + 1;

213      stored_map!("late" => true, "late_mark" => Stored::Int(ordinal))
214    } else {
215      SymHashMap::default()
216    })

218  // Field provenance (frontmatter model M3, stage S1): `\lx@field@mark<prefix><store>{<field>}` — the store, a macro
219  // without parameters, redefined (`<prefix>` `\def` or `\gdef`, the store's own scope at the setter) with the same
220  // tokens, each now carrying out-of-band provenance "token <i> of field <field>" (`Token::prov`). Nothing is added to
221  // the store: every token operation sees exactly the tokens pdflatex sees (`\ifx\@title\same`, `\equal`, xstring's
222  // `\StrLen`/`\StrLeft`, `\pdfstrcmp`, `\@gobble`, `\@car`, `\csname`, `\write`), since token equality and every
223  // character form ignore the provenance. What expands from a stamped token inherits it and what is digested from
224  // one carries it (`fieldProv`), so the document's own print of the store is known for the field's ink by
225  // construction (core `Document::absorb`'s field runs; [`lift_marked_field_ink`]). The kernel `\title` and `\author`
226  // (sect05.rs) call it after the frontmatter took its copy. An empty or blank value is left as it is; anything but
227  // a parameterless macro is left alone. OXIDIZED_DESIGN #715.
228  DefMacro!("\\lx@field@mark Token DefToken {}", sub[(prefix, store, field)] {
229    let field = field.to_string();
230    let Some(field_id) = field_id(field.trim()) else {
231      return Ok(Tokens::default());
232    };
233    let Some(defn) = lookup_definition(&store)? else {
234      return Ok(Tokens::default());
235    };
236    if defn.get_parameters().is_some_and(|p| !p.get_parameters().is_empty()) {
237      return Ok(Tokens::default());
238    }
239    let Some(ExpansionBody::Tokens(value)) = defn.get_expansion() else {
240      return Ok(Tokens::default());
241    };
242    if value.unlist_ref().iter().all(|t| t.get_catcode() == Catcode::SPACE) {
243      return Ok(Tokens::default());
244    }
245    // Each call's positions follow the previous call's (`lx_field_base_<id>`), so a copy of an earlier value
246    // (`\let\old\@title`, then `\title{Short}`) is never taken for a print of this one (review finding 7).
247    let base = usize::try_from(lookup_int(&s!("lx_field_base_{field_id}"))).unwrap_or(0);
248    let len = value.unlist_ref().len();
249    if base + len > usize::from(FIELD_PROV_POS_MASK) {
250      // out of positions: this value is not stamped, and no earlier copy can be whole of it
251      assign_value(&s!("lx_field_extent_{field_id}"), String::new(), Some(Scope::Global));
252      assign_value(&s!("lx_field_range_{field_id}"), String::new(), Some(Scope::Global));
253      // (the frontmatter observer's ledger: OXIDIZED_DESIGN_DIVERGENCES #837)
254      note_field_stamp(field_id, None, len);
255      return Ok(Tokens::default());
256    }
257    assign_value(&s!("lx_field_base_{field_id}"), (base + len) as i64, Some(Scope::Global));
258    // this call's positions: a print is of the value it sets only within them (an earlier call's print or copy, r12,
259    // r13, is not joined with it, [`field_ink_runs`])
260    assign_value(
261      &s!("lx_field_range_{field_id}"),
262      s!("{base}:{}", base + len),
263      Some(Scope::Global),
264    );
265    // The field's extent: the position of every token of the value that bears ink or makes it (a character, a
266    // control sequence, an active character; a macro's expansion and a primitive's arguments are covered from its
267    // position, core `gullet::field_cover_of`), which a whole print covers, all of them: a fragment the document
268    // takes of the value (`\StrLeft{\@title}{5}` of `About \nm`, `\@car`, `\@gobble`) leaves some out (review
269    // finding 2). Not a space, a brace, `$`, `^`, `_`, `&`, `#`: what they make is covered by the tokens around them.
270    // (as `start:end` intervals, end exclusive, joined by `,`: the form of a run's cover, [`field_ink_runs`])
271    // The value's characters (a letter, an other character) are kept apart too: a print digests each once, so a run
272    // that holds one twice read the value again ([`field_ink_runs`]).
273    let mut extent: Vec<(usize, usize)> = Vec::new();
274    let mut chars: Vec<(usize, usize)> = Vec::new();
275    let push = |intervals: &mut Vec<(usize, usize)>, at: usize| match intervals.last_mut() {
276      Some((_, end)) if *end == at => *end += 1,
277      _ => intervals.push((at, at + 1)),
278    };
279    for (i, t) in value.unlist_ref().iter().enumerate() {
280      let code = t.get_catcode();
281      if matches!(
282        code,
283        Catcode::LETTER | Catcode::OTHER | Catcode::CS | Catcode::ACTIVE
284      ) {
285        push(&mut extent, base + i);
286      }
287      if matches!(code, Catcode::LETTER | Catcode::OTHER) {
288        push(&mut chars, base + i);
289      }
290    }
291    if extent.is_empty() {
292      return Ok(Tokens::default());
293    }
294    let joined = |intervals: &[(usize, usize)]| {
295      intervals
296        .iter()
297        .map(|(start, end)| s!("{start}:{end}"))
298        .collect::<Vec<_>>()
299        .join(",")
300    };
301    assign_value(&s!("lx_field_extent_{field_id}"), joined(&extent), Some(Scope::Global));
302    assign_value(&s!("lx_field_chars_{field_id}"), joined(&chars), Some(Scope::Global));
303    document::note_field_prov_stamped();
304    note_field_stamp(field_id, Some(base), len);
305    let mut out = vec![prefix, store, T_BEGIN!()];
306    for (i, t) in value.unlist_ref().iter().enumerate() {
307      let stamped = Token { prov: field_prov(field_id, base + i), ..*t };
308      // A `#` in the value is text here, not a parameter of the definition (as `\lx@deposit@field`).
309      if t.get_catcode() == Catcode::PARAM {
310        out.push(stamped);
311      }
312      out.push(stamped);
313    }
314    out.push(T_END!());
315    Ok(Tokens::new(out))
316  });
317  // The frontmatter observer (FM S2, OXIDIZED_DESIGN_DIVERGENCES #837): `\lx@field@stamp{<field>}{<tokens>}` — the
318  // tokens of an argument, each stamped with its position in field <field> as `\lx@field@mark` stamps a store's
319  // value, the positions following the field's earlier stamps (`lx_field_base_<id>`): M3's "marked field argument",
320  // for a binding's `\lx@add@authors*` call, whose argument no store holds. Nothing else changes (the extent and the
321  // range of the field's current call are a store's, `\lx@field@mark`'s). An unknown field, a blank argument or one
322  // past the field's positions comes back unstamped. No call site yet: the observer's ledger projects these
323  // positions for the bindings' calls (`record_author_call`), and stage S2b stamps them.
324  DefMacro!("\\lx@field@stamp{}{}", sub[(field, tokens)] {
325    let Some(field_id) = field_id(field.to_string().trim()) else {
326      return Ok(tokens);
327    };
328    if tokens.unlist_ref().iter().all(|t| t.get_catcode() == Catcode::SPACE) {
329      return Ok(tokens);
330    }
331    let len = tokens.unlist_ref().len();
332    let base = usize::try_from(lookup_int(&s!("lx_field_base_{field_id}"))).unwrap_or(0);
333    if base + len > usize::from(FIELD_PROV_POS_MASK) {
334      return Ok(tokens);
335    }
336    assign_value(&s!("lx_field_base_{field_id}"), (base + len) as i64, Some(Scope::Global));
337    document::note_field_prov_stamped();
338    Ok(field_stamped(field_id, base, tokens.unlist_ref()))
339  });

954/// An address's formula as TeX prints it where the formula is glyphs: a pair of math shifts — read as characters (a
955/// Semiverbatim argument made `$` other: the API's `\lx@add@url`/`\lx@add@email`, Perl's `neutralize`) or as shifts —
956/// round a run of characters, of character commands ([`text_glyph`]: `\_`, `\{`) and of math symbols that each print
957/// one glyph ([`math_symbol_glyph`]: `\sim`, `\cdot`) is the run, each command its glyph, the shifts and spaces dropped as math drops spaces:
958/// `http://www.zebra.edu/$\sim$ann` prints "http://www.zebra.edu/∼ann", as pdflatex, where the dollars printed
959/// (1301.1466's `\ead[label=u2,url]{http://www.math.duke.edu/$\sim$omer}`, math0608476's
960/// `www.math.wisc.edu/$\sim$swanson`). The glyph is a character of the address, in its font (TeX takes it from the
961/// math font), so the address stays one text and its link whole. The shifts pair left to right, as TeX pairs them: a
962/// pair round a script (`^`, `_`: `$^*$x@y`'s mark), a command of another kind or nothing (`$$`) stays as it is,
963/// whole, and the next pair opens after its closing shift — `$^a$a@x, $^b$b@y` keeps both marks and the text between
964/// them (2508.20055's `ej$_$dresselhaus@berkeley.edu, s$_$g@berkeley.edu`). A pair round spacing alone
965/// ([`spacing_formulas_as_space`]: `$~\quad$`, `$\,$`) is glue, no glyph: a space between printed text, none at an
966/// edge (2205.14113's `$~\quad~~$`).
967pub(in crate::base_utilities) fn math_glyphs_as_printed(address: &Tokens) -> Tokens {
968  let address = spacing_formulas_as_space(address.clone());
969  let v = address.unlist_ref();
970  let shift = |t: &Token| *t == T_OTHER!("$") || t.get_catcode() == Catcode::MATH;
971  let script = |t: &Token| [T_OTHER!("^"), T_OTHER!("_")].contains(t);
972  let glyphs = |t: &Token| -> Option<Vec<Token>> {
973    match t.get_catcode() {
974      Catcode::SPACE => Some(Vec::new()),
975      Catcode::LETTER | Catcode::OTHER if !shift(t) && !script(t) => Some(vec![*t]),
976      Catcode::CS => text_glyph(t)
977        .map(|glyph| vec![glyph])
978        .or_else(|| math_symbol_glyph(t).map(|glyph| text_tokens(&glyph))),
979      _ => None,
980    }
981  };
982  let mut out: Vec<Token> = Vec::with_capacity(v.len());
983  let mut at = 0;
984  while at < v.len() {
985    // the shifts pair left to right, as TeX pairs them: a pair that is not glyphs (a script, the empty `$$`) stays as
986    // read, whole, and the next pair starts after its closing shift (2508.20055's `ej$_$d@x, s$_$g@y`)
987    if shift(&v[at])
988      && let Some(len) = v[at + 1..].iter().position(shift)
989    {
990      let pair = &v[at..at + len + 2];
991      match pair[1..=len]
992        .iter()
993        .map(glyphs)
994        .collect::<Option<Vec<Vec<Token>>>>()
995      {
996        Some(printed) if printed.iter().any(|glyph| !glyph.is_empty()) => {
997          out.extend(printed.into_iter().flatten())
998        },
999        _ => out.extend_from_slice(pair),
1000      }
1001      at += len + 2;
1002      continue;
1003    }
1004    out.push(v[at]);
1005    at += 1;
1006  }
1007  Tokens::new(out)
1008}
