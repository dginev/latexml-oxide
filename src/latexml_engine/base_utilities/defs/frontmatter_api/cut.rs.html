11/// The frontmatter API's Cut (FRONTMATTER_SINGLE_ENGINE_PLAN "The target model", placement step 0: "one item per
12/// printed entry; marked lists cut at their marks"), for every binding: an affiliation, alternate affiliation or address
13/// contact (`\lx@annotate@frontmatter{ltx:creator}{ltx:contact}[role=…]{…}`, which `\lx@add@affiliation`,
14/// `\lx@add@address`, `\lx@add@altaffiliation` and `\lx@add@contact` queue through) whose content holds two or more
15/// entries each led by a printed mark that a queued name requests is queued as one contact per entry, each labelled by
16/// its mark (`\lx@affiliation@withsup`), so the names showing a mark take its entry (step 1) and an entry no name
17/// requests lands once at the block (step 4, OD #159). The calls, or `None` for a contact to queue as it is: any other
18/// role; a call whose binding placed it (`label=`, `_unsplit`); a list already cut (its content `\lx@affiliation@withsup`);
19/// content no mark leads, or holding fewer than two requested marks (`Joint Institute, \textsuperscript{1}Univ A and
20/// \textsuperscript{2}Univ B` is one entry; `$^1$Univ A` one).
21///
22/// The cutter is the author blocks' ([`affiliation_calls`]: lines at `\\`, `\quad` and the `\and` family, a line at each
23/// space-preceded mark, a wrapper repeated on each piece, an address line placed as the list's), and every spelling of
24/// a mark is read by the shared readers first: lines set in one-argument line wrappers, a one-column tabular's rows and
25/// commands set side by side whose arguments marks lead (`relocate::wrapped_lines_as_lines`: 2606.28962, 2006.16977;
26/// OD #964), a superscript on a spacing nucleus, in `\( … \)` or in a box of its own
27/// (`relocate::spaced_marks_as_marks`, `spaced_mark_at`: `${}^{a}$`, `${\,\!}^1$`, `\mbox{${\ }^{1}$}`), and a class's
28/// `\inst{n}` (aa.cls:759 `\def\inst#1{\unskip\hbox{$^{#1}$}}`, mem.cls:318, llncs) written as the
29/// `\textsuperscript{n}` it prints. Witnesses (census127 cluster 2): acmart's `\institution{$^1$ Tsinghua University,
30/// China; $^2$ Tencent Inc., China}` (2602.11605), `\institution{\textsuperscript{1}… \quad \textsuperscript{2}…}`
31/// (2506.18881, cidr's 2408.03013), a raw class's `\affil`/`\institute` store (asp2006's 1002.4968, mem's
32/// astro-ph0701509: 40 authors each given all 16 institutes), aa's sole `\institute{\inst{1}…\\ \inst{2}…\\
33/// \inst{*}\email{…}}` (2410.00577). OXIDIZED_DESIGN_DIVERGENCES #954.
34pub(in crate::base_utilities) fn cut_marked_contact(
35  parenttag: &Tokens,
36  tag: &Tokens,
37  kv: Option<&KeyVals>,
38  content: &Tokens,
39) -> Result<Option<Tokens>> {
40  if tag.to_string() != "ltx:contact" {
41    return Ok(None);
42  }
43  let Some(kv) = kv else {
44    return Ok(None);
45  };
46  let role = kv
47    .get_value("role")
48    .map(ToString::to_string)
49    .unwrap_or_default();
50  if !CUT_ROLES.contains(&role.as_str())
51    || kv.get_value("label").is_some()
52    || kv.get_value("_unsplit").is_some()
53  {
54    return Ok(None);
55  }
56  cut_list(parenttag, tag, &role, kv.revert()?, content, &API_CUT)
57}

63/// Whether a cut `entry` is marked (`\lx@affiliation@withsup{…}` whole), and `attr` with `_markshown={shown}` where it is
64/// — but not on an address line, whose address becomes the e-mail contact of the authors showing its mark, which leaves the
65/// mark no entry (aa's `\inst{*}\email{…}`, 2410.00577).
66pub(super) fn mark_shown_attr(
67  attr: Option<Tokens>,
68  entry: &Tokens,
69  shown: &str,
70) -> (bool, Option<Tokens>) {
71  let v = entry.unlist_ref();
72  let marked = v.len() >= 3
73    && v[0] == T_CS!("\\lx@affiliation@withsup")
74    && v[1].get_catcode() == Catcode::BEGIN
75    && skip_group(v, 1) == v.len();
76  let address = || {
77    leading_line_label(&Tokens::new(v[2..v.len() - 1].to_vec()))
78      .is_some_and(|(_, _, rest)| line_is_email_list(&rest))
79  };
80  if !marked || address() {
81    return (marked, attr);
82  }
83  (marked, Some(with_key(attr, s!("_markshown={shown}"))))
84}

86/// `attr` with `key` after its own keys.
87pub(super) fn with_key(attr: Option<Tokens>, key: String) -> Tokens {
88  let mut keys = attr.map(Tokens::unlist).unwrap_or_default();
89  if !keys.is_empty() {
90    keys.push(T_OTHER!(","));
91  }
92  keys.extend(mouth::tokenize_internal(TeXString::assembled(key)).unlist());
93  Tokens::new(keys)
94}

110/// The queued copies of the printed list `key` (of a contact of `role` annotating `parenttag`, its entries' labels the
111/// names request `requested`, [`requested_entry_labels`]) that the Cut left uncut — queued while too few names were
112/// queued to request two of its marks
113/// (amsart's `\author{Ann Able${}^{1}$}\address{${}^{1}$Univ A\\ ${}^{2}$Univ B}\author{Bob Baker${}^{2}$}` and the same
114/// `\address` again: review rev99a2 d2; 2312.02779's REVTeX `\affiliation` after each name) — give way to its entries:
115/// each is taken out of the queue when every name it reaches requests one of the list's marks, so each of them takes
116/// its entry by its mark instead of the whole list; an entry stands in its place that records it for those names, so
117/// the entry each takes keeps the copy's place among its contacts (user ruling R8, print order:
118/// `relocate::at_queue_position`). The names a copy reaches are its run's — those queued after the last contact of its
119/// role before it, which covers one placed on the name before it and a run's (`annotate=run`, `new`) — or every name
120/// queued before it for one annotating them all (`annotate=all`, a number); a copy reaching a name that requests none of
121/// the list's marks, or no name, stays as it is (no name loses the list it prints).
122pub(super) fn place_uncut_copies(
123  parenttag: &Tokens,
124  role: &str,
125  key: &str,
126  requested: &[(String, Vec<usize>)],
127) -> Result<()> {
128  // (the authors requesting one of the list's marks, as the Cut's decision reads them)
129  let requesters: Vec<usize> = requested
130    .iter()
131    .flat_map(|(_, at)| at.iter().copied())
132    .collect();
133  let queue: Vec<RawFrontmatter> = with_value("frontmatter_raw", |v| match v {
134    Some(Stored::FrontmatterRaw(queue)) => queue.clone(),
135    _ => Vec::new(),
136  });
137  let mut removed: Vec<Tokens> = Vec::new();
138  let mut places: Vec<(usize, Tokens)> = Vec::new();
139  {
140    let mut authors: Vec<(usize, usize)> = Vec::new();
141    let mut last_of_role: Option<usize> = None;
142    for (at, (tag, attr, command)) in queue.iter().enumerate() {
143      if tag == "ltx:creator" && attr.get("role").map(String::as_str) == Some("author") {
144        authors.push((at, authors.len() + 1));
145        continue;
146      }
147      if tag != "ltx:contact" || attr.get("role").map(String::as_str) != Some(role) {
148        continue;
149      }
150      let since = last_of_role.replace(at);
151      let v = command.unlist_ref();
152      if attr.contains_key("label")
153        || attr.contains_key("_unsplit")
154        || attr.contains_key(CUT_LIST)
155        || v.first() != Some(&T_CS!("\\lx@annotate@frontmatter@now"))
156        || !annotates(v, parenttag)
157      {
158        continue;
159      }
160      let (_, copy) = within_inst(&last_group(command));
161      if cut_list_key(parenttag, role, &printed_marks_as_marks(copy)) != key {
162        continue;
163      }
164      let all = attr
165        .get("annotate")
166        .is_some_and(|a| a == "all" || a.parse::<usize>().is_ok());
167      let reach: Vec<&(usize, usize)> = authors
168        .iter()
169        .filter(|(position, _)| all || since.is_none_or(|since| *position > since))
170        .collect();
171      if reach.is_empty() || !reach.iter().all(|(_, n)| requesters.contains(n)) {
172        continue;
173      }
174      let names: Vec<String> = reach.iter().map(|(_, n)| n.to_string()).collect();
175      DebugFeature!(
176        "frontmatter",
177        "FRONT cut {role}: an uncut copy of the list gives way to it, for authors {}",
178        names.join(",")
179      );
180      removed.push(last_group(command));
181      let mut place = vec![T_CS!("\\lx@cut@copy@place"), T_BEGIN!()];
182      place.extend(text_tokens(key));
183      place.extend([T_END!(), T_BEGIN!()]);
184      place.extend(text_tokens(&names.join(",")));
185      place.extend([T_END!(), T_BEGIN!()]);
186      place.extend(text_tokens(attr.get("_zone").map_or("", String::as_str)));
187      place.push(T_END!());
188      places.push((at, Tokens::new(place)));
189    }
190  }
191  if places.is_empty() {
192    return Ok(());
193  }
194  // (each copy's queue entry, the same one: nothing was queued since it was read)
195  with_value_mut("frontmatter_raw", |val_opt| {
196    if let Some(&mut Stored::FrontmatterRaw(ref mut queue)) = val_opt {
197      for (at, place) in places {
198        if let Some(entry) = queue.get_mut(at) {
199          *entry = (CUT_COPY_TAG.to_string(), TagAttrs::default(), place);
200        }
201      }
202    }
203  });
204  // (the frontmatter observer's ledger: the calls taken out, as a clear's, `ledger::record_clear`)
205  for content in removed {
206    record_cut_copy(role, &content);
207  }
208  Ok(())
209}

211/// Does the queued `\lx@annotate@frontmatter@now{parenttag}…` call `command` annotate `parenttag`?
212fn annotates(command: &[Token], parenttag: &Tokens) -> bool {
213  if command
214    .get(1)
215    .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
216  {
217    return false;
218  }
219  let end = skip_group(command, 1);
220  end > 2 && command[2..end - 1] == *parenttag.unlist_ref()
221}

223/// `entry` (an entry of the Cut, `\lx@affiliation@withsup{…}` or the line itself) with the list's opening declarations
224/// at its head, where it does not open with them already (the first entry).
225pub(super) fn with_opening(entry: Tokens, opening: &[Token]) -> Tokens {
226  if opening.is_empty() {
227    return entry;
228  }
229  let v = entry.unlist_ref();
230  let wrapped = v.len() >= 3
231    && v[0] == T_CS!("\\lx@affiliation@withsup")
232    && v[1].get_catcode() == Catcode::BEGIN
233    && skip_group(v, 1) == v.len();
234  let inner = if wrapped { &v[2..v.len() - 1] } else { &v[..] };
235  let lead = inner
236    .iter()
237    .position(|t| *t != T_SPACE!())
238    .unwrap_or(inner.len());
239  if inner[lead..].starts_with(opening) {
240    return entry;
241  }
242  let mut out: Vec<Token> = Vec::with_capacity(v.len() + opening.len() + 1);
243  if wrapped {
244    out.extend_from_slice(&v[..2]);
245  }
246  out.extend_from_slice(opening);
247  out.extend_from_slice(&inner[lead..]);
248  if wrapped {
249    out.push(v[v.len() - 1]);
250  }
251  Tokens::new(out)
252}

254/// `entry` (an entry of the Cut, `\lx@affiliation@withsup{…}` or the line itself) with the parts every entry of the list
255/// shares at its end ([`shared_closing_parts`]), inside the mark's wrapper as [`with_opening`] puts the declarations.
256pub(super) fn with_closing(entry: Tokens, closing: &[Token]) -> Tokens {
257  if closing.is_empty() {
258    return entry;
259  }
260  let mut v = entry.unlist();
261  let wrapped = v.len() >= 3
262    && v[0] == T_CS!("\\lx@affiliation@withsup")
263    && v[1].get_catcode() == Catcode::BEGIN
264    && skip_group(&v, 1) == v.len();
265  // (an entry that prints nothing, the empty piece before a list's first mark, `\institution{ $^1$…}`, stays as it is)
266  let inner = if wrapped { &v[2..v.len() - 1] } else { &v[..] };
267  if !inner
268    .iter()
269    .any(|t| matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER) || is_author_mark(t))
270  {
271    return Tokens::new(v);
272  }
273  let close = if wrapped { v.pop() } else { None };
274  v.extend_from_slice(closing);
275  v.extend(close);
276  Tokens::new(v)
277}

279/// `list` led by a wrapper holding its marked entries and followed by parts of its own (`\institution{$^1$A \quad
280/// $^2$B}\country{X}`, acmart's address parts; pdflatex prints "X" on a line under both entries): the wrapper alone, the
281/// parts every entry shares (an address part), and the parts the block holds once (one holding a link or an address,
282/// `\country{Project page: \href{…}{…}}`, 2506.18881: no author's address); a part whose argument is spacing alone (a
283/// space, a tie, `\relax`, a spacing command) is none of them.
284/// Each part is a command and its argument, none a mark; any other list is returned as it is.
285pub(super) fn shared_closing_parts(list: Tokens) -> (Tokens, Vec<Token>, Vec<Tokens>) {
286  let v = list.unlist_ref();
287  let start = v.iter().position(|t| *t != T_SPACE!()).unwrap_or(v.len());
288  let wrapper = v.get(start).is_some_and(|t| {
289    t.get_catcode() == Catcode::CS && !is_mark_command(t) && !cs_in(t, &LINE_BOXES)
290  }) && v
291    .get(start + 1)
292    .is_some_and(|t| t.get_catcode() == Catcode::BEGIN);
293  if !wrapper {
294    return (list, Vec::new(), Vec::new());
295  }
296  let mut after = skip_group(v, start + 1);
297  // (the wrappers set side by side, one line each, `\institution{$^1$A}\\ \institution{$^2$B}` as
298  // `relocate::wrapped_lines_as_lines` reads them, hold the entries alike: 2006.16977)
299  loop {
300    let next = (after..v.len())
301      .find(|&i| v[i] != T_SPACE!())
302      .unwrap_or(v.len());
303    let line = (next + 1..v.len())
304      .find(|&i| v[i] != T_SPACE!())
305      .unwrap_or(v.len());
306    if v.get(next) == Some(&T_CS!("\\\\"))
307      && v.get(line) == v.get(start)
308      && v
309        .get(line + 1)
310        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
311    {
312      after = skip_group(v, line + 1);
313    } else {
314      break;
315    }
316  }
317  let mut parts: Vec<Vec<Token>> = Vec::new();
318  let mut i = after;
319  while i < v.len() {
320    if v[i] == T_SPACE!() {
321      i += 1;
322      continue;
323    }
324    if v[i].get_catcode() != Catcode::CS
325      || is_mark_command(&v[i])
326      || v
327        .get(i + 1)
328        .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
329    {
330      return (list, Vec::new(), Vec::new());
331    }
332    let end = skip_group(v, i + 1);
333    let part = &v[i..end];
334    if part.iter().any(is_author_mark) {
335      return (list, Vec::new(), Vec::new());
336    }
337    parts.push(part.to_vec());
338    i = end;
339  }
340  if parts.is_empty() {
341    return (list, Vec::new(), Vec::new());
342  }
343  let links = |part: &[Token]| {
344    part
345      .iter()
346      .any(|t| *t == T_OTHER!("@") || cs_in(t, &["\\href", "\\url", "\\email", "\\mailto"]))
347  };
348  // (a part whose argument is spacing alone, `\country{~}` written to satisfy the class, is no entry's: repeated, or
349  // left on the last entry, it printed ", ," there — 2605.18872; any other command in it may print, a macro's text
350  // `\city{\cty}`, so the part is shared: review rev99a2 S-a, probe s3d)
351  let inert = |part: &[Token]| {
352    part[1..].iter().all(|t| {
353      matches!(t.get_catcode(), Catcode::BEGIN | Catcode::END)
354        || *t == T_CS!("\\relax")
355        || is_spacing(t)
356        || is_spacing_command(t)
357    })
358  };
359  let (mut shared, mut block) = (Vec::new(), Vec::new());
360  for part in parts {
361    if inert(&part) {
362      continue;
363    } else if links(&part) {
364      block.push(Tokens::new(part));
365    } else {
366      shared.extend(part);
367    }
368  }
369  (Tokens::new(v[..after].to_vec()), shared, block)
370}

372/// The `\lx@annotate@frontmatter{parenttag}{tag}[attr]{entry}` call of an entry of the Cut ([`cut_marked_contact`]), in
373/// the affiliation's own wrapper when the list was (`withinst`).
374pub(super) fn annotate_call(
375  parenttag: &Tokens,
376  tag: &Tokens,
377  attr: Option<Tokens>,
378  entry: Tokens,
379  withinst: bool,
380) -> Result<Vec<Token>> {
381  let entry = if withinst {
382    Invocation!(T_CS!("\\lx@affiliation@withinst"), vec![Some(entry)])
383  } else {
384    entry
385  };
386  let mut call = vec![T_CS!("\\lx@annotate@frontmatter"), T_BEGIN!()];
387  call.extend(parenttag.unlist_ref().iter().copied());
388  call.extend([T_END!(), T_BEGIN!()]);
389  call.extend(tag.unlist_ref().iter().copied());
390  call.push(T_END!());
391  if let Some(attr) = attr {
392    call.push(T_OTHER!("["));
393    call.extend(attr.unlist());
394    call.push(T_OTHER!("]"));
395  }
396  call.push(T_BEGIN!());
397  call.extend(entry.unlist());
398  call.push(T_END!());
399  Ok(call)
