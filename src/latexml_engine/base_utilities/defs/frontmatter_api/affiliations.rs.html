1//! The frontmatter API's entry cutter: the `\lx@add@affiliation` calls for a printed list (`entry_calls`), address

3//! (`\lx@add@affiliation@marked`/`@shared`) and class-numbered institutes the bindings and `frontmatter_stores` call. Moved
4//! unchanged from `author_block/affiliations.rs` (Z18j): the Cut's own closure, rerouted onto the reading by Option C

9/// The `\lx@add@affiliation` calls for an affiliation list: one per line or `\and`-separated
10/// entry, each with its superscript marker as its label when the list has markers (Perl
11/// Base_Utility.pool.ltxml:742-753, the body of `\lx@add@affiliations` after its dequeue). `attr`
12/// is passed to each entry, `marked_attr` instead when the list has markers and `marks_are_labels` — and there
13/// `unmarked_attr`, when given, to an entry no mark labels (no mark in it, no footnote symbol leading it): llncs's one
14/// `\institute`'s `labelseq=affiliation,_share=authors`, its first line every author's as the class prints it (shape
15/// (c), `\lx@llncs@institute@sup`). `authors_queued` says every author is queued, so an unmarked line may be placed by
16/// the marks: an address run by its authors' positions, another line with the affiliation before it. An `\author`
17/// call's tail (`author_tail_calls`) may come before later authors, and then its unmarked lines are its own creators'
18/// rows, as each was before. A marked entry shows its mark where the list's names show theirs, as the Cut's entries do
19/// (`mark_shown_attr`; llncs's `\institute{$^{1}$Univ A \\ $^{2}$Univ B}`, 1209.0041).
20pub fn affiliation_calls(
21  attr: Option<Tokens>,
22  marked_attr: Option<Tokens>,
23  unmarked_attr: Option<Tokens>,
24  stuff: Tokens,
25  marks_are_labels: bool,
26  authors_queued: bool,
27) -> Result<Vec<Token>> {
28  let add = |attr: Option<Tokens>, entry: Tokens| {
29    let (_, attr) = mark_shown_attr(attr, &entry, MARK_SHOWN_WITH_NAMES);
30    Ok(Invocation!(T_CS!("\\lx@add@affiliation"), vec![attr, Some(entry)]).unlist())
31  };
32  entry_calls(
33    &EntryCall {
34      add:                    &add,
35      labelled_address_lines: false,
36    },
37    attr,
38    marked_attr,
39    unmarked_attr,
40    stuff,
41    marks_are_labels,
42    authors_queued,
43  )
44}

46/// How [`entry_calls`] adds an entry: the call made of its attributes and content, and whether an address line behind
47/// an e-mail label is an address line of the list (the frontmatter API's Cut, [`cut_marked_contact`]) or text of the
48/// entry above it (a binding's own list).
49pub(in crate::base_utilities) struct EntryCall<'a> {
50  pub(in crate::base_utilities) add: &'a dyn Fn(Option<Tokens>, Tokens) -> Result<Vec<Token>>,
51  pub(in crate::base_utilities) labelled_address_lines: bool,
52}

54/// [`affiliation_calls`], each entry's call made by `add` (its attributes, its content) — the frontmatter API's Cut
55/// ([`cut_marked_contact`]) adds an entry with the call that held the list, its role and wrapper kept.
56pub(in crate::base_utilities) fn entry_calls(
57  call: &EntryCall,
58  attr: Option<Tokens>,
59  marked_attr: Option<Tokens>,
60  unmarked_attr: Option<Tokens>,
61  stuff: Tokens,
62  marks_are_labels: bool,
63  authors_queued: bool,
64) -> Result<Vec<Token>> {
65  let EntryCall { add, labelled_address_lines } = *call;
66  let mut calls: Vec<Token> = Vec::new();
67  // Consume `\\[len]` row-break optionals before splitting (KNOWN_PERL_ERRORS #75).
68  // (the lines a list sets in wrappers, a one-column tabular's rows, commands side by side, read as lines, as the
69  // frontmatter API's Cut reads them — `relocate::wrapped_lines_as_lines`: IEEEtran's `\IEEEauthorblockA{\begin{tabular}{c}
70  // $^{1}$ Univ A \\ $^{2}$ Univ B \\ …\end{tabular}}`, 2606.28962)
71  let stuff = wrapped_lines_as_lines(&stuff).unwrap_or(stuff);
72  let stuff = spaced_marks_as_marks(&stuff).unwrap_or(stuff);
73  let stuff = rewrite_ordinal_superscripts(strip_linebreak_options(unalias_mark_commands(stuff)));
74  let stuff = blank_line_marks_dropped(stuff);
75  let stuff = glued_line_marks_as_marks(stuff, true);
76  // (the list's marks as digested, as its gate reads them, `affiliation_is_marked`: a store's `\ma` printing
77  // `\textsuperscript{a}`, revP4d S1; a note's label, 1504.01072), and where its entries start, as the Cut's decision
78  // reads them (`list_marks`: `\ua Univ A` with `\newcommand\ua{\textsuperscript{1}}`, 2608.21425's `\affmark{1}`)
79  let marks = marks_are_labels
80    .then(|| list_marks(&stuff, "affiliation_calls"))
81    .transpose()?;
82  let with_sup = marks.as_ref().is_some_and(|marks| marks.prints);
83  let (lines, line_starts) = lines_starts(
84    &stuff,
85    affiliation_lines(&stuff)?,
86    marks.as_ref().filter(|_| with_sup),
87  );
88  // In a marked list, the affiliations it names, each a mark-led line (or a piece of one) with the unmarked lines
89  // that follow it.
90  let mut marked_entries: Vec<Vec<Token>> = Vec::new();
91  // (whether each entry opens with a mark, as its first line's starts say; unread, `None`)
92  let mut entry_opens: Vec<Option<bool>> = Vec::new();
93  // The runs of address lines, placed once the whole list is read (`place_address_runs`); whether a run is being
94  // read, and whether the line just read was an entry or its continuation.
95  let mut address_runs: Vec<AddressRun> = Vec::new();
96  let mut in_run = false;
97  let mut follows_entry = false;
98  // (the notes of the list's address lines, the document's: [`address_line_notes`])
99  let mut address_notes: Vec<Token> = Vec::new();
100  for ((delimiter, line), starts) in lines.into_iter().zip(line_starts) {
101    // Skip empty segments (e.g. a trailing \\ or a line that was wholly
102    // consumed by an \email/\url) so they don't become blank affiliations.
103    // Mirrors \lx@add@authors. So is a line that prints nothing ([`prints_nothing`]: IEEEtran's
104    // `\end{@IEEEauthorhalign}\hfill\strut`, 2309.15562, an empty affiliation of the last author).
105    if line.is_empty() || prints_nothing(&line) {
106      continue;
107    }
108    if with_sup {
109      let across_and = parts_entries(&delimiter);
110      // (in a list the frontmatter API cuts, an address line behind an e-mail label is one too, its addresses the run's:
111      // `email: agupta@…, orion@…` closing an aa `\institute`, astro-ph0405186 — its label the contacts' own name,
112      // `email_line_calls`; a binding's own list keeps such a line in the entry above it, which the observer's writer
113      // reads: IEEEtran's `\IEEEauthorblockA{$^\dag$…\\ … \\ E-mail: …}`, 2308.05546)
114      let is_address_line = |line: &Tokens| {
115        if labelled_address_lines {
116          line_addresses(line).is_some()
117        } else {
118          line_is_email_list(line)
119        }
120      };
121      // (a line no entry starts in: `q2`'s `\affmark{1} Univ A\\ Street A`, the street; unread, no mark in its tokens)
122      let unmarked = line_unmarked(&line, starts.as_ref());
123      // (…and one carrying notes, its notes the document's, after the list's calls: [`address_line_notes`],
124      // `\institute{$^{1}$Univ X \\ $^{2}$Univ Y \\ \texttt{a@x, b@y}\thanks{…}}`, as 2605.23553's author block)
125      let line = match (authors_queued && unmarked)
126        .then(|| address_line_notes(&line, |rest| Ok(is_address_line(rest))))
127        .transpose()?
128        .flatten()
129      {
130        Some((rest, notes)) => {
131          address_notes.extend(notes);
132          rest
133        },
134        None => line,
135      };
136      let address_line = is_address_line(&line);
137      if authors_queued && unmarked && address_line {
138        // (`\and` separates entries, and runs: `Univ A \\ a@x \and b@y` is two)
139        if !in_run || across_and {
140          let under = (follows_entry && !across_and).then(|| {
141            let entry = marked_entries.len() - 1;
142            (entry, marked_entries[entry].len())
143          });
144          address_runs.push(AddressRun { lines: Vec::new(), under });
145          in_run = true;
146        }
147        if let Some(run) = address_runs.last_mut() {
148          run.lines.push((delimiter, line));
149        }
150        follows_entry = false;
151        continue;
152      }
153      in_run = false;
154      // An unmarked line continues the affiliation before it, with its `\\` put back, as an unmarked line continues
155      // an entry in `\lx@add@authors`: `$^1$Department\\University` is one affiliation; on its own, an unlabelled
156      // line was placed by position, not by mark (2609.19448; repro
157      // sectioning-frontmatter/marked_affiliation_continuation_lines_stay_with_it). Not across `\and`, which
158      // separates entries there too (DIVERGENCES #52(g)), nor a footnote-symbol legend (`* Equal contribution`,
159      // `$\dagger$ Corresponding author`), which belongs to the whole block (llncs `\institute`, 2609.06094).
160      if authors_queued
161        && unmarked
162        && !across_and
163        && !starts_with_footnote_symbol(&line)
164        && let Some(last) = marked_entries.last_mut()
165      {
166        last.extend(delimiter);
167        last.extend(line.unlist());
168        follows_entry = true;
169        continue;
170      }
171      // The superscript markers ARE the affiliation labels here, so drop any
172      // caller-supplied labelseq: applying both double-labels each affiliation
173      // and duplicates it onto every \inst{n} author. Mirrors \lx@add@authors,
174      // which likewise passes no attr in its with-superscript branch.
175      let segs = entry_pieces(line, starts.as_ref())?;
176      for (seg, opens) in segs {
177        if spacing_end(seg.unlist_ref()) == 0 {
178          continue;
179        }
180        marked_entries.push(seg.unlist());
181        entry_opens.push(opens);
182        follows_entry = true;
183      }
184    } else {
185      calls.extend(add(attr.clone(), line)?);
186    }
187  }
188  let email_calls = place_address_runs(address_runs, &mut marked_entries, 0)?;
189  // The separator between two affiliations (`$^1$Univ A, $^2$Univ B`) is trimmed from each entry but the last once
190  // its address runs are in place: an entry is whole only then, and a run spliced at the length the entry had when the
191  // run was read (`AddressRun::under`) would otherwise land past the end of an entry trimmed before it
192  // (`\IEEEauthorrefmark{2}…Nankai University,\\ \{…\}@mail.nankai.edu.cn;\\ \IEEEauthorrefmark{3}…`, 2509.13772: a
193  // panic, "range end index 44 out of range for slice of length 43"; repro
194  // sectioning-frontmatter/affiliation_address_run_after_a_trailing_comma).
195  let last = marked_entries.len().saturating_sub(1);
196  for entry in &mut marked_entries[..last] {
197    *entry = trim_trailing_separator(std::mem::take(entry));
198  }
199  for (entry, opens) in marked_entries.into_iter().zip(entry_opens) {
200    // (an entry no mark labels, in a list whose caller says how such an entry is placed: as that call places it, not
201    // labelled by a mark — llncs's `\institute{Univ A, Town \\ \textsuperscript{*}Equal contribution}`, whose `Univ A,
202    // Town` went to the last author alone, every author's in pdflatex: dev20k 2603.18089, 2608.22772, 2207.13038,
203    // 2609.31726, 1905.07852)
204    let entry = Tokens::new(entry);
205    if let Some(unmarked) = &unmarked_attr
206      && !opens.unwrap_or_else(|| author_mark_position(&entry).is_some())
207      && !starts_with_footnote_symbol(&entry)
208    {
209      calls.extend(add(Some(unmarked.clone()), entry)?);
210      continue;
211    }
212    let withsup = Invocation!(T_CS!("\\lx@affiliation@withsup"), vec![Some(entry)]);
213    calls.extend(add(marked_attr.clone(), withsup)?);
214  }
215  calls.extend(email_calls);
216  calls.extend(address_notes);
217  Ok(calls)
218}

220/// `stuff` without the marks opening a line that print no label — a superscript of spacing alone (`$^{\;}$`, `$^{~}$`,
221/// `\textsuperscript{\,}`), braces, spaces and `\relax` aside: such a line is unmarked, so it continues the entry above
222/// it as any unmarked line does, not an entry of its own (mnras 2303.08793's `$^{8}$Institute for Quantum Computing …,
223/// Waterloo, \\ $^{\;}$Ontario N2L 3G1, Canada`, which pdflatex prints under entry 8 with no label; the line had gone to
224/// the last name). Read by every list's cutter (and, until DEL2, by the old author block reader `dispatch::add_authors_calls_sectioned`)
225/// ([`entry_calls`]). A blank mark elsewhere is left to the mark readers (`email_lines::drop_empty_marks`).
226pub(in crate::base_utilities) fn blank_line_marks_dropped(stuff: Tokens) -> Tokens {
227  let v = stuff.unlist_ref();
228  let blank = |range: &[Token]| {
229    range.iter().all(|t| {
230      matches!(
231        t.get_catcode(),
232        Catcode::BEGIN | Catcode::END | Catcode::SPACE
233      ) || *t == T_CS!("\\relax")
234        || is_spacing(t)
235    })
236  };
237  // (the mark at `i`, a line's first token: where it ends when it prints no label)
238  let blank_mark_end = |i: usize| -> Option<usize> {
239    let (operand, closing) =
240      if v[i].get_catcode() == Catcode::MATH && v.get(i + 1) == Some(&T_SUPER!()) {
241        (i + 2, true)
242      } else if is_mark_command(&v[i]) {
243        (
244          i + 1 + v[i + 1..].iter().take_while(|t| **t == T_SPACE!()).count(),
245          false,
246        )
247      } else {
248        return None;
249      };
250    if v.get(operand)?.get_catcode() != Catcode::BEGIN {
251      return None;
252    }
253    let close = skip_group(v, operand);
254    if close > v.len() || !blank(&v[operand..close]) {
255      return None;
256    }
257    if closing {
258      (v.get(close)?.get_catcode() == Catcode::MATH).then_some(close + 1)
259    } else {
260      Some(close)
261    }
262  };
263  let mut out: Vec<Token> = Vec::with_capacity(v.len());
264  let mut line_start = true;
265  let mut i = 0;
266  let mut dropped = false;
267  while i < v.len() {
268    let t = v[i];
269    if line_start && t == T_SPACE!() {
270      out.push(t);
271      i += 1;
272      continue;
273    }
274    if line_start && let Some(end) = blank_mark_end(i) {
275      i = end;
276      dropped = true;
277      line_start = false;
278      continue;
279    }
280    line_start = t == T_CS!("\\\\") || t == T_CS!("\\newline");
281    out.push(t);
282    i += 1;
283  }
284  if dropped { Tokens::new(out) } else { stuff }
285}

287/// The key of a printed list the Cut reads (`list`, its marks written as the plain marks they print,
288/// [`printed_marks_as_marks`]) of a contact of `role` annotating `parenttag`: two calls holding the same list have the
289/// same key.
290pub(in crate::base_utilities) fn cut_list_key(
291  parenttag: &Tokens,
292  role: &str,
293  list: &Tokens,
294) -> String {
295  use std::hash::{Hash, Hasher};
296  let mut hasher = std::hash::DefaultHasher::new();
297  (parenttag.to_string(), role, list.to_string()).hash(&mut hasher);
298  s!("{:016x}", hasher.finish())
299}

301/// `list` with the marks spelled otherwise written as the plain marks they print, for the Cut's readers
302/// ([`cut_marked_contact`]): lines in one-argument line wrappers as lines (`relocate::wrapped_lines_as_lines`), a
303/// superscript spelled otherwise as `$^{…}$` (`relocate::spaced_marks_as_marks`), a class's `\inst{n}` as
304/// `\textsuperscript{n}`.
305pub(in crate::base_utilities) fn printed_marks_as_marks(list: Tokens) -> Tokens {
306  // (a list wholly in one group, `{\bfseries $^3$Lab C\\ $^4$Lab D}`, is its content: the contact is a group of its own)
307  let v = list.unlist_ref();
308  let start = v.iter().position(|t| *t != T_SPACE!()).unwrap_or(v.len());
309  let end = v
310    .iter()
311    .rposition(|t| *t != T_SPACE!())
312    .map_or(0, |i| i + 1);
313  let list = if start < end
314    && v[start].get_catcode() == Catcode::BEGIN
315    && skip_group(v, start) == end
316    && split_top_level_lines(&v[start + 1..end - 1]).len() > 1
317  {
318    Tokens::new(v[start + 1..end - 1].to_vec())
319  } else {
320    list
321  };
322  let list = wrapped_lines_as_lines(&list).unwrap_or(list);
323  let list = spaced_marks_written(&list, true).unwrap_or(list);
324  // (`\inst` by its meaning, the one-argument mark it prints: `line_marks::inst_prints_its_mark`)
325  if !list.unlist_ref().contains(&T_CS!("\\inst")) || !inst_prints_its_mark() {
326    return list;
327  }
328  Tokens::new(
329    list
330      .unlist()
331      .into_iter()
332      .map(|t| {
333        if t == T_CS!("\\inst") {
334          T_CS!("\\textsuperscript")
335        } else {
336          t
337        }
338      })
339      .collect(),
340  )
341}

343/// The frontmatter API's Cut decision (`cut_marked_contact`), the one predicate every reader of a store's marked
344/// entries asks: does `list` hold two or more entries led by marks of distinct labels that queued names request, its
345/// first ink a mark — its marks as digested ([`requested_entry_labels`]), read as the Cut reads them
346/// (`printed_marks_as_marks`)?
347pub fn holds_requested_marked_entries(list: &Tokens) -> Result<bool> {
348  let list = printed_marks_as_marks(list.clone());
349  Ok(
350    requested_entry_labels(&list, "holds_requested_marked_entries")?
351      .labels
352      .len()
353      >= 2,
354  )
355}

357/// A class-numbered institute list as its title page prints it: llncs.cls:870-885 `\institutename` (copied by
358/// preprint.cls:233-256, 2405.16615) redefines `\and` to step the institute counter and print `$^{\theinst}$` before the
359/// next entry, the first numbered 1 — so the n-th `\and` entry of the `\@institute` store is affiliation n, which the
360/// authors' numeric marks pick (`$^{1}$`, `$^2$`). `stuff` with each entry led by its number as a superscript, when the
361/// class numbers the entries of its `store` (named by its control sequence, `\@institute`), the store holds at least two
362/// `\and` entries and no mark of its own, both read by meaning (the kernel's `\and` a macro holds, a mark a macro prints:
363/// [`split_at_kernel_separators`]; a mark a name requests, [`prints_requested_marks`]), and the queued authors' marks request numbers
364/// ([`numbered_marks_requested`]); else `None` (the entries annotate every author, `\lx@add@store@affiliations`).
365/// Whether the class numbers them is what its own title page printed in the speculative run frontmatter_stores.rs
366/// `note_and_numbering` made of the store (`lx_and_numbered_<store>`: the sentinel entries each printed after its
367/// number, in `\and` order — sab.cls:404-426's and edpsci.cls's `\refstepcounter{inst}`, 2308.05831, 2308.05830,
368/// 2501.08874, baaa.cls 2206.02830, comsis2.cls 2207.09249; batch 104b); with no such run (no store, none could be
369/// made), the class's `\institutename` read as before: a `\def\and` stepping a counter (`\stepcounter`). Was each
370/// author given every institute (2405.16615; SYNC_STATUS row 153; the sab family, SYNC_STATUS row "104b").
371pub fn class_numbered_institutes(stuff: &Tokens, store: Option<&str>) -> Result<Option<Tokens>> {
372  let run = store.and_then(|store| match lookup_value(&s!("lx_and_numbered_{store}")) {
373    Some(Stored::Bool(numbered)) => Some(numbered),
374    _ => None,
375  });
376  let numbers_its_entries = run.unwrap_or_else(|| {
377    matches!(lookup_definition_stored(&T_CS!("\\institutename")),
378    Ok(Some(Stored::Expandable(definition)))
379      if matches!(definition.get_expansion(), Some(ExpansionBody::Tokens(body))
380        if body.unlist_ref().windows(2).any(|w| w[0] == T_CS!("\\def") && w[1] == T_CS!("\\and"))
381          && body.unlist_ref().contains(&T_CS!("\\stepcounter"))))
382  });
383  if !numbers_its_entries
384    || prints_requested_marks(stuff, "class_numbered_institutes")?
385    || !numbered_marks_requested()?
386  {
387    return Ok(None);
388  }
389  let (entries, _) = split_at_kernel_separators(stuff.clone(), vec![T_CS!("\\and").into()])?;
390  if entries.len() < 2 {
391    return Ok(None);
392  }
393  let mut out: Vec<Token> = Vec::new();
394  for (n, (_, entry)) in entries.into_iter().enumerate() {
395    let and = if n > 0 { "\\and" } else { "" };
396    let lead = s!("{and}\\textsuperscript{{{}}}", n + 1);
397    out.extend(mouth::tokenize_internal(TeXString::assembled(lead)).unlist());
398    out.extend(entry.unlist());
399  }
400  Ok(Some(Tokens::new(out)))
401}

403/// Do the queued authors' marks request numbered entries: each mark a comma list of numbers (`\inst{1,2}`), at least
404/// one, beside which only symbols stand — a legend's `\textsuperscript{\dag}`, `*` (comsis2.cls's names, 2207.09249), an
405/// empty piece (edpsci's `\inst{1,2}$^{,}$`, 2501.08874), a command (a `\ref{…}` key, an unread name's spelling) —,
406/// never a letter label (`\inst{a}`), which no number answers? The marks by printed text (`queued_author_marks`).
407pub fn numbered_marks_requested() -> Result<bool> {
408  let mut numbers = false;
409  for mark in queued_author_marks()?.iter().flatten() {
410    for piece in mark.split(',') {
411      let piece = piece.trim();
412      if piece.starts_with('\\') || !piece.chars().any(char::is_alphanumeric) {
413        continue;
414      }
415      if !piece.chars().all(|c| c.is_ascii_digit()) {
416        return Ok(false);
417      }
418      numbers = true;
419    }
420  }
421  Ok(numbers)
422}

424/// Whether an affiliation a class gives after its authors is a marked list: it prints a mark, read as digested

427pub(in crate::base_utilities) fn affiliation_is_marked(stuff: &Tokens) -> Result<bool> {
428  Ok(

431  )

434/// The calls for an unmarked address shared by the authors before it (`\lx@add@affiliation@shared`): each of its
435/// entries (separated by the `\and` family, `\And` the column break of the conference templates) one affiliation,
436/// attributed by `attr` — one for entries printed alike (1302.5348's `\UMDAddress \\ … \And \UMDAddress \\ …`) —
437/// without its e-mail lines, whose addresses, all of them, are one run placed as a marked list's are
438/// ([`place_address_runs`]: one for each author of the row, in their order, else the block's, one contact an address —
439/// the row the authors queued after the last affiliation, as `annotate=run` attaches one: 2302.10910's four e-mails of
440/// a later row go to its four authors, not to the block for not being one for each queued author) — not lines of the
441/// affiliation (1806.02924's `Carnegie Mellon University\\ \{asuggala, adarshp, vaishnavh, pradeepr\}@cs.cmu.edu`,
442/// 1311.1894's `\texttt{\small \{honda,takemura\}@…}`), as the authors' own addresses are in a paired block
443/// (2610.11580).
444pub(in crate::base_utilities) fn shared_affiliation_calls(
445  attr: Option<Tokens>,
446  stuff: Tokens,
447) -> Result<Vec<Token>> {
448  let authors_queued = queued_creator_count() > 0;
449  let mut entries: Vec<Vec<Token>> = Vec::new();
450  let mut run = AddressRun { lines: Vec::new(), under: None };
451  for entry in split_tokens(strip_linebreak_options(stuff), author_and_splits()) {
452    let mut kept: Vec<Token> = Vec::new();
453    for (delimiter, line) in split_tokens_delimited(entry, author_line_breaks()) {
454      if authors_queued && line_is_email_list(&line) {
455        run.lines.push((delimiter, line));
456      } else if spacing_end(line.unlist_ref()) > 0 {
457        if !kept.is_empty() {
458          kept.extend(delimiter);
459        }
460        kept.extend(line.unlist());
461      }
462    }
463    let lead = kept
464      .iter()
465      .take_while(|t| t.get_catcode() == Catcode::SPACE)
466      .count();
467    kept.drain(..lead);
468    while kept
469      .last()
470      .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
471    {
472      kept.pop();
473    }
474    if !kept.is_empty() && !entries.contains(&kept) {
475      entries.push(kept);
476    }
477  }
478  let mut calls: Vec<Token> = Vec::new();
479  for entry in entries {
480    calls.extend(
481      Invocation!(T_CS!("\\lx@add@affiliation"), vec![
482        attr.clone(),
483        Some(Tokens::new(entry))
484      ])
485      .unlist(),
486    );
487  }
488  if !run.lines.is_empty() {
489    calls.extend(place_address_runs(vec![run], &mut [], queued_row_start()?)?);
490  }
491  Ok(calls)
492}

494/// The addresses of an address line of a marked list: a line of addresses only (`email_addresses`), or one behind an
495/// e-mail label (`email: a@x, b@y`: `split_email_label`), whose label `email_line_calls` keeps as the contacts' name.
496/// Read on the tokens the e-mail calls are cut from, at the binding's call: `resolver::prints_only_addresses` on a reading
497/// says only whether (P6f).
498pub(in crate::base_utilities) fn line_addresses(line: &Tokens) -> Option<Vec<String>> {
499  email_addresses(line).or_else(|| match split_email_label(line) {
500    (Some(_), rest) => email_addresses(&rest),
501    _ => None,
502  })
503}

505/// A run of consecutive address lines in a marked affiliation list, with the entry it follows (when the line before
506/// it was that entry, not across `\and`) and the length that entry had when the run was read: where a continuation
507/// goes.
508struct AddressRun {
509  lines: Vec<(Vec<Token>, Tokens)>,
510  under: Option<(usize, usize)>,
511}
512
513/// The `\lx@add@email` calls for the address runs of a marked affiliation list (OXIDIZED_DESIGN #52(j)); a run that
514/// stays with its affiliation is spliced into that entry. An address claims an author only where the list says
515/// whose it is:
516/// - when the list puts addresses under its affiliations — it has one, or a run sits between two (2609.19448
517///   `$^1$Department\\University\\\texttt{david.abel@ed.ac.uk}\\[2ex]$^2$…`) — one address under an affiliation
518///   exactly one author requests is that author's; any other run under an affiliation goes back into that entry, whose
519///   Kind step takes it out as the entry's e-mails and whose Link step places them (the author each spells, the entry's
520///   only carrier, …, else the block once: [`kind_entry_addresses`], OD #956);
521/// - else the run is the block's: one address for each author goes to the authors in order (llncs
522///   `\institute{$^1$A \\ $^2$B \\ \email{a@x} \\ \email{b@y}}`), any other is one contact an address on the shared
523///   creator below the authors (#159) — by order, 2609.06094's four addresses for six authors had given Jindong
524///   Gu's to Runjia Li, and by the mark of the affiliation they follow, three of them to the wrong author.
525///
526/// `row_start` is the queue position the authors the list belongs to start after (0: every queued author).
527fn place_address_runs(
528  runs: Vec<AddressRun>,
529  entries: &mut [Vec<Token>],
530  row_start: usize,
531) -> Result<Vec<Token>> {
532  let authors = queued_author_positions()?;
533  let row_start = row_start.min(authors);
534  // (a legend line, `* Equal contribution`, is an entry with no mark; nor does a footnote-symbol mark after a run,
535  // `$^\dagger$ Corresponding author`, show the run sits under an affiliation — as an iopart `\address`'s one mark it
536  // still labels one)
537  let marks: Vec<Option<Vec<Token>>> = entries.iter().map(|entry| entry_mark(entry)).collect();
538  let affiliation_follows = |entry: usize| {
539    marks[entry + 1..]
540      .iter()
541      .flatten()
542      .any(|mark| !is_symbol_mark(mark))
543  };
544  let under_affiliations = marks.iter().flatten().count() == 1
545    || runs.iter().any(|run| {
546      run
547        .under
548        .is_some_and(|(entry, _)| affiliation_follows(entry))
549    });
550  let mut placed: Vec<Vec<Token>> = Vec::new();
551  // In reverse, so a continuation spliced into an entry leaves the places of the runs before it there.
552  for run in runs.into_iter().rev() {
553    let counts: Vec<usize> = run
554      .lines
555      .iter()
556      .map(|(_, line)| line_addresses(line).map_or(0, |a| a.len()))
557      .collect();
558    let total: usize = counts.iter().sum();
559    let mut calls = Vec::new();
560    match run.under {
561      // (a run under an affiliation is that entry's: it goes back into it, whose Kind step takes its addresses out as
562      // the entry's e-mails and whose Link step places them — the author each spells, the entry's only carrier, every
563      // carrier of a symbol-led one, one per carrier in order, else the block once: `kind_entry_addresses`,
564      // `owners::entry_address_owners`, OXIDIZED_DESIGN_DIVERGENCES #956; it no longer stays in the entry's text,
565      // 2609.19448's one address under one author's affiliation still that author's)
566      Some((entry, at)) if under_affiliations => {
567        let label = entry_mark(&entries[entry]).map_or(Ok(None), |m| mark_set_label(&m))?;
568        let requesting = queued_label_requesters(label.as_deref().unwrap_or_default())?;
569        // (one address under an affiliation one author requests is that author's, its line made the contact here: an
570        // e-mail command a raw class's store holds, `\email{hyf@…}` under `\inst{1}`'s institute, prints nowhere if
571        // left in the entry — astro-ph0701846), with the mark that placed it: a call the reading parts into several
572        // names gives it to the one showing that mark (`Resolver::entry_owners`; IEEEtran's one names block, 2510.10414)
573        if let (Some(label), [author], 1) = (label, requesting.as_slice(), total) {
574          let by_mark =
575            mouth::tokenize_internal(TeXString::assembled(s!("_placedbymark={{{label}}}")));
576          for (_, line) in run.lines {
577            let placed = email_line_calls(line, AddressPlacement::Authors(vec![*author]))?;
578            calls.extend(optioned_email_calls(placed, by_mark.unlist_ref()));
579          }
580        } else {
581          let continuation: Vec<Token> = run
582            .lines
583            .into_iter()
584            .flat_map(|(delimiter, line)| delimiter.into_iter().chain(line.unlist()))
585            .collect();
586          entries[entry].splice(at..at, continuation);
587        }
588      },
589      _ if total == authors - row_start && run_in_author_order(&run.lines, row_start)? => {
590        let mut positions = row_start + 1..;
591        for ((_, line), count) in run.lines.into_iter().zip(counts) {
592          let authors = positions.by_ref().take(count).collect();
593          calls.extend(email_line_calls(line, AddressPlacement::Authors(authors))?);
594        }
595      },
596      _ => {
597        // (an address the run's place leaves to no author still goes to the one it spells, before the block: the
598        // placement order's step 2 before step 4 — aa's `\institute{$^{1}$Univ X\\ $^{2}$Univ Y\\ \email{ann.able@y.org}}`
599        // under three names, 1507.05963, cut at its marks by the frontmatter API, OXIDIZED_DESIGN_DIVERGENCES #954)
600        let keys = queued_author_name_keys()?;
601        for (_, line) in run.lines {
602          let owners: Vec<Option<usize>> = line_addresses(&line)
603            .unwrap_or_default()
604            .iter()
605            .map(|address| spelled_owner(&keys, address).map(|owner| owner + 1))
606            .collect();
607          let placement = if owners.iter().any(Option::is_some) {
608            AddressPlacement::Settled(owners)
609          } else {
610            AddressPlacement::Shared
611          };
612          calls.extend(email_line_calls(line, placement)?);
613        }
614      },
615    }
616    placed.push(calls);
617  }
618  Ok(placed.into_iter().rev().flatten().collect())
619}

621/// Where the addresses of a line in an author block go.
622pub(in crate::base_utilities) enum AddressPlacement {
623  /// Each address to the author at that queue position, by the `author:N` label the creator digests with.
624  Authors(Vec<usize>),
625  /// Each address a name settles to the author at that queue position, as [`AddressPlacement::Authors`]; each other
626  /// the block's, as [`AddressPlacement::Shared`] places it (2609.15028's `qq@…`-like address no name spells beside
627  /// `zihanxue@…`, `serhii@…`; user ruling R10 2026-10-20). The line itself when no address is settled.
628  Settled(Vec<Option<usize>>),
629  /// The block's, on the shared creator below the authors (#159): its label names no creator. One contact an address
630  /// when the line lists several (1802.07182's `\{jrr41,$\,$wct23,…\}@cam.ac.uk`, whose raw line a Semiverbatim
631  /// contact showed as `{jrr41,$ $wct23,…}`), else the line.
632  Shared,
633  /// Each address its own contact, with no placement of its own: an address line of an affiliation-kind entry, digested
634  /// inside it, the entry's address — its owners the entry's Link step gives ([`kind_entry_addresses`],
635  /// `owners::entry_address_owners`).
636  Inner,
637}
