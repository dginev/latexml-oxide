1//! The frontmatter API's line readers for an entry's marks: the mark-command aliases, a line's
2//! notes and leading footnote symbol, and whether `\inst` prints its mark. Moved unchanged from `author_block/marks.rs`,
3//! `separators.rs`, `symbols.rs` and `notes.rs` (Z18j).

7/// `tokens` with every alias of a mark command written as the command it means ([`mark_command_meant`]), so the
8/// readers of an author or affiliation block that match a command by name (`\textsuperscript`'s symbol and ordinal
9/// rewrites) see the alias as the command.
10pub(in crate::base_utilities) fn unalias_mark_commands(tokens: Tokens) -> Tokens {
11  if !tokens.unlist_ref().iter().any(|t| {
12    matches!(t.get_catcode(), Catcode::CS | Catcode::ACTIVE)
13      && !mark_commands().contains(t)
14      && is_mark_command(t)
15  }) {
16    return tokens;
17  }
18  Tokens::new(
19    tokens
20      .unlist()
21      .into_iter()
22      .map(|t| mark_command_meant(&t).unwrap_or(t))
23      .collect(),
24  )
25}

27/// Whether the document's `\inst` is the inst convention's mark, which takes back the space before it: its meaning starts
28/// with `\unskip` (llncs.cls:894 `\def\inst#1{\unskip$^{#1}$}`, aa.cls:759, svjour3.cls, read raw), or it is the
29/// request the convention's bindings give it for that definition (`\lx@request@frontmatter@annotation`: llncs_cls.rs,
30/// inst_support_sty.rs, sv_support_sty.rs). Beamer's (beamerbasetitle.sty:169 `\textsuperscript{#1}`) and a document's
31/// own `\newcommand{\inst}[1]{$^{#1}$}` are none.
32/// Inside author content, where `\lx@author@withinst` makes `\inst` its mark reader, the one it saved decides; a class
33/// that defines `\inst` only inside its title, where it prints the names (epl2.cls:725 `\def\inst##1{\unskip${}^{##1}$…}`,
34/// 1201.1900), has its binding declare that meaning as `\lx@title@inst`, which decides first.
35pub(in crate::base_utilities) fn inst_takes_back_space() -> bool {
36  let first = |cs: &str| match lookup_definition(&T_CS!(cs)) {
37    Ok(Some(defn)) => match defn.get_expansion() {
38      Some(ExpansionBody::Tokens(body)) => body
39        .unlist_ref()
40        .iter()
41        .find(|t| **t != T_SPACE!())
42        .copied(),
43      _ => None,
44    },
45    _ => None,
46  };
47  let lead = match first("\\lx@title@inst").or_else(|| first("\\inst")) {
48    Some(t) if t == T_CS!("\\lx@inst@mark") => first("\\lx@saved@inst"),
49    lead => lead,
50  };
51  lead.is_some_and(|t| t == T_CS!("\\unskip") || t == T_CS!("\\lx@request@frontmatter@annotation"))
52}

54/// Does the document's `\inst` print its one argument as a raised mark — the inst convention's
55/// ([`inst_takes_back_space`]: aa.cls:759 `\def\inst#1{\unskip\hbox{$^{#1}$}}`, mem.cls:318, llncs), or a definition of
56/// one parameter that raises it (beamerbasetitle.sty:169 `\textsuperscript{#1}`, a document's `$^{#1}$`)? By its
57/// meaning, not its name: a document's `\newcommand{\inst}{Institute}` is a word (review rev99a S2).
58pub(in crate::base_utilities) fn inst_prints_its_mark() -> bool {
59  if inst_takes_back_space() {
60    return true;
61  }
62  let Ok(Some(defn)) = lookup_definition(&T_CS!("\\inst")) else {
63    return false;
64  };
65  let one_argument = defn.get_parameters().is_some_and(|p| {
66    let params = p.get_parameters();
67    params.len() == 1 && !params[0].optional
68  });
69  one_argument
70    && matches!(defn.get_expansion(), Some(ExpansionBody::Tokens(body))
71      if body.unlist_ref().iter().any(|t| *t == T_SUPER!() || is_mark_command(t)))
72}

74/// Does `line` open (after spaces, math shifts and braces) with a footnote symbol (`* Equal contribution`,
75/// `$\dagger$ Corresponding author`, `\dag Corresponding author`) — a legend for the authors' symbol marks, not an
76/// affiliation line? The text-mode `\dag`/`\ddag` count here only: as superscript marks (`$^\dag$`) they still label
77/// affiliations (`is_footnote_symbol_operand` leaves them out, so `rewrite_symbol_superscripts` keeps them as marks).
78pub(in crate::base_utilities) fn starts_with_footnote_symbol(line: &Tokens) -> bool {
79  line
80    .unlist_ref()
81    .iter()
82    .find(|t| {
83      !matches!(
84        t.get_catcode(),
85        Catcode::SPACE | Catcode::MATH | Catcode::BEGIN | Catcode::END
86      )
87    })
88    .is_some_and(|t| {
89      *t == T_CS!("\\dag")
90        || *t == T_CS!("\\ddag")
91        || is_footnote_symbol_operand(std::slice::from_ref(t))
92    })
93}

95/// The notes of a line of notes only, each the span `(start, end)` of `line` from its command
96/// through its closing brace; `None` when the line holds other ink.
97pub(in crate::base_utilities) fn line_notes(line: &[Token]) -> Option<Vec<(usize, usize)>> {
98  let spacing = |t: &Token| *t == T_SPACE!() || *t == T_ACTIVE!('~') || *t == T_CS!("\\par");
99  let mut notes = Vec::new();
100  let mut i = 0;
101  while i < line.len() {
102    if spacing(&line[i]) {
103      i += 1;
104    } else {
105      let end = note_call_end(line, i)?;
106      notes.push((i, end));
107      i = end;
108    }
109  }
110  Some(notes)
111}
