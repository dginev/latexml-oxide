9/// The Kind step for the addresses printed inside an affiliation-kind entry (FRONTMATTER_SINGLE_ENGINE_PLAN "The structure
10/// model"; census127 cluster 1, the address family), at the frontmatter API's one call site after the Cut
11/// (`\lx@annotate@frontmatter`, each piece of the Cut passing once): an affiliation, alternate affiliation or address
12/// contact ([`CUT_ROLES`]) holding a line of e-mail addresses — a list (`\{aable, bbaker, ccole\}@a.edu`, `a@x, b@y`),
13/// one behind an e-mail label (`Email: a@x, b@y`, its label the contacts' name), an e-mail command's (`\email{…}`), or
14/// the rest of the entry's first line behind the mark that labels it (`{\tt $^1$\{aable, ccole\}@a.edu}`, the marked
15/// entry the author block cut it as) — is that entry's content without those lines, each address an e-mail call digested
16/// inside it ([`AddressPlacement::Inner`], `email_line_calls`): an e-mail of the entry's, whose owners the Link step
17/// gives (`owners::entry_address_owners`, by the `_entry` key `\lx@annotate@frontmatter@now` gives the entry and
18/// its inner contacts). An entry of addresses alone stays as their carrier, empty, and is dropped once they are placed
19/// (`insert::drop_empty_contacts`). `None` for any other contact, or one holding no such line. The lines are read by the
20/// author block's reader of an address line ([`line_addresses`]): an address inside a line of other words (`Univ X,
21/// Town; ann@x.org`) is the frontmatter observer's text reading's alone (`Resolver::take_out_entry_addresses`), and a
22/// line these tokens do not show — a document's macro, `\authorcr`, a connective "and" — the flush's Kind step's, which
23/// reads it as printed (`insert::kind_address_lines`, OD #966; 2608.22602, 2111.01631, 2302.02638). A
24/// template (`\{firstname.lastname\}@x.org`, [`is_pattern_address`]) or a list the reader could not part
25/// ([`lists_local_parts`]) stays where it prints. Witnesses: authblk's `\affil{… \\ \{…\}@x}` (2502.01754) and
26/// `\affil[ ]{\texttt{\{…\}@x}}` (2602.21641, 2307.02047), the kernel's `{\tt $^1$\{…\}@x}` lines (1901.09107), llncs's
27/// `\email{\{ivan|carolin\}@x}` (0902.3918, 1909.00276), aaspp4's `$^3$…;\\ Email: dzg@, xywang@, hyf@`
28/// (astro-ph0512555).
29pub(in crate::base_utilities) fn kind_entry_addresses(
30  tag: &Tokens,
31  kv: Option<&KeyVals>,
32  content: &Tokens,
33) -> Result<Option<Tokens>> {
34  if tag.to_string() != "ltx:contact"
35    || !content
36      .unlist_ref()
37      .iter()
38      .any(|t| t.with_str(|s| s == "@"))
39  {
40    return Ok(None);
41  }
42  let role = kv
43    .and_then(|kv| kv.get_value("role"))
44    .map(ToString::to_string)
45    .unwrap_or_default();
46  // (an entry a class shares among every author, llncs's one `\institute`, `_share=authors`: its lines of addresses are
47  // the shared row's, placed by the reading's chain (`Resolver::answer_shared_emails`; the old split's `row_email_calls` until P27b), OD #901;
48  // 1103.2027)
49  if !CUT_ROLES.contains(&role.as_str()) || kv.is_some_and(|kv| kv.get_value("_share").is_some()) {
50    return Ok(None);
51  }
52  let (withinst, inner) = within_inst(content);
53  let (withsup, inner) = within_wrapper(&inner, "\\lx@affiliation@withsup");
54  // (a footnote of the entry's holding addresses alone, behind a label or none — REVTeX's `\affiliation{Univ A,
55  // Town\footnote{Electronic address: aable@a.edu}}`, the note the class's own `\email` prints (revtex4-2.cls:2404):
56  // its addresses the entry's, as an address line's; Perl keeps the footnote a `<pubnote>` of the document,
57  // Base_Utility.pool.ltxml:331 — 1002.3223, cond-mat0202268)
58  let (inner, footnotes) = take_address_footnotes(inner);
59  // (a line the reader takes out: its addresses read, none a template or a list it could not part, and its groups
60  // balanced, so the entry left is)
61  // (a line of e-mail commands is e-mails already, the class's own contacts with the label it prints — llncs's
62  // `\email{…}` "E-mail", 2008.05861's `\email{\{jlwang,yhliu\}@…}, \email{jianbo@…}` —, digested inside the entry
63  // and so its addresses: left as it is)
64  // (a line holding any other command — an ORCID, a note, a symbol — is no line of addresses alone: it stays in the
65  // entry as it prints, its addresses the text reading's, [`prints_addresses_only`]; 2308.02269's
66  // `\url{{arim,sue}@ist.hokudai.ac.jp}\orcidID{0000-0002-2701-0271}` had read the ORCID's digits into the domain and
67  // lost the ORCID, review rev100b B1)
68  let addresses_of = |line: &Tokens| {
69    let command = line
70      .unlist_ref()
71      .iter()
72      .any(|t| t.get_catcode() == Catcode::CS && is_email_wrapper(t));
73    (!command && prints_addresses_only(line))
74      .then(|| line_addresses(line))
75      .flatten()
76      .filter(|addresses| {
77        line.is_balanced()
78          && addresses
79            .iter()
80            .all(|a| !is_pattern_address(a) && !lists_local_parts(a))
81      })
82  };
83  let mut kept: Vec<(Vec<Token>, Vec<Token>)> = Vec::new();
84  // (each address line, and for a mailbox link line the label before its link: [`mailbox_link_line`])
85  let mut address_lines: Vec<(Tokens, Option<Option<Tokens>>)> = Vec::new();
86  // (for each address line, whether no line the entry keeps prints before it)
87  let mut leading: Vec<bool> = Vec::new();
88  let mut first = true;
89  // (whether the entry prints anything but addresses, its mark aside)
90  let mut prints_else = false;
91  for (delimiter, line) in affiliation_lines(&inner)? {
92    let printed = !line.is_empty() && !prints_nothing(&line);
93    // (an address line read as it prints: a formula of spacing alone a space, no text of an address —
94    // `$~\quad~~$ cbray@mines.edu`, 2205.14113; `a@x $\quad$ b@y`, 2302.05182; a line that is no address line
95    // stays as written)
96    let spaced = spacing_formulas_as_space(line.clone());
97    // (the first line of a marked entry: the mark that labels it kept, the addresses behind it the entry's)
98    if printed
99      && first
100      && withsup
101      && let Some((mark, rest)) = mark_led_addresses(&spaced)
102      && addresses_of(&rest).is_some()
103    {
104      kept.push((delimiter, mark));
105      address_lines.push((rest, None));
106      leading.push(false);
107      first = false;
108      continue;
109    }
110    // (a line a mark leads is no address line: its mark is no letter of an address, `$^2$bob@y.org`)
111    if printed && author_mark_position(&line).is_none() && addresses_of(&spaced).is_some() {
112      address_lines.push((spaced, None));
113      leading.push(!prints_else);
114      first = false;
115      continue;
116    }
117    if printed
118      && author_mark_position(&line).is_none()
119      && let Some((label, link)) = mailbox_link_line(&line)
120    {
121      address_lines.push((link, Some(label)));
122      leading.push(!prints_else);
123      first = false;
124      continue;
125    }
126    first &= !printed;
127    prints_else |= printed;
128    kept.push((delimiter, line.unlist()));
129  }
130  if address_lines.is_empty() && footnotes.is_empty() {
131    return Ok(None);
132  }
133  let mut entry: Vec<Token> = Vec::new();
134  for (delimiter, line) in kept {
135    if !entry.is_empty() {
136      entry.extend(delimiter);
137    }
138    entry.extend(line);
139  }
140  let mut entry = trim_trailing_separator(entry);
141  // (an entry of addresses alone that the class names, `\mail{…}`'s "Correspondence to", REVTeX's
142  // `\altaffiliation[E-mail address: ]{…}`: its name the e-mails' — where no label of their own names them; 0709.0811,
143  // astro-ph0108379, cond-mat0703234, 1402.3092)
144  let entry_name: Option<Vec<Token>> = (!prints_else && footnotes.is_empty())
145    .then(|| kv.and_then(|kv| kv.revert().ok()))
146    .flatten()
147    .and_then(|attrs| {
148      split_tokens(attrs, vec![T_OTHER!(",").into()])
149        .into_iter()
150        .find(|item| item.to_string().trim_start().starts_with("name="))
151        .map(Tokens::unlist)
152    })
153    // (not the role's own default, `\lx@contact@altaffiliation@name` — "Alternate Affiliation:" names no e-mail, 0812.4856
154    // —, nor an empty one, which names nothing: aa's `\mail` copy, 97b)
155    .filter(|name| {
156      let value = name.iter().skip_while(|t| **t != T_OTHER!("=")).skip(1);
157      !name
158        .iter()
159        .any(|t| t.get_catcode() == Catcode::CS && t.with_str(|s| s.starts_with("\\lx@contact@")))
160        && value.clone().any(|t| {
161          matches!(
162            t.get_catcode(),
163            Catcode::LETTER | Catcode::OTHER | Catcode::CS
164          )
165        })
166    });
167  // (each e-mail the Kind step made says so, `ENTRY_KINDED`: an e-mail command the author wrote inside the entry is
168  // placed by its own call where it holds one address, `insert::link_entry_addresses`)
169  let kinded = mouth::tokenize_internal(TeXString::assembled(s!("{ENTRY_KINDED}=true"))).unlist();
170  // (a line printed before every line the entry keeps, its e-mails placed before the entry: user ruling R8,
171  // `\IEEEauthorblockA{\href{mailto:ann@a.org}{ann@a.org}\\ \url{https://orcid.org/…}\\ …}`, 2206.10577 — review
172  // rev107c N1, `insert::LINE_FIRST`)
173  let kinded_first = mouth::tokenize_internal(TeXString::assembled(s!(
174    "{ENTRY_KINDED}=true,{LINE_FIRST}=true"
175  )))
176  .unlist();
177  for ((line, link_label), leading) in address_lines.into_iter().zip(leading) {
178    let kinded = if leading && prints_else {
179      &kinded_first
180    } else {
181      &kinded
182    };
183    // (the line's own label — a mailbox link line's text before its link, else its e-mail label — names its contacts,
184    // after the class's name of the entry where both print, one rule: [`entry_line_label`]; else the class's name)
185    let labelled = match link_label {
186      Some(label) => label.map(|label| (label, line.clone())),
187      None => email_line_label(&line),
188    };
189    let calls = match (labelled, &entry_name) {
190      (Some((label, rest)), name) => named_email_calls(
191        email_line_calls(rest, AddressPlacement::Inner)?,
192        &entry_line_label(name.as_deref(), label),
193      ),
194      (None, Some(name)) => {
195        optioned_email_calls(email_line_calls(line, AddressPlacement::Inner)?, name)
196      },
197      (None, None) => email_line_calls(line, AddressPlacement::Inner)?,
198    };
199    entry.extend(optioned_email_calls(calls, kinded));
200  }
201  for (label, line) in footnotes {
202    // (the note's label, then the line's own: `\footnote{Corresponding author e-mail: Email:zz@x.org}`, revGL1 B1)
203    let (label, line) = joined_line_label(label, line);
204    let calls = match label {
205      Some(label) => named_email_calls(email_line_calls(line, AddressPlacement::Inner)?, &label),
206      None => email_line_calls(line, AddressPlacement::Inner)?,
207    };
208    entry.extend(optioned_email_calls(calls, &kinded));
209  }
210  let mut entry = Tokens::new(entry);
211  if withsup {
212    entry = Invocation!(T_CS!("\\lx@affiliation@withsup"), vec![Some(entry)]);
213  }
214  if withinst {
215    entry = Invocation!(T_CS!("\\lx@affiliation@withinst"), vec![Some(entry)]);
216  }
217  DebugFeature!(
218    "frontmatter",
219    "FRONT kind {role}: its address lines the entry's e-mails: {entry}"
220  );
221  Ok(Some(entry))
222}

231/// Whether a footnote symbol leads an entry's content (in the affiliation's own wrappers): a text symbol (`* Equal
232/// contribution`, `\dag …`, [`starts_with_footnote_symbol`]) or a superscript one (`$^\dagger$Univ X`).
233pub(in crate::base_utilities) fn led_by_footnote_symbol(content: &Tokens) -> bool {
234  let (_, inner) = within_inst(content);
235  let (_, inner) = within_wrapper(&inner, "\\lx@affiliation@withsup");
236  starts_with_footnote_symbol(&inner)
237    || (marker_leads(&inner)
238      && entry_mark(inner.unlist_ref()).is_some_and(|mark| is_symbol_mark(&mark)))
239}

241/// `content` without the notes (`\footnote{…}`, `\thanks{…}`, an optional mark aside) that print a line of addresses
242/// alone, behind a label or none, and those lines, each with the label its contacts are named by when the line's own
243/// reader does not read it ([`note_addresses`]; [`kind_entry_addresses`]).
244fn take_address_footnotes(content: Tokens) -> (Tokens, Vec<(Option<Tokens>, Tokens)>) {
245  let v = content.unlist_ref();
246  let note_cs = [T_CS!("\\footnote"), T_CS!("\\thanks")];
247  if !v.iter().any(|t| note_cs.contains(t)) {
248    return (content, Vec::new());
249  }
250  let mut kept: Vec<Token> = Vec::with_capacity(v.len());
251  let mut lines: Vec<(Option<Tokens>, Tokens)> = Vec::new();
252  let mut i = 0;
253  while i < v.len() {
254    if note_cs.contains(&v[i]) {
255      let mut open = i + 1;
256      while v.get(open) == Some(&T_SPACE!()) {
257        open += 1;
258      }
259      // (its optional mark, `\footnote[2]{…}`)
260      if v.get(open) == Some(&T_OTHER!("["))
261        && let Some(close) = v[open..].iter().position(|t| *t == T_OTHER!("]"))
262      {
263        open += close + 1;
264        while v.get(open) == Some(&T_SPACE!()) {
265          open += 1;
266        }
267      }
268      if v
269        .get(open)
270        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
271      {
272        let end = skip_group(v, open);
273        if end <= v.len()
274          && end >= open + 2
275          && let Some(line) = note_addresses(&Tokens::new(v[open + 1..end - 1].to_vec()))
276        {
277          lines.push(line);
278          i = end;
279          continue;
280        }
281      }
282    }
283    kept.push(v[i]);
284    i += 1;
285  }
286  (Tokens::new(kept), lines)
287}

289/// The addresses line a note prints when it prints addresses alone: one [`line_addresses`] reads (a list, one behind
290/// an e-mail label), or one behind a label ending in an e-mail label's words before its colon (`Corresponding authors
291/// e-mail: cecilia.pennetta@unile.it`, cond-mat0202268; REVTeX's `Electronic address: …`, 1002.3223) — the label then
292/// given apart, the contacts' name. None a template or a list the reader could not part.
293fn note_addresses(note: &Tokens) -> Option<(Option<Tokens>, Tokens)> {
294  let real = |addresses: Vec<String>| {
295    addresses
296      .iter()
297      .all(|a| !is_pattern_address(a) && !lists_local_parts(a))
298  };
299  if line_addresses(note).is_some_and(real) {
300    return Some((None, note.clone()));
301  }
302  let v = note.unlist_ref();
303  let colon = v.iter().position(|t| *t == T_OTHER!(":"))?;
304  let label: Vec<Token> = v[..colon]
305    .iter()
306    .copied()
307    .skip_while(|t| *t == T_SPACE!())
308    .collect();
309  let text = visible_name_text(&label);
310  let words: Vec<&str> = text.split_whitespace().collect();
311  // (a label of a few words — `Corresponding authors e-mail`; a sentence before its colon is the note's text, `X. Zhang
312  // and L. Duan are with …, Singapore. E-mail`, 1710.05616)
313  let labelled = words.len() <= 4
314    && (1..=2.min(words.len())).any(|n| is_email_label(&words[words.len() - n..].join(" ")));
315  let rest = Tokens::new(
316    v[colon + 1..]
317      .iter()
318      .copied()
319      .skip_while(is_spacing)
320      .collect(),
321  );
322  (labelled && email_addresses(&rest).is_some_and(real)).then(|| {
323    let mut label = label;
324    label.push(T_OTHER!(":"));
325    (Some(Tokens::new(label)), rest)
326  })
327}

365/// Whether `line` holds no command but those a line of addresses alone prints them with ([`ADDRESS_LINE_COMMANDS`], the
366/// font, size and spacing commands; a command whose arguments print no letter, `\color{…}`, `\hspace{…}`, aside:
367/// [`without_non_printing_arguments`]). Any other — `\orcidID{…}` (2308.02269), `\thanks{…}`, `$^\dagger$` — prints
368/// more than the addresses, and the line stays in its entry as it prints (review rev100b B1). On the tokens the line is
369/// taken out of, as [`line_addresses`] reads its addresses (`resolver::prints_only_addresses` reads printed text; P6f).
370fn prints_addresses_only(line: &Tokens) -> bool {
371  without_non_printing_arguments(line.unlist_ref())
372    .iter()
373    .all(|t| {
374      t.get_catcode() != Catcode::CS
375        || is_spacing(t)
376        || is_spacing_command(t)
377        || is_line_separator(t)
378        || name_text_command(t)
379        || t.with_str(|s| FONT_SWITCH.is_match(s) || ADDRESS_LINE_COMMANDS.contains(&s))
380    })
381}

383/// A line led by a printed mark (`$^1$…`, `\textsuperscript{1}…`, the font switches before it kept: `{\tt $^1$\{aable,
384/// ccole\}@a.edu}`, a group round the whole line read inside), parted into the mark and what follows it, each in the
385/// line's group: `None` for a line no printed mark leads (`relocate::leading_line_label`; a number typed by hand is no
386/// mark here).
387fn mark_led_addresses(line: &Tokens) -> Option<(Vec<Token>, Tokens)> {
388  let v = line.unlist_ref();
389  let grouped = v.len() >= 2 && v[0].get_catcode() == Catcode::BEGIN && skip_group(v, 0) == v.len();
390  let inner = if grouped { &v[1..v.len() - 1] } else { v };
391  let (_, typed, rest) = leading_line_label(&Tokens::new(inner.to_vec()))?;
392  if typed {
393    return None;
394  }
395  // (the rest is the switches before the mark and the tokens after it, those a tail of the line)
396  let lead = inner
397    .iter()
398    .take_while(|t| **t == T_SPACE!() || mark_switch(t))
399    .count();
400  let tail = &rest.unlist_ref()[lead.min(rest.unlist_ref().len())..];
401  let mark = &inner[..inner.len().checked_sub(tail.len())?];
402  let wrap = |tokens: Vec<Token>| {
403    if grouped {
404      let mut out = vec![v[0]];
405      out.extend(tokens);
406      out.push(v[v.len() - 1]);
407      out
408    } else {
409      tokens
410    }
411  };
412  let mut address = inner[..lead].to_vec();
413  address.extend_from_slice(tail);
414  Some((wrap(mark.to_vec()), Tokens::new(wrap(address))))
415}

417/// The content of `content` inside the one-argument wrapper `cs` round all of it (`\lx@affiliation@withsup{…}`), and
418/// whether it was.
419fn within_wrapper(content: &Tokens, cs: &str) -> (bool, Tokens) {
420  let v = content.unlist_ref();
421  if v.len() >= 3
422    && v[0] == T_CS!(cs)
423    && v[1].get_catcode() == Catcode::BEGIN
424    && skip_group(v, 1) == v.len()
425  {
426    (true, Tokens::new(v[2..v.len() - 1].to_vec()))
427  } else {
428    (false, content.clone())
429  }
430}

432/// The content of a contact inside the affiliation's own wrapper (`\lx@affiliation@withinst{…}`, which
433/// `\lx@add@affiliation` puts around it), and whether it was in it.
434pub(super) fn within_inst(content: &Tokens) -> (bool, Tokens) {
435  let within = content.unlist_ref();
436  let withinst = within.len() >= 3
437    && within[0] == T_CS!("\\lx@affiliation@withinst")
438    && within[1].get_catcode() == Catcode::BEGIN
439    && skip_group(within, 1) == within.len();
440  if withinst {
441    (true, Tokens::new(within[2..within.len() - 1].to_vec()))
442  } else {
443    (false, content.clone())
444  }
