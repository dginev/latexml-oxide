9/// The Kind step for the addresses printed inside an affiliation-kind entry (FRONTMATTER_SINGLE_ENGINE_PLAN "The structure
10/// model"; census127 cluster 1, the address family), at the frontmatter API's one call site after the Cut
11/// (`\lx@annotate@frontmatter`, each piece of the Cut passing once): an affiliation, alternate affiliation or address
12/// contact ([`CUT_ROLES`]) holding a line of e-mail addresses — a list (`\{aable, bbaker, ccole\}@a.edu`, `a@x, b@y`),
13/// one behind an e-mail label (`Email: a@x, b@y`, its label the contacts' name), an e-mail command's (`\email{…}`), or
14/// the rest of the entry's first line behind the mark that labels it (`{\tt $^1$\{aable, ccole\}@a.edu}`, the marked
15/// entry the author block cut it as) — is that entry's content without those lines, each address an e-mail call digested
16/// inside it ([`AddressPlacement::Inner`], `email_line_calls`): an e-mail of the entry's, whose owners the Link step
17/// gives (`owners::entry_address_owners`, by the `_entry` key `\lx@annotate@frontmatter@now` gives the entry and
18/// its inner contacts). An entry of addresses alone stays as their carrier, empty, and is dropped once they are placed
19/// (`insert::drop_empty_contacts`). `None` for any other contact, or one holding no such line. The lines are read by the
20/// author block's reader of an address line ([`line_addresses`]): an address inside a line of other words (`Univ X,
21/// Town; ann@x.org`) is the frontmatter observer's text reading's alone (`Resolver::take_out_entry_addresses`), and a
22/// line these tokens do not show — a document's macro, `\authorcr`, a connective "and" — the flush's Kind step's, which
23/// reads it as printed (`insert::kind_address_lines`, OD #966; 2608.22602, 2111.01631, 2302.02638). A
24/// template (`\{firstname.lastname\}@x.org`, [`is_pattern_address`]) or a list the reader could not part
25/// ([`lists_local_parts`]) stays where it prints. Witnesses: authblk's `\affil{… \\ \{…\}@x}` (2502.01754) and
26/// `\affil[ ]{\texttt{\{…\}@x}}` (2602.21641, 2307.02047), the kernel's `{\tt $^1$\{…\}@x}` lines (1901.09107), llncs's
27/// `\email{\{ivan|carolin\}@x}` (0902.3918, 1909.00276), aaspp4's `$^3$…;\\ Email: dzg@, xywang@, hyf@`
28/// (astro-ph0512555).
29pub(in crate::base_utilities) fn kind_entry_addresses(
30  tag: &Tokens,
31  kv: Option<&KeyVals>,
32  content: &Tokens,
33  email_keys: &str,
34) -> Result<Option<Tokens>> {
35  if tag.to_string() != "ltx:contact"
36    || !content
37      .unlist_ref()
38      .iter()
39      .any(|t| t.with_str(|s| s == "@"))
40  {
41    return Ok(None);
42  }
43  let role = kv
44    .and_then(|kv| kv.get_value("role"))
45    .map(ToString::to_string)
46    .unwrap_or_default();
47  // (an entry a class shares among every author, llncs's one `\institute`, `_share=authors`: its lines of addresses are
48  // the shared row's, placed by the reading's chain (`Resolver::answer_shared_emails`; the old split's `row_email_calls` until P27b), OD #901;
49  // 1103.2027)
50  if !CUT_ROLES.contains(&role.as_str()) || kv.is_some_and(|kv| kv.get_value("_share").is_some()) {
51    return Ok(None);
52  }
53  let (withinst, inner) = within_inst(content);
54  let (withsup, inner) = within_wrapper(&inner, "\\lx@affiliation@withsup");
55  // (a footnote of the entry's holding addresses alone, behind a label or none — REVTeX's `\affiliation{Univ A,
56  // Town\footnote{Electronic address: aable@a.edu}}`, the note the class's own `\email` prints (revtex4-2.cls:2404):
57  // its addresses the entry's, as an address line's; Perl keeps the footnote a `<pubnote>` of the document,
58  // Base_Utility.pool.ltxml:331 — 1002.3223, cond-mat0202268)
59  let (inner, footnotes) = take_address_footnotes(inner);
60  // (a line the reader takes out: its addresses read, none a template or a list it could not part, and its groups
61  // balanced, so the entry left is)
62  // (a line of e-mail commands is e-mails already, the class's own contacts with the label it prints — llncs's
63  // `\email{…}` "E-mail", 2008.05861's `\email{\{jlwang,yhliu\}@…}, \email{jianbo@…}` —, digested inside the entry
64  // and so its addresses: left as it is)
65  // (a line holding any other command — an ORCID, a note, a symbol — is no line of addresses alone: it stays in the
66  // entry as it prints, its addresses the text reading's, [`prints_addresses_only`]; 2308.02269's
67  // `\url{{arim,sue}@ist.hokudai.ac.jp}\orcidID{0000-0002-2701-0271}` had read the ORCID's digits into the domain and
68  // lost the ORCID, review rev100b B1)
69  let addresses_of = |line: &Tokens| {
70    let command = line
71      .unlist_ref()
72      .iter()
73      .any(|t| t.get_catcode() == Catcode::CS && is_email_wrapper(t));
74    (!command && prints_addresses_only(line))
75      .then(|| line_addresses(line))
76      .flatten()
77      .filter(|addresses| {
78        line.is_balanced()
79          && addresses
80            .iter()
81            .all(|a| !is_pattern_address(a) && !lists_local_parts(a))
82      })
83  };
84  let mut kept: Vec<(Vec<Token>, Vec<Token>)> = Vec::new();
85  // (each address line, and for a mailbox link line the label before its link: [`mailbox_link_line`])
86  let mut address_lines: Vec<(Tokens, Option<Option<Tokens>>)> = Vec::new();
87  // (for each address line, whether no line the entry keeps prints before it)
88  let mut leading: Vec<bool> = Vec::new();
89  let mut first = true;
90  // (whether the entry prints anything but addresses, its mark aside)
91  let mut prints_else = false;
92  for (delimiter, line) in affiliation_lines(&inner)? {
93    let printed = !line.is_empty() && !prints_nothing(&line);
94    // (an address line read as it prints: a formula of spacing alone a space, no text of an address —
95    // `$~\quad~~$ cbray@mines.edu`, 2205.14113; `a@x $\quad$ b@y`, 2302.05182; a line that is no address line
96    // stays as written)
97    let spaced = spacing_formulas_as_space(line.clone());
98    // (the first line of a marked entry: the mark that labels it kept, the addresses behind it the entry's)
99    if printed
100      && first
101      && withsup
102      && let Some((mark, rest)) = mark_led_addresses(&spaced)
103      && addresses_of(&rest).is_some()
104    {
105      kept.push((delimiter, mark));
106      address_lines.push((rest, None));
107      leading.push(false);
108      first = false;
109      continue;
110    }
111    // (a line a mark leads is no address line: its mark is no letter of an address, `$^2$bob@y.org`)
112    if printed && author_mark_position(&line).is_none() && addresses_of(&spaced).is_some() {
113      address_lines.push((spaced, None));
114      leading.push(!prints_else);
115      first = false;
116      continue;
117    }
118    if printed
119      && author_mark_position(&line).is_none()
120      && let Some((label, link)) = mailbox_link_line(&line)
121    {
122      address_lines.push((link, Some(label)));
123      leading.push(!prints_else);
124      first = false;
125      continue;
126    }
127    first &= !printed;
128    prints_else |= printed;
129    kept.push((delimiter, line.unlist()));
130  }
131  if address_lines.is_empty() && footnotes.is_empty() {
132    return Ok(None);
133  }
134  let mut entry: Vec<Token> = Vec::new();
135  for (delimiter, line) in kept {
136    if !entry.is_empty() {
137      entry.extend(delimiter);
138    }
139    entry.extend(line);
140  }
141  let mut entry = trim_trailing_separator(entry);
142  // (an entry of addresses alone that the class names, `\mail{…}`'s "Correspondence to", REVTeX's
143  // `\altaffiliation[E-mail address: ]{…}`: its name the e-mails' — where no label of their own names them; 0709.0811,
144  // astro-ph0108379, cond-mat0703234, 1402.3092)
145  let entry_name: Option<Vec<Token>> = (!prints_else && footnotes.is_empty())
146    .then(|| kv.and_then(|kv| kv.revert().ok()))
147    .flatten()
148    .and_then(|attrs| {
149      split_tokens(attrs, vec![T_OTHER!(",").into()])
150        .into_iter()
151        .find(|item| item.to_string().trim_start().starts_with("name="))
152        .map(Tokens::unlist)
153    })
154    // (not the role's own default, `\lx@contact@altaffiliation@name` — "Alternate Affiliation:" names no e-mail, 0812.4856
155    // —, nor an empty one, which names nothing: aa's `\mail` copy, 97b)
156    .filter(|name| {
157      let value = name.iter().skip_while(|t| **t != T_OTHER!("=")).skip(1);
158      !name
159        .iter()
160        .any(|t| t.get_catcode() == Catcode::CS && t.with_str(|s| s.starts_with("\\lx@contact@")))
161        && value.clone().any(|t| {
162          matches!(
163            t.get_catcode(),
164            Catcode::LETTER | Catcode::OTHER | Catcode::CS
165          )
166        })
167    });
168  // (each e-mail the Kind step made says so, `ENTRY_KINDED`: an e-mail command the author wrote inside the entry is
169  // placed by its own call where it holds one address, `insert::link_entry_addresses`; and the keys its writer gives them,
170  // `email_keys`: the class of the author block's line it decided, issue #783)
171  let kinded_keys = s!("{ENTRY_KINDED}=true{email_keys}");
172  let kinded = mouth::tokenize_internal(TeXString::assembled(kinded_keys.clone())).unlist();
173  // (a line printed before every line the entry keeps, its e-mails placed before the entry: user ruling R8,
174  // `\IEEEauthorblockA{\href{mailto:ann@a.org}{ann@a.org}\\ \url{https://orcid.org/…}\\ …}`, 2206.10577 — review
175  // rev107c N1, `insert::LINE_FIRST`)
176  let kinded_first =
177    mouth::tokenize_internal(TeXString::assembled(s!("{kinded_keys},{LINE_FIRST}=true"))).unlist();
178  for ((line, link_label), leading) in address_lines.into_iter().zip(leading) {
179    let kinded = if leading && prints_else {
180      &kinded_first
181    } else {
182      &kinded
183    };
184    // (the line's own label — a mailbox link line's text before its link, else its e-mail label — names its contacts,
185    // after the class's name of the entry where both print, one rule: [`entry_line_label`]; else the class's name)
186    let labelled = match link_label {
187      Some(label) => label.map(|label| (label, line.clone())),
188      None => email_line_label(&line),
189    };
190    let calls = match (labelled, &entry_name) {
191      (Some((label, rest)), name) => named_email_calls(
192        email_line_calls(rest, AddressPlacement::Inner)?,
193        &entry_line_label(name.as_deref(), label),
194      ),
195      (None, Some(name)) => {
196        optioned_email_calls(email_line_calls(line, AddressPlacement::Inner)?, name)
197      },
198      (None, None) => email_line_calls(line, AddressPlacement::Inner)?,
199    };
200    entry.extend(optioned_email_calls(calls, kinded));
201  }
202  for (label, line) in footnotes {
203    // (the note's label, then the line's own: `\footnote{Corresponding author e-mail: Email:zz@x.org}`, revGL1 B1)
204    let (label, line) = joined_line_label(label, line);
205    let calls = match label {
206      Some(label) => named_email_calls(email_line_calls(line, AddressPlacement::Inner)?, &label),
207      None => email_line_calls(line, AddressPlacement::Inner)?,
208    };
209    entry.extend(optioned_email_calls(calls, &kinded));
210  }
211  let mut entry = Tokens::new(entry);
212  if withsup {
213    entry = Invocation!(T_CS!("\\lx@affiliation@withsup"), vec![Some(entry)]);
214  }
215  if withinst {
216    entry = Invocation!(T_CS!("\\lx@affiliation@withinst"), vec![Some(entry)]);
217  }
218  DebugFeature!(
219    "frontmatter",
220    "FRONT kind {role}: its address lines the entry's e-mails: {entry}"
221  );
222  Ok(Some(entry))
223}

232/// Whether a footnote symbol leads an entry's content (in the affiliation's own wrappers): a text symbol (`* Equal
233/// contribution`, `\dag …`, [`starts_with_footnote_symbol`]) or a superscript one (`$^\dagger$Univ X`).
234pub(in crate::base_utilities) fn led_by_footnote_symbol(content: &Tokens) -> bool {
235  let (_, inner) = within_inst(content);
236  let (_, inner) = within_wrapper(&inner, "\\lx@affiliation@withsup");
237  starts_with_footnote_symbol(&inner)
238    || (marker_leads(&inner)
239      && entry_mark(inner.unlist_ref()).is_some_and(|mark| is_symbol_mark(&mark)))
240}

242/// `content` without the notes (`\footnote{…}`, `\thanks{…}`, an optional mark aside) that print a line of addresses
243/// alone, behind a label or none, and those lines, each with the label its contacts are named by when the line's own
244/// reader does not read it ([`note_addresses`]; [`kind_entry_addresses`]).
245fn take_address_footnotes(content: Tokens) -> (Tokens, Vec<(Option<Tokens>, Tokens)>) {
246  let v = content.unlist_ref();
247  let note_cs = [T_CS!("\\footnote"), T_CS!("\\thanks")];
248  if !v.iter().any(|t| note_cs.contains(t)) {
249    return (content, Vec::new());
250  }
251  let mut kept: Vec<Token> = Vec::with_capacity(v.len());
252  let mut lines: Vec<(Option<Tokens>, Tokens)> = Vec::new();
253  let mut i = 0;
254  while i < v.len() {
255    if note_cs.contains(&v[i]) {
256      let mut open = i + 1;
257      while v.get(open) == Some(&T_SPACE!()) {
258        open += 1;
259      }
260      // (its optional mark, `\footnote[2]{…}`)
261      if v.get(open) == Some(&T_OTHER!("["))
262        && let Some(close) = v[open..].iter().position(|t| *t == T_OTHER!("]"))
263      {
264        open += close + 1;
265        while v.get(open) == Some(&T_SPACE!()) {
266          open += 1;
267        }
268      }
269      if v
270        .get(open)
271        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
272      {
273        let end = skip_group(v, open);
274        if end <= v.len()
275          && end >= open + 2
276          && let Some(line) = note_addresses(&Tokens::new(v[open + 1..end - 1].to_vec()))
277        {
278          lines.push(line);
279          i = end;
280          continue;
281        }
282      }
283    }
284    kept.push(v[i]);
285    i += 1;
286  }
287  (Tokens::new(kept), lines)
288}

290/// The addresses line a note prints when it prints addresses alone: one [`line_addresses`] reads (a list, one behind
291/// an e-mail label), or one behind a label ending in an e-mail label's words before its colon (`Corresponding authors
292/// e-mail: cecilia.pennetta@unile.it`, cond-mat0202268; REVTeX's `Electronic address: …`, 1002.3223) — the label then
293/// given apart, the contacts' name. None a template or a list the reader could not part.
294fn note_addresses(note: &Tokens) -> Option<(Option<Tokens>, Tokens)> {
295  let real = |addresses: Vec<String>| {
296    addresses
297      .iter()
298      .all(|a| !is_pattern_address(a) && !lists_local_parts(a))
299  };
300  if line_addresses(note).is_some_and(real) {
301    return Some((None, note.clone()));
302  }
303  let v = note.unlist_ref();
304  let colon = v.iter().position(|t| *t == T_OTHER!(":"))?;
305  let label: Vec<Token> = v[..colon]
306    .iter()
307    .copied()
308    .skip_while(|t| *t == T_SPACE!())
309    .collect();
310  let text = visible_name_text(&label);
311  let words: Vec<&str> = text.split_whitespace().collect();
312  // (a label of a few words — `Corresponding authors e-mail`; a sentence before its colon is the note's text, `X. Zhang
313  // and L. Duan are with …, Singapore. E-mail`, 1710.05616)
314  let labelled = words.len() <= 4
315    && (1..=2.min(words.len())).any(|n| is_email_label(&words[words.len() - n..].join(" ")));
316  let rest = Tokens::new(
317    v[colon + 1..]
318      .iter()
319      .copied()
320      .skip_while(is_spacing)
321      .collect(),
322  );
323  (labelled && email_addresses(&rest).is_some_and(real)).then(|| {
324    let mut label = label;
325    label.push(T_OTHER!(":"));
326    (Some(Tokens::new(label)), rest)
327  })
328}

366/// Whether `line` holds no command but those a line of addresses alone prints them with ([`ADDRESS_LINE_COMMANDS`], the
367/// font, size and spacing commands; a command whose arguments print no letter, `\color{…}`, `\hspace{…}`, aside:
368/// [`without_non_printing_arguments`]). Any other — `\orcidID{…}` (2308.02269), `\thanks{…}`, `$^\dagger$` — prints
369/// more than the addresses, and the line stays in its entry as it prints (review rev100b B1). On the tokens the line is
370/// taken out of, as [`line_addresses`] reads its addresses (`resolver::prints_only_addresses` reads printed text; P6f).
371fn prints_addresses_only(line: &Tokens) -> bool {
372  without_non_printing_arguments(line.unlist_ref())
373    .iter()
374    .all(|t| {
375      t.get_catcode() != Catcode::CS
376        || is_spacing(t)
377        || is_spacing_command(t)
378        || is_line_separator(t)
379        || name_text_command(t)
380        || t.with_str(|s| FONT_SWITCH.is_match(s) || ADDRESS_LINE_COMMANDS.contains(&s))
381    })
382}

384/// A line led by a printed mark (`$^1$…`, `\textsuperscript{1}…`, the font switches before it kept: `{\tt $^1$\{aable,
385/// ccole\}@a.edu}`, a group round the whole line read inside), parted into the mark and what follows it, each in the
386/// line's group: `None` for a line no printed mark leads (`relocate::leading_line_label`; a number typed by hand is no
387/// mark here).
388fn mark_led_addresses(line: &Tokens) -> Option<(Vec<Token>, Tokens)> {
389  let v = line.unlist_ref();
390  let grouped = v.len() >= 2 && v[0].get_catcode() == Catcode::BEGIN && skip_group(v, 0) == v.len();
391  let inner = if grouped { &v[1..v.len() - 1] } else { v };
392  let (_, typed, rest) = leading_line_label(&Tokens::new(inner.to_vec()))?;
393  if typed {
394    return None;
395  }
396  // (the rest is the switches before the mark and the tokens after it, those a tail of the line)
397  let lead = inner
398    .iter()
399    .take_while(|t| **t == T_SPACE!() || mark_switch(t))
400    .count();
401  let tail = &rest.unlist_ref()[lead.min(rest.unlist_ref().len())..];
402  let mark = &inner[..inner.len().checked_sub(tail.len())?];
403  let wrap = |tokens: Vec<Token>| {
404    if grouped {
405      let mut out = vec![v[0]];
406      out.extend(tokens);
407      out.push(v[v.len() - 1]);
408      out
409    } else {
410      tokens
411    }
412  };
413  let mut address = inner[..lead].to_vec();
414  address.extend_from_slice(tail);
415  Some((wrap(mark.to_vec()), Tokens::new(wrap(address))))
416}

418/// The content of `content` inside the one-argument wrapper `cs` round all of it (`\lx@affiliation@withsup{…}`), and
419/// whether it was.
420fn within_wrapper(content: &Tokens, cs: &str) -> (bool, Tokens) {
421  let v = content.unlist_ref();
422  if v.len() >= 3
423    && v[0] == T_CS!(cs)
424    && v[1].get_catcode() == Catcode::BEGIN
425    && skip_group(v, 1) == v.len()
426  {
427    (true, Tokens::new(v[2..v.len() - 1].to_vec()))
428  } else {
429    (false, content.clone())
430  }
431}

433/// The content of a contact inside the affiliation's own wrapper (`\lx@affiliation@withinst{…}`, which
434/// `\lx@add@affiliation` puts around it), and whether it was in it.
435pub(super) fn within_inst(content: &Tokens) -> (bool, Tokens) {
436  let within = content.unlist_ref();
437  let withinst = within.len() >= 3
438    && within[0] == T_CS!("\\lx@affiliation@withinst")
439    && within[1].get_catcode() == Catcode::BEGIN
440    && skip_group(within, 1) == within.len();
441  if withinst {
442    (true, Tokens::new(within[2..within.len() - 1].to_vec()))
443  } else {
444    (false, content.clone())
445  }
