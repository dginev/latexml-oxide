9pub(in crate::base_utilities) fn load() -> Result<()> {
10  //======================================================================
11  // LaTeX has a very particular, but useful, notion of "Undefined",
12  //    so let's get that squared away at the outset; it's useful for TeX, too!
13  //
14  // Naturally, it uses \csname to check, which ends up DEFINING the possibly undefined macro as
15  // \relax
16  // Perl Base_Utility.pool.ltxml L23-31
17  DefMacro!("\\lx@ifundefined{}{}{}", sub[(name, if_token, else_token)] {
18    let cs = T_CS!(s!("\\{}", Expand!(name).to_string()));
19    // Autoload triggers (declared via `def_autoload` at tex.rs:238-247)
20    // install a closure under the trigger CS so the package auto-loads on
21    // first invocation. Perl scopes the equivalent `DefAutoload` entries
22    // to `OmniBus.cls.ltxml`, so for non-OmniBus papers Perl sees these
23    // CSes as truly undefined. Mirror Perl by treating an unfired
24    // autoload trigger as "undefined" in `\@ifundefined`. We must NOT
25    // overwrite the trigger CS with `\relax` in this case (the kernel
26    // `\csname X\endcsname \ifx \relax` idiom DOES overwrite, but doing
27    // so here would destroy the autoload — subsequent use of the trigger
28    // CS would no-op instead of loading its package). Driver:
29    // arXiv:2507.23241v1 (smfart.cls) — line 373's
30    // `\@ifundefined{numberwithin}` branches wrong when our preloaded
31    // `\numberwithin` autoload makes the CS "look" defined, then the
32    // `\@gobbletwo` branch eats `\ifx \relax` and orphans `\else` / `\fi`.
33    // An autoload TRIGGER counts as "undefined" only while UNFIRED — i.e. its
34    // target package has not yet loaded. `def_autoload` stores the package name
35    // (`.sty`) as a String; once `<pkg>.sty_loaded`/`_raw_loaded` is set the
36    // trigger CS holds the package's real definition and must read as DEFINED
37    // (fixes `\@ifundefined{align}` after `\usepackage{amsmath}`). `.pool`
38    // triggers keep the legacy Bool form and stay "undefined until used".
39    let is_autoload = cs.with_cs_name(|cs_name| {
40      let flagged = match lookup_value(&s!("{cs_name}:autoload")) {
41        Some(Stored::String(pkg_sym)) => {
42          let pkg = with(pkg_sym, |s| s.to_string());
43          !load_recorded(&s!("{pkg}.sty"))
44        },
45        Some(Stored::Bool(b)) => b,
46        _ => false,
47      };
48      // The flag is only meaningful while the CS still HOLDS the trigger
49      // definition. The kernel dump redefines some trigger CSes itself
50      // (`\ProvidesExplPackage` et al. are real latex.ltx macros) — then
51      // the CS is genuinely defined regardless of the package flag. Compare
52      // definition identity against the snapshot def_autoload stored.
53      flagged
54        && match lookup_value(&s!("{cs_name}:autoload_trigger")) {
55          Some(Stored::Expandable(trigger)) => with_meaning(&cs, |m| {
56            matches!(m, Some(Stored::Expandable(current)) if Rc::ptr_eq(current, &trigger))
57          }),
58          None => true, // legacy trigger without snapshot: old behavior
59          _ => false,
60        }
61    });
62    if IsDefined!(&cs) && !is_autoload {
63      Ok(else_token)
64    } else {
65      // No `\relax` pollution of the probed name. latex.ltx:1729-1737 defines
66      // `\@ifundefined` with `\ifcsname` precisely so a probe leaves the name
67      // undefined (the `\csname…\endcsname\relax` form at :1738 is only the
68      // pre-`\ifcsname` fallback, and Perl Base_Utility.pool:23-31 still
69      // pollutes). polyglossia's gloss-*.ldf:591 `\@ifundefined
70      // {initiate@active@char}{\input{babelsh.def}}{}` then hands babelsh.def:1
71      // `\ifx\initiate@active@char\@undefined\else\bbl@afterfi\endinput\fi` a
72      // `\relax`-valued name, so the shorthand file early-exits with
73      // `\bbl@afterfi` undefined (hang, sample; 19 gloss files load it this
74      // way). Guard: `perfect_kernel_batch54::ifundefined_does_not_define_the_name`.
75      Ok(if_token)
76    }
77  }, locked=>true);
78  // \@ifundefined is a LaTeX-kernel macro, but our amsppt / amssymb
79  // bindings invoke it from Plain-TeX/AMSTeX context too — and those
80  // contexts don't load latex_constructs (which has the same Let).
81  // Surfacing it here makes it available regardless of which constructs
82  // pool is loaded. Surpasses Perl's Base_Utility.pool.ltxml gap.
83  Let!("\\@ifundefined", "\\lx@ifundefined");
84
85  // Dash and space primitives used by ligatures and other mechanisms.
86  // Perl Base_Utility.pool.ltxml L44-45
87  DefPrimitive!("\\lx@endash", {
88    Tbox::new(
89      pin!("\u{2013}"),
90      None,
91      None,
92      Tokens!(T_CS!("\\lx@endash")),
93      SymHashMap::default(),
94    )
95  });
96  // Perl Base_Utility.pool.ltxml L46-47
97  DefPrimitive!("\\lx@emdash", {
98    Tbox::new(
99      pin!("\u{2014}"),
100      None,
101      None,
102      Tokens!(T_CS!("\\lx@emdash")),
103      SymHashMap::default(),
104    )
105  });
106  // Perl Base_Utility.pool.ltxml L50-52: stand-in for T_ACTIVE('~').
107  DefPrimitive!("\\lx@NBSP", {
108    Tbox::new(
109      pin!("\u{00A0}"),
110      None,
111      None,
112      Tokens!(T_ACTIVE!('~')),
113      // `~` is a penalty and glue (latex.ltx:9411-9419): `\unskip` removes the glue.
114      stored_map!("isSpace" => true, "isSkip" => true, "width" => Dimension::from_str("0.333em")?),
115    )
116  }, locked => true);
117  // Perl Base_Utility.pool.ltxml L53-55
118  DefPrimitive!("\\lx@nobreakspace", {
119    Tbox::new(
120      pin!("\u{00A0}"),
121      None,
122      None,
123      Tokens!(T_CS!("\\lx@nobreakspace")),
124      // `~` is a penalty and glue (latex.ltx:9411-9419): `\unskip` removes the glue.
125      stored_map!("isSpace" => true, "isSkip" => true, "width" => Dimension::from_str("0.333em")?),
126    )
127  });

129  // Perl Base_Utility.pool.ltxml L57-65
130  DefPrimitive!("\\lx@ignorehardspaces", {
131    let mut boxes = Vec::new();
132    while let Some(token) = read_x_token(None, false, None)? {
133      boxes = invoke_token(&token)?;
134      if boxes.is_empty() {
135        break;
136      }
137      while !boxes.is_empty() {
138        if match boxes[0].get_property("isSpace") {
139          Some(Cow::Borrowed(Stored::Bool(space_bool))) => *space_bool,
140          Some(Cow::Owned(Stored::Bool(ref space_bool))) => *space_bool,
141          _ => false,
142        } {
143          boxes = boxes[1..].to_vec();
144        } else {
145          break;
146        }
147      }
148      if !boxes.is_empty() {
149        break;
150      }
151    }
152    Ok(boxes)
153  });

158  });

168      let key = args[0].as_ref().map(ToString::to_string).unwrap_or_default();
169      let value = args[1].as_ref().map(ToString::to_string).unwrap_or_default();

171        document.set_attribute(&mut element, &key, &value)?;
172      }
173    }, sizer => 0);

175  // Perl Base_Utility.pool.ltxml (PR #2767): split #3 on the delimiter
176  // tokens in #2, invoking #1 on each piece.
177  // DefMacro('\lx@splitting{}{}{}', ...)
178  DefMacro!("\\lx@splitting{}{}{}", sub[(op, delimiters, tokens)] {
179    // Note: Perl tests `$delimiters ?` — a Tokens object is always truthy,
180    // so the split always applies (an empty delimiter list yields one piece).
181    let delims: Vec<SplitDelim> = delimiters.unlist().into_iter().map(SplitDelim::from).collect();
182    Ok(split_calls(&op, split_tokens_delimited(tokens, delims)))
183  });
184  // Rust-only (Option C P2-2, OXIDIZED_DESIGN_DIVERGENCES #995): `\lx@splitting@by@meaning{op}{delims}{tokens}` —
185  // `\lx@splitting`, cut first at the kernel's `\and` by meaning, then each segment at the delimiters by spelling
186  // (`split_at_kernel_separators`: a macro holding only `\and`, a `\let` copy of the kernel's `\and` where the document
187  // renewed `\and`; no token rewritten), one `op` per piece; inst_support's `\institute` list and llncs's marked `\author`
188  // list (P2-3).
189  DefMacro!("\\lx@splitting@by@meaning{}{}{}", sub[(op, delimiters, tokens)] {
190    let delims: Vec<SplitDelim> = delimiters.unlist().into_iter().map(SplitDelim::from).collect();
191    let (pieces, _) = split_at_kernel_separators(tokens, delims)?;
192    Ok(split_calls(&op, pieces))
193  });

195  // Rust-only (OXIDIZED_DESIGN_DIVERGENCES #805): `\lx@split@addresses{op}{list}` — `op{address}` for each address of a
196  // class's address list (jheppub's `\emailAdd{a@x, b@y}`, iopart's `\ead{a@x and b@y}`); a list in plain text that
197  // is a list by `email_addresses` (a braced group of local parts before a shared domain, `\ead{\{yangqq, yangjh\}
198  // @ipp.ac.cn}`, 2607.04241; a semicolon list) each of its addresses written out, as an author block's line gives
199  // them, not "{yangqq" and "yangjh}@ipp.ac.cn"; `op{list}` whole when it is no list of addresses (2011.05971's
200  // `\emailAdd{nikolay.bobev, anthony.charles, valentin.reys @kuleuven.be}`, the bare local parts no address).
201  DefMacro!("\\lx@split@addresses{}{}", sub[(op, list)] {
202    // (a brace set in math is the brace it prints: `$\{$a, b$\}$@x`, `math_braces_as_printed`, OD #972)
203    let list = math_braces_as_printed(&list);
204    // (at the commas and at an " and ": `\ead{cpires@fisica.ufpb.br and psilva@fisica.ufpb.br}`, 0710.2104, 1003.0073)
205    let pieces: Vec<Tokens> = split_tokens(list.clone(), vec![
206      SplitDelim::Token(T_OTHER!(",")),
207      SplitDelim::Tokens(Tokens::new(vec![
208        T_SPACE!(),
209        T_LETTER!("a"),
210        T_LETTER!("n"),
211        T_LETTER!("d"),
212        T_SPACE!(),
213      ])),
214    ])
215    .into_iter()
216    .filter(|piece| !piece.is_empty())
217    .collect();
218    let addresses = pieces.iter().all(|piece| address_text(piece).contains('@'));
219    // (only a list of plain addresses is written out — their letters, the printed braces `\{ \}` (`\lbrace`, `\textbraceleft`) and `\_`, the spacing
220    // and line breaks between them and an e-mail command's address font, `\mailto` (1105.4742's `\ead{\mailto{idan.oren@…}
221    // \\ \mailto{\ \ uzy.smilansky@…}}`): a braced group of local parts, a semicolon list, 0711.4000's `leticia@…;
222    // jorge@…` —, so a list in other markup keeps it: `\href{mailto:a@x,b@y}{a@x, b@y}` one contact, rev78f probe j06)
223    let plain = list.unlist_ref().iter().all(|t| {
224      t.get_catcode() != Catcode::CS
225        || is_spacing_command(t)
226        || is_line_separator(t)
227        || is_email_wrapper(t)
228        || t.with_str(|name| {
229          matches!(
230            name,
231            "\\{" | "\\}" | "\\_" | "\\lbrace" | "\\rbrace" | "\\textbraceleft" | "\\textbraceright" | "\\,"
232          )
233        })
234    });
235    let pieces = if pieces.len() > 1 && addresses {
236      pieces
237    } else {
238      // (a shared domain's addresses each written out, as an author block's line gives them, `email_line_calls`)
239      match email_addresses(&list) {
240        // (each written as the characters it prints, an underscore as the author block's line writes it, `\_`: a bare
241        // `_` prints the OT1 dot accent, `scs˙lzg@tju.edu.cn` for 2304.11989's `\{scs$\_$lzg, …\}@tju.edu.cn`;
242        // `address_tokens`; the others, a Semiverbatim `\emailAdd`'s `~` or `&` among them, as they are)
243        Some(each) if each.len() > 1 && plain => each
244          .iter()
245          .map(|address| {
246            Tokens::new(
247              text_tokens(address)
248                .into_iter()
249                .map(|t| if t == T_OTHER!("_") { T_CS!("\\_") } else { t })
250                .collect(),
251            )
252          })
253          .collect(),
254        _ => vec![list],
255      }
256    };
257    let mut result: Vec<Token> = Vec::new();
258    for piece in pieces {
259      result.extend(op.unlist_ref().iter().copied());
260      result.push(T_BEGIN!());
261      result.extend(piece.unlist());
262      result.push(T_END!());
263    }
264    Ok(Tokens::new(result))
265  });
266  Ok(())
267}

269/// `op{piece}` for each piece of a split list, each paired with the delimiter before it (`split_tokens_delimited`,
270/// `split_at_kernel_separators`): `\lx@splitting`'s expansion, and `\lx@splitting@by@meaning`'s, aa's
271/// (`aa_cls.rs` `\lx@aa@institute@dispatch`) and llncs's (`llncs_cls.rs` `\lx@llncs@institute@dispatch`).
272pub fn split_calls(op: &Tokens, pieces: DelimitedPieces) -> Tokens {
273  let mut result: Vec<Token> = Vec::new();
274  let line_break = T_CS!("\\\\");
275  for (delimiter, piece) in pieces {
276    // Rust-only (OXIDIZED_DESIGN_DIVERGENCES #804): a piece after a `\\` delimiter (a split at line breaks, which no
277    // Perl binding makes: llncs's `\inst` list; webofc's `\author` until P2-1, now one block the reading cuts, OD #993)
278    // starts without the break's `*` and `[…]`, which are no text (`B\inst{2}\\*[2mm] C\inst{1}` made a creator
279    // "*[2mm] C"; the rev78e review's finding 6)
280    let piece = if delimiter == [line_break] {
281      let with_break: Vec<Token> = std::iter::once(line_break).chain(piece.unlist()).collect();
282      let mut stripped = strip_linebreak_options(Tokens::new(with_break)).unlist();
283      stripped.remove(0);
284      while stripped.first() == Some(&T_SPACE!()) {
285        stripped.remove(0);
286      }
287      Tokens::new(stripped)
288    } else {
289      piece
290    };
291    result.extend(op.unlist_ref().iter().copied());
292    result.push(T_BEGIN!());
293    result.extend(piece.unlist());
294    result.push(T_END!());
295  }
296  Tokens::new(result)
