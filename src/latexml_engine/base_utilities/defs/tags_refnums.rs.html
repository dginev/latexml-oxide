6pub(in crate::base_utilities) fn load() -> Result<()> {
7  //======================================================================
8  // Tags & Titles
9  // The reference numbers, titles, captions etc, for various objects have
10  // different styling conventions, and the styling various depending on context.
11  // We'll use ltx:tags as a container for the various forms of ltx:tag with different @role's.
12  // The role=refnum form is simply formatted by \the<counter> and used by \ref;
13  // An ltx:tag w/o @role are for the numbers, often formatted differently, which
14  // appear alongside the object; Such a tag also may be embedded within the title or caption.
15  // Cross-references automatically generated by LaTeXML benefit from a bit more context:
16  // these are the role=typerefnum forms.
17  // Additional forms are needed for bibliographies, hyperref's autoref, etc.
18  // An additional complication is that while the "type" determines the formatting
19  // of the various forms, some types (eg. theorems) share the same counter.
20  // LaTeX defines this handling on an adhoc basis; defines \fnum@table, \fnum@figure for some types
21  // but \labelenumi, etc for others.
22
23  // This section synthesizes a more uniform support for reference numbers,
24  // references to reference numbers, title formatting etc.
25  // It allows you to customize each of the forms for each type encountered.
26  // The design reflects LaTeX needs, more than TeX, but support starts here!
27
28  // This collects up the various declared ltx:tag's into an ltx:tags
29  DefMacro!("\\lx@make@tags {}", sub[(ttype)] {
30    let mut lx_tags = vec![T_CS!("\\lx@tags"), T_BEGIN!()];
31    for (_printed_here, form) in tag_forms(&ttype)? {
32      lx_tags.extend(form.unlist());
33    }
34    lx_tags.push(T_END!());
35    Ok(Tokens::new(lx_tags))
36  });
37  DefMacro!("\\lx@tag@texttilde", "\\lx@NBSP", protected => true);
38
39  // Remove the last closed node, if it's empty.
40  let remove_empty_element: Vec<ConstructionClosure> = construct!(document, _whatsit, {
41    if let Some(node) = document.get_node().get_last_child() {
42      // This should be the wrapper just added.
43      if node.get_first_child().is_none() {
44        document.remove_node(node);
45      }
46    }
47  });
48
49  // \lx@tag[open][close]{stuff}
50  let remove_empty_element_1 = remove_empty_element.clone();
51  DefConstructor!("\\lx@tag[][][]{}", "<ltx:tag open='#1' close='#2'>#4</ltx:tag>",
52    mode => "restricted_horizontal",
53    after_construct => remove_empty_element_1
54  );

56  // \lx@tag@intags{role}{stuff}
57  // The role is an identifier (`refnum`, `autoref`, …), read undigested — Perl digests `[]` (Base_Utility.pool.ltxml:
58  // 1001) after `neutralizeFont`, whose OT1 leaves ASCII alone; the neutral font here keeps the document's encoding
59  // (`neutralize_font`), and under babel greek's LGR (`\selectlanguage{greek}`, greek.ldf:151-153) a digested `refnum`
60  // read back `ρεφνυμ`. Guard `perfect_kernel_batch60::identifiers_are_not_font_decoded`.
61  // (an empty tag holding diagnostics, `_deferred`, stays until its target, or the document's end, ties them to its
62  // labels: `note_empty_held_tag`)
63  DefConstructor!("\\lx@tag@intags OptionalUndigested {}", "<ltx:tag role='#1' _deferred='#deferred'>#2</ltx:tag>",
64    mode => "restricted_horizontal",
65    before_digest => sub { neutralize_font(); },
66    after_construct => sub[document, _whatsit] {
67      if let Some(node) = document.get_node().get_last_child()
68        && node.get_first_child().is_none()
69      {
70        if node.has_attribute("_deferred") {
71          note_empty_held_tag();
72        } else {
73          document.remove_node(node);
74        }
75      }
76    }
77  );
78  // A tag typeset only where a reference prints it — hyperref's `autoref` name (`type_tag_deferred`): TeX expands
79  // `\<type>autorefname` only in `\autoref` (hyperref.sty:8202-8278), so the name's diagnostics belong there. Built
80  // here, at the target (ruling 2026-10-02 (A)), its diagnostics are held and deferred (ruling 7e,
81  // logger.rs `DeferredDiagnostics`): the tag carries their id (`_deferred`), the labelled node ties it to its labels
82  // as it closes (sect11.rs `tie_held_tags`), and they are replayed once the document is built if a reference shows
83  // the tag's role (`replay_shown_deferred`). Witness biblatex-gost-examples (hyperref's Russian `\cyr…` names under TU).
84  // Repro singletons/autoref_name_evaluated_at_every_target.
85  DefPrimitive!("\\lx@tag@intags@held[]{}", sub[(role, stuff)] {
86    let role = role.unwrap_or_default();
87    let hold = util::logger::DiagnosticsHold::begin();
88    // (the tag's text is not the document's input: digested in a group of its own, which undoes what it leaves open,
89    // the alignment brace count too; a text that does not close what it opens errs with the tag, as at a reference)
90    let (tag, balance) = digest_in_own_group(|| {
91      digest(Invocation!(T_CS!("\\lx@tag@intags"), vec![role.clone(), stuff]))
92    })?;
93    let tag = tag?;
94    // (one Error per text: a text whose own typesetting already erred — "Attempt to end mode", the `}` TeX inserts —
95    // raises no summary of the same fault, as `\lx@cref@docformat` does; guard
96    // `perfect_kernel_batch78j::autoref_name_opening_a_group_errs_once`)
97    if balance != 0 && hold.errors_raised() == 0 {
98      Error!(
99        "unexpected",
100        role.to_string(),
101        s!("The text of the {role} tag does not close the groups it opens"),
102        get_source_location().unwrap_or_default()
103      );
104    }
105    // (a tag whose typesetting raised only Debug messages, which are counted but never written, holds nothing: the
106    // plain label format's empty `creflabel` tag is removed where it is built)
107    if hold.is_empty_but_debug() {
108      hold.commit();
109    } else if let Some(whatsit) = held_tag_whatsit(&tag)
110      && let DigestedData::Whatsit(w) = whatsit.data()
111    {
112      if let Some(deferred) = hold.defer() {
113        let id = util::logger::keep_deferred(deferred);
114        w.borrow_mut().set_property("deferred", Stored::String(pin(id.to_string())));
115      }
116    } else {
117      // A tag with no whatsit to carry the id raises its diagnostics here: never lost.
118      hold.commit();
119    }
120    Ok(vec![tag])
121  });
122  DefConstructor!("\\lx@tags{}","<ltx:tags>#1</ltx:tags>",
123    after_construct => remove_empty_element
124  );

148  // `\expandafter` before the first `\endcsname`: ctex's `\labelformat`
149  // makes `\p@section` ARGUMENT-TAKING (`\p@section#1->\CTEX@thesection`,
150  // ctex-heading-article.def:747) and patches every kernel `\p@#1\the#1`
151  // site to that shape (:770-771) — a site it cannot reach here let
152  // `\p@section` swallow the following `\csname` ("Extra \endcsname" on
153  // every heading: caspervector 23, sduthesis 36, tabular2 28, inkpaper-en;
154  // Perl Base_Utility.pool.ltxml:1028 identical, KPE #149). Guard:
155  // `perfect_kernel_batch54::ctex_argument_taking_p_macro_keeps_the_refnum`.
156  DefMacro!("\\lx@@therefnum@@{}", "{\\normalfont\\lx@@p@the@@{#1}}");
157  // `\p@<ctr>\the<ctr>` in latex.ltx:14976's exact shape, resolved through
158  // `counter_for_type` like `\lx@the@@`. Shared by the refnum and typerefnum
159  // formatters so an argument-taking `\p@<ctr>` (`\labelformat`, latex.ltx:14978)
160  // always receives the single `\the<ctr>` token: the typerefnum's former
161  // `\csname p@#1\endcsname\lx@the@@{#1}` handed it `\lx@the@@` and left
162  // `{sentence}` behind ("You can't use } after \the" on every contract.sty
163  // sentence; Perl Base_Utility.pool.ltxml:1080-1084 identical, KPE #160).
164  // Guard: `perfect_kernel_batch54::labelformat_is_a_kernel_macro`.
165  DefMacro!(
166    "\\lx@p@the@@{}",
167    "\\expandafter\\lx@@p@the@@\\expandafter{\\lx@counterfor{#1}}"
168  );
169  // The form is what latex.ltx:14963-14964 `\refstepcounter` stores in `\@currentlabel` and a `\ref` prints:
170  // `\protected@edef` of `\csname p@<ctr>\expandafter\endcsname\csname the<ctr>\endcsname`, so `\p@<ctr>` is
171  // applied once, to the expanded `\the<ctr>`, inside the definition's braces. Typed in place (Perl
172  // Base_Utility.pool.ltxml:1028), a `\p@equation` taking nine arguments (`\renewcommand{\p@equation}
173  // {\gobblesomeargs}`, a robust command, 1902.09345) read past the form's group: 18 errors where pdflatex has 0;
174  // protected, it stays, and gobbles "Equation" of the expanded `\theequation` as the `\ref` does ("S1").
175  // Only a `\p@<ctr>` that does something is applied so: with the kernel's empty one (`\@definecounter`, latex.ltx:10144) the
176  // stored text is `\the<ctr>`'s expansion, which is typeset the same, and the number is typed as written — an
177  // itemize's `\tikz` bullet (a number LaTeX never stores) does not survive an `\edef`.
178  // OXIDIZED_DESIGN_DIVERGENCES #778; guards `perfect_kernel_batch77u::{p_form_applies_to_the_expanded_number,
179  // p_forms_of_the_kernel_keep_their_reference}`, `perfect_kernel_batch56::itemize_label_is_used_as_written`.
180  DefMacro!("\\lx@@p@the@@{}", sub[(ctr)] {
181    let ctr = ctr.to_string();
182    let mut form = vec![T_CS!("\\csname")];
183    form.extend(Explode!(s!("p@{ctr}")));
184    form.extend([T_CS!("\\expandafter"), T_CS!("\\endcsname"), T_CS!("\\csname")]);
185    form.extend(Explode!(s!("the{ctr}")));
186    form.push(T_CS!("\\endcsname"));
187    let form = Tokens::new(form);
188    let applies = match lookup_definition(&T_CS!(s!("\\p@{ctr}")))? {
189      Some(defn) => !(defn.is_expandable()
190        && defn.get_parameters().is_none_or(|params| params.get_num_args() == 0)
191        && match defn.get_expansion() {
192          None => true,
193          Some(ExpansionBody::Tokens(body)) => body.is_empty(),
194          Some(_) => false,
195        }),
196      None => false,
197    };
198    if applies {
199      with_unexpandable_protect(|| do_expand_partially(form))?
200    } else {
201      form
202    }
203  });

218  // The `{}` after `\endcsname` is OXIDIZED_DESIGN #85, a deliberate divergence
219  // from Perl's otherwise byte-identical `Base_Utility.pool.ltxml` L1041-1043.
220  // LaTeX's `\@makecaption` is `\sbox\@tempboxa{#1: #2}`, so the widely-copied
221  // "Fig. 1: -> Fig. 1." hack —
222  //   \renewcommand*{\fnum@figure}[1]{\figurename~\thefigure.}
223  // — works under pdflatex by having the hook eat that `:` TOKEN. LaTeXML has
224  // no `:` to eat: the separator is a tag ATTRIBUTE (`\lx@tag[][: ]`,
225  // `latex_constructs.rs::format@title@figure`). So the argument scan ran past
226  // the hook and swallowed the caption group's closing brace — the `<figure>`
227  // never closed and every following section, bibliography included, was
228  // absorbed into it. The empty group gives an arg-taking hook something
229  // harmless to consume and is inert for the 0-arg hooks that are the normal
230  // case. The `\lx@@fnum@@` branch below — what fires when no `\fnum@<type>`
231  // exists at all, i.e. for nearly every caption — is untouched.
232  // The `fnum@font@<type>` value WRAPS the number as a braced argument
233  // (real enumitem: `\enit@format{<label>}`, enumitem.sty:451/1478) — a bare
234  // prefix let an argument-taking font command (`\textbf`, `\meta`,
235  // tcbdocumentation's `\docAuxKey`; `\setlist[description]{font=…}`) grab
236  // the following `\@ifundefined` instead (non-decimal-units manual, 40
237  // LR-mode/boxing errors; Perl Base_Utility.pool:1041 shares the bare
238  // prefix). The group is transparent for the declaration-style values.
239  DefMacro!(
240    r"\lx@fnum@@{}",
241    r"{\normalfont\@ifundefined{fnum@font@#1}{}{\csname fnum@font@#1\endcsname}{\@ifundefined{fnum@#1}{\lx@@fnum@@{#1}}{\csname fnum@#1\endcsname{}}}}"
242  );

395    sub[document, args, _props] {
396      if let Some(Some(preamble_arg)) = args.first() {
397        let preamble_text = preamble_arg.untex()?;
