30/// One whole block queued for the call (OD #980), as no reader cuts it at the call, which the emitter
31/// takes at the flush (`observer::emitter::emit`): Perl's whole-block creator (`fallback::whole_block_author`). A `late`
32/// call, made while a flush digests its queue after reading it (the ledger's `AuthorBlock::late`), queues it for a flush
33/// of its own (`\lx@frontmatter@late`): Perl's `\lx@add@authors` there runs `@now`, split at the call
34/// (Base_Utility.pool.ltxml:799-806).
35fn whole_block_call(
36  stuff: Tokens,
37  replace: bool,
38  keyvals: Option<Tokens>,
39  late: bool,
40) -> Result<Tokens> {
41  if replace {
42    replace_queued_authors();
43  }
44  let whole = whole_block_author(keyvals, stuff)?;
45  if late {
46    return Ok(Invocation!(T_CS!("\\lx@frontmatter@late"), vec![Some(
47      whole
48    )]));
49  }
50  Ok(whole)
51}

53pub(in crate::base_utilities) fn load() -> Result<()> {
54  // This corresponds to standard LaTeX,
55  // The command replaces any previous authors/creators;
56  // It defining several creators separated by \and;
57  // and can use \\ to separate affiliation from each author.
58  // BUT ALSO, this markup is commonly abused by putting authors & affiliations in
59  // seemingly random orders, but adding superscript markers to connect them.
60  // Assumption: superscript near END is used for authors; near FRONT for affiliation.
61  // NOTE: This is a mess! really should use role, so could apply to editors also
62  // AND, matching \\ this way fails to catch \\[1em], so really should Let it
63
64  // 56fl (OXIDIZED_DESIGN #253) — the raw-class `\author` surplus. `\author` is
65  // locked to the kernel shape `[short]{name}` (Perl latex_constructs.pool.ltxml
66  // :1076 too), so a raw class's own redefinition is refused and recorded as
67  // `\author:redefined` (state.rs `install_definition`). Such a class declared
68  // MORE arguments — a trailing `[keyval]` (cas-common.sty:895 `O{} m O{}`:
69  // `[type=editor, orcid=…]`) or a second mandatory `{affiliation}`
70  // (cnbwp.cls:273 `\def\author{\@ifnextchar[…}` → `\CNB@authorLong#1#2`) —
71  // which the kernel shape left in the stream to typeset as a `<para>` before
72  // the frontmatter (schema-invalid, faux content; Perl leaks identically).
73  // Running the class body instead would LOSE the authors: cas-common/cnbwp/
74  // aomart store names in class-private accumulators laid out only by their
75  // own (locked) `\maketitle` (verified: zero <creator>). So the locked
76  // `\author` keeps the semantics and, ONLY when a class redefined it, (a)
77  // absorbs the trailing argument into the frontmatter API — `orcid=` →
78  // `ltx:contact role=orcid`, a trailing group → `role=affiliation`, other
79  // keys (type, auid, bioid, prefix, suffix, role, style) are presentational
80  // and dropped — and (b) APPENDS creators (`\lx@add@author`) instead of the
81  // LaTeX-faithful dequeue-replace, since such a class calls `\author` once
82  // per author (cnbwp: 3 creators, each with its affiliation). A document that
83  // never redefined `\author` is byte-identical: `\@ifnextchar` is never even
84  // reached. Twins carrying the tail: kernel `\author` (sect05.rs),
85  // inst_support, sv_support, llncs. Residual risk, documented: with a same-
86  // shape class redefinition, a brace group immediately after `\author{…}` on
87  // the same paragraph is read as the class's affiliation argument (`\par`
88  // stops `\@ifnextchar`).
89  DefMacro!("\\lx@add@authors@adaptive{}", sub[(stuff)] {
90    let replace = !lookup_bool("\\author:redefined");
91    let ijcai = ijcai_block(&stuff)?;
92    let late = record_author_call("\\lx@add@authors@adaptive", &stuff, replace, ijcai, None, false);
93    if ijcai {
94      return ijcai_split_call(stuff, replace, late);
95    }
96    whole_block_call(stuff, replace, None, late)
97  });
98  // The late block's flush (`whole_block_call`): its whole entry queued — the flush digesting it let the API to `@now`,
99  // locally — then read by a flush of its own (re-entrant, `digest_front_matter`), which emits its split creators or
100  // writes it whole with its Warning, digested here into the frontmatter (1502.01001, 0807.4720, astro-ph9808358).
101  DefPrimitive!("\\lx@frontmatter@late{}", sub[(whole)] {
102    bgroup();
103    let_i(&T_CS!("\\lx@add@frontmatter"), &T_CS!("\\lx@frontmatter@queue@add"), None);
104    let_i(&T_CS!("\\lx@annotate@frontmatter"), &T_CS!("\\lx@frontmatter@queue@annotate"), None);
105    let queued = digest(whole);
106    let closed = egroup();
107    let queued = queued?;
108    closed?;
109    digest_front_matter()?;
110    Ok(vec![queued])
111  });
112  DefMacro!("\\lx@author@trailing", sub[()] {
113    if lookup_bool("\\author:redefined") {
114      Ok(TokenizeInternal!(
115        r"\@ifnextchar[{\lx@author@trailing@opt}{\@ifnextchar\bgroup{\lx@author@trailing@mand}{}}"
116      ))
117    } else {
118      Ok(Tokens!())
119    }
120  });
121  DefMacro!(
122    "\\lx@author@trailing@opt[]",
123    r"\lx@add@author@keyvals{#1}\lx@author@trailing"
124  );
125  // An author keyval block (`type=editor, auid=000, orcid=0000-…`, the
126  // cas-common `stm/author` key family, .sty:686) → frontmatter contacts:
127  // `orcid=` is identity (`ltx:contact role=orcid`); `type`, `auid`, `bioid`,
128  // `alt`, `style`, `prefix`, `suffix`, `role` are layout/production keys
129  // and are dropped. Shared by the raw-class tail above and the cas-dc/cas-sc
130  // binding's `\author[marks]{name}[keyvals]` (cas_dc_cls.rs).
131  DefMacro!("\\lx@add@author@keyvals{}", sub[(keyvals)] {
132    let mut out: Vec<Token> = Vec::new();
133    let keyvals = keyvals.to_string();
134    for item in keyvals.split(',') {
135      if let Some((key, value)) = item.split_once('=')
136        && key.trim() == "orcid"
137      {
138        let value = value.trim().trim_matches(|c| c == '{' || c == '}').trim();
139        if !value.is_empty() {
140          let value = mouth::tokenize(TeXString::assembled(value.to_string()));
141          out.extend(Invocation!(T_CS!("\\lx@add@orcid"), vec![None, Some(value)]).unlist());
142        }
143      }
144    }
145    Ok(Tokens::new(out))
146  });
147  DefMacro!(
148    "\\lx@author@trailing@mand{}",
149    r"\lx@add@affiliation{#1}\lx@author@trailing"
150  );

152  DefMacro!("\\lx@add@authors{}", sub[(stuff)] {
153    let ijcai = ijcai_block(&stuff)?;
154    let late = record_author_call("\\lx@add@authors", &stuff, true, ijcai, None, false);
155    if ijcai {
156      return ijcai_split_call(stuff, true, late);
157    }
158    whole_block_call(stuff, true, None, late)
159  });
160  // A class's own author command, called once per author or author group and each adding to the ones before
161  // (informs4.cls:1198 `\AUTHOR{Ann Able$^{a}$, Bob Baker$^{b}$}`): the author-line parsing of `\lx@add@authors`
162  // without its replacing, so the names' superscript marks request the affiliations labelled with them.
163  // A per-author class's `\author` (amsart, aastex, revtex) parses its body as one more author block too: a name list
164  // there names several people (`\author{Klemens Fellner and Bao Quoc Tang}`, 1708.01427; aastex's
165  // `Name\altaffilmark{1}, Name\altaffilmark{2}`, 1010.1318), and a `\\` line under a name is its affiliation
166  // (`\author{Masao Ishikawa\\ \small Faculty of Education, …}`, math0606082). The optional keyvals go to each author
167  // (revtex's `annotations=`).
168  DefMacro!("\\lx@add@authors@append[]{}", sub[(keyvals, stuff)] {
169    let late = record_author_call("\\lx@add@authors@append", &stuff, false, false, keyvals.as_ref(), true);
170    whole_block_call(stuff, false, keyvals, late)
171  });
172  // A class's own author-list command (informs4.cls:1198 `\AUTHOR{Ann Able$^{a}$, Bob Baker$^{b}$}`, documented as a
173  // name list): appended like `\lx@add@authors@append`, but split at every separator as the article `\author` is.
174  // (where the list starts in the queue: the row an address shared by its authors falls back to, `queued_row_start`)
175  DefMacro!("\\lx@add@authors@list{}", sub[(stuff)] {
176    assign_value("lx_authors_list_start", queued_author_positions()? as i64, Some(Scope::Global));
177    let ijcai = ijcai_block(&stuff)?;
178    let late = record_author_call("\\lx@add@authors@list", &stuff, false, ijcai, None, false);
179    if ijcai {
180      return ijcai_split_call(stuff, false, late);
181    }
182    whole_block_call(stuff, false, None, late)
183  });
184  // A name list where a binding has no lines or affiliations to read: each name the reading finds (comma,
185  // " and ", " \& ") is an author of its own, the keyvals given to each (`\author{R. Braun\inst{1} and W. B.
186  // Burton\inst{2}}`, astro-ph9810433; `\IEEEauthorblockN{Clemens Paul Zengler, Niels Troldborg, Mac Gaunaa.}`,
187  // 2410.19527). A collaboration the list names beside its names is a contact on them, no part of a name (batch 103b:
188  // webofc's `… on behalf of the KLOE-2 Collaboration` after the notes, 1502.05517). Queued whole, as the emitter reads it.
189  DefMacro!("\\lx@add@author@split[]{}", sub[(keyvals, names)] {
190    let late = record_author_call("\\lx@add@author@split", &names, false, false, keyvals.as_ref(), false);
191    whole_block_call(names, false, keyvals, late)
192  });
193  // That class's affiliation command, under its authors (informs4.cls:1199 `\AFF{$^a$Univ A}`): one affiliation call,
194  // placed as the optional attributes say (apa7's unnumbered list, `annotate=all`); a marked list, the API's Cut cuts as
195  // the binding declares it (`marked_list_calls`), each mark shown and its entry the requesters' (2609.17368, 22690, 28084).
196  DefMacro!("\\lx@add@affiliation@marked[]{}", sub[(attr, stuff)] {
197    let calls = if affiliation_is_marked(&stuff)? {
198      marked_list_calls(attr.as_ref(), &stuff)?
199    } else {
200      None
201    };
202    match calls {
203      Some(calls) => Ok(calls),
204      None => Ok(Invocation!(T_CS!("\\lx@add@affiliation"), vec![attr, Some(stuff)])),
205    }
206  });
207  // Rust-only: an address a class prints under a row of names it shares (aistats's `\aistatsaddress` under
208  // `\aistatsauthor{A \And B \And C}` when its columns are not the names', 1806.02924, 2510.09534): marked,
209  // `\lx@add@affiliation@marked`'s; else each entry an affiliation of the authors the optional attributes place
210  // (`annotate=run`), without its e-mail lines, whose addresses are the authors' e-mails as a marked list's are
211  // (`shared_affiliation_calls`).
212  DefMacro!("\\lx@add@affiliation@shared[]{}", sub[(attr, stuff)] {
213    if affiliation_is_marked(&stuff)? {
214      match marked_list_calls(attr.as_ref(), &stuff)? {
215        Some(calls) => Ok(calls),
216        None => Ok(Invocation!(T_CS!("\\lx@add@affiliation"), vec![attr, Some(stuff)])),
217      }
218    } else {
219      Ok(Tokens::new(shared_affiliation_calls(attr, stuff)?))
220    }
221  });
222  DefMacro!("\\lx@ijcai@names{}", sub[(stuff)] {
223    let late = record_author_call("\\lx@ijcai@names", &stuff, true, false, None, false);
224    whole_block_call(stuff, true, None, late)
225  });

227  // Shared "sectioned author block" machinery for the IJCAI author idiom
228  // (ijcai97.sty and its derivatives): one `\author{}` holding names, then
229  // `\affiliations`, then a comma-separated `\emails` list. Used both by the
230  // `ijcai_sty` binding's `\author` override and by the `\lx@add@authors`
231  // marker-branch above (so raw-loaded derivatives like ttm.sty work too).
232  // `\lx@ijcai@authorsplit` reads the names up to `\affiliations` and runs them
233  // through `\lx@ijcai@names` (`\lx@add@authors` without the marker branch), then splits the remainder into
234  // affiliations (up to `\emails`) and the comma-separated emails. Ported from Perl ijcai.sty.ltxml (PR #2767).
235  DefMacro!(
236    "\\lx@ijcai@authorsplit Until:\\affiliations Until:\\done",
237    "\\lx@ijcai@names{#1}\\ifx.#2.\\else\\lx@ijcai@affilsplit#2\\emails\\affiliations\\done\\fi"
238  );
239  DefMacro!(
240    "\\lx@ijcai@affilsplit  Until:\\emails Until:\\affiliations Until:\\done",
241    "\\ifx.#1.\\else\\expandafter\\lx@ijcai@affiliations\\expandafter{\\lx@strip@braces{#1}}\\fi\\ifx.#2.\\else\\expandafter\\lx@ijcai@emails\\expandafter{\\lx@strip@braces{#2}}\\fi"
242  );
243  DefMacro!(
244    "\\lx@ijcai@affiliations{}",
245    "\\lx@add@affiliations[labelseq=author]{#1}"
246  );
247  // The `\emails` list, which ijcai19.sty:127-130 prints as one line under the affiliations, attributing none
248  // (`{aable, ccole}@x.edu, bob.baker@y.org`). Perl's ijcai.sty.ltxml:50-53 splits it at every comma and gives the
249  // n-th piece to the n-th author (`labelseq=author`): a grouped address `\{a, b\}@x` was cut into `{a` and
250  // `b}@x` and every piece after it went one author on (2105.02711, 1905.13566, 2308.08563; the ttm.sty derivative's
251  // `\{creddy,wmgifford,jayant\}@us.ibm.com`, 2401.03955). Rust-only (OXIDIZED_DESIGN_DIVERGENCES #908,
252  // KNOWN_PERL_ERRORS #1200): the list is a class's address list, as jheppub's `\emailAdd` is (`owner=sequence`,
253  // `\lx@add@contact@list`): each address goes to the author it spells, else to the n-th author while the list is one
254  // address per author in their order, else to the author block (`sequenced_contact_labels`); a grouped address is
255  // written out, one address each (`\lx@split@addresses`), or kept whole where it is no plain list of addresses. The
256  // section is one call of the list (`\lx@sequence@call`, `\lx@add@contact@list`'s), read line by line: a `\\` and a
257  // further `\emails` start a line in the class (ijcai19.sty:127 `\def\emails{\egroup\par\bgroup\rm}`), so an
258  // address after one is an address of its own (2002.11711's `1871161@stu.neu.edu.cn\\zelei.liu@ntu.edu.sg`;
259  // 2609.15094's `\emails {$^{*}$Corresponding author}` after its list) — [`ijcai_email_lines`].
260  DefMacro!("\\lx@ijcai@emails{}", sub[(list)] {
261    let mut out =
262      mouth::tokenize_internal("\\lx@clear@frontmatter{ltx:contact}[role=email]\\lx@sequence@call").unlist();
263    for line in ijcai_email_lines(&list) {
264      // (a line with no address, `$^{*}$Corresponding author`, is no place of the list and no e-mail: a note, as
265      // jheppub's `\note` files its "Corresponding author." (`role=note`, notes.rs `is_note_macro`), placed by its own
266      // evidence, else the author block's, as `\lx@add@contact@placed` places a contact; 2609.15094)
267      if address_text(&Tokens::new(line.clone())).contains('@') {
268        out.extend([T_CS!("\\lx@split@addresses"), T_BEGIN!(), T_CS!("\\lx@ijcai@email"), T_END!()]);
269      } else {
270        out.extend(mouth::tokenize_internal("\\lx@add@contact@placed[role=note]{}").unlist());
271      }
272      out.push(T_BEGIN!());
273      out.extend(line);
274      out.push(T_END!());
275    }
276    Ok(Tokens::new(out))
277  });
278  DefMacro!(
279    "\\lx@ijcai@email{}",
280    "\\lx@add@contact@placed[role=email,owner=sequence]{}{#1}"
281  );

283  // What an author reference mark (IEEEtran's `\IEEEauthorrefmark{n}`) prints for its operand, `\lx@author@refmark@symbol`:
284  // the operand itself (`\lx@author@refmark@arabic`), unless the class binding declares its own (ieeetran_cls.rs: the
285  // footnote symbol sequence, IEEEtran.cls:4543; IEEEoj's binding lets it back to the number, IEEEoj.cls:3186). A label reader (`queue_digest::printed_label`) reads its number: the symbols a class's reference
286  // marks print are one to one with their numbers, so the marks pair among themselves as they print, and a label a
287  // superscript sets in another form is told apart by the class's author block, which writes a reference mark mixed
288  // with other marks as the symbol it prints (OXIDIZED_DESIGN_DIVERGENCES #968; 2312.10807).
289  DefMacro!("\\lx@author@refmark@arabic{}", "#1");
290  Let!("\\lx@author@refmark@symbol", "\\lx@author@refmark@arabic");
291  DefMacro!("\\lx@author@refmark@printed{}", sub[(operand)] {
292    Ok(if reading_label() {
293      operand
294    } else {
295      let symbol = Invocation!(T_CS!("\\lx@author@refmark@symbol"), vec![Some(operand.clone())]);
296      Invocation!(T_CS!("\\lx@author@refmark@shown"), vec![Some(operand), Some(symbol)])
297    })
298  });
299  // (the symbol as it prints, its superscript keyed by the number, `_refmark`: the readers of the marks a name shows —
300  // `relocate::shown_marks`, the Kind steps' — read the mark by its number, as its label is, where its symbol is no
301  // label mark: 2311.03600's `Jakob Hollenstein\IEEEauthorrefmark{1}` answers `\IEEEauthorblockA{\IEEEauthorrefmark{1}…}`)
302  DefConstructor!("\\lx@author@refmark@shown Semiverbatim {}", "#2",
303  after_construct => sub[document, whatsit] {
304    let key = whatsit.get_arg(1).map(ToString::to_string).unwrap_or_default();
305    // (the superscript holding it, past the font wrappers inside it)
306    let mut node = Some(document.get_node().clone());
307    while let Some(at) = node.as_ref()
308      && !matches!(at.get_name().as_str(), "sup" | "personname" | "contact" | "creator" | "note")
309    {
310      node = at.get_parent();
311    }
312    if let Some(mut sup) = node.filter(|at| at.get_name() == "sup")
313      && !key.trim().is_empty()
314    {
315      let key = match sup.get_attribute("_refmark") {
316        Some(before) => s!("{before},{}", key.trim()),
317        None => key.trim().to_string(),
318      };
319      document.set_attribute(&mut sup, "_refmark", &key)?;
320    }
321  });

323  // Superscript markers in author/affiliation blocks carry the affiliation
324  // number. Both sides use the "affiliation" prefix so the author's requested
325  // annotation ("affiliation:N") exactly matches the affiliation's label
326  // ("affiliation:N") in relocate_annotations -- avoiding the prefix-stripped
327  // fallback that would otherwise conflate them with a creator's own "author:N"
328  // sequence label and double-attach.
329  // OXIDIZED_DESIGN #129 (html_feedback#1021, arXiv:2403.11905): these two are
330  // `\let` onto `^`/`\textsuperscript` inside an author/affiliation line, so they
331  // must consume their operand the way a real math superscript does — grabbing a
332  // FULL nucleus (`\text{...}` kept with its group, any `$...$` nested inside it
333  // captured whole+undigested). The old `[affiliation]{}` read grabbed only the
334  // leading `\text`, orphaning its `{...}`; inside inline math that stray group
335  // left a brace-group frame on top and the closing `$` fired
336  // `\lx@end@inline@math` against it ("Attempt to end mode math"), arbitrarily
337  // deep for `$^\text{$...$}$` markers. Both engines erred (SHARED-FAILURE);
338  // reading the whole operand keeps the surrounding math balanced.
339  // A note the operand holds (`\textsuperscript{1,2\thanks{…}}`) is no part of a label: it is read after the mark, the
340  // note its author's or its line's (2512.05464; an author block takes it out before, `notes_out_of_marks`).
341  DefPrimitive!(T_CS!("\\lx@sup@request@affiliation"), None, {
342    let (operand, notes) = frontmatter_sup_operand_and_notes()?;
343    if reading_label() {
344      return Ok(Vec::new());
345    }
346    // (keyed by the operand's printed text: refactor stage 4, OXIDIZED_DESIGN_DIVERGENCES #919)
347    let (label, pictured) = frontmatter_label_request(&operand, "affiliation")?;
348    // (a requested asterisk shows as it prints: one requested in text is `*` — `show_unanswered_marks`' text marks —
349    // unless the operand prints another form, which its key reads `*` too: icrc2009's `\IEEEauthorrefmark{1}` sets the
350    // math `∗`, 0909.3754, 0912.0896, 2609.25589; `one_asterisk`, review rev100a B1)
351    let in_text =
352      !(label.iter().any(|l| l == "affiliation:*") && prints_another_asterisk(&operand)?);
353    // (an author reference mark keys by its number and prints its symbol, which a request nothing answers shows:
354    // 2408.02464's `Gus Gray\IEEEauthorrefmark{1}` stays "Gus Gray∗"; OXIDIZED_DESIGN_DIVERGENCES #968)
355    let mut shown = None;
356    if let [first, rest @ ..] = operand.unlist_ref().as_slice()
357      && *first == T_CS!("\\lx@author@refmark@printed")
358      && let Some(key) = label.first()
359    {
360      let mut symbol = vec![T_CS!("\\lx@author@refmark@symbol")];
361      symbol.extend_from_slice(rest);
362      if let Some(printed) = printed_label(&Tokens::new(symbol))? {
363        shown = Some(s!("{key}={}", printed.text));
364      }
365    }
366    add_annotation_labels(label.join(","), in_text, pictured);
367    if let Some(shown) = shown {
368      with_pending_entry_attr(move |attr| {
369        let mut all = attr.get("_refmarkshown").cloned().unwrap_or_default();
370        if !all.is_empty() {
371          all.push(';');
372        }
373        all.push_str(&shown);
374        attr.insert("_refmarkshown".to_string(), all);
375      });
376    }
377    if let Some(notes) = notes {
378      unread(notes);
379    }
380    Ok(Vec::new())
381  });
382  DefPrimitive!(T_CS!("\\lx@sup@setlabel@affiliation"), None, {
383    // (whether the operand is a group, `^{1}`, or a token, `^*`: a shown mark is read again as it was written, so it

790      // <abstract> after every cover block, TOC and pagebreak the body had emitted by
791      // then — schema-invalid, and later than the source order, RUST-ONLY
792      // (tikz-mirror-lens, tipfr-doc, pgf-interference, axodraw2-man, derivative,
793      // russ_doc, schulmathematik, jourcl, isosigns-docs, bootstrapicons-docs — 16
794      // s106 docs; OXIDIZED_DESIGN #246).
795      // First flush only: promote an unambiguous hand-made title, then let the pass
796      // wrap the rest of the cover (before the abstract's marker) as <titlepage>. Once
797      // `\maketitle` placed the frontmatter, a later abstract is a LATE abstract: the
798      // title-page text printed between `\maketitle` and it — the class's own title page
799      // (the deposit), cover lines set apart, never body prose, a sectional unit, a TOC
800      // or a float — is its cover, wrapped as <titlepage> with the abstract behind it at
801      // its own marker (`\lx@frontmatter@mark`, `place_abstract_after_cover`, OXIDIZED_DESIGN_DIVERGENCES #903;
802      // 2011.04405, 1302.5352); what is not cover stays body after the abstract, which
803      // the placed frontmatter holds (or `insert_late_frontmatter` files, streamed).
804      // (`frontmatter_done` is reset by that path itself.)
805      let first_flush = !lookup_bool("frontmatter_done");
806      if first_flush {
807        maybe_promote_leading_title(document)?;
808      }
809      place_frontmatter(document, first_flush, FrontmatterAnchor::LeadingFrontmatter)?;
810    } else {
811      // Where the abstract sits among the document's in-place setters (`\lx@frontmatter@mark`), read before the flush
812      // sweeps its marker: the root child after it, or the end.
813      let abstract_bound = abstract_mark_bound(document);
814      place_frontmatter(document, false, FrontmatterAnchor::Top)?;
815      // The marked print of a field first (`\lx@field@mark`); the text match only for a copy no mark explains.
816      if lift_marked_field_ink(document)? {
817        assign_value("lx_title_ink_dedup_done", true, Some(Scope::Global));
818      }
819      maybe_dedup_leading_title_ink(document)?;
820      // A document without `\maketitle` whose setters print in place before its abstract (arxiv.cls's `\affiliation`,
821      // 1207.1161; ae100prg's `\address`, 1303.3004; newpasp's `\affil`, astro-ph0210637): their cover is wrapped and
822      // the abstract follows it, as a late abstract does (OXIDIZED_DESIGN_DIVERGENCES #903).
823      if let Some(bound) = abstract_bound {
824        place_abstract_after_cover(document, bound.as_ref(), false)?;
825      }
826    }
827  },
828  after_digest => {
829    // A raw class's stores the document set, read where the frontmatter is, before its queue is
830    // digested (frontmatter_stores.rs): the authors' marks are still there to match.
831    crate::frontmatter_stores::harvest_stores(false)?;
832    // What the fallback flushes is sealed provisionally: a class that prints its frontmatter without `\maketitle`
833    // (imsart's `{frontmatter}`) printed it, and a supplement's later frontmatter must not replace it; but where a
834    // `\maketitle` follows a starred section, the PDF printed only that one ([`seal_digested_frontmatter`]).
835    let outer = lookup_bool("lx_frontmatter_provisional");
836    assign_value("lx_frontmatter_provisional", true, Some(Scope::Global));
837    let flushed = digest_front_matter();
838    assign_value("lx_frontmatter_provisional", outer, Some(Scope::Global));
839    flushed?;
840    assign_value("frontmatter_deferred", true, Some(Scope::Global));
841  },
842  // An internal hook (`\@startsection@hook`, the end-document hook): nothing of it belongs in a
843  // reverted `tex=` (a math left open to `\end{document}` reverts through it).
844  reversion => "");
845
846  // Maintain a list of classes that apply to the document root.
847  // This might involve global style options, like leqno.
848  Tag!("ltx:document", after_open_late => sub[document, root] {
849    let classes = with_mapping_keys("DOCUMENT_CLASSES", |keys| join(&keys," "));
850    if !classes.is_empty()  {
851      document.add_class(root, &classes)?;
852    }
853  });
854  Ok(())
855}

857/// The lines of an IJCAI `\emails` section (`\lx@ijcai@emails`): parted at the top level by a line break (`\\`, its
858/// `*` and `[skip]`, `\newline`) and by a further section marker (`\emails`, `\affiliations`), each a new line of the
859/// class's title block (ijcai19.sty:124-130); a line that prints nothing is none. The top level is outside a TeX group
860/// and outside a printed one (`\{ … \}`, `\lbrace … \rbrace`, `\textbraceleft … \textbraceright`): `\{aable, bbaker,\\
861/// ccole\}@x.edu` is one group of three addresses broken over two lines, which `\lx@split@addresses` writes out (review
862/// of 88b r1, probe p1; r2, probe i7).
863fn ijcai_email_lines(list: &Tokens) -> Vec<Vec<Token>> {
864  let breaks = [
865    T_CS!("\\\\"),
866    T_CS!("\\newline"),
867    T_CS!("\\emails"),
868    T_CS!("\\affiliations"),
869  ];
870  let v = list.unlist_ref();
871  let mut lines: Vec<Vec<Token>> = vec![Vec::new()];
872  let mut depth = 0usize;
873  let mut printed = 0usize;
874  let mut at = 0;
875  while at < v.len() {
876    let t = v[at];
877    match t.get_catcode() {
878      Catcode::BEGIN => depth += 1,
879      Catcode::END => depth = depth.saturating_sub(1),
880      _ if t == T_CS!("\\{") || t == T_CS!("\\lbrace") || t == T_CS!("\\textbraceleft") => {
881        printed += 1
882      },
883      _ if t == T_CS!("\\}") || t == T_CS!("\\rbrace") || t == T_CS!("\\textbraceright") => {
884        printed = printed.saturating_sub(1)
885      },
886      _ => {},
887    }
888    if depth == 0 && printed == 0 && breaks.contains(&t) {
889      at += 1;
890      if t == T_CS!("\\\\") {
891        // (`\\*`, `\\[1ex]`: the break's own arguments)
892        let mut next = at + v[at..].iter().take_while(|t| **t == T_SPACE!()).count();
893        if v.get(next) == Some(&T_OTHER!("*")) {
894          next += 1;
895          at = next;
896        }
897        if v.get(next) == Some(&T_OTHER!("["))
898          && let Some(close) = v[next..].iter().position(|t| *t == T_OTHER!("]"))
899        {
900          at = next + close + 1;
901        }
902      }
903      lines.push(Vec::new());
904      continue;
905    }
906    lines.last_mut().expect("a line").push(t);
907    at += 1;
908  }
909  // (without the spaces around it, the end of the line before: `…,\\` then `wenqian@u.nus.edu`, 2304.11989)
910  lines
911    .into_iter()
912    .map(|line| {
913      let lead = line.iter().take_while(|t| **t == T_SPACE!()).count();
914      let trail = line[lead..]
915        .iter()
916        .rev()
917        .take_while(|t| **t == T_SPACE!())
918        .count();
919      line[lead..line.len() - trail].to_vec()
920    })
921    .filter(|line| !prints_nothing(&Tokens::new(line.clone())))
922    .collect()
923}
