7/// Where a contact placed apart from the names goes ([`placed_contact_owner`]).
8pub(in crate::base_utilities) enum ContactOwner {
9  /// to the authors requesting the mark's label, which pairs them as an affiliation's mark does (and so the mark,
10  /// answered, is not left standing on the name)
11  Mark(String),
12  /// to the author at this queue position (from 1)
13  Author(usize),
14  /// to the whole block, the shared creator below the authors (#159)
15  Block,
16}

18/// The owner of a contact placed apart from the names (`\lx@add@contact@placed`), in this order:
19/// 1. the mark written on the contact itself (`$^*$x@y`), when one queued author requests it — the contact's own mark
20///    is the strongest evidence, stronger than the spelling (`*b@x` under `A$^*$` and `B` is A's); when several
21///    request it, the one of them the address spells (1312.7046);
22/// 2. the author whose surname the address's local part spells ([`spelled_owners`]: 0901.2914), the names read as they
23///    print (`queued_author_name_keys`: a macro-held name, a `\let` copy's; P5c);
24/// 3. the only author of the current group (`\lx@frontmatter@group@start`; the whole queue when no class marks
25///    groups);
26/// 4. else the block's.
27///
28/// An author block's labelled email lines rank otherwise (`block_email_calls`): there the spelling outranks the mark
29/// of a correspondence NOTE before the line (1710.00802's `aleder@` under another author's "Corresponding Author"),
30/// a mark on a nearby note being weaker evidence than one on the address.
31///
32/// A `weak` mark is not written on the contact but on the address block the class prints it under (iopart's `\ead`
33/// after an `\address`, `\lx@address@mark@set`): it places only an address that spells no one (1111.5264's `johnwc@`
34/// under `$^6$`, which two authors request, is J W Conklin's `$^1$`; 2407.14305's `rockfish3141@` under `$^1$ Waseda`,
35/// Hidetomo Hoshino's alone — as the PDF prints them; today both sit on the block's creator, RED
36/// `iopart_ead_under_its_address_mark_red`).
37pub(in crate::base_utilities) fn placed_contact_owner(
38  mark: &Tokens,
39  content: &Tokens,
40  weak: bool,
41) -> Result<ContactOwner> {
42  let authors = queued_author_positions()?;
43  let keys = queued_author_name_keys()?;
44  let spelled: Vec<usize> = address_owners(&keys, content)
45    .into_iter()
46    .map(|at| at + 1)
47    .collect();
48  let requesting = if mark.is_empty() || (weak && !spelled.is_empty()) {
49    Vec::new()
50  } else {
51    queued_requesters(mark.unlist_ref())?
52  };
53  if let [author] = requesting.as_slice() {
54    // (a footnote-symbol mark stays shown on the name and pairs by no label of its own: the author's position)
55    return Ok(match mark_label(mark.unlist_ref()) {
56      Some(label) if !is_symbol_mark(mark.unlist_ref()) => ContactOwner::Mark(label),
57      _ => ContactOwner::Author(*author),
58    });
59  }
60  if let [author] = requesting
61    .iter()
62    .copied()
63    .filter(|author| spelled.contains(author))
64    .collect::<Vec<_>>()
65    .as_slice()
66  {
67    return Ok(ContactOwner::Author(*author));
68  }
69  if let [author] = spelled.as_slice() {
70    return Ok(ContactOwner::Author(*author));
71  }
72  let first = frontmatter_group_first();
73  if authors == first {
74    return Ok(ContactOwner::Author(first));
75  }
76  Ok(ContactOwner::Block)
77}

79/// `text` as tokens of its characters, each as it is: an ASCII letter a letter, anything else other (no `#` parameter, no
80/// `%` comment, no `{`/`}` group: a keyval value that is text).
81pub(in crate::base_utilities) fn text_tokens(text: &str) -> Vec<Token> {
82  text
83    .chars()
84    .map(|c| {
85      let c = c.to_string();
86      if c.chars().all(|c| c.is_ascii_alphabetic()) {
87        T_LETTER!(&c)
88      } else if c == " " {
89        T_SPACE!()
90      } else {
91        T_OTHER!(&c)
92      }
93    })
94    .collect()
95}

97/// The key a footnote-symbol mark (`$^\dagger$` on `$^\dagger$baker@y.edu`) answers on the author it places a contact
98/// for ([`symbol_mark_key`]). `None` for any other mark.
99pub(in crate::base_utilities) fn answered_symbol_key(mark: &[Token]) -> Option<String> {
100  if mark.is_empty() || !is_symbol_mark(mark) {
101    return None;
102  }
103  let label = mark_label(mark)?;
104  symbol_mark_key(label.strip_prefix("affiliation:").unwrap_or(&label))
105}

107/// The queue position of the current author group's first author (`\lx@frontmatter@group@start`; 1 when no class marks
108/// groups).
109pub(in crate::base_utilities) fn frontmatter_group_first() -> usize {
110  match lookup_value("lx_frontmatter_group_first") {
111    Some(Stored::Number(n)) => usize::try_from(n.value_of()).unwrap_or(1).max(1),
112    _ => 1,
113  }
114}

116/// An address's tokens as TeX prints them where a brace is set in math: `$\{$`, `$\lbrace$` the brace `\{`, `$\}$`,
117/// `$\rbrace$` the brace `\}` — the math shifts read as characters (a class's verbatim `\email` argument: llncs's
118/// `\email{$\{$fargyriou,mikebekos,symvonis$\}$@math.ntua.gr}`, 1106.2694, 2308.00146, 1602.04934) or as shifts —, so
119/// a list of local parts in math braces is the list in text braces (`\lx@add@email`, `\lx@split@addresses`; census138
120/// ADDR-b, OXIDIZED_DESIGN_DIVERGENCES #972). The shifts pair left to right, as TeX pairs them; any other formula
121/// stays as it is.
122pub(in crate::base_utilities) fn math_braces_as_printed(address: &Tokens) -> Tokens {
123  let v = address.unlist_ref();
124  let shift = |t: &Token| *t == T_OTHER!("$") || t.get_catcode() == Catcode::MATH;
125  let brace = |t: &Token| {
126    if *t == T_CS!("\\{") || *t == T_CS!("\\lbrace") {
127      Some(T_CS!("\\{"))
128    } else if *t == T_CS!("\\}") || *t == T_CS!("\\rbrace") {
129      Some(T_CS!("\\}"))
130    } else {
131      None
132    }
133  };
134  let mut out: Vec<Token> = Vec::with_capacity(v.len());
135  let mut at = 0;
136  while at < v.len() {
137    // the shifts pair left to right, as TeX pairs them: any other pair stays as read, whole, and the next pair opens
138    // after its closing shift (`$x$\}$y$` keeps both formulas)
139    if shift(&v[at])
140      && let Some(len) = v[at + 1..].iter().position(shift)
141    {
142      let pair = &v[at..at + len + 2];
143      let mut inner = pair[1..=len].iter().filter(|t| **t != T_SPACE!());
144      match (inner.next().and_then(brace), inner.next()) {
145        (Some(printed), None) => out.push(printed),
146        _ => out.extend_from_slice(pair),
147      }
148      at += len + 2;
149      continue;
150    }
151    out.push(v[at]);
152    at += 1;
153  }
154  Tokens::new(out)
155}
