9/// Whether `text` is a professional membership grade as an author writes it after the name and its comma: the words of
10/// one ([`grade_words`]: `Fellow, IEEE`, `Senior Member, IEEE`, `Member, OSA`), or an abbreviated grade before its
11/// society (`M.ASCE`, `A. M. ASCE`; [`NAME_GRADE_ABBREVIATIONS`]).
12pub(in crate::base_utilities) fn is_membership_grade(text: &str) -> bool {
13  let text = strip_glued_marks(text);
14  let compact: String = text.chars().filter(|c| !c.is_whitespace()).collect();
15  grade_words(text, true)
16    || NAME_GRADE_ABBREVIATIONS.iter().any(|grade| {
17      compact
18        .strip_prefix(grade)
19        .is_some_and(|society| NAME_GRADE_SOCIETIES.contains(&society))
20    })
21}

23/// A name with the membership grade written after its comma taken out as the author's `membership` contact: `Cat
24/// Cole\textsuperscript{1}, \textit{Fellow, IEEE}` is "Cat Cole" and the contact "Fellow, IEEE" in its italics, the
25/// contact IEEEtran's `\IEEEmembership` makes of the grade it prints (ieeetran_cls.rs `\lx@IEEE@grade`;
26/// 2107.05093, 2302.11403, 2602.16174), and so are `Bob Baker, Member, OSA` and ascelike's `Matthew R. Kuhn, M.ASCE`
27/// (ascexmpl.tex:48). The grade is the text after the first comma of the name's own (outside its groups and math) after
28/// which only a grade is printed ([`is_membership_grade`]); the marks and notes written after it stay the name's, after
29/// the contact, as they print after the grade. `None` when the name ends in no such grade (`John Smith, Jr.`, `Jane
30/// Doe, MD, PhD` qualify the name).
31pub(in crate::base_utilities) fn written_membership_out(name: &[Token]) -> Option<Vec<Token>> {
32  let (mut depth, mut in_math, mut found) = (0, false, None);
33  let commas: Vec<usize> = (name.iter().enumerate())
34    .filter_map(|(c, t)| {
35      depth += t.group_step();
36      in_math ^= t.get_catcode() == Catcode::MATH;
37      (depth == 0 && !in_math && *t == T_OTHER!(",")).then_some(c)
38    })
39    .collect();
40  // (a grade is a suffix of grade words, so its comma is read from the last leftward, the first tail holding a word no grade
41  // has ending the reading: every tail before it holds that word too. The first comma after an alphabetic head opening a
42  // grade is the one read from the first comma on, each tail read once — not the rest of the name per comma, quadratic on
43  // Perl's whole-block creator of a block of 418 names, 1507.02068)
44  for c in commas.into_iter().rev() {
45    let (grade, annotations) = grade_and_annotations(&name[c + 1..]);
46    let text = visible_name_text(&grade);
47    if is_membership_grade(&text)
48      && visible_name_text(&name[..c])
49        .chars()
50        .any(char::is_alphabetic)
51    {
52      found = Some((c, grade, annotations));
53    } else if text.chars().any(char::is_alphanumeric)
54      && !grade_words(strip_glued_marks(&text), false)
55    {
56      break;
57    }
58  }
59  // (one grade, the line's last: a names line printing a grade before it holds a name after a grade, `Ann Able,
60  // \textit{Member, IEEE}; Bob Baker, \textit{Fellow, IEEE}`, whose grades all stay — taking out the last alone would
61  // leave the names before it one creator: 1510.03162, 2507.19818)
62  let (c, grade, annotations) = found.filter(|(c, ..)| !holds_grade(&name[..*c]))?;
63  let head = &name[..c];
64  let end = head
65    .iter()
66    .rposition(|t| !is_spacing(t))
67    .map_or(0, |k| k + 1);
68  let mut out = head[..end].to_vec();
69  out.extend(TokenizeInternal!(r"\lx@add@contact[role=membership]").unlist());
70  out.push(T_BEGIN!());
71  out.extend(grade);
72  out.push(T_END!());
73  out.extend(annotations);
74  Some(out)
75}

77/// Whether a names line prints a membership grade or a part of one ([`is_membership_part`]) between its names'
78/// separators — a comma, a semicolon, an "and".
79fn holds_grade(line: &[Token]) -> bool {
80  visible_name_text(line)
81    .replace('~', " ")
82    .split([',', ';'])
83    .flat_map(|piece| piece.split(" and "))
84    .map(str::trim)
85    .any(|piece| !piece.is_empty() && is_membership_part(piece))
86}

88/// Where the name line a creator's content holds begins and ends: inside the block reader's `\lx@author@withsup{…}` and
89/// a command the name line is set in as its one argument (a raw IEEE class's `\IEEEauthorblockN{…}`, which the kernel's
90/// split puts round each name: icrc2009, 0907.0548, 1111.2718), each holding the whole content; the whole content
91/// otherwise.
92pub(in crate::base_utilities) fn name_line_span(v: &[Token]) -> (usize, usize) {
93  let (mut start, mut end) = (0, v.len());
94  for _ in 0..2 {
95    while end > start && v[end - 1] == T_SPACE!() {
96      end -= 1;
97    }
98    if end - start < 3 {
99      break;
100    }
101    let cs = v[start];
102    if cs.get_catcode() != Catcode::CS
103      || v[start + 1].get_catcode() != Catcode::BEGIN
104      || skip_group(&v[..end], start + 1) != end
105      || (cs != T_CS!("\\lx@author@withsup")
106        && (is_name_annotation(&cs)
107          || is_footnote_mark(&cs)
108          || is_author_mark(&cs)
109          || cs.with_str(|n| n.starts_with("\\lx@"))))
110    {
111      break;
112    }
113    start += 2;
114    end -= 1;
115  }
116  (start, end)
117}

119/// A creator's content as `\lx@add@author` reads it (frontmatter_fields.rs): the membership grade its name line writes
120/// after the name's comma the author's contact ([`written_membership_out`]), inside what the line is set in
121/// ([`name_line_span`]) as in a bare name.
122pub(in crate::base_utilities) fn membership_read_out(content: Tokens) -> Tokens {
123  let v = content.unlist_ref();
124  let (start, end) = name_line_span(v);
125  match written_membership_out(&v[start..end]) {
126    Some(name) => {
127      let mut out = v[..start].to_vec();
128      out.extend(name);
129      out.extend_from_slice(&v[end..]);
130      Tokens::new(out)
131    },
132    None => content,
133  }
134}

136/// What follows a name's comma parted into the text it prints, its spacing trimmed, and the marks and notes written in
137/// it (a note or mark command with its arguments, [`is_name_annotation`], [`is_footnote_mark`], [`is_mark_command`]; a
138/// formula, `$^{1}$`; a block reader's contact call after the name, `\lx@add@affiliation[annotate=3]{…}`: 1611.04834's
139/// `and Warren J. Gross, \emph{Senior Member, IEEE}\\ Department …`).
140fn grade_and_annotations(tail: &[Token]) -> (Vec<Token>, Vec<Token>) {
141  let kernel_call = |t: &Token| t.with_str(|s| s.starts_with("\\lx@add@"));
142  let (mut text, mut annotations, mut i) = (Vec::new(), Vec::new(), 0);
143  while i < tail.len() {
144    let t = tail[i];
145    let start = i;
146    i += 1;
147    if t.get_catcode() == Catcode::MATH {
148      while i < tail.len() && tail[i].get_catcode() != Catcode::MATH {
149        i += 1;
150      }
151      i = (i + 1).min(tail.len());
152      annotations.extend_from_slice(&tail[start..i]);
153    } else if t.get_catcode() == Catcode::CS
154      && (is_name_annotation(&t) || is_footnote_mark(&t) || is_mark_command(&t) || kernel_call(&t))
155    {
156      if tail.get(i) == Some(&T_OTHER!("["))
157        && let Some(close) = tail[i..].iter().position(|u| *u == T_OTHER!("]"))
158      {
159        i += close + 1;
160      }
161      if kernel_call(&t) {
162        while tail
163          .get(i)
164          .is_some_and(|u| u.get_catcode() == Catcode::BEGIN)
165        {
166          i = skip_group(tail, i).min(tail.len());
167        }
168      } else if !is_footnote_mark(&t) {
169        i = skip_group(tail, i).min(tail.len());
170      }
171      annotations.extend_from_slice(&tail[start..i]);
172    } else if t.get_catcode() == Catcode::BEGIN {
173      i = skip_group(tail, start).min(tail.len());
174      text.extend_from_slice(&tail[start..i]);
175    } else {
176      text.push(t);
177    }
178  }
179  let from = text
180    .iter()
181    .position(|t| !is_spacing(t))
182    .unwrap_or(text.len());
183  let to = text
184    .iter()
185    .rposition(|t| !is_spacing(t))
186    .map_or(from, |k| k + 1);
187  (text[from..to].to_vec(), annotations)
188}

190/// Whether `tokens` hold nothing to read but marks and notes: spaces (`~`, `\ `, `\,`), a note or mark command with
191/// its arguments ([`NAME_ANNOTATIONS`], `\footnotemark[…]`, the mark commands of [`mark_commands`]:
192/// `\textsuperscript`, `\IEEEauthorrefmark`, `\authorrefmark`), a superscript or subscript, or math holding only those
193/// (`$^{*}$`). An author written as a macro (`\A`) is something to read.
194pub(in crate::base_utilities) fn only_marks_and_notes(tokens: &[Token]) -> bool {
195  let mut i = 0;
196  let mut any = false;
197  while i < tokens.len() {
198    let t = tokens[i];
199    match t.get_catcode() {
200      Catcode::SPACE => i += 1,
201      Catcode::ACTIVE if t.with_str(|s| s == "~") => i += 1,
202      Catcode::CS if t == T_CS!("\\ ") || t == T_CS!("\\,") => i += 1,
203      // `${}^{1}$`'s empty group
204      Catcode::BEGIN
205        if tokens
206          .get(i + 1)
207          .is_some_and(|n| n.get_catcode() == Catcode::END) =>
208      {
209        i += 2
210      },
211      // (a mark command too: ieeeaccess's `\authorrefmark{2}` after a list's closing period, 2201.07394)
212      Catcode::CS if is_footnote_mark(&t) || is_name_annotation(&t) || is_mark_command(&t) => {
213        any = true;
214        i += 1;
215        if tokens.get(i) == Some(&T_OTHER!("[")) {
216          while i < tokens.len() && tokens[i] != T_OTHER!("]") {
217            i += 1;
218          }
219          i += 1;
220        }
221        if !is_footnote_mark(&t) {
222          i = skip_group(tokens, i);
223        }
224      },
225      Catcode::SUPER | Catcode::SUB => {
226        any = true;
227        i = skip_group(tokens, i + 1);
228      },
229      Catcode::MATH => {
230        let Some(close) = tokens[i + 1..]
231          .iter()
232          .position(|t| t.get_catcode() == Catcode::MATH)
233        else {
234          return false;
235        };
236        if !only_marks_and_notes(&tokens[i + 1..i + 1 + close])
237          || tokens[i + 1..i + 1 + close].is_empty()
238        {
239          return false;
240        }
241        any = true;
242        i += close + 2;
243      },
244      _ => return false,
245    }
246  }
247  any
248}

250/// How the collaboration a line of an author block names is framed ([`collaboration_line_framing`]), `None` for a line
251/// naming none or marked.
252pub(in crate::base_utilities) fn collaboration_line_reading(line: &Tokens) -> Option<Framing> {
253  if author_mark_position(line).is_some() {
254    return None;
255  }
256  collaboration_line_framing(&visible_name_text(line.unlist_ref()))
