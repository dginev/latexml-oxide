10/// Whether a sectioned caller's block (`\lx@add@authors`, `@adaptive`, `@list`) is an IJCAI sectioned one
11/// (`has_ijcai_section_marker`, after the sole-author and separator macro expansions): it goes to the section split
12/// ([`ijcai_split_call`]), whose names `\lx@ijcai@names` then queues whole — a piece cut through `\affiliations` is
13/// digested where a class defines it only in its `\@maketitle` (kr.sty, 2202.05511). The ledger records it too
14/// (`AuthorBlock::sectioned`), whose held Errors are an IJCAI block's section markers (`resolver::Input::Block`'s
15/// `markers`) — never a raw `\@maketitle`'s own `\name`/`\email` held in a plain `\author` (1908.02923's JAIR rows;
16/// `perfect_kernel_batch63::author_markup_title_code_forms`), whose reading ended at the marker, its address a note.
17pub(super) fn ijcai_block(stuff: &Tokens) -> Result<bool> {
18  has_ijcai_section_marker(&expand_separator_macros(expand_sole_author_macro(
19    stuff.clone(),
20  )))
21}

23/// An IJCAI sectioned block's split ([`ijcai_block`]), the old `\lx@add@authors` driver's first step, kept as the one
24/// path that still cuts a block at the call. Beyond-Perl (surpasses Perl; KNOWN_PERL_ERRORS #100): IJCAI-style author
25/// blocks — ijcai97.sty and its derivatives (e.g. the ttm.sty in arXiv:2401.03955) — pack names, `\affiliations` and a
26/// comma-separated `\emails` list into ONE `\author{}`. Perl's splitter does not recognise those section markers, so the
27/// comma-joined email list is shredded into phantom author creators and the affiliation is dropped (Perl 0.8.8, which
28/// raises `\affiliations`/`\emails` as undefined). The block goes to the shared sectioned-author machinery
29/// (`\lx@ijcai@authorsplit`, also used by `ijcai_sty`): it consumes the markers as `Until:` delimiters (so they no longer
30/// error) and splits names / affiliations / emails, each email placed as a class's address list is (OD #908); the names
31/// re-enter as `\lx@ijcai@names`, queued whole. Witness html_feedback#1361 + #1362. A marker counts at the block's top
32/// level only, and only as a separator: undefined there or the ijcai binding's no-op. A document's own
33/// `\emails`/`\affiliations` with text is author text, which LaTeX expands — counted, the split handed it back unsplit,
34/// without end (`\emails` before any `\affiliations`, 2407.10582; `\affiliations` inside `\thanks`, 2505.05474;
35/// `PushbackLimit`). The block is read as the driver read it first: a `\let` alias of a mark command is the command
36/// (`\let\up\textsuperscript`, 2209.14855; [`unalias_mark_commands`]); a note written inside a mark is taken out after
37/// it (`\textsuperscript{\rm 1,\rm 2\thanks{…}}`, 2512.05464; [`notes_out_of_marks`]); an empty mark is none
38/// ([`drop_empty_marks`]); a mark opening a line that prints no label is none (2303.08793;
39/// [`blank_line_marks_dropped`]); a line of notes only after a `\\` joins the line above, its notes no name requests
40/// the document's, added after the split (2609.17106, 2411.05212, 2312.10807; [`notes_line_joins_names`],
41/// OXIDIZED_DESIGN_DIVERGENCES #957). Where those steps leave no top-level marker, the block is queued whole
42/// ([`whole_block_call`]), as any other.
43pub(super) fn ijcai_split_call(stuff: Tokens, replace: bool, late: bool) -> Result<Tokens> {
44  let read = unalias_mark_commands(expand_separator_macros(expand_sole_author_macro(
45    stuff.clone(),
46  )));
47  let read = blank_line_marks_dropped(drop_empty_marks(notes_out_of_marks(read)));
48  let (read, document_notes) = notes_line_joins_names(read);
49  if !has_ijcai_section_marker(&read)? {
50    return whole_block_call(stuff, replace, None, late);
51  }
52  let mut out = vec![T_CS!("\\lx@ijcai@authorsplit")];
53  out.extend(read.unlist());
54  out.push(T_CS!("\\affiliations"));
55  out.push(T_CS!("\\done"));
56  out.extend(document_notes);
57  Ok(Tokens::new(out))
58}

62/// An author argument that is one parameterless macro, alone or with an empty group (`\author{\Authors}`,
63/// 2401.13568; `\IEEEauthorblockN{\hpcaauthors{}}`, 2407.02944), read as TeX typesets it: expanded once, the macro's body
64/// in its place, so its names and separators are seen; a body that is a `\newif` toggle through the branch TeX takes, any
65/// other conditional body left whole (KNOWN_PERL_ERRORS #552).
66pub(in crate::base_utilities) fn expand_sole_author_macro(stuff: Tokens) -> Tokens {
67  let content: Vec<Token> = stuff
68    .unlist_ref()
69    .iter()
70    .copied()
71    .filter(|t| *t != T_SPACE!())
72    .collect();
73  if let Some(read) = sole_input_file(&content) {
74    return read;
75  }
76  let cs = match content.as_slice() {
77    [cs] => *cs,
78    [cs, open, close]
79      if open.get_catcode() == Catcode::BEGIN && close.get_catcode() == Catcode::END =>
80    {
81      *cs
82    },
83    _ => return stuff,
84  };
85  if cs.get_catcode() != Catcode::CS || lookup_conditional(&cs).is_some() {
86    return stuff;
87  }
88  let Ok(Some(defn)) = lookup_definition(&cs) else {
89    return stuff;
90  };
91  // (no parameter text at all: `\def\Authors.{…}` reads a delimiter TeX would demand)
92  if !defn.is_expandable()
93    || defn
94      .get_parameters()
95      .is_some_and(|p| !p.get_parameters().is_empty())
96  {
97    return stuff;
98  }
99  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
100    return stuff;
101  };
102  // a body holding a conditional is read through the branch a `\newif` toggle takes (`\ifanon Anonymous\else Ann Able,
103  // Bob Baker\fi`, as the PDF shows it); any other is left whole: split at its commas, the `\if` and its `\else`/`\fi`
104  // landed in different authors (`Error:expected:\fi`, the other branch's names leaking)
105  let body = body.unlist_ref();
106  let read = if body.iter().any(|t| lookup_conditional(t).is_some()) {
107    match toggled_branch(body) {
108      Some(branch) => branch,
109      None => return stuff,
110    }
111  } else {
112    body.to_vec()
113  };
114  if read.iter().any(|t| t.get_catcode() == Catcode::LETTER) {
115    Tokens::new(read)
116  } else {
117    stuff
118  }
119}

121/// An author argument that is one `\input{file}` (`\author{\input{author_grid.tex}}`, 2609.38746, its names and
122/// affiliations in the file), read as TeX typesets it, as [`expand_sole_author_macro`] reads a sole macro: the file's
123/// content in its place, read by a file mouth under the document's catcodes, so its names and separators are seen; read whole,
124/// the file's names were one author. A file not found, or one that changes how it is read (`\catcode`,
125/// `\makeatletter`, `\endinput`, a further `\input`), is left to `\input` as written.
126fn sole_input_file(content: &[Token]) -> Option<Tokens> {
127  let [input, open, name @ .., close] = content else {
128    return None;
129  };
130  if *input != T_CS!("\\input")
131    || open.get_catcode() != Catcode::BEGIN
132    || close.get_catcode() != Catcode::END
133    || name.is_empty()
134    || name
135      .iter()
136      .any(|t| matches!(t.get_catcode(), Catcode::CS | Catcode::BEGIN | Catcode::END))
137  {
138    return None;
139  }
140  let name: String = name.iter().map(|t| t.to_string()).collect();
141  let path = find_file(&name, None).or_else(|| find_file(&s!("{name}.tex"), None))?;
142  // (its bytes decoded as a mouth decodes them, a Latin-1 line as its bytes — `mouth::decode_input_bytes` —, and each
143  // command matched whole: `\inputencoding`, `\includegraphics` are none of them; the rev107b review's N3)
144  let text = mouth::decode_input_bytes(&std::fs::read(&path).ok()?);
145  let holds = |cs: &str| {
146    text.match_indices(cs).any(|(at, _)| {
147      !text[at + cs.len()..]
148        .chars()
149        .next()
150        .is_some_and(|c| c.is_ascii_alphabetic())
151    })
152  };
153  if [
154    "\\catcode",
155    "\\makeatletter",
156    "\\makeatother",
157    "\\endinput",
158    "\\input",
159    "\\include",
160  ]
161  .iter()
162  .any(|cs| holds(cs))
163  {
164    return None;
165  }
166  // (read as `\input` reads it, a file mouth: its tokens located in the file)
167  let read = Mouth::create(&path, MouthOptions::default())
168    .ok()?
169    .read_tokens();
170  read
171    .unlist_ref()
172    .iter()
173    .any(|t| t.get_catcode() == Catcode::LETTER)
174    .then_some(read)
175}

177/// An author line's parameterless macros that print only a name separator — `\newcommand{\authcomma}{\textmd{,}
178/// \hskip0.5em}` (2403.07809), tibop-article.cls:41's `\lastand` = `, and ` (2401.13365) — written as that separator,
179/// where the split reads it: hidden in the macro, the names around it were one author. The separator is written as the
180/// macro prints it, a space where its body has spacing and no more (`AT\amp T Labs` stays "AT&T", `NSF\comma NIH`
181/// "NSF,NIH"), and an empty group ending the macro's name (`\amp{}`) prints nothing.
182pub(in crate::base_utilities) fn expand_separator_macros(stuff: Tokens) -> Tokens {
183  let v = stuff.unlist_ref();
184  if !v.iter().any(|t| t.get_catcode() == Catcode::CS) {
185    return stuff;
186  }
187  let mut out: Vec<Token> = Vec::with_capacity(v.len());
188  let mut changed = false;
189  let mut i = 0;
190  while i < v.len() {
191    let t = v[i];
192    i += 1;
193    let Some(sep) = separator_macro_text(&t) else {
194      out.push(t);
195      continue;
196    };
197    changed = true;
198    if sep.lead && out.last() != Some(&T_SPACE!()) {
199      out.push(T_SPACE!());
200    }
201    for c in sep.text.chars() {
202      out.push(match c {
203        ' ' => T_SPACE!(),
204        '&' => T_CS!("\\&"),
205        ',' => T_OTHER!(","),
206        _ => T_LETTER!(&c.to_string()),
207      });
208    }
209    if v.get(i) == Some(&T_BEGIN!()) && v.get(i + 1) == Some(&T_END!()) {
210      i += 2;
211    }
212    if sep.trail {
213      out.push(T_SPACE!());
214      while v.get(i) == Some(&T_SPACE!()) {
215        i += 1;
216      }
217    }
218  }
219  if changed { Tokens::new(out) } else { stuff }
220}

235/// What a separator macro prints: `text` (`,`, `and`, `&` or `, and`), and whether spacing comes before (`lead`) and
236/// after it (`trail`) in the macro's body — written as one space.
237struct SeparatorText {
238  text:  &'static str,
239  lead:  bool,
240  trail: bool,
241}

243/// The separator a parameterless macro prints, read through its body past spacing (`\hskip0.5em`, `\quad`, ties), the
244/// font commands around a group (`\textmd{,}`) and `\ignorespaces`, its letters in the case a `\MakeLowercase`,
245/// `\MakeUppercase`, `\lowercase` or `\uppercase` gives its group (cimento's `\atque`); `None` for any other macro.
246fn separator_macro_text(cs: &Token) -> Option<SeparatorText> {
247  if cs.get_catcode() != Catcode::CS || lookup_conditional(cs).is_some() {
248    return None;
249  }
250  // (an author-block delimiter is split on itself, whatever it prints: a class's `\renewcommand{\and}{{\normalfont
251  // and}}`, ecca.cls:192, still parts its groups, as Perl splits the unexpanded `\and`, Base_Utility.pool.ltxml:682)
252  let delimiter = |d: SplitDelim| matches!(d, SplitDelim::Token(t) if t == *cs);
253  if author_group_splits()
254    .into_iter()
255    .chain(author_and_splits())
256    .chain(author_affil_splits())
257    .any(delimiter)
258  {
259    return None;
260  }
261  let defn = lookup_definition(cs).ok().flatten()?;
262  if !defn.is_expandable()
263    || defn
264      .get_parameters()
265      .is_some_and(|p| !p.get_parameters().is_empty())
266  {
267    return None;
268  }
269  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
270    return None;
271  };
272  let body = body.unlist_ref();
273  let mut text = String::new();
274  let (mut lead, mut spaced) = (false, false);
275  // the case changes in force, each with the group depth it ends at (`\MakeLowercase{and\ \ignorespaces}`), and the
276  // one a case command just asked of the group after it
277  let mut cases: Vec<(usize, bool)> = Vec::new();
278  let mut pending_case: Option<bool> = None;
279  let mut depth = 0usize;
280  let mut i = 0;
281  while i < body.len() {
282    let t = body[i];
283    i += 1;
284    // (a case command's argument that is no group is its one token: `\MakeUppercase a`)
285    let single_case = if t.get_catcode() == Catcode::BEGIN {
286      None
287    } else {
288      pending_case.take()
289    };
290    let spacing = if t.get_catcode() == Catcode::BEGIN {
291      depth += 1;
292      if let Some(upper) = pending_case.take() {
293        cases.push((depth, upper));
294      }
295      continue;
296    } else if t.get_catcode() == Catcode::END {
297      if cases.last().is_some_and(|(at, _)| *at == depth) {
298        cases.pop();
299      }
300      depth = depth.saturating_sub(1);
301      continue;
302    } else if t == T_CS!("\\ignorespaces") {
303      // (prints nothing: cimento.cls:1525 `\atque` = `{\normalfont\MakeLowercase{and\ \ignorespaces}}`, every shipped
304      // copy; 0711.2584, 0801.3557, 1402.1844, 2109.08173, 2309.14694, hep-th0703055)
305      continue;
306    } else if let Some(upper) = t.with_str(|s| match s {
307      "\\MakeLowercase" | "\\lowercase" => Some(false),
308      "\\MakeUppercase" | "\\uppercase" => Some(true),
309      _ => None,
310    }) {
311      // (the case its group's letters print in)
312      pending_case = Some(upper);
313      continue;
314    } else if is_spacing(&t) {
315      true
316    } else if t == T_CS!("\\hskip") || t == T_CS!("\\kern") {
317      i = past_glue(body, i);
318      true
319    } else if t == T_CS!("\\hspace") {
320      if body.get(i) == Some(&T_OTHER!("*")) {
321        i += 1;
322      }
323      i = skip_group(body, i);
324      true
325    } else {
326      false
327    };
328    if spacing {
329      if text.is_empty() {
330        lead = true;
331      } else {
332        spaced = true;
333      }
334      continue;
335    }
336    if t.get_catcode() == Catcode::CS && t != T_CS!("\\&") {
337      // an upright-roman font command around the separator's group (`\textmd{,}`, `\rm`) prints nothing of its own;
338      // a visible one (`\textbf{\&}`, `\large`) is the separator's look, kept as written
339      if !t.with_str(|s| PLAIN_FONT_COMMANDS.contains(&s)) {
340        return None;
341      }
342      continue;
343    }
344    if !(t == T_CS!("\\&") || matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER)) {
345      // (an alignment `&`, a parameter, math: no printed separator)
346      return None;
347    }
348    if std::mem::take(&mut spaced) {
349      text.push(' ');
350    }
351    if t == T_CS!("\\&") {
352      text.push('&');
353    } else {
354      t.with_str(
355        |s| match single_case.or(cases.last().map(|(_, upper)| *upper)) {
356          Some(true) => text.push_str(&s.to_uppercase()),
357          Some(false) => text.push_str(&s.to_lowercase()),
358          None => text.push_str(s),
359        },
360      );
361    }
362  }
363  let text = match text.as_str() {
364    "," => ",",
365    "and" => "and",
366    "&" => "&",
367    ", and" => ", and",
368    _ => return None,
369  };
370  Some(SeparatorText { text, lead, trail: spaced })
371}

373/// The branch of `body` = `\ifX … [\else …] \fi` whose `\ifX` means `\iftrue` or `\iffalse` (a `\newif` toggle) that TeX
374/// takes, when that branch holds no conditional of its own; `None` for any other shape.
375fn toggled_branch(body: &[Token]) -> Option<Vec<Token>> {
376  let first = body.iter().position(|t| *t != T_SPACE!())?;
377  let last = body.iter().rposition(|t| *t != T_SPACE!())?;
378  let test = lookup_definition(&body[first]).ok().flatten()?;
379  let truth = if Some(&test)
380    == lookup_definition(&T_CS!("\\iftrue"))
381      .ok()
382      .flatten()
383      .as_ref()
384  {
385    true
386  } else if Some(&test)
387    == lookup_definition(&T_CS!("\\iffalse"))
388      .ok()
389      .flatten()
390      .as_ref()
391  {
392    false
393  } else {
394    return None;
395  };
396  if last <= first || lookup_conditional(&body[last]) != Some(ConditionalType::Fi) {
397    return None;
398  }
399  let inner = &body[first + 1..last];
400  let otherwise = inner
401    .iter()
402    .position(|t| lookup_conditional(t) == Some(ConditionalType::Else));
403  let branch = match (truth, otherwise) {
404    (true, Some(e)) => &inner[..e],
405    (true, None) => inner,
406    (false, Some(e)) => &inner[e + 1..],
407    (false, None) => &[][..],
408  };
409  if branch.iter().any(|t| lookup_conditional(t).is_some()) {
410    return None;
411  }
412  Some(branch.to_vec())
413}

417/// Whether a mark's operand, its notes taken out, prints nothing: braces, spaces, `\relax` and font switches only
418/// (`^{\thanks{…}}`, `^{\mathbf{}}`).
419pub(in crate::base_utilities) fn mark_prints_nothing(mark: &[Token]) -> bool {
420  mark.iter().all(|t| {
421    matches!(
422      t.get_catcode(),
423      Catcode::BEGIN | Catcode::END | Catcode::SPACE
424    ) || *t == T_CS!("\\relax")
425      || (t.get_catcode() == Catcode::CS && t.with_str(|name| FONT_SWITCH.is_match(name)))
426  })
427}

429/// An author block's notes written inside a mark (`Nataliia Babina\textsuperscript{\rm 1,\rm 2\thanks{Work done …}}`,
430/// `Ann$^{1,\thanks{Corresponding author.}}$`, `$^{1\footnotemark[1]}$`) taken out after it: LaTeX prints the labels
431/// and the note's own mark in the superscript, its text as a footnote, so the note is the author's as one written
432/// after the mark is (`\textsuperscript{\rm 1,\rm 2}\thanks{…}`), and the mark's labels are read without the note's
433/// text in them — the request was the label `2\thanks{…}`, which no list answers and which printed nothing, so the
434/// mark `2` and the note were lost (2512.05464, 1911.03118, 1912.00833, 2211.09622, 2305.16614, 2306.17472,
435/// 2309.12862, 2402.18447, 2403.04164, 2410.12564, 2501.01423, 2502.15412, 2503.05905, 2603.17529, 2605.17958). A
436/// mark holding nothing else (`$^{\thanks{Topical Group Conveners}}$`, 2212.00809) goes, its note kept. The mark
437/// readers keep any note left in an operand all the same ([`split_mark_notes`], `\lx@sup@request@affiliation`). A
438/// macro the mark holds that makes a note is taken out too (`\rm 1,2,\equalcontrib`, 2603.17529; `1\samethanks`,
439/// 2309.12862). Perl reads the operand as the label too (Base_Utility.pool.ltxml:729-734; KNOWN_PERL_ERRORS #1136,
440/// OXIDIZED_DESIGN_DIVERGENCES #881).
441pub(in crate::base_utilities) fn notes_out_of_marks(stuff: Tokens) -> Tokens {
442  let v = stuff.unlist_ref();
443  if !v
444    .iter()
445    .any(|t| t.get_catcode() == Catcode::CS && !is_author_mark(t))
446    || !v.iter().any(is_author_mark)
447  {
448    return stuff;
449  }
450  let mut out: Vec<Token> = Vec::with_capacity(v.len());
451  let mut changed = false;
452  let mut i = 0;
453  while i < v.len() {
454    if is_author_mark(&v[i]) {
455      let open = i + 1 + v[i + 1..].iter().take_while(|t| **t == T_SPACE!()).count();
456      if v
457        .get(open)
458        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
459      {
460        let close = skip_group(v, open);
461        if close <= v.len()
462          && v[close - 1].get_catcode() == Catcode::END
463          && let Some((mark, notes)) = split_mark_notes(&v[open + 1..close - 1])
464        {
465          if !mark_prints_nothing(&mark) {
466            out.push(v[i]);
467            out.push(T_BEGIN!());
468            out.extend(mark);
469            out.push(T_END!());
470          }
471          out.extend(notes);
472          i = close;
473          changed = true;
474          continue;
475        }
476      }
477    }
478    out.push(v[i]);
479    i += 1;
480  }
481  if changed { Tokens::new(out) } else { stuff }
482}

484/// An author block's line break that opens a line of notes only (`line_notes`) taken out, with its `*` and
485/// `[len]` and any spacer `\\` after it: the notes join the line above, as written without the break. Such a line
486/// prints only its notes' marks; its notes are the names' above (`Ann Able$^{1,*}$, … and Cat Cole$^{1}$\\
487/// \thanks{$^{1}$Ann Able and Cat Cole are with Univ A, Town.}…`: 2609.17106, 2411.05212, 2312.10807's `\\[3.0ex]`).
488/// Read as a line of its own, it led with the mark inside its first note — `author_mark_position` (marks.rs) counts a
489/// mark inside a `\thanks`, `has_author_marks` skips it, and the two disagree — and became one affiliation answering
490/// mark 1 while its notes went to the document: Ann's and Cat's affiliation lost, all three notes left unowned (Perl
491/// the same, Base_Utility.pool.ltxml:702-712; KNOWN_PERL_ERRORS #1206, OXIDIZED_DESIGN_DIVERGENCES #957). Only that
492/// misreading: a block whose names carry marks (`has_author_marks`, outside its notes) and a line with a note whose
493/// leading mark a name requests ([`notes_the_names_request`]: its other notes the document's, returned to be added after
494/// the block's calls — review rev100c S1); a line of unmarked notes stays a line — in a marked block it continues the entry above as joined it would, and in an
495/// unmarked block it is the document's notes, no author's by position (ieeetran_cls.rs `without_stop_after_grade`, batch
496/// 86d: 2112.14209, 2409.11755, 2510.15773, 2605.22853). And only after a names line: its `\and` group's first, or one
497/// whose marks follow its text — after an affiliation or an `Emails: …` line the notes stay a line of their own
498/// (joined, 1911.03356's address list no longer read as one). Only a break at the block's top level, outside braces
499/// and environments (a tabular's row end is no line), and only up to the next line break or `\and`.
500pub(in crate::base_utilities) fn notes_line_joins_names(stuff: Tokens) -> (Tokens, Vec<Token>) {
501  let v = stuff.unlist_ref();
502  let brk = T_CS!("\\\\");
503  if !v.contains(&brk) || !has_author_marks(&stuff) {
504    return (stuff, Vec::new());
505  }
506  let ends_line = |t: &Token| {
507    *t == brk
508      || author_and_splits()
509        .iter()
510        .any(|split| matches!(split, SplitDelim::Token(cs) if cs == t))
511  };
512  let past_spaces = |mut at: usize| {
513    while v.get(at) == Some(&T_SPACE!()) {
514      at += 1;
515    }
516    at
517  };
518  // (past a `\\` at `at`: its `*` and `[…]`, the spaces before the `[` skipped as `\@ifnextchar` skips them)
519  let past_break = |at: usize| {
520    let mut j = at + 1;
521    if v.get(j) == Some(&T_OTHER!("*")) {
522      j += 1;
523    }
524    let open = past_spaces(j);
525    if v.get(open) == Some(&T_OTHER!("["))
526      && let Some(close) = v[open..].iter().position(|t| *t == T_OTHER!("]"))
527    {
528      j = open + close + 1;
529    }
530    j
531  };
532  // (the labels the block's names request, outside their notes)
533  let requested: Vec<String> = author_mark_operands(v)
534    .iter()
535    .flat_map(|operand| mark_keys(operand))
536    .collect();
537  let mut out: Vec<Token> = Vec::with_capacity(v.len());
538  let mut document_notes: Vec<Token> = Vec::new();
539  let mut changed = false;
540  let (mut depth, mut environments) = (0usize, 0usize);
541  // (where the line above starts in `out`, and whether it is its `\and` group's first)
542  let (mut above_start, mut first_line) = (0usize, true);
543  let mut i = 0;
544  while i < v.len() {
545    let t = v[i];
546    if t == brk && depth == 0 && environments == 0 {
547      // (the line above a names line: its group's first, or one whose marks follow its text — not an affiliation its
548      // mark leads, nor an unmarked line, `Emails: …` whose address list a note joined would no longer read as one,
549      // 1911.03356)
550      let above = Tokens::new(out[above_start..].to_vec());
551      let names_above = first_line || (has_author_marks(&above) && !marker_leads(&above));
552      let mut j = past_break(i);
553      // (a spacer `\\` right after it, `\\ \\`)
554      while v.get(past_spaces(j)) == Some(&brk) {
555        j = past_break(past_spaces(j));
556      }
557      let mut level = 0usize;
558      let line_end = (j..v.len())
559        .find(|&k| {
560          match v[k].get_catcode() {
561            Catcode::BEGIN => level += 1,
562            Catcode::END => level = level.saturating_sub(1),
563            _ => {},
564          }
565          level == 0 && ends_line(&v[k])
566        })
567        .unwrap_or(v.len());
568      let line = &v[j..line_end];
569      if names_above
570        && let Some(notes) = line_notes(line)
571        && let Some((joined, document)) = notes_the_names_request(line, &notes, &requested)
572      {
573        out.extend(joined);
574        document_notes.extend(document);
575        changed = true;
576        i = line_end;
577        continue;
578      }
579    }
580    match t.get_catcode() {
581      Catcode::BEGIN => depth += 1,
582      Catcode::END => depth = depth.saturating_sub(1),
583      _ if depth == 0 && t == T_CS!("\\begin") => environments += 1,
584      _ if depth == 0 && t == T_CS!("\\end") => environments = environments.saturating_sub(1),
585      _ => {},
586    }
587    out.push(t);
588    i += 1;
589    if depth == 0 && environments == 0 && ends_line(&t) {
590      above_start = out.len();
591      first_line = t != brk;
592    }
593  }
594  if changed {
595    (Tokens::new(out), document_notes)
596  } else {
597    (stuff, document_notes)
598  }
599}

601/// A line of notes only ([`line_notes`]: its `notes`, their spans) as it joins the names above
602/// ([`notes_line_joins_names`]) — the line's tokens to join, and the notes taken out of the block as the document's —, or
603/// `None` when it stays a line: a note joins when the mark leading it is one the names request (`requested`,
604/// [`mark_keys`]); a note no name requests (`\thanks{This work was funded by …}`, a symbol only a note prints,
605/// 2312.10807's `$^{*}$Equal contribution` under `\IEEEauthorrefmark{1}`) is the document's, as the line read whole made
606/// it — digested in a contact, `\thanks` is `\lx@add@pubnote@thanks` and `\footnote` `\lx@add@pubnote` (queue_digest.rs,
607/// [`document_note_call`]) —, not the last name's (FIDELITY_CHARTER R1: 2609.17106's funding note on Markus Vincze, review rev100c S1); taken
608/// out of the block, so no mark its text prints is read as a name's (`$^{\dagger}$Corresponding author`). No note
609/// requested, or one of the document's that is no plain `\thanks{…}` or `\footnote{…}`: the line stays.
610fn notes_the_names_request(
611  line: &[Token],
612  notes: &[(usize, usize)],
613  requested: &[String],
614) -> Option<(Vec<Token>, Vec<Token>)> {
615  let asked: Vec<bool> = notes
616    .iter()
617    .map(|&(start, end)| {
618      leading_note_mark(&line[start..end]).is_some_and(|operand| {
619        mark_keys(&operand)
620          .iter()
621          .any(|key| requested.contains(key))
622      })
623    })
624    .collect();
625  if !asked.contains(&true) {
626    return None;
627  }
628  let (mut joined, mut document) = (Vec::with_capacity(line.len()), Vec::new());
629  let mut at = 0;
630  for (&(start, end), asked) in notes.iter().zip(asked) {
631    joined.extend_from_slice(&line[at..start]);
632    at = end;
633    if asked {
634      joined.extend_from_slice(&line[start..end]);
635      continue;
636    }
637    // (no `[…]` before its text: an optional argument is no pubnote's)
638    document.extend(document_note_call(&line[start..end])?);
639  }
640  joined.extend_from_slice(&line[at..]);
641  Some((joined, document))
642}

644/// The operand of the mark leading a note's text (`\thanks{$^{1}$Univ A}`, `\thanks{$*$ Equal contribution}`): a mark
645/// before any letter of it, or a math group of no letter it opens with (`$*$`, `$\dagger$`).
646fn leading_note_mark(note: &[Token]) -> Option<Vec<Token>> {
647  let open = note
648    .iter()
649    .position(|t| t.get_catcode() == Catcode::BEGIN)?;
650  let text = &note[open + 1..note.len().saturating_sub(1)];
651  let text = &text[text.iter().take_while(|t| **t == T_SPACE!()).count()..];
652  let tokens = Tokens::new(text.to_vec());
653  if let Some(at) = author_mark_position(&tokens)
654    && !name_precedes_marker(&tokens, at)
655  {
656    return mark_operands(text).into_iter().next();
657  }
658  let close = text
659    .iter()
660    .skip(1)
661    .position(|t| t.get_catcode() == Catcode::MATH)?
662    + 1;
663  let inner = &text[1..close];
664  (text.first()?.get_catcode() == Catcode::MATH
665    && !inner.is_empty()
666    && inner.iter().all(|t| t.get_catcode() != Catcode::LETTER))
667  .then(|| inner.to_vec())
668}

670/// The labels a mark's operand keys (`clean_frontmatter_labels`, no prefix), each comma piece's, and a piece's number
671/// or letter and its symbol apart as well when they are glued ([`split_glued_symbol`]: `$^{1*}$` keys `1*`, `1` and
672/// `*`, as 2411.05212's `$^*$Work done …` asks).
673fn mark_keys(operand: &[Token]) -> Vec<String> {
674  let mut keys = Vec::new();
675  for piece in operand.split(|t| *t == T_OTHER!(",")) {
676    let piece: Vec<Token> = piece
677      .iter()
678      .filter(|t| **t != T_SPACE!())
679      .copied()
680      .collect();
681    let mut parts = vec![piece.clone()];
682    if let Some((lead, symbol)) = split_glued_symbol(&piece) {
683      parts.extend([lead, symbol]);
684    }
685    for part in parts {
686      keys.extend(clean_frontmatter_labels(&Tokens::new(part).to_string(), ""));
687    }
688  }
689  keys
690}

692// (moved unchanged from `author_block/separators.rs` by Z18m, which deleted `author_block/`: Author-line separators:
693// line parts, words, `and` splits, alignment and ACM column unwrapping.)
694/// Whether an author block is an IJCAI sectioned one: `\affiliations` or `\emails` at its top level (the split's
695/// `Until:` delimiters match nothing deeper), each undefined, `\relax` or a no-op there — the ijcai binding's
696/// (`ijcai_sty.rs`), or a raw derivative's that defines them only in `\@maketitle` (ttm.sty).
697pub(in crate::base_utilities) fn has_ijcai_section_marker(stuff: &Tokens) -> Result<bool> {
698  let markers = [T_CS!("\\affiliations"), T_CS!("\\emails")];
699  let mut depth = 0i32;
700  for token in stuff.unlist_ref() {
701    match token.get_catcode() {
702      Catcode::BEGIN => depth += 1,
703      Catcode::END => depth -= 1,
704      _ if depth == 0 && markers.contains(token) => {
705        let separator = token.defined_as(&TOKEN_RELAX)
706          || match lookup_definition_stored(token)? {
707            None => true,
708            Some(Stored::Expandable(e)) => {
709              e.paramlist
710                .as_ref()
711                .is_none_or(|p| p.get_parameters().is_empty())
712                && matches!(&e.expansion, Some(ExpansionBody::Tokens(t)) if t.is_empty())
713            },
714            Some(_) => false,
715          };
716        if separator {
717          return Ok(true);
718        }
719      },
720      _ => {},
721    }
722  }
723  Ok(false)
724}
