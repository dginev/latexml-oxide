8/// Helper used by `\leftline`/`\rightline`/`\centerline` and friends.
9/// Perl `TeX_Paragraph.pool.ltxml:75` `sub alignLine`.
10pub fn align_line(
11  document: &mut Document,
12  line: &[Option<Digested>],
13  alignment: &str,
14) -> Result<()> {
15  if document.is_openable("ltx:p") {
16    let line_content = line.iter().filter_map(|c| c.as_ref()).collect();
17    document.insert_element(
18      "ltx:p",
19      line_content,
20      Some(string_map!("class" => s!("ltx_align_{alignment}"))),
21    )?;
22  } else if document.is_openable("ltx:text") {
23    let line_content = line.iter().filter_map(|c| c.as_ref()).collect();
24    document.insert_element(
25      "ltx:text",
26      line_content,
27      Some(string_map!("class" => s!("ltx_align_{alignment}"))),
28    )?;
29    document.insert_element("ltx:break", Vec::new(), None)?;
30  } else if let Some(Some(line_content)) = line.first() {
31    document.absorb(line_content, None)?;
32  }
33  Ok(())
34}

41  //======================================================================
42  // Spacing tweaks
43  //----------------------------------------------------------------------
44  // \ignorespaces           c  makes TeX read and expand tokens but do nothing until a nonspace
45  // token is reached. \noboundary             c  if present, breaks ligatures and kerns.
46  // \vadjust                c  inserts a vertical list between two lines in a paragraph.
47
48  // tex.web §1045: `\ignorespaces` runs `get_x_token` until a non-space —
49  // it EXPANDS what follows (Perl `SkipSpaces` peeks unexpanded, so
50  // `\ignorespaces\if@ignore…\fi <space>` — latex.ltx:15392's `\end`
51  // epilogue behind comment.sty's `\end{Included}`→`\ignorespaces` — left
52  // the end-of-line space in the paragraph; pdflatex eats it).
53  DefPrimitive!("\\ignorespaces", {
54    if let Some(t) = read_x_non_space()? {
55      unread_one(t);
56    }
57  });
58  def_primitive_noop("\\noboundary")?;
59  // \vadjust<filler>{<vertical mode material>}
60  // TeX forbids `\vadjust` in vertical mode (tex.web §1098); as Perl, it is queued for the next paragraph's end.
61  // The material is read live after its `{` and built there, in a group of its own whose paragraph, if it begins one,
62  // ends at the group's end (§1099-1100 `begin_insert_or_adjust`, `insert_group`: `end_graf`; base_utilities.rs
63  // `predigest_insert_group_contents`); its own `\vadjust`s go to that paragraph (`with_own_adjust_queue`). What it
64  // built waits for the paragraph's `\par` (§889 sets it after the line). Perl (TeX_Paragraph.pool.ltxml:32) queues
65  // the tokens of a macro argument and digests them at the `\par`: a short verbatim in it ran its `\ifx`, a box
66  // register a group restored was empty (KNOWN_PERL_ERRORS #441; repros boxes-groups/vadjust_material_is_read_live,
67  // vadjust_material_is_built_where_it_is_read, vadjust_material_is_a_vertical_list; guard `box_primitives::vadjust`).
68  DefPrimitive!("\\vadjust", {
69    // pdfTeX's `\vadjust pre {…}` sets the material before the line; here it is set after it (KNOWN_PERL_ERRORS #441).
70    read_keyword(&["pre"])?;
71    // §403 `scan_left_brace`: the material runs to the group's `}`.
72    scan_left_brace()?;
73    // The material is a list of its own: its paragraph is indented by the `\parindent` in force there, and the
74    // class the host paragraph's `\par` will read (deferred by the `\par` before it) waits for it.
75    let host_class = lookup_value("next_para_class");
76    let parindent_zero =
77      lookup_register("\\parindent", Vec::new())?.is_some_and(|r| r.value_of() == 0);
78    assign_value(
79      "next_para_class",
80      if parindent_zero {
81        Stored::from("ltx_noindent")
82      } else {
83        Stored::None
84      },
85      None,
86    );
87    let built = with_own_adjust_queue(predigest_insert_group_contents);
88    assign_value("next_para_class", host_class.unwrap_or(Stored::None), None);
89    if let Some(material) = built? {
90      push_digested("vAdjust", material);
91    }
92  });

97  // \everypar               pt holds tokens added at the beginning of every paragraph.
98  // \indent                 c  begins a new paragraph indented by \parindent.
99  // \noindent               c  begins a new paragraph that is not indented.
100  // \par                    c  is an explicit command to end a paragraph.
101  DefRegister!("\\everypar", Tokens!());
102  // These determine whether the _next_ paragraph gets indented!
103  // thus it needs \par to check whether such indentation has been set.
104  DefConstructor!("\\indent", sub[document] {
105    if let Some(mut node) = document.get_element() {
106      let tag = document::get_node_qname(&node);
107      let para_tag = pin!("ltx:para");
108      if tag == para_tag {
109        node.set_attribute("class","ltx_indent")?;
110      } else if document::sym_can_contain_somehow(tag, para_tag).is_some() {
111        // Perversely ignore indent on 1st para after sectioning titles (Perl parity)
112        let prev_is_title = node.get_last_child().map(|prev| {
113          let prev_qname = document::get_node_qname(&prev);
114          with(prev_qname, |s| s == "ltx:title" || s == "ltx:toctitle")
115        }).unwrap_or(false);
116        if prev_is_title {
117          document.open_element("ltx:para", None, None)?;
118        } else {
119          document.open_element("ltx:para", Some(string_map!("class"=>"ltx_indent")), None)?;
120        }
121      }
122      // Otherwise ignore.
123    }
124  },
125  properties => { stored_map!("isSpace" => true) },
126  // Perl: enterHorizontal => 1; the whatsit is the paragraph's indent box (tex.web §1091).
127  before_digest => { enter_horizontal_listed_indent(); });
128  DefConstructor!("\\noindent", sub[document] {
129    if let Some(mut node) = document.get_element() {
130      let tag = document::get_node_qname(&node);
131      let para_tag = pin!("ltx:para");
132      if tag == para_tag {
133        node.set_attribute("class","ltx_noindent")?;
134      } else if document::sym_can_contain_somehow(tag, para_tag ).is_some() {
135        // Used in a position where a paragraph can be started, start
136        document.open_element("ltx:para", Some(string_map!("class"=>"ltx_noindent")), None)?;
137      }
138      // Otherwise ignore.
139    }
140  },
141  properties => { stored_map!("isSpace" => true) },
142  // Perl: enterHorizontal => 1; a paragraph begun without its indent box (tex.web §1091).
143  before_digest => { enter_horizontal_listed_indent(); });

145  // <ltx:para> represents a Logical Paragraph, whereas <ltx:p> is a `physical paragraph'.
146  // A para can contain both p and displayed equations and such.
147
148  // Remember; \par _closes_, not opens, paragraphs!
149  // Here, we want to close both an open p and para (if either are open).
150  // \par is a NOOP only in the RAW preamble; everywhere else it closes the paragraph
151  // being built (the exact rule + rationale are in after_digest, where the `noop`
152  // property is set). This is the pragmatic realization of Perl's note ("\par should
153  // be a NOOP in vertical mode ... if we can be sure we're tracking modes correctly"):
154  // we can't rely on mode (LaTeXML stays vertical after display math), so we key on
155  // CONTEXT — whether `\begin{document}` has opened the document environment — instead.
156  // The old code gated the no-op purely on the latexml-only `inPreamble` flag, which
157  // made a blank line inside `\AtBeginDocument` a no-op and failed to split paragraphs
158  // (upstream #2754). So #2846's early `inPreamble=0` — and its `\@onlypreamble`
159  // collateral (#2848) — are unnecessary: `\begin{document}` leaves the preamble AFTER
160  // the hooks again, keeping a deferred `\RequirePackage`/`\usepackage` legal.
161  let mut skippable_props: SymHashMap<Stored> = SymHashMap::default();
162  skippable_props.insert("alignmentSkippable", Stored::Bool(true));

164  DefConstructor!("\\lx@normal@par",
165    sub[document, _args, props] {
166      // (in a frontmatter item whose element holds only inline content, a line break: OXIDIZED_DESIGN_DIVERGENCES #766)
167      let item = prop_string!(props, "frontmatter_item");
168      if !item.is_empty() && frontmatter_paragraph_break(document, &item)? {
169        return Ok(());
170      }
171      if !prop_bool!(props, "noop") {
172        document.maybe_close_element("ltx:p")?;
173        // A source paragraph break between a float's panels ends their row (the panel grid rule,
174        // `note_panel_separator`); not the one leaving horizontal mode inserts (2605.00347, below).
175        if !prop_bool!(props, "internal_par") {
176          crate::latex_constructs::note_panel_separator(document, "par", None)?;
177        }
178        let element = document.get_element();
179        if let Some(mut node) = element {
180          let qname = document::get_node_qname(&node);
181          // Only set on the para about to close, if unknown!
182          if qname == pin!("ltx:para") && node.get_attribute("class").is_none() {
183            let class_sym = prop_str!(props,"class");
184            if class_sym != pin!("") {
185              let class_s = with(class_sym, |s| s.to_string());
186              document.set_attribute(&mut node, "class", &class_s)?;
187            } else if prop_bool!(props, "first_para_noindent")
188              && !document::helpers::preceding_para_sibling(&node)
189            {
190              // Surpass Perl (OXIDIZED_DESIGN #143, issue #719): the deferred
191              // `next_para_class` above never reaches the document's FIRST
192              // paragraph — no prior `\par` recorded a class for it — so with
193              // `\parindent=0` it kept the stylesheet's default first-line indent
194              // where pdflatex is flush. `after_digest` flags the candidate when
195              // `\parindent==0` and no deferred class applies; we stamp it here,
196              // in the constructor, gated on the STRUCTURAL "first paragraph of
197              // its parent" test (no preceding `ltx:para` sibling). That is robust
198              // to how many stray `\par`s fired before the first content (which
199              // varies by TeX Live year / dump vs no-dump) — a state one-shot was
200              // not (issue #719 first landed with `seen_first_para`, which a
201              // begin-document `\par` consumed under the no-dump/tl2023 path).
202              document.set_attribute(&mut node, "class", "ltx_noindent")?;
203            }
204          }
205          // NOTE: Perl's \par (\lx@normal@par) does NOT insert figure-separating
206          // breaks — figure row breaks are computed by WIDTH in
207          // `arrange_panels_and_breaks` (ltx:figure afterClose). A prior Rust-only
208          // branch here inserted an <ltx:break> whenever \par fired in an
209          // ltx:figure, which mis-fired for the internal \par that `leaveHorizontal`
210          // triggers after inter-panel glue (\hfill/\quad) — producing one spurious
211          // break per subfigure, so an intended 4-per-row grid collapsed to 1
212          // panel per row (arXiv 2605.00347). Removed; see arrange_panels.
213        }
214        // A `\par` ends a paragraph, never a list (a TeX list is a paragraph shape, not a group): a `\trivlist`
215        // begun inside the paragraph, `_autoclose` as Perl's, stays open (csquotes' `{quotesample}`, a blank line
216        // before its first `\item`; KNOWN_PERL_ERRORS #456). Nor the inline sectional block an item's heading opens
217        // in its paragraph, with the section in it (`list_or_block_open_inside_para`; 1805.06896).
218        if !prop_bool!(props, "internal_par") && !document::helpers::list_or_block_open_inside_para(document) {
219          document.maybe_close_element("ltx:para")?;
220        }
221      }
222    },
223    before_digest => {
224      // A `\par` in an inline formula the math after it continues (a formula split by a blank line, hep-th9402135):
225      // the formula goes on, as Perl's, and this whatsit reverts as nothing (`lx@par@void`). It still clears the
226      // paragraph shape, as every `\par` (below); the `next_para_class` and `\vadjust` work of its after-digest is
227      // left to the `\par` that ends the paragraph, which this one does not. Otherwise tex.web §1046-1047
228      // `mmode+par_end`: "Missing $ inserted", the inline formula ended and this `\par` read again after it; this
229      // whatsit, left in the formula, reverts as nothing. OXIDIZED_DESIGN #553.
230      if formula_continues_past_par() {
231        assign_value("parshape", Stored::None, None);
232        assign_value("interlinepenalties", Stored::None, None);
233        assign_value("lx@par@void", true, None);
234        return Ok(Vec::new());
235      }
236      let again = get_current_token().unwrap_or_else(|| T_CS!("\\par"));
237      if insert_dollar_sign(Tokens!(again))? {
238        assign_value("lx@par@void", true, None);
239        return Ok(Vec::new());
240      }
241      // Perl: combine any digested horizontal material into a horizontal List
242      let mode = lookup_string_from_sym(pin!("MODE"));
243      let bound = lookup_string_from_sym(pin!("BOUND_MODE"));
244      if mode == "horizontal" && bound.ends_with("vertical") {
245        // Perl: $stomach->repackHorizontal; an explicit `\par` ends the paragraph (tex.web §1096 `end_graf`).
246        if LookupBool!("INTERNAL_PAR") {
247          repack_horizontal();
248        } else {
249          end_graf();
250        }
251        assign_value_inplace_sym(pin!("MODE"), bound); // Resume vertical/internal_vertical
252      }
253      assign_value("parshape", Stored::None, None);
254      assign_value("interlinepenalties", Stored::None, None);
255    },
256    after_digest => sub[whatsit] {
257      if lookup_bool("lx@par@void") {
258        assign_value("lx@par@void", false, None);
259        whatsit.set_property("noop", true);
260        whatsit.set_property("reversion", Tokens!());
261        return Ok(Vec::new());
262      }
263      whatsit.set_property("mode", lookup_string_from_sym(pin!("MODE")));
264      // When invoked by leave_horizontal: no reversion, don't close ltx:para
265      if LookupBool!("INTERNAL_PAR") {
266        whatsit.set_property("internal_par", true);
267        whatsit.set_property("reversion", Tokens!());
268      } else {
269        // the frontmatter field whose own paragraph break this is (`frontmatter_paragraph_field`), whose element decides
270        // at construction what it makes (`frontmatter_paragraph_break`)
271        if let Some(item) = frontmatter_paragraph_field() {
272          whatsit.set_property("frontmatter_item", item);
273        }
274      }
275      // \par is a NOOP only in the RAW preamble — before `\begin{document}` opens the
276      // document environment (package loading, stray `\str_lowercase` output in the
277      // preamble, …). Everywhere else it closes the paragraph being built. Two signals,
278      // both existing state, no mode test:
279      //   * `inPreamble` is set from the preamble start until we leave it (AFTER the
280      //     begindocument hooks — see `\begin{document}`); and
281      //   * `current_environment` gains `document` at the START of `\begin{document}`
282      //     (before any hook), so it is on the stack throughout the hooks and the body.
283      // NOOP iff `inPreamble && document NOT on the env stack`. That is exactly the raw
284      // preamble — so a blank line inside `\AtBeginDocument` (which runs inside the
285      // `document` env while `inPreamble` is still set) splits paragraphs (upstream
286      // #2754), while a deferred `\RequirePackage`/`\usepackage` stays legal (inPreamble
287      // still 1 → onlyPreamble guard passes). Because this no-op no longer hinges on
288      // clearing `inPreamble` before the hooks, upstream #2846's early `inPreamble=0`
289      // and the `inBeginDocumentHook` guard-decouple it forced (#2848) are unnecessary.
290      //
291      // Why NOT the Perl note's literal "no-op in vertical mode"? LaTeXML's mode
292      // tracking isn't faithful enough (it stays vertical after a display equation, so a
293      // mode test would drop the blank line between `$$…$$` groups — spacing.xml,
294      // verb.xml; and raw-preamble text is horizontal, so mode can't tell it from a hook
295      // `\par` — expl3 str/text-case fixtures). Context (are we in the document?), not
296      // mode, is the stable discriminator.
297      //
298      // We check the whole env STACK (Perl `grep {…} lookupStackedValues`), not just the
299      // current env, so a hook that opens a nested environment (e.g.
300      // `\AtBeginDocument{\begin{center}…}`) still counts as "in document". The stack
301      // walk only runs when `inPreamble` is set (a tiny window) — `&&` short-circuits in
302      // the body, where `\par` is hot.
303      let in_document = with_stacked_values("current_environment", |envs| {
304        envs
305          .iter()
306          .any(|v| matches!(&**v, Stored::String(s) if with(*s, |e| e == "document")))
307      });
308      let noop = LookupBool!("inPreamble") && !in_document;
309      if noop {
310        whatsit.set_property("noop", true);
311        Ok(Vec::new())
312      } else {
313        // Did the para closing NOW inherit a class from the PRIOR \par (the
314        // deferred mechanism below)? The very first body paragraph never does.
315        let mut class_from_prior = false;
316        if let Some(c) = lookup_value("next_para_class") {
317          // Check if flags were set by prior \par:
318          whatsit.set_property("class", c);
319          { assign_value("next_para_class", Stored::None, None); }
320          class_from_prior = true;
321        }
322        // Per eTeX spec, \interlinepenalties (like \parshape) is reset after each paragraph.
323        { assign_value("interlinepenalties", Stored::None, None); }
324        // `\parindent` is normally defined; if it isn't (None), don't assume zero
325        // and force noindent — skip the override. Witness: 1502.07281.
326        let parindent_zero =
327          lookup_register("\\parindent", Vec::new())?.is_some_and(|r| r.value_of() == 0);
328        // Surpass Perl (OXIDIZED_DESIGN #143, issue #719): flag this paragraph as
329        // the first-paragraph `ltx_noindent` candidate when `\parindent==0` and no
330        // deferred class already applies. The constructor makes the final decision
331        // — it has the DOM and confirms this really is the first paragraph — but
332        // the `\parindent` register must be sampled HERE, at digest time, where it
333        // holds the value in force as the paragraph closed (a paragraph's indent
334        // is fixed by `\parindent` when it begins; the deferred rule below carries
335        // that to the NEXT paragraph, and the first paragraph has no prior `\par`
336        // to have carried it). Witness: issue #719 MWE (`first_para_noindent_719.tex`).
337        if parindent_zero && !class_from_prior {
338          whatsit.set_property("first_para_noindent", true);
339        }
340        // Fish out flags for next ltx:para, to be used when the next \par closes:
341        if parindent_zero {
342          // respect \parindent if no overrides are given
343          { assign_value("next_para_class", "ltx_noindent", None); }
344        }
345        // Vertical adjustments
346        // Each queued `\vadjust` is a list already built, its paragraph ended (the primitive above). A `\par` in a
347        // restricted horizontal box or in math ends no paragraph (tex.web §1096 `end_graf` acts in `hmode` only), so the
348        // queue waits: `Juliet\vadjust{Kilo} \mbox{Lima\par Mike}` sets Kilo after Juliet's line.
349        if !lookup_string_from_sym(pin!("MODE")).ends_with("vertical") {
350          return Ok(Vec::new());
351        }
352        match remove_value("vAdjust") { Some(Stored::VecDigested(material)) => {
353          assign_value("vAdjust", Stored::VecDigested(Vec::new()), Some(Scope::Global));
354          if LookupBool!("INTERNAL_PAR") {
355            material.iter().for_each(keep_inside_paragraph);
356          }
357          Ok(material)
358        } _ => {
359          Ok(Vec::new())
360        }}
361      }
362    },
363    properties => skippable_props,
364    alias => "\\par"
365  );
366  Let!("\\par", "\\lx@normal@par");
367  Tag!("ltx:para", auto_close => true, auto_open => true,
368    after_close => sub[document, node] {
369      prune_empty_para(document, node)?;
370  });
371  Tag!("ltx:p", auto_close => true, auto_open => true,
372    after_close => sub[document, node] {
373      document.trim_node_whitespace(node)?;
374  });

377  // Paragraph Shape
378  //----------------------------------------------------------------------
379  // \parshape               iq specifies an arbitrary paragraph shape.
380  // Acts like a Number register (returns count of lines when read).
381  // Setter reads n pairs of dimensions and stores them in state.
382  DefRegister!("\\parshape", Number::new(0),
383    getter => sub[_args] {
384      with_value("parshape", |val_opt| {
385        if let Some(Stored::VecDequeStored(shape)) = val_opt {
386          Some(RegisterValue::Number(Number::new((shape.len() / 2) as i64)))
387        } else {
388          Some(RegisterValue::Number(Number::new(0)))
389        }
390      })
391    },
392    setter => sub[value, _scope, _args] {
393      let n_val = value.value_of();
394      let n = if n_val < 0 { 0 } else { n_val } as usize;
395      let mut shape = VecDeque::new();
396      for _ in 0..n {
397        let indent = read_dimension().unwrap_or_default();
398        let length = read_dimension().unwrap_or_default();
399        shape.push_back(Stored::Dimension(indent));
400        shape.push_back(Stored::Dimension(length));
401      }
402      assign_value(
403        "parshape",
404        if n > 0 { Stored::VecDequeStored(shape) } else { Stored::None },
405        Some(Scope::Global),
406      );
407    }

461/// A `\vadjust`'s material set while the paragraph around it goes on — at the `\par` that `leave_horizontal` runs
462/// before a display (`INTERNAL_PAR`) — ends its own paragraph as that `\par` does there: inside the paragraph, which
463/// it does not close (`internal_par`), so the display and the text after it stay in that paragraph. Boxes keep their
464/// own paragraphs.
465fn keep_inside_paragraph(material: &Digested) {
466  match material.data() {
467    DigestedData::List(list) => {
468      if let Ok(list) = list.try_borrow() {
469        list.boxes.iter().for_each(keep_inside_paragraph);
470      }
471    },
472    DigestedData::Whatsit(whatsit) => {
473      if let Ok(mut whatsit) = whatsit.try_borrow_mut()
474        && whatsit.get_definition().get_cs_name() == "\\lx@normal@par"
475      {
476        whatsit.set_property("internal_par", true);
477        whatsit.set_property("reversion", Tokens!());
478      }
479    },
480    _ => {},
481  }
