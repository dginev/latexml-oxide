1//! Loading the LaTeX format on demand — the eager trigger list's safety net.
2//!
3//! `TeX.pool` installs the Perl-faithful autoload triggers (`TeX.pool.ltxml`
4//! L33-56: `\documentclass`, `\newcommand`, `\begin`, …), each a macro that
5//! expands to `\@load@latex@pool <trigger>`. Those fire *eagerly*, on a
6//! legitimate use, before any error is raised — and they stay exactly as Perl
7//! writes them.
8//!
9//! But a curated list is incomplete by construction: real LaTeX has no "before
10//! the kernel" (`latex.ltx` *is* the format), so a document may legitimately
11//! open with any kernel command. Anything not on the list was simply undefined
12//! — stubbed as `<ltx:ERROR/>`, with the document derailing from there. The
13//! canonical witness is the "use this class if installed" idiom
14//!
15//! ```tex
16//! \IfFileExists{proc-l.cls}{\documentclass{proc-l}}{\documentclass{amsproc}}
17//! ```
18//!
19//! (arXiv 2605.25877 line 4, 2606.06905, 2606.09693, 2606.16723): with
20//! `\IfFileExists` undefined the conditional collapses, *no class is ever
21//! selected*, and the run cascades into `Fatal:TooManyErrors`. Same-host Perl
22//! LaTeXML fails identically — see `docs/parity/KNOWN_PERL_ERRORS.md`.
23//!
24//! [`autoload_latex_kernel`] closes the gap generally: registered into
25//! `latexml_core::binding::kernel_autoload`, it is consulted at the two
26//! undefined-CS paths (gullet `read_x_token`, stomach `invoke_token_undefined`)
27//! *instead of* erroring, and answers "is this a LaTeX kernel command?" — if so
28//! the format loads and the token is retried. It subsumes the two Rust-only,
29//! witness-driven trigger accretions that used to sit in `tex.rs`'s list:
30//! `\UseRawInputEncoding` (witness 2403.19280) and `\DocumentMetadata`
31//! (witness 2305.08034), which are now just two instances of "a kernel CS used
32//! before the format loaded".
33//!
34//! # The membership oracle is the DUMP, not `latex.ltx`
35//!
36//! The question that has to be answered *before* committing to a load is "what
37//! will be defined **after** `LaTeX.pool` loads?" — because "load it and see if
38//! that helped" would drag the kernel into genuinely plain-TeX documents on
39//! their first undefined CS.
40//!
41//! The right oracle for that is `latex.YYYY.dump.txt`, and deliberately **not**
42//! the host's `latex.ltx`. The dump is not a "YYYY artifact": it is generated by
43//! *our current code* inside a pinned TL-YYYY container, and it is the
44//! **executed** result of that bootstrap — the state delta this engine actually
45//! reaches. `latex.ltx` is source text, so answering the same question from it
46//! would mean re-deciding every `\def` / `\let` / `\csname` / `\@namedef` /
47//! `\ifx` in the file, i.e. re-running the load we are trying to avoid. It is
48//! also the very artifact `LaTeX.pool` goes on to replay, so a catalogue built
49//! from it cannot disagree with what the load installs.
50//! **Do not "correct" this to `latex.ltx`.**
51//!
52//! Only the backward direction exists. A control sequence newer than the
53//! ambient dump year is out of scope by design — we cannot predict a TeX Live
54//! that has not shipped, and the host TeX Live of that age does not have the
55//! command either, so reporting it undefined is correct. There is no
56//! forward-compat seam and none is wanted.
57//!
58//! # Degraded (no-dump) branch
59//!
60//! `LoadFormat('latex')` has two mutually exclusive branches (CLAUDE.md durable
61//! parity rule 1). On the degraded one — `$LATEXML_NODUMP`, or no dump
62//! reachable at all — there is no key set to test against, so
63//! [`latex_kernel_defines`] is always `false` and this mechanism never fires.
64//! Behaviour there is bit-identical to before: the Perl `TeX.pool` L33-56
65//! trigger list remains the only mechanism.
66

74// Process-once cached env var (see WISDOM #56 — getenv hot-path race).
75static INI_MODE: Lazy<bool> = Lazy::new(|| std::env::var_os("LATEXML_INI_MODE").is_some());

77/// Per-session flag: the kernel autoload has already been attempted.
78///
79/// Necessary and not redundant with `LaTeX.pool_loaded`, which
80/// `input_definitions` only sets *after* the pool body has run — so without
81/// this the load could re-enter itself through its own undefined CSes. It is a
82/// State value, so it resets with the session (a test binary runs many
83/// conversions on one thread) while surviving group pops (assigned globally).
84const AUTOLOAD_ATTEMPTED: &str = "LaTeX.pool_autoload_attempted";
85
86thread_local! {
87  /// Control-sequence names the ambient kernel dump defines. Built at most once
88  /// per process, and only on the cold path below — never at startup, so the
89  /// ~161 ms per-process bootstrap is untouched. ~22k boxed names (<1 MB) that
90  /// only ever materialise in a session that reached an undefined CS with no
91  /// LaTeX format loaded.
92  static KERNEL_CS: OnceCell<FxHashSet<Box<str>>> = const { OnceCell::new() };
93}
94
95/// Load `LaTeX.pool`, the way the `\@load@latex@pool` trigger primitive does.
96///
97/// Single definition of "pull the LaTeX format in", shared by that primitive
98/// and by [`autoload_latex_kernel`], so the post-load fixups can never diverge
99/// between the eager and the on-undefined route.
100pub fn load_latex_pool() -> Result<()> {
101  input_definitions("LaTeX", InputDefinitionOptions {
102    extension: Some(Cow::Borrowed("pool")),
103    ..InputDefinitionOptions::default()
104  })?;
105  // Restore our `\documentstyle` impl after the LaTeX pool load. The
106  // latex_dump unconditionally redefines `\documentstyle` to the
107  // kernel-style `\input{latex209.def}\documentclass`; in Perl that's
108  // overridden by latex_constructs.pool.ltxml's DefConstructor, but
109  // our latex_constructs.rs port doesn't redefine it. Re-Let to our
110  // backup to win. See tex_job.rs `\lx@documentstyle@impl` for the
111  // full diagnostic context (hep-th9912229 witness).
112  Let!("\\documentstyle", "\\lx@documentstyle@impl");
113  Ok(())
114}

116/// Does loading `LaTeX.pool` define `cs`? See the module docs for why the
117/// ambient kernel dump is the oracle. `false` on the degraded no-dump branch.
118pub fn latex_kernel_defines(cs: &str) -> bool {
119  KERNEL_CS.with(|cell| cell.get_or_init(build_kernel_catalogue).contains(cs))
120}
121
122fn build_kernel_catalogue() -> FxHashSet<Box<str>> {
123  // Same resolver `latex_dump::load_definitions` uses (env overrides, ambient
124  // year, embedded fallback), so the catalogue can never name a CS from a
125  // different dump than the one the pool would load. Key scan only — no record
126  // is parsed and no State is touched.
127  match crate::latex_dump::resolve_dump_text() {
128    Some((content, _label, _year, _from_embedded)) => dump_reader::collect_meaning_keys(&content),
129    None => FxHashSet::default(),
130  }
131}

133/// The `latexml_core::binding::kernel_autoload` hook: `true` means the LaTeX
134/// format was just loaded *and* `token` now has a meaning, so the caller must
135/// retry it; `false` means take the ordinary bounded `Error:undefined` path.
136pub fn autoload_latex_kernel(token: &Token) -> bool {
137  if token.get_catcode() != Catcode::CS {
138    return false;
139  }
140  // Dump-build (`--init=plain.tex` / `--init=latex.ltx`) raw-loads a format
141  // file against a deliberately minimal state and writes the delta. Pulling
142  // `LaTeX.pool` in from an undefined CS there would put the whole pool into
143  // the snapshot — i.e. a previous run's dump would leak into the next one.
144  // The dump must be built from the format file alone.
145  if *INI_MODE {
146    return false;
147  }
148  // The caller has declared that undefined CSes in this stretch are expected
149  // forward references that later fixups resolve (expl3-code.tex and friends).
150  // Acting on one would load the format on a name that is about to be defined
151  // anyway. Same gate the undefined-CS `Error!` sites use.
152  if lookup_bool("SUPPRESS_UNDEFINED_ERRORS") {
153    return false;
154  }
155  // Once the format is in, the question is moot: the CS is either defined or
156  // genuinely undefined.
157  if lookup_bool("LaTeX.pool_loaded") {
158    return false;
159  }
160  if lookup_bool(AUTOLOAD_ATTEMPTED) {
161    return false;
162  }
163  if !token.with_cs_name(latex_kernel_defines) {
164    return false;
165  }
166  // Claim the single attempt BEFORE loading. The pool load runs arbitrary
167  // definition code that can itself reach an undefined CS, and re-entering
168  // here would recurse; a trigger that re-emits itself spins the gullet
169  // forever, silently, with no error for `too_many_errors` to cap (witness
170  // 2606.21610, see the `def_autoload` closure in `tex.rs`).
171  assign_value(AUTOLOAD_ATTEMPTED, true, Some(Scope::Global));
172  let cs_name = token.with_cs_name(|cs| cs.to_string());
173  Info!(
174    "autoload",
175    "LaTeX",
176    s!("{cs_name} is a LaTeX kernel command — loading LaTeX.pool")
177  );
178  // Hoist whatever the load installs at the current frame up to global: this
179  // can fire at any group depth (unlike the eager triggers, which in practice
180  // fire at top level), and definitions popped by the enclosing `}` would
181  // leave `LaTeX.pool_loaded` set with the format gone. Same guard
182  // `def_autoload` uses in `tex.rs`.
183  let pre_keys = snapshot_top_frame_meaning_keys();
184  if let Err(e) = load_latex_pool() {
185    Warn!(
186      "autoload",
187      "LaTeX",
188      s!("failed to load LaTeX.pool for {cs_name}: {e}")
189    );
190    return false;
191  }
192  hoist_top_frame_meaning_delta(&pre_keys);
193  // The format may still not define it (e.g. a dump key our own LaTeX pool
194  // later clears). Then this was a no-op and the caller reports it undefined,
195  // exactly as before.
196  has_meaning(token)
197}
