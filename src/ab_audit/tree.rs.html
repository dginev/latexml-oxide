1//! Output trees: finding the document directories under a root, loading one document's files, and the
2//! parallel map the subcommands share.
3//!
4//! A document directory holds a conversion's XML and log under one of the names the harnesses use:
5//! `out.xml` + `out.log` (`ab_audit run`, the payload A/B runners), or `<name>.xml` + `<name>.latexml.log`
6//! (the manual sweeps). `meta.tsv` (`ab_audit run`), `verdict.tsv` (the manual sweeps) or `exit.txt` (a
7//! runner's exit code) give the exit code, the wall time and the timeout; the key is the directory's path
8//! under the root. A directory holding only a run record or a stderr/stdout capture is a document too.
9
10use std::{
11  fs,
12  path::{Path, PathBuf},
13  sync::{
14    Mutex,
15    atomic::{AtomicUsize, Ordering},
16  },
17};

19/// One conversion's run record (`meta.tsv`, or `exit.txt`).
20#[derive(Debug, Default, Clone, PartialEq)]
21pub struct RunMeta {
22  pub rc:        Option<i32>,
23  pub secs:      Option<f64>,
24  pub timed_out: bool,
25  /// `input-missing` when the paper's archive was not found.
26  pub note:      String,
27  pub stamp:     String,
28  pub bin_hash:  String,
29  /// A hash of the run settings (`runner.rs` `settings_hash`): a resume needs the same.
30  pub settings:  String,
31  /// The archive's `size:mtime` when it was converted: a replaced archive reruns.
32  pub input:     String,
33}
34
35impl RunMeta {
36  /// Parse `meta.tsv`: `key<TAB>value` lines.
37  pub fn parse(text: &str) -> RunMeta {
38    let mut meta = RunMeta::default();
39    for line in text.lines() {
40      let (k, v) = line.split_once('\t').unwrap_or((line, ""));
41      match k {
42        "rc" => meta.rc = v.parse().ok(),
43        "secs" => meta.secs = v.parse().ok(),
44        "timed_out" => meta.timed_out = v == "1",
45        "note" => meta.note = v.to_string(),
46        "stamp" => meta.stamp = v.to_string(),
47        "bin_hash" => meta.bin_hash = v.to_string(),
48        "settings" => meta.settings = v.to_string(),
49        "input" => meta.input = v.to_string(),
50        _ => {},
51      }
52    }
53    // `timeout` exits 124 on its TERM and 137 when it had to KILL.
54    meta.timed_out |= matches!(meta.rc, Some(124 | 137));
55    meta
56  }
57  pub fn render(&self) -> String {
58    format!(
59      "stamp\t{}\nbin_hash\t{}\nsettings\t{}\ninput\t{}\nrc\t{}\nsecs\t{}\ntimed_out\t{}\nnote\t{}\n",
60      self.stamp,
61      self.bin_hash,
62      self.settings,
63      self.input,
64      self.rc.map(|r| r.to_string()).unwrap_or_default(),
65      self.secs.map(|s| format!("{s:.2}")).unwrap_or_default(),
66      u8::from(self.timed_out),
67      self.note
68    )
69  }
70}

72/// One document's files as read from disk.
73#[derive(Debug, Default, Clone, PartialEq)]
74pub struct DocInput {
75  /// The directory exists in this tree.
76  pub present: bool,
77  pub xml:     Option<Vec<u8>>,
78  pub log:     Option<String>,
79  pub meta:    Option<RunMeta>,
80  /// The captured output — stderr (`stderr.txt`, a sweep's `<name>.log`) then stdout (`stdout.txt`,
81  /// `<name>.stdout`), every capture present — read when the conversion log is missing, has no
82  /// `Status:conversion:` line, or the run timed out or was killed (exit 124/137): its `Fatal:` lines say
83  /// why. A runner that sends stderr into its stdout capture leaves the watchdog's `Fatal:timeout` only in
84  /// `stdout.txt` (diagnostic-honesty audit, 2026-10-11).
85  pub stderr:  Option<String>,
86}

88/// The files a document directory may hold. A directory with any of them is a document, even with no
89/// XML and no log: a manual-sweep timeout leaves only `<name>.log` (stderr), `<name>.stdout` and
90/// `verdict.tsv` (lie-hasse, pgf-PeriodicTableManual, tcolorbox in s148), and such a document must be a
91/// flagged row, not a missing one.
92const MARKERS: &[&str] = &[
93  "out.xml",
94  "out.log",
95  "meta.tsv",
96  "verdict.tsv",
97  "exit.txt",
98  "stderr.txt",
99  "stdout.txt",
100];
101
102/// A document directory's XML and conversion log, if any.
103struct DocFiles {
104  xml: Option<PathBuf>,
105  log: Option<PathBuf>,
106}
107
108/// Whether `dir` is a document directory, and its XML and log.
109fn doc_files(dir: &Path) -> Option<DocFiles> {
110  let name = dir.file_name()?.to_string_lossy().into_owned();
111  let first = |candidates: [String; 2]| {
112    candidates
113      .into_iter()
114      .map(|c| dir.join(c))
115      .find(|p| p.is_file())
116  };
117  let xml = first(["out.xml".into(), format!("{name}.xml")]);
118  let log = first(["out.log".into(), format!("{name}.latexml.log")]);
119  let named = [format!("{name}.log"), format!("{name}.stdout")];
120  let marked =
121    MARKERS.iter().any(|m| dir.join(m).is_file()) || named.iter().any(|m| dir.join(m).is_file());
122  (xml.is_some() || log.is_some() || marked).then_some(DocFiles { xml, log })
123}

125/// The document directories under a root.
126#[derive(Debug, Default, PartialEq)]
127pub struct Discovery {
128  /// Relative paths, sorted.
129  pub keys:         Vec<String>,
130  /// Directories that hold none of the document files and are leaves (no subdirectory) or at the depth
131  /// limit: rows of their own (`unrecognised`), since a layout this tool does not know would otherwise
132  /// vanish from the comparison.
133  pub unrecognised: Vec<String>,
134}
135
136/// The document directories under `root`, searching at most `depth` levels down.
137pub fn discover(root: &Path, depth: usize) -> Discovery {
138  let mut found = Discovery::default();
139  let mut queue = vec![(root.to_path_buf(), 0)];
140  while let Some((dir, level)) = queue.pop() {
141    let rel = || {
142      dir
143        .strip_prefix(root)
144        .map(|r| r.to_string_lossy().into_owned())
145        .unwrap_or_default()
146    };
147    if level > 0 && doc_files(&dir).is_some() {
148      found.keys.push(rel());
149      continue;
150    }
151    let subdirs: Vec<PathBuf> = fs::read_dir(&dir)
152      .map(|entries| {
153        entries
154          .flatten()
155          .filter(|e| e.file_type().is_ok_and(|t| t.is_dir()))
156          .map(|e| e.path())
157          .collect()
158      })
159      .unwrap_or_default();
160    if level > 0 && (subdirs.is_empty() || level >= depth) {
161      // A leaf without document files, or a directory the depth limit cuts: either would vanish from the
162      // comparison, so both are listed.
163      found.unrecognised.push(rel());
164      continue;
165    }
166    queue.extend(subdirs.into_iter().map(|d| (d, level + 1)));
167  }
168  found.keys.sort();
169  found.unrecognised.sort();
170  found
171}

173/// The run record of `verdict.tsv` (`tools/perfect_kernel/run_doc.sh`: bundle, name, status, exit, errors,
174/// fatals, warnings, seconds) or `exit.txt` (an exit code).
175fn sweep_meta(dir: &Path) -> Option<RunMeta> {
176  let (rc, secs) = if let Ok(text) = fs::read_to_string(dir.join("verdict.tsv")) {
177    let cells: Vec<&str> = text.lines().next().unwrap_or("").split('\t').collect();
178    (
179      cells.get(3).and_then(|v| v.parse().ok()),
180      cells.last().and_then(|v| v.parse().ok()),
181    )
182  } else {
183    let text = fs::read_to_string(dir.join("exit.txt")).ok()?;
184    (
185      text.split_whitespace().next().and_then(|v| v.parse().ok()),
186      None,
187    )
188  };
189  Some(RunMeta {
190    rc,
191    secs,
192    timed_out: matches!(rc, Some(124 | 137)),
193    ..RunMeta::default()
194  })
195}

197/// Load one document's files; an unreadable file counts as missing.
198pub fn load(root: &Path, key: &str) -> DocInput {
199  let dir = root.join(key);
200  if !dir.is_dir() {
201    return DocInput::default();
202  }
203  // A directory that exists but holds none of the files is still present: a failed run.
204  let files = doc_files(&dir).unwrap_or(DocFiles { xml: None, log: None });
205  let meta = fs::read_to_string(dir.join("meta.tsv"))
206    .ok()
207    .map(|t| RunMeta::parse(&t))
208    .or_else(|| sweep_meta(&dir));
209  let read_text = |p: PathBuf| {
210    fs::read(p)
211      .ok()
212      .map(|b| String::from_utf8_lossy(&b).into_owned())
213  };
214  let log = files.log.and_then(read_text);
215  let cut_short = matches!(meta.as_ref().and_then(|m| m.rc), Some(124 | 137));
216  let unfinished = log
217    .as_deref()
218    .is_none_or(|l| !l.lines().any(|line| line.contains("Status:conversion:")));
219  let stderr = if unfinished || cut_short {
220    let name = dir
221      .file_name()
222      .map(|n| n.to_string_lossy().into_owned())
223      .unwrap_or_default();
224    let captures: Vec<String> = [
225      "stderr.txt".to_string(),
226      format!("{name}.log"),
227      "stdout.txt".to_string(),
228      format!("{name}.stdout"),
229    ]
230    .into_iter()
231    .map(|f| dir.join(f))
232    .filter(|p| p.is_file())
233    .filter_map(read_text)
234    .collect();
235    (!captures.is_empty()).then(|| captures.concat())
236  } else {
237    None
238  };
239  DocInput {
240    present: true,
241    xml: files.xml.and_then(|p| fs::read(p).ok()),
242    log,
243    meta,
244    stderr,
245  }
246}

248/// A tree's `STAMP` file (written by `ab_audit run`): `key<TAB>value` lines, shown in the report header.
249pub fn stamp_lines(root: &Path) -> Vec<String> {
250  match fs::read_to_string(root.join("STAMP")) {
251    Ok(text) => text.lines().map(|l| l.replace('\t', "=")).collect(),
252    Err(_) => vec!["stamp=unknown (no STAMP file: not produced by `ab_audit run`)".to_string()],
253  }
254}

256/// Map `f` over `items` on `jobs` threads, results in input order. A panic in `f` becomes `on_panic(item)`.
257pub fn par_map<T: Sync, R: Send>(
258  items: &[T],
259  jobs: usize,
260  f: impl Fn(&T) -> R + Sync,
261  on_panic: impl Fn(&T) -> R + Sync,
262  progress: impl Fn(usize) + Sync,
263) -> Vec<R> {
264  let next = AtomicUsize::new(0);
265  let done = AtomicUsize::new(0);
266  let results: Mutex<Vec<Option<R>>> = Mutex::new((0..items.len()).map(|_| None).collect());
267  std::thread::scope(|scope| {
268    for _ in 0..jobs.max(1) {
269      scope.spawn(|| {
270        loop {
271          let i = next.fetch_add(1, Ordering::Relaxed);
272          let Some(item) = items.get(i) else { break };
273          let r = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| f(item)))
274            .unwrap_or_else(|_| on_panic(item));
275          results.lock().unwrap_or_else(|e| e.into_inner())[i] = Some(r);
276          progress(done.fetch_add(1, Ordering::Relaxed) + 1);
277        }
278      });
279    }
280  });
281  results
282    .into_inner()
283    .unwrap_or_else(|e| e.into_inner())
284    .into_iter()
285    .map(|r| r.expect("every item mapped"))
286    .collect()
287}

293  #[test]
294  fn discovery_finds_both_layouts_and_load_reads_them() {
295    let root = std::env::temp_dir().join(format!("ab_audit_tree_{}", std::process::id()));
296    let _ = fs::remove_dir_all(&root);
297    fs::create_dir_all(root.join("1001.0001")).unwrap();
298    fs::write(root.join("1001.0001/out.xml"), "<a/>").unwrap();
299    fs::write(root.join("1001.0001/out.log"), "Status:conversion:0\n").unwrap();
300    fs::write(root.join("1001.0001/meta.tsv"), "rc\t124\nsecs\t300.00\n").unwrap();
301    fs::create_dir_all(root.join("bundle/name")).unwrap();
302    fs::write(
303      root.join("bundle/name/name.latexml.log"),
304      "Status:conversion:3\n",
305    )
306    .unwrap();
307    assert_eq!(discover(&root, 3).keys, vec![
308      "1001.0001".to_string(),
309      "bundle/name".to_string()
310    ]);
311    let one = load(&root, "1001.0001");
312    assert!(one.present && one.xml.is_some() && one.log.is_some());
313    assert!(one.meta.as_ref().is_some_and(|m| m.timed_out), "{one:?}");
314    let two = load(&root, "bundle/name");
315    assert!(two.present && two.xml.is_none() && two.log.is_some());
316    assert!(!load(&root, "nope").present);
317    fs::remove_dir_all(&root).unwrap();
318  }

320  #[test]
321  fn a_sweep_timeout_with_no_xml_and_no_log_is_a_document() {
322    // lie-hasse in s148: `timeout 180` killed the run before any log or XML was written.
323    let root = std::env::temp_dir().join(format!("ab_audit_timeout_{}", std::process::id()));
324    let _ = fs::remove_dir_all(&root);
325    let doc = root.join("lie-hasse/lie-hasse");
326    fs::create_dir_all(&doc).unwrap();
327    fs::write(doc.join("lie-hasse.log"), "contour: Using real outline\n").unwrap();
328    fs::write(doc.join("lie-hasse.stdout"), "").unwrap();
329    fs::write(
330      doc.join("verdict.tsv"),
331      "lie-hasse\tlie-hasse\t124\t124\t1\t0\t5\t180.1\n",
332    )
333    .unwrap();
334    fs::create_dir_all(root.join("odd/leaf")).unwrap();
335    fs::write(root.join("odd/leaf/something.dat"), "").unwrap();
336    let found = discover(&root, 3);
337    assert_eq!(found.keys, vec!["lie-hasse/lie-hasse".to_string()]);
338    assert_eq!(found.unrecognised, vec!["odd/leaf".to_string()]);
339    // A directory the depth limit cuts is listed, not dropped.
340    assert_eq!(discover(&root, 1).unrecognised, vec![
341      "lie-hasse".to_string(),
342      "odd".to_string()
343    ]);
344    let input = load(&root, "lie-hasse/lie-hasse");
345    assert!(
346      input
347        .stderr
348        .as_deref()
349        .is_some_and(|e| e.contains("contour")),
350      "{input:?}"
351    );
352    assert!(
353      input.present && input.xml.is_none() && input.log.is_none(),
354      "{input:?}"
355    );
356    let meta = input.meta.expect("verdict.tsv read");
357    assert_eq!(
358      (meta.rc, meta.secs, meta.timed_out),
359      (Some(124), Some(180.1), true)
360    );
361    fs::remove_dir_all(&root).unwrap();
362  }

364  #[test]
365  fn a_fatal_only_in_the_stdout_capture_is_read() {
366    // A runner that sends stderr into stdout.txt: the watchdog's `Fatal:timeout` is there and nowhere else,
367    // and the log stops before its `Status:conversion:` line.
368    let root = std::env::temp_dir().join(format!("ab_audit_stdout_{}", std::process::id()));
369    let _ = fs::remove_dir_all(&root);
370    let doc = root.join("2101.00001");
371    fs::create_dir_all(&doc).unwrap();
372    fs::write(doc.join("out.log"), "(Loading article.cls)\n").unwrap();
373    fs::write(
374      doc.join("stdout.txt"),
375      "Fatal:timeout:wallclock latexml-oxide: main-level wall-clock timeout after 240s\n",
376    )
377    .unwrap();
378    let input = load(&root, "2101.00001");
379    assert!(
380      input
381        .stderr
382        .as_deref()
383        .is_some_and(|e| e.contains("Fatal:timeout:wallclock")),
384      "{input:?}"
385    );
386    // A finished log needs no capture.
387    fs::write(doc.join("out.log"), "Status:conversion:0\n").unwrap();
388    assert!(load(&root, "2101.00001").stderr.is_none());
389    fs::remove_dir_all(&root).unwrap();
390  }
