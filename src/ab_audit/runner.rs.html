1//! `ab_audit run`: convert a paper list with two binaries into `<out>/A/<id>/` and `<out>/B/<id>/`, then
2//! compare them.
3//!
4//! It refuses to start unless each binary's stamped revision (`--version`'s `(revision <sha>)`, or the
5//! conversion banner of a binary built before that) is the commit the caller expects: 64x r1 and 64ra
6//! ran their A/Bs on stale binaries. A historical replay names the stamp it accepts with
7//! `--accept-stamp <sha>`, and the TSV header prints it; there is no silent bypass. It also refuses:
8//! * two binaries with one stamp but different expected commits, unless `--accept-same-stamp` (the stamp
9//!   then tells them apart by nothing; a loud line is printed either way);
10//! * a binary older than its expected commit (file mtime before the commit's committer time, when
11//!   `git -C <repo>` resolves the commit);
12//! * two byte-identical binaries, unless `--allow-same-binary`;
13//! * an empty paper list.
14//!
15//! **What the stamp cannot prove.** `build.rs` re-runs only when `LATEXML_GIT_SHA` changes, not when HEAD
16//! moves (#528), so a target directory keeps the stamp of its first build: five distinct binaries of
17//! 2026-10-10/11 (64u, 64ufix, 64zb, 64zc, 64w) all say `9569ca27a7`. A build must set
18//! `LATEXML_GIT_SHA=$(git rev-parse --short HEAD)` for the stamp to mean anything, and even then a dirty
19//! tree is not reflected. The mtime check catches only a binary built before its commit, not one built
20//! after it from other sources; the binary hash (recorded per conversion) tells two builds apart but not
21//! which source each came from. The mtime is weak evidence: `cp` without `-p` or a `touch` gives any old
22//! binary a fresh mtime, and an amend or rebase moves the commit's committer time past a binary that does
23//! hold its code (refused until rebuilt, the safe side).
24//!
25//! Per paper and binary, the payload brief's standard command: a fresh directory, `ulimit -v 8912896`,
26//! `timeout 300`, `taskset -c <cores>`, `--preload=ar5iv.sty --timeout=240 --dest=out.xml --log=out.log`.
27//! A finished conversion leaves `meta.tsv` (stamp, binary hash, settings hash, exit code, seconds,
28//! timeout); a rerun skips every paper whose `meta.tsv` names the same stamp, binary hash and settings
29//! (dump dir, timeouts, memory, TeX Live bin, corpus).
30

67/// The inherited environment that shapes a conversion, by prefix: `LATEXML_*` (the dump directory, the
68/// no-dump switch …), `LXML_*`, `TEX*` (TeX Live's search paths), plus `SOURCE_DATE_EPOCH` and `PATH`.
69/// Sorted.
70fn inherited_env() -> Vec<(String, String)> {
71  let mut vars: Vec<(String, String)> = std::env::vars()
72    .filter(|(k, _)| {
73      ["LATEXML_", "LXML_", "TEX"]
74        .iter()
75        .any(|p| k.starts_with(p))
76        || matches!(k.as_str(), "SOURCE_DATE_EPOCH" | "PATH" | "HOME")
77    })
78    .collect();
79  vars.sort();
80  vars
81}

83/// A hash of the settings that shape a conversion's output (the run's options and the inherited
84/// environment), so a resume under other settings reruns.
85fn settings_hash(args: &RunArgs, env: &[(String, String)]) -> String {
86  let mut h = FxHasher::default();
87  h.write(
88    format!(
89      "{:?}|{}|{}|{}|{:?}|{:?}|{env:?}",
90      args.dump_dir, args.timeout, args.wall, args.memory_kb, args.texlive_bin, args.corpus
91    )
92    .as_bytes(),
93  );
94  format!("{:016x}", h.finish())
95}

97/// The committer time of `sha` in `repo`, when git resolves it.
98fn commit_time(repo: &Path, sha: &str) -> Option<i64> {
99  let out = Command::new("git")
100    .arg("-C")
101    .arg(repo)
102    .args(["log", "-1", "--format=%ct", sha, "--"])
103    .output()
104    .ok()?;
105  if !out.status.success() {
106    return None;
107  }
108  String::from_utf8_lossy(&out.stdout).trim().parse().ok()
109}

111/// The binary-age check: `Err` when the binary's mtime is before the expected commit's committer time
112/// (it cannot hold that commit's code), else the note for the report header.
113fn age_verdict(committed: Option<i64>, built: Option<i64>) -> Result<String, String> {
114  match (committed, built) {
115    (Some(c), Some(b)) if b < c => Err(format!(
116      "was built before its expected commit (mtime {b} < committer time {c}): it cannot hold that commit's code"
117    )),
118    (Some(_), Some(_)) => Ok("; built after the expected commit".to_string()),
119    _ => Ok("; mtime not checked (git cannot resolve the expected commit)".to_string()),
120  }
121}

123/// One stamp on both binaries: a loud line always; `Err` (refused) when their expected commits differ and
124/// `--accept-same-stamp` was not given, since the stamp then tells them apart by nothing.
125fn same_stamp_verdict(
126  stamps: [&str; 2],
127  expect: [&str; 2],
128  accept: bool,
129) -> Result<Option<String>, String> {
130  if stamps[0] != stamps[1] {
131    return Ok(None);
132  }
133  let different = expect[0] != expect[1] && !stamp_matches(expect[0], expect[1]);
134  let line = format!(
135    "WARNING: A and B carry ONE stamp {} (expected {} and {}): the stamp does not tell them apart, only the \
136     binary hashes do",
137    stamps[0], expect[0], expect[1]
138  );
139  match (different, accept) {
140    (true, false) => Err(format!(
141      "{line}\nREFUSED: different expected commits behind one stamp; pass --accept-same-stamp to replay such \
142       binaries"
143    )),
144    (true, true) => Ok(Some(format!("{line} (ACCEPTED by --accept-same-stamp)"))),
145    (false, _) => Ok(Some(line)),
146  }
147}

149/// The paper lines of a list file, one per paper key (the first occurrence kept).
150pub fn paper_lines(text: &str) -> Vec<String> {
151  let mut seen = std::collections::HashSet::new();
152  text
153    .lines()
154    .map(|l| l.split('\t').next().unwrap_or("").trim().to_string())
155    .filter(|l| !l.is_empty() && !l.starts_with('#'))
156    .filter(|l| seen.insert(paper_key(l)))
157    .collect()
158}

160/// A paper's key: an archive path's file stem, an old-style `hep-th/9703192` without its slash, else the
161/// line itself.
162pub fn paper_key(line: &str) -> String {
163  if line.ends_with(".zip") {
164    return Path::new(line)
165      .file_stem()
166      .map(|s| s.to_string_lossy().into_owned())
167      .unwrap_or_else(|| line.to_string());
168  }
169  match line.split_once('/') {
170    Some((archive, number))
171      if number.len() == 7
172        && number.bytes().all(|b| b.is_ascii_digit())
173        && !archive.contains('/') =>
174    {
175      format!("{archive}{number}")
176    },
177    _ => line.to_string(),
178  }
179}

181/// The archive of a paper line: the path itself when it names a `.zip`, else
182/// `<corpus>/<yymm>/<id>/<id>.zip` with `yymm` the first four digits of the id (`hep-th9703192` → 9703).
183pub fn paper_zip(line: &str, corpus: &Path) -> PathBuf {
184  if line.ends_with(".zip") {
185    return PathBuf::from(line);
186  }
187  let id = paper_key(line);
188  let digits: String = id
189    .chars()
190    .skip_while(|c| !c.is_ascii_digit())
191    .take_while(char::is_ascii_digit)
192    .collect();
193  let yymm = digits.get(..4).unwrap_or(&digits);
194  corpus.join(yymm).join(&id).join(format!("{id}.zip"))
195}

197/// Whether a stamp is the expected commit: one a prefix of the other, at least 7 hex digits each.
198pub fn stamp_matches(expected: &str, stamp: &str) -> bool {
199  let (e, s) = (
200    expected.trim().to_ascii_lowercase(),
201    stamp.trim().to_ascii_lowercase(),
202  );
203  e.len() >= 7
204    && s.len() >= 7
205    && s.bytes().all(|b| b.is_ascii_hexdigit())
206    && (e.starts_with(&s) || s.starts_with(&e))
207}

209/// The revision in `… (revision <stamp>)` or `…; revision <stamp>) started …`.
210pub fn parse_revision(text: &str) -> Option<String> {
211  let at = text.find("revision ")? + "revision ".len();
212  let stamp: String = text[at..]
213    .chars()
214    .take_while(|c| !matches!(c, ')' | ' ' | '\n'))
215    .collect();
216  (!stamp.is_empty()).then_some(stamp)
217}

219/// A binary's stamped revision: `--version`, else the banner of a one-line conversion (binaries built
220/// before `--version` printed it).
221/// `scratch` is the run's own `<root>/.stamp_<tag>` (checked by [`scratch_dir`]), never a shared
222/// temporary directory.
223fn read_stamp(
224  bin: &Path,
225  texlive_bin: &Path,
226  scratch: &Path,
227) -> Result<(String, &'static str), String> {
228  let version = Command::new(bin)
229    .arg("--version")
230    .output()
231    .map_err(|e| format!("{}: {e}", bin.display()))?;
232  if let Some(stamp) = parse_revision(&String::from_utf8_lossy(&version.stdout)) {
233    return Ok((stamp, "--version"));
234  }
235  let dir = scratch;
236  let _ = fs::remove_dir_all(dir);
237  fs::create_dir_all(dir).map_err(|e| e.to_string())?;
238  fs::write(
239    dir.join("stamp.tex"),
240    "\\documentclass{article}\\begin{document}x\\end{document}\n",
241  )
242  .map_err(|e| e.to_string())?;
243  let status = Command::new("timeout")
244    .args(["120"])
245    .arg(bin)
246    .args(["--dest=stamp.xml", "--log=stamp.log", "stamp.tex"])
247    .current_dir(dir)
248    .env("PATH", tl_path(texlive_bin))
249    .output();
250  let log = fs::read_to_string(dir.join("stamp.log")).unwrap_or_default();
251  let _ = fs::remove_dir_all(dir);
252  status.map_err(|e| format!("{}: {e}", bin.display()))?;
253  parse_revision(log.lines().next().unwrap_or(""))
254    .map(|s| (s, "banner"))
255    .ok_or_else(|| {
256      format!(
257        "{}: no `revision` in `--version` or in the conversion banner",
258        bin.display()
259      )
260    })
261}

263/// A hash of the binary's bytes, so two copies of one stamp (a rebuild that did not move the stamp) are
264/// still told apart.
265fn hash_file(path: &Path) -> Result<String, String> {
266  let bytes = fs::read(path).map_err(|e| format!("{}: {e}", path.display()))?;
267  let mut h = FxHasher::default();
268  for chunk in bytes.chunks(1 << 16) {
269    h.write(chunk);
270  }
271  Ok(format!("{:016x}", h.finish()))
272}

277    texlive_bin.display(),

279  )

292/// A paper's directory, `<root>/<tag>/<key>`, provably inside the (canonical) output root: the key must be
293/// one plain path component (no `..`, `.`, separator or absolute path), so the `remove_dir_all` that gives
294/// each conversion a fresh directory can never reach outside the root.
295pub fn paper_dir(root: &Path, tag: &str, key: &str) -> Result<PathBuf, String> {
296  let mut parts = Path::new(key).components();
297  let one_plain =
298    matches!(parts.next(), Some(std::path::Component::Normal(_))) && parts.next().is_none();
299  if !one_plain || key.contains('/') || key.contains('\\') {
300    return Err(format!("paper key {key:?} is not one plain path component"));
301  }
302  let dir = root.join(tag).join(key);
303  if !root.is_absolute() || !dir.starts_with(root) || dir.parent() != Some(&root.join(tag)) {
304    return Err(format!(
305      "{} is not inside the output root {}",
306      dir.display(),
307      root.display()
308    ));
309  }
310  Ok(dir)
311}

313/// Create `<root>/<tag>` and prove it is a real directory directly inside the canonical `root`: a
314/// symlinked `A`/`B` would carry every per-paper `remove_dir_all` outside it.
315pub fn tag_dir(root: &Path, tag: &str) -> Result<PathBuf, String> {
316  let dir = root.join(tag);
317  fs::create_dir_all(&dir).map_err(|e| format!("{}: {e}", dir.display()))?;
318  let link = fs::symlink_metadata(&dir)
319    .map(|m| m.file_type().is_symlink())
320    .unwrap_or(true);
321  let real = fs::canonicalize(&dir).map_err(|e| format!("{}: {e}", dir.display()))?;
322  if link || real.parent() != Some(root) {
323    return Err(format!(
324      "{} is a symlink or not directly inside the output root {}",
325      dir.display(),
326      root.display()
327    ));
328  }
329  Ok(dir)
330}

332/// The stamp-reading scratch `<root>/.stamp_<tag>`, inside the canonical root (not a shared temporary
333/// directory), refused when it is a symlink.
334pub fn scratch_dir(root: &Path, tag: &str) -> Result<PathBuf, String> {
335  let dir = root.join(format!(".stamp_{tag}"));
336  // `dir` is `root` joined with one plain name (the tag is `A` or `B`), so its parent is `root` by
337  // construction; a symlink is what could carry the removal elsewhere.
338  if fs::symlink_metadata(&dir).is_ok_and(|m| m.file_type().is_symlink()) {
339    return Err(format!("{} is a symlink", dir.display()));
340  }
341  Ok(dir)
342}

344/// Run one paper on one side unless done for the same stamp and binary.
345fn run_one(
346  args: &RunArgs,
347  root: &Path,
348  side: &Side,
349  line: &str,
350  settings: &str,
351) -> std::io::Result<bool> {
352  let key = paper_key(line);
353  let dir = paper_dir(root, side.tag, &key).map_err(std::io::Error::other)?;
354  let zip = paper_zip(line, &args.corpus);
355  // The archive's size and mtime: a replaced archive reruns.
356  let input = fs::metadata(&zip)
357    .ok()
358    .map(|m| {
359      let mtime = m
360        .modified()
361        .ok()
362        .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok());
363      format!("{}:{}", m.len(), mtime.map_or(0, |d| d.as_nanos()))
364    })
365    .unwrap_or_default();
366  if let Ok(text) = fs::read_to_string(dir.join("meta.tsv")) {
367    let meta = RunMeta::parse(&text);
368    // A missing archive is never a finished conversion: it reruns once the archive is there.
369    if meta.stamp == side.stamp
370      && meta.bin_hash == side.hash
371      && meta.settings == settings
372      && meta.input == input
373      && meta.note != "input-missing"
374    {
375      return Ok(false);
376    }
377  }
378  let _ = fs::remove_dir_all(&dir);
379  fs::create_dir_all(&dir)?;
380  let mut meta = RunMeta {
381    stamp: side.stamp.clone(),
382    bin_hash: side.hash.clone(),
383    settings: settings.to_string(),
384    input,
385    ..RunMeta::default()
386  };
387  if !zip.is_file() {
388    meta.note = "input-missing".to_string();
389    fs::write(dir.join("meta.tsv"), meta.render())?;
390    return Ok(true);
391  }
392  let start = Instant::now();
393  let mut cmd = Command::new("bash");
394  cmd
395    .arg("-c")
396    .arg(SCRIPT)
397    .arg("ab_audit")
398    .arg(args.memory_kb.to_string())
399    .arg(args.wall.to_string())
400    .arg(&args.cores)
401    .arg(&side.bin)
402    .arg(args.timeout.to_string())
403    .arg(&zip)
404    .current_dir(&dir)
405    .env("PATH", tl_path(&args.texlive_bin));
406  if let Some(dumps) = &args.dump_dir {
407    cmd.env("LATEXML_DUMP_DIR", dumps);
408  }
409  let status = cmd.status()?;
410  meta.secs = Some(start.elapsed().as_secs_f64());
411  meta.rc = Some(status.code().unwrap_or(-1));
412  meta.timed_out = matches!(meta.rc, Some(124 | 137));
413  fs::write(dir.join("meta.tsv"), meta.render())?;
414  Ok(true)
415}

417/// `ab_audit run`. Returns the process exit code: 2 refused, 1 a regression or review row, 0 otherwise.
418pub fn run(args: &RunArgs) -> std::io::Result<i32> {
419  let lines = paper_lines(&fs::read_to_string(&args.papers)?);
420  // The output root, canonical, so every per-paper directory removed below is checked to lie inside it.
421  fs::create_dir_all(&args.out)?;
422  let root = fs::canonicalize(&args.out)?;
423  let unsafe_keys: Vec<String> = lines
424    .iter()
425    .filter_map(|l| paper_dir(&root, "A", &paper_key(l)).err())
426    .collect();
427  if !unsafe_keys.is_empty() {
428    eprintln!("ab_audit run: REFUSED: {}", unsafe_keys.join("; "));
429    return Ok(2);
430  }
431  if lines.is_empty() {
432    eprintln!(
433      "ab_audit run: REFUSED: no papers in {}",
434      args.papers.display()
435    );
436    return Ok(2);
437  }
438  let settings = settings_hash(args, &inherited_env());
439  let mut sides = Vec::new();
440  let mut verdicts = Vec::new();
441  for (i, tag) in ["A", "B"].into_iter().enumerate() {
442    let bin = fs::canonicalize(&args.bins[i]).unwrap_or_else(|_| args.bins[i].clone());
443    let scratch = match scratch_dir(&root, tag) {
444      Ok(dir) => dir,
445      Err(e) => {
446        eprintln!("ab_audit run: REFUSED: {e}");
447        return Ok(2);
448      },
449    };
450    let (stamp, source) = match read_stamp(&bin, &args.texlive_bin, &scratch) {
451      Ok(s) => s,
452      Err(e) => {
453        eprintln!("ab_audit run: REFUSED: {tag}: cannot read the stamp of {e}");
454        return Ok(2);
455      },
456    };
457    let hash = hash_file(&bin).map_err(std::io::Error::other)?;
458    let expected = &args.expect[i];
459    let mut verdict = if stamp_matches(expected, &stamp) {
460      "matches the expected commit".to_string()
461    } else if args.accept.iter().any(|a| a == &stamp) {
462      format!("ACCEPTED by --accept-stamp {stamp} (expected {expected})")
463    } else {
464      eprintln!(
465        "ab_audit run: REFUSED: {tag} {} is stamped {stamp} ({source}), not the expected {expected}.\n  Rebuild \
466         it from {expected} with LATEXML_GIT_SHA=$(git rev-parse --short HEAD) (build.rs keeps a stale stamp \
467         otherwise), or replay a historical binary with --accept-stamp {stamp} (printed in the report header).",
468        bin.display()
469      );
470      return Ok(2);
471    };
472    let mtime = fs::metadata(&bin)
473      .and_then(|m| m.modified())
474      .ok()
475      .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
476      .map(|d| d.as_secs() as i64);
477    match age_verdict(commit_time(&args.repo, expected), mtime) {
478      Ok(note) => verdict.push_str(&note),
479      Err(why) => {
480        eprintln!("ab_audit run: REFUSED: {tag} {} {why}", bin.display());
481        return Ok(2);
482      },
483    }
484    if verdict.ends_with("(git cannot resolve the expected commit)") {
485      verdict.push_str(&format!(": {expected} in {}", args.repo.display()));
486    }
487    verdicts.push(format!("{source}: {verdict}"));
488    sides.push(Side { tag, bin, stamp, hash });
489  }
490  let mut header = Vec::new();
491  let stamps = [sides[0].stamp.as_str(), sides[1].stamp.as_str()];
492  match same_stamp_verdict(
493    stamps,
494    [&args.expect[0], &args.expect[1]],
495    args.accept_same_stamp,
496  ) {
497    Ok(Some(line)) => {
498      eprintln!("ab_audit run: {line}");
499      header.push(line);
500    },
501    Ok(None) => {},
502    Err(line) => {
503      eprintln!("ab_audit run: {line}");
504      return Ok(2);
505    },
506  }
507  if sides[0].hash == sides[1].hash && !args.allow_same_binary {
508    eprintln!(
509      "ab_audit run: REFUSED: A and B are the same binary (hash {}); pass --allow-same-binary for a noise baseline",
510      sides[0].hash
511    );
512    return Ok(2);
513  }
514  header.push(format!(
515    "command: ulimit -v {} timeout {} taskset -c {} <bin> --preload=ar5iv.sty --timeout={} -j {}{} (settings {settings})",
516    args.memory_kb,
517    args.wall,
518    args.cores,
519    args.timeout,
520    args.jobs,
521    args
522      .dump_dir
523      .as_ref()
524      .map(|d| format!(" LATEXML_DUMP_DIR={}", d.display()))
525      .unwrap_or_default()
526  ));
527  for (side, verdict) in sides.iter().zip(&verdicts) {
528    if let Err(e) = tag_dir(&root, side.tag) {
529      eprintln!("ab_audit run: REFUSED: {e}");
530      return Ok(2);
531    }
532    fs::write(
533      root.join(side.tag).join("STAMP"),
534      format!(
535        "binary\t{}\nstamp\t{}\nbin_hash\t{}\nverdict\t{verdict}\n",
536        side.bin.display(),
537        side.stamp,
538        side.hash
539      ),
540    )?;
541  }
542  let jobs: Vec<(usize, &str)> = lines
543    .iter()
544    .flat_map(|l| [(0, l.as_str()), (1, l.as_str())])
545    .collect();
546  let total = jobs.len();
547  let start = Instant::now();
548  let ran = par_map(
549    &jobs,
550    args.jobs,
551    |(side, line)| {
552      run_one(args, &root, &sides[*side], line, &settings).unwrap_or_else(|e| {
553        eprintln!("ab_audit run: {line} {}: {e}", sides[*side].tag);
554        true
555      })
556    },
557    |_| true,
558    |done| {
559      if done % 50 == 0 || done == total {
560        eprintln!(
561          "ab_audit run: {done}/{total} conversions ({:.0} s)",
562          start.elapsed().as_secs_f64()
563        );
564      }
565    },
566  );
567  header.push(format!(
568    "{} conversions run, {} resumed as done",
569    ran.iter().filter(|r| **r).count(),
570    ran.iter().filter(|r| !**r).count()
571  ));
572  let keys_file = root.join("papers.txt");
573  fs::write(
574    &keys_file,
575    lines
576      .iter()
577      .map(|l| paper_key(l) + "\n")
578      .collect::<String>(),
579  )?;
580  let out = Output { prefix: root.join("compare") };
581  let regression = run_compare(
582    &root.join("A"),
583    &root.join("B"),
584    Some(&keys_file),
585    &out,
586    args.jobs,
587    &header,
588  )?;
589  Ok(i32::from(regression))
590}
