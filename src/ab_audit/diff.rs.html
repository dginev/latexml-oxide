21/// Severity tiers, most severe first.
22#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
23pub enum Tier {
24  Regression = 1,
25  Review = 2,
26  Change = 3,
27  Same = 4,
28}

36      Tier::Same => "same",
37    }
38  }

111/// A flag's tier and its rank within all flags (lower is more severe).
112pub fn rank(flag: &str) -> (Tier, usize) {
113  // The same problem on both sides keeps the problem's tier; a regression one ranks after every row that B
114  // made worse.
115  if let Some(problem) = flag.strip_suffix(":both") {
116    return match rank(problem) {
117      (Tier::Regression, _) => (Tier::Regression, REGRESSION.len() - 1),
118      other => other,
119    };
120  }
121  let (base, detail) = flag.split_once(':').unwrap_or((flag, ""));
122  // Text or attributes leaving an `ERROR` (an undefined macro now defined) is not a regression.
123  if matches!(base, "lost-text" | "moved-text" | "attrs-changed") && detail.starts_with("ERROR") {
124    return (Tier::Change, REGRESSION.len() + REVIEW.len());
125  }
126  let base = if flag == "gained-elements:ERROR" {
127    flag
128  } else {
129    base
130  };
131  if let Some(i) = REGRESSION.iter().position(|f| *f == base || *f == flag) {
132    (Tier::Regression, i)
133  } else if let Some(i) = REVIEW.iter().position(|f| *f == base) {
134    (Tier::Review, REGRESSION.len() + i)
135  } else {
136    (Tier::Change, REGRESSION.len() + REVIEW.len())
137  }
138}

140/// A finished row.
141#[derive(Debug, Default, Clone)]
142pub struct Row {
143  pub key:     String,
144  pub tier:    Option<Tier>,
145  pub flags:   Vec<String>,
146  /// Already formatted TSV cells after `key`, `tier`, `flags`.
147  pub cells:   Vec<String>,
148  /// The detail text for the per-paper file.
149  pub details: String,
150}
151
152impl Row {
153  /// Sort, dedup and tier the flags.
154  pub fn finish(&mut self) {
155    self.flags.sort_by_key(|f| (rank(f).1, f.clone()));
156    self.flags.dedup();
157    self.tier = Some(
158      self
159        .flags
160        .iter()
161        .map(|f| rank(f).0)
162        .min()
163        .unwrap_or(Tier::Same),
164    );
165  }
166  pub fn severity(&self) -> usize { self.flags.first().map(|f| rank(f).1).unwrap_or(usize::MAX) }
167}

184/// Read one side, parsing its XML once for the facts and, with `want_canon`, the canonical lines.
185fn read_side(input: &DocInput, want_canon: bool, reuse: Option<&Result<XmlFacts, String>>) -> Side {
186  let mut canon = None;
187  let xml = match (&input.xml, reuse) {
188    (_, Some(xml)) => xml.clone(),
189    (None, None) => Err("missing".to_string()),
190    (Some(bytes), None) => parse(bytes).map(|doc| {
191      if want_canon {
192        canon = Some(canonical(&doc));
193      }
194      read_xml(&doc)
195    }),
196  };
197  Side {
198    log: input.log.as_deref().map(read_log),
199    xml,
200    canon,
201    meta: input.meta.clone(),
202    present: input.present,
203    stderr: input.stderr.as_deref().map(read_log),
204    stderr_fatals: input
205      .stderr
206      .as_deref()
207      .map(|t| {
208        t.lines()
209          .map(|l| strip_ansi(l).into_owned())
210          .filter(|l| l.starts_with("Fatal:"))
211          .take(5)
212          .collect()
213      })
214      .unwrap_or_default(),
215  }
216}

218/// The problems of one side that make its row a regression: missing pieces, unreadable XML, no status,
219/// a timeout.
220fn side_problems(side: &Side, tag: &str, flags: &mut Vec<String>) {
221  if !side.present {
222    flags.push(format!("doc-missing:{tag}"));
223    return;
224  }
225  if side
226    .meta
227    .as_ref()
228    .is_some_and(|m| m.note == "input-missing")
229  {
230    flags.push(format!("input-missing:{tag}"));
231  }
232  match &side.xml {
233    Err(e) if e == "missing" => flags.push(format!("xml-missing:{tag}")),
234    Err(_) => flags.push(format!("xml-unparsable:{tag}")),
235    Ok(_) => {},
236  }
237  match &side.log {
238    None => {
239      flags.push(format!("log-missing:{tag}"));
240      if side.stderr.as_ref().is_some_and(|e| e.fatals > 0) {
241        flags.push(format!("fatal-in-stderr:{tag}"));
242      }
243    },
244    Some(log) if log.status_bad => flags.push(format!("status-unparsable:{tag}")),
245    Some(log) if log.status.is_none() => {
246      flags.push(format!("no-status:{tag}"));
247      if side.stderr.as_ref().is_some_and(|e| e.fatals > 0) {
248        flags.push(format!("fatal-in-stderr:{tag}"));
249      }
250    },
251    Some(log) => {
252      if log.status_mismatch() {
253        flags.push(format!("status-diag-mismatch:{tag}"));
254      }
255      // A `Fatal:`/`Error:` printed inside a line (after a `(Loading …` mark) is as real as one at a
256      // line start, though cortex does not count it; a mid-line warning only needs a look.
257      if log.midline_fatals > 0 {
258        flags.push(format!("fatal-midline:{tag}"));
259      }
260      if log.midline_errors > 0 {
261        flags.push(format!("error-midline:{tag}"));
262      }
263      if log.midline_warnings > 0 {
264        flags.push(format!("diag-midline:{tag}"));
265      }
266    },
267  }
268  if side.meta.as_ref().is_some_and(|m| m.timed_out) {
269    // Exit 137 with a `Fatal:oom` (the allocation-failure exit, `bin/latexml_oxide.rs` `alloc_error`) is
270    // out of memory, not a timeout.
271    let oom = |l: &LogFacts| l.classes.keys().any(|c| c.starts_with("Fatal:oom"));
272    let is_oom = side.meta.as_ref().and_then(|m| m.rc) == Some(137)
273      && (side.log.as_ref().is_some_and(oom) || side.stderr.as_ref().is_some_and(oom));
274    flags.push(format!("{}:{tag}", if is_oom { "oom" } else { "timeout" }));
275  }
276  // A Fatal (status 3) must exit non-zero (`bin/latexml_oxide.rs`, Perl bin/latexml:151).
277  if side.meta.as_ref().and_then(|m| m.rc) == Some(0)
278    && side.log.as_ref().is_some_and(|l| l.status == Some(3))
279  {
280    flags.push(format!("contract-violation:{tag}"));
281  }
282  // An exit code that neither a clean run (0), a Fatal (1 with status 3: `bin/latexml_oxide.rs`, Perl
283  // bin/latexml:151), nor a timeout (124 TERM, 137 KILL or the allocation-failure exit) explains: a
284  // crash, a signal (-1), a refused start.
285  let fatal_status = side.log.as_ref().is_some_and(|l| l.status == Some(3));
286  if let Some(rc) = side.meta.as_ref().and_then(|m| m.rc)
287    && !matches!(rc, 0 | 124 | 137)
288    && !(rc == 1 && fatal_status)
289  {
290    flags.push(format!("crash:{tag}"));
291  }
292}

300/// The words of segments, a boundary taken as a space: a word that gains a footnote mark or an index
301/// entry beside it is still the same word (spacing is compared apart, [`boundary_spacing`]).
302fn tokens<'a>(segments: impl Iterator<Item = &'a String>) -> FxHashMap<&'a str, i64> {
303  let mut m: FxHashMap<&str, i64> = FxHashMap::default();
304  for s in segments {
305    for t in s.split([' ', BOUNDARY]).filter(|t| !t.is_empty()) {
306      *m.entry(t).or_default() += 1;
307    }
308  }
309  m
310}

320/// Per boundary (a non-wrapper child inside the text), keyed by the words before and after it: whether it
321/// is glued to the word before and to the word after. `see <Math/> and` is `(see, and) → (false, false)`,
322/// `see<Math/> and` is `(see, and) → (true, false)`.
323fn boundary_spacing(segments: &[String]) -> FxHashMap<(String, String), Vec<(bool, bool)>> {
324  #[derive(PartialEq)]
325  enum Piece<'a> {
326    Word(&'a str),
327    Space,
328    Boundary,
329  }
330  let mut out: FxHashMap<(String, String), Vec<(bool, bool)>> = FxHashMap::default();
331  for segment in segments {
332    if !segment.contains(BOUNDARY) {
333      continue;
334    }
335    let mut pieces = Vec::new();
336    let mut word_start = None;
337    for (i, c) in segment.char_indices() {
338      if c == ' ' || c == BOUNDARY {
339        if let Some(start) = word_start.take() {
340          pieces.push(Piece::Word(&segment[start..i]));
341        }
342        pieces.push(if c == ' ' {
343          Piece::Space
344        } else {
345          Piece::Boundary
346        });
347      } else if word_start.is_none() {
348        word_start = Some(i);
349      }
350    }
351    if let Some(start) = word_start {
352      pieces.push(Piece::Word(&segment[start..]));
353    }
354    let word = |p: &Piece<'_>| match p {
355      Piece::Word(w) => Some(w.to_string()),
356      _ => None,
357    };
358    for (i, piece) in pieces.iter().enumerate() {
359      if *piece != Piece::Boundary {
360        continue;
361      }
362      let before = pieces[..i].iter().rev().find_map(word).unwrap_or_default();
363      let after = pieces[i + 1..].iter().find_map(word).unwrap_or_default();
364      let glued_before = i > 0 && matches!(pieces[i - 1], Piece::Word(_));
365      let glued_after = matches!(pieces.get(i + 1), Some(Piece::Word(_)));
366      out
367        .entry((before, after))
368        .or_default()
369        .push((glued_before, glued_after));
370    }
371  }
372  out
373}

375/// Whether a boundary present on both sides lost a space before or after it.
376fn space_lost(a: &[String], b: &[String]) -> Vec<String> {
377  // Per key, the totals of glued sides: one boundary splitting into two (`see<Math/><Math/>`, s1e) still
378  // compares.
379  let totals = |v: &Vec<(bool, bool)>| {
380    (
381      v.iter().filter(|x| x.0).count(),
382      v.iter().filter(|x| x.1).count(),
383    )
384  };
385  let (sa, sb) = (boundary_spacing(a), boundary_spacing(b));
386  let mut lost = Vec::new();
387  for (key, va) in &sa {
388    if let Some(vb) = sb.get(key) {
389      let ((ga, aa), (gb, ab)) = (totals(va), totals(vb));
390      if gb > ga || ab > aa {
391        lost.push(format!("{:?}␣{:?}", key.0, key.1));
392      }
393    }
394  }
395  lost.sort();
396  lost
397}

399/// A token's pieces for the global-loss check: alphanumeric runs and single other characters, so
400/// `foo,` that became `foo` + `,` (a run moved into its own element) loses nothing.
401fn pieces(token: &str) -> Vec<&str> {
402  let mut out = Vec::new();
403  let mut run: Option<usize> = None;
404  for (i, c) in token.char_indices() {
405    if c.is_alphanumeric() {
406      run.get_or_insert(i);
407    } else {
408      if let Some(start) = run.take() {
409        out.push(&token[start..i]);
410      }
411      out.push(&token[i..i + c.len_utf8()]);
412    }
413  }
414  if let Some(start) = run {
415    out.push(&token[start..]);
416  }
417  out
418}

420/// The multiset of [`pieces`] of every token of some segments.
421fn piece_counts<'a>(segments: impl Iterator<Item = &'a String>) -> FxHashMap<&'a str, i64> {
422  let mut m: FxHashMap<&str, i64> = FxHashMap::default();
423  for s in segments {
424    for t in s.split([' ', BOUNDARY]).filter(|t| !t.is_empty()) {
425      for p in pieces(t) {
426        *m.entry(p).or_default() += 1;
427      }
428    }
429  }
430  m
431}

433/// Pair lost and gained tokens that differ only by numbers within [`JITTER`] of each other (`3.80254pt` and
434/// `3.80252pt` in a `Math@tex`): removed from both, and counted.
435fn take_jitter<'a>(lost: &mut BTreeMap<&'a str, i64>, gained: &mut BTreeMap<&'a str, i64>) -> i64 {
436  let mut pairs = 0;
437  let numeric: Vec<&str> = lost
438    .keys()
439    .copied()
440    .filter(|t| t.bytes().any(|b| b.is_ascii_digit()))
441    .collect();
442  for l in numeric {
443    let (nl, dl, sl) = numbers_with_digits(l);
444    let partners: Vec<&str> = gained
445      .keys()
446      .copied()
447      .filter(|g| {
448        let (ng, dg, sg) = numbers_with_digits(g);
449        // An equal pair always agrees; a differing one is rounding only between computed values printed
450        // with at least 3 fractional digits on both sides (`3.80254pt`, mhchem `28.04677pt`/`28.04684pt`)
451        // within [`JITTER`] and within ten units of the last printed place (so `2101.00001`, five places, is
452        // not `2102.00001`). Everywhere, attributes included: `2101.00001`→`2102.00001` in an href,
453        // `Smith2019`→`Smith2020`, `fig1000`→`fig1001` and `2019`→`2020` are changed content.
454        sg == sl
455          && ng.len() == nl.len()
456          && (0..ng.len()).all(|k| {
457            let (p, q) = (nl[k], ng[k]);
458            let last_place = 10f64.powi(-(dl[k].min(dg[k]) as i32));
459            p == q
460              || (dl[k] >= 3
461                && dg[k] >= 3
462                && (p - q).abs() <= JITTER * p.abs().max(q.abs())
463                && (p - q).abs() <= 10.0 * last_place)
464          })
465      })
466      .collect();
467    for g in partners {
468      let n = lost
469        .get(l)
470        .copied()
471        .unwrap_or(0)
472        .min(gained.get(g).copied().unwrap_or(0));
473      if n <= 0 {
474        continue;
475      }
476      for (m, t) in [(&mut *lost, l), (&mut *gained, g)] {
477        let left = m.get(t).copied().unwrap_or(0) - n;
478        if left > 0 {
479          m.insert(t, left);
480        } else {
481          m.remove(t);
482        }
483      }
484      pairs += n;
485    }
486  }
487  pairs
488}

490/// The relative difference below which two printed values are one value rounded differently (0.1 %):
491/// `3.80254pt` and `3.80252pt` are, `0.001` and `0.005` are not.
492const JITTER: f64 = 1e-3;

494/// What a gained token is when it is a number: `Some(true)` a TeX dimension or a dotted register value
495/// (`24pt`, `-36.83386pt`, `0.70.2`), `Some(false)` a bare number (`3`, `0.5`, `2003`), `None` other text.
496/// Surrounding `()[],;:` and a trailing `.` are stripped first (`24pt,`, `(24pt)`, `2003.`).
497fn number_kind(token: &str) -> Option<bool> {
498  let t = token.trim_matches(|c| matches!(c, '(' | ')' | '[' | ']' | ',' | ';' | ':'));
499  let t = t.strip_suffix('.').unwrap_or(t);
500  let body = t.strip_prefix(['-', '+']).unwrap_or(t);
501  let digits_end = body
502    .find(|c: char| !c.is_ascii_digit() && c != '.')
503    .unwrap_or(body.len());
504  let (number, unit) = body.split_at(digits_end);
505  if !number.bytes().any(|b| b.is_ascii_digit()) {
506    return None;
507  }
508  let tex_unit = matches!(
509    unit,
510    "pt" | "em" | "ex" | "bp" | "mm" | "cm" | "in" | "sp" | "pc" | "dd" | "cc" | "mu"
511  );
512  match unit {
513    _ if tex_unit => Some(true),
514    "" => Some(number.matches('.').count() >= 2),
515    _ => None,
516  }
517}

519/// [`number_kind`] of a token, or else of its [`pieces`]: `heading,24pt` and `24pt\relax` hold a dimension.
520/// A dotted section number (`1.2.3`) reads as a dotted register value: a false flag the rules accept, since
521/// a section number newly typeset in prose is worth a look either way.
522fn token_number_kind(token: &str) -> Option<bool> {
523  number_kind(token).or_else(|| pieces(token).into_iter().filter_map(number_kind).max())
524}

538/// `a - b` as a multiset, in a stable order.
539fn minus<'a>(a: &FxHashMap<&'a str, i64>, b: &FxHashMap<&'a str, i64>) -> BTreeMap<&'a str, i64> {
540  a.iter()
541    .filter_map(|(t, n)| {
542      let d = n - b.get(t).copied().unwrap_or(0);
543      (d > 0).then_some((*t, d))
544    })
545    .collect()
546}

548fn show_tokens(m: &BTreeMap<&str, i64>, limit: usize) -> String {
549  let mut parts: Vec<String> = m
550    .iter()
551    .take(limit)
552    .map(|(t, n)| {

566/// Text by class: `lost-text:<class>` for tokens gone from the class and from the whole document,
567/// `moved-text:<class>` for tokens that left the class (or a class reordered or resegmented),
568/// `gained-text:<class>` for new tokens.
569fn diff_text(
570  a: &XmlFacts,
571  b: &XmlFacts,
572  flags: &mut Vec<String>,
573  cell: &mut Vec<String>,
574  details: &mut String,
575) {
576  let is_attr = |c: &&String| c.contains('@');
577  // The document-wide multisets, by piece (`pieces`).
578  fn all(f: &XmlFacts, attr: bool) -> FxHashMap<&str, i64> {
579    piece_counts(
580      f.text
581        .iter()
582        .filter(move |(c, _)| c.contains('@') == attr)
583        .flat_map(|(_, s)| s.iter()),
584    )
585  }
586  let (ta, tb, aa, ab) = (all(a, false), all(b, false), all(a, true), all(b, true));
587  let delta = |x: &FxHashMap<&str, i64>, y: &FxHashMap<&str, i64>| -> FxHashMap<String, i64> {
588    minus(x, y)
589      .into_iter()
590      .map(|(t, n)| (t.to_string(), n))
591      .collect()
592  };
593  let (lost_text, gained_text) = (delta(&ta, &tb), delta(&tb, &ta));
594  let (lost_attr, gained_attr) = (delta(&aa, &ab), delta(&ab, &aa));
595  let mut classes: Vec<&String> = a.text.keys().chain(b.text.keys()).collect();
596  classes.sort();
597  classes.dedup();
598  let empty = Vec::new();
599  for class in classes {
600    let (sa, sb) = (
601      a.text.get(class).unwrap_or(&empty),
602      b.text.get(class).unwrap_or(&empty),
603    );
604    if sa == sb {
605      continue;
606    }
607    let spaces = space_lost(sa, sb);
608    if !spaces.is_empty() {
609      flags.push(format!("space-lost:{class}"));
610      details.push_str(&format!(
611        "space lost {class}: around the element between {}\n",
612        spaces.join(", ")
613      ));
614      cell.push(format!(
615        "{class}: space lost {}",
616        spaces.iter().take(3).cloned().collect::<Vec<_>>().join(" ")
617      ));
618    }
619    if words_only(sa) == words_only(sb) {
620      // Only element positions inside the text moved (a footnote mark, an index entry): the element
621      // counts and the spacing say the rest.
622      continue;
623    }
624    let (global_lost, global_gained) = if is_attr(&class) {
625      (&lost_attr, &gained_attr)
626    } else {
627      (&lost_text, &gained_text)
628    };
629    let (ca, cb) = (tokens(sa.iter()), tokens(sb.iter()));
630    let (mut lost, mut gained) = (minus(&ca, &cb), minus(&cb, &ca));
631    let jitter = take_jitter(&mut lost, &mut gained);
632    if jitter > 0 {
633      flags.push(format!("text-jitter:{class}"));
634    }
635    // A token is lost (or gained) for good when one of its pieces is lost (gained) from the whole
636    // document: a word, not just a split of `foo,` into `foo` + `,`.
637    let truly = |tokens: &BTreeMap<&str, i64>, global: &FxHashMap<String, i64>| -> Vec<String> {
638      tokens
639        .keys()
640        .filter(|t| {
641          pieces(t)
642            .iter()
643            .any(|p| global.get(*p).is_some_and(|n| *n > 0))
644        })
645        .map(|t| t.to_string())
646        .collect()
647    };
648    let (truly_lost, truly_gained) = (truly(&lost, global_lost), truly(&gained, global_gained));
649    if !truly_lost.is_empty() {
650      flags.push(format!("lost-text:{class}"));
651    }
652    if !truly_gained.is_empty() {
653      flags.push(format!("gained-text:{class}"));
654      // A number typeset from nowhere in prose: a TeX dimension (a register printed by mistake) is a
655      // regression, a bare number needs a look. Not when the class lost a number too (a value changed),
656      // not in math, attribute, generated-number or code classes, and not amid recovered prose: when the
657      // numbers are under half of what the class gained (a document that recovered a section), they are
658      // reviewed, not called a leak.
659      let prose = !is_attr(&class)
660        && !class.starts_with("XM")
661        && !NOT_PROSE_CLASSES.contains(&class.as_str());
662      // A number moved elsewhere (a `3` now inside a `ref`) is not a lost number: `truly_lost` only.
663      let lost_number = truly_lost.iter().any(|t| token_number_kind(t).is_some());
664      if prose {
665        let kinds: Vec<bool> = truly_gained
666          .iter()
667          .filter_map(|t| token_number_kind(t))
668          .collect();
669        let mostly_numbers = 2 * kinds.len() >= truly_gained.len();
670        if !lost_number && mostly_numbers && kinds.iter().any(|k| *k) {
671          flags.push(format!("leaked-dimension:{class}"));
672        } else if !kinds.is_empty() {
673          flags.push(format!("gained-number:{class}"));
674        }
675      }
676    }
677    if truly_lost.len() < lost.len()
678      || truly_gained.len() < gained.len()
679      || (lost.is_empty() && gained.is_empty() && (jitter == 0 || sa.len() != sb.len()))
680    {
681      flags.push(format!("moved-text:{class}"));
682    }
683    let mut summary = format!("{class}:");
684    if !lost.is_empty() {
685      summary.push_str(&format!(" -[{}]", show_tokens(&lost, 6)));
686    }
687    if !gained.is_empty() {
688      summary.push_str(&format!(" +[{}]", show_tokens(&gained, 6)));
689    }
690    if lost.is_empty() && gained.is_empty() {
691      summary.push_str(&format!(" segments {}>{}", sa.len(), sb.len()));
692    }
693    details.push_str(&format!(
694      "text {class}: lost [{}] gained [{}] segments {}>{}\n",
695      show_tokens(&lost, 200),
696      show_tokens(&gained, 200),
697      sa.len(),
698      sb.len()
699    ));
700    cell.push(summary);
701  }
702}

704/// Element counts: every qname whose count moved, losses first.
705fn diff_elements(a: &XmlFacts, b: &XmlFacts, flags: &mut Vec<String>) -> String {
706  let mut names: Vec<&String> = a.elements.keys().chain(b.elements.keys()).collect();
707  names.sort();
708  names.dedup();
709  let mut deltas: Vec<(&String, i64)> = names
710    .into_iter()
711    .map(|q| {
712      let n = |f: &XmlFacts| i64::from(f.elements.get(q).copied().unwrap_or(0));
713      (q, n(b) - n(a))
714    })
715    .filter(|(_, d)| *d != 0)
716    .collect();
717  deltas.sort_by_key(|(q, d)| (*d, (*q).clone()));
718  for (q, d) in &deltas {
719    let math = q.starts_with("XM") || q.starts_with("m:");
720    let wrapper = matches!(q.as_str(), "text" | "emph");
721    let flag = match (*d < 0, q.as_str()) {
722      (true, "ERROR") => None,
723      (false, "ERROR") => Some("gained-elements:ERROR".to_string()),
724      _ if math => Some("math-changed".to_string()),
725      // A lost font or emphasis run is lost markup (a bold word no longer bold): review. A gained one is
726      // a change.
727      _ if wrapper && *d < 0 => Some("wrappers-lost".to_string()),
728      _ if wrapper => Some("wrappers-changed".to_string()),
729      (true, _) => Some(format!("lost-elements:{q}")),
730      (false, _) => Some("gained-elements".to_string()),
731    };
732    flags.extend(flag);
733  }
734  deltas
735    .iter()
736    .map(|(q, d)| format!("{q}:{d:+}"))
737    .collect::<Vec<_>>()
738    .join(",")
739}

741/// Attribute values: the multiset of `qname@attr=value` over every non-id attribute, so a lost class,
742/// font or `name="E-mail: "` shows even when no count and no text moved. One `attrs-changed:<qname@attr>`
743/// per attribute whose values changed.
744fn diff_attrs(a: &XmlFacts, b: &XmlFacts, flags: &mut Vec<String>, details: &mut String) -> String {
745  let mut keys: Vec<&String> = a.attrs.keys().chain(b.attrs.keys()).collect();
746  keys.sort();
747  keys.dedup();
748  let mut by_attr: BTreeMap<&str, (Vec<String>, Vec<String>)> = BTreeMap::new();
749  for key in keys {
750    let n = |f: &XmlFacts| i64::from(f.attrs.get(key).copied().unwrap_or(0));
751    let d = n(b) - n(a);
752    if d == 0 {
753      continue;
754    }
755    let (attr, value) = key.split_once('=').unwrap_or((key.as_str(), ""));
756    let entry = by_attr.entry(attr).or_default();
757    let shown = if d.abs() > 1 {
758      format!("{value:?}×{}", d.abs())
759    } else {
760      format!("{value:?}")
761    };
762    if d < 0 {
763      entry.0.push(shown)
764    } else {
765      entry.1.push(shown)
766    }
767  }
768  let mut cells = Vec::new();
769  for (attr, (lost, gained)) in &by_attr {
770    flags.push(format!("attrs-changed:{attr}"));
771    details.push_str(&format!(
772      "attr {attr}: lost [{}] gained [{}]\n",
773      lost.join(" "),
774      gained.join(" ")
775    ));
776    let mut cell = format!("{attr}:");
777    if !lost.is_empty() {
778      cell.push_str(&format!(
779        " -[{}]",
780        lost.iter().take(4).cloned().collect::<Vec<_>>().join(" ")
781      ));
782    }
783    if !gained.is_empty() {
784      cell.push_str(&format!(
785        " +[{}]",
786        gained.iter().take(4).cloned().collect::<Vec<_>>().join(" ")
787      ));
788    }
789    cells.push(cell);
790  }
791  cells.join(" | ")
792}

794/// Whether two numbers are equal within the [`tolerance`].
795fn near(p: f64, q: f64) -> bool { (p - q).abs() <= tolerance(p, q) }
796
797/// Whether a picture or SVG canvas size, or a path, collapsed between `x` (A) and `y` (B): a size
798/// (`picture`/`svg:svg` width or height, the viewBox extent) shrinking by more than half or to within the
799/// tolerance of zero, or a path whose coordinates all became one point (`M 0 0 L 0 0`, or none).
800fn collapsed(key: &str, x: &str, y: &str) -> bool {
801  let ((nx, _), (ny, _)) = (numbers(x), numbers(y));
802  let size = |n: &[f64]| -> Vec<f64> {
803    if key.ends_with("@viewBox") {
804      n.iter().skip(2).take(2).copied().collect()
805    } else {
806      n.first().copied().into_iter().collect()
807    }
808  };
809  if key == "svg:path@d" {
810    // Coordinates are (x, y) pairs: `M 7 3 L 7 3` is one point. An odd trailing number is ignored; no
811    // pair at all (an empty path) is no extent.
812    let one_point = |n: &[f64]| {
813      let (pairs, _) = n.as_chunks::<2>();
814      pairs.first().is_none_or(|f| pairs.iter().all(|p| p == f))
815    };
816    return !one_point(&nx) && one_point(&ny);
817  }
818  if matches!(
819    key,
820    "picture@width" | "picture@height" | "svg:svg@width" | "svg:svg@height" | "svg:svg@viewBox"
821  ) {
822    let (sx, sy) = (size(&nx), size(&ny));
823    return sx
824      .iter()
825      .zip(&sy)
826      .any(|(p, q)| *p > 0.0 && (*q < 0.5 * p || q.abs() <= tolerance(*p, *q)));
827  }
828  false
829}

831/// Geometry (`svg:*` coordinates, a picture's size), in document order: numbers within the [`tolerance`]
832/// are equal; a collapse ([`collapsed`]) is `geometry-collapsed` (a regression: the drawing is gone or a
833/// dot); anything else is a `geometry-changed` change.
834fn diff_geometry(a: &XmlFacts, b: &XmlFacts, flags: &mut Vec<String>, details: &mut String) {
835  let mut keys: Vec<&String> = a.geometry.keys().chain(b.geometry.keys()).collect();
836  keys.sort();
837  keys.dedup();
838  let empty = Vec::new();
839  let (mut moved, mut gone) = (Vec::new(), Vec::new());
840  for key in keys {
841    let (va, vb) = (
842      a.geometry.get(key).unwrap_or(&empty),
843      b.geometry.get(key).unwrap_or(&empty),
844    );
845    if va.iter().zip(vb).any(|(x, y)| collapsed(key, x, y)) {
846      gone.push(key.as_str());
847    }
848    let same = va.len() == vb.len()
849      && va.iter().zip(vb).all(|(x, y)| {
850        let ((nx, sx), (ny, sy)) = (numbers(x), numbers(y));
851        sx == sy && nx.len() == ny.len() && nx.iter().zip(&ny).all(|(p, q)| near(*p, *q))
852      });
853    if !same {
854      moved.push(key.as_str());
855    }
856  }
857  if !gone.is_empty() {
858    flags.push("geometry-collapsed".into());
859    details.push_str(&format!("geometry collapsed: {}\n", gone.join(" ")));
860  }
861  if !moved.is_empty() {
862    flags.push("geometry-changed".into());
863    details.push_str(&format!(
864      "geometry changed beyond the tolerance: {}\n",
865      moved.join(" ")
866    ));
867  }
868}

870/// Diagnostic class deltas, Fatal then Error then Warning, larger moves first.
871fn diag_delta(a: &LogFacts, b: &LogFacts) -> Vec<String> {
872  let mut names: Vec<&String> = a.classes.keys().chain(b.classes.keys()).collect();
873  names.sort();
874  names.dedup();
875  let mut deltas: Vec<(usize, i64, &String)> = names
876    .into_iter()
877    .filter_map(|c| {
878      let n = |f: &LogFacts| i64::from(f.classes.get(c).copied().unwrap_or(0));
879      let d = n(b) - n(a);
880      let level = ["Fatal:", "Error:", "Warning:"]
881        .iter()
882        .position(|l| c.starts_with(l))
883        .unwrap_or(3);
884      (d != 0).then_some((level, d, c))
885    })
886    .collect();
887  deltas.sort_by_key(|(level, d, c)| (*level, -*d, (*c).clone()));
888  deltas.iter().map(|(_, d, c)| format!("{c}{d:+}")).collect()
889}

891/// Compare one paper's two conversions.
892pub fn compare(key: &str, a_in: &DocInput, b_in: &DocInput) -> Row {
893  let same_bytes = a_in.xml.is_some() && a_in.xml == b_in.xml;
894  let want_canon = !same_bytes && a_in.xml.is_some() && b_in.xml.is_some();
895  let a = read_side(a_in, want_canon, None);
896  // Byte-identical XML: its facts once, for both sides.
897  let b = read_side(b_in, want_canon, same_bytes.then_some(&a.xml));
898  let mut row = Row {
899    key: key.to_string(),
900    ..Row::default()
901  };
902  let mut flags = Vec::new();
903  side_problems(&a, "A", &mut flags);
904  side_problems(&b, "B", &mut flags);
905  // A problem on both sides is one `<problem>:both` flag.
906  let twins: Vec<String> = flags
907    .iter()
908    .filter_map(|f| f.strip_suffix(":A"))
909    .filter(|p| flags.iter().any(|g| g.strip_suffix(":B") == Some(*p)))
910    .map(str::to_string)
911    .collect();
912  for p in twins {
913    flags.retain(|f| f != &format!("{p}:A") && f != &format!("{p}:B"));
914    flags.push(format!("{p}:both"));
915  }
916  // Mid-line Fatals and Errors compare by count, like `errors+`: more in B is a regression, fewer a fix.
917  for kind in ["fatal", "error"] {
918    let count = |s: &Side| {
919      s.log.as_ref().map_or(0, |l| {
920        if kind == "fatal" {
921          l.midline_fatals
922        } else {
923          l.midline_errors
924        }
925      })
926    };
927    let (x, y) = (count(&a), count(&b));
928    let prefix = format!("{kind}-midline:");
929    flags.retain(|f| !f.starts_with(&prefix));
930    let flag = match y.cmp(&x) {
931      std::cmp::Ordering::Greater if x == 0 => Some(format!("{kind}-midline:B")),
932      std::cmp::Ordering::Greater => Some(format!("{kind}-midline+")),
933      std::cmp::Ordering::Less if y == 0 => Some(format!("{kind}-midline-fixed")),
934      std::cmp::Ordering::Less => Some(format!("{kind}-midline-")),
935      std::cmp::Ordering::Equal if x > 0 => Some(format!("{kind}-midline:both")),
936      std::cmp::Ordering::Equal => None,
937    };
938    flags.extend(flag);
939  }
940  // A timeout or out-of-memory of the base alone is a fix; the base without XML while B has one is a gain
941  // to review.
942  for problem in ["timeout", "oom"] {
943    if !flags.iter().any(|f| *f == format!("{problem}:B"))
944      && let Some(i) = flags.iter().position(|f| *f == format!("{problem}:A"))
945    {
946      flags[i] = format!("{problem}-fixed");
947    }
948  }
949  if b.xml.is_ok()
950    && let Some(i) = flags.iter().position(|f| f == "xml-missing:A")
951  {
952    flags[i] = "xml-gained".into();
953  }
954  let mut details = String::new();
955  // Diagnostics.
956  let (la, lb) = (
957    a.log.clone().unwrap_or_default(),
958    b.log.clone().unwrap_or_default(),
959  );
960  let diags = if let (Some(x), Some(y)) = (&a.log, &b.log) {
961    let worse = |p: u32, q: u32, up: &str, down: &str, flags: &mut Vec<String>| {
962      if q > p {
963        flags.push(up.into());
964      } else if q < p {
965        flags.push(down.into());
966      }
967    };
968    worse(x.fatals, y.fatals, "fatal+", "fatal-", &mut flags);
969    worse(x.errors, y.errors, "errors+", "errors-", &mut flags);
970    worse(x.warnings, y.warnings, "warnings+", "warnings-", &mut flags);
971    if let (Some(s), Some(t)) = (x.status, y.status) {
972      worse(
973        u32::from(s),
974        u32::from(t),
975        "status-worse",
976        "status-better",
977        &mut flags,
978      );
979    }
980    let delta = diag_delta(x, y);
981    if !delta.is_empty() && x.errors == y.errors && x.fatals == y.fatals && x.warnings == y.warnings
982    {
983      flags.push("diag-classes-changed".into());
984    }
985    if !delta.is_empty() {
986      details.push_str(&format!("diagnostics: {}\n", delta.join(" ")));
987    }
988    delta
989  } else {
990    Vec::new()
991  };
992  // Content.
993  let (mut elem_cell, mut attr_cell, mut text_cells, mut front_cell, mut xml_cell) = (
994    String::new(),
995    String::new(),
996    Vec::new(),
997    String::new(),
998    String::from("-"),
999  );
1000  if let (Ok(x), Ok(y)) = (&a.xml, &b.xml) {
1001    if same_bytes {
1002      xml_cell = "same".into();
1003    } else {
1004      let before_content = flags.len();
1005      elem_cell = diff_elements(x, y, &mut flags);
1006      attr_cell = diff_attrs(x, y, &mut flags, &mut details);
1007      diff_geometry(x, y, &mut flags, &mut details);
1008      diff_text(x, y, &mut flags, &mut text_cells, &mut details);
1009      let fd = diff_front(&x.front, &y.front);
1010      flags.extend(fd.flags.iter().cloned());
1011      front_cell = fd
1012        .notes
1013        .iter()
1014        .take(4)
1015        .cloned()
1016        .collect::<Vec<_>>()
1017        .join("; ");
1018      for note in &fd.notes {
1019        details.push_str(&format!("frontmatter: {note}\n"));
1020      }
1021      if let (Some(ca), Some(cb)) = (&a.canon, &b.canon) {
1022        let d = canon_diff(ca, cb, 20);
1023        xml_cell = if d.deleted + d.inserted == 0 && d.ids_renamed == 0 {
1024          "equivalent".to_string()
1025        } else if d.deleted + d.inserted == 0 {
1026          format!("ids-only({})", d.ids_renamed)
1027        } else {
1028          format!(
1029            "-{}/+{}{}",
1030            d.deleted,
1031            d.inserted,
1032            if d.approximate { " approx" } else { "" }
1033          )
1034        };
1035        if d.refs_retargeted > 0 {
1036          flags.push("ref-retargeted".into());
1037          xml_cell.push_str(&format!(" refs-retargeted:{}", d.refs_retargeted));
1038        }
1039        if d.deleted + d.inserted == 0 {
1040          if flags.len() == before_content {
1041            // Canonically equal: ids renumbered, or only what the canonical form normalises (geometry
1042            // within the tolerance, host paths).
1043            flags.push(
1044              if d.ids_renamed > 0 {
1045                "ids-only"
1046              } else {
1047                "equivalent"
1048              }
1049              .into(),
1050            );
1051          }
1052        } else if flags[before_content..]
1053          .iter()
1054          .all(|f| rank(f).0 == Tier::Change)
1055        {
1056          // The canonical XML moved and nothing that needs a reason explains it (no content flag, or only
1057          // change-tier ones such as `math-changed`): review, never a silent change.
1058          flags.push("xml-only".into());
1059        }
1060        for hunk in &d.hunks {
1061          details.push_str(hunk);
1062        }
1063      }
1064    }
1065  }
1066  // Wall time.
1067  let secs = |m: &Option<RunMeta>| m.as_ref().and_then(|m| m.secs);
1068  if let (Some(p), Some(q)) = (secs(&a.meta), secs(&b.meta))
1069    && q > 1.5 * p
1070    && q - p > 5.0
1071  {
1072    flags.push("slower".into());
1073  }
1074  row.flags = flags;
1075  row.finish();
1076  let fmt_secs = |m: &Option<RunMeta>| secs(m).map(|s| format!("{s:.1}"));
1077  row.cells = vec![
1078    pair(
1079      la.status.filter(|_| a.log.is_some()),
1080      lb.status.filter(|_| b.log.is_some()),
1081    ),
1082    pair(
1083      a.log.as_ref().map(|l| l.errors),
1084      b.log.as_ref().map(|l| l.errors),
1085    ),
1086    pair(
1087      a.log.as_ref().map(|l| l.fatals),
1088      b.log.as_ref().map(|l| l.fatals),
1089    ),
1090    pair(
1091      a.log.as_ref().map(|l| l.warnings),
1092      b.log.as_ref().map(|l| l.warnings),
1093    ),
1094    clip(&diags.join(" "), 400),
1095    clip(&elem_cell, 400),
1096    clip(&attr_cell, 400),
1097    clip(&text_cells.join(" | "), 600),
1098    clip(&front_cell, 600),
1099    xml_cell,
1100    pair(fmt_secs(&a.meta), fmt_secs(&b.meta)),
1101    pair(
1102      a.log.as_ref().map(status_diag_cell),
1103      b.log.as_ref().map(status_diag_cell),
1104    ),
1105  ];
1106  for (tag, side) in [("A", &a), ("B", &b)] {
1107    if let Err(e) = &side.xml
1108      && side.present
1109    {
1110      details.push_str(&format!("xml {tag}: {e}\n"));
1111    }
1112    for line in &side.stderr_fatals {
1113      details.push_str(&format!("stderr {tag}: {line}\n"));
1114    }
1115  }
1116  row.details = details;
1117  row
1118}

1136/// Audit one conversion: absolute flags (a missing or unparsable piece, a Fatal, errors, a nameless
1137/// creator, an empty, split or fragment-linked contact) and its counts.
1138pub fn audit(key: &str, input: &DocInput) -> Row {
1139  let side = read_side(input, false, None);
1140  let mut row = Row {
1141    key: key.to_string(),
1142    ..Row::default()
1143  };
1144  let mut flags = Vec::new();
1145  side_problems(&side, "", &mut flags);
1146  for f in flags.iter_mut() {
1147    if let Some(stripped) = f.strip_suffix(':') {
1148      *f = stripped.to_string();
1149    }
1150  }
1151  let log = side.log.clone().unwrap_or_default();
1152  // The status alone raises the severity too: a status 3 without a `Fatal:` line is still a Fatal run
1153  // (with `status-diag-mismatch` as the reason).
1154  if log.fatals > 0 || log.status == Some(3) {
1155    flags.push("fatal".into());
1156  }
1157  if log.errors > 0 || log.status == Some(2) {
1158    flags.push("errors".into());
1159  }
1160  let mut top: Vec<(&String, &u32)> = log
1161    .classes
1162    .iter()
1163    .filter(|(c, _)| !c.starts_with("Warning:"))
1164    .collect();
1165  top.sort_by_key(|(c, n)| (std::cmp::Reverse(**n), (*c).clone()));
1166  let top = top
1167    .iter()
1168    .take(5)
1169    .map(|(c, n)| format!("{c}×{n}"))
1170    .collect::<Vec<_>>()
1171    .join(" ");
1172  let (mut counts, mut front_cell, mut words) = (String::new(), String::new(), String::from("-"));
1173  if let Ok(x) = &side.xml {
1174    let f = &x.front;
1175    for (n, flag) in [
1176      (f.nameless(), "creator-nameless"),
1177      (f.empty_contacts(), "contact-empty"),
1178      (f.split_contacts(), "contact-split"),
1179      (f.fragment_links(), "contact-link-fragment"),
1180    ] {
1181      if n > 0 {
1182        flags.push(flag.into());
1183      }
1184    }
1185    if x.elements.get("ERROR").is_some_and(|n| *n > 0) {
1186      flags.push("error-elements".into());
1187    }
1188    counts = AUDIT_ELEMENTS
1189      .iter()
1190      .map(|q| format!("{q}:{}", x.elements.get(*q).copied().unwrap_or(0)))
1191      .collect::<Vec<_>>()
1192      .join(",");
1193    front_cell = format!(
1194      "creators:{} nameless:{} empty-contacts:{} split:{} fragment-links:{} title:{} abstract:{}",
1195      f.creators.len(),
1196      f.nameless(),
1197      f.empty_contacts(),
1198      f.split_contacts(),
1199      f.fragment_links(),
1200      !f.title.is_empty(),
1201      !f.abstract_text.is_empty()
1202    );
1203    words = x.words.to_string();
1204  } else if let Err(e) = &side.xml {
1205    row.details = format!("xml: {e}\n");
1206  }
1207  for line in &side.stderr_fatals {
1208    row.details.push_str(&format!("stderr: {line}\n"));
1209  }
1210  row.flags = flags;
1211  row.finish();
1212  row.cells = vec![
1213    opt(side.log.as_ref().and_then(|l| l.status)),
1214    opt(side.log.as_ref().map(|l| l.errors)),
1215    opt(side.log.as_ref().map(|l| l.fatals)),
1216    opt(side.log.as_ref().map(|l| l.warnings)),
1217    clip(&top, 400),
1218    counts,
1219    front_cell,
1220    words,
1221    opt(
1222      side
1223        .meta
1224        .as_ref()
1225        .and_then(|m| m.secs)
1226        .map(|s| format!("{s:.0}")),
1227    ),
1228    opt(side.log.as_ref().map(status_diag_cell)),
1229  ];
1230  row
1231}

1253/// One side's status/diagnostics agreement, `ok` or `mismatch`.
1254fn status_diag_cell(log: &LogFacts) -> &'static str {
1255  if log.status_mismatch() {
1256    "mismatch"
1257  } else {
1258    "ok"
1259  }
1260}

1262/// Cut a cell to `n` characters, tabs and newlines made spaces.
1263pub fn clip(s: &str, n: usize) -> String {
1264  let s = s.replace(['\t', '\n', '\r'], " ");
1265  if s.chars().count() <= n {
1266    s
1267  } else {
1268    format!("{}…", s.chars().take(n).collect::<String>())
1269  }
1270}
