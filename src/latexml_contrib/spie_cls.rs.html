5/// Does the class code `text` define `\author` itself (`\def\author`, `\renewcommand{\author}`, …; not `\authorinfo`)?
6fn defines_author(text: &str) -> bool {
7  let code = crate::shipped_class::class_code(text);
8  [
9    "\\def\\author",
10    "\\gdef\\author",
11    "\\let\\author",
12    "\\renewcommand\\author",
13    "\\renewcommand{\\author}",
14    "\\renewcommand*{\\author}",
15    "\\newcommand{\\author}",
16    "\\DeclareRobustCommand{\\author}",
17  ]
18  .iter()
19  .any(|form| {
20    code.match_indices(form).any(|(at, _)| {
21      code[at + form.len()..]
22        .chars()
23        .next()
24        .is_none_or(|c| !c.is_ascii_alphabetic() && c != '@')
25    })
26  })
27}

29LoadDefinitions!({
30  // spie.cls defines no `\author` — it only `\LoadClassWithOptions{article}` (:74) —, so its documented author block
31  // (:58-64: names with `\supit` marks, `\skiplinehalf`, then `\\`-separated affiliation lines) is the kernel's `\author`,
32  // whose splitter reads the names and the marked affiliation lines; OmniBus's `\author` (inst_support) splits only at
33  // `\and`, and read every affiliation line as more names (2210.16202's CHRONOS_s.cls, `\ProvidesClass{spie}`: 39
34  // creators, no contact, where the PDF prints 5 authors over 10 affiliation lines). A class that defines its own
35  // `\author` (a spieman.cls copy's `\author[a]`/`\affil[a]`) keeps OmniBus's. Guard
36  // `perfect_kernel_batch77w::spie_author_block_is_the_kernel_author`.
37  //
38  // A copy that loads authblk — SPIE's 2015 v3.4 spie.cls, `\RequirePackage{authblk}` (its :31; the TeX Live v3.25
39  // copy loads none) — writes its author block with authblk's `\author[a]{…}`/`\affil[a]{…}`: it loads authblk as the
40  // copy does, whose `\author` reads the marks (the kernel's one-argument `\author` took "[" as the name and left
41  // every name in the body, 2406.08304: one creator "[", 30 empty affiliations). Guard
42  // `perfect_kernel_batch78m::spie_copy_loading_authblk_reads_authblk_authors`.
43  let copy = crate::shipped_class::shipped_class_text("spie")?;
44  let loads_authblk = copy
45    .as_deref()
46    .is_some_and(|text| crate::shipped_class::class_text_loads(text, "authblk"));
47  let kernel_author = copy.is_some_and(|text| !defines_author(&text)) && !loads_authblk;
48  if kernel_author {
49    Let!("\\lx@spie@kernel@author", "\\author");
50  }
51  LoadClass!("OmniBus");
52  if loads_authblk {
53    RequirePackage!("authblk");
54  }
55  if kernel_author {
56    // (the kernel's splitter reads the argument as written, where TeX expands the class's author-block macros when
57    // `\maketitle` prints it: spie.cls:123-124 `\skiplinehalf` (`\medskip\\`) and `\skipline` (`\\`) end a line,
58    // :128 `\supit` raises the mark "to cross reference author and affiliations" — the splitter's line break and
59    // superscript mark)
60    DefMacro!("\\author{}", sub[(names)] {
61      let mut out = vec![T_CS!("\\lx@spie@kernel@author"), T_BEGIN!()];
62      out.extend(names.unlist_ref().iter().map(|t| {
63        if *t == T_CS!("\\skiplinehalf") || *t == T_CS!("\\skipline") {
64          T_CS!("\\\\")
65        } else if *t == T_CS!("\\supit") {
66          T_CS!("\\textsuperscript")
67        } else {
68          *t
69        }
70      }));
71      out.push(T_END!());
72      Ok(Tokens::new(out))
73    });
74  }
75  // spie.cls (TeX Live's) and spieman.cls (39 copies in a 1/20 arXiv sample) load no amsmath: amsmath's commands for
76  // papers that use them unloaded, its "Old form" error only once the paper loads it (1003.3524's `\pmatrix{…}`, no
77  // error in pdflatex; `amsmath_sty::require_amsmath_beyond_original`, KNOWN_PERL_ERRORS #993).
78  amsmath_sty::require_amsmath_beyond_original(
79    latexml_package::shipped_class::shipped_copy_loads("cls", "amsmath")?,
80    &[],
81  )?;
82  RequirePackage!("amsthm");
83  // Pre-load xcolor with [dvipsnames, table] so a paper's later
84  // `\usepackage[table]{xcolor}` doesn't silently option-clash and
85  // leave colortbl unloaded → `\cellcolor` undefined. spie.cls itself
86  // does NOT load xcolor (only `\LoadClassWithOptions{article}`), so
87  // in Perl the user's `[table]{xcolor}` is the first load and colortbl
88  // comes in via the `table` option; matching that outcome here. Same
89  // anti-clash pattern as svproc_cls / mnras_cls / quantumarticle_cls.
90  // Witness: 1807.04749 ("undefined:\\cellcolor").
91  RequirePackage!("xcolor", options => vec!["dvipsnames".to_string(), "table".to_string()]);
92  RequirePackage!("hyperref");
93  // spie.cls L92: `\RequirePackage[superscript]{cite}[2003/11/04]` — SPIE uses
94  // the cite package for superscript numeric citations. cite.sty defines the
95  // no-brackets `\citen`/`\citenum`/`\citeonline` aliases SPIE templates use
96  // directly. Our OmniBus base autoloads natbib's `\citenum` but NOT cite's
97  // `\citen`, so without this `\citen` is undefined where Perl (raw-loads
98  // spie.cls → cite) is clean. Faithful (spie.cls genuinely loads cite — not an
99  // eager preload). Witness 1808.10428 (`{spie}` + `\citen`).
100  RequirePackage!("cite", options => vec!["superscript".to_string()]);
101
102  // spie.cls L107: \authorinfo{...} for author footnote — preserve.
103  DefMacro!(
104    "\\authorinfo{}",
105    "\\@add@frontmatter{ltx:note}[role=authorinfo]{#1}"
106  );
107  // spie.cls:123 `\def\skiplinehalf{\medskip\\}`: in the kernel's author block it ends the names line (the
108  // `\medskip` is the page's spacing); under OmniBus's `\author` it prints nothing
109  if kernel_author {
110    DefMacro!("\\skiplinehalf", "\\\\");
111  } else {
112    def_macro_noop("\\skiplinehalf")?;
113  }
114  DefMacro!("\\supit{}", "\\textsuperscript{#1}");
115  // spieman.cls, SPIE's journal manuscript class, which the loader gives this binding ("Can't find binding for class
116  // spieman (using spie)"; 87 of 99 sampled copies are its v1.10, 1703.03292's): `\linkable` (:82), an address or URL
117  // set blue and underlined but not linked (run 340: 640 first errors; 2604.17554, 2604.05004); the 2006 v1.00 copies'
118  // `\supscr`/`\supscrsm`, raised small text (1408.0407). spie.cls defines neither, so they are guesses a paper's own
119  // definition replaces (`DefinitionOrigin::Fallback`).
120  {
121    let _guess = OriginGuard::new(DefinitionOrigin::Fallback);
122    RawTeX!(
123      r"\newcommand{\linkable}[1]{\textcolor{blue}{\underline{#1}}}
124\def\supscr#1{\raisebox{0.8ex}{\small #1}\hspace{0.05em}}
125\def\supscrsm#1{\raisebox{0.8ex}{\footnotesize #1}\hspace{0.05em}}"
126    );
127  }
