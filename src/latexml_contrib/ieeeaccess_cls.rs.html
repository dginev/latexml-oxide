1//! Stub for IEEE Access journal class (ieeeaccess.cls).
2//!
3//! ieeeaccess.cls is a derivative of IEEEtran. Route to the IEEEtran
4//! binding and stub the IEEE Access-specific frontmatter macros.
5use latexml_package::prelude::*;
6
7use crate::shipped_class::{class_option_given, shipped_class_loads_or};
8
9LoadDefinitions!({
10  LoadClass!("IEEEtran");
11  // ieeeaccess.cls:453 (2106.14273's copy) `\def\authorrefmark#1{\textsuperscript{#1}}`: the class's own reference mark
12  // prints the number (IEEEtran's `\IEEEauthorrefmark` the footnote symbol; OXIDIZED_DESIGN_DIVERGENCES #968).
13  DefMacro!("\\authorrefmark{}", "\\textsuperscript{#1}");
14  // ieeeaccess.cls:212-245 (1910.07402's copy) `\@makecaption`: a table's caption "TABLE 1. text" on one line, as a
15  // figure's "FIGURE 1. text", in place of IEEEtran's label over the text (ieeetran_cls.rs); caption, when loaded,
16  // replaces it (caption_sty.rs). The names and numbers (:244-262): figures "FIGURE", tables arabic, Roman under the
17  // `jtehm` option, in place of IEEEtran's "Fig." and Roman tables. The label's `\figcapheadfont` (bold sans serif in
18  // the class's blue) is not set (caption_sty.rs `CLASS_CAPTION_FONTS`). Guards
19  // `perfect_kernel_batch73a::ieeeaccess_figure_is_a_figure`, `perfect_kernel_batch78y::ieeeaccess_caption_names`.
20  caption_sty::class_makecaption_label("", ". ", false)?;
21  if class_option_given("jtehm")? {
22    RawTeX!(r"\def\thetable{\@Roman\c@table}\def\figurename{FIGURE}");
23  } else {
24    RawTeX!(r"\def\thetable{\@arabic\c@table}\def\figurename{FIGURE}");
25  }
26  RequirePackage!("ifpdf");
27  // ieeeaccess.cls:121-131 (1901.05553's copy; 241, 251 in others): `\RequirePackage{color}`, then, unless the
28  // `jtehm` option (:36), `\RequirePackage{spotcolor}` for the PANTONE blue, whose `\NewSpotColorSpace`/`\AddSpotColor`/
29  // `\SetPageColorSpace` lines the template's papers copy into their preamble (72 papers of run 353 with them
30  // undefined, 48 with `\textcolor`/`\definecolor`: 1901.05553, 1907.00310, 1712.00216, 1902.03817). spotcolor has no
31  // binding and loads raw; a copy that dropped it (1 of 60 sampled) loads none.
32  RequirePackage!("color");
33  let jtehm = class_option_given("jtehm")?;
34  if !jtehm && shipped_class_loads_or("ieeeaccess", "spotcolor", true)? {
35    RequirePackage!("spotcolor");
36  }
37  // The class's colours (ieeeaccess.cls:121-133): `accessblue` is CMYK .24,1,1,0 under `jtehm`, else the PANTONE 3015 C
38  // spot colour `\definecolor{accessblue}{spotcolor}{PANTONE3015C,1.0}`, whose alternate is CMYK 1 0.3 0 0.2 (the
39  // `\AddSpotColor` line). color has no `spotcolor` model here ("Unrecognized color model"), so the alternate stands in:
40  // the blue a viewer without the spot ink shows. Papers name it in their caption setup
41  // (`\DeclareCaptionFont{ieeeblue}{\color{accessblue}}` + `labelfont={bf,ieeeblue}`: 2405.14505, 2205.15416 — an Error
42  // per caption once caption's keys took effect, 68h; pdflatex 0). The class's `black` (CMYK 0,0,0,1) is left out: it
43  // would turn every black run into a CMYK colour change.
44  if jtehm {
45    RawTeX!(r"\definecolor{accessblue}{cmyk}{.24,1,1,0}\definecolor{greycolor}{cmyk}{0,0,0,1}");
46  } else {
47    RawTeX!(r"\definecolor{accessblue}{cmyk}{1,0.3,0,0.2}\definecolor{greycolor}{cmyk}{0,0,0,.8}");
48  }
49  RawTeX!(r"\definecolor{grey}{cmyk}{0,0,0,.1}");
50  // The caption fonts (ieeeaccess.cls:75-79), which papers call in their own caption formats
51  // (`\DeclareCaptionLabelFormat{myformat}{\raggedright\figcapfont{\textbf{#1}\textbf{#2}}}`, 2405.14505).
52  RawTeX!(
53    r"\def\figcapheadfont{\sffamily\fontencoding{T1}\fontseries{b}\fontsize{7}{8.4}\selectfont}%
54\def\figcapfont{\def\textit##1{{\sffamily\selectfont\itshape##1}}\sffamily\fontencoding{T1}\fontseries{m}\fontsize{7}{8.4}\selectfont}%
55\def\tablecapheadfont{\sffamily\fontencoding{T1}\fontseries{b}\fontsize{7}{8.4}\selectfont}%
56\def\tableheadfont{\sffamily\fontencoding{T1}\fontseries{m}\fontsize{7}{8.4}\selectfont}%
57\def\tablecapfont{\def\textit##1{{\sffamily\selectfont\itshape##1}}\sffamily\fontencoding{T1}\fontseries{m}\fontsize{7}{8.4}\selectfont}%"
58  );
59
60  // IEEE Access frontmatter (ieeeaccess.cls L270, L448-453, L839) —
61  // preserve author-supplied content as ltx:note frontmatter.
62  DefMacro!(
63    "\\history{}",
64    "\\@add@frontmatter{ltx:note}[role=history]{#1}"
65  );
66  DefMacro!(
67    "\\corresp{}",
68    "\\@add@frontmatter{ltx:note}[role=corresponding]{#1}"
69  );
70  def_macro_noop("\\EOD")?;
71  // `\Figure[pos](keys)[gopts]{file}{caption}` (ieeeaccess.cls:270-300): a `{figure}` (`{figure*}` when the image is wider
72  // than the column — page layout, not kept here) holding `\includegraphics[gopts]{file}` and `\caption{caption}`; the
73  // `figkeys` (`topskip`/`midskip`/`botskip`, :262-267) are vertical spacing, so the class's `\setkeys{figkeys}{#2}`
74  // (:275) is not run. As `\@gobble` it dropped the figure and
75  // spilled the file name, whose `_` erred (1803.04935, 1910.07402). Guard:
76  // `perfect_kernel_batch73a::ieeeaccess_figure_is_a_figure`.
77  RawTeX!(
78    r"\def\Figure{\@ifnextchar[{\@Figure}{\@Figure[t!]}}
79\def\@Figure[#1]{\@ifnextchar({\@@Figure[#1]}{\@@Figure[#1](topskip=0pt)}}
80\def\@@Figure[#1](#2){\@ifnextchar[{\@@@Figure[#1](#2)}{\@@@Figure[#1](#2)[scale=1]}}
81\def\@@@Figure[#1](#2)[#3]#4#5{\begin{figure}[#1]\centering\includegraphics[#3]{#4}\caption{#5}\end{figure}}"
82  );
83  DefRegister!("\\titlepgskip" => Dimension!("0pt"));
84  // Metadata — preserve author values.
85  DefMacro!("\\doi{}", "\\@add@frontmatter{ltx:note}[role=doi]{#1}");
86  DefMacro!(
87    "\\address[]{}",
88    "\\@add@frontmatter{ltx:note}[role=address]{#2}"
89  );
90  DefMacro!(
91    "\\tfootnote{}",
92    "\\@add@frontmatter{ltx:note}[role=titlenote]{#1}"
93  );
94});
