1//! Stub for World Scientific journal classes (ws-ijgmmp, ws-ijmpd, etc.).
2//!
3//! These classes share frontmatter macros like `\catchline{vol}{no}{year}{pgs}{pgsE}`,
4//! `\title{...}`, `\Author{name}`. Raw load of the cls may fail before
5//! defining these (cls uses complex font/page-layout machinery that
6//! doesn't transfer to HTML conversion). Stubs preserve substantive
7//! content as frontmatter notes.
8//!
9//! Witness: 2306.12455 (ws-ijgmmp.cls), 2306.15982 (ws-ijmpd.cls).
10use latexml_core::definition::origin::{DefinitionOrigin, OriginGuard};
11use latexml_package::prelude::*;

14  LoadClass!("OmniBus");
15  // Some copies load amsmath (ws-procs9x6.cls 78 of 90 copies in a 1/20 arXiv sample, ws-ijmpa.cls :1087
16  // `\usepackage{amsfonts,amssymb,amsmath}` in 1912.08245's), the older ones none (nucl-th0206027's ws-procs9x6,
17  // quant-ph0607049's ws-ijmpb: their `\pmatrix{…}` is Plain TeX's in pdflatex, no "Old form" error). The paper's copy
18  // decides; without one, no amsmath of the class's (`amsmath_sty::require_amsmath_beyond_original`; KNOWN_PERL_ERRORS
19  // #993).
20  amsmath_sty::require_amsmath_beyond_original(
21    latexml_package::shipped_class::shipped_copy_loads("cls", "amsmath")?,
22    &[],
23  )?;

25  RequirePackage!("amssymb");
26  RequirePackage!("graphicx");
27  RequirePackage!("xcolor");
28  RequirePackage!("hyperref");
29  // ws-ijmpe.cls:226-230 (ws-ijmpcs.cls:155, ws-mpla.cls:147, ws-rv9x6.cls:160 the same): `\def\bm#1{\mathchoice
30  // {\mbox{\boldmath$#1$}}…}`, bold math the class defines itself, its argument in a box — 1308.3116, 2304.11343,
31  // hep-ph0401137 (ws-procs9x6). A `\left` without its `\right` in the argument (`\bm{\left\langle} x
32  // \bm{\right\rangle}`) is TeX's error at the box's end; it is set at the outer level with a Warning, as bm.sty
33  // would set it (bm_sty.rs `define_grouped_bm`, OXIDIZED_DESIGN #544).
34  bm_sty::define_grouped_bm()?;
35  // The figure packages the ws journal and proceedings classes input (ws-ijmpa.cls:551-555, ws-ijmpe.cls:658-662,
36  // ws-procs9x6.cls:614-618 `\InputIfFileExists{psfig.sty}…{epsfsafe.tex}…{epsfig.sty}…{epsf.sty}`; psfig.sty and
37  // epsfsafe.tex are not in TeX Live, so epsfig's `\psfig` serves), so `\psfig`, `\epsfig` and `\epsfbox` need no
38  // package of the paper's own (run 340 ~650 papers; witnesses 0709.0836, 1310.3924, 0709.3996). A class version that
39  // inputs none of them (ws-mpla.cls of 2508.15289) gets them too: one binding serves the family's union.
40  RequirePackage!("epsfig");
41  RequirePackage!("epsf");
42  // ws-mpla.cls:277-282 (ws-ijmpa.cls:288 the same) `\pub{received}{revised}`: the publication history under the
43  // author block, the template's "Received (Day Month Year)" kept as the paper gives it (run 340 ~500 papers;
44  // witnesses 2508.15289, gr-qc0206056, astro-ph0505113). An empty argument (`\pub{}{}`) makes no note.
45  DefMacro!(
46    "\\pub{}{}",
47    "\\if\\relax\\detokenize{#1}\\relax\\else\\lx@add@pubnote[role=received]{#1}\\fi\
48     \\if\\relax\\detokenize{#2}\\relax\\else\\lx@add@pubnote[role=revised]{#2}\\fi"
49  );
50  // ws-ijmpa.cls:152 (ws-ijmpe.cls:1184, ws-procs9x6.cls:348) `\cdash`: a raised en dash, as in "C\cdash H".
51  DefMacro!("\\cdash", "\\ensuremath{^{\\hbox{--}}}");
52  // ws-mpla.cls:451-482 the class's lists: `{alphlist}`, `{arabiclist}` and `{romanlist}` are numbered (a), (1), (i)
53  // lists with their second level, `{itemlist}` a bulleted one (its enumerate takes `\labelitemi` for every label).
54  RawTeX!(
55    r"\newenvironment{alphlist}{\def\theenumi{\alph{enumi}}\def\theenumii{\roman{enumii}}\def\labelenumi{(\theenumi)}\def\labelenumii{(\theenumii)}\begin{enumerate}}{\end{enumerate}}
56\newenvironment{romanlist}{\def\theenumi{\roman{enumi}}\def\theenumii{\alph{enumii}}\def\labelenumi{(\theenumi)}\def\labelenumii{(\theenumii)}\begin{enumerate}}{\end{enumerate}}
57\@ifundefined{arabiclist}{\newenvironment{arabiclist}{\def\theenumi{\arabic{enumi}}\def\theenumii{\alph{enumii}}\def\labelenumi{(\theenumi)}\def\labelenumii{(\theenumii)}\begin{enumerate}}{\end{enumerate}}}{}
58\newenvironment{itemlist}{\begin{itemize}}{\end{itemize}}"
59  );
60
61  // \catchline{vol}{issue}{year}{pageStart}{pageEnd} — bibliographic
62  // running-header metadata. Preserve as frontmatter note.
63  DefMacro!(
64    "\\catchline{}{}{}{}{}",
65    "\\@add@frontmatter{ltx:note}[role=catchline]{Vol. #1, No. #2 (#3) #4--#5}"
66  );
67
68  // \Journal{...} {...}{...}{...} — citation format helper (4-arg per WS conv).
69  // Render the args inline as a generic citation string.
70  DefMacro!("\\Journal{}{}{}{}", "{#1} {\\bf #2}, #3 (#4)");
71
72  // No-op layout helpers.
73  def_macro_noop("\\copyrightheading{}")?;
74  def_macro_noop("\\paperBack")?;
75  DefMacro!("\\catchlinefont", "\\footnotesize");
76
77  // \ccode{...} — WS classification-codes block (PACS-style codes in
78  // ws-ijmpe and friends). Real def is styled paragraph; preserve as
79  // a classification note. Witness 2307.12748, 2307.16467.
80  DefMacro!(
81    "\\ccode{}",
82    "\\@add@frontmatter{ltx:classification}[scheme=PACS]{#1}"
83  );
84  // \category{code}{name} — ws classification 2-arg variant.
85  DefMacro!(
86    "\\category{}{}",
87    "\\@add@frontmatter{ltx:classification}[scheme=#1]{#2}"
88  );
89
90  // {history} env — publication-history wrapper (received/revised/
91  // accepted dates). Preserve as a frontmatter note. Witness
92  // 2307.12748, 2307.16467 + 2 stage-3/4 ws papers.
93  DefEnvironment!("{history}",
94    "<ltx:note role='history'>#body</ltx:note>",
95    mode => "internal_vertical");
96  // \received{date}, \revised{date}, \accepted{date}, \comby{name}
97  // — used inside {history}. Preserve as inline notes.
98  DefMacro!("\\received{}", "Received #1\\par");
99  DefMacro!("\\revised{}", "Revised #1\\par");
100  DefMacro!("\\accepted{}", "Accepted #1\\par");
101  // ws-*.cls: \def\published#1{Published~#1\par}. Witness 2411.01389 (ws-ijmpa).
102  DefMacro!("\\published{}", "Published~#1\\par");
103  DefMacro!("\\comby{}", "Communicated by #1\\par");
104  // \email / \http / \uurl — render contact info inline.
105  DefMacro!("\\email{}", "\\textit{#1}\\par");
106  DefMacro!("\\http{}", "\\textit{http://#1}\\par");
107  DefMacro!("\\uurl{}", "\\textit{#1}\\par");
108
109  // ws-mpla.cls and other ws classes: \tbl{caption}{body} — table
110  // caption + tabular body wrapper. Real def is essentially
111  // \caption{#1}<body #2 in tabular>. Bridge to \caption{caption}
112  // followed by the body content. Witness 2404.18954 (ws-mpla).
113  DefMacro!("\\tbl{}{}", "\\caption{#1}#2");
114  // \ttbl{label}{caption}{body} — 3-arg variant with explicit label.
115  DefMacro!("\\ttbl{}{}{}", "\\caption{#2}\\label{#1}#3");
116  // \tcaption{caption} — single-arg variant in some ws families.
117  DefMacro!("\\tcaption{}", "\\caption{#1}");
118  // ws-*.cls L406 `\def\refcite{\@ifnextchar[...\@refcitex...}` — a bare-number
119  // citation variant (used as "Ref.~\refcite{key}" → "Ref. 5"). The ws_journal
120  // binding lacked it → undefined (Perl defines it). Map to \cite (a linked
121  // citation; the bare-vs-bracketed number is a post-processing nuance). Accepts
122  // comma-separated keys like \cite. Witness 2306.15982 (ws-ijmpd).
123  DefMacro!("\\refcite{}", "\\cite{#1}");
124
125  // ws-procs9x6.cls:721-731, ws-ijmpa.cls:685-694 (every ws class, the lengths differing; ws-ijmpb.cls:653-658 in
126  // cond-mat0509290's copy; ws-rv9x6.cls:618-620 an `\hline` for `\Hline`): the table rules. `\Hline` is a
127  // double-weight `\hline`; the others add a rule (or `\crule`'s argument, a `\cline`) between rows that are empty and
128  // of negative height, which the alignment drops, so each is the rule it draws. Witness 0704.0076 (ws-ijmpa).
129  RawTeX!(
130    r"\def\Hline{\noalign{\ifnum0=`}\fi\hrule \@height 2\arrayrulewidth \futurelet\@tempa\@xhline}
131\def\toprule{\Hline\\[-6.5pt]}
132\def\colrule{\\[-7pt]\hline\\[-6pt]}
133\def\TMcolrule{\\[-9.5pt]\hline\\[-9.5pt]}
134\def\botrule{\\[-7pt]\Hline}
135\def\crule#1{\\[-7pt]#1\\[-6pt]}"
136  );
137  // The table and title-block commands every WS class defines (ws-ijmpe.cls v1.0 as 0711.3504 ships it, ws-procs9x6.cls
138  // v1.2u as 0704.0369 does, ws-procs961x669.cls v1.5r 1605.04791's, ws-rv9x6.cls 0706.1148's), papers ship-
139  // relying on them (run 347: 46 ws-procs9x6 / ws-procs961x669 / ws-ijmpe papers): ws-ijmpe :58-59, :69-71 the fonts
140  // `\titlefont`, `\authorfont`, `\tablefont`, `\tabnotefont` (`\title{\titlefont{…}}`, 1111.6854); :740-751 the
141  // table note `{tabnote}` (`[heading]`, "Note" by default: "Note: …" under the table; 0711.3504, 0704.0369, 1605.04791;
142  // the procs classes :714-719 the same text), :754-763 `{tabfootnote}` with its `\tabmark{a}` marks (0712.0330,
143  // 1404.6651, 1403.1664), the table's box layout their own; :334-340 `{otherinfo}`, a centred line under the authors
144  // (0811.0496's "Published in: …"); :951 (ws-procs9x6 :389) `\citelow`, `\cite` set on the line instead of raised
145  // (1302.0454, hep-ph0301130); :667 (ws-procs9x6 :651) `\figurebox{height}{width}{file}`, the figure's EPS from
146  // `\ArtDir`, or an empty frame for a figure to come (nucl-th0412108, astro-ph0210021). A class of the family that
147  // lacks one leaves it to the document (`DefinitionOrigin::Fallback`).
148  {
149    let _class = OriginGuard::new(DefinitionOrigin::Fallback);
150    RawTeX!(
151      r"\def\titlefont{\fontsize{10}{13}\bfseries\boldmath\selectfont\centering{}}
152\def\authorfont{\footnotesize}
153\def\tablefont{\footnotesize}
154\def\tabnotefont{\footnotesize}
155\def\tabnotehead{Note}
156\def\tabsourcehead{Source}
157\newenvironment{tabnote}{\par\tabnotefont\@ifnextchar[{\@tabnote}{\@tabnote[\tabnotehead]}}{\par}
158\def\@tabnote[#1]{\noindent\def\@Tempa{#1}\ifx\@Tempa\@empty\else{\it#1}:\ \ignorespaces\fi}
159\newenvironment{tabfootnote}{\par\tabnotefont}{\par}
160\def\tabmark#1{\noindent\hbox{\normalsize$\m@th^{#1}$}\ignorespaces}
161\newenvironment{otherinfo}{\authorfont\begin{center}}{\end{center}}
162\def\citelow{\cite}
163\def\ArtDir{\jobname/}
164\def\figurebox#1#2#3{\def\arg{#3}\ifx\arg\empty{\hfill\vbox{\hsize#2\hrule\hbox to #2{\vrule\hfill\vbox to #1{\hsize#2\vfill}\vrule}\hrule}\hfill}\else\hbox to \hsize{\hfill\epsfbox{\ArtDir#3.eps}\hfill}\fi}"
165    );
166  }
