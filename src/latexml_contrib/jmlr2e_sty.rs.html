164/// The heading a jmlr2e.sty copy's `\acks` prints: the text of its bold group, `\long\def\acks#1{\vskip 0.3in\noindent
165/// {\large\bf Acknowledgments}…`.
166fn acks_heading(code: &str) -> Option<String> {
167  let body = &code[code.find("\\def\\acks")?..];
168  let body = &body[..body.find("#1}").map_or(body.len(), |end| end + 3)];
169  let text = &body[body.find("\\bf")? + 3..];
170  let text = text[..text.find('}')?].trim();
171  (!text.is_empty()).then(|| text.to_string())
172}
