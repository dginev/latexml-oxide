1//! mathspec.sty — XeLaTeX math font selection over fontspec (no Perl binding).
2//!
3//! mathspec.sty:9-10 requires XeTeX and :60 passes its options to fontspec; its
4//! setters choose the fonts of math letters and digits, which LaTeXML does not
5//! model (it keeps its own math fonts). The raw file checks `\XeTeXversion`
6//! (:14) and raises its own errors under the Unicode-native default persona
7//! (arXiv 2605.01803 ×63, 2605.02644), so the setters are bound as the
8//! argument-reading no-ops they are here. Guard:
9//! `perfect_kernel_batch56::xetex_only_packages_load_under_the_default_persona`.
10use latexml_package::prelude::*;
11
12LoadDefinitions!({
13  RequirePackage!("fontspec");
14  // mathspec.sty:223-233 `\setmathsfont(<sets>)[<features>]{<font>}` (the set
15  // list is optional: `\setmathsfont[<features>]{<font>}` is the `Special`
16  // branch), `\setmathfont` its alias; :916-931 `\setallmainfonts(<sets>)…`,
17  // `\setprimaryfont`, `\setallsansfonts`/`\setallmonofonts`; :150-156
18  // `\setsansfonts`/`\setmonofonts`; :694 `\setminwhitespace[<n>]`.
19  RawTeX!(
20    r"\def\lx@mathspec@sets{\@ifnextchar(\lx@mathspec@setlist{\lx@mathspec@setlist()}}
21\def\lx@mathspec@setlist(#1){\@ifnextchar[\lx@mathspec@font{\lx@mathspec@font[]}}
22\def\lx@mathspec@font[#1]#2{}
23\def\setmathsfont{\lx@mathspec@sets}
24\let\setmathfont\setmathsfont
25\def\setallmainfonts{\lx@mathspec@sets}
26\def\setprimaryfont{\lx@mathspec@sets}
27\def\setallsansfonts{\@ifnextchar[\lx@mathspec@font{\lx@mathspec@font[]}}
28\let\setallmonofonts\setallsansfonts
29\let\setsansfonts\setallsansfonts
30\let\setmonofonts\setallsansfonts
31\providecommand\setmathcal[2][]{}
32\providecommand\setmathbb[2][]{}
33\providecommand\setmathfrak[2][]{}
34\newcommand\setminwhitespace[1][500]{}"
35  );
36});
