1//! multibib: the raw package, with each list's bibliography style read at that list.
2//!
3//! multibib.sty:234 defines `\bibliographystyle<suffix>` to write `\bibstyle` to the side aux file bibtex reads
4//! (`<suffix>.aux`); it never calls `\bibliographystyle`, so `BIBSTYLE`, which `\bibliographystyle` records for the
5//! `.bib` route (`\lx@record@bibstyle`, sect11.rs), stayed empty. A document with only such lists then read no `.bst`
6//! (`bst_declared_fields`, OXIDIZED_DESIGN_DIVERGENCES #823): the fields its style declares stayed raw data
7//! (`urldate`, quantum.bst:44, printed "(accessed:~12.07.2019)" at :719; the quantum-bibliographystyle-demo manual).
8//!
9//! The record is keyed by the suffix `\newcites` declares, and `\bibliography<suffix>` reads its own in a group around
10//! the list: `\bibliographystyle{alpha}\bibliographystyleweb{plain}` gives `alpha` on the main list and `plain` on
11//! `web`, as pdflatex's `[Zet01]` and `[1]`, in either order of the lists; opticajnl.cls:288's and osajnl's
12//! `\bibliographystylefullrefs{opticajnl}` beside the document's `opticajnlnt` (1908.11477, 2006.07118, 2008.09542,
13//! 2205.11481, 2308.15957, 2607.03975). Where `\bibstyle` is still the
14//! kernel's the list takes the style's citation style too; a package that took `\bibstyle` owns that (natbib's
15//! `\@gobble`, natbib_sty.rs), and the list takes the style's name and order (`record_bibstyle`, as `\bibliographystyle`). `BIBSTYLE` after the lists stays the main list's, or with none the first declared.
16//! Limit: a style declared after its own list reaches `BIBSTYLE` only, not that list's attribute.
17use latexml_engine::latex_constructs::{record_bibstyle, set_bibstyle};
18use latexml_package::prelude::*;
19
20LoadDefinitions!({
21  InputDefinitions!("multibib", noltxml => true, extension => Some(Cow::Borrowed("sty")));
22  // `\bibliographystyle<suffix>`: the suffix's record; `BIBSTYLE` too while nothing holds it (a `\bibliographystyle` always may).
23  DefPrimitive!("\\lx@mb@declare{}{}", sub[(suffix, style)] {
24    let suffix = Expand!(suffix).to_string();
25    let style = pin(Expand!(style).to_string());
26    assign_value(&s!("lx@mb@style@{suffix}"), style, Some(Scope::Global));
27    if lookup_string("BIBSTYLE").is_empty() {
28      assign_value("BIBSTYLE", style, Some(Scope::Global));
29    }
30  });
31  // `\bibliography<suffix>`, in the list's group: its style while the list opens.
32  DefPrimitive!("\\lx@mb@open{}", sub[(suffix)] {
33    let style = lookup_string(&s!("lx@mb@style@{}", Expand!(suffix).to_string()));
34    if style.is_empty() {
35      // (none declared: the standing one, as before)
36    } else if matches!(lookup_meaning(&T_CS!("\\bibstyle")), Some(Stored::Constructor(_))) {
37      set_bibstyle(&style);
38    } else {
39      record_bibstyle(&style, None);
40    }
41  });
42  RawTeX!(
43    r"\def\lx@mb@record#1{\@ifundefined{bibliographystyle#1}{}{%
44    \expandafter\let\csname lx@mb@style@#1\expandafter\endcsname\csname bibliographystyle#1\endcsname
45    \expandafter\let\csname lx@mb@list@#1\expandafter\endcsname\csname bibliography#1\endcsname
46    \@namedef{bibliographystyle#1}##1{\lx@mb@declare{#1}{##1}\@nameuse{lx@mb@style@#1}{##1}}%
47    \@namedef{bibliography#1}##1{{\lx@mb@open{#1}\@nameuse{lx@mb@list@#1}{##1}}}}}
48\let\lx@mb@newcites\newcites
49\def\newcites#1#2{\lx@mb@newcites{#1}{#2}%
50  \@for\@tempa:=#1\do{\expandafter\lx@mb@record\expandafter{\@tempa}}}"
51  );
52});
