1//! Binding for apa7.cls (APA 7th edition manuscripts, TeX Live apa7 2022/07/25).
2//!
3//! The class itself is read for its modes (man, stu, jou, doc), the packages it loads and its commands; the bibliography
4//! package is the one its option names (none by default, :179-216). Its author block — names, numbered affiliations,
5//! author note, note — is kept in etoolbox lists for its own title page (:780-890), which the frontmatter replaces, so
6//! those setters are mapped onto the frontmatter API here. With the `mask` option the class prints no author identity
7//! (:131, 1257-1310), and neither does the binding.
8use latexml_package::prelude::*;
9
10/// The items of an etoolbox comma list (`\forcsvlist`, etoolbox.sty:1373): split at top-level commas, each item's
11/// surrounding spaces and one enclosing brace group removed (`{1,3}` → `1,3`); empty items are dropped.
12fn csv_items(list: Tokens) -> Vec<Tokens> {
13  split_tokens(list, vec![T_OTHER!(",").into()])
14    .into_iter()
15    .filter_map(|item| {
16      let mut toks = item.unlist();
17      while toks.first() == Some(&T_SPACE!()) {
18        toks.remove(0);
19      }
20      while toks.last() == Some(&T_SPACE!()) {
21        toks.pop();
22      }
23      if toks.len() >= 2
24        && toks[0].get_catcode() == Catcode::BEGIN
25        && toks[toks.len() - 1].get_catcode() == Catcode::END
26      {
27        let mut depth = 0usize;
28        let closes_last = toks.iter().enumerate().all(|(i, t)| {
29          match t.get_catcode() {
30            Catcode::BEGIN => depth += 1,
31            Catcode::END => depth -= 1,
32            _ => {},
33          }
34          depth > 0 || i == toks.len() - 1
35        });
36        if closes_last {
37          toks = toks[1..toks.len() - 1].to_vec();
38        }
39      }
40      (!toks.is_empty()).then(|| Tokens::new(toks))
41    })
42    .collect()
43}

45LoadDefinitions!({
46  // apa7.cls:1041-1075 redefines `\appendix` to set each appendix `\section` as a centred heading under its own
47  // `appendix` counter; the kernel's `\section` refuses the redefinition, so the class's `\appendix` left the
48  // sections numbered on in arabic ("Appendix 9" for pdflatex's "Appendix A", 2609.00670). The kernel's appendix, its
49  // sections lettered, stands (semantic markup before PDF fidelity, user 2026-10-06).
50  // article (apa7.cls:159-175 `\LoadClass`, loaded here first so its `\appendix` can be kept).
51  LoadClass!("article");
52  Let!("\\lx@apa@kernel@appendix", "\\appendix");
53  InputDefinitions!("apa7", noltxml => true, extension => Some(Cow::Borrowed("cls")));
54  Let!("\\appendix", "\\lx@apa@kernel@appendix");
55
56  // apa7.cls:798-806: `\authorsnames[marks]{names}`, both comma lists, the n-th mark list for the n-th name, printed
57  // as `name\textsuperscript{marks}` (:840-863) over the affiliations numbered in order (:868-881). The names, with
58  // their marks, go to the kernel's author-line parser, which links each mark to the affiliation it numbers
59  // (2609.12869 `[{1,3},{2},{3,5},4]`, 00670, 02899); without marks, as written (2609.19448 `David M. Abel$^1$`).
60  // `\author` is the class's one-name form (:766), which the kernel's `\author` already is.
61  DefMacro!("\\authorsnames[]{}", sub[(marks, names)] {
62    let marked = marks.as_ref().is_some_and(|m| !m.is_empty());
63    let mut line: Vec<Token> = Vec::new();
64    if marked {
65      let marks = csv_items(marks.unwrap());
66      for (i, name) in csv_items(names).into_iter().enumerate() {
67        if i > 0 {
68          line.extend([T_OTHER!(","), T_SPACE!()]);
69        }
70        line.extend(name.unlist());
71        if let Some(mark) = marks.get(i) {
72          line.extend([T_CS!("\\textsuperscript"), T_BEGIN!()]);
73          line.extend(mark.clone().unlist());
74          line.push(T_END!());
75        }
76      }
77    } else {
78      line = names.unlist();
79    }
80    let mut out = TokenizeInternal!(r"\def\lx@apa@marked").unlist();
81    out.push(T_BEGIN!());
82    if marked {
83      out.push(T_OTHER!("1"));
84    }
85    out.push(T_END!());
86    out.extend(TokenizeInternal!(r"\@ifundefined{apaSeven@maskauthoridentity}").unlist());
87    out.push(T_BEGIN!());
88    out.push(T_CS!("\\lx@add@authors"));
89    out.push(T_BEGIN!());
90    out.extend(line);
91    out.push(T_END!());
92    out.push(T_END!());
93    out.extend([T_BEGIN!(), T_END!()]);
94    // the affiliations given before the names, linked now
95    out.extend(TokenizeInternal!(r"\gdef\lx@apa@named{1}\lx@apa@pending\gdef\lx@apa@pending{}").unlist());
96    Ok(Tokens::new(out))
97  });
98  RawTeX!(r"\def\lx@apa@marked{}\def\lx@apa@named{}\def\lx@apa@pending{}");
99  // apa7.cls:808-812: `\authorsaffiliations{{first},{second}}`, numbered in order when the names carry marks (:875-878),
100  // each linked to the names marked with its number; else listed under the names, each for every author.
101  // `\affiliation` is its deprecated one-item form (:767).
102  // apa7.cls builds its title block at `\maketitle` from the stored lists (:780-890), so the affiliations may come
103  // before the names: they are kept until the names are given, which say whether they are numbered.
104  DefMacro!("\\authorsaffiliations{}", sub[(list)] {
105    if do_expand(T_CS!("\\lx@apa@named"))?.is_empty() {
106      let mut kept = TokenizeInternal!(r"\g@addto@macro\lx@apa@pending").unlist();
107      kept.extend([T_BEGIN!(), T_CS!("\\authorsaffiliations"), T_BEGIN!()]);
108      kept.extend(list.unlist());
109      kept.extend([T_END!(), T_END!()]);
110      return Ok(Tokens::new(kept));
111    }
112    let marked = !do_expand(T_CS!("\\lx@apa@marked"))?.is_empty();
113    let mut out: Vec<Token> = Vec::new();
114    for (j, item) in csv_items(list).into_iter().enumerate() {
115      if marked {
116        let mut labelled = vec![T_CS!("\\textsuperscript"), T_BEGIN!()];
117        labelled.extend(mouth::tokenize(TeXString::assembled((j + 1).to_string())).unlist());
118        labelled.extend([T_END!(), T_BEGIN!()]);
119        labelled.extend(item.unlist());
120        labelled.push(T_END!());
121        out.extend(Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![None, Some(Tokens::new(labelled))]).unlist());
122      } else {
123        // Unnumbered: linked by marks of its own (2609.19448 `$^1$Department …\\…`) when the names carry marks,
124        // else listed under the names, each for every author.
125        out.extend(
126          Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![Some(TokenizeInternal!("annotate=all")), Some(item)])
127            .unlist(),
128        );
129      }
130    }
131    let mut masked = TokenizeInternal!(r"\@ifundefined{apaSeven@maskauthoridentity}").unlist();
132    masked.push(T_BEGIN!());
133    masked.extend(out);
134    masked.push(T_END!());
135    masked.extend([T_BEGIN!(), T_END!()]);
136    Ok(Tokens::new(masked))
137  });
138  DefMacro!("\\affiliation{}", "\\authorsaffiliations{{#1}}");
139  // apa7.cls:766: `\author` is the one-name `\authorsnames`, so `mask` hides it too.
140  DefMacro!("\\author{}", "\\authorsnames{#1}");

188  // A DOI stays a link to its record (user 2026-10-06: semantic markup before PDF fidelity), as OmniBus's `\doi` gave
189  // these papers before the binding: apacite's own (`\providecommand{\doi}`, apacite.sty:1812-1815, which this then
190  // leaves) sets it as url-styled text only. Read as a url is, so `_`, `~` and `%` are its own characters — from a
191  // brace only, as OmniBus's and apacite's `\doi` (a bare `\doi` takes its one token, not the text up to the next
192  // brace).
193  DefMacro!(
194    "\\doi",
195    "\\@ifnextchar\\bgroup\\lx@apa@doi@verbatim\\lx@apa@doi@token"
196  );
197  DefMacro!(
198    "\\lx@apa@doi@verbatim HyperVerbatim",
199    "\\href{https://doi.org/#1}{#1}"
200  );
201  DefMacro!("\\lx@apa@doi@token{}", "\\href{https://doi.org/#1}{#1}");
202  // apa7.cls:322-326: the name followed by the ORCID iD logo linking to the record (2609.00670, 02899); the
203  // kernel's ORCID link (`\lx@orcidlink`, as orcidlink.sty's); an iD given as its orcid.org address is the iD
204  // (`\lx@orcid@id`, OXIDIZED_DESIGN_DIVERGENCES #616).
205  DefMacro!(
206    "\\addORCIDlink{}{}",
207    "#1\\ \\expandafter\\lx@orcidlink\\expandafter{\\lx@orcid@id{#2}}{\\lx@orcidlogo}"
208  );
