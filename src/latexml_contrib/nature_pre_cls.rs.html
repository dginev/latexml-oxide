1//! Binding for nature-pre.cls and nature_mod.cls, the copies papers ship of TL's nature.cls (the Nature pre-print
2//! template).
3use latexml_package::prelude::*;
4
5LoadDefinitions!({
6  LoadClass!("OmniBus");
7  RequirePackage!("amsmath");
8  RequirePackage!("amsthm");
9  // Eager xcolor preload removed for Perl parity: it makes a later document
10  // xcolor[table] load a no-op, so colortbl/array never load and array m{}/b{}
11  // columns break (Unrecognized tabular template -> Extra alignment tab). The
12  // document loads xcolor itself; color/definecolor stay via hyperref->color.
13  // See ifacconf_cls.rs and SYNC_STATUS (eager-xcolor cluster).
14  RequirePackage!("hyperref");
15
16  // The class's own list environments, verbatim (TL nature.cls:67-81 and :110-123, byte-identical in the
17  // nature-pre.cls and nature_mod.cls copies papers ship: 2608.12563's nature-pre.cls:67-81 and :110-123,
18  // 1702.00936's and 2402.00473's nature_mod.cls:74-88 and :117-130), so the list typesets the class's labels:
19  // `{affiliations}`, put after `\maketitle`, numbers each institution `\upshape$^{\arabic{enumi}}$` (an `\item[*]`
20  // its own mark) in a `\slshape` body (2608.12563, 1702.00936, 1408.2522, 2412.11966); `{addendum}` labels each item
21  // through `\addendumlabel`, the bold `\item[Competing Interests]` (2402.00473, 2412.11966), the list's
22  // "Acknowledgements" for a bare `\item` (1408.2522, 1702.00936; not bold: the kernel's `\list` sets its default
23  // label without `\makelabel`, as Perl's latex_constructs.pool.ltxml:1642-1643 does — a label's styling is the CSS
24  // theme's, the item and its tag text are the markup, user ruling 2026-10-30). Emptied to nothing (the
25  // former stubs), the items were plain paragraphs without their labels, and the affiliations read as body prose to
26  // the cover rule (`cover_order.rs` `is_body_prose`), so a later abstract stayed above them.
27  RawTeX!(
28    r"\newenvironment{affiliations}{%
29    \setcounter{enumi}{1}%
30    \setlength{\parindent}{0in}%
31    \slshape\sloppy%
32    \begin{list}{\upshape$^{\arabic{enumi}}$}{%
33        \usecounter{enumi}%
34        \setlength{\leftmargin}{0in}%
35        \setlength{\topsep}{0in}%
36        \setlength{\labelsep}{0in}%
37        \setlength{\labelwidth}{0in}%
38        \setlength{\listparindent}{0in}%
39        \setlength{\itemsep}{0ex}%
40        \setlength{\parsep}{0in}%
41        }
42    }{\end{list}\par\vspace{12pt}}
43\newenvironment{addendum}{%
44    \setlength{\parindent}{0in}%
45    \small%
46    \begin{list}{Acknowledgements}{%
47        \setlength{\leftmargin}{0in}%
48        \setlength{\listparindent}{0in}%
49        \setlength{\labelsep}{0em}%
50        \setlength{\labelwidth}{0in}%
51        \setlength{\itemsep}{12pt}%
52        \let\makelabel\addendumlabel}
53    }
54    {\end{list}\normalsize}
55\newcommand*{\addendumlabel}[1]{\textbf{#1}\hspace{1em}}"
56  );
57  // The class defines no `\thanks` and no `\correspondingauthor`: the kernel's `\thanks` is the one the page prints
58  // (inside `\author` a footnote on the name, 2402.00473), and `\correspondingauthor` is no command of the class or
59  // of any copy shipped (the binding's former definitions, dropped: user ruling 2026-10-30, a binding defines only
60  // what its class defines).
61});
