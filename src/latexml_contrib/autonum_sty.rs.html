1//! autonum.sty binding (no Perl binding). OXIDIZED_DESIGN_DIVERGENCES #895.
2//!
3//! autonum numbers only the equations a reference names: its `\ref`/`\eqref`/`\cref` record the labels they name in
4//! the aux file (autonum.sty:330-345), and on the next run its display environments, renewed at `\begin{document}`
5//! (:127-140, :186-280), drop the numbers of the rest. A single pass has no earlier run to read. Raw-loaded, its
6//! renewals also reach only the environments a binding does not lock: `{split}`'s ran inside our locked `{align}`, whose
7//! own renewal — the one that restores `\label` at its end (`\autonum@restoreSubcommands`) — never did, so the display's
8//! `\label` was left global and every later `\label` failed ("\lx@ams@display@outer@label undefined", the label lost:
9//! 1905.00606). The binding loads autonum's packages and options and leaves the displays to amsmath's binding: every
10//! equation keeps its number, as without autonum.
11use latexml_package::prelude::*;
12
13LoadDefinitions!({
14  // autonum.sty:17-25, with its :20-22: etextools is loaded and etoolbox's `\forlistloop` put back. Without that etextools'
15  // own `\forlistloop` broke translations.sty:1129's `\dolistloop` hook, an Extra `\fi` taking the next begin-document
16  // chunk (simpler-wick, tikz-feynman).
17  RawTeX!(r"\PassOptionsToPackage{hypertexnames=false}{hyperref}");
18  RequirePackage!("etoolbox");
19  Let!("\\autonum@etoolbox@forlistloop", "\\forlistloop");
20  RequirePackage!("etextools");
21  Let!("\\forlistloop", "\\autonum@etoolbox@forlistloop");
22  RequirePackage!("amsmath");
23  RequirePackage!("textpos", options => vec![String::from("absolute")]);
24  RequirePackage!("letltxmacro");
25  // autonum.sty:127-133, 176-183: each display environment it renews is kept as `<env>+`, the environment as amsmath
26  // left it (`\newenvironment{#1+}[#2]{\csuse{autonum@#1Old}}{\csuse{autonum@end#1Old}}`): 1405.2249's `{align+}`.
27  // A constructor (`\begin{equation}`) is aliased as one, a macro pair (`\align`/`\endalign`) as a pair.
28  // autonum.sty:302-322: `\autonum@generatePatchedReference{cmd}` makes a reference command mark the labels it names,
29  // for a document's own reference commands too (2211.09537's `\autonum@generatePatchedReference{autoref}`). Every
30  // equation keeps its number here, so a reference marks nothing and the command stays as it is.
31  DefMacro!("\\autonum@generatePatchedReference{}", "");
32  DefMacro!("\\autonum@generatePatchedReferenceCSL{}", "");
33  DefMacro!("\\autonum@generatePatchedReferenceGeneral{}{}", "");
34  for env in [
35    "equation", "gather", "multline", "flalign", "align", "alignat", "split",
36  ] {
37    let (constructor, end) = (s!("\\begin{{{env}}}"), s!("\\end{{{env}}}"));
38    if is_defined(&constructor) {
39      Let!(T_CS!(s!("\\begin{{{env}+}}")), T_CS!(constructor));
40      Let!(T_CS!(s!("\\end{{{env}+}}")), T_CS!(end));
41    } else if is_defined(&s!("\\{env}")) {
42      Let!(T_CS!(s!("\\{env}+")), T_CS!(s!("\\{env}")));
43      Let!(T_CS!(s!("\\end{env}+")), T_CS!(s!("\\end{env}")));
44    }
45  }
46});
