184/// imsart.sty's commands the binding owns: its frontmatter and its keyword blocks, defined before the class reads
185/// imsart.sty raw (so its raw definitions are refused) and by the imsart.sty binding (`imsart_sty.rs`), which a shipped
186/// copy declaring itself imsart.sty reaches (OXIDIZED_DESIGN_DIVERGENCES #986).
187#[rustfmt::skip]
188pub fn imsart_sty_markup() -> Result<()> {
189  // imsart.sty's frontmatter, defined before its raw definitions so they are refused. `\ead[label=e1]{addr}`
190  // (imsart.sty:944 `\DeclareRobustCommand\ead[2][label= ,email]`) is the author's email, or URL with the `url` key
191  // or a `://` address (`\checkead@prefix`, :955); OmniBus's `\ead{}[]` (elsarticle's order) read `[` as the address
192  // and left `label=e1]{…}` in the author's name (2201.11773, 2507.12447). `\printead[presep=…]{e1,u1}` (:965)
193  // reprints, in an address, the addresses `\ead` gave the authors, each already their contact. `\author[A,B]{…}`
194  // and `\address[A]{…}` (:2003, :2088) link an author to the addresses its marks label, as revtex's and elsarticle's
195  // do (revtex4_support_sty.rs, elsart_support_core_sty.rs). The address prints in typewriter type (imsart.sty:1162
196  // `\setattribute{ead}{size}{\upshape\ttfamily}`; 1601.00202's `\ead[url]`).
197  DefMacro!("\\ead[] Semiverbatim", sub[(keys, address)] {
198    // (stored under its label for a `\printead{e1}` in an address printed in place, below)
199    let url = record_ead(keys.as_ref(), &address);
200    let address = Invocation!(T_CS!("\\lx@address@urlfont"), vec![Some(address)]);
201    Ok(if url {
202      Invocation!(T_CS!("\\lx@add@url"), vec![None, Some(address)])
203    } else {
204      Invocation!(T_CS!("\\lx@add@email"), vec![None, Some(address)])
205    })
206  }, locked => true);
207  // In an address printed in place (below), no author's contact, `\printead{e1,u1}` prints what imsart.sty's
208  // `\@@printead` (:506-519 of 1301.1466's copy) prints: each labelled address `\ead` stored, "e-mail: " or "url: " before
209  // it (`\email@text`, `\url@text`; none under `*`), `;~` between, in typewriter type; `\printead[text]{…}` the text alone
210  // (review revCR3b finding 1: 1301.1466's end blocks, `\printead{e1}\\ \printead{u1}`). A label no `\ead` gave prints
211  // "??", as `\saferef`. Inside the frontmatter it prints nothing: the address it stands in is the author's contact.
212  DefMacro!("\\printead OptionalMatch:* []{}", sub[(star, text, labels)] {
213    if !lookup_bool("lx_imsart_address_inplace") {
214      return Ok(Tokens!());
215    }
216    if let Some(text) = text {
217      return Ok(Invocation!(T_CS!("\\lx@address@urlfont"), vec![Some(text)]));
218    }
219    let mut printed: Vec<Token> = Vec::new();
220    for (at, label) in labels.to_string().split(',').map(str::trim).filter(|l| !l.is_empty()).enumerate() {
221      if at > 0 {
222        printed.extend(mouth::tokenize_internal(";~").unlist());
223      }
224      let kind = match lookup_value(&format!("lx_imsart_ead_{label}")) {
225        Some(Stored::Tokens(kind)) => kind.to_string(),
226        _ => String::from("email"),
227      };
228      if star.is_none() {
229        printed.extend(mouth::tokenize_internal(if kind == "url" { "url:" } else { "e-mail:" }).unlist());
230        printed.push(T_SPACE!());
231      }
232      let address = match lookup_value(&format!("lx_imsart_ead_{kind}_{label}")) {
233        Some(Stored::Tokens(address)) => Invocation!(T_CS!("\\lx@address@printed"), vec![Some(address)]),
234        _ => mouth::tokenize_internal("??"),
235      };
236      printed.extend(Invocation!(T_CS!("\\lx@address@urlfont"), vec![Some(address)]).unlist());
237    }
238    Ok(Tokens::new(printed))
239  }, locked => true);
240  // (a name list in one `\author{A and B}` is an author each, every one with the marks: 2401.00578, 2505.07640)
241  // (each `\ead` in it recorded at the call, as imsart.sty's `\safelabel` writes the aux file that a `\printead` reads on
242  // the next run, wherever it stands: an address printed in place before the frontmatter's flush digests the author)
243  DefMacro!("\\author OptionalSemiverbatim {}", sub[(annotations, name)] {
244    record_eads(&name);
245    Ok(Invocation!(T_CS!("\\lx@add@authors@append"), vec![
246      Some(Tokens::new(
247        [mouth::tokenize_internal("annotations=").unlist(), vec![T_BEGIN!()],
248         annotations.map(Tokens::unlist).unwrap_or_default(), vec![T_END!()]].concat()
249      )),
250      Some(name)
251    ]))
252  }, locked => true);
253  // An unlabelled `\address` after `\end{frontmatter}` is no author's: under the journal options that set
254  // `\@addressatendtrue` (`aap`, imsart.sty:150 of 1301.1466's copy) `\address` stores each block (:1242-1252) and the
255  // end of the document prints them all (`\printaddresses`, :1269-1290), each tied to no author; the API's default
256  // owner, the last author, would take every one (1301.1466's three after its `\bibliography`). It is set as
257  // `\printaddressnum` sets a block (`\address@size`, `{tabular}[t]{@{}l@{}}`, :1255-1264), at the end of the document under
258  // `\@addressatendtrue` (`\AtEndDocument`, as imsart.sty's `\enddocument` runs `\printaddresses`; review revCR3b NIT 4),
259  // else where it is called. Inside `{frontmatter}`, or labelled (`\address[A]`, the authors marked A), it is a contact;
260  // unlabelled, of the authors whose `\ead` labels its `\printead{e1}` cites (`citing_authors`: 1407.2451's ETH block
261  // Nandy's and Maathuis's, its Washington block Richardson's; review revCR4 finding 2), else of every author since the
262  // last `\address` (`annotate=run`, as the sty binding's `\affiliation`): imsart.sty:1021 resets `\author@fmt@init`
263  // there, so `\author{A}\and\author{B}\address{Univ AB}` prints the address under both (review revCR3b finding 3).
264  RawTeX!(
265    r"\AddToHook{env/frontmatter/after}{\lx@imsart@frontmatter@closed}
266\def\lx@imsart@address@contact#1#2{\lx@add@contact[label={#1},role=address]{#2}}
267\def\lx@imsart@address@run#1#2{\lx@add@contact[label={#1},role=address,annotate=run]{#2}}
268\def\lx@imsart@address@inplace#1#2{\@ifundefined{if@addressatend}{\lx@imsart@address@print{#2}}{\csname if@addressatend\endcsname\expandafter\g@addto@macro\expandafter\lx@imsart@addresses@end\else\expandafter\@firstofone\fi{\lx@imsart@address@print{#2}}}}
269\def\lx@imsart@address@print#1{\par\begingroup\lx@imsart@address@inplace@on\csname address@size\endcsname\noindent\begin{tabular}[t]{@{}l@{}}#1\end{tabular}\par\endgroup}
270\def\lx@imsart@addresses@end{}
271\AtEndDocument{\lx@imsart@addresses@end}"
272  );
273  DefPrimitive!("\\lx@imsart@address@inplace@on", sub[_args] {
274    assign_value("lx_imsart_address_inplace", true, Some(Scope::Local));
275    Ok(())
276  });
277  DefPrimitive!("\\lx@imsart@frontmatter@closed", sub[_args] {
278    assign_value("lx_imsart_frontmatter_closed", true, Some(Scope::Global));
279    Ok(())
280  });
281  DefMacro!("\\address OptionalSemiverbatim {}", sub[(label, address)] {
282    let label = label.unwrap_or_default();
283    Ok(if label.is_empty() && lookup_bool("lx_imsart_frontmatter_closed") {
284      Invocation!(T_CS!("\\lx@imsart@address@inplace"), vec![Some(label), Some(address)])
285    } else if label.is_empty() {
286      // (the authors whose `\ead` labels its `\printead{e1,e2}` cites, each its owner — `author:N`, the N-th author's
287      // own label —, else, or when they are the run, the run's)
288      let (owners, run) = citing_authors(&printead_labels(&address));
289      if owners.is_empty() || owners == run {
290        Invocation!(T_CS!("\\lx@imsart@address@run"), vec![Some(label), Some(address)])
291      } else {
292        let mut contacts: Vec<Token> = Vec::new();
293        for owner in owners {
294          contacts.extend(
295            Invocation!(T_CS!("\\lx@imsart@address@contact"), vec![
296              Some(mouth::tokenize_internal(TeXString::assembled(format!("author:{owner}")))),
297              Some(address.clone())
298            ])
299            .unlist(),
300          );
301        }
302        Tokens::new(contacts)
303      }
304    } else {
305      Invocation!(T_CS!("\\lx@imsart@address@contact"), vec![Some(label), Some(address)])
306    })
307  }, locked => true);
308  // `\thanksref[mark]{addr1,t1}` (imsart.sty:746) marks the author with the addresses and notes those labels name, and
309  // `\thankstext{t1}{text}` (:789) is such a note: as elsarticle's `\thanksref`/`\thanks` (elsart_support_core_sty.rs).
310  // Read raw, the note went to `\@thanks`, which nothing typesets, and its text was lost (2201.08502).
311  // The `thanks:` prefix on both sides, as elsarticle's, matches a title's `\thanksref{T1}` to its note exactly (the
312  // prefix-stripped match serves creators only, OXIDIZED_DESIGN_DIVERGENCES #461), and an author's to an `\address`.
313  DefMacro!(
314    "\\thanksref[]{}",
315    "\\lx@request@frontmatter@annotation[thanks]{#2}",
316    locked => true
317  );
318  DefMacro!(
319    "\\thankstext[]{}{}",
320    "\\lx@add@contact[label={thanks:#2},role=thanks]{#3}",
321    locked => true
322  );
323  // `\thanksmark[mark]{m1}` (imsart.sty:640-645) prints, in an `\affiliation`, the mark the authors' `\thanksref{m1}`
324  // point at; the authors' marks are not printed here (the `\thanksref` annotation links them), so the address's
325  // mark is dropped too (1202.6182, 1306.1438). Locked as `\thanksref`/`\thankstext` are, so a paper that ships
326  // imsart.sty does not get its `\thanksmark`, which prints `<sup>??</sup>` here.
327  DefMacro!("\\thanksmark[]{}", "", locked => true);
328  // imsart.sty's author markup, defined only inside its `\author@fmt` (:2010-2028 of 2201.11773's copy: `\degs`,
329  // `\fnms`, `\inits`, `\snm`, `\roles`, each printing its argument), which the `\author` above replaces: in force for
330  // the author content (`\lx@author@markup@class`, frontmatter_fields.rs), as a refused class `\author` gives them —
331  // and where no lock records that refusal, a stand-in copy's (`imsart_sty.rs`): "\snm undefined" there (1905.11425,
332  // 2101.10506, 1301.1466, math0608476).
333  RawTeX!(
334    r"\def\lx@author@markup@class{\lx@author@markup@provide\degs\@firstofone\lx@author@markup@provide\fnms\@firstofone\lx@author@markup@provide\inits\@firstofone\lx@author@markup@provide\snm\@firstofone\lx@author@markup@provide\roles\@firstofone}"
335  );
336  // imsart.sty:464-466: `\setattribute[…]{journal}{name}{…}` stores `\journal@name` (`\getattribute` reads it back);
337  // papers blank the journal name with it (1306.0113, 1706.08519).
338  RawTeX!(
339    r"\def\setattribute{\@ifnextchar[\@setattribute{\@setattribute[]}}
340\def\@setattribute[#1]#2#3#4{\expandafter\gdef\csname #2@#3\endcsname{#4}}
341\def\getattribute#1#2{\csname #1@#2\endcsname}"
342  );
343  // imsart.sty's `{keyword}[class=…]` and `\kwd` (:1518-1567 of 1410.4462's copy) are the VTeX classes' shared markup
344  // (`vtex_keyword_markup`, ims_support_sty.rs): every `\kwd` of a block in one list, and a `class=MSC` (`AMS`,
345  // `MSC2010`, `MSC2020`) block the paper's subject classification. Locked, so the raw definitions are refused: imsart.sty
346  // sets its blocks in `\keyword@box` (or, under the journal options' `keywordsasfootnote`, collects them in
347  // `\keyword@toks` through its `\pseudo@keyword`, :1609, which `\begin` reads first), and OmniBus's `\kwd` kept only
348  // the last keyword (KNOWN_PERL_ERRORS #1196): 1410.4462, 2407.13970, 2605.27137 lost every keyword but one and their
349  // MSC codes. The headings are imsart.sty's (:1521-1525), which its own `\setattribute`/`\setpkgattr` lines set again.
350  // Locked only while the class loads — imsart.sty read raw, and a renamed copy of the class read after this binding
351  // with the imsart.sty copy it loads (1401.3408's myimsart.cls, myimsart.sty) — and unlocked at the class's end, as

365/// Records imsart.sty's `\ead[keys]{address}` under its label, as its `\safelabel` writes the aux file (`\ead` :493-498 of
366/// 1301.1466's copy), for `\printead{label}` in an address printed in place ([`imsart_sty_markup`]): the address and
367/// whether it is a URL (the `url` key, or a `://` address, `\checkead@prefix`), which it returns. Past any speculative
368/// run's rollback (`assign_value_past_rollback`), as TeX never takes back what it wrote to a file.
369fn record_ead(keys: Option<&Tokens>, address: &Tokens) -> bool {
370  let keys = keys.map(ToString::to_string).unwrap_or_default();
371  let url = keys
372    .split(',')
373    .any(|key| key.split('=').next().is_some_and(|k| k.trim() == "url"))
374    || address.to_string().contains("://");
375  if let Some(label) = ead_label(&keys) {
376    let kind = if url { "url" } else { "email" };
377    assign_value_past_rollback(
378      &format!("lx_imsart_ead_{kind}_{label}"),
379      Stored::Tokens(address.clone()),
380    );
381    assign_value_past_rollback(
382      &format!("lx_imsart_ead_{label}"),
383      Stored::Tokens(mouth::tokenize_internal(kind)),
384    );
385  }
386  url
387}

389/// Records each `\ead[keys]{address}` written in an `\author`'s argument ([`record_ead`]), at the call.
390fn record_eads(name: &Tokens) {
391  for (keys, address) in ead_calls(name) {
392    record_ead(keys.as_ref(), &address);
393  }
394}

396/// The `label=` of an `\ead`'s keys (`\ead[label=e1]{…}`), when they give one.
397fn ead_label(keys: &str) -> Option<String> {
398  keys.split(',').find_map(|key| {
399    let (k, v) = key.split_once('=')?;
400    (k.trim() == "label" && !v.trim().is_empty()).then(|| v.trim().to_string())
401  })
402}

404/// The group opened at `v[open]` by `open_t` and closed by `close_t` outside braces: its inner tokens and the index after
405/// it.
406fn token_group(v: &[Token], open: usize, open_t: Token, close_t: Token) -> Option<(Tokens, usize)> {
407  if v.get(open) != Some(&open_t) {
408    return None;
409  }
410  let mut braces = usize::from(open_t == T_BEGIN!());
411  for (at, t) in v.iter().enumerate().skip(open + 1) {
412    if *t == T_BEGIN!() {
413      braces += 1;
414    } else if *t == T_END!() {
415      braces = braces.checked_sub(1)?;
416    }
417    if *t == close_t && braces == 0 {
418      return Some((Tokens::new(v[open + 1..at].to_vec()), at + 1));
419    }
420  }
421  None
422}

424/// Each call of `cs` written in `tokens` — an optional `*`, an optional `[…]`, an argument —: its `[…]` and its argument.
425fn calls_of(tokens: &Tokens, cs: Token) -> Vec<(Option<Tokens>, Tokens)> {
426  let v = tokens.unlist_ref();
427  let spaces = |mut at: usize| {
428    while v.get(at) == Some(&T_SPACE!()) {
429      at += 1;
430    }
431    at
432  };
433  let mut calls = Vec::new();
434  let mut at = 0;
435  while at < v.len() {
436    if v[at] == cs {
437      let mut next = spaces(at + 1);
438      if v.get(next) == Some(&T_OTHER!("*")) {
439        next = spaces(next + 1);
440      }
441      let keys = token_group(v, next, T_OTHER!("["), T_OTHER!("]"));
442      if let Some((_, after)) = &keys {
443        next = spaces(*after);
444      }
445      if let Some((argument, after)) = token_group(v, next, T_BEGIN!(), T_END!()) {
446        calls.push((keys.map(|(keys, _)| keys), argument));
447        at = after;
448        continue;
449      }
450    }
451    at += 1;
452  }
453  calls
454}

456/// Each `\ead[keys]{address}` written in `tokens`: its keys and its address.
457fn ead_calls(tokens: &Tokens) -> Vec<(Option<Tokens>, Tokens)> { calls_of(tokens, T_CS!("\\ead")) }
458
459/// The labels an address's `\printead{e1,e2}` calls cite.
460fn printead_labels(address: &Tokens) -> Vec<String> {
461  calls_of(address, T_CS!("\\printead"))
462    .into_iter()
463    .flat_map(|(_, labels)| {
464      labels
465        .to_string()
466        .split(',')
467        .map(|label| label.trim().to_string())
468        .filter(|label| !label.is_empty())
469        .collect::<Vec<_>>()
470    })
471    .collect()
472}

474/// The places (from 1) among the queued authors of those an `\ead` in whose argument gives one of the labels `cited`:
475/// the authors an address block names by printing their e-mails — `\printead{e1}\\ \printead{e2}`, as imsart.sty's
476/// `\printead` reprints the address `\ead` labelled, which the reader of the PDF takes as theirs (1407.2451's two
477/// blocks, Nandy and Maathuis at ETH, Richardson at the University of Washington; 1409.0912, 1504.04814, 1704.07606,
478/// 1810.08264). None when it cites no queued author's label. With them, the places of the authors queued since the last
479/// queued address, the run `annotate=run` gives the block to.
480fn citing_authors(cited: &[String]) -> (Vec<usize>, Vec<usize>) {
481  if cited.is_empty() {
482    return (Vec::new(), Vec::new());
483  }
484  with_value("frontmatter_raw", |queue| {
485    let mut owners = Vec::new();
486    let mut run = Vec::new();
487    if let Some(Stored::FrontmatterRaw(queue)) = queue {
488      let mut place = 0;
489      for entry in queue {
490        let role = entry.1.get("role").map(String::as_str);
491        if entry.0 == "ltx:creator" && role == Some("author") {
492          place += 1;
493          run.push(place);
494          if ead_calls(&entry.2)
495            .iter()
496            .filter_map(|(keys, _)| {
497              ead_label(&keys.as_ref().map(ToString::to_string).unwrap_or_default())
498            })
499            .any(|label| cited.contains(&label))
500          {
501            owners.push(place);
502          }
503        } else if entry.0 == "ltx:contact" && role == Some("address") {
504          run.clear();
505        }
506      }
507    }
508    (owners, run)
509  })
510}

523/// imsart.sty's `\arxiv`, defined over the raw one ([`imsart_sty_markup`]'s note).
524pub fn imsart_arxiv() -> Result<()> {
525  // `\arxiv{id}` (imsart.sty:1186-1189) prints the paper's arXiv identifier as an unmarked title note: the paper's
526  // arXiv pubnote, as jheppub's `\arxivnumber` (jheppub_sty.rs), not an author's note (2201.08502). Defined over the
527  // raw one and not locked: a paper citing with its own `\renewcommand\arxiv` keeps it.
528  DefMacro!(
529    "\\arxiv{}",
530    "\\gdef\\@arxiv{#1}\\lx@add@pubnote[role=arxiv]{#1}"
531  );
532
533  Ok(())
