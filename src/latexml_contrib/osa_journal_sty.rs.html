1//! The Optical Society of America's submission styles for the article class: `opex3.sty` (Optics Express, v1.0
2//! 2003), `ol2.sty` (Optics Letters, v1.0 2006), `osajnl.sty` (the OSA journals, v2.0 2004) and its successor
3//! `osajnl2.sty` (v2.1 2006), loaded as `\documentclass{article}\usepackage{opex3}`. Not in TeX Live and often not
4//! shipped, so the papers stopped at `\address` (759 run-336 papers: 1006.2188, 0901.2165, 0901.2997, 0901.2793); a
5//! shipped copy, read raw, typeset the title, authors and addresses into the text, leaving no title or authors. They
6//! share one document syntax (cited from the copies in 0901.2165 (opex3.sty:43-67), 0901.2997 (ol2.sty:38-89),
7//! 0707.0735 (osajnl.sty:51-90), 0901.2793 (osajnl2.sty:47-86)): `\title`, `\author` (the names, marked `$^1$` by
8//! affiliation), `\address`/`\affil` (the affiliations, marked alike), `\email`, `\homepage`, `\ocis` (`\pacs`), an
9//! `{abstract}`, the reference list as `{thebibliography}` or `{References}`, and the journal abbreviations of the
10//! references; one binding serves them, routing the frontmatter to the frontmatter API. Rust-only: Perl has no OSA
11//! binding.
12use latexml_package::prelude::*;
13
14/// The styles' common definitions; `name` is the style loaded.
15pub(crate) fn osa_journal_style(name: &str) -> Result<()> {
16  // opex3.sty:27-33, ol2.sty:22-26, osajnl.sty:32-39, osajnl2.sty:32-35: the packages loaded (the page geometry and
17  // fonts aside)
18  RequirePackage!("color");
19  RequirePackage!("graphicx");
20  match name {
21    "opex3" => {},
22    // osajnl.sty:32 `\RequirePackage{overcite}`, its citations as superscripts
23    "osajnl" => {
24      RequirePackage!("overcite");
25    },
26    _ => {
27      RequirePackage!("cite");
28    },
29  }
30  // opex3.sty:55 `\newcommand\url[1]{\textcolor{blue}{\underline{#1}}}`: a URL, the link here
31  RequirePackage!("url");
32  // The styles' names and formats: "Fig." (opex3.sty:84, ol2.sty:91, osajnl.sty:93, osajnl2.sty:89), "Table"
33  // (ol2.sty:93, osajnl.sty:95, osajnl2.sty:91), a caption "Fig. 1. text" (`\@makecaption`'s `#1. #2`, opex3.sty:94,
34  // ol2.sty:98, osajnl.sty:104, osajnl2.sty:100), the references headed "References and links" in opex3 (:86),
35  // "References" in the others (ol2.sty:122, osajnl.sty:115, osajnl2.sty:111); headings numbered "1." (`\@seccntformat`,
36  // opex3.sty:156, ol2.sty:159, osajnl.sty:153, osajnl2.sty:149), subsections "1.1" in opex3 (:157-158), "A" in ol2
37  // (:160-161), "1.A" in osajnl and osajnl2 (osajnl.sty:154-155, osajnl2.sty:150-151). (The `{abstract}` label, opex3's
38  // "Abstract:" (:70), is the kernel's abstract name; the copyright line it adds, :71-72, is the publisher's, not
39  // kept.)
40  DefMacro!("\\figurename", "Fig.");
41  DefMacro!("\\tablename", "Table");
42  if name == "opex3" {
43    DefMacro!("\\refname", "References and links");
44  } else {
45    DefMacro!("\\refname", "References");
46  }
47  DefMacro!(
48    "\\format@title@figure{}",
49    "\\lx@tag[][. ]{\\lx@fnum@@{figure}}#1"
50  );
51  DefMacro!(
52    "\\format@title@table{}",
53    "\\lx@tag[][. ]{\\lx@fnum@@{table}}#1"
54  );
55  DefMacro!(
56    "\\format@title@section{}",
57    "\\lx@tag[][.\\space]{\\thesection}#1"
58  );
59  DefMacro!(
60    "\\format@title@subsection{}",
61    "\\lx@tag[][.\\space]{\\thesubsection}#1"
62  );
63  DefMacro!(
64    "\\format@title@subsubsection{}",
65    "\\lx@tag[][.\\space]{\\thesubsubsection}#1"
66  );
67  match name {
68    "opex3" => {
69      RawTeX!(
70        "\\renewcommand\\thesubsection{\\thesection.\\arabic{subsection}}\\renewcommand\\thesubsubsection{\\thesubsection.\\arabic{subsubsection}}"
71      );
72    },
73    "ol2" => {
74      RawTeX!(
75        "\\renewcommand\\thesubsection{\\Alph{subsection}}\\renewcommand\\thesubsubsection{\\arabic{subsubsection}}"
76      );
77    },
78    _ => {
79      RawTeX!(
80        "\\renewcommand\\thesubsection{\\thesection.\\Alph{subsection}}\\renewcommand\\thesubsubsection{\\thesubsection.\\arabic{subsubsection}}"
81      );
82    },
83  }
84  // The kernel's `\title` collects the title, and `\maketitle` (blanked by ol2.sty:87 and osajnl2.sty:84, where
85  // `\title` typesets at once) makes the frontmatter as anywhere.
86  // opex3.sty:46 `\OEauthor` (ol2.sty:41, osajnl2.sty:50 `\OSAJNLauthor`), `\let\author` to it (opex3.sty:63): each
87  // call typesets its names where it stands, so each adds its authors (`\author{D. L. Boiko}\address{…}\email{…}
88  // \author{N. J. Gunther}…`, 0906.4563), split as the article `\author` splits them, their `$^n$` marks requesting the
89  // affiliations so labelled; a footnote symbol in a mark is a mark of its own (`S. Lacour$^{1,2*}$` is marked 1, 2
90  // and *, 0901.2165). `\@author` is the names, as the kernel's `\author` keeps them. Locked as the kernel's is.
91  DefMacro!("\\author{}", sub[(names)] {
92    let names: Tokens = names;
93    let mut call = TokenizeInternal!(r"\def\@author").unlist();
94    call.push(T_BEGIN!());
95    call.extend(names.unlist_ref().iter().copied());
96    call.push(T_END!());
97    // (the group's start, for the contacts placed after it: `\lx@frontmatter@group@start`)
98    call.push(T_CS!("\\lx@frontmatter@group@start"));
99    call.extend(
100      Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(marks_before_commas(
101        symbol_marks_apart(names)
102      ))])
103      .unlist(),
104    );
105    Ok(Tokens::new(call))
106  }, locked => true);
107  // opex3.sty:64-65 `\let\affil\OEaddress \let\address\OEaddress` (ol2.sty:66-67, osajnl2.sty:75-76; ol2.sty:88 and
108  // osajnl2.sty:85 also `\def\affiliation{\address}`): the affiliations, a line each, labelled by the `$^n$` marks the
109  // names carry (`\address{$^1$LESIA, … \\ $^2$School of Physics, …}`, 0901.2165), or the institution of the authors
110  // before it (a second address of theirs too). An email line in it, a line or the piece of one from an inner mark on
111  // (`$^*$Corresponding author: x@y`, 0705.1274, 1006.2188, 1204.5397), is the emails: see `email_line`.
112  DefMacro!("\\address{}", sub[(lines)] {
113    let mut affiliations: Vec<Token> = Vec::new();
114    let mut emails: Vec<Token> = Vec::new();
115    for line in top_level_lines(lines) {
116      // (a piece marked with several labels a line of its own for each: its next piece is not part of it)
117      let mut kept: Vec<Vec<Token>> = vec![Vec::new()];
118      for piece in pieces_at_marks(line) {
119        if let Some(calls) = email_line(&piece)? {
120          emails.extend(calls);
121          continue;
122        }
123        let mut copies = lines_per_label(piece);
124        if copies.len() == 1 {
125          kept.last_mut().into_iter().for_each(|k| k.append(&mut copies[0]));
126        } else {
127          kept.extend(copies);
128          kept.push(Vec::new());
129        }
130      }
131      for line in kept {
132        if line.iter().any(|t| t.get_catcode() != Catcode::SPACE) {
133          if !affiliations.is_empty() {
134            affiliations.push(T_CS!("\\\\"));
135          }
136          affiliations.extend(line);
137        }
138      }
139    }
140    let mut calls = Vec::new();
141    if !affiliations.is_empty() {
142      calls.extend(
143        Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![
144          Some(TokenizeInternal!("annotate=run")),
145          Some(Tokens::new(affiliations))
146        ])
147        .unlist(),
148      );
149    }
150    calls.extend(emails);
151    Ok(Tokens::new(calls))
152  });
153  Let!("\\affil", "\\address");
154  Let!("\\affiliation", "\\address");
155  // opex3.sty:52 `\OEemail`: the email line, text often labelled and marked (`\email{$^*$Corresponding author: …}`,
156  // 0901.2165), so not the verbatim address `\lx@add@email` reads: see `email_line`. Any other is the block's email,
157  // placed by the kernel.
158  DefMacro!("\\email{}", sub[(line)] {
159    let line: Tokens = line;
160    Ok(match email_line(line.unlist_ref())? {
161      Some(calls) => Tokens::new(calls),
162      None => Invocation!(T_CS!("\\lx@add@contact@placed"), vec![
163        Some(TokenizeInternal!("role=email")),
164        Some(Tokens!()),
165        Some(line)
166      ]),
167    })
168  });
169  // opex3.sty:57 `\homepage`: the group's or block's web page, placed as an email is (1004.2645's institute page was
170  // the last author's)
171  DefMacro!(
172    "\\homepage{}",
173    "\\lx@add@contact@placed[role=url]{}{\\url{#1}}"
174  );
175  // opex3.sty:60, :67 `\let\pacs\ocis`: the paper's OCIS (Optics Classification and Indexing Scheme) codes
176  DefMacro!(
177    "\\ocis{}",
178    "\\lx@add@classification[scheme=ocis,name={OCIS codes:~}]{#1}"
179  );
180  Let!("\\pacs", "\\ocis");
181  // opex3.sty:75-78 `{abstract*}`, the abstract of a paper exempt from the copyright line `{abstract}` adds
182  DefMacro!(T_CS!("\\begin{abstract*}"), None, "\\begin{abstract}");
183  DefMacro!(T_CS!("\\end{abstract*}"), None, "\\end{abstract}");
184  // The reference lists the styles define with both ends: `{References}` (opex3.sty:148, :151), `{OEReflinks}`
185  // (:147, :150), `{OSAJNLReferences}` (ol2.sty:151, :154; osajnl2.sty:140, :144). (`\OSAJNLReflinks` is closed by an
186  // `\endOSAJNLlinks` and `\references` by nothing, ol2.sty:152, :155, osajnl2.sty:143: neither is an environment.)
187  for env in ["References", "OEReflinks", "OSAJNLReferences"] {
188    DefMacro!(
189      T_CS!(&format!("\\begin{{{env}}}")),
190      None,
191      "\\begin{thebibliography}{}"
192    );
193    DefMacro!(
194      T_CS!(&format!("\\end{{{env}}}")),
195      None,
196      "\\end{thebibliography}"
197    );
198  }
199  // ol2.sty:89, osajnl.sty:90, osajnl2.sty:86 `\def\onlinecite{\citeonline}` (cite.sty's citation without brackets)
200  if name != "opex3" {
201    Let!("\\onlinecite", "\\citeonline");
202  }
203  // opex3.sty:237-273 (ol2.sty:240-276, osajnl2.sty:230-266 alike): the journal abbreviations of the references, the
204  // ones every shipped version has. The style `\def`s them, so a paper's own `\def` or `\renewcommand` replaces one
205  // (60 of 751 OSA papers) and a `\newcommand` keeps the style's, as in pdflatex (where it is an error too). The later
206  // opex3.sty's additions (`\boe`, `\ome`, `\aop`, `\as`, `\jocn`, `\opn`, `\pr`; 1406.6489 :245-280) are left
207  // out: no OSA paper cites them, and papers define those names for themselves (0909.4968 `\newcommand{\as}`,
208  // 1104.3859 `\as` for arcseconds, 1104.0806 `\pr` for Phys. Rev.), which a binding's earlier definition would hide.
209  for (cs, journal) in [
210    ("\\opex", " Opt.\\ Express "),
211    ("\\ao", " Appl.\\  Opt.\\ "),
212    ("\\ap", " Appl.\\  Phys.\\ "),
213    ("\\apa", " Appl.\\  Phys.\\ A "),
214    ("\\apb", " Appl.\\  Phys.\\ B "),
215    ("\\apl", " Appl.\\ Phys.\\ Lett.\\ "),
216    ("\\apj", " Astrophys.\\ J.\\ "),
217    ("\\bell", " Bell Syst.\\ Tech.\\ J.\\ "),
218    ("\\jqe", " IEEE J.\\ Quantum Electron.\\ "),
219    (
220      "\\assp",
221      " IEEE Trans.\\ Acoust.\\ Speech Signal Process.\\ ",
222    ),
223    ("\\aprop", " IEEE Trans.\\ Antennas Propag.\\ "),
224    ("\\mtt", " IEEE Trans.\\ Microwave Theory Tech.\\ "),
225    ("\\iovs", " Invest.\\ Ophthalmol.\\ Visual\\ Sci.\\ "),
226    ("\\jcp", " J.\\ Chem.\\ Phys.\\ "),
227    ("\\jmo", " J.\\ Mod.\\ Opt.\\ "),
228    ("\\jon", "J.\\ Opt.\\ Netw.\\ "),
229    ("\\josa", " J.\\ Opt.\\ Soc.\\ Am.\\ "),
230    ("\\josaa", " J.\\ Opt.\\ Soc.\\ Am.\\ A "),
231    ("\\josab", " J.\\ Opt.\\ Soc.\\ Am.\\ B "),
232    ("\\jpp", " J.\\ Phys.\\ (Paris) "),
233    ("\\nat", " Nature (London) "),
234    ("\\oc", " Opt.\\ Commun.\\ "),
235    ("\\ol", " Opt.\\ Lett.\\ "),
236    ("\\pl", " Phys.\\ Lett.\\ "),
237    ("\\pra", " Phys.\\ Rev.\\ A "),
238    ("\\prb", " Phys.\\ Rev.\\ B "),
239    ("\\prc", " Phys.\\ Rev.\\ C "),

241    ("\\pre", " Phys.\\ Rev.\\ E "),
242    ("\\prl", " Phys.\\ Rev.\\ Lett.\\ "),
243    ("\\rmp", " Rev.\\ Mod.\\ Phys.\\ "),

256/// Whether `t` is a footnote symbol as a mark writes it: `*`, `\ast`, the daggers, `\S`, `\P`, `\star`, `\#`.
257fn is_symbol_mark(t: &Token) -> bool {
258  *t == T_OTHER!("*")
259    || [
260      "\\ast",
261      "\\dag",
262      "\\dagger",
263      "\\ddag",
264      "\\ddagger",
265      "\\S",
266      "\\P",
267      "\\star",
268      "\\#",
269      "\\sharp",
270    ]
271    .iter()
272    .any(|cs| *t == T_CS!(*cs))
273}

275/// `names` with each mark written after the comma that ends its name moved before it: `E. Hirose,$ ^{1,4,*} $
276/// K. Kawabe,$ ^{2} $` (0909.0010, 1501.05707, 0812.1160) marks Hirose and Kawabe, where the names split at the comma
277/// gave each mark to the name after it; a bare symbol so (`Leibrandt,* Michael J. Thorpe`, 1104.1413) as a superscript.
278fn marks_before_commas(names: Tokens) -> Tokens {
279  let tokens = names.unlist();
280  let mut out: Vec<Token> = Vec::with_capacity(tokens.len());
281  let mut depth = 0usize;
282  let mut i = 0;
283  while i < tokens.len() {
284    let t = tokens[i];
285    match t.get_catcode() {
286      Catcode::BEGIN => depth += 1,
287      Catcode::END => depth = depth.saturating_sub(1),
288      _ => {},
289    }
290    if depth == 0 && t == T_OTHER!(",") {
291      let at = skip_spaces_at(&tokens, i + 1);
292      // a math mark `$ ^{…} $`, its spaces in it
293      if tokens
294        .get(at)
295        .is_some_and(|m| m.get_catcode() == Catcode::MATH)
296        && tokens
297          .get(skip_spaces_at(&tokens, at + 1))
298          .is_some_and(|m| m.get_catcode() == Catcode::SUPER)
299        && let Some(close) = tokens[at + 1..]
300          .iter()
301          .position(|m| m.get_catcode() == Catcode::MATH)
302          .map(|n| at + 1 + n)
303      {
304        out.extend_from_slice(&tokens[at..=close]);
305        out.push(t);
306        i = close + 1;
307        continue;
308      }
309      // a bare footnote symbol (`David R. Leibrandt,* Michael J. Thorpe`, 1104.1413), as the superscript it marks
310      let symbols = tokens[at..]
311        .iter()
312        .take_while(|m| is_symbol_mark(m) && **m != T_CS!("\\#"))
313        .count();
314      if symbols > 0 {
315        out.extend([T_MATH!(), T_SUPER!(), T_BEGIN!()]);
316        out.extend_from_slice(&tokens[at..at + symbols]);
317        out.extend([T_END!(), T_MATH!(), t]);
318        i = at + symbols;
319        continue;
320      }
321    }
322    out.push(t);
323    i += 1;
324  }
325  Tokens::new(out)
326}

328/// `names` with each footnote symbol written into a numbered superscript mark made a mark of its own: `$^{1,2*}$` as
329/// `$^{1,2,*}$`, `$^{*1}$` as `$^{*,1}$`, so that it requests affiliations 1 and 2 and the line marked `*` (0901.2165,
330/// 1011.1336, 1208.1364, 1310.5176).
331fn symbol_marks_apart(names: Tokens) -> Tokens {
332  let tokens = names.unlist();
333  let mut out = Vec::with_capacity(tokens.len() + 4);
334  let mut i = 0;
335  while i < tokens.len() {
336    let t = tokens[i];
337    out.push(t);
338    i += 1;
339    let is_mark = t.get_catcode() == Catcode::SUPER || t == T_CS!("\\textsuperscript");
340    if !is_mark
341      || tokens
342        .get(i)
343        .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
344    {
345      continue;
346    }
347    // the mark's group, its symbols set apart from what comes before them
348    let mut depth = 0usize;
349    while i < tokens.len() {
350      let u = tokens[i];
351      match u.get_catcode() {
352        Catcode::BEGIN => depth += 1,
353        Catcode::END => depth -= 1,
354        _ => {},
355      }
356      if depth == 1
357        && is_symbol_mark(&u)
358        && out.last().is_some_and(|last| {
359          *last != T_OTHER!(",") && last.get_catcode() != Catcode::BEGIN && !is_symbol_mark(last)
360        })
361      {
362        out.push(T_OTHER!(","));
363      }
364      out.push(u);
365      i += 1;
366      // and from what comes after them (`$^{*1}$`, `$^{\ast 2}$`: 1310.5176, 1506.01336)
367      if depth == 1 && is_symbol_mark(&u) {
368        let next = tokens[i..]
369          .iter()
370          .find(|t| t.get_catcode() != Catcode::SPACE);
371        if next.is_some_and(|n| {
372          *n != T_OTHER!(",") && n.get_catcode() != Catcode::END && !is_symbol_mark(n)
373        }) {
374          out.push(T_OTHER!(","));
375        }
376      }
377      if depth == 0 {
378        break;
379      }
380    }
381  }
382  Tokens::new(out)
383}

385/// An affiliation line marked with several labels (`$^{1,2}$Department …`, 1406.0824) as one line for each: an entry
386/// takes one label (Perl's `\lx@set@frontmatter@label`, Base_Utility.pool.ltxml:568, as the kernel's), so the
387/// author marked 2 had no affiliation.
388fn lines_per_label(line: Vec<Token>) -> Vec<Vec<Token>> {
389  let start = skip_spaces_at(&line, 0);
390  let Some((operand, end)) =
391    mark_at(&line, start).filter(|_| line[start].get_catcode() == Catcode::MATH)
392  else {
393    return vec![line];
394  };
395  let labels: Vec<&[Token]> = operand.split(|t| *t == T_OTHER!(",")).collect();
396  if labels.len() < 2
397    || labels
398      .iter()
399      .any(|label| label.iter().all(|t| t.get_catcode() == Catcode::SPACE))
400  {
401    return vec![line];
402  }
403  labels
404    .into_iter()
405    .map(|label| {
406      let mut copy = vec![T_MATH!(), T_SUPER!(), T_BEGIN!()];
407      copy.extend_from_slice(label);
408      copy.extend([T_END!(), T_MATH!()]);
409      copy.extend_from_slice(&line[end..]);
410      copy
411    })
412    .collect()
413}

415/// The lines of `tokens` at its top-level `\\`.
416fn top_level_lines(tokens: Tokens) -> Vec<Vec<Token>> {
417  let mut lines = vec![Vec::new()];
418  let mut depth = 0usize;
419  for t in tokens.unlist() {
420    match t.get_catcode() {
421      Catcode::BEGIN => depth += 1,
422      Catcode::END => depth = depth.saturating_sub(1),
423      _ => {},
424    }
425    // (a blank line between the lines too: 1109.5493)
426    if depth == 0 && (t == T_CS!("\\\\") || t == T_CS!("\\par")) {
427      lines.push(Vec::new());
428    } else if let Some(line) = lines.last_mut() {
429      line.push(t);
430    }
431  }
432  lines
433}

435/// Whether `t` is the character `c`, whatever its category (babel's French makes `:` and `;` active: 1210.1049).
436fn is_char(t: &Token, c: &str) -> bool {
437  matches!(
438    t.get_catcode(),
439    Catcode::OTHER | Catcode::ACTIVE | Catcode::LETTER
440  ) && t.with_str(|s| s == c)
441}

443/// Whether the tokens hold an `@`.
444fn has_at(tokens: &[Token]) -> bool { tokens.iter().any(|t| t.with_str(|s| s == "@")) }

446/// The first index from `i` that is not a space.
447fn skip_spaces_at(tokens: &[Token], mut i: usize) -> usize {
448  while tokens
449    .get(i)
450    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
451  {
452    i += 1;
453  }
454  i
455}

457/// The index after the group opening at `open`, or after the one token there.
458fn after_argument(tokens: &[Token], open: usize) -> Option<usize> {
459  if tokens.get(open)?.get_catcode() != Catcode::BEGIN {
460    return Some(open + 1);
461  }
462  let mut depth = 0usize;
463  for (j, t) in tokens.iter().enumerate().skip(open) {
464    match t.get_catcode() {
465      Catcode::BEGIN => depth += 1,
466      Catcode::END => {
467        depth -= 1;
468        if depth == 0 {
469          return Some(j + 1);
470        }
471      },
472      _ => {},
473    }
474  }
475  None
476}

478/// The operand of the mark at `i` in `tokens`, and the index after it: `$^…$`, `^…`,
479/// `\textsuperscript{…}`, a footnote symbol in math (`$*$`, 1412.3226) or bare (`*peter.banzer@…`, `**3794…`,
480/// 1004.2645, 2001.05892).
481fn mark_at(tokens: &[Token], i: usize) -> Option<(Vec<Token>, usize)> {
482  let t = tokens.get(i)?;
483  let (operand, end): (Vec<Token>, usize) = if t.get_catcode() == Catcode::MATH {
484    let close = i
485      + 1
486      + tokens[i + 1..]
487        .iter()
488        .position(|t| t.get_catcode() == Catcode::MATH)?;
489    let inner = &tokens[i + 1..close];
490    match inner.first() {
491      Some(sup) if sup.get_catcode() == Catcode::SUPER => {
492        let operand = &inner[1..];
493        let operand = match operand {
494          [open, rest @ .., close]
495            if open.get_catcode() == Catcode::BEGIN && close.get_catcode() == Catcode::END =>
496          {
497            rest
498          },
499          _ => operand,
500        };
501        (operand.to_vec(), close + 1)
502      },
503      _ if !inner.is_empty() && inner.iter().all(is_symbol_mark) => (inner.to_vec(), close + 1),
504      _ => return None,
505    }
506  } else if t.get_catcode() == Catcode::SUPER || *t == T_CS!("\\textsuperscript") {
507    let open = skip_spaces_at(tokens, i + 1);
508    let end = after_argument(tokens, open)?;
509    let operand = if tokens[open].get_catcode() == Catcode::BEGIN {
510      &tokens[open + 1..end - 1]
511    } else {
512      &tokens[open..end]
513    };
514    (operand.to_vec(), end)
515  } else if is_symbol_mark(t) && *t != T_CS!("\\#") {
516    // (bare, `\#` is the address's own character: `\email{\#able@x.org}`)
517    let n = tokens[i..].iter().take_while(|t| is_symbol_mark(t)).count();
518    (tokens[i..i + n].to_vec(), i + n)
519  } else {
520    return None;
521  };
522  Some((operand, end))
523}

532/// One address of an email line: its mark, if any, and the address.
533struct Address {
534  mark:    Option<Vec<Token>>,
535  address: Vec<Token>,
536}

538/// An email line (opex3.sty:52 `\OEemail`; or a line of an `\address`): an optional mark, an optional short label (a
539/// few words ending in `:`, no command or math: `Corresponding author:`, `e-mail:`), then addresses separated by
540/// commas or semicolons, each perhaps marked (`*a@x, **b@y`). `None` for anything else — an affiliation naming an
541/// address in its text (`… Milano, Italy (stefano.longhi@polimi.it)`, 1411.7144) is no email line.
542fn email_line_parts(line: &[Token]) -> Option<EmailLine> {
543  if !has_at(line) {
544    return None;
545  }
546  let start = skip_spaces_at(line, 0);
547  let (line_mark, rest) = match mark_at(line, start) {
548    Some((mark, end)) => (Some(mark), end),
549    None => (None, start),
550  };
551  // the words before the first one naming an address are the label
552  let mut words: Vec<(usize, usize)> = Vec::new();
553  let mut i = skip_spaces_at(line, rest);
554  while i < line.len() {
555    let mut j = i;
556    while j < line.len() && line[j].get_catcode() != Catcode::SPACE {
557      j = if line[j].get_catcode() == Catcode::BEGIN {
558        after_argument(line, j)?
559      } else {
560        j + 1
561      };
562    }
563    words.push((i, j));
564    i = skip_spaces_at(line, j);
565  }
566  let first = words.iter().position(|&(a, b)| has_at(&line[a..b]))?;
567  let label_words = &words[..first];
568  let label_ok = label_words.len() <= 4
569    && label_words.iter().all(|&(a, b)| {
570      line[a..b].iter().all(|t| {
571        (matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER) || is_char(t, ":"))
572          && t.with_str(|s| s != "(")
573      })
574    })
575    && label_words
576      .last()
577      .is_none_or(|&(_, b)| is_char(&line[b - 1], ":"));
578  if !label_ok {
579    return None;
580  }
581  let mut label: Vec<Token> = Vec::new();
582  for &(a, b) in label_words {
583    label.extend_from_slice(&line[a..b]);
584    label.push(T_SPACE!());
585  }
586  // the addresses, at the top-level commas and semicolons
587  let tail = &line[words[first].0..];
588  let mut items: Vec<Vec<Token>> = vec![Vec::new()];
589  let mut depth = 0usize;
590  for t in tail {
591    match t.get_catcode() {
592      Catcode::BEGIN => depth += 1,
593      Catcode::END => depth = depth.saturating_sub(1),
594      _ => {},
595    }
596    if depth == 0 && (is_char(t, ",") || is_char(t, ";")) {
597      items.push(Vec::new());
598    } else if let Some(item) = items.last_mut() {
599      item.push(*t);
600    }
601  }
602  let mut addresses = Vec::new();
603  for item in items {
604    let Some(address) = address_item(&item) else {
605      // a bare "and" between the last two (`$^2$a@x, and $^3$b@y`, 1406.0824)
606      if is_and_word(&item) {
607        continue;
608      }
609      return None;
610    };
611    addresses.push(address);
612  }
613  if addresses.is_empty() {
614    return None;
615  }
616  Some(EmailLine {
617    mark: line_mark,
618    label,
619    addresses,
620  })
621}

623/// Whether `tokens` is the word "and", or only spaces.
624fn is_and_word(tokens: &[Token]) -> bool {
625  let word: String = tokens
626    .iter()
627    .filter(|t| t.get_catcode() != Catcode::SPACE)
628    .map(|t| t.to_string())
629    .collect();
630  word.is_empty() || word == "and"
631}

633/// The declarations a style puts around an address (`{\color{blue}$^*$x@y}`, 1303.3906; `\it`): not its text.
634fn skip_declarations(tokens: &[Token], mut i: usize) -> usize {
635  loop {
636    i = skip_spaces_at(tokens, i);
637    match tokens.get(i) {
638      Some(t) if *t == T_CS!("\\color") => {
639        let Some(end) = after_argument(tokens, skip_spaces_at(tokens, i + 1)) else {
640          return i;
641        };
642        i = end;
643      },
644      Some(t)
645        if [
646          "\\it",
647          "\\bf",
648          "\\rm",
649          "\\sl",
650          "\\tt",
651          "\\sf",
652          "\\em",
653          "\\small",
654          "\\footnotesize",
655        ]
656        .iter()
657        .any(|cs| *t == T_CS!(*cs)) =>
658      {
659        i += 1;
660      },
661      _ => return i,
662    }
663  }
664}

666/// One address of an email line ([`email_line_parts`]): a mark before it (`$^*$`, `*`, `\textit{$^3$}` of 1306.0672),
667/// or after it (`x@y$^\dag$`, 1003.2212), the address itself, one word, and a parenthesis after it (`x@y (A. Able)`),
668/// set aside with the item's enclosing group and declarations.
669fn address_item(item: &[Token]) -> Option<Address> {
670  let mut tokens: Vec<Token> = item.to_vec();
671  // the group the whole item is
672  loop {
673    let start = skip_spaces_at(&tokens, 0);
674    let end = tokens
675      .iter()
676      .rposition(|t| t.get_catcode() != Catcode::SPACE)?
677      + 1;
678    if tokens[start].get_catcode() == Catcode::BEGIN && after_argument(&tokens, start) == Some(end)
679    {
680      tokens = tokens[start + 1..end - 1].to_vec();
681    } else {
682      break;
683    }
684  }
685  let at = skip_declarations(&tokens, 0);
686  let (mut mark, from) = match mark_at(&tokens, at) {
687    Some((mark, end)) => (Some(mark), skip_spaces_at(&tokens, end)),
688    // a mark inside a font command (`\textit{$^3$}yinjuan@…`)
689    None => match tokens.get(at) {
690      // (only with its braced argument: a bare `\textasteriskcentered able@x.edu` is the address's text, and a
691      // command without a brace after it sliced backwards, a panic)
692      Some(t)
693        if t.get_catcode() == Catcode::CS
694          && t.with_str(|s| s.starts_with("\\text") || s == "\\emph")
695          && tokens
696            .get(skip_spaces_at(&tokens, at + 1))
697            .is_some_and(|t| t.get_catcode() == Catcode::BEGIN) =>
698      {
699        let open = skip_spaces_at(&tokens, at + 1);
700        let end = after_argument(&tokens, open)?;
701        let inner = &tokens[open + 1..end.saturating_sub(1)];
702        match mark_at(inner, skip_spaces_at(inner, 0)) {
703          Some((operand, e)) if skip_spaces_at(inner, e) == inner.len() => {
704            (Some(operand), skip_spaces_at(&tokens, end))
705          },
706          _ => (None, at),
707        }
708      },
709      _ => (None, at),
710    },
711  };
712  let mut address = tokens[from..].to_vec();
713  // a parenthesis after the address, naming its owner (`x@y (A. Able)`): no part of the address the contact links to
714  if let Some(space) = address
715    .iter()
716    .position(|t| t.get_catcode() == Catcode::SPACE)
717    && let Some(open) = address[space..]
718      .iter()
719      .position(|t| t.get_catcode() != Catcode::SPACE)
720      .map(|n| space + n)
721    && address[open] == T_OTHER!("(")
722    && address
723      .iter()
724      .rposition(|t| t.get_catcode() != Catcode::SPACE)
725      .is_some_and(|last| address[last] == T_OTHER!(")"))
726  {
727    address.truncate(space);
728  }
729  while address
730    .last()
731    .is_some_and(|t| t.get_catcode() == Catcode::SPACE || *t == T_OTHER!("."))
732  {
733    address.pop();
734  }
735  // a mark after the address (`\email{hyunchul.nha@qatar.tamu.edu$^\dag$}`, 1003.2212)
736  if mark.is_none()
737    && let Some(open) = address
738      .iter()
739      .position(|t| t.get_catcode() == Catcode::MATH)
740    && let Some((operand, end)) = mark_at(&address, open)
741    && end == address.len()
742  {
743    mark = Some(operand);
744    address.truncate(open);
745  }
746  // an address is one word that names one
747  if !has_at(&address) || address.iter().any(|t| t.get_catcode() == Catcode::SPACE) {
748    return None;
749  }
750  Some(Address { mark, address })
751}

753/// The calls of an email line ([`email_line_parts`]), the label each contact's name, each address placed by the kernel
754/// (`\lx@add@contact@placed`): with the author carrying its mark, else the one its local part spells, else the group's
755/// only author, else the block's.
756fn email_line(line: &[Token]) -> Result<Option<Vec<Token>>> {
757  let Some(EmailLine {
758    mark: line_mark,
759    label,
760    addresses,
761  }) = email_line_parts(line)
762  else {
763    return Ok(None);
764  };
765  let mut calls = Vec::new();
766  for Address { mark, address } in addresses {
767    let mut keyvals = TokenizeInternal!("role=email").unlist();
768    if !label.is_empty() {
769      keyvals.extend(TokenizeInternal!(",name=").unlist());
770      keyvals.push(T_BEGIN!());
771      keyvals.extend(label.iter().copied());
772      keyvals.push(T_END!());
773    }
774    let mark = mark.or_else(|| line_mark.clone()).unwrap_or_default();
775    calls.extend(
776      Invocation!(T_CS!("\\lx@add@contact@placed"), vec![
777        Some(Tokens::new(keyvals)),
778        Some(Tokens::new(mark)),
779        Some(Tokens::new(address))
780      ])
781      .unlist(),
782    );
783  }
784  Ok(Some(calls))
785}

787/// The pieces of an `\address` line at its inner marks: `$^1$Institute … Switzerland $^*$Corresponding author: x@y`
788/// (1204.5397) is an affiliation and an email line.
789fn pieces_at_marks(line: Vec<Token>) -> Vec<Vec<Token>> {
790  let mut pieces: Vec<Vec<Token>> = vec![Vec::new()];
791  let mut depth = 0usize;
792  let mut i = 0;
793  while i < line.len() {
794    let t = line[i];
795    let starts_mark = depth == 0
796      && (t.get_catcode() == Catcode::MATH
797        && line
798          .get(i + 1)
799          .is_some_and(|n| n.get_catcode() == Catcode::SUPER)
800        || t.get_catcode() == Catcode::SUPER
801          && (i == 0 || line[i - 1].get_catcode() != Catcode::MATH)
802        || t == T_CS!("\\textsuperscript"));
803    // (a superscript glued to the letter or digit before it is no mark: `5$^{th}$ floor`)
804    let glued = i > 0 && matches!(line[i - 1].get_catcode(), Catcode::LETTER | Catcode::OTHER);
805    if starts_mark
806      && !glued
807      && pieces
808        .last()
809        .is_some_and(|p| p.iter().any(|t| t.get_catcode() != Catcode::SPACE))
810    {
811      pieces.push(Vec::new());
812    }
813    match t.get_catcode() {
814      Catcode::BEGIN => depth += 1,
815      Catcode::END => depth = depth.saturating_sub(1),
816      _ => {},
817    }
818    if let Some(piece) = pieces.last_mut() {
819      piece.push(t);
820    }
821    i += 1;
822  }
823  pieces
824}
