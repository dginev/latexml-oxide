333/// atbegshi's user commands with the shipout they hook into absent. What pdflatex prints on a page — a page's
334/// upper-left picture (`\AtBeginShipoutUpperLeft…`) or its background or foreground box (`\AtBeginShipoutAddToBox…`) set
335/// in the material of `\AtBeginShipout` or `…Next`, and the first page's material (`\AtBeginShipoutFirst`) — is the
336/// paper's page notice (2309.06611's IEEE copyright, 2403.15239's conference line, 2302.00362's "Preprint of the paper
337/// which appeared in: …") and is kept once as a publication note; the rest of the hook material is dropped unrun. Used
338/// outside a shipout hook — directly, or in `\AtBeginShipoutFirst`'s own material — the picture and box commands add to
339/// the shipout box when no page is being shipped (latex.ltx:20069, :20200), so pdflatex prints none of it and they do
340/// nothing. `\AtBeginShipoutBox` is the kernel's `\ShipoutBox`, `\AtBeginShipoutInit` and `\AtBeginShipoutDiscard` do
341/// nothing, the shipout box's dimensions are 0pt.
342fn define_atbegshi_hooks() -> Result<()> {
343  def_macro_noop("\\AtBeginShipoutUpperLeft{}")?;
344  def_macro_noop("\\AtBeginShipoutUpperLeftForeground{}")?;
345  def_macro_noop("\\AtBeginShipoutAddToBox{}")?;
346  def_macro_noop("\\AtBeginShipoutAddToBoxForeground{}")?;
347  DefMacro!("\\AtBeginShipout{}", sub[(material)] { Ok(Tokens::new(shipout_notes(&material, false).0)) });
348  DefMacro!("\\AtBeginShipoutNext{}", sub[(material)] { Ok(Tokens::new(shipout_notes(&material, false).0)) });
349  // (the first page's material is printed itself, beside the notes its own hooks hold)
350  DefMacro!("\\AtBeginShipoutFirst{}", sub[(material)] {
351    let (mut notes, rest) = shipout_notes(&material, true);
352    notes.push(T_CS!("\\lx@arxiv@shipout@note"));
353    notes.push(T_BEGIN!());
354    notes.extend(rest);
355    notes.push(T_END!());
356    Ok(Tokens::new(notes))
357  });
358  // The note carries the words, not the page geometry: the material is set in a group where `\put(x,y){…}` keeps only
359  // its `{…}` (atbegshi's `picture(0,0)` is not set up), and material with a picture (a `tikzpicture` overlay, a
360  // `picture`) gives its text alone. Material printing no text (a rule, an image, 1612.04433's tikz highlighter) and
361  // material already noted (a hook registered again, or once per page) make no note.
362  DefPrimitive!("\\lx@arxiv@shipout@note{}", sub[(material)] {
363    let seen = s!("lx@arxiv@shipout@note:{}", material.to_string());
364    if lookup_bool(&seen) {
365      return Ok(Vec::new());
366    }
367    AssignValue!(&seen => true, Scope::Global);
368    let mut body = TokenizeInternal!(r"\hbox{\unitlength=1pt\relax\let\put\lx@arxiv@shipout@put").unlist();
369    body.extend(material.unlist_ref().iter().cloned());
370    body.push(T_END!());
371    let stuff = digest(Tokens::new(body))?;
372    let mut text = String::new();
373    ink_text_printed(&stuff, &mut text);
374    let words = text.split_whitespace().collect::<Vec<_>>().join(" ");
375    if !words.chars().any(char::is_alphanumeric) {
376      return Ok(Vec::new());
377    }
378    let source = material.to_string();
379    let content = if ["\\tikz", "\\pgfpicture", "picture}"].iter().any(|p| source.contains(p)) {
380      ExplodeText!(words)
381    } else {
382      let n = lookup_int("lx_arxiv_shipout_notes") + 1;
383      assign_value("lx_arxiv_shipout_notes", Stored::Int(n), Some(Scope::Global));
384      assign_value(&s!("lx_arxiv_shipout_note_{n}"), Stored::Digested(stuff), Some(Scope::Global));
385      mouth::tokenize_internal(TeXString::assembled(s!("\\lx@arxiv@shipout@box{{{n}}}"))).unlist()
386    };
387    let mut note = TokenizeInternal!(r"\lx@add@pubnote[role=note,name={}]{").unlist();
388    note.extend(content);
389    note.push(T_END!());
390    unread(Tokens::new(note));
391    Ok(Vec::new())
392  });
393  DefMacro!(
394    "\\lx@arxiv@shipout@put SkipSpaces Match:( Until:, Until:) {}",
395    "#4"
396  );
397  // (the `\hbox` `\lx@arxiv@shipout@note` digested, its list given where the note is digested)
398  DefPrimitive!("\\lx@arxiv@shipout@box{}", sub[(n)] {
399    let Some(Stored::Digested(stuff)) = lookup_value(&s!("lx_arxiv_shipout_note_{}", n.to_string().trim())) else {
400      return Ok(Vec::new());
401    };
402    let items = stuff.unlist();
403    Ok(match items.as_slice() {
404      [hbox] => match hbox.data() {
405        DigestedData::Whatsit(w) => w.borrow().get_arg(2).map(Digested::unlist).unwrap_or_default(),
406        _ => items,
407      },
408      _ => items,
409    })
410  });
411  Let!("\\AtBeginShipoutInit", "\\@empty");
412  Let!("\\AtBeginShipoutDiscard", "\\@empty");
413  if !IsDefined!(&T_CS!("\\AtBeginShipoutBox")) {
414    if IsDefined!(&T_CS!("\\ShipoutBox")) {
415      Let!("\\AtBeginShipoutBox", "\\ShipoutBox");
416    } else {
417      raw_tex("\\newbox\\AtBeginShipoutBox")?;
418    }
419  }
420  if IsDefined!(&T_CS!("\\shipout")) {
421    Let!("\\AtBeginShipoutOriginalShipout", "\\shipout");
422  }
423  DefMacro!("\\ShipoutBoxHeight", "0pt");
424  DefMacro!("\\ShipoutBoxDepth", "0pt");
425  DefMacro!("\\ShipoutBoxWidth", "0pt");
426  Ok(())
427}

429/// A shipout hook's material split: the page notes it holds — a `\lx@arxiv@shipout@note{…}` for the argument of each
430/// `\AtBeginShipoutUpperLeft…` / `\AtBeginShipoutAddToBox…` in it at any depth, an `\AtBeginShipoutFirst{…}` for each of
431/// those — and the rest of it, every atbegshi hook taken out with its argument (so nothing is noted twice). In
432/// `\AtBeginShipoutFirst`'s own material (`first_page`), a picture or box command outside a nested hook prints nothing
433/// in pdflatex (see [`define_atbegshi_hooks`]) and is dropped with its argument.
434fn shipout_notes(material: &Tokens, first_page: bool) -> (Vec<Token>, Vec<Token>) {
435  let noted = [
436    T_CS!("\\AtBeginShipoutUpperLeft"),
437    T_CS!("\\AtBeginShipoutUpperLeftForeground"),
438    T_CS!("\\AtBeginShipoutAddToBox"),
439    T_CS!("\\AtBeginShipoutAddToBoxForeground"),
440  ];
441  let first = T_CS!("\\AtBeginShipoutFirst");
442  let hooks = [T_CS!("\\AtBeginShipout"), T_CS!("\\AtBeginShipoutNext")];
443  let tokens = material.unlist_ref();
444  // (the end of the argument read from `i`: a balanced group or one token, spaces before it skipped)
445  let argument = |mut i: usize| -> (usize, usize) {
446    while i < tokens.len() && tokens[i].get_catcode() == Catcode::SPACE {
447      i += 1;
448    }
449    let start = i;
450    if i < tokens.len() && tokens[i].get_catcode() == Catcode::BEGIN {
451      let mut level = 0;
452      while i < tokens.len() {
453        match tokens[i].get_catcode() {
454          Catcode::BEGIN => level += 1,
455          Catcode::END => level -= 1,
456          _ => {},
457        }
458        i += 1;
459        if level == 0 {
460          break;
461        }
462      }
463    } else if i < tokens.len() {
464      i += 1;
465    }
466    (start, i)
467  };
468  let (mut notes, mut rest) = (Vec::new(), Vec::new());
469  // (the hooks' arguments still open in `rest`: what they hold is dropped from it, their notes kept)
470  let mut dropped_until = 0;
471  let mut i = 0;
472  while i < tokens.len() {
473    let token = &tokens[i];
474    if noted.contains(token) || *token == first {
475      let (start, end) = argument(i + 1);
476      if !(first_page && *token != first && i >= dropped_until) {
477        notes.push(if *token == first {
478          first
479        } else {
480          T_CS!("\\lx@arxiv@shipout@note")
481        });
482        notes.extend_from_slice(&tokens[start..end]);
483      }
484      i = end;
485      continue;
486    }
487    if hooks.contains(token) {
488      dropped_until = dropped_until.max(argument(i + 1).1);
489    } else if i >= dropped_until {
490      rest.push(*token);
491    }
492    i += 1;
493  }
494  (notes, rest)
495}
