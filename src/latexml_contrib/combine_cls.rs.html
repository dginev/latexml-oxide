1//! Stub for combine.cls (multi-document combiner).
2//!
3//! combine.cls (TL `latex/combine/combine.cls`) is a class for
4//! merging multiple LaTeX documents into a single PDF. It defines
5//! `{papers}`, `{document*}`, etc. envs that wrap sub-document
6//! content. Our HTML/XML output is single-document; just render the
7//! body transparently, in the class combine's options name. Witness 2310.14794.
8use latexml_package::prelude::*;
9
10LoadDefinitions!({
11  // combine.cls:29 `\c@lclass`, the class each document is typeset in (article, :85-88 `book`, `report`, `letter` or
12  // `memoir`; `colclass=<class>`, :30-50), :105 every other option passed to it, :111 `\LoadClass{\c@lclass}`. The
13  // binding loaded article whatever the option, so `\documentclass[book]{combine}`'s `\chapter` was undefined
14  // (72b review; witness 2310.14794 is an article one). combine's own switches (:89-104) set what this binding does
15  // not model. Guard `perfect_kernel_batch77v::combine_book_option_loads_book`.
16  RawTeX!(r"\providecommand{\c@lclass}{article}");
17  for class in ["book", "report", "letter", "memoir"] {
18    DeclareOption!(class, {
19      RawTeX!(&s!(r"\def\c@lclass{{{class}}}"));
20    });
21  }
22  for option in [
23    "classes",
24    "packages",
25    "layouts",
26    "folios",
27    "notoc",
28    "nolof",
29    "nolot",
30    "maintoc",
31    "date",
32    "noauthor",
33    "notitle",
34    "nomaketitle",
35    "nopubindoc",
36    "nopubintoc",
37    "onebib",
38    "combinedbib",
39  ] {
40    DeclareOption!(option, None);
41  }
42  DeclareOption!(None, {
43    let class = Expand!(T_CS!("\\c@lclass")).to_string();
44    RawTeX!(&s!(r"\PassOptionsToClass{{\CurrentOption}}{{{class}}}"));
45  });
46  // combine.cls:31-50 reads `colclass=<class>` (`colclass` alone: article) from the class options before
47  // `\ProcessOptions`, the last one given winning (`\setkeys`): every other option then goes to that class, and a
48  // class option (:85-88) replaces it. Read in `\DeclareOption*`, in turn, `[11pt,colclass=amsart]` passed 11pt to
49  // article and `[colclass=report,book]` loaded report. Guards `perfect_kernel_batch77v::{
50  // combine_colclass_gets_the_class_options, combine_class_option_after_colclass}`.
51  let options = Expand!(T_CS!("\\@classoptionslist")).to_string();
52  let colclass = options.split(',').rev().find_map(|option| {
53    let (name, value) = option.split_once('=').unwrap_or((option, "article"));
54    (name.trim() == "colclass").then(|| value.trim().to_string())
55  });
56  if let Some(class) = colclass {
57    RawTeX!(&s!(r"\def\c@lclass{{{class}}}"));
58  }
59  ProcessOptions!();
60  let class = Expand!(T_CS!("\\c@lclass")).to_string();
61  LoadClass!(&class);
62  // combine.cls (TeX Live's) loads no amsmath: amsmath's commands for papers that use them unloaded, its "Old form"
63  // error once the paper loads it; the paper's copy decides (`amsmath_sty::require_amsmath_beyond_original`,
64  // KNOWN_PERL_ERRORS #993).
65  amsmath_sty::require_amsmath_beyond_original(
66    latexml_package::shipped_class::shipped_copy_loads("cls", "amsmath")?,
67    &[],
68  )?;
69  RequirePackage!("amssymb");
70  // Eager xcolor preload removed for Perl parity: it makes a later document
71  // xcolor[table] load a no-op, so colortbl/array never load and array m{}/b{}
72  // columns break (Unrecognized tabular template -> Extra alignment tab). The
73  // document loads xcolor itself; color/definecolor stay via hyperref->color.
74  // See ifacconf_cls.rs and SYNC_STATUS (eager-xcolor cluster).
75  RequirePackage!("hyperref");
76
77  // combine.cls L760 redefines `\import` to take ONE arg (the
78  // basename of a sub-document) and `\input`s `<arg>.tex`. This
79  // conflicts with the `import` PACKAGE's `\import{<dir>/}{<file>}`
80  // (two-arg form) so we must NOT RequirePackage{import} here —
81  // instead define `\import` as combine.cls does. Witness 2310.14794.
82  DefMacro!("\\import{}", "\\input{#1}");
83
84  // Sub-documents inside `{papers}` have their own preambles —
85  // `\documentclass` + `\usepackage` calls. Real combine.cls L194-209
86  // saves and re-routes these via `\c@lbdocumentclass` / `\c@lbusepackage`
87  // when wrapped under `\begin{papers}`. For HTML output we simply
88  // discard them inside `{papers}` so the body content survives.
89  // Use a flag toggled at env begin/end; redefine the kernel CS
90  // to a conditional no-op only while `\ifc@lpapers` is true.
91  RawTeX!(r"\newif\ifc@lpapers");
92  DefMacro!(
93    T_CS!("\\begin{papers}"),
94    None,
95    "\\c@lpaperstrue\\let\\c@l@orig@docclass\\documentclass\
96     \\let\\c@l@orig@usepackage\\usepackage\
97     \\def\\documentclass{\\c@lpapers@gobble@docclass}%
98     \\def\\usepackage{\\c@lpapers@gobble@usepackage}"
99  );
100  DefMacro!(
101    T_CS!("\\end{papers}"),
102    None,
103    "\\let\\documentclass\\c@l@orig@docclass\
104     \\let\\usepackage\\c@l@orig@usepackage\
105     \\c@lpapersfalse"
106  );
107  // Helpers gobble optional + mandatory args.
108  DefMacro!("\\c@lpapers@gobble@docclass [] {}", None);
109  DefMacro!("\\c@lpapers@gobble@usepackage [] {}", None);
110
111  // combine.cls L942: `\newenvironment{papers}[1][\cleardoublepage]{...}{...}`
112  // The body just contains `\subimport`'d sub-documents (or their
113  // verbatim content). Render transparently. (Definitions added
114  // below to also gobble inner \documentclass / \usepackage.)
115  // \pretitle / \posttitle / \preauthor / \postauthor / \predate /
116  // \postdate — combine.cls hooks for sub-document title pages.
117  DefMacro!("\\pretitle{}", None);
118  DefMacro!("\\posttitle{}", None);
119  DefMacro!("\\preauthor{}", None);
120  DefMacro!("\\postauthor{}", None);
121  DefMacro!("\\predate{}", None);
122  DefMacro!("\\postdate{}", None);
123  // `\c@l...` family — combine.cls internal hooks. No-ops since we
124  // don't combine multiple sub-documents.
125  DefMacro!("\\c@lbusepackage[]{}", None);
126  DefMacro!("\\c@lbLoadClass", None);
127  DefMacro!("\\c@lbdocumentclass[]{}", None);
128});
