1//! beilstein.cls (Beilstein Journal of Nanotechnology / Organic Chemistry): the class is loaded raw, and its author
2//! block — `\author` (beilstein.cls:586-697: `\author{name}`, `\author[marks]{name}`, `\author[marks][e-mail]{name}`
3//! (:627 `\@@@author`, the e-mail listed with the others, :649-651), the corresponding author's
4//! `\author*[marks]{name}{e-mail}`) and `\affiliation{…}` (:704-737), which number the affiliations in order and give
5//! each to the author just named, building accumulators (`\authors`, `\affiliations`, `\emails`) that only its
6//! `\maketitle` reads (:739-823) — becomes the frontmatter: each author a creator carrying its marks and the numbers of
7//! the affiliations that follow it, the corresponding author's e-mail a contact, each affiliation a contact labelled
8//! by its number. Under the raw class (the rawclasses lever) the kernel's locked `\author` refused the class's, so a
9//! starred author became an author named `*` with its name and e-mail as stray affiliations, and every affiliation
10//! was lost (2203.09951, 2604.06890). An author is handed on when the next one starts or at `\maketitle`, once the
11//! affiliations that follow it are known.
12use latexml_package::prelude::*;
13
14/// A stored token list (`lx_beilstein_<key>`), empty when unset.
15fn tokens_of(key: &str) -> Vec<Token> {
16  match lookup_value(&s!("lx_beilstein_{key}")) {
17    Some(Stored::Tokens(tokens)) => tokens.unlist(),
18    _ => Vec::new(),
19  }
20}
21
22fn set_tokens(key: &str, tokens: Vec<Token>) {
23  assign_value(
24    &s!("lx_beilstein_{key}"),
25    Stored::Tokens(Tokens::new(tokens)),
26    Some(Scope::Global),
27  );
28}
29
30fn braced(tokens: Vec<Token>) -> Vec<Token> {
31  let mut out = vec![T_BEGIN!()];
32  out.extend(tokens);
33  out.push(T_END!());
34  out
35}
36
37LoadDefinitions!({
38  InputDefinitions!("beilstein", noltxml => true, extension => Some(Cow::Borrowed("cls")));
39  // (the class read raw by this binding is a raw class to K11, its title-page stores rerouted at the end of the class
40  // load as under the rawclasses lever; the binding's own setters below stay its own)
41  // The author just named: its marks, name and e-mail, and the affiliations that followed it.
42  DefPrimitive!("\\lx@beilstein@set{}{}{}", sub[(marks, name, email)] {
43    set_tokens("marks", marks.unlist());
44    set_tokens("name", name.unlist());
45    set_tokens("email", email.unlist());
46    set_tokens("affiliations", Vec::new());
47    assign_value("lx_beilstein_pending", true, Some(Scope::Global));
48  });
49  DefMacro!("\\lx@beilstein@flush", sub[_args] {
50    if !lookup_bool("lx_beilstein_pending") {
51      return Ok(Tokens!());
52    }
53    assign_value("lx_beilstein_pending", false, Some(Scope::Global));
54    let marks = tokens_of("marks");
55    let mut out = if marks.is_empty() {
56      TokenizeInternal!("\\lx@add@creator").unlist()
57    } else {
58      let mut call = TokenizeInternal!("\\lx@add@authors@append[annotations=").unlist();
59      call.extend(braced(marks));
60      call.push(T_OTHER!("]"));
61      call
62    };
63    out.extend(braced(tokens_of("name")));
64    let email = tokens_of("email");
65    if email.iter().any(|t| t.get_catcode() != Catcode::SPACE) {
66      out.extend(TokenizeInternal!("\\lx@add@email").unlist());
67      out.extend(braced(email));
68    }
69    out.extend(tokens_of("affiliations"));
70    Ok(Tokens::new(out))
71  });
72  // :704-737: the next affiliation number, given to the author just named, and the affiliation labelled by it.
73  DefPrimitive!("\\lx@beilstein@affiliation{}", sub[(text)] {
74    let n = lookup_int("lx_beilstein_affiliation_count") + 1;
75    assign_value("lx_beilstein_affiliation_count", Stored::Int(n), Some(Scope::Global));
76    let number = ExplodeText!(&n.to_string());
77    let mut marks = tokens_of("marks");
78    if !marks.is_empty() {
79      marks.push(T_OTHER!(","));
80    }
81    marks.extend(number.clone());
82    set_tokens("marks", marks);
83    let mut affiliations = tokens_of("affiliations");
84    affiliations.extend(TokenizeInternal!("\\lx@add@contact[role=affiliation,label=").unlist());
85    affiliations.extend(braced(number));
86    affiliations.push(T_OTHER!("]"));
87    affiliations.extend(braced(text.unlist()));
88    set_tokens("affiliations", affiliations);
89  });
90  RawTeX!(
91    r"\def\author{\lx@beilstein@flush\@ifstar\lx@beilstein@star\lx@beilstein@plain}
92\def\lx@beilstein@plain{\@ifnextchar[\lx@beilstein@plain@{\lx@beilstein@plain@[]}}
93\def\lx@beilstein@plain@[#1]{\@ifnextchar[{\lx@beilstein@plain@@[#1]}{\lx@beilstein@plain@@[#1][]}}
94\def\lx@beilstein@plain@@[#1][#2]#3{\lx@beilstein@set{#1}{#3}{#2}}
95\def\lx@beilstein@star{\@ifnextchar[\lx@beilstein@star@{\lx@beilstein@star@[]}}
96\def\lx@beilstein@star@[#1]#2#3{\lx@beilstein@set{#1}{#2}{#3}}
97\renewcommand\affiliation[2][]{\lx@beilstein@affiliation{#2}}"
98  );
99  AddToMacro!("\\lx@maketitle@begin", "\\lx@beilstein@flush");
100});
