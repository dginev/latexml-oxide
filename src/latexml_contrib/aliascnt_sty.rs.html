1use latexml_core::state::{assign_mapping, with_mapping};
2use latexml_package::prelude::*;
3
4LoadDefinitions!({
5  InputDefinitions!("aliascnt", noltxml => true, extension => Some(Cow::Borrowed("sty")));
6
7  // aliascnt.sty (`\newaliascnt{A}{B}`) `\let`-aliases the standard LaTeX counter
8  // machinery from B onto A: `\c@A`, `\theA`, `\theHA`, `\p@A`, `\cl@A`. That is
9  // everything real LaTeX needs. But LaTeXML additionally keys a counter's element
10  // `xml:id` off an internal `\the<ctr>@ID` macro (built by `NewCounter`,
11  // `Package.pm:695`), and aliascnt has no idea it exists — so an alias counter has
12  // no `@ID` route.
13  //
14  // The bite: `\newtheorem{lemma}[lemma]{Lemma}` where `lemma` is a `\newaliascnt`
15  // of `theorem`. `define_new_theorem` (latex_constructs.rs) only records
16  // `counter_for_type` when the shared-counter name differs from the theorem name
17  // (`[thm]` ≠ `lem`); here they coincide (`[lemma]` == `lemma`), so no mapping is
18  // set and it assumes `lemma` is a real LaTeXML counter with `\thelemma@ID`. It is
19  // not, so `RefStepCounter('lemma')` sees `has_id = false`, the theorem node gets
20  // NO `xml:id`, and `\label` inside it floats up to the nearest ancestor that
21  // already has an id (the enclosing section/subsection). Two lemmas then share the
22  // ancestor's ref number and both `\ref`s point at the wrong anchor.
23  //
24  // Fix: make the alias behave exactly like a normal shared-counter theorem
25  // (`\newtheorem{lem}[thm]{Lemma}`, which works because `define_new_theorem` sets
26  // `counter_for_type('lem') = 'thm'`). We map `counter_for_type(alias) => root`,
27  // where root is B's own root counter (following any alias chain via our already
28  // recorded mapping). Then `RefStepCounter(alias)` resolves through the root
29  // counter's real `\the<root>@ID`, and the theorem node gets an id in the same
30  // `Thm<root>` namespace as its sibling theorems — id, labels, and refs all land
31  // on the theorem. arXiv witness: 2605.20695 (html_feedback#6560), "Proof of
32  // Lemma 2.1" / "Proof of Lemma 2.2" (aliascnt lemmas under amsart).
33  DefPrimitive!("\\lx@aliascnt@register@id{}{}", sub[(alias, base)] {
34    let alias_s = alias.to_string();
35    let base_s = base.to_string();
36    // Only patch a successfully-created alias (aliascnt `\gdef`s `\AC@cnt@A`
37    // exactly when the alias is defined; a name clash leaves it undefined).
38    if is_defined(&s!("\\AC@cnt@{alias_s}")) {
39      // Follow the chain to the root: if B is itself an alias we already recorded
40      // its root under counter_for_type; otherwise B is the root.
41      let root = with_mapping("counter_for_type", &base_s, |m| m.map(ToString::to_string))
42        .unwrap_or(base_s);
43      assign_mapping("counter_for_type", &alias_s, Some(root));
44    }
45  });
46
47  RawTeX!(r"\let\lx@saved@newaliascnt\newaliascnt");
48  RawTeX!(
49    r"\renewcommand*{\newaliascnt}[2]{%
50    \lx@saved@newaliascnt{#1}{#2}%
51    \lx@aliascnt@register@id{#1}{#2}}"
52  );
53});
