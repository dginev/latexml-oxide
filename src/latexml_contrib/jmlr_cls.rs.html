51  // Author block. jmlr.cls L259/374-449:
52  //   \author[short]{ \Name{N1} \Email{E1} \\ \Name{N2} \Email{E2}
53  //                    \\ \addr Affiliation }
54  // The `\Name`/`\Email`/`\addr` sub-macros carry the structure. The generic
55  // \and/comma author splitter mis-reads it — it crams every \Name into a
56  // single <personname> and splits the affiliation's commas into phantom
57  // authors ("Oxford", "UK"). So when the body uses \Name, digest it directly
58  // into structured creators (each \Name → ltx:creator/ltx:personname, \Email →
59  // ltx:contact[role=email], the trailing \addr block → ltx:contact

63  DefMacro!("\\author[]{}", sub[(short, body)] {
64    if body.to_string().contains("\\Name") {
65      Ok(Invocation!(T_CS!("\\lx@jmlr@structauthor"), vec![Some(body)]))
66    } else {
67      Ok(Invocation!(T_CS!("\\lx@jmlr@core@author"), vec![short, Some(body)]))
