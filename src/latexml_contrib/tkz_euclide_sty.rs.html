15    r"\@ifundefined{usetkzobj}{\def\usetkzobj{\pgfutil@ifnextchar[{\use@usetkzobj}{\use@@usetkzobj}}\def\use@usetkzobj[#1]{\use@@usetkzobj{#1}}\def\use@@usetkzobj#1{}}{}
