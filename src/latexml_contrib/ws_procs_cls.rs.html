1//! World Scientific proceedings and review-volume classes (ws-procs9x6, ws-procs961x669, ws-rv9x6): the ws journal
2//! binding, and the `\body` these three classes add to it.
3use latexml_core::definition::origin::{DefinitionOrigin, OriginGuard};
4use latexml_package::{prelude::*, shipped_class::shipped_copy_text};
5
6use crate::shipped_class::{class_code_contains, require_shipped_packages};
7
8LoadDefinitions!({
9  crate::ws_journal_cls::load_definitions()?;
10  // ws-procs9x6.cls:586-591 (ws-procs961x669.cls:596-601, ws-rv9x6.cls:1011 the same; the journal classes define
11  // neither, so a paper's own `\newcommand{\body}` stays theirs): `\body` (and its alias `\bodymatter`) opens the main
12  // matter, restarting the footnotes and lettering them; the class's `\@makefnmark` there (a roman superscript) is the
13  // printed mark's look only. Witnesses 0704.0883, 1008.0906, 1903.10424, 2301.12666, 1308.0373, 2202.11015
14  // (ws-procs961x669), 0902.1904 (ws-rv9x6).
15  RawTeX!(
16    r"\def\body{\setcounter{footnote}{0}\def\thefootnote{\alph{footnote}}}
17\def\bodymatter{\body}"
18  );
19  // ws-rv9x6.cls:27 the chapter-numbering switch `\if@addchapnum` (the class's `addchapnum` option, :44, sets it;
20  // numbering is presentation here), which papers test in their own numbering macros (run 340: 113 papers; witnesses
21  // 2103.07802, 2008.12305, 1609.07109).
22  RawTeX!(r"\newif\if@addchapnum");
23  // ws-rv9x6.cls:497-501 `\newcommand\chapter` (a review volume is a book of chapters; the two proceedings classes
24  // define none): the class has chapters, so `\@ifundefined{chapter}` says so (ws-rv-har.sty, ws-minitoc.sty: 1807.01378)
25  // and a `\chapter` call is LaTeXML's chapter without the chapterless Warning (OXIDIZED_DESIGN #179 (a), #734).
26  // Its chapter counter with it, the sections numbered within it (ws-rv9x6.cls:397 `\newcounter{chapter}`, :399
27  // `\newcounter{section}[chapter]`): 1807.01378's `\setcounter{chapter}{…}` was "Counter 'chapter' was not defined".
28  // The section redeclared within the chapter, ids too (`Ch1.S1`, as book_cls.rs), the class's `\thesection` kept.
29  if Expand!(Tokens!(T_CS!("\\@currname"))).to_string() == "ws-rv9x6" {
30    DefMacro!("\\chapter", "\\@startsection{chapter}{0}{}{}{}{}");
31    NewCounter!("chapter", "document", idprefix => "Ch", nested => vec!["section"]);
32    let thesection = lookup_meaning(&T_CS!("\\thesection"));
33    NewCounter!("section", "chapter", idprefix => "S", nested => vec!["subsection"]);
34    if let Some(meaning) = thesection {
35      assign_meaning(&T_CS!("\\thesection"), meaning, Some(Scope::Global));
36    }
37  }
38  // ws-rv9x6.cls:1524-1528 (0706.1148's copy) `\usepackage[figuresright]{rotating}`, `{amsfonts,amssymb,amsmath}`,
39  // `{boxedminipage}`, `{makeidx}`, `{url}`: a review-volume chapter ends with `\printindex` without loading makeidx
40  // (33 ws-rv9x6 papers of run 353; 0706.1148, 0810.2192, 0901.1361). The paper's copy decides, for these three classes
41  // alike; without one, none.
42  require_shipped_packages("cls", &["rotating", "boxedminipage", "makeidx", "url"])?;
43  // The three classes' cross-reference shorthands (ws-procs9x6.cls v1.2u :984-993 as 0704.0369 ships it,
44  // ws-procs961x669.cls :1139-1148, ws-rv9x6.cls :1234-1243): `\eref` "Eq. (n)", `\sref` "Sec. n", `\fref` "Fig. n",
45  // `\tref` "Table n" and the capitalised `\Eref`…`\Tref` (run 347: 13 papers call `\fref`, 11 `\eref`; 0704.0369,
46  // 1605.04791, 1110.2544); `\figurecaptionfont` (:79; 1911.07780); `\wstoc{title}{authors}` (:338-344), a proceedings
47  // volume's table-of-contents line for the paper, kept out of the document as its `\\addtocontents` lines are (16
48  // papers: 1605.04791, 2110.00344); `\mastertoc` (:218), the
49  // volume's contents (2002.12333); `{unnumlist}` (:548), a list whose items carry no label (1402.0787, 1306.3110); the
50  // theorem-like `{algorithm}` and `{assumption}` (:918-921, numbered within the section; 1109.0078, 2512.24879); and
51  // `\aindx{name}`, an author-index entry — ws-procs `\index{author}{#1}` (:846, ws-multind's two-argument `\index`;
52  // 0809.4982, 1607.03235), ws-rv9x6 `\index[aindx]{#1}` (:1155), as the paper's copy writes it. The cross-reference
53  // names stay a document's own where its class lacks them (`DefinitionOrigin::Fallback`: cleveref's `\cref`, which the
54  // classes also define, is left to cleveref).
55  {
56    let _class = OriginGuard::new(DefinitionOrigin::Fallback);
57    RawTeX!(
58      r"\newcommand{\eref}[1]{Eq.~(\ref{#1})}
59\newcommand{\sref}[1]{Sec.~\ref{#1}}
60\newcommand{\fref}[1]{Fig.~\ref{#1}}
61\newcommand{\tref}[1]{Table~\ref{#1}}
62\newcommand{\Eref}[1]{Equation (\ref{#1})}
63\newcommand{\Sref}[1]{Section~\ref{#1}}
64\newcommand{\Fref}[1]{Figure~\ref{#1}}
65\newcommand{\Tref}[1]{Table~\ref{#1}}
66\def\figurecaptionfont{\footnotesize}
67\newcounter{ptitl}
68\def\wstoc#1#2{\refstepcounter{ptitl}}
69\newcommand\mastertoc{\tableofcontents}
70\newenvironment{unnumlist}{\list{}{}}{\endlist}
71\@ifundefined{algorithm}{\newtheorem{algorithm}{Algorithm}[section]}{}
72\@ifundefined{assumption}{\newtheorem{assumption}{Assumption}[section]}{}"
73    );
74    if shipped_copy_text("cls")?.is_some_and(|text| class_code_contains(&text, "\\index[aindx]")) {
75      RawTeX!(r"\def\aindx#1{\index[aindx]{#1}}");
76    } else {
77      RawTeX!(r"\def\aindx#1{\index{author}{#1}}");
78    }
79  }
80});
