1//! spotcolor.sty (Jens Elstner, 2006; no Perl binding): read raw, as before. Its `\AddSpotColor{<space>}{<name>}
2//! {<PDF name>}{<C M Y K>}` (spotcolor.sty:31-41) writes the spot colour's CMYK alternate only into a PDF object, so
3//! the binding keeps it for color's driver models (`color_sty::driver_alternate_color`): a `\definecolor{<n>}
4//! {spotcolor}{<name>,<tint>}` that defines `<n>` first takes the alternate at that tint — the colour a viewer without
5//! the spot ink shows (the Separation's tint transform, C0 0 0 0 0 to C1 the CMYK, N 1). ieeeaccess.cls:126-131's
6//! `PANTONE3015C` (CMYK 1 0.3 0 0.2).
7use latexml_package::prelude::*;
8
9LoadDefinitions!({
10  InputDefinitions!("spotcolor", noltxml => true, extension => Some(Cow::Borrowed("sty")),
11    handleoptions => true);
12  // (the package's own `\AddSpotColor` runs as it was; its CMYK is kept first)
13  RawTeX!(
14    r"\let\lx@spot@AddSpotColor\AddSpotColor
15\def\AddSpotColor#1#2#3#4{\lx@spot@alternate{#2}{#4}\lx@spot@AddSpotColor{#1}{#2}{#3}{#4}}"
16  );
17  DefPrimitive!("\\lx@spot@alternate{}{}", sub[(name, cmyk)] {
18    let cmyk = do_expand(cmyk)?.to_string();
19    let values: Vec<&str> = cmyk.split_whitespace().collect();
20    if values.len() == 4 && values.iter().all(|v| v.parse::<f64>().is_ok()) {
21      AssignValue!(&s!("spotcolor_alternate:{}", name.to_string().trim()) => values.join(","), Some(Scope::Global));
22    }
23  });
24  // spotcolor.sty:81-82 `\ifhks\input{spotcolorhks}\fi\ifpantone\input{spotcolorpantone}\fi`: the `[hks]`/`[pantone]`
25  // preloads ran in the package's read, before the wrapper above (1,089 `\AddSpotColor`s in spotcolorpantone.tex);
26  // read again for their CMYK only, the colours and spaces added already (the reviewer's probe spot_pantone:
27  // `PANTONE103PC` is the alternate cmyk 0 0.03 1 0.18, not black).
28  RawTeX!(
29    r"\begingroup\let\NewSpotColorSpace\@gobble\def\AddSpotColor#1#2#3#4{\lx@spot@alternate{#2}{#4}}%
30\ifhks\input{spotcolorhks}\fi\ifpantone\input{spotcolorpantone}\fi\endgroup"
31  );
32  // color.sty's driver model `spotcolor` (spotcolor.sty:71 `\color@spotcolor`): `<name>,<tint>` (`\c@lor@@spotcolor`,
33  // :72-75) → the CMYK alternate scaled by the tint, or nothing for a spot colour no `\AddSpotColor` gave.
34  DefMacro!("\\lx@color@alternate@spotcolor{}", sub[(spec)] {
35    let spec = spec.to_string();
36    let (name, tint) = spec.split_once(',').unwrap_or((spec.as_str(), "1"));
37    let tint: f64 = tint.trim().parse().unwrap_or(1.0);
38    let alternate = lookup_string(&s!("spotcolor_alternate:{}", name.trim()));
39    if alternate.is_empty() {
40      return Ok(Tokens::default());
41    }
42    let scaled: Vec<String> = alternate
43      .split(',')
44      .filter_map(|v| v.parse::<f64>().ok())
45      .map(|v| s!("{}", (v * tint).clamp(0.0, 1.0)))
46      .collect();
47    Ok(mouth::tokenize_internal(TeXString::assembled(s!("{{cmyk}}{{{}}}", scaled.join(",")))))
48  });
49});
