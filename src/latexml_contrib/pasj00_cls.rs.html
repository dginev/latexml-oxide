1//! The PASJ classes (Publications of the Astronomical Society of Japan) — `pasj00`, `pasj01`, `pasj02`: shipped with
2//! every paper, not in TeX Live. The shipped class is interpreted raw (in every profile), for its own macros, and what
3//! LaTeXML builds semantically is put back over it: the kernel `\caption`, and the frontmatter its title page only
4//! typesets.
5use latexml_package::prelude::*;

7use crate::and_by_meaning::and_runs;

9/// Interpret the shipped class `name` raw, then restore the semantic definitions over it:
10/// - the kernel `\caption` (`\lx@caption`): PASJ's own (pasj00.cls:814-835, pasj01.cls:898-919, pasj02.cls:887-908)
11///   steps the counter and calls the class's `\@makecaption` directly, bypassing `\@caption`, the one path that builds
12///   `<caption>`, so its captions were paragraphs without their number and their labels dangled (1310.7069,
13///   1505.02769, 2504.06663);
14/// - `\@maketitle` empty: everything its title page lays out (pasj01.cls:1912-1950) is in the frontmatter, and the
15///   deposit of the rest left an empty "[ Received ; Accepted ]" frame in the body (1505.02769, 2005.13750).
16///
17/// A paper whose source lacks the class falls to OmniBus, as before the binding, with the class's own `{longtable}`
18/// ([`pasj_unshipped_longtable`]) and the class's own commands papers call ([`pasj_unshipped_macros`]). Either way the
19/// frontmatter macros the classes store for `\maketitle` to typeset (pasj01.cls:84-159) are the frontmatter's: they were
20/// lost under OmniBus and raw alike.
21pub(crate) fn pasj_class(name: &str) -> Result<()> {
22  let shipped = find_file(
23    name,
24    Some(FindFileOptions {
25      ext_type: Some(Cow::Borrowed("cls")),
26      forbid_ltxml: true,
27      ..Default::default()
28    }),
29  )
30  .is_some();
31  if shipped {
32    InputDefinitions!(name, noltxml => true, extension => Some(Cow::Borrowed("cls")));
33    Let!("\\caption", "\\lx@caption");
34    Let!("\\@maketitle", "\\@empty");
35    // pasj00.cls:4025-4035, pasj01.cls:4098-4110 ("%%% fleqn"): `\equation` is fleqn.clo's flush-left `\hbox to
36    // \linewidth\bgroup $\m@th\displaystyle`, which the raw class cannot give the kernel's `{equation}`, so its hook is
37    // set here (latexml_package fleqn_sty.rs `math_displays_in_hbox`): an author's `$` in it is text in the math
38    // (2311.01721; KNOWN_PERL_ERRORS #907). pasj02.cls:4070-4083 leaves `\equation` commented out.
39    if crate::shipped_class::shipped_class_text(name)?
40      .is_some_and(|text| crate::shipped_class::class_code_contains(&text, "\\def\\equation{"))
41    {
42      fleqn_sty::math_displays_in_hbox()?;
43    }
44  } else {
45    LoadClass!("OmniBus");
46    pasj_unshipped_longtable()?;
47    pasj_unshipped_macros()?;
48    pasj_unshipped_timeform()?;
49  }
50  pasj_frontmatter()
51}

53/// The class's own commands a paper whose source lacks the class calls, which OmniBus does not define (identical in
54/// pasj00.cls of 0704.3654, pasj01.cls of 1505.02769 and pasj02.cls of 2504.06663):
55/// - `\SetRunningHead` (pasj00.cls:52-54, pasj01.cls:68-70, pasj02.cls:53-55) stores the running head, no content;
56/// - `\FigureFile(w,h){file}` and `\RelativeFigurePath` (pasj00.cls:4122-4164, pasj01.cls:4179-4221,
57///   pasj02.cls:4174-4216): the file, under the figure path, at the width `w` (in mm when it has no unit), through
58///   graphicx's `\includegraphics`; the class's `\vtop` and `\kern` around it are print layout, and its framed
59///   file name for a missing file is the binding's missing-graphic warning;
60/// - amssymb for `\lesssim`/`\gtrsim` (the class's `useamsfonts` branch, pasj00.cls:3445-3446; its other branch
61///   composes the same relations), and the solar symbols `\Sol`, `\solar`, `\MO`, `\LO`, `\RO` (pasj00.cls:2303,
62///   3433-3444) with `\odot`, the glyph the class's `\@solar` takes in script styles;
63/// - `\authorcite` and `\yearcite` (pasj00.cls:1636-1637, pasj01.cls:1704-1705, pasj02.cls:1695-1696), the class's
64///   author-only and year-only citations: natbib's `\citeauthor` and `\citeyear`, which OmniBus autoloads.
65///
66/// Without them astro-ph0108490, astro-ph0201077 and astro-ph0204110 had an undefined-macro error per command, their
67/// figures were the error's text, and each `\authorcite{M31_Makishima}` printed its key with an Error per `_`
68/// (astro-ph0108490). Guard `perfect_kernel_batchclustfix::pasj00_without_its_class_keeps_its_figures`.

75\def\FigureFile{\@ifstar\@FigureFile\@FigureFile}
76\def\@FigureFile(#1,#2)#3{\def\@figfilename{#3}\@onelevel@sanitize\@figfilename\begingroup

437/// The class's own `{longtable}`/`{longtable*}` (pasj00.cls:884-916, pasj01.cls:967-1000, pasj02.cls:956-989) for a
438/// paper whose source lacks the class: the longtable binding's table, which keeps the interface the class shares with
439/// longtable.sty (`\endfirsthead`/`\endhead`/`\endfoot`/`\endlastfoot`, pasj00.cls:1241-1244), with the class's own
440/// `\caption` and `\label`. Its `\@lt@init` lets them to `\@lt@setcaption` and `\@lt@label` (pasj00.cls:1068-1070,
441/// pasj01.cls:1152-1154, pasj02.cls:1141-1143): each is a `\noalign` that gobbles a following `\\` if there is one
442/// (`\@lt@remove@cr`; pasj00.cls:1245-1270, pasj01.cls:1329-1354, pasj02.cls:1318-1343), so a caption is no row and
443/// PASJ papers write `\caption{…}\label{…}`, then `\hline` and the head row. The optional argument
444/// (`\@lt@captionnumber`, a number in place of the counter's) is the list entry's, as the binding's kernel `\caption`
445/// reads it everywhere else; the star changes nothing, as in the class. Without the class's `{longtable}` the tables of
446/// astro-ph0407419, 0704.1713, 0705.1365, 1108.4491, astro-ph0511689, astro-ph0605217, astro-ph0612753 and
447/// astro-ph0701701 were text with a misplaced-alignment error per row; with longtable.sty's caption rule (a row ended
448/// by `\\`), the head row after each of their captions was lost.
449/// Guard `perfect_kernel_batch65h::pasj00_without_its_class_keeps_longtable`.
450fn pasj_unshipped_longtable() -> Result<()> {
451  RequirePackage!("longtable");
452  // The binding lets `\caption`/`\label` to these two at the table's start; the class's `\@lt@init` comes after.
453  DefMacro!(
454    "\\lx@pasj@lt@init",
455    r"\let\lx@longtable@caption\lx@pasj@lt@setcaption\let\lx@longtable@label\lx@pasj@lt@label"
456  );
457  DefMacro!("\\longtable", r"\begingroup\lx@pasj@lt@init\LT@array");
458  DefMacro!(
459    "\\csname longtable*\\endcsname",
460    r"\begingroup\lx@pasj@lt@init\LT@array"
461  );
462  Let!("\\lx@pasj@lt@@label", "\\lx@longtable@label");
463  DefMacro!("\\lx@pasj@lt@setcaption OptionalMatch:* []{}", sub[(_star, number, caption)] {
464    pasj_lt_remove_cr()?;
465    Ok(Invocation!(r"\noalign{\lx@longtable@caption@text{#1}{#2}}", vec![number, Some(caption)]))
466  });

468    pasj_lt_remove_cr()?;
469    Ok(Invocation!(r"\noalign{\lx@pasj@lt@@label{#1}}", vec![Some(label)]))
470  });
471  Ok(())

474/// pasj00.cls `\@lt@remove@cr` (:1254-1261): past spaces (`\@ifnextchar`), a `\\` is gobbled; any other token stays.
475/// Read where the class reads it, before its `\noalign` closes; the tokens read are not expanded.
476fn pasj_lt_remove_cr() -> Result<()> {
477  if let Some(next) = read_non_space()?
478    && !x_equals(&next, &T_CS!("\\\\"))
479  {
480    unread_one(next);
481  }
482  Ok(())
483}

485/// PASJ's frontmatter macros as the frontmatter. One `\author` lists every author, separated by commas and `\&`, its
486/// rows broken by `\\` (pasj00.cls:73-104), and the template puts each author's marks after the comma that ends the
487/// name (`Name,\altaffilmark{1}\orcid{…}`, pasj02's usage). The kernel would read each line after a `\\` as an
488/// affiliation (most of 0707.3867's 26 authors) and hand a mark after a comma to the next author (2504.06663,
489/// 0704.3654), so the binding splits the list itself ([`pasj_author_list`]). Documents write `\altaffiltext{n}{text}`
490/// and `\affil{text}` under all three versions, whatever pasj00's argument shapes; pasj00's `\email` takes an optional
491/// name; pasj02's `\altemailmark` only marks the author `\email` belongs to.
492fn pasj_frontmatter() -> Result<()> {
493  Let!("\\lx@pasj@author", "\\author");
494  DefMacro!("\\author[]{}", sub[(label, author)] {
495    let mut call = vec![T_CS!("\\lx@pasj@author")];
496    if let Some(label) = label {
497      call.push(T_OTHER!("["));
498      call.extend(label.unlist());
499      call.push(T_OTHER!("]"));
500    }
501    call.push(T_BEGIN!());
502    call.extend(pasj_author_list(&author.unlist())?);
503    call.push(T_END!());
504    Ok(Tokens::new(call))
505  });
506  DefMacro!("\\KeyWords{}", "\\lx@add@keywords{#1}");
507  Let!("\\kword", "\\KeyWords");
508  DefMacro!("\\affil{}", "\\lx@add@affiliation{#1}");
509  DefMacro!(
510    "\\altaffilmark Semiverbatim",
511    "\\lx@request@frontmatter@annotation[altaffil]{#1}"
512  );
513  // An empty text (a template's placeholder) makes no contact, which would stand as an empty creator (0704.3654).
514  DefMacro!("\\altaffiltext Semiverbatim {}", sub[(label, text)] {
515    if text.unlist_ref().iter().all(|t| t.get_catcode() == Catcode::SPACE) {
516      Ok(Tokens!())
517    } else {
518      Ok(Invocation!("\\lx@add@contact[role=altaffiliation,label={altaffil:#1}]{#2}", vec![Some(label), Some(text)]))
519    }
520  });
521  // pasj00.cls:68-69, the address in `\small\itshape`; the binding keeps the current font, not that size and shape
522  // (1001.5133)
523  DefMacro!("\\email[]{}", "\\lx@add@email{\\lx@address@textfont{#2}}");
524  def_macro_noop("\\altemailmark")?;
525  DefMacro!("\\orcid Semiverbatim", "\\lx@add@orcid{#1}");
526  DefMacro!("\\Received{}", "\\lx@add@date[role=received]{#1}");
527  DefMacro!("\\Accepted{}", "\\lx@add@date[role=accepted]{#1}");
528  Ok(())
529}

531/// The author marks a PASJ name carries, with their arguments: (control sequence, takes an optional argument, braced
532/// arguments) — `\altaffilmark{…}`, `\orcid{…}`, `\thanks[…]{…}` (pasj01.cls:2019), `\email[…]{…}`, `\altemailmark`.
533const PASJ_AUTHOR_MARKS: [(&str, bool, usize); 5] = [
534  ("\\altaffilmark", false, 1),
535  ("\\orcid", false, 1),
536  ("\\thanks", true, 1),
537  ("\\email", true, 1),
538  ("\\altemailmark", false, 0),
539];

541/// A PASJ author list as `\and`-separated names: it splits at depth-0 `,`, `\&`, `\and` and `\\` (with the `*` and a
542/// closed `[<skip>]` that follow it), gives a piece's leading marks (and math superscripts) back to the name before it,
543/// drops a piece's leading "and", and keeps no empty piece. The `\and`s: each written, as before (pasj00.cls:1878
544/// prints its own), and each held in a macro of the document's, which the separator query reads as the kernel's
545/// (`and_by_meaning`, Option C P2-5).
546fn pasj_author_list(toks: &[Token]) -> Result<Vec<Token>> {
547  let ands = and_runs(toks, &[T_CS!("\\and")])?.top_level(toks);
548  let mut pieces: Vec<Vec<Token>> = vec![Vec::new()];
549  let mut depth = 0usize;
550  let mut i = 0;
551  while i < toks.len() {
552    if let Some(end) = ands.get(&i) {
553      i = *end;
554      pieces.push(Vec::new());
555      continue;
556    }
557    let t = toks[i];
558    i += 1;
559    match t.get_catcode() {
560      Catcode::BEGIN => depth += 1,
561      Catcode::END => depth = depth.saturating_sub(1),
562      _ => {},
563    }
564    let separator = depth == 0 && (t == T_OTHER!(",") || t == T_CS!("\\&") || t == T_CS!("\\\\"));
565    if !separator {
566      pieces.last_mut().unwrap().push(t);
567      continue;
568    }
569    if t == T_CS!("\\\\") {
570      let mut k = i;
571      while k < toks.len() && toks[k].get_catcode() == Catcode::SPACE {
572        k += 1;
573      }
574      if k < toks.len() && toks[k] == T_OTHER!("*") {
575        k += 1;
576      }
577      if k < toks.len()
578        && toks[k] == T_OTHER!("[")
579        && let Some(close) = closing_bracket(&toks[k..])
580      {
581        k += close + 1;
582      }
583      i = k;
584    }
585    pieces.push(Vec::new());
586  }
587  let mut names: Vec<Vec<Token>> = Vec::new();
588  for mut piece in pieces {
589    let marks = leading_marks(&piece);
590    if marks > 0
591      && let Some(prev) = names.last_mut()
592    {
593      prev.extend(piece.drain(..marks));
594    }
595    let piece = strip_leading_and(piece);
596    if piece.iter().any(|t| t.get_catcode() != Catcode::SPACE) {
597      names.push(piece);
598    }
599  }
600  let mut list = Vec::new();
601  for (n, name) in names.into_iter().enumerate() {
602    if n > 0 {
603      list.push(T_CS!("\\and"));
604    }
605    list.extend(name);
606  }
607  Ok(list)
608}

610/// The length of a piece's leading run of author marks (with their arguments), math superscripts (`$^{1}$`) and
611/// spaces — 0 when it starts with none: the marks that follow the comma ending the previous name.
612fn leading_marks(piece: &[Token]) -> usize {
613  let mut end = 0;
614  let mut i = 0;
615  while i < piece.len() {
616    let t = piece[i];
617    match t.get_catcode() {
618      Catcode::SPACE => i += 1,
619      Catcode::MATH => {
620        let Some(len) = piece[i + 1..]
621          .iter()
622          .position(|t| t.get_catcode() == Catcode::MATH)
623        else {
624          break;
625        };
626        if piece.get(i + 1).map(|t| t.get_catcode()) != Some(Catcode::SUPER) {
627          break;
628        }
629        i += len + 2;
630        end = i;
631      },
632      Catcode::CS => {
633        let Some((_, optional, braced)) = PASJ_AUTHOR_MARKS
634          .iter()
635          .find(|(cs, ..)| t.with_str(|s| s == *cs))
636        else {
637          break;
638        };
639        i += 1;
640        // Exactly its own arguments: a following brace group belongs to the next name (`{\'A}lvaro`).
641        let skip_spaces = |mut k: usize| {
642          while k < piece.len() && piece[k].get_catcode() == Catcode::SPACE {
643            k += 1;
644          }
645          k
646        };
647        if *optional {
648          let k = skip_spaces(i);
649          if piece.get(k) == Some(&T_OTHER!("[")) {
650            match closing_bracket(&piece[k..]) {
651              Some(close) => i = k + close + 1,
652              None => return end,
653            }
654          }
655        }
656        for _ in 0..*braced {
657          let k = skip_spaces(i);
658          match piece.get(k) {
659            Some(b) if b.get_catcode() == Catcode::BEGIN => {
660              let mut level = 0usize;
661              i = k;
662              while i < piece.len() {
663                match piece[i].get_catcode() {
664                  Catcode::BEGIN => level += 1,
665                  Catcode::END => {
666                    level -= 1;
667                    if level == 0 {
668                      i += 1;
669                      break;
670                    }
671                  },
672                  _ => {},
673                }
674                i += 1;
675              }
676            },
677            // An undelimited argument is one token (`\altaffilmark1`).
678            Some(_) => i = k + 1,
679            None => return end,
680          }
681        }
682        end = i;
683      },
684      _ => break,
685    }
686  }
687  end
688}

690/// The offset of the `]` closing the `[` that `toks` starts with, at brace depth 0 (`\thanks[{a]b}]{…}`).
691fn closing_bracket(toks: &[Token]) -> Option<usize> {
692  let mut depth = 0usize;
693  for (n, t) in toks.iter().enumerate() {
694    match t.get_catcode() {
695      Catcode::BEGIN => depth += 1,
696      Catcode::END => depth = depth.saturating_sub(1),
697      _ if depth == 0 && *t == T_OTHER!("]") => return Some(n),
698      _ => {},
699    }
700  }
701  None
702}

704/// A piece without a leading "and" — the last name of a comma list ("…, and D. Four").
705fn strip_leading_and(piece: Vec<Token>) -> Vec<Token> {
706  let start = piece
707    .iter()
708    .position(|t| t.get_catcode() != Catcode::SPACE)
709    .unwrap_or(piece.len());
710  let word: String = piece[start..]
711    .iter()
712    .take(3)
713    .map(|t| t.to_string())
714    .collect();
715  let after = piece.get(start + 3);
716  if word == "and" && after.is_some_and(|t| t.get_catcode() == Catcode::SPACE) {
717    piece[start + 3..].to_vec()
718  } else {
719    piece
720  }
721}
