1//! The PASJ classes (Publications of the Astronomical Society of Japan) — `pasj00`, `pasj01`, `pasj02`: shipped with
2//! every paper, not in TeX Live. The shipped class is interpreted raw (in every profile), for its own macros, and what
3//! LaTeXML builds semantically is put back over it: the kernel `\caption`, and the frontmatter its title page only
4//! typesets.

9/// Interpret the shipped class `name` raw, then restore the semantic definitions over it:
10/// - the kernel `\caption` (`\lx@caption`): PASJ's own (pasj00.cls:814-835, pasj01.cls:898-919, pasj02.cls:887-908)
11///   steps the counter and calls the class's `\@makecaption` directly, bypassing `\@caption`, the one path that builds
12///   `<caption>`, so its captions were paragraphs without their number and their labels dangled (1310.7069,
13///   1505.02769, 2504.06663);
14/// - `\@maketitle` empty: everything its title page lays out (pasj01.cls:1912-1950) is in the frontmatter, and the
15///   deposit of the rest left an empty "[ Received ; Accepted ]" frame in the body (1505.02769, 2005.13750).
16///
17/// A paper whose source lacks the class falls to OmniBus, as before the binding, with the class's own `{longtable}`
18/// ([`pasj_unshipped_longtable`]). Either way the frontmatter macros the classes store for `\maketitle` to typeset
19/// (pasj01.cls:84-159) are the frontmatter's: they were lost under OmniBus and raw alike.
20pub(crate) fn pasj_class(name: &str) -> Result<()> {
21  let shipped = find_file(
22    name,
23    Some(FindFileOptions {
24      ext_type: Some(Cow::Borrowed("cls")),
25      forbid_ltxml: true,
26      ..Default::default()
27    }),
28  )
29  .is_some();
30  if shipped {
31    InputDefinitions!(name, noltxml => true, extension => Some(Cow::Borrowed("cls")));
32    Let!("\\caption", "\\lx@caption");
33    Let!("\\@maketitle", "\\@empty");
34    // pasj00.cls:4025-4035, pasj01.cls:4098-4110 ("%%% fleqn"): `\equation` is fleqn.clo's flush-left `\hbox to
35    // \linewidth\bgroup $\m@th\displaystyle`, which the raw class cannot give the kernel's `{equation}`, so its hook is
36    // set here (latexml_package fleqn_sty.rs `math_displays_in_hbox`): an author's `$` in it is text in the math
37    // (2311.01721; KNOWN_PERL_ERRORS #907). pasj02.cls:4070-4083 leaves `\equation` commented out.
38    if crate::shipped_class::shipped_class_text(name)?
39      .is_some_and(|text| crate::shipped_class::class_code_contains(&text, "\\def\\equation{"))
40    {
41      fleqn_sty::math_displays_in_hbox()?;
42    }
43  } else {
44    LoadClass!("OmniBus");
45    pasj_unshipped_longtable()?;
46  }
47  pasj_frontmatter()
48}

50/// The class's own `{longtable}`/`{longtable*}` (pasj00.cls:884-916, pasj01.cls:967-1000, pasj02.cls:956-989) for a
51/// paper whose source lacks the class: the longtable binding's table, which keeps the interface the class shares with
52/// longtable.sty (`\endfirsthead`/`\endhead`/`\endfoot`/`\endlastfoot`, pasj00.cls:1241-1244), with the class's own
53/// `\caption` and `\label`. Its `\@lt@init` lets them to `\@lt@setcaption` and `\@lt@label` (pasj00.cls:1068-1070,
54/// pasj01.cls:1152-1154, pasj02.cls:1141-1143): each is a `\noalign` that gobbles a following `\\` if there is one
55/// (`\@lt@remove@cr`; pasj00.cls:1245-1270, pasj01.cls:1329-1354, pasj02.cls:1318-1343), so a caption is no row and
56/// PASJ papers write `\caption{…}\label{…}`, then `\hline` and the head row. The optional argument
57/// (`\@lt@captionnumber`, a number in place of the counter's) is the list entry's, as the binding's kernel `\caption`
58/// reads it everywhere else; the star changes nothing, as in the class. Without the class's `{longtable}` the tables of
59/// astro-ph0407419, 0704.1713, 0705.1365, 1108.4491, astro-ph0511689, astro-ph0605217, astro-ph0612753 and
60/// astro-ph0701701 were text with a misplaced-alignment error per row; with longtable.sty's caption rule (a row ended
61/// by `\\`), the head row after each of their captions was lost.
62/// Guard `perfect_kernel_batch65h::pasj00_without_its_class_keeps_longtable`.
63fn pasj_unshipped_longtable() -> Result<()> {
64  RequirePackage!("longtable");
65  // The binding lets `\caption`/`\label` to these two at the table's start; the class's `\@lt@init` comes after.
66  DefMacro!(
67    "\\lx@pasj@lt@init",
68    r"\let\lx@longtable@caption\lx@pasj@lt@setcaption\let\lx@longtable@label\lx@pasj@lt@label"
69  );
70  DefMacro!("\\longtable", r"\begingroup\lx@pasj@lt@init\LT@array");
71  DefMacro!(
72    "\\csname longtable*\\endcsname",
73    r"\begingroup\lx@pasj@lt@init\LT@array"
74  );
75  Let!("\\lx@pasj@lt@@label", "\\lx@longtable@label");
76  DefMacro!("\\lx@pasj@lt@setcaption OptionalMatch:* []{}", sub[(_star, number, caption)] {
77    pasj_lt_remove_cr()?;
78    Ok(Invocation!(r"\noalign{\lx@longtable@caption@text{#1}{#2}}", vec![number, Some(caption)]))
79  });
80  DefMacro!("\\lx@pasj@lt@label Semiverbatim", sub[(label)] {
81    pasj_lt_remove_cr()?;
82    Ok(Invocation!(r"\noalign{\lx@pasj@lt@@label{#1}}", vec![Some(label)]))
83  });
84  Ok(())
85}

87/// pasj00.cls `\@lt@remove@cr` (:1254-1261): past spaces (`\@ifnextchar`), a `\\` is gobbled; any other token stays.
88/// Read where the class reads it, before its `\noalign` closes; the tokens read are not expanded.
89fn pasj_lt_remove_cr() -> Result<()> {
90  if let Some(next) = read_non_space()?
91    && !x_equals(&next, &T_CS!("\\\\"))
92  {
93    unread_one(next);
94  }
95  Ok(())
96}

98/// PASJ's frontmatter macros as the frontmatter. One `\author` lists every author, separated by commas and `\&`, its
99/// rows broken by `\\` (pasj00.cls:73-104), and the template puts each author's marks after the comma that ends the
100/// name (`Name,\altaffilmark{1}\orcid{…}`, pasj02's usage). The kernel would read each line after a `\\` as an
101/// affiliation (most of 0707.3867's 26 authors) and hand a mark after a comma to the next author (2504.06663,
102/// 0704.3654), so the binding splits the list itself ([`pasj_author_list`]). Documents write `\altaffiltext{n}{text}`
103/// and `\affil{text}` under all three versions, whatever pasj00's argument shapes; pasj00's `\email` takes an optional
104/// name; pasj02's `\altemailmark` only marks the author `\email` belongs to.
105fn pasj_frontmatter() -> Result<()> {
106  Let!("\\lx@pasj@author", "\\author");
107  DefMacro!("\\author[]{}", sub[(label, author)] {
108    let mut call = vec![T_CS!("\\lx@pasj@author")];
109    if let Some(label) = label {
110      call.push(T_OTHER!("["));
111      call.extend(label.unlist());
112      call.push(T_OTHER!("]"));
113    }
114    call.push(T_BEGIN!());
115    call.extend(pasj_author_list(&author.unlist())?);
116    call.push(T_END!());
117    Ok(Tokens::new(call))
118  });
119  DefMacro!("\\KeyWords{}", "\\lx@add@keywords{#1}");
120  Let!("\\kword", "\\KeyWords");
121  DefMacro!("\\affil{}", "\\lx@add@affiliation{#1}");
122  DefMacro!(
123    "\\altaffilmark Semiverbatim",
124    "\\lx@request@frontmatter@annotation[altaffil]{#1}"
125  );
126  // An empty text (a template's placeholder) makes no contact, which would stand as an empty creator (0704.3654).
127  DefMacro!("\\altaffiltext Semiverbatim {}", sub[(label, text)] {
128    if text.unlist_ref().iter().all(|t| t.get_catcode() == Catcode::SPACE) {
129      Ok(Tokens!())
130    } else {
131      Ok(Invocation!("\\lx@add@contact[role=altaffiliation,label={altaffil:#1}]{#2}", vec![Some(label), Some(text)]))
132    }
133  });
134  // pasj00.cls:68-69, the address in `\small\itshape`; the binding keeps the current font, not that size and shape
135  // (1001.5133)
136  DefMacro!("\\email[]{}", "\\lx@add@email{\\lx@address@textfont{#2}}");
137  def_macro_noop("\\altemailmark")?;
138  DefMacro!("\\orcid Semiverbatim", "\\lx@add@orcid{#1}");
139  DefMacro!("\\Received{}", "\\lx@add@date[role=received]{#1}");
140  DefMacro!("\\Accepted{}", "\\lx@add@date[role=accepted]{#1}");
141  Ok(())
142}

144/// The author marks a PASJ name carries, with their arguments: (control sequence, takes an optional argument, braced
145/// arguments) — `\altaffilmark{…}`, `\orcid{…}`, `\thanks[…]{…}` (pasj01.cls:2019), `\email[…]{…}`, `\altemailmark`.
146const PASJ_AUTHOR_MARKS: [(&str, bool, usize); 5] = [
147  ("\\altaffilmark", false, 1),
148  ("\\orcid", false, 1),
149  ("\\thanks", true, 1),
150  ("\\email", true, 1),
151  ("\\altemailmark", false, 0),
152];

154/// A PASJ author list as `\and`-separated names: it splits at depth-0 `,`, `\&`, `\and` and `\\` (with the `*` and a
155/// closed `[<skip>]` that follow it), gives a piece's leading marks (and math superscripts) back to the name before it,
156/// drops a piece's leading "and", and keeps no empty piece. The `\and`s: each written, as before (pasj00.cls:1878
157/// prints its own), and each held in a macro of the document's, which the separator query reads as the kernel's
158/// (`and_by_meaning`, Option C P2-5).
159fn pasj_author_list(toks: &[Token]) -> Result<Vec<Token>> {
160  let ands = and_runs(toks, &[T_CS!("\\and")])?.top_level(toks);
161  let mut pieces: Vec<Vec<Token>> = vec![Vec::new()];
162  let mut depth = 0usize;
163  let mut i = 0;
164  while i < toks.len() {
165    if let Some(end) = ands.get(&i) {
166      i = *end;
167      pieces.push(Vec::new());
168      continue;
169    }
170    let t = toks[i];
171    i += 1;
172    match t.get_catcode() {
173      Catcode::BEGIN => depth += 1,
174      Catcode::END => depth = depth.saturating_sub(1),
175      _ => {},
176    }
177    let separator = depth == 0 && (t == T_OTHER!(",") || t == T_CS!("\\&") || t == T_CS!("\\\\"));
178    if !separator {
179      pieces.last_mut().unwrap().push(t);
180      continue;
181    }
182    if t == T_CS!("\\\\") {
183      let mut k = i;
184      while k < toks.len() && toks[k].get_catcode() == Catcode::SPACE {
185        k += 1;
186      }
187      if k < toks.len() && toks[k] == T_OTHER!("*") {
188        k += 1;
189      }
190      if k < toks.len()
191        && toks[k] == T_OTHER!("[")
192        && let Some(close) = closing_bracket(&toks[k..])
193      {
194        k += close + 1;
195      }
196      i = k;
197    }
198    pieces.push(Vec::new());
199  }
200  let mut names: Vec<Vec<Token>> = Vec::new();
201  for mut piece in pieces {
202    let marks = leading_marks(&piece);
203    if marks > 0
204      && let Some(prev) = names.last_mut()
205    {
206      prev.extend(piece.drain(..marks));
207    }
208    let piece = strip_leading_and(piece);
209    if piece.iter().any(|t| t.get_catcode() != Catcode::SPACE) {
210      names.push(piece);
211    }
212  }
213  let mut list = Vec::new();
214  for (n, name) in names.into_iter().enumerate() {
215    if n > 0 {
216      list.push(T_CS!("\\and"));
217    }
218    list.extend(name);
219  }
220  Ok(list)
221}

223/// The length of a piece's leading run of author marks (with their arguments), math superscripts (`$^{1}$`) and
224/// spaces — 0 when it starts with none: the marks that follow the comma ending the previous name.
225fn leading_marks(piece: &[Token]) -> usize {
226  let mut end = 0;
227  let mut i = 0;
228  while i < piece.len() {
229    let t = piece[i];
230    match t.get_catcode() {
231      Catcode::SPACE => i += 1,
232      Catcode::MATH => {
233        let Some(len) = piece[i + 1..]
234          .iter()
235          .position(|t| t.get_catcode() == Catcode::MATH)
236        else {
237          break;
238        };
239        if piece.get(i + 1).map(|t| t.get_catcode()) != Some(Catcode::SUPER) {
240          break;
241        }
242        i += len + 2;
243        end = i;
244      },
245      Catcode::CS => {
246        let Some((_, optional, braced)) = PASJ_AUTHOR_MARKS
247          .iter()
248          .find(|(cs, ..)| t.with_str(|s| s == *cs))
249        else {
250          break;
251        };
252        i += 1;
253        // Exactly its own arguments: a following brace group belongs to the next name (`{\'A}lvaro`).
254        let skip_spaces = |mut k: usize| {
255          while k < piece.len() && piece[k].get_catcode() == Catcode::SPACE {
256            k += 1;
257          }
258          k
259        };
260        if *optional {
261          let k = skip_spaces(i);
262          if piece.get(k) == Some(&T_OTHER!("[")) {
263            match closing_bracket(&piece[k..]) {
264              Some(close) => i = k + close + 1,
265              None => return end,
266            }
267          }
268        }
269        for _ in 0..*braced {
270          let k = skip_spaces(i);
271          match piece.get(k) {
272            Some(b) if b.get_catcode() == Catcode::BEGIN => {
273              let mut level = 0usize;
274              i = k;
275              while i < piece.len() {
276                match piece[i].get_catcode() {
277                  Catcode::BEGIN => level += 1,
278                  Catcode::END => {
279                    level -= 1;
280                    if level == 0 {
281                      i += 1;
282                      break;
283                    }
284                  },
285                  _ => {},
286                }
287                i += 1;
288              }
289            },
290            // An undelimited argument is one token (`\altaffilmark1`).
291            Some(_) => i = k + 1,
292            None => return end,
293          }
294        }
295        end = i;
296      },
297      _ => break,
298    }
299  }
300  end
301}

303/// The offset of the `]` closing the `[` that `toks` starts with, at brace depth 0 (`\thanks[{a]b}]{…}`).
304fn closing_bracket(toks: &[Token]) -> Option<usize> {
305  let mut depth = 0usize;
306  for (n, t) in toks.iter().enumerate() {
307    match t.get_catcode() {
308      Catcode::BEGIN => depth += 1,
309      Catcode::END => depth = depth.saturating_sub(1),
310      _ if depth == 0 && *t == T_OTHER!("]") => return Some(n),
311      _ => {},
312    }
313  }
314  None
315}

317/// A piece without a leading "and" — the last name of a comma list ("…, and D. Four").
318fn strip_leading_and(piece: Vec<Token>) -> Vec<Token> {
319  let start = piece
320    .iter()
321    .position(|t| t.get_catcode() != Catcode::SPACE)
322    .unwrap_or(piece.len());
323  let word: String = piece[start..]
324    .iter()
325    .take(3)
326    .map(|t| t.to_string())
327    .collect();
328  let after = piece.get(start + 3);
329  if word == "and" && after.is_some_and(|t| t.get_catcode() == Catcode::SPACE) {
330    piece[start + 3..].to_vec()
331  } else {
332    piece
333  }
334}
