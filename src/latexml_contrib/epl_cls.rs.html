284/// An `\institute` list as one line per entry: an entry begins at its `\inst` mark (or epl2's `\[n]`, :726), a `\\`
285/// within it (`… Blegdamsvej 17,\\ DK-2100 Copenhagen \O, Denmark\\ \inst{3}…`, 1803.06254) is a break in the title
286/// block's lines, not a new entry, and an entry may follow the one before it on its line (`… e-mail:
287/// ziaja@mail.desy.de \inst{2} …`). The spaces after a mark, and a `\\`'s star and space (`\\[2pt]`), are not the
288/// entry's; the break is a grouped `{\\}`, which does not end the entry.
289fn entries_at_marks(list: Tokens) -> Tokens {
290  let inst = T_CS!("\\inst");
291  let newline = T_CS!("\\\\");
292  let tokens = bracket_marks_as_inst(list.unlist());
293  let mut entries: Vec<Token> = Vec::with_capacity(tokens.len());
294  let mut depth = 0usize;
295  let mut started = false;
296  // after a mark: reading its argument, then past it (the spaces next are dropped)
297  let (mut in_mark, mut after_mark) = (false, false);
298  let mut i = 0;
299  while i < tokens.len() {
300    let t = tokens[i];
301    i += 1;
302    let catcode = t.get_catcode();
303    if depth == 0
304      && catcode == Catcode::SPACE
305      && (after_mark
306        || !started
307        || entries
308          .last()
309          .is_some_and(|last| last.get_catcode() == Catcode::SPACE))
310    {
311      continue;
312    }
313    after_mark = false;
314    match catcode {
315      Catcode::BEGIN => depth += 1,
316      Catcode::END => {
317        depth = depth.saturating_sub(1);
318        if depth == 0 && in_mark {
319          in_mark = false;
320          after_mark = true;
321        }
322      },
323      _ => {},
324    }
325    if depth == 0 && t == newline {
326      // a line break within the entry, without its `*` and `[<space>]`
327      if tokens.get(i) == Some(&T_OTHER!("*")) {
328        i += 1;
329      }
330      let next = tokens[i..]
331        .iter()
332        .position(|t| t.get_catcode() != Catcode::SPACE)
333        .map_or(i, |n| i + n);
334      if tokens.get(next) == Some(&T_OTHER!("["))
335        && let Some(close) = tokens[next..].iter().position(|t| *t == T_OTHER!("]"))
336      {
337        i = next + close + 1;
338      }
339      while entries
340        .last()
341        .is_some_and(|last| last.get_catcode() == Catcode::SPACE)
342      {
343        entries.pop();
344      }
345      entries.extend([T_BEGIN!(), newline, T_END!()]);
346      // (nor the spaces after it)
347      after_mark = true;
348      continue;
349    }
350    if depth == 0 && t == inst {
351      if started {
352        while entries
353          .last()
354          .is_some_and(|last| last.get_catcode() == Catcode::SPACE)
355        {
356          entries.pop();
357        }
358        entries.push(newline);
359      }
360      in_mark = true;
361    }
362    started = true;
363    entries.push(t);
364  }
365  Tokens::new(entries)
366}

368/// Whether `list` marks its entries (`\inst{n}`, `\[n]`).
369fn has_inst_marks(list: &Tokens) -> bool {
370  list
371    .unlist_ref()
372    .iter()
373    .any(|t| *t == T_CS!("\\inst") || *t == T_CS!("\\["))
374}

376/// epl2's other spelling of the mark, `\[n]` for `\inst{n}` (:726 `\def\[##1]{\inst{##1}}`), as `\inst{n}`.
377fn bracket_marks_as_inst(tokens: Vec<Token>) -> Vec<Token> {
378  let mut out = Vec::with_capacity(tokens.len());
379  let mut i = 0;
380  while i < tokens.len() {
381    if tokens[i] == T_CS!("\\[")
382      && let Some(close) = tokens[i + 1..].iter().position(|t| *t == T_OTHER!("]"))
383    {
384      out.push(T_CS!("\\inst"));
385      out.push(T_BEGIN!());
386      out.extend_from_slice(&tokens[i + 1..i + 1 + close]);
387      out.push(T_END!());
388      i += close + 2;
389      continue;
390    }
391    out.push(tokens[i]);
392    i += 1;
393  }
394  out
395}

397/// `tokens` with each note that is only a label and an `\email` of an address (`\thanks{E-mail: \email{x@y}}`,
398/// `\footnote{\email{E-mail:x@y}}`) replaced by that email, its label the contact's name (`\lx@epl@noted@address`):
399/// among the names it is that author's (1110.0608, 1212.3881), in an institute the institute's authors' (1107.1575).
400fn email_notes_as_emails(names: Tokens) -> Tokens {
401  let tokens = names.unlist();
402  let mut out = Vec::with_capacity(tokens.len());
403  let mut depth = 0usize;
404  let mut i = 0;
405  while i < tokens.len() {
406    if (tokens[i] == T_CS!("\\thanks") || tokens[i] == T_CS!("\\footnote"))
407      && let Some(open) = tokens[i + 1..]
408        .iter()
409        .position(|t| t.get_catcode() != Catcode::SPACE)
410        .map(|n| i + 1 + n)
411      && tokens[open].get_catcode() == Catcode::BEGIN
412      && let Some(close) = group_end(&tokens, open)
413      && let Some((label, address)) = lone_email(&tokens[open + 1..close])
414    {
415      out.push(T_SPACE!());
416      if label.is_empty() {
417        out.push(T_CS!("\\email"));
418      } else {
419        out.push(T_CS!("\\lx@epl@noted@address"));
420        out.push(T_BEGIN!());
421        out.extend(label);
422        out.push(T_END!());
423      }
424      out.push(T_BEGIN!());
425      out.extend(address);
426      out.push(T_END!());
427      i = close + 1;
428      continue;
429    }
430    match tokens[i].get_catcode() {
431      Catcode::BEGIN => depth += 1,
432      Catcode::END => depth = depth.saturating_sub(1),
433      _ => {},
434    }
435    // a bare `\email` after a comma (`\author{V. V. Kisil, \email{kisilv@…}}`, 0907.0855) stands among the names
436    // without it
437    if depth == 0 && tokens[i] == T_CS!("\\email") {
438      while out
439        .last()
440        .is_some_and(|t: &Token| t.get_catcode() == Catcode::SPACE)
441      {
442        out.pop();
443      }
444      if out.last() == Some(&T_OTHER!(",")) {
445        out.pop();
446      }
447      out.push(T_SPACE!());
448      // an address in markup (`\email{\href{mailto:…}{…}}`, 0907.0855), which the kernel reads as no address: the
449      // author's email contact as written
450      if let Some(open) = tokens[i + 1..]
451        .iter()
452        .position(|t| t.get_catcode() != Catcode::SPACE)
453        .map(|n| i + 1 + n)
454        && tokens[open].get_catcode() == Catcode::BEGIN
455        && let Some(close) = group_end(&tokens, open)
456        && tokens[open + 1..close]
457          .iter()
458          .any(|t| t.get_catcode() == Catcode::CS)
459      {
460        out.push(T_CS!("\\lx@epl@address@contact"));
461        out.extend_from_slice(&tokens[open..=close]);
462        i = close + 1;
463        continue;
464      }
465    }
466    out.push(tokens[i]);
467    i += 1;
468  }
469  Tokens::new(out)
470}

472/// The index of the group's closing brace, the group opening at `open`.
473fn group_end(tokens: &[Token], open: usize) -> Option<usize> {
474  let mut depth = 0usize;
475  for (j, t) in tokens.iter().enumerate().skip(open) {
476    match t.get_catcode() {
477      Catcode::BEGIN => depth += 1,
478      Catcode::END => {
479        depth -= 1;
480        if depth == 0 {
481          return Some(j);
482        }
483      },
484      _ => {},
485    }
486  }
487  None
488}

490/// The label and address of a note that holds an `\email` and a label only (no other command, at most a few words):
491/// `E-mail: \email{x@y}`, or `\email{E-mail:x@y}` (1212.3881).
492fn lone_email(note: &[Token]) -> Option<(Vec<Token>, Vec<Token>)> {
493  let at = note.iter().position(|t| *t == T_CS!("\\email"))?;
494  let open = at
495    + 1
496    + note[at + 1..]
497      .iter()
498      .position(|t| t.get_catcode() != Catcode::SPACE)?;
499  if note[open].get_catcode() != Catcode::BEGIN {
500    return None;
501  }
502  let close = group_end(note, open)?;
503  let mut label: Vec<Token> = Vec::new();
504  for t in note[..at].iter().chain(&note[close + 1..]) {
505    if t.get_catcode() == Catcode::CS
506      || t.get_catcode() == Catcode::BEGIN
507      || t.get_catcode() == Catcode::MATH
508    {
509      return None;
510    }
511  }
512  label.extend(note[..at].iter().copied());
513  // a label inside the command's argument, before its address
514  let inner = &note[open + 1..close];
515  let (inner_label, address) = match inner.iter().rposition(|t| *t == T_OTHER!(":")) {
516    Some(colon) => (&inner[..=colon], &inner[colon + 1..]),
517    None => (&inner[..0], inner),
518  };
519  label.extend(inner_label.iter().copied());
520  let trim = |tokens: &[Token]| -> Vec<Token> {
521    let start = tokens
522      .iter()
523      .position(|t| t.get_catcode() != Catcode::SPACE)
524      .unwrap_or(tokens.len());
525    let end = tokens
526      .iter()
527      .rposition(|t| t.get_catcode() != Catcode::SPACE)
528      .map_or(start, |e| e + 1);
529    tokens[start..end].to_vec()
530  };
531  let (label, address) = (trim(&label), trim(address));
532  // an address, one word; a label of a few words at most
533  if address.is_empty()
534    || address.iter().any(|t| t.get_catcode() == Catcode::SPACE)
535    || label.len() > 24
536  {
537    return None;
538  }
539  Some((label, address))
540}
