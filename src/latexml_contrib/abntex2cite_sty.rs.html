1//! abntex2cite.sty — ABNT citations (Brazilian standard), the bibliography end.
2//!
3//! The raw package is loaded with the document's options (`num`/`alf`,
4//! `foot`, …; abntex2cite.sty:158-174); its `\bibliography` redefinition is
5//! refused by the kernel's lock and its `\cite` handed back to the kernel:
6//! abntex2cite.sty:375 redefines `\bibliography` to write `\bibdata` to the `.aux` and
7//! `\@input@{\jobname.bbl}` — a file only a bibtex run produces, which LaTeXML
8//! never has, so the reference list vanished at 0 errors (abntex2cite's manual
9//! is half bibliography; abntex2cite-alf and unbtex share it). The native
10//! `\bibliography` (latex_constructs.pool.ltxml:3895; sect11.rs) prefers a
11//! shipped `.bbl` and otherwise runs the `.bib` session at post
12//! (`\lx@ifusebbl`), and the `.bib` list is live only here, in the
13//! command's argument — the same interception Perl makes for bibunits'
14//! `\bibliography` wrapper (bibunits.sty.ltxml; OXIDIZED_DESIGN_DIVERGENCES
15//! #87); since batch 56cx the kernel macro is locked against that
16//! redefinition (OXIDIZED_DESIGN_DIVERGENCES #236). Perl has no abntex2cite
17//! binding and loses the list too. The
18//! package's own `\cite` (abntex2cite.sty:748/:862, a `\DeclareRobustCommand`
19//! resolving numbers from the `.aux`) typeset `(??)` and registered no
20//! citation, so no entry was ever selected; `\cite` and `\citeonline` (the
21//! textual form) keep the kernel's citation constructor, which the
22//! bibliography stage resolves.
23//!
24//! 57cm (rc131 root-cause of abntex2cite, 80.5 % recall, bibliography-only):
25//! - the package's `\bibliographystyle` (:409-417) only writes to the `.aux`, and its
26//!   `\AtEndDocument` default (:420-424) goes through it, so no style ever reached the
27//!   bibliography (`bibstyle` unset: URLs printed as "Link"). It now forwards to the
28//!   kernel's, and the package's default, `abntex2-\AbntCitetype`, is recorded at load;
29//! - abntex2's own `.bib` fields, which its `.bst`s print (abntex2-num.bst:26-49,
30//!   :585-602, :997-1000, :1399-1400, :1645), were kept as unprinted `ltx:bib-data`;
31//! - `\citeyear`/`\citeauthoronline` (:927, :1019) read values only a real BibTeX run
32//!   leaves in the `.aux` and printed "??"; they are the kernel's bibrefs now, as
33//!   natbib's (natbib_sty.rs), and in `alf` mode `\citeauthor` too (in `num` mode the
34//!   package's own, :1343-1344, builds on them). `\citetext` (a whole reference)
35//!   still prints "??": no bibref show mode prints a whole entry yet.
36//!
37//! 57cm.1 (review): a shipped `.bbl` carries each entry's author and year in
38//! `\abntrefinfo` (abntex2-num.bst:1375), which now tags the bibitem, so a bibref shows them
39//! there too; in `alf` mode `\citen`/`\citenum` are the textual form (:722-723; `num` mode has
40//! neither); `alf` citations of several works are separated by ";" (`\ABCIcitecolondefault`, :453). A
41//! `\bibliographystyle` inside a group, or before the package, loses to the recorded
42//! default (BibTeX keeps the first `\bibstyle` line).
43use latexml_package::prelude::*;
44
45#[rustfmt::skip]
46LoadDefinitions!({
47  RawTeX!(r"\let\lx@abnt@orig@cite\cite\let\lx@abnt@kernel@bibliographystyle\bibliographystyle");
48  InputDefinitions!("abntex2cite", noltxml => true, extension => Some(Cow::Borrowed("sty")),
49    handleoptions => true);
50  // `\bibliography` itself is the kernel's, locked against the package's
51  // `\@input{\jobname.bbl}` redefinition since batch 56cx (sect11.rs).
52  RawTeX!(r"\let\cite\lx@abnt@orig@cite\let\citeonline\lx@abnt@orig@cite");
53  // The style reaches the bibliography: the kernel's `\bibliographystyle`, then the
54  // package's flag (so its end-of-document default still yields to a user choice); the
55  // package's default recorded now, which a later `\bibliographystyle` overrides.
56  RawTeX!(r"\def\bibliographystyle#1{\lx@abnt@kernel@bibliographystyle{#1}\setboolean{ABCIbibtexstyleused}{true}}\lx@abnt@kernel@bibliographystyle{abntex2-\AbntCitetype}");
57  // abntex2's fields, printed as the `.bst`s print them: credits, notes, the part cited,
58  // illustrations, dimensions, the reprint's text — notes; the access date — a date.
59  RawTeX!(concat!(
60    r"\@namedef{bib@field@default@furtherresp}{\bib@@field{ltx:bib-note}[role=furtherresp]}",
61    r"\@namedef{bib@field@default@abnt-note}{\bib@@field{ltx:bib-note}[role=abnt-note]}",
62    r"\@namedef{bib@field@default@section}{\bib@@field{ltx:bib-note}[role=section]}",
63    // an empty `illustrated` is "il." (abntex2-num.bst:597-605)
64    r"\@namedef{bib@field@default@illustrated}#1{\bib@@field{ltx:bib-note}[role=illustrated]",
65    r"{\if\relax\detokenize{#1}\relax il.\else#1\fi}}",
66    r"\@namedef{bib@field@default@dimensions}{\bib@@field{ltx:bib-note}[role=dimensions]}",
67    r"\@namedef{bib@field@default@reprinted-text}{\bib@@field{ltx:bib-note}[role=reprinted-text]}",
68    r"\@namedef{bib@field@default@urlaccessdate}{\bib@@field{ltx:bib-date}[role=accessed]}"));
69  // The citation forms that read BibTeX's `.aux` values are the kernel's bibrefs.
70  DefMacro!("\\citeyear Semiverbatim", "\\lx@@cite[citeyear]{\\@@bibref{Year}{#1}{}{}}");
71  DefMacro!("\\citeauthoronline Semiverbatim",
72    "{\\lx@abnt@citesep\\lx@@cite[citeauthor]{\\@@bibref{Authors}{#1}{}{}}}");
73  // abntex2-num.bst:1375 writes `\abntrefinfo{EXPL}{IMPL}{YEAR}` after every `\bibitem` of a
74  // `.bbl`: the author as a textual citation names it, as a parenthetical one does, and the year
75  // (abntex2cite.sty:557-575 records them in the `.aux` under `\abntnextkey`, which the package's
76  // own `\bibitem` sets — undefined here). The author and year become the bibitem's
77  // `authors`/`year` tags, as natbib's `\citeauthoryear` label does (`\NAT@@wrout`), so a
78  // bibref shows them: without, `\citeyear` showed the refnum ("em 2, 2."). The parenthetical
79  // (upper-case) form is presentation. A bibitem that already carries them keeps them: the
80  // `\abntrefinfo` after `\hiddenbibitem` (abntex2-num.bst:1379-1390, an `@hidden` entry, which
81  // opens no bibitem) would otherwise tag the entry before it (SYNC_STATUS residual). No corpus
82  // witness yet (TeX Live ships the manuals' `.bib`, not a `.bbl`): the fixture's bibtex-made `.bbl`;
83  // guard `abntex2cite_bbl_carries_author_and_year`.
84  DefConstructor!("\\abntrefinfo{}{}{}", sub[document, args, _props] {
85    let here = document.get_node().clone();
86    if let Some(item) = document.findnode("ancestor-or-self::ltx:bibitem", Some(&here))
87      && let Some(tags) = document.findnode("ltx:tags", Some(&item))
88      && document.findnode("ltx:tag[@role='authors']", Some(&tags)).is_none()
89    {
90      document.set_node(&tags);
91      let mut inserted = Ok(());
92      for (role, arg) in [("authors", &args[0]), ("year", &args[2])] {
93        if let Some(arg) = arg.as_ref() && inserted.is_ok() {
94          inserted = document
95            .insert_element("ltx:tag", vec![arg], Some(string_map!("role" => role)))
96            .map(|_| ());
97        }
98      }
99      document.set_node(&here);
100      inserted?;
101    }
102  });
103  // The author-date system (`alf`, NBR 10520): `\cite` is parenthetical, "(FARIA, 1994, p.
104  // 225)", and `\citeonline` textual, "Faria (1994, p. 225)" — the kernel's refnum form
105  // printed "[Faria (1994)]" for both; `\citeauthor` is the author alone. Several works in one
106  // `\cite` are separated by ";" (`\ABCIcitecolondefault`, :453, :637), in one `\citeonline` by
107  // "," (:696); an author list by ";" (:995-999, :1037-1043; a repeated author is printed again
108  // here, SYNC_STATUS). Chosen by the package's own mode test (abntex2cite.sty:600).
109  DefPrimitive!("\\lx@abnt@citesep", {
110    assign_value("CITE_SEPARATOR", Stored::Token(T_OTHER!(";")), None);
111  });
112  DefMacro!("\\lx@abnt@alf@cite[] Semiverbatim", sub[(post, keys)] {
113    let phrase = Invocation!(T_CS!("\\@@citephrase"), vec![Tokens::new(vec![T_OTHER!(","), T_SPACE!()])]);
114    let bibref = Invocation!(T_CS!("\\@@bibref"),
115      vec![Tokens::new(Explode!("AuthorsPhrase1Year")), keys, phrase, Tokens!()]);
116    let mut body = Tokenize!("(").unlist();
117    body.extend(bibref.unlist());
118    if let Some(post) = post.filter(|p| !p.is_empty()) {
119      body.extend(Tokenize!(",").unlist());
120      body.push(T_SPACE!());
121      body.extend(post.unlist());
122    }
123    body.extend(Tokenize!(")").unlist());
124    let mut out = vec![T_BEGIN!(), T_CS!("\\lx@abnt@citesep")];
125    out.extend(Invocation!(T_CS!("\\lx@@cite"), vec![Tokens::new(Explode!("citep")), Tokens::new(body)]).unlist());
126    out.push(T_END!());
127    Ok(Tokens::new(out))
128  });
129  DefMacro!("\\lx@abnt@alf@citeonline[] Semiverbatim", sub[(post, keys)] {
130    let phrase1 = Invocation!(T_CS!("\\@@citephrase"), vec![Tokenize!("(")]);
131    let mut close = Vec::new();
132    if let Some(post) = post.filter(|p| !p.is_empty()) {
133      close.extend(Tokenize!(",").unlist());
134      close.push(T_SPACE!());
135      close.extend(post.unlist());
136    }
137    close.extend(Tokenize!(")").unlist());
138    let phrase2 = Invocation!(T_CS!("\\@@citephrase"), vec![Tokens::new(close)]);
139    let bibref = Invocation!(T_CS!("\\@@bibref"),
140      vec![Tokens::new(Explode!("Authors Phrase1YearPhrase2")), keys, phrase1, phrase2]);
141    Ok(Invocation!(T_CS!("\\lx@@cite"), vec![Tokens::new(Explode!("citet")), bibref]))
142  });
143  DefMacro!("\\lx@abnt@alf@citeauthor Semiverbatim",
144    "{\\lx@abnt@citesep\\lx@@cite[citeauthor]{\\@@bibref{Authors}{#1}{}{}}}");
145  // `\citen`/`\citenum` are the alf branch's aliases of its textual `\citeonline` (:722-723; num
146  // mode has neither — pdflatex stops there).
147  RawTeX!(r"\ifx\AbntCitetype\AbntCitetypeALF
148    \let\cite\lx@abnt@alf@cite \let\citeonline\lx@abnt@alf@citeonline
149    \let\citeauthor\lx@abnt@alf@citeauthor \let\citen\citeonline\let\citenum\citeonline\fi");
150});
