1//! Pygments-color support for `minted` from a committed `_minted/` frozencache
2//! (surpass-Perl; `docs/parity/OXIDIZED_DESIGN_DIVERGENCES.md` #157).
3//!
4//! Perl LaTeXML has no `minted` binding and errors out on the environment; our
5//! binding already routes minted through the `listings` substrate (bold-black
6//! keywords, no color). When a paper is built with
7//! `\usepackage[frozencache]{minted}`, the `_minted/` directory (a sibling of
8//! the main `.tex`) carries Pygments' output on disk as plain `\textcolor`:
9//!
10//! * `default.style.minted` — `\@namedef{PYG@tok@<class>}{…}` per token class,
11//!   each setting some of a color (`\def\PYG@tc##1{\textcolor[rgb]{r,g,b}{##1}}`),
12//!   bold (`\let\PYG@bf=\textbf`), italic (`\let\PYG@it=\textit`).
13//! * `<MD5>.highlight.minted` — one per highlighted snippet: a `MintedVerbatim`
14//!   body of `\PYG{<tokclass>}{<text>}` runs interleaved with literal spaces and
15//!   `\PYGZ*` escapes.
16//!
17//! Rather than replicate minted's fragile MD5 keying, we **content-match**: each
18//! highlight file, with `\PYG` unwrapped and `\PYGZ*` resolved, yields the exact
19//! plain code of a source snippet. We normalize a minted block's raw body the
20//! same way and look it up. On a hit we emit Pygments-colored listing lines
21//! (reusing the listings `\@lst@startline`/`\@lst@endline`/`\@listingGroup`
22//! constructors and xcolor's `\textcolor`); on a miss the caller keeps the exact
23//! current (uncolored) `listings` path. When no `_minted/` exists the whole
24//! feature is a no-op, so non-frozencache papers are unaffected.
25//!
26//! Reading the host source tree's `_minted/` is in scope (it is like reading a
27//! `.sty`, per CLAUDE.md — the ban is on reading latexml-oxide's *own* resources).
28//!
29//! Witness: arXiv:2605.03143 (`\begin{minted}{ocaml|python}` blocks in
30//! `sections/01-introduction.tex`, `02-a-taste-of-pact.tex`, `03-memo.tex`).

37/// A resolved token style: an optional `rgb` triple string (e.g.
38/// `"0.00,0.50,0.00"`, passed verbatim to `\textcolor[rgb]{…}`), plus bold /
39/// italic flags. Mirrors what `default.style.minted`'s `PYG@tok@<class>` sets.
40#[derive(Clone, Default)]
41struct Style {
42  rgb:    Option<String>,
43  bold:   bool,
44  italic: bool,
45}

47/// One `\PYG{tok}{text}` run (or a literal run with `tok = None`).
48struct Seg {
49  tok:  Option<String>,
50  text: String,
51}

53type Line = Vec<Seg>;

55/// A loaded `_minted/` cache, keyed on the resolved directory path so a later
56/// document on the same thread (a different `_minted/`, or none) never reuses it.
57struct CacheData {
58  dir:          String,
59  styles:       HashMap<String, Style>,
60  /// normalized plaincode → the highlighted lines that produced it.
61  by_plaincode: HashMap<String, Vec<Line>>,
62}

65  static CACHE: RefCell<Option<Rc<CacheData>>> = const { RefCell::new(None) };

68/// Map `\PYGZxx` escape names to the literal character they stand for
69/// (`default.style.minted` L81-99).
70fn pygz_char(name: &str) -> Option<&'static str> {
71  Some(match name {
72    "PYGZbs" => "\\",
73    "PYGZus" => "_",
74    "PYGZob" => "{",
75    "PYGZcb" => "}",
76    "PYGZca" => "^",
77    "PYGZam" => "&",
78    "PYGZlt" => "<",
79    "PYGZgt" => ">",
80    "PYGZsh" => "#",
81    "PYGZpc" => "%",
82    "PYGZdl" => "$",
83    "PYGZhy" => "-",
84    "PYGZsq" => "'",
85    "PYGZdq" => "\"",
86    "PYGZti" => "~",
87    "PYGZat" => "@",
88    "PYGZlb" => "[",
89    "PYGZrb" => "]",
90    _ => return None,
91  })
92}

94/// Resolve every `\PYGZxx` / `\PYGZxx{}` occurrence in `s` to its literal char.
95fn resolve_pygz(s: &str) -> String {
96  static RE: Lazy<Regex> = Lazy::new(|| Regex::new(r"\\(PYGZ[a-z][a-z])(\{\})?").unwrap());
97  RE.replace_all(s, |c: &regex::Captures| pygz_char(&c[1]).unwrap_or(""))
98    .into_owned()
99}

101/// Read a `{…}`-balanced group. `s` must start with `{`. Returns
102/// `(inner, rest_after_close)`. `\PYGZob{}`/`\PYGZcb{}` contribute balanced
103/// `{}` pairs, so brace counting stays correct.
104fn read_group(s: &str) -> Option<(String, &str)> {
105  let mut depth = 0i32;
106  for (idx, ch) in s.char_indices() {
107    match ch {
108      '{' => depth += 1,
109      '}' => {
110        depth -= 1;
111        if depth == 0 {
112          return Some((s[1..idx].to_string(), &s[idx + 1..]));
113        }
114      },
115      _ => {},
116    }
117  }
118  None
119}

121/// Parse one raw highlight line into `\PYG{tok}{text}` and literal segments.
122fn parse_line(line: &str) -> Line {
123  let mut segs: Line = Vec::new();
124  let mut lit = String::new();
125  let mut rest = line;
126  while !rest.is_empty() {
127    if let Some(after) = rest.strip_prefix("\\PYG{") {
128      // `tok}{content}` — tok never contains a brace.
129      if let Some(tok_end) = after.find('}') {
130        let tok = &after[..tok_end];
131        let after_tok = &after[tok_end + 1..];
132        if let Some((content, rest2)) = read_group(after_tok) {
133          if !lit.is_empty() {
134            segs.push(Seg {
135              tok:  None,
136              text: std::mem::take(&mut lit),
137            });
138          }
139          segs.push(Seg {
140            tok:  Some(tok.to_string()),
141            text: resolve_pygz(&content),
142          });
143          rest = rest2;
144          continue;
145        }
146      }
147      // Malformed \PYG — treat the backslash literally and move on.
148      lit.push('\\');
149      rest = &rest[1..];
150    } else if let Some(after) = rest.strip_prefix("\\PYGZ") {
151      // A bare escape outside \PYG (e.g. in an escapeinside remnant).
152      if after.len() >= 2 {
153        let name = format!("PYGZ{}", &after[..2]);
154        if let Some(ch) = pygz_char(&name) {
155          lit.push_str(ch);
156          rest = &after[2..];
157          rest = rest.strip_prefix("{}").unwrap_or(rest);
158          continue;
159        }
160      }
161      lit.push('\\');
162      rest = &rest[1..];
163    } else {
164      let c = rest.chars().next().unwrap();
165      lit.push(c);
166      rest = &rest[c.len_utf8()..];
167    }
168  }
169  if !lit.is_empty() {
170    segs.push(Seg { tok: None, text: lit });
171  }
172  segs
173}

175/// The concatenated plain text of a line.
176fn line_text(line: &Line) -> String { line.iter().map(|s| s.text.as_str()).collect() }

178/// Drop leading/trailing all-whitespace lines (in place).
179fn strip_blank_edges(lines: &mut Vec<Line>) {
180  while lines
181    .first()
182    .is_some_and(|l| line_text(l).trim().is_empty())
183  {
184    lines.remove(0);
185  }
186  while lines.last().is_some_and(|l| line_text(l).trim().is_empty()) {
187    lines.pop();
188  }
189}

191/// Normalized-plaincode key for a set of highlight lines: each line's trailing
192/// whitespace trimmed, joined by `\n` (blank edges already stripped).
193fn lines_key(lines: &[Line]) -> String {
194  lines
195    .iter()
196    .map(|l| line_text(l).trim_end().to_string())
197    .collect::<Vec<_>>()
198    .join("\n")
199}
200
201/// Normalize a minted block's raw source body to the same key form: rstrip each
202/// line, drop leading/trailing blank lines.
203fn normalize_source(text: &str) -> String {
204  let mut v: Vec<String> = text.split('\n').map(|l| l.trim_end().to_string()).collect();
205  while v.first().is_some_and(|l| l.is_empty()) {
206    v.remove(0);
207  }
208  while v.last().is_some_and(|l| l.is_empty()) {
209    v.pop();
210  }
211  v.join("\n")
212}

214/// Extract the `MintedVerbatim` body of a highlight file → its lines.
215fn parse_highlight(text: &str) -> Option<Vec<Line>> {
216  let begin = text.find("\\begin{MintedVerbatim}")?;
217  let body_start = text[begin..].find('\n')? + begin + 1;
218  let end = text.rfind("\\end{MintedVerbatim}")?;
219  if end < body_start {
220    return None;
221  }
222  let body = &text[body_start..end];
223  let mut lines: Vec<Line> = body.split('\n').map(parse_line).collect();
224  strip_blank_edges(&mut lines);
225  Some(lines)
226}

228/// Parse `default.style.minted` into a `tokclass → Style` map.
229fn parse_styles(text: &str) -> HashMap<String, Style> {
230  static NAME_RE: Lazy<Regex> =
231    Lazy::new(|| Regex::new(r"\\@namedef\{PYG@tok@([A-Za-z0-9]+)\}").unwrap());
232  static COLOR_RE: Lazy<Regex> =
233    Lazy::new(|| Regex::new(r"\\textcolor\[rgb\]\{([0-9.,]+)\}").unwrap());
234  let mut map: HashMap<String, Style> = HashMap::default();
235  for line in text.lines() {
236    if let Some(caps) = NAME_RE.captures(line) {
237      let class = caps[1].to_string();
238      let style = Style {
239        rgb:    COLOR_RE.captures(line).map(|c| c[1].to_string()),
240        bold:   line.contains(r"\PYG@bf=\textbf"),
241        italic: line.contains(r"\PYG@it=\textit"),
242      };
243      map.insert(class, style);
244    }
245  }
246  map
247}

249/// Load (and memoize) the `_minted/` cache for the current document. Keyed on
250/// the resolved directory, so repeated blocks reuse it and a different document
251/// reloads. Returns `None` when no frozencache is present.
252fn load_cache() -> Option<Rc<CacheData>> {
253  let style_path = find_file("_minted/default.style.minted", None)?;
254  let dir = Path::new(&style_path)
255    .parent()?
256    .to_string_lossy()
257    .into_owned();
258
259  if let Some(cached) = CACHE.with(|c| c.borrow().as_ref().filter(|d| d.dir == dir).cloned()) {
260    return Some(cached);
261  }
262
263  let styles = std::fs::read_to_string(&style_path)
264    .ok()
265    .map(|s| parse_styles(&s))?;
266  let mut by_plaincode: HashMap<String, Vec<Line>> = HashMap::default();
267  if let Ok(entries) = std::fs::read_dir(&dir) {
268    for entry in entries.flatten() {
269      let name = entry.file_name();
270      let name = name.to_string_lossy();
271      if !name.ends_with(".highlight.minted") {
272        continue;
273      }
274      if let Ok(text) = std::fs::read_to_string(entry.path())
275        && let Some(lines) = parse_highlight(&text)
276        && !lines.is_empty()
277      {
278        by_plaincode.entry(lines_key(&lines)).or_insert(lines);
279      }
280    }
281  }
282
283  let data = Rc::new(CacheData { dir, styles, by_plaincode });
284  CACHE.with(|c| *c.borrow_mut() = Some(data.clone()));
285  Some(data)
286}

288/// True when a `_minted/` frozencache is present for the current document.
289pub fn cache_available() -> bool { load_cache().is_some() }

291/// Resolve a (possibly compound, e.g. `n+nf`, `l+m+mf`) token class to a style.
292/// Mirrors `\PYG@toks`: each `+`-separated sub-class is applied in order — color
293/// takes the LAST sub-class that sets one, bold/italic accumulate.
294fn resolve_style(cache: &CacheData, tok: &Option<String>) -> Style {
295  let mut st = Style::default();
296  if let Some(t) = tok {
297    for sub in t.split('+') {
298      if let Some(s) = cache.styles.get(sub) {
299        if s.rgb.is_some() {
300          st.rgb = s.rgb.clone();
301        }
302        st.bold |= s.bold;
303        st.italic |= s.italic;
304      }
305    }
306  }
307  st
308}

310/// Map a single code character to its emit tokens: TeX-special chars → the
311/// text-command escape (mirrors listings' `lst_char_mapping`, upquote off);
312/// everything else → a literal `OTHER` char (as the listings char path does).
313fn map_char(c: char) -> Vec<Token> {
314  let cs: &'static str = match c {
315    '#' => "\\#",
316    '$' => "\\textdollar",
317    '&' => "\\&",
318    '\'' => "\\textquoteright",
319    '*' => "\\textasteriskcentered",
320    '<' => "\\textless",
321    '>' => "\\textgreater",
322    '\\' => "\\textbackslash",
323    '^' => "\\textasciicircum",
324    '_' => "\\textunderscore",
325    '`' => "\\textquoteleft",
326    '{' => "\\textbraceleft",
327    '}' => "\\textbraceright",
328    '%' => "\\%",
329    '|' => "\\textbar",
330    '~' => "\\textasciitilde",
331    _ => {
332      let mut tmp = [0u8; 4];
333      return vec![T_OTHER!(c.encode_utf8(&mut tmp))];
334    },
335  };
336  vec![T_CS!(cs)]
337}

339/// Emit a run of `n` spaces as a `white-space:pre` group, exactly like the
340/// listings space path (`\@listingGroup{ltx_lst_space}{ … }` with control-space
341/// tokens that survive TeX space-collapsing).
342fn emit_space_run(n: usize, out: &mut Vec<Token>) {
343  out.push(T_CS!("\\@listingGroup"));
344  out.push(T_BEGIN!());
345  out.extend(ExplodeText!("ltx_lst_space"));
346  out.push(T_END!());
347  out.push(T_BEGIN!());
348  for _ in 0..n {
349    out.push(T_CS!(" "));
350  }
351  out.push(T_END!());
352}
353
354/// Emit the characters of a text run, wrapping maximal space runs in a
355/// `ltx_lst_space` group so indentation and interior spacing are preserved
356/// (`.ltx_listingline` is `white-space:nowrap`, which otherwise collapses runs).
357fn emit_text(text: &str, out: &mut Vec<Token>) {
358  let mut chars = text.chars().peekable();
359  while let Some(c) = chars.next() {
360    if c == ' ' || c == '\t' {
361      let mut n = 1usize;
362      while matches!(chars.peek(), Some(' ') | Some('\t')) {
363        chars.next();
364        n += 1;
365      }
366      emit_space_run(n, out);
367    } else {
368      out.extend(map_char(c));
369    }
370  }
371}

373/// Wrap `inner` in `\cs{…}`.
374fn wrap_cs(cs: &'static str, inner: Vec<Token>) -> Vec<Token> {
375  let mut out = vec![T_CS!(cs), T_BEGIN!()];
376  out.extend(inner);
377  out.push(T_END!());
378  out
379}

381/// Wrap `inner` in `\textcolor[rgb]{r,g,b}{…}`.
382fn wrap_color(rgb: &str, inner: Vec<Token>) -> Vec<Token> {
383  let mut out = vec![T_CS!("\\textcolor"), T_OTHER!("[")];
384  out.extend(ExplodeText!("rgb"));
385  out.push(T_OTHER!("]"));
386  out.push(T_BEGIN!());
387  out.extend(ExplodeText!(rgb));
388  out.push(T_END!());
389  out.push(T_BEGIN!());
390  out.extend(inner);
391  out.push(T_END!());
392  out
393}

395/// Emit one segment: its text (with space runs preserved), wrapped in bold /
396/// italic / color per its style. Order mirrors `\PYG@do`: color outside italic
397/// outside bold. Pure-whitespace segments skip the style wrappers (a colored
398/// space is invisible and only bloats the tree).
399fn emit_seg(cache: &CacheData, seg: &Seg, out: &mut Vec<Token>) {
400  let mut inner = Vec::new();
401  emit_text(&seg.text, &mut inner);
402  if seg.text.trim().is_empty() {
403    out.extend(inner);
404    return;
405  }
406  let style = resolve_style(cache, &seg.tok);
407  if style.bold {
408    inner = wrap_cs("\\textbf", inner);
409  }
410  if style.italic {
411    inner = wrap_cs("\\textit", inner);
412  }
413  if let Some(rgb) = style.rgb.as_ref() {
414    inner = wrap_color(rgb, inner);
415  }
416  out.extend(inner);
417}

419/// Build the per-line block body tokens (`\@lst@startline{}` … `\@lst@endline`
420/// per line) for a set of highlighted lines.
421fn build_block_body(cache: &CacheData, lines: &[Line]) -> Vec<Token> {
422  let mut out = Vec::new();
423  for line in lines {
424    out.push(T_CS!("\\@lst@startline"));
425    out.push(T_BEGIN!());
426    out.push(T_END!()); // empty line-number tags
427    for seg in line {
428      emit_seg(cache, seg, &mut out);
429    }
430    out.push(T_CS!("\\@lst@endline"));
431  }
432  out
433}

435/// If `raw_body` (a minted block's raw source body) matches a frozencache
436/// highlight file, return the Pygments-colored block body tokens (to hand to
437/// `listings_sty::lst_process_display_with`). Otherwise `None`.
438pub fn colored_display_body(raw_body: &str) -> Option<Vec<Token>> {
439  let cache = load_cache()?;
440  let key = normalize_source(raw_body);
441  let lines = cache.by_plaincode.get(&key)?;
442  Some(build_block_body(&cache, lines))
443}

445/// Inline body tokens for a `\mintinline` snippet, for use as the argument of
446/// `\@listings@inline`. On a frozencache hit the segments are Pygments-colored;
447/// on a miss (cache present but snippet not found) the code is emitted plain
448/// (uncolored, but text/spacing preserved). Call only when [`cache_available`].
449pub fn inline_body(raw_code: &str) -> Vec<Token> {
450  let key = normalize_source(raw_code);
451  if let Some(cache) = load_cache() {
452    if let Some(lines) = cache.by_plaincode.get(&key) {
453      let mut out = Vec::new();
454      for (i, line) in lines.iter().enumerate() {
455        if i > 0 {
456          out.push(T_CS!(" "));
457        }
458        for seg in line {
459          emit_seg(&cache, seg, &mut out);
460        }
461      }
462      return out;
463    }
464    // Miss: emit plain (uncolored) text, spacing preserved.
465    let mut out = Vec::new();
466    emit_text(&normalize_source(raw_code), &mut out);
467    return out;
468  }
469  let mut out = Vec::new();
470  emit_text(raw_code, &mut out);
471  out
