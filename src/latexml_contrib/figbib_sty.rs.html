1//! figbib.sty — figure-source lists (the bibliography end).
2//!
3//! `figbib.sty:330` defines `\fbList{bibs}` to write its private
4//! `\figbib@aux` stream and `\InputIfFileExists` the `\jobname.figbib.bbl`
5//! (`:122`, `:335`) that only a bibtex run produces; the per-figure `\fbEpsfig` citations (`:271`)
6//! go to that private stream, never the kernel's citation set, so from
7//! LaTeXML's view nothing was cited and the list vanished at 0 errors (Perl
8//! raw-loads the same package, 11 errors, no list). Each figure command now also
9//! cites its key in the kernel's set where it stands (`\lx@mark@nocite`, the mark
10//! `\nocite` defers to the document's end — the entry's "Cited by" is the
11//! figure, as pdflatex's "Fig. n (p. n)"), and the command is the kernel's
12//! `\lx@bibliography` (`latex_constructs.pool.ltxml:3942`), the bibunits
13//! interception (OXIDIZED_DESIGN_DIVERGENCES #87): the figures used, in figure
14//! order, as bibtex lists them, under the list's own header (`\figbibListHeader`,
15//! `:65`, the `\section*` of `thefigbiblist`, `:348`). The figures are cited in a
16//! list of their own (`figbib`, the bibunits mechanism, #87), which the list reads,
17//! so the document's citations stay out of it. Residuals: every citation also
18//! reaches the main list (Scan.pm:379-380), so a regular `\bibliography` beside the
19//! list sees the figure keys ("Missing bibkeys", or a figure entry listed when the
20//! two share a `.bib`; RED index-bib/figbib_beside_a_bibliography); a later
21//! `\bibliographystyle` restyles the first bibliography, the list (sect11.rs
22//! `\bibstyle`).
23//!
24//! An `@fig` entry's fields are what figbib.bst writes into `\figbibitem`
25//! (figbib.bst:19-35; figbib.sty:365-381 prints "`main`, `add`. \figbibFrom
26//! `source`"), which no BibTeX handler knew, so each entry printed as its number
27//! alone (`[1] [2] [3]`; figbib_sample recall 54 %; Perl has no list at all): the
28//! entry is a `misc`, `main` its title, `add` and `source` its notes, the source
29//! only `\if@figbibsource` (`nosource`, `:104-105`), an empty one not at all, the
30//! list in citation order (figbib.bst has no SORT), as Perl's handler dispatch
31//! (BibTeX.pool.ltxml:135-157) finds `\bib@field@fig@<field>`. Repro
32//! index-bib/figbib_fig_entries_print_their_fields.
33use latexml_package::prelude::*;
34
35#[rustfmt::skip]
36LoadDefinitions!({
37  InputDefinitions!("figbib", noltxml => true, extension => Some(Cow::Borrowed("sty")),
38    handleoptions => true);
39  // The list's own style, local to it (figbib.bst has no SORT: citation order).
40  DefPrimitive!("\\lx@figbib@bibstyle", { latexml_engine::latex_constructs::set_bibstyle("figbib"); });
41  // The figures' citation list (`inlist`), as a bibunit's.
42  DefPrimitive!("\\lx@figbib@unit", { assign_value("CITE_UNIT", "figbib", None); });
43  DefMacro!("\\fbList{}", "\\begingroup\\let\\refname\\figbibListHeader\\lx@figbib@bibstyle\
44\\lx@bibliography[figbib]{#1}\\endgroup");
45  // Each figure command writes `\citation{key}` to the private aux (figbib.sty:271, :284, :299, :310), so bibtex lists
46  // exactly the figures used, in figure order: here the same citation is marked at the figure, in the figures' list.
47  RawTeX!(r"\def\lx@figbib@cite#1{{\lx@figbib@unit\lx@mark@nocite{#1}}}
48\let\lx@figbib@Epsfig\fbEpsfig \def\fbEpsfig#1{\lx@figbib@cite{#1}\lx@figbib@Epsfig{#1}}
49\let\lx@figbib@EpsfigM\fbEpsfigM \def\fbEpsfigM#1{\lx@figbib@cite{#1}\lx@figbib@EpsfigM{#1}}
50\let\lx@figbib@Eps\fbEps \def\fbEps#1{\lx@figbib@cite{#1}\lx@figbib@Eps{#1}}
51\let\lx@figbib@EpsM\fbEpsM \def\fbEpsM#1{\lx@figbib@cite{#1}\lx@figbib@EpsM{#1}}");
52  RawTeX!(r"\def\bib@entry@fig@alias{misc}");
53  RawTeX!(r"\def\bib@field@fig@main{\bib@@field{ltx:bib-title}}");
54  // An empty `add` or `source` prints nothing (figbib.sty:378, :380 `\ifx #5\@empty`).
55  RawTeX!(r"\def\bib@field@fig@add#1{\if\relax\detokenize{#1}\relax\else\bib@field@default@note{#1}\fi}");
56  RawTeX!(r"\def\bib@field@fig@source#1{\if@figbibsource\if\relax\detokenize{#1}\relax\else
57    \bib@field@default@note{\figbibFrom\ #1}\fi\fi}");
58});
