1//! The Wiley NJD family of classes: WileyNJD-v1, WileyNJD-v2, WileyNJDv5 (and `wileyNJDv5`), and WileyASNA-v1
2//! ([`crate::wileyasna_cls`]), one binding for every version (`wiley_njd_family`).
3//!
4//! These Wiley journal classes share a common set of frontmatter macros
5//! (\corres, \authormark, \jnlcitation, \cname, \cyear, \vol, \DOI,
6//! \papertype, ...). Route to OmniBus and gobble the frontmatter so
7//! downstream content renders cleanly.

11  class_code, class_code_contains, class_option_given, class_text_loads,
12  require_shipped_algorithm_packages, shipped_class_text,
13};

16  let class = Expand!(Tokens!(T_CS!("\\@currname"))).to_string();
17  wiley_njd_family(&class)?;
18});

20/// The definitions every Wiley NJD-family binding shares; `class` is the class the paper asked for (`\@currname`:
21/// `WileyNJD-v2`, `WileyNJDv5`, `WileyASNA-v1`, …). The versions agree but for the box environments and the
22/// theorem set, where the paper's copy decides (the class as its name says without one).
23pub(crate) fn wiley_njd_family(class: &str) -> Result<()> {
24  LoadClass!("OmniBus");
25  RequirePackage!("amsmath");
26  // amssymb pulls in \gtrsim/\lesssim and other relation symbols Wiley
27  // journal papers commonly use without an explicit \usepackage{amssymb}.
28  // Witness 2406.06228 (WileyASNA-v1 paper).
29  RequirePackage!("amssymb");
30  RequirePackage!("amsthm");
31  RequirePackage!("xcolor");
32  // Wiley journals frequently load hyperref; mirror so cross-refs work.
33  RequirePackage!("hyperref");
34  // Wiley templates assume booktabs / graphicx are available —
35  // eager-load so docs that use \toprule/\midrule/\bottomrule and
36  // \includegraphics without explicit \usepackage don't error.
37  // Witness 2504.02281. (The algorithm packages are the paper's copy's, below: a fixed algorithm/algorithmic
38  // conflicts with algpseudocode.)
39  RequirePackage!("booktabs");
40  RequirePackage!("graphicx");
41  // wrapfig: WileyNJD-v2 authors routinely use \begin{wrapfigure} for
42  // figure-with-side-text layouts. The Wiley cls itself doesn't load
43  // wrapfig; mirror by eager-loading. Witness 2203.16535.
44  RequirePackage!("wrapfig");
45  // WileyNJDv5.cls:357-376, 413, 620-621 (the WileyNJD-v2 and WileyASNA-v1 copies this binding also stands in for load
46  // the same: 2203.16535, 2406.06228): the tabular, float and caption packages the class loads, in its order
47  // (`\multirow` and `\captionsetup` were undefined, 2609.06025), lettersp (:377) where the copy loads it, below. Left
48  // out: ulem (without `normalem` it underlines `\emph`), enumerate, soul, `babel[english]`, `footmisc[bottom]`, and the
49  // page-layout packages.
50  RequirePackage!("multicol");
51  RequirePackage!("multirow");
52  RequirePackage!("float");
53  RequirePackage!("rotating", options => vec![s!("figuresright")]);
54  RequirePackage!("caption");
55  // WileyNJDv5.cls:377 `\usepackage{lettersp}`, after caption and long before the algorithms (:6820): LETTERSP.STY:3-12
56  // defines its `\Comment … \EndComment` block only while `\Comment` is undefined, keeping an existing one as its own;
57  // loaded after algpseudocode it took algorithmicx's `\Comment` for it and ran its self-documentation as text
58  // (2408.07575, 2501.12471, 2602.03829, 2609.08425, 2609.19994). lettersp is no TeX Live package: only the copy the
59  // paper ships, where the paper's copy of the class loads it.
60  let (_, class_text) = wiley_version(class)?;
61  let lettersp = || {
62    find_file(
63      "lettersp",
64      Some(FindFileOptions {
65        ext_type: Some(Cow::Borrowed("sty")),
66        forbid_ltxml: true,
67        ..Default::default()
68      }),
69    )
70    .is_some()
71  };
72  if class_text
73    .as_deref()
74    .is_some_and(|text| class_text_loads(text, "lettersp"))
75    && lettersp()
76  {
77    RequirePackage!("lettersp");
78  }
79  RequirePackage!("tabularx");
80  RequirePackage!("varwidth");
81  RequirePackage!("dcolumn");
82  // WileyNJD-v2.cls:3331, WileyNJDv5.cls:6820 and WileyASNA-v1.cls:2555 `\usepackage{algorithm, algorithmicx,
83  // algpseudocode}`, a copy `algorithm2e` (2002.02812:3338): `{algorithm}` and `\State` were undefined in papers that
84  // rely on the class for them (1810.03033, 1810.12033, 2106.06046, 2106.14215, 2210.02694, 2402.11361, 2402.13183,
85  // 2506.07013). The paper's copy decides; without one, none (Perl's OmniBus has no copy to scan either).
86  require_shipped_algorithm_packages("cls")?;
87  // WileyNJD-v2.cls:63 `\DeclareOption{APA}{\global\@APArefstyletrue}` and :2722-2723 `\if@APArefstyle
88  // \usepackage[natbibapa]{NJDapacite}`: the APA reference style's bibliography is apacite's (`\APACrefYearMonthDay`,
89  // `{APACrefauthors}`, `\BBOQ`; 22 papers of run 353 with them undefined: 1811.00964, 1902.09230, 1904.10792).
90  // NJDapacite.sty, shipped with the class, is apacite.sty of 2013 under another name (its :46 `\ProvidesPackage
91  // {apacite}`), so the apacite binding stands for it, as for sn-jnl's `sn-apa` (sn_jnl_cls.rs). The class's
92  // re-styling of the reference parts (:2725-2760, fonts and punctuation) is not followed. The load follows the
93  // version (`wiley_njd_class_body`).
94  DeclareOption!("APA", {
95    AssignValue!("wileynjd@APA" => true);
96  });
97  ProcessOptions!();
98
99  // WileyNJD-v2.cls:1217-1221 (WileyNJDv5.cls:3636-3640, WileyASNA-v1 through WileyNJD-v1): the address parts of an
100  // affiliation, each printing its argument; OmniBus has the others, but leaves `\state` out (omnibus_cls.rs: a
101  // `\newcount\state` in some classes) — here it is only `\orgaddress{\state{NY}, \country{USA}}`, as in mrm_cls.rs.
102  // Witnesses 1705.06379, 1910.04517, 2210.01555, 2605.24284.
103  def_macro_identity("\\state{}")?;
104
105  // Wiley frontmatter — preserve author content as ltx:note.
106  DefMacro!("\\authormark{}", "\\textsuperscript{#1}");
107  DefMacro!(
108    "\\corres{}",
109    "\\@add@frontmatter{ltx:note}[role=corresponding]{#1}"
110  );
111  DefMacro!(
112    "\\jnlcitation OptionalMatch:* []{}{}",
113    "\\@add@frontmatter{ltx:note}[role=citation]{#3 #4}"
114  );
115  DefMacro!(
116    "\\presentadd[]{}",
117    "\\@add@frontmatter{ltx:note}[role=present-address]{#2}"
118  );
119  DefMacro!(
120    "\\fundingInfo{}",
121    "\\@add@frontmatter{ltx:note}[role=funding]{#1}"
122  );
123  DefMacro!(
124    "\\papertype{}",
125    "\\@add@frontmatter{ltx:note}[role=papertype]{#1}"
126  );
127  DefMacro!(
128    "\\paperfield{}",
129    "\\@add@frontmatter{ltx:note}[role=paperfield]{#1}"
130  );
131  DefMacro!(
132    "\\jname{}",
133    "\\@add@frontmatter{ltx:note}[role=journal]{#1}"
134  );
135  DefMacro!("\\jvol{}", "\\@add@frontmatter{ltx:note}[role=volume]{#1}");
136  DefMacro!("\\jnum{}", "\\@add@frontmatter{ltx:note}[role=issue]{#1}");
137  DefMacro!(
138    "\\cname{}{}",
139    "\\@add@frontmatter{ltx:note}[role=copyright]{#1 #2}"
140  );
141  DefMacro!("\\cyear{}", "\\@add@frontmatter{ltx:note}[role=year]{#1}");
142  // \cjournal{name} / \cvol{N} / \ctitle{text} — citation-line fields
143  // also in WileyNJDv5 frontmatter (alongside \cname / \cyear). Witness
144  // 2504.02281 — preserve as ltx:note.
145  DefMacro!(
146    "\\cjournal{}",
147    "\\@add@frontmatter{ltx:note}[role=cjournal]{#1}"
148  );
149  DefMacro!("\\cvol{}", "\\@add@frontmatter{ltx:note}[role=cvolume]{#1}");
150  DefMacro!(
151    "\\ctitle{}",
152    "\\@add@frontmatter{ltx:note}[role=ctitle]{#1}"
153  );
154  DefMacro!(
155    "\\Copyrightline{}",
156    "\\@add@frontmatter{ltx:note}[role=copyright]{#1}"
157  );
158  DefMacro!(
159    "\\artmonth{}",
160    "\\@add@frontmatter{ltx:note}[role=month]{#1}"
161  );
162  DefMacro!("\\DOI{}", "\\@add@frontmatter{ltx:note}[role=doi]{#1}");
163  DefMacro!("\\doiline{}", "\\@add@frontmatter{ltx:note}[role=doi]{#1}");
164  DefMacro!(
165    "\\runningheads{}{}",
166    "\\@add@frontmatter{ltx:note}[role=runningheads]{#1 / #2}"
167  );
168  DefMacro!(
169    "\\receiveddate{}",
170    "\\@add@frontmatter{ltx:note}[role=received]{#1}"
171  );
172  DefMacro!(
173    "\\reviseddate{}",
174    "\\@add@frontmatter{ltx:note}[role=revised]{#1}"
175  );
176  DefMacro!(
177    "\\accepteddate{}",
178    "\\@add@frontmatter{ltx:note}[role=accepted]{#1}"
179  );
180  // wileyNJDv5.cls (newer template) adds these. Witness 2407.00139.
181  DefMacro!(
182    "\\copyyear{}",
183    "\\@add@frontmatter{ltx:note}[role=copyright-year]{#1}"
184  );
185  DefMacro!(
186    "\\titlemark{}",
187    "\\@add@frontmatter{ltx:note}[role=titlemark]{#1}"
188  );
189  DefMacro!(
190    "\\startpage{}",
191    "\\@add@frontmatter{ltx:note}[role=startpage]{#1}"
192  );
193  wiley_njd_class_body(class)
194}

196/// The version-sensitive definitions are chosen by the copy the paper ships, else by the class name.
197#[derive(Clone, Copy, Debug, PartialEq, Eq)]
198enum WileyVersion {
199  /// WileyNJD-v1.cls (2015-2017): its 20 theorems, `{boxtext}` an mdframed box.
200  V1,
201  /// WileyNJD-v2.cls and WileyASNA-v1.cls: the 47 theorems, `{boxtext}` without arguments.
202  V2,
203  /// WileyNJDv5.cls: v2's theorems, `{boxtext}[1]` and `{boxwithhead}[2]` (boites' `{breakbox}`).
204  V5,
205}
206
207/// The version of the class the paper asked for, and the text of the copy it ships (`None` without one).
208fn wiley_version(class: &str) -> Result<(WileyVersion, Option<String>)> {
209  let name = class
210    .rsplit(['/', '\\'])
211    .next()
212    .unwrap_or_default()
213    .to_ascii_lowercase();
214  let by_name = if name.contains("v5") {
215    WileyVersion::V5
216  } else if name.starts_with("wileynjd-v1") {
217    WileyVersion::V1
218  } else {
219    WileyVersion::V2
220  };
221  // `shipped_class_text` reads the copy of the class the load resolved to (`\@currname`, or the request a fallback
222  // answered with this binding: `WileyNJDv5_preprint`, `WileyNJD-v2a`)
223  let binding = if name.starts_with("wileyasna") {
224    "WileyASNA-v1"
225  } else {
226    class
227  };
228  let text = shipped_class_text(binding)?;
229  let version = text
230    .as_deref()
231    .and_then(version_from_text)
232    .unwrap_or(by_name);
233  Ok((version, text))
234}

236/// The version a copy of the class is, by the definitions where the versions differ: WileyNJDv5.cls:592 `{boxtext}[1]`
237/// (2302.02310), the v2 theorem set's `{eexample}` (WileyNJD-v2.cls:3012, 1712.08341; WileyASNA-v1.cls and the
238/// WileyNJD-v1.cls of 2020, 2010.09463, have it too), the v1 set's `{example}` (WileyNJD-v1.cls:4564, 1511.08547).
239/// `None` for a copy with none of them.
240fn version_from_text(text: &str) -> Option<WileyVersion> {
241  if class_code_contains(text, "\\newenvironment{boxtext}[1]") {
242    Some(WileyVersion::V5)
243  } else if class_code_contains(text, "\\newtheorem{eexample}") {
244    Some(WileyVersion::V2)
245  } else if class_code_contains(text, "\\newtheorem{example}{Example}") {
246    Some(WileyVersion::V1)
247  } else {
248    None
249  }
250}

252/// The commands, environments and theorems the Wiley NJD classes define that papers use and OmniBus does not have
253/// (the class bodies are the same in WileyNJD-v2.cls of 2017-10-25 (1712.08341, 312 copies), WileyNJDv5.cls
254/// (2302.02310, 89 copies), WileyASNA-v1.cls (1804.11014, 158 copies) and WileyNJD-v1.cls (1511.08547, 12 copies) where
255/// not said otherwise; line numbers are these copies').
256fn wiley_njd_class_body(class: &str) -> Result<()> {
257  let (version, class_text) = wiley_version(class)?;
258  let asna = class.to_ascii_lowercase().contains("wileyasna");
259  // WileyNJD-v2.cls:1781 (WileyNJDv5.cls:5066, WileyASNA-v1.cls:1195, WileyNJD-v1.cls:956)
260  // `\def\presentaddress#1{\gdef\@presentaddress{…\textbf{Present Address}\par #1}}`, printed under the
261  // correspondence block of the title page (:2289): a frontmatter note, as sn-jnl's (sn_jnl_cls.rs). Undefined in 156
262  // WileyNJD-v2, 68 WileyASNA-v1 and 38 WileyNJDv5 papers of run 353 (1701.02349, 1705.09846, 1712.08341, 1801.09541;
263  // WileyNJD-v1 1709.02675).
264  DefMacro!(
265    "\\presentaddress{}",
266    "\\@add@frontmatter{ltx:note}[role=present-address]{#1}"
267  );
268  // The title-page stores (WileyNJD-v2.cls:1812, 1890-1891, 1198; WileyNJDv5.cls:5099, 5182-5183; WileyASNA-v1.cls:1226,
269  // 1319-1320, 624): the history line, the funding block's headings and the article id are page furniture
270  // `\@maketitle` lays out (`\historydates` 1902.09230, 2605.19910; `\fundinginfohead` 1806.09518; `\artid` 2101.04467,
271  // 2608.01016). The running head's DOI text (WileyNJD-v2.cls:1820, WileyNJDv5.cls:5107, WileyASNA-v1.cls:1234), a
272  // paper's DOI where it is given (`\doiheadtext{DOI: 10.1002/fld.4504}`, 1803.06134; 2306.05470, 2505.23984,
273  // 2512.05601), is a DOI note, as `\DOI`.
274  // `\uns` (WileyNJD-v2.cls:1180 `\def\uns{\ifmmode\,\else$\,$\fi}`) is a thin space in or out of math (2208.08472,
275  // 2211.16474), as the class defines it (until 78a `\thinspace` out of math: the formula holding only `\,` ended the
276  // paragraph's `--` ligatures in the kernel, 2208.08472's "1989--2008"; fixed in document.rs, guard
277  // `perfect_kernel_batch78a::text_after_a_space_only_formula_is_ligatured`). `\subtitlefont`
278  // (WileyNJD-v2.cls:1436, 1506, WileyNJDv5.cls:3990, the size options' title fonts) the font of a subtitle in
279  // `\title{…\\\subtitlefont …}` (1902.01961, 2108.10040), its margins left out.
280  RawTeX!(
281    r"\def\historydates#1{\gdef\@history@dates{#1}}%
282\def\subtitlefont{\rmfamily\fontsize{16}{21}\bfseries\selectfont}%
283\def\fundinginfohead#1{\gdef\@fundinginfo@head{#1}}%
284\def\fundinginfoheadtext#1{\gdef\@fundinginfo@head@text{#1}}%
285\def\artid#1{\gdef\@artid{#1}}%
286\def\uns{\ifmmode\,\else$\,$\fi}%"
287  );
288  DefMacro!(
289    "\\doiheadtext{}",
290    "\\@add@frontmatter{ltx:note}[role=doi]{#1}"
291  );
292  // WileyNJD-v1.cls's reference styles (1511.08547's copy: APA :1104-1106, AMS :1850-1851, AMA :2498-2499, Vancouver
293  // :3080-3081, Chemistry :3684-3685) each load amsrefs, whose `{bibdiv}`, `{biblist}`, `\bib` and `\ndash` the papers'
294  // reference lists use (1511.08547, 1707.00398, 1709.02675, 1709.08526, 1806.01802: 9 papers of run 353), each load
295  // inside its `\if@…refstyle`, which only the option sets (:33-43, no `\ExecuteOptions`): no amsrefs without one of
296  // APA, AMS, ACS, AMA, VANCOUVER. Those copies load no apacite, so their APA style is amsrefs' author-year, not the
297  // later versions' NJDapacite (WileyNJD-v2.cls:2722-2723, WileyNJDv5.cls, WileyASNA-v1.cls and the WileyNJD-v1.cls of
298  // 2020, 2010.09463, 2208.12482, which load no amsrefs and are the v2 version here). The 2020 copies' title-page
299  // stores (2010.09463's copy :1825-1842): the journal's name, logo and URL in the running head; `\openaccess` (:1871)
300  // sets its licence text as an unmarked footnote of the title page, a licence pubnote here (as
301  // dmtcs_episciences_cls.rs, acmart_cls.rs).
302  let copy_loads = |package: &str, without_copy: bool| {
303    class_text
304      .as_deref()
305      .map_or(without_copy, |text| class_text_loads(text, package))
306  };
307  let refstyle = ["APA", "AMS", "ACS", "AMA", "VANCOUVER"]
308    .into_iter()
309    .map(class_option_given)
310    .collect::<Result<Vec<_>>>()?
311    .contains(&true);
312  if refstyle && copy_loads("amsrefs", version == WileyVersion::V1) {
313    RequirePackage!("amsrefs");
314  }
315  if lookup_bool("wileynjd@APA")
316    && (copy_loads("NJDapacite", version != WileyVersion::V1) || copy_loads("apacite", false))
317  {
318    RequirePackage!("apacite", options => vec![s!("natbibapa")]);
319    // The copy's own `\usepackage[apaciteclassic]{NJDapacite}` (WileyNJDv5.cls:6067; v2's `[natbibapa]`) is a repeat
320    // of this load, NJDapacite.sty being the apacite.sty the binding stands for: marked loaded on the class's behalf,
321    // as amsart marks amsthm (ams_support_sty.rs). Read raw after the binding, its `apaciteclassic` skip of the
322    // `\if@APAC@natbib@emu` branch (NJDapacite.sty:587-987) met the natbib binding's `\ifciteindex` at :931's
323    // `\newif\ifciteindex` — a conditional to the skip, where TeX, with no natbib loaded, sees an undefined name — and
324    // ran on to :1878-1882's `\let\ifBR@verbose\iffalse\let\fi\fi`: three "Attempt to close a group" Errors
325    // (2606.10545, 2602.12288; pdflatex 0).
326    AssignValue!("NJDapacite.sty_loaded" => true, Some(Scope::Global));
327    RawTeX!(
328      r"\expandafter\let\csname ver@NJDapacite.sty\expandafter\endcsname\csname ver@apacite.sty\endcsname"
329    );
330  }
331  // Every version loads listings and alltt (WileyNJD-v2.cls:3346-3347, WileyNJDv5.cls:6822, WileyASNA-v1.cls:2557,
332  // WileyNJD-v1.cls:4712; `\lstset` undefined: 1802.08557, 2206.13356). WileyASNA-v1.cls:250 and the older
333  // WileyNJD-v1.cls:256 load `[labelformat=empty]{subfig}` (`\subfloat` undefined: 1511.08547, 1707.00398); WileyNJD-v2
334  // (:287), WileyNJDv5 (:397) and the WileyNJD-v1 of 2020 (2010.09463:281) only `\IfFileExists{subfigure.sty}{}{…}`,
335  // and TeX Live has subfigure.sty, so none.
336  RequirePackage!("listings");
337  RequirePackage!("alltt");
338  let subfig = match class_text.as_deref() {
339    Some(text) => {
340      class_text_loads(text, "subfig")
341        && !class_code_contains(text, "\\IfFileExists{subfigure.sty}")
342    },
343    None => version == WileyVersion::V1 || asna,
344  };
345  if subfig {
346    RequirePackage!("subfig", options => vec![s!("labelformat=empty")]);
347  }
348  RawTeX!(
349    r"\def\journalname#1{\gdef\@journalname{#1}}%
350\def\journallogo#1#2{\gdef\@journallogo{\textsc{#1}\ \textcolor{blue}{#2}}}%
351\def\jurl#1{\gdef\@jurl{#1}}"
352  );
353  DefMacro!("\\openaccess{}", "\\lx@add@pubnote[role=license]{#1}");
354  // WileyNJD-v2.cls:1857-1864 (WileyNJDv5.cls:5149, WileyASNA-v1.cls:1271, WileyNJD-v1.cls:978, every copy)
355  // `\renewcommand\abstract[2][]{…\absheadfont#1…#2…}`: the abstract and, optionally, its heading
356  // (`\abstract[Summary]{…}`, the templates' form: 45 of 60 sampled papers of the family, 1701.02349, 1712.08341; the
357  // kernel's `\abstract` read `[Summary]` as text of the abstract in 219 of 490 papers). The heading names the
358  // abstract; without one it keeps the kernel's name.
359  RawTeX!(
360    r"\def\lx@wiley@abstract#1#2{\if\relax\detokenize{#1}\relax\lx@add@abstract{#2}\else%
361\lx@add@abstract[name={#1}]{#2}\fi}"
362  );
363  DefMacro!("\\abstract[]{}", "\\lx@wiley@abstract{#1}{#2}");
364  // WileyNJD-v2.cls:1877-1878 (WileyNJDv5.cls:5169-5170): the parts of a `\fundingInfo` entry, printed as
365  // "Agency, Number."; WileyASNA-v1 sets them in running text (wileyasna_cls.rs).
366  RawTeX!(r"\def\fundingNumber#1{#1.}\def\fundingAgency#1{#1,\space}");
367  // WileyNJDv5.cls:5762-5764: the class's own sectioning commands, `\@startsection`s of the section levels, so with
368  // their star and optional title (`\bmsection*{Acknowledgments}`, 2405.03603, 2511.08532: the former `\bmsection{}`
369  // read the star as the title) and the section counter after `\appendix` (2511.08532's `\bmsection{Details …}`):
370  // `\section` as it is when called. `\bmsubsubsection`: 2309.00837, 2311.07331, 2602.03894.
371  RawTeX!(
372    r"\def\bmsection{\section}\def\bmsubsection{\subsection}\def\bmsubsubsection{\subsubsection}"
373  );
374
375  // WileyNJD-v2.cls:2816-2828 (WileyNJDv5.cls:6188-6200, WileyASNA-v1.cls:2315-2327, WileyNJD-v1.cls:4321-4333): the
376  // notes under a table, a list whose first item is the text before any `\item` (`\begin{tablenotes} a) Eq. 1, …`,
377  // 1808.09739; `\item[$\dagger$] …`, 1808.05540, 1901.00057, 1903.09736), and `\tnote`, which prints its mark as it
378  // is (1910.13464, 1911.09684). threeparttable, loaded after the class, replaces both. The class's leading
379  // `\item\relax` opens an item only where text precedes the first `\item` (OXIDIZED_DESIGN_DIVERGENCES #601: the
380  // empty first item the class sets before `\item[$\dagger$]` is a blank, no glyph, in the PDF, and would be an empty
381  // `<ltx:item>` here).
382  RawTeX!(
383    r"\newenvironment{tablenotes}{\list{}{\setlength{\labelsep}{0pt}%
384\setlength{\labelwidth}{0pt}%
385\setlength{\leftmargin}{0pt}%
386\setlength{\rightmargin}{0pt}%
387\setlength{\topsep}{2pt}%
388\setlength{\itemsep}{0pt}%
389\setlength{\partopsep}{0pt}%
390\setlength{\listparindent}{0em}%
391\setlength{\parsep}{0pt}}%
392\@ifnextchar\item{}{\item\relax}}{\endlist}%
393\def\tnote#1{#1}"

506/// The class's four theorem styles.
507const THEOREM_STYLES: [&str; 4] = ["plain", "WBstyleone", "WBstyletwo", "WBstylethree"];

509/// The settings of WileyNJD-v1.cls's (2015-2018) theorem bodies that set their math upright (1511.08547's copy
510/// :4505, :4527, :4537 `{\itshape\everymath{\rm}\let\text\mathrm}`), left out: the formulas of a theorem would be
511/// `\rm` formulas, whose letters the math parser reads as one word (`A_{ij}`'s subscript the identifier "ij", 1511.08547)
512/// and whose `tex` starts with `\rm`; the body stays italic.
513const UPRIGHT_MATH: [&str; 2] = ["\\everymath{\\rm}", "\\let\\text\\mathrm"];
514
515/// The theorem styles as the paper's copy of the class defines them: each `\newtheoremstyle{<name>}` of
516/// [`THEOREM_STYLES`] with its eight arguments, the last definition outside a comment; `None` where the copy lacks
517/// one. (The 578 copies of the family's classes shipped by the 817 papers of the 67w inventory define them four ways,
518/// [`theorem_styles`].)
519fn copy_theorem_styles(text: &str) -> Option<String> {
520  let code = class_code(text);
521  let mut styles = String::new();
522  for name in THEOREM_STYLES {
523    let head = format!("\\newtheoremstyle{{{name}}}");
524    let at = code.rfind(&head)?;
525    let mut rest = &code[at + head.len()..];
526    styles.push_str(&head);
527    for _ in 0..8 {
528      rest = rest.trim_start();
529      let group = balanced_group(rest)?;
530      styles.push_str(group);
531      rest = &rest[group.len()..];
532    }
533    styles.push_str("%\n");
534  }
535  Some(
536    UPRIGHT_MATH
537      .iter()
538      .fold(styles, |styles, setting| styles.replace(setting, "")),
539  )
540}
541
542/// The brace group `text` starts with, its braces included.
543fn balanced_group(text: &str) -> Option<&str> {
544  if !text.starts_with('{') {
545    return None;
546  }
547  let mut depth = 0usize;
548  for (at, ch) in text.char_indices() {
549    match ch {
550      '{' => depth += 1,
551      '}' => {
552        depth -= 1;
553        if depth == 0 {
554          return Some(&text[..=at]);
555        }
556      },
557      _ => {},
558    }
559  }
560  None
561}
