1//! Stub for midl.cls (MIDL — Medical Imaging with Deep Learning), a jmlr
2//! derivative. midl.cls does `\LoadClass[pmlr]{jmlr}` and defines its author wrapper
3//! in the cls body: `\newcommand{\midlauthor}[1]{\author{#1}}` (midl.cls:57).
4//! As an unbound class Rust takes the OmniBus fallback, which dep-scans the cls
5//! (so jmlr loads) but does NOT execute the cls body, leaving `\midlauthor`
6//! undefined; the undefined wrapper then fails to consume its `{...}`, so the
7//! jmlr binding's `\addr Until:\lx@jmlr@endaddr` scanner (jmlr_cls.rs) runs to
8//! EOF and aborts the whole document (`Fatal:Mouth:EoF`). Define the wrapper
9//! (routing through jmlr's structured-author path, which lays the sentinel) and
10//! the few body helpers so the author block is clean. Witness 2605.00538.
11use latexml_package::prelude::*;
12
13LoadDefinitions!({
14  LoadClass!("jmlr");
15  DefMacro!("\\midlauthor{}", "\\author{#1}");
16  // midl.cls body author helpers (midl.cls:50-57,106).
17  DefMacro!("\\midljointauthortext{}", "\\nametag{\\thanks{#1}}");
18  def_macro_noop("\\midlotherjointauthor")?;
19  DefMacro!("\\midlacknowledgments{}", "\\acks{#1}");
20  // midl.cls:106 `\orcid#1`: a linked ORCID logo, no text of the id — the binding's declared drop, which the title-page
21  // check wants none of (OXIDIZED_DESIGN_DIVERGENCES #953 "106t").
22  DefMacro!("\\orcid{}", "", drops => [1]);
23  DefMacro!("\\jmlrpages{}", ""); // jmlr.cls:559 is 1-arg; swallow it (no body leak)
24});
