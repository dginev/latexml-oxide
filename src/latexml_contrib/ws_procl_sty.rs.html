1//! World Scientific's proceedings styles for the article class: `sprocl.sty` (8.5 × 6 in), `mprocl.sty` (9.75 × 6.5
2//! in) and `stwol.sty` (two-column, smaller size), "based on a style file by Lukas Nellen", LaTeX 2.09 option styles
3//! (`\documentstyle[sprocl]{article}`). Not in TeX Live and not always shipped: missing, their `\author` was article's,
4//! which keeps the last call, so a paper with an `\author` per author group kept only the last group (hep-ph9611287,
5//! hep-ph9611291, hep-ph9602339, hep-ex9910019, hep-ex9608006, nucl-th9809087, gr-qc9806116, hep-ph9905201; mprocl
6//! gr-qc9710030). The census of the shipped copies in the 92xx-99xx arXiv (704 papers, 203 variants of sprocl.sty; 59
7//! papers, 23 of mprocl.sty; 42 papers, 10 of stwol.sty; cited from hep-ph9701265's sprocl.sty) shows one document
8//! syntax: `\author` and `\address` append a line each to the title block (sprocl.sty:18-21, `\@aabuffer`), so an
9//! address is that of the authors given since the one before, `\abstracts{…}` the abstract (:32-38),
10//! `\citeup`/`\citelow` citations (:103-110), an `{arabiclist}` (:82-89), footnotes marked `\alph` (:65). Single
11//! variants add more: `\collaboration` (1 of the 10 stwol variants, stwol.sty:23-24, a title-block line as `\author`'s;
12//! hep-th9610159), and in sprocl `\authora` (= `\author`), `\id` (indexed text), `\CaP` (small caption text),
13//! `\clearemptypage` (1 variant each; hep-ph9708375's family), bound as conveniences a document's own definition
14//! replaces; `\pnasus`/`\jtb`/`\jeem` appear only commented out (hep-ph9708375's sprocl.sty:170-178), so they are left
15//! unbound. The rest is page layout. Rust-only: Perl has no binding.
16use latexml_core::definition::origin::{DefinitionOrigin, OriginGuard};
17use latexml_package::prelude::*;
18
19/// The styles' common definitions.
20fn ws_procl_style() -> Result<()> {
21  // sprocl.sty:18-21: an `\author` line holds one or more names (`\author{N. G. DESHPANDE, B. DUTTA}`); an `\address`
22  // line goes to the authors given since the previous one.
23  DefMacro!("\\author{}", "\\lx@add@authors@append{#1}");
24  // The address is the authors' affiliation, as Perl takes iopart's `\address` (iopart_support.sty.ltxml:53
25  // `\lx@add@affiliations{#1}`) and REVTeX's `\affiliation` (revtex4_support_sty.rs): one led by an author's mark
26  // (`\author{A. ABLE$^a$, B. BAKER$^b$}\address{$^a$X}\address{$^b$Y}`; hep-lat9806004's
27  // `\author{C. Alexandrou$^a$, A. Bori\c{c}i$^b$, …}\address{$^a$Department of Natural Sciences, …}`) goes to the
28  // authors showing that mark, any other to the authors given since the previous one.
29  DefMacro!("\\address{}", sub[(content)] {
30    if leads_with_label(&content, "ws_procl address")?
31      && !queued_group_shares_the_mark(&content_mark_labels(&content, "ws_procl address")?)?
32    {
33      Ok(Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![None, Some(content)]))
34    } else {
35      Ok(Invocation!(
36        T_CS!("\\lx@add@affiliation"),
37        vec![Some(mouth::tokenize_internal("annotate=new")), Some(content)]
38      ))
39    }
40  });
41  // sprocl.sty:32-38: the abstract, a centered minipage after the title block
42  DefMacro!("\\abstracts{}", "\\lx@add@abstract{#1}");
43  // sprocl.sty:103-110: `\cite` as a superscript, `\citelow` on the line; both cite
44  Let!("\\citeup", "\\cite");
45  Let!("\\citelow", "\\cite");
46  // sprocl.sty:65: footnotes marked a, b, …
47  DefMacro!("\\thefootnote", "\\alph{footnote}");
48  // sprocl.sty:82-89: an arabic-numbered list
49  RawTeX!(
50    "\\newcounter{arabiclistc}\
51     \\newenvironment{arabiclist}{\\setcounter{arabiclistc}{0}\\begin{list}{\\arabic{arabiclistc}}\
52     {\\usecounter{arabiclistc}\\setlength{\\parsep}{0pt}\\setlength{\\itemsep}{0pt}}}{\\end{list}}"
53  );
54  // the additions of single variants (hep-ph9708375's sprocl.sty:153-168 and kin): conveniences the paper's copy may not
55  // have, which yield to a document's own `\newcommand` of the name (`\newcommand{\id}{{\bf 1}}`), as a fallback
56  // class's guesses do (`DefinitionOrigin::Fallback`)
57  {
58    let _guess = OriginGuard::new(DefinitionOrigin::Fallback);
59    // stwol.sty:23-24: a collaboration's name, a line of the title block among the authors (as REVTeX's
60    // `\collaboration`)
61    Let!("\\collaboration", "\\author");
62    Let!("\\authora", "\\author");
63    DefMacro!("\\id{}", "#1\\index{#1}");
64    // (hep-ph9708375's family: `\def\CaP#1{\footnotesize\parindent=0pt #1}`, no group)
65    DefMacro!("\\CaP{}", "\\footnotesize\\parindent=0pt #1");
66    Let!("\\clearemptypage", "\\clearpage");
67  }
68  // sprocl.sty:112: sections and subsections numbered
69  RawTeX!("\\setcounter{secnumdepth}{2}");
70  Ok(())
71}
