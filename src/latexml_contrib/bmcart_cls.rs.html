1//! bmcart.cls (BioMed Central's article class): the papers ship the class (it is not in TeX Live), and it is read raw
2//! (OXIDIZED_DESIGN_DIVERGENCES #444). Its `{frontmatter}` lets `\title`, `\author` and `\address` to its printers
3//! (:966-984 `\set@frontmatter@cmd`): `\title@fmt` typesets the title in the `{fmbox}` (:1034-1053), `\author@fmt`
4//! the name with the marks of its `addressref` keys (:1198-1255), and `\address@fmt` stores the address by its `id`
5//! (:1287-1296) for the "Author details" block that `\thebibliography` prints (:1708-1712 `\printaddresses`), which
6//! the kernel's bibliography constructor never runs. Read raw, the document had no title, no creator and no address
7//! (1711.01177: the title page as boxes, the marks "??"). So, after the read, the three printers are the frontmatter's,
8//! as elsart_support_core's `\author[labels]`/`\address[label]` are (elsart_support_core.sty.ltxml:35-42): each
9//! `\author` a creator requesting its `addressref`/`presentaddressref` ids, with its `email` addresses; each `\address`
10//! an affiliation labelled by its `id` (the class's `aff<n>` when it has none); and the `{fmbox}` the box only holds
11//! them, its "Full list of author information …" line (:1002 `\auto@set@thanksbox`) a print flourish. Perl has no
12//! binding: OmniBus, whose `\author` read "[" as the name. Guard `perfect_kernel_batch78m::bmcart_frontmatter_is_frontmatter`.
13use latexml_package::{
14  prelude::*,
15  shipped_class::{RawRead, read_raw},
16};

18const BMCART: RawRead = RawRead {
19  shipped: true,
20  ..RawRead::new("bmcart", "cls")

23/// The `key=value` pairs of a keyval list at its top level, each value without the spaces and the one pair of braces
24/// around it (`addressref={cmu1,eth}`: `cmu1,eth`).
25fn keyval_pairs(list: &Tokens) -> Vec<(String, Vec<Token>)> {
26  let mut items: Vec<Vec<Token>> = vec![Vec::new()];
27  let mut depth = 0usize;
28  for t in list.unlist_ref() {
29    match t.get_catcode() {
30      Catcode::BEGIN => depth += 1,
31      Catcode::END => depth = depth.saturating_sub(1),
32      _ if depth == 0 && *t == T_OTHER!(",") => {
33        items.push(Vec::new());
34        continue;
35      },
36      _ => {},
37    }
38    if let Some(item) = items.last_mut() {
39      item.push(*t);
40    }
41  }
42  items
43    .into_iter()
44    .filter_map(|item| {
45      let eq = item.iter().position(|t| *t == T_OTHER!("="))?;
46      let key = Tokens::new(item[..eq].to_vec())
47        .to_string()
48        .trim()
49        .to_string();
50      let mut value: &[Token] = &item[eq + 1..];
51      while let [first, rest @ ..] = value
52        && first.get_catcode() == Catcode::SPACE
53      {
54        value = rest;
55      }
56      while let [rest @ .., last] = value
57        && last.get_catcode() == Catcode::SPACE
58      {
59        value = rest;
60      }
61      if let [open, inner @ .., close] = value
62        && open.get_catcode() == Catcode::BEGIN
63        && close.get_catcode() == Catcode::END
64      {
65        value = inner;
66      }
67      Some((key, value.to_vec()))
68    })
69    .collect()
70}

72/// `tokens` in braces.
73fn braced(tokens: &[Token]) -> Vec<Token> {
74  let mut out = Vec::with_capacity(tokens.len() + 2);
75  out.push(T_BEGIN!());
76  out.extend_from_slice(tokens);
77  out.push(T_END!());
78  out
79}

81LoadDefinitions!({
82  if read_raw(&BMCART)? {
83    // :1034-1053: the title, its optional keys (`\setkeys{title}`) the page's
84    DefMacro!("\\title@fmt[]{}", "\\lx@add@title{#2}");
85    // :1198-1255: the author, its name parts as the printed name reads them (:1247-1249: `\inits` dropped, the others
86    // as written), requesting the addresses its `addressref` and `presentaddressref` keys name; its `email` addresses
87    // (`email={a@x, b@y}`, 1711.01177) its contacts
88    DefMacro!("\\author@fmt[]{}", sub[(keys, name)] {
89      let pairs = keys.as_ref().map(keyval_pairs).unwrap_or_default();
90      let mut labels: Vec<Token> = Vec::new();
91      for (key, value) in &pairs {
92        if (key == "addressref" || key == "presentaddressref") && !value.is_empty() {
93          if !labels.is_empty() {
94            labels.push(T_OTHER!(","));
95          }
96          labels.extend_from_slice(value);
97        }
98        // (:1131-1143: the article notes its `noteref` key, and `thanksref`, whose alias it is, names; each id the
99        // `note:`-labelled contact `\@bmcartnote` makes)
100        if key == "noteref" || key == "thanksref" {
101          for id in value.split(|t| *t == T_OTHER!(",")) {
102            let id: Vec<Token> =
103              id.iter().copied().filter(|t| t.get_catcode() != Catcode::SPACE).collect();
104            if id.is_empty() {
105              continue;
106            }
107            if !labels.is_empty() {
108              labels.push(T_OTHER!(","));
109            }
110            labels.extend(ExplodeText!("note:"));
111            labels.extend(id);
112          }
113        }
114      }
115      let mut out = TokenizeInternal!("\\lx@add@creator[role=author,annotations=").unlist();
116      out.extend(braced(&labels));
117      out.push(T_OTHER!("]"));
118      let mut shown = TokenizeInternal!(
119        "\\def\\inits#1{}\\def\\fnm#1{#1}\\def\\snm#1{#1}\\def\\particle#1{#1}\\def\\suffix#1{#1}"
120      )
121      .unlist();
122      shown.extend(name.unlist());
123      out.extend(braced(&braced(&shown)));
124      for (key, value) in &pairs {
125        if key != "email" {
126          continue;
127        }
128        for address in value.split(|t| *t == T_OTHER!(",") || *t == T_OTHER!(";")) {
129          let address: Vec<Token> =
130            address.iter().copied().filter(|t| t.get_catcode() != Catcode::SPACE).collect();
131          if !address.is_empty() {
132            out.extend(TokenizeInternal!("\\lx@add@email").unlist());
133            out.extend(braced(&address));
134          }
135        }
136      }
137      Ok(Tokens::new(out))
138    });
139    // :1287-1296: the address, labelled by its `id` key, else the class's `aff<n>`; its parts as the class prints them
140    // (:1298-1304 `\address@definitions`, without the space its body opens with; a `\\` the kernel's line break)
141    DefMacro!("\\address@fmt[]{}", sub[(keys, text)] {
142      let n = lookup_int("lx_bmcart_address_count") + 1;
143      assign_value("lx_bmcart_address_count", Stored::Int(n), Some(Scope::Global));
144      let id = keys
145        .as_ref()
146        .map(keyval_pairs)
147        .unwrap_or_default()
148        .into_iter()
149        .find(|(key, _)| key == "id")
150        .map(|(_, value)| value)
151        .unwrap_or_else(|| ExplodeText!(&s!("aff{n}")));
152      let mut out = TokenizeInternal!("\\lx@add@contact[role=affiliation,label=").unlist();
153      out.extend(braced(&id));
154      out.push(T_OTHER!("]"));
155      let mut shown = TokenizeInternal!(
156        "\\def\\orgdiv#1{#1}\\def\\orgname#1{#1}\\def\\street#1{#1}\\def\\postcode#1{#1}\\def\\postbox#1{#1}\\def\\city#1{#1}\\def\\state#1{#1}\\def\\cny#1{#1}\\let\\interref\\surl"
157      )
158      .unlist();
159      shown.extend(text.unlist());
160      out.extend(braced(&braced(&shown)));
161      Ok(Tokens::new(out))
162    });
163    // :1391-1395: an article note (`\note[id=n1]{Corresponding author}` in `{artnotes}`, :1371-1378), which the class
164    // stores by its `id` for `\printartnotes` (:1406-1409) to print under the title block's rule (:999-1002
165    // `\auto@set@thanksbox`, :1440-1447), a note of the authors whose `noteref` names it, as elsart_support_core's
166    // `\fntext[label]` is (1507.06515 "Corresponding author", 2007.00437 "Second corresponding author: …", 2112.10068
167    // "Equal contributor"); its `symbol` key the mark the class prints
168    DefMacro!("\\@bmcartnote[]{}", sub[(keys, text)] {
169      let id = keys
170        .as_ref()
171        .map(keyval_pairs)
172        .unwrap_or_default()
173        .into_iter()
174        .find(|(key, _)| key == "id")
175        .map(|(_, value)| value)
176        .filter(|value| !value.is_empty());
177      let mut out = TokenizeInternal!("\\lx@add@contact[role=note").unlist();
178      if let Some(id) = id {
179        out.extend(TokenizeInternal!(",label=").unlist());
180        let mut label = ExplodeText!("note:");
181        label.extend(id);
182        out.extend(braced(&label));
183      }
184      out.push(T_OTHER!("]"));
185      out.extend(braced(&text.unlist()));
186      Ok(Tokens::new(out))
187    });
188    // :993-1012: the `{fmbox}` only holds the frontmatter calls
189    DefMacro!("\\bmc@fmbox", "");
190    DefMacro!("\\bmc@endfmbox", "");
191  }
192});
