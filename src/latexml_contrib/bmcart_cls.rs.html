1//! bmcart.cls (BioMed Central's article class): the papers ship the class (it is not in TeX Live), and it is read raw
2//! (OXIDIZED_DESIGN_DIVERGENCES #444). Its `{frontmatter}` lets `\title`, `\author` and `\address` to its printers
3//! (:966-984 `\set@frontmatter@cmd`): `\title@fmt` typesets the title in the `{fmbox}` (:1034-1053), `\author@fmt`
4//! the name with the marks of its `addressref` keys (:1198-1255), and `\address@fmt` stores the address by its `id`
5//! (:1287-1296) for the "Author details" block that `\thebibliography` prints (:1708-1712 `\printaddresses`), which
6//! the kernel's bibliography constructor never runs. Read raw, the document had no title, no creator and no address
7//! (1711.01177: the title page as boxes, the marks "??"). So, after the read, the three printers are the frontmatter's,
8//! as elsart_support_core's `\author[labels]`/`\address[label]` are (elsart_support_core.sty.ltxml:35-42): each
9//! `\author` a creator requesting its `addressref`/`presentaddressref` ids, with its `email` addresses; each `\address`
10//! an affiliation labelled by its `id` (the class's `aff<n>` when it has none); and the `{fmbox}` the box only holds
11//! them. Its end prints the thanksbox (:1002 `\auto@set@thanksbox`), whose marks the binding settles there: each
12//! article note under the glyph the class prints, the corresponding author's e-mail named "Correspondence: "
13//! (2112.10068; OD #842), the "Full list of author information …" line a print flourish. Perl has no binding: OmniBus,
14//! whose `\author` read "[" as the name. Guards `perfect_kernel_batch78m::bmcart_frontmatter_is_frontmatter`,
15//! `perfect_kernel_batchbmcart`.

26/// The `key=value` pairs of a keyval list at its top level, each value without the spaces and the one pair of braces
27/// around it (`addressref={cmu1,eth}`: `cmu1,eth`); a key given no value (`corref`, :1153 its default) an empty one.
28fn keyval_pairs(list: &Tokens) -> Vec<(String, Vec<Token>)> {
29  let mut items: Vec<Vec<Token>> = vec![Vec::new()];
30  let mut depth = 0usize;
31  for t in list.unlist_ref() {
32    match t.get_catcode() {
33      Catcode::BEGIN => depth += 1,
34      Catcode::END => depth = depth.saturating_sub(1),
35      _ if depth == 0 && *t == T_OTHER!(",") => {
36        items.push(Vec::new());
37        continue;
38      },
39      _ => {},
40    }
41    if let Some(item) = items.last_mut() {
42      item.push(*t);
43    }
44  }
45  items
46    .into_iter()
47    .filter_map(|item| {
48      let eq = item
49        .iter()
50        .position(|t| *t == T_OTHER!("="))
51        .unwrap_or(item.len());
52      let key = Tokens::new(item[..eq].to_vec())
53        .to_string()
54        .trim()
55        .to_string();
56      if key.is_empty() {
57        return None;
58      }
59      let mut value: &[Token] = item.get(eq + 1..).unwrap_or_default();
60      while let [first, rest @ ..] = value
61        && first.get_catcode() == Catcode::SPACE
62      {
63        value = rest;
64      }
65      while let [rest @ .., last] = value
66        && last.get_catcode() == Catcode::SPACE
67      {
68        value = rest;
69      }
70      if let [open, inner @ .., close] = value
71        && open.get_catcode() == Catcode::BEGIN
72        && close.get_catcode() == Catcode::END
73      {
74        value = inner;
75      }
76      Some((key, value.to_vec()))
77    })
78    .collect()
79}

107/// What a mark prints (`*`, `\textdagger`, `\^{}`; a `symbol` key's `$\star$`, its math symbol's glyph; `**`), as
108/// digested.
109fn printed_mark(mark: Tokens) -> Result<String> {
110  let ink: Vec<&Token> = mark
111    .unlist_ref()
112    .iter()
113    .filter(|t| {
114      !matches!(
115        t.get_catcode(),
116        Catcode::SPACE | Catcode::BEGIN | Catcode::END
117      )
118    })
119    .collect();
120  if let [open, symbol, close] = ink.as_slice()
121    && open.get_catcode() == Catcode::MATH
122    && close.get_catcode() == Catcode::MATH
123    && let Some(glyph) = math_symbol_glyph(symbol)
124  {
125    return Ok(glyph);
126  }
127  Ok(digest_text(mark)?.to_string().trim().to_string())
128}

130/// The glyph of the class's `thanks` counter at `n` (:1095-1097 `\thethanks`, `\getitemvalue{bmcsymbol}{n}`: *, †,
131/// ^), none past its list, where pdflatex prints none (and errs on the label it writes).
132fn thanks_mark(n: usize) -> Result<String> {
133  Ok(
134    match lookup_definition(&T_CS!(s!("\\bmcsymbol@item@{n}")))? {
135      Some(defn) => match defn.get_expansion() {
136        Some(ExpansionBody::Tokens(body)) => printed_mark(body.clone())?,
137        _ => String::new(),
138      },
139      None => String::new(),
140    },
141  )
142}

144/// Whether the class's macro `cs` holds `pattern` in its body.
145fn body_holds(cs: &str, pattern: &[Token]) -> Result<bool> {
146  Ok(match lookup_definition(&T_CS!(cs))? {
147    Some(defn) => matches!(defn.get_expansion(), Some(ExpansionBody::Tokens(body))
148      if body.unlist_ref().windows(pattern.len()).any(|at| at == pattern)),
149    None => false,
150  })
151}

153/// `tokens` in braces.
154fn braced(tokens: &[Token]) -> Vec<Token> {
155  let mut out = Vec::with_capacity(tokens.len() + 2);
156  out.push(T_BEGIN!());
157  out.extend_from_slice(tokens);
158  out.push(T_END!());
159  out
160}

406                "frontmatter",

410            }
411            mark
412          },
413        };
414        // (a repeated id keeps the label its first mark wrote, :334-338 `\safelabel`)

417        }

419        if !printed.is_empty() {

421        }
422      }
