1//! ragslab.sty (Resilient AI + Grounded Sensing preprint style, loaded on `article`), whose title API "mirrors the
2//! FAIR/Meta template": the paper's copy read raw, for its palette, logos, layout and `\beginappendix`; then the
3//! fairmeta family's frontmatter (`meta_class.rs`), so `\author[1,2]{…}` links each author to the `\affiliation[1]{…}`
4//! its marks cite, and the metadata band's lines (`\metadata[<label>]{…}`, `\date`, `\correspondence`, `\code`,
5//! `\projectpage`) are frontmatter with the label the copy prints (ragslab.sty:140-161; 2610.22523; OD #1070).
6use latexml_package::{prelude::*, shipped_class::input_shipped_package};
7
8LoadDefinitions!({
9  input_shipped_package()?;
10  // The copy's `\maketitle` (:278-284) typesets its lists in a tcolorbox card; the lock refused it, and its replay
11  // would print the card, its lists now empty, beside the frontmatter (2610.22523: a picture after the abstract).
12  // Only the card's logo strip (`\supportlogo`/`\lablogo`, :137-165) is printed nowhere else: it stays, a block of
13  // its own (revRAGSLAB B1; 2610.22523's three logos).
14  DefMacro!(
15    "\\lx@dropped@maketitle",
16    "\\ifdefined\\reslabrenderlogos\\par\\noindent\\reslabrenderlogos\\par\\fi"
17  );
18  // (the copy's `\beginappendix` prints its "Appendix" heading, :286; the shared one is a bare `\appendix`)
19  Let!("\\lx@ragslab@beginappendix", "\\beginappendix");
20  crate::meta_class::install_meta_class_frontmatter()?;
21  if IsDefined!(&T_CS!("\\lx@ragslab@beginappendix")) {
22    Let!("\\beginappendix", "\\lx@ragslab@beginappendix");
23  }
24  // A metadata line prints "<label>: <value>" (:151-152): a publication note named by its label. `\code` and
25  // `\projectpage` are the copy's calls of it (:160-161), `\correspondence` too (:159), its role the family's.
26  DefMacro!(
27    "\\metadata[]{}",
28    "\\lx@add@pubnote[role=metadata,name={#1:~}]{#2}"
29  );
30  DefMacro!(
31    "\\correspondence{}",
32    "\\lx@add@pubnote[role=correspondence,name={Correspondence:~}]{#1}"
33  );
34  // `\date` is the copy's "Date:" line (:158), the document's date.
35  DefMacro!(
36    "\\date{}",
37    "\\gdef\\@date{#1}\\lx@add@date[role=creation,name={Date:}]{#1}"
38  );
39});
