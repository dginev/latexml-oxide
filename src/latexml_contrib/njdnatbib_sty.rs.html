1//! NJDnatbib.sty — Wiley's renamed copy of natbib 8.31b (`\ProvidesPackage{NJDnatbib}`, NJDnatbib.sty:29), which
2//! WileyNJD-v2/v5/MRM.cls load in natbib's place (WileyNJDv5.cls:6021-6085 `\usepackage[numbers,super,…]{NJDnatbib}`,
3//! MRM.cls:2441-2442 `\usepackage[numbers,super]{NJDnatbib}`). A stand-in for the author's file (author_files.rs
4//! `AUTHOR_FILE_STAND_INS`): it is natbib, so the natbib binding loads with the options the class gave the copy
5//! ([`natbib_fork_options`]), and the paper's copy, read after it, only provides (OXIDIZED_DESIGN_DIVERGENCES #731):
6//! read raw, its `\bibsection`'s `\@ifundefined{chapter}` took the `\chapter*` branch, heading the references
7//! "Bibliography" with `bib.bibxN` ids (2609.08425, 2101.06297, 2211.16474, 2411.11493, 2206.08798, 2602.03829,
8//! 2408.07575, 2506.14995).
9//!
10//! The defaults its copies change that the PDF shows, by version (the class-family rule's dispatch on a conflict):
11//! - v5 (md5 277c41da: 2602.03829, 2408.07575, 2609.08425): a class without chapters heads the bibliography
12//!   "REFERENCES" (:1050-1053 `\bmsection*{…{{REFERENCES}}…}`; xelatex 2609.08425; the reviewer's probe njd_direct_v5),
13//!   an amsart one `\refname` (:1065);
14//! - v2 and MRM (2fa96b7a: 2101.06297, 2211.16474; 6838c57e: 2411.11493, 2206.08798; 4b333a37: 2506.14995): natbib's
15//!   own `\section*{\refname…}` (:1052; probe njd_direct_v2: pdflatex "References"; MRM.cls:3191's `\refname`
16//!   "REFERENCES" is the class's);
17//! - every version: the superscript citation carries its postnote inside, ", p. 5" (:366-368; natbib's switch
18//!   `NATBIB_SUPER_POSTNOTE_INSIDE`).
19//!
20//! The version is v5's class (`\@ifclassloaded{WileyNJDv5}`) or the copy's own `\bibsection` set by `\bmsection`, v5's
21//! marker; a class without chapters is decided by the one "has chapters" test, as the copy's `\@ifundefined{chapter}`
22//! does (`state::class_defines_chapter`, OXIDIZED_DESIGN #179): not by a `chapter` counter, which the Wiley classes
23//! declare without a `\chapter` (WileyNJDv5.cls:5728, MRM.cls:2160 `\newcounter{chapter}[part]`). The WileyNJDv5
24//! binding's OmniBus base keeps no kernel `\chapter` (a named binding is no unknown class), so the copy's test is TeX's
25//! (2609.08425: "REFERENCES").
26use latexml_package::prelude::*;
27
28/// A renamed natbib copy's options are natbib's (natbib.sty:234-285): the natbib binding loads next with each one
29/// the class gives the copy. A natbib loaded already (a class binding's, or the document's) takes them as a repeat
30/// load does, as natbib's options, with an Info (natbib_sty.rs `\lx@natbib@repeat@options`) — not dropped silently
31/// (MRM's binding had loaded natbib optionless before MRM.cls's `[numbers,super]{NJDnatbib}`: 2506.14995's
32/// author-year where its PDF has superscripts).
33pub fn natbib_fork_options() -> Result<()> {
34  DefMacro!(T_CS!("\\lx@natfork@options"), None, Tokens!(), scope => Some(Scope::Global));
35  DeclareOption!(None, {
36    Digest!("\\xdef\\lx@natfork@options{\\lx@natfork@options,\\CurrentOption}")?;
37  });
38  ProcessOptions!();
39  let options: Vec<String> = Expand!(T_CS!("\\lx@natfork@options"))
40    .to_string()
41    .split(',')
42    .map(|option| option.trim().to_string())
43    .filter(|option| !option.is_empty())
44    .collect();
45  // (the copy's own options, which its `\ProcessOptions` takes in TeX, even after a natbib loaded already:
46  // `natbib.sty:options_of_a_renamed_copy`, binding/content.rs `apply_new_options_on_reload`)
47  AssignValue!("natbib.sty:options_of_a_renamed_copy" => true, Some(Scope::Global));
48  let loaded = require_package("natbib", NewDefault!(RequireOptions, options => options));
49  AssignValue!("natbib.sty:options_of_a_renamed_copy" => false, Some(Scope::Global));
50  loaded
51}
52
53LoadDefinitions!({
54  natbib_fork_options()?;
55  AssignValue!("NATBIB_SUPER_POSTNOTE_INSIDE" => true, Some(Scope::Global));
56  // (after the copy, which is read at the end of this load: its `\bibsection` tells v5)
57  RawTeX!(r"\AtEndOfPackage{\lx@njd@bibsection@title}");
58  DefPrimitive!("\\lx@njd@bibsection@title", {
59    let class = |name: &str| has_meaning(&T_CS!(s!("\\ver@{name}.cls")));
60    let marked = matches!(
61      stand_in_copy_meaning("NJDnatbib.sty", &T_CS!("\\bibsection")),
62      Some(Stored::Expandable(defn)) if matches!(&defn.expansion,
63        Some(ExpansionBody::Tokens(body)) if body.unlist_ref().contains(&T_CS!("\\bmsection")))
64    );
65    let v5 = class("WileyNJDv5") || class("wileyNJDv5") || marked;
66    if v5 && !class_defines_chapter() && !class("amsart") {
67      DefMacro!(T_CS!("\\lx@bibliography@title"), None, Tokenize!("REFERENCES"), scope => Some(Scope::Global));
68    }
69  });
70});
