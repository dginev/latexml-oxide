1//! an.cls (Astronomische Nachrichten, Wiley-VCH; January 2004, revised 2006-01-13): the papers ship the class (it is
2//! not in TeX Live), which is read raw. A paper whose source lacks the class falls to OmniBus, with the loader's
3//! missing-file Warning.
4use latexml_package::prelude::*;
5
6const AN: crate::shipped_class::RawRead = crate::shipped_class::RawRead {
7  shipped: true,
8  setters: &[("\\aaabstract", r"\lx@add@abstract{#1}")],
9  after: r"\ifdefined\aaabstract\ifdefined\@abstract\else\let\@abstract\@empty\fi\fi",
10  ..crate::shipped_class::RawRead::new("an", "cls")
11};
12
13LoadDefinitions!({
14  if !crate::shipped_class::read_raw(&AN)? {
15    return Ok(());
16  }
17  // an.cls:414 `\def\aaabstract#1{\gdef\@abstract{#1}}`, which the class makes the document's `\abstract` at its end
18  // (:98 `\AtEndOfClass{\let\abstract=\aaabstract}`), stores the abstract for its title page (`\@maketitle` :718
19  // `{\@abstract}`): the text was one plain paragraph of the deposited title block, no abstract. The setter is the
20  // frontmatter's abstract where the class defines it, before its end-of-class hook copies it onto `\abstract` (the
21  // hook runs at the end of this binding's load); the title page reads an empty store, which the class never sets.
22  // A class only named like this one has no such setter and is left as it is: `andp2012.cls`, which the prefix
23  // fallback answers with this binding (1504.05091, 1810.06303), and the 2014 an.cls v1.12, which defines `\abstract`
24  // itself (:422; 1810.02388, 2103.06625). Witness astro-ph0611016 (`\abstract{We consider the contribution of
25  // microlensing …}`).
26  //
27  // v1.03 (2002/07/04, astro-ph0306340) prints the class's label before the store (`\@maketitle` :662
28  // `\noindent{\bfseries\abstractname}\ \@abstract`; v1.09a and later comment the line out): the label is the
29  // abstract's name, and the title-page deposit takes the line out of `\@maketitle` (`note_title_page_part`), so no
30  // lone bold label stays behind; `\abstractname` itself is left as it is.
31  const LABELLED_STORE: &str = r"\noindent{\bfseries\abstractname}\ \@abstract";
32  let labelled = crate::shipped_class::shipped_class_text("an")?
33    .is_some_and(|text| crate::shipped_class::class_code_contains(&text, LABELLED_STORE));
34  if labelled {
35    RawTeX!(
36      r"\ifdefined\aaabstract\def\aaabstract#1{\lx@add@abstract[name={\abstractname}]{#1}}\fi"
37    );
38    latexml_engine::latex_constructs::note_title_page_part(
39      "\\@maketitle",
40      TokenizeInternal!(LABELLED_STORE),
41    );
42  }
43});
