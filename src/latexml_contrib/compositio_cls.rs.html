1//! The Foundation Compositio Mathematica class family — `compositio`, `alggeom` and `moduli` (`CPM@` internals) and the
2//! older `hha`, `jhrs` and `jcm` built from the same code (`HHA@`…): shipped with each paper, not in TeX Live. One
3//! binding serves them all, the union of their frontmatter commands.
4use latexml_package::prelude::*;
5
6/// Interpret the shipped class `name` raw (as the document asked for it, a path-prefixed copy included:
7/// [`latexml_package::shipped_class::requested_shipped_class`]), then route the frontmatter it stores for its own title page and closing
8/// address block to the frontmatter:
9/// - `\address`, `\curraddr`, `\email`, `\homepage` (compositio.cls:141-165, hha.cls:104-137) append to the current
10///   author's `\@curr@author` and the `\AtEndDocument` `\@addresses` block, and error without an `\author` the kernel
11///   lock had refused (182 compositio, 34 alggeom, 3 hha, 2 jhrs, 2 jcm, 1 moduli of run 336's papers: 2606.19241,
12///   1109.5540, math0409234, 0805.3030, quant-ph0412067, 2111.06381); here each is the last author's contact, labelled
13///   as the class prints it (`\textit{Current address:}`, compositio.cls:149-150);
14/// - `\keywords` and `\classification` (`\subjclass`, compositio.cls:181-189, hha.cls:146-149) store for `\maketitle`
15///   only; here they are the frontmatter's keywords and MSC classification, the label the class's own definition prints
16///   (its MSC year: 2000 in hha, 2010 in compositio and alggeom, 2020 in moduli);
17/// - `\thanks` (compositio.cls:174-176) stores a `\footnotetext` for `\maketitle` only; here it is the kernel's;
18/// - `\received`, `\revised`, `\online` (compositio.cls:192-206), `\published` (hha.cls:161) and `\dedication`
19///   (compositio.cls:179) store for `\maketitle` only; here they are the frontmatter's dates and dedication;
20/// - the class's own `\maketitle` (compositio.cls:209-271) lays out only those fields, so it is not deposited
21///   (OXIDIZED_DESIGN #265): with `\@authors` filled by its replayed `\author` (#482) it typeset the names again in
22///   the body, its `\address\relax` an empty contact, and its emptied stores' warnings ("No MSC classification").
23///
24/// A paper whose source lacks the class falls to OmniBus, with the loader's missing-file Warning, as before the binding
25/// (OXIDIZED_DESIGN #444).
26pub(crate) fn compositio_family_class(name: &str) -> Result<()> {
27  Let!("\\lx@compositio@kernel@thanks", "\\thanks");
28  if !crate::shipped_class::read_raw(&crate::shipped_class::RawRead {
29    shipped: true,
30    ..crate::shipped_class::RawRead::new(name, "cls")
31  })? {
32    return Ok(());
33  }
34  // a class only named like the family (the prefix fallback's `jcmfoo`, `hhafoo`) is left as it is: without the
35  // family's current-author store (`\@curr@author`, compositio.cls:126, hha.cls, jcm.cls:7), it has none of its setters
36  if !IsDefined!(&T_CS!("\\@curr@author")) {
37    return Ok(());
38  }
39  let msc = msc_label();
40  // hha.cls:142-144, jhrs: "Key words and phrases:"; compositio.cls:181-183: "Keywords:"
41  let keywords = class_body_text("\\keywords");
42  let keywords = if keywords.contains("Key words and phrases") {
43    "Key words and phrases:"
44  } else {
45    "Keywords:"
46  };
47  let homepage_args = class_parameter_count("\\homepage");
48  Let!("\\thanks", "\\lx@compositio@kernel@thanks");
49  DefMacro!("\\address[]{}", "\\lx@add@address{#2}");
50  DefMacro!(
51    "\\curraddr[]{}",
52    "\\lx@add@contact[role=current_address,name={\\textit{Current address:}\\ }]{#2}"
53  );
54  DefMacro!("\\email[] Semiverbatim", "\\lx@add@email{#2}");
55  // jcm.cls:157 `\homepage{url}{text}` (`\href{#1}{\tt #2}`); hha.cls:133 `\homepage{url}`
56  if homepage_args == 2 {
57    DefMacro!(
58      "\\homepage Semiverbatim {}",
59      "\\lx@add@contact[role=url]{\\href{#1}{\\texttt{#2}}}"
60    );
61  } else {
62    DefMacro!("\\homepage Semiverbatim", "\\lx@add@url{#1}");
63  }
64  RawTeX!(&s!("\\def\\lx@compositio@keywords{{{keywords}}}"));
65  DefMacro!(
66    "\\keywords{}",
67    "\\lx@add@keywords[name={\\lx@compositio@keywords\\ }]{#1}"
68  );
69  RawTeX!(&s!("\\def\\lx@compositio@msc{{{msc}}}"));
70  DefMacro!(
71    "\\classification[]{}",
72    "\\lx@add@classification[scheme={Mathematics Subject Classification},name={\\lx@compositio@msc\\ }]{#2}"
73  );
74  Let!("\\subjclass", "\\classification");
75  DefMacro!(
76    "\\received{}",
77    "\\lx@add@date[role=received,name={Received }]{#1}"
78  );
79  // compositio.cls:195/:205 `\revised[2][revised]`, `\online[2][published online]`: the label is the author's
80  DefMacro!(
81    "\\revised[Default:revised]{}",
82    "\\lx@add@date[role=revised,name={#1 }]{#2}"
83  );
84  DefMacro!(
85    "\\online[Default:published online]{}",
86    "\\lx@add@date[role=published,name={#1 }]{#2}"
87  );
88  DefMacro!(
89    "\\published{}",
90    "\\lx@add@date[role=published,name={Published }]{#1}"
91  );
92  // hha.cls:164-166, jhrs: `({\itshape communicated by #1})` in the title block
93  DefMacro!(
94    "\\submitted{}",
95    "\\lx@add@pubnote[role=communicated,name={communicated by }]{#1}"
96  );
97  // compositio.cls:179, :244-245: the dedication in italics, no label (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #921) (2501.00323).
98  DefMacro!(
99    "\\dedication{}",
100    "\\lx@add@pubnote[role=dedication,name={}]{#1}"
101  );
102  Let!("\\lx@dropped@maketitle", "\\relax");
103  Let!("\\@maketitle", "\\@empty");
104  Ok(())
105}

107/// The text of the class's own definition of `cs` (empty where it has none).
108fn class_body_text(cs: &str) -> String {
109  match lookup_meaning(&T_CS!(cs)) {
110    Some(Stored::Expandable(ref d)) => match d.get_expansion() {
111      Some(ExpansionBody::Tokens(body)) => body.clone().untex(),
112      _ => String::new(),
113    },
114    _ => String::new(),
115  }
116}
117
118/// How many parameters the class's own `cs` takes (0 where it has none).
119fn class_parameter_count(cs: &str) -> usize {
120  match lookup_meaning(&T_CS!(cs)) {
121    Some(Stored::Expandable(ref d)) => d.get_parameters().map_or(0, |p| p.get_parameters().len()),
122    _ => 0,
123  }
124}

126/// The MSC label the class's own `\classification` prints — "<year> Mathematics Subject Classification" (the year
127/// before the words, in its body), or the words alone.
128fn msc_label() -> String {
129  let words = "Mathematics Subject Classification";
130  let body = class_body_text("\\classification");
131  let Some(at) = body.find(words) else {
132    return words.to_string();
133  };
134  let year: String = body[..at]
135    .trim_end()
136    .chars()
137    .rev()
138    .take(4)
139    .collect::<Vec<_>>()
140    .into_iter()
141    .rev()
142    .collect();
143  // hha.cls:148 prints its colon ("2000 Mathematics Subject Classification:"), compositio.cls:187 none
144  let colon = if body[at + words.len()..].starts_with(':') {
145    ":"
146  } else {
147    ""
148  };
149  if year.len() == 4 && year.chars().all(|c| c.is_ascii_digit()) {
150    s!("{year} {words}{colon}")
151  } else {
152    s!("{words}{colon}")
153  }
154}
