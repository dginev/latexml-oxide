1//! brief.cls (ntgclass, Dutch business letters): the class is loaded raw, and the letter's sender — which the class
2//! prints only in its first-page head and foot (`\ps@firstpage`, brief.cls:285-294: `\briefhoofd` from
3//! `\maakbriefhoofd{<name>}{<address>}`, :437-458, and the `\voetitem{<label>}{<value>}` foot, :459-468), a page style
4//! LaTeXML never typesets — becomes the letter's frontmatter at the class's own print site, the start of `{brief}`
5//! (:321): a `creator` of role `sender` named by the head's name, with the address and each foot item as a `contact`
6//! (the item's label as its `name`). As the g-brief bindings (OXIDIZED_DESIGN_DIVERGENCES #412, user ruling 2026-10-01:
7//! a letter's sender is frontmatter, the rest of the page furniture). SHARED: Perl with raw classes drops it alike. Only
8//! a document's first letter is given (one frontmatter a document). Witness ntgclass/brief-sample. Guard
9//! `perfect_kernel_batch61::brief_letter_sender_is_frontmatter`.
10use latexml_package::prelude::*;
11
12LoadDefinitions!({
13  InputDefinitions!("brief", noltxml => true, extension => Some(Cow::Borrowed("cls")));
14  // A foot label may break its line (`\voetitem{telefoon \\ priv\'e:}`, brief-sample.tex:16); a contact's name is
15  // one line, so the break is a space there.
16  DefMacro!("\\lx@brief@label{}", sub[(label)] {
17    let mut out: Vec<Token> = Vec::new();
18    for t in label.unlist() {
19      let t = if t == T_CS!("\\\\") { T_SPACE!() } else { t };
20      let space = t.get_catcode() == Catcode::SPACE;
21      if space && out.last().is_some_and(|l| l.get_catcode() == Catcode::SPACE) {
22        continue;
23      }
24      out.push(t);
25    }
26    Ok(Tokens::new(out))
27  });
28  RawTeX!(
29    r"\let\lx@brief@maak\@maakbriefhoofd
30\def\@maakbriefhoofd#1#2{\lx@brief@maak{#1}{#2}\gdef\lx@brief@name{#1}\gdef\lx@brief@address{#2}}
31\gdef\lx@brief@contacts{}
32\let\lx@brief@voetitem\voetitem
33\def\voetitem#1#2{\lx@brief@voetitem{#1}{#2}\g@addto@macro\lx@brief@contacts{\lx@add@contact[name={\lx@brief@label{#1}}]{#2}}}
34\let\footitem\voetitem
35\def\lx@brief@frontmatter{\global\let\lx@brief@frontmatter\relax
36  \ifx\lx@brief@name\@undefined\else
37    \lx@add@creator[role=sender]{\lx@brief@name}%
38    \lx@add@contact[role=address]{\lx@brief@address}\fi
39  \lx@brief@contacts}
40\let\lx@brief@begin\brief
41\def\brief{\lx@brief@frontmatter\lx@brief@begin}
42\def\lx@contact@address@name{}"
43  );
44});
