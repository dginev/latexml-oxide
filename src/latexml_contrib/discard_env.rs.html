1//! Shared helper: discard an environment body by reading tokens up to
2//! matching `\end{kind}`. Mirrors the Perl `discard_env_body` closures
3//! used in ar5iv-bindings/nicematrix / forest / diagrams / pb-diagram
4//! stub packages. Here the users are pb-diagram's stub and the body readers
5//! of forest and animate; Paul Taylor's `diagrams` is a native binding
6//! (diagrams_tex.rs, OXIDIZED_DESIGN_DIVERGENCES #589).
7//!
8//! Perl reference (identical across those stubs):
9//! ```perl
10//! sub discard_env_body {
11//!   my ($stomach, $kind) = @_;
12//!   my $gullet = $stomach->getGullet;
13//!   $stomach->bgroup;
14//!   if (!$reported{$kind}) {
15//!     $reported{$kind} = 1;
16//!     Error('undefined', "{$kind}", $gullet,
17//!       "$kind has no support, this is a stub binding."); }
18//!   while (my $_ = $gullet->readUntil(T_CS('\end'))) {
19//!     my $_drop_open = $gullet->readToken;
20//!     my $env        = $gullet->readBalanced;
21//!     last if ToString($env) eq $kind; }
22//!   $stomach->egroup;
23//!   return; }
24//! ```

26use latexml_package::prelude::*;
27use rustc_hash::FxHashSet as HashSet;

29/// One error per kind per conversion run, matching Perl's `our $reported{$kind}` cache inside each stub package. A rollback
30/// puts it back with the diagnostic it discards: a rehearsal's report left the document's own unsaid.
31static REPORTED: DocumentState<HashSet<String>> = DocumentState::new("discard_env_reported");
32
33/// Read the raw (unexpanded) tokens of the current environment body up to
34/// the first `\end{kind}`, which is consumed and not returned; an `\end` of
35/// another environment inside the body is kept. The one mechanism behind
36/// every binding that captures or discards a body wholesale (forest's bracket
37/// parser, animate's frame discard, the stub discards below) — batch 56bc.
38/// `read_balanced(…, false, false)`: the `{` after `\end` was just consumed,
39/// so the balanced read starts inside it (Perl's argless `readBalanced`).
40pub fn read_env_body_tokens(kind: &str) -> Result<Vec<Token>> {
41  let end_delim = Tokens!(T_CS!("\\end"));
42  let mut body: Vec<Token> = Vec::new();
43  loop {
44    if let Some(toks) = read_until(&end_delim)? {
45      body.extend(toks.unlist());
46    }
47    let Some(open) = read_token()? else {
48      break; // end of input
49    };
50    let env = read_balanced(ExpansionLevel::Off, false, false)?;
51    if env.to_string() == kind {
52      break;
53    }
54    body.push(T_CS!("\\end"));
55    body.push(open);
56    body.extend(env.unlist());
57    body.push(T_END!());
58  }
59  Ok(body)
60}
61
62/// Read and discard the body up to and including the matching `\end{kind}`,
63/// with a one-time stub warning per `kind`.
64pub fn discard_env_body(kind: &str, source: &str) -> Result<()> {
65  bgroup();
66  report_stub_once(kind, source)?;
67  let _body = read_env_body_tokens(kind)?;
68  egroup()?;
69  Ok(())
70}
71
72/// The one-per-kind stub diagnostic shared by the discard helpers.
73fn report_stub_once(kind: &str, source: &str) -> Result<()> {
74  let first_time = REPORTED.update(|set| {
75    if set.contains(kind) {
76      false
77    } else {
78      set.insert(kind.to_string());
79      true
80    }
81  });
82  if first_time {
83    let obj = format!("{{{}}}", kind);
84    let msg = format!(
85      "{} has no support in {}, this is a stub binding.",
86      kind, source
87    );
88    // A Warn, not an Error: the body IS discarded cleanly, and the ar5iv
89    // binding's Error (forest.sty.ltxml:37-38) was the only diagnostic of
90    // forest-quickstart, fragoli_doc and milsymb (pdflatex clean; sweep #41).
91    // Guard: `perfect_kernel_batch56::forest_stub_is_a_warning`.
92    Warn!("undefined", &obj, msg);
93  }
94  Ok(())
95}

97/// Discard the body of the BARE-CS environment form `\<kind> … \end<kind>`
98/// (what `\NewDocumentEnvironment` defines alongside `\begin{<kind>}`):
99/// raw, non-expanding scan up to the `end_cs` token, sharing the
100/// one-per-kind stub report with [`discard_env_body`].
101pub fn discard_body_until_cs(kind: &str, end_cs: &str, source: &str) -> Result<()> {
102  bgroup();
103  report_stub_once(kind, source)?;
104  let _body = read_until(&Tokens!(T_CS!(end_cs)))?;
105  egroup()?;
106  Ok(())
107}
