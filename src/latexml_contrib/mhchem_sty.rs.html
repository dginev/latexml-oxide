1//! mhchem.sty — chemical formula typesetting (`\ce`, `\cee`, `\cf`, `\cesplit`,
2//! `\bond`, …).
3//!
4//! **2026-06-27: RAW-LOAD the genuine TL `mhchem.sty`.** Perl LaTeXML ships no
5//! `mhchem.sty.ltxml` and raw-loads the real package; the engine's expl3 /
6//! xparse / chemgreek support is now mature enough that we can do the same, so
7//! this binding simply `InputDefinitions(noltxml)` the real file. Chemistry
8//! therefore renders with proper digit subscripts (`\ce{H2O}` → H₂O), charge
9//! superscripts (`\ce{SO4^2-}` → SO₄²⁻), reaction arrows (`->`/`<=>`/`->[..]`),
10//! bonds, states, `\cesplit`, etc.
11//!
12//! This SUPERSEDES the former minimal "`\ce`-as-roman-text" stub (retired here;
13//! see git history at `latexml_contrib/src/mhchem_sty.rs` before this commit).
14//!
15//! **2026-11-14 (issue 824): a formula's units are chemistry math.** mhchem's own parser
16//! still reads `\ce`; only its output layer's formula hook is bound
17//! (`semantic_formula`): elements, isotopes, counts and charges (OXIDIZED_DESIGN_DIVERGENCES #1040).
18//! Arrows, `+` between species, bonds, states and amounts keep mhchem's output (stage 2).
19//! A formula's elements are juxtaposed (no U+2062) and a charge sign is tight (DG 2026-11-14).
20//! A formula holding a macro that expands to a math shift keeps mhchem's output (2606.03586, `shifts_math`).
21//!
22//! This stage-1 markup follows DG's TENTATIVE rulings of 2026-11-14 and awaits Bruce Miller's feedback and
23//! refinements; stage 2 adds MathML `intent`. DG: "We expect further improvements are both possible and needed."
24//!
25//! Known edge-case residuals under raw-load (tracked in docs/SYNC_STATUS.md):
26//! `\ce` used inside an amsmath `align*` alignment, and a few `$`-toggle /
27//! `\cesplit`-derived example patterns, can still emit `\lx@begin@alignment` /
28//! `\lx@end@inline@math` errors. Simple `\ce` (the overwhelmingly common case)
29//! is clean.

84/// One `\__mhchem_output_coreFive:nnnnnnn` call (mhchem.sty:2774): pre-superscript (mass number),
85/// pre-subscript (atomic number), the element run, oxidation and stacked superscripts, the count, the charge.
86struct CoreFive {
87  presup:  Vec<Token>,
88  presub:  Vec<Token>,
89  element: Vec<Token>,
90  ox:      Vec<Token>,
91  kv:      Vec<Token>,
92  count:   Vec<Token>,
93  charge:  Vec<Token>,
94}
95
96enum FormulaItem {
97  Unit(CoreFive),
98  /// `\__mhchem_output_withFont:n {…}`: the formula's own text, its brackets (`Fe(CN)6`).
99  Text(Vec<Token>),
100  /// `\__mhchem_output_amount:n {…}` (mhchem.sty:2769): a stoichiometric amount, before its species.
101  Amount(Vec<Token>),
102  Raw(Token),
103}
104
105/// The next balanced `{…}` argument's tokens, spaces before it skipped.
106fn braced(tokens: &mut impl Iterator<Item = Token>) -> Option<Vec<Token>> {
107  let mut open = tokens.next()?;
108  while open.get_catcode() == Catcode::SPACE {
109    open = tokens.next()?;
110  }
111  if open.get_catcode() != Catcode::BEGIN {
112    return None;
113  }
114  let mut depth = 0usize;
115  let mut arg = Vec::new();
116  loop {
117    let t = tokens.next()?;
118    match t.get_catcode() {
119      Catcode::BEGIN => depth += 1,
120      Catcode::END if depth == 0 => return Some(arg),
121      Catcode::END => depth -= 1,
122      _ => {},
123    }
124    arg.push(t);
125  }
126}

128/// mhchem's formula output list (`\l__mhchem_cf_result_tl`) with each coreFive unit as chemistry math, in
129/// mhchem's math-mode fonts: an element is an ordinary atom (`\mathord`, so `MoO` stays `Mo`,`O`), its
130/// isotope prescripts sit on its first element, the count on its last, and a closing charge goes on the
131/// whole formula (`\ce{^{95}MoO4^{2-}}`, `\ce{Fe(CN)6^{3-}}`). `None` (mhchem's own output) when the list
132/// has no unit, an argument is not braced, or a macro of the list expands to a math shift.
133fn semantic_formula(result: &[Token]) -> Option<Vec<Token>> {
134  if shifts_math(result, false, &mut Vec::new()) {
135    return None;
136  }
137  let core_five = T_CS!("\\__mhchem_output_coreFive:nnnnnnn");
138  let with_font = T_CS!("\\__mhchem_output_withFont:n");
139  let amount = T_CS!("\\__mhchem_output_amount:n");
140  let mut items = Vec::new();
141  let mut tokens = result.iter().copied();
142  while let Some(t) = tokens.next() {
143    if t == with_font {
144      items.push(FormulaItem::Text(braced(&mut tokens)?));
145    } else if t == amount {
146      items.push(FormulaItem::Amount(braced(&mut tokens)?));
147    } else if t == core_five {
148      items.push(FormulaItem::Unit(CoreFive {
149        presup:  braced(&mut tokens)?,
150        presub:  braced(&mut tokens)?,
151        element: braced(&mut tokens)?,
152        ox:      braced(&mut tokens)?,
153        kv:      braced(&mut tokens)?,
154        count:   braced(&mut tokens)?,
155        charge:  braced(&mut tokens)?,
156      }));
157    } else {
158      items.push(FormulaItem::Raw(t));
159    }
160  }
161  let last_unit = items
162    .iter()
163    .rposition(|i| matches!(i, FormulaItem::Unit(_)))?;
164  // The species: the items after the last amount (`\ce{2e-}`, `\ce{3SO4^{2-}}`).
165  let start = items
166    .iter()
167    .rposition(|i| matches!(i, FormulaItem::Amount(_)))
168    .map_or(0, |a| a + 1);
169  let space =
170    |i: &FormulaItem| matches!(i, FormulaItem::Raw(t) if t.get_catcode() == Catcode::SPACE);
171  let more = items[start.min(last_unit)..last_unit]
172    .iter()
173    .any(|i| !space(i));
174  // A closing charge belongs to the species when it is more than one element (`\ce{\textit{Ln}O^{2-}}`).
175  let lifted = match &items[last_unit] {
176    FormulaItem::Unit(u) if !u.charge.is_empty() && items[last_unit + 1..].iter().all(space) => {
177      (more || element_items(&u.element).len() != 1).then(|| u.charge.clone())
178    },
179    _ => None,
180  };
181  let mut body = Vec::new();
182  let mut species = 0;
183  let mut before = Before::Nothing;
184  let mut run = Run::default();
185  // A raw command with its argument (2407.15836 `\textit{R}`, `\mathrm{Y}`) is an atom of the run; any other raw
186  // token (`\cdot`, mhchem's `\bond{-}` and own output) parts it, its braced arguments with it.
187  let heads: Vec<bool> = (0..items.len())
188    .map(|i| match (&items[i], items.get(i + 1)) {
189      (FormulaItem::Raw(t), Some(FormulaItem::Raw(next))) => {
190        t.get_catcode() == Catcode::CS
191          && next.get_catcode() == Catcode::BEGIN
192          && !t.with_str(|s| s.starts_with("\\__mhchem_") || s == "\\bond")
193      },
194      _ => false,
195    })
196    .collect();
197  let (mut in_raw, mut raw_depth) = (false, 0usize);
198  for (i, item) in items.into_iter().enumerate() {
199    if i == start {
200      species = body.len();
201    }
202    if !matches!(item, FormulaItem::Raw(_)) {
203      in_raw = false;
204    }
205    match item {
206      FormulaItem::Raw(t) => {
207        let code = t.get_catcode();
208        if raw_depth == 0 && code != Catcode::SPACE && !(in_raw && code == Catcode::BEGIN) {
209          if heads[i] {
210            run.open(body.len(), 1);
211          } else {
212            run.close(&mut body);
213          }
214          in_raw = true;
215          before = Before::Atom;
216        }
217        match code {
218          Catcode::BEGIN => raw_depth += 1,
219          Catcode::END => raw_depth = raw_depth.saturating_sub(1),
220          _ => {},
221        }
222        body.push(t);
223      },
224      FormulaItem::Amount(a) => {
225        run.close(&mut body);
226        body.extend(call("\\__mhchem_output_amount:n", a));
227        before = Before::Nothing;
228      },
229      FormulaItem::Text(text) => {
230        run.open(body.len(), 1);
231        body.extend(math_mode(font(text)));
232        before = Before::Atom;
233      },
234      FormulaItem::Unit(mut u) => {
235        if i == last_unit && lifted.is_some() {
236          u.charge.clear();
237        }
238        // A unit with no element scripts the bracket or text it follows (`Fe(CN)6`, 2602.09975 `\textit{LnPn}2`),
239        // after a unit the species so far (`\ce{H2_{(g)}}`; `\ce{SO4^{2-}_{(aq)}}`, the state after the charge).
240        if u.element.is_empty() && before == Before::Unit {
241          run.close(&mut body);
242          let formula = body.split_off(species);
243          body.extend(call("\\mathord", formula));
244          run.open(species, 1);
245        }
246        // A count on raw text (2407.15836 `\ce{\textit{R}2Ir2Al20}`) scripts it, inside the run.
247        if !u.element.is_empty() || run.start.is_some() {
248          run.open(body.len(), element_items(&u.element).len());
249        }
250        unit_tokens(&mut body, u, before != Before::Nothing);
251        before = Before::Unit;
252      },
253    }
254  }
255  run.close(&mut body);
256  match lifted {
257    Some(charge) => {
258      let formula = body.split_off(species);
259      body.extend(call("\\mathord", formula));
260      body.push(T_SUPER!());
261      body.extend(call("", math_mode(script(charge))));
262      Some(body)
263    },
264    None => Some(body),
265  }
266}

268/// Whether a macro of `tokens` expands (as `\edef` would, a protected macro kept) to a math shift, `$`, `\(` or `\)`;
269/// `shift` counts a literal one. mhchem's parser reads a literal `$` of `\ce` itself (`\__mhchem_cg_withinMathTokens:n`,
270/// mhchem.sty:162), a macro's reaches its output unread (2606.03586 `\ce{\alphasic}`, `\alphasic` = `$\alpha$-\ce{SiC}`):
271/// only mhchem's own text-mode layout, `\text{…}` (mhchem.sty:2934), holds it, not the math the units are set in here.
272fn shifts_math(tokens: &[Token], shift: bool, seen: &mut Vec<Token>) -> bool {
273  tokens.iter().any(|t| match t.get_catcode() {
274    Catcode::MATH => shift,
275    Catcode::CS | Catcode::ACTIVE => {
276      if *t == T_CS!("\\(") || *t == T_CS!("\\)") {
277        return shift;
278      }
279      if seen.contains(t)
280        || t.with_str(|s| s.starts_with("\\__mhchem_") || s.starts_with("\\mhchem@"))
281      {
282        return false;
283      }
284      seen.push(*t);
285      match lookup_definition(t) {
286        Ok(Some(defn)) if !defn.is_protected() => match defn.get_expansion() {
287          Some(ExpansionBody::Tokens(body)) => shifts_math(body.unlist_ref(), true, seen),
288          _ => false,
289        },
290        _ => false,
291      }
292    },
293    _ => false,
294  })
295}

297/// A formula's elements and brackets are juxtaposed, not multiplied (DG 2026-11-14): a run of them is
298/// `\lx@kludged{…}`, the unparsed row Perl uses for adjacency (Base_XMath.pool.ltxml:193, MathParser.pm:530).
299#[derive(Default)]
300struct Run {
301  start: Option<usize>,
302  atoms: usize,
303}
304
305impl Run {
306  fn open(&mut self, at: usize, atoms: usize) {
307    self.start.get_or_insert(at);
308    self.atoms += atoms;
309  }
310
311  fn close(&mut self, body: &mut Vec<Token>) {
312    // A run cut inside a raw group (an unbalanced brace) stays as it is.
313    if let Some(start) = self.start.take()
314      && self.atoms > 1
315      && balanced(&body[start..])
316    {
317      let run = body.split_off(start);
318      body.extend(kludged(run));
319    }
320    self.atoms = 0;
321  }
322}
323
324fn kludged(body: Vec<Token>) -> Vec<Token> { call("\\lx@kludged", body) }
325
326fn balanced(tokens: &[Token]) -> bool {
327  let mut depth = 0isize;
328  for t in tokens {
329    match t.get_catcode() {
330      Catcode::BEGIN => depth += 1,
331      Catcode::END => depth -= 1,
332      _ => {},
333    }
334    if depth < 0 {
335      return false;
336    }
337  }
338  depth == 0
339}

349fn is_letter(t: &Token) -> bool {
350  t.get_catcode() == Catcode::LETTER && t.with_str(|s| s.chars().all(|c| c.is_ascii_alphabetic()))
351}

353/// The element run split at its capitals (`MoO` is `Mo`, `O`) when it is letters only; else the run whole.
354fn element_items(element: &[Token]) -> Vec<&[Token]> {
355  if element.is_empty() || !element.iter().all(is_letter) {
356    return if element.is_empty() {
357      Vec::new()
358    } else {
359      vec![element]
360    };
361  }
362  let mut items = Vec::new();
363  let mut start = 0;
364  for (i, t) in element.iter().enumerate().skip(1) {
365    if t.with_str(|s| s.starts_with(|c: char| c.is_ascii_uppercase())) {
366      items.push(&element[start..i]);
367      start = i;
368    }
369  }
370  items.push(&element[start..]);
371  items
372}

374/// One unit, in a group where mhchem's output takes its math-mode branch (`\ensuremath` holds the formula):
375/// `{}^{mass}_{number}\mathord{El}…\mathord{El}_{count}^{ox charge}`; the charge of a unit of several elements
376/// is the unit's (`\ce{OH^-(H2O)}`, 2405.01836). A unit with no element scripts what precedes it when
377/// `scripts_previous`, else an empty base.
378fn unit_tokens(out: &mut Vec<Token>, u: CoreFive, scripts_previous: bool) {
379  let mut unit = Vec::new();
380  if !u.presup.is_empty() || !u.presub.is_empty() {
381    unit.extend([T_BEGIN!(), T_END!()]);
382    if !u.presup.is_empty() {
383      scripted(&mut unit, T_SUPER!(), u.presup);
384    }
385    if !u.presub.is_empty() {
386      scripted(&mut unit, T_SUB!(), u.presub);
387    }
388  }
389  let items = element_items(&u.element);
390  let empty = items.is_empty();
391  let unit_charge = items.len() > 1 && !u.charge.is_empty();
392  let atoms = u.element.iter().all(is_letter);
393  for item in items {
394    let font = font(item.to_vec());
395    if atoms {
396      unit.extend(call("\\mathord", font));
397    } else {
398      unit.extend(font);
399    }
400  }
401  if empty && !scripts_previous {
402    unit.extend([T_BEGIN!(), T_END!()]);
403  }
404  if !u.count.is_empty() {
405    scripted(&mut unit, T_SUB!(), u.count);
406  }
407  let mut sup: Vec<Token> = u.ox.into_iter().chain(u.kv).collect();
408  if !unit_charge {
409    sup.extend(u.charge.iter().copied());
410  }
411  if !sup.is_empty() {
412    scripted(&mut unit, T_SUPER!(), sup);
413  }
414  if unit_charge {
415    unit = call("\\mathord", kludged(unit));
416    scripted(&mut unit, T_SUPER!(), u.charge);
417  }
418  out.extend(math_mode(unit));
419}

421fn scripted(out: &mut Vec<Token>, mark: Token, body: Vec<Token>) {
422  out.push(mark);
423  out.extend(call("", script(body)));
424}

431/// `\cs{body}`, or `{body}` for no `cs`.
432fn call(cs: &str, body: Vec<Token>) -> Vec<Token> {
433  let mut out = Vec::with_capacity(body.len() + 3);
434  if !cs.is_empty() {
435    out.push(T_CS!(cs));
436  }
437  out.push(T_BEGIN!());
438  out.extend(body);
439  out.push(T_END!());
440  out
441}

443/// `body` in a group with `\l__mhchem_output_isMathMode_bool` true.
444fn math_mode(body: Vec<Token>) -> Vec<Token> {
445  let mut out = Vec::from([
446    T_CS!("\\group_begin:"),
447    T_CS!("\\bool_set_true:N"),
448    T_CS!("\\l__mhchem_output_isMathMode_bool"),
449  ]);
450  out.extend(body);
451  out.push(T_CS!("\\group_end:"));
452  out
