33  // \inputminted[opts]{language}{filename} — Perl L43-53 reads the
34  // referenced file via FindFile + Mouth->readRawLine, then wraps the
35  // contents in \begin{minted}{language}...\end{minted}, relying on
36  // mintedEnvBody to read those tokens via `gullet->readUntil(T_CS('\end'))`.
37  //
38  // We can't go through \begin{lstlisting} the same way: lstlisting's
39  // body reader is `listings_read_raw_lines("lstlisting")` which reads
40  // RAW LINES FROM THE MOUTH (the underlying TeX source), not from
41  // tokens we've pushed back into the gullet. Routing \inputminted
42  // through `\begin{lstlisting} ... \end{lstlisting}` therefore
43  // discards the file content and instead consumes lines from the
44  // surrounding `.tex` file until it hits `\end{lstlisting}` —
45  // which never appears, so it eats the rest of the document.
46  // Driver: 1903.09408 (\inputminted inside \begin{listing} broke
47  // section nesting because \end{listing} got swallowed too).
48  //
49  // Solution: bypass \begin{lstlisting} entirely and call
50  // lst_process_display directly with the file contents, mirroring
51  // what `\begin{lstlisting}` does internally but with our string in
52  // place of the read_raw_lines call.
53  use latexml_core::binding::content::find_file;
54  DefMacro!("\\inputminted[]{}{}", sub[(_opts, _lang, file_arg)] {
55    let file_str = file_arg.to_string();
56    let contents = find_file(&file_str, None)
57      .and_then(|path| std::fs::read_to_string(&path).ok())
58      .unwrap_or_default();
59    bgroup();
60    assign_value(
61      "current_environment",
62      Stored::String(pin("lstlisting")),
63      None,
64    );
65    def_macro(
66      T_CS!("\\@currenvir"),
67      None,
68      Tokens!(T_OTHER!("lstlisting")),
69      None,
70    )?;
71    // Beyond-Perl frozencache color path (see \begin{minted}); falls back to the
72    // uncolored listings display on a cache miss / no `_minted/`.
73    let mut result = match crate::minted_frozencache::colored_display_body(&contents) {
74      Some(body) => lst_process_display_with(None, &contents, body),
75      None => lst_process_display(None, &contents),
76    };
77    // lst_process_display ends with T_END to balance bgroup we opened above
78    // (mirrors `\begin{lstlisting}`'s convention).
79    if !matches!(result.last(), Some(t) if t.get_catcode() == Catcode::END) {
80      result.push(T_END!());
81    }
82    Ok(Tokens::new(result))
83  });

87  // Perl minted.sty.ltxml L58-99 dynamically defined new CSes via runtime
88  // DefMacroI closures. The TeX-level equivalent uses `\expandafter\def
89  // \csname <name>\endcsname` so the same user input now binds working aliases.
90  //
91  // Each of `\newmint` / `\newmintinline` / `\newminted` / `\newmintedfile` takes
92  // an OPTIONAL `[name]` before the two mandatory `{language}{options}` args (real
93  // minted.sty: `\newcommand{..}[3][]`, each with its own default name). The prior
94  // port declared the first three as two-mandatory `#1#2`, so
95  // `\newminted[leancode]{lean4}{...}` bound `#1 = "["` and then ran
96  // `\expandafter\def\csname [\endcsname{...}` — but `\csname [\endcsname` IS the
97  // control sequence `\[`, silently redefining display-math open to
98  // `\begin{lstlisting}`; every later `\[` opened a listing that ran to EOF,
99  // swallowing the body and `\bibliography` with no `Error:` (arXiv:2606.05629,
100  // issue #520). Each of the three now branches on `\@ifnextchar[` (as
101  // `\newmintedfile` already did) and consumes all three args, so the optional
102  // env/command name binds correctly and nothing (least of all `\[`) leaks. Default names match
103  // minted: `\newmint{lang}` → `\lang`; `\newmintinline{lang}` → `\langinline`;
104  // `\newminted{lang}` → env `langcode`; the optional form uses the given name
105  // verbatim.
106  //
107  // The inline commands (`\mint`, `\mintinline`, and the `\newmint`/`\newmintinline`
108  // aliases) route to `\lstinline[language=<lang>]`, NOT `\verb`: `\verb` is
109  // delimiter-based, so the common `\mintinline{lang}{code}` BRACE form read `{` as
110  // the delimiter and ran past the closing `}` (garbled code + unbalanced-group
111  // errors that broke following content); `\lstinline` accepts braces and, given the
112  // language, highlights the snippet. listings is the Perl-chosen substrate (L30),
113  // so a `\newminted` env delegates to `\lstnewenvironment{name}` (as the tcolorbox
114  // binding's `\newtcblisting` does) — whose verbatim reader stops at the env's own
115  // `\end{name}` (the prior alias to `\begin{lstlisting}` read raw until the literal
116  // `\end{lstlisting}`, so a `\newminted` env used normally ran to EOF) — and its
117  // start code activates the language via `\lstset{language=<lang>}` for keyword/
118  // string/comment highlighting. Language matching is case-insensitive and an unknown
119  // language (e.g. `lean4`) is a silent no-op, so minted's Pygments lexer name passes
120  // through verbatim and degrades to plain text rather than erroring. `\newmintedfile`
121  // binds either the given optional macro or `\<lang>file` to `\inputminted`.
122  RawTeX!(
123    r#"\def\newmint{\@ifnextchar[\lx@minted@nm@opt\lx@minted@nm@noopt}
124\def\lx@minted@nm@noopt#1#2{\expandafter\def\csname #1\endcsname{\lstinline[language=#1]}}
125\def\lx@minted@nm@opt[#1]#2#3{\expandafter\def\csname #1\endcsname{\lstinline[language=#2]}}
126\def\newmintinline{\@ifnextchar[\lx@minted@nmi@opt\lx@minted@nmi@noopt}
127\def\lx@minted@nmi@noopt#1#2{\expandafter\def\csname #1inline\endcsname{\lstinline[language=#1]}}
128\def\lx@minted@nmi@opt[#1]#2#3{\expandafter\def\csname #1\endcsname{\lstinline[language=#2]}}
129\def\newminted{\@ifnextchar[\lx@minted@nmd@opt\lx@minted@nmd@noopt}
130\def\lx@minted@nmd@noopt#1#2{\lx@minted@nmd@def{#1code}{#1}}
131\def\lx@minted@nmd@opt[#1]#2#3{\lx@minted@nmd@def{#1}{#2}}
132\def\lx@minted@nmd@def#1#2{%
133  \lstnewenvironment{#1}{\lstset{language=#2}}{}%
134  \lstnewenvironment{#1*}[1]{\lstset{language=#2,##1}}{}}
135\def\newmintedfile{\@ifnextchar[\lx@minted@nmf@opt\lx@minted@nmf@noopt}
136\def\lx@minted@nmf@opt[#1]#2{\let#1\inputminted}
137\def\lx@minted@nmf@noopt#1{\expandafter\let\csname #1file\endcsname\inputminted}
138"#

145  // `\mintinline[opts]{lang}{code}` — beyond-Perl frozencache color path.
146  // When NO `_minted/` frozencache is present the code arg is left untouched in
147  // the stream and we reconstruct the historical `\lstinline[language=<lang>]`
148  // expansion (byte-for-byte the old behavior, so non-frozencache papers are
149  // unaffected). When a frozencache IS present we read the `{code}` verbatim
150  // (mirroring `\lx@lstinline`) and emit a colored inline `ltx_lstlisting` span
151  // on a content-match hit, or the plain code on a miss. (OXIDIZED_DESIGN_
152  // DIVERGENCES #157.)
153  {
154    let cs = T_CS!("\\mintinline");
155    let params = parse_parameters("[]{}", &cs, true)?;
156    let expansion: Option<ExpansionBody> = Some(ExpansionBody::Closure(Rc::new(
157      move |args: Vec<ArgWrap>| {
158        let mut it = args.into_iter();
159        let _opts = it.next();
160        let lang = it.next().and_then(|a| a.owned_tokens()).unwrap_or_default();
161        if !crate::minted_frozencache::cache_available() {
162          // Delegate to \lstinline, which reads the following {code} verbatim.
163          let mut out = vec![T_CS!("\\lstinline"), T_OTHER!("[")];
164          out.extend(Tokenize!("language=").unlist());
165          out.extend(lang.unlist());
166          out.push(T_OTHER!("]"));
167          return Ok(Tokens::new(out));
168        }
169        // Frozencache present: read the verbatim code ourselves (mirror
170        // \lx@lstinline listings.sty:2076-2108).
171        let init = read_token()?;
172        let until = init.as_ref().map(|t| {
173          if t.get_catcode() == Catcode::BEGIN {
174            T_END!()
175          } else {
176            *t
177          }
178        });
179        let verbatim_chars = ['%', '\\', '{', '}', '$', '&', '#', '^', '_', '~'];
180        let saved: Vec<(char, Catcode)> = verbatim_chars
181          .iter()
182          .map(|&c| (c, lookup_catcode(c).unwrap_or(Catcode::OTHER)))
183          .collect();
184        push_frame();
185        for c in verbatim_chars {
186          assign_catcode(c, Catcode::OTHER, Some(Scope::Local));
187        }
188        let body = listings_read_raw_string(until.as_ref(), &saved);
189        pop_frame()?;
190        let segs = crate::minted_frozencache::inline_body(&body);
191        let mut out = vec![
192          T_CS!("\\leavevmode"),
193          T_CS!("\\@listings@inline"),
194          T_BEGIN!(),
195        ];
196        out.extend(segs);
197        out.push(T_END!());
198        Ok(Tokens::new(out))
199      },
200    )));
201    def_macro(cs, params, expansion, None)?;
202  }
203  // \begin{minted}[opts]{language} — port of Perl mintedEnvBody
204  // (minted.sty.ltxml L68-85). Read raw input lines until \end{minted},
205  // then dispatch to listings' lst_process_display, mirroring Perl's
206  // `bgroup; current_environment=lstlisting; lstProcessDisplay(...)`.
207  // Without this, the legacy `\begin{minted} -> \begin{lstlisting}`
208  // expansion lets lstlisting swallow the rest of the file because
209  // it never sees its own `\end{lstlisting}` marker (the user wrote
210  // `\end{minted}`).
211  use latexml_core::stomach::bgroup;
212  use latexml_package::package::listings_sty::{
213    listings_read_raw_lines, listings_read_raw_string, lst_process_display,
214    lst_process_display_with,
215  };
216  {
217    let cs = T_CS!("\\begin{minted}");
218    // `\begin{minted}[opts]{language}`. Previously the `[opts]` were dropped, so
219    // `escapeinside=!!` never reached the listings tokenizer: an inline
220    // `!$\label{…}$!` was emitted as literal code, its `\label` never ran, and
221    // `\ref` to that line vanished (html_feedback#1028, arXiv:2308.03276;
222    // OXIDIZED_DESIGN #127). We now feed both the `{language}` and the options
223    // through `\lstset{…}` (the same activation `\begin{lstlisting}` uses), so
224    // minted's `escapeinside`/`mathescape` — shared verbatim with the listings
225    // substrate — take effect and the language activates listings' keyword/string/
226    // comment highlighting; minted-only keys (`linenos`, `fontsize`, …) are stored
227    // harmlessly. Scoped to the environment's `bgroup`.
228    let params = parse_parameters("[]{}", &cs, true)?;
229    let expansion: Option<ExpansionBody> = Some(ExpansionBody::Closure(Rc::new(
230      move |args: Vec<ArgWrap>| {
231        // args[0] = `[opts]` (optional); args[1] = `{language}`.
232        let mut it = args.into_iter();
233        let opts = it.next().and_then(|a| a.owned_tokens());
234        let lang = it.next().and_then(|a| a.owned_tokens());
235        bgroup();
236        assign_value(
237          "current_environment",
238          Stored::String(pin("lstlisting")),
239          None,
240        );
241        def_macro(
242          T_CS!("\\@currenvir"),
243          None,
244          Tokens!(T_OTHER!("lstlisting")),
245          None,
246        )?;
247        // Feed `language=<lang>,<opts>` to `\lstset`. The language activates the
248        // listings language pack for keyword/string/comment highlighting; matching
249        // is case-insensitive and an unknown language (e.g. `lean4`) is a silent
250        // no-op, so passing minted's Pygments lexer name verbatim degrades to the
251        // old plain rendering rather than erroring. The `[opts]` still ride along
252        // for `escapeinside`/`mathescape` (html_feedback#1028); minted-only keys
253        // (`linenos`, `fontsize`, …) are stored harmlessly.
254        let mut kv: Vec<Token> = Vec::new();
255        if let Some(lang) = lang.as_ref().filter(|l| !l.unlist_ref().is_empty()) {
256          kv.extend(Tokenize!("language=").unlist());
257          kv.extend(lang.unlist_ref().iter().cloned());
258        }
259        if let Some(opts) = opts.as_ref().filter(|o| !o.unlist_ref().is_empty()) {
260          if !kv.is_empty() {
261            kv.push(T_OTHER!(","));
262          }
263          kv.extend(opts.unlist_ref().iter().cloned());
264        }
265        if !kv.is_empty() {
266          let _ = digest(Tokens::new(
267            std::iter::once(T_CS!("\\lstset"))
268              .chain(std::iter::once(T_BEGIN!()))
269              .chain(kv)
270              .chain(std::iter::once(T_END!()))
271              .collect(),
272          ));
273        }
274        let text = listings_read_raw_lines("minted");
275        // Beyond-Perl: if a `_minted/` frozencache holds this block's Pygments
276        // colors (content-match on the normalized body), emit colored listing
277        // lines; otherwise keep the exact uncolored listings path.
278        // (OXIDIZED_DESIGN_DIVERGENCES #157.)
279        let result = match crate::minted_frozencache::colored_display_body(&text) {
280          Some(body) => lst_process_display_with(None, &text, body),
281          None => lst_process_display(None, &text),
282        };
283        Ok(Tokens::new(result))
284      },
285    )));
286    def_macro(cs, params, expansion, None)?;
287  }
