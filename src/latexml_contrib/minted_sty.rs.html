215  // the stream and we reconstruct the historical `\lstinline[language=<lang>]`
216  // expansion (byte-for-byte the old behavior, so non-frozencache papers are
217  // unaffected). When a frozencache IS present we read the `{code}` verbatim
218  // (mirroring `\lx@lstinline`) and emit a colored inline `ltx_lstlisting` span
219  // on a content-match hit, or the plain code on a miss. (OXIDIZED_DESIGN_
220  // DIVERGENCES #157.)
221  {
222    let cs = T_CS!("\\mintinline");
223    let params = parse_parameters("[]{}", &cs, true)?;
224    let expansion: Option<ExpansionBody> = Some(ExpansionBody::Closure(Rc::new(
225      move |args: Vec<ArgWrap>| {
226        let mut it = args.into_iter();
227        let _opts = it.next();
228        let lang = it.next().and_then(|a| a.owned_tokens()).unwrap_or_default();
229        if !crate::minted_frozencache::cache_available() {
230          // Delegate to \lstinline, which reads the following {code} verbatim.
231          let mut out = vec![T_CS!("\\lstinline"), T_OTHER!("[")];
232          out.extend(Tokenize!(MINTED_LST_RESET).unlist());
233          out.extend(Tokenize!(",language=").unlist());
234          out.extend(lang.unlist());
235          out.push(T_OTHER!("]"));
236          return Ok(Tokens::new(out));
237        }
238        // Frozencache present: read the verbatim code ourselves (mirror
239        // \lx@lstinline listings.sty:2076-2108).
240        let init = read_token()?;
241        let until = init.as_ref().map(|t| {
242          if t.get_catcode() == Catcode::BEGIN {
243            T_END!()
244          } else {
245            *t
246          }
247        });
248        let verbatim_chars = ['%', '\\', '{', '}', '$', '&', '#', '^', '_', '~'];
249        let saved: Vec<(char, Catcode)> = verbatim_chars
250          .iter()
251          .map(|&c| (c, lookup_catcode(c).unwrap_or(Catcode::OTHER)))
252          .collect();
253        push_frame();
254        for c in verbatim_chars {
255          assign_catcode(c, Catcode::OTHER, Some(Scope::Local));
256        }
257        let (body, close_counted) = listings_read_raw_string(until.as_ref(), &saved);
258        pop_frame()?;
259        // The `{` delimiter's count, as \lx@lstinline (the `{…}` output below re-counts as a pair).
260        settle_verbatim_brace_ledger(init.as_ref(), close_counted);
261        let segs = crate::minted_frozencache::inline_body(&body);
262        let mut out = vec![
263          T_CS!("\\leavevmode"),
264          T_CS!("\\@listings@inline"),
265          T_BEGIN!(),
266        ];
267        out.extend(segs);
268        out.push(T_END!());
269        Ok(Tokens::new(out))
270      },
271    )));
272    def_macro(cs, params, expansion, None)?;
273  }

277  // `bgroup; current_environment=lstlisting; lstProcessDisplay(...)`.
278  // Without this, the legacy `\begin{minted} -> \begin{lstlisting}`

289    // `\begin{minted}[opts]{language}`. Previously the `[opts]` were dropped, so
290    // `escapeinside=!!` never reached the listings tokenizer: an inline
291    // `!$\label{…}$!` was emitted as literal code, its `\label` never ran, and
292    // `\ref` to that line vanished (html_feedback#1028, arXiv:2308.03276;
293    // OXIDIZED_DESIGN #127). We now feed both the `{language}` and the options
294    // through `\lstset{…}` (the same activation `\begin{lstlisting}` uses), so
295    // minted's `escapeinside`/`mathescape` — shared verbatim with the listings
296    // substrate — take effect and the language activates listings' keyword/string/
297    // comment highlighting; minted-only keys (`linenos`, `fontsize`, …) are stored
298    // harmlessly. Scoped to the environment's `bgroup`.
299    let params = parse_parameters("[]{}", &cs, true)?;
300    let expansion: Option<ExpansionBody> = Some(ExpansionBody::Closure(Rc::new(
301      move |args: Vec<ArgWrap>| {
302        // args[0] = `[opts]` (optional); args[1] = `{language}`.
303        let mut it = args.into_iter();
304        let opts = it.next().and_then(|a| a.owned_tokens());
305        let lang = it.next().and_then(|a| a.owned_tokens());
306        bgroup();
307        assign_value(
308          "current_environment",
309          Stored::String(pin("lstlisting")),
310          None,
311        );
312        def_macro(
313          T_CS!("\\@currenvir"),
314          None,
315          Tokens!(T_OTHER!("lstlisting")),
316          None,
317        )?;
318        // Feed `<MINTED_LST_RESET>,language=<lang>,<opts>` to `\lstset`. The language activates the
319        // listings language pack for keyword/string/comment highlighting; matching
320        // is case-insensitive and an unknown language (e.g. `lean4`) is a silent
321        // no-op, so passing minted's Pygments lexer name verbatim degrades to the
322        // old plain rendering rather than erroring. The `[opts]` still ride along
323        // for `escapeinside`/`mathescape` (html_feedback#1028); minted-only keys
324        // (`linenos`, `fontsize`, …) are stored harmlessly.
325        let mut kv: Vec<Token> = Tokenize!(MINTED_LST_RESET).unlist();
326        if let Some(lang) = lang.as_ref().filter(|l| !l.unlist_ref().is_empty()) {
327          kv.extend(Tokenize!(",language=").unlist());
328          kv.extend(lang.unlist_ref().iter().cloned());
329        }
330        if let Some(opts) = opts.as_ref().filter(|o| !o.unlist_ref().is_empty()) {
331          kv.push(T_OTHER!(","));
332          kv.extend(opts.unlist_ref().iter().cloned());
333        }
334        let _ = digest(Tokens::new(
335          std::iter::once(T_CS!("\\lstset"))
336            .chain(std::iter::once(T_BEGIN!()))
337            .chain(kv)
338            .chain(std::iter::once(T_END!()))
339            .collect(),
340        ));
341        // fancyvrb.sty:295-297/386-403: a user environment that wraps minted
342        // with `\VerbatimEnvironment` (RetoMatematico.cls:174 `{codigo}` =
343        // `{\VerbatimEnvironment\begin{minted}[…]{#1}}{\end{minted}}`) names the
344        // environment whose `\end` closes the verbatim body — the OUTERMOST, as
345        // `\FV@EnvironName` is set only while still `\relax` — and the name is
346        // reset once read. Reading to a literal `\end{minted}` ran past
347        // `\end{codigo}` and swallowed the rest of the document
348        // (retomatematico-ejemplo, recall 53 %). Guard:
349        // `perfect_kernel_batch56::verbatim_environment_wrapper_ends_minted`.
350        let fv_name = T_CS!("\\FV@EnvironName");
351        let env_name =
352          if lookup_meaning(&fv_name).is_some() && !x_equals(&fv_name, &T_CS!("\\relax")) {
353            let name = Expand!(Tokens!(fv_name)).to_string();
354            Let!(fv_name, T_CS!("\\relax"), Scope::Global);
355            name
356          } else {
357            "minted".to_string()
358          };
359        let text = listings_read_raw_lines(&env_name);
360        // Beyond-Perl: if a `_minted/` frozencache holds this block's Pygments
361        // colors (content-match on the normalized body), emit colored listing
362        // lines; otherwise keep the exact uncolored listings path.
363        // (OXIDIZED_DESIGN_DIVERGENCES #157.)
364        let result = match crate::minted_frozencache::colored_display_body(&text) {
365          Some(body) => lst_scoped(lst_process_display_with(None, &text, body)),
366          None => lst_process_display_scoped(None, &text),
367        };
368        // As `\inputminted`: close the key-scoping `bgroup()` here and emit
369        // the `\begingroup` token opener for the listing's own group.
370        egroup()?;
371        let mut out = lst_group_opener("minted", None)?;
372        out.extend(result);
373        let mut end_tokens = vec![T_CS!("\\end"), T_BEGIN!()];
374        end_tokens.extend(ExplodeText!(env_name));
375        end_tokens.push(T_END!());
376        unread_expansion(Tokens::new(end_tokens));
377        Ok(Tokens::new(out))
378      },
379    )));
380    def_macro(cs, params, expansion, None)?;
381    DefMacro!(T_CS!("\\end{minted}"), None, Tokens!());

393    .map(|within| s!("within={within},"))
394    .unwrap_or_default();
395  if minted_newfloat {
396    RawTeX!(&s!(
397      r"\DeclareFloatingEnvironment[{within}fileext=lol,placement=tbp]{{listing}}"
398    ));
399  } else {
400    let within = minted_within
401      .map(|within| s!("[{within}]"))
402      .unwrap_or_default();
403    RawTeX!(&s!(
404      r"\newfloat{{listing}}{{tbp}}{{lol}}{within}\floatname{{listing}}{{\listingscaption}}"
405    ));
406  }
407});

409/// minted's float options (minted.sty:155-165, read by `\ProcessPgfOptions{/minted/pkg}`, :282): `newfloat`
410/// (`newfloat/.is if`: alone or `=true`; `=false` turns it off) and the listing counter's `within`, `chapter` or
411/// `section`; each the last one given.
412fn minted_float_options() -> (bool, Option<&'static str>) {
413  let mut newfloat = false;
414  let mut within = None;
415  if let Some(options) = lookup_vecdeque("opt@minted.sty") {
416    for option in options.iter() {
417      let option = option.to_string();
418      let (key, value) = option
419        .split_once('=')
420        .map_or((option.trim(), "true"), |(key, value)| {
421          (key.trim(), value.trim())
422        });
423      match key {
424        "newfloat" => newfloat = value != "false",
425        "chapter" => within = Some("chapter"),
426        "section" => within = Some("section"),
427        _ => {},
428      }
429    }
430  }
431  (newfloat, within)
432}
