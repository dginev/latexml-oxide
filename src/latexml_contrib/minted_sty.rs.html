33  // \inputminted[opts]{language}{filename} — Perl L43-53 reads the
34  // referenced file via FindFile + Mouth->readRawLine, then wraps the
35  // contents in \begin{minted}{language}...\end{minted}, relying on
36  // mintedEnvBody to read those tokens via `gullet->readUntil(T_CS('\end'))`.
37  //
38  // We can't go through \begin{lstlisting} the same way: lstlisting's
39  // body reader is `listings_read_raw_lines("lstlisting")` which reads
40  // RAW LINES FROM THE MOUTH (the underlying TeX source), not from
41  // tokens we've pushed back into the gullet. Routing \inputminted
42  // through `\begin{lstlisting} ... \end{lstlisting}` therefore
43  // discards the file content and instead consumes lines from the
44  // surrounding `.tex` file until it hits `\end{lstlisting}` —
45  // which never appears, so it eats the rest of the document.
46  // Driver: 1903.09408 (\inputminted inside \begin{listing} broke
47  // section nesting because \end{listing} got swallowed too).
48  //
49  // Solution: bypass \begin{lstlisting} entirely and call
50  // lst_process_display directly with the file contents, mirroring
51  // what `\begin{lstlisting}` does internally but with our string in
52  // place of the read_raw_lines call.
53  use latexml_core::binding::content::find_file;
54  DefMacro!("\\inputminted[]{}{}", sub[(_opts, _lang, file_arg)] {
55    let file_str = file_arg.to_string();
56    let contents = find_file(&file_str, None)
57      .and_then(|path| std::fs::read_to_string(&path).ok())
58      .unwrap_or_default();
59    bgroup();
60    assign_value(
61      "current_environment",
62      Stored::String(pin("lstlisting")),
63      None,
64    );
65    def_macro(
66      T_CS!("\\@currenvir"),
67      None,
68      Tokens!(T_OTHER!("lstlisting")),
69      None,
70    )?;
71    let mut result = lst_process_display(None, &contents);
72    // lst_process_display ends with T_END to balance bgroup we opened above
73    // (mirrors `\begin{lstlisting}`'s convention).
74    if !matches!(result.last(), Some(t) if t.get_catcode() == Catcode::END) {
75      result.push(T_END!());
76    }
77    Ok(Tokens::new(result))
78  });

82  // Perl minted.sty.ltxml L58-99 dynamically defined new CSes via runtime
83  // DefMacroI closures. The TeX-level equivalent uses `\expandafter\def
84  // \csname <name>\endcsname` so the same user input now binds working aliases.
85  //
86  // Each of `\newmint` / `\newmintinline` / `\newminted` / `\newmintedfile` takes
87  // an OPTIONAL `[name]` before the two mandatory `{language}{options}` args (real
88  // minted.sty: `\newcommand{..}[3][]`, each with its own default name). The prior
89  // port declared the first three as two-mandatory `#1#2`, so
90  // `\newminted[leancode]{lean4}{...}` bound `#1 = "["` and then ran
91  // `\expandafter\def\csname [\endcsname{...}` — but `\csname [\endcsname` IS the
92  // control sequence `\[`, silently redefining display-math open to
93  // `\begin{lstlisting}`; every later `\[` opened a listing that ran to EOF,
94  // swallowing the body and `\bibliography` with no `Error:` (arXiv:2606.05629,
95  // issue #520). Each of the three now branches on `\@ifnextchar[` (as
96  // `\newmintedfile` already did) and consumes all three args, so the optional
97  // env/command name binds correctly and nothing (least of all `\[`) leaks. Default names match
98  // minted: `\newmint{lang}` → `\lang` (verb-like); `\newmintinline{lang}` →
99  // `\langinline`; `\newminted{lang}` → env `langcode`; the optional form uses the
100  // given name verbatim. listings is the Perl-chosen substrate (L30), so a
101  // `\newminted` env delegates to `\lstnewenvironment{name}` (as the tcolorbox
102  // binding's `\newtcblisting` does) — whose verbatim reader stops at the env's own
103  // `\end{name}`. The prior alias to `\begin{lstlisting}` read raw until the literal
104  // `\end{lstlisting}`, so a `\newminted` env used normally (closed by `\end{name}`)
105  // never saw its marker and ran to EOF, swallowing the rest of the document.
106  // `\newmintedfile` binds either the given optional macro or `\<lang>file` to
107  // `\inputminted`.
108  RawTeX!(
109    r#"\def\newmint{\@ifnextchar[\lx@minted@nm@opt\lx@minted@nm@noopt}
110\def\lx@minted@nm@noopt#1#2{\expandafter\def\csname #1\endcsname{\verb}}
111\def\lx@minted@nm@opt[#1]#2#3{\expandafter\def\csname #1\endcsname{\verb}}
112\def\newmintinline{\@ifnextchar[\lx@minted@nmi@opt\lx@minted@nmi@noopt}
113\def\lx@minted@nmi@noopt#1#2{\expandafter\def\csname #1inline\endcsname{\verb}}
114\def\lx@minted@nmi@opt[#1]#2#3{\expandafter\def\csname #1\endcsname{\verb}}
115\def\newminted{\@ifnextchar[\lx@minted@nmd@opt\lx@minted@nmd@noopt}
116\def\lx@minted@nmd@noopt#1#2{\lx@minted@nmd@def{#1code}}
117\def\lx@minted@nmd@opt[#1]#2#3{\lx@minted@nmd@def{#1}}
118\def\lx@minted@nmd@def#1{%
119  \lstnewenvironment{#1}{}{}%
120  \lstnewenvironment{#1*}[1]{}{}}
121\def\newmintedfile{\@ifnextchar[\lx@minted@nmf@opt\lx@minted@nmf@noopt}
122\def\lx@minted@nmf@opt[#1]#2{\let#1\inputminted}
123\def\lx@minted@nmf@noopt#1{\expandafter\let\csname #1file\endcsname\inputminted}
124"#
125  );

132  // \begin{minted}[opts]{language} — port of Perl mintedEnvBody
133  // (minted.sty.ltxml L68-85). Read raw input lines until \end{minted},
134  // then dispatch to listings' lst_process_display, mirroring Perl's
135  // `bgroup; current_environment=lstlisting; lstProcessDisplay(...)`.
136  // Without this, the legacy `\begin{minted} -> \begin{lstlisting}`
137  // expansion lets lstlisting swallow the rest of the file because
138  // it never sees its own `\end{lstlisting}` marker (the user wrote
139  // `\end{minted}`).
140  use latexml_core::stomach::bgroup;
141  use latexml_package::package::listings_sty::{listings_read_raw_lines, lst_process_display};
142  {
143    let cs = T_CS!("\\begin{minted}");
144    // `\begin{minted}[opts]{language}`. Previously the `[opts]` were dropped, so
145    // `escapeinside=!!` never reached the listings tokenizer: an inline
146    // `!$\label{…}$!` was emitted as literal code, its `\label` never ran, and
147    // `\ref` to that line vanished (html_feedback#1028, arXiv:2308.03276;
148    // OXIDIZED_DESIGN #127). We now feed the options through `\lstset{…}` (the
149    // same activation `\begin{lstlisting}` uses), so minted's `escapeinside`/
150    // `mathescape` — shared verbatim with the listings substrate — take effect;
151    // minted-only keys (`linenos`, `fontsize`, …) are stored harmlessly and
152    // ignored. The `{language}` arg is left unapplied (we keep the current
153    // language-agnostic rendering). Scoped to the environment's `bgroup`.
154    let params = parse_parameters("[]{}", &cs, true)?;
155    let expansion: Option<ExpansionBody> = Some(ExpansionBody::Closure(Rc::new(
156      move |args: Vec<ArgWrap>| {
157        // args[0] = `[opts]` (optional); args[1] = `{language}` (unapplied).
158        let opts = args.into_iter().next().and_then(|a| a.owned_tokens());
159        bgroup();
160        assign_value(
161          "current_environment",
162          Stored::String(pin("lstlisting")),
163          None,
164        );
165        def_macro(
166          T_CS!("\\@currenvir"),
167          None,
168          Tokens!(T_OTHER!("lstlisting")),
169          None,
170        )?;
171        if let Some(opts) = opts
172          && !opts.unlist_ref().is_empty()
173        {
174          let _ = digest(Tokens::new(
175            vec![T_CS!("\\lstset"), T_BEGIN!()]
176              .into_iter()
177              .chain(opts.unlist().iter().cloned())
178              .chain(std::iter::once(T_END!()))
179              .collect(),
180          ));
181        }
182        let text = listings_read_raw_lines("minted");
183        let result = lst_process_display(None, &text);
184        Ok(Tokens::new(result))
185      },
186    )));
187    def_macro(cs, params, expansion, None)?;
188  }
189  // No \end{minted}: mintedEnvBody fully consumed it.
190  DefMacro!(T_CS!("\\begin{listing}"), None, "\\begin{figure}");
191  DefMacro!(T_CS!("\\end{listing}"), None, "\\end{figure}");
