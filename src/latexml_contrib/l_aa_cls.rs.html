1//! `l-aa.sty` (also shipped as `laa.sty`, the same file, and as `l-aa-ps.sty`, the same with PostScript fonts: copy in
2//! astro-ph9606053, used by astro-ph9505054): Springer-Verlag's LaTeX 2.09 style for Astronomy &
3//! Astrophysics, "L-AA vers. 3.0" (1992-1996), loaded as `\documentstyle{l-aa}`. Neither TeX Live nor most papers
4//! using it ship it, so the document fell to OmniBus and stopped at `\thesaurus` (843 run-336 papers: astro-ph9804097,
5//! astro-ph9405007, astro-ph0211472). It is the predecessor of `aa.cls`, which kept its document syntax (`\institute`
6//! with `\inst`, `\offprints`, `\mail`, `\date`, `{abstract}`, `\keywords`, `{acknowledgements}`, the astronomical
7//! symbols, plain TeX's `\cases` and `\pmatrix` in displays: astro-ph9908044), so this is the `aa` class binding with
8//! what L-AA itself adds; a binding for the class, which `\documentstyle` prefers
9//! (latex_constructs.pool.ltxml:105-108).
10//! Rust-only: Perl has no `l-aa` binding. A paper shipping the file gets this binding too, as a binding outranks a raw
11//! file (astro-ph9601070, astro-ph9601165).

17  DeclareOption!(None, {
18    Digest!("\\PassOptionsToClass{\\CurrentOption}{aa}")?;
19  });
20  ProcessOptions!();
21  LoadClass!("aa");
22  // l-aa.sty:208-214 (v3.0, the copies astro-ph9604133 and astro-ph9701125 ship): the appendix renumbers the sections
23  // "A" and the equations within them, "A1" (no dot); the figures and tables keep their running numbers, where aa.cls's
24  // (the aa binding's `\appendix`, aa_cls.rs) numbers all three "A.1". The kernel's `\@appendix` runs first for the
25  // `ltx:appendix` units. pdflatex with the copy: astro-ph9705112 "Fig. 12.", "(A1) (A2)"; astro-ph9607057 "A1" where
26  // the aa numbering gave "A.1". Guard `perfect_kernel_batch78q::appendix_counters_l_aa`.
27  RawTeX!(
28    r"\def\appendix{\@appendix\par\setcounter{section}{0}\setcounter{subsection}{0}\def\thesection{\Alph{section}}%
29  \renewcommand\theequation{\thesection\arabic{equation}}\setcounter{equation}{0}\@addtoreset{equation}{section}}"
30  );
31  // l-aa.sty:453-454 `\def\thesaurus#1{\gdef\@thesaurus{#1}}`: the A&A thesaurus codes, printed in the title's head box
32  // as "Your thesaurus codes are: …" (:502-516 `\makeheadbox`) — the paper's subject classification, which the `aa`
33  // binding (aa.cls, whose head box has none) drops. Read as the `\def` reads it, not `\long`: a `\par` in the argument
34  // ends the call, "Paragraph ended before \thesaurus was complete" (tex.web §396), where a `%`-commented closing brace
35  // (`… 02.19.1) % Shock waves }`, astro-ph9405007) otherwise made the rest of the document the codes.
36  DefMacro!(
37    "\\lx@l@aa@thesaurus{}",
38    "\\lx@add@classification[scheme=thesaurus,name={Thesaurus codes:~}]{#1}"

44    match matching.end_taking_runaway_argument() {
45      None => Ok(Invocation!(T_CS!("\\lx@l@aa@thesaurus"), vec![Some(codes)])),

47        abandon_runaway_call(cs, false, true, runaway)?;
48        Ok(Tokens!())
49      },
50    }
51  });

67  // the text after it, to the paragraph's end (`\keywords ISM: molecules --- …`, astro-ph0001079), or of a
68  // `{keywords}` environment (astro-ph9609156, astro-ph9803302), or a group (`\keywords{supernova remnants --
69  // pulsars}`, astro-ph9405007), the aa binding's argument (aa_support.sty.ltxml:96), which read the first letter of
70  // the other two forms. Labelled as l-aa labels them, "Key words:".
71  DefMacro!(
72    "\\lx@l@aa@keywords{}",
73    "\\lx@add@keywords[name={Key words:~}]{#1}"
74  );
75  DefMacro!("\\keywords", {
76    skip_spaces()?;
77    if if_next(T_BEGIN!())? {
78      Ok(Tokens!(T_CS!("\\lx@l@aa@keywords")))
79    } else {
80      keywords_text(false)

82  });

84  DefMacro!(T_CS!("\\end{keywords}"), None, "");
85  // l-aa.sty:405-427 `\institute` and `\institutename`: the institutes numbered by position, `\and` between them,
86  // which the authors' `\inst{n}` name (inst_support). An `\institute` whose entries carry their own `$^n$` marks
87  // instead (`\institute{$^1$ Ioffe Institute … \\ $^2$ University of Crete …}`, atom-ph9605001, astro-ph9609131) is
88  // an affiliation list labelled by them: as one numbered entry it went whole to the author of `\inst{1}`. One
89  // institute (no `\and`), which l-aa sets unnumbered (:419-424), is every author's (astro-ph9405007, astro-ph9603094),
90  // a mark on it too when the names carry none (astro-ph9609156).
91  Let!("\\lx@l@aa@institute", "\\institute");
92  // Such papers mark their names alike (`A.Y.\,Potekhin$^{1,2,3}$ \and G.G.\,Pavlov$^{4,1}$`, atom-ph9605001): those
93  // marks request the affiliations so labelled, which the `\inst` author list (inst_support) does not read.
94  Let!("\\lx@l@aa@author", "\\author");
95  DefMacro!("\\author[]{}", sub[(short, names)] {
96    let names: Tokens = names;
97    if names.unlist_ref().iter().any(|t| {
98      t.get_catcode() == Catcode::SUPER || *t == T_CS!("\\textsuperscript") || *t == T_CS!("\\inst")
99    }) {
100      AssignValue!("lx@l@aa@marked@names" => true, Some(Scope::Global));
101    }
102    Ok(if names.unlist_ref().iter().any(|t| t.get_catcode() == Catcode::SUPER || *t == T_CS!("\\textsuperscript")) {
103      Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(names)])
104    } else {
105      Invocation!(T_CS!("\\lx@l@aa@author"), vec![short, Some(names)])
106    })
107  }, locked => true);
108  DefMacro!("\\institute{}", sub[(list)] {
109    let list: Tokens = list;
110    // (an `\and` written, as before, or held in a macro of the document's, which the separator query reads as the
111    // kernel's: `and_by_meaning`, Option C P2-5)
112    let ands = and_runs(list.unlist_ref(), &[T_CS!("\\and"), T_CS!("\\And")])?;
113    let several = !ands.runs.is_empty() || ands.inked > 0;
114    let marked = leads_with_superscript_mark(list.unlist_ref());
115    Ok(if marked && lookup_bool("lx@l@aa@marked@names") || marked && several {
116      Invocation!(T_CS!("\\lx@add@affiliations"), vec![None, Some(list)])
117    } else if !several {
118      Invocation!(T_CS!("\\lx@add@affiliation"), vec![Some(TokenizeInternal!("annotate=all")), Some(list)])
119    } else {
120      Invocation!(T_CS!("\\lx@l@aa@institute"), vec![Some(list)])
121    })
122  });
123});

125/// Whether `list` begins with a superscript mark: `$^1$`, `$^{1}$`, `^1` or `\textsuperscript{1}` (an affiliation
126/// list labelled by its marks: also ptephy's `\affil`).
127pub(crate) fn leads_with_superscript_mark(list: &[Token]) -> bool {
128  let mut tokens = list.iter().filter(|t| t.get_catcode() != Catcode::SPACE);
129  match tokens.next() {
130    Some(t) if t.get_catcode() == Catcode::MATH => tokens
131      .next()
132      .is_some_and(|t| t.get_catcode() == Catcode::SUPER),
133    Some(t) => t.get_catcode() == Catcode::SUPER || *t == T_CS!("\\textsuperscript"),
134    None => false,
135  }
136}

138/// The keywords `\keywords` starts, to the paragraph's end: a `\par`, the `\end` of the environment it stands in
139/// (`{abstract}`), a heading or `\begin`; with `in_environment` those of a `{keywords}` environment, to its
140/// `\end{keywords}`, the paragraphs in it one list. What ends them is put back.
141fn keywords_text(in_environment: bool) -> Result<Tokens> {
142  let mut body: Vec<Token> = Vec::new();
143  while let Some(token) = read_token()? {
144    if token.get_catcode() == Catcode::BEGIN {
145      body.push(token);
146      body.extend(read_balanced(ExpansionLevel::Off, false, false)?.unlist());
147      body.push(T_END!());
148    } else if !in_environment
149      && [
150        "\\begin",
151        "\\section",
152        "\\subsection",
153        "\\subsubsection",
154        "\\maketitle",
155      ]
156      .iter()
157      .any(|cs| token == T_CS!(*cs))
158    {
159      unread_vec(vec![token]);
160      break;
161    } else if token == T_CS!("\\par") {
162      if !in_environment {
163        unread_vec(vec![token]);
164        break;
165      }
166      body.push(T_SPACE!());
167    } else if token == T_CS!("\\end") {
168      let env = read_arg(ExpansionLevel::Off)?;
169      let name = env.to_string();
170      if name == "keywords" || !in_environment {
171        let mut end = vec![token, T_BEGIN!()];
172        end.extend(env.unlist());
173        end.push(T_END!());
174        unread_vec(end);
175        break;
176      }
177      body.push(token);
178      body.push(T_BEGIN!());
179      body.extend(env.unlist());
180      body.push(T_END!());
181    } else {
182      body.push(token);
183    }
184  }
185  while body
186    .last()
187    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
188  {
189    body.pop();
190  }
191  let lead = body
192    .iter()
193    .take_while(|t| t.get_catcode() == Catcode::SPACE)
194    .count();
195  body.drain(..lead);
196  Ok(Invocation!(T_CS!("\\lx@l@aa@keywords"), vec![Some(
197    Tokens::new(body)
198  )]))
