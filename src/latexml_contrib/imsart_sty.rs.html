1//! The IMS `imsart.sty` support file, as a binding of its own: the frontmatter and keyword markup the imsart.cls binding
2//! owns over it ([`crate::imsart_cls::imsart_sty_markup`], `\arxiv`), for a paper that loads imsart.sty without that
3//! class. A shipped copy is read after it as the stand-in's copy (OXIDIZED_DESIGN_DIVERGENCES #731), its own extras
4//! added: imsart.sty under its own name beside `\documentclass{article}` (1301.1466), and a renamed copy that declares
5//! itself `\ProvidesFile{imsart.sty}` (the declaration route, `content.rs` `provided_binding`) — 1905.11425's and
6//! 2101.10506's stylefile.sty (loaded by their stylefile.cls, a renamed imsart.cls), math0608476's imsart_mod.sty. Read
7//! raw, `\ead[label=e1,mark]{…}` inside `\author` glued "label=e1" into the name (OmniBus's `\ead{}[]`) or lost the
8//! e-mail; now each is its author's contact (OXIDIZED_DESIGN_DIVERGENCES #986).
9use latexml_package::prelude::*;
10
11use crate::imsart_cls::{imsart_arxiv, imsart_sty_markup};
12
13LoadDefinitions!({
14  imsart_sty_markup()?;
15  imsart_arxiv()?;
16  // `\affiliation[A]{…}` (imsart.sty:1720-1735 of 1905.11425's copy; `\affiliation{…}`, :1028 of 1301.1466's) prints the
17  // affiliation under the authors set since the last one (`\let\author@fmt@init\author@fmt@init@def`), or those its label
18  // marks. Under the class binding the raw class's `\affiliation` reaches the frontmatter through the kernel's class
19  // readers; a stand-in's copy is no raw class read (the binding owns the title page), so it printed in the body: the
20  // binding's, for the run of new authors (1301.1466).
21  DefMacro!(
22    "\\affiliation OptionalSemiverbatim {}",
23    "\\lx@add@contact[label={#1},role=affiliation,annotate=run]{#2}",
24    locked => true
25  );
26});
