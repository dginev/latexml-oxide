1//! crossreftools.sty: every extractor reads a label's data from `\r@<label>`, which a second LaTeX run defines from the
2//! `.aux` (hyperref's five fields: reference, page, name, anchor, unused; crossreftools.sty:293-345), and
3//! `\crtlistoflabels` inputs the `.lla` file the labels wrote (`\crtaddlabeltotoc`, :433-441; `\@crt@listofl@bels`,
4//! :407-421). Neither round trip happens here, so the extractors printed `[UNDEFINED]` or nothing and the list was
5//! empty. The binding is that second pass, per label: `\crtaddlabeltotoc`, which the package's `\label` runs right
6//! after the label (:638-656), defines `\r@<label>` from the values at the label — a backward reference resolves, a
7//! forward one stays undefined, as in a first run — and records the label's `.lla` line (`\numberline{\crtrefnumber
8//! {<label>}}<label>`) as a reference to the label. The list prints the recorded lines when it is built, after the
9//! whole document is read (a streamed conversion builds it with the lines read so far: residual). Links go to the
10//! label (`\hyperref[<label>]`), not to the hyperref anchor name in the data, which LaTeXML does not make. With
11//! cleveref, whose binding sets `\label` after the raw begin-document hooks, the package's wrapper is put back around
12//! it (cleveref loaded after crossreftools keeps cleveref's `\label` unwrapped: residual). Witness
13//! crossreftools/crossreftools_driver. Guards `perfect_kernel_batch61::{crossreftools_label_data_and_list,
14//! crossreftools_without_cleveref}`.
15use latexml_package::prelude::*;
16
17LoadDefinitions!({
18  InputDefinitions!("crossreftools", noltxml => true, extension => Some(Cow::Borrowed("sty")), handleoptions => true);
19  RawTeX!(
20    r"\def\crtaddlabeltotoc#1{\lx@crt@newlabel{#1}%
21  \ifcrtfinal\else\lx@crt@lla{#1}{\lx@crt@entry{#1}{\crtrefnumber{#1}}}\fi}
22\def\lx@crt@newlabel#1{\begingroup
23  \ifdefined\@currentlabelname\else\let\@currentlabelname\@empty\fi
24  \ifdefined\@currentHref\else\let\@currentHref\@empty\fi
25  \ifx\@currentHref\@empty\protected@edef\@currentHref{\lx@crt@counter.\@currentlabel}\fi
26  \protected@xdef\lx@crt@data{{\@currentlabel}{\thepage}{\@currentlabelname}{\@currentHref}{}}%
27  \endgroup\expandafter\global\expandafter\let\csname r@#1\endcsname\lx@crt@data}
28\def\crthypercref#1{\if@crt@hyperrefloaded\hyperref[#1]{\crtcref{#1}}\else\crtcref{#1}\fi}
29\def\crthyperCref#1{\if@crt@hyperrefloaded\hyperref[#1]{\crtCref{#1}}\else\crtCref{#1}\fi}
30\def\@crtlnameref@unstarred#1{\crtifdefinedlabel{#1}{%
31  \if@crt@hyperrefloaded\hyperref[#1]{\@crtlnameref@starred{#1}}\else\@crtlnameref@starred{#1}\fi}{}}
32\def\@crtunameref@unstarred#1{\crtifdefinedlabel{#1}{%
33  \if@crt@hyperrefloaded\hyperref[#1]{\@crtunameref@starred{#1}}\else\@crtunameref@starred{#1}\fi}{}}
34\def\crt@nameref@unstarred#1{\crtifdefinedlabel{#1}{%
35  \if@crt@hyperrefloaded\hyperref[#1]{\crtrefname{#1}}\else\crtrefname{#1}\fi}{}}"
36  );
37  // The package wraps whatever `\label` is at `\begin{document}` (:638-656); a binding's own begin-document `\label`
38  // (cleveref_sty.rs) is installed after the raw hooks ran, so the wrapper is put back around it, the order LaTeX's
39  // hooks give.
40  at_begin_document(TokenizeInternal!(
41    r"\ifx\label\lx@cleverref@label
42  \let\crt@l@bels@fe\label\def\label{\@ifnextchar[{\l@belwithopt@rg}{\l@belwithoutopt@rg}}\fi"
43  ))?;
44  // hyperref's anchor names a label's counter before its first `.` (`chapter.1`; `\crtrefcounter`,
45  // crossreftools.sty:332-343, splits it there); LaTeXML makes none, so the data carries the counter the label
46  // follows.
47  DefMacro!("\\lx@crt@counter", sub[_args] {
48    let counter = lookup_string("current_counter");
49    ExplodeText!(if counter.is_empty() { "Doc-Start".to_string() } else { counter })
50  });
51  // A label's `.lla` line, digested where the label is (as text), kept for the list; a label set twice (material
52  // digested again) is listed once.
53  DefPrimitive!("\\lx@crt@lla Semiverbatim {}", sub[(label, line)] {
54    let seen = s!("crt_lla_seen:{}", label.to_string());
55    if lookup_bool(&seen) {
56      return Ok(Vec::new());
57    }
58    assign_value(&seen, true, Some(Scope::Global));
59    begin_mode("text")?;
60    let digested = digest(line);
61    end_mode("text")?;
62    push_value("crt_lla", digested?)?;
63  });
64  // The line's number, then the label's name as written (its `_` or `^` are the name's characters).
65  DefConstructor!("\\lx@crt@entry Semiverbatim {}",
66  "<ltx:tocentry><ltx:ref labelref='#label'>#2 #1</ltx:ref></ltx:tocentry>",
67  properties => sub[args] {
68    let label = args[0].as_ref().map(|a| a.to_string()).unwrap_or_default();
69    Ok(stored_map!("label" => clean_label(&label, None).into_owned()))
70  });
71  DefConstructor!("\\@crt@listofl@bels", sub[document, _args, _props] {
72    // The `.lla` file is read once (crossreftools.sty:408-421: `\@input`, then `\openout` truncates it).
73    let lines: Vec<Digested> = match remove_value("crt_lla") {
74      Some(Stored::VecDequeStored(lines)) => {
75        lines.into_iter().filter_map(|line| line.into()).collect()
76      },
77      _ => Vec::new(),
78    };
79    let mut attrs: HashMap<String, String> = HashMap::default();
80    attrs.insert("class".into(), "ltx_listoflabels".into());
81    document.open_element("ltx:TOC", Some(attrs), None)?;
82    document.open_element("ltx:toclist", None, None)?;
83    for line in &lines {
84      document.absorb(line, None)?;
85    }
86    document.close_element("ltx:toclist")?;
87    document.close_element("ltx:TOC")?;
88  });
89});
