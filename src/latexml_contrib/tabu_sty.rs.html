141    let eq = Tokens!(T_OTHER!("="));

143    let down = Tokens!(T_SUB!());
144    let set = |reg: &str, d: &Dimension, out: &mut Vec<Token>| {

149    };

155    let _ = read_match(&[&eq])?;
156    if sel.is_none() {

158    }
159    match sel {
160      None => {

191/// The lets of tabu.sty:705-708 (`\tabu@init`, from `\tabu@setup` :687, for every tabu and longtabu) and :907's private
192/// `|` (`\tabu@privatecolumns`, :1132), local to the environment.
193fn tabu_init() -> Result<()> {
194  for (cs, to) in [
195    ("\\centering", "\\tabu@centering"),
196    ("\\raggedright", "\\tabu@raggedright"),
197    ("\\raggedleft", "\\tabu@raggedleft"),
198    ("\\NC@rewrite@|", "\\lx@tabu@vline"),
199  ] {
200    let_i(&T_CS!(cs), &T_CS!(to), None);
201  }
202  Digest!("\\tabu@trivlist")?;
203  Ok(())
