1//! Stub for pnas-new.cls (PNAS journal class).
2//!
3//! pnas-new.cls is the modern PNAS submission class. It is not in TeXLive;
4//! authors bundle it with their paper. The class defines several
5//! frontmatter helpers (\leadauthor, \correspondingauthor, \authorcontributions,
6//! \significancestatement, \authordeclaration) and a \templatetype{...}
7//! command that loads a per-article-type style (pnasresearcharticle.sty,
8//! pnasperspective.sty etc.). Since our binding registry routes
9//! pnas-new.cls → this binding, we provide gobble + frontmatter-preserve
10//! stubs for the author-facing helpers and a \templatetype that defers to
11//! \RequirePackage like the upstream does.
12//!
13//! Witness: 2305.01604 (significant 7-error gap on canvas, all from
14//! undefined pnas-new frontmatter macros).

33  // pnas-new.cls:422 `\newcommand{\doi}[1]{\def\@doi{#1}}` stores the DOI for the page footer
34  // (:268, :369-370): a setter taking a normal argument, often `\url{…}`, not OmniBus's `\doi`
35  // that reads a DOI verbatim as a url (KNOWN_PERL_ERRORS #326), which printed "\urlwww.pnas…".
36  // Witnesses 2605.03599, 2605.07504.
37  // The front matter's DOI only: a `.bbl`'s `\doi` (pnas-new.bst:116-121; plainnat's), inside
38  // `{thebibliography}`, meets the same setter in pdflatex and prints nothing. A body `\doi` after
39  // `\maketitle` is a DOI too (the footer's changes with it); each call adds a note, where the
40  // setter keeps the last.
41  DefMacro!("\\doi{}", sub[(doi)] {
42    let in_bibliography =
43      do_expand(Tokens!(T_CS!("\\@currenvir")))?.to_string().trim() == "thebibliography";
44    let mut tokens = if in_bibliography {
45      vec![T_CS!("\\def"), T_CS!("\\@doi"), T_BEGIN!()]
46    } else {
47      // (the footer prints the DOI alone, pnas-new.cls:268/:369-370 (2602.17158's copy): no label, FIDELITY_CHARTER §4,
48      // OXIDIZED_DESIGN_DIVERGENCES #921)
49      mouth::tokenize_internal(TeXString::assembled("\\lx@add@pubnote[role=doi,name={}]{".to_string())).unlist()
50    };
51    tokens.extend(doi.unlist());
52    tokens.push(T_END!());
53    Ok(Tokens::new(tokens))
54  });
