1//! eptcs.cls (Electronic Proceedings in Theoretical Computer Science; v1.5 2010/11/26, v1.6 2020/12/20, v1.7 2022/05/20
2//! — the same title page in each): the papers ship the class (it is not in TeX Live) and it is read raw, its packages
3//! loaded as it loads them (`\LoadClass[letterpaper,11pt,twoside]{article}`, hyperref, mathptmx, eptcsdata.tex when
4//! the paper has one). Perl has no binding.
5//!
6//! Its title page (eptcs.cls:160-191) sets `\@author` in a centred `tabular`, so `\and` starts a column of its own and
7//! `\institute{…}` and `\email{…}`, written inside `\author` after the names, are each `\\` and a line under them:
8//! `\author{Ann Able \institute{Univ A} \email{ann@a.edu} \and Bob Baker \institute{Univ B}}`. Read as the kernel's
9//! author line, the institute and e-mail stayed in the name ("Ilaria De Crescenzo Università degli Studi di Salerno
10//! Dipartimento di Informatica", 1408.5969; 1909.07676, 2406.16409, 2109.08295's "Jens Lehmann SDA Research Group, … \{grubenmann,
11//! jens.lehmann\}@cs.uni-bonn.de"). The binding writes the block as the class prints it and gives it whole to the
12//! reading (`\lx@add@author@split`, OXIDIZED_DESIGN_DIVERGENCES #984): each cell a column — cut where the block holds a
13//! separator meaning the kernel's `\and`, written, held in a macro or a `\let` copy (the separator query, #995; P2-4,
14//! OD #991 (8)), and followed by that separator's own tokens, which the reading cuts at — its names
15//! (`Tobias Grubenmann and Jens Lehmann`, 2109.08295) and each `\institute` and `\email` a `\\` line under them; the
16//! reading places each line — an institute every name's of the column, an e-mail the author it spells, else, a list of
17//! one address per name in order, the name at its place (1101.4424's `g.stavrinos@… \qquad tassiana98@…`), else the
18//! column's only author, else the block's (FIDELITY_CHARTER §4). A cell that opens with a `tabular` of its
19//! own — the names as its columns, a row of e-mails under them, and lines under the table (1407.7928) — or with an
20//! environment of the paper's own that opens one (2409.15014's `{authors}{5}`, its `\institute` and `\email` lines
21//! spanning the columns) is read the same way: a column's names its cell, a full row's cell a line under its column,
22//! any other line under the table every name's. A block with none of these is the kernel's.

55/// The author separators of the class's `\@maketitle` by spelling, `\and` and its kernel aliases: the cut where the
56/// separator query reads nothing ([`cells`]).
57fn and_delims() -> Vec<SplitDelim> {
58  vec![
59    SplitDelim::Token(T_CS!("\\and")),
60    SplitDelim::Token(T_CS!("\\And")),
61    SplitDelim::Token(T_CS!("\\AND")),
62  ]

65/// A cell of the block: the separator's own tokens written before it (none for the first), and its tokens.
66struct Cell {
67  separator: Vec<Token>,
68  tokens:    Vec<Token>,
69}

71/// The block's cells (its `\\[…]` options stripped), where it is the class's: an `\institute` or `\email` in it, or a
72/// cell opening with a table of names; `None` for a block the kernel's author line reads.
73fn eptcs_structured(body: &Tokens) -> Result<Option<Vec<Cell>>> {
74  let calls = body
75    .unlist_ref()
76    .iter()
77    .any(|t| *t == T_CS!("\\institute") || *t == T_CS!("\\email"));
78  let cells = cells(&strip_linebreak_options(body.clone()))?;
79  let structured = calls
80    || cells
81      .iter()
82      .any(|cell| leading_table(&cell.tokens).is_some());
83  Ok(structured.then_some(cells))
84}

86/// The block's cells, cut where it holds a separator meaning the kernel's `\and` — eptcs.cls's column break (its
87/// `\@maketitle` sets `\@author` in a `tabular[t]{c}`, :170-174, which latex.ltx's `\and` closes, opening the next):
88/// written, held in a macro of the document's, or a `\let` copy, as the separator query reads them by meaning
89/// (`argument_separators`, OXIDIZED_DESIGN_DIVERGENCES #995; a separator of another meaning — a class's or the
90/// document's renewed `\and`, amsmath's `\And` — prints no column, trap T8). A separator is cut only where its own tokens
91/// are whole units of the block's top level (no group or environment holds them, [`unit_end`]) and print nothing else;
92/// one a macro holds together with names stays in its cell, which the reading cuts at its `Sep(And)` (#993): the runs of
93/// `kernel_separator_runs`, the cut the institute lists of inst_support and aa share (P2-2). Where the
94/// query reads nothing (`None`: inside a rehearsal, over its allowance), the cut is by spelling, at `\and`, `\And`,
95/// `\AND` and an alias of `\and`'s own definition (`split_tokens`), as before P2-4. The spaces at a cell's ends are its
96/// readers' to drop ([`leading_table`], [`cell_items`]).
97fn cells(body: &Tokens) -> Result<Vec<Cell>> {
98  let Some(runs) = kernel_separator_runs(body)? else {
99    return Ok(
100      split_tokens_delimited(body.clone(), and_delims())
101        .into_iter()
102        .map(|(separator, tokens)| Cell {
103          separator,
104          tokens: tokens.unlist(),
105        })
106        .collect(),
107    );
108  };
109  let tokens = body.unlist_ref();
110  let mut cells = Vec::new();
111  let mut separator = Vec::new();
112  let mut from = 0;
113  for (start, end) in runs {
114    cells.push(Cell {
115      separator: std::mem::replace(&mut separator, tokens[start..end].to_vec()),
116      tokens:    tokens[from..start].to_vec(),
117    });
118    from = end;
119  }
120  cells.push(Cell {
121    separator,
122    tokens: tokens[from..].to_vec(),
123  });
124  Ok(cells)
125}

127/// The block as the class prints it, for the reading: each cell a column of names and the lines under them (an
128/// `\institute` or `\email` call is a `\\` line of its own, eptcs.cls:188-190 `\newcommand{\institute}[1]{…\\…}`,
129/// :191 `\newcommand{\email}[1]{\\…}`), each column after the separator written before it, and the lines under every
130/// name of a table, each a `_share=authors` entry.
131#[derive(Default)]
132struct Block {
133  /// each column: the separator written before it, its names and lines
134  columns:   Vec<(Vec<Token>, Vec<Token>)>,
135  shared:    Vec<Vec<Token>>,
136  /// the separator of the cell being read, for the column it opens
137  separator: Vec<Token>,
138}
139
140impl Block {
141  /// The cell next read follows `separator` (none for the first).
142  fn follows(&mut self, separator: Vec<Token>) { self.separator = separator; }
143
144  /// A new column holding `names`, after the cell's separator, or a column of a table after another column of it
145  /// (a table's own `&` columns, which the binding reads as names: `\and`); its place.
146  fn add_names(&mut self, names: Vec<Token>) -> usize {
147    let separator = match std::mem::take(&mut self.separator) {
148      separator if separator.is_empty() => vec![T_CS!("\\and")],
149      separator => separator,
150    };
151    self.columns.push((separator, names));
152    self.columns.len() - 1
153  }
154
155  /// A line under the names of column `at`.
156  fn line_under(&mut self, at: usize, line: Vec<Token>) {
157    let column = &mut self.columns[at].1;
158    column.push(T_CS!("\\\\"));
159    column.extend(line);
160  }
161
162  /// The block's whole list to the reading (`\lx@add@author@split`, as llncs's `\author`, OD #984), which cuts it at its
163  /// separators' `Sep(And)` (#993): each column after the separator its cell follows, as written (a literal `\and`, a
164  /// macro holding it, a `\let` copy), or a table's `\and`; then each line every name shares, an affiliation (e-mail
165  /// calls in it are lines of the row, placed by the reading: the address the author it spells, a list in order, else
166  /// the block's — FIDELITY_CHARTER §4).
167  fn calls(self) -> Result<Tokens> {
168    let mut list = Vec::new();
169    for (at, (separator, column)) in self.columns.into_iter().enumerate() {
170      if at > 0 {
171        list.push(T_SPACE!());
172        list.extend(separator);
173        list.push(T_SPACE!());
174      }
175      list.extend(column);
176    }
177    let mut out = Invocation!(T_CS!("\\lx@add@author@split"), vec![
178      None,
179      Some(Tokens::new(list))
180    ])
181    .unlist();
182    for line in self.shared {
183      let options = mouth::tokenize_internal("labelseq={affiliation},_share={authors}");
184      out.extend(
185        Invocation!(T_CS!("\\lx@add@affiliation"), vec![
186          Some(options),
187          Some(Tokens::new(line))
188        ])
189        .unlist(),
190      );
191    }
192    Ok(Tokens::new(out))
193  }
194}

196/// A call the class prints as a line of its own (`\institute{…}`, `\email{…}`), written as it is called.
197fn call_line(cs: Token, argument: Vec<Token>) -> Vec<Token> {
198  let mut line = vec![cs, T_BEGIN!()];
199  line.extend(argument);
200  line.push(T_END!());
201  line
202}

204/// An item of a cell: a line of text (after a `\\` or not), an `\institute`'s argument or an `\email`'s.
205enum Item {
206  Text {
207    after_break: bool,
208    line:        Vec<Token>,
209  },
210  Institute(Vec<Token>),
211  Email(Vec<Token>),
212}

214/// The calls the block makes: each cell ([`cells`]) read in turn.
215fn eptcs_author_calls(cells: Vec<Cell>) -> Result<Tokens> {
216  let mut block = Block::default();
217  for Cell { separator, tokens } in cells {
218    block.follows(separator);
219    match leading_table(&tokens) {
220      Some(table) => read_table(&mut block, table),
221      None => read_cell(&mut block, cell_items(&tokens)),
222    }
223  }
224  block.calls()
225}

227/// A cell's items in order (an environment in it, and a group, one unit of its text).
228fn cell_items(cell: &[Token]) -> Vec<Item> {
229  let mut items = Vec::new();
230  let mut line: Vec<Token> = Vec::new();
231  let mut after_break = false;
232  let push_line = |items: &mut Vec<Item>, line: &mut Vec<Token>, after_break: bool| {
233    let taken = std::mem::take(line);
234    if !is_blank(&taken) {
235      items.push(Item::Text {
236        after_break,
237        line: trim_spaces(taken),
238      });
239    }
240  };
241  let mut at = 0;
242  while at < cell.len() {
243    let t = cell[at];
244    if t == T_CS!("\\\\") {
245      push_line(&mut items, &mut line, after_break);
246      after_break = true;
247      at += 1;
248    } else if t == T_CS!("\\institute") || t == T_CS!("\\email") {
249      push_line(&mut items, &mut line, after_break);
250      let (argument, next) = read_argument(cell, at + 1);
251      items.push(if t == T_CS!("\\institute") {
252        Item::Institute(argument)
253      } else {
254        Item::Email(argument)
255      });
256      after_break = true;
257      at = next;
258    } else {
259      let next = unit_end(cell, at);
260      line.extend_from_slice(&cell[at..next]);
261      at = next;
262    }
263  }
264  push_line(&mut items, &mut line, after_break);
265  items
266}

268/// A cell read as the class prints it: a column, its first line the names and each `\institute`, `\email` and `\\`
269/// line a line under them, each decided by the reading (an e-mail list, an affiliation, else a row of names of its own).
270fn read_cell(block: &mut Block, items: Vec<Item>) {
271  let mut column: Option<usize> = None;
272  for item in items {
273    let line = match item {
274      Item::Institute(argument) => call_line(T_CS!("\\institute"), argument),
275      Item::Email(argument) => call_line(T_CS!("\\email"), argument),
276      Item::Text { line, .. } => line,
277    };
278    match column {
279      Some(at) => block.line_under(at, line),
280      None => column = Some(block.add_names(line)),
281    }
282  }
283}

298/// The environments that set their content as a table: the tabular family, and an environment of the document's own
299/// whose begin code opens one (2409.15014's `\newenvironment{authors}[1]{…\begin{tabular}[t]{…}}`), with the number
300/// of arguments each takes after its name (an optional one first, if any, is read as one).
301fn table_arguments(name: &str) -> Option<usize> {
302  match name {
303    "tabular" | "array" => Some(1),
304    "tabular*" | "tabularx" | "tabulary" => Some(2),
305    _ => {
306      let definition = lookup_definition(&T_CS!(s!("\\{name}"))).ok().flatten()?;
307      let Some(ExpansionBody::Tokens(begin)) = definition.get_expansion() else {
308        return None;
309      };
310      let text = begin.to_string();
311      (text.contains("{tabular}") || text.contains("{tabular*}")).then(|| {
312        definition
313          .get_parameters()
314          .map_or(0, |p| p.get_parameters().len())
315      })
316    },
317  }
318}

320/// The table a cell opens with (after spaces), when it is one ([`table_arguments`]).
321fn leading_table(cell: &[Token]) -> Option<Table> {
322  let start = cell
323    .iter()
324    .take_while(|t| t.get_catcode() == Catcode::SPACE)
325    .count();
326  if cell.get(start) != Some(&T_CS!("\\begin")) {
327    return None;
328  }
329  let (name_tokens, mut at) = read_argument(cell, start + 1);
330  let name = Tokens::new(name_tokens).to_string();
331  let nargs = table_arguments(&name)?;
332  let end = environment_end(cell, start, &name)?;
333  // (its arguments: a position, `[t]`, then the given number of groups)
334  at += cell[at..]
335    .iter()
336    .take_while(|t| t.get_catcode() == Catcode::SPACE)
337    .count();
338  if cell.get(at) == Some(&T_OTHER!("[")) {
339    at = cell[at..end.0]
340      .iter()
341      .position(|t| *t == T_OTHER!("]"))
342      .map_or(at, |close| at + close + 1);
343  }
344  for _ in 0..nargs {
345    at = read_argument(cell, at).1;
346  }
347  let body = &cell[at.min(end.0)..end.0];
348  Some(Table {
349    rows: table_rows(body),
350    tail: cell[end.1..].to_vec(),
351  })
352}

354/// A table's rows: split at its `\\`, each `\institute`/`\email` a row of its own, each row's cells at its `&`.
355fn table_rows(body: &[Token]) -> Vec<Row> {
356  let mut rows = Vec::new();
357  let mut line: Vec<Token> = Vec::new();
358  let push_cells = |rows: &mut Vec<Row>, line: &mut Vec<Token>| {
359    let taken = std::mem::take(line);
360    let cells: Vec<(Vec<Token>, usize)> =
361      split_tokens(Tokens::new(taken), vec![SplitDelim::Token(T_ALIGN!())])
362        .into_iter()
363        .map(|cell| multicolumn(cell.unlist()))
364        .collect();
365    if cells.iter().any(|(cell, _)| !is_blank(cell)) {
366      rows.push(Row::Cells(cells));
367    }
368  };
369  let mut at = 0;
370  while at < body.len() {
371    let t = body[at];
372    if t == T_CS!("\\\\") {
373      push_cells(&mut rows, &mut line);
374      at += 1;
375    } else if t == T_CS!("\\institute") || t == T_CS!("\\email") {
376      push_cells(&mut rows, &mut line);
377      let (argument, next) = read_argument(body, at + 1);
378      rows.push(if t == T_CS!("\\institute") {
379        Row::Institute(argument)
380      } else {
381        Row::Email(argument)
382      });
383      at = next;
384    } else {
385      let next = unit_end(body, at);
386      line.extend_from_slice(&body[at..next]);
387      at = next;
388    }
389  }
390  push_cells(&mut rows, &mut line);
391  rows
392}

394/// A cell's content and how many columns it spans: a `\multicolumn{n}{spec}{content}`'s, else the cell itself.
395fn multicolumn(cell: Vec<Token>) -> (Vec<Token>, usize) {
396  let start = cell
397    .iter()
398    .take_while(|t| t.get_catcode() == Catcode::SPACE)
399    .count();
400  if cell.get(start) == Some(&T_CS!("\\multicolumn")) {
401    let (span, at) = read_argument(&cell, start + 1);
402    let (_, at) = read_argument(&cell, at);
403    let (content, _) = read_argument(&cell, at);
404    let span = Tokens::new(span).to_string().trim().parse().unwrap_or(1);
405    return (content, span);
406  }
407  (cell, 1)
408}

410/// A table of names read as the class prints it: its first row the names, a cell of each column its names; a full row
411/// under them a line under each column, unless it reads as names, a row of names of its own; any other row a line
412/// under every name of the table, a `_share=authors` entry; and after the table, what stands beside it the last name's
413/// (2409.15014's `\thanks` after `\end{authors}`, its mark printed after "Verena Wolf") and each line under it every
414/// name's (1407.7928's institute lines).
415fn read_table(block: &mut Block, table: Table) {
416  let mut columns: Vec<usize> = Vec::new();
417  for row in table.rows {
418    match row {
419      Row::Institute(line) => block.shared.push(call_line(T_CS!("\\institute"), line)),
420      Row::Email(line) => block.shared.push(call_line(T_CS!("\\email"), line)),
421      Row::Cells(cells) => {
422        let full = !columns.is_empty()
423          && cells.len() == columns.len()
424          && cells.iter().all(|(_, span)| *span == 1);
425        let names_row = columns.is_empty()
426          || (full
427            && cells.iter().all(|(cell, _)| {
428              let tokens = Tokens::new(cell.clone());
429              is_blank(cell) || !(line_is_email_list(&tokens) || reads_as_affiliation(&tokens))
430            }));
431        if names_row && (columns.is_empty() || full) {
432          columns = cells
433            .into_iter()
434            .map(|(cell, _)| block.add_names(trim_spaces(cell)))
435            .collect();
436        } else if full {
437          for ((cell, _), &column) in cells.into_iter().zip(&columns) {
438            if !is_blank(&cell) {
439              block.line_under(column, trim_spaces(cell));
440            }
441          }
442        } else {
443          let mut line: Vec<Token> = Vec::new();
444          for (cell, _) in cells {
445            if !is_blank(&cell) {
446              if !line.is_empty() {
447                line.push(T_SPACE!());
448              }
449              line.extend(trim_spaces(cell));
450            }
451          }
452          block.shared.push(line);
453        }
454      },
455    }
456  }
457  for item in cell_items(&table.tail) {
458    match item {
459      Item::Institute(line) => block.shared.push(call_line(T_CS!("\\institute"), line)),
460      Item::Email(line) => block.shared.push(call_line(T_CS!("\\email"), line)),
461      Item::Text { after_break: false, line } => {
462        if let Some((_, last)) = block.columns.last_mut() {
463          last.extend(line);
464        }
465      },
466      Item::Text { line, .. } => block.shared.push(line),
467    }
468  }
469}

471/// The spacing a cell's line may hold beside its text: no text of its own.
472fn is_spacing(t: &Token) -> bool {
473  t.get_catcode() == Catcode::SPACE
474    || [
475      "\\smallskip",
476      "\\medskip",
477      "\\bigskip",
478      "\\hline",
479      "\\par",
480      "\\centering",
481      "\\noindent",
482      "\\relax",
483    ]
484    .iter()
485    .any(|cs| *t == T_CS!(*cs))
486}
487
488/// Whether a line prints nothing: spacing alone.
489fn is_blank(line: &[Token]) -> bool { line.iter().all(is_spacing) }
490
491/// A line without the spacing at its ends.
492fn trim_spaces(line: Vec<Token>) -> Vec<Token> {
493  let start = line.iter().take_while(|t| is_spacing(t)).count();
494  let end = line.len()
495    - line[start..]
496      .iter()
497      .rev()
498      .take_while(|t| is_spacing(t))
499      .count();
500  line[start..end].to_vec()
501}
