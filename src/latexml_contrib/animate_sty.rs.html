1//! animate.sty — PDF/SVG animation from graphics files
2//!
3//! Replaces the unbounded/many-frame loops of animate.sty with a single
4//! representative frame (the first frame), avoiding memory exhaustion on
5//! multi-frame animations (e.g. tikz-among-us with 180 frames).
6//! Exposes the frame count as RDFa `property="ltx:frameCount" content="N"` on
7//! `<ltx:block class="ltx_animate">` (schema-legal Common.attributes).
8

21struct AnimateContext {
22  frame_count: usize,
23  /// animate.sty:3197-3201 `end` key: code run at the END of every frame
24  /// (`\@anim@endframe`, animate.sty:2314-2340); the `begin` code is emitted
25  /// straight into the stream by `\begin{animateinline}`.
26  end:         Vec<Token>,
27  /// The representative frame has been closed (its `end` code emitted).
28  ended:       bool,
29}
30
31/// Discard the rest of the `{animateinline}` body up to its literal
32/// `\end{animateinline}` (unexpanded read, bounded by the end of input) and
33/// count the `\newframe` separators it contained.
34fn discard_remaining_frames() -> Result<usize> {
35  let body = read_env_body_tokens("animateinline")?;
36  Ok(
37    body
38      .iter()
39      .filter(|t| t.get_catcode() == Catcode::CS && t.to_string() == "\\newframe")
40      .count(),
41  )
42}
43
44fn env_tokens(cs: &str, env: &str) -> Vec<Token> {
45  let mut toks = vec![T_CS!(cs), T_BEGIN!()];
46  toks.extend(tok_str(env).unlist());
47  toks.push(T_END!());
48  toks
49}

51/// The open `{animateinline}` contexts, innermost last. Each entry is SHARED
52/// (`Rc`) with the environment whatsit's `anim_ctx` property
53/// (`Stored::Opaque`), so the macros (`\newframe`, `\multiframe`,
54/// `\end{animateinline}`) mutate the innermost one and the whatsit reads its
55/// own at construction — in whatever order deferred whatsits construct. The
56/// stack is popped when the environment ends (`after_digest`), not at
57/// construction.
58type AnimCtx = Rc<RefCell<AnimateContext>>;
59
60/// The open animations, innermost last. A rollback puts it back: a block's `\begin{animateinline}` with no end stood open
61/// when it was a thread-local, a later `\multiframe` counted into it.
62static ANIM_STACK: DocumentState<Vec<AnimCtx>> = DocumentState::new("animate_stack");
63
64/// Run `f` on the innermost open context, if any.
65fn with_top<R>(f: impl FnOnce(&mut AnimateContext) -> R) -> Option<R> {
66  let top: Option<AnimCtx> = ANIM_STACK.read(|s| s.last().cloned());
67  top.map(|ctx| f(&mut ctx.borrow_mut()))
68}
69
70fn tok_str(s: &str) -> Tokens { mouth::tokenize_internal(TeXString::assembled(s.to_string())) }
71
72/// Parse a single variable declaration like `var = init + inc` or `var = init - inc` or `var = init`.
73fn parse_var_decl(decl: &str) -> Option<(String, String)> {
74  let (name, rest) = decl.split_once('=')?;
75  let var_name = name.trim().trim_start_matches('\\');
76  let rest = rest.trim();
77  if var_name.is_empty() || rest.is_empty() {
78    return None;
79  }
80  // Find '+' or '-' separating initial value and increment,
81  // skipping any leading sign of the initial value.
82  let start = if rest.starts_with('+') || rest.starts_with('-') {
83    1
84  } else {
85    0
86  };
87  let sep_idx = rest[start..].find(['+', '-']).map(|i| start + i);
88  let mut init_val = match sep_idx {
89    Some(idx) => rest[..idx].trim(),
90    None => rest,
91  };
92  if init_val.starts_with('{') && init_val.ends_with('}') && init_val.len() >= 2 {
93    init_val = init_val[1..init_val.len() - 1].trim();
94  }
95  Some((var_name.to_string(), init_val.to_string()))
96}

99LoadDefinitions!({
100  // animate.sty:65 reads its options with `\ProcessKeyOptions`, which marks them processed: `\@curroptions` stays as it was.
101  key_options_processed()?;
102  // Per-conversion reset: a mid-body fatal leaves an unbalanced push behind
103  // on a long-running worker thread (chemnum resets the same way).
104  ANIM_STACK.update(Vec::clear);
105  RequirePackage!("graphicx");
106  // animate.sty:23: a document can rely on its calc (OXIDIZED_DESIGN #317).
107  RequirePackage!("calc");
108
109  // animate.sty:3197-3201: the per-frame `begin`/`end` code keys (the ones
110  // the single-frame model consumes; the rest are read and skipped).
111  DefKeyVal!("animate", "begin", "UndigestedKey");
112  DefKeyVal!("animate", "end", "UndigestedKey");
113
114  // `\begin{animateinline}[opts]{fps}` is a macro layer over the block
115  // constructor `{lx@animateinline}`: the option list's `begin`/`end` code is
116  // run at the start/end of the ONE representative frame, exactly where
117  // animate.sty:2314-2340 runs `\@anim@begin`/`\@anim@end` for every frame,
118  // and the first `\newframe` closes that frame and discards the rest of the
119  // body (liftarm.tex:626-641 `\liftarmanimate` wraps every frame in a
120  // tikzpicture through `begin=`/`end=`: without them 72 frames ran outside
121  // any picture, 813 errors, sweep 63).
122  DefMacro!(T_CS!("\\begin{animateinline}"), "[] {}", sub[(opts, _fps)] {
123    // The option list is a keyval list (animate.sty:3197-3201 `begin`/`end`
124    // are `.tl_gset:N` keys): the core `KeyVals` reader (`read_from`, the
125    // `=`/`,` grammar with brace handling; `Tokens::to_keyvals` is only a
126    // token-pairing), silenced for animate's many undeclared keys.
127    let mut kv = KeyVals::new(KeyvalsConfig { keysets: vec!["animate".into()], ..Default::default() });
128    if let Some(list) = opts.owned_tokens() {
129      // `read_from` consumes the opening delimiter itself.
130      let mut toks = vec![T_OTHER!("[")];
131      toks.extend(list.unlist());
132      toks.push(T_OTHER!("]"));
133      reading_from_mouth(Mouth::default(), || {
134        unread(Tokens::new(toks));
135        kv.read_from(T_OTHER!("]"), true)
136      })?;
137    }
138    let key_tokens = |key: &str| -> Vec<Token> {
139      kv.get_value(key)
140        .and_then(|a| a.clone().owned_tokens())
141        .map(|t| t.unlist())
142        .unwrap_or_default()
143    };
144    let begin = key_tokens("begin");
145    let end = key_tokens("end");
146    ANIM_STACK.update(|s| {
147      s.push(Rc::new(RefCell::new(AnimateContext { frame_count: 1, end, ended: false })))
148    });
149    let mut toks = env_tokens("\\begin", "lx@animateinline");
150    toks.extend(begin);
151    Ok(Tokens::new(toks))
152  });
153  DefMacro!(T_CS!("\\end{animateinline}"), None, sub[_args] {
154    let mut toks = with_top(|ctx| {
155      if ctx.ended {
156        Vec::new()
157      } else {
158        ctx.ended = true;
159        ctx.end.clone()
160      }
161    })
162    .unwrap_or_default();
163    toks.extend(env_tokens("\\end", "lx@animateinline"));
164    Ok(Tokens::new(toks))
165  });
166
167  DefEnvironment!(
168    "{lx@animateinline}",
169    sub[document, _args, props] {
170      document.maybe_close_element("ltx:p")?;
171      let frame_count = match props.get("anim_ctx") {
172        Some(Stored::Opaque(payload)) => payload
173          .downcast_ref::<AnimCtx>()
174          .map(|ctx| ctx.borrow().frame_count)
175          .unwrap_or(1),
176        _ => 1,
177      };
178      let mut attrs = HashMap::default();
179      attrs.insert("class".to_string(), "ltx_animate".to_string());
180      // The collapsed animation's frame count rides as RDFa (`property`/`content`
181      // are Common.attributes and the XSLT passes them through) — the ad-hoc
182      // `frame-count` attribute lived only in the RUNTIME model, so every
183      // animate document failed schema validation (chessboard, liftarm, movie15,
184      // tikz-among-us, tikz-mirror-lens ×2; batch 56ez).
185      attrs.insert("property".to_string(), "ltx:frameCount".to_string());
186      attrs.insert("content".to_string(), frame_count.to_string());
187      document.open_element("ltx:block", Some(attrs), None)?;
188      if let Some(Stored::Digested(body)) = props.get("body") {
189        document.absorb(body, None)?;
190      }
191      document.close_element("ltx:block")?;
192      Ok(())
193    },
194    mode => "internal_vertical",
195    properties => {
196      let mut props = stored_map!();
197      if let Some(ctx) = ANIM_STACK.read(|s| s.last().cloned()) {
198        props.insert("anim_ctx", Stored::Opaque(Rc::new(ctx)));
199      }
200      Ok(props)
201    },
202    after_digest => sub[_whatsit] {
203      ANIM_STACK.update(|s| s.pop());
204    }
205  );

207  DefMacro!("\\anim@set@framecount {Number}", sub[(count)] {
208    let count_num = count.value_of().max(1) as usize;
209    with_top(|ctx| ctx.frame_count = count_num);
210    Ok(Tokens::new(vec![]))
211  });
212
213  DefMacro!("\\multiframe {Number} {} {}", sub[(count, vars, body)] {
214    let count_num = count.value_of().max(1) as usize;
215    let is_inside_animate = ANIM_STACK.read(|s| !s.is_empty());
216    if is_inside_animate {
217      // `\multiframe{n}` inside a frame sequence stands for n frames.
218      with_top(|ctx| ctx.frame_count += count_num - 1);
219    }
220
221    let vars_str = vars.to_string();
222    let mut toks = Vec::new();
223    for decl in vars_str.split(',') {
224      if let Some((name, init_val)) = parse_var_decl(decl) {
225        toks.push(T_CS!("\\def"));
226        toks.push(T_CS!(format!("\\{}", name)));
227        toks.push(T_BEGIN!());
228        toks.extend(tok_str(&init_val).unlist());
229        toks.push(T_END!());
230      }
231    }
232
233    toks.extend(body.unlist());
234
235    if !is_inside_animate {
236      let mut wrapped = Vec::new();
237      wrapped.push(T_CS!("\\begin"));
238      wrapped.push(T_BEGIN!());
239      wrapped.extend(tok_str("animateinline").unlist());
240      wrapped.push(T_END!());
241      wrapped.push(T_BEGIN!());
242      wrapped.extend(tok_str("1").unlist());
243      wrapped.push(T_END!());
244
245      wrapped.push(T_CS!("\\anim@set@framecount"));
246      wrapped.push(T_BEGIN!());
247      wrapped.extend(tok_str(&count_num.to_string()).unlist());
248      wrapped.push(T_END!());
249
250      wrapped.extend(toks);
251
252      wrapped.push(T_CS!("\\end"));
253      wrapped.push(T_BEGIN!());
254      wrapped.extend(tok_str("animateinline").unlist());
255      wrapped.push(T_END!());
256      Ok(Tokens::new(wrapped))
257    } else {
258      Ok(Tokens::new(toks))
259    }
260  });
261
262  // The first `\newframe` ends the representative frame: it runs the `end`
263  // code, discards the remaining frames (counting their separators for
264  // `frame-count`) and re-emits the environment end.
265  DefMacro!("\\newframe OptionalMatch:* []", sub[(_star, _fps)] {
266    let end = with_top(|ctx| {
267      ctx.frame_count += 1;
268      if ctx.ended {
269        None
270      } else {
271        ctx.ended = true;
272        Some(ctx.end.clone())
273      }
274    })
275    .flatten();
276    let Some(mut toks) = end else {
277      return Ok(Tokens::new(vec![]));
278    };
279    let rest = discard_remaining_frames()?;
280    with_top(|ctx| ctx.frame_count += rest);
281    toks.extend(env_tokens("\\end", "animateinline"));
282    Ok(Tokens::new(toks))
283  });

285  def_macro_noop("\\multiframebreak")?;

287  DefMacro!(
288    "\\animategraphics [] {} {} {} {}",
289    sub[(opts, fps, basename, first, last)] {
290      let f_str = first.to_string().trim().to_string();
291      let l_str = last.to_string().trim().to_string();
292      let base_str = basename.to_string().trim().to_string();
293      let f_num = f_str.parse::<i64>();
294      let l_num = l_str.parse::<i64>();
295      let frame_count = match (&f_num, &l_num) {
296        (Ok(f), Ok(l)) => ((l - f).abs() + 1) as usize,
297        _ => 1,
298      };
299      let mut file = if f_str.is_empty() {
300        base_str.clone()
301      } else {
302        format!("{base_str}{f_str}")
303      };
304      // Animate reverse playback (animate.sty:1560ff) or padded numbering:
305      // if the first candidate file does not exist, look for an existing frame
306      // among zero-padded variants, the other end of the range, or base_str itself
307      // (multipage document).
308      if !file.is_empty() && util::image::image_candidates(&file).is_empty() {
309        let mut alt_candidates = Vec::new();
310        if let (Ok(f), Ok(l)) = (f_num, l_num) {
311          if l_str.len() > f_str.len() {
312            alt_candidates.push(format!("{base_str}{:0width$}", f, width = l_str.len()));
313          }
314          alt_candidates.push(format!("{base_str}{l_str}"));
315          if f_str.len() > l_str.len() {
316            alt_candidates.push(format!("{base_str}{:0width$}", l, width = f_str.len()));
317          }
318        }
319        alt_candidates.push(base_str);
320        for cand in alt_candidates {
321          if !util::image::image_candidates(&cand).is_empty() {
322            file = cand;
323            break;
324          }
325        }
326      }
327      let mut toks = Vec::new();
328      // \begin{animateinline}[opts]{fps}
329      toks.push(T_CS!("\\begin"));
330      toks.push(T_BEGIN!());
331      toks.extend(tok_str("animateinline").unlist());
332      toks.push(T_END!());
333      if let Some(ref opts) = opts {
334        toks.push(T_OTHER!("["));
335        toks.extend(opts.clone().unlist());
336        toks.push(T_OTHER!("]"));
337      }
338      toks.push(T_BEGIN!());
339      toks.extend(fps.unlist());
340      toks.push(T_END!());
341
342      // \anim@set@framecount{frame_count}
343      toks.push(T_CS!("\\anim@set@framecount"));
344      toks.push(T_BEGIN!());
345      toks.extend(tok_str(&frame_count.to_string()).unlist());
346      toks.push(T_END!());
347
348      // \includegraphics[opts]{file}
349      toks.push(T_CS!("\\includegraphics"));
350      if let Some(opts) = opts {
351        toks.push(T_OTHER!("["));
352        toks.extend(opts.unlist());
353        toks.push(T_OTHER!("]"));
354      }
355      toks.push(T_BEGIN!());
356      toks.extend(tok_str(&file).unlist());
357      toks.push(T_END!());
358
359      // \end{animateinline}
360      toks.push(T_CS!("\\end"));
361      toks.push(T_BEGIN!());
362      toks.extend(tok_str("animateinline").unlist());
363      toks.push(T_END!());
364
365      Ok(Tokens::new(toks))
366    }
367  );
