1//! apacite.sty (TL 2025 apacite.sty v6.03) — APA citations and reference lists.
2//!
3//! Two layers, as the package has them:
4//! * the reference-list formatting layer — every macro apacite.bst writes into a `.bbl`
5//!   (`\APACrefYearMonthDay`, `\APACjournalVolNumPages`, `\BPGS`, `\BCnt`, `{APACrefURL}`, …)
6//!   and the strings they print — is the package's own TeX, run verbatim (`RawTeX!` blocks
7//!   below, each citing its apacite.sty lines);
8//! * the citation layer (`\cite<pre>[post]{keys}`, `\citeA`, `\citeNP`, …) is a semantic
9//!   binding onto natbib's `\cite`: natbib_sty.rs `\lx@NAT@parselabel` splits the bibitem label
10//!   `\citeauthoryear{full}{short}{year}` and digests its apacite macros (`\APACyear`,
11//!   `\APACexlab{\BCnt{1}}`, `\BBA`, `\BOthers`, `\APACciteatitle`, …) into the `ltx:tags` the
12//!   post-processor cites from.
13//!
14//! Perl ships no apacite binding: its default run leaves every macro undefined, and
15//! `--includestyles` raw-loads apacite.sty, whose own `\cite`, `\@lbibitem` and
16//! `\thebibliography` (apacite.sty:259-470, 1152-1242) leave every citation unresolved.
17use latexml_package::{package::natbib_sty::natbib_invocation, prelude::*};
18
19/// An apacite citation form `\<form> [pre] <pre> [post] {keys}` (the signature below) as the natbib command `cs` over the
20/// notes as read: a `[post]` given goes on with the pre-note before it (`[pre][post]`); a `<pre>` alone is the pre-note
21/// before an empty post-note (`[pre][]`); a lone `[note]` stays a lone one, which natbib reads as the post-note
22/// (apacite's `\@cite<>[note]`, apacite.sty:313-327). Both pre-note slots given (`[a]<b>`, no spelling of either
23/// package) are joined.
24fn apacite_delegate(cs: &str, args: Vec<ArgWrap>) -> Result<Tokens> {
25  let mut it = args.into_iter();
26  let bracket_pre: Option<Tokens> = it.next().unwrap().into();
27  let angled_pre: Option<Tokens> = it.next().unwrap().into();
28  let post: Option<Tokens> = it.next().unwrap().into();
29  let keys: Tokens = it.next().unwrap().into();
30  let angled = angled_pre.is_some();
31  let pre = match (bracket_pre, angled_pre) {
32    (Some(a), Some(b)) => Some(Tokens::new([a.unlist(), b.unlist()].concat())),
33    (a, b) => a.or(b),
34  };
35  let post = if angled {
36    Some(post.unwrap_or_default())
37  } else {
38    post
39  };
40  Ok(natbib_invocation(cs, None, pre, post, keys))
41}

584  // apacite citation forms (apacite.sty L328+). Delegate to natbib's \cite
585  // which we wrapped in natbib_sty.rs. Forms:
586  //   \citeA<pre>[post]{key}      — author-only ("Smith")
587  //   \citeNP<pre>[post]{key}     — citation without parens
588  //   \citeyearNP<pre>[post]{key} — year-only without parens
589  // Witness 2407.14158, 2407.18402, 2407.16770 (apacite-using papers).
590  //
591  // apacite's ANGLE-BRACKET pre-note. apacite.sty L259-311 gives every classic
592  // cite form the dispatch `\@ifnextchar< {\@cite} {\@cite<>}`, and L313-327
593  // `\def\@cite<#1>{... \@ifnextchar[ {\@@cite<#1>} {\@@cite<#1>[]}}` — so the
594  // full apacite citation syntax is
595  //   \cite<pre-note>[post-note]{key-list}
596  // Without the `<...>` form the kernel/natbib `\cite` takes the single token
597  // `<` as its key list: the citation renders as a dangling `[<]`, the REAL
598  // keys are never cited (so they are silently absent from the References) and
599  // `see>` leaks into the body text. Witness 2605.10951
600  // (`\cite<see>{Gangopadhyay02,Ferris25}`, agujournal2019), 2606.16518,
601  // 2606.19048, 2606.21531, 2606.24563.
602  //
603  // `OptionalAngled` (NOT `OptionalMatch:< OptionalUntil:>`): `Until` never
604  // checks for the OPENING delimiter, so when no `<` is present it scans to the
605  // next `>` anywhere downstream — `\citeA{Smith} and $a > b$` swallowed the
606  // cite and the math and reported the key as `b`.
607  // The signature `[pre1] OptionalAngled [post]` is UNAMBIGUOUS and so serves
608  // both spellings with one prototype: natbib's `[pre][post]` can only ever
609  // match `[pre1]…[post]` (the angled slot peeks, sees `[`, and yields nothing),
610  // and apacite's `<pre>[post]` can only ever match `OptionalAngled [post]`
611  // (the `[pre1]` slot peeks, sees `<`, and yields nothing). The notes go on to
612  // natbib as read (`apacite_delegate`): an absent post-note stays absent, so
613  // `\cite[p.~5]{k}` is the post-note natbib's swap makes of a lone `[…]` (apacite's
614  // `\@cite<>[p.~5]`, apacite.sty:313-327, "(Smith, 2001, p. 5)"), and a `<pre>` alone
615  // is a pre-note before an empty post-note. 2002.00730, 2005.00039.
616  DefMacro!("\\citeA [] OptionalAngled [] Semiverbatim", sub[args] {
617    apacite_delegate("\\citet", args)
618  });
619  DefMacro!("\\citeNP [] OptionalAngled [] Semiverbatim", sub[args] {
620    apacite_delegate("\\citealp", args)
621  });
622  DefMacro!("\\citeyearNP [] OptionalAngled [] Semiverbatim", sub[args] {
623    apacite_delegate("\\citeyear", args)
624  });

627  // `\def\@BBOP{\BBOP}`, its open parenthesis) is parenthetical with or without notes: natbib's `\cite` in its
628  // parenthetical form, which keeps the `cite` class of the macro the author wrote (natbib_sty.rs
629  // `\lx@natbib@cite@parenthetical`; natbib's own `\cite{k}` is textual in author-year mode).
630  // OXIDIZED_DESIGN_DIVERGENCES #655. `natbibemu` (:587, :883-892) sends `\cite` to `\citep` when a `[` follows and to
631  // `\citet` otherwise (starred alike); `natbibapa` (:995-1037) and `nocitation` define no `\cite`, so natbib's own
632  // stays (textual without notes, natbib.sty:694-701) — the `[natbibapa]{apacite}` of WileyASNA-v1.cls:2120, the
633  // Wiley NJD classes and sn-jnl.cls's `sn-apa` (L1683-1685). Witnesses 2307.11525, 2406.19732.
634  let mode = lookup_string("apacite@citation");
635  if mode == "apaciteclassic" {
636    DefMacro!("\\cite [] OptionalAngled [] Semiverbatim", sub[args] {
637      apacite_delegate("\\lx@natbib@cite@parenthetical", args)
638    });
639  } else if mode == "natbibemu" {
640    DefMacro!("\\cite OptionalMatch:* [][] Semiverbatim", sub[args] {
641      let mut it = args.into_iter();
642      let star: Option<Tokens> = it.next().unwrap().into();
643      let pre: Option<Tokens> = it.next().unwrap().into();
644      let post: Option<Tokens> = it.next().unwrap().into();
645      let keys: Tokens = it.next().unwrap().into();
646      let cs = if pre.is_some() { "\\citep" } else { "\\citet" };
647      Ok(natbib_invocation(cs, star, pre, post, keys))
648    });
649  }
