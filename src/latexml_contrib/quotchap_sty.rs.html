1//! quotchap.sty — quotations that open a chapter. The package typesets each
2//! `savequote` into a box that its own `\chapter` prints at the next chapter
3//! head (quotchap.sty:97-132). Our `\chapter` is locked, as Perl's is
4//! (latex_constructs.pool.ltxml:557), so that `\chapter` never runs and the box
5//! was never printed: the manual lost both quotations and their authors
6//! (recall 44 %, sweep 120; Perl loses them alike). The quotation becomes an
7//! epigraph (epigraph_sty.rs's structure) where it is written, just before the
8//! chapter it opens (DIVERGENCES #287).
9use latexml_package::prelude::*;
10
11LoadDefinitions!({
12  InputDefinitions!("quotchap", noltxml => true, extension => Some(Cow::Borrowed("sty")));
13  // quotchap.sty:97 `\begin{savequote}[<width>]`: the width is the box's.
14  DefEnvironment!(
15    "{savequote}[]",
16    "<ltx:quote class='ltx_epigraph ltx_quotchap'>#body</ltx:quote>"
17  );
18  // quotchap.sty:106 `\qauthor{<source>}`, set flush right under the quotation.
19  DefConstructor!(
20    "\\qauthor{}",
21    "<ltx:block class='ltx_epigraph_source'>#1</ltx:block>"
22  );
23});
