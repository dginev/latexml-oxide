1//! ecai.cls (ECAI / EurAI conference class).
2//!
3//! Covers the class's HTML-relevant frontmatter model (ecai.cls L392-851): its
4//! reference-linked author/affiliation scheme, where `\author[<addr-refs>]{name}`
5//! names each author and `\address[<label>]{text}` defines an affiliation, the
6//! two connected by the shared label (like elsarticle). The rest of the class is
7//! print layout (margins, fonts, two-column, page styles) with no HTML meaning
8//! and is left to OmniBus/the kernel.
9use latexml_package::prelude::*;
10
11use crate::and_by_meaning::and_runs;
12
13LoadDefinitions!({
14  LoadClass!("OmniBus");
15  // ecai.cls loads no amsmath (67 copies in a 1/20 arXiv sample): amsmath's commands for papers that use them unloaded,
16  // its "Old form" error once the paper loads it; the paper's copy decides
17  // (`amsmath_sty::require_amsmath_beyond_original`, KNOWN_PERL_ERRORS #993).
18  amsmath_sty::require_amsmath_beyond_original(
19    latexml_package::shipped_class::shipped_copy_loads("cls", "amsmath")?,
20    &[],
21  )?;
22  RequirePackage!("amsthm");
23  // Eager xcolor preload removed for Perl parity: it makes a later document
24  // xcolor[table] load a no-op, so colortbl/array never load and array m{}/b{}
25  // columns break (Unrecognized tabular template -> Extra alignment tab). The
26  // document loads xcolor itself; color/definecolor stay via hyperref->color.
27  // See ifacconf_cls.rs and SYNC_STATUS (eager-xcolor cluster).
28  RequirePackage!("hyperref");
29  RequirePackage!("url"); // ecai.cls L395: \RequirePackage{url}\urlstyle{rm}
30
31  // --- Reference-linked author/affiliation model (ecai.cls L405-557) ---------
32  // `\author[A,D]{\fnms{First}~\snm{Last}\orcid{..}\thanks{..}}` — one \author
33  // per author, the optional arg a comma list of address labels. `\address[A]{..}`
34  // defines affiliation "A". ecai defers the join to \makeaddresses at frontmatter
35  // time; the kernel's annotation machinery does the same job eagerly: a creator
36  // carries `annotations={A,D}` and each affiliation carries `label={A}`, and
37  // `\lx@annotate@frontmatter` attaches every affiliation to the creator whose
38  // annotations contain its label (identical to elsart_support_core). So author
39  // "First Last" ends up with exactly affiliations A and D — not all of them piled
40  // onto the last author, which is what the generic OmniBus `\author`/`\address`
41  // produced. Witness 2501.13598 (html_feedback#6571: "Broken Affiliations"-class).
42  // The class's earlier versions (ecai.cls 2020) take every name in one `\author`, parted by `\and` (:889
43  // `\def\and{{\enspace \rm and \enspace}}`), each name's affiliation its `\institute{…}` (below):
44  // `\author{Kenneth Skiba\institute{…} \and Matthias Thimm$^1$}`
45  // (2008.02735). So a block parted by a top-level `\and`, or the current class's `\author[A,D]{…}`, is read as a
46  // per-author class's is (`\lx@add@authors@append`); any other block is one `\author` as written, as Perl adds it:
47  // names parted only by `\ ` with a `\\` line of more names (`\author{Martin C. Cooper \ Andreas Herzig \ Faustine
48  // Maffre \\ Fr\'{e}d\'{e}ric Maris \ Pierre R\'{e}gnier\institute{IRIT, …}}`, 1606.03244) read the second line as an
49  // affiliation when split, two authors lost to it.
50  DefMacro!("\\author OptionalSemiverbatim {}", sub[(addresses, names)] {
51    // (an `\and` written, as before, or held in a macro of the document's, which the separator query reads as the
52    // kernel's: `and_by_meaning`, Option C P2-5)
53    let ands = and_runs(names.unlist_ref(), &[T_CS!("\\and")])?;
54    let and_list = !ands.top_level(names.unlist_ref()).is_empty() || ands.inked > 0;
55    let addresses = addresses.unwrap_or_default();
56    Ok(if and_list || !addresses.is_empty() {
57      Invocation!("\\lx@add@authors@append[annotations={#1}]{#2}", vec![Some(addresses), Some(names)])
58    } else {
59      Invocation!("\\lx@add@creator[role=author,annotations={#1}]{#2}", vec![Some(addresses), Some(names)])
60    })
61  });
62  // `\institute{…}` inside `\author` (ecai.cls 2020:450, a `\thanks` printing the institute as the author's
63  // footnote): the affiliation of the author it follows, as the institute it names (1606.03244's "IRIT, University of
64  // Toulouse III, …", 2008.02735). Defined for every class version, as one binding serves them all: the current
65  // ecai.cls comments it out (:559), and its documents do not use it. OmniBus's llncs-style `\institute`, a list of
66  // affiliations numbered for `\inst`, is no ecai command (inside a split author block it left a nameless creator).
67  DefMacro!("\\institute{}", "\\lx@add@affiliation{#1}");
68  DefMacro!(
69    "\\address OptionalSemiverbatim {}",
70    "\\lx@add@contact[label={#1},role=affiliation]{#2}"
71  );
72  // \fnms{first}/\snm{surname} are name-part hints (OmniBus already makes them
73  // identity); keep them printing their argument inside the personname.
74  def_macro_identity("\\fnms{}")?;
75  def_macro_identity("\\snm{}")?;
76
77  // ECAI frontmatter (ecai.cls L1290) — preserve paper ID as note.
78  DefMacro!(
79    "\\paperid{}",
80    "\\@add@frontmatter{ltx:note}[role=paperid]{#1}"
81  );
82  def_macro_noop("\\makepaperid")?;
83  // Deferred frontmatter builders (ecai.cls L657-697): the kernel emits the
84  // creators/affiliations eagerly, so the class's \make* helpers are no-ops.
85  def_macro_noop("\\makeauthors")?;
86  def_macro_noop("\\makeaddresses")?;
87  def_macro_noop("\\makeorcids")?;
88  // ECAI authors use \orcid{<id>} inside \author{…} for the ORCID identifier
89  // (ecai.cls renders it as \url{https://orcid.org/<id>} in the author block).
90  // Route it through the kernel's `\lx@add@orcid`, which annotates the current
91  // creator with an `ltx:contact[role=orcid]` — the XSLT then renders it as a
92  // clickable `https://orcid.org/<id>` link (LaTeXML-structure-xhtml.xsl:767),
93  // the same mechanism revtex4/quantumarticle use. A bare `ltx:note[role=orcid]`
94  // only produced a dagger footnote with the raw number (html_feedback#6571).
95  // Witness 2501.13598 (#6571), 2501.02040 + 3 ecai papers.
96  DefMacro!("\\orcid{}", "\\lx@add@orcid{#1}");
97  // {ack} environment — acknowledgments block. Emit as structural
98  // ltx:acknowledgements (vs flattening into a generic section).
99  // Witness 2408.16081.
100  DefEnvironment!("{ack}", "<ltx:acknowledgements>#body</ltx:acknowledgements>",
101    mode => "internal_vertical");
102  // \ecaisubmission — page-numbering toggle for submission mode. No-op
103  // (ecai.cls L1100-ish flips internal `\if@ecai@subm` then issues
104  // `\pagenumbering{arabic}\setcounter{page}{1}`). The visible effect
105  // is page numbers in print; in HTML the page concept is meaningless.
106  // Witness 2305.13804.
107  def_macro_noop("\\ecaisubmission")?;
108});
