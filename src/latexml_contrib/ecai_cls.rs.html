1//! ecai.cls (ECAI / EurAI conference class).
2//!
3//! Covers the class's HTML-relevant frontmatter model (ecai.cls L392-851): its
4//! reference-linked author/affiliation scheme, where `\author[<addr-refs>]{name}`
5//! names each author and `\address[<label>]{text}` defines an affiliation, the
6//! two connected by the shared label (like elsarticle). The rest of the class is
7//! print layout (margins, fonts, two-column, page styles) with no HTML meaning
8//! and is left to OmniBus/the kernel.
9use latexml_package::prelude::*;
10
11LoadDefinitions!({
12  LoadClass!("OmniBus");
13  RequirePackage!("amsmath");
14  RequirePackage!("amsthm");
15  // Eager xcolor preload removed for Perl parity: it makes a later document
16  // xcolor[table] load a no-op, so colortbl/array never load and array m{}/b{}
17  // columns break (Unrecognized tabular template -> Extra alignment tab). The
18  // document loads xcolor itself; color/definecolor stay via hyperref->color.
19  // See ifacconf_cls.rs and SYNC_STATUS (eager-xcolor cluster).
20  RequirePackage!("hyperref");
21  RequirePackage!("url"); // ecai.cls L395: \RequirePackage{url}\urlstyle{rm}
22
23  // --- Reference-linked author/affiliation model (ecai.cls L405-557) ---------
24  // `\author[A,D]{\fnms{First}~\snm{Last}\orcid{..}\thanks{..}}` — one \author
25  // per author, the optional arg a comma list of address labels. `\address[A]{..}`
26  // defines affiliation "A". ecai defers the join to \makeaddresses at frontmatter
27  // time; the kernel's annotation machinery does the same job eagerly: a creator
28  // carries `annotations={A,D}` and each affiliation carries `label={A}`, and
29  // `\lx@annotate@frontmatter` attaches every affiliation to the creator whose
30  // annotations contain its label (identical to elsart_support_core). So author
31  // "First Last" ends up with exactly affiliations A and D — not all of them piled
32  // onto the last author, which is what the generic OmniBus `\author`/`\address`
33  // produced. Witness 2501.13598 (html_feedback#6571: "Broken Affiliations"-class).
34  DefMacro!(
35    "\\author OptionalSemiverbatim {}",
36    "\\lx@add@creator[role=author,annotations={#1}]{#2}"
37  );
38  DefMacro!(
39    "\\address OptionalSemiverbatim {}",
40    "\\lx@add@contact[label={#1},role=affiliation]{#2}"
41  );
42  // \fnms{first}/\snm{surname} are name-part hints (OmniBus already makes them
43  // identity); keep them printing their argument inside the personname.
44  def_macro_identity("\\fnms{}")?;
45  def_macro_identity("\\snm{}")?;
46
47  // ECAI frontmatter (ecai.cls L1290) — preserve paper ID as note.
48  DefMacro!(
49    "\\paperid{}",
50    "\\@add@frontmatter{ltx:note}[role=paperid]{#1}"
51  );
52  def_macro_noop("\\makepaperid")?;
53  // Deferred frontmatter builders (ecai.cls L657-697): the kernel emits the
54  // creators/affiliations eagerly, so the class's \make* helpers are no-ops.
55  def_macro_noop("\\makeauthors")?;
56  def_macro_noop("\\makeaddresses")?;
57  def_macro_noop("\\makeorcids")?;
58  // ECAI authors use \orcid{<id>} inside \author{…} for the ORCID identifier
59  // (ecai.cls renders it as \url{https://orcid.org/<id>} in the author block).
60  // Route it through the kernel's `\lx@add@orcid`, which annotates the current
61  // creator with an `ltx:contact[role=orcid]` — the XSLT then renders it as a
62  // clickable `https://orcid.org/<id>` link (LaTeXML-structure-xhtml.xsl:767),
63  // the same mechanism revtex4/quantumarticle use. A bare `ltx:note[role=orcid]`
64  // only produced a dagger footnote with the raw number (html_feedback#6571).
65  // Witness 2501.13598 (#6571), 2501.02040 + 3 ecai papers.
66  DefMacro!("\\orcid{}", "\\lx@add@orcid{#1}");
67  // {ack} environment — acknowledgments block. Emit as structural
68  // ltx:acknowledgements (vs flattening into a generic section).
69  // Witness 2408.16081.
70  DefEnvironment!("{ack}", "<ltx:acknowledgements>#body</ltx:acknowledgements>",
71    mode => "internal_vertical");
72  // \ecaisubmission — page-numbering toggle for submission mode. No-op
73  // (ecai.cls L1100-ish flips internal `\if@ecai@subm` then issues
74  // `\pagenumbering{arabic}\setcounter{page}{1}`). The visible effect
75  // is page numbers in print; in HTML the page concept is meaningless.
76  // Witness 2305.13804.
77  def_macro_noop("\\ecaisubmission")?;
78});
