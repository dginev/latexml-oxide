1//! Stub for ascmac.sty — a Japanese LaTeX2e add-on (part of `jsclasses`
2//! family) that provides boxed environments such as `{itembox}`,
3//! `{screen}`, `{shadebox}`, `{boxnote}` (TeX Live: platex-tools/ascmac).
4//!
5//! Witness: 2601.09339 (`\usepackage{ascmac, fancybox}` +
6//! `\begin{itembox}[l]{title}` body `\end{itembox}`), chemobabel-en/-ja
7//! (`{screen}` inside a figure). The boxes are block boxes through
8//! `insert_block`; the itembox title is kept as an `ltx:note`.
9use latexml_package::prelude::*;
10
11LoadDefinitions!({
12  // The four boxed environments are BLOCK BOXES: their body goes through the
13  // kernel's `insert_block` (Perl TeX_Box.pool.ltxml insertBlock; as framed.sty
14  // and mdframed do), which picks the container the context admits — a
15  // `<ltx:block>` inside a `<ltx:figure>`, a `<ltx:para>`-wrapped block in the
16  // flow, an inline-block in a picture. The earlier fixed
17  // `<ltx:para><ltx:p>#body</ltx:p></ltx:para>` was schema-invalid whenever the
18  // box sat in a figure (chemobabel-en/-ja: `\begin{figure}…\begin{screen}`;
19  // RUST-ONLY — Perl runs ascmac.sty raw and the content flows as figure
20  // panels). Guard: `perfect_kernel_batch56::ascmac_screen_inside_a_figure_is_a_block`.
21  // ascmac.sty: `screen` = rounded frame, `shadebox` = shadowed frame,
22  // `boxnote` = note frame, `itembox[align]{title}` = frame with a title.
23  DefEnvironment!(
24    "{itembox}[]{}",
25    sub[document, args, props] {
26      document.maybe_close_element("ltx:p")?;
27      if let Some(Stored::Digested(body)) = props.get("body") {
28        let mut attrs: HashMap<String, String> = HashMap::default();
29        attrs.insert("framed".to_string(), "rectangle".to_string());
30        attrs.insert("class".to_string(), "ltx_ascmac_itembox".to_string());
31        let blocks = insert_block(document, body, attrs)?;
32        // The title becomes the FIRST child of the box's content (as the raw
33        // itembox typesets it — a label above the body), so the author's
34        // caption stays with its box and in order. `insert_block` returns the
35        // box's CONTENT nodes (for a text body: the one `<p>`), so the note is
36        // opened inside the first of them and moved before whatever is there;
37        // with nothing returned (empty body) it is emitted at the current point.
38        // `args` is 0-based: [0] = the optional alignment, [1] = the title.
39        if let Some(Some(title)) = args.get(1) {
40          let saved = document.get_node().clone();
41          let host = blocks.first().cloned();
42          if let Some(h) = host.as_ref() {
43            document.set_node(h);
44          }
45          let attrs = string_map!("role" => "itembox-title");
46          document.open_element("ltx:note", Some(attrs), None)?;
47          document.absorb(title, None)?;
48          if let (Some(mut note), Some(mut h)) = (document.close_element("ltx:note")?, host) {
49            note.unlink();
50            match h.get_first_child() {
51              Some(mut first) => {
52                first.add_prev_sibling(&mut note)?;
53              },
54              None => {
55                h.add_child(&mut note)?;
56              },
57            }
58          }
59          document.set_node(&saved);
60        }
61      }
62      Ok(())
63    },
64    mode => "internal_vertical"
65  );
66  DefEnvironment!(
67    "{screen}",
68    sub[document, _args, props] {
69      document.maybe_close_element("ltx:p")?;
70      if let Some(Stored::Digested(body)) = props.get("body") {
71        let mut attrs: HashMap<String, String> = HashMap::default();
72        attrs.insert("framed".to_string(), "rectangle".to_string());
73        attrs.insert("class".to_string(), "ltx_ascmac_screen".to_string());
74        insert_block(document, body, attrs)?;
75      }
76      Ok(())
77    },
78    mode => "internal_vertical"
79  );
80  DefEnvironment!(
81    "{boxnote}",
82    sub[document, _args, props] {
83      document.maybe_close_element("ltx:p")?;
84      if let Some(Stored::Digested(body)) = props.get("body") {
85        let mut attrs: HashMap<String, String> = HashMap::default();
86        attrs.insert("framed".to_string(), "rectangle".to_string());
87        attrs.insert("class".to_string(), "ltx_ascmac_boxnote".to_string());
88        insert_block(document, body, attrs)?;
89      }
90      Ok(())
91    },
92    mode => "internal_vertical"
93  );
94  DefEnvironment!(
95    "{shadebox}",
96    sub[document, _args, props] {
97      document.maybe_close_element("ltx:p")?;
98      if let Some(Stored::Digested(body)) = props.get("body") {
99        let mut attrs: HashMap<String, String> = HashMap::default();
100        attrs.insert("framed".to_string(), "rectangle".to_string());
101        attrs.insert("class".to_string(), "ltx_ascmac_shadebox".to_string());
102        insert_block(document, body, attrs)?;
103      }
104      Ok(())
105    },
106    mode => "internal_vertical"
107  );
108});
