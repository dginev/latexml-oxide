1//! Stub for achemso.cls (ACS chemistry journals).
2//!
3//! achemso.cls is an article-derivative for ACS journals. Provides
4//! authorship/affiliation primitives (\affiliation, \alsoaffiliation,
5//! \altaffiliation, \email, \phone, \fax). Gobble for now since we
6//! don't render ACS-style title pages.
7use latexml_package::prelude::*;
8
9LoadDefinitions!({
10  LoadClass!("OmniBus");
11  RequirePackage!("amsmath");
12  RequirePackage!("amsthm");
13  // Eager xcolor preload removed for Perl parity: it makes a later document
14  // xcolor[table] load a no-op, so colortbl/array never load and array m{}/b{}
15  // columns break (Unrecognized tabular template -> Extra alignment tab). The
16  // document loads xcolor itself; color/definecolor stay via hyperref->color.
17  // See ifacconf_cls.rs and SYNC_STATUS (eager-xcolor cluster).
18  RequirePackage!("hyperref");
19  RequirePackage!("natbib");
20  // achemso uses setspace internally; user papers also call
21  // \singlespacing / \doublespacing in preambles. Witness 2503.21357.
22  RequirePackage!("setspace");
23  // achemso.cls L308: `\RequirePackage[margin=2.54cm]{geometry}` and L1306
24  // calls `\geometry{...}` for its own layout. The real class (which Perl
25  // raw-loads) thus has `\geometry` defined; our OmniBus stub must mirror
26  // that so authors' preamble `\geometry{...}` resolves. Layout itself is
27  // moot in the XML/HTML paradigm, but the CS must exist. Witness 2407.02650
28  // (`\geometry{voffset=10pt,...}` → undefined without this; Perl: 0 errors).
29  RequirePackage!("geometry");
30
31  // ACS authorship primitives — preserve author content as ltx:note
32  // frontmatter entries.
33  DefMacro!(
34    "\\affiliation[]{}",
35    "\\@add@frontmatter{ltx:note}[role=affiliation]{#2}"
36  );
37  DefMacro!(
38    "\\alsoaffiliation[]{}",
39    "\\@add@frontmatter{ltx:note}[role=affiliation]{#2}"
40  );
41  DefMacro!(
42    "\\altaffiliation[]{}",
43    "\\@add@frontmatter{ltx:note}[role=affiliation]{#2}"
44  );
45  DefMacro!("\\email{}", "\\@add@frontmatter{ltx:note}[role=email]{#1}");
46  DefMacro!("\\phone{}", "\\@add@frontmatter{ltx:note}[role=phone]{#1}");
47  DefMacro!("\\fax{}", "\\@add@frontmatter{ltx:note}[role=fax]{#1}");
48  DefMacro!(
49    "\\suppinfo{}",
50    "\\@add@frontmatter{ltx:note}[role=suppinfo]{#1}"
51  );
52  DefMacro!(
53    "\\manuscript{}",
54    "\\@add@frontmatter{ltx:note}[role=manuscript]{#1}"
55  );
56  DefMacro!(
57    "\\abbreviations{}",
58    "\\@add@frontmatter{ltx:note}[role=abbreviations]{#1}"
59  );
60  // \acsAuthorList — emit the author-list text inline (no frontmatter slot).
61  DefMacro!("\\acsAuthorList{}", "#1");
62  DefMacro!(
63    "\\notetext{}",
64    "\\@add@frontmatter{ltx:note}[role=notetext]{#1}"
65  );
66  // \acsSection — section opener with text becoming heading.
67  DefMacro!("\\acsSection{}", "\\section*{#1}");
68
69  // {tocentry} environment — the journal's table-of-contents graphic, which
70  // does not belong in the body.
71  //
72  // It must NOT be suppressed with `\iffalse`…`\fi`. Conditional skipping
73  // matches `\fi` by MEANING and expands nothing on the way (TeX's rule, and
74  // ours — `skip_conditional_body` in `conditional.rs`), so an
75  // `\end{tocentry}` whose macro *body* is `\fi` is invisible to it: the skip
76  // ran to end of file and swallowed the rest of the paper, `\bibliography`
77  // included. That one line cost 42 of the 342 residual papers in the
78  // 2605+2606 bibliography-absence cohort their entire bibliography, and
79  // showed up as `Error:expected:\fi \iffalse` pointing at EOF — which reads
80  // like a source defect and is not one. Witnesses 2606.14933, 2605.00451,
81  // 2606.00264; audit family F6.
82  //
83  // Perl never reaches this: it has no achemso binding at all and falls back
84  // to OmniBus (`Warning:missing_file:achemso`), leaving `{tocentry}`
85  // undefined. pdflatex of course renders these papers fine.
86  //
87  // Nor may the body simply be DIGESTED and dropped (`DefEnvironment!
88  // ("{tocentry}", "")`): these graphics are TikZ/`\includegraphics` blocks
89  // that error heavily out of context. Measured on the same cohort, that
90  // turned three papers from "no bibliography" into no OUTPUT AT ALL —
91  // 2606.08929, 2606.12056 and 2606.15422 went from 1 error to 513 and a
92  // fatal (`\lxSVG@endscope`, `undefined:\@startsection`).
93  //
94  // So skip the body as RAW LINES, the way comment.sty's excluded
95  // environments do (`comment_sty.rs`) — no digestion, no conditionals. The
96  // `\end{tocentry}` is consumed as text, so it is deliberately left
97  // undefined.
98  DefConstructor!(T_CS!("\\begin{tocentry}"), None, None,
99  after_digest => {
100    // The rest of the `\begin{tocentry}` line — which may already carry the
101    // matching `\end{tocentry}`, as the one-line form does.
102    if let Some(first) = read_raw_line()
103      && first.contains("\\end{tocentry}")
104    {
105      return Ok(Vec::new());
106    }
107    let mut nlines = 0;
108    while let Some(line) = read_raw_line() {
109      if line.contains("\\end{tocentry}") {
110        break;
111      }
112      nlines += 1;
113    }
114    note_progress(&s!("[Skipped tocentry ({nlines} lines)]"));
115    Ok(Vec::new())
116  });
117
118  // {acknowledgement} — ACS-spelt acknowledgement section.
119  DefEnvironment!(
120    "{acknowledgement}",
121    "<ltx:acknowledgements>#body</ltx:acknowledgements>"
122  );
123
124  // achemso.cls extras commonly hit in ACS papers. The class L1087 sets
125  // a section-numbering policy via `\SectionNumbersOn` (preamble only);
126  // HTML rendering inherits LaTeX's default numbering so the toggle is
127  // a no-op. L294 `\providecommand{\latin}[1]{#1}` is an identity
128  // wrapper for italicized Latin abbreviations. Witness 2312.12737.
129  DefMacro!("\\SectionNumbersOn", None);
130  DefMacro!("\\SectionNumbersOff", None);
131  DefMacro!("\\latin{}", "#1");
132});
