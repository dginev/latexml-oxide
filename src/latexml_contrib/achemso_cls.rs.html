1//! Stub for achemso.cls (ACS chemistry journals).
2//!
3//! achemso.cls is an article-derivative for ACS journals. Provides
4//! authorship/affiliation primitives (\affiliation, \alsoaffiliation,
5//! \altaffiliation, \email, \phone, \fax). Gobble for now since we
6//! don't render ACS-style title pages.
7use latexml_package::prelude::*;

9LoadDefinitions!({
10  LoadClass!("OmniBus");
11  // achemso.cls (TeX Live's) loads no amsmath: amsmath's commands for papers that use them unloaded, its "Old form"
12  // error once the paper loads it; the paper's copy decides (`amsmath_sty::require_amsmath_beyond_original`,
13  // KNOWN_PERL_ERRORS #993).
14  amsmath_sty::require_amsmath_beyond_original(
15    latexml_package::shipped_class::shipped_copy_loads("cls", "amsmath")?,
16    &[],
17  )?;
18  RequirePackage!("amsthm");
19  // Eager xcolor preload removed for Perl parity: it makes a later document
20  // xcolor[table] load a no-op, so colortbl/array never load and array m{}/b{}
21  // columns break (Unrecognized tabular template -> Extra alignment tab). The
22  // document loads xcolor itself; color/definecolor stay via hyperref->color.
23  // See ifacconf_cls.rs and SYNC_STATUS (eager-xcolor cluster).
24  RequirePackage!("hyperref");
25  // achemso.cls:1396-1398 `\RequirePackage[sort&compress,numbers,super]{natbib}` (`round` for a non-superscript
26  // journal): a citation's numbers sorted and compressed ("1–3"; natbib_sty.rs, KNOWN_PERL_ERRORS #1072). The
27  // numeric styles are not passed: the binding's citations stay as before.
28  RequirePackage!("natbib", options => vec![s!("sort&compress")]);
29  // achemso uses setspace internally; user papers also call
30  // \singlespacing / \doublespacing in preambles. Witness 2503.21357.
31  RequirePackage!("setspace");
32  // achemso.cls L308: `\RequirePackage[margin=2.54cm]{geometry}` and L1306
33  // calls `\geometry{...}` for its own layout. The real class (which Perl
34  // raw-loads) thus has `\geometry` defined; our OmniBus stub must mirror
35  // that so authors' preamble `\geometry{...}` resolves. Layout itself is
36  // moot in the XML/HTML paradigm, but the CS must exist. Witness 2407.02650
37  // (`\geometry{voffset=10pt,...}` → undefined without this; Perl: 0 errors).
38  RequirePackage!("geometry");
39  // achemso.cls:311 `\RequirePackage{...,float,...}` and :1022-1030 — the
40  // class's own float types; without them `\begin{scheme}` is undefined and
41  // its `\caption` cascades into "outside any known float" (achemso-demo;
42  // Perl raw-loads the class). Guard:
43  // `perfect_kernel_batch54::achemso_declares_its_scheme_floats`.
44  RequirePackage!("float");
45  // achemso.cls:35 xkeyval and :309-315 the rest of the class's package list, caption, graphicx and url: documents
46  // use `\captionsetup`/`\captionof` without loading caption (witness 2202.05883).
47  RequirePackage!("xkeyval");
48  RequirePackage!("caption");
49  RequirePackage!("graphicx");
50  RequirePackage!("url");
51  // achemso.cls:1413-1414 loads mciteplus when installed (`\mciteSubRef`).
52  RequirePackage!("mciteplus");
53  // achemso.cls:144-165 (its embedded notes2bib): `\bibnote[label]{text}` files
54  // the text as a bibliography entry `Note-N` and cites it; the bibliography
55  // is bibtex's, so the note is rendered here as a numbered note in place
56  // (`\lx@note{bibnote}`), the mark/text halves as the footnote pair;
57  // `\printbibnotes` is the .bbl step (achemso-demo). Guard:
58  // `perfect_kernel_batch54::achemso_bibnote_is_a_numbered_note`.
59  RawTeX!(
60    r"\newcounter{bibnote}\def\thebibnote{Note-\the\value{bibnote}}
61\newcommand*\bibnote[2][\thebibnote]{\stepcounter{bibnote}\lx@note{bibnote}[\thebibnote]{#2}}
62\newcommand*\bibnotemark[1][\thebibnote]{\stepcounter{bibnote}\lx@notemark{bibnote}[\thebibnote]}
63\newcommand*\bibnotetext[2][\thebibnote]{\lx@notetext{bibnote}[\thebibnote]{#2}}
64\newcommand*\printbibnotes{}\def\bibnotetyperefname{note}\def\ext@bibnote{}"
65  );
66  RawTeX!(
67    r"\newfloat{scheme}{htbp}{los}\floatname{scheme}{Scheme}
68\newfloat{chart}{htbp}{loc}\floatname{chart}{Chart}
69\newfloat{graph}{htbp}{loh}\floatname{graph}{Graph}
70\newcommand*\schemename{Scheme}\newcommand*\chartname{Chart}\newcommand*\graphname{Graph}"
71  );

74  // frontmatter entries.
75  DefMacro!(
76    "\\affiliation[]{}",
77    "\\@add@frontmatter{ltx:note}[role=affiliation]{#2}"
78  );
79  DefMacro!(
80    "\\alsoaffiliation[]{}",
81    "\\@add@frontmatter{ltx:note}[role=affiliation]{#2}"
82  );
83  DefMacro!(
84    "\\altaffiliation[]{}",
85    "\\@add@frontmatter{ltx:note}[role=affiliation]{#2}"
86  );
87  // achemso.cls:513-520 `\email` stores the address for the author's footnote (`\@email@<n>`): an address, read as
88  // the other classes' are (`\lx@address@textfont`, revtex4_support_sty.rs), so `ranojoy_bose@apple.com` is no
89  // subscript ("Script _ can only appear in math mode", 1803.03591; OXIDIZED_DESIGN #551). Guard
90  // `perfect_kernel_batch65m::file_arguments_are_not_typeset`.
91  DefMacro!(
92    "\\email Semiverbatim",
93    "\\lx@add@email{\\lx@address@textfont{#1}}"
94  );
95  DefMacro!("\\phone{}", "\\@add@frontmatter{ltx:note}[role=phone]{#1}");
96  DefMacro!("\\fax{}", "\\@add@frontmatter{ltx:note}[role=fax]{#1}");
97  // achemso.cls:1115-1124: `suppinfo` is an environment that opens an unnumbered "Supporting Information Available"
98  // section (`\acs@section*{\suppinfoname}`), not a one-argument frontmatter command — which took the `\begin` of
99  // the environment's first list as its argument (2609.03387, 2609.10829).
100  DefMacro!("\\suppinfoname", "Supporting Information Available");
101  DefMacro!("\\suppinfo", "\\section*{\\suppinfoname}");
102  DefMacro!("\\endsuppinfo", "");
103  DefMacro!(
104    "\\manuscript{}",
105    "\\@add@frontmatter{ltx:note}[role=manuscript]{#1}"
106  );
107  DefMacro!(
108    "\\abbreviations{}",
109    "\\@add@frontmatter{ltx:note}[role=abbreviations]{#1}"
110  );
111  // \acsAuthorList — emit the author-list text inline (no frontmatter slot).
112  DefMacro!("\\acsAuthorList{}", "#1");
113  DefMacro!(
114    "\\notetext{}",
115    "\\@add@frontmatter{ltx:note}[role=notetext]{#1}"
116  );
117  // \acsSection — section opener with text becoming heading.
118  DefMacro!("\\acsSection{}", "\\section*{#1}");

120  // {tocentry} environment — the journal's table-of-contents graphic, which
121  // does not belong in the body.
122  //
123  // It must NOT be suppressed with `\iffalse`…`\fi`. Conditional skipping
124  // matches `\fi` by MEANING and expands nothing on the way (TeX's rule, and
125  // ours — `skip_conditional_body` in `conditional.rs`), so an
126  // `\end{tocentry}` whose macro *body* is `\fi` is invisible to it: the skip
127  // ran to end of file and swallowed the rest of the paper, `\bibliography`
128  // included. That one line cost 42 of the 342 residual papers in the
129  // 2605+2606 bibliography-absence cohort their entire bibliography, and
130  // showed up as `Error:expected:\fi \iffalse` pointing at EOF — which reads
131  // like a source defect and is not one. Witnesses 2606.14933, 2605.00451,
132  // 2606.00264; audit family F6.
133  //
134  // Perl never reaches this: it has no achemso binding at all and falls back
135  // to OmniBus (`Warning:missing_file:achemso`), leaving `{tocentry}`
136  // undefined. pdflatex of course renders these papers fine.
137  //
138  // Nor may the body simply be DIGESTED and dropped (`DefEnvironment!
139  // ("{tocentry}", "")`): these graphics are TikZ/`\includegraphics` blocks
140  // that error heavily out of context. Measured on the same cohort, that
141  // turned three papers from "no bibliography" into no OUTPUT AT ALL —
142  // 2606.08929, 2606.12056 and 2606.15422 went from 1 error to 513 and a
143  // fatal (`\lxSVG@endscope`, `undefined:\@startsection`).
144  //
145  // So skip the body as RAW LINES, the way comment.sty's excluded
146  // environments do (`comment_sty.rs`) — no digestion, no conditionals. The
147  // `\end{tocentry}` is consumed as text, so it is deliberately left
148  // undefined.
149  DefConstructor!(T_CS!("\\begin{tocentry}"), None, None,
150  after_digest => {
151    // The rest of the `\begin{tocentry}` line — which may already carry the
152    // matching `\end{tocentry}`, as the one-line form does.
153    if let Some(first) = read_raw_line()
154      && first.contains("\\end{tocentry}")
155    {
156      return Ok(Vec::new());
157    }
158    let mut nlines = 0;
159    while let Some(line) = read_raw_line() {
160      if line.contains("\\end{tocentry}") {
161        break;
162      }
163      nlines += 1;
164    }
165    note_progress(&s!("[Skipped tocentry ({nlines} lines)]"));
166    Ok(Vec::new())
167  });
168
169  // {acknowledgement} — ACS-spelt acknowledgement section.
170  DefEnvironment!(
171    "{acknowledgement}",
172    "<ltx:acknowledgements>#body</ltx:acknowledgements>"
173  );

175  // achemso.cls extras commonly hit in ACS papers. The class L1087 sets
176  // a section-numbering policy via `\SectionNumbersOn` (preamble only);
177  // HTML rendering inherits LaTeX's default numbering so the toggle is
178  // a no-op. L294 `\providecommand{\latin}[1]{#1}` is an identity
179  // wrapper for italicized Latin abbreviations. Witness 2312.12737.
180  DefMacro!("\\SectionNumbersOn", None);
181  DefMacro!("\\SectionNumbersOff", None);
182  // achemso.cls:1091-1102 `\SectionsOff`/`\SectionsOn` and :1160-1169 `\AbstractOff`/`\AbstractOn` switch the
183  // section headings and the abstract per manuscript type (preamble only); the content is kept either way, so
184  // both are no-ops here. :1266-1269 the TOC graphic's box size, which documents set (`{tocentry}` is skipped).
185  // Witnesses 1710.02414 (`\AbstractOn`), 2006.15922 and 2112.07721 (`\acs@tocentry@height`).
186  DefMacro!("\\SectionsOn", None);
187  DefMacro!("\\SectionsOff", None);
188  DefMacro!("\\AbstractOn", None);
189  DefMacro!("\\AbstractOff", None);
190  RawTeX!(
191    r"\newlength{\acs@tocentry@height}\newlength{\acs@tocentry@width}
192\setlength{\acs@tocentry@height}{1.75in}\setlength{\acs@tocentry@width}{3.25in}"
193  );
194  DefMacro!("\\latin{}", "#1");
195});
