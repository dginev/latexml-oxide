1//! tex-label.sty: the package is loaded raw; `\labels{<keywords>}` sets the token register the page foot prints
2//! (`\rfoot{… \the\labels}`, tex-label.sty:30-32), a page style LaTeXML never typesets (fancyhdr's `\rfoot` is a
3//! no-op here, fancyhdr_sty.rs). The settings become a frontmatter `keywords` entry named "Labels", each page's list
4//! after the last, separated by `;` — its own entry, so an author's `\keywords` stays (a later entry of the same name
5//! replaces the earlier; user ruling 2026-10-03: they are content, not furniture). Witness tex-label/tex-label-demo.
6//! Guard `perfect_kernel_batch61::tex_label_keywords_are_frontmatter`.
7use latexml_package::prelude::*;
8
9LoadDefinitions!({
10  InputDefinitions!("tex-label", noltxml => true, extension => Some(Cow::Borrowed("sty")));
11  RawTeX!(
12    r"\let\lx@texlabel@toks\labels
13\def\labels{\@ifnextchar={\lx@texlabel@eq}{\lx@texlabel@set}}
14\def\lx@texlabel@eq=#1{\lx@texlabel@set{#1}}
15\gdef\lx@texlabel@all{}
16\def\lx@texlabel@set#1{\lx@texlabel@toks{#1}%
17  \ifx\lx@texlabel@all\@empty\gdef\lx@texlabel@all{#1}\else\g@addto@macro\lx@texlabel@all{; #1}\fi
18  \expandafter\lx@texlabel@keywords\expandafter{\lx@texlabel@all}}
19\def\lx@texlabel@keywords#1{\lx@add@frontmatter{ltx:keywords}[name={Labels:~}]{#1}}"
20  );
21});
