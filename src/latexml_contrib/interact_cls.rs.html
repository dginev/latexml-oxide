26  // interact.cls:266-274: `\author{\name{…}\affil{…}}` holds the whole block, and `\name{X}` prints `X\\`, `\affil{Y}`
27  // and `\email{Z}` lines of their own: the names line and the affiliation lines the kernel author parse reads, so a
28  // name list `\name{A\textsuperscript{a}, B\textsuperscript{b}}` is its authors, linked by their marks (Perl maps
29  // `\name` to its argument, one creator; 2112.10522, 2312.11500, 2403.20182). Each later `\name` opens a group of its
30  // own (`\name{A}\affil{X}\name{B}\affil{Y}`), and the class's `\and` prints "and ".
31  // The `\and`s: each written, as before (the class's whatever the document made it: `\author` defines it), and each held
32  // in a macro of the document's, which the separator query reads as the kernel's (`and_by_meaning`, Option C P2-5).
33  DefMacro!("\\author{}", sub[(body)] {
34    let toks = body.unlist();
35    let ands = and_runs(&toks, &[T_CS!("\\and")])?.runs;
36    let mut lines: Vec<Token> = Vec::new();
37    let mut named = false;
38    let mut i = 0;
39    while i < toks.len() {
40      let t = toks[i];
41      i += 1;
42      if let Some(end) = ands.get(&(i - 1)) {
43        i = *end;
44        // (before a later `\name`, which opens a group of its own, it separates nothing more)
45        if toks[i..].iter().find(|u| **u != T_SPACE!()) != Some(&T_CS!("\\name")) {
46          lines.extend(Tokenize!("and").unlist());
47          lines.push(T_SPACE!());
48        }
49        continue;
50      }
51      if t != T_CS!("\\name") && t != T_CS!("\\affil") && t != T_CS!("\\email") {
52        lines.push(t);
53        continue;
54      }
55      if t == T_CS!("\\name") {
56        if named {
57          lines.push(T_CS!("\\and"));
58        }
59        named = true;
60      }
61      while toks.get(i) == Some(&T_SPACE!()) {
62        i += 1;
63      }
64      if toks.get(i).is_none_or(|u| u.get_catcode() != Catcode::BEGIN) {
65        lines.push(t);
66        continue;
67      }
68      let mut depth = 0usize;
69      let start = i;
70      while i < toks.len() {
71        match toks[i].get_catcode() {
72          Catcode::BEGIN => depth += 1,
73          Catcode::END => depth -= 1,
74          _ => {},
75        }
76        i += 1;
77        if depth == 0 {
78          break;
79        }
80      }
81      // (the argument read on its own: an `\affil`'s or `\email`'s is a note's, whose separators the block's query
82      // leaves to the note)
83      let argument = &toks[start + 1..i - 1];
84      let inner = and_runs(argument, &[T_CS!("\\and")])?.runs;
85      let mut k = 0;
86      while k < argument.len() {
87        if let Some(end) = inner.get(&k) {
88          lines.extend(Tokenize!("and").unlist());
89          lines.push(T_SPACE!());
90          k = *end;
91        } else {
92          lines.push(argument[k]);
93          k += 1;
94        }
95      }
96      lines.push(T_CS!("\\\\"));
97    }
98    let mut out = vec![T_CS!("\\gdef"), T_CS!("\\@author"), T_BEGIN!()];
99    out.extend(toks.iter().copied());
100    out.push(T_END!());
101    out.extend(Invocation!(T_CS!("\\lx@add@authors"), vec![Some(Tokens::new(lines))]).unlist());
102    Ok(Tokens::new(out))
103  }, locked => true);
