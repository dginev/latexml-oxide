317/// The marks a cas `\author[marks]` requests, as the letters cas-common.sty:907-914 prints them: a list opening with a
318/// letter reads each item as an entry's label, the letter that entry prints (`\processAffRef`, `\stmRef`, :197-201,
319/// :278-286; the letter when the entry came first, else `\lx@label@ref`, read at the flush once the entries are known,
320/// as cas reads it from the `.aux` of the run before; nothing for a label no entry has, where cas prints "0"), any
321/// other each number N as the N-th letter (`\processAffNum`, `\int_to_alph:n`, :288-296), the letter
322/// the entry `\address[N]` and the N-th lettered entry print (`cas_labelled_entry`). (an item cas cannot read so, a
323/// `*` among letters or a template's `<num>` among numbers, as written)
324fn cas_mark_labels(marks: Option<Tokens>) -> Vec<Token> {
325  let marks = marks.map(Tokens::unlist).unwrap_or_default();
326  let by_label = cas_letter_headed(&marks);
327  let mut out: Vec<Token> = Vec::new();
328  for item in marks.split(|t| *t == T_OTHER!(",")) {
329    let text = Tokens::new(item.to_vec()).to_string();
330    let text = text.trim();
331    if text.is_empty() {
332      continue;
333    }
334    if !out.is_empty() {
335      out.push(T_OTHER!(","));
336    }
337    if by_label && cas_letter_headed(item) {
338      // (the letter, when the entry came first; else read once the entries are known)
339      let key = cas_label_key(text);
340      match label_ref_text(&key) {
341        Some(letter) => out.extend(mouth::tokenize_internal(TeXString::assembled(letter)).unlist()),
342        None => out.extend(label_ref_tokens(&key)),
343      }
344    } else {
345      let letter = (!by_label)
346        .then(|| cas_number(text))
347        .flatten()
348        .filter(|n| *n > 0);
349      let text = letter.map_or_else(|| text.to_string(), radix::radix_alpha);
350      out.extend(mouth::tokenize_internal(TeXString::assembled(text)).unlist());
351    }
352  }
353  out
354}

356/// `\lx@add@contact[label={…},role=affiliation]` for a cas affiliation-list entry (`\affiliation[label]{…}[keys]`,
357/// `\address[label]{…}[keys]`), labelled by the letter it prints, as cas-common.sty:1228-1242 (and :1274-1290) prints
358/// it: a label opening with a letter steps the entry counter and prints its letter, the letter the label stands for
359/// (`\int_gincr:N \g_stm_aff_int`, `\stmLabel`); a number N sets the counter and prints the N-th letter
360/// (`\int_gset:Nn`). So `\address[b]{Univ X}\address[a]{Univ Y}` prints "a Univ X", "b Univ Y", the entries an
361/// `\author[1]` and an `\author[2]` ask for, and `\address[1]{X}\affiliation[a]{…}` prints X and the affiliation as
362/// "a" and "b". An entry the `foot` key sets as a footnote (:1294-1326, :1328-1373) leaves the counter and labels
363/// nothing: a number N prints the N-th letter, a lettered one the letter the entries before it printed last
364/// (`\@currentlabel`, `\lx@label@ref{cas@current}`). (a label cas cannot read so, a `*` or `1a`, is kept as written and leaves
365/// the counter; no label, none)
366fn cas_labelled_entry(label: Option<Tokens>, keys: Option<Tokens>) -> Vec<Token> {
367  let label = label.map(Tokens::unlist).unwrap_or_default();
368  let text = Tokens::new(label.clone()).to_string();
369  let text = text.trim();
370  let lettered = !text.is_empty() && cas_letter_headed(&label);
371  let foot = keys.is_some_and(|keys| cas_foot_key(&keys));
372  let printed = if text.is_empty() {
373    None
374  } else if foot {
375    if lettered {
376      Some(label_ref_tokens(CAS_CURRENT_LABEL))
377    } else {
378      cas_number(text).map(|n| cas_letter_tokens(n, &label))
379    }
380  } else {
381    let counter = if lettered {
382      Some(lookup_int(CAS_ENTRY_COUNTER) + 1)
383    } else {
384      cas_number(text)
385    };
386    counter.map(|counter| {
387      assign_value(CAS_ENTRY_COUNTER, Stored::Int(counter), Some(Scope::Global));
388      let letter = radix::radix_alpha(counter);
389      set_label_ref(CAS_CURRENT_LABEL, &letter);
390      if lettered {
391        // (the first entry under a label keeps it: `\NewLabel` defines `X@<label>:` globally only when it is free,
392        // and LaTeX reads the `.aux` in a group, latex.ltx:9486-9490, so the second's local definition goes)
393        set_label_ref_once(&cas_label_key(text), &letter);
394      }
395      cas_letter_tokens(counter, &label)
396    })
397  };
398  // (a label read as the letter cas prints before the line is one that prints, `_labelprints`: the mark a legend's note
399  // shows, OXIDIZED_DESIGN_DIVERGENCES #961, 2303.06763)
400  let prints = printed.is_some();
401  // (an unlabelled entry under no marked author: cas prints the entry list under the whole row of names, so the
402  // group's one entry is every author's, before it and after it — `\author{Ann}\affiliation{organization={Univ
403  // A}}\author{Bob}\author{Cat}` prints "Ann, Bob and Cat / Univ A", 2201.00009, 2511.22218, 2106.02306, 2606.24054 —;
404  // with two or more the PDF cannot tell whose each is, and each is its run's, the authors since the entry before it,
405  // read by authoring order as elsarticle reads an unlabelled `\address` (`annotate=run`, `\lx@els@shared`):
406  // `\author{Liu}\address{RC}\author{Sun}\author{Zhang}\address{UCAS}` is Liu's RC, Sun's and Zhang's UCAS
407  // (2202.06096). Declared `_share=authors,annotate=run`: the reading shares such an entry only when it is the
408  // only one (`Resolver::sole_shared_run`), OXIDIZED_DESIGN_DIVERGENCES #901. Once an author is marked, the marks link the entries (FIDELITY_CHARTER
409  // §4 R1) and an unlabelled one stays the author's before it, as elsarticle's does)
410  let shared = printed.is_none()
411    && text.is_empty()
412    && !foot
413    && lookup_meaning(&T_CS!("\\lx@cas@labelled")).is_none();
414  let mut out = mouth::tokenize_internal("\\lx@add@contact[label=").unlist();
415  out.push(T_BEGIN!());
416  out.extend(printed.unwrap_or(label));
417  out.push(T_END!());
418  out.extend(
419    mouth::tokenize_internal(if prints {
420      ",role=affiliation,_labelprints=true]"
421    } else if shared {
422      ",role=affiliation,_share={authors},annotate=run]"
423    } else {
424      ",role=affiliation]"
425    })
426    .unlist(),
427  );
428  out
429}

431/// The tokens of the letter `\int_to_alph:n` prints for `number`, else (0, which prints none) `label` as written.
432fn cas_letter_tokens(number: i64, label: &[Token]) -> Vec<Token> {
433  let letter = radix::radix_alpha(number);
434  if letter.is_empty() {
435    label.to_vec()
436  } else {
437    mouth::tokenize_internal(TeXString::assembled(letter)).unlist()
438  }
439}

441/// Whether an entry's `stm/address` keys set it as a footnote: `foot` (a `.bool_gset:N`, cas-common.sty:1180, true
442/// when given no value) set true, the last setting read. Each entry reads its own keys from `foot = false`
443/// (`\__reset_addr:`, :1200).
444fn cas_foot_key(keys: &Tokens) -> bool {
445  let text = keys.to_string();
446  let mut foot = false;
447  for item in text.split(',') {
448    let (key, value) = item.split_once('=').unwrap_or((item, "true"));
449    if key.trim() == "foot" {
450      foot = value.trim().trim_matches(|c| c == '{' || c == '}').trim() == "true";
451    }
452  }
453  foot
454}

456/// Whether a cas label opens with a letter (`\tl_if_head_eq_catcode:nNTF {#1} a`, spaces skipped as the head is read).
457fn cas_letter_headed(label: &[Token]) -> bool {
458  label
459    .iter()
460    .find(|t| **t != T_SPACE!())
461    .is_some_and(|t| t.get_catcode() == Catcode::LETTER)
462}

464/// A cas label written as a number in digits, the value `\int_gset:Nn` and `\int_to_alph:n` read.
465fn cas_number(text: &str) -> Option<i64> {
466  if text.is_empty() || !text.bytes().all(|b| b.is_ascii_digit()) {
467    return None;
468  }
469  text.parse().ok()
