1//! Stub for jgcc.cls (JGCC journal class).
2//!
3//! jgcc.cls L429-450 defines `\jgccdoi{vol}{issue}{paper}{paperid}`
4//! and `\jgccheading{vol}{issue}{year}{pages}{subm}{publ}{rev}`. Same
5//! pattern as LMCS — raw cls preamble (font/page-layout machinery)
6//! fails mid-load, leaving these undefined. Witness 2308.10812,
7//! 2308.02610, 2309.06144.
8//!
9//! The base class is the one jgcc.cls:34 loads, `\LoadClass[11pt,reqno]{amsart}` (OXIDIZED_DESIGN_DIVERGENCES #736):
10//! the paper's copy, read after the binding, finds it loaded and its amsart calls (`\address`, `\email`,
11//! `\copyrightinfo`) meet amsart (2309.03819, 2401.15504).
12use latexml_package::prelude::*;
13
14LoadDefinitions!({
15  LoadClass!("amsart", vec![s!("11pt"), s!("reqno")], Tokens!());
16  RequirePackage!("amsmath");
17  RequirePackage!("amsthm");
18  RequirePackage!("amssymb");
19  RequirePackage!("hyperref");
20
21  // JGCC publication metadata.
22  DefMacro!(
23    "\\jgccdoi{}{}{}{}",
24    "\\@add@frontmatter{ltx:note}[role=jgcc-doi]{Volume #1, Issue #2, Paper #3 (#4)}"
25  );
26  // \jgccheading uses 7-9 args depending on variant; capture the
27  // common 7 explicitly (vol/issue/year/pages/subm/publ/rev).
28  def_macro_noop("\\jgccheading{}{}{}{}{}{}{}")?;
29
30  DefMacro!("\\jgccorcid{}", "\\href{https://orcid.org/#1}{ORCID:#1}");
31
32  DefConditional!("\\ifjgccheadingcalled");
33  def_macro_noop("\\jgccheadingcalledtrue")?;
34  def_macro_noop("\\jgccheadingcalledfalse")?;
35});
