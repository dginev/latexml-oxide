1//! imakeidx.sty — several indexes, each with its own title. Raw, the package writes each index's entries to
2//! `<name>.idx` and `\printindex[name]` inputs `<name>.ind`, the output of an external makeindex run (imakeidx.sty
3//! `\imki@makeindex`, `\imki@printindex`), which never runs here: every named index was lost (biblatex-sbl's author
4//! index, sbl-paper's subject, scripture and author indexes; Perl has no binding, alike). The binding keeps the model
5//! and hands it to the kernel's: `\makeindex[name=<name>,title=<title>]` records the index's title (other keys —
6//! `intoc`, `columns`, `program`, `options`, `noautomatic`, `columnsep`, `columnseprule` — are layout or the external
7//! run); `\index[<name>]{<entry>}` files the entry in list `<name>` (`\lx@index@inlist`, latex_constructs sect11), the
8//! default index (`\jobname`, or no name) being the kernel's `\index`; `\printindex[<name>]` is an `ltx:index` listing
9//! `<name>` under its title, which MakeIndex fills, and the default one the kernel's `{theindex}` under its title.
10//! The raw package is loaded first, so its switches, options and required packages stay. User ruling 2026-10-03 (not a
11//! bibliography-rendering residual). Residual: `\indexprologue` text is not typeset.
12//! Witnesses biblatex-sbl/biblatex-sbl, biblatex-sbl/sbl-paper. Guard `perfect_kernel_batch61::imakeidx_named_indexes`.
13use latexml_package::{
14  engine::latex_constructs::{adjust_backmatter_element, note_backmatter_element},
15  prelude::*,
16};

18/// The `key=value` pairs of a `\makeindex[…]` option list, split at top-level commas.
19fn keyval_pairs(options: &Tokens) -> Vec<(String, Tokens)> {
20  let mut pairs = Vec::new();
21  let mut depth = 0usize;
22  let mut item: Vec<Token> = Vec::new();
23  let flush = |item: &mut Vec<Token>, pairs: &mut Vec<(String, Tokens)>| {
24    let eq = item
25      .iter()
26      .position(|t| t.get_catcode() == Catcode::OTHER && t.to_string() == "=");
27    let (key, value) = match eq {
28      Some(i) => (
29        Tokens::new(item[..i].to_vec()).to_string(),
30        item[i + 1..].to_vec(),
31      ),
32      None => (Tokens::new(item.clone()).to_string(), Vec::new()),
33    };
34    let key = key.trim().to_string();
35    if !key.is_empty() {
36      let mut value = value;
37      while value
38        .first()
39        .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
40      {
41        value.remove(0);
42      }
43      while value
44        .last()
45        .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
46      {
47        value.pop();
48      }
49      if value.len() >= 2
50        && value[0].get_catcode() == Catcode::BEGIN
51        && value[value.len() - 1].get_catcode() == Catcode::END
52      {
53        value = value[1..value.len() - 1].to_vec();
54      }
55      pairs.push((key, Tokens::new(value)));
56    }
57    item.clear();
58  };
59  for t in options.unlist_ref().iter().cloned() {
60    match t.get_catcode() {
61      Catcode::BEGIN => depth += 1,
62      Catcode::END => depth = depth.saturating_sub(1),
63      Catcode::OTHER if depth == 0 && t.to_string() == "," => {
64        flush(&mut item, &mut pairs);
65        continue;
66      },
67      _ => {},
68    }
69    item.push(t);
70  }
71  flush(&mut item, &mut pairs);
72  pairs
73}

75/// The index a `[name]` names: `None` for the default index (no name, or `\jobname`, spelled or expanded).
76fn named_index(name: Tokens) -> Result<Option<String>> {
77  let name = Expand!(name).to_string();
78  let name = name.trim();
79  let jobname = Expand!(T_CS!("\\jobname")).to_string();
80  Ok((!name.is_empty() && name != jobname).then(|| name.to_string()))
81}

83LoadDefinitions!({
84  // The raw package first — its switches, options and the packages it loads (xkeyval, ifxetex/ifluatex, multicol,
85  // imakeidx.sty:23-47, :219), on which documents rely (atableau, tikz-ext-manual, genealogy-profiles) — then the
86  // three commands that reach the external run are replaced. The kernel's `\index` and its `\@index` constructor are
87  // kept across the load: imakeidx.sty:159-160 redefine both for its `.idx` writer.
88  RawTeX!(r"\let\lx@imakeidx@kernel@index\index\let\lx@imakeidx@kernel@@index\@index");
89  InputDefinitions!("imakeidx", noltxml => true, extension => Some(Cow::Borrowed("sty")),
90    handleoptions => true);
91  RawTeX!(r"\let\@index\lx@imakeidx@kernel@@index");
92  DefMacro!("\\makeindex[]", sub[(options)] {
93    let options = options.unwrap_or_default();
94    let pairs = keyval_pairs(&options);
95    let name = pairs.iter().find(|(k, _)| k == "name").map(|(_, v)| v.clone()).unwrap_or_default();
96    let key = named_index(name)?.unwrap_or_default();
97    if let Some((_, title)) = pairs.iter().find(|(k, _)| k == "title") {
98      assign_value(&s!("imakeidx_title:{key}"), Stored::Tokens(title.clone()), Some(Scope::Global));
99    }
100    Ok(TokenizeInternal!(r"\ifdefined\@indexfile\else\csname newwrite\endcsname\@indexfile\fi"))
101  });
102  DefMacro!(
103    "\\index",
104    "\\@ifnextchar[{\\lx@imakeidx@index}{\\lx@imakeidx@kernel@index}"
105  );
106  DefMacro!("\\lx@imakeidx@index[]", sub[(name)] {
107    Ok(match named_index(name.unwrap_or_default())? {
108      Some(list) => {
109        let mut out = vec![T_CS!("\\lx@index@inlist"), T_BEGIN!()];
110        out.extend(ExplodeText!(list));
111        out.push(T_END!());
112        Tokens::new(out)
113      },
114      None => Tokens!(T_CS!("\\lx@imakeidx@kernel@index")),
115    })
116  });
117  DefMacro!("\\printindex[]", sub[(name)] {
118    let list = named_index(name.unwrap_or_default())?;
119    let key = list.clone().unwrap_or_default();
120    let mut out = vec![T_BEGIN!()];
121    if let Some(Stored::Tokens(title)) = lookup_value(&s!("imakeidx_title:{key}")) {
122      out.extend([T_CS!("\\def"), T_CS!("\\indexname"), T_BEGIN!()]);
123      out.extend(title.unlist());
124      out.push(T_END!());
125    }
126    match list {
127      Some(list) => {
128        out.extend([T_CS!("\\lx@imakeidx@print"), T_BEGIN!()]);
129        out.extend(ExplodeText!(list));
130        out.push(T_END!());
131      },
132      None => out.extend(TokenizeInternal!(r"\begin{theindex}\end{theindex}").unlist()),
133    }
134    out.push(T_END!());
135    Ok(Tokens::new(out))
136  });
137  // The list name is read Semiverbatim, so it is its characters under any encoding (an LGR document's `subject` would
138  // read back as Greek and miss the marks' `inlist`, which `\@index` takes from the source tokens).
139  DefConstructor!("\\lx@imakeidx@print Semiverbatim",
140  "<ltx:index xml:id='#id' lists='#list'><ltx:title font='#titlefont' _force_font='true'>#title</ltx:title></ltx:index>",
141  after_digest => sub[whatsit] {
142    note_backmatter_element(whatsit, "ltx:index");
143    let list = whatsit.get_arg(1).map(|a| a.to_string()).unwrap_or_default();
144    // The named index is printed: its deferred entries are typeset (OXIDIZED_DESIGN_DIVERGENCES #738).
145    resolve_deferred_prints(Some(list.trim()))?;
146    let docid: String = Expand!(T_CS!("\\thedocument@ID")).to_string();
147    let name = clean_id(&list);
148    let id = if docid.is_empty() { s!("idx.{name}") } else { s!("{docid}.idx.{name}") };
149    whatsit.set_property("id", id);
150    whatsit.set_property("list", list);
151    if let Some(title) = digest_in_group_if(T_CS!("\\indexname"))? {
152      if let Some(titlefont) = title.get_font()? {
153        whatsit.set_property("titlefont", titlefont);
154      }
155      whatsit.set_property("title", title);
156    }
157  },
158  before_construct => sub[doc, whatsit] {
159    adjust_backmatter_element(doc, whatsit)?;
160  });
161});
