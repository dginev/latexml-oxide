3// gauss.sty (gmatrix with row/column operations). Raw, `\end{gmatrix}` →
4// `\g@endmatrix` (gauss.sty:1733) measures the collected matrix with two
5// recursive `\lastbox` passes whose termination is a physical width
6// (`\g@measureCols`, :1011: `\ifdim\wd\g@trash=100cm` on a sentinel `\hbox to
7// 100cm`) — box geometry no LaTeXML has, so it recurses without end
8// (gauss/gauss-ex, `PushbackLimit`; Perl loops too; pdflatex clean). The
9// measurement only positions arrows LaTeXML does not draw, so the binding
10// renders `gmatrix[d]` as the amsmath matrix its delimiter names (`[p]` →
11// `pmatrix` … `[X]` → the user's `\newmatrix` env) and the `\rowops`/`\colops`
12// operations as a trailing math annotation in the manual's own semantics
13// (`\mult{i}{f}` = R_i ← f R_i, `\add[a]{i}{j}` = R_j ← R_j + a R_i, `\swap{i}
14// {j}` = R_i ↔ R_j; C for column operations; indices 0-based as in the
15// package). Arrow geometry and label fontifiers are accepted and discarded.
16// Guard: `perfect_kernel_batch54::gauss_gmatrix_renders_with_operation_lines`.
17#[rustfmt::skip]
18LoadDefinitions!({
19  RequirePackage!("amsmath");
20  // Environment `gmatrix`: renders `gmatrix[d]` natively through the
21  // corresponding amsmath matrix environment (`pmatrix`, `bmatrix`, etc.),
22  // avoiding gullet delimited-scan failures when nested inside outer alignments
23  // (such as `alignat*`). `\rowops` and `\colops` close the inner matrix alignment
24  // if still open, and switch the operation prefix (\lx@gauss@RC) to R or C.
25  DefMacro!("\\gmatrix []", sub[args] {
26    let delim = args.first().filter(|a| !a.is_none()).map(|d| d.to_string()).unwrap_or_default();
27    let env = if delim.is_empty() { "matrix".to_string() } else { format!("{delim}matrix") };
28    // The matrix is the environment, `\begin{pmatrix}`, as gmatrix's `[p]` stands for it: amsmath's `\pmatrix` and
29    // `\matrix` called inside another environment are their Plain-TeX old forms (amsmath.sty:1077-1082
30    // `\matrix@check`, amsmath_sty.rs `\lx@ams@matrix@check`).
31    Ok(TokenizeInternal!(TeXString::assembled(format!(
32      "\\def\\lx@gauss@env{{{env}}}\\let\\lx@gauss@matrix@open\\relax\\begin{{{env}}}"
33    ))))
34  });
35  DefMacro!("\\rowops",
36    "\\ifx\\lx@gauss@matrix@open\\relax\
37       \\expandafter\\end\\expandafter{\\lx@gauss@env}\
38       \\let\\lx@gauss@matrix@open\\undefined\
39     \\fi\
40     \\def\\lx@gauss@RC{R}"
41  );
42  DefMacro!("\\colops",
43    "\\ifx\\lx@gauss@matrix@open\\relax\
44       \\expandafter\\end\\expandafter{\\lx@gauss@env}\
45       \\let\\lx@gauss@matrix@open\\undefined\
46     \\fi\
47     \\def\\lx@gauss@RC{C}"
48  );
49  DefMacro!("\\endgmatrix",
50    "\\ifx\\lx@gauss@matrix@open\\relax\
51       \\expandafter\\end\\expandafter{\\lx@gauss@env}\
52       \\let\\lx@gauss@matrix@open\\undefined\
53     \\fi"
54  );
55  DefMacro!("\\g@matrix", "\\gmatrix[]");
56  DefMacro!("\\endg@matrix", "\\endgmatrix");
57  // The operations, as math annotation fragments (each preceded by `\;`).
58  DefMacro!("\\mult{}{}", "\\;\\lx@gauss@RC_{#1}\\leftarrow #2\\,\\lx@gauss@RC_{#1}");
59  DefMacro!("\\add[][]{}{}", "\\;\\lx@gauss@RC_{#4}\\leftarrow\\lx@gauss@RC_{#4}+#1\\,\\lx@gauss@RC_{#3}");
60  DefMacro!("\\swap[][]{}{}", "\\;\\lx@gauss@RC_{#3}\\leftrightarrow\\lx@gauss@RC_{#4}");
61  DefMacro!("\\lx@gauss@RC", "R");
62  // gauss.sty `\newmatrix{l}{r}{X}`: defines the `Xmatrix` amsmath-style environment
63  // and enables `\begin{gmatrix}[X]`.
64  DefMacro!("\\newmatrix{}{}{}", sub[(l, r, x)] {
65    let (l, r, x) = (l.to_string(), r.to_string(), x.to_string());
66    if x.is_empty() || x == "g" { return Ok(Tokens!()); }
67    Ok(TokenizeInternal!(TeXString::assembled(format!(
68      "\\def\\{x}matrix{{\\lx@ams@matrix{{name={x}matrix,datameaning=matrix,left=\\lx@left{l},right=\\lx@right{r}}}}}\
69       \\def\\end{x}matrix{{\\lx@end@ams@matrix}}\
70       \\newenvironment{{{x}matrix}}{{\\{x}matrix}}{{\\end{x}matrix}}"
71    ))))
72  });
73  // Arrow geometry and label fontifiers: presentational.
74  DefRegister!("\\rowarrowsep" => Dimension::new(0));
75  DefRegister!("\\colarrowsep" => Dimension::new(0));
76  DefRegister!("\\opskip" => Dimension::new(0));
77  DefRegister!("\\labelskip" => Dimension::new(0));
78  DefRegister!("\\rowopminsize" => Dimension::new(0));
79  DefRegister!("\\colopminsize" => Dimension::new(0));
80  DefMacro!("\\rowmultlabel{}", "#1");
81  DefMacro!("\\colmultlabel{}", "#1");
82  DefMacro!("\\rowswapfromlabel{}", "#1");
83  DefMacro!("\\rowswaptolabel{}", "#1");
84  DefMacro!("\\colswapfromlabel{}", "#1");
85  DefMacro!("\\colswaptolabel{}", "#1");
86  DefMacro!("\\rowaddfromlabel{}", "#1");
87  DefMacro!("\\rowaddtolabel{}", "#1");
88  DefMacro!("\\coladdfromlabel{}", "#1");
89  DefMacro!("\\coladdtolabel{}", "#1");
90});
