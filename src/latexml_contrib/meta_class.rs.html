1//! Shared helper: the frontmatter interface common to the "addtolist ML
2//! meta-class" family (`fairmeta` / `selfevolagent` / `openmoss` and future
3//! siblings) — pre-print classes whose `\author`/`\affiliation`/`\contribution`/
4//! `\correspondence`/`\abstract`/`\beginappendix` commands are defined on an
5//! `\addtolist` accumulator in the class BODY. An unknown `.cls` body is not
6//! raw-loaded (OmniBus extracts dependencies only), so every one of those is
7//! `Error:undefined` without a binding. Mirrors the shared-helper pattern of
8//! [`crate::discard_env`].
9use latexml_package::{package::authblk_sty::author_names_a_list, prelude::*};
10
11/// Install the identical, order-independent frontmatter routing shared by every
12/// sibling class: route `\author`/`\affiliation`/`\contribution` through
13/// LaTeXML's author-annotation / contact-label plan (so the superscript marks
14/// link each author to its institutions/notes) and the rest to
15/// `\@add@frontmatter`/`\lx@add@abstract` so they reach `<ltx:document>`
16/// frontmatter. Call once from inside the class's `LoadDefinitions!` block.
17///
18/// The per-class parts stay in each `*_cls.rs`: the (order-sensitive) dependency
19/// list, the colour palette, and the class-specific labeled field — `\metadata`
20/// vs `\checkdata`, and whether its label routes to a `role` attribute
21/// (attribute-safe labels, `fairmeta`) or to note content (arbitrary-markup
22/// labels, `selfevolagent`/`openmoss`).
23pub fn install_meta_class_frontmatter() -> Result<()> {
24  // Accumulator lists → no-ops; the `\@add@frontmatter` sink accumulates.
25  def_macro_noop("\\authorlist")?;
26  def_macro_noop("\\affiliationlist")?;
27  def_macro_noop("\\contributionlist")?;
28  // The class's `\author[mark]{name}` / `\affiliation[mark]{inst}` /
29  // `\contribution[mark]{text}` link authors to institutions/notes by superscript
30  // MARK (`\author[1,2,*]{…}` cites `\affiliation[1]{…}` and `\contribution[*]{…}`).
31  // Route the marks through LaTeXML's author-annotation / contact-label plan — the
32  // authblk.sty.ltxml idiom (`\lx@add@creator[annotations={#1}]` +
33  // `\lx@add@contact[role,annotate,label={#1}]`), whose `relocate_annotations`
34  // step attaches each contact to the creators that cite its mark. Dropping the
35  // mark (the old `\lx@add@author{#2}` + detached `ltx:note`) lost every
36  // author↔institution association (arXiv/html_feedback#1396 and the fairmeta
37  // family #662/#3512/#4707/#4971/#5035/#5466); Perl has no binding and mangles
38  // them under OmniBus. An uncited mark has no target and is dropped. An unmarked
39  // call naming several authors, as authblk's `\author` reads one, is their list
40  // (13 names and their legend in one `\author{}`, 2605.17829).
41  DefMacro!("\\author[]{}", sub[(marks, name)] {
42    let marks = marks.unwrap_or_default();
43    if marks.is_empty() && author_names_a_list(&name)? {
44      return Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(name)]));
45    }
46    Ok(Invocation!("\\lx@add@creator[role=author,annotations={#1}]{#2}", vec![Some(marks), Some(name)]))
47  });
48  // (an unmarked contact is the new authors': `annotate=new` written out, which the queue's reading of a list
49  // above reads, as it reads no conditional)
50  DefMacro!("\\affiliation[]{}", sub[(mark, text)] { meta_contact("affiliation", mark, text) });
51  DefMacro!("\\contribution[]{}", sub[(mark, text)] { meta_contact("contribution", mark, text) });
52  DefMacro!(
53    "\\correspondence{}",
54    "\\@add@frontmatter{ltx:note}[role=correspondence]{#1}"
55  );
56  DefMacro!("\\abstract{}", "\\lx@add@abstract{#1}");
57  DefMacro!("\\email{}", "\\href{mailto:#1}{\\texttt{#1}}");
58  DefMacro!("\\beginappendix", "\\appendix");
59  Ok(())
60}

62/// `\lx@add@contact[role=<role>,annotate=…,label={mark}]{text}`: the contact labelled `mark` (`annotate=1`), or, unmarked,
63/// the new authors' (`annotate=new`).
64fn meta_contact(role: &str, mark: Option<Tokens>, text: Tokens) -> Result<Tokens> {
65  let mark = mark.unwrap_or_default();
66  let annotate = if mark.is_empty() { "new" } else { "1" };
67  let keys = TokenizeInternal!(TeXString::assembled(s!(
68    "role={role},annotate={annotate},label="
69  )));
70  Ok(Invocation!("\\lx@add@contact[#1{#2}]{#3}", vec![
71    Some(keys),
72    Some(mark),
73    Some(text)
74  ]))
75}
