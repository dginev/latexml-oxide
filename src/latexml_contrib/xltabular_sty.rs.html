1use latexml_package::prelude::*;
2
3LoadDefinitions!({
4  // xltabular.sty — tabularx's `X` columns inside a page-breaking longtable.
5  //
6  // The real environment (xltabular.sty L118-135 `\newenvironment{xltabular}[1][x]`)
7  // collects the body through tabularx's trial loop and then runs
8  // `\expandafter\longtable\the\toks@\endlongtable` (L94-96), restoring
9  // `\caption`/`\endhead`/`\endfirsthead`/`\endfoot`/`\endlastfoot` to
10  // longtable's versions first (L86-91). So semantically it IS a longtable
11  // whose column spec may contain `X`; `X` is a global column type once
12  // tabularx is loaded, so longtable's alignment machinery accepts it as is.
13  //
14  // The former alias `\xltabular → \tabularx` left `\caption`/`\endhead`
15  // bound to the class versions: under a binding class that gave `Use of
16  // \caption outside any known float`, and under raw KOMA (tocbasic's expl3
17  // `\caption` reads `\@captype`, xltabular-doc, hvfloat: 36→101 errors) a
18  // fatal cascade. Guard: `perfect_kernel_batch53::xltabular_is_a_longtable`.
19  //
20  // Signature: `\begin{xltabular}[l|r|c|x]{width}[vpos]{cols}` — the leading
21  // optional is longtable's horizontal alignment (L120-128 sets
22  // `\LTleft`/`\LTright`), the width is presentational (the X columns fill
23  // it), the second optional is tabularx's vertical position.
24  RequirePackage!("tabularx");
25  RequirePackage!("longtable");
26  // `X` is tabularx's own column here, as xltabular.sty:28 runs `\TX@newcol` before the preamble is read
27  // (KNOWN_PERL_ERRORS #463).
28  DefMacro!(
29    "\\xltabular[]{}[]{}",
30    r"\let\NC@rewrite@X\lx@tabularx@X\lx@longtable@bindings{#4}\@@longtable[#1]{#4}\lx@begin@alignment"
31  );
32  DefMacro!("\\endxltabular", r"\lx@end@alignment\@end@tabular");
33  // xltabular.sty:19-21 — `\newif\ifXLT@normalPB` plus the two user toggles
34  // that flip it; the boolean only steers longtable's page-break patches
35  // (L103-104), which have no XML counterpart. The binding replaces the raw
36  // .sty, so it must carry the user surface (witness: xltabular-doc, 2
37  // `undefined` errors). Guard: `perfect_kernel_batch54::xltabular_pagebreak_toggles`.
38  RawTeX!(r"\newif\ifXLT@normalPB \XLT@normalPBtrue");
39  DefMacro!("\\normalLTpagebreak", r"\global\XLT@normalPBtrue");
40  DefMacro!("\\specialLTpagebreak", r"\global\XLT@normalPBfalse");
41});
