1//! The author separators of a binding's argument, found by meaning as well as by spelling (Option C step P2-5,
2//! OXIDIZED_DESIGN_DIVERGENCES #995): the gates and the cuts of l_aa, ecai, interact, pasj00, gretsi and lipics part
3//! at each `\and` written in the argument, as before, and at each separator the query (`argument_separators`) reads
4//! meaning the kernel's `\and` — held in a macro of the document's, a `\let` copy —, which the spelling missed. A
5//! written `\and` stays a separator whatever the document made it (trap T8): these classes set their own `\and` where
6//! they print the block ([`and_runs`]), and where they do not (ecai, gretsi) today's cut is kept. P2-4's eptcs `cells`
7//! (`eptcs_cls.rs`) and P2-2's institute lists cut at the kernel meaning alone through the engine's shared
8//! `kernel_separator_runs` (`observer/separators.rs`, P22b); [`and_runs`] is written apart and moves onto it at the
9//! helpers' unification.
10use std::collections::BTreeMap;
11
12use latexml_package::prelude::*;
13
14/// The `\and`s of an argument a binding parts at.
15pub struct AndRuns {
16  /// each separator's own run of the argument, `start → end`: a token spelled as one of the binding's, and a kernel
17  /// separator's tokens where they print nothing else (`FmArgSeparator::span`)
18  pub runs:  BTreeMap<usize, usize>,
19  /// the kernel separators whose tokens print ink too (a macro holding names and the `\and`, `None` spans): no cut by
20  /// tokens parts them, the reading's cut does; a gate counts them
21  pub inked: usize,
22}
23
24impl AndRuns {
25  /// The runs that stand at `tokens`' own brace level (no group holds them), `start → end`.
26  pub fn top_level(&self, tokens: &[Token]) -> BTreeMap<usize, usize> {
27    let mut depth = 0usize;
28    let mut at_top = Vec::with_capacity(tokens.len());
29    for t in tokens {
30      at_top.push(depth == 0);
31      match t.get_catcode() {
32        Catcode::BEGIN => depth += 1,
33        Catcode::END => depth = depth.saturating_sub(1),
34        _ => {},
35      }
36    }
37    self
38      .runs
39      .iter()
40      .filter(|(start, _)| at_top.get(**start).copied().unwrap_or(false))
41      .map(|(start, end)| (*start, *end))
42      .collect()
43  }
44}
45
46/// The `\and`s of `tokens` ([`AndRuns`]): each token spelled as one of `spelled`, whatever it means where the binding
47/// reads it (the parts the binding cut before P2-5). l_aa, interact and pasj00 print their own `\and` where they set the
48/// block, over a document's renewed one (l-aa.sty:412 inside `\institutename`'s group, interact.cls:266
49/// `\def\author#1{\def\and{and }…}`, pasj00.cls:1878 `\def\and{…and…}` in `\maketitle`). lipics (v2019+: a global
50/// `\def\and{\newline}`, lipics-v2021.cls:758), ecai (ecai.cls:889, global) and gretsi (no `\and` of its own) honour a
51/// renewal, so a written renewed `\and` should not cut there: today's spelled cut is kept for them pending the helpers'
52/// unification (Kernel-only, eptcs `cells`' rule; revP25 finding 1). And each separator the query reads meaning the
53/// kernel's `\and`: held in a macro of the document's, a `\let` copy (its own run), which the spelling missed.
54pub fn and_runs(tokens: &[Token], spelled: &[Token]) -> Result<AndRuns> {
55  let mut runs: BTreeMap<usize, usize> = tokens
56    .iter()
57    .enumerate()
58    .filter(|(_, t)| spelled.contains(t))
59    .map(|(at, _)| (at, at + 1))
60    .collect();
61  let mut inked = 0;
62  if let Some(separators) = argument_separators(&Tokens::new(tokens.to_vec()))? {
63    for separator in separators
64      .iter()
65      .filter(|found| found.meaning == FmSepMeaning::Kernel)
66    {
67      match separator.span {
68        Some((start, end)) => {
69          runs.insert(start, end);
70        },
71        None => inked += 1,
72      }
73    }
74  }
75  Ok(AndRuns { runs, inked })
76}
