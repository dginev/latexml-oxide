1//! PRIMEarxiv.sty — a paper-bundled fork of `arxiv.sty` ("based on the
2//! Arxiv style of George Kour", its own first line). It differs from the
3//! upstream file only cosmetically: no `\headeright`/`\undertitle`/
4//! `\shorttitle`, a rule-less header, and `\keywordname` written
5//! `{\bfseries \emph Keywords}` — an unbraced `\emph` that italicises the
6//! single letter `K` (PRIMEarxiv.sty L39). That quirk is reproduced here
7//! rather than corrected: it is what the raw file renders under
8//! `--includestyles` (`<em>K</em><span class="ltx_font_bold">eywords</span>`).
9//!
10//! Same rationale, same configuration gate and same Perl-parity position
11//! as [`crate::arxiv_sty`] — see that module's header. Witness 2605.10111
12//! (`\keywords{...}` at templateArxiv.tex L54); same-host Perl reports the
13//! identical single `undefined:\keywords`.
14use latexml_package::prelude::*;
15
16LoadDefinitions!({
17  if lookup_bool("INCLUDE_STYLES") {
18    // Raw style loading is on: the paper's own PRIMEarxiv.sty is authoritative.
19    InputDefinitions!("PRIMEarxiv", noltxml => true, extension => Some(Cow::Borrowed("sty")));
20  } else {
21    // PRIMEarxiv.sty drops arxiv.sty's `\headeright` / `\undertitle` /
22    // `\shorttitle` (L27-32 of the upstream file), so only the keyword
23    // pair — plus the two `\RequirePackage` dependencies — is ported.
24    RequirePackage!("geometry");
25    RequirePackage!("fancyhdr");
26    crate::arxiv_sty::define_keywords(r"{\bfseries \emph Keywords}")?;
27  }
28});
