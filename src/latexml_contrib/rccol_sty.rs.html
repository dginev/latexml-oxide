1//! rccol.sty — right-centered numeric columns (`R{<int>}{<frac>}`), formatted by fltpoint. No Perl binding.
2use latexml_package::prelude::*;
3
4LoadDefinitions!({
5  // The cell code (`\rc@begin` … `\rc@end`, `\rc@write`, fltpoint's arithmetic) runs raw.
6  InputDefinitions!("rccol", noltxml => true, extension => Some(Cow::Borrowed("sty")));
7  // rccol.sty:46-69 `\NC@rewrite@R` rewrites `R[-][<in>][<out>]{<int>}{<frac>}` by assignments and an `\edef`
8  // (`\rc@digits{…}`, `\def\rc@withsign{…}`, `\edef\rc@rewrite@scratch{\the\@temptokena>{…}c<{…}}`), run by
9  // array.sty's column-type search (array.sty:351-362). The template reader expands a column type and does not run
10  // it, so the rewrite is the `\edef`'s result, as tokens: `>{<in sign><sign box>\rc@begin}c<{\rc@end<out
11  // sign>\rc@write{<int>}{<frac>}}`. Read as raw tokens it was the cell code taken for column letters, and, with
12  // natbib's `\appdef` leaving `\@temptokena` filled, `Fatal:Timeout:PushbackLimit` (1101.1265, 0812.2309,
13  // 2609.37147; repro alignment-bindings/rccol_with_natbib_pushback_loop).
14  DefMacro!("\\NC@rewrite@R OptionalMatch:- [][]{}{}", sub[(minus, input, output, int, frac)] {
15    let mut toks = vec![T_OTHER!(">"), T_BEGIN!()];
16    if let Some(input) = input.as_ref() {
17      // `\rc@digits{\rcDecimalSignInput{#1}}` (rccol.sty:58)
18      toks.push(T_CS!("\\rcDecimalSignInput"));
19      toks.push(T_BEGIN!());
20      toks.extend(input.unlist_ref().iter().copied());
21      toks.push(T_END!());
22    }
23    if minus.is_some() {
24      // `\def\rc@withsign{{\setbox0\hbox{$-$}\hskip\wd0}}` (rccol.sty:52)
25      toks.extend(TokenizeInternal!(r"{\setbox0\hbox{$-$}\hskip\wd0}").unlist());
26    }
27    toks.push(T_CS!("\\rc@begin"));
28    toks.push(T_END!());
29    toks.push(T_LETTER!("c"));
30    toks.push(T_OTHER!("<"));
31    toks.push(T_BEGIN!());
32    toks.push(T_CS!("\\rc@end"));
33    // `\rc@preothertoks{\rcDecimalSignOutput{…}}`: the second optional, else the first (rccol.sty:59-62)
34    if let Some(sign) = output.as_ref().or(input.as_ref()) {
35      toks.push(T_CS!("\\rcDecimalSignOutput"));
36      toks.push(T_BEGIN!());
37      toks.extend(sign.unlist_ref().iter().copied());
38      toks.push(T_END!());
39    }
40    toks.push(T_CS!("\\rc@write"));
41    toks.push(T_BEGIN!());
42    toks.extend(int.unlist_ref().iter().copied());
43    toks.push(T_END!());
44    toks.push(T_BEGIN!());
45    toks.extend(frac.unlist_ref().iter().copied());
46    toks.push(T_END!());
47    toks.push(T_END!());
48    Ok(Tokens::new(toks))
49  });
50});
