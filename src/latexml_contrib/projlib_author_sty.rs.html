1//! projlib-author.sty (ProjLib, TeX Live 2025, 2025/10/21): the package is loaded raw, and the author-block commands it
2//! provides — `\institute`, `\address`, `\curraddr`, `\email` (projlib-author.sty:121-139 `\ProvideDocumentCommand`,
3//! each `\projlib_author_affil_generic:nn`, :98-119) — become the frontmatter: each line the package adds under the
4//! author it follows (`\hook_gput_code:nnn { cmd/@author/after }`, run where `\@maketitle` prints `\@author`) is that
5//! author's affiliation, address, current address or e-mail. Under the raw package the kernel's locked `\maketitle`
6//! never ran the hook, so the lines were lost (2609.02711, colorart.cls: "Laboratoire de Mathématiques Jean Leray …").
7//! Provided only as the package provides them: a command defined before the package loaded stays (an AMS class's
8//! `\address`; the package also ends at once under one, :38 `\cs_if_exist:NT \subjclass { \endinput }`).
9use latexml_package::prelude::*;
10
11/// The commands projlib-author.sty provides and the frontmatter call each line is.
12const PROVIDED: &[(&str, &str)] = &[
13  ("institute", "\\lx@add@affiliation{#1}"),
14  ("address", "\\lx@add@address{#1}"),
15  ("curraddr", "\\lx@add@currentaddress{#1}"),
16  ("email", "\\lx@add@email{#1}"),
17];
18
19LoadDefinitions!({
20  let before: Vec<bool> = PROVIDED
21    .iter()
22    .map(|(name, _)| IsDefined!(&T_CS!(&s!("\\{name}"))))
23    .collect();
24  InputDefinitions!("projlib-author", noltxml => true, extension => Some(Cow::Borrowed("sty")));
25  if !IsDefined!(&T_CS!("\\projlib_author_affil_generic:nn")) {
26    return Ok(());
27  }
28  for ((name, call), defined_before) in PROVIDED.iter().zip(before) {
29    if !defined_before {
30      RawTeX!(&s!(r"\long\def\{name}#1{{{call}}}"));
31    }
32  }
33});
