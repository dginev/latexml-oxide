1//! Paul Taylor's `diagrams` package (diagrams.tex v3.97, 2024-11-20; www.PaulTaylor.EU/diagrams,
2//! CTAN macros/generic/diagrams/taylor). Not in TeX Live (its licence forbids alteration), but in
3//! arXiv's TeX tree, so ~470 arXiv papers build with it (`\diagramstyle` 467, `\newarrow` 441,
4//! `\rTo` 362 papers undefined in run 349). Perl's ar5iv-bindings `diagrams.tex.ltxml` discards
5//! every `{diagram}` as `<ltx:ERROR>{diagram}</ltx:ERROR>`; this binding keeps the diagram, after
6//! the model of amscd's `{CD}` (amscd_sty.rs): a math array of objects and labelled arrows
7//! (OXIDIZED_DESIGN_DIVERGENCES #589). Witnesses: math0001049, math-ph0001006, math0003019,
8//! math-ph0007001, math0010298, math0012073, math0102115, math0104065, math0104175, math0109209,
9//! math0310311, math0404431, cs0204036; plain TeX `\input diagrams`: math0105166; the `amstex`
10//! option: math0104206, math0403203.
11//!
12//! The macro surface is the manual's (manual.tex v3.97, "Typing the diagram", "Labels",
13//! "Parallel arrows", "Defining arrow styles", "Diagonal lines", "Options"), checked against the
14//! use in the 100 witness papers:
15//! - `\begin{diagram}[options] … \end{diagram}` (plain `\diagram … \enddiagram`): cells `&`, rows
16//!   `\\` (the package's `\cr`, with no `[dim]`). In maths it is a matrix of the formula; outside
17//!   maths it is displayed ("it is better to remove such enclosings", manual FAQ "Display
18//!   options"), but in a table cell or an LR box, as with `[inline]`, it stays in the text line;
19//!   `[LaTeXeqno]` numbers it with LaTeX's `equation` counter ("You can put a \label inside").
20//!   The options are read only when `[` follows at once (manual "Options": after
21//!   `\begin{diagram}` and a new line a first cell may begin with a bracket).
22//! - An arrow `\<dir><Name>` for the directions r l u d ru rd lu ld, followed by `[options]`,
23//!   `(x,y)` (the cells a diagonal spans), positional `{f}{g}` (very old versions) and the labels
24//!   `^` above, `_` below, `<` left, `>` right, `~` middle, each a single token or a braced group.
25//!   "For reverse compatibility, above=left and below=right for vertical arrows"; for negative
26//!   gradient diagonals (rd, lu) above=right and below=left. Outside a diagram the source reads
27//!   no span and no `<`/`>` (`\switch@arg`, diagrams.tex:301-306). Horizontal arrows "may also be
28//!   used in text, `$A \rTo^f B$`"; one outside maths is the package's error, as is a vertical
29//!   arrow outside a diagram (in maths or not); a diagonal one outside a diagram raises nothing.
30//! - `\newarrow{Name}{tail}{filler}{middle}{filler}{head}` defines the eight arrows of `Name`
31//!   (diagrams.tex:449, the definitions made at the caller's group level with `\aftergroup`; a tip
32//!   or middle equal to its filler is empty; ignored within maths), and the arrows the source
33//!   declares in its section (24) are predefined, with its synonyms (`\rInto`, `\hLine`, `\NW`, …).
34//! - `\pile{\rTo^f \\ \rTo_g}`: parallel horizontal arrows, a one-column array.
35//! - Layout only, no output: `\diagramstyle[…]` (but its `amstex` loads amscd's `{CD}`),
36//!   `\newarrowhead/tail/middle/filler`, `\newdiagramgrid`, `\HonV`, `\VonH`, `\HmeetV`. The
37//!   pullback corners `\SEpbk` etc. are a visible mark not emitted (open, OD #589;
38//!   math0203179, math0306018).

41/// The eight arrow directions of `\newarrow` (manual "Defining arrow styles").
42const DIRECTIONS: [&str; 8] = ["r", "l", "u", "d", "ru", "rd", "lu", "ld"];

44/// The direction an arrow drawn backwards (a tail that is an arrowhead, `<----`) points.
45fn reverse_direction(dir: &str) -> &'static str {
46  match dir {
47    "r" => "l",
48    "l" => "r",
49    "u" => "d",
50    "d" => "u",
51    "ru" => "ld",
52    "ld" => "ru",
53    "rd" => "lu",
54    _ => "rd",
55  }
56}

58fn direction_index(dir: &str) -> usize { DIRECTIONS.iter().position(|d| *d == dir).unwrap_or(0) }

86/// The glyph `\<dir><Name>` is drawn with, from the five components `\newarrow` was given
87/// (diagrams.tex section (23) names them: heads `>` `vee` `LaTeX` `->` `>>` `->>` `doublevee`
88/// `=>` `harpoon` `triangle` …, tails `|` `mapsto` `C` `hook` `boldhook` `>` `<` `<=` …, fillers
89/// `-` `=` `.` `dash` `3`). A component the package does not know (a document's own
90/// `\newarrowhead`) draws as the plain arrow of its shaft; an arrow of no components
91/// (`\newarrow{Nul}{}{}{}{}{}`, math0310311) draws nothing.
92fn arrow_glyph(dir: &str, tail: &str, filler1: &str, filler2: &str, head: &str) -> &'static str {
93  if [tail, filler1, filler2, head]
94    .iter()
95    .all(|part| part.is_empty())
96  {
97    return "";
98  }
99  let filler = if filler1.is_empty() { filler2 } else { filler1 };
100  let shaft = match filler {
101    "=" => "double",
102    "." => "dotted",
103    "dash" => "dashed",
104    "3" => "triple",
105    _ => "solid",
106  };
107  let tail_kind = match tail {
108    "" => "",
109    "|" | "mapsto" => "bar",
110    "C" | "hook" | "hooka" | "hookb" | "boldhook" | "boldhooka" | "boldhookb" | "sqhook" => "hook",
111    ">" | "vee" | "littlevee" | "boldlittlevee" | "curlyvee" | "LaTeX" | "triangle" => "tail",
112    "<" | "<<" | "<=" | "<-" | "<<-" => "head",
113    _ => "",
114  };
115  let head_kind = match head {
116    "" => "",
117    ">>" | "->>" | "doublevee" | "doubleLaTeX" => "two",
118    "=>" => "double",
119    "harpoon" => "harpoon",
120    "triangle" => "open",
121    _ => "simple",
122  };
123  let (kind, dir) = match (tail_kind, head_kind) {
124    ("head", "") => {
125      let kind = if shaft == "double" || tail == "<=" {
126        "implies"
127      } else if tail.starts_with("<<") {
128        "onto"
129      } else {
130        "to"
131      };
132      (kind, reverse_direction(dir))
133    },
134    ("head", _) => (
135      if shaft == "double" || tail == "<=" || head_kind == "double" {
136        "iff"
137      } else {
138        "both"
139      },
140      dir,
141    ),
142    (_, "") => (
143      match shaft {
144        "double" => "eq",
145        "dotted" => "dots",
146        "dashed" => "dashes",
147        "triple" => "triple",
148        _ => "line",
149      },
150      dir,
151    ),
152    (_, "double") => ("implies", dir),
153    (..) if shaft == "double" => ("implies", dir),
154    (..) if shaft == "triple" => ("tripleto", dir),
155    ("tail", "two") => ("tailonto", dir),
156    ("bar", "two") => ("mapsonto", dir),
157    (_, "two") => ("onto", dir),
158    (_, "harpoon") => ("harpoon", dir),
159    (_, "open") => ("open", dir),
160    ("bar", _) => ("mapsto", dir),
161    ("hook", _) => ("hook", dir),
162    ("tail", _) => ("tail", dir),
163    _ if shaft == "dotted" || shaft == "dashed" => ("dashto", dir),
164    _ => ("to", dir),
165  };
166  kind_glyphs(kind)[direction_index(dir)]
167}

169/// `\newarrow{Name}{tail}{filler}{middle}{filler}{head}` as definitions, one per direction:
170/// `\def\<dir><Name>{\lx@dg@arrow\<dir><Name>{<dir>}{<glyph>}}` (diagrams.tex:449-470).
171fn newarrow_definitions(
172  name: &str,
173  tail: &str,
174  filler1: &str,
175  filler2: &str,
176  head: &str,
177) -> Tokens {
178  // diagrams.tex:450-454: a tail equal to the first filler, a head equal to the second, is empty.
179  let tail = if tail == filler1 { "" } else { tail };
180  let head = if head == filler2 { "" } else { head };
181  let mut tokens = Vec::new();
182  for dir in DIRECTIONS {
183    let cs = T_CS!(&*s!("\\{dir}{name}"));
184    let glyph = arrow_glyph(dir, tail, filler1, filler2, head);
185    tokens.extend([
186      T_CS!("\\def"),
187      cs,
188      T_BEGIN!(),
189      T_CS!("\\lx@dg@arrow"),
190      cs,
191      T_BEGIN!(),
192    ]);
193    tokens.extend(Explode!(dir));
194    tokens.extend([T_END!(), T_BEGIN!()]);
195    tokens.extend(glyph.chars().map(|c| CharToken!(c)));
196    tokens.extend([T_END!(), T_END!()]);
197  }
198  Tokens::new(tokens)
199}

201/// Is `token` the label marker `^` / `_`? The manual ("Labels", footnote): "The sub- and
202/// superscript characters are recognised by their `\catcode`s, so `\sp` and `\sb` will work".
203fn script_catcode(token: &Token) -> Option<Catcode> {
204  let code = match lookup_meaning(token) {
205    Some(Stored::Token(meaning)) => meaning.get_catcode(),
206    _ => token.get_catcode(),
207  };
208  matches!(code, Catcode::SUPER | Catcode::SUB).then_some(code)
209}

211/// Read a delimited group `open … close` after its opener was read; it is kept when `accept` says
212/// the text is the package's syntax, else the tokens go back to the input (`\rTo(\calf.\pi)^c`,
213/// `$f\colon X\rTo [0,1]$` are an object after a text arrow, not a span or options).
214fn read_bracketed(open: Token, close: &str, accept: fn(&str) -> bool) -> Result<bool> {
215  let mut read = vec![open];
216  let mut depth = 0usize;
217  loop {
218    let Some(token) = read_token()? else { break };
219    read.push(token);
220    match token.get_catcode() {
221      Catcode::BEGIN => depth += 1,
222      Catcode::END if depth == 0 => break,
223      Catcode::END => depth -= 1,
224      _ if depth == 0
225        && token.get_catcode() == Catcode::OTHER
226        && token.with_str(|s| s == close) =>
227      {
228        let text = Tokens::new(read[1..read.len() - 1].to_vec()).to_string();
229        if accept(&text) {
230          return Ok(true);
231        }
232        break;
233      },
234      _ => {},
235    }
236    if read.len() > 200 {
237      break;
238    }
239  }
240  unread(Tokens::new(read));
241  Ok(false)
242}

244/// `(x,y)`: two numbers (manual "Diagonal lines"; `\rdTo(5.225,1.9)` in math0104175-era papers).
245fn is_span(text: &str) -> bool {
246  let parts: Vec<&str> = text.split(',').map(str::trim).collect();
247  parts.len() == 2
248    && parts.iter().all(|p| {
249      !p.is_empty()
250        && p
251          .chars()
252          .all(|c| c.is_ascii_digit() || c == '.' || c == '-')
253    })
254}

256/// `[options]`: a key list, which begins with a letter (`[abut]`, `[leftshortfall=.425cm]`).
257fn is_options(text: &str) -> bool {
258  text
259    .trim_start()
260    .chars()
261    .next()
262    .is_some_and(|c| c.is_ascii_alphabetic())
263}

265/// The labels of an arrow: above / below (left / right of a vertical) and middle.
266#[derive(Default)]
267struct Labels {
268  first:  Option<Tokens>,
269  second: Option<Tokens>,
270  mid:    Option<Tokens>,
271}

273/// Read what follows an arrow command (manual "Labels", "Options", "Diagonal lines"). Inside a
274/// diagram that is the source's `\CD@PB` (diagrams.tex:290-299, made `\switch@arg` at :729):
275/// `^`, `_`, `<`, `>`, `~`, `(`, `[`, `.`, a space or a `{`. Outside one it is `\switch@arg`
276/// (:301-306), which reads `^`, `_`, `[`, `.`, `~`, a space or a `{` and leaves a `(`, `<` or
277/// `>` to the formula (`$X\rTo (0,1)$` prints "X → (0, 1)"). A `.` is the dotted-shaft switch
278/// (`\CD@JJ`, :583), read and not drawn. A `{` is a positional label (`\positional@`, :578): the
279/// upper one while that is unread, then the lower one; after a lower label it ends the arrow.
280fn read_arrow_labels(dir: &str, in_diagram: bool) -> Result<Labels> {
281  let mut labels = Labels::default();
282  let mut any_label = false;
283  // For a negative-gradient diagonal `<` is below and `>` above (manual "Labels").
284  let negative = matches!(dir, "rd" | "lu");
285  loop {
286    let mut spaces = Vec::new();
287    let token = loop {
288      match read_token()? {
289        Some(t) if t.get_catcode() == Catcode::SPACE => spaces.push(t),
290        other => break other,
291      }
292    };
293    let Some(token) = token else {
294      unread(Tokens::new(spaces));
295      break;
296    };
297    let other = |text: &str| token.get_catcode() == Catcode::OTHER && token.with_str(|s| s == text);
298    if other(".") {
299      continue;
300    }
301    let slot: Option<u8> = match script_catcode(&token) {
302      Some(Catcode::SUPER) => Some(1),
303      Some(_) => Some(2),
304      None if in_diagram && other("<") => Some(if negative { 2 } else { 1 }),
305      None if in_diagram && other(">") => Some(if negative { 1 } else { 2 }),
306      None if token.get_catcode() == Catcode::ACTIVE && token.with_str(|s| s == "~") => Some(3),
307      None => None,
308    };
309    if let Some(slot) = slot {
310      let place = match slot {
311        1 => &mut labels.first,
312        2 => &mut labels.second,
313        _ => &mut labels.mid,
314      };
315      if place.is_some() {
316        unread_one(token);
317        unread(Tokens::new(spaces));
318        break;
319      }
320      *place = Some(read_arg(ExpansionLevel::Off)?);
321      any_label = true;
322      continue;
323    }
324    if !any_label && token.get_catcode() == Catcode::OTHER {
325      let opener = token.with_str(|s| match s {
326        "(" if in_diagram => Some((")", is_span as fn(&str) -> bool)),
327        "[" => Some(("]", is_options as fn(&str) -> bool)),
328        _ => None,
329      });
330      if let Some((close, accept)) = opener {
331        if read_bracketed(token, close, accept)? {
332          continue;
333        }
334        unread(Tokens::new(spaces));
335        break;
336      }
337    }
338    // Very old versions' positional labels, `\rTo{f}{g}`, `\rTo^f{g}` (manual "Labels").
339    if labels.second.is_none() && token.get_catcode() == Catcode::BEGIN {
340      let group = read_balanced(ExpansionLevel::Off, false, false)?;
341      let place = if labels.first.is_none() {
342        &mut labels.first
343      } else {
344        &mut labels.second
345      };
346      *place = Some(group);
347      any_label = true;
348      continue;
349    }
350    unread_one(token);
351    unread(Tokens::new(spaces));
352    break;
353  }
354  Ok(labels)
355}

357fn braced(tokens: Option<Tokens>) -> Vec<Token> {
358  let mut out = vec![T_BEGIN!()];
359  if let Some(tokens) = tokens {
360    out.extend(tokens.unlist());
361  }
362  out.push(T_END!());
363  out
364}

366#[rustfmt::skip]
367LoadDefinitions!({
368  // diagrams.tex section (25): labels are `\textstyle` in a diagram and `\scriptstyle` in text;
369  // objects are `\displaystyle` (documents `\renewcommand{\labelstyle}`, math0012073).
370  RawTeX!(r"\newif\ifincommdiag
371\def\labelstyle{\ifincommdiag\textstyle\else\scriptstyle\fi}
372\let\objectstyle\displaystyle");
373
374  // `\lx@dg@arrow <cs>{<dir>}{<glyph>}`: an arrow command. Unexpandable, as the source's arrows
375  // begin with `\relax` (`\CD@nB`, `\CD@pB`): the mode is tested when the arrow is digested, not
376  // when an alignment's look-ahead for `\omit` expands it at a cell's start, before the cell's
377  // math begins (`&\rTo` in `align*`, math0404431). A horizontal arrow belongs in a formula
378  // (manual "Typing the diagram": "You still need the dollars"); typed in text it is the
379  // package's error (diagrams.tex `\CD@nB` → `\CD@E`, "Horizontal arrows may be used outside
380  // diagrams, but must still be in maths"), and, where TeX's inserted `$` runs on to the
381  // paragraph's end, the arrow alone is set as a formula (`$`: plain TeX has no `\ensuremath`,
382  // math0105166). A vertical arrow outside a diagram, in maths or text, is the package's other
383  // error (`\CD@pB` sets `\CD@ZD` to `\CD@LK`, diagrams.tex:488, "vertical map ignored outside
384  // diagram", :497-498), with no `$` test; TeX drops the arrow, we keep the author's arrow, flagged
385  // by the Error, as for a horizontal one. A diagonal goes through `\Vector@Map`, which raises
386  // nothing: in text it is set as a formula without an Error.
387  DefPrimitive!("\\lx@dg@arrow Token {}{}", sub[(cs, dir, glyph)] {
388    let in_math = lookup_bool("IN_MATH");
389    match dir.to_string().as_str() {
390      "u" | "d" => {
391        if !if_condition(&T_CS!("\\ifincommdiag"))?.unwrap_or(false) {
392          Error!("unexpected", cs,
393            "Commutative Diagram: vertical map ignored outside diagram");
394        }
395      },
396      "r" | "l" if !in_math => {
397        Error!("unexpected", cs,
398          "Commutative Diagram: \"$\" inserted before horizontal map");
399      },
400      _ => {},
401    }
402    let mut tokens = Vec::new();
403    if !in_math {
404      tokens.push(T_MATH!());
405    }
406    tokens.extend([T_CS!("\\lx@dg@read"), cs, T_BEGIN!()]);
407    tokens.extend(dir.unlist());
408    tokens.extend([T_END!(), T_BEGIN!()]);
409    tokens.extend(glyph.unlist());
410    tokens.extend([T_END!(), T_BEGIN!()]);
411    if !in_math {
412      tokens.push(T_MATH!());
413    }
414    tokens.push(T_END!());
415    unread(Tokens::new(tokens));
416    Ok(Vec::new())
417  });
418  // `\lx@dg@read <cs>{<dir>}{<glyph>}{<close>}`, in maths: the arrow reads its options, span and
419  // labels, then `<close>` (the `$` of a text arrow).
420  DefMacro!("\\lx@dg@read Token {}{}{}", sub[(cs, dir, glyph, close)] {
421    let dir = dir.to_string();
422    let in_diagram = if_condition(&T_CS!("\\ifincommdiag"))?.unwrap_or(false);
423    let labels = read_arrow_labels(&dir, in_diagram)?;
424    let constructor = if matches!(dir.as_str(), "u" | "d") { "\\lx@dg@side" } else { "\\lx@dg@stack" };
425    let mut tokens = vec![T_CS!(constructor), T_BEGIN!()];
426    tokens.push(cs);
427    tokens.extend([T_END!(), T_BEGIN!(), T_CS!("\\lx@dg@glyph"), T_BEGIN!()]);
428    tokens.extend(Explode!(dir));
429    tokens.extend([T_END!(), T_BEGIN!()]);
430    tokens.extend(glyph.unlist());
431    tokens.extend([T_END!(), T_END!()]);
432    tokens.extend(braced(labels.first));
433    tokens.extend(braced(labels.second));
434    tokens.extend(braced(labels.mid));
435    tokens.extend(close.unlist());
436    Ok(Tokens::new(tokens))
437  });
438
439  // The arrow itself, a stretchy operator as amscd's `\lx@amscd@rightarrow`.
440  DefPrimitive!("\\lx@dg@glyph{}{}", sub[(dir, glyph)] {
441    let dir = dir.to_string();
442    let glyph = glyph.to_string();
443    let props = match dir.as_str() {
444      "r" | "l" => stored_map!(
445        "role" => "ARROW",
446        "stretchy" => "true",
447        "class" => "ltx_horizontally_stretchy",
448        "width" => Dimension!("30pt")
449      ),
450      "u" | "d" => stored_map!("role" => "ARROW", "stretchy" => "true"),
451      _ => stored_map!("role" => "ARROW"),
452    };
453    if glyph.is_empty() {
454      return Ok(Vec::new());
455    }
456    let reversion = Tokens::new(glyph.chars().map(|c| CharToken!(c)).collect());
457    Ok(vec![Digested::from(Tbox::new(pin(&glyph), None, None, reversion, props))])
458  });
459
460  // A horizontal or diagonal arrow with its labels above and below, as amscd's
461  // `\lx@amscd@stack`; a middle label (`~`) takes the place above when that is free.
462  DefConstructor!("\\lx@dg@stack Undigested {} ScriptStyle ScriptStyle ScriptStyle",
463    sub[document, args, props] {
464      let scriptpos = props.get("scriptpos").map(|v| v.to_string()).filter(|s| !s.is_empty());
465      let present = |i: usize| args.get(i).and_then(|a| a.as_ref())
466        .filter(|a| !a.is_empty().unwrap_or(true));
467      let op = args.get(1).and_then(|a| a.as_ref());
468      let mut over: Vec<&Digested> = present(2).into_iter().collect();
469      let mut under: Vec<&Digested> = present(3).into_iter().collect();
470      if let Some(mid) = present(4) {

695/// The source of an arrow: the command, then its labels in the package's syntax (`^`/`_` on a
696/// horizontal or diagonal, `<`/`>` on a vertical, `~` in the middle).
697fn revert_arrow(args: &[Option<Digested>], vertical: bool) -> Result<Tokens> {
698  let mut tokens = match args.first() {
699    Some(Some(cs)) => cs.revert()?.unlist(),
700    _ => Vec::new(),
701  };
702  let markers = if vertical {
703    [T_OTHER!("<"), T_OTHER!(">"), T_ACTIVE!('~')]
704  } else {
705    [T_SUPER!(), T_SUB!(), T_ACTIVE!('~')]
706  };
707  for (i, marker) in markers.into_iter().enumerate() {
708    if let Some(Some(label)) = args.get(i + 2)
709      && !label.is_empty().unwrap_or(true)
710    {
711      tokens.push(marker);
712      tokens.push(T_BEGIN!());
713      tokens.extend(label.revert()?.unlist());
714      tokens.push(T_END!());
715    }
716  }
717  Ok(Tokens::new(tokens))
718}
