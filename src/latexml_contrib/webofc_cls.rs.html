1//! webofc.cls (EDP Sciences' Web of Conferences: EPJ, MATEC, E3S, ITM, SHS and BIO Web of Conferences): the papers ship
2//! the class (it is not in TeX Live) and it is read raw, its packages loaded as it loads them. Its title page is the
3//! A&A/Springer `\inst` form, which the shared `inst_support` binding gives aa, llncs and the Springer classes: an
4//! `\author` list split at `\and`, `\inst{n}` naming the `n`-th of the `\institute` list (webofc.cls:434-438, where
5//! `\inst` typesets `$^{#1}$` and the institutes are an `\and` list the title page numbers; read raw, the name kept the
6//! superscripts and every author got every institute: 2301.07438 "Andrea Shindler,1", 1411.7792, 2610.23498). The
7//! e-mail is its author's contact (webofc.cls:440 prints "e-mail: " and the address, inside a `\thanks` or not; a
8//! `\thanks` holding only the e-mail leaves no note, the kernel's `\lx@annotate@frontmatter@now`: 2409.01693), and
9//! `\fnsep`, the comma between two footnote marks (:217, aa's `\fnmsep`), sets nothing before a `\thanks` in the
10//! frontmatter (the kernel's `digest_frontmatter_item`). `\sidecaption` (:824-857) sets the float's body beside its
11//! caption, which is layout: the caption stays the float's `<caption>`, as `\sidecaption` in aa_support and
12//! sv_support (read raw it typeset "Figure 1. …" as text in a `\parbox`; the html_feedback report on 2610.23498). Its
13//! `\@makecaption` (:788-812) prints the label and its period bold, "**Figure 1.** …", the figure's and table's title
14//! format here (`caption_sty::class_makecaption_label`), unless the document loads caption, which replaces it. A paper
15//! whose source lacks the class falls to OmniBus, with the loader's missing-file Warning.
16use latexml_package::prelude::*;
17
18const WEBOFC: crate::shipped_class::RawRead = crate::shipped_class::RawRead {
19  shipped: true,
20  ..crate::shipped_class::RawRead::new("webofc", "cls")
21};
22
23/// This binding's frontmatter conventions (the observer's, OXIDIZED_DESIGN_DIVERGENCES #837), declared once the class is
24/// read: the kernel's, but every `\\` line of the `\author` paragraph is names (`break_parts_names`).
25pub const FRONTMATTER_CONVENTIONS: FmConventions = FmConventions {
26  name: "webofc",
27  binding: "webofc.cls",
28  break_parts_names: true,
29  cite: "webofc.cls:497-552 (the `\\@author` paragraph); OD #993 (1609.01040, 2610.23498)",
30  ..FM_KERNEL_CONVENTIONS
31};
32
33LoadDefinitions!({
34  if !crate::shipped_class::read_raw(&WEBOFC)? {
35    return Ok(());
36  }
37  // (scoped to the class read: a paper lacking webofc.cls is OmniBus's, inst_support's conventions)
38  declare_frontmatter_conventions(&FRONTMATTER_CONVENTIONS);
39  RequirePackage!("inst_support");
40  // (a `\\` line under the names, the template's `\\The SND Collaboration` — 1609.01040, 2610.23498's `\\ for the
41  // NA61/SHINE Collaboration` —, is a line of the author paragraph, webofc.cls:497-552: a creator of its own, as
42  // llncs's split has it, where inst_support's split at `\and` only kept it in the last name — the whole list one
43  // block the reading cuts at its `\and` and, by this binding's `break_parts_names`, at its `\\` (P2-1, OD #993); the
44  // break's `*` and `[…]` are no text)
45  // (and a second `\author` replaces the first, as the kernel's `\author` the class keeps does: webofc.cls defines none,
46  // its `\@parseauthor` reads `\@author`, :431-466 — OXIDIZED_DESIGN #920; inst_support's own `\author` stacks, OD #53)
47  DefMacro!(
48    "\\author[]{}",
49    "\\lx@clear@creators[role=author]\\lx@add@author@split{#2}\\lx@author@trailing"
50  );
51  DefMacro!("\\email Semiverbatim", "\\lx@add@email{#1}");
52  def_macro_noop("\\sidecaption")?;
53  caption_sty::class_makecaption_label("\\bfseries", ". ", true)?;
54});
