1//! atlasdoc.cls — the ATLAS Collaboration paper/note class (author-bundled, very
2//! large; not raw-loaded). A full binding (hundreds of physics-notation macros,
3//! dozens of bundled `atlas*.sty`) is out of scope. This binds only the
4//! FRONT-MATTER macros that otherwise leak as literal text — the reported issue
5//! (witness 2508.20929 → `\AtlasTitle \AtlasAbstract` shown raw, no title/abstract).
6//!
7//! The ATLAS author list is `\input` in the document BODY (a `flushleft`
8//! `{\Large The ATLAS Collaboration}` + `\AtlasOrcid[orcid]{Name}` list), not in
9//! the pre-`\maketitle` frontmatter, so those names render as the collaboration
10//! block in the body rather than as title-page creators; bind `\AtlasOrcid` so
11//! the names show cleanly instead of leaking the macro.
12//!
13//! The base class is the KOMA class atlasdoc.cls loads (OXIDIZED_DESIGN_DIVERGENCES #736), chosen by its options as
14//! the class chooses it (`main_class`); OmniBus, the base this binding had before, is provided over it
15//! ([`provide_over_base_class`]). The paper's copy, read after the binding, finds its `\LoadClass{\ADOC@mainclass}`
16//! loaded and its KOMA calls (`\KOMAoptions`, `\setkomafont`, `\addtokomafont`, `\titlehead`, …) meet KOMA
17//! (2509.01219; census 1605.06035, 1805.01649, 1806.07355, 1807.02012, 1902.05892, 1904.03536).
18use latexml_package::{
19  package::omnibus_cls::provide_over_base_class,
20  prelude::*,
21  shipped_class::{class_code_contains, shipped_class_text},
22};

24/// The class atlasdoc.cls loads, by the version the paper ships (or the latest, with no copy) and the options the
25/// document gave (`setting`: the last `key` or `key=true`/`key=false` wins; `None` when the option is not given). v13
26/// (2111.06712's copy, 2022/05/24, :73-185): `cernpreprint` (or `preprint`), `PAPER`, `CONF` and `PUB` turn `BOOK` and
27/// `REPORT` off; `BOOK` loads scrbook (:176), `REPORT` (`\DeclareBoolOption[true]{REPORT}`, :39) scrreprt (:180),
28/// otherwise scrartcl (:183), `\LoadClass[titlepage, 11pt]{\ADOC@mainclass}` (:382). v3 (1609.03920's, 2017/03/02,
29/// :84-127, :249-254) has no `REPORT`: scrbook for `BOOK`, else scrartcl. v1 (1503.03243's, 2015/02/23, :42, :58-61,
30/// :231-237) scrartcl when `koma` or `atlasstyle` (true by default), else article.
31fn main_class(copy: Option<&str>, setting: &dyn Fn(&str) -> Option<bool>) -> &'static str {
32  let on = |key: &str| setting(key).unwrap_or(false);
33  let declares = |option: &str| copy.is_none_or(|text| class_code_contains(text, option));
34  if copy.is_some_and(|text| class_code_contains(text, "{koma}")) {
35    return if on("koma") || setting("atlasstyle").unwrap_or(true) {
36      "scrartcl"
37    } else {
38      "article"
39    };
40  }
41  let other_kind = ["cernpreprint", "preprint", "PAPER", "CONF", "PUB"]
42    .iter()
43    .any(|kind| on(kind));
44  if on("BOOK") && !other_kind {
45    "scrbook"
46  } else if declares("{REPORT}") && !other_kind && setting("REPORT").unwrap_or(true) {
47    "scrreprt"
48  } else {
49    "scrartcl"
50  }
51}

53LoadDefinitions!({
54  // atlasdoc.cls:52-54 (2207.12246's copy) the font boolean options (`newtx` by default, `txfonts`, `libertine`,
55  // `palatino`) and :516-546 the packages each loads, the math font with them: papers write `\checkmark`, `\upmu`
56  // (23 of run 353 with newtxmath's commands undefined: 2207.12246, 2210.05415, 2304.03053). `<option>=false` keeps the
57  // default.
58  for option in ["txfonts", "libertine", "palatino"] {
59    DeclareOption!(option, {
60      AssignValue!(&s!("atlasdoc@{option}") => true);
61    });
62    DeclareOption!(&s!("{option}=true"), {
63      AssignValue!(&s!("atlasdoc@{option}") => true);
64    });
65  }
66  ProcessOptions!();
67  // The class's kvoptions (`preprint`, `preprint=true`, `preprint = false`, …; the last setting wins).
68  let options = ["\\opt@atlasdoc.cls", "\\@classoptionslist"]
69    .iter()
70    .find_map(|list| {
71      lookup_definition(&T_CS!(list))
72        .ok()
73        .flatten()
74        .and_then(|defn| match defn.get_expansion() {
75          Some(ExpansionBody::Tokens(body)) => Some(body.to_string()),
76          _ => None,
77        })
78    });
79  let setting = |key: &str| -> Option<bool> {
80    options
81      .as_deref()
82      .unwrap_or("")
83      .split(',')
84      .fold(None, |set, option| {
85        let option: String = option.chars().filter(|c| !c.is_whitespace()).collect();
86        match option.split_once('=') {
87          None if option == key => Some(true),
88          Some((name, value)) if name == key => Some(value == "true"),
89          _ => set,
90        }
91      })
92  };
93  let copy = shipped_class_text("atlasdoc")?;
94  LoadClass!(
95    main_class(copy.as_deref(), &setting),
96    vec![s!("titlepage"), s!("11pt")],
97    Tokens!()
98  );
99  provide_over_base_class()?;
100  if lookup_bool("atlasdoc@txfonts") {
101    RequirePackage!("txfonts", options => vec![s!("varg")]);
102    RequirePackage!("fontenc", options => vec![s!("T1")]);
103  } else if lookup_bool("atlasdoc@libertine") {
104    RequirePackage!("libertine");
105    RequirePackage!("fontenc", options => vec![s!("T1")]);
106    RequirePackage!("newtxmath", options => vec![s!("libertine")]);
107  } else if lookup_bool("atlasdoc@palatino") {
108    RequirePackage!("mathpazo");
109    RequirePackage!("helvet", options => vec![s!("scaled")]);
110    RequirePackage!("courier");
111    RequirePackage!("fontenc", options => vec![s!("T1")]);
112  } else {
113    RequirePackage!("fontenc", options => vec![s!("T1")]);
114    RequirePackage!("newtxtext");
115    RequirePackage!("newtxmath");
116  }
117
118  // atlasdoc.cls L296: `\newcommand{\AtlasTitle}[1]{\title{#1}…}`.
119  DefMacro!("\\AtlasTitle{}", "\\title{#1}");
120  // atlasdoc.cls L360: `\AtlasAbstract{#1}` stashes the abstract; surface it as
121  // the document abstract.
122  DefMacro!("\\AtlasAbstract{}", "\\lx@add@abstract{#1}");
123  // atlasdoc.cls L515: `\NewDocumentCommand \AtlasOrcid { o m }` — optional ORCID
124  // + author name. Render the name (the trailing `$^{…}$` affiliation marker in
125  // the source follows naturally); drop the ORCID decoration.
126  DefMacro!("\\AtlasOrcid[]{}", "#2");
127  // atlasdoc.cls (v15, 2504.04805's copy :286-345) the document-data setters, which store what the title page prints
128  // and what the cover pages and `\AtlasPaperWebpage` read; each stores as the class does, the last call winning. At
129  // `\maketitle` (:1074) the class picks one of four title headers (:991-1006) by its options (:82-91): with
130  // `preprint`/`cernpreprint` (64 of 67 sampled papers) the CERN preprint header (:835-856), the journal reference or
131  // else "Submitted to: <journal>", the DOI and the preprint number; otherwise with `HEPDATA` the HepData header
132  // (:866-893), the preprint number and the arXiv id with its category (:878); with `BOOK` (:894ff; `PAPER`, `CONF` and
133  // `PUB` turn it off, :140-192) none of them; otherwise the default header (:746-747), "<\subprepend> <journal>"
134  // ("Journal:" by default). The same, as the frontmatter notes, each prefix the note's label. Run 340: nearly every
135  // atlasdoc paper calls them (`\AtlasRefCode` 470, `\PreprintIdNumber` 147, `\AtlasJournalRef` 32, `\AtlasVersion` 52;
136  // witnesses 2508.20929, 2103.01029, 1904.03536, 2407.10879).
137  RawTeX!(
138    r"\newcommand*{\ADOC@draftversion}{}\newcommand*{\draftversion}[1]{\renewcommand*{\ADOC@draftversion}{#1}}
139\newcommand*{\AtlasVersion}[1]{\renewcommand*{\ADOC@draftversion}{#1}}
140\newcommand*{\ADOC@refcode}{}\newcommand*{\AtlasRefCode}[1]{\renewcommand*{\ADOC@refcode}{#1}}
141\newcommand*{\AtlasPaperWebpage}{https://atlas.web.cern.ch/Atlas/GROUPS/PHYSICS/PAPERS/\ADOC@refcode}
142\newcommand*{\ADOC@atlasnote}{}\newcommand*{\AtlasNote}[1]{\renewcommand*{\ADOC@atlasnote}{#1}}
143\newcommand*{\ADOC@journal}{}
144\newcommand*{\AtlasJournal}[1]{\renewcommand*{\ADOC@journal}{#1}}\def\journal#1{\renewcommand*{\ADOC@journal}{#1}}
145\newcommand*{\ADOC@preprintnumber}{}
146\newcommand*{\PreprintIdNumber}[1]{\renewcommand*{\ADOC@preprintnumber}{#1}}
147\newcommand*{\ADOC@journalref}{}
148\newcommand*{\AtlasJournalRef}[1]{\renewcommand*{\ADOC@journalref}{#1}}
149\newcommand*{\ADOC@doi}{}\newcommand*{\AtlasDOI}[1]{\renewcommand*{\ADOC@doi}{#1}}
150\newif\iflx@atlas@cernpreprint\newif\iflx@atlas@hepdata\newif\iflx@atlas@book
151\def\lx@atlas@titlepage{\iflx@atlas@cernpreprint
152  \ifx\ADOC@journalref\@empty\ifx\ADOC@journal\@empty\else
153    \lx@add@pubnote[role=journal,name={Submitted to:~}]{\ADOC@journal}\fi
154  \else\lx@add@pubnote[role=journalref]{\ADOC@journalref}\fi
155  \ifx\ADOC@doi\@empty\else\lx@add@pubnote[role=doi]{\ADOC@doi}\fi
156  \ifx\ADOC@preprintnumber\@empty\else\lx@add@pubnote[role=preprint,name={}]{\ADOC@preprintnumber}\fi
157\else\iflx@atlas@hepdata
158  \ifx\ADOC@preprintnumber\@empty\else\lx@add@pubnote[role=preprint,name={}]{\ADOC@preprintnumber}\fi
159  \ifx\AtlasArXivId\@empty\else\lx@add@pubnote[role=arxiv]{\AtlasArXivId\ \AtlasArXivCat}\fi
160\else\iflx@atlas@book\else
161  \ifx\ADOC@journal\@empty\else\lx@add@pubnote[role=journal,name={\ADOC@subprepend~}]{\ADOC@journal}\fi
162\fi\fi\fi}
163\newcommand*{\AtlasArXivId}{}\newcommand*{\AtlasArXivURL}{}
164\newcommand*{\arXivId}[1]{\renewcommand*{\AtlasArXivId}{#1}\renewcommand*{\AtlasArXivURL}{https://arXiv.org/abs/#1}}
165\newcommand*{\AtlasArXivCat}{[hep-ex]}
166\newcommand*{\AtlasHEPDataRecord}{}
167\newcommand*{\HepDataRecord}[1]{\renewcommand*{\AtlasHEPDataRecord}{https://www.hepdata.net/record/ins#1}}
168\newcommand*{\HEPDataRecord}[1]{\renewcommand*{\AtlasHEPDataRecord}{https://www.hepdata.net/record/ins#1}}
169\newcommand*{\ADOC@subprepend}{Journal:}\newcommand*{\subprepend}[1]{\renewcommand*{\ADOC@subprepend}{#1}}"
170  );
171  // The options choose the header: `preprint` or `cernpreprint` the CERN preprint one, which also turns `HEPDATA` off
172  // (:82-91).
173  let setting = |key: &str| setting(key).unwrap_or(false);
174  let cern_preprint = setting("preprint") || setting("cernpreprint");
175  if cern_preprint {
176    RawTeX!(r"\lx@atlas@cernpreprinttrue");
177  } else if setting("HEPDATA") {
178    RawTeX!(r"\lx@atlas@hepdatatrue");
179  } else if setting("BOOK") && !["PAPER", "CONF", "PUB"].iter().any(|kind| setting(kind)) {
180    RawTeX!(r"\lx@atlas@booktrue");
181  }
182  AddToMacro!("\\lx@maketitle@body", "\\lx@atlas@titlepage");
183  // :304-305 `\AtlasDate` (the title page's date, `\today` by default) as the document's date.
184  DefMacro!("\\AtlasDate{}", "\\date{#1}");
185  // The cover-page commands (the class's own, :339ff, for the `coverpage` option): the analysis team, editors,
186  // editorial board, supporting notes and comment deadline print on the internal cover page only (run 340: 14 papers).
187  for cover in [
188    "\\AtlasCoverSupportingNote{}{}",
189    "\\AtlasCoverAnalysisTeam{}",
190    "\\AtlasCoverEditor{}{}",
191    "\\AtlasCoverEdBoardMember{}",
192    "\\AtlasCoverReaderMember{}",
193    "\\AtlasCoverEgroupAnalysisTeam{}",
194    "\\AtlasCoverEgroupEditors{}",
195    "\\AtlasCoverEgroupEdBoard{}",
196    "\\AtlasCoverCommentsDeadline{}",
197    "\\AtlasCoverTwikiURL{}",
198  ] {
199    def_macro_noop(cover)?;
200  }
201});
