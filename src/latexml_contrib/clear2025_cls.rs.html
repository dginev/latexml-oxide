1//! clear2025.cls (CLeaR 2025 proceedings, PMLR; author-bundled): `\LoadClass[pmlr]{jmlr}` after its own `anon` and
2//! `final` options, and `\clearauthor`, the review form `\newcommand{\clearauthor}[1]{}` or the final form
3//! `\newcommand{\clearauthor}[1]{\author{#1}}` (2412.17080, 2503.16206). The base class is jmlr, as the class loads it
4//! (OXIDIZED_DESIGN_DIVERGENCES #736): the paper's copy, read after the binding, finds it loaded. `\clearauthor` is the
5//! final form, as colt2024_cls.rs's `\coltauthor`.
6use latexml_package::prelude::*;
7
8LoadDefinitions!({
9  DeclareOption!("anon", None);
10  DeclareOption!("final", None);
11  ProcessOptions!();
12  LoadClass!("jmlr", vec![s!("pmlr")], Tokens!());
13  DefMacro!("\\clearauthor{}", "\\author{#1}");
14});
