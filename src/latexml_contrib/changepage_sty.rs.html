3/// `\ch@ngetext` in LaTeXML's one column. changepage's (changepage.sty:80-88), chngpage's
4/// (chngpage.sty:79-88) and memoir's (memoir.cls:11237-11245), which `\changetext`/`\changepage`
5/// end with, halve `\columnwidth` under `\if@twocolumn`; LaTeXML sets one column, so
6/// `\columnwidth` is the changed `\textwidth` and `\linewidth` follows it (every later
7/// `width=\linewidth` graphic at half width otherwise). OXIDIZED_DESIGN_DIVERGENCES #341.
8pub const ONE_COLUMN_CH_NGETEXT: &str = r"\DeclareRobustCommand{\ch@ngetext}{%
9  \setlength{\@colht}{\textheight}\setlength{\@colroom}{\textheight}%
10  \setlength{\vsize}{\textheight}\setlength{\columnwidth}{\textwidth}%
11  \setlength{\hsize}{\columnwidth}\setlength{\linewidth}{\hsize}}";
12
13LoadDefinitions!({
14  // memoir provides changepage's commands itself (memoir.cls:12216 `\EmulatedPackage{changepage}`,
15  // which a later `\RequirePackage` then skips), and changepage.sty stops under memoir
16  // (changepage.sty:8-11): memoir's `\checkoddpage` and page macros stay. Its `{adjustwidth}`
17  // (memoir.cls:11267-11297) is the same list as changepage's, so the environments below are
18  // defined either way.
19  if !class_loaded("memoir") {
20    // The page-check and page-layout macros are the package's own TeX: `[strict]` (:22),
21    // `\ifoddpage`, the `cp@cntr` counter and `\cp@tempcnt` (:29-31; the dgruyter.sty that arXiv
22    // 2605.13539 and 2605.16066 ship reads it), `\cplabel` (:32), `\checkoddpage` (:59-67; odd, on
23    // LaTeXML's one page).
24    InputDefinitions!("changepage", noltxml => true, extension => Some(Cow::Borrowed("sty")));
25    RawTeX!(ONE_COLUMN_CH_NGETEXT);
26  }
27  // `{adjustwidth}` and `{adjustwidth*}` are lists (:110-139): the list's end closes the paragraph
28  // (`\endtrivlist`'s `\ifhmode\unskip\par`, latex.ltx:15915-15926) and the text after it goes on in
29  // the same logical paragraph (`\@doendpe`, :15939). The body stays in the flow — loaded raw it is
30  // a tagless one-item <itemize>, in which a bibliography is malformed — and its last paragraph
31  // closes before the end. `internal_vertical` keeps `$$` inside a display (witness 2305.09826);
32  // `{adjustwidth*}` is its own environment (:122; witness 2006.09676). The margins are read, not
33  // typeset: digested, `{-0.005\linewidth}` assigned `\linewidth` = 0pt (witness 2605.02723, a
34  // figure at zero width). The ar5iv binding this replaced stubbed all of it (KPE #318); the
35  // transparent body is a divergence from the raw list (OXIDIZED_DESIGN_DIVERGENCES #341).
36  DefEnvironment!("{adjustwidth} Undigested Undigested", "#body",
37    mode => "internal_vertical", before_digest_end => { leave_horizontal()?; });
38  DefEnvironment!("{adjustwidth*} Undigested Undigested", "#body",
39    mode => "internal_vertical", before_digest_end => { leave_horizontal()?; });
40});
