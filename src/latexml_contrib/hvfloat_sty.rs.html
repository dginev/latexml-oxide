5  // hvfloat.sty — captions beside/after/around float objects.
6  //
7  // Raw-impossibility justification (perfect-kernel README protocol): raw
8  // hvfloat BUILDS its floats manually out of (mini)boxes with `\@captype`
9  // set (hvfloat.sty L545+ `\do@hvFloat` box assembly), so `\caption` runs
10  // with NO float element open — our (and Perl's) `^^<ltx:caption>` float-up
11  // finds no legal ancestor and every caption errors
12  // `malformed:ltx:(toc)caption isn't allowed in <ltx:block>` (43 docs, the
13  // largest single malformed cluster of sweep 16; real LaTeX only requires
14  // `\@captype`). The box choreography is pure page layout; the SEMANTIC
15  // content is exactly a float + object + caption + label — map it to the
16  // real environment.
17  //
18  // \hvFloat*?[keys]{figure|table}{object}[shortcap]{caption}{label}
19  // (hvfloat.sty L535-550). Placement keys (capPos, rotation, fullpage…)
20  // are presentational — dropped.
21  // hvfloat.sty's real dependency chain (caption/graphicx/…).
22  RequirePackage!("caption");
23  // hvfloat.sty:45 — documents call atbegshi's `\AtBeginShipoutNext`
24  // without loading it themselves (hvfloat fullpage1s1c/fullpage1s2c:
25  // `Error:undefined:\AtBeginShipoutNext`).
26  RequirePackage!("atbegshi");
27  RequirePackage!("graphicx");
28  // \hvFloat*?[keys]{type}{object}[shortcap]{caption}{label}
29  // (hvfloat.sty L535-550), plus the multiFloat form where each sub-float
30  // arrives as `+{type}{object}[short]{caption}` and ONE trailing {label}
31  // (multi-default2s1c.tex L26-33; the naive `{}` prototype grabbed the
32  // bare `+` as the environment name → `undefined:{+}` ×15 docs).
33  // Placement keys are presentational — dropped.
34  DefMacro!("\\hvFloat OptionalMatch:* []", sub[(_star, _opts)] {
35    let plus = Tokens!(T_OTHER!("+"));
36    let mut out: Vec<Token> = Vec::new();
37    let mut subfloats: Vec<(Tokens, Tokens, Option<Tokens>, Tokens)> = Vec::new();
38    let multi = loop {
39      if read_match(&[&plus])?.is_some() {
40        let ftype = read_arg(ExpansionLevel::Off)?;
41        let obj = read_arg(ExpansionLevel::Off)?;
42        let short = read_optional(None)?;
43        let cap = read_arg(ExpansionLevel::Off)?;
44        subfloats.push((ftype, obj, short, cap));
45      } else if subfloats.is_empty() {
46        break false;
47      } else {
48        break true;
49      }
50    };
51    if !multi {
52      let ftype = read_arg(ExpansionLevel::Off)?;
53      let obj = read_arg(ExpansionLevel::Off)?;
54      let short = read_optional(None)?;
55      let cap = read_arg(ExpansionLevel::Off)?;
56      subfloats.push((ftype, obj, short, cap));
57    }
58    let label = read_arg(ExpansionLevel::Off)?;
59    for (i, (ftype, obj, short, cap)) in subfloats.into_iter().enumerate() {
60      let ftype_str = ftype.to_string();
61      let ftype_name = ftype_str.trim();
62      if ftype_name.is_empty() {
63        // hvfloat.sty L630-636 `\hvFloat@ii`: an EMPTY float type forces
64        // `nonFloat,onlyText` — no float, no counter, no "Figure:" label;
65        // the object is typeset with the caption as plain text beside it
66        // (L780-782 `\ifhv@nonFloat\ifhv@onlyText \hv@longCap`). The former
67        // `\begin{}` emission gave `undefined:{}` and ran the class
68        // `\caption` outside any float (hvfloat.tex L1475 `onlyText=true`
69        // demo; under raw KOMA that ERROR-defined `\@captype` and every later
70        // `\captionof` recursed).
71        out.push(T_CS!("\\par"));
72        out.extend(obj.unlist());
73        out.push(T_CS!("\\par"));
74        out.extend(cap.unlist());
75        if i == 0 && !label.to_string().trim().is_empty() {
76          out.push(T_CS!("\\label"));
77          out.push(T_BEGIN!());
78          out.extend(label.clone().unlist());
79          out.push(T_END!());
80        }
81        out.push(T_CS!("\\par"));
82        continue;
83      }
84      out.extend(Tokenize!(TeXString::assembled(s!(
85        "\\begin{{{0}}}\\centering ", ftype_name
86      ))).unlist());
87      out.extend(obj.unlist());
88      out.extend(Tokenize!(TeXString::assembled("\\caption[".to_string())).unlist());
89      if let Some(sc) = short {
90        out.extend(sc.unlist());
91      }
92      out.push(T_OTHER!("]"));
93      out.push(T_BEGIN!());
94      out.extend(cap.unlist());
95      out.push(T_END!());
96      if i == 0 {
97        out.extend(Tokenize!(TeXString::assembled("\\label".to_string())).unlist());
98        out.push(T_BEGIN!());
99        out.extend(label.clone().unlist());
100        out.push(T_END!());
101      }
102      out.extend(Tokenize!(TeXString::assembled(s!(
103        "\\end{{{0}}}", ftype.to_string().trim()
104      ))).unlist());
105    }
106    Ok(Tokens::new(out))
107  });
108  // Companion setup macros (presentational).
109  def_macro_noop("\\hvFloatSet{}")?;
110  def_macro_noop("\\hvFloatSetDefaults")?;
111  def_macro_noop("\\hvSet{}")?;
112  // hvfloat.sty L309/L328: `\setDefaults` is the real name, `\hvFloatSetDefaults`
113  // its `\let` alias; L19-22 `\fileversion`/`\hvFloatFileVersion`.
114  def_macro_noop("\\setDefaults")?;
115  DefMacro!("\\hvFloatFileVersion", "2.54");
116  // hvfloat.sty L381-400: `\figcaption`/`\tabcaption`/`\tabcaptionbelow` set
117  // `\@captype` in a group and run `\caption` — the `\captionof` shape
118  // (caption.sty binding), which is what our `\caption` needs to find a type.
119  DefMacro!("\\figcaption[]{}", r"\captionof{figure}[#1]{#2}");
120  DefMacro!("\\tabcaption[]{}", r"\captionof{table}[#1]{#2}");
121  DefMacro!("\\tabcaptionbelow[]{}", r"\captionof{table}[#1]{#2}");
122  // Two-object variant \hvFloat with sub-floats is rare in the manuals;
123  // the main form covers the corpus. Registers used by the demos
124  // (hvfloat.sty L86-99 `\newlength`/`\newsavebox`):
125  DefRegister!("\\hvObjectWidth" => Dimension!("0pt"));
126  DefRegister!("\\hvCapWidth" => Dimension!("0pt"));
127  DefRegister!("\\hvWideWidth" => Dimension!("0pt"));
128  DefRegister!("\\hvMaxCapWidth" => Dimension!("0pt"));
129  DefRegister!("\\hvFloatFullWidth" => Dimension!("0pt"));
130  DefRegister!("\\hvMultiFloatSkip" => Dimension!("0pt"));
131  DefRegister!("\\hvNonFloatTopSkip" => Dimension!("0pt"));
132  RawTeX!(r"\newsavebox\hvObjectBox\newsavebox\hvCaptionBox\newsavebox\hvOBox");
133  // hvfloat.sty L24-36: the package options are `\newif` switches
134  // (`\hv@fboxfalse` is used by the manual's demos, L70-71 loads hyperref on
135  // `[hyperref]`); L306-307 `\defhvstyle{name}{keys}` = `\@namedef{hv@name}`
136  // with `\hvDefFloatStyle` as its alias.
137  RawTeX!(r"\newif\ifhv@fbox\newif\ifhv@hyperref\newif\ifhv@nostfloats\newif\ifhv@tugboat\newif\ifhv@forceLeft
138    \def\defhvstyle#1#2{\@namedef{hv@#1}{#2}}\let\hvDefFloatStyle\defhvstyle");
139  DeclareOption!("fbox", "\\hv@fboxtrue");
140  DeclareOption!("hyperref", "\\hv@hyperreftrue");
141  DeclareOption!("nostfloats", "\\hv@nostfloatstrue");
142  DeclareOption!("no-stfloats", "\\hv@nostfloatstrue");
143  ProcessOptions!();
144  RawTeX!(r"\ifhv@hyperref\RequirePackage{hyperref}\fi");
145  // hvfloat.sty L55.
146  RequirePackage!("ifoddpage");
147  // hvfloat.sty L1264-1266: `\newenvironment{hvFloatEnv}[1][\textwidth]
148  // {\minipage{#1}}{\endminipage}` — a plain minipage the manual wraps
149  // `\captionof` demos in (hvfloat.tex L1531; was `undefined:{hvFloatEnv}`).
150  DefMacro!("\\hvFloatEnv[]", r"\minipage{\ifx.#1.\textwidth\else#1\fi}");
151  DefMacro!("\\endhvFloatEnv", r"\endminipage");
