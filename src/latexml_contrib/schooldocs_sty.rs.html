3#[rustfmt::skip]
4LoadDefinitions!({
5  // A registered binding REPLACES the raw file, so the real schooldocs.sty
6  // must be loaded here first: it `\RequirePackage{xcolor}` (:32), defines
7  // `titlecolor` (:100), `\subject`/`\institute`/`\seprule`… — all of which
8  // the former stub skipped (`\definecolor` undefined, schooldocs-examples
9  // 17 errors; Perl raw-loads it clean). The patches below are applied on
10  // top. Guard: `perfect_kernel_batch54::schooldocs_binding_loads_the_real_style`.
11  InputDefinitions!("schooldocs", noltxml => true, extension => Some(Cow::Borrowed("sty")));
12  // schooldocs.sty defines \subtitle only INSIDE \fancypagestyle{exam}{...}
13  // (schooldocs.sty L312-315: `\newcommand\subtitle[1]{\long\def\@subtitle{##1}}`,
14  // a store for its own title code), which Perl discards (fancyhdr.sty.ltxml L56
15  // makes \fancypagestyle a no-op) and which selecting the style runs nothing of
16  // here (fancyhdr_sty.rs stores it as `\ps@exam` to be read, not run). Hoist the
17  // semantic part: route to the standard frontmatter subtitle. Witness:
18  // doc/latex/schooldocs/schooldocs-examples.tex L227-230 (perfect-kernel corpus).
19  DefMacro!("\\subtitle{}", "\\lx@add@subtitle{#1}");
20  // Presentational style hooks (schooldocs.sty \*style family).
21  def_macro_noop("\\titlestyle")?;
22  def_macro_noop("\\subtitlestyle")?;
23  def_macro_noop("\\sectionstyle")?;
24});
