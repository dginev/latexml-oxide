1//! svproc.cls (Springer Proceedings template, sister of svjour/svmult).
2//!
3//! Base = `article` + `sv_support` (Springer support), mirroring Perl's
4//! svjour.cls.ltxml / svmult.cls.ltxml which both
5//! `RequirePackage('sv_support')`. sv_support pulls inst_support + natbib
6//! and provides the Springer frontmatter (`\email`/`\keywords`/`\institute`)
7//! plus `\spnewtheorem` and the standard theorem set — so we do NOT need
8//! OmniBus (its last-resort grab-bag) nor an eager amsthm preload. See
9//! WISDOM #55. The paper's own `\usepackage{amsthm}` (if any) is then the
10//! first real amsthm load, so the `\let\proof\relax` idiom works (witness
11//! 1707.03222 {proof}).
12use latexml_package::prelude::*;
13
14LoadDefinitions!({
15  LoadClass!("article");
16  // svproc.cls has no class binding in Perl, so Perl falls back to OmniBus,
17  // whose `RequirePackage('graphicx')` (OmniBus.cls.ltxml L44 / our
18  // omnibus_cls.rs L46 — "loaded by random cls files 10-to-1") defines
19  // `\includegraphics`. This contrib binding intercepts svproc BEFORE the
20  // OmniBus fallback, so without this the class's figures hit
21  // `undefined:\includegraphics` and cascade (`_` in text mode). Mirror the
22  // OmniBus outcome. Witness 2004.07524 (svproc proceedings with `\includegraphics`).
23  RequirePackage!("graphicx");
24  RequirePackage!("sv_support");
25  // svproc.cls:369-379 (every census copy, 1705.10638's) `\DeclareMathSymbol{\Gamma}{\mathalpha}{letters}{"00}`…: the
26  // capital Greek letters are the math italic's, italic as a Latin letter unless a math alphabet is in force (KNOWN_PERL_ERRORS
27  // #1147).
28  declare_capital_greek("", Some(LettersClass::Variable))?;
29  // svproc.cls:389-404 (every copy of the census's 17): `\vec` is the upright bold `\mbox{\boldmath$\mathbf{#1}$}`, not
30  // the kernel's arrow accent (KNOWN_PERL_ERRORS #1143). Its `vecphys` (bold italic) and `vecarrow` (the accent) options,
31  // :36-37, are not read: this binding declares no options, and no census paper passes them or uses `\vec`.
32  RawTeX!(r"\def\vec#1{\ensuremath{\lx@sv@vec@mathbf{#1}}}");
33  // svproc.cls loads no amsmath (103 copies in a 1/20 arXiv sample): amsmath's commands for papers that use them
34  // unloaded, its "Old form" error once the paper loads it; the paper's copy decides
35  // (`amsmath_sty::require_amsmath_beyond_original`, KNOWN_PERL_ERRORS #993).
36  amsmath_sty::require_amsmath_beyond_original(
37    latexml_package::shipped_class::shipped_copy_loads("cls", "amsmath")?,
38    &[],
39  )?;
40  // Pre-load xcolor with [dvipsnames, table] so a paper's later
41  // `\usepackage[table]{xcolor}` doesn't silently option-clash and
42  // leave colortbl unloaded → `\cellcolor` undefined. svproc.cls
43  // itself does NOT load xcolor (only uses the kernel `\normalcolor`),
44  // so in Perl the user's `[table]{xcolor}` is the first load and
45  // colortbl comes in via the `table` option; matching that outcome
46  // here. Same anti-clash pattern as mnras_cls / quantumarticle_cls.
47  // Witness: 1706.04315, 1804.09301, 1807.04749 ("undefined:\\cellcolor").
48  RequirePackage!("xcolor", options => vec!["dvipsnames".to_string(), "table".to_string()]);
49  RequirePackage!("hyperref");
50  // svproc.cls L1387-1466 inlines the full standard AAS journal-abbreviation
51  // set (`\newcommand*\apj{ApJ}`, `\aj`, `\mnras`, `\sovast`, … — 79 macros,
52  // the same set as aas_macros.sty). Bibliographies generated by AAS/natbib
53  // bst styles emit these directly in `\bibitem` text. Our binding intercepts
54  // svproc.cls, so those inline defs never run; mirror them via aas_macros
55  // (which Perl-faithfully wraps each as `\ref@jnl{…}`). Without this,
56  // `\apj`/`\sovast` are undefined where Perl (which raw-loads the bundled
57  // svproc.cls) is clean. Witness 2110.04152 (`\apj`, `\sovast` in thebibliography).
58  RequirePackage!("aas_macros");
59
60  // svproc.cls L864: \newtoks\tocauthor / \toctitle for TOC entries.
61  // Preserve author content as ltx:note.
62  DefMacro!(
63    "\\tocauthor{}",
64    "\\@add@frontmatter{ltx:note}[role=tocauthor]{#1}"
65  );
66  DefMacro!(
67    "\\toctitle{}",
68    "\\@add@frontmatter{ltx:note}[role=toctitle]{#1}"
69  );
70  // (no `\institute` of its own: sv_support's stands, splitting it at `\and` into numbered affiliations the `\inst{n}`
71  // marks answer, as Perl's — no svproc binding, OmniBus's inst_support `\institute`, inst_support.sty.ltxml:36-38; a
72  // document `<note role=institute>` here held them all in no author: dev20k 2101.04527, 2504.04157, 2206.03859,
73  // 1911.11805, 2409.01900, repro sectioning-frontmatter/svproc_institute_per_author)
74  // (`\inst{N}`, the superscript marker keyed to those affiliations, is sv_support's too)
75  // svproc.cls L240-245 defines \frontmatter / \mainmatter / \backmatter as
76  // page-numbering + \@mainmatter-flag toggles (`\pagenumbering{Roman}` etc.).
77  // All three are pure page-layout — moot in our HTML paradigm — so Perl
78  // (which raw-loads svproc.cls and runs them) produces no visible output for
79  // them. Our binding intercepts svproc.cls, so the raw defs never run;
80  // `\mainmatter` was already stubbed but `\frontmatter`/`\backmatter` were
81  // left undefined where Perl is clean. No-op all three (matching their net
82  // HTML effect). Witness 1902.03320 (`\documentclass{svproc}`, `\frontmatter`).
83  def_macro_noop("\\frontmatter")?;
84  def_macro_noop("\\mainmatter")?;
85  def_macro_noop("\\backmatter")?;
86});
