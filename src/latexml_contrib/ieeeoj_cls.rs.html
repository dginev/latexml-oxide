1//! Stub for IEEEoj.cls (IEEE Open Journal template).
2//!
3//! IEEEoj is IEEEtran-derived but uses authblk-style `\author{}`/`\affil{}` for
4//! the author/affiliation block. It was previously dispatched to
5//! `ieeeaerospace_cls` (IEEEtran + `\acknowledgments`), which omits authblk — so
6//! `\affil` was undefined while Perl defines it. Give IEEEoj its OWN binding:
7//! ieeeaerospace's provisions (IEEEtran base + `\acknowledgments`) PLUS authblk
8//! (which supplies `\affil`/`\affilmark`/`\author`), mirroring the sibling
9//! `ieeeojcsys_cls`. Kept separate from `ieeeaerospace_cls` so the IEEE Aerospace
10//! conference papers (which use IEEEtran's own author style) are unaffected by
11//! authblk's `\author` redefinition. Witness 2203.03906.
12//!
13//! NOTE: the OJ-family siblings IEEEapm/IEEEtai (still on ieeeaerospace) likely
14//! share the same `\affil` gap; route them here too once witnessed.
15use latexml_package::prelude::*;
16
17LoadDefinitions!({
18  LoadClass!("IEEEtran");
19  // IEEEoj.cls:3183-3186 (2309.03038's copy): `\IEEEauthorrefmark{n}` prints the number, IEEEtran's footnote symbols
20  // commented out — the kernel's arabic form of the reference mark (author_macros.rs, OXIDIZED_DESIGN_DIVERGENCES #968).
21  Let!("\\lx@author@refmark@symbol", "\\lx@author@refmark@arabic");
22  // ieeeoj.cls:1809-1827 (2608.19980's copy) `\@makecaption`: a table's caption "TABLE 1. text" on one line, as a
23  // figure's "FIGURE 1. text", in place of IEEEtran's label over the text (ieeetran_cls.rs); caption, when loaded,
24  // replaces it (caption_sty.rs). The names and numbers (:1667, 1867-1869): figures "FIGURE", tables arabic, in place of
25  // IEEEtran's "Fig." and Roman tables. Guard `perfect_kernel_batch78y::ieeeoj_caption_names`.
26  caption_sty::class_makecaption_label("", ". ", false)?;
27  RawTeX!(r"\def\thetable{\@arabic\c@table}\def\figurename{FIGURE}");
28  RequirePackage!("authblk");
29  // Mirror ieeeaerospace_cls: IEEE OJ/Aerospace templates define a no-arg
30  // `\acknowledgments` opening an unnumbered "Acknowledgments" section.
31  DefMacro!("\\acknowledgments", "\\section*{Acknowledgments}");
32  // IEEEoj.cls:3447-3451, 3713-3717, 4877: the class's front-matter setters — the author note and the
33  // corresponding-author line as notes, the dates and the DOI as the first page prints them (:3457-3458 "Received …;
34  // revised …; accepted …; Date of publication …; date of current version …", "Digital Object Identifier 10.1109/…"),
35  // a template's "XX Month, XXXX" placeholders included, as pdflatex prints them (2609.01380, 05811, 11359).
36  // Each setter keeps its last value, as the class's `\gdef` does (the dates' `\lx@add@date` clears its role too).
37  DefMacro!(
38    "\\authornote{}",
39    "\\lx@clear@frontmatter{ltx:note}[role=note]\\lx@add@frontmatter{ltx:note}[role=note]{#1}"
40  );
41  DefMacro!(
42    "\\corresp{}",
43    "\\lx@clear@frontmatter{ltx:note}[role=corresponding]\\lx@add@frontmatter{ltx:note}[role=corresponding]{#1}"
44  );
45  DefMacro!("\\receiveddate{}", "\\lx@add@date[role=received]{#1}");
46  DefMacro!("\\reviseddate{}", "\\lx@add@date[role=revised]{#1}");
47  DefMacro!("\\accepteddate{}", "\\lx@add@date[role=accepted]{#1}");
48  DefMacro!("\\publisheddate{}", "\\lx@add@date[role=published]{#1}");
49  DefMacro!("\\currentdate{}", "\\lx@add@date[role=current]{#1}");
50  // Labelled as the class prints it (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #915): IEEEoj.cls:3458 (2503.03013's copy) "Digital Object
51  // Identifier 10.1109/…" — not the role's "DOI: ".
52  // An empty `\doiinfo{}` (the class's own default) prints no DOI line (IEEEoj.cls:3458 `\ifx\@doiinfo\@empty`; 2609.27083).
53  DefMacro!(
54    "\\doiinfo{}",
55    "\\lx@clear@frontmatter{ltx:pubnote}[role=doi]\\def\\lx@ieee@doi{#1}\\ifx\\lx@ieee@doi\\@empty\\else\\lx@add@pubnote[role=doi,name={Digital Object Identifier~}]{10.1109/#1}\\fi"
56  );
57  // The names the first page prints them with (IEEEoj.cls:3457; Perl looks a date's name up as `\lx@date@<role>@name`).
58  DefMacro!("\\lx@date@published@name", "Date of publication~");
59  DefMacro!("\\lx@date@current@name", "Date of current version~");
60});
