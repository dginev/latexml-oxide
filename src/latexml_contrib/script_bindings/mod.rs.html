1//! Runtime script bindings via embedded **Rhai** (see `docs/parity/script_bindings_plan.md`).
2//!
3//! Feature-gated behind `runtime-bindings` (ON by default and in the distribution
4//! build — downstream single-binary consumers customize without recompiling; the
5//! module keeps its original name, the feature does not). The ONLY module in the
6//! workspace that embeds Rhai; core/engine/package are untouched. Pure-Rust, no
7//! FFI, no ABI — the reliable successor to the abandoned libperl approach.
8//!
9//! ## Model (the "fine seam", Model A)
10//!
11//! A binding is a Rhai script that calls registration functions (`DefMacro`,
12//! `DefConstructor`, …) with the binding body as a Rhai function value. Each
13//! registration installs a **native** latexml-oxide `Definition` whose body
14//! closure trampolines into the Rhai engine. Prototype parsing, argument
15//! reading, the gullet/stomach/document machinery all stay native; Rhai runs
16//! only the body.
17//!
18//! ## Seams implemented
19//!
20//! * **`DefMacro`** (expandable) — body receives Perl's `($gullet, @args)`, the arguments as Tokens, and returns
21//!   Tokens or TeX source (re-tokenized via `mouth::tokenize_internal`, so `\textit{x}` yields a control sequence).
22//! * **`DefConstructor`** (construction) — two forms. A string XML template, or an imperative body
23//!   that reads like the Perl original: it receives a `Document` **proxy** as its first argument
24//!   (Perl's `$_[0]`), each digested argument as an opaque handle, then Perl's `%props` as one map, e.g. `|document, x| {
25//!   document.openElement("ltx:emph"); document.absorb(x); document.closeElement("ltx:emph"); }`.
26//!   The proxy's methods resolve a thread-local **active context** (the live `&mut Document` +
27//!   props, published for the duration of the call; never borrowed across a re-entrant call).
28//!
29//! ## Lifecycle
30//!
31//! `load_script` compiles the script to an `AST`, runs it once to collect the
32//! registrations, wraps engine + AST in `Rc`, and installs one native definition
33//! per registration — each capturing `Rc<Engine>`/`Rc<AST>` so a body stays
34//! callable for the whole conversion (a deferred `FnPtr::call` needs both alive).
35

148/// The logged Fatal a registered function raised, if the script's failure `e` is that Fatal; always taken, so a
149/// Fatal the script caught never stands in for a later, unrelated error (see `RAISED_FATAL`). Matched by text: a
150/// later `throw` that repeats the caught Fatal's whole internal message is taken for it (only a hand-written copy).
151pub(super) fn take_raised_fatal(e: &EvalAltResult) -> Option<Error> {
152  let (fatal, text) = RAISED_FATAL.with(|r| r.borrow_mut().take())?;
153  let raised = match e.unwrap_inner() {
154    EvalAltResult::ErrorRuntime(v, _) => v.is_string() && v.to_string() == text,
155    _ => false,
156  };
157  (raised && fatal.is_logged_fatal()).then_some(fatal)
158}

248/// One `Warning:deprecated:<old>` per binding load for a deprecated alias, naming its replacement; the binding is
249/// the running script.
250pub(super) fn warn_deprecated(old: &'static str) {
251  let script = CURRENT_SCRIPT.with(|c| {
252    c.borrow()
253      .last()
254      .map_or(0, |(_, ast)| Rc::as_ptr(ast) as usize)
255  });
256  warn_deprecated_once(script, old);
257}
258
259/// [`warn_deprecated`] from a trampoline, outside its script's run: the binding is the one the closure belongs to.
260pub(super) fn warn_deprecated_in(ast: &Rc<AST>, old: &'static str) {
261  warn_deprecated_once(Rc::as_ptr(ast) as usize, old);
262}
263
264fn warn_deprecated_once(script: usize, old: &'static str) {
265  let first = DEPRECATED_WARNED.with(|w| {
266    let mut warned = w.borrow_mut();
267    let first = !warned.contains(&(script, old));
268    if first {
269      warned.push((script, old));
270    }
271    first
272  });
273  if first {
274    let new = DEPRECATED_ALIASES
275      .iter()
276      .find(|(name, _)| *name == old)
277      .map_or("", |(_, new)| *new);
278    // (the call's line is not known here: the locator names the binding file)
279    let message = format!("{old} is deprecated; use {new}");
280    latexml_core::Warn!(
281      "deprecated",
282      old,
283      with_script_locator(&message, current_script_locator(rhai::Position::NONE))
284    );
285  }
286}

294/// Generate an RAII push/pop guard for one of the active-context stacks.
295///
296/// Why RAII and not manual `push`…`pop()`: Rhai does NOT `catch_unwind` its
297/// native calls, so a panic in a script body (or a deep `RefCell` borrow panic,
298/// incl. the B1 re-entrancy guard) skips a trailing manual `pop()` — leaving a
299/// **stale** entry on the stack. `reset_thread_state` doesn't clear these
300/// `latexml_contrib` thread-locals, so the dangling entry survives the cortex
301/// per-paper `catch_unwind` and leaks into the next paper. A `Drop` guard pops
302/// on EVERY scope exit — normal return, `?` early-return, or unwind. Review M1.
303macro_rules! ctx_stack_guard {
304  ($guard:ident, $stack:ident, $elem:ty) => {
305    #[must_use = "the guard pops on drop; binding it to `_` would pop immediately"]
306    struct $guard(());
307    impl $guard {
308      fn new(entry: $elem) -> Self {
309        $stack.with(|c| c.borrow_mut().push(entry));
310        Self(())
311      }
312    }
313    impl Drop for $guard {
314      fn drop(&mut self) {
315        $stack.with(|c| {
316          c.borrow_mut().pop();
317        });
318      }
319    }
320  };
321}
322
323ctx_stack_guard!(CtorCtxGuard, CTOR_CTX, CtorCtx);
324ctx_stack_guard!(WhatsitCtxGuard, WHATSIT_CTX, (*mut Whatsit, bool));
325ctx_stack_guard!(RevertingGuard, REVERTING, usize);
326ctx_stack_guard!(ScriptCtxGuard, CURRENT_SCRIPT, (Rc<Engine>, Rc<AST>));

341/// Wire immediately from inside a registration closure: resolve the current
342/// script handles, run the wiring fn, count, map errors to Rhai.
343fn wire_now(
344  wire: impl FnOnce(&Rc<Engine>, &Rc<AST>) -> Result<()>,
345) -> std::result::Result<(), Box<EvalAltResult>> {
346  let (engine, ast) = current_script()?;
347  wire(&engine, &ast).map_err(rhai_err)?;
348  note_wired();
349  Ok(())
350}

352/// Call a DEFERRED Rhai body (a primitive/macro/constructor/… closure) during
353/// digestion, with its `(engine, AST)` pushed onto `CURRENT_SCRIPT` for the
354/// duration. This lets the body itself call a registration (`DefPrimitive`,
355/// `DefMacro`, …): the nested `wire_now` → `current_script()` resolves this
356/// pushed context instead of failing "registration called outside a script
357/// load" (#316). Mirrors Perl, where a `def*` sub is callable from anywhere —
358/// script load OR digestion. The RAII guard nests correctly for a body that
359/// runs another body.
360pub(super) fn call_deferred_body<A: rhai::FuncArgs>(
361  engine: &Rc<Engine>,
362  ast: &Rc<AST>,
363  body: &FnPtr,
364  args: A,
365) -> std::result::Result<Dynamic, Box<EvalAltResult>> {
366  let _script_guard = ScriptCtxGuard::new((engine.clone(), ast.clone()));
367  let mut args_vec: Vec<Dynamic> = Vec::new();
368  args.parse(&mut args_vec);
369  // A Perl sub takes any number of arguments: a closure's missing ones are `()` (undef), extra ones dropped.
370  // A closure's captured variables are curried ahead of its own parameters.
371  if let Some(params) = closure_arity(ast, body.fn_name()) {
372    args_vec.resize(params.saturating_sub(body.curry().len()), Dynamic::UNIT);
373  }
374  body
375    .call::<Dynamic>(engine, ast, args_vec)
376    .inspect_err(|_| {
377      let path = ast.source().unwrap_or_default().to_string();
378      FAILED_BODY.with(|f| *f.borrow_mut() = Some(path));
379    })
380}

389/// A type name as a script author reads it: Rhai's own names (`string`, `int`, `float`, `bool`, `array`, `map`,
390/// `Fn`) and ours (`Tokens`, `Node`), never the Rust path or Rhai's internal string trio.
391pub(super) fn user_type_name(name: &str) -> &str {
392  match name.trim() {
393    "&str | ImmutableString | String" | "&str" | "String" | "ImmutableString" | "string" => {
394      "string"
395    },
396    "i64" | "i32" | "INT" | "int" => "int",
397    "f64" | "f32" | "FLOAT" | "float" => "float",
398    "Array" | "array" => "array",
399    "Map" | "map" => "map",
400    "FnPtr" | "Fn" => "Fn",
401    other => other.rsplit("::").next().unwrap_or(other),
402  }
403}

405/// The script-function calls a failure `e` passed through, innermost first, one per line, as Rhai prints them
406/// (`in call to function 'helper' (line 5, position 3)`); `None` for a failure in the body itself.
407fn failure_chain(e: &EvalAltResult) -> Option<String> {
408  let mut frames = Vec::new();
409  let mut cur = e;
410  while let EvalAltResult::ErrorInFunctionCall(name, _, inner, pos) = cur {
411    // (a closure frame called from Rust, the body itself, has no call site)
412    if !pos.is_none() {
413      let call = if name.starts_with("anon$") {
414        "in closure call".to_string()
415      } else {
416        format!("in call to function '{name}'")
417      };
418      frames.push(format!("{call} ({pos})"));
419    }
420    cur = inner;
421  }
422  frames.reverse();
423  (!frames.is_empty()).then(|| frames.join("\n\t"))
424}

426/// A Rhai failure as a binding's author reads it: the innermost error without Rhai's ` (line L, position C)`
427/// suffix, its types in [`user_type_name`] terms; where it happened; and the function-call chain above it.
428pub(super) fn describe_failure(e: &EvalAltResult) -> (String, rhai::Position, Option<String>) {
429  let chain = failure_chain(e);
430  let inner = e.unwrap_inner();
431  let pos = inner.position();
432  let text = match inner {
433    EvalAltResult::ErrorFunctionNotFound(sig, _) => {
434      let sig = match sig.split_once(" (") {
435        Some((name, args)) => {
436          let args = args.strip_suffix(')').unwrap_or(args);
437          let args: Vec<&str> = if args.is_empty() {
438            Vec::new()
439          } else {
440            args.split(", ").map(user_type_name).collect()
441          };
442          format!("{name} ({})", args.join(", "))
443        },
444        None => sig.clone(),
445      };
446      format!("Function not found: {sig}")
447    },
448    EvalAltResult::ErrorMismatchDataType(want, got, _) => format!(
449      "Data type incorrect: {} (expecting {})",
450      user_type_name(got),
451      user_type_name(want)
452    ),
453    EvalAltResult::ErrorMismatchOutputType(want, got, _) => format!(
454      "Output type incorrect: {} (expecting {})",
455      user_type_name(got),
456      user_type_name(want)
457    ),
458    other => {
459      let text = other.to_string();
460      match text.strip_suffix(&format!(" ({pos})")) {
461        Some(bare) if !pos.is_none() => bare.to_string(),
462        _ => text,
463      }
464    },
465  };
466  (text, pos, chain)
467}

469/// A binding file's locator, in the TeX locator's form: `at <path>; line L col C` (`None` without a path).
470pub(super) fn script_locator(path: &str, pos: rhai::Position) -> Option<String> {
471  if path.is_empty() {
472    return None;
473  }
474  Some(match (pos.line(), pos.position()) {
475    (Some(line), Some(col)) => format!("at {path}; line {line} col {col}"),
476    (Some(line), None) => format!("at {path}; line {line}"),
477    _ => format!("at {path}"),
478  })
479}

499/// [`closure_arity`]'s entry: the AST it was read from (held weakly: a dropped script's entry is recomputed) and
500/// the arity.
501type ArityEntry = (std::rc::Weak<AST>, Option<usize>);
502
503thread_local! {
504  static ARITY_CACHE: RefCell<HashMap<(usize, String), ArityEntry>> = RefCell::new(HashMap::new());
505}
506
507/// The parameter count of the script function `fn_name` (a closure's `anon$…`), `None` when the name is
508/// overloaded; looked up once per AST and name.
509fn closure_arity(ast: &Rc<AST>, fn_name: &str) -> Option<usize> {
510  let key = (Rc::as_ptr(ast) as usize, fn_name.to_string());
511  ARITY_CACHE.with(|cache| {
512    if let Some((weak, arity)) = cache.borrow().get(&key)
513      && weak.upgrade().is_some_and(|live| Rc::ptr_eq(&live, ast))
514    {
515      return *arity;
516    }
517    let mut arities = ast
518      .iter_functions()
519      .filter(|f| f.name == fn_name)
520      .map(|f| f.params.len());
521    let arity = match (arities.next(), arities.next()) {
522      (Some(params), None) => Some(params),
523      _ => None,
524    };
525    cache.borrow_mut().insert(key, (Rc::downgrade(ast), arity));
526    arity
527  })
528}

542/// audited site that re-mints the `&mut Document` the core published for the
543/// in-flight body (review B1: consolidated here so the `unsafe` has one
544/// documented home instead of being scattered across every proxy op).
545///
546/// RE-ENTRANCY SOUNDNESS (B1 — RESOLVED 2026-06-27, verified, not a band-aid):
547/// for a NESTED script construct — `\wrap{\myemph{..}}`, where the outer body's
548/// `Document::absorb` re-enters the bridge while its own `&mut self` is still
549/// live — this re-mints a *second* `&mut` from a raw pointer while the outer one
550/// is parked on the stack. The earlier B1 review feared this was aliasing UB; a
551/// careful reborrow analysis (and a Miri model — see below) shows it is **sound**:
552/// the nested pointer is a reborrow **descendant** of the outer one, not an alias.
553/// The chain is linear: the outer body's `with_doc` reborrows the top pointer →
554/// `Document::absorb(&mut self)` reborrows that → the core threads a reborrow of
555/// `&mut self` down to the nested constructor (`absorb` → `be_absorbed(self)` →
556/// the nested trampoline's `&mut Document`) → that body's `with_doc` reborrows
557/// *it*. Because `CTOR_CTX` is a STACK and `current_ctx()` always returns the
558/// **innermost** published pointer, every `with_doc` re-mints from a genuine
559/// descendant of all parked outer `&mut`s; a descendant reborrow never
560/// invalidates its ancestors. The native path is sound for the same reason
561/// (reborrow down the call chain); the Rhai closure boundary erases the
562/// lifetime, so the round-trip through `*mut` is unavoidable here, but it
563/// preserves the reborrow lineage.
564///
565/// VERIFIED: the exact pattern (thread-local `*mut` stack + RAII guard +
566/// `with_doc` re-mint + nested `absorb` reborrowing down) is modeled over a
567/// libxml2-free `Doc` in `latexml_core::runtime_bindings_reentrancy_model` and
568/// passes Miri under **both Stacked and Tree Borrows, 0 UB** (the real path can't
569/// be Miri-checked directly because `Document` is libxml2/FFI). The earlier
570/// checked-guard "fix" was correctly REJECTED — there is no UB to guard against,
571/// and failing the re-entrant op would deadlock `Document::absorb`'s loop, which
572/// requires the nested construction to SUCCEED. Tracked in `docs/SYNC_STATUS.md`
573/// "Open tasks #3 / PR #248 B1".
574fn with_doc<R>(
575  f: impl FnOnce(&mut Document, &SymHashMap<Stored>) -> std::result::Result<R, Box<EvalAltResult>>,
576) -> std::result::Result<R, Box<EvalAltResult>> {
577  let ctx = current_ctx()?;
578  // SAFETY: `ctx.document` is the `&mut Document` the core published for this
579  // body (valid for its duration). For the non-nested case this is the unique
580  // live `&mut`; for the nested case see the CAVEAT above. `ctx.props` is read
581  // only — never mutated through this `*const`.
582  let doc = unsafe { &mut *ctx.document };
583  let props = unsafe { &*ctx.props };
584  f(doc, props)
585}

588/// value instead of a `Result`. Outside a constructor body there is no document
589/// to consult, and a script has nothing useful to do with that fact, so the
590/// caller's `Default` (an empty array, a `()`) is the answer.
591fn with_doc_infallible<R: Default>(f: impl FnOnce(&mut Document) -> R) -> R {

604/// Unwrap a script-held node for a document operation that SPLICES it, refusing
605/// one that came from `ParseXML`.
606///
607/// A parsed node is still owned by the throwaway document it was parsed into, so
608/// moving it into ours would leave our tree pointing at memory freed the moment
609/// the script drops its handle — a use-after-free across FFI, the failure mode
610/// that has already cost this project SIGSEGVs and that `catch_unwind` cannot
611/// contain. `insertXML` is the supported route for parsed markup: it RE-CREATES
612/// the subtree through `append_tree` rather than moving it.
613fn in_tree(
614  node: NodeProxy,
615  method: &str,
616) -> std::result::Result<libxml::tree::Node, Box<EvalAltResult>> {
617  if node.1.is_some() {
618    return Err(Box::<EvalAltResult>::from(format!(
619      "{method}: this node came from ParseXML and belongs to its own parsed \
620       document; insert it with insertXML (which re-creates it) instead of \
621       splicing it in directly"
622    )));
623  }
624  Ok(node.0)
625}

627/// [`in_tree`] over an array argument, rejecting non-node entries by position —
628/// never silently skipping one, which would drop part of what a script asked for.
629fn in_tree_array(
630  nodes: rhai::Array,
631  method: &str,
632) -> std::result::Result<Vec<libxml::tree::Node>, Box<EvalAltResult>> {
633  let mut out = Vec::with_capacity(nodes.len());
634  for (i, entry) in nodes.into_iter().enumerate() {
635    let kind = type_label(&entry);
636    let proxy = entry.try_cast::<NodeProxy>().ok_or_else(|| {
637      Box::<EvalAltResult>::from(format!(
638        "{method}: entry {i} of the array is a {kind}, not a Node"
639      ))
640    })?;
641    out.push(in_tree(proxy, method)?);
642  }
643  Ok(out)
644}

647/// panic from an untrusted script aborts the conversion (the process, under

691/// Rhai proxy for a document-tree node: wraps the clonable libxml handle
692/// directly — no lifetimes, no active context. Handed to closure-form matcher
693/// bodies (`DefMathLigature(|node| …)`, read-only) and rewrite-`replace` bodies
694/// (mutable — `setAttribute`/`setContent`/`unlink`).
695///
696/// ⚠ LIFETIME FOOTGUN (review m5): the wrapped handle aliases a live C `xmlNode`
697/// owned by the conversion's document tree, and is valid ONLY for the duration
698/// of the body that received it. A script that stows a `NodeProxy` in a variable
699/// and dereferences it AFTER `unlink()` has detached the node, or after the
700/// conversion ends and the tree is freed (`reset_thread_engine`), is a C-level
701/// use-after-free (cf. WISDOM #58). Unlike `DocProxy`/`WhatsitProxy` — which
702/// resolve their target through the active-context stack on each call and carry
703/// no pointer — `NodeProxy` is the only proxy holding a raw tree handle. Use it
704/// within the body; never retain it across calls or past `unlink()`.
705/// A node may instead OWN its document, which lifts that restriction for parsed
706/// markup: `ParseXML` builds its nodes in a throwaway document, and since libxml's
707/// `Document` is refcounted, holding a clone here keeps the node valid for exactly
708/// as long as the script keeps the handle. One type serves both provenances on
709/// purpose — every node method then works on in-tree and parsed nodes alike, and a
710/// new one is a single registration rather than a parallel API.
711#[derive(Clone)]
712pub(crate) struct NodeProxy(
713  pub(crate) libxml::tree::Node,
714  /// `Some` when this handle keeps its own (parsed) document alive; `None` for a
715  /// node inside the conversion's document, whose lifetime the caller guarantees.
716  pub(crate) Option<libxml::tree::Document>,
717);
718
719impl NodeProxy {
720  /// A node inside the live conversion document — valid only for the body that
721  /// received it (see the caveat above).
722  pub(crate) fn borrowed(node: libxml::tree::Node) -> Self { Self(node, None) }
723
724  /// A node whose owning document this handle keeps alive.
725  pub(crate) fn owning(node: libxml::tree::Node, doc: libxml::tree::Document) -> Self {
726    Self(node, Some(doc))
727  }
728
729  /// A node reached FROM this one (parent, sibling, child): it lives in the same
730  /// document, so it must inherit the same provenance — otherwise walking a
731  /// parsed tree would hand back nodes that no longer keep their owner alive.
732  pub(crate) fn derived(&self, node: libxml::tree::Node) -> Self { Self(node, self.1.clone()) }
733}

804/// Load a binding script: compile once (cached by source), then RUN the script
805/// against the current State. Every `Def…` registration wires its native
806/// definition IMMEDIATELY as the script executes (sequential, Perl `.ltxml`
807/// semantics — so `DeclareOption` precedes a later `ProcessOptions()`), and
808/// top-level side-effect calls (`RawTeX`, `Let`, `NewCounter`, `AssignValue`,
809/// `DefRegister`, …) re-apply on every load — i.e. per conversion, not just
810/// the first one that happened to compile the script. Returns the number of
811/// definitions installed.
812pub fn load_script(src: &str) -> Result<usize> { load_script_from(src, "") }
813
814/// [`load_script`] for the binding file at `path` (empty: an anonymous script), which its diagnostics name.
815///
816/// A failure is one `Error:script:<file>` whose locator is the `.rhai` line: a syntax error loads nothing; a run
817/// error keeps what the statements before it defined and skips the rest (the loader then marks the file loaded, so
818/// its raw `.sty`/`.cls` is not read instead).
819pub fn load_script_from(src: &str, path: &str) -> Result<usize> {
820  let key = (path.to_string(), src.to_string());
821  // Fast path: a previously-compiled script with the same source.
822  let cached = SCRIPT_CACHE.with(|c| c.borrow().get(&key).cloned());
823  let cached = match cached {
824    Some(c) => c,
825    None => {
826      let engine = make_engine();
827      let mut ast = engine.compile(src).map_err(|e| {
828        let pos = e.position();
829        let text = e.to_string();
830        let text = text.strip_suffix(&format!(" ({pos})")).unwrap_or(&text);
831        load_failure(
832          path,
833          &format!("Syntax error: {text}; the binding is not loaded"),
834          pos,
835          None,
836        )
837      })?;
838      if !path.is_empty() {
839        ast.set_source(path);
840      }
841      let cs = CachedScript {
842        engine: Rc::new(engine),
843        ast:    Rc::new(ast),
844      };
845      SCRIPT_CACHE.with(|c| {
846        c.borrow_mut().insert(key, cs.clone());
847      });
848      cs
849    },
850  };
851
852  // Publish the script handles for the registration closures, run, unpublish
853  // (a stack, so a script loading another script nests correctly). RAII pop on
854  // every exit incl. a panic out of `run_ast` (review M1).
855  let before = WIRED_COUNT.with(|c| *c.borrow());
856  let script = Rc::as_ptr(&cached.ast) as usize;
857  DEPRECATED_WARNED.with(|w| w.borrow_mut().retain(|(s, _)| *s != script));
858  // A dropped script's closure arities go with it.
859  ARITY_CACHE.with(|c| c.borrow_mut().retain(|_, (ast, _)| ast.strong_count() > 0));
860  let run = {
861    let _script_guard = ScriptCtxGuard::new((cached.engine.clone(), cached.ast.clone()));
862    cached.engine.run_ast(&cached.ast).map_err(|e| {
863      take_raised_fatal(&e).unwrap_or_else(|| {
864        let (text, pos, chain) = describe_failure(&e);
865        let text = format!("{text}; the rest of the binding is skipped");
866        load_failure(path, &text, pos, chain)
867      })
868    })
869  };
870  run?;
871  Ok(WIRED_COUNT.with(|c| *c.borrow()) - before)
872}

874/// A binding that failed to load, as `Error:script:<file>` with the `.rhai` locator: an error naming its own
875/// category and object, which the loader logs as spelled (`content::input_definitions`).
876fn load_failure(path: &str, message: &str, pos: rhai::Position, chain: Option<String>) -> Error {
877  use latexml_core::common::error::{ErrorCategory, ErrorData, ErrorTarget};
878  let file = std::path::Path::new(path).file_name().map_or_else(

881  );

883  Error::from(ErrorData {
884    target:   ErrorTarget::Named("script".to_string()),

886    message:  match chain {
887      Some(chain) => format!("{located}\n\t{chain}"),
888      None => located,
889    },
890  })

932    }

937/// Convert a Rhai object map into a whatsit property map, each value with its type ([`dynamic_to_stored`]); a
938/// `Digested` under the `font` key contributes its FONT (the IEEEproof "titlefont" idiom).
939fn rhai_map_to_props(map: Map) -> SymHashMap<Stored> {
940  let mut props: SymHashMap<Stored> = SymHashMap::default();
941  for (k, v) in map {
942    match v.clone().try_cast::<Digested>() {
943      Some(d) => {
944        if k.as_str() == "font"
945          && let Ok(Some(f)) = d.get_font()
946        {
947          props.insert("font", Stored::Font(f));
948          continue;
949        }
950        props.insert(k.as_str(), Stored::Digested(d));
951      },
952      // A constructor reads a token property as its ToString, as Perl's absorb (Object.pm:169) and attributes do.
953      _ if v.is::<Tokens>() || v.is::<Token>() => {
954        props.insert(k.as_str(), dynamic_to_string(v).into())
955      },
956      _ => props.insert(k.as_str(), dynamic_to_stored(v)),
957    }
958  }
959  props
960}

1015/// Marshal a `Stored` out to a Rhai value with its type (Perl hands back the stored object, Package.pm
1016/// LookupValue :257, LookupMapping :283): a string, bool, int, float, `Token`, `Tokens` or digested handle as

1024    Stored::Bool(b) => Dynamic::from(b),

1026    Stored::Number(n) => Dynamic::from(n),

1030    Stored::Float(f) => Dynamic::from(f.0),
1031    Stored::Tokens(t) => Dynamic::from(t),
1032    Stored::Token(t) => Dynamic::from(t),
1033    Stored::Digested(d) => Dynamic::from(d),
1034    Stored::Strings(ss) => Dynamic::from_array(
1035      ss.iter()
1036        .map(|s| Dynamic::from(arena::to_string(*s)))
1037        .collect(),
1038    ),
1039    Stored::VecDequeStored(items) => {
1040      Dynamic::from_array(items.into_iter().map(stored_to_dynamic).collect())
1041    },
1042    Stored::VecDigested(items) => {
1043      Dynamic::from_array(items.into_iter().map(Dynamic::from).collect())
1044    },
1045    Stored::HashStored(h) => Dynamic::from_map(
1046      h.0
1047        .into_iter()
1048        .map(|(k, v)| (arena::to_string(k).into(), stored_to_dynamic(v)))
1049        .collect(),
1050    ),
1051    Stored::HashString(h) => Dynamic::from_map(
1052      h.into_iter()
1053        .map(|(k, v)| (k.into(), Dynamic::from(v)))
1054        .collect(),
1055    ),
1056    Stored::Charcode(c) => Dynamic::from(i64::from(c)),
1057    Stored::Catcode(c) => Dynamic::from(c as i64),
1058    // A stored definition is Perl's definition object: a `LookupDefinition` handle.
1059    Stored::Expandable(_)
1060    | Stored::Conditional(_)
1061    | Stored::Primitive(_)
1062    | Stored::MathPrimitive(_)
1063    | Stored::Constructor(_)
1064    | Stored::Register(_) => match definition_cs(&v) {
1065      Some(cs) => Dynamic::from(DefinitionProxy { cs, meaning: v }),
1066      None => Dynamic::from(v.to_string()),
1067    },
1068    other => Dynamic::from(other.to_string()),
1069  }
1070}

1072/// The CS a stored definition defines.
1073fn definition_cs(v: &Stored) -> Option<Token> {
1074  use latexml_core::definition::Definition;
1075  Some(match v {
1076    Stored::Expandable(d) => d.get_cs().into_owned(),
1077    Stored::Conditional(d) => d.get_cs().into_owned(),
1078    Stored::Primitive(d) => d.get_cs().into_owned(),
1079    Stored::MathPrimitive(d) => d.get_cs().into_owned(),
1080    Stored::Constructor(d) => d.get_cs().into_owned(),
1081    Stored::Register(d) => d.get_cs().into_owned(),
1082    _ => return None,
1083  })
1084}

1086/// Perl's `ToString($definition)` (Definition.pm:116): the CS, then its parameters after a space.
1087fn definition_string(d: &DefinitionProxy) -> String {
1088  use latexml_core::{common::object::Object, definition::Definition};
1089  let params = match &d.meaning {
1090    Stored::Expandable(x) => x.get_parameters().map(Object::stringify),
1091    Stored::Conditional(x) => x.get_parameters().map(Object::stringify),
1092    Stored::Primitive(x) => x.get_parameters().map(Object::stringify),
1093    Stored::MathPrimitive(x) => x.get_parameters().map(Object::stringify),
1094    Stored::Constructor(x) => x.get_parameters().map(Object::stringify),
1095    Stored::Register(x) => x.get_parameters().map(Object::stringify),
1096    _ => None,
1097  };
1098  match params {
1099    Some(p) => format!("{} {p}", d.cs),
1100    None => d.cs.to_string(),
1101  }
1102}

1115/// Marshal a Rhai value INTO a `Stored` (Perl's value table is untyped, Package.pm AssignValue :261,
1116/// AssignMapping :287, PushValue): string, int, float, bool, `Token`, `Tokens`, a value handle (Number, Dimension,
1117/// Glue, MuGlue) and a digested handle keep their type, an array is a list and a map a hash of such (Perl's arrayref/hashref), `()` is `None` (undef); anything
1118/// else falls back to its string form (#540). The inverse of [`stored_to_dynamic`].
1119fn dynamic_to_stored(v: Dynamic) -> Stored {
1120  if v.is_unit() {
1121    Stored::None
1122  } else if v.is_string() {
1123    v.into_string().unwrap_or_default().into()
1124  } else if let Ok(i) = v.as_int() {
1125    Stored::Int(i)
1126  } else if let Ok(f) = v.as_float() {
1127    Stored::Float(Float(f))
1128  } else if let Ok(b) = v.as_bool() {
1129    Stored::Bool(b)
1130  } else if let Some(tokens) = v.clone().try_cast::<Tokens>() {
1131    Stored::Tokens(tokens)
1132  } else if let Some(token) = v.clone().try_cast::<Token>() {
1133    Stored::Token(token)
1134  } else if let Some(digested) = v.clone().try_cast::<Digested>() {
1135    Stored::Digested(digested)
1136  } else if let Some(def) = v.clone().try_cast::<DefinitionProxy>() {
1137    def.meaning
1138  } else if let Some(value) = values::dynamic_to_register_value(&v) {
1139    match value {
1140      RegisterValue::Number(n) => Stored::Number(n),
1141      RegisterValue::Dimension(d) => Stored::Dimension(d),
1142      RegisterValue::Glue(g) => Stored::Glue(g),
1143      RegisterValue::MuGlue(g) => Stored::MuGlue(g),
1144      other => other.to_string().into(),
1145    }
1146  } else if v.is_array() {
1147    Stored::VecDequeStored(
1148      v.cast::<rhai::Array>()
1149        .into_iter()
1150        .map(dynamic_to_stored)
1151        .collect(),
1152    )
1153  } else if v.is_map() {
1154    let mut hash = SymHashMap::default();
1155    for (k, v) in v.cast::<Map>() {
1156      hash.insert(k.as_str(), dynamic_to_stored(v));
1157    }
1158    Stored::HashStored(hash)
1159  } else {
1160    v.to_string().into()
1161  }
1162}

1164/// A closure's returned expansion or reversion as Tokens: a `Token` or `Tokens` as itself (a Perl macro sub
1165/// returns tokens), `()` as nothing, a string (the Rhai shorthand) TokenizeInternal'd as before.
1166fn dynamic_to_tokens(d: Dynamic) -> Tokens {
1167  if let Some(tokens) = d.clone().try_cast::<Tokens>() {
1168    tokens
1169  } else if let Some(token) = d.clone().try_cast::<Token>() {
1170    Tokens::new(vec![token])
1171  } else if d.is_unit() {
1172    Tokens::default()
1173  } else {
1174    mouth::tokenize_internal(TeXString::assembled(dynamic_to_string(d)))
1175  }
1176}

1181/// Read a Rhai value back as a string, Perl's `ToString` (Object.pm:59): a `Token`/`Tokens` as `Tokens->toString`
1182/// (`\foo bar` → `\foobar`, as [`rhai_map_to_props`] stores a token property), a digested handle or a value handle
1183/// (Dimension, …) its `ToString` (Rhai's own printing gives a custom type's Rust name).
1184fn dynamic_to_string(d: Dynamic) -> String {
1185  if d.is_string() {
1186    d.into_string().unwrap_or_default()
1187  } else if let Some(tokens) = d.clone().try_cast::<Tokens>() {
1188    tokens.to_string()
1189  } else if let Some(token) = d.clone().try_cast::<Token>() {
1190    token.to_string()
1191  } else if let Some(digested) = d.clone().try_cast::<Digested>() {
1192    digested.to_string()
1193  } else if let Some(value) = values::value_string(&d) {
1194    value
1195  } else if d.is::<WhatsitProxy>() {
1196    // (`to_string`, `ToString` and `absorbString` raise its error; see `whatsit_string`)
1197    whatsit_string(rhai::Position::NONE).unwrap_or_default()
1198  } else if let Some(def) = d.clone().try_cast::<DefinitionProxy>() {
1199    definition_string(&def)
1200  } else if d.is_array() {
1201    let items: Vec<String> = d
1202      .cast::<rhai::Array>()
1203      .into_iter()
1204      .map(element_string)
1205      .collect();
1206    format!("[{}]", items.join(", "))
1207  } else {
1208    d.to_string()
1209  }
1210}

1212/// The hook's whatsit as Perl's `ToString($whatsit)`: its reversion (Whatsit.pm:163). Inside that whatsit's own
1213/// script reversion, where Perl recurses without end, an Error and `""` (Rhai's `+` would swallow a raised one).
1214fn whatsit_string(pos: rhai::Position) -> std::result::Result<String, Box<EvalAltResult>> {
1215  let (ptr, _) = current_whatsit_entry()?;
1216  if REVERTING.with(|r| r.borrow().contains(&(ptr as usize))) {
1217    report_self_reversion(pos).map_err(rhai_err)?;
1218    return Ok(String::new());
1219  }
1220  with_whatsit(|w| w.revert().map(|t| t.to_string()).map_err(rhai_err))?
1221}
1222
1223fn report_self_reversion(pos: rhai::Position) -> Result<()> {
1224  latexml_core::Error!(
1225    "script",
1226    "reversion",
1227    with_script_locator(
1228      "a reversion's own whatsit in a string reverts it again; use whatsit.getArg(n)",
1229      current_script_locator(pos)
1230    )
1231  );
1232  Ok(())
1233}

1235/// Perl's truthiness (`if ($v)`): undef, "", "0" and 0 are false, a bool is itself, and every object (Token, Tokens,
1236/// a box, a whatsit, an array, a map) is true, as Perl LaTeXML's objects overload no bool (Common/Object.pm:120).
1237fn perl_truthy(v: &Dynamic) -> bool {
1238  if v.is_unit() {
1239    false
1240  } else if let Ok(b) = v.as_bool() {
1241    b
1242  } else if let Ok(i) = v.as_int() {
1243    i != 0
1244  } else if let Ok(f) = v.as_float() {
1245    f != 0.0
1246  } else if let Ok(s) = v.as_immutable_string_ref() {
1247    !(s.is_empty() || s.as_str() == "0")
1248  } else if let Ok(c) = v.as_char() {
1249    c != '0'
1250  } else {
1251    true
1252  }
1253}
