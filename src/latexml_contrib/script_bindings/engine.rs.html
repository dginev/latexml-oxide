1//! The sandboxed Rhai engine: every `register_fn` of the binding-language
2//! surface (state, tokens, counters, definitions, package/class machinery,
3//! document/whatsit proxies). Split from `mod.rs` for navigability; each
4//! registration lowers to the same native function its compile-time macro
5//! does (`setup_binding_language.rs`).

9/// Mini-DSL to expose a `document.<rhai>(qname)` → `Document::<rust>(qname)`
10/// method whose result is discarded — the common side-effect-on-element shape.
11/// Adding a new such method is then one line; see the registrations in
12/// `make_engine`.
13macro_rules! doc_qname_method {
14  ($engine:ident, $rhai:literal, $rust:ident) => {
15    $engine.register_fn(
16      $rhai,
17      |_d: &mut DocProxy, qname: &str| -> std::result::Result<(), Box<EvalAltResult>> {
18        with_doc(|doc, _props| {
19          doc.$rust(qname).map_err(rhai_err)?;
20          Ok(())
21        })
22      },
23    );
24  };
25}

27/// Build a sandboxed Rhai engine with the binding API registered.
28pub(super) fn make_engine() -> Engine {
29  let mut engine = Engine::new();
30  engine.set_max_operations(MAX_OPERATIONS);
31  engine.set_max_call_levels(MAX_CALL_LEVELS);
32  engine.set_max_string_size(MAX_STRING_SIZE);
33
34  // ── registration API (wired to native defs IMMEDIATELY, in script order) ──
35  engine.register_fn(
36    "DefMacro",
37    |proto: &str, body: FnPtr| -> std::result::Result<(), Box<EvalAltResult>> {
38      wire_now(|e, a| wire_macro(e, a, proto, body))
39    },
40  );
41  // String-body form — Perl's most common `DefMacro('\foo', 'bar')`, where the
42  // body is TeX source that becomes the expansion directly (no Rhai closure).
43  // Wires the same native expandable as the compile-time `DefMacro!` string
44  // form, so a `.rhai` binding reads like the `.ltxml` original. The wiring
45  // ignores the script handles, hence the throwaway `|_, _|`.
46  engine.register_fn(
47    "DefMacro",
48    |proto: &str, body: &str| -> std::result::Result<(), Box<EvalAltResult>> {
49      wire_now(|_, _| wire_macro_string(proto, body))
50    },
51  );
52  // Option-bag form (Perl's trailing `key => value`s): scalars onto
53  // `ExpandableOptions` via the shared mapper. Both the closure-body and the
54  // string-body variants accept it.
55  engine.register_fn(
56    "DefMacro",
57    |proto: &str, body: FnPtr, opts: Map| -> std::result::Result<(), Box<EvalAltResult>> {
58      wire_now(|e, a| wire_macro_opts(e, a, proto, body, opts))
59    },
60  );
61  engine.register_fn(
62    "DefMacro",
63    |proto: &str, body: &str, opts: Map| -> std::result::Result<(), Box<EvalAltResult>> {
64      wire_now(|_, _| wire_macro_string_opts(proto, body, opts))
65    },
66  );
67  engine.register_fn(
68    "DefPrimitive",
69    |proto: &str, body: FnPtr| -> std::result::Result<(), Box<EvalAltResult>> {
70      wire_now(|e, a| wire_primitive(e, a, proto, body, PrimitiveOptions::default()))
71    },
72  );
73  engine.register_fn(
74    "DefPrimitive",
75    |proto: &str, body: FnPtr, opts: Map| -> std::result::Result<(), Box<EvalAltResult>> {
76      wire_now(|e, a| wire_primitive(e, a, proto, body, primitive_options_from_map(opts, e, a)))
77    },
78  );
79  // Class/package option. Mirrors the `DeclareOption!` macro's lowering
80  // (setup_binding_language.rs): push the name onto `@declaredoptions` and
81  // define a `\ds@<opt>` primitive carrying the body. Installed immediately,
82  // so a following `ProcessOptions()` in the same script sees it.
83  engine.register_fn(
84    "DeclareOption",
85    |opt: &str, body: FnPtr| -> std::result::Result<(), Box<EvalAltResult>> {
86      wire_now(|e, a| wire_option(e, a, opt, body))
87    },
88  );
89  // String-body form: `DeclareOption('opt', '\xdef...')` — the body is TeX,
90  // installed as a token-expansion macro (Perl's `DeclareOption($o, $string)`).
91  engine.register_fn(
92    "DeclareOption",
93    |opt: &str, body: &str| -> std::result::Result<(), Box<EvalAltResult>> {
94      wire_now(|_, _| wire_option_string(opt, body))
95    },
96  );
97  // Default-option handler: `DeclareOption(sub {...})` with no name — Perl's
98  // `DeclareOption(undef, sub {...})`, defining `\default@ds` for any option not
99  // otherwise declared.
100  engine.register_fn(
101    "DeclareOption",
102    |body: FnPtr| -> std::result::Result<(), Box<EvalAltResult>> {
103      wire_now(|e, a| wire_option_default(e, a, body))
104    },
105  );
106
107  // ── free-function helpers (prelude names) over core datatypes ──
108  // `Tokens` is registered as an opaque pass-through handle (Tier-2): a script
109  // produces it with Tokenize and hands it to Digest without inspecting it.
110  engine.register_type_with_name::<Tokens>("Tokens");
111  // `assembled`: the argument is TeX the script author wrote — the runtime twin
112  // of a `Tokenize!("…")` literal. A script cannot hand us a flattened `Tokens`
113  // (Rhai `Tokens` is opaque and has no string conversion registered).
114  engine.register_fn("Tokenize", |s: &str| -> Tokens {
115    mouth::tokenize(TeXString::assembled(s.to_string()))
116  });
117  engine.register_fn("TokenizeInternal", |s: &str| -> Tokens {
118    mouth::tokenize_internal(TeXString::assembled(s.to_string()))
119  });
120  // `Digest` returns the digested handle (Perl returns the digested box), so a
121  // script can `ToString(Digest(...))` — e.g. a default-option handler reading
122  // `\CurrentOption`. Callers that ignore the return (the common side-effect
123  // use) are unaffected.
124  engine.register_fn(
125    "Digest",
126    |t: Tokens| -> std::result::Result<Digested, Box<EvalAltResult>> {
127      latexml_core::stomach::digest(t).map_err(rhai_err)
128    },
129  );
130  // `T_CS('\foo')` — a single control-sequence token (wrapped as `Tokens` so it
131  // composes with `Digest`/`Expand`). The dominant token-constructor use.
132  engine.register_fn("T_CS", |name: &str| -> Tokens {
133    Tokens::new(vec![latexml_core::T_CS!(name)])
134  });
135
136  // ── state API (for primitive side-effects / value-reading macros) ──
137  // `assign_value` is group-local (TeX default); `assign_global` persists past
138  // the enclosing group. (Untrusted-script key-namespacing is a documented
139  // follow-up — see the plan's critical re-eval §5.)
140  engine.register_fn("assign_value", |key: &str, val: &str| {
141    latexml_core::state::assign_value(key, val.to_string(), Some(Scope::Local));
142  });
143  engine.register_fn("assign_global", |key: &str, val: &str| {
144    latexml_core::state::assign_value(key, val.to_string(), Some(Scope::Global));
145  });
146  // Curated pool helpers, registered 1:1 under their Perl/Rust names so binding
147  // bodies read like the originals (e.g. `beforeDigest: || neutralize_font()`).
148  engine.register_fn(
149    "neutralize_font",
150    latexml_engine::base_utilities::neutralize_font,
151  );
152  engine.register_fn("lookup_value", |key: &str| -> String {
153    match latexml_core::state::lookup_value(key) {
154      Some(Stored::String(s)) => arena::to_string(s),
155      Some(other) => other.to_string(),
156      None => String::new(),
157    }
158  });
159
160  // ── the binding-language surface, 1:1 under its macro names ──
161  // Each registration lowers to the SAME native function its compile-time
162  // macro does (setup_binding_language.rs), so a Rhai binding reads like the
163  // `.pool`/`_sty.rs` original. String values in, strings/handles out.
164  engine.register_fn("AssignValue", |k: &str, v: &str| {
165    latexml_core::state::assign_value(k, v.to_string(), None);
166  });
167  engine.register_fn("AssignValue", |k: &str, v: &str, scope: &str| {
168    latexml_core::state::assign_value(k, v.to_string(), scope_of(scope));
169  });
170  // Typed `Assign*` counterparts to the typed `Lookup*` readers below (#543).
171  // Before these, a binding could only ever assign a *string*, so a value
172  // written from a script was invisible to `LookupNumber`/`LookupBool`/
173  // `LookupFloat`. Each stores the matching `Stored` variant explicitly — in
174  // particular `AssignFloat` builds a `Float` (keeping the fraction) rather than
175  // an integer `Number`, which a bare `f64 -> Stored` would floor. 2-arg = TeX
176  // default scope (as `AssignValue`), 3-arg takes "local"/"global".
177  engine.register_fn("AssignNumber", |k: &str, v: i64| {
178    latexml_core::state::assign_value(k, Number(v), None);
179  });
180  engine.register_fn("AssignNumber", |k: &str, v: i64, scope: &str| {
181    latexml_core::state::assign_value(k, Number(v), scope_of(scope));
182  });
183  engine.register_fn("AssignFloat", |k: &str, v: f64| {
184    latexml_core::state::assign_value(k, Float(v), None);
185  });
186  engine.register_fn("AssignFloat", |k: &str, v: f64, scope: &str| {
187    latexml_core::state::assign_value(k, Float(v), scope_of(scope));
188  });
189  engine.register_fn("AssignBool", |k: &str, v: bool| {
190    latexml_core::state::assign_value(k, v, None);
191  });
192  engine.register_fn("AssignBool", |k: &str, v: bool, scope: &str| {
193    latexml_core::state::assign_value(k, v, scope_of(scope));
194  });
195  // Explicit string counterpart to `LookupString` (equivalent to `AssignValue`,
196  // named for a symmetric typed family).
197  engine.register_fn("AssignString", |k: &str, v: &str| {
198    latexml_core::state::assign_value(k, v.to_string(), None);
199  });
200  engine.register_fn("AssignString", |k: &str, v: &str, scope: &str| {
201    latexml_core::state::assign_value(k, v.to_string(), scope_of(scope));
202  });
203  // The Perl `@values` list-value family (Package.pm L265-279 -> State.pm L218):
204  // a value-table key holds a queue that these push/pop/unshift/shift. Always
205  // GLOBAL scope, auto-vivifying the list if absent — matching Perl, which takes
206  // no scope argument. Perl's `PushValue` is untyped and the Rust State layer
207  // holds any `Stored`, so `PushValue`/`UnshiftValue` take any Rhai-representable
208  // value (string/int/float/bool/`Tokens`/digested handle) and `PopValue`/
209  // `ShiftValue` return it with its type preserved (#540). `LookupValue` reads
210  // the whole queue back as a Rhai array. As of #561 `SEARCHPATHS` is itself a
211  // group-scoped value-table list (like `GRAPHICSPATHS`), so `PushValue`/
212  // `UnshiftValue` reach it too; the dedicated `PrependSearchPath`/
213  // `AppendSearchPath` below are the ergonomic front for input dirs.
214  engine.register_fn("PushValue", |k: &str, v: Dynamic| {
215    // Returns Ok(()) always; the wrong-type BUG path self-reports via Error!.
216    let _ = latexml_core::state::push_value(k, dynamic_to_stored(v));
217  });
218  engine.register_fn("UnshiftValue", |k: &str, v: Dynamic| {
219    latexml_core::state::unshift_value(k, vec![dynamic_to_stored(v)]);
220  });
221  engine.register_fn("PopValue", |k: &str| -> Dynamic {
222    match latexml_core::state::pop_value(k) {
223      Ok(Some(popped)) => popped_stored_to_dynamic(popped),
224      _ => Dynamic::UNIT,
225    }
226  });
227  engine.register_fn("ShiftValue", |k: &str| -> Dynamic {
228    match latexml_core::state::shift_value(k) {
229      Ok(Some(shifted)) => popped_stored_to_dynamic(shifted),
230      _ => Dynamic::UNIT,
231    }
232  });
233  // Input search paths. Perl keeps these in the `SEARCHPATHS` value and prepends
234  // via `UnshiftValue`; as of #561 the Rust port does the same — `SEARCHPATHS` is
235  // a group-scoped value-table list read by `find_file` — so these dedicated
236  // bindings are a convenience over `PushValue`/`UnshiftValue`. Prepend = searched
237  // first (Perl's `SEARCHPATHS` semantics).
238  engine.register_fn("PrependSearchPath", |dir: &str| {
239    latexml_core::state::search_paths_push_front(dir.to_string());
240  });
241  engine.register_fn("AppendSearchPath", |dir: &str| {
242    latexml_core::state::add_search_path(dir.to_string());
243  });
244  engine.register_fn("LookupString", |k: &str| -> String {
245    latexml_core::state::lookup_string(k)
246  });
247  // Perl-style `LookupValue`: returns the stored value in its natural shape. A
248  // list value (e.g. `class_options`, a pushed queue) becomes a Rhai array —
249  // mirroring Perl's `LookupValue` returning the arrayref (#315) — rather than
250  // the leaked Debug repr `LookupString` used to emit. Scalars stringify as
251  // before; a missing key is unit (Perl `undef`).
252  engine.register_fn("LookupValue", |k: &str| -> Dynamic {
253    match latexml_core::state::lookup_value(k) {
254      None => Dynamic::UNIT,
255      Some(v) if v.is_list() => {
256        let arr: rhai::Array = v
257          .list_items()
258          .unwrap_or_default()
259          .into_iter()
260          .map(Dynamic::from)
261          .collect();
262        Dynamic::from_array(arr)
263      },
264      Some(Stored::String(s)) => arena::to_string(s).into(),
265      Some(other) => other.to_string().into(),
266    }
267  });
268  engine.register_fn("LookupNumber", |k: &str| -> i64 {
269    latexml_core::state::lookup_number(k)
270      .map(|n| n.0)
271      .unwrap_or(0)
272  });
273  // The fractional reader `LookupNumber` cannot be: a value assigned with
274  // `AssignFloat` (e.g. NOMINAL_FONT_SIZE = 10.95) would read back whole
275  // through `LookupNumber`, so scripts that need the fraction use this (#542/#543).
276  engine.register_fn("LookupFloat", |k: &str| -> f64 {
277    latexml_core::state::lookup_float(k)
278      .map(|f| f.0)
279      .unwrap_or(0.0)
280  });
281  engine.register_fn("LookupBool", |k: &str| -> bool {
282    latexml_core::state::lookup_bool(k)
283  });
284  // Perl's `$LaTeXML::VERSION` — the running latexml-oxide version as a bare
285  // `X.Y.Z`, so a binding can gate on it (BookML does `$LaTeXML::VERSION <= …`).
286  // It is OUR crate version, not the emulated Perl LaTeXML's. The top crate
287  // publishes it to state (`LATEXML_VERSION`) at session init (#320); `""` if a
288  // conversion hasn't been initialized (e.g. a bare unit test).
289  engine.register_fn("LaTeXMLVersion", || -> String {
290    latexml_core::state::lookup_string("LATEXML_VERSION")
291  });
292  engine.register_fn("LookupTokens", |k: &str| -> Tokens {
293    latexml_core::state::lookup_tokens(k).unwrap_or_default()
294  });
295  // Catcode access: the char as a 1-char string, the catcode as its TeX int.
296  engine.register_fn("LookupCatcode", |c: &str| -> i64 {
297    c.chars()
298      .next()
299      .and_then(latexml_core::state::lookup_catcode)
300      .map(|cc| cc as i64)
301      .unwrap_or(12)
302  });
303  engine.register_fn("AssignCatcode", |c: &str, code: i64| {
304    if let Some(ch) = c.chars().next() {
305      latexml_core::state::assign_catcode(ch, Catcode::from(code as u8), None);
306    }
307  });
308  // StartSemiverbatim/EndSemiverbatim (Perl `Package.pm` exports, L1002-1008):
309  // read the following content raw — the default neutralizes the math/special
310  // catcodes (`^ _ ~ & $ # '`) to `OTHER` so they are literal, keeping `\ { }`
311  // special so e.g. `\end{env}` still parses (#653). The canonical way to give a
312  // Rhai `DefEnvironment` a verbatim body: `beforeDigest: || StartSemiverbatim()`
313  // + `afterDigestBody: || EndSemiverbatim()`. Pass extra single-char strings to
314  // neutralize them too (e.g. `StartSemiverbatim(["%"])`).
315  engine.register_fn("StartSemiverbatim", || {
316    latexml_core::state::begin_semiverbatim(None);
317  });
318  engine.register_fn("StartSemiverbatim", |extra: rhai::Array| {
319    let chars: Vec<char> = extra
320      .iter()
321      .filter_map(|d| d.clone().into_string().ok())
322      .filter_map(|s| s.chars().next())
323      .collect();
324    latexml_core::state::begin_semiverbatim(Some(&chars));
325  });
326  engine.register_fn(
327    "EndSemiverbatim",
328    || -> std::result::Result<(), Box<EvalAltResult>> {
329      latexml_core::state::end_semiverbatim().map_err(rhai_err)
330    },
331  );
332  // LookupMeaning: the meaning of a CS as its display string ("" if none).
333  engine.register_fn("LookupMeaning", |cs: &str| -> String {
334    match latexml_core::state::lookup_meaning(&latexml_core::T_CS!(cs)) {
335      Some(m) => m.to_string(),
336      None => String::new(),
337    }
338  });
339  engine.register_fn("Let", |a: &str, b: &str| {
340    latexml_core::state::let_i(&latexml_core::T_CS!(a), &latexml_core::T_CS!(b), None);
341  });
342  engine.register_fn("XEquals", |a: &str, b: &str| -> bool {
343    latexml_core::state::x_equals(&latexml_core::T_CS!(a), &latexml_core::T_CS!(b))
344  });
345  engine.register_fn("IsDefined", |cs: &str| -> bool {
346    latexml_core::binding::def::dialect::is_defined_token(&latexml_core::T_CS!(cs))
347  });
348  // RawTeX: process literal TeX as definitions input (the raw-`\def` escape
349  // hatch every nontrivial binding uses). TeX: tokenize-internal + digest.
350  engine.register_fn(
351    "RawTeX",
352    |text: &str| -> std::result::Result<(), Box<EvalAltResult>> {
353      latexml_core::stomach::raw_tex(text).map_err(rhai_err)
354    },

356  // Document-lifecycle hooks (Perl `Package.pm` `AtBeginDocument`/`AtEndDocument`,
357  // L2815/L2830). Each pushes a Tokens body onto the value-table list the engine
358  // digests at `\begin{document}`/`\end{document}` — mirroring the native macros
359  // in `latex_constructs.rs` (which `push_value` onto `@at@begin@document`/
360  // `@at@end@document`). Accept TeX source (via `TokenizeInternal`, so `@` is a
361  // letter as Perl's string path does) or a `Tokens` value (#539). The Perl
362  // CODE-ref form is not exposed; a script can `DefMacro` a CS and hook that.
363  // `AtEndOfPackage`/`AtEndOfClass`/`AtBeginDvi` are TeX-macro-level (`AddToMacro`
364  // / no-op), reachable via `RawTeX`.
365  engine.register_fn(
366    "AtBeginDocument",
367    |text: &str| -> std::result::Result<(), Box<EvalAltResult>> {
368      latexml_core::state::push_value(
369        "@at@begin@document",
370        mouth::tokenize_internal(TeXString::assembled(text.to_string())),
371      )
372      .map_err(rhai_err)
373    },
374  );
375  engine.register_fn(
376    "AtBeginDocument",
377    |tokens: Tokens| -> std::result::Result<(), Box<EvalAltResult>> {
378      latexml_core::state::push_value("@at@begin@document", tokens).map_err(rhai_err)
379    },
380  );

382    "AtEndDocument",
383    |text: &str| -> std::result::Result<(), Box<EvalAltResult>> {
384      latexml_core::state::push_value(
385        "@at@end@document",
386        mouth::tokenize_internal(TeXString::assembled(text.to_string())),
387      )
388      .map_err(rhai_err)
389    },
390  );
391  engine.register_fn(
392    "AtEndDocument",
393    |tokens: Tokens| -> std::result::Result<(), Box<EvalAltResult>> {
394      latexml_core::state::push_value("@at@end@document", tokens).map_err(rhai_err)
395    },
396  );
397  engine.register_fn(
398    "TeX",
399    |text: &str| -> std::result::Result<(), Box<EvalAltResult>> {
400      latexml_core::stomach::digest(mouth::tokenize_internal(TeXString::assembled(
401        text.to_string(),
402      )))
403      .map_err(rhai_err)?;
404      Ok(())
405    },
406  );

408    "Expand",

472      let res: Result<()> = (|| {

478    },

650  // argument reaches a constructor *closure* body as a digested `KeyVals` handle
651  // (or unit `()` when an optional set was omitted); it reaches other contexts
652  // as its TeX-source string ("k=v,k2={v2}"). `GetKeyVal` accepts both — plus
653  // unit — so a script body can `GetKeyVal(#1, "width")` uniformly. Missing key
654  // (or unit) → "".
655  engine.register_fn("GetKeyVal", |kv: Dynamic, key: &str| -> String {
656    if let Some(d) = kv.clone().try_cast::<Digested>() {
657      return match d.data() {
658        latexml_core::digested::DigestedData::KeyVals(keyval) => keyval
659          .get_value_digested(key)
660          .map(|v| v.to_string())
661          .unwrap_or_default(),
662        _ => String::new(),
663      };
664    }
665    if kv.is_string() {
666      let s = kv.into_string().unwrap_or_default();
667      return latexml_core::keyval::split_keyval_source(&s)
668        .into_iter()
669        .find(|(k, _)| k == key)

671        .unwrap_or_default();

676  // argument reaches a constructor closure as a digested `KeyVals` handle, and
677  // reaches other contexts as its `k=v,…` source string. Accept both (plus unit
678  // for an omitted optional set) so a script can `GetKeyVals(#1)` uniformly and
679  // turn the arg into a Rhai map (issue #627). Values are the digested value's
680  // string form (falling back to the raw value pre-digestion).

692      }
693      return m;
694    }
695    if kv.is_string() {
696      let s = kv.into_string().unwrap_or_default();
697      for (k, v) in latexml_core::keyval::split_keyval_source(&s) {
698        m.insert(k.into(), Dynamic::from(v));
699      }
700    }
701    m

1702  engine.register_fn(
1703    "insertXML",
1704    |_d: &mut DocProxy, nodes: rhai::Array| -> std::result::Result<(), Box<EvalAltResult>> {
1705      let mut owned = Vec::with_capacity(nodes.len());
1706      for (i, entry) in nodes.into_iter().enumerate() {

1711          ))
1712        })?);
1713      }
1714      with_doc(move |doc, _props| {
1715        let raw = owned.iter().map(|n| n.0.clone()).collect();
1716        doc.insert_nodes(raw).map_err(rhai_err)?;
1717        Ok(())

1721  // ParseXML(markup) -> [Node] — the parser exposed as its own primitive (#350),
1722  // so a script can inspect or edit markup BEFORE it reaches the document. Each
1723  // returned node OWNS the throwaway document it was parsed into (see
1724  // `NodeProxy`), so it is safe to hold, walk and mutate for as long as the
1725  // script keeps it — unlike an in-tree node from a rewrite callback. Returns an
1726  // array because a chunk may legitimately be a fragment of several siblings.
1727  engine.register_fn(
1728    "ParseXML",
1729    |markup: &str| -> std::result::Result<rhai::Array, Box<EvalAltResult>> {
1730      // Malformed markup is reported as a latexml `Error:` and yields an EMPTY
1731      // array — deliberately NOT a Rhai runtime error. A script error raised here
1732      // would propagate out of the constructor and ABORT THE WHOLE CONVERSION
1733      // (`wire.rs` maps it to a hard `Error::from`), so a single bad snippet in one
1734      // contrib binding would cost the entire document. That contradicts the
1735      // failure-isolation contract (`script_bindings_plan.md`: a bad binding
1736      // "degrades only that package"), and it would also make these two siblings
1737      // behave oppositely on the same bad input — `insertXML(markup)` already
1738      // degrades this way. Loud, but contained: the author still gets the `Error:`.
1739      match latexml_core::common::xml::parse_fragment(markup) {
1740        Ok(parsed) => {
1741          let doc = parsed.document().clone();
1742          Ok(
1743            parsed
1744              .nodes()
1745              .into_iter()
1746              .map(|n| Dynamic::from(NodeProxy::owning(n, doc.clone())))
1747              .collect(),
1748          )
1749        },
1750        Err(e) => {
1751          // The one thing that MAY still abort is `Error!` escalating to Fatal
1752          // past MAX_ERRORS — same shape as the `Error` registration above.
1753          let res: Result<()> = (|| {
1754            latexml_core::Error!(
1755              "malformed",
1756              "ParseXML",
1757              format!("could not parse XML markup: {e}")
1758            );
1759            Ok(())
1760          })();
1761          res.map_err(rhai_err)?;
1762          Ok(rhai::Array::new())
1763        },
1764      }
1765    },
1766  );

1856  // e.g. plain `\footnote`'s afterDigest routing its mark arg to `mark`/`prenote`).
1857  // The value lands as a string `Stored`, which the template interpreter renders
1858  // at attribute (`to_attribute`), content (absorb), and truth-test positions.
1859  engine.register_fn(
1860    "setProperty",
1861    |_w: &mut WhatsitProxy, key: &str, val: &str| -> std::result::Result<(), Box<EvalAltResult>> {
1862      let (ptr, mutable) = current_whatsit_entry()?;
1863      if !mutable {
1864        return Err(Box::<EvalAltResult>::from(
1865          "setProperty in a construction hook (whatsit is read-only there)",
1866        ));
1867      }
1868      // SAFETY: `ptr` is the in-flight whatsit the core published onto
1869      // WHATSIT_CTX for THIS digestion-hook body, and the `mutable` flag (just
1870      // checked) confirms it was published as writable. It is the sole live
1871      // `&mut` to this whatsit, so the re-mint is unique and non-aliasing.
1872      // AUDITED (2026-06-27, sibling of PR #248 B1): the core runs after-digest
1873      // hooks ONE-PASS and sequentially on a FRESH-LOCAL whatsit
1874      // (`definition.rs::execute_after_digest`: `for post in … { post(&mut
1875      // whatsit) }`), never re-entering a hook on the SAME whatsit — so unlike
1876      // the Document case there is no nested same-object re-mint at all. This is
1877      // the single-body parked-`&mut` + brief-re-mint pattern proven sound by
1878      // the Miri model `latexml_core::runtime_bindings_reentrancy_model`
1879      // (`reentrancy_model_single_body_sound`). (A hook that digests other
1880      // content builds DIFFERENT whatsits → different `WHATSIT_CTX` entries, not
1881      // an alias of this one.) The read-only ops above (`&*`) need no such
1882      // argument — shared refs may alias freely.
1883      let w = unsafe { &mut *ptr };
1884      w.set_property(key, val.to_string());
1885      Ok(())
1886    },
