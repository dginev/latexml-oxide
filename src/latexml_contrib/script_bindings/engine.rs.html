1//! The sandboxed Rhai engine: every `register_fn` of the binding-language
2//! surface (state, tokens, counters, definitions, package/class machinery,
3//! document/whatsit proxies). Split from `mod.rs` for navigability; each
4//! registration lowers to the same native function its compile-time macro
5//! does (`setup_binding_language.rs`).

9/// Build a sandboxed Rhai engine with the binding API registered.
10pub(super) fn make_engine() -> Engine {
11  let mut engine = Engine::new();
12  engine.set_max_operations(MAX_OPERATIONS);
13  engine.set_max_call_levels(MAX_CALL_LEVELS);
14  engine.set_max_string_size(MAX_STRING_SIZE);
15  // Registered operators must win over the built-ins for `()` vs a string (the deprecated string tests below).
16  engine.set_fast_operators(false);
17
18  // ── registration API (wired to native defs IMMEDIATELY, in script order) ──
19  engine.register_fn(
20    "DefMacro",
21    |proto: &str, body: FnPtr| -> std::result::Result<(), Box<EvalAltResult>> {
22      wire_now(|e, a| wire_macro(e, a, proto, body))
23    },
24  );
25  // String-body form — Perl's most common `DefMacro('\foo', 'bar')`, where the
26  // body is TeX source that becomes the expansion directly (no Rhai closure).
27  // Wires the same native expandable as the compile-time `DefMacro!` string
28  // form, so a `.rhai` binding reads like the `.ltxml` original. The wiring
29  // ignores the script handles, hence the throwaway `|_, _|`.
30  engine.register_fn(
31    "DefMacro",
32    |proto: &str, body: &str| -> std::result::Result<(), Box<EvalAltResult>> {
33      wire_now(|_, _| wire_macro_string(proto, body))
34    },
35  );
36  // Option-bag form (Perl's trailing `key => value`s): scalars onto
37  // `ExpandableOptions` via the shared mapper. Both the closure-body and the
38  // string-body variants accept it.
39  engine.register_fn(
40    "DefMacro",
41    |proto: &str, body: FnPtr, opts: Map| -> std::result::Result<(), Box<EvalAltResult>> {
42      wire_now(|e, a| wire_macro_opts(e, a, proto, body, opts))
43    },
44  );
45  engine.register_fn(
46    "DefMacro",
47    |proto: &str, body: &str, opts: Map| -> std::result::Result<(), Box<EvalAltResult>> {
48      wire_now(|_, _| wire_macro_string_opts(proto, body, opts))
49    },
50  );
51  engine.register_fn(
52    "DefPrimitive",
53    |proto: &str, body: FnPtr| -> std::result::Result<(), Box<EvalAltResult>> {
54      wire_now(|e, a| wire_primitive(e, a, proto, body, PrimitiveOptions::default()))
55    },
56  );
57  engine.register_fn(
58    "DefPrimitive",
59    |proto: &str, body: FnPtr, opts: Map| -> std::result::Result<(), Box<EvalAltResult>> {
60      wire_now(|e, a| {
61        wire_primitive(
62          e,
63          a,
64          proto,
65          body,
66          primitive_options_from_map(proto, opts, e, a)?,
67        )
68      })
69    },
70  );
71  // Class/package option. Mirrors the `DeclareOption!` macro's lowering
72  // (setup_binding_language.rs): push the name onto `@declaredoptions` and
73  // define a `\ds@<opt>` primitive carrying the body. Installed immediately,
74  // so a following `ProcessOptions()` in the same script sees it.
75  engine.register_fn(
76    "DeclareOption",
77    |opt: &str, body: FnPtr| -> std::result::Result<(), Box<EvalAltResult>> {
78      wire_now(|e, a| wire_option(e, a, opt, body))
79    },
80  );
81  // String-body form: `DeclareOption('opt', '\xdef...')` — the body is TeX,
82  // installed as a token-expansion macro (Perl's `DeclareOption($o, $string)`).
83  engine.register_fn(
84    "DeclareOption",
85    |opt: &str, body: &str| -> std::result::Result<(), Box<EvalAltResult>> {
86      wire_now(|_, _| wire_option_string(opt, body))
87    },
88  );
89  // Default-option handler: `DeclareOption(sub {...})` with no name — Perl's
90  // `DeclareOption(undef, sub {...})`, defining `\default@ds` for any option not
91  // otherwise declared.
92  engine.register_fn(
93    "DeclareOption",
94    |body: FnPtr| -> std::result::Result<(), Box<EvalAltResult>> {
95      wire_now(|e, a| wire_option_default(e, a, body))
96    },
97  );
98
99  // ── free-function helpers (prelude names) over core datatypes ──
100  // `Tokens` is registered as an opaque pass-through handle (Tier-2): a script
101  // produces it with Tokenize and hands it to Digest without inspecting it.
102  engine.register_type_with_name::<Tokens>("Tokens");
103  // `assembled`: the argument is TeX the script author wrote — the runtime twin
104  // of a `Tokenize!("…")` literal. A script cannot hand us a flattened `Tokens`
105  // (Rhai `Tokens` is opaque and has no string conversion registered).
106  engine.register_fn("Tokenize", |s: &str| -> Tokens {
107    mouth::tokenize(TeXString::assembled(s.to_string()))
108  });
109  engine.register_fn("TokenizeInternal", |s: &str| -> Tokens {
110    mouth::tokenize_internal(TeXString::assembled(s.to_string()))
111  });
112  // `Digest` returns the digested handle (Perl returns the digested box), so a
113  // script can `ToString(Digest(...))` — e.g. a default-option handler reading
114  // `\CurrentOption`. Callers that ignore the return (the common side-effect
115  // use) are unaffected.
116  engine.register_fn(
117    "Digest",
118    |t: Tokens| -> std::result::Result<Digested, Box<EvalAltResult>> {
119      latexml_core::binding::content::binding_digest(t).map_err(rhai_err)
120    },
121  );
122  engine.register_fn(
123    "Digest",
124    |t: Token| -> std::result::Result<Digested, Box<EvalAltResult>> {
125      latexml_core::binding::content::binding_digest(Tokens::new(vec![t])).map_err(rhai_err)
126    },
127  );
128  // A string is TeX source, TokenizeInternal'd (Package.pm:393).
129  engine.register_fn(
130    "Digest",
131    |s: &str| -> std::result::Result<Digested, Box<EvalAltResult>> {
132      latexml_core::binding::content::binding_digest(mouth::tokenize_internal(
133        TeXString::assembled(s.to_string()),
134      ))
135      .map_err(rhai_err)
136    },
137  );
138  // Perl's `Tokens(@tokens)` (Tokens.pm:29): an array of `Token`s and `Tokens`, flattened; also
139  // `Tokens()` and `Tokens(a, …)` with up to four items, Rhai having no variadic functions.
140  engine.register_fn("Tokens", || Tokens::new(Vec::new()));
141  engine.register_fn("Tokens", |a: Dynamic| tokens_of(vec![a]));
142  engine.register_fn("Tokens", |a: Dynamic, b: Dynamic| tokens_of(vec![a, b]));
143  engine.register_fn("Tokens", |a: Dynamic, b: Dynamic, c: Dynamic| {
144    tokens_of(vec![a, b, c])
145  });
146  engine.register_fn(
147    "Tokens",
148    |a: Dynamic, b: Dynamic, c: Dynamic, d: Dynamic| tokens_of(vec![a, b, c, d]),
149  );
150  engine.register_fn("Tokens", tokens_of);
151  // Perl's token constructors and catcode constants (Token.pm:26-37, :45-84); a Perl
152  // constant (`T_MATH`, `CC_ACTIVE`) is a zero-argument function here.
153  engine.register_type_with_name::<Token>("Token");
154  values::register_values(&mut engine);
155  // A token value in a string context (`"(" + ReadArg() + ")"`, `${…}`) is its TeX source, as `UnTeX` gives it.
156  engine.register_fn("to_string", |t: &mut Token| Tokens::new(vec![*t]).untex());
157  engine.register_fn("to_string", |t: &mut Tokens| t.clone().untex());
158  engine.register_fn("T_CS", |name: &str| -> Token { latexml_core::T_CS!(name) });
159  engine.register_fn("T_LETTER", |c: &str| -> Token {
160    Token::new(c, Catcode::LETTER)
161  });
162  engine.register_fn("T_OTHER", |c: &str| -> Token {
163    Token::new(c, Catcode::OTHER)
164  });
165  engine.register_fn("T_ACTIVE", |c: &str| -> Token {
166    Token::new(c, Catcode::ACTIVE)
167  });
168  engine.register_fn(
169    "Token",
170    |s: &str, cc: i64| -> std::result::Result<Token, Box<EvalAltResult>> {
171      match u8::try_from(cc) {
172        Ok(cc) if cc <= Catcode::ARG as u8 => Ok(Token::new(s, Catcode::from(cc))),
173        _ => Err(format!("Token: no catcode {cc}").into()),
174      }
175    },
176  );
177  engine.register_fn("Token", |s: &str| -> Token {
178    Token::new(s, Catcode::OTHER)
179  });
180  // Token.pm:81-95: `T_COMMENT` prefixes the `%`; `T_ARG` takes 1-9 or a digit Token.
181  engine.register_fn("T_COMMENT", |c: &str| -> Token {
182    Token::new(format!("%{c}"), Catcode::COMMENT)
183  });
184  engine.register_fn("T_MARKER", |t: &str| -> Token {
185    Token::new(t, Catcode::MARKER)
186  });
187  engine.register_fn(
188    "T_ARG",
189    |v: Dynamic| -> std::result::Result<Token, Box<EvalAltResult>> {
190      let n = match v.clone().try_cast::<Token>() {
191        Some(t) => t.to_string().parse::<i64>().unwrap_or(0),
192        None => v
193          .as_int()
194          .map_err(|got| format!("T_ARG: expected an int or a Token, got {got}"))?,
195      };
196      if (1..=9).contains(&n) {
197        Ok(Token::new(n.to_string(), Catcode::ARG))
198      } else {
199        Err(format!("T_ARG: value should be #1-#9, got {n}").into())
200      }
201    },
202  );
203  for (name, token) in [
204    ("T_BEGIN", latexml_core::T_BEGIN!()),
205    ("T_END", latexml_core::T_END!()),
206    ("T_MATH", latexml_core::T_MATH!()),
207    ("T_ALIGN", latexml_core::T_ALIGN!()),
208    ("T_PARAM", latexml_core::T_PARAM!()),
209    ("T_SUB", latexml_core::T_SUB!()),
210    ("T_SUPER", latexml_core::T_SUPER!()),
211    ("T_SPACE", latexml_core::T_SPACE!()),
212    ("T_CR", latexml_core::T_CR!()),
213  ] {
214    engine.register_fn(name, move || -> Token { token });
215  }
216  for (name, cc) in [
217    ("CC_ESCAPE", Catcode::ESCAPE),
218    ("CC_BEGIN", Catcode::BEGIN),
219    ("CC_END", Catcode::END),
220    ("CC_MATH", Catcode::MATH),
221    ("CC_ALIGN", Catcode::ALIGN),
222    ("CC_EOL", Catcode::EOL),
223    ("CC_PARAM", Catcode::PARAM),
224    ("CC_SUPER", Catcode::SUPER),
225    ("CC_SUB", Catcode::SUB),
226    ("CC_IGNORE", Catcode::IGNORE),
227    ("CC_SPACE", Catcode::SPACE),
228    ("CC_LETTER", Catcode::LETTER),
229    ("CC_OTHER", Catcode::OTHER),
230    ("CC_ACTIVE", Catcode::ACTIVE),
231    ("CC_COMMENT", Catcode::COMMENT),
232    ("CC_INVALID", Catcode::INVALID),
233    ("CC_CS", Catcode::CS),
234    ("CC_MARKER", Catcode::MARKER),
235    ("CC_ARG", Catcode::ARG),
236  ] {
237    engine.register_fn(name, move || -> i64 { cc as i64 });
238  }

293  // The Perl `@values` list-value family (Package.pm L265-279 -> State.pm L218):
294  // a value-table key holds a queue that these push/pop/unshift/shift. Always
295  // GLOBAL scope, auto-vivifying the list if absent — matching Perl, which takes
296  // no scope argument. Perl's `PushValue` is untyped and the Rust State layer
297  // holds any `Stored`, so `PushValue`/`UnshiftValue` take any Rhai-representable
298  // value (string/int/float/bool/`Tokens`/digested handle) and `PopValue`/
299  // `ShiftValue` return it with its type preserved (#540). `LookupValue` reads
300  // the whole queue back as a Rhai array. As of #561 `SEARCHPATHS` is itself a
301  // group-scoped value-table list (like `GRAPHICSPATHS`), so `PushValue`/
302  // `UnshiftValue` reach it too; the dedicated `PrependSearchPath`/
303  // `AppendSearchPath` below are the ergonomic front for input dirs.
304  engine.register_fn("PushValue", |k: &str, v: Dynamic| {
305    // Returns Ok(()) always; the wrong-type BUG path self-reports via Error!.
306    let _ = latexml_core::state::push_value(k, dynamic_to_stored(v));
307  });
308  engine.register_fn("UnshiftValue", |k: &str, v: Dynamic| {
309    latexml_core::state::unshift_value(k, vec![dynamic_to_stored(v)]);
310  });
311  // Perl's `PushValue($key, @values)` / `UnshiftValue($key, @values)` (Package.pm:265/275), up to four values.
312  for arity in 2..=4 {

314  }

316    match latexml_core::state::pop_value(k) {

324      _ => Dynamic::UNIT,

2435              .nodes()

2446              "malformed",

2663/// Perl's `Tokens(@tokens)` (Tokens.pm:29): `Token`s, `Tokens` and arrays of them, flattened.

2677          item.type_name()
