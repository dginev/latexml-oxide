1//! xr.sty and xr-hyper.sty — references into another document. Perl ships no binding.
2use latexml_core::{
3  binding::content::{FindFileOptions, find_file},
4  common::error::emit_warn,
5};
6use latexml_engine::latex_constructs::{brace_groups, record_newlabel};
7use latexml_package::prelude::*;
8
9#[rustfmt::skip]
10LoadDefinitions!({
11  // xr.sty:44-47 (v6; xr-hyper.sty:38-41 alike): `\externaldocument[prefix][nocite]{file}[url]` reads `file.aux` and
12  // every `.aux` it `\@input`s, each `\newlabel{l}{{text}{page}…}` defining `\r@<prefix>l` (`\XR@test`), so `\ref{l}`
13  // prints the other document's number. The raw package's `\r@…` never reached our `\ref`, a constructor post resolves:
14  // each such reference was a missing label where pdflatex prints "Fig. S3" (1307.3388, 1608.05314; 21 of the census's 125
15  // `\externaldocument` papers ship the external `.aux`, the rest print "??" in pdflatex too). The labels are recorded
16  // for `\ref` and `\eqref`, their text typeset as TeX (sect11.rs `record_external_label`), and cleveref's `<label>@cref`
17  // records and hyperref's anchors for `\cref` and `\autoref`, which name the type (2412.14256, 2508.18808, 2402.05535);
18  // the `\bibcite`s v6 imports too (unless the second argument says `nocite`) are not. Where the URL is given its anchors are linked; the default, `file.pdf`, names no
19  // page of ours. The file is Semiverbatim so a name with `_` (`ex_supplement`) does not leak the underscore into text
20  // mode (witness 2402.12241); xr v5 read `[prefix]{file}` only (xr-2023-07-04.sty:46-48), which this reads too.
21  DefPrimitive!("\\externaldocument[][] Semiverbatim OptionalSemiverbatim", sub[(prefix, _nocite, file, url)] {
22    let prefix = prefix.map(|tokens| do_expand(tokens).map(|t| t.to_string())).transpose()?.unwrap_or_default();
23    let url = url.map(|tokens| tokens.to_string());
24    import_external_document(prefix.trim(), file.to_string().trim(), url.as_deref().map(str::trim));
25  });
26  Let!("\\externalcitedocument", "\\externaldocument");
27});
28
29/// `\XR@@`/`\XR@loop` (xr.sty:57-80): the labels of `file.aux` and of the `.aux` files it `\@input`s (each named from
30/// the directory of the file that inputs it, `\filename@area`), recorded under `prefix`; a missing file warns, as xr's
31/// "No file …  LABELS NOT IMPORTED" does.
32fn import_external_document(prefix: &str, file: &str, url: Option<&str>) {
33  // `\set@curr@file` unquotes the name (latex.ltx:9654-9666); `\XR@next\@curr@file.aux` appends `.aux` to all of it
34  let mut queue = vec![s!("{}.aux", file.replace('"', ""))];
35  let mut seen = Vec::new();
36  while let Some(aux) = queue.pop() {
37    if seen.contains(&aux) {
38      continue;
39    }
40    seen.push(aux.clone());
41    let found = find_file(
42      &aux,
43      Some(FindFileOptions {
44        ext_type: Some("aux".into()),
45        ..FindFileOptions::default()
46      }),
47    );
48    let Some(text) = found.and_then(|path| std::fs::read_to_string(path).ok()) else {
49      emit_warn(
50        "missing_file",
51        &aux,
52        &s!("No file {aux}; labels not imported (xr)"),
53      );
54      continue;
55    };
56    let area = aux.rfind('/').map_or("", |slash| &aux[..=slash]);
57    for line in text.lines() {
58      if let Some(rest) = line.trim_start().strip_prefix("\\newlabel") {
59        let mut groups = brace_groups(rest);
60        if let (Some(key), Some(value)) = (groups.next(), groups.next()) {
61          record_newlabel(&s!("{prefix}{key}"), value, url);
62        }
63      } else if let Some(rest) = line.trim_start().strip_prefix("\\@input")
64        && let Some(name) = brace_groups(rest).next()
65      {
66        queue.push(s!("{area}{name}"));
67      }
68    }
69  }
70}
