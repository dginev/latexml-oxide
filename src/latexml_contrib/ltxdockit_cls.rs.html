1//! ltxdockit.cls — Philipp Lehman's "LaTeX documentation kit" class
2//! (TL `texmf-dist/tex/latex/ltxdockit/ltxdockit.cls`, v1.2d 2011/03/25),
3//! the class behind the biblatex-ecosystem manuals (biblatex, etoolbox,
4//! abraces, … — 12+ TL doc bundles in the perfect-kernel corpus).
5//!
6//! Why a binding exists (user directive 2026-08-31): the raw class does
7//! `\renewrobustcmd*{\titlepage}[1]{\setkeys{ltd@ttp}{#1}}`, but `\titlepage`
8//! is a LOCKED kernel CS, so under raw interpretation the redefinition is
9//! refused ("Ignoring redefinition of \titlepage") and the whole keyval
10//! title machinery breaks (`undefined \ltd@title@title` ×6 per document;
11//! Perl 0.8.8 fails identically, plus a fatal on some members). Locks stay
12//! (user decision, same day) — the resolution for lock-conflicting classes
13//! is a `_cls.rs` binding that sets the desired definitions explicitly from
14//! the Rust layer, where binding loads run UNLOCKED.
15//!
16//! The mapping is SEMANTIC, not a replay of the print title page: the
17//! `\titlepage` keys feed standard frontmatter (`\title`/`\author`/`\date`,
18//! subtitle as `ltx:subtitle`, url/email/revision as classed notes) and
19//! `\printtitlepage` becomes `\maketitle` — so the manuals get real
20//! `<ltx:title>`/`<ltx:creator>` markup instead of a centered group.
21//!
22//! Base class: the real ltxdockit.def passes `11pt,a4paper,DIV9,…` to
23//! scrartcl and loads it; we route to our scrartcl binding the same way.
24//! The `ltxdockit` PACKAGE (ltxdockit.sty — lstnewenvironment-based example
25//! envs, doc markup commands) has no binding and raw-loads exactly as
26//! before; only the lock-conflicting CLASS layer is bound here.
27use latexml_package::prelude::*;
28
29LoadDefinitions!({
30  LoadClass!("scrartcl");
31
32  // ltxdockit.cls dependency block (L26-33). fontenc[T1]/hypcap options are
33  // presentation-only; the packages themselves carry the definitions the
34  // manuals use (`\href`, `\Hurl`, etoolbox's test/hook suite, multicols).
35  RequirePackage!("etoolbox");
36  RequirePackage!("multicol");
37  RequirePackage!("keyval");
38  RequirePackage!("textcomp");
39  RequirePackage!("ltxdockit"); // the .sty half — raw-loads (no binding)
40  RequirePackage!("hyperref");
41  RequirePackage!("hypcap");
42
43  // L38-39: \email{addr} → mailto link (the raw version goes through
44  // \hyper@normalise, an href-internal we don't expose).
45  DefMacro!("\\email{}", "\\href{mailto:#1}{#1}");
46  // L35-36: \fnurl[pre]{url} — footnote with a linked URL.
47  DefMacro!("\\fnurl[]{}", "\\footnote{#1\\url{#2}}");
48
49  // L57-67: the keyval title page, set explicitly at the Rust layer (the
50  // reason this binding exists — see module docs). Each key lands as real
51  // frontmatter; \printtitlepage (L69-91) then just fires \maketitle.
52  RawTeX!(
53    r"\define@key{ltd@ttp}{title}{\title{#1}}
54      \define@key{ltd@ttp}{subtitle}{\@add@frontmatter{ltx:subtitle}{#1}}
55      \define@key{ltd@ttp}{url}{\@add@frontmatter{ltx:note}[role=url]{\url{#1}}}
56      \define@key{ltd@ttp}{author}{\author{#1}}
57      \define@key{ltd@ttp}{email}{\@add@frontmatter{ltx:note}[role=email]{\email{#1}}}
58      \define@key{ltd@ttp}{revision}{\@add@frontmatter{ltx:note}[role=revision]{Version #1}}
59      \define@key{ltd@ttp}{date}{\date{#1}}"
60  );
61  DefMacro!("\\titlepage{}", "\\setkeys{ltd@ttp}{#1}");
62  DefMacro!("\\printtitlepage", "\\maketitle");
63  def_macro_noop("\\titlefont")?;
64
65  // L95-131: RCS tag extraction (`\rcsid{$Id: … $}` → \rcsfile /
66  // \rcsrevision / \rcsdate / …). Faithful raw port — delimiter-matched
67  // \defs, no lock conflicts — minus the `\write\@auxout` indirection: the
68  // aux round-trip only exists so the values survive to the NEXT LaTeX run;
69  // we define them directly in this one — the revision's date too
70  // (`\ltd@setdate`, ltxdockit.cls:129-137, read back from the aux: `\year`,
71  // `\month`, `\day`, `\time`, which `\rcstoday` = `\today` prints; it
72  // printed the conversion day; set where `\rcsid` stands, pdflatex at
73  // `\begin{document}`: a `\today` printed before a late `\rcsid` differs).
74  // Repro sectioning-frontmatter/ltxdockit_rcsid_sets_the_date.
75  RawTeX!(
76    r"\providecommand*{\rcsfile}{[rcsfile]}
77      \providecommand*{\rcsrevision}{[revision]}
78      \providecommand*{\rcsdate}{[date]}
79      \providecommand*{\rcstime}{[time]}
80      \providecommand*{\rcsstate}{[state]}
81      \providecommand*{\rcsauthor}{[author]}
82      \providecommand*{\rcslocker}{[unlocked]}
83      \providecommand*{\rcstoday}{\today}
84      \providecommand*{\rcsid}[1]{\ifblank{#1}{}{\ltd@rcsid@i#1}}
85      \def\ltd@rcsid@i$Id#1${\ifblank{#1}{}{\ltd@rcsid@ii#1&}}
86      \def\ltd@rcsid@ii#1#2&{\ifblank{#1}{}{\ltd@rcsid@iii#2&}}
87      \def\ltd@rcsid@iii#1 #2 #3 #4&{\gdef\rcsfile{#2}\gdef\rcsrevision{#3}\ltd@rcsid@iv#4&}
88      \def\ltd@rcsid@iv#1/#2/#3 #4:#5:#6 #7&{\gdef\rcsdate{#1/#2/#3}\gdef\rcstime{#4:#5:#6}%
89        \ltd@setdate{#1}{#2}{#3}{#4}{#5}\ltd@rcsid@v#7&}
90      \def\ltd@rcsid@v#1 #2 #3&{\gdef\rcsauthor{#1}\gdef\rcsstate{#2}\ifblank{#3}{}{\ltd@rcsid@vi#3&}}
91      \def\ltd@rcsid@vi#1 &{\gdef\rcslocker{#1}}
92      \def\ltd@setdate#1#2#3#4#5{\global\year=#1 \global\month=#2 \global\day=#3 \global\time=#4
93        \global\multiply\time by 60 \global\advance\time by #5 }"
94  );
95
96  // ltxdockit.def L17-37: \AtBeginToc/\AtEndToc/\AtBeginLot/\AtEndLot hook
97  // registries. Defined for real (documents may append to them); the def's
98  // own use — wrapping the TOC in a 2-column multicols — is print layout and
99  // deliberately not replayed.
100  RawTeX!(
101    r"\newcommand*{\@begintochook}{}\newcommand*{\@endtochook}{}
102      \newcommand*{\@beginlothook}{}\newcommand*{\@endlothook}{}
103      \newcommand*{\AtBeginToc}{\g@addto@macro\@begintochook}
104      \newcommand*{\AtEndToc}{\g@addto@macro\@endtochook}
105      \newcommand*{\AtBeginLot}{\g@addto@macro\@beginlothook}
106      \newcommand*{\AtEndLot}{\g@addto@macro\@endlothook}"
107  );
108
109  // L135 (end of class): the DFSG config file — it defines the name/logo
110  // macro suite the manuals use in running text (\etex, \latex, \lppl,
111  // \biber, …) and appends to the Toc hooks, so it must load AFTER they
112  // exist (real class order: ltxdockit.def first, cfg last). Raw-loads from
113  // the texmf tree exactly as the real class does.
114  RawTeX!(r"\InputIfFileExists{ltxdockit.cfg}{}{}");
115});
