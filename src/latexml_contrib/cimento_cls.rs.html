1//! cimento.cls (Il Nuovo Cimento, Società Italiana di Fisica): the papers ship the class (it is not in TeX Live) —
2//! or a renamed copy (`cimento_modif.cls`, 2203.13736), which the prefix fallback answers with this binding — and it is
3//! read raw, with what the kernel's frontmatter needs of it. A paper whose source lacks the class falls to OmniBus, with
4//! the loader's missing-file Warning.
5use latexml_package::prelude::*;
6
7const CIMENTO: crate::shipped_class::RawRead = crate::shipped_class::RawRead {
8  shipped: true,
9  ..crate::shipped_class::RawRead::new("cimento", "cls")
10};
11
12LoadDefinitions!({
13  if !crate::shipped_class::read_raw(&CIMENTO)? {
14    return Ok(());
15  }
16  // cimento.cls:746 `\def\ETC{\unskip\unscrunch|}`, inside its `\maketitle`: in an author list (or a title) `\ETC` marks
17  // where the running head cuts the list short with "etc." (:766 `\def\ETC{\unskip\ etc.\scrunch|}`), and on the title
18  // page it sets the whole list (`\unscrunch|#1|{#1}` over the rest of `\@author@`, which `\author` ends with `|`,
19  // :680). The kernel's `\author` and `\title` keep the whole text, so there it is the `\unskip`. A variant's own
20  // `\ETC` (:1361, :1679) stands. Witnesses astro-ph0502331 (`\author{K.-I.~Nishikawa\from{ins:x}\ETC, …}`),
21  // astro-ph0506233, 1804.04677 (`\title{… physics \ETC\ from …}`).
22  RawTeX!(r"\providecommand{\ETC}{\unskip}");
23});
