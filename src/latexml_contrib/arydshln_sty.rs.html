111/// The between-column material of arydshln's dashed column rules (`:` and
112/// `;{..}`): a `\vrule` classed `ltx_border_r_dashed`, which the stylesheet
113/// draws dashed (ar5iv-bindings arydshln.sty.ltxml:21-24).
114fn dashed_rule_between_columns() -> Vec<Token> {
115  vec![
116    T_CS!("\\vrule"),
117    T_CS!("\\@ADDCLASS"),
118    T_BEGIN!(),
119    T_OTHER!("ltx_border_r_dashed"),
120    T_END!(),
121    T_CS!("\\relax"),
122  ]
123}
