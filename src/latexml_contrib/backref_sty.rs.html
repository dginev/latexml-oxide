1//! backref.sty — back-references from bibliography to citations
2//! (part of the Hyperref Bundle).
3//!
4//! backref adds a "Cited on pages X, Y" link at the end of each
5//! bibliography entry, pointing back to the citation locations.
6//! It does this by redefining `\@bibitem` / `\@lbibitem` to wrap
7//! the bibitem body with a `\BR@bibitem` hook.
8//!
9//! Raw-loading backref.sty into our engine is fragile: backref's
10//! redefinitions of `\@bibitem`/`\@lbibitem` chain with paper-
11//! bundled bibliography helpers (e.g. myharvard's `\harvarditem`
12//! which writes to `.aux` then calls `\item[]`) — the resulting
13//! expansion loops indefinitely (witness 1107.0498: amsart +
14//! `\usepackage[hyperpageref]{backref}` + paper-local myharvard.sty
15//! → Convert TIMEOUT in `.bbl` processing; Perl handles the same
16//! input with the same 4 undefined-macro errors but completes).
17//!
18//! Perl LaTeXML has no backref binding either; with default
19//! `INCLUDE_STYLES=false`, raw backref.sty is not loaded — Perl
20//! emits "missing binding" and continues without back-references.
21//!
22//! Match Perl: stub the user-facing API as no-ops. We lose the
23//! back-reference links, but the bibliography itself renders
24//! correctly.
25use latexml_package::prelude::*;
26
27LoadDefinitions!({
28  // backref.sty:196 reads its options with `\ProcessKeyvalOptions`, which marks them processed: `\@curroptions` stays as it was.
29  key_options_processed()?;
30  // Loaded through hyperref's `backref`/`pagebackref` option (hyperref.sty:4587-4589, which the hyperref binding
31  // flags), the missing back-reference lists are generated navigation the author did not write, and Perl ignores
32  // the option silently: an Info. A document's own `\usepackage{backref}` keeps the Warning (OXIDIZED_DESIGN_DIVERGENCES
33  // #547; witness 2203.07162).
34  let message = "backref.sty is minimally stubbed — back-references from bibliography to citations are not rendered.";
35  if lookup_bool("BACKREF_LOADED_BY_HYPERREF") {
36    Info!("missing_file", "backref.sty", message);
37  } else {
38    Warn!("missing_file", "backref.sty", message);
39  }
40  // backref.sty:61-62 and :483 load kvoptions, kvsetkeys and rerunfilecheck.
41  RequirePackage!("kvoptions");
42  RequirePackage!("kvsetkeys");
43  // rerunfilecheck has no binding: read raw only where raw loading is on (see mdframed_sty.rs).
44  if lookup_bool("INCLUDE_STYLES") {
45    RequirePackage!("rerunfilecheck");
46  }
47  // backref's user-facing API: setup, format, print. backref.sty:67 `\backrefsetup` = `\kvsetkeys{backref}`.
48  DefMacro!("\\backrefsetup", "\\kvsetkeys{backref}");
49  DefMacro!("\\backrefprint", "");
50  // backref.sty:95-148 the language hooks, which set the words of the back-reference line (documents
51  // `\renewcommand` them, `\backrefsep` and the like); :195 English is the default. :207-208 the
52  // `\ifbackrefparscan` switch a bibliography sets (witness 1809.07574: `\backrefparscanfalse` in
53  // `{thebibliography}`, with `[pagebackref]{hyperref}` loading backref). :78-92 `\backref` by the options
54  // `pageref` (the default, :195), `ref`, `hyperref` and `hyperpageref` (the two hyperref's options pass), other
55  // options and `\backrefsetup` keys accepted and dropped; :270-277 `\backrefalt`/`\backrefentrycount`, which
56  // documents redefine to format the line; the line itself is not rendered, so they are never called here.
57  RawTeX!(
58    r"\def\backrefenglish{\def\backrefpagesname{pages}\def\backrefsectionsname{sections}\def\backrefsep{, }%
59\def\backreftwosep{ and~}\def\backreflastsep{, and~}}
60\def\backrefgerman{\def\backrefpagesname{Seiten}\def\backrefsectionsname{Abschnitte}\def\backrefsep{, }%
61\def\backreftwosep{ und~}\def\backreflastsep{ und~}}
62\def\backreffrench{\def\backrefpagesname{pages}\def\backrefsectionsname{sections}\def\backrefsep{, }%
63\def\backreftwosep{ et~}\def\backreflastsep{ et~}}
64\def\backrefspanish{\def\backrefpagesname{p\'aginas}\def\backrefsectionsname{secciones}\def\backrefsep{, }%
65\def\backreftwosep{ y~}\def\backreflastsep{ y~}}
66\def\backrefbrazil{\def\backrefpagesname{p\'aginas}\def\backrefsectionsname{se\c c\~oes}\def\backrefsep{, }%
67\def\backreftwosep{ e~}\def\backreflastsep{ e~}}
68\def\backrefafrikaans{\def\backrefpagesname{bladsye}\def\backrefsectionsname{afdelings}\def\backrefsep{, }%
69\def\backreftwosep{ en~}\def\backreflastsep{ en~}}
70\def\backrefitalian{\def\backrefpagesname{pagine}\def\backrefsectionsname{sezioni}\def\backrefsep{, }%
71\def\backreftwosep{ e~}\def\backreflastsep{ e~}}
72\backrefenglish
73\SetupKeyvalOptions{family=backref,prefix=BR@}
74\DeclareBoolOption[false]{verbose}
75\long\def\page@backref#1#2#3{#1}
76\long\def\section@backref#1#2#3{#2}
77\long\def\hyper@section@backref#1#2#3{\hyperlink{#3}{#2}}
78\long\def\hyper@page@backref#1#2#3{\hyperlink{page.#1}{#1}}
79\DeclareVoidOption{pageref}{\def\backref{\backrefpagesname\ }\let\backrefxxx\page@backref}
80\DeclareVoidOption{ref}{\def\backref{\backrefsectionsname\ }\let\backrefxxx\section@backref}
81\DeclareVoidOption{hyperref}{\def\backref{}\let\backrefxxx\hyper@section@backref}
82\DeclareVoidOption{hyperpageref}{\def\backref{}\let\backrefxxx\hyper@page@backref}
83\DeclareBoolOption[true]{enable}
84\DeclareComplementaryOption{disable}{enable}
85\kv@set@family@handler{backref}{}
86\DeclareDefaultOption{}
87\backrefsetup{pageref}
88\ProcessKeyvalOptions*\relax
89\newif\ifbackrefparscan\backrefparscantrue
90\@ifundefined{backrefalt}{\def\backrefalt#1#2#3#4{}}{}
91\@ifundefined{backrefentrycount}{\def\backrefentrycount#1#2{#1}}{}"
92  );
93  // backref.sty:457-482 `\Hy@backout`, the hook each citation calls (hyperref's `\hyper@natlinkstart`, backref's
94  // `\BR@citex`): it writes the back-cite record — "Doc-Start" where no label is current — to the .aux for the next run.
95  // The records are never read back here (the lists are not rendered), but documents patch it: `\patchcmd{\Hy@backout}
96  // {Doc-Start}{\@currentHref}{}{\errmessage{I can't seem to patch backref}}` failed on hyperref's `\@gobble` stand-in
97  // (2405.13143, 2601.06492, 2004.14852).
98  RawTeX!(
99    r"\def\Hy@backout#1{\@bsphack
100\ifBR@enable\ifBR@verbose\PackageInfo{backref}{Back cite \string`#1\string'}\fi
101\ifx\@empty\@currentlabel\protected@write\@auxout{}{\string\@writefile{brf}{\string\backcite{#1}{{\thepage}{(document)}{Doc-Start}}}}%
102\else\protected@write\@auxout{}{\string\@writefile{brf}{\string\backcite{#1}{{\thepage}{\@currentlabel}{\@currentHref}}}}\fi
103\else\ifBR@verbose\PackageInfo{backref}{Disabled back cite \string`#1\string'}\fi\fi
104\@esphack}"
105  );
106});
