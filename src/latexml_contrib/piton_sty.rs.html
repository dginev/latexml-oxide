1//! piton.sty — François Pantigny's LPEG-based code highlighter, a
2//! LuaLaTeX-ONLY package (its .sty lives under `tex/lualatex/` and raises
3//! "LuaLaTeX is mandatory" as a FATAL on every other engine — which killed
4//! whole documents here, e.g. the nicematrix manual, whose `{Code}` example
5//! environment is built with `\NewPitonEnvironment`).
6//!
7//! Binding scope (user directive 2026-08-31 on LuaTeX-only environments):
8//! CONTENT-preserving verbatim treatment. A piton environment's body is
9//! program text; it is captured raw and rendered through the listings
10//! display engine, exactly as our minted/fancyvrb-class degradations do —
11//! syntax COLORING is presentation (a future refinement could run piton.lua
12//! itself through the texlua bridge — it is pure LPEG, which texlua ships —
13//! and consume its `tex.sprint` stream; the texlua bridge's tiers are in
14//! docs/perfect_kernel/LUA_REBINDING.md).
15use latexml_package::{
16  package::listings_sty::{listings_read_raw_lines, lst_process_display},
17  prelude::*,
18};
19
20#[rustfmt::skip]
21LoadDefinitions!({
22  // piton.sty:127 reads its options with `\ProcessKeyOptions`, which marks them processed: `\@curroptions` stays as it was.
23  key_options_processed()?;
24  RequirePackage!("listings");
25
26  // Shared engine for every piton-style environment: absorb an optional
27  // `[keys]`, capture the body raw to the matching \end, display it as a
28  // listing, and re-inject the \end so the environment machinery balances.
29  DefPrimitive!("\\lx@piton@env{}", sub[(name)] {
30    let env_name = name.to_string();
31    let _opt = read_optional(None)?;
32    bgroup();
33    let text = listings_read_raw_lines(&env_name);
34    unread(Tokenize!(TeXString::assembled(s!("\\end{{{env_name}}}"))));
35    unread(Tokens::new(lst_process_display(None, &text)));
36  });
37
38  // \NewPitonEnvironment{name}{argspec}{before}{after} (piton doc §"New
39  // environments"): creates a verbatim environment whose body piton
40  // highlights; before/after are layout wrappers (minipage, width setup) —
41  // presentation, not absorbed into the captured code. The created env's
42  // own args (argspec, typically `O{}`) are covered by the shared engine's
43  // optional-absorb.
44  DefPrimitive!("\\NewPitonEnvironment{}{}{}{}", sub[(name, _argspec, _before, _after)] {
45    let env = name.to_string().trim().to_string();
46    def_macro(
47      T_CS!(s!("\\{env}")),
48      None,
49      Tokens!(T_CS!("\\lx@piton@env"), T_BEGIN!(), ExplodeText!(&env), T_END!()),
50      None,
51    )?;
52    def_macro(T_CS!(s!("\\end{env}")), None, Tokens!(), None)?;
53  });
54
55  // {Piton} — the package's own default environment.
56  RawTeX!(r"\NewPitonEnvironment{Piton}{}{}{}");
57
58  // \piton{code} — inline code (braced form; verbatim-delimited forms
59  // degrade through Semiverbatim).
60  DefMacro!("\\piton Semiverbatim", "\\texttt{#1}");
61  // \PitonInputFile[opt]{file} — display a source file.
62  DefMacro!("\\PitonInputFile []{}", "\\lstinputlisting{#2}");
63  // Configuration surface — styling keys (language=, width=, splittable=…).
64  def_macro_noop("\\PitonOptions{}")?;
65  def_macro_noop("\\SetPitonStyle{}")?;
66  def_macro_noop("\\SetPitonIdentifier{}{}")?;
67  def_macro_noop("\\PitonClearUserStyles")?;
68  def_macro_noop("\\NewPitonLanguage{}{}")?;
69});
