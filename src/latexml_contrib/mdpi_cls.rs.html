1//! Stub for MDPI journal class (Definitions/mdpi.cls, bundled by users).
2//!
3//! Real mdpi.cls L20-50 loads article + many packages including hyperref,
4//! url, booktabs, ragged2e (for \justify), cleveref (the copies since 2023). Mirror those so
5//! papers using \href, \hypersetup, \url, \justify, \crefrangelabelformat
6//! don't error out. Witness 2410.21443.

671/// The DOI the copy `class` (of mdpi.cls) gives an article by default: the journals whose DOI is the article number
672/// alone (2208.12066's :515-528 `\ifthenelse{\equal{\@journal}{molbank} \OR …}{\doinum{10.3390/\@articlenumber}}`), and
673/// every other journal's, the last default `\doinum` (:529), when it is made of the journal's DOI abbreviation, the
674/// volume, issue and article number and the class's padding only (6 copies pad with siunitx's `\num`, which no `\edef`
675/// expands: none).
676fn default_doi(class: &str) -> Option<(Vec<String>, String)> {
677  const ARTICLE_ONLY: &str = "10.3390/\\@articlenumber";
678  const PARTS: [&str; 7] = [
679    "@doiabbr",
680    "@pubvolume",
681    "@issuenum",
682    "@articlenumber",
683    "twodigits",
684    "fourdigits",
685    "fivedigits",
686  ];
687  let code = crate::shipped_class::class_code(class);
688  let defaults: Vec<(usize, String)> = code
689    .match_indices("\\doinum{10.3390/")
690    .filter_map(|(at, _)| {
691      crate::shipped_class::class_code_argument(&code[at..], "\\doinum")
692        .map(|arg| (at, arg.trim().to_string()))
693    })
694    .collect();
695  let (_, formula) = defaults.last()?.clone();
696  if formula.contains(['#', '%']) {
697    return None;
698  }
699  let mut rest = formula.as_str();
700  while let Some(at) = rest.find(['\\', '[']) {
701    let after = rest[at..].strip_prefix('\\')?;
702    let len = after
703      .find(|c: char| !c.is_ascii_alphabetic() && c != '@')
704      .unwrap_or(after.len());
705    if !PARTS.contains(&&after[..len]) {
706      return None;
707    }
708    rest = &after[len..];
709  }
710  let article_only = match defaults.as_slice() {
711    [(at, first), .., _] if first == ARTICLE_ONLY => {
712      let condition = &code[code[..*at].rfind("\\ifthenelse")?..*at];
713      condition
714        .match_indices("\\equal{\\@journal}{")
715        .filter_map(|(i, call)| {
716          let name = &condition[i + call.len()..];
717          Some(name[..name.find('}')?].trim().to_string())
718        })
719        .collect()
720    },
721    _ => Vec::new(),
722  };
723  Some((article_only, formula))
724}

726/// The text of the macro `cs`'s expansion (empty for an empty one, `\gdef\@doiabbr{}`: the `preprints` row), if `cs`
727/// is defined.
728fn macro_text(cs: &str) -> Option<String> {
729  let defn = lookup_definition(&T_CS!(cs)).ok()??;
730  Some(match defn.get_expansion() {
731    Some(ExpansionBody::Tokens(body)) => body.to_string(),
732    _ => String::new(),
733  })
734}

736/// The class's default DOI (`default_doi`) for the paper's journal, volume, issue and article number, when the journal
737/// table named the journal and each number is one (`\cutdigits` of digits; a Molbank article number as it is): what
738/// the class's `\@doinum` expands to.
739fn default_doi_tokens() -> Option<Tokens> {
740  let formula = macro_text("\\lx@mdpi@doi@default")?;
741  let article_only = macro_text("\\@journal").is_some_and(|journal| {
742    macro_text("\\lx@mdpi@doi@articleonly")
743      .is_some_and(|list| list.contains(&s!(",{},", journal.trim())))
744  });
745  let number = |cs: &str| {
746    macro_text(cs).is_some_and(|text| {
747      let digits: String = text
748        .replace("\\cutdigits", "")
749        .chars()
750        .filter(|c| !matches!(c, '{' | '}') && !c.is_whitespace())
751        .collect();
752      !digits.is_empty() && digits.bytes().all(|b| b.is_ascii_digit())
753    })
754  };
755  if article_only {
756    return macro_text("\\@articlenumber")
757      .is_some_and(|text| !text.trim().is_empty())
758      .then(|| TokenizeInternal!("10.3390/\\@articlenumber"));
759  }
760  let complete = (!formula.contains("\\@doiabbr") || macro_text("\\@doiabbr").is_some())
761    && ["\\@pubvolume", "\\@issuenum", "\\@articlenumber"]
762      .iter()
763      .all(|cs| !formula.contains(cs) || number(cs));
764  complete.then(|| Tokens!(T_CS!("\\lx@mdpi@doi@default")))
765}

767/// How the paper's copy of mdpi.cls sets `\conflictofinterests`'s argument.
768enum ConflictForm {
769  /// The 2012-2015 versions: the argument is the section's title, set bold (`\textbf{#1}`).
770  Title,
771  /// The later versions: the statement under a bold "Conflicts of Interest:" label.
772  Label,

774  Plain,

777/// The form of the `\conflictofinterests` definition in `class` (a copy of mdpi.cls); the label form when it has none.
778fn conflict_of_interests_form(class: &str) -> ConflictForm {
779  let Some((_, rest)) = class.split_once("\\newcommand{\\conflictofinterests}") else {
780    return ConflictForm::Label;
781  };
782  let body: String = rest.chars().take(300).collect();
783  if body.contains("Conflicts of Interest:") {
784    ConflictForm::Label
785  } else if body.contains("\\textbf{#1}") {
786    ConflictForm::Title
787  } else {
788    ConflictForm::Plain
789  }
