1//! ccaption.sty — extended captioning by Peter Wilson (Herries Press).
2//!
3//! ccaption provides:
4//!  * Non-numbered/`*` captions (`\bottomcaption`, `\namedlegend`)
5//!  * Continued captions (`\contcaption`)
6//!  * Bilingual captions (`\bicaption`)
7//!  * Sidecap-style captions, custom caption fonts/labels
8//!  * `[subfigure]` compatibility: redefines `\caption`, `\@subcaption`, `\@float`, `\@dbflt` to
9//!    track main-vs-sub state.
10//!
11//! Raw-loading ccaption.sty into our engine is fragile: with the
12//! `[subfigure]` option, ccaption redefines `\@subcaption` to use
13//! `\xdef\@subfigcaptionlist{... {\protect\numberline {\@currentlabel}
14//! ...}}`. Our `\@currentlabel` expansion produces tokens like
15//! `\the\c@figure` that the `\xdef` mis-handles, surfacing the
16//! "You can't use `}` after `\the`" cascade at every figure +
17//! `\caption{...}\label{a}` pair (driver: 1105.3285).
18//!
19//! Perl LaTeXML has no ccaption binding either; with default
20//! `INCLUDE_STYLES=false`, raw ccaption.sty is not loaded. Match
21//! Perl in not loading the file: the binding defines its user-facing
22//! macros — the continued and bilingual captions as the file defines
23//! them, the rest (`\bottomcaption`, `\legend`, the caption styling)
24//! as approximations its load warning names — leaving the kernel
25//! `\caption`/`\label`/`\@subcaption` untouched.
26use latexml_package::prelude::*;
27

33  );

49  // ccaption.sty:46-48 and :66-96: the caption-shape commands as the file defines them — the width a caption is set
50  // in (`\captionwidth{<len>}`, 24 papers: 0803.1137, 0806.4323, 0811.1937; `\changecaptionwidth`, 20 papers: 0905.4349,

52  // a caption, 1706.01843), and `\precaption`/`\postcaption`/`\midbicaption{<code>}`, stored for the caption builder
53  // (0711.3402, 1001.1088, 1012.0535, 1408.3003); the captions keep LaTeXML's layout. Perl has no ccaption binding
54  // (KNOWN_PERL_ERRORS #807).

72  // ccaption.sty:204-217 `\bitwonumcaption[<label>]{<lof>}{<caption>}{<name>}{<lof2>}{<caption2>}`: a bilingual caption,
73  // each language numbered — the second caption under the float name `<name>`, the counter stepped back between them
74  // (1302.5179: "Figure 1: …" then "Tableau 1: …"); :219-259 its siblings with one number, `\bionenumcaption`,
75  // `\bicaption[<label>]{<lof>}{<caption>}{<name>}{<caption2>}` (the binding's former four-argument `\bicaption` kept
76  // only the first caption and left the rest as text) and `\bicontcaption`, the second caption a `\contcaption`.
77  // `\@if@contemptyarg` (:199-202) tells an empty list entry.
78  RawTeX!(
79    r"\begingroup
80\catcode`\Q=3
81\long\gdef\@if@contemptyarg#1{\@xif@contmt#1QQ\@secondoftwo\@firstoftwo\@nil}
82\long\gdef\@xif@contmt#1#2Q#3#4#5\@nil{#4}
83\endgroup
84\newcommand{\bitwonumcaption}[6][\@empty]{%
85  \begingroup
86  \@if@contemptyarg{#2}{\caption{#3}}{\caption[#2]{#3}}%
87  \ifx\@empty#1\else\label{#1}\fi
88  \edef\@conttempc{#4}%
89  \expandafter\renewcommand\csname \@captype name\endcsname{\@conttempc}%
90  \addtocounter{\@captype}{-1}%
91  \@contmidbi
92  \@if@contemptyarg{#5}{\caption{#6}}{\caption[#5]{#6}}%
93  \endgroup}
94\newcommand{\bionenumcaption}[6][\@empty]{%
95  \begingroup
96  \@if@contemptyarg{#2}{\caption{#3}}{\caption[#2]{#3}}%
97  \ifx\@empty#1\else\label{#1}\fi
98  \edef\@conttempc{#4}%
99  \expandafter\renewcommand\csname \@captype name\endcsname{\@conttempc}%
100  \@contmidbi
101  \contcaption{#6}%
102  \endgroup}
103\newcommand{\bicaption}[5][\@empty]{%
104  \begingroup
105  \@if@contemptyarg{#2}{\caption{#3}}{\caption[#2]{#3}}%
106  \ifx\@empty#1\else\label{#1}\fi
107  \edef\@conttempc{#4}%
108  \expandafter\renewcommand\csname \@captype name\endcsname{\@conttempc}%
109  \@contmidbi
110  \contcaption{#5}%
111  \endgroup}
112\newcommand{\bicontcaption}[3]{%
113  \begingroup
114  \contcaption{#1}%
115  \edef\@conttempc{#2}%
116  \expandafter\renewcommand\csname \@captype name\endcsname{\@conttempc}%
117  \@contmidbi
118  \contcaption{#3}%
119  \endgroup}"
120  );
121  // ccaption.sty:309-312 and :382-430 under `subfigure`: a continued figure's subfloats go on counting from the
122  // previous float's (`\contsubbottom`/`\contsubtop`, and `\contsubfigure`/`\contsubtable` for them: 0706.2285,
123  // 0907.2099, 0907.2405), the count kept after every subfloat (`\@contkeep`, through the subfigure binding's
124  // `\lx@subfloat@done`) and restored before a continued one (`\@contset`); the continued subfloat is numbered within
125  // the float it continues, so the figure step a bottom-captioned subfloat anticipates (subfigure_sty.rs) is not
126  // taken (`\lx@cc@contsub@step`, defined outside the option's conditional: its `\iftrue` would unbalance the skipped
127  // text). `\subconcluded` ends the count; `\subbottom`/`\subtop` are the subfigure package's subfloat of the float
128  // type.
129  RawTeX!(
130    r"\def\lx@cc@contsub@step#1{\expandafter\ifx\csname if\@captype topcap\endcsname\iftrue\else
131    \addtocounter{\@captype}{#1}\fi}"
132  );
133  RawTeX!(
134    r"\if@contsubfig
135  \newcounter{@contsubnum}
136  \newcommand{\@contkeep}{\setcounter{@contsubnum}{\value{sub\@captype}}}
137  \newcommand{\@contset}{\setcounter{sub\@captype}{\value{@contsubnum}}}
138  \newcommand{\subconcluded}{\setcounter{sub\@captype}{0}}
139  \def\lx@subfloat@done{\@contkeep}
140  \newcommand{\subbottom}{\csname sub\@captype\endcsname}
141  \newcommand{\subtop}{\csname sub\@captype\endcsname}
142  \newcommand{\contsubbottom}[2][]{\@contset\lx@cc@contsub@step{-1}%
143    \csname sub\@captype\endcsname[#1]{#2}\lx@cc@contsub@step{1}}
144  \let\contsubtop\contsubbottom
145  \let\contsubfigure\contsubbottom
146  \let\contsubtable\contsubbottom
147\fi"
148  );
149  // ccaption.sty:138-141 `\contcaption{<text>}`: the caption of a continued float, under the number of the float it
150  // continues (the counter stepped back before it is stepped; 0907.2405 "Figure 1: Continued."). Split and bilingual
151  // captions otherwise fall back to the regular caption.
152  DefMacro!(
153    "\\contcaption{}",

155  );
