1//! luacode.sty — Lua-in-LaTeX helpers (Manuel Pégourié-Gonnard), implemented
2//! on the engine's `\directlua` texlua bridge (`latexml_engine::lua_bridge`;
3//! user directive 2026-08-31 on first-principles LuaTeX-escape support).
4//!
5//! The real package aborts on non-LuaTeX engines ("LuaTeX is required"), so
6//! raw-loading is impossible by design — this binding stands in, mapping the
7//! public API onto `\directlua`, which is exactly how the real package
8//! implements it on LuaTeX (luacode.sty: `\luacode@dbg@exec` →
9//! `\directlua`). The catcode-taming differences between the variants are
10//! faithfully reduced to their observable effect: how much expansion the
11//! body sees before reaching Lua.
12use latexml_package::prelude::*;
13
14LoadDefinitions!({
15  // \luadirect{<lua>}: \directlua with luacode's debug wrapper — same
16  // evaluation. \luaexec{<lua>}: additionally makes \\, \%, \# usable;
17  // the expansion path already yields their character forms.
18  DefMacro!("\\luadirect{}", "\\lx@directlua{#1}");
19  DefMacro!("\\luaexec{}", "\\lx@directlua{#1}");
20
21  // \luastring{<text>}   — full expansion, then a quoted Lua string literal;
22  // \luastringO{<text>}  — expand once;
23  // \luastringN{<text>}  — no expansion (detokenized literal).
24  // All three reduce to "…escaped…" via the engine's \luaescapestring.
25  DefMacro!("\\luastring{}", "\"\\lx@luaescapestring{#1}\"");
26  DefMacro!("\\luastringO{}", "\"\\lx@luaescapestring{#1}\"");
27  DefMacro!(
28    "\\luastringN{}",
29    "\"\\lx@luaescapestring{\\detokenize{#1}}\""
30  );
31
32  // Debug-log toggles: affect console tracing only.
33  def_macro_noop("\\LuaCodeDebugOn")?;
34  def_macro_noop("\\LuaCodeDebugOff")?;
35
36  // {luacode} / {luacode*}: the body is Lua source, read RAW line-by-line to
37  // the matching \end (luacode* fully verbatim; plain luacode allows macros
38  // in the body — the corpus's bodies are Lua either way, and any TeX macro
39  // in them would have been flattened by \directlua's expansion too), then
40  // executed as ONE chunk; whatever it tex.print()s re-enters the input
41  // stream here. Mirrors the filecontents capture shape (the \begin group is
42  // closed manually, standing in for the consumed \end).
43  fn run_luacode_env(env: &str) -> Result<()> {
44    let end_marker = s!("\\end{{{env}}}");
45    read_raw_line(); // remainder of the \begin{...} line — not Lua
46    let mut lines: Vec<String> = Vec::new();
47    loop {
48      match read_raw_line() {
49        Some(line) if !line.contains(end_marker.as_str()) => lines.push(line),
50        _ => break,
51      }
52    }
53    let chunk = lines.join("\n");
54    // Re-inject the consumed `\end{...}` so the normal environment-close
55    // machinery balances the `\begin` (raw capture ate the whole line), then
56    // the chunk's output ABOVE it so the printed tokens are read first.
57    unread(Tokenize!(TeXString::assembled(s!("\\end{{{env}}}"))));
58    match latexml_engine::lua_bridge::lua_exec(&chunk) {
59      Ok(out) if !out.is_empty() => {
60        unread(Tokenize!(TeXString::assembled(out)));
61      },
62      Ok(_) => {},
63      Err(msg) => {
64        Info!("lua", env, s!("{{{env}}} chunk not evaluated: {msg}"));
65      },
66    }
67    Ok(())
68  }
69  DefPrimitive!(T_CS!("\\luacode"), None, {
70    run_luacode_env("luacode")?;
71  });
72  DefPrimitive!(T_CS!("\\luacode*"), None, {
73    run_luacode_env("luacode*")?;
74  });
75});
