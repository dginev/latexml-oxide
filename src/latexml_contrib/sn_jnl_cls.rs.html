563/// The frontmatter call of `\author[*][labels]{name}`: the labels, and `*` for a starred name, its `annotations`; the
564/// name, then its marks macro (`\lx@sn@marks@<n>`), the requests of the notes after it. Whether the `*` prints is the
565/// class's decision as the names print, from the last `\author` (`\ifx\@authfrstarg\empty`, :498, :510): `print_notes`
566/// makes it. The notes given inside the name run last at the print, so their text is the one printed (`inner_notes`).
567fn author_call(starred: bool, labels: Option<Tokens>, name: &Tokens) -> Tokens {
568  let n = lookup_int(AUTHORS) + 1;
569  assign_value(AUTHORS, n, Scope::Global);
570  assign_value(STARRED, starred, Scope::Global);
571  assign_value(PRINTED, false, Scope::Global);
572  assign_value(FLUSHED, false, Scope::Global);
573  let mut annotations = labels.map(Tokens::unlist).unwrap_or_default();
574  assign_value(LAST_LABELLED, !annotations.is_empty(), Scope::Global);
575  let name = name.unlist_ref();
576  if name.contains(&T_CS!("\\email")) {
577    assign_value(NAME_EMAIL, true, Scope::Global);
578  }
579  let marks = T_CS!(s!("\\lx@sn@marks@{n}"));
580  let mut call = vec![T_CS!("\\gdef"), marks, T_BEGIN!(), T_END!()];
581  // (the label `*`, as an `\author[*]`'s, which an `\affil[*]` line answers too: 2305.11920)
582  if starred {
583    assign_value(STARRED_NAMES, lookup_int(STARRED_NAMES) + 1, Scope::Global);
584    if !annotations.is_empty() {
585      annotations.push(T_OTHER!(","));
586    }
587    annotations.push(T_OTHER!("*"));
588  }
589  call.extend(inner_notes(name));
590  call.extend(tokenized(r"\lx@add@authors@append[annotations={"));
591  call.extend(annotations);
592  call.extend(tokenized("}]{"));
593  call.extend(name.iter().copied());
594  call.extend([marks, T_END!()]);
595  Tokens::new(call)
596}

598/// The text of each author note given inside `name`, the last of each kind, kept for `print_notes`
599/// (`\lx@sn@inner@<command>`): the class runs it as the names print, after every note outside, so its `\gdef` wins
600/// (twotext: "† Second text."); its call there is counted to emit nothing (`lx@sn@inside@<command>`).
601fn inner_notes(name: &[Token]) -> Vec<Token> {
602  let mut out = Vec::new();
603  for &(command, ..) in &AUTHOR_NOTES {
604    let cs = T_CS!(s!("\\{command}"));
605    let calls: Vec<usize> = (0..name.len()).filter(|&i| name[i] == cs).collect();
606    let Some(&last) = calls.last() else {
607      continue;
608    };
609    let inside = s!("lx@sn@inside@{command}");
610    assign_value(
611      &inside,
612      lookup_int(&inside) + calls.len() as i64,
613      Scope::Global,
614    );
615    out.extend([
616      T_CS!("\\gdef"),
617      T_CS!(s!("\\lx@sn@inner@{command}")),
618      T_BEGIN!(),
619    ]);
620    out.extend(note_argument(command, &name[last + 1..]));
621    out.push(T_END!());
622  }
623  out
624}

626/// The text a note command reads after it in `rest`, as its definition does: the copy's optional form `[text]`, or its
627/// default (`\@equalconttxt`, 2504.21788's); else one argument, a group's content or one token.
628fn note_argument(command: &str, rest: &[Token]) -> Vec<Token> {
629  let mut rest = rest
630    .iter()
631    .copied()
632    .skip_while(|t| t.get_catcode() == Catcode::SPACE);
633  let default = lookup_string(&s!("lx@sn@default@{command}"));
634  if !default.is_empty() {
635    let mut rest = rest.peekable();
636    if rest.peek() != Some(&T_OTHER!("[")) {
637      return vec![T_CS!(s!("\\{default}"))];
638    }
639    rest.next();
640    let mut depth = 0;
641    return rest
642      .take_while(|t| {
643        match t.get_catcode() {
644          Catcode::BEGIN => depth += 1,
645          Catcode::END => depth -= 1,
646          _ => {},
647        }
648        !(depth == 0 && *t == T_OTHER!("]"))
649      })
650      .collect();
651  }
652  match rest.next() {
653    Some(t) if t.get_catcode() == Catcode::BEGIN => {
654      let mut depth = 1;
655      rest
656        .take_while(|t| {
657          match t.get_catcode() {
658            Catcode::BEGIN => depth += 1,
659            Catcode::END => depth -= 1,
660            _ => {},
661          }
662          depth > 0
663        })
664        .collect()
665    },
666    Some(t) => vec![t],
667    None => Vec::new(),
668  }
669}

671/// An author note's text stored and its switch set, as the class's command does; the names written so far end with
672/// its request (the class's `\g@addto@macro\artauthors{$^{\dagger}$}`). Before any name, or after a section digested
673/// them, none; inside a name, run as the names print, its text already queued (`inner_notes`); else, after the print,
674/// the document's note.
675fn author_note(command: &str, text: Tokens) -> Tokens {
676  let Some(&(_, store, switch, role)) = AUTHOR_NOTES.iter().find(|note| note.0 == command) else {
677    return Tokens::new(Vec::new());
678  };
679  if lookup_bool(PRINTED) {
680    let pending = lookup_int(&s!("lx@sn@pending@{command}"));
681    if pending > 0 {
682      assign_value(&s!("lx@sn@pending@{command}"), pending - 1, Scope::Global);
683      return Tokens::new(Vec::new());
684    }
685    let mut call = tokenized(&s!("\\@add@frontmatter{{ltx:note}}[role={role}]{{"));
686    call.extend(text.unlist());
687    call.push(T_END!());
688    return Tokens::new(call);
689  }
690  assign_value(&s!("lx@sn@given@{command}"), true, Scope::Global);
691  let mut call = vec![T_CS!("\\gdef"), T_CS!(s!("\\{store}")), T_BEGIN!()];
692  call.extend(text.unlist());
693  call.push(T_END!());
694  call.extend(tokenized(&s!("\\global\\{switch}true")));
695  let n = lookup_int(AUTHORS);
696  // (after a section's flush the names are digested: the note is the document's, unmarked, revSNJNLb B2)

699    call.extend([
700      T_CS!("\\g@addto@macro"),

702      T_BEGIN!(),
703    ]);
704    call.extend(tokenized(&s!(
705      "\\lx@request@frontmatter@annotation[sn]{{{command}}}"
706    )));
707    call.push(T_END!());
708  }
709  Tokens::new(call)
710}

712/// The notes `\@maketitle` prints under the addresses (sn-jnl.cls:749-756), each queued once with its label, which the
713/// names carrying it request. The starred names (label `*`) share the correspondent note where the class prints the
714/// line (an `\email` after a starred name, or in a name printed under a starred last author; :535-546), marked `*` where
715/// the last `\author` has labels (:498, :510); a printed `*` that neither it nor an `\affil[*]` line explains shows on
716/// the name (`declare_unexplained_name_labels`, 2509.19318), an unprinted one nothing (2606.13079). Then each author note
717/// given, with its last text and the glyph the copy appends; one given before any name is the document's.
718fn print_notes() -> Tokens {
719  let mut tex = String::new();
720  let starred = lookup_int(STARRED_NAMES);
721  let text = lookup_string(CORRESPONDING_TEXT);
722  let line = !text.is_empty()
723    && (lookup_bool(CORRESPONDING_EMAIL) || (lookup_bool(NAME_EMAIL) && lookup_bool(STARRED)));
724  let star_prints = lookup_bool(LAST_LABELLED) && lookup_bool(STAR_PRINTS);
725  let mut shown = Vec::new();
726  if starred > 0 && line {
727    tex.push_str(&creators_note(
728      "*",
729      "correspondent",
730      star_prints.then_some("*"),
731      false,
732      &text,
733    ));
734  } else if starred > 0 && !lookup_bool(AFFIL_STAR) {
735    // (one the class prints no `*` for, under an unlabelled last `\author` or a copy's `{}`, prints nothing: 2606.13079)
736    shown.push((s!("*"), if star_prints { s!("*") } else { s!("") }));
737  }
738  declare_unexplained_name_labels(&shown);
739  for &(command, store, _, role) in &AUTHOR_NOTES {
740    let inside = lookup_int(&s!("lx@sn@inside@{command}"));
741    let given = lookup_bool(&s!("lx@sn@given@{command}")) || inside > 0;
742    let requested = lookup_bool(&s!("lx@sn@requested@{command}"));
743    for key in ["given", "requested"] {
744      assign_value(&s!("lx@sn@{key}@{command}"), false, Scope::Global);
745    }
746    assign_value(&s!("lx@sn@inside@{command}"), 0i64, Scope::Global);
747    assign_value(&s!("lx@sn@pending@{command}"), inside, Scope::Global);
748    if !given {
749      continue;
750    }
751    let text = if inside > 0 {
752      s!("\\lx@sn@inner@{command}")
753    } else {
754      s!("\\{store}")
755    };
756    if requested {
757      // (its command prints the glyph after the names: no glyph typed on a name, nor the star's, is it — revSNJNLb S3)
758      let glyph = note_glyph(command);
759      tex.push_str(&creators_note(
760        &s!("sn:{command}"),

766    } else {

769  }
770  for key in [CORRESPONDING_EMAIL, NAME_EMAIL, AFFIL_STAR] {
771    assign_value(key, false, Scope::Global);
772  }
773  assign_value(STARRED_NAMES, 0i64, Scope::Global);
774  assign_value(PRINTED, true, Scope::Global);
775  Tokens::new(tokenized(&tex))
776}

827/// `printed_label_pieces`); none, with a Warning, where it prints no one glyph — its names then link it unmarked.
828fn note_glyph(command: &str) -> Option<String> {
829  let mark = lookup_string(&s!("lx@sn@mark@{command}"));
830  let pieces = printed_label_pieces(&Tokens::new(tokenized(&mark))).unwrap_or_default();
831  if let [glyph] = pieces.as_slice() {
832    // (the pieces key every asterisk `*`: a math one, or `\textasteriskcentered`, prints ∗, which the starred names'
833    // text `*` is not — 2501.01157's `$^{*}$`, its "1*∗" two marks, revSNJNLb S3)
834    let math = ["$", "\\ast", "\\textasteriskcentered"];
835    if glyph == "*" && math.iter().any(|m| mark.contains(m)) {
836      return Some(s!("\u{2217}"));
837    }
838    return Some(glyph.clone());
839  }
840  Warn!(
841    "frontmatter",
842    "unanswered_mark_lost",
843    s!(
844      "sn-jnl's \\{command} prints {mark} after its names, read as no one mark: the names show none"
845    )
846  );
847  None
848}

850/// The mark `command`'s definition in a copy appends to the names (`\g@addto@macro\artauthors{$^{\dagger}$}`,
851/// 2412.01641's :586; `$^{*}$` 2501.01157's :587, `$^{\S}$` 2307.08214's :595, `\textsuperscript{\dag}`); `None`
852/// where it marks no name (`{}`, 2209.11450's) or the copy does not define the command.
853fn note_mark(text: &str, command: &str) -> Option<String> {
854  let code = class_code(text);
855  let at = code.find(&s!("\\newcommand{{\\{command}}}"))?;
856  let body = &code[at + 1..];
857  let body = &body[..body.find("\\newcommand").unwrap_or(body.len())];
858  let mark = class_code_argument(body, "\\g@addto@macro\\artauthors")?;
859  let bare: String = mark
860    .chars()
861    .filter(|c| !matches!(c, '$' | '^' | '{' | '}') && !c.is_whitespace())
862    .collect();
863  (!bare.is_empty()).then(|| mark.trim().to_string())
864}

866/// Whether a copy's starred name prints `*` after its labels (`{*}` in its `\@@corrauthor`, 2412.01641's :511; `{}` in
867/// 2209.11450's).
868fn star_prints(text: &str) -> bool {
869  let code = class_code(text);
870  let Some(at) = code.find("\\newcommand{\\@@corrauthor}") else {
871    return true;
872  };
873  let body = &code[at + 1..];
874  body[..body.find("\\newcommand").unwrap_or(body.len())].contains("{*}")
875}

877/// The words of a copy's corresponding line before its e-mails, after its `{\ifnum\aucount>1*\fi}` (2412.01641's :751
878/// "Corresponding author(s).", 2211.08503's "Corresponding author.", 2509.24703's "Corresponding authors.", 2606.20205's
879/// "Correspondence:").
880fn corresponding_text(text: &str) -> Option<String> {
881  let code = class_code(text);
882  let line = match code.find("\\ifnum\\aucount>1*\\fi}") {
883    Some(at) => &code[at + "\\ifnum\\aucount>1*\\fi}".len()..],
884    None => &code[code.find("Corresponding author")?..],
885  };
886  let end = line
887    .find("E-mail")
888    .or_else(|| line.find("\\corrauthemail"))?;
889  let words = line[..end].trim();
890  (!words.is_empty() && !words.contains('\\')).then(|| words.to_string())
891}
