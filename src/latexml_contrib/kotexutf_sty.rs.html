1//! kotexutf.sty — ko.TeX for pdfTeX (the UTF-8 byte path).
2//!
3//! The real package runs under the pdfTeX byte mouth (KERNEL_CAPABILITIES
4//! K10, `inputenc_sty::enable_pdftex_byte_mouth`): kotexutf-core.tex:100-136
5//! makes every UTF-8 lead byte an active reader of its continuation bytes,
6//! and kotexutf.sty's josa commands (`\은`/`\를`/`\가`…, the byte-model
7//! `\DeclareRobustCommand*\^^ec[2]` family) are control symbols over the
8//! lead byte that read the next two bytes. Every non-ASCII character then
9//! funnels through kotexutf-core.tex:213 `\protected\def\unihangulchar#1`
10//! (the code point as a number) → `\unihangulchar@@` (line-break/josa
11//! lookahead) → a glyph in an LUC subfont. Bindings outrank raw: the funnel
12//! records the code point for the josa logic (kotexutf.sty:358 `\makejosa`
13//! reads `\@josa`, set at kotexutf-core.tex:134) and emits the character
14//! itself. Witnesses: kotex-utf-doc (27 josa errors), kotex-doc, cjk-ko-doc
15//! via dhucs.sty:44. Guard:
16//! `perfect_kernel_batch56::kotexutf_runs_under_the_byte_mouth`.
17use latexml_package::prelude::*;
18
19LoadDefinitions!({
20  inputenc_sty::enable_pdftex_byte_mouth()?;
21  InputDefinitions!("kotexutf", noltxml => true, extension => Some(Cow::Borrowed("sty")));
22  // kotexutf-core.tex:213-217 + :134: keep the code point for `\makejosa`,
23  // emit the character.
24  RawTeX!(
25    r"\protected\def\unihangulchar#1{\unih@ngulpoint#1\relax\global\@josa\unih@ngulpoint\lx@kotex@char{#1}}"
26  );
27  DefPrimitive!("\\lx@kotex@char{Number}", sub[(n)] {
28    if let Some(ch) = u32::try_from(n.value_of()).ok().and_then(char::from_u32) {
29      unread(Tokens!(CharToken!(ch)));
30    }
31  });
32});
