1//! aaai.sty and its years (aaai16-aaai2027, the versioned fallback strips each to `aaai`): the paper's copy read raw, the
2//! marks its `\maketitle` sets declared, the author notes only its `\@maketitle` defines (2501.02471, 2511.07192,
3//! 2103.05277, 2111.06599, 2006.08767, 2609.00420; OD #1053).

13LoadDefinitions!({
14  // As AAAI's: each prints its footnote once, "These authors contributed equally." and "Corresponding author.", the
15  // first use making the note, a later one its `\footnotemark[<n>]` (2501.02471, 2508.07026, 2511.17927, 2511.15266,
16  // 2511.15807; `aaai_note`, OD #1053); in a document where a use is followed by ".", each use its own note
17  // (2509.03329). As stubs (`DefinitionOrigin::Stub`): the document's or a raw class's own `\newcommand` still defines
18  // them (copernicus.cls:1673 `\newcommand\equalcontrib[1]`, its equal-contribution note on the affiliations).
19  AAAI_FIRST_NOTES.with(|notes| notes.borrow_mut().clear());
20  AAAI_DOTTED.set(false);
21  DefMacro!("\\lx@aaai@note{}{}", sub[(key, text)] { aaai_note(&key.to_string(), text) });
22  DefMacro!("\\lx@aaai@noted{}{}{}", sub[(key, field, before)] {
23    let number = |tokens: &Tokens| tokens.to_string().trim().parse::<i64>().unwrap_or(0);
24    let (field, before) = (number(&field), number(&before));
25    let now = title_note_number()?;
26    if now > before {
27      note_aaai_first(&key.to_string(), u16::try_from(field).unwrap_or(0), now);
28    }
29    Ok(Tokens::new(Vec::new()))
30  });
31  with_origin(DefinitionOrigin::Stub, || -> Result<()> {
32    DefMacro!(
33      "\\equalcontrib",
34      r"\lx@aaai@note{equalcontrib}{These authors contributed equally.}"
35    );
36    DefMacro!(
37      "\\corresponding",
38      r"\lx@aaai@note{corresponding}{Corresponding author.}"
39    );
40    Ok(())
41  })?;
42  // (after the stubs: a copy's own `\equalcontrib` replaces them, revMTREDUCE S3)
43  input_shipped_package_title_marks(FNSYMBOL_TITLE_MARKS)?;
44});

46thread_local! {
47  /// AAAI's notes made once (`\equalcontrib`, `\corresponding`): each kind's first use, by its position in the author
48  /// field (`Token::prov`, 0 for none), and the footnote number its note took ([`aaai_note`]).
49  static AAAI_FIRST_NOTES: RefCell<Vec<(String, u16, i64)>> = const { RefCell::new(Vec::new()) };
50  /// A use followed by "." was read ([`aaai_note`]): the emitter's cut at ". " gives the next name the use before it,
51  /// so every use is then a note of its own, as before the first-use reading (c1/c6/c7 of review revPRESBXb, 2509.03329).
52  static AAAI_DOTTED: Cell<bool> = const { Cell::new(false) };
53}
54
55/// `\lx@aaai@note{<kind>}{<text>}`: the note `<text>` at the kind's first use, `\footnotemark[<n>]` after it. The rehearsal
56/// tells the first by its position in the field (`Token::prov`); a use digested with none (a name the emitter digests
57/// again, an `\affiliations` field) is a later one once the kind's note is made (OD #1053).
58/// Once a use is followed by "." ([`AAAI_DOTTED`]), every use is `\thanks{<text>}`.
59fn aaai_note(kind: &str, text: Tokens) -> Result<Tokens> {
60  let dotted =
61    peek_token()?.is_some_and(|t| t.get_catcode() == Catcode::OTHER && t.with_str(|s| s == "."));
62  if dotted || AAAI_DOTTED.get() {
63    // (the fail-safe: no note lost and no name given another's mark; each name's note marked as its own)
64    AAAI_DOTTED.set(true);
65    let mut note = vec![T_CS!("\\thanks"), T_BEGIN!()];
66    note.extend(text.unlist());
67    note.push(T_END!());
68    return Ok(Tokens::new(note));
69  }
70  let field = get_current_token().map_or(0, |t| t.prov);
71  let now = title_note_number()?;
72  let first = AAAI_FIRST_NOTES.with(|notes| {
73    notes
74      .borrow()
75      .iter()
76      .find(|(k, at, _)| k == kind && (field == 0 || (*at != 0 && *at < field)))
77      .map(|(_, _, number)| *number)
78  });
79  if let Some(number) = first {
80    // (a name digested again carries the kind's mark once: a segment that also holds the name before's,
81    // `Bernal\equalcontrib. Raigoso\equalcontrib`, 2509.03329)
82    let marked = s!("lx_aaai_marked@{kind}");
83    if field == 0 && rehearsal_footnote_value().is_none() {
84      if lookup_bool(&marked) {
85        return Ok(Tokens::new(Vec::new()));
86      }
87      assign_value(&marked, true, None);
88    }
89    let mut mark = vec![T_CS!("\\footnotemark"), T_OTHER!("[")];
90    mark.extend(Explode!(number));
91    mark.push(T_OTHER!("]"));
92    return Ok(Tokens::new(mark));
93  }
94  let before = now;
95  let mut note = vec![T_CS!("\\thanks"), T_BEGIN!()];
96  note.extend(text.unlist());
97  note.extend([T_END!(), T_CS!("\\lx@aaai@noted"), T_BEGIN!()]);
98  note.extend(Explode!(kind));
99  note.extend([T_END!(), T_BEGIN!()]);
100  note.extend(Explode!(field));
101  note.extend([T_END!(), T_BEGIN!()]);
102  note.extend(Explode!(before));
103  note.push(T_END!());
104  Ok(Tokens::new(note))
105}

107/// The note of `kind` made at field position `field` (0: none) took number `number` ([`aaai_note`]): kept unless one is
108/// kept for an earlier position, or for any where this one has none; at the same position, a rehearsal's count replaces
109/// the one before (the last rehearsal's base counts the title's notes digested ahead of the names: "Title∗", "Ann†",
110/// "Bob†"), a later run's (the title-page check's) does not.
111fn note_aaai_first(kind: &str, field: u16, number: i64) {
112  let rehearsing = rehearsal_footnote_value().is_some();
113  AAAI_FIRST_NOTES.with(|notes| {
114    let mut notes = notes.borrow_mut();
115    if notes.iter().any(|(k, at, _)| {
116      k == kind && (field == 0 || (*at != 0 && (*at < field || (*at == field && !rehearsing))))
117    }) {
118      return;
119    }
120    notes.retain(|(k, ..)| k != kind);
121    notes.push((kind.to_string(), field, number));
122  });
