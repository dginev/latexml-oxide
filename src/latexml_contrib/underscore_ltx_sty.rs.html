1//! underscore-ltx.sty — the LaTeX kernel's first aid for underscore.sty.
2//!
3//! The kernel loads it right after underscore.sty
4//! (latex2e-first-aid-for-external-files.ltx:176-177); Perl has no binding, so
5//! its active `_` stays underscore.sty's `\ifmmode\sb\else\textunderscore\fi`.
6//! The first-aid `_` is `\protected` and a literal `_` inside a `\csname`
7//! (underscore-ltx.sty:38-51), so a file name keeps its underscore:
8//! `\input{sections/logic_T}` read `sections/logic\textunderscoreT` (arXiv
9//! 2606.31852, once PDF output ran its `\ifpdf\usepackage{underscore}`; the
10//! file name is expanded as `\set@curr@file` does it,
11//! `gullet::expand_as_csname_text`). The `\lowercase` turns `~` into the active
12//! `_` the body needs. Guard: `class_census::underscore_first_aid_file_name`.
13use latexml_package::prelude::*;
14
15#[rustfmt::skip]
16LoadDefinitions!({
17  RawTeX!(r"\begingroup\lccode`\~=`\_\lowercase{\endgroup
18\protected\gdef~{\ifincsname\string~\else\ifx\protect\@typeset@protect\ifmmode\sb\else\BreakableUnderscore\fi\else\ifx\protect\@unexpandable@protect\noexpand~\else\protect~\fi\fi\fi}%
19\global\let\ActiveUnderscore=~}");
20});
