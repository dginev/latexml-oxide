1//! algpseudocodex.sty — LaTeXML binding for algpseudocodex package.
2//!
3//! algpseudocodex builds on `algorithmicx` / `algpseudocode` and provides pseudocode
4//! typesetting with customizable indentation guides, single-line and multiline comments,
5//! and boxed code blocks.
6//!
7//! On `perfect_kernel`, raw loading already renders `\Comment` on one line (`float:right`)
8//! with 0 Fatal; the binding provides full semantic support for `\LComment`, `\BeginBox`,
9//! `\EndBox`, and `\BoxedString` (which raise undefined control sequence errors under raw loading)
10//! and implements clean in-flow box wrappers.
11//!
12//! This binding (beyond-Perl surpass; see docs/parity/OXIDIZED_DESIGN_DIVERGENCES.md #214):
13//! 1. Loads `algpseudocode` (which loads `algorithmicx`).
14//! 2. Defensively guards against the path where old `algorithmic.sty` was loaded first
15//!    and `algorithmicx` bailed (witness: 2410.03000).
16//! 3. Honors `italicComments`, `rightComments`, `commentColor` (default `gray`),
17//!    `beginComment`, `endComment`, `beginLComment`, `endLComment`, and `noEnd` (default `true`).
18//!    TikZ visual-only options (`indLines`, `spaceRequire`) are parsed and safely dropped.
19//! 4. Preserves `\Statex` in-line break semantics (`<break/>` within the open line box)
20//!    matching raw TeX varwidth-in-open-box execution.
21//! 5. Replaces TikZ overlay boxes with in-flow styled wrappers (`ltx:text`)
22//!    capturing `draw=<color>`, `dashed`/`dotted`, `thick` options from
23//!    `\BeginBox`, `\EndBox`, and `\BoxedString`. Multi-line boxes open only on the
24//!    pending->open transition to prevent double-opening.
25//! 6. Declares dummy counters and stubs for internal registers (`algpx@codeBoxCount`, etc.).
26

30fn parse_box_options(raw: &str) -> (String, String) {
31  let mut classes = vec!["ltx_algpx_boxed".to_string()];
32  let mut color = "black".to_string();
33  let mut style = "solid".to_string();
34  let mut width = "1px".to_string();
35
36  for part in raw.split(',') {
37    let part = part.trim();
38    if part.is_empty() || part == "algpxDefaultBox" {
39      continue;
40    }
41    if let Some((k, v)) = part.split_once('=') {
42      let k = k.trim();
43      let v = v.trim();
44      if k == "draw" {
45        color = v.to_string();
46        classes.push(format!("ltx_border_{v}"));
47      }
48    } else {
49      match part {
50        "dashed" => {
51          style = "dashed".to_string();
52          classes.push("ltx_dashed".to_string());
53        },
54        "dashdotted" => {
55          style = "dashed".to_string();
56          classes.push("ltx_dashdotted".to_string());
57        },
58        "dotted" => {
59          style = "dotted".to_string();
60          classes.push("ltx_dotted".to_string());
61        },
62        "thick" => {
63          width = "2px".to_string();
64          classes.push("ltx_thick".to_string());
65        },
66        "very thick" => {
67          width = "3px".to_string();
68          classes.push("ltx_very_thick".to_string());
69        },
70        "thin" => {
71          width = "0.5px".to_string();
72          classes.push("ltx_thin".to_string());
73        },
74        other => {
75          if !other.is_empty() {
76            classes.push(format!("ltx_{other}"));
77          }
78        },
79      }
80    }
81  }
82
83  let css = format!("border: {width} {style} {color};");
84  (classes.join(" "), css)
85}

297  // Code command line box:
298  // In raw algpseudocodex.sty:185, \algpx@startCodeCommand opens a varwidth box
299  // and \algpx@endCodeCommand closes it. \Statex has no hook, so its text sits
300  // inside the open box, causing \lx@algorithmicx@@item to insert <ltx:break/>
301  // rather than a nested <ltx:listingline> (preserved by statex_continues_the_open_line_box).
302  // Marking with _noautoclose="1" prevents maybe_close_element from prematurely
303  // auto-closing the line box when \Statex runs.
304  DefConstructor!("\\algpx@startCodeCommand", sub [document] {
305    if lookup_value("algpx@code_open").is_some_and(|s| matches!(s, Stored::Bool(true))) {
306      let _ = document.maybe_close_element("ltx:text");
307      assign_value("algpx@code_open", Stored::Bool(false), Some(Scope::Global));
308    }
309    let mut attrs = FxHashMap::default();
310    attrs.insert("class".into(), "ltx_algpx_code".into());
311    attrs.insert("_noautoclose".into(), "1".into());
312    document.open_element("ltx:text", Some(attrs), None)?;
313    assign_value("algpx@code_open", Stored::Bool(true), Some(Scope::Global));
314  });

316  fn unwind_code_command(document: &mut Document) -> Result<()> {
317    if lookup_value("algpx@code_open").is_some_and(|s| matches!(s, Stored::Bool(true))) {
318      let mut curr = Some(document.get_node().clone());
319      let mut target = None;
320      while let Some(node) = curr {
321        if document::get_node_qname(&node) == pin!("ltx:text")
322          && node.get_attribute("class").as_deref() == Some("ltx_algpx_code")
323        {
324          target = Some(node);
325          break;
326        }
327        if document::get_node_qname(&node) == pin!("ltx:listingline")
328          || document::get_node_qname(&node) == pin!("ltx:listing")
329          || document::get_node_qname(&node) == pin!("ltx:document")
330        {
331          break;
332        }
333        curr = node.get_parent();
334      }
335      if let Some(code_node) = target {
336        document.close_node_internal(&code_node)?;
337      }
338      assign_value("algpx@code_open", Stored::Bool(false), Some(Scope::Global));
339    }
340    Ok(())
341  }

343  fn unwind_one_box(document: &mut Document) -> Result<()> {
344    unwind_code_command(document)?;
345    let mut count = match lookup_value("algpx@open_box_depth") {
346      Some(Stored::Number(n)) => n.value_of(),
347      _ => 0,
348    };
349    if count > 0 {
350      let mut curr = Some(document.get_node().clone());
351      let mut target = None;
352      while let Some(node) = curr {
353        if document::get_node_qname(&node) == pin!("ltx:text")
354          && node
355            .get_attribute("class")
356            .is_some_and(|c| c.contains("ltx_framed"))
357        {
358          target = Some(node);
359          break;
360        }
361        if document::get_node_qname(&node) == pin!("ltx:listingline")
362          || document::get_node_qname(&node) == pin!("ltx:listing")
363          || document::get_node_qname(&node) == pin!("ltx:document")
364        {
365          break;
366        }
367        curr = node.get_parent();
368      }
369      if let Some(box_node) = target {
370        document.close_node_internal(&box_node)?;
371        count -= 1;
372        assign_value(
373          "algpx@open_box_depth",
374          Stored::Number(Number::new(count)),
375          Some(Scope::Global),
376        );
377      }
378    }
379    assign_value(
380      "algpx@pending_boxes",
381      Stored::String(pin("")),
382      Some(Scope::Global),
383    );
384    Ok(())
385  }

437  // In-flow boxes:
438  // \BeginBox[opts] stores pending box options.
439  // Multiple nested boxes can be pending and are opened at the next item.
440  // \EndBox closes one nested box.
441  DefConstructor!("\\BeginBox []", sub [_document, args] {
442    let raw = args
443      .first()
444      .and_then(|a| a.as_ref())
445      .map(|d| d.to_string())
446      .unwrap_or_else(|| "algpxDefaultBox".to_string());
447    let raw_str = if raw.is_empty() { "algpxDefaultBox" } else { &raw };
448    let prev = lookup_value("algpx@pending_boxes");
449    let queue_str = if let Some(Stored::String(s)) = prev && !to_string(s).is_empty() {
450      format!("{}|||{}", to_string(s), raw_str)
451    } else {
452      raw_str.to_string()
453    };
454    assign_value("algpx@pending_boxes", Stored::String(pin(&queue_str)), Some(Scope::Global));
455  });

457  DefConstructor!("\\algpx@check@box", sub [document] {
458    let pending = lookup_value("algpx@pending_boxes");
459    if let Some(Stored::String(p)) = pending && !to_string(p).is_empty() {
460      let spec_str = to_string(p).to_string();
461      assign_value("algpx@pending_boxes", Stored::String(pin("")), Some(Scope::Global));
462      let mut count = match lookup_value("algpx@open_box_depth") {
463        Some(Stored::Number(n)) => n.value_of(),
464        _ => 0,
465      };
466      for spec in spec_str.split("|||") {
467        let (classes, cssstyle) = parse_box_options(spec);
468        let mut attrs = FxHashMap::default();
469        attrs.insert("class".into(), format!("ltx_framed {classes}"));
470        attrs.insert("cssstyle".into(), cssstyle);
471        document.open_element("ltx:text", Some(attrs), None)?;
472        count += 1;
473      }
474      assign_value("algpx@open_box_depth", Stored::Number(Number::new(count)), Some(Scope::Global));
475    }
476  });
