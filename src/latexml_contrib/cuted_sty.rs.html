1use latexml_package::prelude::*;

3LoadDefinitions!({
4  // No upstream binding: cuted.sty is read raw (its options, `\stripsep`, the column switches), then `{strip}` is
5  // redefined. cuted.sty sets the strip in the global box `\@viper` (TL2025's v2.7: `\strip` :208, `\endstrip` :225),
6  // which only its output routine ships (`\@viperoutput`, :241; v2.10 :229, :250, :272) — pages our engine never builds, so the strip's content
7  // was lost (Perl with raw styles errs "Not in outer par mode" and loses it too; without them `{strip}` is undefined).
8  // The strip is a vertical block where it is written, as `{center}` is (its paragraphs end at both edges, a
9  // `\captionof` takes the material beside it); the PDF prints it across both columns at the top of the next page.
10  // Witnesses 2508.07251 (a teaser figure), 2609.08929 (a table), 2609.22546 (a nomenclature box), 2404.07191,
11  // 2404.00122, 2406.06521, 2502.12371, 2502.19582 (OXIDIZED_DESIGN_DIVERGENCES #476).
12  InputDefinitions!("cuted", noltxml => true, extension => Some(Cow::Borrowed("sty")));
13  DefEnvironment!(
14    "{strip}",
15    sub[document, _args, props] {
16      document.maybe_close_element("ltx:p")?;
17      if let Some(Stored::Digested(body)) = props.get("body") {
18        insert_block(document, body, HashMap::default())?;
19      }
20      Ok(())
21    },
22    mode => "internal_vertical"
23  );
24});
