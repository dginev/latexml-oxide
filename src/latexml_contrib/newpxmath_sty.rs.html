1//! newpxmath.sty (Palatino math fonts). The font itself is presentational;
2//! what documents reach is its SYMBOL SET — newpxmath.sty:187 loads the
3//! `amssymbols` option by default and :1075-1258 `\re@DeclareMathSymbol`s
4//! the AMS set (`\square`, `\blacksquare`, `\blacktriangleright`, `\nmid`
5//! … uantwerpenexam-example2), which the amssymb binding already carries.
6//! Guard: `perfect_kernel_batch56::font_symbol_packages_carry_amssymb`.
7//!
8//! newpxmath.sty is newtxmath.sty's Palatino twin: the same symbol names over the same symbol-font layout (its
9//! `symbolsC` is pxfonts' `pxsyc`, txfonts' `txsyc`), so the binding carries newtxmath's: txfonts' symbols (`\nequiv`
10//! :1374, `\fint` :967, `\mapsfrom` :1313, `\circleddot` :644, `\llbracket` :527, `\multimapinv` :1345) without loading
11//! txfonts (newpxmath.sty:23-33 never does; `\@ifpackageloaded{txfonts}` stays false) and without the txfonts names it
12//! does not declare (`\varv`, `\varw`, `\vary`, the clockwise double and triple contour integrals), then newtxmath's
13//! own layer (`\coloneq` and `\eqcolon` :1541-1542 as ≔ and ≕, `\varg` :1858 as an italic g, `\uppartial` :1860,
14//! `\transp` :570, `\Angstrom` :551, `\cdotB` :1646, `\vv` :833, `\laplac` :745, the upright Greek `\Gammaup`…
15//! :1662-1701 and their `\upGamma`… names :1703-1743).
16//! Witnesses 1709.08558, 1902.06558, 2001.06387, 2003.04827, 2003.05278, 2305.12589, 2508.05134, 2603.09013.
17//! Guards: `perfect_kernel_batch68j::newpxmath_carries_its_symbol_set`, `::newpxmath_coloneq_over_mathtools`.
18use latexml_package::{
19  package::{
20    newtxmath_sty::{declare_newtx_greek_options, def_newtx_math_symbols, newtx_capital_greek},
21    txfonts_sty::def_txfonts_symbols,
22  },
23  prelude::*,
24};
25
26LoadDefinitions!({
27  // newpxmath.sty:23-33 `\RequirePackage{amsmath}`, `{etoolbox}`, `{xkeyval}`, `{centernot}`: the binding shadows
28  // the raw load, so a document that relies on them (`\text`, `\tfrac` via a style that loads only newpxmath —
29  // arXiv 2605.27258) flooded undefined.
30  RequirePackage!("amsmath");
31  RequirePackage!("etoolbox");
32  RequirePackage!("xkeyval");
33  RequirePackage!("centernot");
34  RequirePackage!("amssymb");
35  // The symbol set, as the newtxmath binding defines it (txfonts' symbols, then newtxmath's own layer).
36  def_txfonts_symbols()?;
37  def_newtx_math_symbols()?;
38  // newpxmath.sty:2118-2119: `\varmathbb` is a blackboard alphabet of its own (`\vmathbb` the
39  // same), not a copy of `\mathbb`, which a document may have `\renewcommand`ed to `\varmathbb`
40  // before the load (the two then expanded into each other; KPE #355).
41  DefConstructor!("\\varmathbb{}", "#1", bounded => true, require_math => true,
42    font => { family => "blackboard", series => "medium", shape => "upright" });
43  Let!("\\vmathbb", "\\varmathbb");
44  Let!("\\vvmathbb", "\\varmathbb");
45  Let!("\\vvarmathbb", "\\varmathbb");
46  // newpxmath.sty:173-175, :212 and :595-619: its Greek options and capital Greek letters, as newtxmath's; but :188
47  // `\ProcessOptions*` runs them in the order given, so the last of `slantedGreek`/`uprightGreek` wins (pdflatex:
48  // `[slantedGreek,uprightGreek]` U/npxmia, the upright `lettersA`). Witness 2508.05134 (`slantedGreek`).
49  declare_newtx_greek_options()?;
50  ProcessOptions!(*);
51  newtx_capital_greek()?;
52});
