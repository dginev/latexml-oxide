1//! entics.cls (Electronic Notes in Theoretical Informatics and Computer Science) and its predecessor entcs.cls (ENTCS):
2//! the papers ship the class (it is not in TeX Live) and it is read raw. Both are generated from Elsevier's esl.dtx, as
3//! elsart.cls is, and keep its frontmatter interface: `\author[keys]{…}`, `\address[label]{…}`, `\thanks[key]{…}` and
4//! `\thanksref{key}` (entics.cls:707-735, :860-930), so that interface is the frontmatter API as the elsart binding
5//! gives it (elsart_support_core_sty.rs): a `\thanksref{ALL}` in an author is the request for the `\thanks[ALL]{…}`
6//! note, a contact of the authors who make it, not the key printed after the name (2212.08177 "Willem HeijltjesALL",
7//! the kernel's `\thanksref` a superscript; 1812.00040, entcs). A paper whose source lacks the class falls to OmniBus,
8//! with the loader's missing-file Warning.
9use latexml_package::prelude::*;
10
11LoadDefinitions!({
12  // (the class this binding answers for: entcs, or entics)
13  let name = if Expand!(Tokens!(T_CS!("\\@currname"))).to_string() == "entcs" {
14    "entcs"
15  } else {
16    "entics"
17  };
18  let class = crate::shipped_class::RawRead {
19    shipped: true,
20    ..crate::shipped_class::RawRead::new(name, "cls")
21  };
22  if !crate::shipped_class::read_raw(&class)? {
23    return Ok(());
24  }
25  // (the frontmatter interface only: the class's own `\qed`, entics.cls:449's open box `$\Box$`, stays — elsart's
26  // `\ltx@qed` "∎" replaced it in every proof of 2204.12376, 2504.08349, 1905.09181 (entcs); repro
27  // class-bindings/entics/entics_qed_is_the_class_box)
28  let class_qed = is_defined("\\qed");
29  if class_qed {
30    Let!("\\lx@entics@qed", "\\qed");
31  }
32  RequirePackage!("elsart_support_core");
33  if class_qed {
34    Let!("\\qed", "\\lx@entics@qed");
35  }
36});
