47  // memoir's auto-table family (memoir.cls:5477-5719; manual §"Automatic
48  // tables", memman.tex:20627-20886). All four build low-level alignments the
49  // engine cannot intercept: `\autocols` does `\let\@sharp ##` + `\valign`
50  // (:5654-5657; our `\valign` is a no-op — Perl TeX_Tables.pool:555 — so the
51  // PARAM tokens reach the Stomach), `\autorows` the `\halign` analogue
52  // (:5677), `\ctabular` (:5491) runs `\@mkpream` on array.sty internals
53  // (`\@classz`/`\@acol`/`\col@sep`, deliberately absent — TeX_Tables.pool:627)
54  // and borrows tabularx's `\TX@cols`/`\TX@col@width` scratch registers.
55  // Reduce them to `\tabular`, the way tabularx_sty.rs reduces tabularx:
56  //   * `\ctabular[pos]{fmt}` — a tabular that may break across pages; `pos`
57  //     is HORIZONTAL (l/c/r), not tabular's vertical slot → dropped;
58  //   * `\autorows[width]{pos}{num}{style}{items}` — `num` COLUMNS, filled
59  //     row-major (memman.tex:1761-1765), last row short;
60  //   * `\autocols[width]{pos}{num}{style}{items}` — `num` COLUMNS, filled
61  //     column-major with `\linespercol` heights (:5665-5675: greedy
62  //     `ceil(remaining / cols_left)` per column — column 0 tallest), then
63  //     transposed to rows; `width` only sizes the columns.
64  // Items are a brace-aware comma list (a literal comma needs `\comma`,
65  // memman.tex:20852). Witnesses: memman (4 `\autorows`, 2 `\autocols`,
66  // 2 `{ctabular}` → ~157 of its errors), AroundTheBend. Guards
67  // `perfect_kernel_batch54::memoir_auto_tables_reduce_to_tabular`.
68  DefMacro!("\\ctabular[Default:c]{}", "\\tabular{#2}");
69  DefMacro!("\\endctabular", "\\endtabular");
70  DefMacro!(
71    T_CS!("\\ctabular*"),
72    "[Default:c]{}{}",
73    "\\csname tabular*\\endcsname{#2}{#3}"
74  );
75  DefMacro!(
76    T_CS!("\\endctabular*"),
77    None,
78    "\\csname endtabular*\\endcsname"
79  );
80  DefMacro!("\\autorows[Default:0pt]{}{Number}{}{}", sub[(_width, _pos, num, style, items)] {
81    let cols = (num.value_of().max(1)) as usize;
82    let items = split_top_level_commas(items.into_tokens_result()?);
83    let rows: Vec<Vec<Tokens>> = items.chunks(cols).map(|c| c.to_vec()).collect();
84    Ok(auto_table(cols, style.into_tokens_result()?, rows))
85  });
86  DefMacro!("\\autocols[Default:0pt]{}{Number}{}{}", sub[(_width, _pos, num, style, items)] {
87    let cols = (num.value_of().max(1)) as usize;
88    let items = split_top_level_commas(items.into_tokens_result()?);
89    // memoir.cls:5665-5675 `\linespercol`: column heights are assigned
90    // greedily left to right, ceil(remaining / columns left).
91    let mut heights = Vec::with_capacity(cols);
92    let mut remaining = items.len();
93    for left in (1..=cols).rev() {
94      let h = if remaining > 0 { remaining.div_ceil(left) } else { 0 };
95      heights.push(h);
96      remaining -= h;
97    }
98    let mut offsets = Vec::with_capacity(cols);
99    let mut acc = 0;
100    for h in &heights {
101      offsets.push(acc);
102      acc += h;
103    }
104    let nrows = heights.first().copied().unwrap_or(0);
105    let rows: Vec<Vec<Tokens>> = (0..nrows)
106      .map(|r| {
107        (0..cols)
108          .map(|c| {
109            if r < heights[c] {
110              items[offsets[c] + r].clone()
111            } else {
112              Tokens::new(Vec::new())
113            }
114          })
115          .collect()
116      })
117      .collect();
118    Ok(auto_table(cols, style.into_tokens_result()?, rows))
119  });

146/// Split a brace-aware comma list into its items (top-level catcode-OTHER
147/// commas only; a trailing whitespace-only item is dropped, as memoir's
148/// `\@for` would yield an empty entry there).
149fn split_top_level_commas(arg: Tokens) -> Vec<Tokens> {
150  let comma = T_OTHER!(",");
151  let mut parts: Vec<Tokens> = Vec::new();
152  let mut current: Vec<Token> = Vec::new();
153  let mut depth: i32 = 0;
154  for t in arg.unlist() {
155    match t.get_catcode() {
156      Catcode::BEGIN => depth += 1,
157      Catcode::END => depth -= 1,
158      _ => {},
159    }
160    if depth == 0 && t == comma {
161      parts.push(Tokens::new(std::mem::take(&mut current)));
162    } else {
163      current.push(t);
164    }
165  }
166  if !current.iter().all(|t| t.get_catcode() == Catcode::SPACE) {
167    parts.push(Tokens::new(current));
168  }
169  parts
170}

172/// `\tabular{<style>×cols} row & … \\ … \endtabular`, splicing the raw item
173/// tokens (never a re-tokenized string, so `\cmd{\footnote}`-style items
174/// survive).
175fn auto_table(cols: usize, style: Tokens, rows: Vec<Vec<Tokens>>) -> Tokens {
176  let style_tok = style
177    .unlist()
178    .into_iter()
179    .find(|t| t.get_catcode() != Catcode::SPACE)
180    .unwrap_or_else(|| T_LETTER!("l"));
181  let mut out = vec![T_CS!("\\par"), T_CS!("\\tabular"), T_BEGIN!()];
182  out.extend(std::iter::repeat_n(style_tok, cols));
183  out.push(T_END!());
184  for (i, row) in rows.iter().enumerate() {
185    if i > 0 {
186      out.push(T_CS!("\\\\"));
187    }
188    for (j, cell) in row.iter().enumerate() {
189      if j > 0 {
190        out.push(T_ALIGN!());
191      }
192      out.extend(cell.clone().unlist());
193    }
194  }
195  out.push(T_CS!("\\endtabular"));
196  out.push(T_CS!("\\par"));
197  Tokens::new(out)
198}
