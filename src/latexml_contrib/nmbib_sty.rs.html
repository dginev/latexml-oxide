1use latexml_package::{package::natbib_sty::natbib_invocation, prelude::*};

3// nmbib.sty raw-loads over the natbib binding, then reimplements natbib's
4// LOW-LEVEL citation engine (`\NAT@citexnum`, nmbib.sty:141) and adds
5// `\citeall` (:343 → `\@citeall` → `\@@@citeall`, :347), which opens with
6// natbib.sty:780's `\NAT@reset@parser` and the full parser/sort/state surface
7// that the natbib binding (a high-level `<ltx:cite>` emulation, ~20 `\NAT@*`)
8// does not carry — 57 undefined internals, malformed text-mode citations
9// (nmbib-sample 22 errors; Perl's natbib.sty.ltxml is the same emulation and
10// has no nmbib binding). `\citealn` is already `[\citenum{#1}]` (:338).
11// `\citeall` cites the entry with all authors and the year — `\citet*`'s
12// content — so it is emulated as that constructor, exactly how natbib's
13// public commands are. Guard: `perfect_kernel_batch54::nmbib_citeall_is_a_cite`.
14//
15// The bibliography end: `\multibibliography{bibs}` (nmbib.sty:72) only
16// writes `\bibdata` to one `.aux` per view type, and each
17// `\printbibliography{type}` (:383) inputs `\jobname-<type>.bbl` — bibtex
18// products LaTeXML never has, so the three views (timeline, sequence,
19// authors) vanished at 0 bibitems (Perl raw-loads the same package, 9
20// errors, no list). The views are one cited-entry list sorted three ways,
21// so the saved `.bib` list runs the kernel's `\lx@bibliography`
22// (`latex_constructs.pool.ltxml:3942`) once, at the first
23// `\printbibliography`; natbib's `\citep`/`\citet` select the entries.
24// The list is captured in the preamble, so the override must be in place
25// at package load, not `\AtBeginDocument`. nmbib.sty:92 also
26// `\let\@biblabel\NAT@biblabel`, a natbib internal the natbib emulation
27// does not define; the kernel default (latex.ltx `\@biblabel`, sect11.rs)
28// is restored. Guard: `06_cluster_bibliography::nmbib_prints_the_saved_bib_list_once`.
29#[rustfmt::skip]
30LoadDefinitions!({
31  InputDefinitions!("nmbib", noltxml => true, extension => Some(Cow::Borrowed("sty")));
32  // its notes as read (`natbib_invocation`): a lone `[p.~5]` stays the post-note natbib makes of it
33  DefMacro!("\\citeall OptionalMatch:* [][]{}", sub[args] {
34    let mut it = args.into_iter().skip(1);
35    let pre: Option<Tokens> = it.next().unwrap().into();
36    let post: Option<Tokens> = it.next().unwrap().into();
37    let keys: Tokens = it.next().unwrap().into();
38    Ok(natbib_invocation("\\citet", Some(Tokens!(T_OTHER!("*"))), pre, post, keys))
39  });
40  DefMacro!("\\@biblabel{}", "[#1]");
41  RawTeX!(r"\newif\ifNMBIB@rendered");
42  DefMacro!("\\multibibliography{}", "\\gdef\\NMBIB@saved@bib@list{#1}");
43  DefMacro!("\\printbibliography{}",
44    "\\ifNMBIB@rendered\\else\\expandafter\\lx@bibliography\\expandafter{\\NMBIB@saved@bib@list}\\global\\NMBIB@renderedtrue\\fi");
45});
