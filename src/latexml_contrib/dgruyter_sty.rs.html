1//! dgruyter.sty (De Gruyter's journal style, shipped by the papers that use it; not in TeX Live) — a stand-in: the
2//! paper's copy is read after it ("a stand-in's copy provides", OXIDIZED_DESIGN_DIVERGENCES #731), so the copy's
3//! definitions of names the kernel's semantic layer or this binding owns hold only inside its own code — its copy of
4//! array.sty's `\@array` (dgruyter.sty:2003-2029, calling array.sty's `\@arrayleft`), its `\let\@@array\@array` (:2031)
5//! and `\@@array`, `\@tabular`, `\@classz`, `\endtabular` wrappers (:2032-2056) — and the document's alignments stay the
6//! native ones, while the copy's extras stand as it makes them. Read raw with no binding (the ar5iv preload reads a
7//! shipped style that has none), the copy's array code took over every `array`: 2510.17333's bordered block matrices,
8//! 274 errors and two section labels lost; 2301.07931 101 errors and a Fatal. Guard
9//! `perfect_kernel_batch80a::dgruyter_copy_keeps_the_native_array`.
10//!
11//! The copy's `\maketitle` is the kernel's too, so the frontmatter dgruyter stores for its title page (:2665-2760:
12//! `\subtitle`, `\keywords`, `\classification`, the history dates, `\communicated`, `\dedication`, `\DOI`) is
13//! declared here as the frontmatter's elements; the running heads and the journal's issue data are page furniture.
14//! Guard `perfect_kernel_batch80a::dgruyter_frontmatter_is_kept`.
15use latexml_package::prelude::*;
16
17LoadDefinitions!({
18  // dgruyter.sty:2392-2475 builds its author block on authblk (`\RequirePackage[noblocks]{authblk}`, :2394): under
19  // `article`, `\author*[1]{Name}` is authblk's `\author[1]{Name}` (:2397) whose star (`\@ifstar`, :2472-2475) marks the
20  // corresponding author — a `*` after the name, "*Corresponding author:" before the affiliation note (:2418-2421). The
21  // copy's redefinitions are refused (the kernel's `\author` is locked), so authblk's binding read the star as the name:
22  // a creator "*", "[1]Karoline Busse" a paragraph after the abstract (1910.07269). authblk is loaded here, before the
23  // copy (whose own request then finds it loaded), and its `\author` takes the star: the corresponding-author contact
24  // after the creator. Under another class the copy keeps the kernel's `\author` (:2476-2481). Guard
25  // `perfect_kernel_batch86c::dgruyter_author_star_and_affiliation_labels`.
26  RequirePackage!("authblk", options => vec![s!("noblocks")]);
27  if has_meaning(&T_CS!("\\ver@article.cls")) {
28    Let!("\\lx@dg@blk@author", "\\author");
29    DefMacro!("\\author OptionalMatch:* []{}", sub[(star, label, name)] {
30      let mut call = Invocation!(T_CS!("\\lx@dg@blk@author"), vec![label, Some(name)]).unlist();
31      if star.is_some() {
32        call.extend(
33          TokenizeInternal!("\\lx@add@contact[role=corresponding]{\\@ifundefined{CorrAuth}{Corresponding author}{\\CorrAuth}}")
34            .unlist(),
35        );
36      }
37      Ok(Tokens::new(call))
38    });
39  }
40  DefMacro!("\\subtitle{}", "\\lx@add@subtitle{#1}");
41  DefMacro!("\\keywords{}", "\\lx@add@keywords{#1}");
42  // `\transabstract[<language>]{…}`, `\transkeywords[<language>]{…}` (:2733-2742): the abstract and keywords again, in
43  // the second language
44  DefMacro!("\\transabstract[]{}", "\\lx@add@abstract{#2}");
45  DefMacro!("\\transkeywords[]{}", "\\lx@add@keywords{#2}");
46  // `\classification[<scheme name>]{…}` (:2743-2745, "PACS" by default)
47  DefMacro!(
48    "\\classification[]{}",
49    "\\lx@add@classification[scheme={#1}]{#2}"
50  );
51  // with the labels dgruyter prints them under (:2886-2958), in its language's names (:1771-1810), which its captions set
52  DefMacro!(
53    "\\received{}",
54    "\\lx@add@date[role=received,name={\\@ifundefined{receivedname}{Received}{\\receivedname}~}]{#1}"
55  );
56  DefMacro!(
57    "\\revised{}",
58    "\\lx@add@date[role=revised,name={\\@ifundefined{revisedname}{revised}{\\revisedname}~}]{#1}"
59  );
60  DefMacro!(
61    "\\accepted{}",
62    "\\lx@add@date[role=accepted,name={\\@ifundefined{acceptedname}{accepted}{\\acceptedname}~}]{#1}"
63  );
64  DefMacro!(
65    "\\communicated{}",
66    "\\lx@add@pubnote[role=communicated,name={\\@ifundefined{communicatedname}{Communicated by}{\\communicatedname}:~}]{#1}"
67  );
68  DefMacro!("\\dedication{}", "\\lx@add@pubnote[role=dedication]{#1}");
69  // dgruyter.sty:370 `\def\DOI#1{\gdef\@DOI{https://doi.org/#1}}`
70  DefMacro!(
71    "\\DOI{}",
72    "\\@add@frontmatter{ltx:note}[role=doi,name={DOI:~}]{https://doi.org/#1}"
73  );
74  for proto in [
75    "\\runningtitle{}",
76    "\\runningauthor{}",
77    "\\journalname{}",
78    "\\journalyear{}",
79    "\\journalvolume{}",
80    "\\journalissue{}",
81    "\\startpage{}",
82    "\\articletype{}",
83    "\\articlesubtype{}",
84  ] {
85    def_macro_noop(proto)?;
86  }
87});
