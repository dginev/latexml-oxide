1use latexml_package::prelude::*;

3LoadDefinitions!({
4  // stix2.sty defines the AMS symbol set under its own names (`\nmid`
5  // stix2.sty:1316, `\varnothing` :1294, `\twoheadrightarrow` :896, …), so a
6  // document that loads stix2 instead of amssymb (rbt-mathnotes.sty:129)
7  // needs the same math tokens: reuse amssymb's Unicode bindings, then add
8  // the stix-only names. The former stub shadowed the raw file and left them
9  // all undefined. Guard: `perfect_kernel_batch56::sweep46_single_name_gaps`.
10  RequirePackage!("amssymb");
11  DefMath!("\\twoheadrightarrowtail", "\u{2916}", role => "ARROW");
12  DefMath!("\\twoheadleftarrowtail", "\u{2B3B}", role => "ARROW");
13  // stix2.sty:116-126 `\stix@MathSymbol{\Gamma}{\mathalpha}{operators}{"00}`…: the capital Greek letters upright, over
14  // a class's italic ones (llncs.cls:379-389; KNOWN_PERL_ERRORS #1147), unless `nomath` leaves the math fonts alone
15  // (:41, :59-571). Review rev79e finding 1 (witness 2606.09074). Any other option is not the binding's concern.
16  DeclareOption!("nomath", {
17    AssignValue!("stix2_nomath" => true);
18  });
19  DeclareOption!(None, {});
20  ProcessOptions!();
21  if !lookup_bool("stix2_nomath") {
22    declare_capital_greek("", None)?;
23  }
24});
