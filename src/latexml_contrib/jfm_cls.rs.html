326/// Does this copy of jfm.cls set `\le`/`\ge` to the slanted ⩽ ⩾ when it finds the AMS symbol fonts (an uncommented
327/// `\let\le=\leqslant`, 2103.10260's :1361)?
328fn slants_inequalities(class: &str) -> bool {
329  class.lines().any(|line| {
330    line
331      .split('%')
332      .next()
333      .unwrap_or_default()
334      .contains("\\let\\le=\\leqslant")
335  })
