1//! apa6.cls (APA 6th edition manuscripts, TeX Live apa6 2023/08/07 v2.35): the class is loaded raw, and its numbered author
2//! forms — `\twoauthors` … `\sixauthors`, `\onetwoauthors`/`\twooneauthors` (apa6.cls:745-759) and `\twoaffiliations` …
3//! `\sixaffiliations`, `\onetwoaffiliations`/`\twooneaffiliations` (:761-777), which store each name in `\@authorOne` …
4//! `\@authorSix` and each affiliation in `\@affilOne` … `\@affilSix` — become the frontmatter as its title page lays
5//! them out: every layout (man :1143-1214; jou and doc `\put@one@authaffil`/`\put@two@authaffil` :1321-1365 at :1378-1414)
6//! sets the n-th name over the n-th affiliation, so each name is a creator carrying the affiliation of its number.
7//! Under the raw class (the rawclasses lever) only the class's refused `\maketitle` read those stores, so the names and
8//! affiliations were lost (2102.04789: "Najko Jahn, Lisa Matthias, Mikael Laakso" and their Göttingen address;
9//! 1710.04173 `\sixauthors`/`\sixaffiliations`). The one-author `\author`/`\affiliation` stay the kernel's and K11's.

12/// The numbered author and affiliation forms of apa.cls and apa6.cls (apa.cls:427-459 is apa6.cls:745-777 line for
13/// line), handed to the frontmatter at `\maketitle`, where the class reads them: `mask` names the class's
14/// author-masking flag (apa6.cls:126 the `mask` option, read at :1143 and :1378; apa.cls has none, `None`).
15///
16/// The title page picks its layout by which numbered names are defined (:1145-1214, :1381-1414: `\@authorTwo` …
17/// `\@authorSix`, each checked only when the one before it is). A name slot may hold a group of names over its one
18/// affiliation (1401.2054 `\twoauthors{Zhiyong Zhang, Kaifeng Jiang, Haiyan Liu}{In-Sue Oh}`, 1402.6720
19/// `{Edgar C.\ Merkle and Dongjun You}`): the slot is split as the kernel's `\author` splits its names,
20/// appended to the slots before it (`\lx@add@authors@list`), and its affiliation given to each of them (`annotate=new`: the creators of this slot, the
21/// earlier ones having theirs). A numbered form also sets `\@author` and `\@affil` to
22/// its first name and affiliation (:747, :762), so once `\@authorTwo` is defined the numbered names and affiliations
23/// are the title page's whatever `\author` and `\affiliation` said: the creators the kernel's `\author` made are
24/// cleared (2011.06172 `\author{Author}` "hack around some bugs in apa.cls" beside `\threeauthors`), and so is the
25/// one-author affiliation store, which K11 would otherwise give every author at `\maketitle` (`harvest_stores`;
26/// 1811.01821 `\affiliation{University of Cambridge}` beside `\twoaffiliations`). Without numbered names the title
27/// page sets `\@author` over `\@affil`, a numbered affiliation form's first one included (1605.09288, 1609.04156
28/// `\author{…}` with `\twoaffiliations`; the class warns "More affiliations than authors defined", :785): K11's own.
29pub fn define_numbered_author_forms(mask: Option<&str>) -> Result<()> {
30  let (open, close) = match mask {
31    Some(flag) => (s!(r"\@ifundefined{{{flag}}}{{"), "}{}"),
32    None => (String::new(), ""),
33  };
34  RawTeX!(&s!(
35    r"\def\lx@apa@pair#1{{\expandafter\expandafter\expandafter\lx@add@authors@list\expandafter\expandafter\expandafter{{\csname @author#1\endcsname}}\@ifundefined{{@affil#1}}{{}}{{\expandafter\expandafter\expandafter\lx@apa@affiliation\expandafter\expandafter\expandafter{{\csname @affil#1\endcsname}}}}}}
36\def\lx@apa@affiliation#1{{\lx@add@affiliation[annotate=new]{{#1}}}}
37\def\lx@apa@numbered@authors{{\@ifundefined{{@authorTwo}}{{}}{{\global\let\@affil\@empty{open}\lx@clear@creators\lx@apa@pair{{One}}\lx@apa@pair{{Two}}\@ifundefined{{@authorThree}}{{}}{{\lx@apa@pair{{Three}}\@ifundefined{{@authorFour}}{{}}{{\lx@apa@pair{{Four}}\@ifundefined{{@authorFive}}{{}}{{\lx@apa@pair{{Five}}\@ifundefined{{@authorSix}}{{}}{{\lx@apa@pair{{Six}}}}}}}}}}{close}}}\global\let\lx@apa@numbered@authors\relax}}"
38  ));
39  AddToMacro!("\\lx@maketitle@begin", "\\lx@apa@numbered@authors");
40  Ok(())
41}

43/// Interpret the class `name` raw (as the document asked for it: [`latexml_package::shipped_class::requested_shipped_class`], so
44/// a class only named like the family — TeX Live's apa6e.cls, a paper's `apa6mod.cls`, which the prefix fallback
45/// answers with this binding — is read as itself), then give its numbered author forms to the frontmatter
46/// ([`define_numbered_author_forms`]) when it has them. A paper whose source lacks the class falls to OmniBus, with the
47/// loader's missing-file Warning.
48pub fn apa_family_class(name: &str, mask: Option<&str>) -> Result<()> {
49  // (the class read raw by a binding is still a raw class to K11, as when the rawclasses lever reads it: its title-page
50  // stores — `\shorttitle`, `\affiliation`, `\keywords`, `\abstract`, … — reach the frontmatter, `input_definitions`
51  // firing the reroutes at the end of this class load)
52  if !crate::shipped_class::read_raw(&crate::shipped_class::RawRead {
53    shipped: true,
54    ..crate::shipped_class::RawRead::new(name, "cls")
55  })? {
56    return Ok(());
57  }
58  // (a class only named like the family, without the numbered stores, is left as it is)
59  if !IsDefined!(&T_CS!("\\twoauthors")) || !IsDefined!(&T_CS!("\\twoaffiliations")) {
60    return Ok(());
61  }
62  define_notes(mask)?;
63  define_numbered_author_forms(mask)
64}

66/// The title page's notes as apa7's binding makes them (apa7_cls.rs): `\note{…}` (apa6.cls:776, apa.cls:459 `\long\def
67/// \note#1{\long\def\@note{#1}}`), printed under the authors and affiliations (apa6.cls:1215-1217, :1416-1418,
68/// :1677-1679; apa.cls:987-989, :1236-1238), and apa6's `\authornote{…}` (apa6.cls:779 → `\@acks`), printed under its
69/// "Author Note" heading (:1222-1234, :1460-1466, :1700-1707), neither under `mask` (the class's `\@ifundefined
70/// {apaSix@maskauthoridentity}` around them). `\note` is no store K11's table names (a note means something else in
71/// most classes), and K11's `\authornote` does not know the mask: 2406.08022 writes its author note ("Author Note
72/// \newline Correspondence concerning this article …") as `\note`, lost under the raw class.
73/// apa.cls's `\acknowledgements{…}` (:461 `\long\def\acknowledgements#1{\long\def\@acks{#1}}`; apa6.cls has none) is
74/// the paper's acknowledgements under the class's "Author Note" heading (`\acksname`, :194): the doc and jou title
75/// pages print the store (:1262-1267, :1549-1557), where the refused `\maketitle` lost it with 0 errors, and man mode
76/// as a closing section (:760-764 `\AtEndDocument`, `\section{\acksname}\@acks`), which an unset store skips.
77fn define_notes(mask: Option<&str>) -> Result<()> {
78  let (open, close) = match mask {
79    Some(flag) => (s!(r"\@ifundefined{{{flag}}}{{"), "}{}"),
80    None => (String::new(), ""),
81  };
82  RawTeX!(&s!(
83    r"\long\def\note#1{{{open}\lx@add@frontmatter{{ltx:note}}[role=note]{{#1}}{close}}}"
84  ));
85  if IsDefined!(&T_CS!("\\authornote")) {
86    RawTeX!(&s!(
87      r"\long\def\authornote#1{{{open}\lx@add@frontmatter{{ltx:note}}[role=authornote]{{#1}}{close}}}"
88    ));
89  }
90  if IsDefined!(&T_CS!("\\acknowledgements")) {
91    RawTeX!(&s!(
92      r"\long\def\acknowledgements#1{{{open}\lx@add@frontmatter{{ltx:acknowledgements}}[name={{\acksname}}]{{#1}}{close}}}"
93    ));
94  }
95  Ok(())
96}
