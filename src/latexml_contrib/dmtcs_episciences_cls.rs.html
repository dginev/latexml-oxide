1//! Stub for dmtcs-episciences.cls (Discrete Mathematics & Theoretical
2//! Computer Science journal class, Episciences platform).
3//!
4//! The raw cls preamble has font/page-layout machinery and conditional
5//! `\@firsthead` formatting (real `\publicationdata` builds running
6//! headers via tikz). Mid-load fails before reaching the publication
7//! macros, leaving `\publicationdata`, `\papertype`, `\fundinginfo`,
8//! `\accepted` undefined. Witness 2309.05874, 2309.12265.
9use latexml_package::prelude::*;
10
11LoadDefinitions!({
12  // Base on OmniBus, NOT article: Perl ships no dmtcs binding and falls back to
13  // OmniBus, which provides the generic journal frontmatter (`\received`,
14  // `\revised`, `\acknowledgements`/`\acknowledgments`, …). Basing on plain
15  // `article` left those undefined where Perl (OmniBus) is clean. The
16  // dmtcs-specific setters below are defined AFTER and override OmniBus.
17  // Witness 1904.12329 (`\received{}`/`\revised{}`/`\acknowledgements`).
18  LoadClass!("OmniBus");
19  // dmtcs-episciences.cls loads no amsmath (1904.05533's, 2209.09223's copies): amsmath's commands for papers that use
20  // them unloaded, its "Old form" error once the paper loads it; the paper's copy decides
21  // (`amsmath_sty::require_amsmath_beyond_original`, KNOWN_PERL_ERRORS #993).
22  amsmath_sty::require_amsmath_beyond_original(
23    latexml_package::shipped_class::shipped_copy_loads("cls", "amsmath")?,
24    &[],
25  )?;
26  RequirePackage!("amsthm");
27  RequirePackage!("amssymb");
28  RequirePackage!("graphicx");
29  RequirePackage!("hyperref");
30
31  // \publicationdata{volume}{year}{issue}{doi}{dates}{accepted} —
32  // 6-arg setter for running headers. Preserve as frontmatter note.
33  DefMacro!(
34    "\\publicationdata{}{}{}{}{}{}",
35    "\\@add@frontmatter{ltx:note}[role=dmtcs-publicationdata]{Volume #1 (#2), \\##3, doi:#4}"
36  );
37  // \papertype{T} — single-arg paper-type marker (Article/Editorial/…).
38  DefMacro!(
39    "\\papertype{}",
40    "\\@add@frontmatter{ltx:note}[role=papertype]{#1}"
41  );
42  // \fundinginfo{text} — funding statement.
43  DefMacro!(
44    "\\fundinginfo{}",
45    "\\@add@frontmatter{ltx:note}[role=funding]{#1}"
46  );
47  // \accepted{date} — accepted-date setter.
48  DefMacro!(
49    "\\accepted{}",
50    "\\@add@frontmatter{ltx:note}[role=accepted]{Accepted: #1}"
51  );
52  // \processdates / \endprocessdates — internal parser hooks; no-op.
53  def_macro_noop("\\processdates")?;
54  def_macro_noop("\\endprocessdates")?;
55  def_macro_noop("\\getinfo")?;
56
57  // \affiliation{text} — dmtcs-episciences.cls L251: stores affil text
58  // (`\gdef\@affil{#1}`). Preserve as ltx:contact frontmatter.
59  // Witness 2403.03614.
60  DefMacro!(
61    "\\affiliation{}",
62    "\\@add@to@frontmatter{ltx:creator}{\\@@@affiliation{#1}}"
63  );
64  DefConstructor!(
65    "\\@@@affiliation{}",
66    "^ <ltx:contact role='affiliation'>#1</ltx:contact>"
67  );
68  // \affiliationmark{tag} — L229: superscript marker on author/affil.
69  def_macro_noop("\\affiliationmark{}")?;
70  // \keywords{text} — L311: stores keywords (`\gdef\@keywords{#1}`).
71  // Preserve as frontmatter keywords.
72  DefMacro!("\\keywords{}", "\\@add@frontmatter{ltx:keywords}{#1}");
73  // dmtcs-episciences.cls (1211.0978's copy :225-236, as 115 of 120 sampled):
74  // `\publicationdetails{volume}{year}{issue}{number}{id}` sets the first page's head, "DMTCS vol. <volume>:<issue>,
75  // <year>, #<number>", linked to the volume's page, and the copyright line with its license (:231-235): the reference
76  // and the license, as pubnotes. (Under the
77  // class's `ojs` option a four-argument form (:56-70) replaces it; no sampled paper passes `ojs`.) The number sets of
78  // the class, :483-487. Run 340: 113 of 120 sampled papers call `\publicationdetails` (~320 first errors; 1211.0978).
79  // The id (#5) is only the head's link target, no printed text: the binding's declared drop, so the title-page check
80  // wants none of it (OXIDIZED_DESIGN_DIVERGENCES #953 "106t").
81  DefMacro!(
82    "\\publicationdetails{}{}{}{}{}",
83    "\\lx@add@pubnote[role=journalref]{DMTCS vol. #1:#3, #2, \\##4}\
84     \\lx@add@copyright{#2 by the author(s)}\
85     \\lx@add@pubnote[role=license]{Distributed under a Creative Commons Attribution 4.0 International License}",
86    drops => [5]
87  );
88  RawTeX!(
89    r"\newcommand{\reals}{\ensuremath{\mathbb R}}\newcommand{\rationals}{\ensuremath{\mathbb Q}}
90\newcommand{\naturals}{\ensuremath{\mathbb N}}\newcommand{\integers}{\ensuremath{\mathbb Z}}
91\newcommand{\complexes}{\ensuremath{\mathbb C}}\newcommand{\primes}{\ensuremath{\mathbb P}}"
92  );
