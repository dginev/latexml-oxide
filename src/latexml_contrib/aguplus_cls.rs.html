1//! aguplus.cls (AGU++, P. W. Daly, 1999/08/19 1.6b): the class is loaded raw — it reads the AGUTeX package its option
2//! names (aguplus.cls:47-71 `\getAGU`, `\InputIfFileExists{\AGUpack.sty}`: agupp.sty, agums.sty, jgrga.sty, …) — and
3//! the title-page commands the AGUTeX packages share with AASTeX, which they derive from, become the frontmatter as the
4//! AAS binding makes them (aas_support_sty.rs, Perl aas_support.sty.ltxml:121-134). Under the raw class (the
5//! rawclasses lever) they were the class's typesetting: `\affil` set its text into the `\@title` box (aguplus.cls:236)
6//! that `\xmaketitle` typesets at the first `\section` (:192, :225), so the affiliations were body paragraphs, and
7//! `\authoraddr` (agupp.sty:154-158) gathered `\authaddr@list`, printed only by the class's own end of the references
8//! (agupp.sty:266-306 `\endreferences`, `\endthebibliography`), which the kernel's bibliography replaces, so every
9//! author address was lost (1107.5305, 1203.5042, physics9702009, 1010.3998, 0811.3118).
10//!
11//! The class also carries a copy of natbib 7.0 (aguplus.cls:861-1173, `\ModuleVersion{natbib}[1999/05/28 7.0 (PWD)]`)
12//! with square brackets (:864 `\NAT@open{[} \NAT@close{]}`): it loads natbib's binding with `square`, as the copy
13//! is natbib's. Without it `\citet`/`\citep` were the copy's, printing a bold "?" for every reference (the kernel's
14//! bibliography writes no `\bibcite`; 1010.3998: 75), and `\cite[][]{key}` the kernel's (1107.5305).
15use latexml_package::prelude::*;
16
17LoadDefinitions!({
18  InputDefinitions!("aguplus", noltxml => true, extension => Some(Cow::Borrowed("cls")));
19  RequirePackage!("natbib", options => vec![s!("square")]);
20  // The class's `\title`, `\author` and `\affil` only build its title-page box (aguplus.cls:227-238 `\setbox\@title=
21  // \vbox{… \box\@title \aguauthor{#1}}`), which the frontmatter replaces: the refused `\author` is not replayed
22  // for its side effects (`replay_class_setter`, sect05.rs), as the box it would build is the kernel's `\@title` text
23  // ("Missing number" for every author, 1010.3998).
24  RawTeX!(r"\global\let\lx@dropped@author\relax\global\let\lx@dropped@title\relax");
25  // An affiliation is every preceding author's that has none yet (`\author{…}\affil{…}` per author,
26  // physics9702009), as the AAS binding's (aas_support_sty.rs).
27  DefMacro!("\\affil{}", "\\lx@add@affiliation[annotate=new]{#1}");
28  // The author addresses the class gathers (agupp.sty:154-158 `\authoraddress`, `\let\authoraddr`) and prints in a
29  // block after the references, each naming its author ("S. G. Claudepierre, Laboratory …"; 1010.3998): each is
30  // placed with the author it spells, else the only author, else the block (`\lx@add@contact@placed`). They are
31  // gathered as the class gathers them, `\authoraddr` being given before `\author` too (0811.3118), and handed on
32  // where the authors are queued and the frontmatter not yet made: at `\maketitle`, else at the first section (ahead
33  // of the kernel's frontmatter fallback on `\@startsection@hook`), else at the end of the document.
34  RawTeX!(
35    r"\let\lx@agu@authoraddrs\@empty
36\def\lx@agu@hand@authoraddrs{\lx@agu@authoraddrs\global\let\lx@agu@authoraddrs\@empty}
37\begingroup\toks0=\expandafter{\@startsection@hook}\xdef\@startsection@hook{\noexpand\lx@agu@hand@authoraddrs\the\toks0}\endgroup
38\AtEndDocument{\lx@agu@hand@authoraddrs}"
39  );
40  AddToMacro!("\\lx@maketitle@begin", "\\lx@agu@hand@authoraddrs");
41  DefMacro!(
42    "\\authoraddress{}",
43    "\\g@addto@macro\\lx@agu@authoraddrs{\\lx@add@contact@placed[role=address]{}{#1}}"
44  );
45  Let!("\\authoraddr", "\\authoraddress");
46  // `\altaffilmark` marks an author, `\altaffiltext{n}{…}` gives the marked authors the affiliation (agupp.sty:150,
47  // aguplus.cls:239-242 a title footnote), as the AAS binding's.
48  aas_support_sty::altaffil_markup()?;
49  // The dates printed under the references (agupp.sty:68-70, :119-126 `\@dates`), and the slug comment over the
50  // title (agupp.sty:110-117 `\@makeslugcmmnt`; aguplus.cls:245 `\let\date=\slugcomment`), as the AAS binding's.
51  DefMacro!(
52    "\\received{}",
53    "\\lx@add@date[role=received,name=Received]{#1}"
54  );
55  DefMacro!(
56    "\\revised{}",
57    "\\lx@add@date[role=revised,name=Revised]{#1}"
58  );
59  DefMacro!(
60    "\\accepted{}",
61    "\\lx@add@date[role=accepted,name=Accepted]{#1}"
62  );
63  // agupp.sty:109 `\slugcomment` (and `\date`, aguplus.cls:245): the slug line, its text alone (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #921)
64  DefMacro!("\\slugcomment{}", "\\lx@add@pubnote[role=note,name={}]{#1}");
65  Let!("\\date", "\\slugcomment");
66});
