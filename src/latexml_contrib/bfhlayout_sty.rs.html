1//! bfhlayout.sty (bfh-ci, Bern University of Applied Sciences): the package is loaded raw, and the title page's foot —
2//! `\department` and `\institute` (`\titlefooterleft`, bfhlayout.sty:735-753) and `\titlefooterright` (:755-756),
3//! printed only by the title page's scrlayer footer layer (`title.BFH.footer`), a page style LaTeXML never typesets —
4//! becomes frontmatter notes (roles `department`, `institute`, `titlefooter`) after the class's print site,
5//! `\maketitle`, with the values set then (the demos set them after `\begin{document}`). A store the kernel already
6//! handed to the authors as an affiliation there (`\department`/`\institute` read by a raw class's `\@maketitle`,
7//! frontmatter_stores.rs; DEMO-BFHThesis) is not repeated. User ruling 2026-10-03: the department is frontmatter, as
8//! a letter's sender (OXIDIZED_DESIGN_DIVERGENCES #412). SHARED: Perl with raw classes drops them alike. Residual: a
9//! document's own `\titlefooterleft{…}`, replacing the department and institute there, is not read. Witnesses
10//! bfh-ci DEMO-BFHFactsheet, DEMO-BFHProjektProposal, DEMO-BFHThesis. Guard
11//! `perfect_kernel_batch61::bfh_title_footer_is_frontmatter`.
12use latexml_package::prelude::*;
13
14LoadDefinitions!({
15  InputDefinitions!("bfhlayout", noltxml => true, extension => Some(Cow::Borrowed("sty")));
16  DefMacro!("\\lx@bfh@frontmatter", sub[_args] {
17    let mut notes = String::from(r"\global\let\lx@bfh@frontmatter\relax");
18    for (store, role) in [("department", "department"), ("institute", "institute"), ("titlefooterright", "titlefooter")] {
19      if !lookup_bool(&s!("lx_store_handed_{store}")) {
20        notes.push_str(&s!(r"\ifx\@{store}\@empty\else\lx@add@frontmatter{{ltx:note}}[role={role}]{{\@{store}}}\fi"));
21      }
22    }
23    Ok(TokenizeInternal!(TeXString::assembled(notes)))
24  });
25  RawTeX!(
26    r"\AtBeginDocument{\let\lx@bfh@maketitle\maketitle\def\maketitle{\lx@bfh@maketitle\lx@bfh@frontmatter}}"
27  );
28});
