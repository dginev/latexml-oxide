1//! The JHEP class family as shipped with papers and not in TeX Live: `JINST` (Journal of Instrumentation) and `CECS`,
2//! both JHEP3-derived (`\auto@maketitle` title page, `\abstract@cs`, `\Jrece@cs`). One binding serves them, the union
3//! of their frontmatter commands; `JHEP`, `JHEP2` and `JHEP3` themselves have the Perl-ported emulating bindings
4//! (latexml_package `jhep_cls.rs`), whose routing this follows.
5use latexml_package::prelude::*;
6
7/// Interpret the shipped class `name` raw (as the document asked for it, a path-prefixed copy included:
8/// [`latexml_package::shipped_class::requested_shipped_class`]), then route the frontmatter it stores for its own title page to the
9/// frontmatter, as the JHEP binding does (Perl JHEP.cls.ltxml):
10/// - `\received`, `\revised`, `\accepted`, `\published` (JINST.cls:764-774) and the copyright dates (`\JINSTcopydate`
11///   :777, `\JHEPcopydate`, `\CECScopydate`) are frontmatter dates, after the class's own setter has run for the flags
12///   its `\AtBeginDocument` checks in `published` mode (:1592-1607);
13/// - `\keywords` (:826-832, a catcode-changing reader storing `\@keywords`) is the frontmatter's keywords, its flag set;
14/// - `\dedicated`, `\preprint`, `\conference` are publication notes and `\email` (:798) the author's email contact, as
15///   the JHEP binding has them (JHEP.cls.ltxml:43);
16/// - the frontmatter is made (the kernel `\maketitle`) where `\auto@maketitle` typesets the title page at
17///   `\begin{document}`: it then kills the commands that page used (:488-506: `\global\let\email\@gobble`,
18///   `\thanks`, `\author`, `\title`, …), and the queued `\author` blocks, digested any later, lost their emails
19///   (0704.3706, gr-qc0303113, 41 of 2,918 papers);
20/// - `\@maketitle` (:462-560), the title page `\auto@maketitle` typesets at `\begin{document}`, is empty: everything it
21///   lays out is in the frontmatter, and it typeset a second title, author block and abstract in the body (2601.17161,
22///   0704.3706, 0801.2206, 0805.3984); the rest of `\auto@maketitle` (footnote style, the commands it kills after the
23///   title) runs as the class has it. The class's own `\maketitle` only warns that it is ignored (:461), and is not
24///   deposited.
25///
26/// The `\author` and `\abstract` checks of `\AtBeginDocument` read flags the kernel's locked `\author`/`\abstract` set
27/// by replaying the class's own (OXIDIZED_DESIGN #482). A paper whose source lacks the class falls to OmniBus, with
28/// the loader's missing-file Warning.
29pub(crate) fn jhep_family_class(name: &str) -> Result<()> {
30  if !crate::shipped_class::read_raw(&crate::shipped_class::RawRead {
31    shipped: true,
32    ..crate::shipped_class::RawRead::new(name, "cls")
33  })? {
34    return Ok(());
35  }
36  // a class only named like the family (the prefix fallback's `JINSTfoo`) is left as it is: without the JHEP title
37  // page (`\auto@maketitle`), nothing it stores is laid out only there
38  if !IsDefined!(&T_CS!("\\auto@maketitle")) {
39    return Ok(());
40  }
41  for (cs, frontmatter) in [
42    (
43      "received",
44      "\\lx@add@date[role=received,name={\\receivedname}]{#1}",
45    ),
46    (
47      "revised",
48      "\\lx@add@date[role=revised,name={\\revisedname}]{#1}",
49    ),
50    (
51      "accepted",
52      "\\lx@add@date[role=accepted,name={\\acceptedname}]{#1}",
53    ),
54    (
55      "published",
56      "\\lx@add@date[role=published,name={\\publishedname}]{#1}",
57    ),
58    ("JINSTcopydate", "\\lx@add@date[role=copydate]{#1}"),
59    ("JHEPcopydate", "\\lx@add@date[role=copydate]{#1}"),
60    ("CECScopydate", "\\lx@add@date[role=copydate]{#1}"),
61  ] {
62    route_after_class_setter(cs, frontmatter)?;
63  }
64  DefMacro!(
65    "\\keywords{}",
66    "\\lx@add@keywords[name={\\keywordsname~}]{#1}\\global\\csname @keywordstrue\\endcsname"
67  );
68  DefMacro!("\\email Semiverbatim", "\\lx@add@email{#1}");
69  // The frontmatter is made where the class typesets its title page: `\auto@maketitle` then kills the commands that
70  // page has used (`\global\let\email\@gobble`, `\thanks`, `\author`, …), which the queued `\author` blocks,
71  // digested any later, still need.
72  Let!("\\lx@jhep@auto@maketitle", "\\auto@maketitle");
73  RawTeX!(r"\def\auto@maketitle{\maketitle\lx@jhep@auto@maketitle}");
74  // No label, as the class prints none (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #921): JINST.cls, JHEP.cls's sibling (1508.06575's copy),
75  // sets the dedication in italics, the report number in the preprint header and the conference line alone — the
76  // role stays.
77  DefMacro!(
78    "\\dedicated{}",
79    "\\lx@add@pubnote[role=dedication,name={}]{#1}"
80  );
81  DefMacro!(
82    "\\preprint{}",
83    "\\lx@add@pubnote[role=preprint,name={}]{#1}"
84  );
85  DefMacro!(
86    "\\conference{}",
87    "\\lx@add@pubnote[role=conference,name={}]{#1}"
88  );
89  Let!("\\@maketitle", "\\@empty");
90  Let!("\\lx@dropped@maketitle", "\\relax");
91  Ok(())
92}

94/// `\<cs>{…}` as the frontmatter's `frontmatter` (its `#1` the argument), after the class's own `\<cs>` where it has one.
95fn route_after_class_setter(cs: &str, frontmatter: &str) -> Result<()> {
96  let token = T_CS!(s!("\\{cs}"));
97  if IsDefined!(&token) {
98    let class_setter = s!("\\lx@jhep@class@{cs}");
99    Let!(&T_CS!(&class_setter), token);
100    RawTeX!(&s!(
101      "\\def\\{cs}#1{{\\csname lx@jhep@class@{cs}\\endcsname{{#1}}{frontmatter}}}"
102    ));
103  } else {
104    RawTeX!(&s!("\\def\\{cs}#1{{{frontmatter}}}"));
105  }
106  Ok(())
107}

110  jhep_family_class("JINST")?;
