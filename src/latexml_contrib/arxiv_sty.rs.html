1//! arxiv.sty — the widely-bundled "arXiv preprint" style
2//! (George Kour, <https://github.com/kourgeorge/arxiv-style>), shipped
3//! *inside* the paper's own source rather than installed in texmf.
4//!
5//! No Perl LaTeXML binding exists (the file is not a CTAN package at all),
6//! so same-host Perl reports the same `Error:undefined:\keywords` whenever
7//! the raw .sty is not read. Witness 2605.02338: Perl and Rust both emit
8//! exactly one error, `undefined:\keywords`.
9//!
10//! **Why a binding when the file is right there on disk.** `\keywords` is
11//! only undefined in configurations that do NOT raw-load style files
12//! (`INCLUDE_STYLES` off — the bare default). Under `--includestyles` /
13//! the ar5iv profile the paper's own arxiv.sty loads and everything works;
14//! a binding that *replaced* that load would throw away the file's
15//! `\@maketitle`, its `abstract`/`table` environment redefinitions and its
16//! section formatting for no gain. So this binding is deliberately
17//! **configuration-aware**: when raw style loading is available it does
18//! nothing but hand control back to the real file; otherwise it supplies
19//! the small set of definitions a document actually calls by name. The
20//! gate is the same `lookup_bool("INCLUDE_STYLES")` the raw-load path
21//! itself uses (`latexml_core/src/binding/content.rs` L776).
22//!
23//! The bare-mode definitions are a verbatim port of arxiv.sty L10, L29-31,
24//! L33 and L44-47 — including `\and`'s local rebinding to `$\cdot$`, which
25//! is what separates the keyword list in the rendered output. Getting the
26//! ARITY right is the point: an undefined `\keywords` is recovered as a
27//! zero-argument `<ltx:ERROR/>` and its braced argument then leaks into
28//! the body as an unlabelled paragraph.
29//!
30//! Witnesses: 2605.02338 (`arxiv.sty` + `\keywords{a \and b \and c}`),
31//! 2605.10111 (the `PRIMEarxiv.sty` fork — see `primearxiv_sty.rs`).
32use latexml_package::prelude::*;
33
34/// arxiv.sty L44-47, verbatim. `\keywordname` differs between the upstream
35/// file and its PRIMEarxiv fork, so it is passed in. Kept as raw TeX (not
36/// a `DefMacro!` body) so that the space arxiv.sty's line break leaves at
37/// the end of `\and` survives exactly as the real file has it.
38pub(crate) fn define_keywords(keywordname: &str) -> Result<()> {
39  RawTeX!(&s!(r"\def\keywordname{{{}}}", keywordname));
40  RawTeX!(
41    r"\def\keywords#1{\par\addvspace\medskipamount{\rightskip=0pt plus1cm
42\def\and{\ifhmode\unskip\nobreak\fi\ $\cdot$
43}\noindent\keywordname\enspace\ignorespaces#1\par}}"
44  );
45  Ok(())
46}
47
48/// The shared bare-mode fallback for the arxiv.sty family: everything a
49/// document calls by name, and nothing that only affects page layout.
50pub(crate) fn define_bare_fallbacks(keywordname: &str) -> Result<()> {
51  // arxiv.sty L10 / L33. On the miss-handler path these two are picked up
52  // by the `\RequirePackage` dependency scan of the raw file; a registered
53  // binding bypasses that scan, so request them explicitly.
54  RequirePackage!("geometry");
55  RequirePackage!("fancyhdr");
56  // arxiv.sty L29-31.
57  RawTeX!(r"\newcommand{\headeright}{A Preprint}");
58  RawTeX!(r"\newcommand{\undertitle}{A Preprint}");
59  RawTeX!(r"\newcommand{\shorttitle}{\@title}");
60  define_keywords(keywordname)
61}
62
63LoadDefinitions!({
64  if lookup_bool("INCLUDE_STYLES") {
65    // Raw style loading is on: the paper's own arxiv.sty is authoritative.
66    InputDefinitions!("arxiv", noltxml => true, extension => Some(Cow::Borrowed("sty")));
67  } else {
68    define_bare_fallbacks(r"{\bfseries \emph{Keywords}}")?;
69  }
70});
