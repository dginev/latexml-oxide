1//! Stub for CEUR-WS ceurart.cls.
2//!
3//! ceurart.cls is built on top of scrartcl + expl3/xparse. The class
4//! defines `\sep` via `\tex_def:D \sep{\unskip,}` inside an expl3 block,
5//! which our raw-load can't reliably execute. Most user-facing
6//! frontmatter macros (`\ead`, `\fnmark`, etc.) use `\NewDocumentCommand`
7//! with expl3 bodies that don't fully unfurl either.
8//!
9//! Provide content-preserving stubs that route the author-supplied text
10//! into either frontmatter notes (\tnotetext, \fntext, \cortext) or
11//! contact entries (\ead, \orcidauthor), so no body material is dropped.
12//!
13//! Witness: 2501.13802, 2501.14238, 2501.16855, 2501.17381, 2502.01404,
14//! 2502.02753, 2502.06743 — all `Error:undefined:\sep`.
15//!
16//! The base class is the one ceurart.cls loads, scrartcl (OXIDIZED_DESIGN_DIVERGENCES #736): the paper's copy, read
17//! after the binding, finds it loaded, so its own `\LoadClass{scrartcl}` is a repeat and its KOMA calls
18//! (`\KOMAoptions{DIV=10}`, :63-68) meet KOMA (2509.11496, 2608.23214, 2310.02658, 2505.16276). OmniBus, the base
19//! this binding had before, is provided over it ([`provide_over_base_class`]): its `{keywords}` environment and the
20//! frontmatter setters scrartcl leaves undefined.
21use latexml_package::{package::omnibus_cls::provide_over_base_class, prelude::*};

183  // ceurart.cls:381-500 `\title[mode=<mode>]{…}`: every call adds a line to the title block (`\g_ceur_title_seq`, printed
184  // in order by `\@title`), styled by its mode — `title` (the default, `O{mode = title}`), `sub`, `alt`, `trans`,
185  // `transsub`, `ascii` — and `\subtitle{…}` is `\title[mode=sub]{…}` (:500). The optional argument is that key list,
186  // never a short title. The main title is the `title` one; a `sub` call is the subtitle, a translated one a note, as
187  // acmart's `\translatedtitle`/`\translatedsubtitle` (acmart_cls.rs). Under the kernel `\title` (locked; the copy's
188  // `\RenewDocumentCommand\title` is refused) a `\title[mode=sub]{…}` replaced the main title (19 of the 112 ceurart
189  // papers of 2506+2509: 2509.11496, 2509.11492).

191  DefMacro!("\\title[]{}", sub[(keys, text)] {
192    let mode = keys.as_ref().and_then(|keys| ceur_title_mode(&keys.to_string()));
193    let head = match mode.as_deref() {
194      Some("sub") => "\\@add@frontmatter{ltx:subtitle}",
195      Some("transsub") => "\\@add@frontmatter{ltx:note}[role=translatedsubtitle]",
196      Some("trans") => "\\@add@frontmatter{ltx:note}[role=translatedtitle]",
197      Some("alt") => "\\@add@frontmatter{ltx:note}[role=alternativetitle]",
198      Some("ascii") => "\\@add@frontmatter{ltx:note}[role=asciititle]",
199      _ => "\\lx@ceur@title",
200    };
201    let mut out = TokenizeInternal!(head).unlist();
202    out.push(T_BEGIN!());
203    out.extend(text.unlist());
204    out.push(T_END!());
205    Ok(Tokens::new(out))
206  }, locked => true);

254/// The `mode` key of a ceurart `\title[<keys>]` (ceurart.cls:381-382, a `.choice:`), trimmed; `None` without one.
255fn ceur_title_mode(keys: &str) -> Option<String> {
256  keys.split(',').find_map(|pair| {
257    let (key, value) = pair.split_once('=')?;
258    (key.trim() == "mode").then(|| value.trim().to_string())
259  })
