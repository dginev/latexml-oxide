1//! Stub binding for SIAM siamart-family classes (siamart, siamart220329, ...).
2//!
3//! Activated via prefix-match: any class name starting with "siamart" routes
4//! here. Defines the high-level macros papers use (\newsiamthm, \newsiamremark,
5//! \headers, \dedicatory) on top of OmniBus's article-like behaviour.
6use latexml_core::definition::origin::{DefinitionOrigin, OriginGuard};
7use latexml_package::prelude::*;
8
9use crate::shipped_class::class_option_given;
10
11LoadDefinitions!({
12  LoadClass!("OmniBus");
13  // siamart220329.cls L58 (siamart250211.cls:64): \RequirePackage[leqno]{amsmath}
14  // — SIAM numbers equations on the left (DIVERGENCES #336; 2605.02838).
15  RequirePackage!("amsmath", options => vec!["leqno".to_string()]);
16  RequirePackage!("amsthm");
17  // Many siamart papers pre-define colors in their macros.tex before
18  // their own `\usepackage{xcolor}`. Defensive xcolor load matches Perl
19  // behaviour. Witness 2405.17955 (EPCID).
20  // Eager xcolor preload removed for Perl parity: it makes a later document
21  // xcolor[table] load a no-op, so colortbl/array never load and array m{}/b{}
22  // columns break (Unrecognized tabular template -> Extra alignment tab). The
23  // document loads xcolor itself; color/definecolor stay via hyperref->color.
24  // See ifacconf_cls.rs and SYNC_STATUS (eager-xcolor cluster).
25  // siamart220329 L1361: \RequirePackage{algorithm}.
26  RequirePackage!("algorithm");
27  RequirePackage!("url");
28  // siamart220329 L1285: \RequirePackage{hyperref}[6.83] (unconditional).
29  // Mirror so papers using \hidelinks/\href/\hypersetup don't error.
30  // hyperref MUST come before cleveref (cleveref errors out otherwise).
31  // Witness 2407.00765 (siamart220329 with `[hidelinks,…]` class option).
32  RequirePackage!("hyperref");
33  // siamart220329 L1371: \RequirePackage[capitalize,nameinlink]{cleveref}.
34  // Loaded AFTER hyperref to satisfy cleveref's ordering check.
35  // Witness 2501.11060 (Error:latex:cleveref must be loaded after hyperref!).
36  RequirePackage!("cleveref");
37  // siamonline220329 L1676: \RequirePackage[mathlines]{lineno}.
38  RequirePackage!("lineno");
39  // ifpdf is auto-loaded inside epstopdf; our binding triggers
40  // \ifpdf usage during epstopdf raw-load, so preload it here.
41  RequirePackage!("ifpdf");
42  RawTeX!(
43    r"\newtheorem{theorem}{Theorem}
44\newtheorem{lemma}[theorem]{Lemma}
45\newtheorem{corollary}[theorem]{Corollary}
46\newtheorem{proposition}[theorem]{Proposition}
47\newtheorem{definition}[theorem]{Definition}"
48  );
49
50  // \newsiamthm{name}{title}, \newsiamremark{name}{title}: siamart220329 L1452,
51  // L1469. Both reduce to `\newtheorem{name}[theorem]{title}`; the theoremstyle
52  // tweaks are visual and don't affect XML.
53  DefMacro!("\\newsiamthm{}{}", r"\newtheorem{#1}[theorem]{#2}");
54  DefMacro!("\\newsiamremark{}{}", r"\newtheorem{#1}[theorem]{#2}");
55
56  // siamart frontmatter primitives. Round-34 surpass-Perl: preserve
57  // \dedicatory, \fundingsource, and \funding as author-typed
58  // frontmatter notes (the funding text is often a real funding
59  // statement worth keeping). \headers{left}{right} → running-head
60  // text, preserve as ltx:note.
61  DefMacro!(
62    "\\headers{}{}",
63    "\\@add@frontmatter{ltx:note}[role=runningheads]{#1 / #2}"
64  );
65  DefMacro!(
66    "\\dedicatory{}",
67    "\\@add@frontmatter{ltx:note}[role=dedicatory]{#1}"
68  );
69  DefMacro!(
70    "\\fundingsource{}",
71    "\\@add@frontmatter{ltx:note}[role=funding-source]{#1}"
72  );
73  // siamart220329.cls:1266 loads xr-hyper, whose `\externaldocument[][nocite]{ex_supplement}` (the SIAM template's
74  // article ↔ supplement idiom) imports the other document's labels from its `.aux` (xr_sty.rs; a file name with `_`
75  // stays Semiverbatim, witness 2402.12241).
76  RequirePackage!("xr-hyper");
77  // siamart220329 L1130: \funding{...} writes a marked line in the
78  // titlepage. Preserve as a frontmatter ltx:acknowledgements via
79  // \@add@frontmatter so the element lands at top-level no matter
80  // where the macro is invoked. Papers commonly nest \funding{...}
81  // inside \thanks{...} (which is rendered as ltx:note); an inline
82  // DefConstructor of ltx:acknowledgements there fires
83  // Error:malformed:ltx:acknowledgements isn't allowed in <ltx:note>.
84  // Witness 2311.08549.
85  DefMacro!(
86    "\\funding{}",
87    "\\@add@frontmatter{ltx:acknowledgements}[name=Funding]{#1}"
88  );
89  // {MSCcodes} env — siamart220329 L743 wraps content in an "@abssec"
90  // (frontmatter section). Mirror as keywords-like classification block.
91  DefEnvironment!(
92    "{MSCcodes}",
93    "<ltx:classification scheme='MSC'>#body</ltx:classification>"
94  );
95
96  // {AMS} env — siamart190516 L742 same pattern (AMS subject
97  // classification). Some older SIAM templates use {AMS} instead of
98  // {MSCcodes}. Witness 2306.11286, 2306.13351 (siamart190516).
99  DefEnvironment!(
100    "{AMS}",
101    "<ltx:classification scheme='AMS'>#body</ltx:classification>"
102  );
103
104  // {@abssec}{title} — siamonline190516.cls L810 (`\newenvironment{@abssec}[1]`):
105  // a titled frontmatter paragraph (`#1.` bold heading, then body). The
106  // public abstract/keywords/AMS/keyword envs wrap it, but papers also
107  // use `\begin{@abssec}{Author Information…}…\end{@abssec}` directly.
108  // `\begin{@abssec}` resolves via `\csname @abssec\endcsname`, so it
109  // works regardless of `@` catcode. siamonline190516.cls is reported
110  // missing-file in Perl (24 undefined-macro errors there); our siamart
111  // binding covers everything else, so defining this env makes the
112  // conversion clean. `{@doisec}` (L820) is the same shape with a
113  // trailing rule. Witness 2005.11911 (`\documentclass{siamonline190516}`).
114  // Use `inline-logical-block` (Misc.class) rather than `ltx:para`:
115  // papers place `\begin{@abssec}` inside frontmatter note contexts
116  // (e.g. an author/acknowledgements block already wrapped in
117  // `ltx:note`), where a block `ltx:para` is rejected
118  // (`malformed:ltx:para isn't allowed in <ltx:note>`). Misc.class is
119  // accepted in both inline and block positions.
120  DefEnvironment!(
121    "{@abssec}{}",
122    "<ltx:inline-logical-block class='ltx_abssec'><ltx:text font='bold'>#1. </ltx:text>#body</ltx:inline-logical-block>"
123  );
124  DefEnvironment!(
125    "{@doisec}{}",
126    "<ltx:inline-logical-block class='ltx_doisec'><ltx:text font='bold'>#1. </ltx:text>#body</ltx:inline-logical-block>"
127  );
128  // What every siamart/siamonline release papers ship defines (1116 to 251216; 2006.12120's siamart220329.cls, as the
129  // 169 siamart and 55 siamonline copies have it) and papers call (run 347: 31+19 siamart and 26 siamonline papers), a
130  // copy without one leaving it to the paper (`DefinitionOrigin::Fallback`):
131  // - :1426 `\proofbox`, the proofs' hollow box (`\theoremsymbol{\proofbox}`, 1805.01916; 30 papers);
132  // - :491-498 `\slugger{journal}{year}{vol}{no}{pages}`, the title page's slug line, page furniture (1311.2828;
133  //   13 papers); :517-522 `\siamprelabel` (`SM` under the `supplement` option), the supplement's label prefix;
134  // - :749 `{DOI}` "DOI." (1901.02925; 11+19 papers), :747 `{AM}` "AMS subject classification" (1801.00197), :744
135  //   `{MSCcode}` "MSC code" (2101.02911), :746 `{ACM}` (2006.12120), as `{MSCcodes}` and `{AMS}`;
136  // - :710-718 `{romannum}`, `{remunerate}`, lists numbered (i), (ii)… and 1., 2.… (1902.07263, 1601.00044);
137  // - :924-930 `\appendixnotitle`, an appendix numbered but untitled (1703.08345);
138  // - siamonline250211.cls :837 `{Code}` "Code repository" (10 copies, 2306.16314, 2509.14771), siamonline :550
139  //   `\dedication{text}`, a dedication under the title (2210.02418).
140  {
141    let _class = OriginGuard::new(DefinitionOrigin::Fallback);
142    let supplement = class_option_given("supplement")?;
143    RawTeX!(if supplement {
144      r"\def\siamprelabel{SM}\def\siampretitle{Supplementary Materials: }"
145    } else {
146      r"\def\siamprelabel{}\def\siampretitle{}"
147    });
148    Let!("\\proofbox", "\\openbox");
149    RawTeX!(
150      r"\def\slugger#1#2#3#4#5{}
151\newcommand\DOIname{DOI\@}
152\newenvironment{DOI}{\begin{@doisec}{\DOIname}}{\end{@doisec}}
153\newcounter{rmnum}
154\newenvironment{romannum}{\begin{list}{{\upshape (\roman{rmnum})}}{\usecounter{rmnum}}}{\end{list}}
155\newcounter{muni}
156\newenvironment{remunerate}{\begin{list}{{\upshape \arabic{muni}.}}{\usecounter{muni}}}{\end{list}}
157\newcommand{\appendixnotitle}{\section{}}
158\newcommand\codelink{Code repository\@}
159\newenvironment{Code}{\begin{@abssec}{\codelink}}{\end{@abssec}}"
160    );
161    DefEnvironment!(
162      "{AM}",
163      "<ltx:classification scheme='AMS'>#body</ltx:classification>"
164    );
165    DefEnvironment!(
166      "{MSCcode}",
167      "<ltx:classification scheme='MSC'>#body</ltx:classification>"
168    );
169    DefEnvironment!(
170      "{ACM}",
171      "<ltx:classification scheme='ACM'>#body</ltx:classification>"
172    );
173    DefMacro!(
174      "\\dedication{}",
175      "\\@add@frontmatter{ltx:note}[role=dedication]{#1}"
176    );
177  }
178});
