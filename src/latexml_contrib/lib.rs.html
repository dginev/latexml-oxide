915/// Runtime lookup: route `filename` (e.g. `"MnSymbol.sty"`) through its
916/// compiled `load_definitions` fn, or return `None` when the filename has no
917/// registered binding. Splits on the *first* `.` so a multi-part extension
918/// like `("foo", "code.tex", …)` matches correctly — mirrors
919/// `latexml_package::dispatch`. (The former `hobby.code.tex` refusal stub was
920/// retired in perfect-kernel batch 54: the real file raw-loads with 0 errors;
921/// guard `perfect_kernel_batch54::hobby_code_tex_raw_loads`.)
922#[inline]
923fn matches_entry(target: &str, name: &str, ext: &str) -> bool {
924  target.len() == name.len() + 1 + ext.len()
925    && target.starts_with(name)
926    && target.as_bytes()[name.len()] == b'.'
927    && target.ends_with(ext)
928}
929
930#[inline]
931fn matches_entry_nocase(target: &str, name: &str, ext: &str) -> bool {
932  target.len() == name.len() + 1 + ext.len()
933    && target[..name.len()].eq_ignore_ascii_case(name)
934    && target.as_bytes()[name.len()] == b'.'
935    && target[name.len() + 1..].eq_ignore_ascii_case(ext)
936}
937
938pub fn dispatch(filename: &str) -> Option<Result<()>> {
939  if !filename.contains('.') {
940    return None;
941  }
942  // Strip directory prefix: `\documentclass{Definitions/mdpi}` →
943  // basename `mdpi`. Perl Package.pm L2167-2170 (FindFile_fallback)
944  // does the same. Without this, paper-bundled classes like
945  // `Definitions/mdpi.cls` miss the registered binding and cascade.
946  // See latexml_package::dispatch.
947  let base_only = filename
948    .rsplit_once(['/', '\\'])
949    .map_or(filename, |(_, b)| b);
950  // Perl pathname_find L383-389: strict-case first, case-insensitive fallback
951  // (matches `\documentclass{jhep}` against `JHEP.cls.ltxml`-style entries).
952  // See latexml_package::dispatch for the parallel comment.
953  BINDINGS
954    .iter()
955    .find(|(name, extension, _)| matches_entry(filename, name, extension))
956    .or_else(|| {
957      BINDINGS
958        .iter()
959        .find(|(name, extension, _)| matches_entry(base_only, name, extension))
960    })
961    .or_else(|| {
962      BINDINGS
963        .iter()
964        .find(|(name, extension, _)| matches_entry_nocase(filename, name, extension))
965    })
966    .or_else(|| {
967      BINDINGS
968        .iter()
969        .find(|(name, extension, _)| matches_entry_nocase(base_only, name, extension))
970    })
971    .map(|(name, extension, loader)| {
972      // K1 provenance: a binding's own definitions are `Binding`; a `.pool`
973      // entry is a pool; a binding a fallback class requires stays part of its
974      // guess, `Fallback` (`input_definitions`), and one a binding borrows stays
975      // `Borrowed` (`require_package_borrowed`).
976      use ::latexml_core::definition::origin::{DefinitionOrigin, current_origin, with_origin};
977      let origin = if *extension == "pool" {
978        DefinitionOrigin::Pool
979      } else if matches!(
980        current_origin(),
981        DefinitionOrigin::Fallback | DefinitionOrigin::Borrowed
982      ) {
983        current_origin()
984      } else {
985        DefinitionOrigin::Binding
986      };
987      // (the frontmatter observer notes the binding whose load defines `\author`, its conventions' key; under its
988      // shadow switch only: OXIDIZED_DESIGN_DIVERGENCES #837 — jpconf, jcappub, jinstpub are contrib bindings)
989      ::latexml_engine::base_utilities::note_binding_load(name, extension, || {
990        with_origin(origin, loader)
991      })
992    })
993    // biblatex style/variant packages (`biblatex-chicago`, `biblatex-apa`,
994    // `biblatex-ieee`, `biblatex-nature`, `biblatex-science`, `biblatex-mla`,
995    // `biblatex-phys`, `biblatex-chem`, …) each begin their real `.sty` with
996    // `\RequirePackage{biblatex}` and then only *configure* it; the
997    // configuration commands (`\DeclareFieldFormat`, `\renewbibmacro`, …) run
998    // in the preamble. Without a binding for the variant name, that
999    // `\RequirePackage{biblatex}` never happens, so those commands are
1000    // undefined the moment they are used — before the `\addbibresource`
1001    // autoload trigger fires — and the whole bibliography collapses (0
1002    // bibitems). Route every `biblatex-<style>.sty` to the biblatex binding,
1003    // reproducing the `\RequirePackage{biblatex}` the variant would have done.
1004    // Perl LaTeXML ships no biblatex binding at all, so this is surpass-Perl,
1005    // consistent with the existing biblatex contrib. Witness arXiv 2605.11180
1006    // (`\usepackage[authordate,backend=biber]{biblatex-chicago}` via
1007    // `biblatex-aer.tex`; html_feedback #6601): `\DeclareFieldFormat`
1008    // undefined, empty References, before this fallback.
1009    .or_else(|| {
1010      let lower = base_only.to_ascii_lowercase();
1011      (lower.ends_with(".sty")
1012        && lower.len() > "biblatex-.sty".len()
1013        && lower.starts_with("biblatex-"))
1014      .then(|| biblatex_sty::load_variant(&lower[..lower.len() - ".sty".len()]))
1015    })
1016}

1018/// All registered (name, extension) pairs for this crate's BINDINGS.
1019/// Mirror of `latexml_package::binding_names`. Consumed by
1020/// `find_file(notex=true)` to detect compiled-binding existence across
1021/// all registered extensions (cls/sty/def/pool/code.tex/...). The class
1022/// names (entries with `ext == "cls"`) flow into `load_class`'s
1023/// Perl-parity prefix-match fallback via the
1024/// `state::get_class_binding_names()` filtered view.
1025pub fn binding_names() -> &'static [(&'static str, &'static str)] {
1026  use std::sync::OnceLock;
1027  static NAMES: OnceLock<Vec<(&'static str, &'static str)>> = OnceLock::new();
1028  NAMES
1029    .get_or_init(|| {
1030      BINDINGS
1031        .iter()
1032        .map(|(name, ext, _)| (*name, *ext))
1033        .collect()
1034    })
1035    .as_slice()
1036}
