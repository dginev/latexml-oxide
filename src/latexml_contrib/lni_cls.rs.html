1//! lni.cls (GI Lecture Notes in Informatics): the class is loaded raw. From v2.0 (TL 2025 ships 2025/02/05 v2.0) it
2//! builds its author block by copying authblk's code (lni.cls:362 `\RequirePackage{authblk}`, :384-435
3//! `\renewcommand\author[4][]`, :444-464 `\renewcommand\affil[2][]`) into accumulators only its `\maketitle` reads; that
4//! block becomes the frontmatter as authblk's binding makes it: `\author[marks]{name}{e-mail}{ORCID}` an author carrying
5//! its affiliation marks, its e-mail and ORCID as contacts (the class prints them in the affiliation footnote,
6//! :425-432), `\affil[mark]{…}` authblk's. Under the raw class the copy's `\affil` replaced the binding's and the
7//! kernel lock refused its `\author`, so the affiliations, e-mails and ORCIDs were lost and the last two arguments were
8//! left as stray preamble text (2608.21389). A paper's own older copy (v0.5, v1.7) loads no authblk and keeps LaTeX's
9//! `\author[]{}` (`\renewcommand{\author}{\@dblarg\@@author}`), which the kernel's reads (2203.09493, 1705.02823,
10//! 2210.04484).
11use latexml_package::{package::authblk_sty::define_author_and_affil, prelude::*};
12
13LoadDefinitions!({
14  // (the kernel's `\author`, before the class's authblk replaces it: a call written for lni v1 is its, below)
15  Let!("\\lx@lni@kernel@author", "\\author");
16  InputDefinitions!("lni", noltxml => true, extension => Some(Cow::Borrowed("cls")));
17  // (the title's marks lni's own `\maketitle` prints, every version's: lni.cls:475-477 `\renewcommand\thefootnote
18  // {\@arabic\c@footnote}`, v0.5's :89-93; 2407.04435 "1, 1, 1,2, 3", 1705.02823, 1705.08889)
19  if lookup_bool("\\maketitle:redefined@class") {
20    DefMacro!("\\lx@title@thefootnote", "\\@arabic\\c@footnote");
21    // (its `\thanks` LaTeX's, each stepping the counter; set back after the title only by a `\maketitle` that says so,
22    // v0.5's, article's copy (:116), not v1.7's or v2.0's (lni.cls:475-514): a body note after a v2.0 title's
23    // `\thanks` 3 is 4)
24    latexml_package::engine::latex_constructs::declare_title_footnote_counting(
25      maketitle_resets_footnotes(),
26    )?;
27  }
28  // (the class read raw by this binding is a raw class to K11, its title-page stores rerouted at the end of the class
29  // load: lni.cls `\subtitle`; the binding's `\author`/`\affil` below stay its own)
30  if package_loaded("authblk") {
31    // (a call written for lni v1, `\author[running names]{names}` with no e-mail and ORCID arguments after it, is read
32    // as v1's `\@dblarg\@@author` reads it, by the kernel's `\author[short]{names}`: v2.0's `[4][]` would take the next
33    // tokens for them and the running names for authblk labels; pdflatex stops there, 2407.04435)
34    define_author_and_affil()?;
35    Let!("\\lx@lni@author", "\\author");
36    // (and lni's `\affil` sets the footnote counter to its label, so a title `\thanks` is numbered after the last one:
37    // lni.cls:463, v1.10's :468, 2412.04178)
38    Let!("\\lx@lni@affil", "\\affil");
39    DefMacro!("\\affil OptionalSemiverbatim {}", r"\lx@lni@affil[#1]{#2}\setcounter{footnote}{#1}", robust => true);
40    DefMacro!(
41      "\\author[]{}",
42      r"\@ifnextchar\bgroup{\lx@lni@author@v{#1}{#2}}{\lx@lni@kernel@author[#1]{#2}}"
43    );
44    DefMacro!(
45      "\\lx@lni@author@v{}{}{}{}",
46      r"\if\relax\detokenize{#1}\relax\lx@lni@author{#2}\else\lx@lni@author[#1]{#2}\fi\if\relax\detokenize{#3}\relax\else\lx@add@email{#3}\fi\if\relax\detokenize{#4}\relax\else\lx@add@orcid{#4}\fi"
47    );
48  }
49});

51/// Whether the class's own `\maketitle`, which the lock refused (`\lx@dropped@maketitle`), sets the footnote counter back
52/// to 0: v0.5's copy's own `\setcounter{footnote}{0}` (its :116), read as spelled; a copy writing `\c@footnote\z@` is
53/// missed, its body notes counting on.
54fn maketitle_resets_footnotes() -> bool {
55  lookup_definition(&T_CS!("\\lx@dropped@maketitle"))
56    .ok()
57    .flatten()
58    .is_some_and(|definition| match definition.get_expansion() {
59      Some(ExpansionBody::Tokens(body)) => body
60        .to_string()
61        .replace(' ', "")
62        .contains("\\setcounter{footnote}{0}"),
63      _ => false,
64    })
65}
