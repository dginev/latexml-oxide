1//! gtpart.cls — Geometry & Topology Publications' class (Colin Rourke, 2006; not in TeX Live). A paper that ships it has
2//! it read raw; one whose source lacks it gets OmniBus with the loader's missing-file Warning, and the class's
3//! publication metadata setters, which print nothing (gtpart.cls:573-600, `\def\corresponding#1{\relax}` …): the
4//! template asks for them left empty (`\corresponding{}`, the corresponding editor's, beside `\proposed` and
5//! `\seconded`; no author's correspondence). OmniBus's `\doi`, `\received`, `\revised` and `\accepted` stay its own.
6//! 1108.0474, 1201.1579, 1301.0616, 1301.5693, 1410.6974, 1512.00353 (no copy); 0901.0208, 2404.01499 (a copy).
7use latexml_package::prelude::*;
8
9const GTPART: crate::shipped_class::RawRead = crate::shipped_class::RawRead {
10  shipped: true,
11  ..crate::shipped_class::RawRead::new("gtpart", "cls")
12};
13
14LoadDefinitions!({
15  if crate::shipped_class::read_raw(&GTPART)? {
16    return Ok(());
17  }
18  // gtpart.cls:573-600 (`\MR`'s link at :212 is redefined so at :576)
19  RawTeX!(
20    r"\def\givenname#1{\relax}\def\surname#1{\relax}\def\MR#1{\relax}\def\Zbl#1{\relax}
21\def\issuenumber#1{\relax}\def\publishedonline#1{\relax}\def\proposed#1{\relax}\def\seconded#1{\relax}
22\def\corresponding#1{\relax}\def\editor#1{\relax}\def\version#1{\relax}\def\arxivreference#1{\relax}
23\def\arxivpassword#1{\relax}\def\volumenumber#1{\relax}\def\volumename#1{\relax}\def\publicationyear#1{\relax}
24\def\papernumber#1{\relax}\def\startpage#1{\relax}\def\endpage#1{\relax}\def\published#1{\relax}"
25  );
26});
