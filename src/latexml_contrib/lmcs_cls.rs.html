1//! Stub for lmcs.cls (Logical Methods in Computer Science journal class).
2//!
3//! lmcs.cls is NOT distributed in TeX Live (papers bundle it, but the corpus
4//! copies don't reach our search path), so Perl LaTeXML — which ships no lmcs
5//! binding — emits `Can't find binding for class lmcs (using OmniBus)` and
6//! falls back to **OmniBus**. The base class is the one lmcs.cls:33 loads,
7//! `\LoadClass[11pt,reqno]{amsart}` (OXIDIZED_DESIGN_DIVERGENCES #736: the paper's copy, read after the binding, finds
8//! it loaded), with OmniBus, the base this binding had before, provided over it; and the class's own theorem set (lmcs.cls:696-761, below)
9//! eagerly: a paper whose preamble does `\newtheorem{remark}[thm]{…}` and whose
10//! body uses `\begin{thm}`/`\begin{lem}` shares the class's `thm` counter,
11//! numbered within the section as pdflatex numbers it (OmniBus's lazy autoloads
12//! once supplied `thm`/`lem` with document-wide numbers). Witness 1607.01886. The raw cls relies on pgfmath/tikz/XeTeXLinkBox for
13//! ORCID-logo rendering (fails mid-load), so `\lmcsdoi`/`\lmcsheading`/
14//! `\lmcsorcid` get content-preserving stubs below. Witness 2305.14448,
15//! 2305.19985, 1607.04128 (graphicx).

155  // `\author`): `\author[short]{Name}[a,b]` prints the name with the letters raised after it (`\rsuper`, :398-404), and

158  // it. The letters are the link: a name's letters request the addresses printed with them (the `\inst` request of
159  // llncs's binding, a frontmatter annotation, so the name is the name alone, FIDELITY_CHARTER §4), and an address is
160  // labelled by its place in the list. The papers of the earlier classes write the letters themselves, `Name\rsuper a`
161  // and `\address{{\lsuper a}Univ}` (1103.0510, cs0701140, 1202.6352's `{\lsuper{b,c}}`), which mean the same. A paper

270/// An author's name as the frontmatter reads it: each `\rsuper{a,b}` the letters' address requests ([`inst_requests`]),
271/// each `\lmcsorcid` the author's ORCID contact. `labelled` is set when a letter is requested.
272fn lmcs_author_name(name: &[Token], labelled: &mut bool) -> Vec<Token> {
273  let mut out = Vec::with_capacity(name.len());
274  let mut at = 0;
275  while at < name.len() {
276    let t = name[at];
277    if t == T_CS!("\\rsuper") {
278      let (argument, next) = read_argument(name, at + 1);
279      out.extend(inst_requests(&argument, labelled));
280      at = next;
281      continue;
282    }
283    out.push(if t == T_CS!("\\lmcsorcid") {
284      T_CS!("\\lx@add@orcid")
285    } else {
286      t
287    });
288    at += 1;
289  }
290  out
291}

293/// `\inst{…}` for each label of a comma list, its letters lowercased as the class's `\MakeLowercase` prints them: a
294/// letter or number requests the address it labels; a footnote symbol (`\rsuper{a,*}`, 2012.13129) stays shown.
295fn inst_requests(labels: &[Token], labelled: &mut bool) -> Vec<Token> {
296  let mut out = Vec::new();
297  for item in label_items(labels) {
298    *labelled |= item.iter().any(|t| {
299      matches!(t.get_catcode(), Catcode::LETTER)
300        || t.with_str(|s| s.chars().all(|c| c.is_ascii_digit()))
301    });
302    out.push(T_CS!("\\inst"));
303    out.push(T_BEGIN!());
304    out.extend(item);
305    out.push(T_END!());
306  }
307  out
308}

310/// The labels of a comma list, each without its spaces and its letters lowercased.
311fn label_items(labels: &[Token]) -> Vec<Vec<Token>> {
312  split_tokens(Tokens::new(labels.to_vec()), vec![SplitDelim::Token(
313    T_OTHER!(","),
314  )])
315  .into_iter()
316  .map(|item| {
317    item
318      .unlist()
319      .into_iter()
320      .filter(|t| t.get_catcode() != Catcode::SPACE)
321      .map(|t| {
322        let lower = t.with_str(|s| s.to_ascii_lowercase());
323        if t.get_catcode() == Catcode::LETTER && t.with_str(|s| s != lower) {
324          T_LETTER!(&lower)
325        } else {
326          t
327        }
328      })
329      .collect::<Vec<_>>()
330  })
331  .filter(|item| !item.is_empty())
332  .collect()
333}

335/// The argument of a macro at `at`: a brace group's content, else the one token there; and where it ends.
336fn read_argument(tokens: &[Token], at: usize) -> (Vec<Token>, usize) {
337  let mut start = at;
338  while start < tokens.len() && tokens[start].get_catcode() == Catcode::SPACE {
339    start += 1;
340  }
341  if start >= tokens.len() {
342    return (Vec::new(), start);
343  }
344  if tokens[start].get_catcode() != Catcode::BEGIN {
345    return (vec![tokens[start]], start + 1);
346  }
347  let mut depth = 0usize;
348  for (end, t) in tokens.iter().enumerate().skip(start) {
349    match t.get_catcode() {
350      Catcode::BEGIN => depth += 1,
351      Catcode::END => {
352        depth -= 1;
353        if depth == 0 {
354          return (tokens[start + 1..end].to_vec(), end + 1);
355        }
356      },
357      _ => {},
358    }
359  }
360  (tokens[start + 1..].to_vec(), tokens.len())
361}

363/// The letters an address opens with (`{\lsuper a}Univ`, `\lsuper{b,c} Univ`, `{\lsuper{a}}Univ`) and the address
364/// without them; `None` when it opens with none. An address of spacing alone (`\vskip-6 pt`) is left empty.
365fn leading_lsuper(body: Vec<Token>) -> (Option<Vec<Token>>, Vec<Token>) {
366  let start = body
367    .iter()
368    .take_while(|t| t.get_catcode() == Catcode::SPACE)
369    .count();
370  let mut letters = None;
371  let mut rest_at = start;
372  if body.get(start) == Some(&T_CS!("\\lsuper")) {
373    let (argument, next) = read_argument(&body, start + 1);
374    letters = Some(argument);
375    rest_at = next;
376  } else if body
377    .get(start)
378    .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
379  {
380    let (group, next) = read_argument(&body, start);
381    let inner = group
382      .iter()
383      .take_while(|t| t.get_catcode() == Catcode::SPACE)
384      .count();
385    if group.get(inner) == Some(&T_CS!("\\lsuper")) {
386      let (argument, after) = read_argument(&group, inner + 1);
387      if group[after..]
388        .iter()
389        .all(|t| t.get_catcode() == Catcode::SPACE)
390      {
391        letters = Some(argument);
392        rest_at = next;
393      }
394    }
395  }
396  let rest: Vec<Token> = body[rest_at..].to_vec();
397  // (spacing alone is no address)
398  static SPACING: Lazy<Regex> = Lazy::new(|| {
399    Regex::new(
400      r"\\(?:[vh]skip\s*[-+]?\s*[0-9.]*\s*(?:pt|em|ex|cm|mm|in|bp|sp|pc|dd|cc|mu)|[vh]space\*?\s*\{[^}]*\}|(?:small|med|big)skip|par)",
401    )
402    .unwrap()
403  });
404  let spacing = SPACING
405    .replace_all(&Tokens::new(rest.clone()).to_string(), "")
406    .trim()
407    .is_empty();
408  if spacing {
409    return (letters, Vec::new());
410  }
411  (letters, rest)
