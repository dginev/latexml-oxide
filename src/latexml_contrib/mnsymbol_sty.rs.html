1//! MnSymbol.sty — Achim Blumensath's symbol fonts (TeX Live `tex/latex/mnsymbol/MnSymbol.sty`, v1.4 2007/01/21).
2//!
3//! No Perl binding exists (Perl reads the package raw, where its seven NFSS symbol fonts give no glyphs). MnSymbol
4//! replaces the math symbol fonts and declares every name in them, one font slot after another, with
5//! `\Decl@Mn@Char`/`\Decl@Mn@Alias` (`\DeclareMathSymbol`), the big operators with `\Decl@Mn@Op`/`\Decl@Mn@Int`, and
6//! the delimiters with `\Decl@Mn@Open`/`\Decl@Mn@Close` (`\DeclareMathDelimiter`). The binding gives each declared name
7//! a math symbol: the tables below hold all 1,307 `\Decl@Mn@Char`/`\Decl@Mn@Alias` names of MnSymbol.sty:250-1584 in
8//! declaration order, the 55 `\Decl@Mn@Op`/`\Decl@Mn@Int` operators of :1586-1650, and the delimiters, accents and
9//! commands of :1651-1853 are bound by hand below.
10//!
11//! **Glyphs.** A name's code point is unicode-math's for the same name (`unicode-math-table.tex`), else the Unicode
12//! TeX4ht gives the name's font slot (`tex4ht/ht-fonts/unicode/MnSymbol/MnSymbol{A,B,C,D,F}-Bold10.htf`, HTML entities
13//! decoded: `&gt;` is `\gtr`'s >; a slot is the running `\Set@Mn@Sym` count), else the base glyph with U+0338 for a negated `\n…` name, else a glyph chosen
14//! from MnSymbol.pdf's tables, checked against its drawings (the footlines' bar on the side the name says,
15//! `\rightfootline` ⟞; `\upmodels` ⫫, 2604.25507; `\ocirc` ⊚, where unicode-math's `\ocirc` is the ring accent). A glyph Unicode lacks (most of the diagonal arrows, the filled spoons, the
16//! smile/frown stacks) takes the code point of its family's right-pointing member — the XMTok keeps the command's
17//! name. **Roles** follow the declared class as the sibling symbol bindings do: `\mathrel` an ARROW for an
18//! arrow/harpoon/spoon/mapsto family, else RELOP; `\mathbin` ADDOP for the plus/minus/cup/cap/vee/wedge/triangle/
19//! diamond families (math_common's `\cup`, `\vee`, `\diamond`), else MULOP; `\mathord` none (ID for the dots);
20//! `\mathop` SUMOP or INTOP.
21//!
22//! **Names already defined keep their definition** (OXIDIZED_DESIGN_DIVERGENCES #545). MnSymbol redeclares the whole
23//! AMS set (`\Decl@Mn@Sym` first `\let`s the name undefined); here a name the kernel, amsmath or amssymb already
24//! defines keeps that definition, whose role and meaning are richer than the slot's class, and a name sharing an
25//! already defined name's slot and class is `\let` to it (`\nin` to `\notin`, `\rightvdash` to `\vdash`). So
26//! `\Mn@Fake@Missing` (:29-78) has nothing left to fake, and its undefining of `\eth`, `\mho`, `\Finv`… is not
27//! carried out: a document that uses them keeps them. The package's own declarators stay available to a document
28//! under the same rule (2410.20575 `\Decl@Mn@Open{\lsem}{MNlargesymbols}{'102}` after loading MnSymbol); a new name
29//! takes the glyph the tables give its slot (`slot_glyph`), and a slot they do not map goes to the kernel's
30//! `\DeclareMathSymbol` with an `Info:unmapped`.
31//!
32//! Witnesses (run 340: 13,470 papers load MnSymbol, 3,724 ended in Error, their first errors `\nin` 175, `\udots`
33//! 173, `\sumint` 165, `\cupdot` 130, `\strokedint` 113, `\lsem` 108): 2410.20575, 2210.15345 (`\let\vec\undefined`
34//! before MnSymbol, whose `\DeclareMathAccent\vec` defines it again), 2508.12496 (`\checkmark`), 2609.07645
35//! (`\llangle`), 2604.25507 (`\upmodels`), atableau (`\bigcircle`), biblatex-apa6-test (`\rcurvearrowse`),
36//! univie-ling (`\blacktriangleright`).

73/// Bind one `\Decl@Mn@Char`: an already defined name keeps its definition; a name whose slot-and-class sibling is
74/// defined becomes that sibling (`\let`); any other is a math symbol of its glyph and role.
75fn declare_char(&(cs, glyph, role, same_as, ..): &MnChar) -> Result<()> {
76  if !unclaimed(cs) {
77    return Ok(());
78  }
79  if let Some(sibling) = same_as.iter().find(|name| !unclaimed(name)) {
80    let_i(&T_CS!(cs), &T_CS!(*sibling), None);
81    return Ok(());
82  }
83  def_math(T_CS!(cs), None, glyph.to_string(), MathPrimitiveOptions {
84    role: role_of(role),
85    ..Default::default()
86  })
87}

89/// The math class a table role stands for (the inverse of the module comment's role rule).
90fn class_of_role(role: &str) -> &'static str {
91  match role {
92    "ARROW" | "RELOP" => "\\mathrel",
93    "ADDOP" | "MULOP" => "\\mathbin",
94    "SUMOP" | "INTOP" => "\\mathop",
95    "PUNCT" => "\\mathpunct",
96    "OPEN" => "\\mathopen",
97    "CLOSE" => "\\mathclose",
98    _ => "\\mathord",
99  }
100}

102/// The role of a document-declared name: the slot's own role when the declared class is the slot's, else the
103/// class's role as `\DeclareMathSymbol` gives it (latex_constructs `\DeclareMathSymbol`).
104fn declared_role(class: &str, slot_role: &str) -> Option<String> {
105  if slot_role == "VERTBAR" || class == class_of_role(slot_role) {
106    return role_of(slot_role);
107  }
108  match class {
109    "\\mathord" | "0" | "\\mathalpha" | "7" => Some("ID"),
110    "\\mathop" | "1" => Some("BIGOP"),
111    "\\mathbin" | "2" => Some("BINOP"),
112    "\\mathrel" | "3" => Some("RELOP"),
113    "\\mathopen" | "4" => Some("OPEN"),
114    "\\mathclose" | "5" => Some("CLOSE"),
115    "\\mathpunct" | "6" => Some("PUNCT"),
116    _ => None,
117  }
118  .map(str::to_string)
119}

121/// The glyph and role the tables give a slot of MnSymbol's symbol fonts (`MnSyA`…`MnSyD`, `symbols`,
122/// `largesymbols`), for a name a document declares with the package's own declarators.
123fn slot_glyph(font: &str, slot: u8) -> Option<(&'static str, &'static str)> {
124  match font {
125    "symbols" => MN_SIZED_OPS
126      .iter()
127      .find(|row| row.4 == slot)
128      .map(|row| (row.1, row.2)),
129    "largesymbols" => MN_DELIMS
130      .iter()
131      .find(|row| row.3 == slot)
132      .map(|row| (row.1, row.2)),
133    _ => MN_CHARS
134      .iter()
135      .find(|row| row.4 == font && row.5 == slot)
136      .map(|row| (row.1, row.2)),
137  }
138}

152/// `\cmd{arg}…{slot}`: the kernel declaration for a slot the tables do not map (`cmd` empty for a bare argument list).
153fn kernel_declaration(cmd: &str, args: &[&Tokens], slot: i64) -> Tokens {
154  let mut toks = Vec::new();
155  if !cmd.is_empty() {
156    toks.push(T_CS!(cmd));
157  }
158  for arg in args {
159    toks.push(T_BEGIN!());
160    toks.extend((*arg).clone().unlist());
161    toks.push(T_END!());
162  }
163  toks.push(T_BEGIN!());
164  toks.extend(Explode!(slot.to_string()));
165  toks.push(T_END!());
166  Tokens::new(toks)
167}

1833    matches!(c as u32, 0x0300..=0x036F | 0x1AB0..=0x1AFF | 0x1DC0..=0x1DFF | 0x20D0..=0x20FF | 0xFE20..=0xFE2F)

1838  /// MnSymbol's ⊚).
