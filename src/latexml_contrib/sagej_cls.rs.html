6LoadDefinitions!({
7  LoadClass!("OmniBus");
8  // sagej.cls L108-L109 load these math-symbol/font packages:
9  //   L108: \RequirePackage{latexsym,ifthen,rotating,calc,textcase,...}
10  //   L109: \RequirePackage{amsfonts,amssymb,amsbsy,amsmath,amsthm}
11  // Perl ships NO sagej binding → raw-loads the bundled sagej.cls → all of
12  // these load, so e.g. `\leqslant` (amssymb), `\mathbb` (amsfonts) and the
13  // latexsym glyphs are defined. Our OmniBus binding intercepts the class,
14  // so the `\RequirePackage` line never runs — replicate the AMS cluster
15  // here (in cls order: amsfonts before amssymb, which depends on it).
16  // Pure math-symbol/font packages — no layout side effects. Witness
17  // 1802.07225 (`\leqslant` in a paper that loads sagej but not amssymb).
18  RequirePackage!("latexsym");
19  RequirePackage!("amsfonts");
20  RequirePackage!("amssymb");
21  RequirePackage!("amsbsy");
22  RequirePackage!("amsmath");
23  RequirePackage!("amsthm");
24  // Do NOT eager-load xcolor (Perl ships no sagej binding → OmniBus, no
25  // preload). A preloaded xcolor makes a later `\usepackage[table]{xcolor}`
26  // a no-op → colortbl/array never load → array `m{}`/`b{}` columns are
27  // "Unrecognized tabular template" → "Extra alignment tab". The document
28  // loads xcolor with its own options; `\color`/`\definecolor` stay
29  // available via hyperref→color. See ifacconf_cls.rs / SYNC_STATUS.
30  RequirePackage!("hyperref");
31  // sagej templates use \toprule / \midrule / \bottomrule from booktabs.
32  // The raw cls relies on the user `\usepackage{booktabs}` but many
33  // papers don't load it explicitly. Eager-load. Witness 2309.01813.
34  RequirePackage!("booktabs");
35  // sagej.cls L108: `\RequirePackage{latexsym,ifthen,rotating,calc,textcase,
36  // booktabs,color,endnotes}`. Perl ships no sagej binding → raw-loads the
37  // bundled sagej.cls → endnotes loaded → `\endnote`/`\theendnotes` defined.
38  // Our binding intercepts sagej.cls (so that `\RequirePackage` never runs),
39  // leaving `\endnote` undefined where Perl is clean. Load endnotes to match
40  // the real .cls (SAGE papers use `\endnote{…}` for footnote-style notes).
41  // Witness 1901.10968 (`\endnote`, `\theendnotes`).
42  RequirePackage!("endnotes");
43  // The rest of that line (79 of the 80 sampled copies; one loads xcolor for color): ifthen's `\newboolean`/
44  // `\setboolean` (2204.02460), rotating's `{sideways}` (2 papers of the sample: 1412.4039), calc, textcase, color.
45  RequirePackage!("ifthen");
46  RequirePackage!("rotating");
47  RequirePackage!("calc");
48  RequirePackage!("textcase");
49  RequirePackage!("color");
50
51  // sagej frontmatter — preserve author content.
52  DefMacro!(
53    "\\corrauth{}",
54    "\\@add@frontmatter{ltx:note}[role=corresponding]{#1}"
55  );
56  DefMacro!(
57    "\\affiliation{}",
58    "\\@add@frontmatter{ltx:note}[role=affiliation]{#1}"
59  );
60  // sagej.cls:271 `\def\affilnum#1{${}^{\text{{#1}}}$}` — the author's and the affiliation's
61  // superscript mark (2609.04585).
62  DefMacro!("\\affilnum{}", "\\textsuperscript{#1}");
63  DefMacro!(
64    "\\runninghead{}",
65    "\\@add@frontmatter{ltx:note}[role=runninghead]{#1}"
66  );
67
68  // {acks}, {funding}, {dci} envs (sagej L456-470). Use
69  // internal_vertical mode so block-level body (paragraphs, lists,
70  // funding-statement prose) is accepted — restricted_horizontal
71  // default tripped `Attempt to end mode restricted_horizontal in
72  // internal_vertical` on multi-paragraph bodies.
73  // Each end-matter statement is headed by the class's own text (`\subsection*{\normalsize\sagesf\bfseries
74  // <heading>}`), which is the statement's `name`: the XHTML prints `@name` as its visible heading
75  // (LaTeXML-structure-xhtml.xsl:174-182, `ltx:acknowledgements`). 2101.10072's sagej.cls:460-470 `{funding}`
76  // "Funding", `{dci}` "Declaration of conflicting interests" (every one of the 79 sampled copies); 2505.13309's
77  // :460-462 `{credit}` "Author contributions" (3 copies); 2406.06038's :458-461 `{conflict}` "Declaration of
78  // conflicting interests"; 2204.02460's :584-586 `{das}` "Data Accessibility Statement"; 2101.10072's :464-466 `{sm}`
79  // "Supplemental material" (78 copies; 2503.14633's "Supplemental Material"). `{sm}` 43 papers of the inventory,
80  // `{credit}` 3 (1412.4039, 1801.09386); `{acks}` keeps the kernel's acknowledgements heading.
81  let shipped = shipped_class_text("sagej")?;
82  DefEnvironment!("{acks}", "<ltx:acknowledgements>#body</ltx:acknowledgements>",
83    mode => "internal_vertical");
84  DefEnvironment!("{funding}", "<ltx:acknowledgements name='Funding'>#body</ltx:acknowledgements>",
85    mode => "internal_vertical");
86  DefEnvironment!("{dci}",
87    "<ltx:acknowledgements name='Declaration of conflicting interests'>#body</ltx:acknowledgements>",
88    mode => "internal_vertical");
89  DefEnvironment!("{credit}", "<ltx:acknowledgements name='Author contributions'>#body</ltx:acknowledgements>",
90    mode => "internal_vertical");
91  DefEnvironment!("{conflict}",
92    "<ltx:acknowledgements name='Declaration of conflicting interests'>#body</ltx:acknowledgements>",
93    mode => "internal_vertical");
94  DefEnvironment!("{das}", "<ltx:acknowledgements name='Data Accessibility Statement'>#body</ltx:acknowledgements>",
95    mode => "internal_vertical");
96  if shipped
97    .as_deref()
98    .is_some_and(|text| class_code_contains(text, "\\bfseries Supplemental Material}"))
99  {
100    DefEnvironment!("{sm}", "<ltx:acknowledgements name='Supplemental Material'>#body</ltx:acknowledgements>",
101      mode => "internal_vertical");
102  } else {
103    DefEnvironment!("{sm}", "<ltx:acknowledgements name='Supplemental material'>#body</ltx:acknowledgements>",
104      mode => "internal_vertical");
105  }
106  // 2101.10072's sagej.cls:118-120, 231, 235, 237: `\sagesf` the class's sans-serif switch (`\rm` under the `PCfour`
107  // option; 17 papers: 2101.10072), `{refsize}` and `\titlesize` its type sizes, `\absbox` the abstract's box;
108  // :38 `\journalclass` the class's name, printed in its page foot.
109  RawTeX!(
110    r"\let\sagesf\sf
111\newcommand{\refsize}{\fontsize{9}{12pt}\selectfont}
112\newcommand{\titlesize}{\fontsize{18.0}{20pt}\selectfont}
113\newbox\absbox
114\newcommand{\journalclass}{sagej.cls}\newcommand{\journalclassshort}{sagej}"
115  );
116  if class_option_given("PCfour")? {
117    RawTeX!(r"\let\sagesf\rm");
118  }
119  // 2101.10072's :480-486 `{biog}` and `{biogs}`, the author biographies under the heading "Author biography" or
120  // "Author Biographies" (13 and 2 papers: 1801.00336, 1912.06398).
121  RawTeX!(
122    r"\newenvironment{biog}{\subsection*{Author biography}\begin{refsize}\noindent}{\end{refsize}}
123\newenvironment{biogs}{\subsection*{Author Biographies}\begin{refsize}\noindent}{\end{refsize}}"
124  );
125  // 2401.17393's :246-248 `\Highlights{…}`, the highlights the title page prints with the keywords, and 2309.06697's
126  // :257-260 `\twitter{handle}`, a title-page footnote (2 and 1 papers).
127  DefMacro!(
128    "\\Highlights{}",
129    "\\@add@frontmatter{ltx:note}[role=highlights]{#1}"
130  );
131  DefMacro!(
132    "\\twitter{}",
133    "\\@add@frontmatter{ltx:note}[role=twitter]{#1}"
134  );
