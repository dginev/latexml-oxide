4/// The argument of `tokens` at `i` — a brace group's content, or one token — and the index after it.
5fn argument_at(tokens: &[Token], i: usize) -> Option<(Vec<Token>, usize)> {
6  let first = tokens.get(i)?;
7  if first.get_catcode() != Catcode::BEGIN {
8    return Some((vec![*first], i + 1));
9  }
10  let mut depth = 0usize;
11  for (j, t) in tokens.iter().enumerate().skip(i) {
12    match t.get_catcode() {
13      Catcode::BEGIN => depth += 1,
14      Catcode::END => {
15        depth -= 1;
16        if depth == 0 {
17          return Some((tokens[i + 1..j].to_vec(), j + 1));
18        }
19      },
20      _ => {},
21    }
22  }
23  None
24}

26/// The first index from `i` that is not a space.
27fn skip_spaces(tokens: &[Token], mut i: usize) -> usize {
28  while tokens.get(i) == Some(&T_SPACE!()) {
29    i += 1;
30  }
31  i
32}

34/// One `\affil` of an `\address`: its mark (`\affil{mark}` or `\affil[mark]`), its text when a brace group follows
35/// (ptephy.cls:1390 `\affil#1#2`), and the index after it.
36struct Affil {
37  mark: Vec<Token>,
38  text: Option<Vec<Token>>,
39  end:  usize,
40}
41
42/// The `\affil` at `i`.
43fn affil_at(tokens: &[Token], i: usize) -> Option<Affil> {
44  let at = skip_spaces(tokens, i + 1);
45  let (mark, after) = if tokens.get(at) == Some(&T_OTHER!("[")) {
46    let close = tokens[at..].iter().position(|t| *t == T_OTHER!("]"))? + at;
47    (tokens[at + 1..close].to_vec(), close + 1)
48  } else {
49    argument_at(tokens, at)?
50  };
51  let next = skip_spaces(tokens, after);
52  match tokens.get(next) {
53    Some(t) if t.get_catcode() == Catcode::BEGIN => {
54      let (text, end) = argument_at(tokens, next)?;
55      Some(Affil { mark, text: Some(text), end })
56    },
57    _ => Some(Affil { mark, text: None, end: after }),
58  }
59}

61/// `tokens` without the calls of `commands` among them, each with its argument; and those calls. A name's marks
62/// without their notes (`\name{D.~Tomono}{8, \thanks{Present Address: …}}`, 1306.3810), an affiliation without its
63/// emails (`\affil{}{Department of Physics, … \email{jido@tmu.ac.jp}}`, 1605.07339).
64fn split_out(tokens: &[Token], commands: &[&str]) -> (Vec<Token>, Vec<Token>) {
65  let (mut rest, mut calls) = (Vec::new(), Vec::new());
66  let mut i = 0;
67  while i < tokens.len() {
68    if commands.iter().any(|cs| tokens[i] == T_CS!(*cs))
69      && let Some((_, next)) = argument_at(tokens, skip_spaces(tokens, i + 1))
70    {
71      calls.extend_from_slice(&tokens[i..next]);
72      i = next;
73      continue;
74    }
75    rest.push(tokens[i]);
76    i += 1;
77  }
78  (rest, calls)
79}

81/// A footnote symbol, however typed, as the frontmatter labels it: `∗` (the class's `\ast`, `*`), `†`, `‡`, `§`, `¶`,
82/// `⋆`, `•`, `◦` — the symbols `is_footnote_symbol_operand` (base_utilities.rs) shows as marks, `★`, `♦`, `\diamond`,
83/// `\sharp`, `\#`, `\|` among them.
84fn footnote_symbol(t: &Token) -> Option<Token> {
85  let any =
86    |names: &[&str], chars: &[Token]| names.iter().any(|cs| *t == T_CS!(*cs)) || chars.contains(t);
87  if any(&["\\ast"], &[T_OTHER!("*"), T_OTHER!("\u{2217}")]) {
88    Some(T_OTHER!("\u{2217}"))
89  } else if any(&["\\dag", "\\dagger", "\\textdagger"], &[T_OTHER!(
90    "\u{2020}"
91  )]) {
92    Some(T_OTHER!("\u{2020}"))
93  } else if any(&["\\ddag", "\\ddagger", "\\textdaggerdbl"], &[T_OTHER!(
94    "\u{2021}"
95  )]) {
96    Some(T_OTHER!("\u{2021}"))
97  } else if any(&["\\S", "\\textsection"], &[T_OTHER!("\u{A7}")]) {
98    Some(T_OTHER!("\u{A7}"))
99  } else if any(&["\\P", "\\textparagraph"], &[T_OTHER!("\u{B6}")]) {
100    Some(T_OTHER!("\u{B6}"))
101  } else if any(&["\\star"], &[T_OTHER!("\u{22C6}")]) {
102    Some(T_OTHER!("\u{22C6}"))
103  } else if any(&["\\bullet", "\\textbullet"], &[T_OTHER!("\u{2022}")]) {
104    Some(T_OTHER!("\u{2022}"))
105  } else if any(&["\\circ"], &[T_OTHER!("\u{25E6}")]) {
106    Some(T_OTHER!("\u{25E6}"))
107  } else if any(&[], &[T_OTHER!("\u{2605}"), T_OTHER!("\u{2666}")]) {
108    Some(*t)
109  } else if any(&["\\diamond", "\\sharp", "\\#", "\\|"], &[]) {
110    // (no character the frontmatter's symbol test takes for these: kept as written)
111    Some(*t)
112  } else {
113    None
114  }
115}

117/// Spacing between marks (`{1 \, \ast}`, `{2,~3}`), none of the label.
118fn is_mark_spacing(t: &Token) -> bool {
119  *t == T_SPACE!()
120    || t.get_catcode() == Catcode::ACTIVE
121    || ["\\,", "\\;", "\\:", "\\!", "\\ ", "\\quad", "\\thinspace"]
122      .iter()
123      .any(|cs| *t == T_CS!(*cs))
124}

126/// Marks as the frontmatter compares them: expanded (`\newcommand{\AFFicrr}{1}`, on the names and the affiliations
127/// alike: 2209.07273, 1409.7469 `\affil{\INSTC}{…}`), each footnote symbol written as `footnote_symbol` labels it, and
128/// a symbol glued to the mark before it a mark of its own (`{1\ast}` is `1,∗`, `{1\dagger*}` `1,†,∗`; 1512.04524,
129/// 1504.04965, 1408.5182, 1502.00715) — spacing and braces dropped.
130fn expanded_marks(marks: Vec<Token>) -> Result<Vec<Token>> {
131  let normalized = |tokens: Vec<Token>| -> Vec<Token> {
132    tokens
133      .into_iter()
134      .filter(|t| {
135        !is_mark_spacing(t)
136          && !matches!(
137            t.get_catcode(),
138            Catcode::MATH | Catcode::BEGIN | Catcode::END
139          )
140      })
141      .map(|t| footnote_symbol(&t).unwrap_or(t))
142      .collect()
143  };
144  let mut out: Vec<Token> = Vec::new();
145  for t in normalized(do_expand(Tokens::new(normalized(marks)))?.unlist()) {
146    if let Some(prev) = out.last()
147      && *prev != T_OTHER!(",")
148      && t != T_OTHER!(",")
149      && (footnote_symbol(prev).is_some() || footnote_symbol(&t).is_some())
150    {
151      out.push(T_OTHER!(","));
152    }
153    out.push(t);
154  }
155  Ok(out)
156}

158/// The frontmatter call adding `email` (its own label text, `E-mail:`, as the contact's name) for the author marked
159/// `symbol`, or, with none, for the whole author block.
160fn email_call(symbol: Option<&str>, email: Vec<Token>) -> Result<Vec<Token>> {
161  // a font group around it (`{\rm E-mail: …}`, 1304.2086), the font aside
162  let email = match email.as_slice() {
163    [open, inner @ .., close]
164      if open.get_catcode() == Catcode::BEGIN
165        && close.get_catcode() == Catcode::END
166        && inner
167          .iter()
168          .all(|t| !matches!(t.get_catcode(), Catcode::BEGIN | Catcode::END)) =>
169    {
170      let fonts = [
171        "\\rm",
172        "\\it",
173        "\\tt",
174        "\\sf",
175        "\\bf",
176        "\\normalfont",
177        "\\upshape",
178      ];
179      let start = inner
180        .iter()
181        .position(|t| *t != T_SPACE!() && !fonts.iter().any(|cs| *t == T_CS!(*cs)))
182        .unwrap_or(inner.len());
183      inner[start..].to_vec()
184    },
185    _ => email,
186  };
187  let text = Tokens::new(email.clone()).to_string();
188  let lower = text.trim_start().to_lowercase();
189  let labelled = ["e-mail", "email"]
190    .iter()
191    .find_map(|word| lower.strip_prefix(word))
192    .is_some_and(|after| after.trim_start().starts_with(':'));
193  let colon = email.iter().position(|t| *t == T_OTHER!(":"));
194  let (name, email) = match colon {
195    Some(colon) if labelled => {
196      let name = Tokens::new(email[..=colon].to_vec())
197        .to_string()
198        .trim()
199        .to_string();
200      (Some(name), email[skip_spaces(&email, colon + 1)..].to_vec())
201    },
202    _ => (None, email),
203  };
204  let mut options = match symbol {
205    Some(symbol) => s!("label=affiliation:{symbol}"),
206    None => s!("label=addresses:block"),
207  };
208  if let Some(name) = name {
209    options.push_str(&s!(",name={{{name} }}"));
210  }
211  Ok(
212    // (in the current font, ptephy.cls:1318 `E-mail: #1`; 1206.6924, 1408.5182)
213    Invocation!(T_CS!("\\lx@add@email"), vec![
214      Some(mouth::tokenize_internal(TeXString::assembled(options))),
215      Some(Invocation!(T_CS!("\\lx@address@textfont"), vec![Some(
216        Tokens::new(email)
217      )]))
218    ])
219    .unlist(),
220  )
221}

223/// An `\author` line with each `\name{name}{marks}`'s marks expanded (`expanded_marks`: `\name{H.~Sekiya}{\AFFicrr,
224/// \AFFipmu}`, 2209.07273; `{\Tokushima,}`, 1801.03251), its notes after it, and a comma between two names nothing
225/// but spaces, notes and line breaks separates (`\name{…}{1,2}` on one line, `\name{…}{1,\ast}` on the next: the class
226/// prints them side by side, two people; 2210.05569, 1512.04524, 2404.08725, a commented-out comma in 2101.03480; a
227/// `\\` before the closing `\name{(The T2K Collaboration)}{}`, 1409.7469); and whether a name is marked `∗`, the
228/// corresponding author, whose email ptephy.cls:1318 prints `$^{\ast}$E-mail:`.
229fn author_line(list: &Tokens) -> Result<(Tokens, bool)> {
230  let tokens = list.unlist_ref();
231  let mut out: Vec<Token> = Vec::new();
232  let mut starred = false;
233  // what came since the last name, while it is only spaces and line breaks (`\\`, `\newline`); a `~`, `\quad` or
234  // `\par` (blank line) between two names still joins them, as no paper of the 204 writes it
235  // (`ptephy_names_without_separator`)
236  let mut since_name: Option<Vec<Token>> = None;
237  let mut i = 0;
238  while i < tokens.len() {
239    // (`\collaborator{X}`, ptephy_v1.cls:1403, a name of its own: 2101.03480, 2404.08725)
240    let named = if tokens[i] == T_CS!("\\name") {
241      argument_at(tokens, skip_spaces(tokens, i + 1)).and_then(|(name, after)| {
242        argument_at(tokens, skip_spaces(tokens, after)).map(|(marks, next)| (name, marks, next))
243      })
244    } else if tokens[i] == T_CS!("\\collaborator") {
245      argument_at(tokens, skip_spaces(tokens, i + 1)).map(|(name, next)| (name, Vec::new(), next))
246    } else {
247      None
248    };
249    if let Some((name, marks, next)) = named {
250      // (the line breaks between two names are the line of names going on)
251      if since_name.take().is_some() {
252        out.extend([T_OTHER!(","), T_SPACE!()]);
253      }
254      let (marks, notes) = split_out(&marks, &["\\thanks", "\\footnote"]);
255      let marks = expanded_marks(marks)?;
256      starred |= marks.contains(&T_OTHER!("\u{2217}"));
257      out.extend(
258        Invocation!(T_CS!("\\name"), vec![
259          Some(Tokens::new(name)),
260          Some(Tokens::new(marks))
261        ])
262        .unlist(),
263      );
264      out.extend(notes);
265      since_name = Some(Vec::new());
266      i = next;
267      continue;
268    }
269    if (tokens[i] == T_CS!("\\thanks") || tokens[i] == T_CS!("\\footnote"))
270      && let Some((_, next)) = argument_at(tokens, skip_spaces(tokens, i + 1))
271    {
272      out.extend(since_name.as_mut().map(std::mem::take).unwrap_or_default());
273      out.extend_from_slice(&tokens[i..next]);
274      i = next;
275      continue;
276    }
277    if let Some(pending) = since_name.as_mut() {
278      if tokens[i] == T_SPACE!() {
279        pending.push(tokens[i]);
280        i += 1;
281        continue;
282      }
283      if tokens[i] == T_CS!("\\\\") || tokens[i] == T_CS!("\\newline") {
284        // the break, its `*` and `[length]`
285        let mut next = i + 1;
286        if tokens.get(next) == Some(&T_OTHER!("*")) {
287          next += 1;
288        }
289        if tokens.get(next) == Some(&T_OTHER!("["))
290          && let Some(k) = tokens[next..].iter().position(|t| *t == T_OTHER!("]"))
291        {
292          next += k + 1;
293        }
294        pending.extend_from_slice(&tokens[i..next]);
295        i = next;
296        continue;
297      }
298      out.append(pending);
299      since_name = None;
300    }
301    out.push(tokens[i]);
302    i += 1;
303  }
304  out.extend(since_name.unwrap_or_default());
305  Ok((Tokens::new(out), starred))
306}

308/// The index past a spacing command at `i` and its argument (`\vspace{1mm}`, `\\[2mm]`, `\vskip 2pt`, `\kern1pt`), whose
309/// letters are units, not text; `None` for any other token.
310fn past_spacing(tokens: &[Token], i: usize) -> Option<usize> {
311  let is = |names: &[&str]| names.iter().any(|cs| tokens[i] == T_CS!(*cs));
312  let star = |j: usize| {
313    if tokens.get(j) == Some(&T_OTHER!("*")) {
314      j + 1
315    } else {
316      j
317    }
318  };
319  if is(&["\\vspace", "\\hspace"]) {
320    argument_at(tokens, skip_spaces(tokens, star(i + 1))).map(|(_, next)| next)
321  } else if is(&["\\\\", "\\newline"]) {
322    let j = skip_spaces(tokens, star(i + 1));
323    if tokens.get(j) == Some(&T_OTHER!("[")) {
324      tokens[j..]
325        .iter()
326        .position(|t| *t == T_OTHER!("]"))
327        .map(|k| j + k + 1)
328    } else {
329      Some(i + 1)
330    }
331  } else if is(&["\\vskip", "\\hskip", "\\kern", "\\mskip"]) {
332    // the glue or dimension (tex.web §1057, §1061): a run of digits, signs, points and unit letters
333    let mut j = skip_spaces(tokens, i + 1);
334    while tokens.get(j).is_some_and(|t| {
335      t.get_catcode() == Catcode::LETTER
336        || (t.get_catcode() == Catcode::OTHER
337          && t.with_str(|s| s.chars().all(|c| c.is_ascii_digit() || ".,+-".contains(c))))
338    }) {
339      j += 1;
340    }
341    Some(j)
342  } else {
343    None
344  }
345}

347/// A mark at `i` in an address's text — `{\dag}`, `$^\dagger$`, `${}^\ast$`, `\textsuperscript{\ddag}`, a bare `\dag` —
348/// its first footnote symbol and the index after it; balanced, so the text around it stays whole.
349fn mark_run(tokens: &[Token], i: usize) -> Option<(Token, usize)> {
350  let opening =
351    |t: &Token| matches!(t.get_catcode(), Catcode::BEGIN | Catcode::SUPER) || is_mark_spacing(t);
352  let mut j = i;
353  let mut symbol = None;
354  while let Some(t) = tokens.get(j) {
355    if let Some(sym) = footnote_symbol(t) {
356      symbol.get_or_insert(sym);
357    } else if !(matches!(
358      t.get_catcode(),
359      Catcode::MATH | Catcode::SUPER | Catcode::BEGIN | Catcode::END
360    ) || *t == T_CS!("\\textsuperscript")
361      || is_mark_spacing(t))
362    {
363      break;
364    }
365    j += 1;
366  }
367  while j > i && opening(&tokens[j - 1]) {
368    j -= 1;
369  }
370  let run = &tokens[i..j];
371  let count = |code: Catcode| run.iter().filter(|t| t.get_catcode() == code).count();
372  (count(Catcode::BEGIN) == count(Catcode::END) && count(Catcode::MATH) % 2 == 0)
373    .then_some(symbol?)
374    .map(|symbol| (symbol, j))
375}

377/// The frontmatter calls adding an `\email`'s argument, split where a footnote symbol leads a piece of it
378/// (`\email{knabe@…, $^\dagger$E-mail: hisao@…}`, 1408.5182): each piece the email of the author its symbol marks, the
379/// lead `lead`'s (`email_call`).
380fn email_calls(tokens: &[Token], lead: Option<&str>) -> Result<Vec<Token>> {
381  let (mut first, mut pieces): (Vec<Token>, Vec<(Token, Vec<Token>)>) = (Vec::new(), Vec::new());
382  let mut i = 0;
383  while i < tokens.len() {
384    if let Some((symbol, next)) = mark_run(tokens, i) {
385      pieces.push((symbol, Vec::new()));
386      i = next;
387      continue;
388    }
389    match pieces.last_mut() {
390      Some((_, piece)) => piece.push(tokens[i]),
391      None => first.push(tokens[i]),
392    }
393    i += 1;
394  }
395  let balanced = |v: &[Token]| {
396    v.iter()
397      .filter(|t| t.get_catcode() == Catcode::BEGIN)
398      .count()
399      == v.iter().filter(|t| t.get_catcode() == Catcode::END).count()
400  };
401  if !balanced(&first) || pieces.iter().any(|(_, piece)| !balanced(piece)) {
402    return email_call(lead, tokens.to_vec());
403  }
404  let mut calls = Vec::new();
405  if let Some(first) = trimmed_text(first) {
406    calls.extend(email_call(lead, first)?);
407  }
408  for (symbol, piece) in pieces {
409    if let Some(piece) = trimmed_text(piece) {
410      calls.extend(email_call(Some(&symbol.to_string()), piece)?);
411    }
412  }
413  Ok(calls)
414}

416/// `tokens` without the separators around it (spaces, spacing, `,`, `;`, `\\`), when it holds text (letters).
417fn trimmed_text(mut tokens: Vec<Token>) -> Option<Vec<Token>> {
418  let separator = |t: &Token| {
419    is_mark_spacing(t) || *t == T_OTHER!(",") || *t == T_OTHER!(";") || *t == T_CS!("\\\\")
420  };
421  while tokens.first().is_some_and(separator) {
422    tokens.remove(0);
423  }
424  while tokens.last().is_some_and(separator) {
425    tokens.pop();
426  }
427  tokens
428    .iter()
429    .any(|t| t.get_catcode() == Catcode::LETTER)
430    .then_some(tokens)
431}

433/// What an `\address` of `\affil{mark}{text}`s prints beside them, its `\email`s and spacing aside (`\vspace{1mm}`,
434/// 1407.3513), when it holds text: each piece led by a footnote symbol the email (or note) of the author that symbol
435/// marks (`\email{a@x}, {\dag}b@y, {\ddag}c@z`, 1401.4647, 1507.04527, 1610.06306; `$^\dagger$E-mail: …`, 1408.5182),
436/// the rest an address of the whole block (`${}^\ast$ {\rm E-mail: …}` is the starred author's, 1304.7885).
437fn remainder_calls(tokens: &[Token]) -> Result<Vec<Token>> {
438  let mut rest: Vec<Token> = Vec::new();
439  let mut i = 0;
440  while i < tokens.len() {
441    let skip = if tokens[i] == T_CS!("\\affil") {
442      affil_at(tokens, i).map(|affil| affil.end)
443    } else if tokens[i] == T_CS!("\\email") {
444      argument_at(tokens, skip_spaces(tokens, i + 1)).map(|(_, next)| next)
445    } else {
446      past_spacing(tokens, i)
447    };
448    match skip {
449      Some(next) => i = next,
450      None => {
451        rest.push(tokens[i]);
452        i += 1;
453      },
454    }
455  }
456  let (mut lead, mut pieces): (Vec<Token>, Vec<(Token, Vec<Token>)>) = (Vec::new(), Vec::new());
457  let mut i = 0;
458  while i < rest.len() {
459    if let Some((symbol, next)) = mark_run(&rest, i) {
460      pieces.push((symbol, Vec::new()));
461      i = next;
462      continue;
463    }
464    match pieces.last_mut() {
465      Some((_, piece)) => piece.push(rest[i]),
466      None => lead.push(rest[i]),
467    }
468    i += 1;
469  }
470  let balanced = |v: &[Token]| {
471    v.iter()
472      .filter(|t| t.get_catcode() == Catcode::BEGIN)
473      .count()
474      == v.iter().filter(|t| t.get_catcode() == Catcode::END).count()
475  };
476  if !balanced(&lead) || pieces.iter().any(|(_, piece)| !balanced(piece)) {
477    pieces.clear();
478    lead = rest;
479  }
480  let mut calls: Vec<Token> = Vec::new();
481  if let Some(lead) = trimmed_text(lead) {
482    calls.extend(
483      Invocation!(T_CS!("\\lx@ptephy@remainder"), vec![Some(Tokens::new(
484        lead
485      ))])
486      .unlist(),
487    );
488  }
489  for (symbol, piece) in pieces {
490    let Some(piece) = trimmed_text(piece) else {
491      continue;
492    };
493    let symbol = symbol.to_string();
494    let text = Tokens::new(piece.clone()).to_string();
495    if text.contains('@') || text.contains("(at)") {
496      calls.extend(email_call(Some(&symbol), piece)?);
497    } else {
498      calls.extend(
499        Invocation!(T_CS!("\\lx@add@address"), vec![
500          Some(mouth::tokenize_internal(TeXString::assembled(s!(
501            "label=affiliation:{symbol}"
502          )))),
503          Some(Tokens::new(piece))
504        ])
505        .unlist(),
506      );
507    }
508  }
509  Ok(calls)
510}

512/// An `\address` with a single-argument `\affil`, its `\email`s taken out: each the email of the author marked by the
513/// mark-only `\affil` right before it (`\affil{\dag}\email{…}`, 1601.07691), which is dropped from the lines, else of
514/// the starred author (1703.03659 `\name{T. Fukuda}{1,*}`), else of the whole block.
515fn routed_emails(tokens: &[Token], starred: bool) -> Result<(Vec<Token>, Vec<Token>)> {
516  let (mut rest, mut calls) = (Vec::new(), Vec::new());
517  let mut i = 0;
518  while i < tokens.len() {
519    if tokens[i] == T_CS!("\\affil")
520      && let Some(Affil { mark, text: None, end }) = affil_at(tokens, i)
521      && !mark.iter().any(|t| t.get_catcode() == Catcode::LETTER)
522      && let at = skip_spaces(tokens, end)
523      && tokens.get(at) == Some(&T_CS!("\\email"))
524      && let Some((email, next)) = argument_at(tokens, skip_spaces(tokens, at + 1))
525    {
526      let marks = expanded_marks(mark)?;
527      let label: Vec<Token> = marks
528        .into_iter()
529        .take_while(|t| *t != T_OTHER!(","))
530        .collect();
531      calls.extend(email_calls(&email, Some(&Tokens::new(label).to_string()))?);
532      i = next;
533      continue;
534    }
535    if tokens[i] == T_CS!("\\email")
536      && let Some((email, next)) = argument_at(tokens, skip_spaces(tokens, i + 1))
537    {
538      calls.extend(email_calls(&email, starred.then_some("\u{2217}"))?);
539      i = next;
540      continue;
541    }
542    rest.push(tokens[i]);
543    i += 1;
544  }
545  Ok((rest, calls))
546}

548/// An `\address` whose `\affil`s are not all `\affil{mark}{text}`, as the lines it prints, for the frontmatter's
549/// affiliation-list parser: an `\affil{mark}{text}` as `\textsuperscript{mark}text\\`, a single-argument one as the
550/// mark it prints (ptephy.cls:1390 `$^{#1}$`, 1703.03659 `\affil{1}$^{1}${Nagoya University}`), or as its line when it
551/// is an affiliation (letters: ptephy_v1's authblk-TI `\affil{text}`).
552fn address_lines(tokens: &[Token]) -> Result<Tokens> {
553  let mut out: Vec<Token> = Vec::new();
554  let mut i = 0;
555  while i < tokens.len() {
556    if tokens[i] == T_CS!("\\affil")
557      && let Some(Affil { mark, text, end }) = affil_at(tokens, i)
558    {
559      let is_text = mark.iter().any(|t| t.get_catcode() == Catcode::LETTER);
560      let mark = if is_text { mark } else { expanded_marks(mark)? };
561      match text {
562        Some(text) => {
563          out.extend([T_CS!("\\textsuperscript"), T_BEGIN!()]);
564          out.extend(mark);
565          out.push(T_END!());
566          out.extend(text);
567          out.push(T_CS!("\\\\"));
568        },
569        None if is_text => {
570          out.extend(mark);
571          out.push(T_CS!("\\\\"));
572        },
573        None => {
574          out.extend([T_CS!("\\textsuperscript"), T_BEGIN!()]);
575          out.extend(mark);
576          out.push(T_END!());
577        },
578      }
579      i = end;
580      continue;
581    }
582    out.push(tokens[i]);
583    i += 1;
584  }
585  Ok(Tokens::new(out))
586}

814/// An author list split at its first top-level `\\` (with its `*` and `[length]`): the names before it, and the lines
815/// after it, each `\\`-separated line or brace group standing alone a line of its own (`\\ {${}^{\ast}$Deceased,
816/// August 2020. } {${}^{\dag}$Deceased, April 2021. }`, 2203.01270), spaces trimmed, empty ones dropped.
817fn names_and_legends(tokens: &[Token]) -> (Vec<Token>, Vec<Vec<Token>>) {
818  let is_break = |t: &Token| *t == T_CS!("\\\\") || *t == T_CS!("\\newline");
819  // the break's `*` and `[length]`
820  let past_break = |mut i: usize| {
821    i += 1;
822    if tokens.get(i) == Some(&T_OTHER!("*")) {
823      i += 1;
824    }
825    if tokens.get(i) == Some(&T_OTHER!("["))
826      && let Some(k) = tokens[i..].iter().position(|t| *t == T_OTHER!("]"))
827    {
828      i += k + 1;
829    }
830    i
831  };
832  let mut depth = 0usize;
833  let mut first_break = None;
834  for (i, t) in tokens.iter().enumerate() {
835    match t.get_catcode() {
836      Catcode::BEGIN => depth += 1,
837      Catcode::END => depth = depth.saturating_sub(1),
838      _ if depth == 0 && is_break(t) => {
839        first_break = Some(i);
840        break;
841      },
842      _ => {},
843    }
844  }
845  let Some(at) = first_break else {
846    return (tokens.to_vec(), Vec::new());
847  };
848  let trimmed = |line: Vec<Token>| -> Option<Vec<Token>> {
849    let start = line
850      .iter()
851      .position(|t| t.get_catcode() != Catcode::SPACE)?;
852    let end = line
853      .iter()
854      .rposition(|t| t.get_catcode() != Catcode::SPACE)?;
855    Some(line[start..=end].to_vec())
856  };
857  let mut legends = Vec::new();
858  let mut line: Vec<Token> = Vec::new();
859  let mut i = past_break(at);
860  while i < tokens.len() {
861    let t = tokens[i];
862    if is_break(&t) {
863      legends.extend(trimmed(std::mem::take(&mut line)));
864      i = past_break(i);
865    } else if t.get_catcode() == Catcode::BEGIN
866      && let Some((group, next)) = argument_at(tokens, i)
867    {
868      if line.iter().all(|t| t.get_catcode() == Catcode::SPACE) {
869        legends.extend(trimmed(group));
870        line.clear();
871      } else {
872        line.extend_from_slice(&tokens[i..next]);
873      }
874      i = next;
875    } else {
876      line.push(t);
877      i += 1;
878    }
879  }
880  legends.extend(trimmed(line));
881  // only legends, every line led by its footnote symbol: a line of names after the break (`B.~Two$^{2}$,\\
882  // C.~Three$^{1}$`, as pdflatex prints it) keeps the whole list names
883  if legends.is_empty() || !legends.iter().all(|line| leads_with_footnote_symbol(line)) {
884    return (tokens.to_vec(), Vec::new());
885  }
886  (tokens[..at].to_vec(), legends)
887}

889/// Whether `line` opens with a footnote symbol (`${}^{\ast}$Deceased`, `$^{\dag}$`, `\dag Deceased`, `* Equal
890/// contribution`, `\textsuperscript{*}Deceased.`) past the spacing, math shifts, braces and superscript that carry it.
891fn leads_with_footnote_symbol(line: &[Token]) -> bool {
892  const SYMBOL_CS: &[&str] = &[
893    "\\ast",
894    "\\star",
895    "\\dag",
896    "\\dagger",
897    "\\ddag",
898    "\\ddagger",
899    "\\S",
900    "\\P",
901    "\\|",
902    "\\#",
903    "\\bullet",
904    "\\diamond",
905    "\\circ",
906    "\\sharp",
907  ];
908  const SYMBOL_CHARS: &[&str] = &[
909    "*", "\u{2217}", "\u{2020}", "\u{2021}", "\u{A7}", "\u{B6}", "\u{22C6}", "#",
910  ];
911  line
912    .iter()
913    .find(|t| {
914      !matches!(
915        t.get_catcode(),
916        Catcode::SPACE | Catcode::MATH | Catcode::BEGIN | Catcode::END | Catcode::SUPER
917      ) && **t != T_CS!("\\textsuperscript")
918    })
919    .is_some_and(|t| match t.get_catcode() {
920      Catcode::CS => t.with_str(|s| SYMBOL_CS.contains(&s)),
921      Catcode::OTHER => t.with_str(|s| SYMBOL_CHARS.contains(&s)),
922      _ => false,
923    })
924}

926/// Whether `list` names its authors with superscript marks of their own that label affiliations (`A.~One$^{1}$`,
927/// `B.~Two\textsuperscript{2,a}`: a digit or letter among them) rather than with ptephy's `\name{name}{marks}`; a
928/// superscript in a note (`\thanks{…$^{3}$He…}`) is no mark, and a corresponding-author `$^{*}$` alone requests no
929/// affiliation (1810.00422: the kernel's author line keeps it as printed).
930fn superscript_marked_names(list: &Tokens) -> bool {
931  let (tokens, _notes) = split_out(list.unlist_ref(), &["\\thanks", "\\footnote"]);
932  if tokens
933    .iter()
934    .any(|t| *t == T_CS!("\\name") || *t == T_CS!("\\collaborator"))
935  {
936    return false;
937  }
938  let labels_affiliation = |mark: &[Token]| {
939    mark.iter().any(|t| {
940      t.get_catcode() == Catcode::LETTER
941        || (t.get_catcode() == Catcode::OTHER
942          && t.with_str(|s| s.chars().all(|c| c.is_ascii_digit())))
943    })
944  };
945  (0..tokens.len()).any(|i| {
946    let at = if tokens[i].get_catcode() == Catcode::SUPER {
947      i + 1
948    } else if tokens[i] == T_CS!("\\textsuperscript") {
949      skip_spaces(&tokens, i + 1)
950    } else {
951      return false;
952    };
953    match argument_at(&tokens, at) {
954      Some((mark, _)) => labels_affiliation(&mark),
955      None => false,
956    }
957  })
958}
