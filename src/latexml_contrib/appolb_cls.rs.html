1//! Binding for appolb.cls (Acta Physica Polonica B; not in TeX Live, and the papers that use it do not ship it, so
2//! without a binding the class is OmniBus's; OXIDIZED_DESIGN_DIVERGENCES #719).
3use latexml_package::prelude::*;
4
5LoadDefinitions!({
6  // appolb.cls (v3.5, 2210.03396's copy) is a class of its own, not on article.cls: OmniBus serves its frontmatter
7  // (`\title`, `\author`, `\address`, `\thanks`, `{abstract}`), as it did with no binding.
8  LoadClass!("OmniBus");
9  // appolb.cls:322 `\def\PACS#1{\par\noindent\strut\kern18pt{\PACSsize\rm PACS numbers: #1}\par}`, the classification
10  // after the abstract (13 papers of the int19 census with `\PACS` undefined: 0712.3606, 1007.2277, 1101.3125,
11  // 1108.2782, 1605.06550, nucl-th0308018, physics0501160); :321 `\received`, the date line.
12  DefMacro!(
13    "\\PACS{}",
14    "\\lx@add@classification[scheme=pacs,name={PACS numbers: }]{#1}"
15  );
16  DefMacro!("\\received{}", "\\date{#1}");
17  // appolb.cls:578-582 `\eqsec`, which the class's template invites ("uncomment this line to get equations numbered by
18  // (sec.num)"): equations numbered within sections.
19  RawTeX!(
20    r"\def\eqsec{\def\theequation{\thesection.\@arabic\c@equation}\@addtoreset{equation}{section}}"
21  );
22  // appolb.cls:299, :304 `\headtitle`, `\headauthor`, the running head's title and authors (hep-ph0403117), as the
23  // running-head notes of other journal bindings; :1019-1033 its text abbreviations (`\ie`: hep-ph0403117), each
24  // followed by a space unless punctuation follows (`\comsp`); :1059 `\e`, the upright e.
25  DefMacro!(
26    "\\headtitle{}",
27    "\\@add@frontmatter{ltx:note}[role=runningtitle]{#1}"
28  );
29  DefMacro!(
30    "\\headauthor{}",
31    "\\@add@frontmatter{ltx:note}[role=runningauthor]{#1}"
32  );
33  RawTeX!(
34    r"\def\comsp{\@ifnextchar,\relax{\@ifnextchar\ \relax{\@ifnextchar:\relax{\@ifnextchar:\relax{\ }}}}}
35\DeclareRobustCommand{\ie}{{\it i.e.}\comsp}\DeclareRobustCommand{\eg}{{\it e.g.}\comsp}
36\DeclareRobustCommand{\cf}{{\it cf.}\comsp}\DeclareRobustCommand{\etal}{{\it et~al.}\comsp}
37\DeclareRobustCommand{\ibid}{{\it ibid.}\comsp}"
38  );
39  DefMath!("\\e", "e", font => { shape => "upright" });
40  // appolb.cls:1036-1048 `\lambdabar`, a barred lambda (U+019B, as newtxmath's).
41  DefMath!("\\lambdabar", "\u{019B}", role => "ID");
42  // The rest of the class's public commands, as the class defines them, since the binding also stands in for a copy a
43  // paper ships (0801.2427, cond-mat0205657): :83 `\ifRedakcja` (the `reda` option's editorial mode, whose `redakcja`
44  // package is the journal's own), :278-282 `\preprint` with its `\ifPreprint` and `\prepNo` (the preprint number the
45  // title page sets in place of the journal heading, :399 `\ifPreprint…\prepNo` in `\@maketitle`; here a preprint note,
46  // as revtex's `\preprint`, made where `\maketitle` sets it, so a `\preprint` after `\maketitle` shows none), :283-285 the
47  // `\volume` and `\No` counts and `\pageno` (the journal heading's), :311 `\WERS`, :1019 `\anymode`, :1053-1057
48  // `\half`/`\Half` (LaTeX's `\frac` form), :1061 `\Tr`, :1063 `\Eq`, :1065-1066 `\seteqprefix`/`\reseteqprefix`.
49  // Guard `perfect_kernel_batch71a::appolb_class_commands`.
50  RawTeX!(
51    r"\newif\ifRedakcja \Redakcjafalse
52\newif\ifPreprint \Preprintfalse
53\def\prepNo{}
54\long\def\preprint#1{\Preprinttrue\long\gdef\prepNo{#1}}
55\def\lx@appolb@preprint{\ifPreprint\lx@add@pubnote[role=preprint,name={}]{\prepNo}\fi}
56\newcount\volume \volume=0
57\let\pageno=\c@page
58\newcount\No \No=0
59\let\WERS\MakeUppercase
60\def\anymode#1{\ifmmode #1\else $#1$\fi}
61\def\half{{\textstyle\frac12}}
62\def\Half{\frac12}
63\def\Tr{\mathop{\operator@font Tr}\nolimits}
64\def\Eq#1{Eq. (\ref{#1})}
65\def\seteqprefix#1{\renewcommand{\p@equation}{#1}}
66\def\reseteqprefix#1{\setcounter{equation}{0}\renewcommand{\p@equation}{#1}}"
67  );
68  AddToMacro!("\\lx@maketitle@begin", "\\lx@appolb@preprint");
69});
