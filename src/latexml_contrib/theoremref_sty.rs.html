1//! theoremref.sty binding (no Perl binding).
2//!
3//! theoremref names a theorem by its environment where it is referred to: `\thlabel{key}` writes
4//! `\newlabel{key}{{\@currentlabel}{<theorem name>}}` to the `.aux` file (theoremref.sty:23-26) and `\thref{key,…}`
5//! prints each name with its number, read from that record on the next run (:47-66). Raw-loaded, `\thlabel` labelled
6//! nothing the document holds: every `\thref` printed "?THM?" and a missing label where pdflatex prints "Theorem 1.1"
7//! (1211.4329). `\thlabel` labels the theorem as `\label` does, and `\thref` refers to it by the type and number its
8//! tags give (`typerefnum`, ntheorem's `\thref`, ntheorem_sty.rs). theoremref's grouping of a run of references to one
9//! environment under its plural ("Lemmas 5.7.7 and 2.1.5") is not made: each reference names its own.
10use latexml_package::prelude::*;
11
12LoadDefinitions!({
13  // The options and `\thnameref`.
14  InputDefinitions!("theoremref", noltxml => true, extension => Some(Cow::Borrowed("sty")));
15  // The theorem's label. theoremref's own record names the environment by `\thmref@currname`, which its `\@thm`
16  // patch sets and our theorem environments never run, so `\thnameref` (one census paper) still prints "?THM?".
17  DefMacro!("\\thlabel{}", "\\label{#1}", locked => true);
18  // `\thref{a,b,c}`: "A 1, B 2, and C 3", "A 1 and B 2" (`\thmref@flush` with each key of its own environment).
19  DefMacro!("\\thref{}", sub[(keys)] {
20    let keys = keys.to_string();
21    let keys: Vec<&str> = keys.split(',').map(str::trim).filter(|key| !key.is_empty()).collect();
22    let mut tokens = Vec::new();
23    for (i, key) in keys.iter().enumerate() {
24      if i > 0 {
25        let separator = match (keys.len(), i + 1 == keys.len()) {
26          (2, _) => " and ",
27          (_, true) => ", and ",
28          _ => ", ",
29        };
30        tokens.extend(ExplodeText!(separator));
31      }
32      tokens.extend([T_CS!("\\lx@thref"), T_BEGIN!()]);
33      tokens.extend(ExplodeText!(*key));
34      tokens.push(T_END!());
35    }
36    Ok(Tokens::new(tokens))
37  }, locked => true);
38  DefConstructor!("\\lx@thref Semiverbatim",
39  "<ltx:ref labelref='#label' show='typerefnum' _force_font='true'/>",
40  enter_horizontal => true,
41  properties => sub[args] {
42    let key = args[0].as_ref().map(|a| a.to_string()).unwrap_or_default();
43    Ok(stored_map!("label" => clean_label(&key, None).into_owned()))
44  });
45});
