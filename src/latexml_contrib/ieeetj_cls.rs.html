1//! ieeetj.cls — IEEE Transactions journal template (author-bundled, ~4900 lines;
2//! not raw-loaded). The class is IEEEtran V1.7a copied inline (its header,
3//! ieeetj.cls:209) plus its own `\affil{…}` store (:4856-4858, numbered
4//! affiliations after a flat `\author` list), so it is IEEEtran with that
5//! `\affil` (authblk's would attach every affiliation to every author). As an
6//! OmniBus stub, IEEEtran's `\ifCLASSOPTION…`
7//! conditionals, `{IEEEkeywords}`, `\IEEEPARstart` and `\IEEEpubidadjcol` were
8//! undefined (arXiv 2605.01773). The class's late-defined journal-metadata
9//! frontmatter macros are bound below; they leaked as literal text before
10//! (witness 2405.01673, 2603.04284 → `\corresp \authornote \receiveddate …`).
11use latexml_package::prelude::*;
12
13LoadDefinitions!({
14  LoadClass!("IEEEtran");
15  // ieeetj.cls:1792-1810 (2510.05542's copy) `\@makecaption`: a table's caption "TABLE 1. text" on one line, as a
16  // figure's "FIGURE 1. text", in place of IEEEtran's label over the text (ieeetran_cls.rs); caption, when loaded,
17  // replaces it (caption_sty.rs). The names and numbers (:1650, 1850-1852): figures "FIGURE", tables arabic, in place of
18  // IEEEtran's "Fig." and Roman tables. Guard `perfect_kernel_batch78y::ieeetj_caption_names`.
19  caption_sty::class_makecaption_label("", ". ", false)?;
20  RawTeX!(r"\def\thetable{\@arabic\c@table}\def\figurename{FIGURE}");
21  // The author list is flat, with `\\` as a line break inside it (2405.01673);
22  // inst_support's `\author` reads it that way, as under OmniBus.
23  RequirePackage!("inst_support");
24  // ieeetj.cls:4858 `\affil{text}` stores one numbered affiliation block.
25  DefMacro!("\\affil{}", "\\lx@add@affiliation{#1}");
26
27  // Corresponding-author byline and author/funding note — preserve as notes, each keeping its last value as the
28  // class's `\gdef` does.
29  DefMacro!(
30    "\\corresp{}",
31    "\\lx@clear@frontmatter{ltx:note}[role=corresponding]\\lx@add@frontmatter{ltx:note}[role=corresponding]{#1}"
32  );
33  DefMacro!(
34    "\\authornote{}",
35    "\\lx@clear@frontmatter{ltx:note}[role=note]\\lx@add@frontmatter{ltx:note}[role=note]{#1}"
36  );
37  // `\affil{…}` — a numbered affiliation block. inst_support already defines
38  // `\affil`; keep that. (Listed here only for the record.)
39
40  // Editorial dates and the DOI, as the first page prints them (ieeetj.cls:3439-3440 "Received …; revised …;
41  // accepted …; Date of publication …; date of current version …", "Digital Object Identifier 10.1109/…"; a
42  // template's "XX Month, XXXX" placeholders included, as pdflatex prints them); gobbling them as template noise
43  // dropped real dates and DOIs too.
44  DefMacro!("\\receiveddate{}", "\\lx@add@date[role=received]{#1}");
45  DefMacro!("\\reviseddate{}", "\\lx@add@date[role=revised]{#1}");
46  DefMacro!("\\accepteddate{}", "\\lx@add@date[role=accepted]{#1}");
47  DefMacro!("\\publisheddate{}", "\\lx@add@date[role=published]{#1}");
48  DefMacro!("\\currentdate{}", "\\lx@add@date[role=current]{#1}");
49  // Labelled as the class prints it (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #915): ieeetj.cls:3440 (2510.05542's copy) "Digital Object
50  // Identifier 10.1109/…" — not the role's "DOI: ".
51  // An empty `\doiinfo{}` (the class's own default) prints no DOI line (ieeetj.cls:3440 `\ifx\@doiinfo\@empty`; 2609.27083).
52  DefMacro!(
53    "\\doiinfo{}",
54    "\\lx@clear@frontmatter{ltx:pubnote}[role=doi]\\def\\lx@ieee@doi{#1}\\ifx\\lx@ieee@doi\\@empty\\else\\lx@add@pubnote[role=doi,name={Digital Object Identifier~}]{10.1109/#1}\\fi"
55  );
56  // The names the first page prints them with (ieeetj.cls:3439; Perl looks a date's name up as `\lx@date@<role>@name`).
57  DefMacro!("\\lx@date@published@name", "Date of publication~");
58  DefMacro!("\\lx@date@current@name", "Date of current version~");
59  def_macro_noop("\\history{}")?;
60  def_macro_noop("\\articletype{}")?;
61  // Author-supplied funding acknowledgement — preserve rather than gobble.
62  DefMacro!(
63    "\\fundingtext{}",
64    "\\lx@add@frontmatter{ltx:note}[role=funding]{#1}"
65  );
66});
