7/// A rectangle of cells to fill, in nicematrix's 1-based MAIN-matrix
8/// coordinates (the `first-row`/`first-col` label line is excluded). `r2`/`c2`
9/// may be `i32::MAX` to mean "through the last row/column" (`\rowcolor`,
10/// `\columncolor`, `\arraycolor`); clamped to the real cell count at apply time.
11#[derive(Clone)]
12struct NiceRect {
13  bg: String,
14  r1: i32,
15  c1: i32,
16  r2: i32,
17  c2: i32,
18}

20thread_local! {
21  /// `\CodeBefore` color rectangles recorded for the matrix currently being
22  /// built. Reset at each `\begin{<x>NiceMatrix}` (`\lx@nice@setopts`), drained
23  /// and painted onto the built `ltx:XMArray` by `\lx@nicematrix@applycolors`.
24  /// thread_local + `RefCell` per project rule (never a `Mutex`).
25  static NICE_RECTS: RefCell<Vec<NiceRect>> = const { RefCell::new(Vec::new()) };
26  /// Whether the current matrix declared `first-row` / `first-col`: shifts the
27  /// coordinate origin by one and marks the label line `thead`.
28  static NICE_FIRST_ROW: Cell<bool> = const { Cell::new(false) };
29  static NICE_FIRST_COL: Cell<bool> = const { Cell::new(false) };
30}

32/// Parse an `i-j` cell coordinate ("row-col").
33fn nice_parse_cell(s: &str) -> Option<(i32, i32)> {
34  let (a, b) = s.trim().split_once('-')?;
35  Some((a.trim().parse().ok()?, b.trim().parse().ok()?))
36}
37
38/// Parse a comma list of `i` / `i-j` items into inclusive `(lo, hi)` ranges
39/// (`\rowcolor{2,4-5}` → `[(2,2),(4,5)]`).
40fn nice_ranges(list: &str) -> Vec<(i32, i32)> {
41  list
42    .split(',')
43    .filter_map(|item| {
44      let item = item.trim();
45      if item.is_empty() {
46        return None;
47      }
48      match item.split_once('-') {
49        Some((a, b)) => Some((a.trim().parse().ok()?, b.trim().parse().ok()?)),
50        None => {
51          let n = item.parse().ok()?;
52          Some((n, n))
53        },
54      }
55    })
56    .collect()
57}

59/// Record a fill rectangle, normalizing corner order.
60fn nice_push_rect(bg: &str, r1: i32, c1: i32, r2: i32, c2: i32) {
61  NICE_RECTS.with(|cell| {
62    cell.borrow_mut().push(NiceRect {
63      bg: bg.to_string(),
64      r1: r1.min(r2),
65      c1: c1.min(c2),
66      r2: r1.max(r2),
67      c2: c1.max(c2),
68    });
69  });
70}

72// The `\rectanglecolor` recorders run at DIGESTION (DefPrimitive); the color-walk
73// runs at CONSTRUCTION (DefConstructor) — two separate phases. So the whole
74// document is digested (every matrix's CodeBefore recording into NICE_RECTS,
75// cleared between matrices) BEFORE any color-walk runs; by then the thread_local
76// holds only the last matrix's rects. To bridge the phase boundary, each matrix's
77// `\lx@nicematrix@applycolors` snapshots ITS rects + flags into a whatsit property
78// (in the digest-time `properties` closure) and reads them back at construction.
79// `#` is disallowed in the encoding fields anyway (hex has no `,`/`|`/`;`).
80
81/// Encode `(first_row, first_col, rects)` as a flat whatsit-property string.
82fn nice_encode(fr: bool, fc: bool, rects: &[NiceRect]) -> String {
83  let body: Vec<String> = rects
84    .iter()
85    .map(|r| format!("{},{},{},{},{}", r.r1, r.c1, r.r2, r.c2, r.bg))
86    .collect();
87  format!("{};{};{}", u8::from(fr), u8::from(fc), body.join("|"))
88}

90/// Whether a comma-separated key list contains `flag` as its own key — matching
91/// the exact key (before any `=`), so `first-row` is NOT satisfied by
92/// nicematrix's unrelated `code-for-first-row` styling key.
93fn nice_opts_has(opts: &str, flag: &str) -> bool {
94  opts.split(',').any(|tok| {
95    let key = tok.split('=').next().unwrap_or("").trim();
96    key == flag
97  })
98}

100/// Inverse of [`nice_encode`].
101fn nice_decode(data: &str) -> (bool, bool, Vec<NiceRect>) {
102  let mut it = data.splitn(3, ';');
103  let fr = it.next() == Some("1");
104  let fc = it.next() == Some("1");
105  let rects = it
106    .next()
107    .unwrap_or("")
108    .split('|')
109    .filter(|s| !s.is_empty())
110    .filter_map(|s| {
111      let f: Vec<&str> = s.splitn(5, ',').collect();
112      if f.len() != 5 {
113        return None;
114      }
115      Some(NiceRect {
116        r1: f[0].parse().ok()?,
117        c1: f[1].parse().ok()?,
118        r2: f[2].parse().ok()?,
119        c2: f[3].parse().ok()?,
120        bg: f[4].to_string(),
121      })
122    })
123    .collect();
124  (fr, fc, rects)
125}

236  // Paint the recorded \CodeBefore rectangles onto the just-built XMArray, and mark
237  // first-row/first-col label cells thead. A no-output DefConstructor (like
238  // colortbl's \lxsetcellcolor) run right after \lx@end@ams@matrix: the ams matrix
239  // constructor has already emitted the whole XMDual>XMWrap>XMArg>XMArray tree
240  // synchronously, so the array is in the document and navigable here. The
241  // digest-time `properties` closure drains THIS matrix's rects+flags (before the
242  // next matrix's \lx@nice@setopts clears the thread_local) and stashes them on the
243  // whatsit; the constructor reads them back — the only reliable bridge across the
244  // digest→construct phase gap. Cells carry `backgroundcolor`/`thead` on XMCell
245  // (schema LaTeXML-math.rnc:330,352). The array is this math's direct
246  // XMDual>XMWrap>XMArg>XMArray child (needXMDual always set via datameaning=matrix).
247  DefConstructor!("\\lx@nicematrix@applycolors",
248    sub[document, _args, props] {
249      let data = match props.get("nice_data") {
250        Some(Stored::String(s)) => to_string(*s),
251        _ => String::new(),
252      };
253      let (first_row, first_col, rects) = nice_decode(&data);
254      // `current` is the shared math parent (ltx:XMath), not the matrix just
255      // closed — a display can hold several matrix XMDuals (`\[ A \quad B \]`,
256      // or a plain `pmatrix` beside a Nice matrix). The matrix we just built is
257      // the LAST matrix-XMDual child at construct time (later matrices aren't
258      // constructed yet), so pick `[last()]` — NOT the first, which would paint
259      // an earlier (possibly unrelated) matrix. PR-review witness.
260      let current = document.get_node().clone();
261      let array = document
262        .findnode(
263          "ltx:XMDual[ltx:XMWrap/ltx:XMArg/ltx:XMArray][last()]/ltx:XMWrap/ltx:XMArg/ltx:XMArray",
264          Some(&current),
265        )
266        .or_else(|| document.findnode("(descendant::ltx:XMArray)[last()]", Some(&current)));
267      if let Some(array) = array {
268        let rows = document.findnodes("ltx:XMRow", Some(&array));
269        let n_rows = rows.len() as i32;
270        let n_cols = rows
271          .first()
272          .map(|r| document.findnodes("ltx:XMCell", Some(r)).len() as i32)
273          .unwrap_or(0);
274        let row_off = i32::from(first_row);
275        let col_off = i32::from(first_col);
276        // Main-matrix extent (label line excluded).
277        let main_rows = n_rows - row_off;
278        let main_cols = n_cols - col_off;
279        for rect in &rects {
280          let r_hi = rect.r2.min(main_rows);
281          let c_hi = rect.c2.min(main_cols);
282          for i in rect.r1.max(1)..=r_hi {
283            let ri = (i - 1 + row_off) as usize;
284            let Some(row) = rows.get(ri) else { continue };
285            let cells = document.findnodes("ltx:XMCell", Some(row));
286            for j in rect.c1.max(1)..=c_hi {
287              let ci = (j - 1 + col_off) as usize;
288              if let Some(cell) = cells.get(ci) {
289                let mut cell = cell.clone();
290                document.set_attribute(&mut cell, "backgroundcolor", &rect.bg)?;
291              }
292            }
293          }
294        }
295        // first-row → column headers; first-col → row headers; corner → both.
296        if first_row || first_col {
297          for (ri, row) in rows.iter().enumerate() {
298            let cells = document.findnodes("ltx:XMCell", Some(row));
299            for (ci, cell) in cells.iter().enumerate() {
300              let col_head = first_row && ri == 0;
301              let row_head = first_col && ci == 0;
302              if col_head || row_head {
303                let mut thead = String::new();
304                if col_head {
305                  thead.push_str("column");
306                }
307                if col_head && row_head {
308                  thead.push(' ');
309                }
310                if row_head {
311                  thead.push_str("row");
312                }
313                let mut cell = cell.clone();
314                document.set_attribute(&mut cell, "thead", &thead)?;
315              }
316            }
317          }
318        }
319      }
320    },
321    // Digest-time: snapshot THIS matrix's rects+flags off the thread_local before
322    // the next \begin's \lx@nice@setopts clears it.
323    properties => {
324      let fr = NICE_FIRST_ROW.with(|c| c.get());
325      let fc = NICE_FIRST_COL.with(|c| c.get());
326      let rects = NICE_RECTS.with(|c| std::mem::take(&mut *c.borrow_mut()));
327      let mut props = stored_map!();
328      props.insert("nice_data", Stored::String(pin(nice_encode(fr, fc, &rects))));
329      Ok(props)
330    },
331    alias => "");
