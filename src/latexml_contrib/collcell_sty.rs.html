1use latexml_package::prelude::*;
2
3LoadDefinitions!({
4  // collcell.sty — hand a tabular cell's content to a macro as ONE argument:
5  //   \newcolumntype{B}{>{\collectcell\Xfer}c<{\endcollectcell}}
6  // means every B cell is typeset as `\Xfer{<cell>}` (collcell.sty:89
7  // `\ccell@swap`).
8  //
9  // Raw-impossibility justification (perfect-kernel README protocol): the
10  // real `\collectcell#1#2\ignorespaces` (collcell.sty:76) is DELIMITED by
11  // the `\ignorespaces` that the LaTeX kernel inserts into every l/c/r cell
12  // template (latex.ltx:16671-16675 `\ignorespaces\@sharp\unskip`,
13  // array.sty:97-112 `\insert@column`). Perl's alignment (Alignment.pm:203-217)
14  // assembles cells WITHOUT that wrapping, so the raw macro scans for a
15  // delimiter that never comes: Perl reports `Missing argument
16  // Until:\ignorespaces` and a boxing-group cascade. Ours gives the `>{…}`
17  // tokens their `\ignorespaces` since 63v (template.rs, OXIDIZED_DESIGN #473),
18  // but its trim (`\lx@column@trimright`) and row end (`\lx@hidden@cr`) are not
19  // the `\unskip` and `\tabularnewline`/`\cr` the raw scanner tests for
20  // (collcell.sty:184-212) (witness onedown-ref.tex:469 `\begin{bidding}`,
21  // onedown.sty:1326 column `B`, once a `Fatal:Mouth:EoF`).
22  // Guards: `perfect_kernel_batch53::collcell_hands_cell_to_macro`,
23  // `perfect_kernel_batch65e::collcell_cell_starts_after_the_template_ignorespaces`.
24  RequirePackage!("array");
25  // `verb`/`robustcr` (collcell.sty:12-32) only tune the raw scanner.
26  DeclareOption!("verb", "");
27  DeclareOption!("noverb", "");
28  DeclareOption!("robustcr", "");
29  DeclareOption!("norobustcr", "");
30  ProcessOptions!();
31  // The scanner mirrors `\collect@cell@look` (collcell.sty:91-146): the cell
32  // is collected token by token UNEXPANDED (`\futurelet` + `#1` grabs, groups
33  // kept as `{…}`), until the `<{\endcollectcell}` column-after template
34  // arrives. TeX inserts that v-part when `&`/`\cr` is scanned at
35  // `align_state`=0 (tex.web §342); our `read_token` does the same for
36  // `&`/`\cr`/`\lx@hidden@cr` (gullet.rs `is_column_end`+`handle_template`).
37  // A row end spelled `\\`/`\tabularnewline` is a MACRO, though, invisible to
38  // an unexpanding scan (a plain `Until:\endcollectcell` swallowed the rest
39  // of the document); collcell handles it by `\ifx`-matching `\tabularnewline`
40  // and letting it expand down to `\cr` (L184-192 `\collect@cell@cr`) — we
41  // expand it once so `\lx@hidden@cr` reaches the column-end check.
42  // `\collectcell#1#2\ignorespaces` (collcell.sty:76): its delimiter is the `\ignorespaces` the cell template puts
43  // after the `>{…}` tokens (array.sty:97-101 `\insert@column`, template.rs `add_before_column`), so the cell begins
44  // after it (what precedes it, `#2`, is scanned as the cell's start: `\collect@cell@look#2`). The scan skips the
45  // template's `\unskip` after the cell (collcell.sty:207-212, with the spaces before it), whose role our template's
46  // `\lx@column@trimright` plays (Perl Template.pm:60/97), and the macro gets the cell between `\ignorespaces` and
47  // `\unskip` (collcell.sty:62-64 `\collcell@beforeuser`/`@afteruser`). Since the template's `\ignorespaces`
48  // (63v, OXIDIZED_DESIGN #473) the macro was handed it and the trim marks, `\ifdim\ignorespaces 0.2pt<…`
49  // (1912.00582, 2604.02352).
50  DefMacro!("\\collectcell Token", sub[(xfer)] {
51    let end = T_CS!("\\endcollectcell");
52    let newline = T_CS!("\\lx@alignment@newline");
53    let cci = T_CS!("\\cci");
54    let delimiter = T_CS!("\\ignorespaces");
55    let unskips = [T_CS!("\\unskip"), T_CS!("\\lx@column@trimright")];
56    let mut delimited = false;
57    let mut trimmed = false;
58    let mut cell: Vec<Token> = Vec::new();
59    while let Some(t) = read_token()? {
60      if t == end || t.defined_as(&end) {
61        break;
62      }
63      match t.get_catcode() {
64        // the `#2\ignorespaces` delimiter, once
65        Catcode::CS if !delimited && t == delimiter => delimited = true,
66        // collcell.sty:207-212: an `\unskip` is not collected, nor the spaces before it
67        Catcode::CS if unskips.contains(&t) => {
68          trimmed |= t == unskips[1];
69          while cell.last().is_some_and(|t| t.get_catcode() == Catcode::SPACE) {
70            cell.pop();
71          }
72        },
73        // collcell.sty:231-233: `\cci` is skipped by the scanner, so
74        // `\cci{ ...}` hands the macro a braced group it will treat as
75        // literal text (onedown.sty:1338 bidding headers).
76        Catcode::CS if t == cci => {},
77        Catcode::BEGIN => {
78          cell.push(t);
79          cell.extend(read_balanced(ExpansionLevel::Off, false, false)?.unlist());
80          cell.push(T_END!());
81        },
82        Catcode::CS | Catcode::ACTIVE if t.defined_as(&newline) => {
83          if let Some(defn) = lookup_expandable(&t, None)? {
84            unread(defn.invoke(true)?);
85          } else {
86            cell.push(t);
87          }
88        },
89        _ => cell.push(t),
90      }
91    }
92    // `\ignorespaces` before and `\unskip` after the cell (`\collcell@beforeuser`/`@afteruser`), the latter the
93    // template's trim it stands for.
94    while cell.last().map(|t| t.get_catcode() == Catcode::SPACE).unwrap_or(false) {
95      cell.pop();
96    }
97    let mut out = vec![xfer, T_BEGIN!()];
98    out.extend(cell.into_iter().skip_while(|t| t.get_catcode() == Catcode::SPACE));
99    out.push(T_END!());
100    if trimmed {
101      out.push(unskips[1]);
102    }
103    Ok(Tokens::new(out))
104  });
105  // Reached only outside a scan (collcell.sty:90 gobbles its own name).
106  DefMacro!("\\endcollectcell", "");
107  // collcell.sty:178-182: user-level `\protected` no-op marker and `\unskip`.
108  DefMacro!("\\cci", "");
109  DefMacro!("\\ccunskip", r"\unskip");
110});
