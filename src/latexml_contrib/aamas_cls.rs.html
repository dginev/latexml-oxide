1//! Stub for AAMAS conference class.
2//!
3//! aamas.cls is an ACM-style class. Reuse acmart's macros for all the
4//! ACM frontmatter (\setcopyright, \acmConference, \acmDOI, etc.) and
5//! treat content like amsart.
6use latexml_package::prelude::*;
7
8LoadDefinitions!({
9  // (the copy read before acmart's load, while `\@currname` is this class)
10  let shipped = crate::shipped_class::shipped_class_text("aamas")?;
11  // An acmproc-era aamas class — aamas2013, "based on acmproc.cls V1.3", whose authors are `\alignauthor` tabular
12  // columns (1401.1086's confSt.cls:342, `\ProvidesClass{aamas2013}`; `\numberofauthors` :333) — is no acmart class:
13  // the paper's copy is read raw, as the class it is (acmart's `\author` met its columns: 3-6 Errors and one merged
14  // creator, also for a `\documentclass{aamas2013}` whose copy the version fallback answered with this binding). Guard
15  // `perfect_kernel_batch77w::acmproc_era_aamas_reads_raw`.
16  if shipped
17    .as_deref()
18    .is_some_and(|text| crate::shipped_class::class_code_contains(text, "\\def\\alignauthor"))
19  {
20    crate::shipped_class::read_raw(&crate::shipped_class::RawRead {
21      shipped: true,
22      ..crate::shipped_class::RawRead::new("aamas", "cls")
23    })?;
24    return Ok(());
25  }
26  let defines_doi =
27    shipped.is_none_or(|text| crate::shipped_class::class_code_contains(&text, "\\def\\doi#1"));
28  LoadClass!("acmart");
29  // aamas.cls 2026 (2602.10598's copy :1435-1436, 65 of the 69 copies papers ship): `\doi{SUFFIX}` sets the paper's
30  // DOI under the IFAAMAS prefix, `10.65109/SUFFIX`, which the ACM reference line prints (:1903, :2492; `\acmDOI` stays
31  // unchanged, the template says). A preamble `\doi` with a suffix is the paper's DOI note (65 papers: 2602.03799,
32  // 2602.05679, 2602.06599); elsewhere, or empty, it only sets `\@doi`, as the class does. Only that version defines
33  // it: the paper's own copy decides (the 2021 v1.78 copies of 2510.21200 and 2602.05888 have none, 2608.28553's has
34  // the lines commented out), the newest version without one, so a reference list's own
35  // `\providecommand{\doi}[1]{doi: #1}` stands where the class has no `\doi` (its DOIs printed nothing).
36  if defines_doi {
37    RawTeX!(r"\def\@doi{10.65109/}");
38    DefMacro!("\\doi{}", sub[(suffix)] {
39      let mut out = TokenizeInternal!("\\def\\@doi{10.65109/").unlist();
40      out.extend(suffix.unlist_ref().iter().copied());
41      out.push(T_END!());
42      if lookup_bool("inPreamble") && !suffix.unlist_ref().iter().all(|t| *t == T_SPACE!()) {
43        // (printed as its https://doi.org/ link, aamas.cls:1674, with no label: FIDELITY_CHARTER §4,
44        // OXIDIZED_DESIGN_DIVERGENCES #921)
45        out.extend(TokenizeInternal!("\\lx@add@pubnote[role=doi,name={}]{10.65109/").unlist());
46        out.extend(suffix.unlist());
47        out.push(T_END!());
48      }
49      Ok(Tokens::new(out))
50    });
51  }
52});
