1//! Stub for IEEEAerospaceCLS (IEEE Aerospace conference template).
2//!
3//! IEEEAerospaceCLS is an IEEEtran-derived class for the IEEE Aerospace
4//! Conference. Route to IEEEtran which supplies \appendices, \PARstart
5//! and similar IEEE-template macros. Witness 2408.05924, 2408.06274.
6use latexml_package::prelude::*;
7
8LoadDefinitions!({
9  LoadClass!("IEEEtran");
10  // IEEEAerospaceCLS.cls L290-292 defines its own sectioning
11  // `\def\acknowledgments{\section*{Acknowledgments}\addcontentsline{toc}
12  // {section}{Acknowledgments}}` — a NO-arg command that opens an
13  // "Acknowledgments" section (content runs until the next section; there is
14  // no `\endacknowledgments`, so this is NOT the bounded ltx:acknowledgements
15  // environment form). Perl ships no IEEEAerospaceCLS binding → raw-loads the
16  // bundled cls and gets this def. Our binding intercepts the cls (routing to
17  // IEEEtran), so `\acknowledgments` was undefined where Perl is clean. Mirror
18  // the real cls: an unnumbered "Acknowledgments" section (the `\addcontents
19  // line` is TOC-only, moot in HTML). Witness 1610.07252. RUST 1 → 0.
20  DefMacro!("\\acknowledgments", "\\section*{Acknowledgments}");
21  // The current class (2026 template) spells it `\acknowledgements` (IEEEAerospaceCLS.cls:267-272) and opens the
22  // biographies the same way, `\thebiography` (:403-408); each biography is `{biography}{<name>}` (a framed photo
23  // placeholder, :528-536), `{biographywithpic}{<name>}{<file>}` (:541-549, the name bold italic, :549
24  // `\textit{\textbf{#1}}`) or `{biographywithjpg}{<name>}{<file>}` (:554-562, its file name as the class writes it,
25  // `{\fighome #2}` :556, `\fighome` the document's to define): the IEEEtran biography, photo and bold name before the
26  // text. Witnesses 2601.06000, 2601.00238, 2603.17229.
27  DefMacro!("\\acknowledgements", "\\section*{Acknowledgements}");
28  DefMacro!("\\thebiography", "\\section*{Biography}");
29  DefMacro!(
30    "\\biographywithpic{}{}",
31    "\\IEEEbiography[{\\includegraphics[width=25mm,height=32mm]{#2}}]{\\textit{#1}}"
32  );
33  Let!("\\endbiographywithpic", "\\endIEEEbiography");
34  DefMacro!(
35    "\\biographywithjpg{}{}",
36    "\\IEEEbiography[{\\includegraphics[height=32mm]{\\fighome #2}}]{#1}"
37  );
38  Let!("\\endbiographywithjpg", "\\endIEEEbiography");
39});
