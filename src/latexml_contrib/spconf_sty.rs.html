1//! spconf.sty / INTERSPEECH2021.sty — the ICASSP/older-Interspeech frontmatter
2//! package (loaded on top of `\documentclass{article}`).
3//!
4//! Both define the single-argument conference convention
5//!   `\name{Author1$^1$, Author2$^2$, …}`  `\address{$^1$Inst … $^2$ …}`
6//!   `\email{…}`
7//! (spconf.sty L170-172, INTERSPEECH2021.sty L171-173). With no binding the
8//! raw `.sty` `\def\name#1{\gdef\@name{…}}` merely stashes the names and
9//! article's structural `\maketitle` never emits them → zero creators (witness
10//! 2309.14838, 2405.13379, 2605.10272). Route `\name` through the standard
11//! author machinery so the comma/superscript-marked list becomes structured
12//! creators, and keep `\address`/`\email` as frontmatter.
13//!
14//! The same file's `keywords` block (the "Index Terms" list) is a bare
15//! `\def\keywords`/`\def\endkeywords` pair, so `\begin{keywords}` found no
16//! environment at all — the single largest `undefined` cluster in the sandbox
17//! corpora (94 papers in sandbox-arxiv-2605, 49 in 2606; witnesses 2605.00480,
18//! 2605.00698, 2605.00721, 2605.01187). See the `\keywords` binding below.
19use latexml_package::prelude::*;
20
21LoadDefinitions!({
22  // `\name{names}` — the whole author list in one argument. Hand it to `\author`
23  // (→ `\lx@add@authors`), which splits the comma / superscript-marked list into
24  // individual creators and links their affiliation superscripts.
25  DefMacro!("\\name{}", "\\author{#1}");
26  // `\address{affils}` — the (superscript-numbered) affiliation block. Preserve
27  // as a frontmatter note so the numbered institutions are kept.
28  DefMacro!(
29    "\\address{}",
30    "\\lx@add@frontmatter{ltx:note}[role=address]{#1}"
31  );
32  DefMacro!(
33    "\\email{}",
34    "\\lx@add@frontmatter{ltx:note}[role=email]{#1}"
35  );
36  // `\twoauthors{names1}{affil1}{names2}{affil2}` (spconf.sty L183-190) — the
37  // side-by-side two-author title block, typeset as two `tabular`s that
38  // overwrite `\@name` and blank `\@address`. Feed the same author machinery
39  // as `\name`: `\and` separates the two groups, and within a group the `\\`
40  // line after the names is that group's affiliation (`\lx@add@authors`,
41  // base_utilities.rs L950-960). Witnesses 2605.05692, 2605.18923, 2605.26747.
42  DefMacro!(
43    "\\twoauthors{}{}{}{}",
44    "\\author{#1 \\\\ #2 \\and #3 \\\\ #4}"
45  );
46  // The "Index Terms" block (spconf.sty L211-214):
47  //   \def\keywords{\vspace{.5em}{\bfseries\textit{Index Terms}---\,\relax}}
48  //   \def\endkeywords{\par}
49  // — a plain-TeX environment pair, not a `\newenvironment`, so
50  // `\begin{keywords}` runs `\keywords` and `\end{keywords}` runs
51  // `\endkeywords`. Mirror the pair rather than declaring a `DefEnvironment!`,
52  // both because that is what the `.sty` does and because it keeps a document
53  // that calls the two macros directly (without `\begin`/`\end`) working.
54  //
55  // spconf.sty says this section was "adapted from IEEEtrans", and IEEEtran.cls
56  // L5286-5288 typesets it identically (`\textit{\IEEEkeywordsname}---`). Perl
57  // LaTeXML binds that very construct in IEEEtran.cls.ltxml L147-148 as the
58  // structured `ltx:keywords` frontmatter, carrying the label in `@name` and
59  // normalizing the print-only `---` separator to `:~`. Follow that precedent
60  // exactly: the label is metadata (the XSLT renders `@name` as the block's
61  // `<h6 class="ltx_title_keywords">` title), not content of the keyword list.
62  // Raw-loaded spconf leaves it as inline body text instead — OXIDIZED_DESIGN #82.
63  DefMacro!("\\spconf@keywordsname", "Index Terms");
64  DefMacro!(
65    "\\spconf@begin@keywords",
66    "\\lx@begin@keywords[name={\\spconf@keywordsname:~}]"
67  );
68  // `\keywords` is argument-less in the `.sty`, so `\keywords{a, b}` is also
69  // legal there — the group simply typesets after the label. Routed straight to
70  // the environment opener, that form has no `\endkeywords` to stop at and
71  // `\lx@add@frontmatter@until` scans to EOF, pulling the rest of the document
72  // inside `<ltx:keywords>` (loudly — `malformed:ltx:section` and friends — but
73  // the body is wrecked, where before it was one undefined error with the text
74  // intact). Peek for a `{` and dispatch, exactly as Perl does for the same
75  // legacy pair in IEEEtran.cls.ltxml L398-404 (`\keywords@onearg`).
76  DefMacro!("\\keywords", sub[_args] {
77    if let Some(t) = read_token()? {
78      unread(Tokens!(t));
79      if t.get_catcode() == Catcode::BEGIN {
80        return Ok(Tokens!(T_CS!("\\spconf@keywords@onearg")));
81      }
82    }
83    Ok(Tokens!(T_CS!("\\spconf@begin@keywords")))
84  });
85  DefMacro!(
86    "\\spconf@keywords@onearg{}",
87    "\\spconf@begin@keywords #1\\lx@end@keywords"
88  );
89  DefMacro!("\\endkeywords", "\\lx@end@keywords");
90  // spconf uppercases the title; keep LaTeX's `\title` semantics (no forced
91  // uppercase — casing is presentational and belongs in CSS).
92  DefMacro!("\\ninept", "");
93  def_macro_noop("\\copyrightnotice{}")?;
94  def_macro_noop("\\toappear{}")?;
95});
