33/// The `\newtheorem{name}[counter]{Title}` / `\newtheorem{name}{Title}[within]` declarations of a copy of autart.cls,
34/// outside comments and other than `{thm}`'s (whose numbering the `secthm` option chooses), in the copy's order: each
35/// environment's name and its declaration.
36fn class_theorem_declarations(text: &str) -> Vec<(String, String)> {
37  let re =
38    Regex::new(r"\\newtheorem\{([A-Za-z]+)\}(\[[A-Za-z]+\])?\{([^{}]*)\}(\[[A-Za-z]+\])?").unwrap();
39  let mut out = Vec::new();
40  for line in text.lines() {
41    let code = line.split('%').next().unwrap_or_default();
42    for cap in re.captures_iter(code) {
43      let name = &cap[1];
44      if name == "thm" {
45        continue;
46      }
47      out.push((name.to_string(), cap[0].to_string()));
48    }
49  }
50  out
51}

54  LoadClass!("OmniBus");
55  // autart.cls loads no amsmath (110 copies in a 1/20 arXiv sample): amsmath's commands for papers that use them
56  // unloaded, its "Old form" error once the paper loads it; the paper's copy decides
57  // (`amsmath_sty::require_amsmath_beyond_original`, KNOWN_PERL_ERRORS #993).
58  amsmath_sty::require_amsmath_beyond_original(
59    latexml_package::shipped_class::shipped_copy_loads("cls", "amsmath")?,
60    &[],
61  )?;
62  // Do NOT eager-load amsthm here. Perl ships no autart binding (→ OmniBus,
63  // no amsthm preload), and OmniBus already installs LAZY theorem-env
64  // autoload stubs (`\begin{theorem}`/`\begin{proof}`/… each `require`
65  // amsthm on first use). Preloading amsthm eagerly breaks the common
66  // pattern of a document that clears a class-defined `\proof` and then
67  // (re)loads amsthm to get amsthm's version:
68  //     \let\proof\relax        % drop autart/class \proof
69  //     \usepackage{amsthm}     % expect amsthm to (re)define \proof
70  // With amsthm pre-loaded, the `\usepackage{amsthm}` is a no-op (already
71  // loaded), so amsthm's `\let\proof\@proof` never re-runs and `\proof`
72  // stays `\relax` → `\begin{proof}` → "{proof} environment not defined".
73  // Witness arXiv:2009.00150 (autart + `\let\proof\relax` + amsthm). The
74  // lazy OmniBus stub still covers papers that DON'T load amsthm themselves.
75
76  // autart.cls L317-323: \def\ack{\section*{Acknowledgements}}
77  // with \let\endack\par. Bind {ack}/{ack*} as structural
78  // ltx:acknowledgements (post-processors map to canonical
79  // role/styling).
80  DefEnvironment!("{ack}",  "<ltx:acknowledgements>#body</ltx:acknowledgements>",
81    mode => "internal_vertical");
82  DefEnvironment!("{ack*}", "<ltx:acknowledgements>#body</ltx:acknowledgements>",
83    mode => "internal_vertical");
84
85  // autart.cls L537 `\newenvironment{pf}{…{\bfseries\Elproofname}…}{…}` with
86  // `\def\Elproofname{PROOF.}`, plus `\@namedef{pf*}#1{…custom label…}`. autart's
87  // own proof environment — distinct from amsthm's {proof} (handled lazily via
88  // OmniBus), so the OmniBus theorem-env autoload never covers it and `{pf}` was
89  // undefined (Perl raw-executes the cls). Bind as a semantic ltx:proof (mirrors
90  // mn2e_support's {proof} pattern; self-contained, no amsthm dep — avoids the
91  // eager-amsthm hazard noted below). Witness 2309.12476 (autart, `\begin{pf}` in
92  // \input'd subfiles).
93  // A constructor pair, as autart.cls:537-543 makes `{pf}` a plain environment and `{pf*}` a macro around `\pf`: the
94  // body stays in the mode around it, so a `$$` there is display math (a DefEnvironment boxed it in restricted
95  // horizontal mode, where the formula ran as text, "Script _ can only appear in math mode": 2101.05047, 14 of 40
96  // sampled run-336 papers with `$$` and frontmatter). The end closes the `ltx:proof` only when one is open.
97  DefMacro!("\\Elproofname", "PROOF.");
98  DefConstructor!("\\pf", "<ltx:proof><ltx:title>#title</ltx:title>",
99  properties => sub[_args] {
100    // autart.cls:539 `{\bfseries\Elproofname}`.
101    let title = digest(Tokens!(T_BEGIN!(), T_CS!("\\bfseries"), T_CS!("\\Elproofname"), T_END!()))?;
102    Ok(stored_map!("title" => title))
103  });
104  DefConstructor!("\\endpf", sub[document, _args] {
105    document.maybe_close_element("ltx:proof")?;
106  });
107  RawTeX!(
108    r"\@namedef{pf*}#1{\par\begingroup\def\Elproofname{#1}\pf\endgroup\ignorespaces}
109\expandafter\let\csname endpf*\endcsname=\endpf"
110  );
111
112  // autart.cls carries elsart.cls (its last line: "End of file `elsart.cls'"), and with it elsart's `{keyword}`
113  // (autart.cls:1248-1258 of 1106.2518's copy): `\sep` between the keywords, `\PACS` and `\MSC` heading the
114  // classifications that follow them (`\sep` 40 papers of the inventory, `\PACS` 5, `\MSC` 3: 1106.2518, 0807.2769),
115  // the markup the elsart binding gives them (elsart_support_core_sty.rs `define_keyword_markup`).
116  elsart_support_core_sty::define_keyword_markup()?;
117  RawTeX!(r"\def\sep{\unskip, }");
118  // autart.cls:1429-1456 (1112.0879's copy, every copy sampled): the class's math shorthands — `\e` the upright
119  // exponential (35 papers: 1207.4104), `\half`/`\threehalf`/`\quart` the text-style fractions (11: 1112.0879), the number
120  // sets `\Rset`, `\Nset`, `\Zset`, … (7, 1, 1: 1611.06692, 1705.01052, 2404.11194), `\nuc`/`\itnuc` the nuclide
121  // superscripts and `\pol` the arrow vector, saved as `\old@vec`.
122  RawTeX!(
123    r"\def\nuc#1#2{\relax\ifmmode{}^{#1}{\protect\text{#2}}\else${}^{#1}$#2\fi}
124\def\itnuc#1#2{\setbox\@tempboxa=\hbox{\scriptsize\it #1}\def\@tempa{{}^{\box\@tempboxa}\!\protect\text{\it #2}}\relax\ifmmode \@tempa \else $\@tempa$\fi}
125\let\old@vec\vec\def\pol#1{\old@vec{#1}}
126\def\half{{\textstyle\frac{1}{2}}}\def\threehalf{{\textstyle\frac{3}{2}}}\def\quart{{\textstyle\frac{1}{4}}}
127\def\e{\mathop{\mathrm{e}}\nolimits}
128\def\Cset{\mathbb{C}}\def\Hset{\mathbb{H}}\def\Nset{\mathbb{N}}\def\Qset{\mathbb{Q}}\def\Rset{\mathbb{R}}\def\Zset{\mathbb{Z}}"
129  );
130  // autart.cls:722-723: `\cv`, the authors' curriculum vitae text, and `{cv*}` with its heading (2204.06912).
131  RawTeX!(r"\def\cv{\small}\@namedef{cv*}{\section*{Curriculum Vitae}\cv}");
132  // autart.cls:545-580: the class's theorem set, defined when the class loads with the kernel's `\newtheorem` (amsthm
133  // only under the `amsthm` option, :90-91, :518-519): `{thm}` numbered within sections under `secthm` (:545-550), the
134  // statements numbered with it, `{alg}`, `{note}`, `{summ}`, `{case}` on their own counters (`{case}` 4 papers:
135  // 2001.10125). A copy's own additions in that list are defined as it declares them (`{assumption}` 4 papers,
136  // 1912.06343's copy :558 `\newtheorem{assumption}[thm]{Assumption}`; `{property}` 2, 1804.10435's :567), which a
137  // fallback set defined lazily could not number with `thm`.
138  let secthm = class_option_given("secthm")?;
139  RawTeX!(if secthm {
140    r"\@ifundefined{thm}{\newtheorem{thm}{Theorem}[section]}{}"
141  } else {
142    r"\@ifundefined{thm}{\newtheorem{thm}{Theorem}}{}"
143  });
144  let declared = shipped_class_text("autart")?
145    .map(|text| class_theorem_declarations(&text))
146    .unwrap_or_default();
147  let declared = if declared.is_empty() {
148    AUTART_THEOREMS
149      .iter()
150      .flat_map(|d| class_theorem_declarations(d))
151      .collect()
152  } else {
153    declared
154  };
155  // Each only where the name is free, as OmniBus's fallback set (omnibus_cls.rs): its lazy `{cor}`, `{lem}`, … are
156  // the class's, numbered with `{thm}`.
157  for (name, declaration) in declared {
158    raw_tex(&format!(r"\@ifundefined{{{name}}}{{{declaration}}}{{}}"))?;
159  }

178  // autart.cls L516 defines `\qed` directly at class level (a `\Box` at the
179  // end of a proof). Perl — which OmniBus-fallbacks autart and dep-scans
180  // autart.cls's `\if@amsthm \RequirePackage{amsthm}` (the regex dep-scan
181  // ignores the `\if@amsthm` guard) — ends up loading amsthm and so produces
182  // amsthm's `\qed` (∎). We do NOT eager-load amsthm here (see the amsthm note
183  // above re: 2009.00150), so mirror amsthm's `\qed`/`\ltx@qed` (∎) directly to
184  // match Perl's ground-truth output. A paper that later `\usepackage{amsthm}`
185  // simply re-installs the identical definitions. Used via the common
186  // `\def\epf{\hfill\mbox{\qed}}` idiom OUTSIDE any proof env, so OmniBus's
187  // lazy theorem-env autoload never fires. Witness 1703.03101. Sized by amsthm's own sizer (`openbox_size`), so
188  // `\mbox{\qed}` alone in a cell is kept.
189  // autart.cls:568-577 defines the class's own algorithm float: counter `algorithm`, listed in `lof`, its caption

192  float_sty::define_float_environment("algorithm", "lof", "")?;

194  DefMacro!("\\fname@algorithm", "Algorithm");
195  DefMacro!("\\fnum@font@algorithm", "\\bfseries");

197    "\\format@title@algorithm{}",

202    "?#isMath(<ltx:XMTok role='PUNCT'>\u{220E}</ltx:XMTok>)(\u{220E})",
203    enter_horizontal => true,
204    sizer => sub[whatsit] { Ok(amsthm_sty::openbox_size(whatsit)) },
205    reversion => "\\qed"
206  );
207  Let!("\\mathqed", "\\qed");
208  Let!("\\textsquare", "\\qed");
209  Let!("\\qedsymbol", "\\qed");
210  Let!("\\openbox", "\\qed");
