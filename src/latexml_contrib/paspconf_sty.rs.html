1//! paspconf.sty — the PASP conference-proceedings substyle of AAS RevTeX 3 (v1.6, 12 Aug 93:
2//! `\revtex@org{PASP}`, `\revtex@genre{conference proceedings}`, L1-8), loaded as a
3//! `\documentstyle[11pt,paspconf]{article}` option by ASP Conference Series papers. Not in
4//! TeX Live; often not shipped either.
5//!
6//! It is aaspp.sty's sibling: the same RevTeX-3 frontmatter (`\affil` L72, `\altaffilmark`
7//! L77, `\altaffiltext` L78), table notes (L85-104), `\acknowledgments` (L159-162),
8//! `references` (L174-177), journal abbreviations (L195-227), astronomical symbols
9//! (L228-254) and `\plotone`/`\plottwo`/`\plotfiddle` (L258-265) — so it loads the aaspp
10//! binding (aas_support) and adds what is paspconf's own. Perl has no binding for the name
11//! (its FindFile_fallback, Package.pm:2177-2189, strips no part of it), so a shipped copy is
12//! raw-loaded and a missing one is an unmet `\documentstyle` option that loads OmniBus (its
13//! `\affil`, a `\reference` keyed by the entry's first token, no `{answer}`); ours had mapped the
14//! name to the unrelated ICASSP spconf binding, which defines no `\affil` (~860 papers of arXiv
15//! run 336, 298 shipping the file). Witnesses astro-ph9712155, astro-ph9512088, astro-ph9605141.
16use latexml_package::prelude::*;
17
18#[rustfmt::skip]
19LoadDefinitions!({
20  RequirePackage!("aaspp");
21  // paspconf.sty:121-122 `\let\keywords=\@gobble`, `\let\subjectheadings=\@gobble`: the PDF
22  // prints no keywords, so they are kept as metadata only (RDFa `dcterms:subject` through
23  // hyperref's `pdfkeywords`, as icml_support_sty.rs), not as a displayed keywords block.
24  DefMacro!("\\keywords{}", "\\hypersetup{pdfkeywords={#1}}");
25  Let!("\\subjectheadings", "\\keywords");
26  // paspconf.sty:41-42: line spacing; layout.
27  def_macro_noop("\\loosenlines")?;
28  // paspconf.sty:124-125: `\upper` makes `\sec@upcase` upper-case the titles; letter case is
29  // typography, the titles keep the author's text.
30  def_macro_noop("\\upper")?;
31  RawTeX!(r"\def\sec@upcase#1{\relax#1}");
32  // paspconf.sty:228 `\def\deg{\hbox{$^\circ$}}`, the degree sign (the kernel's `\deg` is the
33  // operator).
34  Let!("\\deg", "\\arcdeg");
35  // paspconf.sty:163-170: the conference discussion — the first `question` opens an unnumbered
36  // "Discussion" section; each question and answer starts with its speaker in italics.
37  RawTeX!(r"\def\qanda@heading{Discussion}
38\newif\if@firstquestion \@firstquestiontrue
39\newenvironment{question}[1]{\if@firstquestion
40\section*{\qanda@heading}\global\@firstquestionfalse\fi
41\par\vskip 1ex
42\noindent{\it#1\/}:}{\par}
43\newenvironment{answer}[1]{\par\vskip 1ex
44\noindent{\it#1\/}:}{\par}");
45  // paspconf.sty:171-173: equations numbered within sections, `<section>-<equation>`.
46  RawTeX!(r"\def\mathwithsecnums{\@addtoreset{equation}{section}%
47\def\theequation{\arabic{section}-\arabic{equation}}}");
48});
