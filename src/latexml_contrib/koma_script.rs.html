1//! Shared patches around the raw load of the KOMA-Script classes (scrartcl /
2//! scrbook / scrreprt). The classes themselves are raw-interpreted (see
3//! `scrartcl_cls.rs`); this module gives them the standard classes' semantic
4//! layer (ids, `\appendix`, the kernel sectioning check: OXIDIZED_DESIGN_DIVERGENCES
5//! #737) and re-targets the handful of commands whose real definitions are
6//! print-only and would otherwise LOSE author content or bury a heading in a
7//! presentational paragraph.
8use latexml_core::binding::counter::dialect::{NewCounterOptions, define_counter_ids};
9use latexml_package::prelude::*;
10
11/// Which KOMA class is loaded: the standard class whose binding gives its semantic layer (ids, appendix level).
12#[derive(Clone, Copy, PartialEq, Eq)]
13pub(crate) enum KomaClass {
14  /// scrartcl, as article.cls (article_cls.rs)
15  Article,
16  /// scrreprt and scrbook, as report.cls and book.cls (report_cls.rs, book_cls.rs)
17  Chaptered,
18}

20/// Applied before the raw `.cls` is read. KOMA checks that the kernel's sectioning workers are still latex.ltx's
21/// before it builds on them (scrartcl.cls:3292 + :3326, scrreprt/scrbook alike: its copy of the kernel
22/// `\@startsection` against `\@startsection`; :3520 + :3552 its copy of `\@sect` against `\@sect`) and warns when
23/// they are not ("`\@startsection' has been changed", "Unexpected definition of \@sect!"). Here the kernel's are
24/// ours — the native `\@startsection` dispatcher (sect04.rs) and the `\@sect` router
25/// (latex_constructs_rust_only.rs) — so under pdflatex neither test fails and neither Warning is written (minimal
26/// `\documentclass{scrartcl}` log, TL 2025), while every KOMA paper here had both. KOMA's copies of the kernel
27/// workers are therefore the kernel's own here: each is set to ours and kept against the class's definition of it,
28/// and the class takes the branch it takes under pdflatex. Neither copy is called on that branch (the dispatcher
29/// KOMA installs over `\@startsection` is refused by its lock, and `\scr@startsection` is ours, `koma_post_load`).
30/// Witnesses: every scrartcl/scrreprt/scrbook paper (2509.11496, 2608.23214 via ceurart). Guard
31/// `perfect_kernel_batch72a::koma_classes_take_the_kernel_branch_without_warnings`.
32pub(crate) fn koma_pre_load() -> Result<()> {
33  for (copy, kernel) in [
34    ("\\scr@saved@startsection", "\\@startsection"),
35    ("\\scr@latex@sect", "\\@sect"),
36  ] {
37    Let!(T_CS!(copy), T_CS!(kernel), Scope::Global);
38    assign_value(&s!("{copy}:locked"), true, Some(Scope::Global));
39  }
40  Ok(())
41}

43/// Applied after the raw `.cls` finished loading, so every definition here
44/// deliberately overrides the class's own.
45pub(crate) fn koma_post_load(class: KomaClass) -> Result<()> {
46  koma_counter_ids(class)?;
47  koma_appendix(class)?;
48  // Title-page pieces. scrartcl.cls L2768-2803 store these into `\@extratitle`
49  // … `\@dedication` for KOMA's own `\maketitle` (L2815) to typeset — but
50  // `\maketitle` is a locked constructor here (the class's redefinition is
51  // ignored, like `\@maketitle`), so the stored text would never reach the
52  // XML. Re-target the setters at the frontmatter. `\subtitle` is a real
53  // ltx:subtitle; the rest keep the `ltx:note[role]` shape the former stub
54  // established (witness 2305.01582, ar5iv #498: a `\titlehead` banner with
55  // the software name + `\giturl`). Neither Perl nor upstream LaTeXML binds
56  // these (no scrartcl.cls.ltxml) — surpass-Perl content recovery.
57  DefMacro!("\\subtitle{}", "\\@add@frontmatter{ltx:subtitle}{#1}");
58  for role in [
59    "titlehead",
60    "subject",
61    "publishers",
62    "extratitle",
63    "frontispiece",
64    "uppertitleback",
65    "lowertitleback",
66    "dedication",
67  ] {
68    RawTeX!(&s!(
69      r"\def\{role}#1{{\@add@frontmatter{{ltx:note}}[role={role}]{{#1}}}}"
70    ));
71    // Inside a maketitle deposit the setter fills KOMA's own field, as
72    // scrartcl.cls's `\gdef\@<role>{#1}` does (uni-titlepage's `subject=`
73    // key reads `\@subject`; sect05.rs `\lx@deposit@setters`). The field is
74    // set globally and outlives the deposit, unlike `\@title`/`\@author`/`\@date`
75    // that the kernel `\maketitle` empties: only a replayed class body reads it,
76    // and the frontmatter never takes it from there.
77    RawTeX!(&s!(
78      r"\def\lx@deposit@{role}{{\lx@deposit@field\@{role}}}\g@addto@macro\lx@deposit@setters{{\let\{role}\lx@deposit@{role}}}"
79    ));
80  }
81  RawTeX!(
82    r"\def\lx@deposit@subtitle{\lx@deposit@field\@subtitle}\g@addto@macro\lx@deposit@setters{\let\subtitle\lx@deposit@subtitle}"
83  );
84  // `\minisec{title}` (scrartcl.cls L5081-5100): an unnumbered, un-TOC'd
85  // freestanding heading. The real body is `\usekomafont{minisec}{#1\par}`
86  // inside a `\parbox`-free group — a bold sans paragraph in the XML, no
87  // heading semantics. The starred paragraph heading is the closest
88  // structural equivalent and keeps the title as
89  // `<ltx:paragraph><ltx:title>` (TL doc corpus: 17 bundles).
90  DefMacro!("\\minisec{}", "\\paragraph*{#1}");
91  // `\DeclareNewSectionCommand[style=section,level=N]{foo}` routes every KOMA
92  // `style=section` heading — standard AND author-declared — through
93  // `\scr@startsection` (scrartcl.cls:3316), a 6-arg positional drop-in for
94  // `\@startsection` `{name}{level}{indent}{beforeskip}{afterskip}{font}`. Our
95  // sectioning recognition lives only in the native (locked) `\@startsection`
96  // (sect04.rs:89-156), which standard `\section`/`\subsection` still reach; but
97  // a newly-declared command (`\task`) takes KOMA's route and never reaches it,
98  // so the heading degrades to a bold para. Aliasing the single choke point maps
99  // it back onto the SECTION_ELEMENT machinery (reads `level` as #2 → the right
100  // element). RUST-ONLY surpass (no Perl koma binding). Do NOT alias
101  // `\scr@startpart` — different signature.
102  Let!("\\scr@startsection", "\\@startsection");
103  koma_bibliography_preamble()?;
104  Ok(())
105}

107/// `\setbibpreamble{<text>}` (scrartcl.cls:6752-6754, scrreprt/scrbook alike) sets `\bibpreamble`, which KOMA's
108/// `thebibliography` typesets after the heading when it is neither `\relax` (its default) nor empty (:6762-6763,
109/// `\noindent\bibpreamble\par\nobreak`). The bibliography here is the kernel's (the class's `thebibliography` gives way
110/// to it), which typesets `\lx@bibliography@preamble` there (sect11.rs): it is KOMA's test and its text. Before, the text
111/// was dropped without a diagnostic. natbib, loaded after the class, points the preamble at its own `\bibpreamble`
112/// (natbib_sty.rs), as its `thebibliography` replaces KOMA's. Neither Perl (no KOMA binding) nor the former binding
113/// printed it. Guard `perfect_kernel_batch77v::koma_setbibpreamble_is_printed`.
114fn koma_bibliography_preamble() -> Result<()> {
115  RawTeX!(
116    r"\def\lx@bibliography@preamble{\ifx\bibpreamble\relax\else\ifx\bibpreamble\@empty\else\noindent\bibpreamble\par\fi\fi}"
117  );
118  Ok(())
119}

121/// scrbook's `\frontmatter`, `\mainmatter` and `\backmatter` (scrbook.cls:5272-5283) switch `\if@mainmatter`, by which
122/// KOMA's chapter numbers (scrbook.cls:5573-5576 `\if@mainmatter \@tempswatrue`): a front- or back-matter chapter is
123/// unnumbered and leaves the chapter counter alone. LaTeXML's sectioning numbers by `no_number_sections` instead
124/// (sect04.rs `\@startsection`), which book.cls's binding sets in its `\frontmatter`/`\mainmatter` (book_cls.rs; Perl
125/// book.cls.ltxml:112-118). Here the two also set it, before KOMA's own code (the page clearing, `\@mainmatterfalse`,
126/// `\pagenumbering`), which stays as written. `\backmatter` is left to KOMA: the flag unnumbers every level, where KOMA
127/// unnumbers only the chapter and keeps a section's number without the chapter's (scrbook.cls:6030-6036
128/// `\if@mainmatter\thechapter.\fi`), and a back-matter appendix's sections are referenced by those numbers
129/// (2508.01496: "Appendix" with sections 1-24; set at `\backmatter`, every reference to them lost its number). Book's
130/// `no_number_chapters` (book_cls.rs, 78a) unnumbers the chapter alone but does not reset the counters within it, as
131/// an unnumbered KOMA chapter does (scrbook.cls:5590-5600 from v3.15): 2508.01496's appendix figures would be off by
132/// one, so its appendix chapter stays "Appendix A" (PDF "Appendix"). Witness 2506.17485: "Abstract" unnumbered, then "1 Introduction", as its PDF (before: "Chapter 1
133/// Abstract", every later number one too high). Guard
134/// `perfect_kernel_batchint23::scrbook_front_matter_chapters_are_unnumbered`.
135pub(crate) fn koma_book_matter() -> Result<()> {
136  DefPrimitive!("\\lx@koma@unnumbered", {
137    AssignValue!("no_number_sections" => true);
138  });
139  DefPrimitive!("\\lx@koma@numbered", {
140    AssignValue!("no_number_sections" => false);
141  });
142  RawTeX!(
143    r"\let\lx@koma@frontmatter\frontmatter\def\frontmatter{\lx@koma@unnumbered\lx@koma@frontmatter}%
144\let\lx@koma@mainmatter\mainmatter\def\mainmatter{\lx@koma@numbered\lx@koma@mainmatter}"
145  );
146  Ok(())
147}

149/// The ids of the sectioning and float counters, as the standard class's binding gives them (article_cls.rs:86-107,
150/// report_cls.rs:94-130, book_cls.rs:100-136; Perl article.cls.ltxml:52-63): the raw class declares the counters with
151/// `\newcounter` (scrartcl.cls:5045-5127, scrreprt.cls:5530-6051, tocbasic's `\DeclareNewTOC` for figure and table),
152/// which names their ids after the counter (`section1.subsection1`, `equation1`), where an article paper's are
153/// `S1.SS1` and `S1.E1`. Perl, which has no KOMA binding, loads OmniBus (article) for these classes and gives the
154/// article's ids. Only the ids are set: the registers and reset lists stay the class's. A counter the class did not
155/// declare is left alone. Witnesses: 124 of the 126 ceurart/atlasdoc papers of 2506+2509 and 97 of the KOMA 100 had
156/// counter-named ids (2509.11496, 2506.00694, 2509.01219). Guard
157/// `perfect_kernel_batch72a::koma_ids_and_appendix_follow_the_standard_classes`.
158fn koma_counter_ids(class: KomaClass) -> Result<()> {
159  let opts = |idprefix, idwithin, nested| NewCounterOptions { idprefix, idwithin, nested };
160  let table: Vec<(&str, &str, NewCounterOptions)> = match class {
161    KomaClass::Article => vec![
162      ("part", "document", opts("Pt", "", vec!["section"])),
163      ("section", "document", opts("S", "", vec!["subsection"])),
164      (
165        "subsection",
166        "section",
167        opts("SS", "", vec!["subsubsection"]),
168      ),
169      (
170        "subsubsection",
171        "subsection",
172        opts("SSS", "", vec!["paragraph"]),
173      ),
174      (
175        "paragraph",
176        "subsubsection",
177        opts("P", "", vec!["subparagraph"]),
178      ),
179      (
180        "subparagraph",
181        "paragraph",
182        opts("SP", "", vec!["equation", "figure", "table"]),
183      ),
184      ("equation", "document", opts("E", "section", vec![])),
185      ("figure", "document", opts("F", "section", vec![])),
186      ("table", "document", opts("T", "section", vec![])),
187    ],
188    KomaClass::Chaptered => vec![
189      ("part", "document", opts("Pt", "", vec!["chapter"])),
190      ("chapter", "document", opts("Ch", "", vec!["section"])),
191      ("section", "chapter", opts("S", "", vec!["subsection"])),
192      (
193        "subsection",
194        "section",
195        opts("SS", "", vec!["subsubsection"]),
196      ),
197      (
198        "subsubsection",
199        "subsection",
200        opts("SSS", "", vec!["paragraph"]),
201      ),
202      (
203        "paragraph",
204        "subsubsection",
205        opts("P", "", vec!["subparagraph"]),
206      ),
207      (
208        "subparagraph",
209        "paragraph",
210        opts("SP", "", vec!["equation", "figure", "table"]),
211      ),
212      ("equation", "chapter", opts("E", "", vec![])),
213      ("figure", "chapter", opts("F", "", vec![])),
214      ("table", "chapter", opts("T", "", vec![])),
215    ],
216  };
217  for (ctr, within, options) in table {
218    if lookup_definition(&T_CS!(s!("\\c@{ctr}")))?.is_some() {
219      define_counter_ids(ctr, within, Some(&options))?;
220    }
221  }
222  Ok(())
223}

225/// `\appendix` starts the appendices, as the standard class's binding does (article_cls.rs:148-152,
226/// report_cls.rs:178-182; Perl article.cls.ltxml `\@appendix` → `startAppendices('section')`), and then runs the
227/// class's own (scrartcl.cls:4842-4867, scrreprt.cls:5305-5335, scrbook.cls:5284-5314): KOMA's counter resets, its
228/// `\thesection`/`\thechapter` lettering, `\@chapapp` = `\appendixname`, its one-time hook and the author's
229/// `\appendixmore` stay as written, after `start_appendices` has put the section (scrartcl) or chapter
230/// (scrreprt/scrbook) level aside for the appendix. Before, a KOMA appendix was more sections (`ltx:section`, its
231/// ids colliding with the body's: `section1a`), in 26 of 27 ceurart/atlasdoc papers with one (2509.11496,
232/// 2506.00694). A second `\appendix` runs KOMA's alone (`\@appendix` is `\relax` after the first, as in the
233/// standard classes).
234fn koma_appendix(class: KomaClass) -> Result<()> {
235  let level = match class {
236    KomaClass::Article => "section",
237    KomaClass::Chaptered => "chapter",
238  };
239  DefPrimitive!("\\@appendix", {
240    start_appendices(level);
241  });
242  RawTeX!(r"\let\lx@koma@appendix\appendix\def\appendix{\@appendix\lx@koma@appendix}");
243  Ok(())
