1//! aistats.sty — the AISTATS conference styles (aistats2012.sty, aistats2014.sty … aistats2027.sty, which the versioned
2//! fallback strips to `aistats`; aistats2026.sty, and the 2010-2011 aistats2e.sty and its margin variant aistats2eM.sty,
3//! by their own names): one family binding (as Perl's icml.sty.ltxml serves the ICML years, icml_support.sty.ltxml:34ff).
4//! The paper's copy is read raw, then its title-block setters, which print in place inside `\twocolumn[…]` and never
5//! reach `\title`/`\author` (aistats2027.sty:173/248/355, aistats2eM.sty:102/171/252, the same in every year), are the
6//! frontmatter's:
7//! - `\aistatstitle{T}` is the title (`\title`); a later one prints in place as the copy does (the supplement's
8//!   "Supplementary Material" heading, 2610.11580), where a second `\title` would be dropped;
9//! - `\aistatsauthor{A₁ \And A₂}` followed by `\aistatsaddress{a₁ \And a₂}` are the authors, each column of the address
10//!   under the author of its column when the row has several columns and the address as many (`A₁ \\ a₁ \And A₂ \\ a₂`:
11//!   name, affiliation and e-mail lines, as the kernel `\author` reads them; 2604.21432, 2605.03921, 2610.11580,
12//!   1002.4058) — also across material between them that prints nothing (`\runningauthor`, `\vspace`, a definition:
13//!   2003.03080, 2402.12508, 1302.5348), which runs first; otherwise the names, and the address shared by them (`\lx@add@affiliation@shared`): each column an
14//!   affiliation of the row, its e-mail lines the authors' (1806.02924's four names over one Carnegie Mellon address,
15//!   2510.09534's one column of names apart by `\quad`), or, marked, each line to the authors whose marks it names
16//!   (1910.09056's `$^1$University of British Columbia, $^2$…`, a row of one column). Each `\aistatsauthor` row adds to
17//!   the ones before. Their arguments are read where they stand (`in_place`: 1302.5348's `\UMDAddress`).
18//!
19//! What the PDF shows decides (user ruling 2026-10-06): a submission under review (`\statePaper` 0, no `accepted` or
20//! 2027 `preprint` option) prints "Anonymous Author" and "Anonymous Institution" where the copy says so (aistats2020.sty
21//! … aistats2027.sty; 2e, 2012, 2014 and 2019 print the names either way). Rebinding `\aistatsauthor` also retires the
22//! copy's running-head measurement, whose `\PackageError{Document}{Running heading author exceeds size limitations}`
23//! (aistats2026.sty:289-295) the former aistats2026 stub pre-empted with a placeholder `\@runningauthor` (47 papers,
24//! stages 15-20). Perl has no aistats binding: the setters print in place there too, or are undefined (3 errors).
25//! OXIDIZED_DESIGN_DIVERGENCES #772. Witnesses 1002.4058, 1302.5134, 1302.5348, 1303.0449, 1103.0941 (2e), 1202.1568
26//! (2eM), 1102.1492 (2012), 1311.1894 (2014), 1806.02924 (2019), 2604.21432 (2020), 1910.09056 (2022), 2405.15172 (2025),
27//! 2605.03921, 2510.09534 (2026), 2610.11580 (2027); 2101.00218 ships aistats2021.sty unused.

34LoadDefinitions!({
35  // The style as requested (`aistats2027`, also through the versioned fallback: `fallback_request`, possibly with a
36  // directory).
37  let fallback_request = lookup_string("fallback_request");
38  let request = match fallback_request.rsplit('/').next() {
39    Some(name) if name.to_ascii_lowercase().starts_with("aistats") => fallback_request.clone(),
40    _ => do_expand(T_CS!("\\@currname"))?.to_string(),
41  };
42  let shipped = !request.is_empty()
43    && find_file(
44      &request,
45      Some(FindFileOptions {
46        ext_type: Some(Cow::Borrowed("sty")),
47        forbid_ltxml: true,
48        ..Default::default()
49      }),
50    )
51    .is_some();
52  let mut anonymous_under_review = false;
53  if shipped {
54    InputDefinitions!(&request, noltxml => true, extension => Some(Cow::Borrowed("sty")));
55    anonymous_under_review =
56      shipped_copy_text("sty")?.is_some_and(|text| class_code_contains(&text, "Anonymous Author"));
57  }
58  assign_value(
59    "lx_aistats_anonymous",
60    anonymous_under_review,
61    Some(Scope::Global),
62  );
63  // Each setter where the copy defines it (every year's does), or, without a copy, as the family's interface.
64  let owns = |cs: &str| !shipped || IsDefined!(&T_CS!(cs));
65  if owns("\\aistatstitle") {
66    if IsDefined!(&T_CS!("\\aistatstitle")) {
67      Let!("\\lx@aistats@title@inplace", "\\aistatstitle");
68    } else {
69      // (aistats2027.sty:225-226)
70      DefMacro!(
71        "\\lx@aistats@title@inplace{}",
72        "{\\centering{\\Large\\bfseries #1\\par}}"
73      );
74    }
75    // (aistats2022.sty:172 `\gdef\@supplementtitle{#1}`, which its `\makesupplementtitle` prints, :366-368; 1910.09056)
76    DefMacro!(
77      "\\aistatstitle{}",
78      "\\gdef\\@supplementtitle{#1}\\lx@aistats@title{#1}"
79    );
80    DefMacro!("\\lx@aistats@title{}", sub[(title)] {
81      if lookup_bool("lx_aistats_titled") {
82        Ok(Invocation!(T_CS!("\\lx@aistats@title@inplace"), vec![Some(title)]))
83      } else {
84        assign_value("lx_aistats_titled", true, Some(Scope::Global));
85        Ok(Invocation!(T_CS!("\\title"), vec![None, Some(title)]))
86      }
87    });
88  }
89  if owns("\\aistatsauthor") {
90    DefMacro!("\\aistatsauthor{}", sub[(names)] { aistats_authors(names) });
91    DefMacro!("\\lx@aistats@pair{}{}", sub[(names, address)] { aistats_pair(names, address) });
92  }
93  if owns("\\aistatsaddress") {
94    DefMacro!("\\aistatsaddress{}", sub[(address)] { aistats_address(address) });
95  }
96});

98/// Whether the copy prints the anonymous block: a submission (`\statePaper` is not `\acceptedPaper`'s 1) to a year
99/// whose copy has one.
100fn under_review() -> bool {
101  lookup_bool("lx_aistats_anonymous")
102    && IsDefined!(&T_CS!("\\statePaper"))
103    && do_expand(T_CS!("\\statePaper")).is_ok_and(|state| state.to_string().trim() != "1")
104}

106/// The columns of an author row or an address, split at the copy's column separators outside braces: `\And`, and
107/// `\AND`, `\ANDD`, which also start a new row (aistats2027.sty:315-325, the same in every year).
108fn columns(tokens: Tokens) -> Vec<Tokens> {
109  split_tokens(tokens, vec![
110    T_CS!("\\And").into(),
111    T_CS!("\\AND").into(),
112    T_CS!("\\ANDD").into(),
113  ])
114  .into_iter()
115  .filter(|column| {
116    !column
117      .unlist_ref()
118      .iter()
119      .all(|t| t.get_catcode() == Catcode::SPACE)
120  })
121  .collect()
122}

124/// Tokens that assign or test a meaning (`\renewcommand{\Uni}{…}`, `\ifx\Uni\relax`, `\string\Uni`): a column of a
125/// setter argument holding one is not read in place ([`read_columns`]), as the meaning it names is the one the column
126/// itself makes or asks about where it is digested — reading the target in place garbled it (review probe `g_renewin`: "rstSecondFirst"
127/// for pdflatex's "Second"; 2511.05163's `\renewcommand{\arraystretch}{1.0}` inside its names).
128fn assigns_or_tests(tokens: &[Token]) -> bool {
129  tokens.iter().any(|token| {
130    token.get_catcode() == Catcode::CS
131      && token.with_str(|name| {
132        matches!(
133          name,
134          "\\def"
135            | "\\gdef"
136            | "\\edef"
137            | "\\xdef"
138            | "\\let"
139            | "\\futurelet"
140            | "\\global"
141            | "\\newcommand"
142            | "\\renewcommand"
143            | "\\providecommand"
144            | "\\DeclareRobustCommand"
145            | "\\ifx"
146            | "\\ifdefined"
147            | "\\ifcsname"
148            | "\\noexpand"
149            | "\\string"
150            | "\\meaning"
151            | "\\csname"
152            | "\\detokenize"
153            | "\\unexpanded"
154        )
155      })
156  })
157}

159/// `tokens` as the setter prints them where it stands: each macro without parameters whose meaning the paper made
160/// inside a group open now (`\twocolumn[…]`'s, whose `]` ends it; [`meaning_is_group_local`]) replaced by its expansion,
161/// as the frontmatter that holds them is digested after that group, where the meaning is gone (1302.5348's
162/// `\newcommand{\UMDAddress}{University of Maryland \\ College Park, MD 20742, USA}` between `\aistatsauthor` and
163/// `\aistatsaddress`: "undefined"; a global `\Uni` renewed inside the block, review probe `g_globlocal`: the global
164/// meaning). A column that assigns or tests a meaning is not read in place ([`read_columns`]), and the body of such a
165/// macro that does is not read further (review probe `g_hash`'s `\newcommand{\Uni}{\def\z##1{[##1]}\z{Hash Univ}}`).
166/// Other macros stay; a local macro with parameters (`\def\Uni#1{…}`, probe `g_args`) or a local `\let` of a primitive
167/// (`\let\Uni\relax`, probe `g_letprim`) is not read in place (SYNC residual).
168fn in_place(tokens: Tokens) -> Tokens {
169  fn expand(tokens: &[Token], depth: usize, out: &mut Vec<Token>) {
170    for token in tokens {
171      if depth < 8
172        && meaning_is_group_local(token)
173        && let Some(Stored::Expandable(definition)) = lookup_meaning(token)
174        && definition.get_parameters().is_none()
175        && let Some(ExpansionBody::Tokens(body)) = definition.get_expansion()
176      {
177        if assigns_or_tests(body.unlist_ref()) {
178          out.extend_from_slice(body.unlist_ref());
179        } else {
180          expand(body.unlist_ref(), depth + 1, out);
181        }
182      } else {
183        out.push(*token);
184      }
185    }
186  }
187  let mut out = Vec::new();
188  expand(tokens.unlist_ref(), 0, &mut out);
189  Tokens::new(out)
190}

192/// `columns` joined by `\And`, the kernel's author separator.
193fn joined(columns: Vec<Tokens>) -> Tokens {
194  let mut out: Vec<Token> = Vec::new();
195  for (i, column) in columns.into_iter().enumerate() {
196    if i > 0 {
197      out.push(T_CS!("\\And"));
198    }
199    out.extend(column.unlist());
200  }
201  Tokens::new(out)
202}

204/// `\aistatsaddress{a₁ \And a₂}` not paired with the names before it: each column an affiliation of the authors of the
205/// row before it (`annotate=run`; columns printed alike, one), as the copy prints them centred under the row — split,
206/// not one affiliation "a₁ & a₂" (review probes `a_mismatch`, `a_addrfirst`) —, their e-mail lines the authors' e-mails
207/// (`\lx@add@affiliation@shared`); marked, each affiliation goes to the authors whose marks it names (1910.09056's
208/// `$^1$University of British Columbia, $^2$…`).
209fn aistats_address(address: Tokens) -> Result<Tokens> {
210  let address = if under_review() {
211    vec![Tokens::new(Explode!("Anonymous Institution"))]
212  } else {
213    read_columns(address)
214  };
215  shared_address(address)
216}

218/// The address columns as one shared address ([`aistats_address`]).
219fn shared_address(columns: Vec<Tokens>) -> Result<Tokens> {
220  Ok(Invocation!(T_CS!("\\lx@add@affiliation@shared"), vec![
221    Some(Tokens::new(Explode!("annotate=run"))),
222    Some(joined(columns))
223  ]))
224}

226/// The columns of a setter argument ([`columns`]), each read in place ([`in_place`]) unless it assigns or tests a
227/// meaning ([`assigns_or_tests`]) — column by column, so a renewal in one leaves the next read in place (pdflatex's
228/// "Second" and "Local Univ" for `\renewcommand{\Uni}{Second}\Uni \And \Glob`); a macro that expands to several
229/// columns is split again.
230fn read_columns(tokens: Tokens) -> Vec<Tokens> {
231  columns(tokens)
232    .into_iter()
233    .flat_map(|column| {
234      if assigns_or_tests(column.unlist_ref()) {
235        vec![column]
236      } else {
237        columns(in_place(column))
238      }
239    })
240    .collect()
241}

243/// How a command that prints nothing in a title block reads what follows it ([`unprinted`]).
244#[derive(Clone, Copy)]
245enum Unprinted {
246  /// Nothing (`\medskip`, `\par`, a size or font declaration, `\selectfont`, `\begingroup`).
247  Bare,
248  /// An optional `*`, then that many arguments (`\vspace*{…}`, `\fontsize{…}{…}`, the copy's `\runningauthor{…}`).
249  Arguments(usize),
250  /// An optional `*` and `[…]` (`\\`).
251  LineBreak,
252  /// Glue (`\vskip 12pt`, `\vskip\baselineskip`).
253  Glue,
254  /// An optional `*`, the command, optional `[…]`s, the body (`\newcommand{\cs}[1]{…}`).
255  NewCommand,
256  /// The command, its parameter text, the body (`\def\cs#1{…}`).
257  Def,
258  /// The command, an optional `=`, the meaning (`\let\cs\relax`).
259  Let,
260}

262/// The commands that print nothing where they stand between an author row and its address, and how each reads its
263/// arguments: vertical space, definitions, size and font declarations, the copy's running-head setters
264/// (aistats2027.sty:170-171 `\def\runningauthor#1{\gdef\@runningauthor{#1}}`), groups — what the 2,843 papers with
265/// `\aistatsauthor` put there (77o review scan: `\runningauthor` 40 rows, `\vspace` 79, `\medskip`, `\vskip`,
266/// `\newcommand`, `\renewcommand\thefootnote`, `\fontsize…\selectfont`, `~\\`, `\begingroup…\endgroup`).
267fn unprinted(token: &Token) -> Option<Unprinted> {
268  if token.get_catcode() != Catcode::CS {
269    return None;
270  }
271  token.with_str(|name| {
272    Some(match name {
273      "\\par" | "\\medskip" | "\\smallskip" | "\\bigskip" | "\\relax" | "\\noindent"
274      | "\\begingroup" | "\\endgroup" | "\\global" | "\\selectfont" | "\\normalsize"
275      | "\\small" | "\\footnotesize" | "\\scriptsize" | "\\tiny" | "\\large" | "\\Large"
276      | "\\LARGE" | "\\huge" | "\\Huge" | "\\normalfont" | "\\bf" | "\\it" | "\\rm" | "\\sf"
277      | "\\tt" | "\\sc" | "\\sl" | "\\bfseries" | "\\mdseries" | "\\itshape" | "\\upshape"
278      | "\\slshape" | "\\scshape" | "\\rmfamily" | "\\sffamily" | "\\ttfamily" => Unprinted::Bare,
279      "\\vspace" | "\\addvspace" | "\\runningauthor" | "\\runningtitle" | "\\newlength" => {
280        Unprinted::Arguments(1)
281      },
282      "\\fontsize" | "\\setlength" | "\\addtolength" | "\\setcounter" => Unprinted::Arguments(2),
283      "\\\\" | "\\newline" => Unprinted::LineBreak,
284      "\\vskip" => Unprinted::Glue,
285      "\\newcommand" | "\\renewcommand" | "\\providecommand" | "\\DeclareRobustCommand" => {
286        Unprinted::NewCommand
287      },
288      "\\def" | "\\gdef" | "\\edef" | "\\xdef" => Unprinted::Def,
289      "\\let" => Unprinted::Let,
290      _ => return None,
291    })
292  })
293}

295/// The next token, the spaces before it read into `read`; `None` at the end of the input.
296fn next_non_space(read: &mut Vec<Token>) -> Result<Option<Token>> {
297  while let Some(token) = read_token()? {
298    if token.get_catcode() == Catcode::SPACE {
299      read.push(token);
300    } else {
301      return Ok(Some(token));
302    }
303  }
304  Ok(None)
305}

307/// The rest of a group whose `{` is read, into `read`: whether its `}` is.
308fn read_group_rest(read: &mut Vec<Token>) -> Result<bool> {
309  let mut depth = 1usize;
310  while let Some(token) = read_token()? {
311    read.push(token);
312    match token.get_catcode() {
313      Catcode::BEGIN => depth += 1,
314      Catcode::END => {
315        depth -= 1;
316        if depth == 0 {
317          return Ok(true);
318        }
319      },
320      _ => {},
321    }
322  }
323  Ok(false)
324}

326/// One argument, a group or a token, into `read`: whether there was one (not a `}` or the end of the input).
327fn read_argument(read: &mut Vec<Token>) -> Result<bool> {
328  let Some(token) = next_non_space(read)? else {
329    return Ok(false);
330  };
331  read.push(token);
332  match token.get_catcode() {
333    Catcode::BEGIN => read_group_rest(read),
334    Catcode::END => Ok(false),
335    _ => Ok(true),
336  }
337}

339/// An optional `*` into `read`.
340fn read_star(read: &mut Vec<Token>) -> Result<()> {
341  if let Some(token) = next_non_space(read)? {
342    if token == T_OTHER!("*") {
343      read.push(token);
344    } else {
345      unread_one(token);
346    }
347  }
348  Ok(())
349}

351/// An optional `[…]` into `read`: whether its `]` is read (or there is none).
352fn read_bracketed(read: &mut Vec<Token>) -> Result<bool> {
353  let Some(token) = next_non_space(read)? else {
354    return Ok(true);
355  };
356  if token != T_OTHER!("[") {
357    unread_one(token);
358    return Ok(true);
359  }
360  read.push(token);
361  let mut depth = 0usize;
362  while let Some(token) = read_token()? {
363    read.push(token);
364    match token.get_catcode() {
365      Catcode::BEGIN => depth += 1,
366      Catcode::END if depth == 0 => return Ok(false),
367      Catcode::END => depth -= 1,
368      _ if depth == 0 && token == T_OTHER!("]") => return Ok(true),
369      _ => {},
370    }
371  }
372  Ok(false)
373}

375/// The arguments a command of [`unprinted`] reads, into `read`: whether they are all there.
376fn read_unprinted_arguments(shape: Unprinted, read: &mut Vec<Token>) -> Result<bool> {
377  Ok(match shape {
378    Unprinted::Bare => true,
379    Unprinted::Arguments(count) => {
380      read_star(read)?;
381      for _ in 0..count {
382        if !read_argument(read)? {
383          return Ok(false);
384        }
385      }
386      true
387    },
388    Unprinted::LineBreak => {
389      read_star(read)?;
390      read_bracketed(read)?
391    },
392    Unprinted::Glue => {
393      // a signed number, then a register or a unit
394      let mut next = next_non_space(read)?;
395      while let Some(token) = next.filter(|t| {
396        t.get_catcode() == Catcode::SPACE
397          || (t.get_catcode() == Catcode::OTHER && t.with_str(|s| "+-0123456789.,".contains(s)))
398      }) {
399        read.push(token);
400        next = read_token()?;
401      }
402      match next {
403        Some(token) if token.get_catcode() == Catcode::CS => read.push(token),
404        Some(token) if token.get_catcode() == Catcode::LETTER => {
405          read.push(token);
406          let mut next = read_token()?;
407          while let Some(token) = next.filter(|t| t.get_catcode() == Catcode::LETTER) {
408            read.push(token);
409            next = read_token()?;
410          }
411          if let Some(token) = next {
412            unread_one(token);
413          }
414        },
415        Some(token) => unread_one(token),
416        None => return Ok(false),
417      }
418      true
419    },
420    Unprinted::NewCommand => {
421      read_star(read)?;
422      read_argument(read)? && read_bracketed(read)? && read_bracketed(read)? && read_argument(read)?
423    },
424    Unprinted::Def => {
425      let Some(name) = read_token()? else {
426        return Ok(false);
427      };
428      read.push(name);
429      loop {
430        let Some(token) = read_token()? else {
431          return Ok(false);
432        };
433        read.push(token);
434        match token.get_catcode() {
435          Catcode::BEGIN => break read_group_rest(read)?,
436          Catcode::END => return Ok(false),
437          _ => {},
438        }
439      }
440    },
441    Unprinted::Let => {
442      let Some(name) = read_token()? else {
443        return Ok(false);
444      };
445      read.push(name);
446      let Some(mut meaning) = next_non_space(read)? else {
447        return Ok(false);
448      };
449      if meaning == T_OTHER!("=") {
450        read.push(meaning);
451        let Some(next) = read_token()? else {
452          return Ok(false);
453        };
454        meaning = next;
455        if meaning.get_catcode() == Catcode::SPACE {
456          read.push(meaning);
457          let Some(next) = read_token()? else {
458            return Ok(false);
459          };
460          meaning = next;
461        }
462      }
463      read.push(meaning);
464      true
465    },
466  })
467}

469/// The `\aistatsaddress{…}` an author row stands over, read ahead through what prints nothing between them: spaces,
470/// `~` and paragraph ends (the template's lines), the commands of [`unprinted`] with their arguments, groups (2501.00988's
471/// `{\fontsize{9.6pt}{12pt}\selectfont` before it). With the material between, in order, without its spaces, `~`
472/// and paragraph ends. `None` — what was read is read again — when something that may print comes first (a character, a
473/// command not of [`unprinted`]: a paragraph of text between them keeps the address unpaired), the next
474/// `\aistatsauthor`, or the end of the block (the `}` closing the group the row stands in).
475fn address_ahead() -> Result<Option<(Vec<Token>, Tokens)>> {
476  let mut read: Vec<Token> = Vec::new();
477  let mut kept: Vec<Token> = Vec::new();
478  let mut depth = 0usize;
479  loop {
480    let Some(token) = read_token()? else {
481      break;
482    };
483    if token == T_CS!("\\aistatsaddress") {
484      return Ok(Some((kept, read_arg(ExpansionLevel::Off)?)));
485    }
486    read.push(token);
487    let code = token.get_catcode();
488    if code == Catcode::SPACE
489      || token == T_CS!("\\par")
490      || (code == Catcode::ACTIVE && token.with_str(|s| s == "~"))
491    {
492      continue;
493    }
494    // (a source comment kept under `--comments` stays with the material)
495    if code == Catcode::COMMENT {
496      kept.push(token);
497      continue;
498    }
499    let start = read.len() - 1;
500    let complete = match code {
501      Catcode::BEGIN => {
502        depth += 1;
503        true
504      },
505      Catcode::END if depth > 0 => {
506        depth -= 1;
507        true
508      },
509      _ => match unprinted(&token) {
510        Some(shape) => read_unprinted_arguments(shape, &mut read)?,
511        None => false,
512      },
513    };
514    if !complete {
515      break;
516    }
517    kept.extend_from_slice(&read[start..]);
518  }
519  unread_vec(read);
520  Ok(None)
521}

523/// `\aistatsauthor{names}`: the authors, with the `\aistatsaddress{…}` that stands under them when only what prints
524/// nothing comes between ([`address_ahead`]), that material first, so a macro it defines is read in place
525/// (1302.5348's `\newcommand{\UMDAddress}`; 2003.03080's `\vspace`, 2402.12508's `\runningauthor{…}`, 2602.12181's
526/// `\medskip`: [`aistats_pair`]); else the names, one author list across the columns, and an address after other
527/// material goes on its own ([`aistats_address`]).
528fn aistats_authors(names: Tokens) -> Result<Tokens> {
529  if let Some((mut before, address)) = address_ahead()? {
530    before
531      .extend(Invocation!(T_CS!("\\lx@aistats@pair"), vec![Some(names), Some(address)]).unlist());
532    return Ok(Tokens::new(before));
533  }
534  if under_review() {
535    return Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(
536      Tokens::new(Explode!("Anonymous Author"))
537    )]));
538  }
539  Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(
540    joined(read_columns(names))
541  )]))
542}

544/// An author row and the address under it: each column of names over the column of the address under it when the row
545/// has several columns and the address as many, in as many rows (`\AND`), and the two are not marked — what stands
546/// between them aside (2003.03080, 2402.12508, 2602.12181,
547/// 2305.16147, 1804.04378, 1302.5348, 1902.01239, 1603.09584) — (`A₁ \\ a₁ \And A₂ \\ a₂`: name, affiliation and
548/// e-mail lines, as the kernel `\author` reads them; 2604.21432, 2605.03921, 2610.11580, 1002.4058); else the names,
549/// one author list across the columns, and the address after them ([`aistats_address`]) — a row of one column is names
550/// the address is printed under together (2510.09534's `Jeong$^\ast$ \quad Zhai$^\ast$ \quad Ročková` over one
551/// University of Chicago; 1910.09056's marked names and addresses), as is a row of other columns than its address
552/// (1806.02924's four names over one Carnegie Mellon address) or in other rows, and a marked row over a marked address,
553/// whose marks pair them (2312.06786's `$^1$Carnegie Mellon University \And $^2$Microsoft Research \AND {…}@… \And …`
554/// under four marked names: four columns, two of them e-mails on a second row).
555fn aistats_pair(names: Tokens, address: Tokens) -> Result<Tokens> {
556  if under_review() {
557    let mut block = Explode!("Anonymous Author");
558    block.push(T_CS!("\\\\"));
559    block.extend(Explode!("Anonymous Institution"));
560    return Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(
561      Tokens::new(block)
562    )]));
563  }
564  // (rows of the two tables alike, and no marks to pair them by: an address is a key the names' marks index only when it
565  // shows two or more distinct marks — one mark is a note's, `$^*$equal contribution`, and the PDF still prints column i
566  // under author i; 77o r3 review, 1905.11190, 2405.18281)
567  let row_breaks = |tokens: &Tokens| {
568    tokens
569      .unlist_ref()
570      .iter()
571      .filter(|t| **t == T_CS!("\\AND") || **t == T_CS!("\\ANDD"))
572      .count()
573  };
574  let by_column = row_breaks(&names) == row_breaks(&address)
575    && !(has_author_marks(&names) && distinct_author_marks(&address) >= 2);
576  let name_columns = read_columns(names);
577  let address_columns = read_columns(address);
578  if by_column && name_columns.len() > 1 && name_columns.len() == address_columns.len() {
579    let mut block: Vec<Token> = Vec::new();
580    for (i, (name, place)) in name_columns.into_iter().zip(address_columns).enumerate() {
581      if i > 0 {
582        block.push(T_CS!("\\And"));
583      }
584      block.extend(name.unlist());
585      if !place.is_empty() {
586        block.push(T_CS!("\\\\"));
587        block.extend(place.unlist());
588      }
589    }
590    return Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(
591      Tokens::new(block)
592    )]));
593  }
594  let mut out = Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(joined(
595    name_columns
596  ))])
597  .unlist();
598  out.extend(shared_address(address_columns)?.unlist());
599  Ok(Tokens::new(out))
600}
