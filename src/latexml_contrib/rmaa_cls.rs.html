1//! rmaa.cls (Revista Mexicana de Astronomía y Astrofísica, J. Henney's class, 2000-2007): the papers ship the class (it
2//! is not in TeX Live), which is read raw; a paper whose source lacks it falls to OmniBus, with the loader's
3//! missing-file Warning. Its title page keeps stores only its own `\maketitle` prints (rmaa.cls:914-940, :943-999),
4//! which the kernel's locked `\maketitle` refuses, so their text was lost: the affiliations `\altaffiltext` appends to
5//! `\@thanks` (:500-503, printed as footnotes by :932, which the kernel's `\maketitle` empties), the Spanish abstract
6//! (:450-453 `\resumen`, printed above the abstract, :975) and the keywords (:509-541 `\addkeyword`, `\keywords`, the
7//! "Key Words:" line, :981-993); an `\affil` inside an `\author` (:491-494) stayed in the name; the running head's
8//! `\shorttitle` (:550) was lost too. They are the frontmatter's here, defined before the class is read, whose
9//! `\newcommand`s of the same names are then ignored as those of any name LaTeXML defines. Perl has no binding and
10//! reads rmaa through OmniBus, whose `\altaffilmark`/`\altaffiltext` and `\shorttitle` these are; it lacks
11//! `\addkeyword` (an undefined macro) and keeps the Spanish abstract in place of the English one. Witnesses 0903.4819,
12//! astro-ph0208329, 1301.7157 (OXIDIZED_DESIGN_DIVERGENCES #904).
13use latexml_package::prelude::*;
14
15const RMAA: crate::shipped_class::RawRead = crate::shipped_class::RawRead {
16  shipped: true,
17  ..crate::shipped_class::RawRead::new("rmaa", "cls")
18};
19
20LoadDefinitions!({
21  // `\author{R.E. Mennickent\altaffilmark{1} and Z. Ko{\l}aczkowski\altaffilmark{1,2}}`: each mark (a list, `1,2`) is
22  // the author's request for the affiliations `\altaffiltext{n}{…}` labels, the footnotes the class prints under the
23  // title page (0903.4819, astro-ph0208329); an `\altaffiltext` no author marks is the author block's (1301.7157).
24  aas_support_sty::altaffil_markup()?;
25  // :491-494 `\affil{…}` (every copy, v1.9 to v3.26): inside an `\author`, the affiliation the title page sets on a
26  // line of its own under the names (`\\[1.5ex]…\\[1.5ex]`), which read raw stayed in the name ("S. Kimeswenger
27  // Institut für Astrophysik, …", astro-ph0105072): the affiliation of the authors before it.
28  DefMacro!("\\affil{}", "\\lx@add@affiliation[annotate=new]{#1}");
29  // :450 `\resumen{…}` and :454 `\abstract{…}` store the abstracts in Spanish and in English, which the title page
30  // prints in that order, under "Resumen" and "Abstract" (:974-980), whichever the paper gives first (1301.7157 gives
31  // `\resumen` first, 0903.4819 `\abstract`): two elements, handed on in the title page's order where the authors are
32  // queued and the frontmatter not yet made — at `\maketitle`, else at the first section (ahead of the kernel's
33  // frontmatter fallback on `\@startsection@hook`), else at the end of the document (aguplus_cls.rs's hand-off).
34  RawTeX!(
35    r"\let\lx@rmaa@resumen\@empty\let\lx@rmaa@abstract\@empty
36\def\lx@rmaa@hand@abstracts{\ifx\lx@rmaa@resumen\@empty\else
37    \expandafter\lx@rmaa@add@abstract\expandafter{\lx@rmaa@resumen}{Resumen}\global\let\lx@rmaa@resumen\@empty\fi
38  \ifx\lx@rmaa@abstract\@empty\else
39    \expandafter\lx@rmaa@add@abstract\expandafter{\lx@rmaa@abstract}{Abstract}\global\let\lx@rmaa@abstract\@empty\fi}
40\begingroup\toks0=\expandafter{\@startsection@hook}\xdef\@startsection@hook{\noexpand\lx@rmaa@hand@abstracts\the\toks0}\endgroup
41\AtEndDocument{\lx@rmaa@hand@abstracts}"
42  );
43  AddToMacro!("\\lx@maketitle@begin", "\\lx@rmaa@hand@abstracts");
44  DefMacro!("\\resumen{}", "\\gdef\\lx@rmaa@resumen{#1}");
45  DefMacro!("\\abstract{}", "\\gdef\\lx@rmaa@abstract{#1}");
46  DefMacro!(
47    "\\lx@rmaa@add@abstract{}{}",
48    "\\lx@add@frontmatter{ltx:abstract}[name={#2}]{#1}"
49  );
50  // :509-524 `\addkeyword{…}` adds one keyword to those before it, `\@kwdsep` (" --- ") between them; :543 `\keywords{…}`
51  // gives them all. The title page prints them after "Key Words:" (:513 `\@keywordheader`).
52  RawTeX!(r"\gdef\lx@rmaa@keywords{}");
53  DefMacro!(
54    "\\addkeyword{}",
55    "\\ifx\\lx@rmaa@keywords\\@empty\\gdef\\lx@rmaa@keywords{#1}\\else\\g@addto@macro\\lx@rmaa@keywords{ --- #1}\\fi\\lx@rmaa@add@keywords"
56  );
57  DefMacro!(
58    "\\keywords{}",
59    "\\gdef\\lx@rmaa@keywords{#1}\\lx@rmaa@add@keywords"
60  );
61  DefMacro!(
62    "\\lx@rmaa@add@keywords",
63    "\\expandafter\\lx@rmaa@add@keywords@list\\expandafter{\\lx@rmaa@keywords}"
64  );
65  DefMacro!(
66    "\\lx@rmaa@add@keywords@list{}",
67    "\\lx@add@keywords[name={Key Words:}]{#1}"
68  );
69  // :550 `\shorttitle{…}` (`\@shorttitle`, the running head, :421): the toc title, as OmniBus's (Perl reads rmaa so and
70  // keeps it, 0903.4819; 9 of the 11 rmaa papers give one). Only until the title page is made: the class's own
71  // `abstracts` environment (:1366) re-sets the running head with `\shorttitle{Abstracts}`, no toc title of the paper's.
72  DefMacro!("\\shorttitle{}", "\\lx@add@toctitle{#1}");
73  AddToMacro!("\\lx@maketitle@begin", "\\global\\let\\shorttitle\\@gobble");
74  crate::shipped_class::read_raw(&RMAA)?;
75});
