5use crate::discard_env::discard_env_body;

7#[rustfmt::skip]
8LoadDefinitions!({
9  // The real package allocates the registers and macros a document uses
10  // outside its pictures (pb-diagram.sty:45 `\newskip\dgARROWLENGTH`,
11  // `\dgARROWPARTS`, `\dgMAXSQUARE`, …; pb-manual.tex:374 `\divide
12  // \dgARROWLENGTH by2`). The stub used to refuse the raw load, so every such
13  // use was "undefined" + "expected a Variable" while Perl (no binding on the
14  // default path) raw-loads it cleanly. Batch 56bf: raw-load, then keep the
15  // ar5iv override for the picture environment below.
16  InputDefinitions!("pb-diagram", noltxml => true, extension => Some(Cow::Borrowed("sty")));
17  // Perl ar5iv-bindings/pb-diagram.sty.ltxml L22-41: \begin{diagram} emits
18  // <ltx:ERROR> and swallows the body via discard_env_body (the commutative
19  // diagram is drawn with `\put`-arithmetic pictures; an SVG rendering is
20  // future work).
21  DefConstructor!(
22    T_CS!("\\begin{diagram}"), None,
23    "<ltx:ERROR>{diagram}</ltx:ERROR>",
24    bounded => true,
25    mode    => "text",
26    locked  => true,
27    before_digest => { discard_env_body("diagram", "pb-diagram.sty.ltxml")?; }
28  );
29  DefMacro!("\\enddiagram", "\\relax");
30});
