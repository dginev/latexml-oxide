1//! bibentry.sty — bibliography entries in the text (Patrick W. Daly, 2007/10/30 v1.5).
2//!
3//! `\nobibliography{<db>}` (or `\nobibliography*` before a later `\bibliography`) reads the `.bbl` in a store mode:
4//! each `\bibitem` keeps its entry's text as `\BR@r@<key>` and the list prints nothing (bibentry.sty:44-62,
5//! `\BR@nobib`'s `{thebibliography}` and `\BR@c@bibitem#1 #2 \par`); `\bibentry{<key>}` cites the key and prints that
6//! text without its final period (:63-66, `\BR@nodot`). Raw, the store mode cannot run: its `\renewenvironment
7//! {thebibliography}` meets the kernel's locked bibliography (sect11.rs), so `\nobibliography*` printed the whole list
8//! and each `\bibentry` printed nothing ("Paper: ." for pdflatex's "Paper: A. Able. First paper. 2001."). Perl has no
9//! binding and leaves both commands undefined (KNOWN_PERL_ERRORS #1059).
10//!
11//! The binding reads the `<jobname>.bbl` itself, as `\bibliography`'s `\@input@{\jobname.bbl}` would, and keeps each
12//! entry's tokens up to the blank line that ends it. `\nobibliography{<db>}` without the star still builds the list
13//! where it stands: in TeX its `\bibitem`s write the labels that resolve every `\cite`, and here the list is what the
14//! citations resolve to (OXIDIZED_DESIGN_DIVERGENCES #831). Witnesses 2210.16098 (AAAI, 48 `\bibentry` lines after
15//! `\nobibliography{aaai23}`), 2505.07606 (`\nobibliography*` then `\bibliography`, 29 entries). Guards
16//! `perfect_kernel_batch78k::bibentry_*`.
17use latexml_package::prelude::*;
18
19/// The entries of a `.bbl`'s token list, as `(key, text)`: each `\bibitem[<label>]{<key>}` and the tokens after it up
20/// to the paragraph end (`\par`, a blank line), the next `\bibitem` or the list's `\end`, at brace depth 0; the text
21/// trimmed of spaces and of one final period (bibentry.sty:63 `\BR@nodot`).
22fn bbl_entries(tokens: &[Token]) -> Vec<(String, Vec<Token>)> {
23  let bibitem = T_CS!("\\bibitem");
24  let ends = [
25    T_CS!("\\par"),
26    bibitem,
27    T_CS!("\\end"),
28    T_CS!("\\endthebibliography"),
29  ];
30  let is_space = |t: &Token| t.get_catcode() == Catcode::SPACE;
31  let mut entries = Vec::new();
32  let mut i = 0;
33  while i < tokens.len() {
34    if tokens[i] != bibitem {
35      i += 1;
36      continue;
37    }
38    i += 1;
39    while i < tokens.len() && is_space(&tokens[i]) {
40      i += 1;
41    }
42    // the optional label, `[…]` at brace depth 0
43    if i < tokens.len() && tokens[i] == T_OTHER!("[") {
44      let mut depth = 0i32;
45      while i < tokens.len() {
46        match tokens[i].get_catcode() {
47          Catcode::BEGIN => depth += 1,
48          Catcode::END => depth -= 1,
49          _ if depth == 0 && tokens[i] == T_OTHER!("]") => {
50            i += 1;
51            break;
52          },
53          _ => {},
54        }
55        i += 1;
56      }
57    }
58    while i < tokens.len() && is_space(&tokens[i]) {
59      i += 1;
60    }
61    // the key, a group or a single token
62    let mut key = Vec::new();
63    if i < tokens.len() && tokens[i].get_catcode() == Catcode::BEGIN {
64      let mut depth = 1i32;
65      i += 1;
66      while i < tokens.len() {
67        match tokens[i].get_catcode() {
68          Catcode::BEGIN => depth += 1,
69          Catcode::END => depth -= 1,
70          _ => {},
71        }
72        if depth == 0 {
73          i += 1;
74          break;
75        }
76        key.push(tokens[i]);
77        i += 1;
78      }
79    } else if i < tokens.len() {
80      key.push(tokens[i]);
81      i += 1;
82    }
83    let key = Tokens::new(key).to_string().trim().to_string();
84    // the entry's text
85    let mut text = Vec::new();
86    let mut depth = 0i32;
87    while i < tokens.len() {
88      let t = tokens[i];
89      if depth == 0 && ends.contains(&t) {
90        break;
91      }
92      match t.get_catcode() {
93        Catcode::BEGIN => depth += 1,
94        Catcode::END => depth -= 1,
95        _ => {},
96      }
97      text.push(t);
98      i += 1;
99    }
100    while text.first().is_some_and(is_space) {
101      text.remove(0);
102    }
103    while text.last().is_some_and(is_space) {
104      text.pop();
105    }
106    if text.last() == Some(&T_OTHER!(".")) {
107      text.pop();
108    }
109    if !key.is_empty() {
110      entries.push((key, text));
111    }
112  }
113  entries
114}

116LoadDefinitions!({
117  // bibentry.sty:42-82: the commands, the store mode a Rust primitive (`\lx@BR@store`, below). The store mode's own
118  // `\BR@b@bibitem`, `\BR@c@bibitem` and `\BR@nodot` (:58-63) are defined as there, for code that redefines them:
119  // tufte-common.def:876-882 `\renewcommand\BR@b@bibitem` before its own `\nobibliography*`. The star records that a
120  // `\bibliography` follows; bibentry's `\def\bibliography` wrapper that resets it (:74-75) is refused by the kernel's
121  // lock, so the `\AtEndDocument` warning (:76-79) asks the kernel's count of `\bibliography` calls instead
122  // (`lx_bibliography_calls`, sect11.rs `\lx@ifusebbl`) whether one came after the last `\nobibliography`.
123  RawTeX!(
124    r"\newif\ifBR@star
125\newcommand\nobibliography{\@ifstar{\BR@nobib{}}{\BR@nobib}}
126\newcommand\BR@nobib[1]{\ifx\relax#1\relax\global\BR@startrue\else\global\BR@starfalse\fi
127\lx@BR@store\ifBR@star\else\bibliography{#1}\fi\lx@BR@mark}
128\newcommand\bibentry[1]{\nocite{#1}{\frenchspacing
129\hyper@natanchorstart{#1\@extra@b@citeb}\lx@BR@entry{#1\@extra@b@citeb}\hyper@natanchorend}}
130\newcommand\BR@b@bibitem[2][]{\ifx\relax#1\relax\BR@bibitem{#2}\else \BR@bibitem[#1]{#2}\fi \BR@c@bibitem{#2}}
131\def\BR@c@bibitem#1 #2 \par{{\let\protect\@unexpandable@protect
132\expandafter \gdef\csname BR@r@#1\@extra@b@citeb\endcsname{\BR@nodot#2\relax.\relax\relax}}}
133\def\BR@nodot#1.\relax#2\relax{#1}
134\AtEndDocument{\ifBR@star\lx@BR@ifnobibliography{\PackageWarningNoLine{bibentry}
135{You have used \string\nobibliography* \MessageBreak
136without a following \string\bibliography.\MessageBreak
137You may not be able to run BibTeX}}\fi}
138\providecommand\@extra@b@citeb{}
139\providecommand\hyper@natanchorstart[1]{}
140\providecommand\hyper@natanchorend{}
141\AtBeginDocument{\providecommand*{\urlprefix}{URL }\providecommand*{\url}[1]{\texttt{#1}}}
142\AtBeginDocument{\@ifpackageloaded{url}{\providecommand{\doi}{doi: \begingroup \urlstyle{rm}\Url}}
143{\providecommand{\doi}[1]{doi: #1}}}"
144  );
145
146  // The store mode: the `<jobname>.bbl` `\bibliography` would read (`\lx@ifusebbl`'s bbl route), each entry kept
147  // globally as `\BR@r@<key>` (bibentry.sty:60-62 `\gdef\csname BR@r@#1\endcsname{…}`). Without a `.bbl` (a paper
148  // shipping only its `.bib`) nothing is stored, and `\bibentry` reports each key it cannot print.
149  DefPrimitive!("\\lx@BR@store", sub[_args] {
150    let jobname = Expand!(T_CS!("\\jobname")).to_string();
151    let Some(path) = find_file(
152      &jobname,
153      Some(FindFileOptions {
154        ext_type: Some(Cow::Borrowed("bbl")),
155        ..Default::default()
156      }),
157    ) else {
158      Info!("expected", "bbl", s!("bibentry: no {jobname}.bbl, so no entry is stored for \\bibentry"));
159      return Ok(Vec::new());
160    };
161    let text = vfs_read(&path).or_else(|| {
162      std::fs::read(&path)
163        .ok()
164        .map(|b| String::from_utf8_lossy(&b).into_owned())
165    }).unwrap_or_default();
166    let tokens = mouth::tokenize(TeXString::assembled(text));
167    for (key, entry) in bbl_entries(tokens.unlist_ref()) {
168      def_macro(T_CS!(s!("\\BR@r@{key}")), None, Tokens::new(entry), Some(ExpandableOptions { scope: Some(Scope::Global), ..Default::default() }))?;
169    }
170  });
171
172  // The count of `\bibliography` calls when `\nobibliography` ran, and whether none came since.
173  DefPrimitive!("\\lx@BR@mark", sub[_args] {
174    assign_value("BR_bibliography_calls", Stored::Int(lookup_int("lx_bibliography_calls")), Some(Scope::Global));
175  });
176  DefMacro!("\\lx@BR@ifnobibliography{}", sub[(warning)] {
177    Ok(if lookup_int("lx_bibliography_calls") == lookup_int("BR_bibliography_calls") { warning } else { Tokens!() })
178  });
179
180  // bibentry.sty:66 `\@nameuse{BR@r@#1\@extra@b@citeb}`: the stored text; a key with none prints nothing, as there,
181  // and is reported (pdflatex reports the undefined citation of its `\nocite`).
182  DefMacro!("\\lx@BR@entry{}", sub[(key)] {
183    let key = Expand!(key).to_string();
184    let cs = T_CS!(s!("\\BR@r@{key}"));
185    if has_meaning(&cs) {
186      Ok(Tokens!(cs))
187    } else {
188      Warn!("missing", "bibentry", s!("bibentry: no stored entry for key '{key}' (no \\nobibliography or no .bbl entry)"));
189      Ok(Tokens!())
190    }
191  });
192});
