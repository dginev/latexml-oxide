1//! jpconf.cls (Journal of Physics: Conference Series, IOP's conference class, 2004/05/04 v1.0): the papers ship the
2//! class (it is not in TeX Live), and it is read raw (OXIDIZED_DESIGN_DIVERGENCES #444). Its address and e-mail blocks
3//! (:104-110: `\eads`, `\ead`, `\mailto`, `\address`, each an `{indented}` list item where it stands) are iopart's
4//! (iopart.cls:240-248, 363-364), so after the read they are the frontmatter iopart_support makes of them
5//! (`iopart_support_sty::iop_address_and_email`): each `\address` an affiliation of the authors before it, each `\ead`
6//! an e-mail placed with its author, where read raw they were bullet lists after the abstract (0705.0202, 1011.0523).
7//! Perl has no binding: OmniBus's `\address` and `\email`. Guard
8//! `perfect_kernel_batch78m::jpconf_address_and_email_are_frontmatter`.
9use latexml_package::{
10  prelude::*,
11  shipped_class::{RawRead, read_raw},
12};
13
14const JPCONF: RawRead = RawRead {
15  shipped: true,
16  ..RawRead::new("jpconf", "cls")
17};
18
19/// This binding's frontmatter conventions (the observer's, OXIDIZED_DESIGN_DIVERGENCES #837), declared at its load
20/// (`declare_frontmatter_conventions`, user ruling 2026-10-28: each binding declares its own):
21/// jpconf: its notes print the body's arabic `\thefootnote` as a superscript (jpconf.cls:139, 153 `\@thefnmark`,
22/// `\def\@fnsymbol#1{…^{\thefootnote}…}`; no `\maketitle` of its own changes `\thefootnote`): 1105.2659's
23/// `\footnote[3]{Speaker}` shows "3" — the kernel's `\@fnsymbol\c@footnote` there recurses through `\thefootnote`
24/// (review revfms2a2_7 finding 4); its `\address`/`\ead` are iopart's
25pub const FRONTMATTER_CONVENTIONS: FmConventions = FmConventions {
26  name: "jpconf",
27  binding: "jpconf.cls",
28  email_owner: FmEmailOwner::SpelledElseOrder,
29  call_is_group: true,
30  title_marks: FmTitleMarks::Body,
31  cite: "jpconf.cls:104-110 (iopart's `\\address`/`\\ead`), :139, 153 (its notes' marks); OD #841 (0705.0202, \
32         1011.0523)",
33  ..FM_KERNEL_CONVENTIONS
34};
35
36LoadDefinitions!({
37  // (this binding's frontmatter conventions, first: OXIDIZED_DESIGN_DIVERGENCES #837)
38  declare_frontmatter_conventions(&FRONTMATTER_CONVENTIONS);
39  if read_raw(&JPCONF)? {
40    iopart_support_sty::iop_address_and_email()?;
41  }
42});
