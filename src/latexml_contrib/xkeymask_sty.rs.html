5  // xkeymask.sty:33 reads its options with `\ProcessKeyvalOptions`, which marks them processed: `\@curroptions` stays as it was.
6  key_options_processed()?;
7  // xkeymask.sty (Ramkumar Ramachandra, 2022-2023) — an xkeyval extension
8  // that "masks" keys so `\setkeys` ignores them. No Perl LaTeXML binding
9  // exists; the raw package works under real TeX but relies on `\XKV@resa`
10  // surviving across `\KV@@sp@def` calls — keyval.tex's `\KV@@sp@def`
11  // clobbers `\XKV@resa` via `\futurelet` (keyval.tex L41), so xkeymask's
12  // accumulator ends up as the self-referential `macro:->\XKV@resa ,…`
13  // (verified with pdflatex `\meaning`). Real TeX shrugs it off because the
14  // walk only ever expands it one level at a time; our (and Perl LaTeXML's)
15  // recursion guard reports `Error:recursion:\XKV@resa` at every masked
16  // `\setkeys`. The binding provides the same user surface on clobber-safe
17  // `\XKM@…` scratch names instead — observable behavior matches pdflatex
18  // (mask honored, unmasked keys applied, no diagnostics).
19  //
20  // Witness: doc/latex/xkeymask/xkeymask.tex (perfect-kernel sweep 12).
21  RequirePackage!("xkeyval");
22
23  RawTeX!(r"\ProvidesPackage{xkeymask}[v1.0 An extension of xkeyval with a mask]");
24
25  // Real xkeymask takes its `prefix` option through kvoptions
26  // (`\DeclareStringOption{prefix}` + `\ProcessKeyvalOptions*`); the mask
27  // machinery is only activated when the option is present. Parse the
28  // key=value option with a catch-all handler (geometry_sty.rs pattern).
29  RawTeX!(
30    r"\def\XKM@prefix{}
31\def\XKM@grab#1=#2\@nil{\def\XKM@key{#1}\XKM@grabv#2\@nil}
32\def\XKM@grabv#1=#2\@nil{\def\XKM@val{#1}}
33\DeclareOption*{%
34  \expandafter\XKM@grab\CurrentOption==\@nil
35  \def\XKM@tempc{prefix}%
36  \ifx\XKM@key\XKM@tempc\global\let\XKM@prefix\XKM@val\fi
37}
38\ProcessOptions\relax"
39  );
40
41  // The mask machinery — xkeymask.sty L36-113, on `\XKM@…` scratch names.
42  // `\appendmask`/`\ifinmask` keep the real package's xkeyval front-end
43  // (`\XKV@testopta{\XKV@testoptc …}`), including `\XKM@ifinmask`'s
44  // per-family×key branch execution. `\XKM@setkeys` filters masked keys out
45  // of the list and hands the survivors to `\XKV@s@tkeys`, exactly like the
46  // original; only the accumulator names differ (see module comment).
47  RawTeX!(
48    r"\ifx\XKM@prefix\@empty\else
49  \def\XKM@mask{}
50  \long\def\XKM@appendmask[#1]{%
51    \XKV@for@o\XKV@fams\XKV@tfam{%
52      \xdef\XKM@hdr{\XKV@prefix\XKV@tfam @}%
53      \XKV@for@n{#1}\XKM@tempa{%
54        \expandafter\KV@@sp@def\expandafter\XKM@tempa\expandafter{\XKM@tempa}%
55        \ifx\XKM@mask\@empty\xdef\XKM@mask{\XKM@hdr\XKM@tempa}%
56        \else\xdef\XKM@mask{\XKM@mask,\XKM@hdr\XKM@tempa}\fi
57      }%
58    }%
59  }
60  \def\appendmask{\XKV@testopta{\XKV@testoptc\XKM@appendmask}}
61  \long\def\XKM@ifinmask[#1]#2#3{%
62    \XKV@checksanitizea{#1}\XKM@tempb
63    \XKV@for@o\XKV@fams\XKV@tfam{%
64      \XKV@for@o\XKM@tempb\XKM@tempa{%
65        \xdef\XKM@hdr{\XKV@prefix\XKV@tfam @}%
66        \xdef\XKM@tempd{\XKV@prefix\XKV@tfam @\XKM@tempa}%
67        \@expandtwoargs\in@\XKM@tempd\XKM@mask
68        \ifin@#2\else#3\fi
69      }%
70    }%
71  }
72  \def\ifinmask{\XKV@testopta{\XKV@testoptc\XKM@ifinmask}}
73  \def\clearmask{\global\let\XKM@mask\@empty}
74  \long\def\XKM@setkeys[#1]#2{%
75    \XKV@checksanitizea{#2}\XKM@resb
76    \let\XKM@filtered\@empty
77    \XKV@for@o\XKV@fams\XKV@tfam{%
78      \XKV@for@o\XKM@resb\XKM@tempb{%
79        \expandafter\XKV@g@tkeyname\XKM@tempb=\@nil\XKM@tempc
80        \expandafter\KV@@sp@def\expandafter\XKM@tempc\expandafter{\XKM@tempc}%
81        \xdef\XKM@tempd{\XKV@prefix\XKV@tfam @\XKM@tempc}%
82        \@expandtwoargs\in@\XKM@tempd\XKM@mask
83        \ifin@\else\XKV@addtolist@o\XKM@filtered\XKM@tempb\fi
84      }%
85    }%
86    \ifnum\XKV@depth=\z@\let\XKV@rm\@empty\fi
87    \expandafter\XKV@s@tkeys\expandafter{\XKM@filtered}{#1}%
88    \let\CurrentOption\@empty
89  }
90  \long\def\XKM@setkeys@dispatch{%
91    \xdef\XKM@tempa{\XKM@prefix @}%
92    \ifx\XKV@prefix\XKM@tempa
93      \expandafter\XKM@setkeys
94    \else
95      \expandafter\XKV@setkeys
96    \fi
97  }
98  \def\setkeys{\XKV@testopta{\XKV@testoptc\XKM@setkeys@dispatch}}
99\fi"
100  );
101});
