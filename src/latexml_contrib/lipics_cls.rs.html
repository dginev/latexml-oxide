406/// Whether `tokens` holds nothing but spaces (the class's `\if!#n!` test).

410    .iter()
411    .all(|t| t.get_catcode() == Catcode::SPACE)

414/// `tokens` without its leading and trailing spaces.
415fn trimmed(tokens: &[Token]) -> Vec<Token> {
416  let start = tokens
417    .iter()
418    .position(|t| t.get_catcode() != Catcode::SPACE)
419    .unwrap_or(tokens.len());
420  let end = tokens
421    .iter()
422    .rposition(|t| t.get_catcode() != Catcode::SPACE)
423    .map_or(start, |i| i + 1);
424  tokens[start..end].to_vec()
425}

427/// The parts of `tokens` between its top-level `\and`s: each written, as before, and each held in a macro of the
428/// document's, which the separator query reads as the kernel's (`and_by_meaning`, Option C P2-5).
429fn split_on_and(tokens: &Tokens) -> Result<Vec<Vec<Token>>> {
430  let tokens = tokens.unlist_ref();
431  let cuts = and_runs(tokens, &[T_CS!("\\and")])?.top_level(tokens);
432  let mut parts = Vec::new();
433  let mut from = 0;
434  for (start, end) in cuts {
435    if start < from {
436      continue;
437    }
438    parts.push(tokens[from..start].to_vec());
439    from = end;
440  }
441  parts.push(tokens[from..].to_vec());
442  Ok(parts)
443}

445/// An ORCID iD (`0000-0002-1825-0097`, the last character a digit or `X`) or its orcid.org address, with or without
446/// the scheme or `www.`; an argument holding a macro is taken as one (`\lx@add@orcid` keeps it as written).
447fn is_orcid(tokens: &[Token]) -> bool {
448  if tokens.iter().any(|t| t.get_catcode() == Catcode::CS) {
449    return true;
450  }
451  let text = Tokens::new(tokens.to_vec())
452    .to_string()
453    .trim()
454    .to_ascii_lowercase();
455  let mut rest = text.as_str();
456  for scheme in ["https://", "http://"] {
457    if let Some(r) = rest.strip_prefix(scheme) {
458      rest = r;
459      break;
460    }
461  }
462  rest = rest.strip_prefix("www.").unwrap_or(rest);
463  if rest.starts_with("orcid.org/") {
464    return true;
465  }
466  let groups: Vec<&str> = rest.split('-').collect();
467  groups.len() == 4
468    && groups.iter().enumerate().all(|(i, group)| {
469      group.len() == 4
470        && group
471          .chars()
472          .enumerate()
473          .all(|(j, c)| c.is_ascii_digit() || (i == 3 && j == 3 && c == 'x'))
474    })
475}

477/// The address an ORCID slot holds when it is an email address (`name@host`, or a `mailto:` link to one): one `@`
478/// with text before it, a dotted host after it, no white space and no URL scheme.
479fn email_address(tokens: &[Token]) -> Option<String> {
480  if tokens.iter().any(|t| t.get_catcode() == Catcode::CS) {
481    return None;
482  }
483  let text = Tokens::new(tokens.to_vec()).to_string();
484  let text = text.trim();
485  let address = text.strip_prefix("mailto:").unwrap_or(text);
486  let (local, host) = address.split_once('@')?;
487  (!local.is_empty()
488    && host.contains('.')
489    && !host.contains('@')
490    && !address.contains(char::is_whitespace)
491    && !local.contains("://"))
492  .then(|| address.to_string())
493}

503/// The five-part `\author` of lipics-v2019/v2021 (`\author` in `load_definitions`): the creator, its affiliations and
504/// homepage, email, ORCID and funding note, through the frontmatter API.
505fn lipics_author(
506  name: Tokens,
507  affiliations: Tokens,
508  email: Tokens,
509  orcid: Tokens,
510  funding: Tokens,
511) -> Result<Tokens> {
512  let anonymous = lookup_bool("lx_lipics_anonymous");
513  let mut out = Vec::new();
514  if anonymous {
515    out.extend(invoke("\\lx@add@creator", Explode!("Anonymous author")));
516    if !is_blank(&affiliations) {
517      out.extend(invoke(
518        "\\lx@add@affiliation",
519        Explode!("Anonymous affiliation"),
520      ));
521    }
522    return Ok(Tokens::new(out));
523  }
524  out.extend(invoke("\\lx@add@creator", name.unlist_ref().to_vec()));
525  for part in split_on_and(&affiliations)? {
526    let part = trimmed(&part);
527    if part.is_empty() {
528      continue;
529    }
530    // `\and \url{homepage}`: the class's homepage, linked from the author's name (:796)
531    if part[0] == T_CS!("\\url") {
532      let rest = trimmed(&part[1..]);
533      let url = match (rest.first(), rest.last()) {
534        (Some(first), Some(last))
535          if first.get_catcode() == Catcode::BEGIN
536            && last.get_catcode() == Catcode::END
537            && rest.len() >= 2 =>
538        {
539          rest[1..rest.len() - 1].to_vec()
540        },
541        _ => rest,
542      };
543      out.extend(invoke("\\lx@add@url", url));
544    } else {
545      out.extend(invoke("\\lx@add@affiliation", part));
546    }
547  }
548  let email = (!is_blank(&email)).then(|| trimmed(email.unlist_ref()));
549  if let Some(email) = &email {
550    out.extend(invoke("\\lx@add@email", email.clone()));
551  }
552  // the class links the slot as written (lipics-v2021.cls:797 `\if!#4!\else{\,\href{#4}{\orcidsymbol}}\fi`); what the
553  // author put there is read by its type: an ORCID iD, bare or its orcid.org address, is the author's ORCID
554  // (`\lx@add@orcid`); an email address (all five authors of 2607.06446, `{}{gerhard@uni-bremen.de}`) is the author's
555  // email (`\lx@add@email`, once when the email slot holds it too), which as a link would be a relative one to nowhere;
556  // any other address is the class's link to it
557  if !is_blank(&orcid) {
558    let orcid = trimmed(orcid.unlist_ref());
559    if is_orcid(&orcid) {
560      out.extend(invoke("\\lx@add@orcid", orcid));
561    } else if let Some(address) = email_address(&orcid) {
562      let same = email
563        .as_ref()
564        .is_some_and(|email| Tokens::new(email.clone()).to_string().trim() == address);
565      if !same {
566        // as written, or the address a `mailto:` link names
567        let written = Tokens::new(orcid.clone()).to_string().trim() == address;
568        let address = if written {
569          orcid
570        } else {
571          mouth::tokenize_internal(TeXString::assembled(address)).unlist()
572        };
573        out.extend(invoke("\\lx@add@email", address));
574      }
575    } else {
576      let mut link = vec![T_CS!("\\href"), T_BEGIN!()];
577      link.extend(orcid);
578      link.extend([T_END!(), T_BEGIN!(), T_CS!("\\lx@orcidlogo"), T_END!()]);
579      out.extend(
580        mouth::tokenize_internal(TeXString::assembled(
581          r"\lx@annotate@frontmatter{ltx:creator}{ltx:contact}[role=orcid]".to_string(),
582        ))
583        .unlist(),
584      );
585      out.extend(invoke_group(link));
586    }
587  }
588  if !is_blank(&funding) {
589    let mut note = vec![
590      T_CS!("\\textit"),
591      T_BEGIN!(),
592      T_BEGIN!(),
593      T_CS!("\\let"),
594      T_CS!("\\footnote"),
595      T_CS!("\\@gobble"),
596    ];
597    note.extend(name.unlist_ref().iter().copied());
598    note.extend([T_END!(), T_END!(), T_OTHER!(":"), T_CS!("\\space")]);
599    note.extend(funding.unlist_ref().iter().copied());
600    let mut tokens = vec![T_CS!("\\lx@add@frontmatter"), T_BEGIN!()];
601    tokens.extend(Explode!("ltx:note"));
602    tokens.push(T_END!());
603    tokens.push(T_OTHER!("["));
604    tokens.extend(Explode!("role=funding"));
605    tokens.push(T_OTHER!("]"));
606    tokens.extend(invoke_group(note));
607    out.extend(tokens);
608  }
609  Ok(Tokens::new(out))
610}
