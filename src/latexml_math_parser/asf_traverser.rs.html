1//! ASF-based math parser traversal.
2//!
3//! ## What this does
4//!
5//! Marpa builds an ASF (Abstract Syntax Forest) — a DAG where shared
6//! sub-parses collapse to a single **glade**. `MathTraverser` is the
7//! per-glade callback: it produces the set of alternative XM trees
8//! that can terminate at that glade's input position. Outputs are
9//! memoized by the marpa driver, so a glade reached by multiple
10//! parents is evaluated once.
11//!
12//! ## Glade categories
13//!
14//! Marpa classifies glades for us; this callback only routes:
15//!
16//! 1. **Token glade** (`glade.is_token()`): a ByteScanner byte, read as its own terminal or as a byte
17//!    class holding it (`lex_char`, `digit`); its token value is the byte + 1. Emits one
18//!    `XM::Lexeme("x")` for the byte.
19//! 2. **Lexeme rule** (`builder.is_token(rule_id)`): RELOP / ADDOP / NUMBER and friends — rule
20//!    whose body is a byte sequence that rolls up into a single lexeme name, then is
21//!    `.specialize`d.
22//! 3. **Discard rule** (`builder.is_discard(rule_id)`): whitespace. Emits one `None`.
23//! 4. **No-action rule**: either a grammar passthrough (`is_rule` → forward the first child's
24//!    alternatives) or an internal byte- passthrough scaffolding piece (concatenate child bytes
25//!    into a Lexeme).
26//! 5. **Rule with action**: cartesian-product child alternatives across RHS positions, dispatch
27//!    `Actions::action_on` per combo, accumulate the surviving results.
28//!
29//! ## Pruning
30//!
31//! `Actions::action_on` returns `Ok(Some(_))` (good parse),
32//! `Ok(None)` (this position contributes nothing — passed up
33//! unchanged), or `Err(_)` (semantic pragma rejection). An `Err`
34//! drops that combo from the glade's alternatives. If every alt at
35//! a glade is rejected, the glade's Vec becomes empty and the
36//! parent's cartesian product yields zero combos — pruning cascades.
37

53thread_local! {
54  /// Every byte as the char of the same number (U+0000-U+00FF): `byte_lexeme`'s cache.
55  static BYTE_LEXEMES: RefCell<Vec<Rc<str>>> = RefCell::new(
56    (0u8..=255).map(|b| Rc::<str>::from(char::from(b).to_string())).collect()
57  );
58}
59
60/// A byte-token glade's lexeme. Below a lexeme rule the rollup carries each input byte as the char
61/// of the same number, so a multi-byte character survives it whole, as the tree builder's
62/// `rollup_token_rec` keeps the token's raw bytes; `token_text` decodes them once the lexeme rule
63/// completes. (Bytes above 127 were dropped: ASF named the tree route's `ATOM:Γ:33` `ATOM::33`,
64/// and the two routes' readings of one formula never compared equal.)
65#[inline]
66fn byte_lexeme(byte: u8) -> Rc<str> {
67  BYTE_LEXEMES.with(|cache| cache.borrow()[byte as usize].clone())
68}
69
70/// A lexeme rule's text: its raw bytes (`byte_lexeme`) decoded as UTF-8, as the tree route decodes a
71/// token's bytes (`Actions::translate_node`, semantics.rs).
72fn token_text(raw: Rc<str>) -> Rc<str> {
73  if raw.is_ascii() {
74    return raw;
75  }
76  let bytes: Vec<u8> = raw
77    .chars()
78    .map(|c| u8::try_from(u32::from(c)).unwrap_or(b'?'))
79    .collect();
80  String::from_utf8(bytes).map_or_else(|_| Rc::from("malformed-utf8"), Rc::from)
81}

83/// Alternatives at a single glade. Wrapped in `Rc` so the marpa ASF
84/// driver's per-glade `cache.insert(_, output.clone())` and `cache
85/// .get(&id).clone()` become refcount bumps instead of deep tree
86/// copies.
87pub type GladeAlts = Rc<Vec<Option<XM>>>;

89/// Per-glade traversal callback. Holds the math parser's actions,
90/// pragmas, builder, document — the same dependencies the legacy
91/// `translate_node` path uses for bottom-up action dispatch.
92pub struct MathTraverser<'a> {
93  pub actions:              &'a Actions,
94  pub pragmas:              &'a [ValidationPragmatics],
95  pub builder:              &'a TreeBuilder,
96  pub nodes:                &'a [Node],
97  pub document:             &'a mut Document,
98  /// `action_on(...) -> Err(_)` count. Surfaces in `PARSE_AUDIT`
99  /// diagnostics analogous to the legacy `pruned_trees` counter.
100  pub pruned_count:         usize,
101  /// The work bound of the tree iterator's second chance (`MathParser::parse_marpa`); `None` on
102  /// the ordinary ASF route, whose bocages are small.
103  pub budget:               Option<AsfBudget>,
104  /// Does the formula hold an expectation lexeme that the grammar also reads as a letter
105  /// (`EXPECTATION:𝔼.letter`, the letter retry's, `expectation_letter`, M3)? Each glade then keeps its
106  /// alternatives with the fewest letter readings (`keep_fewest_letter_readings`).
107  pub letter_readings:      bool,
108  /// Does the formula hold an integral's letter `d` (`XDIFFUNK`/`XDIFFID`)? Each glade then keeps its alternatives
109  /// with the fewest letter `d`s before a variable their differential takes (`keep_fewest_letter_differentials`).
110  pub differential_letters: bool,
111}

113/// A bound on one traversal's work: the actions it attempts (pruned ones too), the alternatives its
114/// glades produce, and a deadline, a safety net only. What a traversal costs follows the readings
115/// that survive at each glade, not the bocage's size — bar pairs multiply them (`|a||b|…|k|`) — so
116/// the counts are what bound it. A traversal cut short anywhere is `exhausted`: its readings are an
117/// arbitrary part of the whole, so the caller discards them — the formula has no parse, or keeps
118/// the tree iterator's sample if that was cut short at `max_unique`.
119pub struct AsfBudget {
120  pub attempts:     usize,
121  pub alternatives: usize,
122  pub deadline:     std::time::Instant,
123  pub exhausted:    bool,
124}

126impl Traverser for MathTraverser<'_> {
127  type ParseTree = GladeAlts;
128  type ParseState = ();
129
130  fn traverse_glade(
131    &mut self,
132    glade: &mut Glade,
133    children: &[Option<Self::ParseTree>],
134    _state: &mut Self::ParseState,
135  ) -> MarpaResult<Self::ParseTree> {
136    if !self.within_budget() {
137      return Ok(Rc::new(Vec::new()));
138    }
139    // Case 1: byte-token glade. Its value is the ByteScanner byte + 1, whether the byte was read as
140    // its own terminal or through a byte class holding it (marpa-asf 0.4 `inverse_string_set`,
141    // `char_range`: `lex_char`, `digit`), whose terminal is the glade's symbol. A nulling token has
142    // no value and no byte.
143    if glade.is_token() {
144      let alt = glade.token_value().map(|value| {
145        debug_assert!(
146          (1..=256).contains(&value),
147          "a byte token's value is its byte + 1, got {value}"
148        );
149        XM::Lexeme(byte_lexeme((value - 1) as u8), Meta::default())
150      });
151      return Ok(Rc::new(vec![alt]));
152    }
153
154    let mut alts: Vec<Option<XM>> = Vec::new();
155    // Symch loop — runs once per grammar-level alternative at this
156    // position. For unambiguous parses this body runs exactly once.
157    loop {
158      let rule_id = glade.rule_id();
159      let rh_len = glade.rh_length();
160      let has_action = self.actions.has_action(rule_id);
161      let is_lex_rule = self.builder.is_token(rule_id);
162
163      if self.builder.is_discard(rule_id) {
164        // Case 3: whitespace-discard rule.
165        alts.push(None);
166      } else if is_lex_rule || (!has_action && !self.builder.is_rule(rule_id)) {
167        // Cases 2 + 4b: byte-rollup. The lexeme-rule branch (2) also
168        // calls `.specialize` on the result; the bare byte-passthrough
169        // (4b) doesn't.
170        if let Some(raw) = collect_lexeme(glade, rh_len, children) {
171          if is_lex_rule {
172            let lexeme = XM::Lexeme(token_text(raw), Meta::default());
173            match lexeme.specialize(Meta::default(), self.pragmas) {
174              Ok(x) => alts.push(Some(x)),
175              Err(_) => self.pruned_count += 1,
176            }
177          } else {
178            alts.push(Some(XM::Lexeme(raw, Meta::default())));
179          }
180        } else {
181          alts.push(None);
182        }
183      } else if !has_action {
184        // Case 4a: grammar passthrough — forward the first child's
185        // alts unchanged. Mirrors legacy `args.remove(0)`.
186        let first_id = glade.rh_glade_id(0).expect("rh 0");
187        let first = children
188          .get(first_id)
189          .and_then(|o| o.as_ref())
190          .expect("child precomputed");
191        for alt in first.iter() {
192          if !self.within_budget() {
193            break;
194          }
195          self.spend();
196          alts.push(alt.clone());
197        }
198      } else {
199        // Case 5: rule with action — cartesian-product children's
200        // alts across RHS positions, run `action_on` per combo.
201        self.dispatch_action(glade, rule_id, rh_len, children, &mut alts);
202      }
203
204      if glade.next().is_none() {
205        break;
206      }
207    }
208    if self.letter_readings && alts.len() > 1 {
209      keep_fewest_letter_readings(&mut alts);
210    }
211    if self.differential_letters && alts.len() > 1 {
212      keep_fewest_letter_differentials(&mut alts);
213    }
214    Ok(Rc::new(alts))
215  }
216}

218impl MathTraverser<'_> {
219  /// Cartesian-product children, dispatch `action_on` per combo,
220  /// push surviving results into `out`. Pull this out of
221  /// `traverse_glade` so the loop body in case 5 doesn't compete
222  /// with the simpler cases for visual real estate.
223  fn dispatch_action(
224    &mut self,
225    glade: &Glade,
226    rule_id: i32,
227    rh_len: usize,
228    children: &[Option<GladeAlts>],
229    out: &mut Vec<Option<XM>>,
230  ) {
231    // Resolve children once.
232    let mut per_pos: Vec<&Vec<Option<XM>>> = Vec::with_capacity(rh_len);
233    for ix in 0..rh_len {
234      let cid = glade.rh_glade_id(ix).expect("rh position has child glade");
235      let child = children
236        .get(cid)
237        .and_then(|o| o.as_ref())
238        .expect("child precomputed");
239      per_pos.push(child.as_ref());
240    }
241    let total: usize = per_pos.iter().map(|p| p.len()).product();
242    if total == 0 {
243      return;
244    }
245    if total == 1 {
246      // Common case: every RHS position has one alternative. Build
247      // one combo without the odometer machinery.
248      let combo: Vec<Option<XM>> = per_pos.iter().map(|p| p[0].clone()).collect();
249      self.run_action(rule_id, combo, out);
250      return;
251    }
252    // General case: stream combinations via an odometer over per-
253    // position indices instead of materialising the full
254    // `Vec<Vec<Option<XM>>>` accumulator. The accumulator approach
255    // re-cloned every prefix as it grew (O(total × rh_len) extra
256    // clones on top of the inevitable per-combo clones); the
257    // odometer pays only the unavoidable `total × rh_len` clones
258    // and a single fixed-size `indices` buffer. Memory drops from
259    // O(total × rh_len) to O(rh_len).
260    let mut indices = vec![0usize; rh_len];
261    'odometer: loop {
262      if !self.within_budget() {
263        break;
264      }
265      let combo: Vec<Option<XM>> = indices
266        .iter()
267        .enumerate()
268        .map(|(pos, &ix)| per_pos[pos][ix].clone())
269        .collect();
270      self.run_action(rule_id, combo, out);
271      // Advance: increment the rightmost position; on wrap, carry
272      // left. If the leftmost wraps too, we've enumerated every
273      // combination.
274      for pos in (0..rh_len).rev() {
275        indices[pos] += 1;
276        if indices[pos] < per_pos[pos].len() {
277          continue 'odometer;
278        }
279        indices[pos] = 0;
280      }
281      break;
282    }
283  }
284
285  #[inline]
286  fn run_action(&mut self, rule_id: i32, combo: Vec<Option<XM>>, out: &mut Vec<Option<XM>>) {
287    if !self.within_budget() {
288      return;
289    }
290    self.spend_attempt();
291    let ctxt = ActionContext {
292      nodes:    self.nodes,
293      document: &mut *self.document,
294    };
295    match self.actions.action_on(rule_id, combo, self.pragmas, ctxt) {
296      Ok(opt_xm) => {
297        self.spend();
298        out.push(opt_xm)
299      },
300      Err(_) => self.pruned_count += 1,
301    }
302  }
303
304  /// Is there work left in the budget (always, without one)? The first time there is not, the
305  /// traversal is cut short: `exhausted`.
306  fn within_budget(&mut self) -> bool {
307    let Some(budget) = self.budget.as_mut() else {
308      return true;
309    };
310    if !budget.exhausted {
311      budget.exhausted = budget.attempts == 0
312        || budget.alternatives == 0
313        || std::time::Instant::now() >= budget.deadline;
314    }
315    !budget.exhausted
316  }
317
318  /// Was the traversal cut short, its readings incomplete?
319  pub fn budget_spent(&self) -> bool { self.budget.as_ref().is_some_and(|budget| budget.exhausted) }
320
321  /// Count one attempted action, pruned or not, against the budget.
322  fn spend_attempt(&mut self) {
323    if let Some(budget) = self.budget.as_mut() {
324      budget.attempts = budget.attempts.saturating_sub(1);
325    }
326  }
327
328  /// Count one produced alternative against the budget.
329  fn spend(&mut self) {
330    if let Some(budget) = self.budget.as_mut() {
331      budget.alternatives = budget.alternatives.saturating_sub(1);
332    }
333  }
334}

336/// An expectation's letter reading (`expectation_letter`, M3) yields to an operator reading of the same
337/// symbol over the same span: a glade keeps the alternatives with the fewest letter readings, so the
338/// letter twin of each expectation does not multiply through the Cartesian products above it (the
339/// root's `ExpectationLettersAreFallbacks` ranks whole trees the same way, for the tree iterator's). A hard
340/// drop, unlike the root's: it assumes no parent refuses the typed alternative yet accepts the letter one
341/// (none of the M3.1 review's ~150 probes did).
342fn keep_fewest_letter_readings(alts: &mut Vec<Option<XM>>) {
343  // (a `None` alternative, an action's empty result, is no reading: kept, and not ranked; M3 review)
344  let counts: Vec<Option<usize>> = alts
345    .iter()
346    .map(|alt| alt.as_ref().map(crate::semantics::expectation_letter_count))
347    .collect();
348  let Some(fewest) = counts.iter().flatten().copied().min() else {
349    return;
350  };
351  if counts.iter().flatten().all(|&count| count == fewest) {
352    return;
353  }
354  let mut counts = counts.into_iter();
355  alts.retain(|_| counts.next().flatten().is_none_or(|count| count == fewest));
356}

358/// An integral's letter `d` before a variable yields to the differential of that variable over the same span
359/// (`diffop_apply`, `differential_power_apply`; Perl's IntFactor tries `diffd ATOM_OR_ID addScripts` first,
360/// MathGrammar:640-647): a glade keeps the alternatives with the fewest such letters, so each differential's letter
361/// twin does not multiply through the Cartesian products above it (the root's
362/// `LetterDsBeforeVariablesAreDifferentials` ranks whole trees the same way, for the tree iterator's; 57cj.20; 2605.28900,
363/// 2605.08899, 2605.29990). A hard drop, as `keep_fewest_letter_readings`.
364fn keep_fewest_letter_differentials(alts: &mut Vec<Option<XM>>) {
365  let pragma = ValidationPragmatics::LetterDsBeforeVariablesAreDifferentials;
366  let counts: Vec<Option<usize>> = alts
367    .iter()
368    .map(|alt| alt.as_ref().map(|alt| pragma.rank_violations(alt)))
369    .collect();
370  let Some(fewest) = counts.iter().flatten().copied().min() else {
371    return;
372  };
373  if counts.iter().flatten().all(|&count| count == fewest) {
374    return;
375  }
376  let mut counts = counts.into_iter();
377  alts.retain(|_| counts.next().flatten().is_none_or(|count| count == fewest));
378}

380/// Build a lexeme's raw bytes (`byte_lexeme`) from the first alternative of each child glade,
381/// as the tree builder's `rollup_token_rec` concatenates a token's bytes: byte-passthrough
382/// intermediate rules pre-roll their subtree into one Lexeme, so the outer rule just chains. The
383/// one-child case is common for byte-passthrough scaffolding, so return the child's existing
384/// `Rc<str>` rather than allocating a second one.
385fn collect_lexeme(glade: &Glade, rh_len: usize, children: &[Option<GladeAlts>]) -> Option<Rc<str>> {
386  let child_lexeme = |ix: usize| -> Option<&Rc<str>> {
387    let cid = glade.rh_glade_id(ix).expect("rh position has child glade");
388    let alts = children
389      .get(cid)
390      .and_then(|o| o.as_ref())
391      .expect("child precomputed");
392    match alts.first() {
393      Some(Some(XM::Lexeme(s, _))) => Some(s),
394      _ => None,
395    }
396  };
397  if rh_len == 1 {
398    return child_lexeme(0).cloned();
399  }
400  let mut raw = String::with_capacity(rh_len * 2);
401  for ix in 0..rh_len {
402    if let Some(s) = child_lexeme(ix) {
403      raw.push_str(s);
404    }
405  }
406  (!raw.is_empty()).then(|| Rc::from(raw))
407}

413  /// A lexeme's bytes survive the rollup: each byte a char of its own number, decoded once as
414  /// UTF-8 at the lexeme rule — the tree route's `ATOM:Γ:33`, not `ATOM::33` (57at).
415  #[test]
416  fn a_lexeme_keeps_its_non_ascii_bytes() {
417    let raw: String = "ATOM:Γ¯z:33"
418      .bytes()
419      .map(|byte| byte_lexeme(byte).to_string())
420      .collect();
421    assert_eq!(&*token_text(Rc::from(raw)), "ATOM:Γ¯z:33");
422    // ASCII passes through untouched.
423    assert_eq!(&*token_text(Rc::from("UNKNOWN:x:1")), "UNKNOWN:x:1");
