5// Marpa SLIF-style grammar DSL inside `grammar!()` / `production!()`
6// macros. A handful of long alternation lists (`qm_bracket`,
7// `floatsuperscript`) push past 100 chars; `rustfmt::skip` is applied
8// to the entire builder so the BNF reads as a flat table rather than
9// being re-flowed mid-rule.
10#[rustfmt::skip]
11pub fn init_grammar() -> Result<(MarpaGrammar, Actions, TreeBuilder)> {
12  // We create a declarative macro language of our own, in the spirit of the Marpa SLIF
13  default_registry!();
14  // Tokens, to be used in rules directly
15  token!(atom ~ "ATOM");
16  // A fraction or root of numbers (util.rs, `is_numeric_constant`): an ATOM to the actions, a number to
17  // `numeric_monomial` (57cj.1 review: an `atom` lead refused every other ATOM's twin).
18  token!(numeric_atom ~ "ATOM_NUMBER");
19  token!(unknown ~ "UNKNOWN");
20  token!(id ~ "ID");
21  // M4: Specialized tokens for "d" that could be differential operators.
22  // Lexer emits XDIFFUNK/XDIFFID instead of UNKNOWN/ID for "d" content.
23  // These are added as alternatives everywhere unknown/id appear, plus in the diffop rule.
24  token!(diffunk ~ "XDIFFUNK");
25  token!(diffid ~ "XDIFFID");
26  token!(array ~ "ARRAY");
27  token!(number ~ "NUMBER");
28  token!(punct ~ "PUNCT");
29  token!(period ~ "PERIOD");
30  // PUNCT with `\quad`-class spacing (rpadding ≥ 5pt). The lexer
31  // (util.rs::punct_followed_by_wide_space) emits these as
32  // `WIDE_PUNCT:,:idx` so the grammar can prefer formulae_apply
33  // for the `formula , \quad condition` arXiv idiom without
34  // enumerating the `list_apply` alternatives that the pragma
35  // would only reject post-hoc. See docs/archive/MATH_AMBIGUITY_AUDIT_2026-05-21.md §2.
36  token!(wide_punct ~ "WIDE_PUNCT");
37  // A comma alone, not a `\quad` hint (`PUNCT:quad`): the #37 hole's list separator.
38  token!(comma = "PUNCT:,");
39  // The ellipsis IDs (`\ldots`, `\hdots` and `...` lex as `ldots`; `\dots` and amsmath's variants):
40  // the #18 chain goes on across them and no other ID.

42  token!(dots_id = "ID:dots");
43  token!(dotsc_id = "ID:dotsc");
44  token!(dotsb_id = "ID:dotsb");
45  token!(dotsm_id = "ID:dotsm");
46  token!(dotsi_id = "ID:dotsi");
47  token!(dotso_id = "ID:dotso");
48  token!(addop_t ~ "ADDOP");
49  token!(mulop_t ~ "MULOP");
50  token!(relop_t ~ "RELOP");
51  token!(elideop ~ "ELIDEOP");
52  token!(langle_rel = "RELOP:less-than");
53  token!(langle_open = "OPEN:langle");
54  token!(langle = [langle_rel langle_open]);
55  token!(rangle_rel = "RELOP:greater-than");
56  token!(rangle_close = "CLOSE:rangle");
57  token!(rangle =[rangle_rel rangle_close]);
58  token!(vertbar ~ "VERTBAR");
59  token!(singlevertbar = "VERTBAR:|");
60  // `\|` and `\Vert` lex to a single doubled-bar token `VERTBAR:||` (vs four
61  // single `VERTBAR:|` for `||x||`). Used by the `\|x\|` norm rule below.
62  token!(doublevertbar = "VERTBAR:||");
63  // `\left|...\right|` produces VERTBAR tokens tagged `stretchy="true"`.
64  // The lexer (util.rs) further distinguishes those emitted by `\lx@delim@left`
65  // from those emitted by `\lx@delim@right` via the `lx@side` property set in
66  // the respective constructors (tex_math.rs). Side-distinct lexemes
67  // (LEFT_STRETCHY_VERTBAR / RIGHT_STRETCHY_VERTBAR) eliminate
68  // combinatorial pairing ambiguity for the kerned-stack norm idioms
69  // (\vertii, \vertiii, …) where multiple identical bars appear in
70  // sequence. `stretchy_vertbar` is the union — used by legacy rules
71  // (e.g. eval_at) that don't care which side a stretchy bar came from.
72  // See docs/archive/MATH_AMBIGUITY_AUDIT_2026-05-21.md §2 (delimiter-pairing) and Task #263.
73  token!(left_stretchy_vertbar ~ "LEFT_STRETCHY_VERTBAR");
74  token!(right_stretchy_vertbar ~ "RIGHT_STRETCHY_VERTBAR");
75  // Fallback for stretchy bars that arrived without a side tag
76  // (legacy DOM input, or a code path that bypassed `\lx@delim@left`/`\lx@delim@right`).
77  // The lexer in util.rs emits this when `role_side` is absent.
78  // Rules that legitimately accept any side (eval_at) enumerate both
79  // alternatives explicitly rather than relying on a union token —
80  // the Marpa tree builder doesn't roll up alternation-of-tokens cleanly.
81  token!(undirected_stretchy_vertbar ~ "STRETCHY_VERTBAR");
82  // A single stretchy bar, not a `\left\|` double one (`LEFT_STRETCHY_VERTBAR:||`, util.rs): the
83  // kerned stacks (`stretchy_norm_fenced`) are made of single bars only — `\left|\!\left\|A\right\|
84  // \!\right|` is |‖A‖|, as Perl, never a merged glyph that drops two bars (57am review round 7).
85  token!(left_stretchy_single_bar = "LEFT_STRETCHY_VERTBAR:|");
86  token!(right_stretchy_single_bar = "RIGHT_STRETCHY_VERTBAR:|");
87  token!(close_pipe = "CLOSE:|");
88  token!(middle_bar = "MIDDLE:|");
89  token!(middle_parallel = "MIDDLE:parallel-to");
90  token!(midbar = [vertbar middle_bar middle_parallel]);
91  // amsmath's `\lvert` and `\rvert` are an OPEN and a CLOSE (amsmath.sty.ltxml:1150-1152), lexed
92  // `OTHER_OPEN:lvert` / `OTHER_CLOSE:rvert` (util.rs): a group's delimiters, and a ket's opening and a bra's
93  // closing bar (`ket_bar`, `bra_bar`, 57bt).
94  token!(lvert_open = "OTHER_OPEN:lvert");
95  token!(rvert_close = "OTHER_CLOSE:rvert");
96  token!(lbrace = "OPEN:{");
97  token!(rbrace = "CLOSE:}");
98  token!(lparen = "OPEN:(");
99  token!(rparen = "CLOSE:)");
100  token!(lbracket = "OPEN:[");
101  // A group holding only integrals (`util::integral_operator_group`): its OPEN
102  token!(intop_group_open ~ "INTOP_GROUP_OPEN");
103  token!(rbracket = "CLOSE:]");
104  token!(relop_equals = "RELOP:equals");
105  token!(metarelop ~ "METARELOP");
106  token!(colon_metarelop = "METARELOP:colon");
107  token!(modifierop ~ "MODIFIEROP");
108  token!(modifier ~ "MODIFIER");
109  token!(arrow_t ~ "ARROW");
110  token!(binop_t ~ "BINOP");
111  token!(postfix ~ "POSTFIX");
112  token!(function ~ "FUNCTION");
113  token!(plain_opfunction ~ "OPFUNCTION");
114  // An expectation or probability, an OPFUNCTION of its own lexeme category
115  // (`parser::type_expectation_lexemes`): `opfunction` below takes it with the others, the
116  // operator nests take none (57cd).
117  token!(expectation_e = "EXPECTATION:\u{1D53C}");
118  token!(expectation_p = "EXPECTATION:\u{2119}");
119  // … and one that also reads as the letter it was lexed as (`expectation_letter`, M3): the letter retry's
120  // spelling, for a formula no typed stream parses (`parser::spell_letter_readings`).
121  token!(expectation_e_letter = "EXPECTATION:\u{1D53C}.letter");
122  token!(expectation_p_letter = "EXPECTATION:\u{2119}.letter");
123  // The OPFUNCTIONs whose scripts are limits — TeX's `\mathop` without `\nolimits` (plain.tex:1073-1074,
124  // :1083-1084), LaTeXML's `scriptpos => \&doScriptpos` (math_common.pool.ltxml:742, :759-764) — and the
125  // named `argmin`/`argmax` and the expectation, which bind a variable as they do: their subscript binds it
126  // (`limits_opfunction`, `bound_application`, 57cg; `semantics::binds_its_subscript` reads the same
127  // lexemes). `\log_a`, `\exp_p` are `\nolimits` (plain.tex:1054, :1082): a base, a parameter.
128  token!(maximum_opfunction = "OPFUNCTION:maximum");
129  token!(minimum_opfunction = "OPFUNCTION:minimum");
130  token!(gcd_opfunction = "OPFUNCTION:gcd");
131  token!(pr_opfunction = "OPFUNCTION:Pr");
132  token!(argmin_opfunction = "OPFUNCTION:argmin");
133  token!(argmax_opfunction = "OPFUNCTION:argmax");
134  token!(trigfunction ~ "TRIGFUNCTION");
135  token!(applyop ~ "APPLYOP");
136  token!(composeop ~ "COMPOSEOP");
137  token!(supop ~ "SUPOP");
138  // `open`/`close` match GENERIC delimiters (\lfloor, \lceil, \llbracket, etc.)
139  // Specific delimiters (paren, bracket, brace, langle) have their own tokens.
140  // The lexer (util.rs) emits OTHER_OPEN:/OTHER_CLOSE: for generic delimiters
141  // so Marpa doesn't ambiguously match both `open` AND `lparen` for `OPEN:(:N`.
142  token!(open ~ "OTHER_OPEN");
143  token!(close ~ "OTHER_CLOSE");
144  token!(middle ~ "MIDDLE");
145  token!(bigop ~ "BIGOP");
146  token!(sumop ~ "SUMOP");
147  token!(intop ~ "INTOP");
148  token!(limitop ~ "LIMITOP");
149  token!(diffop ~ "DIFFOP");
150  token!(operator ~ "OPERATOR");
151  token!(start_postsubscript ~ "start_POSTSUBSCRIPT");
152  token!(end_postsubscript ~ "end_POSTSUBSCRIPT");
153  token!(start_postsuperscript ~ "start_POSTSUPERSCRIPT");
154  token!(end_postsuperscript ~ "end_POSTSUPERSCRIPT");
155  // Separated bigop script tokens — prevents earley chart competition
156  // between scripted_bigop and scripted_factor rules
157  token!(start_bigopsub ~ "start_BIGOPSUB");
158  token!(end_bigopsub ~ "end_BIGOPSUB");
159  token!(start_bigopsup ~ "start_BIGOPSUP");
160  token!(end_bigopsup ~ "end_BIGOPSUP");
161  token!(start_floatsuperscript ~ "start_FLOATSUPERSCRIPT");
162  token!(end_floatsuperscript ~ "end_FLOATSUPERSCRIPT");
163  token!(start_floatsubscript ~ "start_FLOATSUBSCRIPT");
164  token!(end_floatsubscript ~ "end_FLOATSUBSCRIPT");
165  token!(start_arrow ~ "start_ARROW");
166  token!(end_arrow ~ "end_ARROW");

169      // The operators, each extended below with its decorating scripts (`addOpDecoration`).
170      relop = relop_t;
171      arrow = arrow_t;
172      addop = addop_t;
173      mulop = mulop_t;
174      binop = binop_t;
175      // An OPFUNCTION: a named one or an expectation or probability (the actions read both as
176      // OPFUNCTIONs, `semantics::operator_category`). An operator nests over a named one only
177      // (`plain_opfunction`), and takes an expectation's application (`expectation_application`):
178      // one derivation, no refused nest (57cd review).
179      expectation_head = expectation_e | expectation_p | expectation_e_letter | expectation_p_letter;
180      opfunction = plain_opfunction | expectation_head;
181      // (a nonterminal: a token group of exact-text tokens does not roll up in the ASF builder)
182      limits_opfunction = maximum_opfunction | minimum_opfunction | gcd_opfunction | pr_opfunction
183        | argmin_opfunction | argmax_opfunction | expectation_e | expectation_e_letter;
184      // Factors
185      // opfunction/function/trigfunction are NOT factors — they require arguments.
186      // Standalone usage is handled at the term level (term += function | ...).
187      // `2 \sin` is handled via dedicated tight_term rules below.
188      // Perl MathGrammar L315: ATOM_OR_ID : ATOM | ID | ARRAY
189      // XMArray elements (role="ARRAY") should parse as atoms/factors, like matrices in equations
190      // M4: diffunk/diffid are added as factor_base alternatives so "d" tokens
191      // can appear anywhere unknown/id appear. The diffop rule only uses diffunk/diffid.
192      // An expectation or probability also reads as the letter it was lexed as (M3, user directive
193      // 2026-09-30: an unparsed variant of a lexeme is a grammatical category of its own, the foundation
194      // intentionally ambiguous and disambiguated by pruning): `expectation_letter`, which its action
195      // presents as the UNKNOWN lexeme the lexer gave it (`expectation_as_letter`), a factor as any
196      // letter is. It survives only where no reading takes it as an operator (the soft prune
197      // `ExpectationLettersAreFallbacks`, first at the root, and per glade in the ASF traverser):
198      // `\sin\mathbb{E}\sum_i X_i` sine@(E)·∑…, `\nabla\log\mathbb{E}\sum_i X_i`, `\log x\cdot\mathbb{E}\sum_i X_i`
199      // (57cf review), one expectation at a time: `\sin\mathbb{E}\sum_i X_i+\mathbb{E}Y` keeps 𝔼@(Y). Only the
200      // letter retry's stream spells it (`expectation_e_letter`, parser.rs `spell_letter_readings`: a formula no
201      // typed stream parses; offered everywhere, the twin doubled the tree iterator's trees per expectation), and a
202      // letter's group application is none of its readings: the expectation's own (the token keeps its role, so
203      // the markup is the same).
204      expectation_letter = expectation_e_letter => expectation_as_letter
205        | expectation_p_letter => expectation_as_letter;
206      factor_base = unknown | number | id | atom | numeric_atom | array | diffunk | diffid | expectation_letter;
207      // Perl MathGrammar L277: OPEN ARRAY CLOSE -> Fence (e.g. \{ array \} or ( array ))
208      // Also handle unmatched delimiters for cases-like patterns.
209      // Perf: `open` is now narrowed to OTHER_OPEN (non-paren/bracket/brace), so
210      // we add specific rules for each standard delimiter.
211      fenced_array = open array close => fenced
212        | open array => open_fenced
213        | array close => close_fenced
214        | lparen array rparen => fenced
215        | lparen array => open_fenced
216        | array rparen => close_fenced
217        | lbracket array rbracket => fenced
218        | lbracket array => open_fenced
219        | array rbracket => close_fenced
220        | lbrace array rbrace => fenced
221        | lbrace array => open_fenced
222        | array rbrace => close_fenced;
223      // FUNCTION is a factor (participates in implicit multiplication).
224      // OPFUNCTION is intentionally NOT here. Including OPFUNCTION in
225      // `factor` would let `tight_term factor → apply_invisible_times`
226      // admit a bare OPFUNCTION as its LEFT operand — a reading the
227      // pragma `apply_invisible_times: left is OPFUNCTION/.../FUNCTION,
228      // prefer prefix_apply` correctly rejects, but only after the
229      // grammar has enumerated thousands of such derivations
230      // (~52% of trees on complex math-heavy equations such as
231      // 1911.09517 eq.993). Excluding OPFUNCTION from `factor`
232      // pushes the constraint upstream into the grammar: bare
233      // OPFUNCTION cannot anchor an invisible-times chain on the LEFT.
234      // To keep legitimate trailing-OPFUNCTION cases (`c \not`,
235      // `a b \not`, ...), `bare_opfunction_term` (below, 57cb) admits
236      // OPFUNCTION as a product-terminating RIGHT operand, a term no
237      // factor follows. OPFUNCTION as the head of an
238      // applied form (`\log x`, `\sin x`) enters via `applied_func`
239      // (`opfunction group_factor`, a group; `opfunction op_bare_item|op_bare_arg`, Perl's
240      // bare argument, after `op_bare_arg`).
241      // An ellipsis stands among juxtaposed factors and as a MulOp/BinOp operand (user ruling
242      // 2026-09-29: `\cdots` keeps ELIDEOP, divergence #3, with product rules): `a_1a_2\cdots a_n`,
243      // `a\times\cdots\times b`, `x_{i_1\cdots i_k}`, as Perl's `\cdots` (an ID, math_common.pool.ltxml:479)
244      // does (57bs; ~1,035 formulas in 236 A/B papers were unparsed). Not a `factor_base`: a limit-from and a
245      // differential take no ellipsis, a trig bare argument only as its first item (`trig_arg`, 57cj.12); an
246      // OPFUNCTION's or operator's bare argument does (`op_bare_item`, 57bx).
247      factor = factor_base | function | fenced_array | elideop;
248      // Perl: limit-from@(number, sign) — directional limits: 0+, 1-
249      // A "left-only term": on the left behaves as a term (for comma lists),
250      // on the right terminates at the addop (like expression-level postfix).
251      limit_from_term = factor_base addop => limit_from_apply;
252      // Terms
253      // Perl: bigop = BIGOP | SUMOP | INTOP | LIMITOP | DIFFOP (MathGrammar:717). A differential operator
254      // takes one factor (`diffop_application`, divergence #374), the others a term (`bigop_application`).
255      summation_bigop = bigop | sumop | intop | limitop;
256      any_bigop = summation_bigop | diffop;
257      // Adjacent bigops apply in turn, as Perl's `Factor : preScripted['bigop'] addOpArgs`
258      // (MathGrammar:292) does: `\partial\partial f` is ∂@(∂@(f)) through `diffop_operand`
259      // (a Rust-only `composed_bigop` gave (∂@∂)@(f); golden
260      // tests/parse/bigop_operands.tex#stacked_bigops_apply_in_turn).
261
262      // Compound operators: OPERATOR composed with functions/other operators (right-recursive)
263      // D sin => Apply(D, sin), D D sin => Apply(D, Apply(D, sin)), and D D => Apply(D, D):
264      // Perl `nestOperators` (MathGrammar:663-671) nests operators until a function, and
265      // `recApply` (MathParser.pm:1313-1315) applies each to the rest — `\nabla\nabla f` is
266      // (∇@∇)@(f).
267      compound_operator = operator trigfunction => prefix_apply
268        | operator function => prefix_apply
269        // (an expectation or probability is none: `\nabla\mathbb{E}[X]` applies ∇ to 𝔼's application)
270        | operator plain_opfunction => prefix_apply
271        | operator operator => prefix_apply
272        | operator compound_operator => prefix_apply;
273
274      // tight_term includes single factors (for left-recursive chaining)
275      // and all compound constructs (invisible times, prefix application, etc.)
276      // The `\log x` → `log*x` issue is handled by semantic pruning in
277      // apply_invisible_times, not at the grammar level.
278      tight_term = factor
279        | tight_term factor => factor_product
280        // Note: FUNCTION does NOT absorb bare args — only parens or APPLYOP.
281        // `fga` = f*g*a, but `f(a)` = f@(a). OPFUNCTION absorbs: `Fga` = F@(g*a).
282        // FUNCTION only chains via opfunction's factor status or APPLYOP.
283        // trigfunction uses trigbarearg via applied_func (absorbs MulOp chains)
284        // NOTE: bigop rules moved to += section (after `term` is defined) so they
285        // can absorb full term (mulop chains like x² * dx), not just tight_term.
286        // An operator applied to its argument is `op_application`, below.
287        | factor_base applyop tight_term => prefix_apply_applyop
288        // Perl: FUNCTION/OPFUNCTION/TRIGFUNCTION + explicit APPLYOP + argument
289        // Handles \lxDeclare-annotated tokens: f⁡(x) where ⁡ is APPLYOP
290        | function applyop tight_term => prefix_apply_applyop
291        | opfunction applyop tight_term => prefix_apply_applyop
292        | trigfunction applyop tight_term => prefix_apply_applyop;
293
294      // Perl MathGrammar L258: Factor moreFactors — consecutive function
295      // applications chain with invisible times.
296      // e.g. \sin x \cos y => sin(x) * cos(y)
297      //
298      // A trig function's bare argument is `trig_arg` (below): Perl's greedy `trigBarearg`
299      // (MathGrammar:340-357), MulOp chains included (`\sin x\cdot y` sin@(x·y)), ended only by
300      // explicit space or a differential `d` (#367).
301      // applied_func and tight_term augmentations moved below trig_arg definition
302
303      // Composed functions: f∘g, sin∘cos — these can then be applied as functions
304      // COMPOSEOP operates on function-level operands (curry level 2)
305      // Left-to-right associative (matching Perl): f∘g∘h = (f∘g)∘h
306      composed_term = function composeop function => infix_apply
307        | function composeop trigfunction => infix_apply
308        | function composeop opfunction => infix_apply
309        | trigfunction composeop function => infix_apply
310        | trigfunction composeop trigfunction => infix_apply
311        | trigfunction composeop opfunction => infix_apply
312        | opfunction composeop function => infix_apply
313        | opfunction composeop trigfunction => infix_apply
314        | opfunction composeop opfunction => infix_apply
315        // Left-recursive for left-to-right associativity
316        | composed_term composeop function => infix_apply
317        | composed_term composeop trigfunction => infix_apply
318        | composed_term composeop opfunction => infix_apply;
319
320      // Composed functions can be applied like regular functions
321      tight_term += composed_term tight_term => prefix_apply;
322

324      term = tight_term
325      | term mulop tight_term => infix_apply_nary
326      | term mulop tight_term elideop => infix_apply_and_elide
327      // Perl: BINOP matches both AddOp and MulOp (ambiguous precedence from \mathbin)
328      | term binop tight_term => infix_apply_nary
329      | term binop tight_term elideop => infix_apply_and_elide
330      // Fallback: COMPOSEOP on general terms (for non-function-level composition)
331      | term composeop term => infix_apply
332      | operator applyop term => prefix_apply_applyop;
333

336      // Allow standalone functions/trigfunctions/opfunctions/operators as terms
337      // This is needed for (f*g)(x) where f and g are FUNCTION tokens
338      // opfunction here allows standalone \operatorname{R} to parse
339      // an operator as a term, `D - 1`, `D + G`, is `bare_op_term` (below)
340      term += function | trigfunction | opfunction | composed_term;
341      // (An ellipsis in a sum, `y + i + \cdots + y_n`, is a term through `factor`.)
342
343      // Higher-order operator terms: functions as standalone objects multiplied by factors
344      // `2\sin` = `2 * sin`, `2\sin\cos` = `2 * sin * cos`
345      // These are term-level (not tight_term) so they don't interfere with
346      // function application: `2\sin x` = `2 * sin(x)` (not `(2*sin) * x`)
347      tight_opterm = factor function => apply_invisible_times
348        | factor trigfunction => apply_invisible_times
349        // Consecutive functions multiply: fgh → f·g·h (Perl Factor moreFactors); a bare OPFUNCTION after
350        // a factor is `bare_opfunction_term` (57cb)
351        | function function => apply_invisible_times
352        | function trigfunction => apply_invisible_times
353        | tight_opterm function => apply_invisible_times
354        | tight_opterm trigfunction => apply_invisible_times
355        | tight_opterm opfunction => apply_invisible_times;
356      term += tight_opterm;
357
358      // Expressions
359      expression = term
360        | expression addop term => infix_apply_nary
361        | expression addop term elideop => infix_apply_and_elide
362        // Perl MathGrammar:246 `SignedTerm : AddOp Term`: the sign takes a whole term — a mulop
363        // chain (`-a/b`, `x^{-1/2}`) or a bigop application (`-J\sum_{ij}…`, `-\int f`). A
364        // `tight_term` left those with no derivation (RUST-ONLY; golden
365        // tests/parse/bigop_operands.tex#signed_term_is_a_whole_term).
366        | addop term => prefix_apply
367        // A trailing sign, `0+`, `a^2+` (a factor is an expression: one rule, 57bi)
368        | expression addop => postfix_apply
369        // Perl MathGrammar L236: addExpressionModifier: MODIFIEROP Expression
370        // => Apply(modifierop, expr, expr2). Handles infix `a mod b`.
371        | expression modifierop expression => infix_apply
372        // Perl MathGrammar L236: addExpressionModifier: MODIFIER
373        // Standalone postfix modifier (e.g. `8\pmod{3}` → annotated(8, pmod(3)))
374        // Placed at expression level so MODIFIER binds BEFORE RELOP.
375        // e.g. `5 ≡ 8 \pmod{3}` → `5 ≡ annotated(8, pmod(3))` not `annotated(5≡8, pmod(3))`
376        | expression modifier => postfix_modifier_apply
377        // Perl MathGrammar L224-233: OPEN relop/modifierop Expression balancedClose
378        // Parenthesized modifier expressions: x(>0) → annotated(x, Fence(>0))
379        | expression lparen relop expression rparen => annotated_fenced_modifier
380        | expression lparen modifierop expression rparen => annotated_fenced_modifier
381        // Perl's `relop` includes the arrows (MathGrammar:713 `ARROW addOpDecoration`), which this
382        // grammar keeps apart: `\mathrm{prob}(\rightarrow j^*)` is annotated(prob, absent → j^*)
383        // (2605.13374 S2.E12; golden
384        // tests/parse/decorated_relations.tex#arrow_operand_in_parentheses_parses).
385        | expression lparen arrow expression rparen => annotated_fenced_modifier
386        // Perl MathGrammar L223: PUNCT? OPEN relop Expression CLOSE
387        // Semicolon annotation: a;(<e) → annotated(a, absent < e)
388        | expression punct lparen relop expression rparen => annotated_punct_fenced_modifier
389        | expression punct lparen modifierop expression rparen => annotated_punct_fenced_modifier
390        | expression punct lparen arrow expression rparen => annotated_punct_fenced_modifier;
