21/// Same as `node_to_grammar_lexemes` but with pre-filtered child nodes.
22/// Used when `filter_hints` has already been called (to avoid double-filtering). The document, when given, decodes the
23/// tokens' fonts (a `d` token's identity, `DifferentialEvidence`).
24pub fn node_to_grammar_lexemes_from(
25  mathnode: &Node,
26  child_nodes: Vec<Node>,
27  idx: &mut usize,
28  document: Option<&Document>,
29) -> (Vec<String>, Vec<Node>) {
30  let (mut lexemes, nodes) = node_to_grammar_lexemes_ctx(mathnode, child_nodes, idx, false);
31  // Which letter `d`s can be an integral's differential: the grammar's differential branches (`diffunk`/`diffid`:
32  // `diffop_apply`, `raised_differential_d`, grammar/builder.rs) are the only place `XDIFFUNK`/`XDIFFID` differ from a
33  // plain `unknown`/`id`, so a `d` that is none is lexed as the plain letter and the branch is never built (M4: ≈71
34  // and-nodes per `d<var>` in an INTOP-free formula, `\frac{dx}{dt}`, `d` as a variable). A `d` is a differential only
35  // inside an integral's operand (user ruling 2026-10-01, SYNC (17); `in_an_integral_operand`; Perl's `diffd`, an
36  // INTOP's argument only, MathGrammar:633-638), not anywhere in a formula holding an integral: `\int f\,dx\leq
37  // d\pi^d` reads the second `d` a letter (2605.03853), as the raised `d` of divergence #395 was since 57cj.20 — unless
38  // the formula or its document shows a differential outside (user rulings 2026-10-01: an SDE, a differential form, a
39  // measure's set; the document's own uses, `DifferentialMap`; `DifferentialEvidence`).
40  // A closed group holding only integrals is an integral operator (user ruling 2026-10-01; `integral_operator_group`):
41  // its OPEN lexes `INTOP_GROUP_OPEN`, a big operator the grammar applies to the integrand after it
42  // (`integral_operator_group`, grammar/builder.rs), and the `d`s of that integrand are its differentials
43  // (`in_an_integral_operand`).
44  if lexemes.iter().any(|l| l.starts_with("INTOP:"))
45    && lexemes.iter().any(|l| l.starts_with("OPEN:"))
46  {
47    let levels = operand_levels(&nodes, true);
48    for (open, lexeme) in lexemes.iter_mut().enumerate() {
49      if lexeme.starts_with("OPEN:") && integral_operator_group(&nodes, &levels, open).is_some() {
50        *lexeme = lexeme.replacen("OPEN:", "INTOP_GROUP_OPEN:", 1);
51      }
52    }
53  }
54  if lexemes.iter().any(|l| l.starts_with("XDIFF")) {
55    let levels = operand_levels(&nodes, true);
56    // (whether an earlier row left an integral open is asked only of a `d` the formula's own walk leaves open at its
57    // start, once: `continues_an_open_integral` reads the document)
58    let opened = std::cell::OnceCell::new();
59    // (a root's or a fraction part's content inside an integral's operand is all in it, `in_an_enclosing_integrand`)
60    let in_an_integrand = in_an_enclosing_integrand(mathnode);
61    let outside: Vec<usize> = lexemes
62      .iter()
63      .enumerate()
64      .filter(|(at, lex)| {
65        lex.starts_with("XDIFF")
66          && !in_an_integrand
67          && !in_an_integral_operand(&nodes, &levels, *at, false)
68          && !(in_an_integral_operand(&nodes, &levels, *at, true)
69            && *opened.get_or_init(|| continues_an_open_integral(mathnode)))
70      })
71      .map(|(at, _)| at)
72      .collect();
73    let evidence = DifferentialEvidence::of(&nodes, &levels, &outside, document, mathnode, true);
74    for at in outside {
75      if evidence.reads_a_differential(at) {
76        continue;
77      }
78      let lex = &mut lexemes[at];
79      if let Some(rest) = lex.strip_prefix("XDIFFUNK:") {
80        *lex = format!("UNKNOWN:{rest}");
81      } else if let Some(rest) = lex.strip_prefix("XDIFFID:") {
82        *lex = format!("ID:{rest}");
83      }
84    }
85  }
86  (lexemes, nodes)
87}

89/// Is the node at `at` inside an integral's operand (user ruling 2026-10-01, SYNC (17))? Perl reads `diffd` only among
90/// an INTOP's arguments (`moreIntOpArgFactors`, MathGrammar:633-638); here, as the token stream shows it: an INTOP
91/// before the node at its own level, with no relation, arrow or wide punctuation between at that level. The `d` keeps its
92/// differential reading past a sign, where Perl's integrand ends (divergence #398; `\int_M a+b\,dV`,
93/// `\int|\mathrm{d}P-\mathrm{d}Q|`, 2605.17001, 2605.15207: 697 such sites in 129 of the 3,003 A/B papers, all sum
94/// integrands), past a colon, a product in an operand (the Frobenius `\int_\Omega\mathbf{H}:\mathbf{Q}_t\,d\mathbf{x}`,
95/// 2605.24758, 2605.12883, 2605.19415), and past narrow punctuation, mostly a comma typed for `\,`
96/// (`\int_0^L\bm u\cdot\bm e_x,dx`, 2605.21567, 2605.04013) — even where the grammar ends the integral's tree before it
97/// (`\int f-g\,dx` reads integral@(f) − g·dx); wide punctuation (`,\quad`, a `\qquad`) separates formulas (2605.02034). A
98/// group closed before the node is skipped whole — an integral inside it is no operand the node is in,
99/// `(\int_0^1 f\,dx)\,d\pi`, `\left|\int f\,dx\right|d\pi`, nor does a relation inside it end anything,
100/// `\int\mathbb{1}\{x\le y\}\,d^2x` —, and a group enclosing the node is left for the level outside it: the measures
101/// `\int f\,P(y,dx')` (2605.08485), `\int K\tilde\mu(ds,de)` (2605.20593). A node inside a script is in none (an exponent's
102/// `d`), nor is a `d` that a big operator in between binds, its index (`\int f\,dx+\sum_{d=1}^D d\,w_d`, the 57cj.20 review;
103/// until a sign at that level ends the big operator's operand; `binds_the_letter_d`). The realized role is read, as a
104/// gathered/split row's content branch lexes XMRefs whose INTOP is their target's (golden
105/// tests/parse/integrals_and_differentials.tex#gathered_row_keeps_its_differential). A row that continues an integral an
106/// earlier row of its alignment left open (`opened`, `continues_an_open_integral`) is inside it up to its first relation
107/// (user ruling 2026-10-01: `&\times\Bigl[…\Bigr]\mathrm d\xi` after `=\int_0^T\Bigl[…`, 2605.15926).
108pub(crate) fn in_an_integral_operand(
109  nodes: &[Node],
110  levels: &OperandLevels,
111  at: usize,
112  opened: bool,
113) -> bool {
114  integral_operand_reach(nodes, levels, at, opened).is_some()
115}

117/// `in_an_integral_operand`, and if so whether the walk alone reaches the node — a sign, punctuation or a plain bar
118/// between the integral and the node at the integral's level (`\int f\,dx + C\,dh`, `\int f\,dx;\,d\pi`,
119/// `|\int f\,dx|\,d\pi`: the grammar ends the integral's tree there; the node is in its operand by this walk only).
120fn integral_operand_reach(
121  nodes: &[Node],
122  levels: &OperandLevels,
123  at: usize,
124  opened: bool,
125) -> Option<bool> {
126  let operand_levels = levels;
127  let OperandLevels { roles, levels } = levels;
128  let &(mut level) = levels.get(at)?;
129  let mut ended = false;
130  let mut signed = false;
131  let mut weak = false;
132  for at in (0..at).rev() {
133    let (node, node_level, role) = (&nodes[at], levels[at], roles[at].as_str());
134    if node_level > level {
135      // inside a group closed before the node
136      continue;
137    }
138    if node_level < level {
139      // the opening of a group enclosing the node: an OPEN, a left bar, a script's start (an exponent's `d` is no
140      // differential) or another role-carrying application's
141      if matches!(
142        role,
143        "POSTSUBSCRIPT" | "POSTSUPERSCRIPT" | "FLOATSUBSCRIPT" | "FLOATSUPERSCRIPT"
144      ) {
145        return None;
146      }
147      level = node_level;
148      ended = false;
149      signed = false;
150      weak = false;
151      continue;
152    }
153    match role {
154      "INTOP" if !ended => return Some(signed || weak),
155      // (a group closed before the node that holds only integrals is one, `integral_operator_group`)
156      "CLOSE"
157        if !ended
158          && (0..at)
159            .rev()
160            .find(|&open| levels[open] == node_level && roles[open] == "OPEN")
161            .and_then(|open| integral_operator_group(nodes, operand_levels, open))
162            == Some(at) =>
163      {
164        return Some(signed || weak);
165      },
166      "RELOP" | "ARROW" => ended = true,
167      "METARELOP" if !is_a_colon(node) => ended = true,
168      "PUNCT" if node.get_name() == "XMHint" || punct_followed_by_wide_space(node) => ended = true,
169      "PUNCT" | "VERTBAR" => weak = true,
170      "ADDOP" => signed = true,
171      "SUMOP" | "BIGOP" | "LIMITOP" if !signed && binds_the_letter_d(nodes, at) => return None,
172      _ => {},
173    }
174  }
175  // … or before the formula, an integral an earlier row left open (`continues_an_open_integral`)
176  (opened && !ended).then_some(signed || weak)
177}

179/// Is `container`, the content of a root or of a fraction's part, parsed on its own, inside an integral's operand in the
180/// formula around it (57cj.23.7; the integrand outranks every scalar context, user ruling 2026-10-01e) —
181/// `\int\sqrt{dP\,dQ}` (2605.09119), `\int\frac{dx}{2+x}` (ruling 2026-10-01c)? A sub-parse runs before the formula
182/// around it, which is read raw; a script's content stays out (an exponent's `d`, `\int e^{-dx}\,dx`).
183pub(crate) fn in_an_enclosing_integrand(container: &Node) -> bool {
184  if container.get_name() != "XMArg" {
185    return false;
186  }
187  let Some(app) = container.get_parent() else {
188    return false;
189  };
190  let root_or_fraction = app.get_name() == "XMApp"
191    && app.get_child_elements().first().is_some_and(|op| {
192      matches!(
193        op.get_attribute("meaning").as_deref(),
194        Some("square-root" | "nth-root")
195      ) || op.get_attribute("role").as_deref() == Some("FRACOP")
196    });
197  let Some(outer) = app.get_parent().filter(|_| root_or_fraction) else {
198    return false;
199  };
200  let nodes: Vec<Node> = outer
201    .get_child_elements()
202    .into_iter()
203    .filter(|node| node.get_name() != "XMHint")
204    .collect();
205  let Some(at) = nodes.iter().position(|node| *node == app) else {
206    return false;
207  };
208  integral_operand_reach(&nodes, &operand_levels(&nodes, false), at, false).is_some()
209    || in_an_enclosing_integrand(&outer)
210}

212/// The evidence that a letter `d` before a variable outside an integral's operand is a differential all the same (user
213/// ruling 2026-10-01, widening SYNC (17); Perl reads it a letter, `diffd` being an INTOP's argument only,
214/// MathGrammar:633-638). The lexer offers the differential (XDIFF) wherever the evidence holds — the grammar keeps both
215/// readings and `LetterDsBeforeVariablesAreDifferentials` prefers the differential — and the plain letter elsewhere. The
216/// evidence, each a shape of the corpus (A/B km21's 92 lost readings; precision sampled over the 3,003 papers), then the
217/// document's (`DifferentialMap`; an upright `d` is no evidence alone since 57cj.23.6, user rulings 2026-10-01c/d/e):
218/// - a scalar context of its own formula is none (`ArgContext::is_scalar`, `scalar_by_shape`, `in_an_order_argument`);
219/// - a Leibniz or Radon-Nikodym quotient's part, italic too (`\frac{dy}{dx}`), a Fréchet derivative applied to a direction
220///   (`\mathrm{d}H(u)[\psi]`, 2605.04766);
221/// - two or more `d<var>` in the formula, no other `d` a variable there: an SDE or a differential form
222///   (`dX_t=\mu\,dt+\sigma\,dW_t`, 2605.14643; `ds^2=dX^2+dY^2`, 2605.27600; `dJ/dK`; ~92 % of 70);
223/// - a whole item of a group applied to a letter, a measure's set (`\pi(du):=`, 2605.02070; `F(du\mid x)`, 2605.18724;
224///   `\lambda(\mathrm{d}z)`, `\widetilde N(ds,dz)`) — not after an order symbol, `O(dk^3)` a dimension (~85 % of 30);
225/// - the opening of a formula's first side before a relation, with an integral in the formula (`dU(z)=-\int…`,
226///   2605.26170; 45 of 45);
227/// - beside a wedge, a differential form (`dx\wedge dy`; 29 of 29).
228/// - after `\in`, an infinitesimal set (`\tau_1\in dt`; 2605.03594, 2605.11264, 2605.15276) — checked after the formula's own
229///   scalar `d`; not before a set's letter the `d` scales (`k\in d\mathbb Z`, `y\in dZ`), nor where the document uses the
230///   same `d` as a number unless the set is a measure's argument (`d\geq 2` … `q\in dn`; `P(X_t\in dx)`).
231///
232/// No `d` reads so that heads an italic word (`dist`: its one-letter variable runs on into another letter) or is Pearl's
233/// `do(` (2605.31254); nor, as the only evidence of an upright or relation-opening `d`, one whose variable runs on into a
234/// letter unspaced (a time step `\mathrm dt^2L_R^2`, 2605.29194); a dimension stays a letter (`\leq d\pi^d`, 2605.03853),
235/// as does a `d` whose token the formula itself uses as a scalar (`scalar_here`). Past the formula: the same `d<var>` a
236/// scalar elsewhere in the document is a letter, a differential elsewhere a differential (`\mathrm dS` beside
237/// `\int_\Sigma f\,\mathrm dS`, 2605.16486); an upright `d` the document never uses as a scalar is a differential.
238/// A raised `d` takes a variable only after an integer or one-letter order (`d^2x`, not `d^\top`).
239struct DifferentialEvidence<'a> {
240  nodes:                 &'a [Node],
241  levels:                &'a OperandLevels,
242  /// the lexer's stream (a script opens and closes around its content), or a formula's element children (a script is
243  /// one node: the document map's reading, `DifferentialMap::read`)
244  stream:                bool,
245  /// the outside sites that are differential candidates (a variable after them, not in a script, no word head)
246  candidates:            Vec<Candidate>,
247  /// two or more candidates and no other `d` a variable in the formula
248  a_form:                bool,
249  /// an INTOP in the formula
250  integral:              bool,
251  /// a wedge past the formula's first top-level relation (`\mathrm d\eta^a=-\frac12 f\eta^b\wedge\eta^c`)
252  wedge_past_a_relation: bool,
253  document:              Option<&'a Document>,
254  /// where the formula sits, a sub-parse's `XMArg` (`ArgContext`)
255  context:               ArgContext,
256  /// the `d` tokens (`d_identity`) this formula itself uses as a scalar: a lone `d` (`d\pi^d`'s exponent, a
257  /// big operator's index `\sum_{d=1}`, `x\in\mathbb R^d`), a `d<var>` in a script outside an integrand
258  scalar_here:           rustc_hash::FxHashSet<String>,
259}

271/// What a `d<var>`'s own formula shows (`DifferentialEvidence::local`), before its document is asked.
272#[derive(Clone, Copy, PartialEq)]
273enum Local {
274  /// a differential: a Leibniz quotient's part, a measure's argument, a form, a wedge, a Fréchet direction, …
275  Differential,
276  /// a scalar context (a root, a script, a d-less denominator's part) or a scalar shape (`\to 0`, a number assigned):
277  /// the document map records the `d` a scalar
278  Scalar,
279  /// a letter for this formula only (an order symbol's argument, a run-on variable, a bold `d`, the formula's own
280  /// scalar `d`)
281  Letter,
282  /// nothing decides: `\in d<var>`, then the document
283  Open,
284}

286impl<'a> DifferentialEvidence<'a> {
287  fn of(
288    nodes: &'a [Node],
289    levels: &'a OperandLevels,
290    outside: &[usize],
291    document: Option<&'a Document>,
292    mathnode: &Node,
293    stream: bool,
294  ) -> Self {
295    let mut candidates = Vec::new();
296    let mut a_variable = false;
297    for &at in outside {
298      match differential_variable_in(nodes, at, stream) {
299        Some((variable, end)) => {
300          if heads_a_word(nodes, at, end) || inside_a_word(nodes, at, document) {
301            continue;
302          }
303          let runs_on = nodes.get(end).is_some_and(|next| {
304            is_a_letter(next)
305              && !(token_text(next) == "d"
306                && differential_variable_in(nodes, end, stream).is_some())
307          }) && !has_trailing_space(&nodes[end - 1]);
308          let candidate = Candidate { at, variable, end, runs_on };
309          // (a script's `d<var>` is none, but a measure's argument, which outranks the script: user ruling
310          // 2026-10-02, `\|F\|_{L^1(d\mu)}`)
311          if in_a_script(nodes, levels, at) && !is_a_measure_s_item(nodes, levels, &candidate) {
312            continue;
313          }
314          candidates.push(candidate);
315        },
316        // a `d` with no variable is a variable itself, unless a group, an operator or a structure follows it
317        // (`d(\mu_1\times\mu_2)`, `\mathrm d\frac{\partial L}{\partial M}`, `\mathrm d\langle W\rangle`)
318        None => {
319          a_variable |= !in_a_script(nodes, levels, at)
320            && !nodes.get(at + 1).is_some_and(|next| {
321              next.get_name() == "XMApp"
322                || matches!(
323                  levels.roles[at + 1].as_str(),
324                  "OPEN" | "OPERATOR" | "OPFUNCTION" | "TRIGFUNCTION" | "FUNCTION" | "FRACOP"
325                )
326            })
327        },
328      }
329    }
330    let a_form = candidates.len() >= 2 && !a_variable;
331    let integral = levels.roles.iter().any(|role| role == "INTOP");
332    let wedge_past_a_relation = (0..nodes.len())
333      .find(|&k| levels.levels[k] == 0 && levels.roles[k] == "RELOP")
334      .is_some_and(|relation| nodes[relation + 1..].iter().any(is_a_wedge));
335    let mut scalar_here = rustc_hash::FxHashSet::default();
336    for k in 0..nodes.len() {
337      if !matches!(nodes[k].get_name().as_str(), "XMTok" | "XMRef") || token_text(&nodes[k]) != "d"
338      {
339        continue;
340      }
341      // (a lone upright `d` is an operator or a label, never a scalar: `\frac{\mathrm d}{\mathrm dt}`, `\mathrm d^2=0`,
342      // `M_{\rm d}`)
343      let lone = differential_variable_in(nodes, k, stream).is_none();
344      let scalar = if lone && is_upright(&nodes[k], document) {
345        false
346      } else if in_a_script(nodes, levels, k) {
347        // (an integrand's `d` in an exponent is a differential, and a measure's argument one)
348        integral_operand_reach(nodes, levels, k, false).is_none()
349          && !candidates.iter().any(|candidate| candidate.at == k)
350      } else {
351        lone
352          && !nodes.get(k + 1).is_some_and(|next| {
353            next.get_name() == "XMApp"
354              || matches!(
355                levels.roles[k + 1].as_str(),
356                "OPEN" | "OPERATOR" | "OPFUNCTION" | "TRIGFUNCTION" | "FUNCTION" | "FRACOP"
357              )
358          })
359      };
360      if scalar {
361        scalar_here.insert(d_identity(&nodes[k], document));
362      }
363    }
364    DifferentialEvidence {
365      nodes,
366      levels,
367      stream,
368      candidates,
369      a_form,
370      integral,
371      wedge_past_a_relation,
372      document,
373      context: ArgContext::of(mathnode),
374      scalar_here,
375    }
376  }

378  /// The `(d token, variable)` pair of a candidate (`DifferentialMap`).
379  fn pair(&self, candidate: &Candidate) -> (String, String) {
380    (
381      d_identity(&self.nodes[candidate.at], self.document),
382      token_text(&self.nodes[candidate.variable]),
383    )

386  /// Is the candidate a differential by the strong evidence of its formula's shape (user ruling 2026-10-01e): a form,
387  /// a measure's item, a relation's first side with an integral or a wedge past the relation, beside a wedge, a Fréchet
388  /// derivative applied to a direction?
389  fn strong(&self, candidate: &Candidate) -> bool {
390    let (nodes, levels, at) = (self.nodes, self.levels, candidate.at);
391    self.a_form
392      || is_a_measure_s_item(nodes, levels, candidate)
393      || (self.integral || self.wedge_past_a_relation)
394        && !candidate.runs_on
395        && opens_the_first_side(levels, candidate)
396      || at
397        .checked_sub(1)
398        .is_some_and(|before| is_a_wedge(&nodes[before]))
399      || nodes.get(candidate.end).is_some_and(is_a_wedge)
400      || applies_to_a_direction(nodes, levels, candidate)
401  }

403  /// What the candidate's own formula shows. The order of the evidence (user rulings 2026-10-01d/e and 2026-10-02,
404  /// after the intent study of the 377 formulas the upright font alone made differentials): a Leibniz quotient's part,
405  /// a fraction's or a slash's, whatever its container (`\frac{\sigma}{dE/dx}`, `\frac{1}{\mathrm ds/\mathrm d\lambda}`);
406  /// a measure's argument in a script; a scalar context of the formula itself; then evidence its shape gives, whatever
407  /// the font; then what the formula itself shows of its `d` token. The font decides nothing on its own: a document
408  /// keeps one `d` token for its differentials.
409  fn local(&self, candidate: &Candidate) -> Local {
410    let (nodes, levels, at) = (self.nodes, self.levels, candidate.at);
411    if self.context.is_leibniz() || in_a_slash_quotient(nodes, levels, at, self.stream) {
412      return Local::Differential;
413    }
414    if self.context == ArgContext::Script && is_a_measure_s_item(nodes, levels, candidate) {
415      return Local::Differential;
416    }
417    if self.context.is_scalar()
418      || scalar_by_shape(nodes, at, candidate.end, self.candidates.len() > 1)
419    {
420      return Local::Scalar;
421    }
422    // (an order symbol's argument is a dimension or a step, unless the same `d<var>` is a differential elsewhere in the
423    // formula: user ruling 2026-10-02, `\phi+d\phi … o(\|d\phi\|^2)`)
424    if in_an_order_argument(nodes, levels, candidate) {
425      let pair = self.pair(candidate);
426      let elsewhere = self.candidates.iter().any(|other| {
427        other.at != at
428          && !in_an_order_argument(nodes, levels, other)
429          && self.pair(other) == pair
430          && self.strong(other)
431      });
432      return if elsewhere {
433        Local::Differential
434      } else {
435        Local::Letter
436      };
437    }
438    if self.strong(candidate) {
439      return Local::Differential;
440    }
441    // A bold `d` bolder than its variable is the vector convention (`\mathbf{d}x`): only shape evidence makes it a
442    // differential (user ruling 2026-10-01d: the bold font is not evidence), never its token's use in the document;
443    // under `\boldmath` both are bold and the font stays evidence. What the formula itself shows of its `d` token
444    // outranks the document (`\int_{[-1,1]^d}f\,dx\leq d\pi^d`, 2605.03853; `\sum_{d=1}^D d\,w_d`).
445    if candidate.runs_on
446      || self.bold_contrast(candidate)
447      || self
448        .scalar_here
449        .contains(&d_identity(&nodes[at], self.document))
450    {
451      return Local::Letter;
452    }
453    Local::Open
454  }
455
456  /// Is the candidate's `d` bold and its variable not (`\mathbf{d}x`)?
457  fn bold_contrast(&self, candidate: &Candidate) -> bool {
458    // (the token's font as the document records it, `_font`, specialized as the serializer does: a plain `d` is italic)
459    let bold = |node: &Node| {
460      token_font(node, self.document)
461        .is_some_and(|font| font.get_series().is_some_and(|series| series == "bold"))
462    };
463    bold(&self.nodes[candidate.at]) && !bold(&self.nodes[candidate.variable])
464  }
465
466  fn reads_a_differential(&self, at: usize) -> bool {
467    let Some(candidate) = self.candidates.iter().find(|candidate| candidate.at == at) else {
468      return false;
469    };
470    match self.local(candidate) {
471      Local::Differential => return true,
472      Local::Scalar | Local::Letter => return false,
473      Local::Open => {},
474    }
475    let nodes = self.nodes;
476    let variable = &nodes[candidate.variable];
477    let variable_font = token_font(variable, self.document);
478    let upright = is_upright(&nodes[at], self.document) && !self.bold_contrast(candidate);
479    // `\in d<var>`, an infinitesimal set (`\tau_1\in dt`; 2605.03594, 2605.11264, 2605.15276) — behind the formula's
480    // own scalar `d` (`d\geq 2,\ q\in dn`), and not before a set's letter, which the `d` scales: a blackboard,
481    // calligraphic, fraktur, script or bold letter, or a capital (`k\in d\mathbb Z`, `x\in d\Lambda`, `y\in dZ`; the
482    // 57cj.23.6 review)
483    let set_letter = token_text(variable)
484      .chars()
485      .next()
486      .is_some_and(char::is_uppercase)
487      || variable_font
488        .as_ref()
489        .is_some_and(|font| font.get_series().is_some_and(|series| series == "bold"))
490      || variable_font.as_ref().is_some_and(|font| {
491        font.get_family().is_some_and(|family| {
492          matches!(
493            family.to_string().as_str(),
494            "blackboard" | "caligraphic" | "fraktur" | "script"
495          )
496        })
497      });
498    if at
499      .checked_sub(1)
500      .is_some_and(|before| token_text(&nodes[before]) == "\u{2208}")
501    {
502      // (… and not where the document uses the same `d` as a number, `d\geq 2` … `q\in dn`, unless the set is a
503      // measure's argument, `P(X_t\in dx)`)
504      let a_number_here = crate::data::with_differential_map(|map| {
505        map.is_some_and(|map| {
506          map
507            .lone_scalar_ids
508            .contains(&d_identity(&nodes[at], self.document))
509        })
510      });
511      return !set_letter
512        && (!a_number_here || measure_group_open(nodes, self.levels, at, true).is_some());
513    }
514    let pair = self.pair(candidate);
515    crate::data::with_differential_map(|map| match map {
516      Some(map) if map.scalar_pairs.contains(&pair) => false,
517      Some(map) if map.differential_pairs.contains(&pair) => true,
518      // The token's identity decides for an upright `d`, a document's convention token (`\mathrm d`, `{\rm d}`; the
519      // ruling's population): never a scalar in the document, a differential. The italic `d` is every document's default
520      // letter as well: alone it needs its own pair integrated elsewhere (`\int f\,dx\le d^2n`, `abcde` stay letters).
521      Some(map) => upright && !map.scalar_ids.contains(&pair.0),
522      // (no document map, a lexer called outside the parser: the font is the document's convention)
523      None => upright,
524    })
525  }
526}

528/// Where a sub-parse's content sits (`ArgContext::of`): the argument of a root or a script, a fraction's numerator or
529/// denominator, or elsewhere.
530#[derive(Clone, Copy, PartialEq)]
531enum ArgContext {
532  Root,
533  Script,
534  /// a fraction's part; `true` when both parts open with a `d` (a Leibniz or Radon-Nikodym quotient)
535  Numerator(bool),
536  /// a fraction's denominator; the flag as for the numerator, and whether the numerator holds a `d` at all
537  Denominator(bool, bool),
538  Other,
539}
540
541impl ArgContext {
542  /// The context of `container`, an `XMArg` whose parent `XMApp` is a root, a script or a fraction.
543  fn of(container: &Node) -> ArgContext {
544    if container.get_name() != "XMArg" {
545      return ArgContext::Other;
546    }
547    let Some(app) = container.get_parent() else {
548      return ArgContext::Other;
549    };
550    if app.get_name() != "XMApp" {
551      return ArgContext::Other;
552    }
553    if matches!(
554      app.get_attribute("role").as_deref(),
555      Some(
556        "POSTSUBSCRIPT"
557          | "POSTSUPERSCRIPT"
558          | "SUBSCRIPT"
559          | "SUPERSCRIPT"
560          | "FLOATSUBSCRIPT"
561          | "FLOATSUPERSCRIPT"
562      )
563    ) {
564      // An under- or overbrace's label restates the braced term: no script (user ruling 2026-10-02;
565      // `\underbrace{…}_{h_{\alpha\beta}\mathrm dy^\alpha\mathrm dy^\beta}`, 2605.15276)
566      let a_brace = app.get_prev_element_sibling().is_some_and(|base| {
567        base.get_name() == "XMApp"
568          && base.get_child_elements().first().is_some_and(|op| {
569            matches!(
570              op.get_attribute("name").as_deref(),
571              Some("underbrace" | "overbrace" | "underbracket" | "overbracket")
572            ) || matches!(
573              token_text(op).as_str(),
574              "\u{23DF}" | "\u{23DE}" | "\u{23B5}" | "\u{23B4}"
575            )
576          })
577      });
578      return if a_brace {
579        ArgContext::Other
580      } else {
581        ArgContext::Script
582      };
583    }
584    let children: Vec<Node> = app.get_child_elements();
585    let Some(op) = children.first() else {
586      return ArgContext::Other;
587    };
588    if matches!(
589      op.get_attribute("meaning").as_deref(),
590      Some("square-root" | "nth-root")
591    ) {
592      return ArgContext::Root;
593    }
594    if op.get_attribute("role").as_deref() == Some("FRACOP") && children.len() == 3 {
595      // (a part parsed already, or unwrapped to its one token, opens with its first visible token)
596      let opens_with_d =
597        |arg: &Node| first_visible_token(arg).is_some_and(|first| token_text(&first) == "d");
598      // (a denominator is lexed before it is parsed: its `d` takes a variable, `\frac{dn}{d+2}` is no quotient)
599      let opens_with_a_differential = |arg: &Node| {
600        if arg.get_name() != "XMArg" {
601          return opens_with_d(arg);
602        }
603        let nodes: Vec<Node> = arg
604          .get_child_elements()
605          .into_iter()
606          .filter(|node| node.get_name() != "XMHint")
607          .collect();
608        nodes.first().is_some_and(|first| token_text(first) == "d")
609          && raw_differential_variable(&nodes, 0).is_some()
610      };
611      // A Leibniz numerator is a differential operator or a differential (`numerator_of_a_quotient`), read raw —
612      // when the numerator is lexed, before it is parsed —, and recorded on the fraction for the denominator, lexed
613      // after the numerator's parse (`_lx_leibniz`, a bookkeeping attribute the finalize drops).
614      let numerator_is_a_differential = |arg: &Node| {
615        // (the verdict recorded when the numerator was read raw, so its two parts always agree)
616        if let Some(recorded) = app.get_attribute("_lx_leibniz") {
617          return recorded == "1";
618        }
619        if arg.get_name() == "XMArg" {
620          let nodes: Vec<Node> = arg
621            .get_child_elements()
622            .into_iter()
623            .filter(|node| node.get_name() != "XMHint")
624            .collect();
625          return numerator_of_a_quotient(&nodes);
626        }
627        // (unwrapped to its one token: a lone `d`)
628        arg.get_name() == "XMTok" && token_text(arg) == "d"
629      };
630      let numerator = numerator_is_a_differential(&children[1]);
631      let leibniz = numerator && opens_with_a_differential(&children[2]);
632      if children[1] == *container {
633        let mut app = app.clone();
634        let _ = app.set_attribute("_lx_leibniz", if numerator { "1" } else { "0" });
635        return ArgContext::Numerator(leibniz);
636      }
637      if children[2] == *container {
638        // (the numerator, parsed by now, anywhere in its tree: `\frac{p\,\mathrm dV}{\mathrm dt}`)
639        let mut tokens = Vec::new();
640        collect_named(&children[1], "XMTok", &mut tokens);
641        let holds_d = children[1].get_name() == "XMTok" && token_text(&children[1]) == "d"
642          || tokens.iter().any(|node| token_text(node) == "d");
643        return ArgContext::Denominator(leibniz, holds_d);
644      }
645    }
646    ArgContext::Other
647  }
648
649  /// A scalar context (user ruling 2026-10-01e): a root, a script, a denominator whose numerator holds no `d`
650  /// (`\sqrt{\mathrm dt}`, `x_{t+\mathrm dt}`, `\frac{3}{2\,\mathrm dt}`, 2605.09779, 2605.00250, 2605.29194).
651  fn is_scalar(self) -> bool {
652    matches!(
653      self,
654      ArgContext::Root | ArgContext::Script | ArgContext::Denominator(false, false)
655    )
656  }
657
658  /// A Leibniz or Radon-Nikodym quotient's part (`\frac{dy}{dx}`, `\frac{\mathrm d}{\mathrm dt}`; ruling 2026-10-01d:
659  /// a d-fraction is evidence, italic too).
660  fn is_leibniz(self) -> bool {
661    matches!(
662      self,
663      ArgContext::Numerator(true) | ArgContext::Denominator(true, _)
664    )
665  }
666}

674/// A token's font as the document records it (`_font`, through an XMRef), specialized for its text as the serializer
675/// does (a plain `d` is italic): the one decoding behind `d_identity`, `is_upright` and the evidence's font tests.
676fn token_font(node: &Node, document: Option<&Document>) -> Option<Font> {
677  let node = resolve_xmref(node).unwrap_or_else(|| node.clone());
678  let hash = node.get_attribute("_font")?;
679  document?
680    .decode_font(&hash)
681    .map(|font| font.specialize(&token_text(&node)))
682}

684/// Is the raw content of a fraction's numerator (its element children) a differential operator or a differential, as
685/// a Leibniz quotient's numerator: a `d`, with its power (`d`, `d^2`), before a variable (`dy`, `d^2y`), an accented
686/// letter (`d\vec r`), a function (`d\ln Z`, `d\sin\phi`) or a group holding no `d` with nothing after it but
687/// factors (`d(uv)`, `d\left(q^2\right)`, `d|\psi\rangle`, `d\langle A\rangle`) — not a sum or a product that merely
688/// opens with `d` (`\frac{d-1}{dk}`, `\frac{d(d+1)}{dn}`; the 57cj.23.6 review).
689fn numerator_of_a_quotient(nodes: &[Node]) -> bool {
690  if !nodes.first().is_some_and(|first| token_text(first) == "d") {
691    return false;
692  }
693  let mut k = 1;
694  while nodes.get(k).is_some_and(is_a_script_marker) {
695    k += 1;
696  }
697  if k == nodes.len() || raw_differential_variable(nodes, 0).is_some() {
698    return true;
699  }
700  let first = &nodes[k];
701  let role = get_grammatical_role(first);
702  let accent = first.get_name() == "XMApp"
703    && first.get_child_elements().first().is_some_and(|op| {
704      matches!(
705        op.get_attribute("role").as_deref(),
706        Some("OVERACCENT" | "UNDERACCENT")
707      )
708    });
709  if accent || matches!(role.as_str(), "OPFUNCTION" | "TRIGFUNCTION" | "OPERATOR") {
710    return true;
711  }
712  if !(matches!(role.as_str(), "OPEN" | "VERTBAR") || token_text(first) == "\u{27E8}") {
713    return false;
714  }
715  let holds_a_d = nodes[k..].iter().any(|node| {
716    let mut tokens = Vec::new();
717    if node.get_name() == "XMTok" {
718      tokens.push(node.clone());
719    } else {
720      collect_named(node, "XMTok", &mut tokens);
721    }
722    tokens.iter().any(|token| token_text(token) == "d")
723  });
724  if holds_a_d {
725    return false;
726  }
727  // nothing but factors after the group: no sign or relation outside it
728  let mut depth: i32 = 0;
729  for node in &nodes[k..] {
730    match get_grammatical_role(node).as_str() {
731      "OPEN" => depth += 1,
732      "CLOSE" => depth -= 1,
733      "ADDOP" | "RELOP" | "METARELOP" | "ARROW" if depth <= 0 => return false,
734      _ => {},
735    }
736  }
737  true
738}

751/// A `d` token's identity in its document: its font's family, series and shape — not its size, which a script
752/// changes —, the font a document keeps for its differential `d` (user ruling 2026-10-01e: the same `d` token never
753/// used as a scalar in the document is a differential).

760        font.get_series().map(|f| f.to_string()).unwrap_or_default(),
761        font.get_shape().map(|f| f.to_string()).unwrap_or_default()
762      )
763    },
764    None => resolve_xmref(node)
765      .unwrap_or_else(|| node.clone())
766      .get_attribute("_font")
767      .unwrap_or_default(),
768  }

771/// Is `d<var>` at `at` (variable ending at `end`) a scalar by its formula's own shape: `d<var>\to 0`, or a side
772/// assigned a nonzero number (`\mathrm dt=1/12`, `{\rm d}r=2.3`) when no other `d<var>` is in the formula — an
773/// equation between differentials outranks it (user ruling 2026-10-01e; 2605.09717 keeps its SDE)?
774fn scalar_by_shape(nodes: &[Node], at: usize, end: usize, other_candidates: bool) -> bool {
775  let next = |k: usize| nodes.get(k).map(token_text).unwrap_or_default();
776  if next(end) == "\u{2192}" && next(end + 1) == "0" {
777    return true;
778  }
779  if other_candidates || at != 0 || next(end) != "=" {
780    return false;
781  }
782  let rest = &nodes[end + 1..];
783  !rest.is_empty()
784    && rest.iter().all(|node| {
785      matches!(
786        get_grammatical_role(node).as_str(),
787        "NUMBER" | "ADDOP" | "MULOP" | "PUNCT"
788      ) || node.get_name() == "XMApp"
789        && node
790          .get_child_elements()
791          .first()
792          .is_some_and(|op| op.get_attribute("role").as_deref() == Some("FRACOP"))
793    })
794    && rest.iter().any(|node| {
795      get_grammatical_role(node) == "NUMBER" && token_text(node) != "0"
796        || node.get_name() == "XMApp"
797    })
798}

800/// What a document's formulas show about its `d` tokens, read before any is parsed (user rulings 2026-10-01d/e):
801/// which `d` identities are ever a scalar — a `d<var>` in a scalar context —, and
802/// which `(identity, variable)` pairs are a scalar or a differential (an integrand, a Leibniz quotient) somewhere.
803#[derive(Default)]
804pub(crate) struct DifferentialMap {
805  scalar_ids:         rustc_hash::FxHashSet<String>,
806  /// the identities a lone `d` is a scalar of somewhere (`d\geq 2`; italic only, an upright lone `d` being an
807  /// operator): the evidence `\in d<var>` gives yields to them
808  lone_scalar_ids:    rustc_hash::FxHashSet<String>,
809  scalar_pairs:       rustc_hash::FxHashSet<(String, String)>,
810  differential_pairs: rustc_hash::FxHashSet<(String, String)>,
811}
812
813impl DifferentialMap {
814  pub(crate) fn of(xmaths: &[Node], document: &Document) -> DifferentialMap {
815    let mut map = DifferentialMap::default();
816    map.absorb(xmaths, document);
817    map
818  }
819
820  /// Add what `xmaths` show.
821  pub(crate) fn absorb(&mut self, xmaths: &[Node], document: &Document) {
822    for xmath in xmaths {
823      let mut containers = vec![xmath.clone()];
824      collect_named(xmath, "XMArg", &mut containers);
825      for container in &containers {
826        self.read(container, document);
827      }
828    }
829  }
830
831  fn read(&mut self, container: &Node, document: &Document) {
832    // (the element children, hints skipped: `filter_hints` accumulates spacing onto the tokens, which the
833    // parse reads later, so this pre-pass must not run it)
834    let nodes: Vec<Node> = container
835      .get_child_elements()
836      .into_iter()
837      .filter(|node| node.get_name() != "XMHint")
838      .collect();
839    // (a `d` token, or a reference to one, as the lexer realizes it)
840    let is_a_d = |node: &Node| {
841      matches!(node.get_name().as_str(), "XMTok" | "XMRef") && token_text(node) == "d"
842    };
843    if !nodes.iter().any(is_a_d) {
844      return;
845    }
846    let context = ArgContext::of(container);
847    let levels = operand_levels(&nodes, false);
848    let pairs: Vec<(usize, usize, usize)> = (0..nodes.len())
849      .filter(|&k| is_a_d(&nodes[k]))
850      .filter_map(|k| {
851        raw_differential_variable(&nodes, k).map(|(variable, end)| (k, variable, end))
852      })
853      .collect();
854    // A lone `d` is a scalar of an italic token (`d\geq 2`, `\mathbb R^d`), unless a group, an operator or a structure
855    // follows it past its own scripts; an upright one is an operator or a label, never a scalar (`\frac{\mathrm
856    // d}{\mathrm dt}`, `\mathrm d^2=0`, `M_{\rm d}`; the 57cj.23.6 review). Only the `\in` evidence reads it.
857    for k in (0..nodes.len()).filter(|&k| is_a_d(&nodes[k])) {
858      // (a Leibniz quotient's lone `d` is its operator, `\frac{d}{dt}`)
859      if pairs.iter().any(|(at, ..)| *at == k)
860        || is_upright(&nodes[k], Some(document))
861        || context.is_leibniz()
862      {
863        continue;
864      }
865      let next = (k + 1..)
866        .find(|&j| nodes.get(j).is_none_or(|node| !is_a_script_marker(node)))
867        .unwrap_or(k + 1);
868      // (a bar introduces a group only when a bar or a `⟩` closes it later: `d|\psi\rangle`, `d|x|` — `d|n` is
869      // divisibility, its `d` a number)
870      let closed_bar = get_grammatical_role(&nodes[next.min(nodes.len().saturating_sub(1))])
871        == "VERTBAR"
872        && nodes.get(next + 1..).is_some_and(|after| {
873          after
874            .iter()
875            .any(|node| get_grammatical_role(node) == "VERTBAR" || token_text(node) == "\u{27E9}")
876        });
877      let introduces = closed_bar
878        || nodes.get(next).is_some_and(|next| {
879          next.get_name() == "XMApp"
880            || token_text(next) == "\u{27E8}"
881            || matches!(
882              get_grammatical_role(next).as_str(),
883              "OPEN" | "OPERATOR" | "OPFUNCTION" | "TRIGFUNCTION" | "FUNCTION" | "FRACOP"
884            )
885        });
886      if !introduces {
887        self
888          .lone_scalar_ids
889          .insert(d_identity(&nodes[k], Some(document)));
890      }
891    }
892    // The formula's own evidence for the `d<var>` outside its integrals, read as the lexer reads it (one verdict,
893    // `DifferentialEvidence::local`): a differential by any local evidence is the document's pair (57cj.23.7: an
894    // equation between differentials, a measure, a wedge, a form, a Leibniz quotient in any container; 2605.00250),
895    // a scalar context's `d` its scalar.
896    let in_an_integrand = in_an_enclosing_integrand(container);
897    let outside: Vec<usize> = pairs
898      .iter()
899      .map(|&(at, ..)| at)
900      .filter(|&at| integral_operand_reach(&nodes, &levels, at, false).is_none())
901      .collect();
902    let evidence =
903      DifferentialEvidence::of(&nodes, &levels, &outside, Some(document), container, false);
904    for &(at, variable, _) in &pairs {
905      let id = d_identity(&nodes[at], Some(document));
906      let pair = (id.clone(), token_text(&nodes[variable]));
907      // An integrand's `d` is a differential wherever the integral sits (an exponent's `e^{-\int r\,ds}`, a root's
908      // `\sqrt{\int|f|^2dx}`, a root or a fraction inside the integrand, `\int\sqrt{dP\,dQ}`), and never a scalar; it is
909      // evidence only where the integral reaches it cleanly — past a sign, punctuation or a plain bar the walk alone
910      // puts it in the operand (`\int f\,dx+C\,dh`, `\int f\,dx;\,d\pi`), and the reading would spread to every
911      // `d<var>` of the document.
912      if let Some(walked) = integral_operand_reach(&nodes, &levels, at, false) {
913        if !walked {
914          self.differential_pairs.insert(pair);
915        }
916        continue;
917      }
918      if in_an_integrand {
919        self.differential_pairs.insert(pair);
920        continue;
921      }
922      let Some(candidate) = evidence
923        .candidates
924        .iter()
925        .find(|candidate| candidate.at == at)
926      else {
927        continue;
928      };
929      match evidence.local(candidate) {
930        Local::Differential => {
931          self.differential_pairs.insert(pair);
932        },
933        Local::Scalar => {
934          self.scalar_ids.insert(id);
935          self.scalar_pairs.insert(pair);
936        },
937        Local::Letter | Local::Open => {},
938      }
939    }
940  }
941}

953/// The realized text of a token (through an XMRef).

1467/// `open` after the integrals and differentials of `formula` (`continues_an_open_integral`).
1468fn integral_balance(formula: &Node, mut open: usize) -> usize {

1470    .get_attribute("class")
1471    .is_some_and(|class| class.contains("ltx_math_unparsed"));
1472  let mut tokens = Vec::new();
1473  collect_tokens(formula, &mut tokens);
1474  for (k, token) in tokens.iter().enumerate() {

1500/// Is the group opening at `open` an integral operator — at its own level only integral signs, each an INTOP or a big
1501/// operator before one (`\sum_{i=1}^6\int_{a_i}^{b_i}`), with their scripts, joined by signs, no operand (user ruling
1502/// 2026-10-01: a closed group holding only integrals counts as an integral; `\left[\int_G+\sum_{i=1}^6\int_{a_i}^{b_i}\right]
1503/// f(\theta)\,\mathrm d\theta`, 2605.15451; `\left(\int_{-\infty}^{-\varepsilon}+\int_\varepsilon^\infty\right)f(z)\,\mathrm dz`,
1504/// 2605.02925)? Its CLOSE's index when it is.
1505fn integral_operator_group(nodes: &[Node], levels: &OperandLevels, open: usize) -> Option<usize> {
1506  // (a parenthesis, bracket or brace group: the grammar's `integral_operator_group` closes no other)
1507  if !matches!(token_text(&nodes[open]).as_str(), "(" | "[" | "{") {
1508    return None;
1509  }
1510  let level = levels.levels[open];
1511  let close = (open + 1..nodes.len()).find(|&k| levels.levels[k] <= level)?;
1512  if levels.roles[close] != "CLOSE" || levels.levels[close] != level {
1513    return None;
1514  }
1515  let mut integral = false;
1516  let mut expects_an_operator = true;
1517  for k in open + 1..close {
1518    // (a big operator's scripts: their bracketing nodes and content — any other application is an operand, a fraction,
1519    // a root, an accent: `\left(\int_\Omega\frac{|f|^2}{w}\right)^{1/2}` is a group, the merge review's finding)
1520    if levels.levels[k] > level + 1 || is_a_script_marker(&nodes[k]) {
1521      continue;
1522    }
1523    match levels.roles[k].as_str() {
1524      "ADDOP" if !expects_an_operator || k == open + 1 => expects_an_operator = true,
1525      "INTOP" => {
1526        integral = true;
1527        expects_an_operator = false;
1528      },
1529      "SUMOP" | "BIGOP" if expects_an_operator => {},
1530      _ => return None,
1531    }
1532  }
1533  (integral && !expects_an_operator).then_some(close)
1534}

1542/// Does the big operator at `at` bind the letter `d` in its subscript (`in_an_integral_operand`): `\sum_d`, `\sum_{d=1}`,
1543/// `\prod_{d\in D}`, `\lim_{d\to\infty}`, `\sum_{1\le d\le D}`, `\sum_{d\mid n}`, `\sum_{d,e}` — the letter, an item of a list,
1544/// or an operand of a relation, a relation chain or a condition other than its last (`\sum_{i\le d}` binds `i`)? Its scripts
1545/// follow it, each a node whose argument is parsed already or, unparsed, its tokens (then: the letter before a relation).
1546/// Read through XMRefs, as a split row's content branch holds them.
1547fn binds_the_letter_d(nodes: &[Node], at: usize) -> bool {
1548  let realized = |node: &Node| resolve_xmref(node).unwrap_or_else(|| node.clone());
1549  let is_d = |node: &Node| {
1550    let token = realized(node);
1551    token.get_name() == "XMTok" && token.get_content() == "d"
1552  };
1553  let relates = |node: &Node| {
1554    matches!(
1555      get_grammatical_role(node).as_str(),
1556      "RELOP" | "ARROW" | "MODIFIEROP"
1557    )
1558  };
1559  let binds = |script: &Node| {
1560    let Some(argument) = script.get_first_element_child() else {
1561      return false;
1562    };
1563    let mut argument = realized(&argument);
1564    if argument.get_name() == "XMArg" {
1565      let items = argument.get_child_elements();
1566      match items.as_slice() {
1567        [only] => argument = only.clone(),
1568        [first, second, ..] => return is_d(first) && relates(second),
1569        [] => return false,
1570      }
1571      argument = realized(&argument);
1572    }
1573    if argument.get_name() == "XMDual" {
1574      match argument.get_first_element_child() {
1575        Some(content) => argument = realized(&content),
1576        None => return false,
1577      }
1578    }
1579    if is_d(&argument) {
1580      return true;
1581    }
1582    if argument.get_name() != "XMApp" {
1583      return false;
1584    }
1585    let children = argument.get_child_elements();
1586    let Some((operator, operands)) = children.split_first() else {
1587      return false;
1588    };
1589    let operator = realized(operator);
1590    match operator.get_attribute("meaning").as_deref() {
1591      Some("list") => operands.iter().any(is_d),
1592      _ if operator.get_attribute("meaning").as_deref() == Some("multirelation")
1593        || relates(&operator) =>
1594      {
1595        operands
1596          .split_last()
1597          .is_some_and(|(_, leading)| leading.iter().any(is_d))
1598      },
1599      _ => false,
1600    }
1601  };
1602  let mut next = at + 1;
1603  // (its two scripts, a subscript and a superscript in either order)
1604  for _ in 0..2 {
1605    let Some(marker) = nodes.get(next) else {
1606      return false;
1607    };
1608    let script = realized(marker);
1609    let role = script.get_attribute("role");
1610    if script.get_name() != "XMApp"
1611      || !matches!(role.as_deref(), Some("POSTSUBSCRIPT" | "POSTSUPERSCRIPT"))
1612    {
1613      return false;
1614    }
1615    if role.as_deref() == Some("POSTSUBSCRIPT") && binds(&script) {
1616      return true;
1617    }
1618    // past the script: its end in a lexer stream, or itself among a formula's children or as an XMRef
1619    next = nodes[next + 1..]
1620      .iter()
1621      .position(|node| node == marker)
1622      .map_or(next + 1, |end| next + 1 + end + 1);
1623  }
1624  false
1625}

1627/// Each node's realized role and nesting level, for `in_an_integral_operand`.
1628pub(crate) struct OperandLevels {
1629  roles:  Vec<String>,
1630  levels: Vec<usize>,
1631}
1632
1633/// The nesting level of each node (`in_an_integral_operand`): an OPEN, a left bar (`\left|`, `role_side`) or, in the
1634/// lexer's `stream`, a bracketing node's first occurrence (a script's or another role-carrying application's start, which
1635/// the lexer recurses into: it occurs again as the end) opens a level its content is on; a CLOSE, a right bar or the
1636/// bracketing node's second occurrence closes it and is on the outer level — a bracketing node closes what is still open
1637/// inside it. A stray CLOSE (an unbalanced alignment cell's `\right)`) closes nothing. A formula's children (`stream`
1638/// false) hold each script as one node.
1639pub(crate) fn operand_levels(nodes: &[Node], stream: bool) -> OperandLevels {
1640  let roles: Vec<String> = nodes.iter().map(get_grammatical_role).collect();
1641  let mut levels = Vec::with_capacity(nodes.len());
1642  let mut open: Vec<Option<&Node>> = Vec::new();
1643  for (node, role) in nodes.iter().zip(&roles) {
1644    let side = || {
1645      (role == "VERTBAR")
1646        .then(|| {
1647          resolve_xmref(node)
1648            .unwrap_or_else(|| node.clone())
1649            .get_attribute("role_side")
1650        })
1651        .flatten()
1652    };
1653    if let Some(at) = open
1654      .iter()
1655      .rposition(|entry| entry.is_some_and(|entry| entry == node))
1656    {
1657      open.truncate(at);
1658      levels.push(open.len());
1659    } else if role == "OPEN"
1660      || side().as_deref() == Some("left")
1661      || (stream
1662        && node.get_name() == "XMApp"
1663        && !node.has_attribute("_rewrite")
1664        && node
1665          .get_attribute("role")
1666          .is_some_and(|role| !matches!(role.as_str(), "ARROW" | "METARELOP" | "RELOP")))
1667    {
1668      levels.push(open.len());
1669      open.push((node.get_name() == "XMApp").then_some(node));
1670    } else if (role == "CLOSE" || side().as_deref() == Some("right"))
1671      && open.last().is_some_and(Option::is_none)
1672    {
1673      open.pop();
1674      levels.push(open.len());
1675    } else {
1676      levels.push(open.len());
1677    }
1678  }
1679  OperandLevels { roles, levels }
1680}

1682fn node_to_grammar_lexemes_ctx(
1683  mathnode: &Node,
1684  child_nodes: Vec<Node>,
1685  idx: &mut usize,
1686  bigop_context: bool,
1687) -> (Vec<String>, Vec<Node>) {
1688  let mut lexemes = Vec::new();
1689  let mut nodes = Vec::new();
1690  let top_role_opt = mathnode.get_attribute("role");
1691  if let Some(ref top_role) = top_role_opt {
1692    *idx += 1;
1693    // When in bigop context, emit BIGOPSUB/BIGOPSUP instead of POSTSUBSCRIPT/POSTSUPERSCRIPT
1694    let mapped_role = if bigop_context {
1695      match top_role.as_str() {
1696        "POSTSUBSCRIPT" => "BIGOPSUB".to_string(),
1697        "POSTSUPERSCRIPT" => "BIGOPSUP".to_string(),
1698        _ => top_role.clone(),
1699      }
1700    } else {
1701      top_role.clone()
1702    };
1703    lexemes.push(format!("start_{mapped_role}:start:{idx}"));
1704    nodes.push(mathnode.clone());
1705  }
1706  // Track whether the last emitted token was a bigop (SUMOP/INTOP/etc.)
1707  // so we can emit bigop-specific script tokens to reduce earley chart ambiguity.
1708  let mut last_was_bigop = false;
1709  for node in child_nodes.into_iter() {
1710    // For XMRef nodes: resolve to get the target's role/meaning for lexing,
1711    // but keep the ORIGINAL XMRef node in the output so the parse tree
1712    // preserves XMRef indirection (matching Perl behavior).
1713    // The get_grammatical_role/get_token_meaning functions already resolve XMRef
1714    // internally, so we just need to handle the case where the target is a
1715    // compound node (XMApp without role) — these need flattening.
1716    // Note: we do NOT replace the node variable — we keep the XMRef.
1717
1718    if node.get_name() == "XMApp" && node.get_attribute("role").is_some() {
1719      let role = node.get_attribute("role").unwrap();
1720      // A complete operator that has been decorated or scripted is still a
1721      // single grammatical operator and must lex as ONE atomic terminal:
1722      //   - ARROW/METARELOP: decorated arrows (\xrightarrow{over},
1723      //     \xleftrightarrow[under]{over}).
1724      //   - RELOP: a scripted relation (\stackrel{?}{=}, \overset/\underset over
1725      //     a relation) — `=^?` is an `<XMApp role=RELOP>` whose recursion would
1726      //     emit `start_RELOP SUPERSCRIPTOP RELOP:= UNKNOWN:? end_RELOP`, a
1727      //     sequence no grammar rule consumes, so the relation failed to parse as
1728      //     a standalone item (e.g. `a \quad \stackrel{?}{=} \quad b` →
1729      //     ltx_math_unparsed). Perl recurses with start_/end_ markers its Marpa
1730      //     grammar handles; the Rust grammar treats decorated operators as
1731      //     atomic terminals instead (the whole node is preserved as the
1732      //     terminal's value, so rendering still shows the script). The lexeme
1733      //     keys on the inner base operator's meaning, so `=^?` lexes like `=`.
1734      if role == "ARROW" || role == "METARELOP" || role == "RELOP" {
1735        let op_meaning = node
1736          .get_child_elements()
1737          .into_iter()
1738          .find(|ch| {
1739            let cr = ch.get_attribute("role");
1740            match role.as_str() {
1741              // A scripted relation's base is a RELOP leaf.
1742              "RELOP" => cr.as_deref() == Some("RELOP"),
1743              // Decorated arrows may carry either an ARROW or METARELOP base.
1744              _ => cr.as_deref() == Some("ARROW") || cr.as_deref() == Some("METARELOP"),
1745            }
1746          })
1747          .and_then(|ch| {
1748            ch.get_attribute("meaning")
1749              .or_else(|| ch.get_attribute("name"))
1750              .or_else(|| Some(ch.get_content()))
1751          })
1752          .unwrap_or_else(|| role.to_string());
1753        *idx += 1;
1754        let lexeme = grammar_lexeme(&role, &op_meaning, *idx);
1755        lexemes.push(lexeme);
1756        nodes.push(node);
1757      } else if node.has_attribute("_rewrite") {
1758        // Rewrite-created or a parse_children replacement: treat as atomic
1759        // token with the assigned role. Don't recurse — the inner structure
1760        // was pre-parsed, and the role on this node overrides whatever the
1761        // children contain. An atomic bigop (e.g. `\mathop{...}` → BIGOP)
1762        // must set bigop context so a following script lexes as
1763        // BIGOPSUB/BIGOPSUP, same as the plain-token arm below.
1764        let gram_role = get_grammatical_role(&node);
1765        let mut text = get_token_meaning(&node);
1766        if text.is_empty() {
1767          text = "UNKNOWN".to_string();
1768        }
1769        *idx += 1;
1770        last_was_bigop = matches!(
1771          gram_role.as_str(),
1772          "SUMOP" | "INTOP" | "LIMITOP" | "DIFFOP" | "BIGOP"
1773        );
1774        lexemes.push(grammar_lexeme(&gram_role, &text, *idx));
1775        nodes.push(node);
1776      } else {
1777        // Only recurse into XMApp nodes that have a role (scripts, etc.)
1778        // Role-less XMApps (e.g. \sqrt, already-parsed structures) are atomic.
1779        // Pass bigop_context for POSTSUBSCRIPT/POSTSUPERSCRIPT following a bigop
1780        let is_script = matches!(role.as_str(), "POSTSUBSCRIPT" | "POSTSUPERSCRIPT");
1781        let ctx = last_was_bigop && is_script;
1782        let children = filter_hints(node.get_child_nodes());
1783        let (mut inner_lexes, mut inner_nodes) =
1784          node_to_grammar_lexemes_ctx(&node, children, idx, ctx);
1785        for (inner_lex, inner_node) in inner_lexes.drain(..).zip(inner_nodes.drain(..)) {
1786          lexemes.push(inner_lex);
1787          nodes.push(inner_node);
1788        }
1789        // Script following a bigop is still "bigop context" for the next script
1790        if !is_script {
1791          last_was_bigop = false;
1792        }
1793      }
1794    } else if node.get_name() == "XMArg" {
1795      // XMArg is a transparent wrapper (e.g. from \lx@post@subscript).
1796      // Recurse into its children so the grammar can parse them individually.
1797      // E.g. _{ij} should emit UNKNOWN:i + UNKNOWN:j, not ATOM:ij.
1798      let arg_children = filter_hints(node.get_child_nodes());
1799      let (mut inner_lexes, mut inner_nodes) =
1800        node_to_grammar_lexemes_ctx(&node, arg_children, idx, false);
1801      for (inner_lex, inner_node) in inner_lexes.drain(..).zip(inner_nodes.drain(..)) {
1802        lexemes.push(inner_lex);
1803        nodes.push(inner_node);
1804      }
1805    } else {
1806      let mut role = get_grammatical_role(&node);
1807      // An accented differential operator is one (57cj review): `\partial\bar\partial\phi` is ∂(∂̄φ)
1808      // (2605.01526, 2605.15276, 2605.01646), where Perl reads the accent's application an ATOM that
1809      // `\partial` multiplies. It keeps its text (`DIFFOP:¯∂`), so it is no plain `\partial`.
1810      if role == "ATOM" && is_accented_differential_operator(&node) {
1811        role = "DIFFOP".to_string();
1812      }
1813      // A fraction of numbers (`\frac12`, `\tfrac{3}{4}`) is an ATOM that is a number: its own category,
1814      // `ATOM_NUMBER`, which the grammar's `numeric_monomial` leads (57cj.1 review, latent: its probes; the
1815      // actions read it as an ATOM, `semantics::lexeme_category`). Perl reads it an ATOM. So is any constant atom
1816      // (`is_numeric_constant`: `\sqrt2`, `\frac{\pi}{2}`, `\sqrt[3]{2}`, `\frac{\pi^2}{6}`; the actions' notion,
1817      // `semantics::is_constant`; 57cj.7, 57cj.8 reviews, latent: their probes).
1818      if role == "ATOM" && is_numeric_constant(&node) {
1819        role = "ATOM_NUMBER".to_string();
1820      }
1821      let mut text = get_token_meaning(&node);
1822      if text.is_empty() {
1823        text = "UNKNOWN".to_string();
1824      }
1825      // Perl's grammar terminal for a bar is content-agnostic (`/VERTBAR:\S*:\d+/`,
1826      // MathGrammar:797) and its fences key on the bar's content (`'|@|'`, MathParser.pm:1361);
1827      // Rust's rules key the bar lexeme, so a bar with a name but no meaning (`\bigl\lvert`,
1828      // `\arrowvert`, `\bigl\lVert`) reads by its glyph: `|` single, `∥`/`‖` double. The
1829      // bar is read through an XMRef (a gathered/split row's content branch), as the role is
1830      // (arXiv 2605.02713 `\arrowvert\psi_n\arrowvert^2`, 2605.19292 `\bigl\lvert
1831      // a_{ij}\bigr\rvert`). Golden
1832      // tests/parse/bar_pairs.tex#named_vertbar_parses_as_absolute_value.
1833      if role == "VERTBAR" {
1834        let bar = resolve_xmref(&node).unwrap_or_else(|| node.clone());
1835        if bar.get_attribute("meaning").is_none() {
1836          match bar.get_content().as_str() {
1837            "|" => text = "|".to_string(),
1838            "\u{2225}" | "\u{2016}" => text = "||".to_string(),
1839            _ => {},
1840          }
1841        }
1842      }
1843      *idx += 1;
1844      // Track bigop tokens for bigop-specific script token emission
1845      let is_bigop = matches!(
1846        role.as_str(),
1847        "SUMOP" | "INTOP" | "LIMITOP" | "DIFFOP" | "BIGOP"
1848      );
1849      last_was_bigop = is_bigop;
1850      // Normalize langle/rangle meaning for consistent grammar matching.
1851      // Do NOT remap to parentheses — langle_open/rangle_close tokens need
1852      // distinct identity for QM bra-ket rules vs conditional probability.
1853      //
1854      // Perf: Split generic OPEN/CLOSE into OTHER_OPEN/OTHER_CLOSE for delimiters
1855      // OTHER than parens/brackets/braces/langle. This prevents the grammar's
1856      // generic `open ~ "OPEN"` prefix match from ALSO matching the specific
1857      // `lparen = "OPEN:("`, etc. — which was producing 2× duplicate grammar
1858      // derivations for every `(x)`, `[x]`, `{x}` expression.
1859      let lexeme = if role == "OPEN" && (text == "⟨" || text == "langle") {
1860        format!("OPEN:langle:{idx}")
1861      } else if role == "CLOSE" && (text == "⟩" || text == "rangle") {
1862        format!("CLOSE:rangle:{idx}")
1863      } else if matches!(role.as_str(), "OPEN" | "CLOSE")
1864        && text == "||"
1865        && resolve_xmref(&node)
1866          .unwrap_or_else(|| node.clone())
1867          .get_attribute("stretchy")
1868          .as_deref()
1869          == Some("true")
1870      {
1871        // A `\left`/`\right` double bar — `\left\|`, `\left\Vert`, `\left\lVert` — is a bar
1872        // pair in Perl: `augmentDelimiterProperties` re-roles its ∥ a VERTBAR on either side
1873        // (DELIMITER_MAP, TeX_Math.pool.ltxml:755, :813-816), so `VERTBAR absExpression
1874        // VERTBAR` reads it as an `aBarearg` (MathGrammar:329-330) — `\exp\left\|x\right\|^2` is
1875        // exp@(‖x‖²), not a group `\exp` applies to. DELIMITER_MAP here keeps its OPEN/CLOSE
1876        // role (`name="||"`) for the markup; its grammar view is the stretchy bar's, sided by
1877        // the role. Read through an XMRef, as a split row's content branch holds it: unlike a
1878        // single `|` (below) no divider or bra/ket rule reads a double bar, and on the XMRef itself
1879        // a split row's `\left\|` would lex `OTHER_OPEN:||`, a group `\exp` applies to (57am
1880        // review round 10). Golden tests/parse/bar_pairs.tex#left_double_bar_is_a_bar_pair.
1881        if role == "OPEN" {
1882          format!("LEFT_STRETCHY_VERTBAR:||:{idx}")
1883        } else {
1884          format!("RIGHT_STRETCHY_VERTBAR:||:{idx}")
1885        }
1886      } else if role == "OPEN" && !matches!(text.as_str(), "(" | "[" | "{") {
1887        grammar_lexeme("OTHER_OPEN", &text, *idx)
1888      } else if role == "CLOSE" && !matches!(text.as_str(), ")" | "]" | "}") {
1889        grammar_lexeme("OTHER_CLOSE", &text, *idx)
1890      } else if role == "UNKNOWN" && text == "d" {
1891        // M4: Emit XDIFFUNK for possible differential-d tokens.
1892        // Only "d" tokens can be diffops; other unknowns skip the diffop rule.
1893        format!("XDIFFUNK:{text}:{idx}")
1894      } else if role == "ID" && text == "d" {
1895        format!("XDIFFID:{text}:{idx}")
1896      } else if role == "VERTBAR"
1897        && text == "|"
1898        && node.get_attribute("stretchy").as_deref() == Some("true")
1899      {
1900        // `\left|...\right|` produces a balanced pair of VERTBAR tokens
1901        // with `stretchy="true"` (whereas bare `|x|` is `stretchy="false"`).
1902        // Further distinguish by side: the `\lx@delim@left` constructor tags
1903        // the emitted XMTok with `role_side="left"`, `\lx@delim@right` with
1904        // `role_side="right"` (tex_math.rs). This refines the `role`
1905        // attribute (delimiter direction) without changing it, so
1906        // the grammar can use distinct LEFT_STRETCHY_VERTBAR /
1907        // RIGHT_STRETCHY_VERTBAR tokens. The kerned-stack norm rules
1908        // (\vertii, \vertiii, …) then don't have to enumerate every
1909        // pairing of identical bars. Eliminates the VERTBAR-pairing
1910        // combinatorial explosion in patterns like
1911        // `\log^+ ∫ \left| f \right|^k dm(z) ≲ ...` (see
1912        // docs/archive/MATH_AMBIGUITY_AUDIT_2026-05-21.md §2) and unlocks task #263's
1913        // norm-fenced grammar rules.
1914        // Defensive fallback: if `role_side` is missing — legacy DOM
1915        // input or a path that bypassed `\lx@delim@left`/`\lx@delim@right` — keep the
1916        // old undirected STRETCHY_VERTBAR lexeme so legacy rules
1917        // (eval_at, modulus fence) still work.
1918        // Read on the node itself: a split row's content branch holds XMRefs, which lex as the
1919        // unsided `VERTBAR:|`. The divider and bra/ket rules take the sided bar too since 57ap
1920        // (`divider_bar`, `bra_bar`, `ket_bar`), so a split row may read its bars sided next — SYNC
1921        // "A split row reads its bars unsided"; 57am review round 8: sided before 57ap, the split rows
1922        // of 2605.11264 and 2605.20326 lost their parse.
1923        match node.get_attribute("role_side").as_deref() {
1924          Some("left") => format!("LEFT_STRETCHY_VERTBAR:|:{idx}"),
1925          Some("right") => format!("RIGHT_STRETCHY_VERTBAR:|:{idx}"),
1926          _ => format!("STRETCHY_VERTBAR:|:{idx}"),
1927        }
1928      } else if role == "VERTBAR" && (text == "mid" || text == "\u{2223}") {
1929        // Canonicalize the single "divides / such-that / given" bar to one
1930        // grammar token. `\mid` (∣ U+2223), bare `|` (U+007C) and `\vert`
1931        // (`\Let` to `|`) are all the SAME bar grammatically, but they reach
1932        // the lexer with different content: `|`/`\vert` carry "|" while `\mid`
1933        // carries either the glyph ∣ or the CS name "mid" — the latter because
1934        // `def_math` infers a name from the CS (`binding/def/dialect.rs`)
1935        // whereas Perl's `DefMathI('\mid', undef, "\x{2223}", role=>'VERTBAR')`
1936        // (math_common.pool.ltxml:290) leaves the name undef. Either way the
1937        // lexeme came out as `VERTBAR:mid` / `VERTBAR:∣`, NOT `VERTBAR:|`.
1938        // Every divides-bar grammar rule (modulus `|x|`, conditional `(a|b)`,
1939        // set-builder `{x|P}`, Dirac bra/ket/braket) keys on the exact
1940        // `singlevertbar = VERTBAR:|` terminal, so `\mid` inside any fence
1941        // failed to parse (→ ltx_math_unparsed) even though `|` did — breaking
1942        // ubiquitous notation: conditional probability `P(A\mid B)`,
1943        // set-builder `\{x\mid x>0\}`, Dirac `\langle a\mid b\rangle`.
1944        // Perl's grammar terminal is content-agnostic (`/VERTBAR:\S*:\d+/`,
1945        // MathGrammar:797), so Perl parses all of these; Rust's is
1946        // glyph-specific, so we canonicalize here instead. The rendered glyph
1947        // is unaffected — the XMTok node keeps its ∣ content; the lexeme is a
1948        // parallel grammar-only view looked up by idx. (Doubled bars
1949        // `\|`/`\Vert` → `VERTBAR:||` and `\parallel` → `VERTBAR:parallel-to`
1950        // are NOT single divides bars; they keep their content for the
1951        // norm / parallel rules.)
1952        format!("VERTBAR:|:{idx}")
1953      } else if role == "PUNCT" && punct_followed_by_wide_space(&node) {
1954        // PUNCT followed by `\quad` / `\qquad` etc. carries an `rpadding`
1955        // attribute. This wide spacing is a strong arXiv idiom for
1956        // "formula separator (with side-condition)" rather than
1957        // "list-element separator". Tag as WIDE_PUNCT so the grammar
1958        // can prefer `formulae_apply` over `list_apply` for this
1959        // separator unambiguously. See docs/archive/MATH_AMBIGUITY_AUDIT_2026-05-21.md §2.
1960        format!("WIDE_PUNCT:,:{idx}")
1961      } else {
1962        grammar_lexeme(&role, &text, *idx)
1963      };
1964      lexemes.push(lexeme);
1965      nodes.push(node);
1966    }
1967  }
1968  if let Some(ref top_role) = top_role_opt {
1969    *idx += 1;
1970    let mapped_end = if bigop_context {
1971      match top_role.as_str() {
1972        "POSTSUBSCRIPT" => "BIGOPSUB".to_string(),
1973        "POSTSUPERSCRIPT" => "BIGOPSUP".to_string(),
1974        _ => top_role.clone(),
1975      }
1976    } else {
1977      top_role.clone()
1978    };
1979    lexemes.push(format!("end_{mapped_end}:end:{idx}"));
1980    nodes.push(mathnode.clone());
1981  }
1982  (lexemes, nodes)
1983}

2094/// Auxiliary separator for ROLE:style-lexeme into ("ROLE:style", '-', lexeme)
2095pub fn distill_lexeme(name: &str) -> (&str, &str, &str) {

2097  // only trying to distill the last pure lexeme

2111  }

2114/// Parse an XMHint `width` attribute string to points.
2115/// `pub(crate)` so semantics.rs can re-use the same parser when
2116/// inspecting `rpadding` (task #263 stretchy_norm_fenced guards).
2117/// Supports "3.0mu" (mu → pt by dividing by 1.8) and "1.667pt"/"0.16667em".
2118/// Handles glue specs like "2.77pt plus 2.77pt" by extracting base dimension.
2119pub(crate) fn get_xmhint_spacing(width: &str) -> f64 {
2120  let width = width.trim();
2121  if width.is_empty() {
2122    return 0.0;
2123  }
2124  // Strip glue stretch/shrink: "2.77pt plus 2.77pt" → "2.77pt"
2125  let base = width
2126    .split_once(" plus")
2127    .or_else(|| width.split_once(" minus"))
2128    .map_or(width, |(base, _)| base)
2129    .trim();
2130  let unit_start = base.find(|c: char| c.is_alphabetic()).unwrap_or(base.len());
2131  let (number_str, unit) = base.split_at(unit_start);
2132  let number: f64 = number_str.trim().parse().unwrap_or(0.0);
2133  match unit.trim() {
2134    "mu" => number / 1.8,
2135    "pt" => number,
2136    "em" => number * 10.0, // assume 10pt font size
2137    _ => 0.0,
2138  }
2139}

2141/// True iff the given PUNCT-like XMTok carries an `rpadding` attribute
2142/// large enough to indicate a `\quad`-class spacing — TeX's idiom for
2143/// separating a main formula from a side-condition (e.g.
2144/// `A = B, \quad r \notin E`). Threshold of ≥10pt catches `\quad` (10pt)

2159  }

2162/// Filter XMHint nodes from a list of child nodes, transferring their spacing
2163/// info to adjacent tokens as `lpadding`/`rpadding` attributes (Perl MathParser.pm
2164/// `filter_hints` L417-491). XMHints are also unlinked from the XML tree so they won't be
2165/// seen again. Large spacings (≥10pt, e.g. \quad) become virtual PUNCT nodes (L476-487),
2166/// unless a phantom (`name` ending in `phantom`, L443) contributed to them (arXiv 2605.24688).
2167/// An APPLYOP token is kept but takes no space (L458-461): "they tend to disappear". A space
2168/// pending at the next node is its `lpadding`, negative (`\!`) too (L466). The OPEN and PUNCT
2169/// tests read the realized role, where Perl reads the node's own (L445, L482-483): in a
2170/// gathered/split row's content branch the nodes are XMRefs, and Perl's `,\quad` there became
2171/// two punctuations, leaving the formula unparsed (KNOWN_PERL_ERRORS #365, OXIDIZED_DESIGN
2172/// #349; golden tests/parse/aligned_content_branch.tex#content_branch_reads_its_delimiters).
2173pub fn filter_hints(nodes: Vec<Node>) -> Vec<Node> {
2174  const HINT_PUNCT_THRESHOLD: f64 = 10.0;
2175  let mut prefiltered: Vec<Node> = Vec::new();
2176  // Perl's `$prev`: the last kept node that is not an APPLYOP.
2177  let mut prev: Option<usize> = None;
2178  let mut pending_space: f64 = 0.0;
2179  let mut pending_phantom = false;
2180  // Save hint nodes that contributed to _space, for possible PUNCT reuse
2181  let mut last_hint_for: Vec<Option<Node>> = Vec::new(); // parallel to prefiltered
2182  // Perl's `_phantom`, parallel to prefiltered
2183  let mut phantom: Vec<bool> = Vec::new();
2184
2185  for mut node in nodes {
2186    if node.get_type() != Some(NodeType::ElementNode) {
2187      continue;
2188    }
2189    if node.get_name() == "XMHint" {
2190      if let Some(width_str) = node.get_attribute("width") {
2191        let pts = get_xmhint_spacing(&width_str);
2192        if pts != 0.0 {
2193          let ph = node
2194            .get_attribute("name")
2195            .is_some_and(|name| name.ends_with("phantom"));
2196          match prev {
2197            Some(idx) if get_grammatical_role(&prefiltered[idx]) != "OPEN" => {
2198              let prev = &mut prefiltered[idx];
2199              let s: f64 = prev
2200                .get_attribute("_space")
2201                .and_then(|v| v.parse().ok())
2202                .unwrap_or(0.0);
2203              let _ = prev.set_attribute("_space", &format!("{}", s + pts));
2204              phantom[idx] |= ph;
2205              // Save this hint node for potential PUNCT reuse
2206              last_hint_for[idx] = Some(node.clone());
2207            },
2208            _ => {
2209              pending_space += pts;
2210              pending_phantom = ph;
2211            },
2212          }
2213        }
2214      }
2215      // Unlink from the XML tree; XMHints are ephemeral
2216      node.unlink();
2217    } else if node.get_name() == "XMTok" && node.get_attribute("role").as_deref() == Some("APPLYOP")
2218    {
2219      prefiltered.push(node);
2220      last_hint_for.push(None);
2221      phantom.push(false);
2222    } else {
2223      let mut ph = false;
2224      if pending_space != 0.0 {
2225        let _ = node.set_attribute("lpadding", &format!("{:.1}pt", pending_space));
2226        ph = pending_phantom;
2227        pending_space = 0.0;
2228        pending_phantom = false;
2229      }
2230      prev = Some(prefiltered.len());
2231      prefiltered.push(node);
2232      last_hint_for.push(None);
2233      phantom.push(ph);
2234    }
2235  }
2236
2237  // Second pass: convert _space to rpadding (or PUNCT XMHint if above threshold)
2238  let mut filtered: Vec<Node> = Vec::new();
2239  for (i, mut node) in prefiltered.into_iter().enumerate() {
2240    filtered.push(node.clone());
2241    if let Some(s_str) = node.get_attribute("_space") {
2242      let _ = node.remove_attribute("_space");
2243      let s: f64 = s_str.parse().unwrap_or(0.0);
2244      if !phantom[i] && s >= HINT_PUNCT_THRESHOLD && get_grammatical_role(&node) != "PUNCT" {
2245        // Perl MathParser.pm L487: create virtual PUNCT XMHint
2246        // Reuse the saved hint node, setting role="PUNCT"
2247        if let Some(Some(mut hint)) = last_hint_for.get(i).cloned() {
2248          let _ = hint.set_attribute("role", "PUNCT");
2249          // Clean width: round to integer if close, matching Perl format
2250          let s_rounded = if (s - s.round()).abs() < 0.01 {
2251            s.round()
2252          } else {
2253            s
2254          };
2255          let width = format!("{}pt", s_rounded);
2256          let _ = hint.set_attribute("width", &width);
2257          // Remove extraneous attributes from the reused hint node
2258          let _ = hint.remove_attribute("depth");
2259          let _ = hint.remove_attribute("height");
2260          let quads = "q".repeat((s / 10.0) as usize);
2261          let _ = hint.set_attribute("name", &format!("{quads}uad"));
2262          filtered.push(hint);
2263        }
2264      } else {
2265        let _ = node.set_attribute("rpadding", &format!("{:.1}pt", s));
2266      }
2267    }
2268  }
2269  filtered
2270}

2272/// Given a list of XML nodes (either libxml nodes, or array representations)
2273/// return a list of XMRef's referring to those nodes;
2274/// ensure each source node has an ID (if already instanciated as XML)
2275/// or _xmkey if still in array rep. since it will get an ID later, and the connection re-made)
2276/// Note that ltx:XMHint nodes are ephemeral and shouldn't be ref'd!
2277/// likewise, we avoid creating XMRefs to XMRefs
2278pub fn create_xmrefs(args: &mut [&mut XM], ctxt: ActionContext) -> Result<Vec<XM>, Box<dyn Error>> {
2279  let nodes = ctxt.nodes;
2280  let document = ctxt.document;
2281  let mut refs = Vec::with_capacity(args.len());
2282  for arg in args {
2283    match arg {
2284      XM::Token(props, _meta) => {
2285        if let Some(id) = props.id.as_ref() {
2286          refs.push(XM::Ref(XProps {
2287            id: Some(id.clone()),
2288            ..XProps::default()
2289          }));
2290        } else if let Some(existing) = props.xmkey.clone() {
2291          // Preserve an existing _xmkey (see the XM::Apply arm) — faithful to
2292          // Perl `createXMRefs`, which never clobbers an arg's `_xmkey`.
2293          refs.push(XM::Ref(XProps {
2294            xmkey: Some(existing),
2295            ..XProps::default()
2296          }));
2297        } else {
2298          // Parser-created token without id — use _xmkey for deferred resolution
2299          let key = get_xmarg_id()?.to_string();
2300          props.xmkey = Some(Cow::Owned(key.clone()));
2301          refs.push(XM::Ref(XProps {
2302            xmkey: Some(Cow::Owned(key)),
2303            ..XProps::default()
2304          }));
2305        }
2306      },
2307      XM::Lexeme(lex, _) => {
2308        // If arg is already XML, it's too late to get automatic ID's.
2309        // lookup_lex_node now returns Err for malformed lex strings instead
2310        // of panicking; skip the arg on failure rather than abort the whole
2311        // ref-building pass.
2312        let node = match lookup_lex_node(lex, nodes) {
2313          Ok(n) => n,
2314          Err(e) => {
2315            // Perl MathParser.pm:151 — Error('expected', 'id', undef,
2316            //   "Cannot find a node with xml:id='$idref'", ...)
2317            // We don't always have an idref string at this layer; the
2318            // lookup error itself carries enough context.
2319            log_math_error!(
2320              "expected",
2321              "id",
2322              "create_xmrefs: skipping lexeme with invalid node lookup: {}",
2323              e
2324            );
2325            continue;
2326          },
2327        };
2328
2329        // Perl `createXMRefs` (Package.pm L1557-1561): "clone an XMRef
2330        // rather than create an XMRef to an XMRef" — reuse the target's
2331        // idref so a content branch never points at a presentation-side
2332        // reference. Without this branch the ref-to-a-ref shape reaches
2333        // the output (Perl's whole golden corpus has zero of them).
2334        if node.get_name() == "XMRef" {
2335          // Carry BOTH spellings, as Perl does — `_xmkey` is the deferred
2336          // form used before ids exist, `idref` the resolved one, and either
2337          // may be absent. Taking this branch unconditionally matters: on a
2338          // key-only XMRef, falling through would mint an xml:id ON the ref
2339          // and reference THAT, which is the ref-to-a-ref this branch exists
2340          // to prevent.
2341          refs.push(XM::Ref(XProps {
2342            id: node.get_attribute("idref").map(Cow::Owned),
2343            xmkey: node.get_attribute("_xmkey").map(Cow::Owned),
2344            ..XProps::default()
2345          }));
2346          continue;
2347        }
2348
2349        // NS-aware read (xml:id = local "id" in XML_NS; the bare
2350        // get_attribute("xml:id") form always returns None, which sent
2351        // every node — id or not — through the generate_id branch).
2352        match node.get_attribute_ns("id", latexml_core::common::xml::XML_NS) {
2353          //  already has id, so refer to it.
2354          Some(id) => refs.push(XM::Ref(XProps {
2355            id: Some(Cow::Owned(id)),
2356            ..XProps::default()
2357          })),
2358          None => {
2359            // Generate xml:id for this node so we can reference it
2360            document.generate_id(&mut node.clone(), "")?;
2361            if let Some(id) = node.get_attribute_ns("id", latexml_core::common::xml::XML_NS) {
2362              refs.push(XM::Ref(XProps {
2363                id: Some(Cow::Owned(id)),
2364                ..XProps::default()
2365              }));
2366            }
2367          },
2368        }
2369      },
2370      XM::Apply(_op, _args, props, _meta) => {
2371        if let Some(id) = props.id.as_ref() {
2372          refs.push(XM::Ref(XProps {
2373            id: Some(id.clone()),
2374            ..XProps::default()
2375          }));
2376        } else if let Some(existing) = props.xmkey.clone() {
2377          // PRESERVE an existing _xmkey. Faithful to Perl `createXMRefs`
2378          // (Package.pm:1527): it `GenerateID`s an XML arg but NEVER clobbers
2379          // its `_xmkey`, so a node that is ALREADY an arg of an outer
2380          // package XMDual (via `\lx@xmarg`, e.g. braket `\braket{a|op|c}`)
2381          // keeps that key while also being referenced by THIS (grammar) dual
2382          // — both content XMRefs resolve to the same shared node. Overwriting
2383          // the key (as before) dangled the package's XMRef →
2384          // `expected:id Cannot find a node` (compound op; witness 2205.06843).
2385          refs.push(XM::Ref(XProps {
2386            xmkey: Some(existing),
2387            ..XProps::default()
2388          }));
2389        } else {
2390          // not yet instanciated, so hasn't had chance to get auto-id; use _xmkey
2391          let key = get_xmarg_id()?.to_string();
2392          props.xmkey = Some(Cow::Owned(key.clone()));
2393          refs.push(XM::Ref(XProps {
2394            xmkey: Some(Cow::Owned(key)),
2395            ..XProps::default()
2396          }));
2397        }
2398      },
2399      // clone an XMRef (w/o any attributes or id ?) rather than create an XMRef to an XMRef:
2400      // only its `_xmkey` and `idref` (Package.pm:1557-1561), not the role/meaning a parse
2401      // annotated it with.
2402      XM::Ref(props) => {
2403        refs.push(XM::Ref(XProps {
2404          id: props.id.clone(),
2405          xmkey: props.xmkey.clone(),
2406          ..XProps::default()
2407        }));
2408      },
2409      XM::Dual(_, _, props, _) | XM::Wrap(_, props, _) => {
2410        if let Some(id) = props.id.as_ref() {
2411          refs.push(XM::Ref(XProps {
2412            id: Some(id.clone()),
2413            ..XProps::default()
2414          }));
2415        } else if let Some(existing) = props.xmkey.clone() {
2416          // Preserve an existing _xmkey (see the XM::Apply arm) — faithful to
2417          // Perl `createXMRefs`, which never clobbers an arg's `_xmkey`.
2418          refs.push(XM::Ref(XProps {
2419            xmkey: Some(existing),
2420            ..XProps::default()
2421          }));
2422        } else {
2423          let key = get_xmarg_id()?.to_string();
2424          props.xmkey = Some(Cow::Owned(key.clone()));
2425          refs.push(XM::Ref(XProps {
2426            xmkey: Some(Cow::Owned(key)),
2427            ..XProps::default()
2428          }));
2429        }
2430      },
2431      _ => {
2432        // XMHint's are ephemeral — clone without id
2433        // Other variants: skip with warning
2434      },
2435    }
2436  }
2437  Ok(refs)
2438}

2440/// A grammar lexeme `ROLE:text:index`, whitespace removed as Perl does (`s/\s//g`, MathParser.pm:
2441/// 803-804): a text token holding a newline or an em-space no longer breaks its lexeme, and a colon
2442/// in the text is the grammar's to allow (`lex_char`), the index being the last `:digits` (57bm;
2443/// `\text{(OCP): }R(s)`, 2605.11979, 2605.03556) — a text's colon, or a parsed script's, which
2444/// reaches the formula as one ATOM holding its content (`\sum_{i:a_i>0}x_i`, `ATOM::i>ai0:3`). A text of whitespace only, which Perl's `\S*`
2445/// matches empty, is the placeholder `␣` (`BLANK_LEXEME_TEXT`) — the grammar's text needs one
2446/// character.
2447pub(crate) fn grammar_lexeme(role: &str, text: &str, idx: usize) -> String {
2448  let strip = |s: &str| s.chars().filter(|c| !c.is_whitespace()).collect::<String>();
2449  let text = strip(text);
2450  let text = if text.is_empty() {
2451    BLANK_LEXEME_TEXT.to_string()
2452  } else {
2453    text
2454  };
2455  format!("{}:{text}:{idx}", strip(role))
2456}
2457
2458/// The lexeme text of a token whose text is whitespace only (`grammar_lexeme`).
2459pub(crate) const BLANK_LEXEME_TEXT: &str = "\u{2423}";
