175/// Parse a preload spec into `(name, ext, options)`.
176///
177/// Mirrors Perl `Core.pm:initializeState` (regexes
178/// `s/^\[([^\]]*)\]//` then `s/\.(\w+)$//`): the option bracket
179/// comes at the *front*, e.g. `[ids,mathlexemes]latexml.sty`.
180/// Defaults to `ext = "sty"` when the spec has no `.<ext>` suffix.
181pub(crate) fn parse_preload_spec(preload: &str) -> (String, String, Vec<String>) {
182  let (base, options) = match preload
183    .strip_prefix('[')
184    .and_then(|rest| rest.find(']').map(|end| (&rest[..end], &rest[end + 1..])))
185  {
186    Some((opts_str, rest)) => {
187      let opts: Vec<String> = opts_str
188        .split(',')
189        .map(|s| s.trim().to_string())
190        .filter(|s| !s.is_empty())
191        .collect();
192      (rest.to_string(), opts)
193    },
194    None => (preload.to_string(), vec![]),
195  };
196  let (name, ext) = match base.rfind('.') {
197    Some(pos) => (base[..pos].to_string(), base[pos + 1..].to_string()),
198    None => (base.clone(), String::from("sty")),
199  };
200  (name, ext, options)
201}

203/// One guarded digestion step: `digest_next_body(None)` under the salvage
204/// policy `digest_internal` has always applied (extracted verbatim so the
205/// eager loop and the streaming driver share ONE policy). `Ok(true)` =
206/// continue; `Ok(false)` = a Fatal was recovered (announced + latched, boxes
207/// salvaged) and digestion must stop; `Err` = a resource fatal that must not
208/// be recovered.
209/// Can we create files in `dir`? Probes by creating and removing a uniquely
210/// named entry — the only answer that is true for the actual operation, since
211/// permission bits, read-only mounts, and full filesystems all present
212/// differently and `metadata().permissions()` sees none of them reliably.
213fn dir_is_writable(dir: &Path) -> bool {
214  let probe = dir.join(format!(".latexml-writable-{}", std::process::id()));
215  match std::fs::File::create(&probe) {
216    Ok(_) => {
217      let _ = std::fs::remove_file(&probe);
218      true
219    },
220    Err(_) => false,
221  }
222}

224/// Diagnostic (`LXML_TRACE_FRAGMENT_RC`): after a fragment is absorbed, every
225/// box in it should be referenced by the fragment alone (plus `node_boxes`,
226/// which the streaming sweep drops). Report the boxes with extra holders, by
227/// kind and depth, so a retention can be attributed.
228fn trace_fragment_holders(boxes: &[Digested]) {
229  use latexml_core::digested::DigestedData;
230  fn walk(
231    d: &Digested,
232    depth: usize,
233    expected: usize,
234    out: &mut Vec<String>,
235    seen: &mut rustc_hash::FxHashSet<usize>,
236  ) {
237    // A whatsit reaches its content through an argument AND a property (the
238    // same handle twice): visit each allocation once, or the walk is
239    // exponential in the nesting depth.
240    if !seen.insert(d.as_ptr() as usize) {
241      return;
242    }
243    let (kind, extra) = match d.data() {
244      DigestedData::TBox(_) => ("TBox", String::new()),
245      DigestedData::List(l) => ("List", format!(" n={}", l.borrow().boxes.len())),
246      DigestedData::Whatsit(w) => ("Whatsit", format!(" {}", w.borrow().definition.get_cs())),
247      DigestedData::Alignment(_) => ("Alignment", String::new()),
248      _ => ("other", String::new()),
249    };
250    let rc = d.strong_count();
251    if rc > expected && out.len() < 40 {
252      out.push(format!(
253        "  depth {depth} rc {rc} (expected {expected}) {kind}{extra}"
254      ));
255    }
256    match d.data() {
257      DigestedData::List(l) => {
258        for c in &l.borrow().boxes {
259          walk(c, depth + 1, 1, out, seen);
260        }
261      },
262      DigestedData::Whatsit(w) => {
263        let w = w.borrow();
264        for a in w.args.iter().flatten() {
265          walk(a, depth + 1, 1, out, seen);
266        }
267        for (_, v) in w.properties.iter() {
268          if let Stored::Digested(c) = v {
269            walk(c, depth + 1, 1, out, seen);
270          }
271        }
272      },
273      _ => {},
274    }
275  }
276  let mut out = Vec::new();
277  let mut seen = rustc_hash::FxHashSet::default();
278  for b in boxes {
279    walk(b, 0, 1, &mut out, &mut seen);
280  }
281  eprintln!(
282    "fragment holders: {} boxes walked, {} with extra holders",
283    seen.len(),
284    out.len()
285  );
286  for l in out {
287    eprintln!("{l}");
288  }
289}

308/// Perl `hardYankProcessing` (Common/Error.pm L320-348): after a Fatal there
309/// is nothing left to do in the main processing — rescue the digested bodies
310/// and reset the gullet (pushback, mouth stack, mouth), so the salvage
311/// `digest_internal` re-collects the rescue and reads NO further input. Until
312/// this existed the salvage pass kept reading the live mouth past the Fatal:
313/// jlreq's `PushbackLimit` fired inside the class, then luatexja and the whole
314/// body were converted with every Error record muted by the post-Fatal latch
315/// (asternote, hideanswer-doc, inlinelabel, jpnedumathsymbols-doc: 8 of 22
316/// undefined macros tallied with no `Error:` line — diagnostics audit
317/// 2026-09-20). Perl also `\relax`es the busy token; with the input gone the
318/// loop cannot resume, so that step is not needed here.
319fn hard_yank_processing(boxes: Vec<Digested>) {
320  RESCUED_BOXES.with(|r| *r.borrow_mut() = boxes);
321  gullet::flush();
322  // (Perl's `hardYankProcessing` empties the token stack too, Common/Error.pm:329: a re-raised resource Fatal —
323  // `Timeout:Convert`, the RSS fuse, `TokenLimit`, `PushbackLimit` — is salvaged and built, and a stale stack would
324  // fail that build's first push as a ghost recursion Fatal, as the recovery arms' clear guards against (2410.07647,
325  // 2105.02599; review rev131fix finding 1). The recursion Fatal's message already holds its stack, stomach.rs.)
326  stomach::clear_token_stack();
327}

329fn digest_step_guarded(boxes: &mut Vec<Digested>) -> Result<bool> {
330  match stomach::digest_next_body(None) {
331    Ok(next_bodies) => {
332      boxes.extend(next_bodies);
333      Ok(true)
334    },
335    Err(e) => {
336      // Re-raise MemoryBudget / wall-clock Timeout (Convert) errors:
337      // those are *resource* failures, not recoverable digestion
338      // hiccups. Catching them here would silently produce empty
339      // output for a runaway-loop paper, masking a real bug and
340      // inflating canvas pass rates with empty conversions.
341      // R35.A: ensure pathological inputs fail loudly (exit 1+)
342      // rather than silently turning into a zero-byte HTML.
343      use latexml_core::common::error::{ErrorCategory, ErrorTarget};
344      // NOTE the target discrimination on `MemoryBudget`, which is
345      // deliberate rather than an oversight. `Timeout`-target is the RSS
346      // fuse: real resident memory is already at the ceiling, so
347      // continuing means allocating straight into an OOM — there is
348      // nothing to do but stop. `Stomach`-target is the box-list ceilings
349      // (`box_count_cap` / `box_bytes_budget` / boxing depth), where the
350      // salvage below CLEARS the offending accumulation and therefore
351      // itself frees the memory; recovering there is safe precisely
352      // because the hard RSS backstop above stays non-recoverable
353      // underneath it. So the stomach's memory guards stay on the recovery
354      // path — a graceful end with as much of the document as was already
355      // digested, and the Fatal announced and latched below.
356      // A streaming-restart watermark stop is not a failure at all: the CLI
357      // reruns the document under --streaming. No salvage, no Fatal line, no
358      // recovery warning — the converter reports it as an Info.
359      if matches!(e.category, ErrorCategory::StreamingRestart) {
360        return Err(e);
361      }
362      if matches!(
363        (&e.target, &e.category),
364        (ErrorTarget::Timeout, ErrorCategory::MemoryBudget)
365          | (ErrorTarget::Timeout, ErrorCategory::Convert)
366          | (ErrorTarget::Timeout, ErrorCategory::TokenLimit)
367          | (ErrorTarget::Timeout, ErrorCategory::PushbackLimit)
368      ) {
369        emit_warn(
370          "recovery",
371          "digest_internal",
372          &format!(
373            "digest_internal: resource failure ({:?}/{:?}) — not recovering",
374            e.target, e.category
375          ),
376        );
377        return Err(e);
378      }
379      // The Err that landed here was raised at Fatal level. We recover
380      // BOXES from it (below) — Perl `finishDigestion` L219-220 — but a
381      // Fatal-level raise stays FATAL in the document's reported outcome:
382      // salvaging content is not licence to reclassify the verdict, and
383      // there is deliberately no auto-upgrade to Error severity here (user
384      // policy 2026-07-28). The one sanctioned demotion in this codebase is
385      // the bibliography's explicit `DEMOTE_FATALS` (`error.rs`), which is
386      // opt-in and scoped.
387      //
388      // `log_fatal()` is the single seam that does BOTH halves: it emits
389      // the standard `Fatal:<target>:<category>` line AND latches
390      // `LogStatus::Fatal`. This used to be a hand-rolled `log::error!`
391      // with a `Fatal:`-prefixed target, avoiding `log_fatal` for fear of
392      // "double-incrementing the counter" — unfounded, since the fatal
393      // status is a sticky BOOL (`error.rs` `note_status`, guarded by
394      // `fatal_status_is_sticky_and_returns_1`). The hand-rolled call used
395      // the raw `log`-crate macro, which never reaches `note_status`, so
396      // guard fatals raised as a plain `Err` by `stomach::check_timeout`
397      // (rather than through `Fatal!`) were never counted at all: the log
398      // carried a `Fatal:` line while the run summarised as `Conversion
399      // complete: No obvious problems`, status code 0 — "ok" to cortex.
400      // Guard: `101_fatal_salvages_partial_document`.
401      e.log_fatal();
402      emit_warn(
403        "recovery",
404        "digest_internal",
405        &format!("digest_internal: error during recovery digestion: {:?}", e),
406      );
407      // (the tokens the failed invocations left on the stack go, as Perl's `hardYankProcessing` empties it for every
408      // Fatal: left there, the next digestion — the build's — failed its first push, 2410.07647, 2105.02599)
409      stomach::clear_token_stack();
410      // Recover what the failed body already digested. Without this the
411      // "still produce partial output" intent above only worked when the
412      // failure landed in a LATER body — a Fatal inside the FIRST one left
413      // `boxes` empty and the run wrote a 39-byte empty document, losing a
414      // whole paper to one bad construct (arXiv:2508.07407 / ar5iv #556,
415      // one pathological `\tikz` picture).
416      //
417      // Scoped to the STOMACH box-cycle guard, deliberately. That guard
418      // fires while the token stream is still healthy — one construct is
419      // piling up boxes — so the surrounding document is sound and worth
420      // keeping, and the innermost level (the 50k-box repeating window) is
421      // exactly the construct to drop.
422      //
423      // It is NOT extended to the gullet's `Timeout:Recursion`, where the
424      // TOKEN stream is the thing looping: measured on arXiv:2605.25400,
425      // salvaging there revived a poisoned state that re-entered the same
426      // loop during build and turned an 8.7 s fatal into a 2 m 12 s
427      // wall-clock timeout writing a ZERO-byte file — strictly worse than
428      // the 39-byte stub, for a 1.7 KB gain on the one paper it helped.
429      //
430      // `TooManyErrors` keeps everything digested before it, as Perl's
431      // `hardYankProcessing` rescues `@LaTeXML::LIST` for every Fatal
432      // (Common/Error.pm:336-338) and the daemon's retry of `finishDigestion`
433      // (LaTeXML.pm:251-259, Core.pm:222-224) builds it: latexmlc writes the
434      // partial document where this wrote a 39-byte stub (acro-manual 120 KB
435      // in Perl; 27 TeX Live manuals and 5 of 7 Fatals in a 3,003-paper arXiv
436      // sample hit this Fatal). Its token stream is healthy — each error ends
437      // in an `<ERROR>` box — so there is no loop to re-enter, and the
438      // current level is honest content (the paragraph in progress). Input
439      // after the Fatal stays unread, as in Perl.
440      //
441      //
442      // The conditional guard (`Timeout:IfLimit`) keeps everything too: a run of conditionals adds no boxes of its own,
443      // so the current level is the document's (for a loop in the body's own paragraph, the whole body), and the
444      // salvage's own ceiling still refuses a loop that piles up boxes. A finite figure-heavy paper reaches the limit as
445      // surely as a loop: 2006.14937's 24 matplotlib `.pgf` figures (5.3 MB) trip it at the ninth, and the whole paper
446      // was a 39-byte stub.
447      let drop_innermost = match (&e.target, &e.category) {
448        (ErrorTarget::Stomach, _) => Some(true),
449        (ErrorTarget::TooManyErrors, _) | (ErrorTarget::Timeout, ErrorCategory::IfLimit) => {
450          Some(false)
451        },
452        _ => None,
453      };
454      // What is kept still has to be built. The Fatal is announced and its document kept, so it leaves the latch the math
455      // parser reads a resource Fatal from (`take_last_resource_fatal`): left there, the first formula a semantic action
456      // rejects took it for its own and dropped the document; `resource_fatal_latched` stays set. A conditional of the
457      // build (a rewrite's, a constructor's) would trip the spent guard again, so the phases after digestion get its
458      // budget afresh, and a loop the build re-enters still ends at the limit.
459      if matches!(e.category, ErrorCategory::IfLimit) {
460        error::take_last_resource_fatal();
461        if state::if_limit() > 0 {
462          state::set_if_limit(state::if_count() + state::if_limit());
463        }
464      }
465      if let Some(drop_innermost) = drop_innermost {
466        let salvaged = stomach::salvage_pending_box_lists(drop_innermost);
467        if !salvaged.is_empty() {
468          emit_info(
469            "recovery",
470            "digest_internal",
471            &format!(
472              "digest_internal: salvaged {} box(es) digested before the fatal",
473              salvaged.len()
474            ),
475          );
476          boxes.extend(salvaged);
477        }
478      }
479      Ok(false)
480    },
481  }
482}

484/// The document head shared by the eager and streaming paths: a fresh
485/// [`Document`] with the schema model loaded and the preload PIs inserted
486/// (the front half of `convert_document`, extracted verbatim).
487fn build_document_head(preloads: &[String]) -> Result<Document> {
488  // (no `d`-token evidence carries over from an earlier document on this thread)
489  latexml_math_parser::clear_differential_evidence();
490  let mut document = Document::new();
491  {
492    // TODO: Can we disentangle the ownership to avoid the clone?
493    let paths_stored = state::get_search_paths();
494    let schema_paths = paths_stored
495      .iter()
496      .map(String::as_str)
497      .collect::<Vec<&str>>();
498    let default_model_load = model::with_schema_data(|schema_opt| match schema_opt {
499      None => true,
500      Some(v) => v.last() == Some(&pin!("LaTeXML")),
501    });
502    if default_model_load {
503      // Compile-time load of model AND indirect model. Single
504      // shared instantiation lives at `crate::load_latexml_default_model`
505      // so LTO can keep exactly one `_ModelLoader::build_model` in
506      // the final binary (~600 KiB per copy otherwise).
507      crate::load_latexml_default_model();
508    } else {
509      // Eager-load at runtime
510      model::load_schema(schema_paths.as_slice())?; // If needed?
511    }
512    if state::has_search_paths() {
513      {
514        if state::lookup_bool("INCLUDE_COMMENTS") {
515          let paths_string = state::with_search_paths(|paths| {
516            paths
517              .iter()
518              .map(String::as_str)
519              .collect::<Vec<&str>>()
520              .join(",")
521          });
522          let attributes = map! {s!("searchpaths") => paths_string};
523          document.insert_pi("latexml", Some(attributes))?;
524        }
525      }
526    }
527  }
528
529  for preload in preloads {
530    if preload.ends_with(".pool") {
531      continue;
532    }
533    // Perl `Core.pm` L268-277 rewrites `$preload` IN PLACE with `s///`, so
534    // the option bracket and the `.cls`/`.sty` suffix are gone from the
535    // string that becomes the attribute value, and the captured options ride
536    // along as a second attribute. `Regex::replace_all` RETURNS a new string
537    // rather than mutating its input, so discarding the result — as this loop
538    // did until 2026-07-29 — stripped nothing at all:
539    // `--preload=[twocolumn,11pt]article.cls` emitted
540    // `<?latexml class="[twocolumn,11pt]article.cls"?>` where Perl emits
541    // `<?latexml class="article" options="twocolumn,11pt"?>`.
542    //
543    // Deliberately NOT routed through `parse_preload_spec` above: that
544    // splits on the LAST `.` and so would also eat a non-package extension
545    // (`--preload=mystyle.tex` -> `package="mystyle"`), while Perl strips
546    // only the two literal suffixes and leaves anything else attached. It
547    // also trims/drops empty options, where Perl passes `$1` verbatim.
548    let mut spec: &str = preload;
549    let mut options: &str = "";
550    if let Some(bracket) = LATEX_OPTION_REGEX.captures(spec) {
551      options = bracket.get(1).map_or("", |m| m.as_str());
552      spec = &spec[bracket.get(0).map_or(0, |m| m.end())..];
553    }
554    let mut attributes: HashMap<String, String> = HashMap::default();
555    // Perl's `($options ? (options => $options) : ())`: an empty bracket
556    // (`[]name.sty`) is falsy, so it contributes no attribute at all.
557    if !options.is_empty() {
558      attributes.insert(s!("options"), options.to_string());
559    }
560    if let Some(class) = spec.strip_suffix(".cls") {
561      attributes.insert(s!("class"), class.to_string());
562    } else {
563      attributes.insert(
564        s!("package"),
565        spec.strip_suffix(".sty").unwrap_or(spec).to_string(),
566      );
567    }
568    document.insert_pi("latexml", Some(attributes))?;
569  }
570  Ok(document)
571}

611/// The whole-document tail shared by both paths: rewrites, `\lxDeclare`
612/// application, math parsing, finalize, XMTok-id cleanup (the back half of
613/// `convert_document`, extracted verbatim). In streaming mode this runs on
614/// the live SPINE — spilled fragments received the same phases fragment-by-
615/// fragment in `streaming_pass2` beforehand.
616fn finish_document(document: &mut Document) -> Result<()> {
617  // Every reference is built (forward ones too, the eager and streaming paths alike): the diagnostics a held tag
618  // deferred (an `\autoref` name, ruling 7e) are raised for the tags a reference prints.
619  // (an empty held tag no target settled is tied to its owner's labels first, `settle_empty_held_tags`)
620  if let Some(root) = document.get_document().get_root_element() {
621    latexml_engine::latex_constructs::settle_empty_held_tags(document, &root);
622  }
623  latexml_core::util::logger::settled_empty_held_tags();
624  latexml_core::util::logger::replay_shown_deferred();
625  // Lines of nothing but float-boxes give way to their floats (57g), once the document is built;
626  // spilled streaming segments get the same pass in `streaming_pass2`.
627  if let Some(root) = document.get_document().get_root_element() {
628    latexml_engine::latex_constructs::settle_float_paragraphs(document, &root)?;
629  }
630  let has_rewrites = state::has_value("DOCUMENT_REWRITE_RULES");
631  if has_rewrites {
632    let _gp_rewrite = latexml_core::telemetry::phase(latexml_core::telemetry::Phase::Rewrite);
633    note_begin("Rewriting");
634    document.mark_xmnode_visibility()?;
635    document.load_labels_for_rewrite()?;
636    // TODO: What is the right way to do rewrites in a daemon-safe manner?
637    if let Some(Stored::VecDequeStored(rules)) = state::remove_value("DOCUMENT_REWRITE_RULES")
638      && let Some(root) = document.get_document().get_root_element()
639    {
640      apply_rewrite_rules(document, rules, &root)?;
641    }
642    note_end("Rewriting");
643  }
644
645  // Apply \lxDeclare declarations: set roles/names/meanings on matching XMTok elements.
646  // Must run BEFORE math parsing so the parser sees the updated roles.
647  apply_lx_declarations(document);
648
649  if !state::get_nomathparse_flag() {
650    // Telemetry: count formulae and time the whole Marpa parse pass.
651    // Per-formula bucket histogram requires per-call instrumentation
652    // inside latexml_math_parser::parser::parse_math; deferred.
653    let xmath_count = document.findnodes("//ltx:XMath", None).len() as u32;
654    // ADD, not set: `finish_document` runs on the streaming SPINE *after*
655    // `streaming_pass2` has already counted every spilled fragment's formulae,
656    // so a plain `set` here would clobber the segment tallies and report only
657    // the spine's own — near zero on a document that spilled most of itself.
658    // Eager is unaffected: the counter starts at 0 and this runs once.
659    latexml_core::telemetry::add_formulae(xmath_count);
660    let _gp = latexml_core::telemetry::phase(latexml_core::telemetry::Phase::MathParse);
661    let mut parser = MathParser::default();
662    let parsed = parser.parse_math(document);
663    // (the spine, or the eager document, is the last parse of the conversion)
664    latexml_math_parser::clear_differential_evidence();
665    parsed?;
666    drop(_gp);
667    // Post-parse: mark failed XMath nodes as unparsed.
668    // The parser's parse_kludge already handles OPEN/CLOSE wrapping + script attachment
669    // (parse_kludgeScripts_rec), so we only need to add the unparsed CSS class here.
670    if !parser.failed_xmath_ids.is_empty() {
671      for mut math_node in document.findnodes("descendant-or-self::ltx:Math[not(@text)]", None) {
672        for xmath_child in document.findnodes("ltx:XMath", Some(&math_node)) {
673          if parser.failed_xmath_ids.contains(&xmath_child.to_hashable()) {
674            document.add_class(&mut math_node, "ltx_math_unparsed")?;
675            break;
676          }
677        }
678      }
679    }
680    // Renumber xml:ids inside parsed XMath subtrees to be sequential in document
681    // order. The Marpa parser explores multiple parse alternatives, consuming ID
682    // counter slots for pruned nodes. This pass reassigns IDs post-parse.
683    renumber_math_ids(document);
684    // Fill in \ltx@count@parses markers with actual parse tree counts.
685    // Each marker is <ltx:text _parsetrees_marker="true">0</ltx:text>.
686    // Find the preceding ltx:Math[@_parsetrees] and copy the count.
687    let markers = document.findnodes("//*[@_parsetrees_marker='true']", None);
688    for mut marker in markers {
689      let count = {
690        let preceding = document.findnodes("preceding::ltx:Math[@_parsetrees][1]", Some(&marker));
691        preceding
692          .into_iter()
693          .last()
694          .and_then(|m| m.get_attribute("_parsetrees"))
695          .unwrap_or_else(|| "0".to_string())
696      };
697      // Replace the text content with the actual count
698      for mut child in marker.get_child_nodes() {
699        child.unlink_node();
700      }
701      let _ = marker.append_text(&count);
702      // Remove the marker attribute
703      let _ = marker.remove_attribute("_parsetrees_marker");
704    }
705  }
706
707  // #683 (xworld21): persist NOMINAL_FONT_SIZE as a
708  // `<?latexml nominal-font-size="X"?>` processing instruction when it differs
709  // from the 10pt default, so post-processing can size font-relative (em)
710  // external SVGs correctly (an `em` is `NOMINAL_FONT_SIZE`pt, not always 10pt).
711  // Perl does not emit this — NOMINAL_FONT_SIZE is digestion-only (`DEFSIZE`),
712  // so this is beyond-Perl. Only a0poster (25), the NNpt class options, and
713  // BookML move it off 10, so a normal document's output stays byte-identical
714  // (no new PI). `insert_pi` places it before the root, alongside the other
715  // `<?latexml …?>` metadata PIs (class/package/graphicspath).
716  if let Some(nominal) = state::lookup_float("NOMINAL_FONT_SIZE")
717    && (nominal.0 - 10.0).abs() > 1e-6
718  {
719    let mut attrs = HashMap::default();
720    attrs.insert(String::from("nominal-font-size"), nominal.0.to_string());
721    document.insert_pi("latexml", Some(attrs))?;
722  }
723
724  note_begin("Finalizing");
725  document.finalize()?;
726  note_end("Finalizing");
727  // Perl core produces role="UNKNOWN" for single-letter math tokens.
728  // Per-document .latexml files set role="ID" via DefMathRewrite BEFORE parsing.
729  // We do NOT apply a blanket conversion — roles are set by rewrite rules only.
730  // Cleanup unreferenced xml:ids on XMTok elements generated by the math parser.
731  // Must run after finalize (which includes prune_xmduals that may transfer ids).
732  document.cleanup_unreferenced_xmtok_ids();
733  Ok(())
734}

736/// Compile and invoke a set of `DefRewrite`/`DefMathRewrite` rules against
737/// `root` (extracted verbatim from the eager Rewriting phase). The streaming
738/// pass 2 calls this once per FRAGMENT with a snapshot of the rule set — the
739/// S2 census showed the production corpus is subtree-local, so per-fragment
740/// application is equivalent; `label:`-scoped rules resolve through the
741/// fragment's `rewrite_labels`, then the document-wide
742/// shared label map (`Document::rewrite_labels_shared`).
743fn apply_rewrite_rules(
744  document: &mut Document,
745  rules: std::collections::VecDeque<Stored>,
746  root: &libxml::tree::Node,
747) -> Result<()> {
748  // Step 1: copy the rules locally through Rc, to be able to invoke them with mutable
749  // state. (TODO: obviously, this could be avoided if they never needed mutable
750  // state. When do they?)
751  let mut rewrites = Vec::new();
752  for rule in rules {
753    if let Stored::Rewrite(mut rewrite_rule) = rule {
754      rewrite_rule.compile_clauses(document);
755      rewrites.push(rewrite_rule);
756    }
757  }
758  // 31 rules compiled for declare test; XPath matching issue prevents application
759  // Step 2: invoke the rewrite rules
760  // R35.D instrumentation: print per-rule timing if
761  // LATEXML_REWRITE_TIMING=1. Logs BEFORE the rule runs so we can
762  // identify the rule that hangs (the timeout watchdog kills
763  // mid-rule otherwise).
764  let trace_all = std::env::var_os("LATEXML_REWRITE_TIMING").is_some();
765  let n_rules = rewrites.len();
766  for (idx, mut rewrite_rule) in rewrites.into_iter().enumerate() {
767    // Build a useful one-line hint from the rule's options. The
768    // Debug impl on RewriteOptions is `<RewriteOptions>` only,
769    // so reach into the fields directly.
770    let opts = &rewrite_rule.options;
771    let mut xpath_hint = format!(
772      "select={:?} xpath={:?} regexp={:?} scope={:?} label={:?} clauses={}",
773      opts
774        .select
775        .as_deref()
776        .map(|s| s.chars().take(60).collect::<String>()),
777      opts
778        .xpath
779        .as_deref()
780        .map(|s| s.chars().take(60).collect::<String>()),
781      opts
782        .regexp
783        .as_deref()
784        .map(|s| s.chars().take(60).collect::<String>()),
785      opts.scope.as_ref().map(|_| "<scope>"),
786      opts.label.as_deref(),
787      rewrite_rule.clauses.len(),
788    );
789    // Dump compiled clauses by op + pattern preview (helps when
790    // the options struct itself is empty after compile_clauses
791    // moved them into the clauses vec).
792    for (ci, c) in rewrite_rule.clauses.iter().enumerate() {
793      use std::fmt::Write;
794      let _ = write!(
795        xpath_hint,
796        "\n    [{ci}] op={:?} pat={:?}",
797        c.op,
798        match &c.pattern {
799          latexml_core::rewrite::RewritePattern::String(s) =>
800            format!("Str({})", s.chars().take(120).collect::<String>()),
801          latexml_core::rewrite::RewritePattern::Tokens(_) => "Tokens(..)".into(),
802          latexml_core::rewrite::RewritePattern::Closure(_) => "Closure(..)".into(),
803          latexml_core::rewrite::RewritePattern::NodeList(n) => format!("NodeList({})", n.len()),
804          _ => "??".into(),
805        }
806      );
807    }
808    if trace_all {
809      eprintln!(
810        "[rewrite-timing] rule #{}/{} START :: {}",
811        idx, n_rules, xpath_hint
812      );
813      // Flush stderr so it appears even if the rule hangs
814      use std::io::Write;
815      let _ = std::io::stderr().flush();
816    }
817    let started = std::time::Instant::now();
818    rewrite_rule.invoke(document, root)?;
819    let elapsed = started.elapsed();
820    if trace_all {
821      eprintln!(
822        "[rewrite-timing] rule #{}/{} END {:.2?}",
823        idx, n_rules, elapsed
824      );
825    } else if elapsed > std::time::Duration::from_secs(5) {
826      eprintln!(
827        "[rewrite-timing] rule #{}/{} SLOW {:.2?} :: {}",
828        idx, n_rules, elapsed, xpath_hint
829      );
830    }
831  }
832  Ok(())
833}

849/// Streaming pass 2: give every spilled fragment the SAME whole-document tail
850/// the spine gets — rewrites, `\lxDeclare`, math parsing, per-fragment
851/// finalize — then store its final serialized text for the assembly splice.
852/// Runs after digestion finished (so the rewrite-rule set is complete) and
853/// before `finish_document` consumes the rules for the spine.
854/// Root-level id counters (`_ID_counter_*` attrs) must run document-wide:
855/// eager mints `id1..idN` across the whole document in one walk, so pass 2
856/// seeds each fragment's parse wrapper with the counters carried out of the
857/// previous one (`counters` in/out), and the caller seeds the spine's tail
858/// from the final state (sweep witness tests/alignment/plainmath.tex, where
859/// every fragment restarted at `id1`). `shared_labels` is the document-wide
860/// label map scope labels resolve through; `root_id` joins every fragment's
861/// ancestors, the root being an ancestor of every segment.
862fn streaming_pass2(
863  store: &mut latexml_core::sxml::SegmentStore,
864  node_fonts: &rustc_hash::FxHashMap<u64, latexml_core::common::font::Font>,
865  counters: &mut Vec<(String, String)>,
866  anchors: &mut rustc_hash::FxHashMap<String, Vec<(String, String)>>,
867  index: &latexml_core::sxml::FragmentIndex,
868  shared_labels: Option<Rc<rustc_hash::FxHashMap<String, String>>>,
869  root_id: Option<String>,
870) -> Result<()> {
871  use latexml_core::common::error::{ErrorCategory, ErrorData, ErrorTarget};
872  // A non-consuming snapshot of the rules: `finish_document` will consume the
873  // live entry for the spine afterwards.
874  let rules_opt = match state::lookup_value("DOCUMENT_REWRITE_RULES") {
875    Some(Stored::VecDequeStored(rules)) => Some(rules),
876    _ => None,
877  };
878  let nomath = state::get_nomathparse_flag();
879  // The segment each spilled anchor is in: its counters there continue what earlier segments minted under it.
880  let mut anchors_in: rustc_hash::FxHashMap<u32, Vec<String>> = rustc_hash::FxHashMap::default();
881  for id in anchors.keys() {
882    if let Some(seg) = index.id_segment(id) {
883      anchors_in.entry(seg.0).or_default().push(id.clone());
884    }
885  }
886  let segments: Vec<_> = store.ids().collect();
887  // Rate-limited exactly like the pass-1 telemetry next door (see the
888  // `is_power_of_two() || is_multiple_of(65536)` gate in `convert_streaming`):
889  // the 131 MB witness spills 459,579 segments, and three ungated `info!` per
890  // segment wrote 1.38 M lines — 44% of a 161 MB log — into the in-RAM
891  // LOG_BUFFER that pass 1 exists to keep bounded. Dense at the start so short
892  // runs and tests still see every line, logarithmic after, 64k floor.
893  //
894  // The ARGUMENTS must stay inside the gate, not just the macro: each probe
895  // reads /proc/self/status, and the `parsed` line re-reads the whole segment
896  // off disk purely to print its size.
897  fn telemetry_due(n: usize) -> bool { n.is_power_of_two() || n.is_multiple_of(65536) }
898  for (seg_idx, seg) in segments.into_iter().enumerate() {
899    let due = telemetry_due(seg_idx + 1);
900    if store.is_retired(seg) {
901      // Inlined into an enclosing segment; its content is processed there.
902      continue;
903    }
904    let meta = store.meta(seg)?.clone();
905    // Segments are bounded (one spill run), so plain parsing is as bounded as
906    // streaming — and unlike `xmlTextReaderExpand`, it does not mint the
907    // libxml2 "default" namespace prefix for default-namespace content
908    // (`append_clone`'s doc-comment records the same trap). The streaming
909    // TextReader stays reserved for genuinely unbounded files (the post
910    // half's pass A over the whole core XML).
911    let frag_xml = libxml::parser::Parser::default()
912      .parse_string(&store.wrapped_segment(seg)?)
913      .map_err(|e| {
914        Error::from(ErrorData {
915          target:   ErrorTarget::Internal,
916          category: ErrorCategory::Libxml,
917          message:  s!("cannot re-parse staged segment {seg}: {e}"),
918        })
919      })?;
920    let mut out = String::new();
921    {
922      let mut frag = Document::from_xml_document(frag_xml, node_fonts.clone())?;
923      if due {
924        emit_info(
925          "streaming",
926          "progress",
927          &format!(
928            "streaming pass2: segment {seg} ({} KB) parsed; RSS ~{} MB",
929            store.read_segment(seg).map(|t| t.len() / 1024).unwrap_or(0),
930            latexml_core::watchdog::process_rss_kb().unwrap_or(0) / 1024,
931          ),
932        );
933      }
934      // A fragment re-emits the (nested) placeholders it contains; only the
935      // final assembly resolves them.
936      frag.literal_placeholders = true;
937      // An ancestor-scoped rewrite covers the whole fragment (field docs).
938      frag.fragment_ancestor_ids = meta.ancestors.iter().cloned().collect();
939      // The root is every segment's ancestor, though its id — minted by the
940      // late `GenerateID` hook, after the spills — is in no segment's record:
941      // a bare `\label` before the first unit lands on it (1703.09326), and a
942      // declaration scoped to that label covers the whole document.
943      frag.fragment_ancestor_ids.extend(root_id.iter().cloned());
944      // Judge the parse wrapper as the segment's REAL parent in schema
945      // decisions (empty-`ltx:text` collapse etc.) — see the field docs.
946      frag.fragment_parent_qname = meta.parent.as_deref().map(arena::pin);
947      // Seed the wrapper root with the carried id counters of the element its top nodes are minted under (the root's
948      // when no element below it has an id), so fragment id minting continues the document-wide sequence.
949      frag.fragment_anchor_id = meta.anchor.clone();
950      let seed = match &meta.anchor {
951        Some(id) => anchors.get(id).map(Vec::as_slice).unwrap_or_default(),
952        None => counters.as_slice(),
953      };
954      if let Some(mut root) = frag.get_document().get_root_element() {
955        for (key, value) in seed {
956          let _ = root.set_attribute(key, value);
957        }
958      }
959      for id in anchors_in.get(&seg.0).into_iter().flatten() {
960        if let Some(carried) = anchors.get(id) {
961          frag.set_id_counters(id, carried);
962        }
963      }
964      // No strip needed: pass 1 serializes segments FLAT (`spill_flat`), so
965      // the indentation text nodes this used to delete are never created.
966      // `strip_indentation_whitespace` unlinked them, and unlink does not
967      // free — they were orphaned for the fragment's lifetime.
968      // Restore empty text children the parse could not represent (spill-time
969      // `_lx_empty_text` markers — see `spill_run`): `<p></p>` must not
970      // collapse to `<p/>` across the round-trip.
971      for mut marked in frag.findnodes("//*[@_lx_empty_text]", None) {
972        let _ = marked.remove_attribute("_lx_empty_text");
973        if marked.get_child_nodes().is_empty() {
974          // `append_text("")` is a no-op (the fork guards on len > 0), so
975          // create a real text node and then empty it in place.
976          let _ = marked.append_text("x");
977          if let Some(mut text) = marked.get_first_child() {
978            let _ = text.set_content("");
979          }
980        }
981      }
982      // The segment's share of `finish_document`'s float-box pass (57g) and of its held-tag settling.
983      if let Some(root) = frag.get_document().get_root_element() {
984        latexml_engine::latex_constructs::settle_empty_held_tags(&mut frag, &root);
985        latexml_engine::latex_constructs::settle_float_paragraphs(&mut frag, &root)?;
986      }
987      if let Some(rules) = &rules_opt {
988        // `Phase::Rewrite` is taken in `finish_document`, which only ever runs
989        // on the SPINE — pass 2 calls `apply_rewrite_rules` directly, so every
990        // fragment's rewrite work was unattributed.
991        let _gp_rewrite = latexml_core::telemetry::phase(latexml_core::telemetry::Phase::Rewrite);
992        frag.mark_xmnode_visibility()?;
993        frag.load_labels_for_rewrite()?;
994        // Share the prebuilt index rather than copying it in; a frag-local
995        // label still wins, exactly as the `or_insert_with` copy ensured.
996        frag.rewrite_labels_shared = shared_labels.clone();
997        if let Some(root) = frag.get_document().get_root_element() {
998          apply_rewrite_rules(&mut frag, rules.clone(), &root)?;
999        }
1000      }
1001      apply_lx_declarations(&mut frag);
1002      if !nomath {
1003        // Only probed when the line will actually be emitted — this read used
1004        // to sit outside the macro, so its /proc cost was paid on every
1005        // segment even at `--quiet`.
1006        let rss0 = if due {
1007          latexml_core::watchdog::process_rss_kb().unwrap_or(0) / 1024
1008        } else {
1009          0
1010        };
1011        // Same telemetry the EAGER path records (see the `//ltx:XMath` count
1012        // and `Phase::MathParse` guard above). Streaming recorded neither,
1013        // while `record_math_parse` inside the parser kept firing — so a
1014        // streamed job's telemetry.json showed thousands of
1015        // `math_parse_attempts` against `formulae: 0` and a near-zero
1016        // `phase_math_parse_us`, systematically, on the biggest papers.
1017        // Accumulated across segments, since pass 2 parses each separately.
1018        latexml_core::telemetry::add_formulae(frag.findnodes("//ltx:XMath", None).len() as u32);
1019        let _gp = latexml_core::telemetry::phase(latexml_core::telemetry::Phase::MathParse);
1020        let mut parser = MathParser::default();
1021        parser.parse_math(&mut frag)?;
1022        drop(_gp);
1023        if due {
1024          emit_info(
1025            "streaming",
1026            "progress",
1027            &format!(
1028              "streaming pass2: segment {seg} math done; RSS {} -> {} MB; arena {} syms",
1029              rss0,
1030              latexml_core::watchdog::process_rss_kb().unwrap_or(0) / 1024,
1031              arena::len(),
1032            ),
1033          );
1034        }
1035        // Mirror the eager tail: mark failed formulae, renumber math ids
1036        // (per-Math, so fragment-local by construction).
1037        if !parser.failed_xmath_ids.is_empty() {
1038          for mut math_node in frag.findnodes("descendant-or-self::ltx:Math[not(@text)]", None) {
1039            for xmath_child in frag.findnodes("ltx:XMath", Some(&math_node)) {
1040              if parser.failed_xmath_ids.contains(&xmath_child.to_hashable()) {
1041                frag.add_class(&mut math_node, "ltx_math_unparsed")?;
1042                break;
1043              }
1044            }
1045          }
1046        }
1047        renumber_math_ids(&mut frag);
1048      }
1049      // The per-fragment share of finalize: XMRef/XMDual pruning and the
1050      // font/bookkeeping walk. The root-only passes (RDFa prefixes, namespace
1051      // declarations, schema PI) belong to the SPINE root and must NOT touch
1052      // a fragment root — they would change its serialization.
1053      frag.prune_dangling_split_xmrefs()?;
1054      frag.prune_xmduals()?;
1055      if let Some(mut root) = frag.get_document().get_root_element() {
1056        let context = meta
1057          .font
1058          .as_deref()
1059          .and_then(|key| frag.decode_font(key))
1060          .cloned()
1061          .unwrap_or_else(latexml_core::common::font::Font::text_default);
1062        frag.finalize_subtree_in(&mut root, context)?;
1063      }
1064      frag.cleanup_unreferenced_xmtok_ids();
1065      // The parsed root is the `_lxfragment` wrapper; the spilled subtrees are
1066      // its children. Serialize each at the recorded position parameters.
1067      if let Some(root) = frag.get_document().get_root_element() {
1068        let _gp_ser = latexml_core::telemetry::phase(latexml_core::telemetry::Phase::Serialize);
1069        for child in root.get_child_nodes() {
1070          out.push_str(&frag.serialize_aux(&child, meta.depth, meta.noindent, false));
1071        }
1072        // Carry the wrapper's id counters to the next fragment under the same element
1073        // (finalize deliberately leaves the wrapper's bookkeeping attrs in
1074        // place — see finalize_rec's PostWork).
1075        let carried = latexml_core::document::id_counters(&root);
1076        match &meta.anchor {
1077          Some(id) => {
1078            anchors.insert(id.clone(), carried);
1079          },
1080          None => *counters = carried,
1081        }
1082      }
1083    }
1084    store.finalize_segment(seg, &out)?;
1085    if due {
1086      emit_info(
1087        "streaming",
1088        "progress",
1089        &format!(
1090          "streaming pass2: segment {seg} finalized+dropped; RSS ~{} MB",
1091          latexml_core::watchdog::process_rss_kb().unwrap_or(0) / 1024
1092        ),
1093      );
1094    }
1095  }
1096  Ok(())
1097}

1194  /// The input-side half of [`DigestionAPI::digest`]: canonicalize the
1195  /// request, seed SOURCEFILE/SOURCEDIRECTORY/SEARCHPATHS/`\jobname`, queue
1196  /// postamble/source/preamble on the gullet. Returns the progress-note label
1197  /// the caller must `note_end` when digestion completes. Shared by the eager
1198  /// path and `convert_streaming` (which drives digestion itself,
1199  /// fragment-by-fragment).
1200  fn digest_setup(
1201    &mut self,
1202    request: String,
1203    preamble: Option<String>,
1204    postamble: Option<String>,
1205    mode: Option<DigestionMode>,
1206  ) -> Result<String> {
1207    DIGEST_SETUP_COUNT.with(|c| c.set(c.get() + 1));
1208    stomach::mark_conversion_rss_baseline();
1209    let mut _ext = match &mode {
1210      Some(m) => Some(m.extension()),
1211      None => Some(DigestionMode::TeX.extension()),
1212    };
1213    let mut dir_opt = None;
1214
1215    // Canonicalize relative paths so `Path::parent()` gives a real directory.
1216    // `Path::new("foo.tex").parent()` returns `Some("")` (empty string) which
1217    // poisons SEARCHPATHS / SOURCEDIRECTORY: an empty-string search-path
1218    // entry resolves files via cwd-name with no normalization, changing the
1219    // order in which resource files (e.g. `ts1enc.def` vs `t1enc.def`) are
1220    // discovered. Concrete symptom: TS1 fontmap leaks into control-sequence
1221    // construction → `cn` characters become `⚮♪` → `\c@cn` undefined →
1222    // 381-error cascade (paper 0709.2868). Canonicalizing matches Perl's
1223    // `File::Spec->splitpath` behavior which always yields a real directory.
1224    let canonical_request = if pathname::is_literaldata(&request) || pathname::is_url(&request) {
1225      request.clone()
1226    } else {
1227      std::fs::canonicalize(&request)
1228        .ok()
1229        .and_then(|p| p.to_str().map(String::from))
1230        .unwrap_or_else(|| request.clone())
1231    };
1232    let name = if pathname::is_literaldata(&request) {
1233      s!("Anonymous String")
1234    } else if pathname::is_url(&request) {
1235      request.clone()
1236    } else {
1237      let path = Path::new(&canonical_request);
1238      dir_opt = path.parent();
1239      match path.file_stem() {
1240        None => String::from("missing_name"),
1241        Some(pf) => pf.to_str().unwrap().to_string(),
1242      }
1243    };
1244    // else {
1245    //   $self->withState(sub {
1246    //       Fatal('missing_file', $request, undef, "Can't find $mode file $request"); }); } }
1247    // };
1248    // Book-scale sources legitimately expand past the arXiv-sized 1G
1249    // runaway-token backstop; scale it to the input (never lowers, env
1250    // override wins — see gullet::scale_token_limit_to_source).
1251    // A file source is counted when its Mouth opens it (`Mouth::open_file`);
1252    // literal data has no file, so count it here.
1253    if pathname::is_literaldata(&request) {
1254      gullet::scale_token_limit_to_source(request.len());
1255    }
1256    let digestion_note = s!("Digesting {}", name);
1257    note_begin(&digestion_note);
1258    // $self->initializestate::$mode . ".pool", @{ $$self{preload} || [] }) unless
1259    // $options{noinitialize};
1260    if !pathname::is_literaldata(&request) {
1261      state::assign_value("SOURCEFILE", arena::pin(&request), None);
1262    }
1263    if let Some(dir) = dir_opt {
1264      let dir_str = dir.to_str().unwrap_or(".");
1265      // Perl Core.pm L195-200 unshifts the SOURCE file's directory onto
1266      // SEARCHPATHS so subsequent `\input`-style lookups resolve relative
1267      // to the main file. When `canonicalize` succeeded `dir_str` is the
1268      // absolute parent; if it failed (e.g. file not on disk yet at the
1269      // time we resolved — unusual for normal latexml CLI invocations
1270      // but possible for `literal:` etc.) `dir_str` may be empty and an
1271      // empty entry on SEARCHPATHS is useless. Fall back to CWD in that
1272      // case so paper-local files (`\input{Chapter/Abstract}` etc.) are
1273      // discoverable. Witness: arXiv:2604.09744, 2603.04457 (papers
1274      // bundling subdirectory `\subimport` chains).
1275      let resolved_dir = if dir_str.is_empty() {
1276        std::env::current_dir()
1277          .ok()
1278          .and_then(|cwd| cwd.to_str().map(String::from))
1279          .unwrap_or_else(|| ".".to_string())
1280      } else {
1281        dir_str.to_string()
1282      };
1283      state::assign_value("SOURCEDIRECTORY", arena::pin(&resolved_dir), None);
1284      establish_pdf_output_mode(&resolved_dir);
1285      // Perl Core.pm L195-200: `$state->unshiftValue(SEARCHPATHS => $dir)`.
1286      // `unshift` puts the source dir at the FRONT so it's the new "lead"
1287      // — the same lead `\lx@append@path` reads as the basis for
1288      // appended subdir paths. Pushing to BACK (`add_search_path`) left
1289      // the lead as whatever the CLI's `--path` provided (typically
1290      // `ar5iv-bindings/bindings`); `\subimport{Chapter/}{Abstract}`
1291      // then appended Chapter/ to ar5iv-bindings/bindings instead of to
1292      // the paper's directory, and `\input{Abstract}` couldn't resolve.
1293      // Witness: arXiv:2604.09744, 2603.04457.
1294      state::search_paths_push_front(resolved_dir);
1295    }
1296    //   if defined $dir && !grep { $_ eq $dir } @{ $state->lookupValue('SEARCHPATHS') };
1297    // $state->unshiftValue(GRAPHICSPATHS => $dir)
1298
1299    // if defined $dir && !grep { $_ eq $dir } @{ $state->lookupValue('GRAPHICSPATHS') };
1300
1301    install_jobname(&name);
1302
1303    // Reverse order, since last opened is first read!
1304    // (Perl: Core.pm L154-157 in `digestFile`.)
1305    if let Some(postamble) = postamble {
1306      self.load_postamble(postamble);
1307    }
1308    input_content(&request, InputOptions::default())?;
1309    if let Some(preamble) = preamble {
1310      self.load_preamble(preamble);
1311    }
1312
1313    // Now for the Hacky part for BibTeX!!!
1314    // Perl `Core.pm` L160-162: drain the .bib mouth via the Pre::BibTeX
1315    // parser, register each entry in the bibtex.rs thread-local
1316    // registry, and push back a `literal:` wrapper that produces a
1317    // `\begin{bibtex@bibliography}...\end{bibtex@bibliography}` block
1318    // for the digester to process.
1319    if matches!(mode, Some(DigestionMode::BibTeX)) {
1320      use latexml_engine::pre_bibtex::PreBibTeX;
1321      let mut bib = PreBibTeX::new_from_gullet(&name);
1322      match bib.to_tex() {
1323        Ok(tex) => {
1324          input_content(&s!("literal:{tex}"), InputOptions::default())?;
1325        },
1326        Err(parse_err) => {
1327          Error!(
1328            "bibtex",
1329            "parse_failed",
1330            s!("Failed to parse BibTeX file {}: {:?}", name, parse_err)
1331          );
1332        },
1333      }
1334    }
1335
1336    Ok(digestion_note)
1337  }

1346  /// The streaming front half: spill store + disk gate, deferred root,
1347  /// literal placeholders, flat spill, the fragment index and the yield
1348  /// knobs. Shared by `convert_streaming` and the adaptive transition.
1349  fn stream_setup(
1350    &mut self,
1351    document: &mut Document,
1352    budget: usize,
1353  ) -> Result<latexml_core::sxml::FragmentIndex> {
1354    use latexml_core::sxml::{FragmentIndex, SegmentStore};
1355    latexml_core::document::reset_spilled_segment_count();
1356    // The spill area lives beside the source when that directory is WRITABLE
1357    // (same volume as the output in the by-far-common layout, so the
1358    // disk-headroom check measures the filesystem the spill actually
1359    // consumes), else the system temp dir. The writability test is not
1360    // hypothetical: auto-activation fires on exactly the large documents that
1361    // get processed from read-only trees — an arXiv bulk mount, a CI
1362    // checkout — and creating the directory unconditionally failed the whole
1363    // conversion there. Literal input has no source directory at all.
1364    let spill_anchor = match state::lookup_value("SOURCEDIRECTORY") {
1365      Some(Stored::String(dir)) => {
1366        let dir = arena::with(dir, |s: &str| std::path::PathBuf::from(s));
1367        if dir_is_writable(&dir) {
1368          dir
1369        } else {
1370          emit_info(
1371            "streaming",
1372            "progress",
1373            &format!(
1374              "streaming: {} is not writable; spilling to {} instead",
1375              dir.display(),
1376              std::env::temp_dir().display()
1377            ),
1378          );
1379          std::env::temp_dir()
1380        }
1381      },
1382      _ => std::env::temp_dir(),
1383    };
1384    let store = SegmentStore::create(&spill_anchor)?;
1385    // Disk-headroom gate (user requirement 2026-07-29): the spill needs
1386    // roughly the core-XML size on disk — measured ~12x the source bytes on
1387    // math-heavy content — and a silent mid-conversion ENOSPC would be the
1388    // OOM-kill failure mode all over again. Verify up front and raise a
1389    // Fatal that NAMES the shortfall and the requirement, so the user can
1390    // free space (or point --dest at a roomier volume) with numbers in hand.
1391    {
1392      use latexml_core::common::error::{ErrorCategory, ErrorData, ErrorTarget};
1393      const SPILL_BYTES_PER_SOURCE_BYTE: u64 = 16; // ~12x measured + slack
1394      let src_bytes = match state::lookup_value("SOURCEFILE") {
1395        Some(Stored::String(f)) => {
1396          arena::with(f, |f| std::fs::metadata(f).map(|m| m.len()).unwrap_or(0))
1397        },
1398        _ => 0,
1399      };
1400      let need = src_bytes.saturating_mul(SPILL_BYTES_PER_SOURCE_BYTE);
1401      if let Some(avail) = latexml_core::watchdog::available_disk_bytes(store.dir())
1402        && avail < need
1403      {
1404        stomach::set_fragment_yield_budget(None);
1405        return Err(Error::from(ErrorData {
1406          target:   ErrorTarget::Timeout,
1407          category: ErrorCategory::MemoryBudget,
1408          message:  s!(
1409            "streaming spill needs ~{} MB free under {} but only {} MB is available — free disk space there, or convert with a destination on a roomier volume",
1410            need / (1024 * 1024),
1411            store.dir().display(),
1412            avail / (1024 * 1024)
1413          ),
1414        }));
1415      }
1416    }
1417    document.set_spill_store(store);
1418    document.set_defer_root_after_open(true);
1419    // Pass 1 serializes placeholders literally (nested spills stay nested;
1420    // the final assembly resolves them recursively — see the field docs).
1421    document.literal_placeholders = true;
1422    // The math parser's `d`-token evidence is the whole document's, read before its rewrites (as the eager path
1423    // reads it): each subtree's as it spills, the spine's before pass 2.
1424    // So are the output's words the title-page check looks for (latexml_engine `title_page_check`): each subtree's
1425    // as it spills, the spine's when the document closes.
1426    document.spill_observer = Some(if state::get_nomathparse_flag() {
1427      observe_spilled_words
1428    } else {
1429      observe_spilled_evidence_and_words
1430    });
1431    // Spill segments are an intermediate that pass 2 re-serializes; the
1432    // indentation pass 1 used to emit was generated, written, read back,
1433    // parsed into ~40M text nodes and then deleted again. See `spill_flat`.
1434    document.spill_flat = true;
1435    let index = FragmentIndex::default();
1436    stomach::set_fragment_yield_budget(Some(budget));
1437    // Soft-RSS yield: fire regardless of box count once RSS crosses the
1438    // watermark — the box budget assumes a per-box footprint, and math-dense
1439    // content blows past it (witness: fuse at 18.8 GB with the box budget
1440    // untouched). `spill_watermark_bytes` owns the policy, including the
1441    // `--max-memory=0` case where there is no fuse to divide.
1442    if let Some(mark_kb) = stomach::spill_mark_kb() {
1443      stomach::set_fragment_yield_rss_soft_kb(Some(mark_kb));
1444    }
1445
1446    Ok(index)
1447  }

1449  /// Streaming pass 1: interleaved digest → absorb → spill, at legal seams only.
1450  /// `seed` = bodies already digested by the eager loop (the adaptive
1451  /// transition), absorbed as fragment 1. Returns whether a fatal stopped it.
1452  fn stream_pass1(
1453    &mut self,
1454    document: &mut Document,
1455    index: &mut latexml_core::sxml::FragmentIndex,
1456    mut seed: Option<Vec<Digested>>,
1457  ) -> Result<bool> {
1458    let mut fatal_stop = false;
1459    loop {
1460      let mut boxes = Vec::new();
1461      // A seeded first iteration (the adaptive eager→streaming transition,
1462      // `digest_adaptive`) absorbs the bodies the eager loop had already
1463      // accumulated instead of digesting: they were cut at a legal yield seam,
1464      // so they are fragment 1, and the gullet continues where it stopped.
1465      let (step, yielded) = if let Some(seed_boxes) = seed.take() {
1466        boxes = seed_boxes;
1467        (Ok(true), true)
1468      } else {
1469        let step = {
1470          // Streaming had NO telemetry phases at all — every `phase()` guard in
1471          // the tree is on the eager path — so `TELEMETRY.md`'s "sum of phase
1472          // wall ≈ total wall, median ≥ 0.92" was silently unmet for every
1473          // streamed conversion. Reuse the eager phases rather than adding
1474          // streaming-specific ones: pass 1 IS digest + build, interleaved.
1475          let _g = latexml_core::telemetry::phase(latexml_core::telemetry::Phase::Digest);
1476          digest_step_guarded(&mut boxes)
1477        };
1478        (step, stomach::take_fragment_yielded())
1479      };
1480      let mut stopped = match step {
1481        Ok(keep_going) => !keep_going,
1482        Err(e) => {
1483          // Under EAGER digestion a resource fatal (the RSS fuse) is
1484          // deliberately not recovered — continuing would allocate straight
1485          // into an OOM. Under STREAMING the calculus inverts: stopping
1486          // digestion here FREES the box list, the spill below releases the
1487          // DOM, and the pipeline finishes on already-spilled content — so
1488          // recovery is not just safe, it is the feature this mode exists
1489          // for. The Fatal contract still holds: announce + latch the
1490          // verdict, keep the partial document (user policy 2026-07-28).
1491          e.log_fatal();
1492          emit_warn(
1493            "internal",
1494            "core_interface",
1495            &format!(
1496              "convert_streaming: digestion stopped by a resource fatal ({:?}/{:?}) — keeping the document built so far",
1497              e.target, e.category
1498            ),
1499          );
1500          // Recover what the interrupted step had digested. Unlike the eager
1501          // Stomach-target salvage, drop_innermost is FALSE: there is no
1502          // runaway construct to excise here — the innermost level IS the
1503          // healthy top-level accumulation, and the budget simply ran out.
1504          // Safe because everything salvaged is absorbed, spilled and freed
1505          // immediately below.
1506          boxes.extend(stomach::salvage_pending_box_lists(false));
1507          // (and the stack of tokens being invoked, which the build's digestion pushes onto again: 2410.07647)
1508          stomach::clear_token_stack();
1509          fatal_stop = true;
1510          true
1511        },
1512      };
1513      let mut fragment: Option<Digested> = None;
1514      if !boxes.is_empty() {
1515        let digested = Digested::from(List::new(boxes));
1516        let _g_build = latexml_core::telemetry::phase(latexml_core::telemetry::Phase::Build);
1517        // Interleaved with digestion: base-level build bindings must not
1518        // pop with the digestion group open at this seam (`with_build_floor`).
1519        if let Err(e) = state::with_build_floor(|| document.absorb(&digested, None)) {
1520          // Same Fatal contract as the eager Build: announce, latch, keep the
1521          // partial document (recovery is a FEATURE of Fatal).
1522          e.log_fatal();
1523          emit_warn(
1524            "internal",
1525            "core_interface",
1526            &format!(
1527              "convert_streaming: build stopped early ({:?}/{:?}) — keeping the document built so far",
1528              e.target, e.category
1529            ),
1530          );
1531          fatal_stop = true;
1532          stopped = true;
1533        }
1534        fragment = Some(digested);
1535      }
1536      // Perl inserts resources directly once a document exists; the eager
1537      // path's root-hook drain is deferred here, so fold fresh arrivals in
1538      // per fragment — mid-digestion consumers (the frontmatter fallback's
1539      // resource[last()] anchor) depend on them being placed.
1540      document.process_pending_resources_at_top()?;
1541      if std::env::var_os("LXML_TRACE_FRAGMENT_RC").is_some()
1542        && let Some(f) = fragment.as_ref()
1543      {
1544        trace_fragment_holders(std::slice::from_ref(f));
1545      }
1546      drop(fragment);
1547      let finishing = stopped || (!yielded && !gullet::has_more_input());
1548      if finishing
1549        && !fatal_stop
1550        && let Err(e) = state::with_build_floor(|| close_abandoned_document(document))
1551      {
1552        e.log_fatal();
1553      }
1554      // Spill policy at the end: a conversion that NEVER yielded fits in RAM
1555      // whole — spilling it would buy no headroom and cost a full
1556      // serialize→reparse→reserialize cycle in pass 2 (measured +38% wall
1557      // time at a roomy cap). But once yields happened, the ceiling is real
1558      // and the FINAL fragment must spill like every other one: retaining it
1559      // hands the eager-style spine tail (rewrites, whole-doc math parse,
1560      // finalize) everything the last fragment held — the witness died in
1561      // exactly that tail at 24.5 GB after a perfectly bounded pass 1.
1562      let bounded_mode = stomach::fragment_yield_count() > 0;
1563      if !finishing || bounded_mode {
1564        let t_spill = std::time::Instant::now();
1565        #[cfg(target_os = "linux")]
1566        let trace_spill = std::env::var_os("LXML_TRACE_NODE_BOXES").is_some();
1567        #[cfg(target_os = "linux")]
1568        let c_live_before = if trace_spill {
1569          unsafe { libc::mallinfo2() }.uordblks / (1024 * 1024)
1570        } else {
1571          0
1572        };
1573        let runs_spilled = document.spill_closed_subtrees(index)?;
1574        let t_spill_dur = t_spill.elapsed();
1575        #[cfg(target_os = "linux")]
1576        if trace_spill {
1577          let c_live_after = unsafe { libc::mallinfo2() }.uordblks / (1024 * 1024);
1578          eprintln!(
1579            "streaming: spill C-live {c_live_before} -> {c_live_after} MB ({runs_spilled} runs)"
1580          );
1581        }
1582        // Self-healing: entries for nodes that build-time discard paths
1583        // detached without purging pin whole Digested box trees (see
1584        // sweep_stale_node_boxes). Spilled subtrees are purged as they spill
1585        // (`index_and_purge_spilled`, `discard_subtree`), so the sweep is only
1586        // this backstop. It marks the whole live DOM, so it runs when
1587        // `node_boxes` has grown by an eighth since the last sweep (at least
1588        // 256 entries, and never more than the absolute threshold). That is
1589        // amortized O(1) per entry while the live DOM is proportional to
1590        // `node_boxes`; resident nodes whose entries were released make each
1591        // sweep cost more. Sweeping after every spill cost O(resident DOM) on
1592        // each of 3,000-5,500 yields once a glossary block of ~3,000
1593        // definitions stayed resident at the root (datatool-user 11.6 s,
1594        // glossaries-extra-manual 82.4 s; roadmap stream C, 2026-09-25). A
1595        // spill that purged live entries lowers the baseline with them, so
1596        // stale entries cannot regrow to the old high-water mark; and the
1597        // finishing yield sweeps, so none is pinned through pass 2 and the
1598        // spine tail.
1599        document.last_swept_node_boxes_len = document
1600          .last_swept_node_boxes_len
1601          .min(document.node_boxes.len());
1602        let growth = document
1603          .node_boxes
1604          .len()
1605          .saturating_sub(document.last_swept_node_boxes_len);
1606        let relative = (document.last_swept_node_boxes_len / 8).max(256);
1607        let should_sweep =
1608          (finishing && runs_spilled > 0) || growth >= relative.min(node_boxes_sweep_threshold());
1609        if should_sweep {
1610          let t_sweep = std::time::Instant::now();
1611          document.sweep_stale_node_boxes();
1612          let t_sweep_dur = t_sweep.elapsed();
1613          if std::env::var_os("LXML_TRACE_NODE_BOXES").is_some() {
1614            eprintln!(
1615              "streaming: spill took {t_spill_dur:?} (spilled {runs_spilled} runs), sweep took {t_sweep_dur:?}"
1616            );
1617          }
1618        }
1619        // Rate-limited: a book-scale run yields MILLIONS of fragments, and
1620        // every log line lands in the in-RAM LOG_BUFFER — per-fragment
1621        // telemetry alone wrote a 1.37 GB log on the 131 MB witness,
1622        // feeding the very creep pass 1 exists to avoid.
1623        let fragments = stomach::fragment_yield_count();
1624        // Hand freed spill memory back to the OS. The spilled DOM lives on
1625        // GLIBC's heap (libxml2 allocates via libc malloc, NOT the Rust
1626        // global allocator), and glibc keeps freed mid-heap pages mapped —
1627        // measured on the 131 MB witness: every probed Rust collection flat,
1628        // C live-heap 4.45 GB peak (heaptrack), yet RSS crept 22→36 GB into
1629        // the fuse. `malloc_trim(0)` madvises free pages away (glibc ≥2.26
1630        // releases mid-heap pages too, not just the top). Rate-limited: a
1631        // trim walks the heap, and a book-scale run yields millions of
1632        // fragments.
1633        #[cfg(target_os = "linux")]
1634        if fragments.is_multiple_of(4096) {
1635          unsafe {
1636            libc::malloc_trim(0);
1637          }
1638        }
1639        // The Rust-side analogue: mimalloc (the global allocator) retains
1640        // freed pages; a forced collect purges them back to the OS. Gated
1641        // off under dhat-heap, whose tracking allocator replaces mimalloc.
1642        #[cfg(not(feature = "dhat-heap"))]
1643        if fragments.is_multiple_of(4096) {
1644          unsafe {
1645            libmimalloc_sys::mi_collect(true);
1646          }
1647        }
1648        if fragments.is_power_of_two() || fragments.is_multiple_of(65536) {
1649          let (index_ids, index_labels, _) = index.sizes();
1650          // C-heap split (glibc only manages libxml2's allocations — Rust
1651          // goes through mimalloc): uordblks = live C bytes, fordblks =
1652          // freed-but-retained. Together with RSS this separates C live
1653          // growth / C fragmentation / Rust growth.
1654          #[cfg(target_os = "linux")]
1655          let (c_live_mb, c_free_mb) = {
1656            let mi = unsafe { libc::mallinfo2() };
1657            (mi.uordblks / (1024 * 1024), mi.fordblks / (1024 * 1024))
1658          };
1659          #[cfg(not(target_os = "linux"))]
1660          let (c_live_mb, c_free_mb) = (0usize, 0usize);
1661          let spine_children = document
1662            .get_document()
1663            .get_root_element()
1664            .map(|r| r.get_child_nodes().len())
1665            .unwrap_or(0);
1666          let (mouths, comments) = gullet::queue_sizes();
1667          emit_info(
1668            "streaming",
1669            "progress",
1670            &format!(
1671              "streaming: undo {}; mouths {}; comments {}; node_boxes {}",
1672              state::undo_depth(),
1673              mouths,
1674              comments,
1675              document.node_boxes.len(),
1676            ),
1677          );
1678          emit_info(
1679            "streaming",
1680            "progress",
1681            &format!(
1682              "streaming: fragment {} absorbed; {} segment(s) staged to disk; RSS ~{} MB; C-live {} MB; C-free {} MB; root-children {}; idstore {}; index {}+{}; arena {}; fonts {}; pending {}",
1683              fragments,
1684              latexml_core::document::spilled_segment_count(),
1685              stomach::last_sampled_rss_kb() / 1024,
1686              c_live_mb,
1687              c_free_mb,
1688              spine_children,
1689              document.idstore.len(),
1690              index_ids,
1691              index_labels,
1692              arena::len(),
1693              document.node_fonts.len(),
1694              document.pending.len(),
1695            ),
1696          );
1697        }
1698      }
1699      if finishing {
1700        break;
1701      }
1702    }
1703    Ok(fatal_stop)
1704  }

1706  /// Streaming tail: root hooks, pass 2 over the spilled segments, finalize.
1707  fn stream_finish(
1708    &mut self,
1709    mut document: Document,
1710    index: latexml_core::sxml::FragmentIndex,
1711    fatal_stop: bool,
1712    phase_clock: std::time::Instant,
1713  ) -> Result<Document> {
1714    document.set_defer_root_after_open(false);
1715    // Late-arrived frontmatter first (canonically positioned after what the
1716    // mid-digestion insertion already placed), then the root's deferred late
1717    // hooks (which perform the whole insertion for documents where nothing
1718    // ran early).
1719    latexml_engine::base_utilities::insert_late_frontmatter(&mut document)?;
1720    document.dispatch_deferred_root_hooks()?;
1721    if fatal_stop {
1722      // CHEAP partial (Fatal contract under real memory pressure): every
1723      // further phase allocates while we are AT the ceiling — the first
1724      // implementation marched from the 18.8 GB cooperative fuse straight
1725      // into the 24 GB hard watchdog (SIGKILL) doing pass 2. Skip pass 2 and
1726      // the spine tail: spilled segments splice in their raw pre-finalize
1727      // form — well-formed XML that still carries `_`-bookkeeping attributes,
1728      // which a salvaged partial is allowed to. The verdict is already
1729      // latched Fatal.
1730      emit_warn(
1731        "internal",
1732        "core_interface",
1733        "convert_streaming: fatal stop — emitting the cheap partial (pass 2 and the finalize tail skipped to avoid allocating at the memory ceiling)",
1734      );
1735      // The partial's serialization must still RESOLVE placeholders (raw
1736      // segments splice recursively) — literal mode was for pass 1 only.
1737      document.literal_placeholders = false;
1738      // Flat serialization was for the spill INTERMEDIATE only — the output
1739      // keeps its formatting.
1740      document.spill_flat = false;
1741      return Ok(document);
1742    }
1743    // RDFa prefixes used inside spilled content: the finalize scan can no
1744    // longer see them, so feed the spill-time record in.
1745    document.add_extra_rdfa_prefixes(index.rdfa_prefixes().map(|(p, _)| p));
1746
1747    // The rule set must be complete before ANY rewrites run: the source's
1748    // `.latexml` file loads only now (as in the eager path), and fragments
1749    // must see it too — which is exactly why pass 1 applies no rewrites.
1750    load_source_latexml_rules();
1751    // Pass 2 mutates segments while fragment documents live independently, so
1752    // it borrows the store OUT of the spine and hands it back for assembly.
1753    let mut store = document
1754      .take_spill_store()
1755      .expect("attached above; nothing detaches it during pass 1");
1756    let node_fonts = document.node_fonts.clone();
1757    // Root-level id counters run document-wide (see streaming_pass2's doc):
1758    // seed pass 2 from the spine root's pass-1 state, and hand the final
1759    // state back to the root so the spine's own tail continues the sequence.
1760    let mut counters: Vec<(String, String)> = document
1761      .get_document()
1762      .get_root_element()
1763      .map(|root| latexml_core::document::id_counters(&root))
1764      .unwrap_or_default();
1765    // So do those of each element a spilled run's top nodes are minted under (SegmentMeta::anchor).
1766    let mut anchors = document.take_spill_anchor_counters();
1767    // The document's labels, for scope resolution in every fragment and on the
1768    // spine (Document::rewrite_labels_shared, consulted on a local miss): the
1769    // spilled content's, from the pass-1 index, and those on nodes that stay on
1770    // the spine (a top-level theorem or table) — read, not minted, so pass 2's id
1771    // sequence is unchanged. Built once and shared: copying it into each
1772    // fragment's own map was quadratic (28,068 labels × 459,579 segments on the
1773    // 131 MB witness = 12.9 billion String allocations, dominating pass 2).
1774    let shared_labels: Option<Rc<rustc_hash::FxHashMap<String, String>>> = matches!(
1775      state::lookup_value("DOCUMENT_REWRITE_RULES"),
1776      Some(Stored::VecDequeStored(_))
1777    )
1778    .then(|| {
1779      let mut labels: rustc_hash::FxHashMap<String, String> = index
1780        .labels()
1781        .map(|(label, id)| (label.to_string(), id.to_string()))
1782        .collect();
1783      for node in document.findnodes("//*[@labels]", None) {
1784        if let (Some(node_labels), Some(id)) =
1785          (node.get_attribute("labels"), node.get_attribute("id"))
1786        {
1787          // A label carried twice (a duplicated `\label`) keeps the spilled copy's
1788          // id here, and each fragment resolves its own copy first; Perl's
1789          // `rewrite` fills its map in document order, so the last copy wins
1790          // (OXIDIZED_DESIGN #360: the index keeps no positions).
1791          for label in node_labels.split_whitespace() {
1792            labels
1793              .entry(label.to_string())
1794              .or_insert_with(|| id.clone());
1795          }
1796        }
1797      }
1798      Rc::new(labels)
1799    });
1800    let pass2_start = phase_clock.elapsed();
1801    let root_id = document
1802      .get_document()
1803      .get_root_element()
1804      .and_then(|root| root.get_attribute("id"));
1805    // The spine's share of the `d`-token evidence: with every spilled subtree's, read by the spill observer, pass 2
1806    // parses each segment against the whole document's, as the eager parse does.
1807    if !state::get_nomathparse_flag()
1808      && let Some(root) = document.get_document().get_root_element()
1809    {
1810      latexml_math_parser::read_differential_evidence(&document, &root);
1811    }
1812    streaming_pass2(
1813      &mut store,
1814      &node_fonts,
1815      &mut counters,
1816      &mut anchors,
1817      &index,
1818      shared_labels.clone(),
1819      root_id,
1820    )?;
1821    emit_info(
1822      "streaming",
1823      "progress",
1824      &format!(
1825        "streaming: PASS 2 done in {:.1?} ({:.1?} cumulative) — {} segment(s), RSS ~{} MB",
1826        phase_clock.elapsed().saturating_sub(pass2_start),
1827        phase_clock.elapsed(),
1828        latexml_core::document::spilled_segment_count(),
1829        latexml_core::watchdog::process_rss_kb().unwrap_or(0) / 1024,
1830      ),
1831    );
1832    if let Some(mut root) = document.get_document().get_root_element() {
1833      for (key, value) in &counters {
1834        let _ = root.set_attribute(key, value);
1835      }
1836    }
1837    for (id, carried) in &anchors {
1838      document.set_id_counters(id, carried);
1839    }
1840    // The spine resolves scope labels against the whole document, as the
1841    // fragments do: a label on a spilled node is not missing (its scope selects
1842    // nothing here and applies in its fragment), so only a label no element
1843    // carries reports Perl's `getLabelID` error.
1844    document.rewrite_labels_shared = shared_labels;
1845    // The spine gets the normal whole-document tail; spilled content is
1846    // invisible to it (placeholders), so root-level passes act on the live
1847    // root exactly as in the eager path.
1848    finish_document(&mut document)?;
1849    // From here serialization splices the processed segments at their
1850    // placeholders (recursively — pass 1 kept nested spills nested).
1851    document.literal_placeholders = false;
1852    // Flat serialization was for the spill INTERMEDIATE only; the spine and
1853    // the spliced segment text must carry the normal output formatting.
1854    document.spill_flat = false;
1855    document.set_spill_store(store);
1856    Ok(document)
1857  }

1859  fn convert_streaming(
1860    &mut self,
1861    request: String,
1862    preamble: Option<String>,
1863    postamble: Option<String>,
1864    mode: Option<DigestionMode>,
1865    budget: usize,
1866  ) -> Result<Document> {
1867    let digestion_note = self.digest_setup(request, preamble, postamble, mode)?;
1868    let mut document = build_document_head(&self.preload)?;
1869    let mut index = self.stream_setup(&mut document, budget)?;
1870    // Phase clock starts after setup (spill store, disk gate), as it always did.
1871    let phase_clock = std::time::Instant::now();
1872    let fatal_stop = self.stream_pass1(&mut document, &mut index, None)?;
1873    stomach::set_fragment_yield_budget(None);
1874    stomach::set_fragment_yield_rss_soft_kb(None);
1875    gullet::flush();
1876    note_end(&digestion_note);
1877    let pass1_elapsed = phase_clock.elapsed();
1878    emit_info(
1879      "streaming",
1880      "progress",
1881      &format!(
1882        "streaming: PASS 1 done in {:.1?} — {} yield(s), {} segment(s) staged to disk, RSS ~{} MB",
1883        pass1_elapsed,
1884        stomach::fragment_yield_count(),
1885        latexml_core::document::spilled_segment_count(),
1886        latexml_core::watchdog::process_rss_kb().unwrap_or(0) / 1024,
1887      ),
1888    );
1889
1890    // The ROOT's after-open hooks were DEFERRED during pass 1
1891    // (`set_defer_root_after_open`): eager semantics guarantee every hook
1892    // runs with digestion complete, and firing them at fragment 1 was not
1893    // merely incomplete but harmful — the frontmatter hook consumed EMPTY
1894    // frontmatter state and marked it done (losing the abstract; sweep
1895    // witness tests/structure/abstract.tex), and the root-classes hook read
1896    // a `DOCUMENT_CLASSES` mapping `\maketitle` had not populated yet
1897    // (dropping `ltx_authors_1line`; gate witness). Dispatch exactly once,
1898    // now, with digestion finished — the eager timing.
1899    self.stream_finish(document, index, fatal_stop, phase_clock)
1900  }

1902  fn digest_adaptive(
1903    &mut self,
1904    request: String,
1905    preamble: Option<String>,
1906    postamble: Option<String>,
1907    mode: Option<DigestionMode>,
1908    transition_budget: usize,
1909  ) -> Result<AdaptiveOutcome> {
1910    use latexml_core::common::error::ErrorCategory;
1911    let digestion_note = self.digest_setup(request, preamble, postamble, mode)?;
1912    // Arm the fragment-yield seam unless a caller already did (the yield
1913    // tests pre-set their own knobs): the box budget is never the trigger
1914    // here — only the soft-RSS watermark (`spill_watermark_bytes`, live at
1915    // `--max-memory=0` as RAM/8) and the constructs that request a seam
1916    // (`\pgfsys@endpicture`). A yield below the watermark just returns the
1917    // accumulated top-level bodies, which this loop keeps collecting: the
1918    // concatenation of the yielded fragments is the eager box list
1919    // (`114_streaming_*`).
1920    // Only the soft-RSS trigger is armed here — never a box budget, so a
1921    // picture seam with room to spare does not yield (the eager path stays
1922    // exactly the eager path until the watermark). A caller that pre-armed
1923    // the knobs (the yield tests) keeps pure eager yield semantics: no
1924    // transition.
1925    let armed_here =
1926      stomach::fragment_yield_budget().is_none() && stomach::fragment_yield_rss_soft_kb().is_none();
1927    if armed_here && let Some(mark_kb) = stomach::spill_mark_kb() {
1928      stomach::set_fragment_yield_rss_soft_kb(Some(mark_kb));
1929      // The growth floor measures the DOCUMENT's accumulation: baseline on
1930      // the resident set now, after the engine/dump load.
1931      stomach::rebaseline_soft_yield_growth();
1932    }
1933    let disarm = |armed: bool| {
1934      if armed {
1935        stomach::set_fragment_yield_rss_soft_kb(None);
1936      }
1937    };
1938    let mut boxes = RESCUED_BOXES.with(|r| std::mem::take(&mut *r.borrow_mut()));
1939    while gullet::has_more_input() {
1940      match digest_step_guarded(&mut boxes) {
1941        Ok(true) => {},
1942        Ok(false) => break,
1943        Err(e) => {
1944          disarm(armed_here);
1945          if !matches!(e.category, ErrorCategory::StreamingRestart) {
1946            hard_yank_processing(boxes);
1947          }
1948          return Err(e);
1949        },
1950      }
1951      // TRANSITION: an RSS-driven yield — the process is over the spill
1952      // watermark at a legal seam. Hand what is accumulated to streaming
1953      // pass 1 as fragment 1 and continue digesting from the same gullet
1954      // position: one digest, bounded memory, no from-scratch restart (which
1955      // re-read the whole source: the >180 s component on source2e/source3/
1956      // datatool/glossaries/pgf-spectra LSE). Yields below the watermark (a
1957      // picture seam with room to spare) keep accumulating — a document that
1958      // never crosses it finishes exactly as before, whole, with no spill.
1959      if stomach::take_fragment_yielded()
1960        && armed_here
1961        && stomach::fragment_yield_rss_soft_kb()
1962          .is_some_and(|soft| stomach::last_sampled_rss_kb() > soft)
1963      {
1964        stomach::set_streaming_restart_watermark(None);
1965        emit_info(
1966          "streaming",
1967          "transition",
1968          &format!(
1969            "streaming: eager digestion crossed the spill watermark at a yield seam (RSS ~{} MB, {} top-level bodies) — continuing as streaming pass 1",
1970            stomach::last_sampled_rss_kb() / 1024,
1971            boxes.len()
1972          ),
1973        );
1974        // Every failure inside the transition (the spill disk gate, a spill or
1975        // resource error in pass 1, pass 2) must leave NO yield knob armed —
1976        // they are thread-locals a persistent worker (cortex_worker) carries
1977        // into its next paper, where an armed knob would silently disable this
1978        // lever (`armed_here` false) — and must not fall into the eager
1979        // `digest_internal` salvage, which would re-digest the gullet tail
1980        // into a fresh document while the spilled front is gone: the gullet is
1981        // flushed, so the salvage finds nothing, and the Fatal is reported
1982        // exactly as `convert_streaming`'s own failure would be.
1983        let seed = std::mem::take(&mut boxes);
1984        let transition = |this: &mut Self| -> Result<(Document, bool)> {
1985          let phase_clock = std::time::Instant::now();
1986          let mut document = build_document_head(&this.preload)?;
1987          let mut index = this.stream_setup(&mut document, transition_budget)?;
1988          let fatal_stop = this.stream_pass1(&mut document, &mut index, Some(seed))?;
1989          stomach::set_fragment_yield_budget(None);
1990          stomach::set_fragment_yield_rss_soft_kb(None);
1991          gullet::flush();
1992          emit_info(
1993            "streaming",
1994            "progress",
1995            &format!(
1996              "streaming: PASS 1 done in {:.1?} — {} yield(s), {} segment(s) staged to disk, RSS ~{} MB",
1997              phase_clock.elapsed(),
1998              stomach::fragment_yield_count(),
1999              latexml_core::document::spilled_segment_count(),
2000              latexml_core::watchdog::process_rss_kb().unwrap_or(0) / 1024,
2001            ),
2002          );
2003          this
2004            .stream_finish(document, index, fatal_stop, phase_clock)
2005            .map(|document| (document, fatal_stop))
2006        };
2007        let outcome = transition(self);
2008        stomach::set_fragment_yield_budget(None);
2009        stomach::set_fragment_yield_rss_soft_kb(None);
2010        note_end(&digestion_note);
2011        return match outcome {

2482/// Where an `\lxDeclare` applies (Perl `Rewrite::compile_clause`'s `scope` clause, Rewrite.pm:298-311):
2483/// the whole document (no scope, or one Perl does not recognize and ignores), the element with an
2484/// id and what it holds (`id:S2`; a `label:` resolved to its element's id), or nowhere — a label no
2485/// element carries, which Perl compiles to an id-less XPath that selects nothing.
2486enum DeclarationScope {
2487  Everywhere,
2488  Within(String),
2489  Nowhere,
2490}

2492impl DeclarationScope {
2493  fn resolve(document: &Document, scope: &str) -> Self {
2494    if scope.is_empty() {
2495      DeclarationScope::Everywhere
2496    } else if let Some(id) = scope.strip_prefix("id:") {
2497      DeclarationScope::Within(id.to_string())

2499      document
2500        .lookup_rewrite_label(label)
2501        .or_else(|| document.lookup_rewrite_label(&format!("LABEL:{label}")))
2502        .map_or(DeclarationScope::Nowhere, DeclarationScope::Within)
2503    } else {
2504      DeclarationScope::Everywhere
2505    }

2509  /// the id is one of the fragment's ancestors (`Document::fragment_ancestor_ids`).
2510  fn covers(&self, document: &Document, node: &libxml::tree::Node) -> bool {
2511    match self {
2512      DeclarationScope::Everywhere => true,
2513      DeclarationScope::Nowhere => false,
2514      DeclarationScope::Within(id) => {
2515        if document.fragment_ancestor_ids.contains(id) {
2516          return true;
2517        }

2521            .get_attribute_ns("id", latexml_core::common::xml::XML_NS)
2522            .as_deref()
2523            == Some(id.as_str())
2524            || element.get_property("id").as_deref() == Some(id.as_str())
2525          {
2526            return true;
2527          }
2528          current = element.get_parent();
2529        }
2530        false
2531      },
2532    }
2533  }

2548  // Parse declarations:
2549  // "token_text\trole\tname\tmeaning\tdecl_id\tmatch_font\tscope".
2550  // match_font (font_attribute_string of the digested pattern, e.g.
2551  // "italic"/"bold") makes matching font-aware: a plain italic `$x$` declaration
2552  // must not annotate a bold `\mathbf{x}` — different fonts denote different
2553  // meanings; empty when the pattern carried no distinguishing font.
2554  // scope is the declaration's (`get_declaration_scope`), resolved here as the
2555  // rewrite rule resolves it (`Rewrite::compile_clause`).
2556  let declarations: Vec<(&str, &str, &str, &str, &str, &str, DeclarationScope)> = decls_str
2557    .lines()
2558    .filter_map(|line| {
2559      let parts: Vec<&str> = line.splitn(7, '\t').collect();
2560      if parts.len() >= 4 {
2561        Some((
2562          parts[0],
2563          parts[1],
2564          parts[2],
2565          parts[3],
2566          *parts.get(4).unwrap_or(&""),
2567          *parts.get(5).unwrap_or(&""),
2568          DeclarationScope::resolve(document, parts.get(6).unwrap_or(&"")),
2569        ))
2570      } else {
2571        None
2572      }
2573    })
2574    .collect();
2575
2576  if declarations.is_empty() {
2577    return;
2578  }
2579
2580  // Find all XMTok elements in the document and apply matching declarations.
2581  // Skip tokens already marked by the rewrite system (_matched) — these were
2582  // handled by subscript/prime/wildcard patterns which take precedence.
2583  let xmtoks = document.findnodes("descendant-or-self::ltx:XMTok", None);
2584  for mut tok in xmtoks {
2585    if tok.has_attribute("_matched") {
2586      continue;
2587    }
2588    let content = tok.get_content();
2589    let tok_name = tok.get_attribute("name").unwrap_or_default();
2590
2591    // The latest declaration wins, as Perl's `UnshiftValue` puts each rule before the earlier ones
2592    // and a matched node is skipped by the rest (latexml.sty.ltxml:564; Rewrite.pm select).
2593    for (pattern, role, name, meaning, decl_id, match_font, scope) in declarations.iter().rev() {
2594      let (pattern, role, name, meaning, decl_id, match_font) =
2595        (*pattern, *role, *name, *meaning, *decl_id, *match_font);
2596      // Match by content text, or by XMTok name attribute (for CS patterns like \circ)
2597      let matches = content == pattern
2598        || (!tok_name.is_empty() && pattern.starts_with('\\') && pattern[1..] == tok_name);
2599      if matches {
2600        // Font-class check (mirrors declare_node_matches in the rewrite path):
2601        // discriminate only on the meaningful font *classes* — bold vs not,
2602        // caligraphic/typewriter family — NOT on an exact font-string match.
2603        // Exact equality is wrong here because the declaration's digested frame
2604        // font (e.g. "italic") differs from an upright operator token's font
2605        // even when they should match (e.g. `$*$` → the ∗ COMPOSEOP token).
2606        // A plain/italic declaration rejects bold/caligraphic/typewriter tokens
2607        // (so italic `$x$` skips bold `\mathbf{x}`); a declaration that is
2608        // itself bold/caligraphic/typewriter requires the token to share it.
2609        {
2610          let tf = document.get_node_font(&tok);
2611          let tok_bold = tf
2612            .get_series()
2613            .map(|s| s.as_ref() == "bold")
2614            .unwrap_or(false);
2615          let tok_fam = tf.get_family();
2616          let tok_cal = tok_fam
2617            .as_ref()
2618            .map(|f| f.as_ref() == "caligraphic")
2619            .unwrap_or(false);
2620          let tok_tt = tok_fam
2621            .as_ref()
2622            .map(|f| f.as_ref() == "typewriter")
2623            .unwrap_or(false);
2624          let decl_bold = match_font.contains("bold");
2625          let decl_cal = match_font.contains("caligraphic");
2626          let decl_tt = match_font.contains("typewriter");
2627          if tok_bold != decl_bold || tok_cal != decl_cal || tok_tt != decl_tt {
2628            continue;
2629          }
2630        }
2631        if !scope.covers(document, &tok) {
2632          continue; // Outside the declaration's scope
2633        }
2634        if !role.is_empty() {
2635          let _ = tok.set_attribute("role", role);
2636        }
2637        if !name.is_empty() {
2638          let _ = tok.set_attribute("name", name);
2639        }
2640        if !meaning.is_empty() {
2641          let _ = tok.set_attribute("meaning", meaning);
2642        }
2643        if !decl_id.is_empty() {
2644          let _ = tok.set_attribute("decl_id", decl_id);
2645        }
2646        break; // The latest matching declaration wins
2647      }
2648    }
2649  }
2650}

2652/// Fallback parser for unparseable math expressions.
2653/// Perl: MathParser.pm parse_kludge().
2654/// Balances OPEN/CLOSE delimiters by wrapping matched groups in XMWrap.
2655/// Uses document.wrap_nodes for proper namespace handling.
2656/// Renumber xml:ids inside parsed XMath subtrees so they are sequential in
2657/// document order. The Marpa parser explores multiple parse alternatives,
2658/// consuming ID counter slots for pruned nodes (e.g. m1.1, m1.7, m1.12
2659/// instead of m1.1, m1.2, m1.3). This pure post-processing pass reassigns
2660/// IDs after all pruning is complete.
2661///
2662/// Optimized: single DFS walk per XMath (not XPath), O(1) parent-prefix
2663/// lookup via ID string parsing, and allocation reuse across Math nodes.
2664fn renumber_math_ids(document: &mut Document) {
2665  let xml_ns = "http://www.w3.org/XML/1998/namespace";
2666  let math_nodes = document.findnodes("descendant-or-self::ltx:Math[@text]", None);
2667
2668  // Reuse allocations across Math nodes
2669  let mut id_entries: Vec<(libxml::tree::Node, String)> = Vec::new();
2670  let mut idref_entries: Vec<(libxml::tree::Node, String)> = Vec::new();
2671  let mut id_map: rustc_hash::FxHashMap<String, String> = rustc_hash::FxHashMap::default();
2672  let mut referenced_ids: rustc_hash::FxHashSet<String> = rustc_hash::FxHashSet::default();
2673
2674  for mut math_node in math_nodes {
2675    let math_id = match math_node.get_attribute_ns("id", xml_ns) {
2676      Some(id) => id,
2677      None => continue,
2678    };
2679
2680    let xmath_nodes = document.findnodes("ltx:XMath", Some(&math_node));
2681    for xmath in xmath_nodes {
2682      id_entries.clear();
2683      idref_entries.clear();
2684      id_map.clear();
2685      referenced_ids.clear();
2686
2687      // Single DFS walk collects both xml:id and idref nodes in document order
2688      renumber_collect_dfs(&xmath, xml_ns, &mut id_entries, &mut idref_entries);
2689      if id_entries.is_empty() {
2690        continue;
2691      }
2692
2693      // Collect all referenced IDs (from XMRef idref attributes)
2694      for (_, idref) in &idref_entries {
2695        referenced_ids.insert(idref.clone());
2696      }
2697
2698      // Strip xml:id from XMTok elements that are not referenced by any XMRef.
2699      // The math parser assigns xml:ids to all tokens during parsing, but only
2700      // structural nodes (XMApp, XMDual) and explicitly referenced tokens need them.
2701      // Orphan XMTok ids inflate the renumbering counter causing ID gaps.
2702      {
2703        let mut stripped = false;
2704        for (node, id) in &mut id_entries {
2705          if node.get_name() == "XMTok" && !referenced_ids.contains(id.as_str()) {
2706            document.unrecord_id(id);
2707            let _ = node.remove_attribute("xml:id");
2708            let _ = node.remove_attribute_ns("id", xml_ns);
2709            id.clear(); // mark for removal
2710            stripped = true;
2711          }
2712        }
2713        if stripped {
2714          id_entries.retain(|(_, id)| !id.is_empty());
2715        }
2716      }
2717
2718      if id_entries.is_empty() {
2719        continue;
2720      }
2721
2722      // Build old→new mapping. Flat sequential numbering under the math_id prefix,
2723      // matching Perl's approach of assigning all IDs at the same level.
2724      let mut counter = 0u32;
2725      let mut any_changed = false;
2726      for (_node, old_id) in &id_entries {
2727        counter += 1;
2728        let new_id = format!("{math_id}.{counter}");
2729        if new_id != *old_id {
2730          any_changed = true;
2731        }
2732        id_map.insert(old_id.clone(), new_id);
2733      }
2734
2735      if !any_changed {
2736        continue;
2737      }
2738
2739      // Apply new xml:ids in TWO passes to avoid idstore collisions.
2740      // A new id like "m1.1" would collide with an old "m1.1" still in the
2741      // idstore if we interleave unrecord+record. Strip all first, then assign.
2742      let mut nodes_to_update: Vec<(libxml::tree::Node, String)> = Vec::new();
2743      for (mut node, old_id) in id_entries.drain(..) {
2744        if let Some(new_id) = id_map.get(&old_id)
2745          && new_id != &old_id
2746        {
2747          document.unrecord_id(&old_id);
2748          let _ = node.remove_attribute("xml:id");
2749          let _ = node.remove_attribute_ns("id", xml_ns);
2750          nodes_to_update.push((node, new_id.clone()));
2751        }
2752      }
2753      for (mut node, new_id) in nodes_to_update {
2754        let _ = document.set_attribute(&mut node, "xml:id", &new_id);
2755      }
2756
2757      // Update idrefs
2758      for (mut node, old_idref) in idref_entries.drain(..) {
2759        if let Some(new_idref) = id_map.get(&old_idref)
2760          && new_idref != &old_idref
2761        {
2762          let _ = node.set_attribute("idref", new_idref);
2763        }
2764      }
2765
2766      // Reset _ID_counter__ on the Math node to the final count
2767      let _ = math_node.set_attribute("_ID_counter__", &counter.to_string());
2768    }
2769  }
2770}

2772/// DFS walk collecting nodes with xml:id and idref attributes in document order.
2773/// Stops at nested `Math` elements (which have their own parsing scope).
2774fn renumber_collect_dfs(
2775  node: &libxml::tree::Node,
2776  xml_ns: &str,
2777  id_entries: &mut Vec<(libxml::tree::Node, String)>,
2778  idref_entries: &mut Vec<(libxml::tree::Node, String)>,
2779) {
2780  if let Some(id) = node.get_attribute_ns("id", xml_ns) {
2781    id_entries.push((node.clone(), id));
2782  }
2783  if let Some(idref) = node.get_attribute("idref") {
2784    idref_entries.push((node.clone(), idref));
2785  }
2786  for child in node.get_child_elements() {
2787    // Skip nested Math elements — they have their own parsing scope
2788    if child.get_name() == "Math" {
2789      continue;
2790    }
2791    renumber_collect_dfs(&child, xml_ns, id_entries, idref_entries);
2792  }
2793}
