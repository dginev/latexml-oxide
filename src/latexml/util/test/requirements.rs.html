2//! `tex_tests!` fixture harness and the guard tests. A test whose requirement the host TeX tree does not meet is
3//! skipped VISIBLY: Perl's `Skip:` message on stderr and one `<test>\t<reason>` line in the skip log.

9/// What a test needs from the host: Perl's `requires` value, in any of its three forms (a file, a list of files,
10/// or `{packages => …, texlive_min => N, env => VAR}`), plus programs on `PATH` (a Rust addition: `jing`).
11#[derive(Debug, Clone, Copy, Default)]
12pub struct Requirements<'a> {
13  /// Files found through the engine's own kpathsea lookup, else a find from the working directory.
14  pub packages:    &'a [&'a str],
15  /// Programs found on `PATH`.
16  pub programs:    &'a [&'a str],
17  /// The minimal TeX Live year (0: any); the host year is `LATEXML_TEST_TEXLIVE` or the engine's detected year.
18  pub texlive_min: u32,
19  /// An environment variable that must be set (Perl's CI-only tests, `env => 'CI'`).
20  pub env:         Option<&'a str>,
21  /// A latex kernel dump on disk ([`super::dump_available`]): without one the engine raw-loads the kernel and the
22  /// error landscape is the raw load's, not the test's.
23  pub format_dump: bool,
24}
25
26impl<'a> Requirements<'a> {
27  /// The files `packages` (Perl's scalar and list forms).
28  pub const fn packages(packages: &'a [&'a str]) -> Self {
29    Requirements {
30      packages,
31      programs: &[],
32      texlive_min: 0,
33      env: None,
34      format_dump: false,
35    }
36  }
37  /// Also require the programs `programs` on `PATH`.
38  pub const fn programs(self, programs: &'a [&'a str]) -> Self { Requirements { programs, ..self } }
39  /// Also require TeX Live `year` or later.
40  pub const fn texlive_min(self, year: u32) -> Self { Requirements { texlive_min: year, ..self } }
41  /// Also require the environment variable `var` to be set.
42  pub const fn env(self, var: &'a str) -> Self { Requirements { env: Some(var), ..self } }
43  /// Also require a latex kernel dump on disk.
44  pub const fn format_dump(self) -> Self { Requirements { format_dump: true, ..self } }
45
46  /// Why `test` cannot run here, in Perl's words; `None` when every requirement is met. Perl's order: the packages,
47  /// (the programs,) then `texlive_min`, then `env`.
48  pub fn unmet(&self, test: &str) -> Option<String> {
49    if let Some(file) = self.packages.iter().find(|file| !file_available(file)) {
50      return Some(format!("Missing requirement {file} for {test}"));
51    }
52    if let Some(program) = self
53      .programs
54      .iter()
55      .find(|program| !program_available(program))
56    {
57      return Some(format!("Missing program {program} for {test}"));
58    }
59    if self.texlive_min > 0 && texlive_version() < self.texlive_min {
60      return Some(format!(
61        "Minimal texlive {} requirement not met for {test}",
62        self.texlive_min
63      ));
64    }
65    if let Some(var) = self.env
66      && std::env::var_os(var).is_none_or(|value| value.is_empty())
67    {
68      return Some(format!(
69        "{test} is only checked in continuous integration. (set {var}=true)"
70      ));
71    }
72    if self.format_dump && !super::dump_available() {
73      return Some(format!(
74        "No latex kernel dump for {test} (run tools/make_formats.sh)"
75      ));
76    }
77    None
78  }
79
80  /// True when `test` may run; otherwise records the skip ([`skip`]) and returns false.
81  pub fn check(&self, test: &str) -> bool {
82    match self.unmet(test) {
83      None => true,
84      Some(reason) => {
85        skip(test, &reason);
86        false
87      },
88    }
89  }
90
91  /// [`Requirements::check`] for the running `#[test]`, named by its thread.
92  pub fn met(&self) -> bool { self.check(&current_test_name()) }
93}

95/// The guard form: `if !check_requirements(&["tcolorbox.sty"]) { return; }` skips the running test, visibly, when a file
96/// is missing from the host TeX tree.
97pub fn check_requirements(files: &[&str]) -> bool { Requirements::packages(files).met() }
98
99/// [`check_requirements`] for a test that needs several load sets ([`loads`]): `check_all_requirements(&[loads::PSTRICKS, &["x.sty"]])`.
100pub fn check_all_requirements(sets: &[&[&str]]) -> bool {
101  let files: Vec<&str> = sets.iter().flat_map(|set| set.iter().copied()).collect();
102  check_requirements(&files)
103}

182/// Perl's `pathname_kpsewhich($reqmt) || pathname_find($reqmt)`, through the lookup the conversion itself uses.
183pub fn file_available(file: &str) -> bool {
184  pathname::kpsewhich(&[file]).is_some()
185    || pathname::find(file, pathname::PathnameFindOptions::default()).is_some()
186}

188/// An executable `program` (or `program.exe`) in a `PATH` directory.
189pub fn program_available(program: &str) -> bool {
190  let Some(path) = std::env::var_os("PATH") else {
191    return false;
192  };
193  std::env::split_paths(&path).any(|dir| {
194    [program.to_string(), format!("{program}.exe")]
195      .iter()
196      .any(|name| dir.join(name).is_file())
197  })
198}

200/// Perl's `texlive_version`: `LATEXML_TEST_TEXLIVE`, else the host TeX Live year the engine resolves its dumps by,
201/// else 0.
202pub fn texlive_version() -> u32 {
203  std::env::var("LATEXML_TEST_TEXLIVE")
204    .ok()
205    .and_then(|year| year.trim().parse().ok())
206    .or_else(latexml_engine::dump_paths::detect_ambient_texlive_year)
207    .unwrap_or(0)
208}

210/// Record a skipped test: Perl's `diag("Skip: …")` on stderr and one line in [`skip_log_path`]. The only writer of
211/// the skip log.
212pub fn skip(test: &str, reason: &str) {
213  eprintln!("Skip: {reason}");
214  let line = format!("{test}\t{reason}\n");
215  let path = skip_log_path();
216  // One `write_all` of one line on an O_APPEND file: nextest's concurrent test processes do not interleave.
217  let written = std::fs::OpenOptions::new()
218    .create(true)
219    .append(true)
220    .open(&path)
221    .and_then(|mut log| log.write_all(line.as_bytes()));
222  if let Err(err) = written {
223    eprintln!("Skip log {}: {err}", path.display());
224  }
225}

227/// `LATEXML_TEST_SKIP_LOG`, else `test-skips.log` in the cargo target directory of the running test binary
228/// (`<target>/<profile>/deps/<binary>`), else in the temporary directory.
229pub fn skip_log_path() -> PathBuf {
230  #[cfg(test)]
231  if let Some(path) = tests::LOG_OVERRIDE.with(|cell| cell.borrow().clone()) {
232    return path;
233  }
234  if let Some(path) = std::env::var_os("LATEXML_TEST_SKIP_LOG") {
235    return PathBuf::from(path);
236  }
237  std::env::current_exe()
238    .ok()
239    .and_then(|exe| {
240      let deps = exe.parent()?;
241      (deps.file_name()? == "deps").then(|| {
242        deps
243          .parent()?
244          .parent()
245          .map(|target| target.join("test-skips.log"))
246      })?
247    })
248    .unwrap_or_else(|| std::env::temp_dir().join("latexml-test-skips.log"))
249}

251/// The running `#[test]`'s path (`module::test`): libtest names each test's thread after it.
252pub fn current_test_name() -> String {
253  std::thread::current()
254    .name()
255    .unwrap_or("<unnamed test>")
256    .to_string()

259/// The fixture harness's table: Perl's `requires => {name => …, '*' => …}`, a test's own key and `'*'` (every
260/// test of the directory).
261pub type RequirementTable = [(&'static str, Requirements<'static>)];
262
263/// Perl's `check_requirements($test, 1, $$requires{'*'}, $$requires{$name})` for fixture `name` of `dirpath`.
264pub fn check_table(dirpath: &str, name: &str, table: Option<&RequirementTable>) -> bool {
265  let test = format!("{dirpath}/{name}");
266  table
267    .into_iter()
268    .flatten()
269    .filter(|(key, _)| *key == "*" || *key == name)
270    .all(|(_, reqs)| reqs.check(&test))
271}

279  // Each row writes its skips to its own log, so the rows run in parallel.
280  fn with_log(tag: &str, run: impl FnOnce() -> bool) -> (bool, String) {
281    let path =
282      std::env::temp_dir().join(format!("latexml-req-test-{}-{tag}.log", std::process::id()));
283    let _ = std::fs::remove_file(&path);
284    let ran = LOG_OVERRIDE.with(|cell| {
285      *cell.borrow_mut() = Some(path.clone());
286      let ran = run();
287      *cell.borrow_mut() = None;
288      ran
289    });
290    let log = std::fs::read_to_string(&path).unwrap_or_default();
291    let _ = std::fs::remove_file(&path);
292    (ran, log)
293  }
294
295  thread_local! {
296    pub(super) static LOG_OVERRIDE: RefCell<Option<PathBuf>> = const { RefCell::new(None) };
297  }

299  #[test]
300  fn requirement_forms() {
301    const PRESENT: &str = "article.cls";
302    const ABSENT: &str = "no-such-package-req1.sty";
303    let (present, absent) = (PRESENT, ABSENT);
304    if !file_available(present) {
305      skip(
306        "requirement_forms",
307        "Missing requirement article.cls (no TeX tree)",
308      );
309      return;
310    }
311    let year = texlive_version();
312    let rows: &[(&str, Requirements, Option<String>)] = &[
313      ("file present", Requirements::packages(&[PRESENT]), None),
314      (
315        "file absent",
316        Requirements::packages(&[ABSENT]),
317        Some(format!("Missing requirement {absent} for t")),
318      ),
319      (
320        "list, first absent wins",
321        Requirements::packages(&[PRESENT, ABSENT, "also-absent.sty"]),
322        Some(format!("Missing requirement {absent} for t")),
323      ),
324      (
325        "texlive_min met",
326        Requirements::packages(&[PRESENT]).texlive_min(year),
327        None,
328      ),
329      (
330        "texlive_min unmet",
331        Requirements::packages(&[PRESENT]).texlive_min(year + 1),
332        Some(format!(
333          "Minimal texlive {} requirement not met for t",
334          year + 1
335        )),
336      ),
337      ("env set", Requirements::default().env("PATH"), None),
338      (
339        "program present",
340        Requirements::default().programs(&["cargo"]),
341        None,
342      ),
343      (
344        "program absent",
345        Requirements::packages(&[PRESENT]).programs(&["no-such-program-req1"]),
346        Some("Missing program no-such-program-req1 for t".into()),
347      ),
348      (
349        "env unset",
350        Requirements::default().env("LATEXML_REQ1_UNSET_VAR"),
351        Some(
352          "t is only checked in continuous integration. (set LATEXML_REQ1_UNSET_VAR=true)".into(),
353        ),
354      ),
355      (
356        "packages before texlive_min",
357        Requirements::packages(&[ABSENT]).texlive_min(year + 1),
358        Some(format!("Missing requirement {absent} for t")),
359      ),
360    ];
361    for (label, reqs, expected) in rows {
362      assert_eq!(&reqs.unmet("t"), expected, "{label}");
363    }
364  }

366  #[test]
367  fn a_skip_is_logged_and_a_met_requirement_runs() {
368    const ABSENT: &str = "no-such-package-req1.sty";
369    let absent = ABSENT;
370    let (ran, log) = with_log("skip", || {
371      Requirements::packages(&[ABSENT]).check("dir/name")
372    });
373    assert!(!ran);
374    assert_eq!(
375      log,
376      format!("dir/name\tMissing requirement {absent} for dir/name\n")
377    );
378
379    let (ran, log) = with_log("met", || Requirements::default().check("dir/name"));
380    assert!(ran);
381    assert_eq!(log, "");
382
383    // The table: `'*'` and the test's own key both apply; another test's key does not.
384    let table: &RequirementTable = &[
385      ("*", Requirements::packages(&[])),
386      (
387        "other",
388        Requirements::packages(&["no-such-package-req1.sty"]),
389      ),
390      (
391        "name",
392        Requirements::default().env("LATEXML_REQ1_UNSET_VAR"),
393      ),
394    ];
395    let (ran, log) = with_log("table", || check_table("dir", "name", Some(table)));
396    assert!(!ran);
397    assert_eq!(
398      log,
399      "dir/name\tdir/name is only checked in continuous integration. (set LATEXML_REQ1_UNSET_VAR=true)\n"
400    );
401    let (ran, _) = with_log("table-other", || check_table("dir", "third", Some(table)));
402    assert!(ran);
403  }
