1//! Process-parallel page rendering for pass B of the post pipeline
2//! (`docs/archive/STREAMING_POST_DESIGN_2026-07-06.md` §6).
3//!
4//! In-process page threads are blocked twice (`ObjectDB` is `!Send`;
5//! libxslt serializes every transform behind a process-wide lock), so the
6//! chosen shape is **process-level page-range workers**: the parent saves the
7//! completed ObjectDB as a SQLite file, partitions the spilled pages into
8//! contiguous chunks, and re-invokes its own binary once per chunk with
9//! `LATEXML_RENDER_WORKER=<manifest.json>`. Each child attaches the db
10//! readonly (WAL — N readers share one page cache via mmap), runs the SAME
11//! per-page pipeline as the serial driver
12//! (`crate::post::render_spilled_page`), and reports its diagnostic tally
13//! as trailing `Status:` lines on stderr. The parent folds child logs and
14//! counts deterministically, in chunk order, into its own `LOG_BUFFER` /
15//! `REPORT` — so the combined verdict and the persisted `--log` stay lossless
16//! (canvas signal-integrity rule: a child that dies without a status line is
17//! never taken for success).
18//!
19//! Env knob: `LATEXML_RENDER_JOBS` (usize; `1` = the serial path). Unset, the
20//! `latexml_oxide` binary renders a split of `MIN_PAGES_FOR_PARALLEL` pages
21//! or more on min(available cores, `AUTO_JOBS_MAX`) workers; every other
22//! embedder (the library, `cortex_worker`) stays serial. The pages are
23//! byte-identical either way (the parent names every page's images before the render and
24//! the workers look the names up, issue 995); a worker that fails to spawn, cannot set up, or
25//! dies without its status report sends the whole document back to the serial
26//! render at once, stopping the rest of the fleet (an Info; a Warning for a
27//! worker killed by a signal, a crash).

42/// Fewer spilled pages than this and the db save + child spawn overhead beats
43/// the parallel win — the serial path is taken regardless of the jobs knob.
44pub(crate) const MIN_PAGES_FOR_PARALLEL: usize = 64;
45
46/// The automatic fleet's ceiling: on Nasser's 2,400-problem book (9,713 pages)
47/// 8 workers took the two-invocation run from 115 s to 79 s.
48pub(crate) const AUTO_JOBS_MAX: usize = 8;
49
50/// Set by a binary that serves the worker mode (`LATEXML_RENDER_WORKER`, see
51/// [`worker_main`]): only it may spawn itself as a worker, so only it renders
52/// in parallel without being asked.
53static AUTO_JOBS: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
54
55/// Let an unset `LATEXML_RENDER_JOBS` pick the fleet size (`render_jobs`).
56pub fn enable_auto_jobs() { AUTO_JOBS.store(true, std::sync::atomic::Ordering::Relaxed); }
57
58/// How many page-render worker processes pass B may spawn: an explicit
59/// `LATEXML_RENDER_JOBS` wins (an unparsable or zero value is 1 = serial);
60/// unset, min(available cores, `AUTO_JOBS_MAX`) where [`enable_auto_jobs`]
61/// was called, else 1. Read once per conversion, so no hot-path caching is
62/// needed.
63pub(crate) fn render_jobs() -> usize {
64  match std::env::var("LATEXML_RENDER_JOBS") {
65    Ok(v) => v.parse::<usize>().ok().filter(|&n| n >= 1).unwrap_or(1),
66    Err(_) if AUTO_JOBS.load(std::sync::atomic::Ordering::Relaxed) => {
67      std::thread::available_parallelism()
68        .map_or(1, std::num::NonZeroUsize::get)
69        .min(AUTO_JOBS_MAX)
70    },
71    Err(_) => 1,
72  }
73}

194/// The parent-side result of a parallel render in which workers were actually
195/// spawned (successfully or not — failures are already folded as diagnostics).
196pub(crate) struct ParallelResult {
197  /// The first page's finalized output, read back from its destination file —
198  /// the same content the serial driver would have returned as `main_output`.
199  pub(crate) main_output:    Option<String>,
200  /// Total pages written, summed from the workers' `Status:pages:` lines.
201  pub(crate) pages_rendered: usize,

203  /// the render leaves no output (#782).
204  pub(crate) discarded:      bool,

206  /// abandoned fleet, every page it may have written ([`pages_the_fleet_may_have_written`]).
207  pub(crate) written:        Vec<String>,
208  /// A worker failed, so nothing was folded and the document renders serially. A page's Fatal
209  /// there removes `written` too (#782); a fuse or timeout stop keeps only the pages it rewrote.
210  pub(crate) abandoned:      bool,
211}

213/// Remove the per-run handoff artifacts (db + WAL sidecars + manifests).
214/// Best-effort: they live in the page-spill tempdir, which is removed wholesale
215/// when the parent drops it, so a failure here only delays the cleanup.
216fn cleanup_handoff(dbfile: &Path, manifests: &[PathBuf]) {
217  let _ = std::fs::remove_file(dbfile);
218  for suffix in ["-wal", "-shm"] {
219    let mut side = dbfile.as_os_str().to_owned();
220    side.push(suffix);
221    let _ = std::fs::remove_file(PathBuf::from(side));
222  }
223  for m in manifests {
224    let _ = std::fs::remove_file(m);
225  }
226}

228/// Strip ANSI `ESC[...m` color sequences (same logic as the logger's private
229/// helper). Child stderr is a pipe, so the TTY-gated logger should emit none —
230/// this is belt-and-suspenders for the log fold (signal-integrity rule).
231fn strip_ansi(s: &str) -> String {
232  let mut result = String::with_capacity(s.len());
233  let mut in_escape = false;
234  for c in s.chars() {
235    if in_escape {
236      if c == 'm' {
237        in_escape = false;
238      }
239    } else if c == '\u{1b}' {
240      in_escape = true;
241    } else {
242      result.push(c);
243    }
244  }
245  result
246}

248/// A worker's stderr, parsed: the trailing `Status:` lines are consumed into
249/// structured fields and everything else is the log text to forward + fold.
250struct ChildReport {
251  log:          String,
252  counts:       Option<ReportCounts>,
253  status:       Option<usize>,
254  pages:        usize,
255  /// `Status:discard:1`: a MathML/XSLT step died, so no page of the run is kept (#782).
256  discard:      bool,
257  /// `Status:written:<path>`: the pages this worker wrote, the only ones a discard removes.
258  written:      Vec<String>,
259  /// `Status:setup-failed:<why>`: the worker rendered nothing (`print_setup_failure`).
260  setup_failed: Option<String>,
261}
262
263fn parse_child_report(stderr_text: &str) -> ChildReport {
264  let mut log = String::new();
265  let mut counts = None;
266  let mut status = None;
267  let mut pages = 0usize;
268  let mut discard = false;
269  let mut written = Vec::new();
270  let mut setup_failed = None;
271  for line in stderr_text.lines() {
272    if let Some(rest) = line.strip_prefix("Status:counts:") {
273      let mut it = rest.split(',').map(|v| v.trim().parse::<usize>().ok());
274      let (d, i, w, e, f) = (
275        it.next().flatten(),
276        it.next().flatten(),
277        it.next().flatten(),
278        it.next().flatten(),
279        it.next().flatten(),
280      );
281      if let (Some(debug), Some(info), Some(warning), Some(error), Some(fatal)) = (d, i, w, e, f) {
282        counts = Some(ReportCounts {
283          debug,
284          info,
285          warning,
286          error,
287          fatal: fatal > 0,
288        });
289      }
290    } else if let Some(rest) = line.strip_prefix("Status:conversion:") {
291      status = rest.trim().parse::<usize>().ok();
292    } else if let Some(rest) = line.strip_prefix("Status:pages:") {
293      pages = rest.trim().parse::<usize>().unwrap_or(0);
294    } else if let Some(rest) = line.strip_prefix("Status:discard:") {
295      discard = rest.trim() == "1";
296    } else if let Some(path) = line.strip_prefix("Status:written:") {
297      written.push(path.to_string());
298    } else if let Some(why) = line.strip_prefix("Status:setup-failed:") {
299      setup_failed = Some(why.to_string());
300    } else {
301      log.push_str(line);
302      log.push('\n');
303    }
304  }
305  ChildReport {
306    log,
307    counts,
308    status,
309    pages,
310    discard,
311    written,
312    setup_failed,
313  }
314}

316/// One spawned (or spawn-failed) worker awaiting its fold.
317enum Pending {
318  /// The drain thread owns the child and returns its full `Output`; the pid
319  /// is kept so a parent-side timeout can kill the fleet. Only read on Unix
320  /// (`libc::kill` in the breach path below); the equivalent Windows fleet-kill
321  /// (OpenProcess + TerminateProcess) is not wired yet, so the pid is dead on
322  /// non-unix — scope the `dead_code` allow there rather than workspace-wide.
323  Spawned(
324    std::thread::JoinHandle<std::io::Result<std::process::Output>>,
325    #[cfg_attr(not(unix), allow(dead_code))] u32,
326  ),
327  Failed(String),
328}
329
330/// A joined worker: its parsed report and exit status, or why it could not be waited for.
331type Joined = Result<(ChildReport, std::process::ExitStatus), String>;
332
333fn join_worker(p: Pending) -> Joined {
334  match p {
335    Pending::Failed(e) => Err(e),
336    Pending::Spawned(handle, _) => match handle.join() {
337      Err(_) => Err("drain thread panicked".to_string()),
338      Ok(Err(e)) => Err(format!("wait failed: {e}")),
339      Ok(Ok(output)) => {
340        let stderr_text = strip_ansi(&String::from_utf8_lossy(&output.stderr));
341        Ok((parse_child_report(&stderr_text), output.status))
342      },
343    },
344  }
345}

347/// Why a worker sends the document to the serial render.
348struct WorkerFailure {
349  why:       String,
350  /// Ended by a signal (no exit code): a crash, not one of the worker's own exits.
351  by_signal: bool,
352}
353
354/// The failure of a joined worker that did not render its range, `None` for one that reported. A death names its
355/// signal and ends with the worker's last stderr lines (its `Fatal:oom:rss`, a panic), which are not folded.
356fn worker_failure(r: &Joined) -> Option<WorkerFailure> {
357  let (why, by_signal) = match r {
358    Err(e) => (e.clone(), false),
359    Ok((report, _)) if let Some(why) = &report.setup_failed => {
360      (format!("setup failed: {why}"), false)
361    },
362    Ok((report, status)) if report.status.is_none() => {
363      let by_signal = status.code().is_none();
364      let why = format!(
365        "{} without a Status:conversion line",
366        describe_death(status)
367      );
368      match stderr_tail(&report.log) {
369        Some(tail) => (format!("{why}; its stderr ended: {tail}"), by_signal),
370        None => (why, by_signal),
371      }
372    },
373    Ok(_) => return None,
374  };
375  Some(WorkerFailure { why, by_signal })
376}

378/// The last three non-blank lines of a failed worker's stderr, on one line (each cut at 300 bytes), so a line the
379/// worker began with `Fatal:` stays inside the parent's message.
380fn stderr_tail(log: &str) -> Option<String> {
381  let mut lines: Vec<&str> = log
382    .lines()
383    .map(str::trim)
384    .filter(|l| !l.is_empty())
385    .rev()
386    .take(3)
387    .map(|l| &l[..l.floor_char_boundary(300)])
388    .collect();
389  lines.reverse();
390  (!lines.is_empty()).then(|| lines.join(" | "))
391}

393/// How a worker ended: `exited (code 137)`, or `killed by SIGSEGV (signal 11)`.
394fn describe_death(status: &std::process::ExitStatus) -> String {
395  match status.code() {
396    Some(code) => format!("exited (code {code})"),
397    None => format!("killed by {}", describe_signal(status)),
398  }
399}

401/// The signal by name and number (`SIGSEGV (signal 11)`), and whether it dumped core.
402#[cfg(unix)]
403fn describe_signal(status: &std::process::ExitStatus) -> String {
404  use std::os::unix::process::ExitStatusExt;
405  let Some(n) = status.signal() else {
406    return "a signal".to_string();
407  };
408  let name = match n {
409    libc::SIGSEGV => "SIGSEGV",
410    libc::SIGBUS => "SIGBUS",
411    libc::SIGABRT => "SIGABRT",
412    libc::SIGILL => "SIGILL",
413    libc::SIGFPE => "SIGFPE",
414    libc::SIGKILL => "SIGKILL",
415    libc::SIGTERM => "SIGTERM",
416    libc::SIGINT => "SIGINT",
417    libc::SIGHUP => "SIGHUP",
418    libc::SIGQUIT => "SIGQUIT",
419    libc::SIGPIPE => "SIGPIPE",
420    libc::SIGTRAP => "SIGTRAP",
421    libc::SIGSYS => "SIGSYS",
422    libc::SIGXCPU => "SIGXCPU",
423    libc::SIGXFSZ => "SIGXFSZ",
424    _ => return format!("signal {n}"),
425  };
426  let core = if status.core_dumped() {
427    ", core dumped"
428  } else {
429    ""
430  };
431  format!("{name} (signal {n}{core})")
432}

437/// A worker's expected peak resident set: it decodes the whole db at attach, and its pages add
438/// little. Measured VmHWM, Nasser's book: 95 MB on the 600-problem part (11 MB db), 260 MB on the
439/// 2,400-problem part (45 MB db), i.e. ~42 MB + 4.85 x db; this adds a quarter's margin. The old
440/// `max(3 x db, 256 MB)` sized k2400 workers below their start, and one died (exit 137).
441fn worker_estimate_bytes(db_bytes: u64) -> u64 {
442  db_bytes.saturating_mul(6).saturating_add(64 << 20)
443}

445/// Pure fleet-sizing calculation for [`parallel_render`].
446///
447/// `soft_cap` is the cooperative RSS fuse (`stomach::resolve_rss_cap`, already
448/// 3/4 of `--max-memory` via `soft_cap_from_ceiling`) — it must NOT be scaled
449/// by 3/4 a second time. Returns `(jobs, affordable, budget, per_worker, max_rss_kb)`.
450fn compute_fleet_budget(
451  available_mem: Option<u64>,
452  soft_cap: Option<u64>,
453  parent_rss_bytes: u64,
454  db_bytes: u64,
455  requested_jobs: usize,
456  total_pages: usize,
457) -> (usize, usize, u64, u64, u64) {
458  let per_worker = worker_estimate_bytes(db_bytes);
459  let mut budget = available_mem.unwrap_or(u64::MAX);
460  if let Some(fuse) = soft_cap {
461    budget = budget.min(fuse.saturating_sub(parent_rss_bytes));
462  }
463  let affordable = (budget / per_worker) as usize;
464  let jobs = requested_jobs.min(total_pages).min(affordable);
465  let max_rss_kb = if budget != u64::MAX && jobs >= 2 {
466    budget / jobs as u64 / 1024
467  } else {
468    0
469  };
470  (jobs, affordable, budget, per_worker, max_rss_kb)
471}

473/// The pages an abandoned fleet may have written, for the serial render to remove if it ends in a
474/// Fatal (#782): a worker that reported (or could not set up) names its pages, and one that died
475/// before its `Status:written:` lines may have written any page of its range. The serial render
476/// rewrites them; one that stops (the fuse, the timeout, a page's Error) removes those it did not.
477fn pages_the_fleet_may_have_written(joined: &[Joined], ranges: &[Vec<String>]) -> Vec<String> {
478  joined
479    .iter()
480    .zip(ranges)
481    .flat_map(|(r, range)| match r {
482      Ok((report, _)) if report.status.is_some() || report.setup_failed.is_some() => {
483        report.written.clone()
484      },
485      Err(e) if e.starts_with("worker spawn failed") => Vec::new(),
486      _ => range.clone(),
487    })
488    .collect()
489}

491/// Spawn `jobs` page-range workers over `pages` and fold their results.
492///
493/// Returns `None` when setup failed or the fleet was declined, and an `abandoned` result when a
494/// worker failed to spawn, could not set up, or died without its status report (an Info, a Warning
495/// when a signal killed it) — the caller falls
496/// back to the serial render, which is still fully correct since the workers
497/// keep the spilled pages. Only a timeout breach folds the killed workers as an
498/// Error + FATAL status (fail toward flagging, never silent success).
499pub(crate) fn parallel_render(
500  mut manifest: RenderManifest,
501  pages: Vec<PageJob>,
502  jobs: usize,
503  spill_dir: &Path,
504  db: &ObjectDB,
505) -> Option<ParallelResult> {
506  let total_pages = pages.len();
507  let exe = match std::env::current_exe() {
508    Ok(e) => e,
509    Err(e) => {
510      Info!(
511        "post",
512        "parallel-render",
513        s!("parallel render disabled (current_exe: {})", e)
514      );
515      return None;
516    },
517  };
518  let dbfile = spill_dir.join("LaTeXML-render.db");
519  if let Err(e) = db.save_as(&dbfile) {
520    Info!(
521      "post",
522      "parallel-render",
523      s!("parallel render disabled (db save: {})", e)
524    );
525    return None;
526  }
527  manifest.dbfile = dbfile.clone();
528
529  // Clamp the fleet to ACTUAL headroom (witness OOM 2026-08-03: a joint run
530  // carries ~15 GB of core residual into post, and 8 workers each
531  // eager-loading a 1.82M-object db blew a 30 GB cgroup at spawn — the
532  // post-only measurement survived only because its parent was fresh). Free
533  // what the allocators will give back first, then estimate each worker at
534  // 64 MiB + 6x the on-disk db (`worker_estimate_bytes`), and size the
535  // fleet from the smaller of MemAvailable and the
536  // distance to this run's own memory fuse. Degrade to fewer workers — or
537  // decline to engage — rather than let the kernel choose a victim.
538  crate::heap_trim::release_free_heap();
539  let db_bytes = std::fs::metadata(&dbfile).map(|m| m.len()).unwrap_or(0);
540  let parent_rss_bytes = latexml_core::watchdog::process_rss_kb()
541    .unwrap_or(0)
542    .saturating_mul(1024);
543  let (jobs, affordable, budget, per_worker, max_rss_kb) = compute_fleet_budget(
544    latexml_core::watchdog::available_memory_bytes(),
545    latexml_core::stomach::resolve_rss_cap(),
546    parent_rss_bytes,
547    db_bytes,
548    jobs,
549    total_pages,
550  );
551  // Each worker's watchdog holds it to its share, so parent + fleet stay
552  // under the fuse; one that outgrows it dies and the document renders serially.
553  manifest.max_rss_kb = max_rss_kb;
554  if jobs < 2 {
555    Info!(
556      "post",
557      "parallel-render",
558      s!(
559        "parallel render declined: headroom {} MB affords {} worker(s) at ~{} MB each — staying serial",
560        budget / (1024 * 1024),
561        affordable,
562        per_worker / (1024 * 1024)
563      )
564    );
565    cleanup_handoff(&dbfile, &[]);
566    return None;
567  }
568
569  // Contiguous chunks preserve page order: worker 0 owns the first pages, so
570  // the fold below (and the first page's main-output read) is deterministic.
571  let chunk_size = total_pages.div_ceil(jobs);
572  let first_destination = pages.first().and_then(|p| p.destination.clone());
573  let mut manifest_paths: Vec<PathBuf> = Vec::with_capacity(jobs);
574  let mut ranges: Vec<Vec<String>> = Vec::with_capacity(jobs);
575  for (i, chunk) in pages.chunks(chunk_size).enumerate() {
576    ranges.push(chunk.iter().filter_map(|p| p.destination.clone()).collect());
577    let mut m = manifest.clone();
578    m.pages = chunk.to_vec();
579    let mpath = spill_dir.join(format!("render-manifest-{i}.json"));
580    let write_result = serde_json::to_string(&m)
581      .map_err(|e| e.to_string())
582      .and_then(|json| std::fs::write(&mpath, json).map_err(|e| e.to_string()));
583    if let Err(e) = write_result {
584      Info!(
585        "post",
586        "parallel-render",
587        s!("parallel render disabled (manifest write: {})", e)
588      );
589      cleanup_handoff(&dbfile, &manifest_paths);
590      return None;
591    }
592    manifest_paths.push(mpath);
593  }
594
595  // Parent-side deadline check before committing to the spawn (the workers
596  // carry no deadline of their own; the parent polls between waits below and
597  // kills the fleet on a breach).
598  if let Err(e) = latexml_core::stomach::check_timeout() {
599    e.log_fatal();
600    cleanup_handoff(&dbfile, &manifest_paths);
601    return Some(ParallelResult {
602      main_output:    None,
603      pages_rendered: 0,
604      discarded:      false,
605      written:        Vec::new(),
606      abandoned:      false,
607    });
608  }
609
610  // The engagement marker (also what the parity test asserts on — without it
611  // a silently-ignored jobs knob would let the test pass vacuously serial).
612  Info!(
613    "post",
614    "parallel-render",
615    s!(
616      "parallel page render engaged: {} worker(s) over {} pages",
617      manifest_paths.len(),
618      total_pages
619    )
620  );
621
622  let mut pending: Vec<Pending> = Vec::with_capacity(manifest_paths.len());
623  for mpath in &manifest_paths {
624    let spawned = std::process::Command::new(&exe)
625      .env("LATEXML_RENDER_WORKER", mpath)
626      .stdin(std::process::Stdio::null())
627      .stdout(std::process::Stdio::piped())
628      .stderr(std::process::Stdio::piped())
629      .spawn();
630    match spawned {
631      Ok(child) => {
632        let pid = child.id();
633        // Register with the watchdog so timeout/OOM exits never orphan the
634        // worker pool across Unix and Windows. Moved into the drain closure so
635        // each worker's PID unregisters immediately when `wait_with_output`
636        // reaps it (never outliving an early-finishing worker).
637        let kill_guard = latexml_core::watchdog::kill_pid_on_exit(pid, false);
638        // Each child gets a drain thread calling `wait_with_output` — piped
639        // stderr MUST be consumed while the parent polls, or a chatty child
640        // blocks on a full pipe and the poll below never sees it exit.
641        let handle = std::thread::Builder::new()
642          .name(format!("render-worker-drain-{pid}"))
643          .spawn(move || {
644            let _kill_guard = kill_guard;
645            child.wait_with_output()
646          });
647        match handle {
648          Ok(h) => pending.push(Pending::Spawned(h, pid)),
649          Err(e) => {
650            // `child` and `kill_guard` were moved into the closure and dropped
651            // on spawn failure, unregistering the PID; kill it explicitly so we
652            // never leak a live worker.
653            latexml_core::watchdog::kill_pid(pid, false);
654            pending.push(Pending::Failed(format!("drain thread spawn failed: {e}")));
655          },
656        }
657      },
658      Err(e) => pending.push(Pending::Failed(format!("worker spawn failed: {e}"))),
659    }
660  }
661
662  // Poll (rather than block) so the parent's cooperative timeout keeps
663  // running between waits, and join each worker as it ends: the first one
664  // that fails sends the document to the serial render at once, and a breach
665  // ends the run, so either way the rest of the fleet is killed rather than
666  // waited for (Nasser k2400: a worker's death was acted on only after its
667  // seven siblings finished).
668  let mut slots: Vec<Option<Pending>> = pending.into_iter().map(Some).collect();
669  let mut joined: Vec<Option<Joined>> = slots.iter().map(|_| None).collect();
670  let failed: Option<(usize, WorkerFailure)> = loop {
671    for (slot, out) in slots.iter_mut().zip(joined.iter_mut()) {
672      if slot
673        .as_ref()
674        .is_some_and(|p| !matches!(p, Pending::Spawned(h, _) if !h.is_finished()))
675      {
676        *out = slot.take().map(join_worker);
677      }
678    }
679    let failure = joined
680      .iter()
681      .enumerate()
682      .find_map(|(i, r)| r.as_ref().and_then(worker_failure).map(|f| (i, f)));
683    if failure.is_some() || slots.iter().all(Option::is_none) {
684      break failure;
685    }
686    if let Err(e) = latexml_core::stomach::check_timeout() {
687      e.log_fatal();
688      break None;
689    }
690    std::thread::sleep(std::time::Duration::from_millis(100));
691  };
692  for slot in slots.iter().flatten() {
693    if let Pending::Spawned(h, pid) = slot
694      && !h.is_finished()
695    {
696      latexml_core::watchdog::kill_pid(*pid, false);
697    }
698  }
699  for (slot, out) in slots.into_iter().zip(joined.iter_mut()) {
700    if let Some(p) = slot {
701      *out = Some(join_worker(p));
702    }
703  }
704
705  // Every worker is joined before any child log is folded: one that could not
706  // run, or died without its status report (a crash, its memory share
707  // exceeded, a broken pipe), sends the whole range to the serial render, so
708  // nothing is counted twice. A worker's own diagnostics, Fatal included, are
709  // the document's and fold as they are. After a timeout breach the killed
710  // workers fold as failures (the Fatal is already logged).
711  let mut joined: Vec<Joined> = joined
712    .into_iter()
713    .map(|r| r.unwrap_or_else(|| Err("not joined".to_string())))
714    .collect();
715  if let Some((i, failure)) = failed {
716    let message = s!(
717      "parallel render abandoned (worker {}: {}) — rendering serially",
718      i,
719      failure.why
720    );
721    // A worker killed by a signal crashed (or the kernel's OOM killer took
722    // it): the serial render recovers the pages, but the crash is counted. Its
723    // own watchdog's memory exit and a setup failure are expected outcomes.
724    if failure.by_signal {
725      Warn!("post", "parallel-render", message);
726    } else {
727      Info!("post", "parallel-render", message);
728    }
729    cleanup_handoff(&dbfile, &manifest_paths);
730    return Some(ParallelResult {
731      main_output:    None,
732      pages_rendered: 0,
733      discarded:      false,
734      written:        pages_the_fleet_may_have_written(&joined, &ranges),
735      abandoned:      true,
736    });
737  }
738
739  // Fold IN CHUNK ORDER (deterministic log/tally, mirroring the graphics
740  // worker fold): forward each child's stderr, replay its log + counts into
741  // the parent LOG_BUFFER/REPORT, and flag anything that did not report.
742  let mut pages_rendered = 0usize;
743  let mut discard = false;
744  let mut written: Vec<String> = Vec::new();
745  for (i, r) in joined.drain(..).enumerate() {
746    match r {
747      Err(e) => {
748        emit_error("post", "render_worker", &format!("worker {i}: {e}"));
749        note_status(LogStatus::Fatal, None);
750      },
751      Ok((report, exit_status)) => {
752        if !report.log.is_empty() {
753          // The child's stderr was piped, so nothing reached the live
754          // stderr yet — forward it now, then fold the same text + counts
755          // into the captured log and the REPORT tally.
756          eprint!("{}", stderr_share(&report.log));
757        }
758        replay_captured(CapturedDiagnostics {
759          log:    report.log,
760          counts: report.counts.unwrap_or_default(),
761        });
762        pages_rendered += report.pages;
763        discard |= report.discard;
764        written.extend(report.written);
765        match report.status {
766          None => {
767            // No status line = the child died before its final report (or
768            // never got that far). Fail toward flagging: this chunk's pages
769            // cannot be assumed rendered.
770            emit_error(
771              "post",
772              "render_worker",
773              &format!(
774                "worker {i} {} without a Status:conversion line",
775                describe_death(&exit_status)
776              ),
777            );
778            note_status(LogStatus::Fatal, None);
779          },
780          Some(s) if s >= 3 && !report.counts.is_some_and(|c| c.fatal) => {
781            // The child declared fatal but its counts line didn't carry the
782            // flag (or was missing) — max-fold the declared status anyway.
783            note_status(LogStatus::Fatal, None);
784          },
785          Some(_) => {},
786        }
787      },
788    }
789  }
790
791  // The serial driver returns the first page's finalized output as
792  // `main_output`; here that page is already on disk (written by worker 0),
793  // so read it back. A missing/unreadable file leaves `None` and the caller's
794  // fallback applies — the diagnostics above already flagged the failure.
795  if discard {
796    // Only this run's pages: a stale page of an earlier run in the destination stays.
797    crate::post::remove_pages(&written);
798    cleanup_handoff(&dbfile, &manifest_paths);
799    return Some(ParallelResult {
800      main_output:    None,
801      pages_rendered: 0,
802      discarded:      true,
803      written:        Vec::new(),
804      abandoned:      false,
805    });
806  }
807  let main_output = first_destination.and_then(|d| std::fs::read_to_string(&d).ok());
808  cleanup_handoff(&dbfile, &manifest_paths);
809  Some(ParallelResult {
810    main_output,
811    pages_rendered,
812    discarded: false,
813    written,
814    abandoned: false,
815  })
816}

818/// Print the worker's canonical trailing status report (the LAST lines on
819/// stderr): `Status:pages:`, `Status:written:` per page, then `Status:counts:`, then
820/// `Status:conversion:` — the parent parses them positionally-independently by prefix. Returns
821/// the process exit code (0 below fatal, 1 at fatal).
822fn print_status_report(written: &[String], discard: bool) -> i32 {
823  let c = snapshot_report_counts();
824  let status = get_status_code();
825  eprintln!("Status:pages:{}", written.len());
826  for path in written {
827    eprintln!("Status:written:{path}");
828  }
829  if discard {
830    eprintln!("Status:discard:1");
831  }
832  eprintln!(
833    "Status:counts:{},{},{},{},{}",
834    c.debug,
835    c.info,
836    c.warning,
837    c.error,
838    usize::from(c.fatal)
839  );
840  eprintln!("Status:conversion:{status}");
841  if status < 3 { 0 } else { 1 }
842}

844/// The lines of a worker's log the parent's stderr shows: its records at the parent's verbosity (`--quiet` hides
845/// `Info` and `Warning`, only `--verbose` shows `Debug`), each with its continuation lines; `Error`/`Fatal` always,
846/// as the logger prints them (OXIDIZED_DESIGN_DIVERGENCES #1158).
847fn stderr_share(log: &str) -> String {
848  stderr_share_at(log, latexml_core::util::logger::stderr_admits)
849}
850
851/// [`stderr_share`] at the verbosity `admits`. A line that opens no record is either a progress note (an `Info`)
852/// or an unindented continuation of a multi-line message, which the log cannot tell apart: it is shown when either
853/// would be, so a shown record keeps every line (a note after it may show under `--quiet`).
854fn stderr_share_at(log: &str, admits: impl Fn(log::Level) -> bool) -> String {
855  use log::Level;
856  let mut shown = String::with_capacity(log.len());
857  let mut admitted = true;
858  let mut record_admitted = false;
859  for line in log.split_inclusive('\n') {
860    let level = [
861      ("Error:", Level::Error),
862      ("Fatal:", Level::Error),
863      ("Warning:", Level::Warn),
864      ("Info:", Level::Info),
865      ("Debug:", Level::Debug),
866      ("Trace:", Level::Trace),
867    ]
868    .into_iter()
869    .find_map(|(head, level)| line.starts_with(head).then_some(level));
870    match level {
871      Some(level) => {
872        admitted = level == Level::Error || admits(level);
873        record_admitted = admitted;
874      },
875      // An indented line (`\tat …`, a blank line) goes with the line before it.
876      None if line.starts_with(char::is_whitespace) => {},
877      None => admitted = record_admitted || admits(Level::Info),
878    }
879    if admitted {
880      shown.push_str(line);
881    }
882  }
883  shown
884}

886/// A worker that could not set up (its manifest or db unreadable) has rendered
887/// nothing and its failure is not the document's: no status report, so the
888/// parent renders serially, and this line names why (an Info there).
889fn print_setup_failure(why: &str) -> i32 {
890  eprintln!("Status:setup-failed:{}", why.replace('\n', " "));
891  1
892}

894/// Entry point of the hidden worker mode (`LATEXML_RENDER_WORKER=<manifest>`),
895/// dispatched from the binary's `main` before any CLI handling. Renders the
896/// manifest's page range and ends its stderr with the status report, or with
897/// the setup-failure line when it could not start, so the parent never
898/// mistakes a broken worker for a clean one.
899pub fn worker_main(manifest_path: &str) -> i32 {
900  latexml_core::util::logger::init(log::LevelFilter::Info).ok();
901  let manifest: RenderManifest = match std::fs::read_to_string(manifest_path)
902    .map_err(|e| e.to_string())
903    .and_then(|s| serde_json::from_str(&s).map_err(|e| e.to_string()))
904  {
905    Ok(m) => m,
906    Err(e) => {
907      return print_setup_failure(&format!(
908        "cannot read the render manifest {manifest_path}: {e}"
909      ));
910    },
911  };
912  // Mirror `api.rs::on_worker`: a 256 MiB-stack thread for deeply nested
913  // math, and an explicit engine reset before the thread exits (the
914  // `#[thread_local]` roots do not Drop on a bare thread exit).
915  #[cfg(feature = "test-utils")]
916  let first_worker = manifest_path.ends_with("render-manifest-0.json");
917  std::thread::Builder::new()
918    .stack_size(256 * 1024 * 1024)
919    .spawn(move || {
920      #[cfg(feature = "test-utils")]
921      match test_fault().as_deref() {
922        None | Some("attach" | "unreported" | "rss") => {},
923        Some("signal") if first_worker => {
924          latexml_core::watchdog::kill_pid(std::process::id(), false)
925        },
926        Some("signal") => std::thread::sleep(std::time::Duration::from_secs(600)),
927        Some(_) => latexml_core::watchdog::exit_now(70),
928      }
929      // Hold this worker to its share of the run's memory budget, and align
930      // `heap_trim::step_kb()` with that ceiling instead of the 4.5 GB default.
931      // The share is the worker's HARD (watchdog) ceiling, where the parent's cap
932      // is the soft 3/4 fuse: a `check_timeout` added to the page loop must not
933      // read it as the cooperative fuse.
934      if manifest.max_rss_kb > 0 {
935        latexml_core::stomach::set_memory_cap(Some(manifest.max_rss_kb.saturating_mul(1024)));
936      }
937      manifest.post_values.restore();
938      let level = manifest.log_level.parse().unwrap_or(log::LevelFilter::Info);
939      latexml_core::util::logger::init(level.max(log::LevelFilter::Info)).ok();
940      let watchdog_kb = manifest.max_rss_kb;
941      #[cfg(feature = "test-utils")]
942      let watchdog_kb = if test_fault().as_deref() == Some("rss") {
943        1
944      } else {
945        watchdog_kb
946      };
947      let _watchdog =
948        (watchdog_kb > 0).then(|| latexml_core::watchdog::Watchdog::with_limits(0, watchdog_kb));
949      let code = match render_manifest_pages(manifest) {
950        #[cfg(feature = "test-utils")]
951        Ok(_) if test_fault().as_deref() == Some("unreported") => {
952          latexml_core::watchdog::exit_now(70)
953        },
954        Ok((written, discard)) => print_status_report(&written, discard),
955        Err(why) => print_setup_failure(&why),
956      };
957      latexml_core::reset_thread_engine();
958      code
959    })
960    .expect("spawn render worker thread")
961    .join()
962    .expect("render worker thread panicked")
963}

965/// Test hook `LATEXML_RENDER_WORKER_TEST_ABORT` (guard: 120_midscale_streaming):
966/// `attach` fails the worker's db attach; `signal` SIGKILLs worker 0 (a crash)
967/// and holds its siblings for ten minutes, unless the parent stops them;
968/// `unreported` renders the range and exits before the status report, as a
969/// worker killed late would; `rss` holds the worker's watchdog to 1 KB, so it
970/// exits 137 at its first poll, while the render attaches; any other value
971/// exits the worker before it renders.
972#[cfg(feature = "test-utils")]
973fn test_fault() -> Option<String> { std::env::var("LATEXML_RENDER_WORKER_TEST_ABORT").ok() }

975/// Rebuild the pass-B processor set from the manifest — the EXACT construction
976/// the parent's `run_post_processing_inner` performs (same MathML
977/// primary/secondary parallel model, same XSLT wiring, same error paths) —
978/// then render every page in the manifest's range through the shared
979/// [`crate::post::render_spilled_page`]. Returns the pages written, and whether a Fatal discards them;
980/// `Err` = the worker could not set up (see `print_setup_failure`).
981fn render_manifest_pages(m: RenderManifest) -> Result<(Vec<String>, bool), String> {
982  use latexml_post::{
983    crossref::{CrossRef, UrlStyle},
984    processor::Processor,
985  };
986  #[cfg(feature = "test-utils")]
987  let m = RenderManifest {
988    dbfile: if test_fault().as_deref() == Some("attach") {
989      m.dbfile.with_extension("missing")
990    } else {
991      m.dbfile
992    },
993    ..m
994  };
995  let db = match ObjectDB::attach(&m.dbfile, DbAttachOptions {
996    readonly: true,
997    clean:    false,
998  }) {
999    Ok(db) => db,
1000    Err(e) => {
1001      return Err(format!(
1002        "cannot attach the render db {}: {e}",
1003        m.dbfile.display()
1004      ));
1005    },
1006  };
1007  let url_style = UrlStyle::from_cli(&m.url_style).unwrap_or(UrlStyle::File);
1008  let mut crossref = CrossRef::new(db, url_style, true);
1009  crossref.set_extension(&m.out_extension);
1010  if let Some(navtoc) = m.navigation_toc.as_deref() {
1011    crossref.set_navigation_toc(navtoc);
1012  }
1013  // The worker's own handle on the destination's cache (Perl's document cache), shared by its pages and Graphics.
1014  let mut page_opts: latexml_post::document::PostDocumentOptions = m.page_opts.clone().into();
1015  let cache = latexml_post::document::DocCache::new(page_opts.nocache);
1016  page_opts.cache = Some(cache.clone());
1017  let graphics = m.graphicimages.then(|| {
1018    latexml_post::graphics::Graphics::new(None, true)
1019      .with_svg_threshold_kb(m.graphics_svg_threshold_kb)
1020      .with_planned_names(m.resource_names.clone())
1021      .with_doc_cache(cache.clone())
1022  });
1023  let post = latexml_post::Post::new();
1024  let mut processors: Vec<Box<dyn Processor>> = Vec::new();
1025  // Same chain as the in-process renderer (`post.rs`): SVG on the live page
1026  // DOM before MathML/XSLT for the HTML formats (Perl `LaTeXML.pm` order).
1027  if m.svg {
1028    processors.push(Box::new(latexml_post::svg::SVG::new()));
1029  }
1030  let math_formats = crate::post::MathFormats {
1031    pmml:            m.pmml,
1032    cmml:            m.cmml,
1033    unicodemath:     m.unicodemath,
1034    keep_xmath:      m.keep_xmath,
1035    invisible_times: m.invisible_times,
1036    plane1:          m.plane1,
1037    hackplane1:      m.hackplane1,
1038    mathtex:         m.mathtex,
1039    intent_literal:  m.intent_literal,
1040  };
1041  if let Some(math) = math_formats.processor() {
1042    processors.push(math);
1043  }
1044  if let Some(xsl_path) = m.stylesheet.as_deref() {
1045    let params: rustc_hash::FxHashMap<String, String> = m.xslt_params.iter().cloned().collect();
1046    match latexml_post::xslt::XSLT::new(
1047      xsl_path,
1048      params,
1049      !m.default_resources,
1050      None,
1051      m.searchpaths.clone(),
1052    ) {
1053      Ok(xslt) => processors.push(Box::new(xslt)),
1054      Err(e) => emit_error("post", "xslt", &format!("XSLT error: {e}")),
1055    }
1056  }
1057  let ctx = crate::post::PageRenderCtx {
1058    page_opts,
1059    is_html_out: m.is_html_out,
1060    schemadocs: m.schemadocs,
1061    whatsout: latexml_post::extract::Whatsout::from_cli(&m.whatsout).unwrap_or_default(),
1062    remove_spilled: false,
1063  };
1064  let mut procs = crate::post::PageProcessors {
1065    crossref,
1066    graphics,
1067    post,
1068    processors,
1069  };
1070  let mut written: Vec<String> = Vec::new();
1071  let mut made_dirs = latexml_post::writer::MadeDirs::default();
1072  for job in &m.pages {
1073    // Same OS give-back cadence as the serial render loop: each page cycles a
1074    // DOM + XSLT result through the C heap, and a chunk can be tens of
1075    // thousands of pages.
1076    if !written.is_empty() && written.len().is_multiple_of(crate::heap_trim::SAMPLE_PAGES) {
1077      crate::heap_trim::sample();
1078    }
1079    match crate::post::render_spilled_page(
1080      &job.path,
1081      &mut procs,
1082      &ctx,
1083      job.destination.clone(),
1084      job.destination_directory.clone(),
1085    ) {
1086      Ok(outputs) => {
1087        for (dest, output) in outputs {
1088          if let Some(path) = dest.as_deref() {
1089            let _ = made_dirs.ensure_parent(path);
1090            written.push(path.to_string());
1091            if let Err(e) = std::fs::write(path, &output) {
1092              emit_error(
1093                "post",
1094                "write",
1095                &format!("failed to write page {path}: {e}"),
1096              );
1097            }
1098          }
1099        }
1100      },
1101      // Already reported inside the pipeline; mirror the serial driver's
1102      // abort-on-page-failure (the parent flags the shortfall through the
1103      // folded error + the fatal-bearing status this worker will report).
1104      Err(crate::post::PageFailure::Error) => break,
1105      // The parent removes every page of the run (#782).
1106      Err(crate::post::PageFailure::Fatal) => return Ok((written, true)),
1107    }
1108  }
1109  Ok((written, false))
1110}

1116  /// At the `Warn` level a Warning keeps its unindented continuation lines (a multi-line message); the notes after an
1117  /// Info record stay hidden. The default verbosity shows every line; `--quiet` (`Error`) only the Error and the
1118  /// Fatal, each with its continuation lines, and so does a console that admits nothing (OD #1158).
1119  #[test]
1120  fn stderr_share_keeps_a_records_continuation_lines() {
1121    let log = "note before\nInfo:a:x one\nMathML[Presentation] S1.html processing\nWarning:b:y first line\nsecond \
1122               line\n\tat loc\nInfo:c:z two\n\tIn here\npost-processing\nError:d:w bad\nits second line\n\tat \
1123               loc2\nWarning:e:v late\nFatal:f:u end\n\tat loc3\n";
1124    let errors = "Error:d:w bad\nits second line\n\tat loc2\nFatal:f:u end\n\tat loc3\n";
1125    assert_eq!(
1126      stderr_share_at(log, |level| level <= log::Level::Warn),
1127      "Warning:b:y first line\nsecond line\n\tat loc\nError:d:w bad\nits second line\n\tat loc2\nWarning:e:v \
1128       late\nFatal:f:u end\n\tat loc3\n"
1129    );
1130    assert_eq!(stderr_share_at(log, |level| level <= log::Level::Info), log);
1131    assert_eq!(
1132      stderr_share_at(log, |level| level <= log::Level::Error),
1133      errors
1134    );
1135    assert_eq!(stderr_share_at(log, |_| false), errors);
1136  }

1212  #[test]
1213  fn fleet_budget_uses_soft_cap_without_double_discount() {
1214    const MIB: u64 = 1024 * 1024;
1215    // `--max-memory=8192` → soft_cap = 6,144 MiB (3/4 of 8 GiB).
1216    // With 4,300 MiB parent RSS and 80 MiB db (per_worker = 64 + 6 x 80 = 544 MiB):
1217    // remaining fuse headroom is 1,844 MiB → 3 workers affordable.
1218    // (With the old double 3/4 bug, fuse dropped to 4,608 MiB → 308 MiB → 0 workers.)
1219    let soft_cap = latexml_core::stomach::soft_cap_from_ceiling(8192);
1220    assert_eq!(soft_cap, 6144 * MIB);
1221    let (jobs, affordable, budget, per_worker, max_rss_kb) = compute_fleet_budget(
1222      Some(16_000 * MIB),
1223      Some(soft_cap),
1224      4300 * MIB,
1225      80 * MIB,
1226      8,
1227      1000,
1228    );
1229    assert_eq!(budget, 1844 * MIB);
1230    assert_eq!(per_worker, 544 * MIB);
1231    assert_eq!(affordable, 3);
1232    assert_eq!(jobs, 3);
1233    assert_eq!(max_rss_kb, (1844 * MIB) / 3 / 1024);
1234  }

1255  #[test]
1256  fn only_a_signal_death_is_a_crash() {
1257    use std::os::unix::process::ExitStatusExt;
1258    let report = || parse_child_report("Info:post:x partial\n");
1259    let killed = worker_failure(&Ok((report(), std::process::ExitStatus::from_raw(9)))).unwrap();
1260    assert!(killed.by_signal);
1261    assert_eq!(
1262      killed.why,
1263      "killed by SIGKILL (signal 9) without a Status:conversion line; its stderr ended: Info:post:x partial"
1264    );
1265    // A crash names its signal and the core dump (wait status 11 | 0x80).
1266    let crashed = worker_failure(&Ok((
1267      parse_child_report(""),
1268      std::process::ExitStatus::from_raw(11 | 0x80),
1269    )))
1270    .unwrap();
1271    assert!(crashed.by_signal);
1272    assert_eq!(
1273      crashed.why,
1274      "killed by SIGSEGV (signal 11, core dumped) without a Status:conversion line"
1275    );
1276    // The worker's own memory exit is not a crash; its last three stderr lines end the message, on one line.
1277    let capped = worker_failure(&Ok((
1278      parse_child_report(
1279        "Info:post:a one\nInfo:post:b two\n\nWarning:post:c three\nFatal:oom:rss latexml-oxide: resident \
1280         memory 900MB exceeded the 800MB ceiling\nStatus:pages:0\n",
1281      ),
1282      std::process::ExitStatus::from_raw(137 << 8),
1283    )))
1284    .unwrap();
1285    assert!(
1286      !capped.by_signal,
1287      "the worker's own memory exit is not a crash"
1288    );
1289    assert_eq!(
1290      capped.why,
1291      "exited (code 137) without a Status:conversion line; its stderr ended: Info:post:b two | Warning:post:c \
1292       three | Fatal:oom:rss latexml-oxide: resident memory 900MB exceeded the 800MB ceiling"
1293    );
1294    let done = parse_child_report("Status:pages:1\nStatus:conversion:0\n");
1295    assert!(worker_failure(&Ok((done, std::process::ExitStatus::from_raw(0)))).is_none());
1296  }
