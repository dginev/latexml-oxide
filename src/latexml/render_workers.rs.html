1//! Process-parallel page rendering for pass B of the post pipeline
2//! (`docs/performance/STREAMING_POST_DESIGN_2026-07-06.md` §6).
3//!
4//! In-process page threads are blocked twice (`ObjectDB` is `!Send`;
5//! libxslt serializes every transform behind a process-wide lock), so the
6//! chosen shape is **process-level page-range workers**: the parent saves the
7//! completed ObjectDB as a SQLite file, partitions the spilled pages into
8//! contiguous chunks, and re-invokes its own binary once per chunk with
9//! `LATEXML_RENDER_WORKER=<manifest.json>`. Each child attaches the db
10//! readonly (WAL — N readers share one page cache via mmap), runs the SAME
11//! per-page pipeline as the serial driver
12//! (`crate::post::render_spilled_page`), and reports its diagnostic tally
13//! as trailing `Status:` lines on stderr. The parent folds child logs and
14//! counts deterministically, in chunk order, into its own `LOG_BUFFER` /
15//! `REPORT` — so the combined verdict and the persisted `--log` stay lossless
16//! (canvas signal-integrity rule: a child that dies without a status line is
17//! folded as FATAL, never as success).
18//!
19//! Env knob: `LATEXML_RENDER_JOBS` (usize). Default 1 = the serial path,
20//! byte-identical to before this module existed.
21

36/// Fewer spilled pages than this and the db save + child spawn overhead beats
37/// the parallel win — the serial path is taken regardless of the jobs knob.
38pub(crate) const MIN_PAGES_FOR_PARALLEL: usize = 64;

40/// The `LATEXML_RENDER_JOBS` knob: how many page-render worker processes pass
41/// B may spawn. Default (and any unparsable value) is 1 = serial. Read once
42/// per conversion, so no hot-path caching is needed.
43pub(crate) fn render_jobs() -> usize {
44  std::env::var("LATEXML_RENDER_JOBS")
45    .ok()
46    .and_then(|v| v.parse::<usize>().ok())
47    .filter(|&n| n >= 1)
48    .unwrap_or(1)
49}

159/// Remove the per-run handoff artifacts (db + WAL sidecars + manifests).
160/// Best-effort: they live in the page-spill tempdir, which is removed wholesale
161/// when the parent drops it, so a failure here only delays the cleanup.
162fn cleanup_handoff(dbfile: &Path, manifests: &[PathBuf]) {
163  let _ = std::fs::remove_file(dbfile);
164  for suffix in ["-wal", "-shm"] {
165    let mut side = dbfile.as_os_str().to_owned();
166    side.push(suffix);
167    let _ = std::fs::remove_file(PathBuf::from(side));
168  }
169  for m in manifests {
170    let _ = std::fs::remove_file(m);
171  }
172}

174/// Strip ANSI `ESC[...m` color sequences (same logic as the logger's private
175/// helper). Child stderr is a pipe, so the TTY-gated logger should emit none —
176/// this is belt-and-suspenders for the log fold (signal-integrity rule).
177fn strip_ansi(s: &str) -> String {
178  let mut result = String::with_capacity(s.len());
179  let mut in_escape = false;
180  for c in s.chars() {
181    if in_escape {
182      if c == 'm' {
183        in_escape = false;
184      }
185    } else if c == '\u{1b}' {
186      in_escape = true;
187    } else {
188      result.push(c);
189    }
190  }
191  result
192}

194/// A worker's stderr, parsed: the trailing `Status:` lines are consumed into
195/// structured fields and everything else is the log text to forward + fold.
196struct ChildReport {
197  log:    String,
198  counts: Option<ReportCounts>,
199  status: Option<usize>,
200  pages:  usize,
201}
202
203fn parse_child_report(stderr_text: &str) -> ChildReport {
204  let mut log = String::new();
205  let mut counts = None;
206  let mut status = None;
207  let mut pages = 0usize;
208  for line in stderr_text.lines() {
209    if let Some(rest) = line.strip_prefix("Status:counts:") {
210      let mut it = rest.split(',').map(|v| v.trim().parse::<usize>().ok());
211      let (d, i, w, e, f) = (
212        it.next().flatten(),
213        it.next().flatten(),
214        it.next().flatten(),
215        it.next().flatten(),
216        it.next().flatten(),
217      );
218      if let (Some(debug), Some(info), Some(warning), Some(error), Some(fatal)) = (d, i, w, e, f) {
219        counts = Some(ReportCounts {
220          debug,
221          info,
222          warning,
223          error,
224          fatal: fatal > 0,
225        });
226      }
227    } else if let Some(rest) = line.strip_prefix("Status:conversion:") {
228      status = rest.trim().parse::<usize>().ok();
229    } else if let Some(rest) = line.strip_prefix("Status:pages:") {
230      pages = rest.trim().parse::<usize>().unwrap_or(0);
231    } else {
232      log.push_str(line);
233      log.push('\n');
234    }
235  }
236  ChildReport { log, counts, status, pages }
237}

239/// One spawned (or spawn-failed) worker awaiting its fold.
240enum Pending {
241  /// The drain thread owns the child and returns its full `Output`; the pid
242  /// is kept so a parent-side timeout can kill the fleet.
243  Spawned(
244    std::thread::JoinHandle<std::io::Result<std::process::Output>>,
245    u32,
246  ),
247  Failed(String),
248}

250/// Spawn `jobs` page-range workers over `pages` and fold their results.
251///
252/// Returns `None` when setup failed BEFORE any child was spawned (reported at
253/// Info severity) — the caller falls back to the serial render, which is still
254/// fully correct since the spilled pages are untouched. Once children run
255/// there is no fallback: a worker that fails, dies, or omits its status line
256/// is folded as an Error + FATAL status (fail toward flagging, never silent
257/// success).
258pub(crate) fn parallel_render(
259  mut manifest: RenderManifest,
260  pages: Vec<PageJob>,
261  jobs: usize,
262  spill_dir: &Path,
263  db: &ObjectDB,
264) -> Option<ParallelResult> {
265  let total_pages = pages.len();
266  let exe = match std::env::current_exe() {
267    Ok(e) => e,
268    Err(e) => {
269      Info!(
270        "post",
271        "parallel-render",
272        s!("parallel render disabled (current_exe: {})", e)
273      );
274      return None;
275    },
276  };
277  let dbfile = spill_dir.join("render.db");
278  if let Err(e) = db.save_as(&dbfile) {
279    Info!(
280      "post",
281      "parallel-render",
282      s!("parallel render disabled (db save: {})", e)
283    );
284    return None;
285  }
286  manifest.dbfile = dbfile.clone();
287
288  // Clamp the fleet to ACTUAL headroom (witness OOM 2026-08-03: a joint run
289  // carries ~15 GB of core residual into post, and 8 workers each
290  // eager-loading a 1.82M-object db blew a 30 GB cgroup at spawn — the
291  // post-only measurement survived only because its parent was fresh). Free
292  // what the allocators will give back first, then estimate each worker at
293  // ~3x the on-disk db (eager JSON decode + libxml holder + one page) plus a
294  // fixed floor, and size the fleet from the smaller of MemAvailable and the
295  // distance to this run's own memory fuse. Degrade to fewer workers — or
296  // decline to engage — rather than let the kernel choose a victim.
297  #[cfg(target_os = "linux")]
298  unsafe {
299    libc::malloc_trim(0);
300  }
301  #[cfg(not(feature = "dhat-heap"))]
302  unsafe {
303    libmimalloc_sys::mi_collect(true);
304  }
305  let db_bytes = std::fs::metadata(&dbfile).map(|m| m.len()).unwrap_or(0);
306  let per_worker = db_bytes.saturating_mul(3).max(256 * 1024 * 1024);
307  let mut budget = latexml_core::watchdog::available_memory_bytes().unwrap_or(u64::MAX);
308  if let Some(cap) = latexml_core::stomach::resolve_rss_cap() {
309    // Budget against the cooperative FUSE (75% of the hard cap), where the
310    // graceful stop fires — not the cap itself, where the watchdog kills.
311    let fuse = cap / 4 * 3;
312    let rss = latexml_core::watchdog::process_rss_kb().unwrap_or(0) * 1024;
313    budget = budget.min(fuse.saturating_sub(rss));
314  }
315  let affordable = (budget / per_worker) as usize;
316  let jobs = jobs.min(total_pages).min(affordable);
317  if jobs < 2 {
318    Info!(
319      "post",
320      "parallel-render",
321      s!(
322        "parallel render declined: headroom {} MB affords {} worker(s) at ~{} MB each — staying serial",
323        budget / (1024 * 1024),
324        affordable,
325        per_worker / (1024 * 1024)
326      )
327    );
328    cleanup_handoff(&dbfile, &[]);
329    return None;
330  }
331
332  // Contiguous chunks preserve page order: worker 0 owns the first pages, so
333  // the fold below (and the first page's main-output read) is deterministic.
334  let chunk_size = total_pages.div_ceil(jobs);
335  let first_destination = pages.first().and_then(|p| p.destination.clone());
336  let mut manifest_paths: Vec<PathBuf> = Vec::with_capacity(jobs);
337  for (i, chunk) in pages.chunks(chunk_size).enumerate() {
338    let mut m = manifest.clone();
339    m.pages = chunk.to_vec();
340    let mpath = spill_dir.join(format!("render-manifest-{i}.json"));
341    let write_result = serde_json::to_string(&m)
342      .map_err(|e| e.to_string())
343      .and_then(|json| std::fs::write(&mpath, json).map_err(|e| e.to_string()));
344    if let Err(e) = write_result {
345      Info!(
346        "post",
347        "parallel-render",
348        s!("parallel render disabled (manifest write: {})", e)
349      );
350      cleanup_handoff(&dbfile, &manifest_paths);
351      return None;
352    }
353    manifest_paths.push(mpath);
354  }
355
356  // Parent-side deadline check before committing to the spawn (the workers
357  // carry no deadline of their own; the parent polls between waits below and
358  // kills the fleet on a breach).
359  if let Err(e) = latexml_core::stomach::check_timeout() {
360    e.log_fatal();
361    cleanup_handoff(&dbfile, &manifest_paths);
362    return Some(ParallelResult {
363      main_output:    None,
364      pages_rendered: 0,
365    });
366  }
367
368  // The engagement marker (also what the parity test asserts on — without it
369  // a silently-ignored jobs knob would let the test pass vacuously serial).
370  Info!(
371    "post",
372    "parallel-render",
373    s!(
374      "parallel page render engaged: {} worker(s) over {} pages",
375      manifest_paths.len(),
376      total_pages
377    )
378  );
379
380  let mut pending: Vec<Pending> = Vec::with_capacity(manifest_paths.len());
381  for mpath in &manifest_paths {
382    let spawned = std::process::Command::new(&exe)
383      .env("LATEXML_RENDER_WORKER", mpath)
384      .stdin(std::process::Stdio::null())
385      .stdout(std::process::Stdio::piped())
386      .stderr(std::process::Stdio::piped())
387      .spawn();
388    match spawned {
389      Ok(child) => {
390        let pid = child.id();
391        // Each child gets a drain thread calling `wait_with_output` — piped
392        // stderr MUST be consumed while the parent polls, or a chatty child
393        // blocks on a full pipe and the poll below never sees it exit.
394        let handle = std::thread::Builder::new()
395          .name(format!("render-worker-drain-{pid}"))
396          .spawn(move || child.wait_with_output());
397        match handle {
398          Ok(h) => pending.push(Pending::Spawned(h, pid)),
399          Err(e) => pending.push(Pending::Failed(format!("drain thread spawn failed: {e}"))),
400        }
401      },
402      Err(e) => pending.push(Pending::Failed(format!("worker spawn failed: {e}"))),
403    }
404  }
405
406  // Poll (rather than block) so the parent's cooperative timeout keeps
407  // running between waits; on a breach, kill the fleet so the join below
408  // returns promptly instead of riding out the children.
409  loop {
410    let all_done = pending
411      .iter()
412      .all(|p| !matches!(p, Pending::Spawned(h, _) if !h.is_finished()));
413    if all_done {
414      break;
415    }
416    if let Err(e) = latexml_core::stomach::check_timeout() {
417      e.log_fatal();
418      #[cfg(unix)]
419      for p in &pending {
420        if let Pending::Spawned(h, pid) = p
421          && !h.is_finished()
422        {
423          // SAFETY: plain kill(2) on a child pid this process spawned.
424          unsafe {
425            libc::kill(*pid as i32, libc::SIGKILL);
426          }
427        }
428      }
429      break;
430    }
431    std::thread::sleep(std::time::Duration::from_millis(100));
432  }
433
434  // Fold IN CHUNK ORDER (deterministic log/tally, mirroring the graphics
435  // worker fold): forward each child's stderr, replay its log + counts into
436  // the parent LOG_BUFFER/REPORT, and flag anything that did not report.
437  let mut pages_rendered = 0usize;
438  for (i, p) in pending.into_iter().enumerate() {
439    match p {
440      Pending::Failed(e) => {
441        emit_error("post", "render_worker", &format!("worker {i}: {e}"));
442        note_status(LogStatus::Fatal, None);
443      },
444      Pending::Spawned(handle, _) => match handle.join() {
445        Err(_) => {
446          emit_error(
447            "post",
448            "render_worker",
449            &format!("worker {i}: drain thread panicked"),
450          );
451          note_status(LogStatus::Fatal, None);
452        },
453        Ok(Err(e)) => {
454          emit_error(
455            "post",
456            "render_worker",
457            &format!("worker {i}: wait failed: {e}"),
458          );
459          note_status(LogStatus::Fatal, None);
460        },
461        Ok(Ok(output)) => {
462          let stderr_text = strip_ansi(&String::from_utf8_lossy(&output.stderr));
463          let report = parse_child_report(&stderr_text);
464          if !report.log.is_empty() {
465            // The child's stderr was piped, so nothing reached the live
466            // stderr yet — forward it now, then fold the same text + counts
467            // into the captured log and the REPORT tally.
468            eprint!("{}", report.log);
469          }
470          replay_captured(CapturedDiagnostics {
471            log:    report.log,
472            counts: report.counts.unwrap_or_default(),
473          });
474          pages_rendered += report.pages;
475          match report.status {
476            None => {
477              // No status line = the child died before its final report (or
478              // never got that far). Fail toward flagging: this chunk's pages
479              // cannot be assumed rendered.
480              emit_error(
481                "post",
482                "render_worker",
483                &format!(
484                  "worker {i} exited (code {:?}) without a Status:conversion line",
485                  output.status.code()
486                ),
487              );
488              note_status(LogStatus::Fatal, None);
489            },
490            Some(s) if s >= 3 && !report.counts.is_some_and(|c| c.fatal) => {
491              // The child declared fatal but its counts line didn't carry the
492              // flag (or was missing) — max-fold the declared status anyway.
493              note_status(LogStatus::Fatal, None);
494            },
495            Some(_) => {},
496          }
497        },
498      },
499    }
500  }
501
502  // The serial driver returns the first page's finalized output as
503  // `main_output`; here that page is already on disk (written by worker 0),
504  // so read it back. A missing/unreadable file leaves `None` and the caller's
505  // fallback applies — the diagnostics above already flagged the failure.
506  let main_output = first_destination.and_then(|d| std::fs::read_to_string(&d).ok());
507  cleanup_handoff(&dbfile, &manifest_paths);
508  Some(ParallelResult { main_output, pages_rendered })
509}

511/// Print the worker's canonical trailing status report (the LAST lines on
512/// stderr): `Status:pages:`, then `Status:counts:`, then `Status:conversion:`
513/// — the parent parses them positionally-independently by prefix. Returns the
514/// process exit code (0 below fatal, 1 at fatal).
515fn print_status_report(pages: usize) -> i32 {
516  let c = snapshot_report_counts();
517  let status = get_status_code();
518  eprintln!("Status:pages:{pages}");
519  eprintln!(
520    "Status:counts:{},{},{},{},{}",
521    c.debug,
522    c.info,
523    c.warning,
524    c.error,
525    usize::from(c.fatal)
526  );
527  eprintln!("Status:conversion:{status}");
528  if status < 3 { 0 } else { 1 }
529}

531/// Entry point of the hidden worker mode (`LATEXML_RENDER_WORKER=<manifest>`),
532/// dispatched from the binary's `main` before any CLI handling. Renders the
533/// manifest's page range and ALWAYS ends its stderr with the status report —
534/// even when the manifest cannot be read — so the parent never mistakes a
535/// broken worker for a clean one.
536pub fn worker_main(manifest_path: &str) -> i32 {
537  latexml_core::util::logger::init(log::LevelFilter::Info).ok();
538  let manifest: RenderManifest = match std::fs::read_to_string(manifest_path)
539    .map_err(|e| e.to_string())
540    .and_then(|s| serde_json::from_str(&s).map_err(|e| e.to_string()))
541  {
542    Ok(m) => m,
543    Err(e) => {
544      emit_fatal(
545        "post",
546        "render_worker",
547        &format!("cannot read the render manifest {manifest_path}: {e}"),
548      );
549      return print_status_report(0);
550    },
551  };
552  // Mirror `api.rs::on_worker`: a 256 MiB-stack thread for deeply nested
553  // math, and an explicit engine reset before the thread exits (the
554  // `#[thread_local]` roots do not Drop on a bare thread exit).
555  std::thread::Builder::new()
556    .stack_size(256 * 1024 * 1024)
557    .spawn(move || {
558      let pages = render_manifest_pages(manifest);
559      let code = print_status_report(pages);
560      latexml_core::reset_thread_engine();
561      code
562    })
563    .expect("spawn render worker thread")
564    .join()
565    .expect("render worker thread panicked")
566}

568/// Rebuild the pass-B processor set from the manifest — the EXACT construction
569/// the parent's `run_post_processing_inner` performs (same MathML
570/// primary/secondary parallel model, same XSLT wiring, same error paths) —
571/// then render every page in the manifest's range through the shared
572/// [`crate::post::render_spilled_page`]. Returns the number of pages written.
573fn render_manifest_pages(m: RenderManifest) -> usize {
574  use latexml_post::{
575    crossref::{CrossRef, UrlStyle},
576    processor::Processor,
577  };
578  let db = match ObjectDB::attach(&m.dbfile, DbAttachOptions {
579    readonly: true,
580    clean:    false,
581  }) {
582    Ok(db) => db,
583    Err(e) => {
584      emit_fatal(
585        "post",
586        "render_worker",
587        &format!("cannot attach the render db {}: {e}", m.dbfile.display()),
588      );
589      return 0;
590    },
591  };
592  let url_style = UrlStyle::from_cli(&m.url_style).unwrap_or(UrlStyle::File);
593  let mut crossref = CrossRef::new(db, url_style, true);
594  crossref.set_extension(&m.out_extension);
595  if let Some(navtoc) = m.navigation_toc.as_deref() {
596    crossref.set_navigation_toc(navtoc);
597  }
598  let graphics = m.graphicimages.then(|| {
599    latexml_post::graphics::Graphics::new(None, true)
600      .with_svg_threshold_kb(m.graphics_svg_threshold_kb)
601  });
602  let post = latexml_post::Post::new();
603  let mut processors: Vec<Box<dyn Processor>> = Vec::new();
604  if m.pmml {
605    let mut presentation = latexml_post::mathml::MathML::new_presentation()
606      .with_keep_xmath(m.keep_xmath)
607      .with_invisible_times(m.invisible_times)
608      .with_plane1(m.plane1, m.hackplane1)
609      .with_mathtex(m.mathtex)
610      .with_intent_literal(m.intent_literal);
611    if m.cmml {
612      presentation = presentation.with_secondaries(vec![Box::new(
613        latexml_post::mathml::MathML::new_content()
614          .with_keep_xmath(m.keep_xmath)
615          .with_invisible_times(m.invisible_times)
616          .with_plane1(m.plane1, m.hackplane1)
617          .secondary(),
618      )]);
619    }
620    processors.push(Box::new(presentation));
621  } else if m.cmml {
622    processors.push(Box::new(
623      latexml_post::mathml::MathML::new_content()
624        .with_keep_xmath(m.keep_xmath)
625        .with_invisible_times(m.invisible_times)
626        .with_plane1(m.plane1, m.hackplane1),
627    ));
628  }
629  if let Some(xsl_path) = m.stylesheet.as_deref() {
630    let params: rustc_hash::FxHashMap<String, String> = m.xslt_params.iter().cloned().collect();
631    match latexml_post::xslt::XSLT::new(
632      xsl_path,
633      params,
634      m.nodefaultresources,
635      None,
636      m.searchpaths.clone(),
637    ) {
638      Ok(xslt) => processors.push(Box::new(xslt)),
639      Err(e) => emit_error("post", "xslt", &format!("XSLT error: {e}")),
640    }
641  }
642  let ctx = crate::post::PageRenderCtx {
643    page_opts:     m.page_opts.clone().into(),
644    is_html_out:   m.is_html_out,
645    svg_fragments: m.svg_fragments.clone(),
646    schemadocs:    m.schemadocs,
647    whatsout:      latexml_post::extract::Whatsout::from_cli(&m.whatsout).unwrap_or_default(),
648  };
649  let mut procs = crate::post::PageProcessors {
650    crossref,
651    graphics,
652    post,
653    processors,
654  };
655  let mut pages_written = 0usize;
656  for job in &m.pages {
657    // Same OS give-back cadence as the serial render loop: each page cycles a
658    // DOM + XSLT result through the C heap, and a chunk can be tens of
659    // thousands of pages.
660    if pages_written > 0 && pages_written.is_multiple_of(512) {
661      #[cfg(target_os = "linux")]
662      unsafe {
663        libc::malloc_trim(0);
664      }
665      #[cfg(not(feature = "dhat-heap"))]
666      unsafe {
667        libmimalloc_sys::mi_collect(true);
668      }
669    }
670    match crate::post::render_spilled_page(
671      &job.path,
672      &mut procs,
673      &ctx,
674      job.destination.clone(),
675      job.destination_directory.clone(),
676    ) {
677      Ok(outputs) => {
678        for (dest, output) in outputs {
679          if let Some(path) = dest.as_deref() {
680            if let Some(parent) = Path::new(path).parent()
681              && !parent.as_os_str().is_empty()
682            {
683              let _ = std::fs::create_dir_all(parent);
684            }
685            pages_written += 1;
686            if let Err(e) = std::fs::write(path, &output) {
687              emit_error(
688                "post",
689                "write",
690                &format!("failed to write page {path}: {e}"),
691              );
692            }
693          }
694        }
695      },
696      // Already reported inside the pipeline; mirror the serial driver's
697      // abort-on-page-failure (the parent flags the shortfall through the
698      // folded error + the fatal-bearing status this worker will report).
699      Err(()) => break,
700    }
701  }
702  pages_written
703}

709  #[test]
710  fn child_report_parses_status_lines_and_keeps_log() {
711    let stderr_text = "Warning:post:x something odd\nInfo:post:y fine\nStatus:pages:41\nStatus:counts:0,2,1,3,1\nStatus:conversion:3\n";
712    let r = parse_child_report(stderr_text);
713    assert_eq!(r.pages, 41);
714    assert_eq!(r.status, Some(3));
715    let c = r.counts.expect("counts parsed");
716    assert_eq!((c.debug, c.info, c.warning, c.error), (0, 2, 1, 3));
717    assert!(c.fatal);
718    assert!(r.log.contains("something odd"));
719    assert!(
720      !r.log.contains("Status:"),
721      "status lines must not leak into the folded log"
722    );
723  }
