23/// Where a runtime `.rhai` binding may be looked up — the two tiers differ in
24/// *cost* and in *authority*, so they sit at opposite ends of the dispatch
25/// chain (see [`install_binding_dispatch`]).
26#[cfg(feature = "runtime-bindings")]
27#[derive(Clone, Copy)]
28enum RhaiScope {
29  /// The local search paths only (source directory + `--path`) — a cheap
30  /// `pathname::find`, no kpsewhich. Checked FIRST, so a `.rhai` a user put
31  /// beside their document *overrides* a compiled binding of the same name.
32  LocalPaths,
33  /// Additionally the host TeX tree, via kpsewhich (`$TEXINPUTS`). Checked
34  /// LAST, only once every compiled dispatcher has declined (#345).
35  TeXTree,
36}

38/// Resolve a runtime `<request>.rhai` binding within `scope` and load it if
39/// present. Returns `None` when no such file exists, so the caller falls
40/// through to the next tier of the chain.
41/// See `docs/parity/script_bindings_plan.md` §7.
42#[cfg(feature = "runtime-bindings")]
43fn rhai_dispatch(request: &str, scope: RhaiScope) -> Option<Result<BindingSource>> {
44  use latexml_core::{
45    binding::content::{FindFileOptions, find_file},
46    state::record_opened_source,
47  };
48  let path = find_file(
49    request,
50    Some(FindFileOptions {
51      // Append `.rhai`, so `foo.sty` resolves `foo.sty.rhai`.
52      ext_type: Some("rhai".into()),
53      // `TeXTree` lets the search fall through to kpsewhich, which is what
54      // honours `$TEXINPUTS` — so a `<pkg>.sty.rhai` distributed in a texmf
55      // tree is found by `\usepackage{pkg}` with no `--path` (#345). kpsewhich
56      // locates it fine (the extension is irrelevant to a `//` recursive
57      // search). This tier is deliberately NOT used for the first-priority
58      // probe: that one runs on EVERY package/class request (64 of them on a
59      // plain acmart paper), and a kpathsea miss is a directory-tree probe —
60      // or a full fork-exec on the subprocess-`kpsewhich` backend. The memo
61      // in `pathname::kpsewhich` is keyed by candidate name, so distinct
62      // package names never share a hit.
63      search_paths_only: matches!(scope, RhaiScope::LocalPaths),
64      ..FindFileOptions::default()
65    }),
66  )?;
67  // Pin the resolved `.rhai` in the opened-sources read-log. `load_file`
68  // below reads it with a raw `std::fs::read_to_string` (it is not opened
69  // through a `Mouth`, so `Mouth::create`'s `record_opened_source` never
70  // fires for it). Without this, an edited binding is invisible to the warm
71  // LSP preamble cache (`warmup_dep_snapshot` / `deps_still_current`) and the
72  // stale macros survive every reconversion. Recording the resolved path lets
73  // the cache invalidate on the file's mtime change.
74  record_opened_source(arena::pin(&path));
75  // #560: report the resolved on-disk path as the load's `BindingSource`, so
76  // the "(Loading …)" note names the real `.rhai` file rather than the
77  // synthesized `<name>_sty.rs` compiled-module proxy name — more useful, and
78  // closer to Perl, which names the actual binding file.
79  Some(latexml_contrib::script_bindings::load_file(&path).map(|_| Some(path)))
80}

101/// `text` as TeX reads its code: each line (ended by CR, LF or CRLF, as the Mouth splits them) up to its first `%`
102/// that an even run of backslashes precedes (`\%` is a character, `\\%` a control symbol then a comment), without the
103/// files its `filecontents` environments write (a shipped `.sty` ends with its own `\endinput`), up to the first
104/// `\endinput` control word (TeX reads no further of the file).
105fn uncommented_code(text: &str) -> String {
106  static FILECONTENTS: Lazy<Regex> = Lazy::new(|| {
107    Regex::new(r"(?s)\\begin\s*\{\s*filecontents\*?\s*\}.*?\\end\s*\{\s*filecontents\*?\s*\}")
108      .unwrap()
109  });
110  static ENDINPUT: Lazy<Regex> = Lazy::new(|| Regex::new(r"\\endinput(?:[^A-Za-z@]|$)").unwrap());
111  let mut code = String::with_capacity(text.len());
112  for line in text.split(['\n', '\r']) {
113    let bytes = line.as_bytes();
114    let mut backslashes = 0;
115    let mut end = bytes.len();
116    for (i, &b) in bytes.iter().enumerate() {
117      match b {
118        b'\\' => backslashes += 1,
119        b'%' if backslashes % 2 == 0 => {
120          end = i;
121          break;
122        },
123        _ => backslashes = 0,
124      }
125    }
126    code.push_str(&line[..end]);
127    code.push('\n');
128  }
129  let mut code = FILECONTENTS.replace_all(&code, "").into_owned();
130  if let Some(at) = ENDINPUT.find(&code) {
131    code.truncate(at.start());
132  }
133  code
134}

136/// Whether `code` names a LaTeX document's cue: `\documentclass`, `\documentstyle` or `\begin{document}`, or a
137/// command plain TeX cannot run (`\usepackage`, `\RequirePackage`, `\include`).
138fn has_latex_cue(code: &str) -> bool {
139  static LATEX_CUE: Lazy<Regex> = Lazy::new(|| {
140    Regex::new(
141      r"\\document(?:class|style)|\\begin\s*\{\s*document\s*\}|\\(?:usepackage|RequirePackage|include)(?:[^A-Za-z@]|$)",
142    )
143    .unwrap()
144  });
145  LATEX_CUE.is_match(code)
146}

148/// The text of `path.tex` or, failing that, of `path` (TeX tries the `.tex` name first): a regular file of at most
149/// 4 MB, so a device or a FIFO named by an `\input` is never read.
150fn read_source_file(path: &std::path::Path) -> Option<String> {
151  const MAX_SOURCE_BYTES: u64 = 4 * 1024 * 1024;
152  [
153    std::path::PathBuf::from(format!("{}.tex", path.display())),
154    path.to_path_buf(),
155  ]
156  .into_iter()
157  .find(|candidate| {
158    std::fs::metadata(candidate).is_ok_and(|m| m.is_file() && m.len() <= MAX_SOURCE_BYTES)
159  })
160  .and_then(|candidate| std::fs::read(candidate).ok())
161  .map(|bytes| String::from_utf8_lossy(&bytes).into_owned())
162}

178/// The [`CompileRoute`] of the main file `source` (a path, `.tex` resolved as the Mouth does, or `literal:` content).
179/// A LaTeX document names `\documentclass` or `\documentstyle` outside a comment
180/// (AutoTeX's tex-vs-latex test) or `\begin{document}` (a main file `\input`ting its preamble), in the main file or,
181/// when it names none, in a file it `\input`s or `\include`s (a wrapper naming its preamble and body). AMSTeX's own
182/// `\documentstyle{amsppt}` (after `\input amstex`) still counts as LaTeX, though Perl reads it plain: kept plain, 64
183/// sampled 1996-99 amsppt papers went 192 → 319 errors (math9806005 0 → TooManyErrors), AMSTeX's plain-mode gaps. The
184/// first `\documentclass`, its options naming a DVI driver, in a file setting no `\pdfoutput=1`, makes it a latex+dvips one:
185/// 0908.4150's `\documentclass[12pt,dvips]{article}` (stopped full-arXiv run 329), which pdflatex stops on with
186/// l3backend's "Backend request inconsistent with engine" for `dvips`, `dvipdfmx` and `dvisvgm` (expl3-code.tex's
187/// backend options) and whose specials it drops for the other graphics drivers. An unreadable source counts as
188/// LaTeX, PDF.
189fn compile_route(source: &str) -> CompileRoute {
190  static PDFOUTPUT_ONE: Lazy<Regex> =
191    Lazy::new(|| Regex::new(r"\\pdfoutput\s*=?\s*1(?:[^0-9]|$)").unwrap());
192  static INPUT: Lazy<Regex> = Lazy::new(|| {
193    Regex::new(r"\\(?:input|include)(?:\s*\{\s*([^}]+?)\s*\}|\s+([^\s{}\\]+))").unwrap()
194  });
195  static IFFALSE: Lazy<Regex> =
196    Lazy::new(|| Regex::new(r"(?s)\\iffalse\b.*?\\(?:else|fi)\b").unwrap());
197  // A followed file's class cue must be one in use, not a dual-mode test (`\ifx\documentclass\undefined`).
198  static CLASS_IN_USE: Lazy<Regex> = Lazy::new(|| {
199    Regex::new(r"\\document(?:class|style)\s*[\[{]|\\begin\s*\{\s*document\s*\}").unwrap()
200  });
201  const MAX_INPUT_DEPTH: usize = 3;
202  let (text, dir) = match source.strip_prefix("literal:") {
203    Some(content) => (content.to_string(), None),
204    None => {
205      let path = std::path::Path::new(source);
206      match read_source_file(path) {
207        Some(text) => (text, path.parent().map(std::path::Path::to_path_buf)),
208        None => return CompileRoute::default(),
209      }
210    },
211  };
212  let mut code = uncommented_code(&text);
213  let figures = figure_output_cue(&code, dir.as_deref());
214  let mut latex = has_latex_cue(&code);
215  // A main file naming no class: the files it inputs (not in `\iffalse … \fi`), breadth first, a few levels deep,
216  // until one names the class; each using a LaTeX cue joins the main file's code for the DVI test.
217  if !code.contains("\\documentclass")
218    && let Some(dir) = dir
219  {
220    let mut visited = std::collections::HashSet::new();
221    let mut level = vec![IFFALSE.replace_all(&code, "").into_owned()];
222    'levels: for _ in 0..MAX_INPUT_DEPTH {
223      let mut next_level = Vec::new();
224      for parent in &level {
225        for captures in INPUT.captures_iter(parent) {
226          let Some(name) = captures.get(1).or_else(|| captures.get(2)) else {
227            continue;
228          };
229          // A name only TeX can resolve (`\input{\pre}`) may be the preamble: lean to LaTeX, the previous default. A `#`
230          // name is a macro's call-time argument (hep-ph9210212 `\def\inputrefs#1{…\input #1…}`, hep-th0703274
231          // `\input #1.aux`), no preamble: leaning to LaTeX there read a plain paper as LaTeX, whose locked `\chapter`
232          // refused its own (`\thechapter` undefined).
233          if name.as_str().contains('#') {
234            continue;
235          }
236          if name.as_str().contains('\\') {
237            latex = true;
238            continue;
239          }
240          if !visited.insert(name.as_str().to_string()) {
241            continue;
242          }
243          let Some(input) = read_source_file(&dir.join(name.as_str())) else {
244            continue;
245          };
246          let input_code = IFFALSE
247            .replace_all(&uncommented_code(&input), "")
248            .into_owned();
249          if CLASS_IN_USE.is_match(&input_code) {
250            latex = true;
251            code = format!("{code}\n{input_code}");
252            if input_code.contains("\\documentclass") {
253              break 'levels;
254            }
255          }
256          next_level.push(input_code);
257        }
258      }
259      level = next_level;
260    }
261  }
262  let class_option = |driver: &dyn Fn(&str) -> bool| {
263    code.find("\\documentclass").is_some_and(|at| {
264      let rest = code[at + "\\documentclass".len()..].trim_start();
265      rest
266        .strip_prefix('[')
267        .and_then(|r| r.split_once(']'))
268        .is_some_and(|(options, _)| options.split(',').any(|o| driver(o.trim())))
269    })
270  };
271  let dvi = !PDFOUTPUT_ONE.is_match(&code) && class_option(&|o| DVI_DRIVER_OPTIONS.contains(&o));
272  // the `pdftex` driver option: a pdflatex document whatever PostScript it also ships (MDPI templates' `logo-*.eps`
273  // beside PNG figures, 2210.04612: expl3's backend check, expl3-code.tex:8036-8041, errs under DVI output)
274  let pdf = !dvi && class_option(&|o| o == "pdftex");
275  CompileRoute {
276    plain_tex: !latex,
277    dvi,
278    pdf,
279    figures,
280  }
281}

283/// The output mode the figures a source includes call for (the K6 ruling's figure cue, OXIDIZED_DESIGN_DIVERGENCES
284/// #537): the targets of `\includegraphics`, `\epsfig`/`\psfig`, `\epsfbox`/`\epsffile` and AASTeX's
285/// `\plotone`/`\plottwo` in the main file's code `main_code` and the files it inputs (not in `\iffalse … \fi`, a few
286/// levels deep), each resolved in the source directory `dir` and the `\graphicspath` directories, as the name stands
287/// and with the extensions pdfTeX and dvips try.
288/// - PDF: a figure that exists only as PDF/PNG/JPEG (latex+dvips cannot include it).
289/// - DVI: a pstricks load (pstricks draws with PostScript specials; pst-pdf, auto-pst-pdf and pdftricks are its
290///   pdflatex front ends), or every included figure resolved and each PostScript only — whatever epstopdf
291///   conversions (`<stem>-eps-converted-to.pdf`) or epstopdf load the source also carries: arXiv compiled 1806.00093,
292///   1806.00188, 1806.00370, 1804.02991, 2104.01314, 1209.6096, 2210.10273 and 1701.08490 with latex+dvips (their
293///   PDFs' Producer is "dvips + GPL Ghostscript"), and 1806.00284 (Producer "GPL Ghostscript", Creator dvips(k)).
294/// - Otherwise undecided, and the tree's files decide: a figure that does not resolve (a name TeX builds at run time,
295///   `\figdir/a`, `#1`; a missing file), or one held in both forms.
296///
297/// A file the source never includes decides nothing: 1012.0815's EPS figures beside an IEEEtran manual PDF are a
298/// latex+dvips document whose `\ifCLASSINFOpdf` (unclosed) swallowed the whole text in PDF mode, and 1002.1751 loads
299/// pstricks only in its `\ifpdf` false branch, its tree's stale PDFs included by nothing.
300fn figure_output_cue(main_code: &str, dir: Option<&std::path::Path>) -> state::FigureOutputCue {
301  use state::FigureOutputCue;
302  static IFFALSE: Lazy<Regex> =
303    Lazy::new(|| Regex::new(r"(?s)\\iffalse\b.*?\\(?:else|fi)\b").unwrap());
304  static INPUT: Lazy<Regex> = Lazy::new(|| {
305    Regex::new(r"\\(?:input|include|subfile)(?:\s*\{\s*([^}]+?)\s*\}|\s+([^\s{}\\]+))").unwrap()
306  });
307  static FIGURE: Lazy<Regex> = Lazy::new(|| {
308    Regex::new(concat!(
309      r"\\includegraphics\*?\s*(?:<[^>]*>\s*)?(?:\[[^\]]*\]\s*){0,2}\{\s*([^}]+?)\s*\}",
310      r"|\\(?:epsfig|psfig)\s*\{[^}]*?\b(?:file|figure)\s*=\s*([^,}\s]+)",
311      r"|\\(?:epsfbox|epsffile)\s*(?:\[[^\]]*\]\s*)?\{\s*([^}]+?)\s*\}",
312      r"|\\plotone\s*\{\s*([^}]+?)\s*\}",
313      r"|\\plottwo\s*\{\s*([^}]+?)\s*\}\s*\{\s*([^}]+?)\s*\}",
314    ))
315    .unwrap()
316  });
317  static GRAPHICSPATH: Lazy<Regex> =
318    Lazy::new(|| Regex::new(r"\\graphicspath\s*\{((?:\s*\{[^{}]*\})+)\s*\}").unwrap());
319  static BRACED: Lazy<Regex> = Lazy::new(|| Regex::new(r"\{([^{}]*)\}").unwrap());
320  static PACKAGES: Lazy<Regex> = Lazy::new(|| {
321    Regex::new(r"\\(?:usepackage|RequirePackage)\s*(?:\[[^\]]*\]\s*)?\{([^}]*)\}|\\input\s*\{?\s*(pstricks)\b").unwrap()
322  });
323  const MAX_INPUT_DEPTH: usize = 4;
324  const MAX_FILES: usize = 200;
325  // A name TeX resolves at run time (`#1`, `\figdir`), or one outside the source tree (absolute, `..`), is not read.
326  let outside_the_tree = |name: &str| {
327    name.contains(['#', '\\']) || name.starts_with('/') || name.split('/').any(|part| part == "..")
328  };
329  const POSTSCRIPT: [&str; 6] = ["eps", "ps", "epsf", "epsi", "eps.gz", "ps.gz"];
330  const NOT_POSTSCRIPT: [&str; 4] = ["pdf", "png", "jpg", "jpeg"];
331  // The source's code: the main file's and that of the files it inputs, breadth first.
332  let mut codes = vec![IFFALSE.replace_all(main_code, "").into_owned()];
333  if let Some(dir) = dir {
334    let mut visited = std::collections::HashSet::new();
335    let mut level = 0..1;
336    for _ in 0..MAX_INPUT_DEPTH {
337      let mut names = Vec::new();
338      for code in &codes[level.clone()] {
339        for captures in INPUT.captures_iter(code) {
340          if let Some(name) = captures.get(1).or_else(|| captures.get(2)) {
341            names.push(name.as_str().to_string());
342          }
343        }
344      }
345      let start = codes.len();
346      for name in names {
347        if codes.len() >= MAX_FILES {
348          break;
349        }
350        if outside_the_tree(&name) || !visited.insert(name.clone()) {
351          continue;
352        }
353        if let Some(input) = read_source_file(&dir.join(&name)) {
354          codes.push(
355            IFFALSE
356              .replace_all(&uncommented_code(&input), "")
357              .into_owned(),
358          );
359        }
360      }
361      level = start..codes.len();
362      if level.is_empty() {
363        break;
364      }
365    }
366  }
367  let mut pstricks = false;
368  let mut pstricks_for_pdf = false;
369  for captures in codes.iter().flat_map(|code| PACKAGES.captures_iter(code)) {
370    if captures.get(2).is_some() {
371      pstricks = true;
372    }
373    for package in captures
374      .get(1)
375      .map_or("", |m| m.as_str())
376      .split(',')
377      .map(str::trim)
378    {
379      match package {
380        "pst-pdf" | "auto-pst-pdf" | "pdftricks" => pstricks_for_pdf = true,
381        _ if package == "pstricks" || package.starts_with("pst-") => pstricks = true,
382        _ => {},
383      }
384    }
385  }
386  // Every included figure, resolved: how many, how many PostScript only, and whether one did not resolve.
387  let (mut figures, mut postscript_only, mut unresolved) = (0usize, 0usize, false);
388  if let Some(dir) = dir {
389    let mut search = vec![dir.to_path_buf()];
390    for captures in codes
391      .iter()
392      .flat_map(|code| GRAPHICSPATH.captures_iter(code))
393    {
394      for path in BRACED.captures_iter(&captures[1]) {
395        let path = path[1].trim();
396        if !path.is_empty() && !outside_the_tree(path) {
397          search.push(dir.join(path));
398        }
399      }
400    }
401    let is_file = |name: &str| search.iter().any(|base| base.join(name).is_file());
402    for captures in codes.iter().flat_map(|code| FIGURE.captures_iter(code)) {
403      for name in captures.iter().skip(1).flatten() {
404        let name = name.as_str().trim_matches('"');
405        if name.is_empty() {
406          continue;
407        }
408        if outside_the_tree(name) {
409          unresolved = true;
410          continue;
411        }
412        figures += 1;
413        let exists = |ext: &str| {
414          is_file(&format!("{name}.{ext}"))
415            || is_file(&format!("{name}.{}", ext.to_ascii_uppercase()))
416        };
417        let named = name.to_ascii_lowercase();
418        let named_ext = POSTSCRIPT
419          .iter()
420          .chain(NOT_POSTSCRIPT.iter())
421          .find(|ext| named.ends_with(&format!(".{ext}")));
422        let (ps, other) = match named_ext {
423          // A name with its extension is that file (graphicx, `\epsfbox`).
424          Some(ext) if is_file(name) => (POSTSCRIPT.contains(ext), NOT_POSTSCRIPT.contains(ext)),
425          Some(_) => (false, false),
426          None => (
427            POSTSCRIPT.iter().any(|ext| exists(ext)),
428            NOT_POSTSCRIPT.iter().any(|ext| exists(ext)),
429          ),
430        };
431        if other && !ps {
432          return FigureOutputCue::Pdf;
433        }
434        if ps && !other {
435          postscript_only += 1;
436        }
437        unresolved |= !ps && !other;
438      }
439    }
440  }
441  if (pstricks && !pstricks_for_pdf) || (figures > 0 && !unresolved && postscript_only == figures) {
442    FigureOutputCue::Dvi
443  } else {
444    FigureOutputCue::Undecided
445  }
446}

448/// Install the binding-resolution **priority chain** as the single dispatcher
449/// and register the `latexml_package` binding-name registry.
450///
451/// This is the one definition of binding-resolution policy, shared by
452/// [`Converter::initialize_session`] and the integration-test harness
453/// (`util::test::process_texfile`) — so a test exercises the *same* chain a real
454/// conversion does, and local `.rhai` fixtures are discovered identically.
455///
456/// Priority, highest first (installed in ONE slot, so call-site ordering can't
457/// reshuffle it):
458///   1. a local `<request>.rhai` in the search paths — lets a user (or a test
459///      fixture next to its `.tex`) OVERRIDE any compiled binding of the same
460///      name (e.g. `article.cls.rhai` shadows `article_cls`). `runtime-bindings`
461///      only.
462///   2. the `extra` dispatcher (`latexml_contrib` for our binaries) — consulted
463///      before `latexml_package` to preserve the prior external-before-internal
464///      order; the two registries are disjoint, so the order is immaterial.
465///      Compiled `.rs` bindings keep precedence even under the raw-
466///      interpretation preloads (`rawstyles`/`rawclasses`) — user directive
467///      2026-08-31 (perfect-kernel mission): raw mode governs what happens for
468///      names with NO binding (raw-load instead of the OmniBus fallback), it
469///      never demotes an existing binding.
470///   3. `latexml_package` — core compiled engine bindings.
471///   4. a `<request>.rhai` on the host TeX tree (`$TEXINPUTS`, via kpsewhich) —
472///      a binding *distributed* with a package, so it FILLS A GAP rather than
473///      overriding: `\usepackage{X}` finds `X.sty.rhai` in a texmf tree with no
474///      `--path` (#345), while a stray `amsmath.sty.rhai` left on that tree
475///      cannot silently displace the compiled `amsmath` binding. Last, so the
476///      kpathsea probe is paid only by requests nothing else could answer.
477///      `runtime-bindings` only.
478pub(crate) fn install_binding_dispatch(extra: Option<BindingDispatcher>) {
479  set_bindings_dispatch(Rc::new(move |request: &str| {
480    #[cfg(feature = "runtime-bindings")]

620    // Perl: $self->initializeState($mode.".pool", @{$$self{preload} || []})
621    // For `--bibtex` (mode = BibTeX), Perl `Common/Config.pm:406`
622    // unshifts ['TeX.pool', 'LaTeX.pool', 'BibTeX.pool'] into the preload
623    // list. `BibTeX.pool` already begins with `LoadPool('LaTeX')` (and
624    // LaTeX with TeX), so we only need the BibTeX entry — the transitive
625    // chain handles the rest, and pool loads are idempotent.
626    // The format's dump must be built from the TeX tree this process reads packages from; one is
627    // built when none is (`format_dumps`), before the first format loads.
628    crate::format_dumps::ensure_format_dumps();

630      Some(DigestionMode::BibTeX) => vec![s!("TeX.pool"), s!("BibTeX.pool")],
631      _ => vec![s!("TeX.pool")],

643    // "1 warning". Running it inline here guarantees the tables are complete
644    // before `convert()` looks anything up, on every thread. Idempotent and a
645    // fast no-op once warm; single-process/single-thread runs (the CLI, the
646    // one-conversion-per-process cortex fleet) are unaffected. Honours the
647    // CLI's `LATEXML_NO_KPATHSEA_PREWARM` benchmarking opt-out.
648    if std::env::var_os("LATEXML_NO_KPATHSEA_PREWARM").is_none() {

650    }

652    // carries it. A dead or degraded kpathsea is otherwise invisible — it looks
653    // exactly like a document referencing files that do not exist — and issue
654    // #304 cost days for want of this one line in the reporter's log.
655    let (backend, why) = latexml_core::util::pathname::kpathsea_backend();
656    Info!("kpathsea", "backend", s!("{} ({why})", backend.as_str()));

658    Ok(())
659  }

1003  /// The SHARED conversion tail: instrumentation flush (ASF stats, the
1004  /// `--source-map` decoder table), the Perl-faithful completion `Note!`, and
1005  /// response assembly. Every arm of `convert` must end here — the streaming
1006  /// arm used to return early with a hand-copy of the verdict fold alone,
1007  /// silently skipping the rest (no `MARPA_ASF_STATS` line, and a streamed
1008  /// `--source-map` run emitted `data:sourcepos` tags with no decoder ring).
1009  ///
1010  /// Recomputes `status`/`status_code` (idempotent reads of the REPORT
1011  /// counters), so diagnostics raised during serialization itself still reach
1012  /// the reported verdict — the streaming arm always did this; the eager path
1013  /// previously froze status before serializing.
1014  fn finish_response(&mut self, serialized: String) -> ConversionResponse {
1015    self.runtime.status = get_status_message();
1016    self.runtime.status_code = get_status_code();
1017    if self.opts.verbosity >= 0 {
1018      Debug!("arena", "strings_allocated", arena::len());
1019      // Final token-read progress: the calibration basis for `token_limit`
1020      // and `CYCLE_GUARD_ACTIVATE` (the read-checkpoint accounting changed
1021      // in PR #249 — read_x_token/read_balanced now count too — so limits
1022      // must be recalibrated against THIS metric, not historical figures).
1023      Debug!("gullet", "progress", latexml_core::gullet::token_progress());
1024    }
1025    // MARPA_ASF_STATS=1: emit ASF instrumentation counters once
1026    // per converted document. Codex instrumentation plan, see
1027    // marpa/docs/ASF_PERFORMANCE_FINDINGS.md. The thread-local
1028    // accumulator is reset after the snapshot so per-document
1029    // figures are independent.
1030    latexml_math_parser::report_and_reset_asf_stats();
1031    // --source-map (#47/#92): serialise the `tag → file` decoder table into the
1032    // `.log` — latexml-oxide's existing conversion-metadata channel — rather than
1033    // inlining it into the output. The output carries only the anonymous integer
1034    // `tag` (in each `data:sourcepos`); this is its decoder ring, Source-Map-v3
1035    // `sources`-style (the array index *is* the tag). Keeping it out of the
1036    // HTML/XML keeps that output anonymisable: a consumer without the source
1037    // files sees only opaque tags. In-process embedders (e.g. the ar5iv-editor
1038    // server) read the same table programmatically via `source_table_snapshot()`.
1039    // Gated on the switch, so a normal conversion emits nothing.
1040    if source_map_enabled() {
1041      for (tag, sym) in source_table_snapshot().iter().enumerate() {
1042        arena::with(*sym, |src| {
1043          Info!("source-map", "source", s!("[{tag}] {src}"));
1044        });
1045      }
1046    }
1047    // Perl: Note("Conversion complete: " . $$runtime{status}); (LaTeXML.pm:315)
1048    // is reached only on success — a Fatal `die`s before it, and bin/latexml:127
1049    // then prints `"Conversion " . ($code == 3 ? 'failed' : 'complete')`. Rust
1050    // recovers from a Fatal (graceful degradation) instead of dying, so it reaches
1051    // this note even when status_code == 3; fold in bin/latexml's verdict here so
1052    // a fatal run reports "failed", never the self-contradictory "complete: N fatal
1053    // error". Success cases (status_code < 3) stay byte-identical.
1054    Note!(s!("{}", conversion_verdict(self.runtime.status_code)));
1055    let log = self.flush_log();
1056    // self->sanitize($log) if ($$runtime{status_code} == 3);
1057
1058    ConversionResponse {
1059      result: Some(serialized),
1060      log,
1061      status: self.runtime.status.clone(),
1062      status_code: self.runtime.status_code,
1063    }
1064  }

1066  /// Convert in-memory `content` under the source name `name`, producing the
1067  /// HTML5-format core XML (the persistent server then post-processes it).
1068  /// Unlike [`Converter::convert`] (`literal:` → anonymous source), the source
1069  /// is *named*, so `--source-map` stamps its locators. Focused on the
1070  /// `Document`/HTML5 path the server uses — no amble wrapping, no TeX/Box
1071  /// output formats.
1072  pub fn convert_content_with_provenance(
1073    mut self,
1074    name: &str,
1075    content: String,
1076  ) -> ConversionResponse {
1077    // Load + digest through the shared top-level loader, so the source-context
1078    // setup is not duplicated here.
1079    let digested = match self.digest_content_with_provenance(name, content) {
1080      Ok(d) => d,
1081      Err(e) => {
1082        if !self.ready {
1083          return ConversionResponse {
1084            result:      None,
1085            log:         self.flush_log(),
1086            status:      s!("Initialization failed."),
1087            status_code: 3,
1088          };
1089        }
1090        report_mut!().status_code = 3;
1091        e.log_fatal();
1092        // Salvage whatever digested before the error (mirrors `convert`).
1093        match self.core.digest_internal() {
1094          Ok(salvaged) if !salvaged.is_empty().unwrap_or(true) => salvaged,
1095          _ => Digested::from(List::new(Vec::new())),
1096        }
1097      },
1098    };
1099
1100    let serialized = {
1101      let _g = telemetry::phase(Phase::Build);
1102      match self.core.convert_document(digested) {
1103        Ok(dom) => {
1104          // No `note_rootless_document` here: this in-process entry is also
1105          // the editor's fallback for a fragment without `\begin{document}`
1106          // (lsp_server/server.rs), a legitimate transient state.
1107          let _g = telemetry::phase(Phase::Serialize);
1108          dom.serialize_to_string()
1109        },
1110        Err(e) => {
1111          // `Error!` expands into a `Result`-returning context; wrap it the
1112          // same way `convert` does so it composes in this `-> ConversionResponse` fn.
1113          // Timeout-target resource fatals get the standard `Fatal:` line
1114          // (see the sibling handler in `convert` — P1-4).
1115          if matches!(e.category, ErrorCategory::StreamingRestart) {
1116            // Not a failure: the CLI reruns this document under --streaming
1117            // (the watermark crossed during the build; the digest arm above
1118            // handles the common digestion crossing).
1119            emit_info("streaming", "restart_watermark", &e.message);
1120          } else if matches!(e.target, ErrorTarget::Timeout) {
1121            e.log_fatal();
1122          } else {
1123            let message = s!("{:?}", e);
1124            let err = || {
1125              Error!("document", "convert", message);
1126              Ok(())
1127            };
1128            err().ok();
1129          }
1130          String::new()
1131        },
1132      }
1133    };
1134
1135    // The verdict counts what building the document raised too — its own diagnostics, and the deferred ones replayed
1136    // once every reference is built (ruling 7e) — as `finish_response` recomputes it for `convert`; read after
1137    // digestion alone, a construction-time error left the status "No obvious problems".
1138    self.runtime.status = get_status_message();
1139    self.runtime.status_code = get_status_code();
1140    let log = self.flush_log();
1141    ConversionResponse {
1142      result: Some(serialized),
1143      log,
1144      status: self.runtime.status.clone(),
1145      status_code: self.runtime.status_code,
1146    }
1147  }

1159  /// Digest in-memory `content` as the **main document** named `name`, leaving
1160  /// the thread-local engine state **live**. Used by the persistent server to
1161  /// warm a preamble once (then resume body digestion in a fork child over the
1162  /// inherited state) and by [`Converter::convert_content_with_provenance`] for
1163  /// the in-process path.
1164  ///
1165  /// This is the in-memory twin of [`crate::core_interface::DigestionAPI::digest_file`]:
1166  /// same top-level spine — establish the source context, open the source,
1167  /// `digest_internal` — but the content is *supplied* rather than read from
1168  /// disk. It shares `core_interface::establish_source_context` with
1169  /// `digest_file` (so `SOURCEFILE`/`SOURCEDIRECTORY`/`SEARCHPATHS`/
1170  /// `GRAPHICSPATHS`/`\jobname` can't drift), making sibling
1171  /// `\usepackage`/`\input`/`\includegraphics` of local files resolve. The
1172  /// source is opened as a *named* mouth (not the anonymous `literal:`
1173  /// protocol) so locators carry `name` — the **provenance** that
1174  /// `--source-map` needs (`stamp_source_locator` only stamps
1175  /// `.tex`/`.ltx`/`.bbl`/`.bib` user sources). Initializes the session if
1176  /// needed; does not finalize a document.
1177  pub fn digest_content_with_provenance(
1178    &mut self,
1179    name: &str,
1180    content: String,
1181  ) -> Result<Digested> {
1182    if !self.ready {
1183      self.initialize_session()?;
1184    }
1185    self.bind_log();
1186    note_format_build_failure();
1187    // Top-level document load: establish the source context (SOURCEFILE,
1188    // SOURCEDIRECTORY, SEARCHPATHS, GRAPHICSPATHS, \jobname) so sibling
1189    // \usepackage/\input/\includegraphics of local files resolve. Shared with
1190    // `digest_file` via `establish_source_context` so the two can't drift.
1191    // (A continuation/nested mouth must NOT do this — see
1192    // `open_named_in_memory_mouth`.)
1193    let path = std::path::Path::new(name);
1194    let dir = path.parent().and_then(|p| p.to_str()).unwrap_or("");
1195    let jobname = path.file_stem().and_then(|s| s.to_str()).unwrap_or(name);
1196    crate::core_interface::establish_source_context(Some(name), jobname, dir);
1197    open_named_in_memory_mouth(name, content)?;
1198    self.core.digest_internal()
1199  }

1225/// Resolve the `(preamble, postamble)` to wrap the source in, based on
1226/// the requested input chunk size. Faithful port of Perl `LaTeXML.pm`
1227/// L165-172 — note both ambles key on **`whatsin`** (not `whatsout`):
1228///
1229/// * `math` → `\begin{document}\ensuremathfollows` … `\ensuremathpreceeds\end{document}` (magic
1230///   math-mode trigger).
1231/// * `fragment` → the caller-supplied `preamble`/`postamble`, defaulting to `standard_preamble.tex`
1232///   / `standard_postamble.tex`.
1233/// * everything else (`document`, `archive`, …) → no wrapping.
1234pub(crate) fn resolve_amble(
1235  whatsin: &DataSize,
1236  preamble: &Option<String>,
1237  postamble: &Option<String>,
1238) -> (Option<String>, Option<String>) {
1239  match whatsin {
1240    DataSize::Math => (
1241      Some(s!("literal:\\begin{{document}}\\ensuremathfollows")),
1242      Some(s!("literal:\\ensuremathpreceeds\\end{{document}}")),
1243    ),
1244    DataSize::Fragment => (
1245      Some(
1246        preamble
1247          .clone()
1248          .unwrap_or_else(|| s!("standard_preamble.tex")),
1249      ),
1250      Some(
1251        postamble
1252          .clone()
1253          .unwrap_or_else(|| s!("standard_postamble.tex")),
1254      ),
1255    ),
1256    _ => (None, None),
1257  }
1258}

1260/// A built document with no root element — `\begin{document}` was never
1261/// digested, typically because an unbalanced `\ifX` (an undefined conditional
1262/// auto-`\newif`ed to `\iffalse`) skipped it through EOF — is a FATAL, not
1263/// an output. Both engines write the bare XML declaration for it (Perl too:
1264/// xwatermark-guide), but reporting it as status 2 with "output" hid a whole-
1265/// document loss behind an `Error:expected:\fi` line (skeyval-pokayoke2,
1266/// thesis-sample, xwatermark-guide in sweeps s109/s110). The messages are the
1267/// success signal, so the loss is named here — on the full-document `convert`
1268/// path (CLI, corpus harness, cortex_worker); the editor's in-process
1269/// fragment fallback is exempt. `Fatal!` logs and latches the sticky fatal at
1270/// the raise; the `Err` is not needed — serialization of the empty document
1271/// proceeds as before. Only a LaTeX document — one whose `\documentclass`
1272/// loaded a class (`document_class_filename`) — has a `\begin{document}` to
1273/// miss: a source that is all comments (arXiv's `%auto-ignore` placeholders,
1274/// 53 papers of sandbox 2605, e.g. 2605.00131; 28 of 2606) or a plain-TeX
1275/// file that typesets nothing lost no content, and stays clean as in Perl.
1276/// Guards
1277/// `perfect_kernel_batch56::document_without_a_root_is_a_fatal`,
1278/// `perfect_kernel_batch56::comment_only_source_is_not_a_fatal`.
1279fn note_rootless_document(dom: &Document) {
1280  // A Fatal already on record (TooManyErrors, a resource fuse) explains the
1281  // loss; do not add a second line for the same event.
1282  if dom.get_document().get_root_element().is_none()
1283    && get_status_code() < 3
1284    && state::has_value("document_class_filename")
1285  {
1286    let _: Result<()> = (|| {
1287      Fatal!(
1288        Document,
1289        Malformed,
1290        "The conversion built a document with no root element: \\begin{document} was never \
1291         reached (an unbalanced conditional skipped it?); the output is empty"
1292      );
1293    })();
1294  }
1295}

1499  #[test]
1500  fn figure_output_cue_of_sources() {
1501    use state::FigureOutputCue::{Dvi, Pdf, Undecided};
1502    let dir = tempfile::tempdir().expect("tempdir");
1503    let write = |name: &str, content: &str| {
1504      let path = dir.path().join(name);
1505      std::fs::create_dir_all(path.parent().unwrap()).unwrap();
1506      std::fs::write(path, content).unwrap();
1507    };
1508    let cue = |code: &str| figure_output_cue(&uncommented_code(code), Some(dir.path()));
1509    write("figs/a.eps", "");
1510    write("figs/b.png", "");
1511    write("figs/c.eps", "");
1512    write("figs/c.pdf", "");
1513    write("doc/manual.pdf", "");
1514    write("img/d.EPS", "");
1515    // An EPS-only figure, by name or extension-less; one that is PNG only; one in both forms decides nothing.
1516    assert_eq!(cue("\\includegraphics[width=1cm]{figs/a.eps}"), Dvi);
1517    assert_eq!(cue("\\includegraphics{figs/a}"), Dvi);
1518    assert_eq!(
1519      cue("\\includegraphics{figs/a}\\includegraphics{figs/b}"),
1520      Pdf
1521    );
1522    assert_eq!(cue("\\includegraphics{figs/c}"), Undecided);
1523    // A missing figure, a macro argument, a commented or `\iffalse`d include: nothing.
1524    assert_eq!(cue("\\includegraphics{figs/none}"), Undecided);
1525    assert_eq!(cue("\\includegraphics{../figs/a}"), Undecided);
1526    assert_eq!(cue("\\def\\f#1{\\includegraphics{#1}}"), Undecided);
1527    assert_eq!(cue("% \\includegraphics{figs/a}\n"), Undecided);
1528    assert_eq!(cue("\\iffalse\\includegraphics{figs/a}\\fi"), Undecided);
1529    // `\graphicspath`, an upper-case extension, the epsf/psfig family, AASTeX's `\plotone`.
1530    assert_eq!(cue("\\graphicspath{{img/}}\\includegraphics{d}"), Dvi);
1531    assert_eq!(cue("\\epsfig{file=figs/a.eps,width=3cm}"), Dvi);
1532    assert_eq!(cue("\\psfig{figure=figs/b.png}"), Pdf);
1533    assert_eq!(cue("\\epsfbox{figs/a.eps}"), Dvi);
1534    assert_eq!(cue("\\plotone{figs/a.eps}"), Dvi);
1535    // pstricks; its pdflatex front ends.
1536    assert_eq!(cue("\\usepackage{pstricks,pst-node}"), Dvi);
1537    assert_eq!(cue("\\usepackage{pst-plot}"), Dvi);
1538    assert_eq!(cue("\\input pstricks"), Dvi);
1539    assert_eq!(
1540      cue("\\usepackage{pstricks}\\usepackage{auto-pst-pdf}"),
1541      Undecided
1542    );
1543    assert_eq!(cue("\\usepackage{pstricks}\\includegraphics{figs/b}"), Pdf);
1544    // Only when every included figure resolves: a missing or run-time-named one leaves the tree's files to decide.
1545    assert_eq!(
1546      cue("\\includegraphics{figs/a}\\includegraphics{figs/none}"),
1547      Undecided
1548    );

1550      cue("\\includegraphics{figs/a}\\includegraphics{\\figdir/b}"),
1551      Undecided
1552    );

1554    // with latex+dvips (1806.00093, 1804.02991: Producer "dvips + GPL Ghostscript").
1555    write("figs/e.eps", "");
1556    write("figs/e-eps-converted-to.pdf", "");
1557    assert_eq!(cue("\\includegraphics{figs/e}"), Dvi);
1558    assert_eq!(cue("\\includegraphics{figs/e.eps}"), Dvi);
1559    assert_eq!(cue("\\usepackage{epstopdf}\\includegraphics{figs/a}"), Dvi);
1560    // A file the main file inputs.
1561    write("sec.tex", "\\includegraphics{figs/b.png}\n");
1562    assert_eq!(cue("\\input{sec}"), Pdf);
1563    // No directory: nothing resolves.
1564    assert_eq!(
1565      figure_output_cue("\\includegraphics{figs/a}", None),
1566      Undecided
1567    );
1568  }
