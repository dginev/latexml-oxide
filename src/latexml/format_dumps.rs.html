1//! A format dump for the TeX tree this process reads.
2//!
3//! A dump is the format: the state `latex.ltx` (and the L3 kernel it loads) or `plain.tex` leaves,
4//! built from one TeX tree's files. Packages come raw from the tree the process reads, so a dump
5//! built from another tree is the wrong format for them — `expl3.sty` refuses a format whose L3
6//! kernel date differs from its own ("Mismatched LaTeX support files", expl3.sty:64-78). The 2609
7//! release ran a dump built from `/usr/local/texlive/2025` with `/usr/share/texlive`'s packages, and
8//! 12,144 papers ended as Error (witnesses 2607.28725, 2609.40175).
9//!
10//! Every dump records the files it was built from, and the loaders pass over one this tree does not
11//! have (`latexml_engine::dump_paths::dump_matches_tree`). When no dump found matches, the format
12//! is built here once, as `--init` and fmtutil build it, into a per-user cache keyed by the engine
13//! build and the tree's files, and that dump is used. When it cannot be built, the format loads
14//! from the engine's own definitions, `LoadFormat`'s branch for a missing dump (Perl
15//! Package.pm:2755-2766).
16

31/// Marks, per format (`.failed.<kind>`), a cache directory where it could not be built, holding the
32/// reason: later processes take the base branch at once instead of building again. The mark lapses
33/// after [`FAILED_LAPSE`]; removing it retries at once.
34const FAILED: &str = ".failed";
35
36/// Counts, per format (`.attempts.<kind>`), the builds into a cache directory that have not
37/// succeeded in a row, with the last one's reason: written before a build, as one that dies with
38/// its process (killed, or a panic in a binary built to abort) records nothing after; removed by a
39/// success, or replaced by the [`FAILED`] mark. A build an interrupt cancels puts it back as it was
40/// ([`cancel_builds_on_interrupt`]).
41const ATTEMPTS: &str = ".attempts";
42
43/// Builds of a format that may fail in a row before it is marked [`FAILED`]: a format that does not
44/// build is not tried by every process, and one passing failure (a full disk, no thread) is not
45/// taken for it.
46const MAX_ATTEMPTS: u32 = 2;
47
48/// How long a [`FAILED`] mark holds: a failure that passes (the disk is cleared, the tree repaired)
49/// is not kept for good.
50const FAILED_LAPSE: Duration = Duration::from_secs(24 * 60 * 60);
51
52/// How long a process waits for another one building into the same directory — far longer than a
53/// build takes, even on a loaded host.
54const LOCK_WAIT: Duration = Duration::from_secs(300);
55
56/// The exit status of a process an interrupt ends while it builds a format: a shell's for Ctrl+C
57/// (128 + SIGINT).
58const INTERRUPTED: i32 = 130;
59
60/// Whether an interrupt cancels a build ([`cancel_builds_on_interrupt`]).
61static CANCEL_ON_INTERRUPT: AtomicBool = AtomicBool::new(false);
62
63/// The build this process is running, as an interrupt would undo it ([`cancel_in_flight_build`]).
64static IN_FLIGHT: Mutex<Option<InFlight>> = Mutex::new(None);

66/// What a running build has left in its cache directory.
67struct InFlight {
68  /// The build's [`ATTEMPTS`] note, and what it held before the build (`None`: no note).
69  attempts: PathBuf,
70  before:   Option<String>,
71  /// The dump being written.
72  partial:  PathBuf,
73}

75/// Makes an interrupt that ends this process while it builds a format dump a cancellation, not a
76/// failed build (issue 988): the build's attempt note goes back to what it held before the build,
77/// its partial dump is removed, and the process exits with status 130. The next process builds the
78/// format again, where an interrupted build used to count towards marking it failed (`.failed.<kind>`), so two
79/// interrupts left every conversion for a day on the engine's own definitions. The interrupts are
80/// Ctrl+C and, on Unix, SIGTERM and SIGHUP (on Windows, Ctrl+Break and closing the console). For a
81/// front end its user interrupts (the `latexml_oxide` CLI's conversions); an embedder that handles
82/// signals itself leaves it off. The handler is installed when the process starts its first build,
83/// so a process that builds nothing keeps its default interrupt. One that builds keeps the handler
84/// to its end: a later SIGTERM or SIGHUP ends it with status 130, not by the signal, since neither
85/// ctrlc nor a safe `signal`/`sigaction` can put the default action back. A process that ignores or
86/// handles any of these signals already (`nohup`, a shell's background job) keeps them as they are,
87/// and an interrupted build there counts as before.
88pub fn cancel_builds_on_interrupt() { CANCEL_ON_INTERRUPT.store(true, Ordering::Release); }
89
90/// Installs, once per process, the interrupt handler [`cancel_builds_on_interrupt`] asks for.
91fn install_interrupt_handler() {
92  static INSTALLED: Once = Once::new();
93  INSTALLED.call_once(|| {
94    // `try_`: a signal the process already ignores or handles stays so (ctrlc's `MultipleHandlers`).
95    let installed = ctrlc::try_set_handler(|| {
96      // The record stays locked until the process ends, so the build cannot record an outcome after
97      // its undo (`set_in_flight` waits).
98      let mut slot = IN_FLIGHT.lock().unwrap_or_else(PoisonError::into_inner);
99      if let Some(build) = slot.take() {
100        undo(build);
101        note(log::Level::Info, format_args!(
102          "\n[format_dumps] interrupted: the format build was cancelled; the next conversion builds it again"
103        ));
104      }
105      // `_exit`: the build's thread still runs, and `exit` would run libxml2's destructor under it (#973). The
106      // note above is written (stderr is unbuffered) and the undo is on disk.
107      latexml_core::watchdog::exit_now(INTERRUPTED);
108    });
109    if let Err(e @ ctrlc::Error::System(_)) = installed {
110      note(log::Level::Warn, format_args!(
111        "[format_dumps] cannot take the interrupt ({e}): a build interrupted now counts as one that failed"
112      ));
113    }
114  });
115}

120  if level <= log::Level::Error || latexml_core::util::logger::stderr_admits(level) {
121    eprintln!("{args}");
122  }

125/// Undoes the format build this process is running, if any (the interrupt handler's undo, without
126/// its exit). Whether a build was running.
127#[cfg(test)]
128fn cancel_in_flight_build() -> bool {
129  let build = IN_FLIGHT
130    .lock()
131    .unwrap_or_else(PoisonError::into_inner)
132    .take();
133  build.map(undo).is_some()
134}

136/// Undoes a format build: its attempt note goes back to what it held before the build and its
137/// partial dump is removed.
138fn undo(build: InFlight) {
139  match build.before {
140    Some(before) => {
141      let _ = std::fs::write(&build.attempts, before);
142    },
143    None => {
144      let _ = std::fs::remove_file(&build.attempts);
145    },
146  }
147  // (On Windows a file the build still has open stays; the next build removes it, `remove_partial_dumps`.)
148  let _ = std::fs::remove_file(&build.partial);
149}

151/// Records the build this process starts or ends ([`IN_FLIGHT`]).
152fn set_in_flight(build: Option<InFlight>) {
153  *IN_FLIGHT.lock().unwrap_or_else(PoisonError::into_inner) = build;
154}

156/// Ensures each format has a dump built from this process's TeX tree, building any that has none
157/// into the cache. Once per process, before the first session loads a format; nothing to do without
158/// a TeX tree, under `LATEXML_NODUMP`, or while a format is being built (`--init`, or the thread
159/// building one here). A format that cannot be built is reported once, as a Warning on stderr (not
160/// under `--quiet`) and in each conversion's log (`converter::note_format_build_failure`), and
161/// loads from the engine's definitions. A build changes the process's working directory while
162/// it runs (see `build`), so it belongs before the process converts anything.
163pub fn ensure_format_dumps() {
164  if dump_paths::building_format() {
165    return;
166  }
167  static ENSURED: OnceLock<()> = OnceLock::new();
168  ENSURED.get_or_init(|| {
169    // Once per TeX tree, not the document's work: off the conversion's `--timeout`, which a busy host's build outlasted
170    // and, killed twice, marked the format failed. Bounded (under a non-zero `--timeout`) by the lock wait and a build.
171    let ensured = latexml_core::watchdog::off_the_clock(
172      "the format build",
173      LOCK_WAIT + Duration::from_secs(300),
174      ensure,
175    );
176    if let Err(reason) = ensured {
177      // A Warning, as each conversion logs it (`converter::note_format_build_failure`).
178      note(log::Level::Warn, format_args!(
179        "[format_dumps] {reason}: the format loads from the engine's own definitions instead of a dump"
180      ));
181      dump_paths::set_format_build_failure(reason);
182    }
183  });
184}

186fn ensure() -> Result<(), String> {
187  if std::env::var_os("LATEXML_NODUMP").is_some() || std::env::var_os("LATEXML_INI_MODE").is_some()
188  {
189    return Ok(());
190  }
191  let kinds: Vec<&'static str> = ["plain", "latex"]
192    .into_iter()
193    .filter(|kind| dump_paths::runtime_source_stamps(kind).is_some())
194    .collect();
195  if kinds.is_empty() {
196    return Ok(());
197  }
198  // A dump the configuration names, installs or keeps beside the sources is used before the cache's;
199  // the embedded one only after it, as reading it back costs every process a 4 MB dump.
200  let wanted: Vec<&'static str> = kinds
201    .into_iter()
202    .filter(|kind| !dump_paths::format_dump_on_disk(kind))
203    .collect();
204  if wanted.is_empty() {
205    return Ok(());
206  }
207  // Only the dump's file name carries the year.
208  let year = dump_paths::detect_ambient_texlive_year().unwrap_or(2000);
209  let dir = match cache_dir() {
210    Ok(dir) => dir,
211    Err(_)
212      if wanted
213        .iter()
214        .all(|kind| dump_paths::format_dump_available(kind)) =>
215    {
216      return Ok(());
217    },
218    Err(reason) => return Err(reason),
219  };
220  // Each kind uses the cache's dump on its own merits.
221  let use_cached = |kind: &str| match cached_dump(&dir, kind, year) {
222    Some((path, found_year)) => {
223      dump_paths::set_rebuilt_dump(kind, path, found_year);
224      true
225    },
226    None => false,
227  };
228  let missing: Vec<&'static str> = wanted
229    .into_iter()
230    .filter(|kind| !use_cached(kind) && !dump_paths::format_dump_available(kind))
231    .collect();
232  if missing.is_empty() {
233    return Ok(());
234  }
235  private_dir(&dir)?;
236  let mut reasons = Vec::new();
237  let missing: Vec<&'static str> = missing
238    .into_iter()
239    .filter(|kind| match failed_before(&dir, kind) {
240      Some(reason) => {
241        reasons.push(reason);
242        false
243      },
244      None => true,
245    })
246    .collect();
247  if !missing.is_empty() {
248    // What the builds so far have left, to tell whether one failed while this process waited. (A
249    // build that fails between this read and its release of the lock goes unseen: the window is
250    // the few instructions between its last write and its unlock.)
251    let seen: Vec<Option<String>> = missing
252      .iter()
253      .map(|kind| attempts_text(&dir, kind))
254      .collect();
255    // One process builds; the others wait for it and find the dump, or its failure.
256    let _lock = match lock_dir(&dir) {
257      Ok(lock) => lock,
258      Err(reason) => {
259        reasons.push(reason);
260        return Err(reasons.join("; "));
261      },
262    };
263    // No other process builds here while this one holds the lock: a partial dump is a killed
264    // build's.
265    remove_partial_dumps(&dir);
266    for (kind, seen) in missing.into_iter().zip(seen) {
267      // Built by the process this one waited for.
268      if use_cached(kind) {
269        continue;
270      }
271      if let Some(reason) = failed_before(&dir, kind) {
272        reasons.push(reason);
273        continue;
274      }
275      // A build that failed while this process waited is not tried again at once: a passing
276      // failure would likely recur, and a later process retries it.
277      let now = attempts_text(&dir, kind);
278      if now.is_some() && now != seen {
279        reasons.push(previous_attempts(&dir, kind).1);
280        continue;
281      }
282      if let Err(reason) = attempt(kind, year, &dir, &use_cached) {
283        reasons.push(reason);
284      }
285    }
286  }
287  if reasons.is_empty() {
288    Ok(())
289  } else {
290    Err(reasons.join("; "))
291  }
292}

294/// Builds the `kind` dump into `dir`, counting it until it succeeds and marking the format
295/// [`FAILED`] after [`MAX_ATTEMPTS`] builds in a row that did not.
296fn attempt(
297  kind: &'static str,
298  year: u32,
299  dir: &Path,
300  use_cached: &dyn Fn(&str) -> bool,
301) -> Result<(), String> {
302  let (failed, last_reason) = previous_attempts(dir, kind);
303  if failed >= MAX_ATTEMPTS {
304    return Err(mark_failed(dir, kind, &last_reason));
305  }
306  // Installed before the build leaves anything to undo, so an interrupt always finds what it left.
307  if CANCEL_ON_INTERRUPT.load(Ordering::Acquire) {
308    install_interrupt_handler();
309  }
310  {
311    // What an interrupt undoes is recorded and its note written under one lock: an interrupt
312    // meanwhile waits, then undoes both (it could otherwise undo before the note was written).
313    let mut slot = IN_FLIGHT.lock().unwrap_or_else(PoisonError::into_inner);
314    *slot = Some(InFlight {
315      attempts: kind_file(dir, ATTEMPTS, kind),
316      before:   attempts_text(dir, kind),
317      partial:  partial_dump(dir, kind, year),
318    });
319    if let Err(reason) = note_attempt(
320      dir,
321      kind,
322      failed + 1,
323      &format!("the {kind} format build died with its process"),
324    ) {
325      *slot = None;
326      return Err(reason);
327    }
328  }
329  note(
330    log::Level::Info,
331    format_args!(
332      "[format_dumps] no {kind} dump matches this TeX tree; building one into {} (once for this tree; later \
333     conversions reuse it)",
334      dir.display()
335    ),
336  );
337  let built = build(kind, year, dir);
338  // The build has ended (a published dump cleared the record as it was published, `publish`): what
339  // follows records its outcome, which an interrupt no longer undoes.
340  set_in_flight(None);
341  let built = built.and_then(|()| {
342    if use_cached(kind) {
343      Ok(())
344    } else {
345      Err(format!(
346        "the {kind} dump built into {} does not match this TeX tree",
347        dir.display()
348      ))
349    }
350  });
351  match built {
352    Ok(()) => {
353      let _ = std::fs::remove_file(kind_file(dir, ATTEMPTS, kind));
354      Ok(())
355    },
356    Err(reason) => {
357      let _ = note_attempt(dir, kind, failed + 1, &reason);
358      if failed + 1 >= MAX_ATTEMPTS {
359        Err(mark_failed(dir, kind, &reason))
360      } else {
361        Err(reason)
362      }
363    },
364  }
365}

367/// `dir`'s `marker` file (`.attempts`, `.failed`) for the `kind` format.
368fn kind_file(dir: &Path, marker: &str, kind: &str) -> PathBuf {
369  dir.join(format!("{marker}.{kind}"))
370}

372/// What the builds of the `kind` format into `dir` that have not succeeded have recorded.
373fn attempts_text(dir: &Path, kind: &str) -> Option<String> {
374  std::fs::read_to_string(kind_file(dir, ATTEMPTS, kind)).ok()
375}

377/// The builds of the `kind` format into `dir` that have not succeeded in a row so far, and the last
378/// one's reason.
379fn previous_attempts(dir: &Path, kind: &str) -> (u32, String) {
380  let Some(text) = attempts_text(dir, kind) else {
381    return (0, String::new());
382  };
383  let (count, reason) = text.split_once('\t').unwrap_or((&text, ""));
384  let reason = if reason.is_empty() {
385    format!("the {kind} format build did not succeed")
386  } else {
387    reason.to_string()
388  };
389  (count.trim().parse().unwrap_or(1), reason)
390}

392/// Records that the `count`th build of the `kind` format into `dir` in a row has not succeeded
393/// (yet), and why.
394fn note_attempt(dir: &Path, kind: &str, count: u32, reason: &str) -> Result<(), String> {
395  let path = kind_file(dir, ATTEMPTS, kind);
396  std::fs::write(&path, format!("{count}\t{reason}"))
397    .map_err(|e| format!("cannot write {}: {e}", path.display()))
398}

400/// Marks the `kind` format as one that could not be built into `dir`, and says so. The mark holds
401/// the reason in place of the count, so removing it gives the format fresh attempts.
402fn mark_failed(dir: &Path, kind: &str, reason: &str) -> String {
403  let path = kind_file(dir, FAILED, kind);
404  if std::fs::write(&path, reason).is_ok() {
405    let _ = std::fs::remove_file(kind_file(dir, ATTEMPTS, kind));
406  }
407  failure_message(&path, reason)
408}

410/// The failure a [`FAILED`] mark at `path` records, with how to retry.
411fn failure_message(path: &Path, reason: &str) -> String {
412  format!(
413    "no dump matches this TeX tree and none could be built ({reason}; remove {} to retry)",
414    path.display()
415  )
416}

418/// Where dumps built here go: `$LATEXML_FORMAT_CACHE`, else the user's cache directory
419/// (`$XDG_CACHE_HOME`, `~/.cache`, when absolute), else the temp dir under the user's id — one
420/// directory per engine build and TeX tree, so a process reuses what an earlier one built. The build
421/// is the executable's size and modification time beside the version: the embedded git revision is
422/// the one `build.rs` last saw, and a dump is the engine's as much as the tree's. Absolute, as the
423/// build runs in it.
424fn cache_dir() -> Result<PathBuf, String> {
425  let absolute_var = |name: &str| {
426    std::env::var_os(name)
427      .map(PathBuf::from)
428      .filter(|dir| dir.is_absolute())
429  };
430  let base = std::env::var_os("LATEXML_FORMAT_CACHE")
431    .map(PathBuf::from)
432    .or_else(|| absolute_var("XDG_CACHE_HOME").map(|dir| dir.join("latexml-oxide/formats")))
433    .or_else(|| absolute_var("HOME").map(|dir| dir.join(".cache/latexml-oxide/formats")))
434    .unwrap_or_else(|| std::env::temp_dir().join(format!("latexml-oxide-formats-{}", user_id())));
435  let base = std::path::absolute(&base)
436    .map_err(|e| format!("cannot place the format cache {}: {e}", base.display()))?;
437  let tree: String = ["plain", "latex"]
438    .into_iter()
439    .filter_map(dump_paths::runtime_source_stamps)
440    .flatten()
441    .map(|stamp| format!("{:08x}", stamp.checksum))
442    .collect::<Vec<_>>()
443    .join("-");
444  let build = std::env::current_exe()
445    .and_then(std::fs::metadata)
446    .map(|meta| {
447      let modified = meta
448        .modified()
449        .ok()
450        .and_then(|time| time.duration_since(std::time::UNIX_EPOCH).ok())
451        .map_or(0, |since| since.as_nanos());
452      format!("{:x}-{modified:x}", meta.len())
453    })
454    .unwrap_or_default();
455  Ok(base.join(format!("{}-{build}-{tree}", crate::identity::VERSION)))
456}

458/// The cache's `kind` dump for this tree and its year, if this user built it: a dump is loaded as
459/// the format, so one in a directory or file another user could have written is not used.
460fn cached_dump(dir: &Path, kind: &str, year: u32) -> Option<(PathBuf, u32)> {
461  if !trusted_dir(dir) {
462    return None;
463  }
464  dump_paths::resolve_versioned_in_dir(dir, kind, Some(year))
465    .filter(|(path, _)| private_to_user(path))
466}

468/// The failure an earlier build of the `kind` format into `dir` recorded, if its mark has not lapsed
469/// ([`FAILED_LAPSE`]); a lapsed mark is removed (no count is kept beside a mark, see `mark_failed`).
470fn failed_before(dir: &Path, kind: &str) -> Option<String> {
471  let path = kind_file(dir, FAILED, kind);
472  let reason = std::fs::read_to_string(&path).ok()?;
473  let lapsed = std::fs::metadata(&path)
474    .and_then(|meta| meta.modified())
475    .ok()
476    .and_then(|marked| marked.elapsed().ok())
477    .is_some_and(|age| age > FAILED_LAPSE);
478  if lapsed {
479    let _ = std::fs::remove_file(&path);
480    return None;
481  }
482  Some(failure_message(&path, &reason))
483}

485/// Takes `dir`'s build lock, waiting up to [`LOCK_WAIT`] for a process building there.
486fn lock_dir(dir: &Path) -> Result<std::fs::File, String> {
487  let file = std::fs::File::create(dir.join(".lock"))
488    .map_err(|e| format!("cannot lock {}: {e}", dir.display()))?;
489  let deadline = Instant::now() + LOCK_WAIT;
490  loop {
491    match file.try_lock() {
492      Ok(()) => return Ok(file),
493      Err(std::fs::TryLockError::WouldBlock) if Instant::now() < deadline => {
494        std::thread::sleep(Duration::from_millis(100));
495      },
496      Err(std::fs::TryLockError::WouldBlock) => {
497        return Err(format!(
498          "another process has been building into {} for {} s",
499          dir.display(),
500          LOCK_WAIT.as_secs()
501        ));
502      },
503      Err(std::fs::TryLockError::Error(e)) => {
504        return Err(format!("cannot lock {}: {e}", dir.display()));
505      },
506    }
507  }
508}

519/// Removes the partial dumps killed builds left in `dir`.
520fn remove_partial_dumps(dir: &Path) {
521  let Ok(entries) = std::fs::read_dir(dir) else {
522    return;
523  };
524  for entry in entries.flatten() {
525    if entry.file_name().to_string_lossy().ends_with(".partial") {
526      let _ = std::fs::remove_file(entry.path());
527    }
528  }

531/// Builds the `kind` dump into `dir` as `--init` does, on a thread of its own: the engine's state is
532/// per thread, so the build starts from a fresh one, leaves the caller's untouched, and is released
533/// when it ends. Like fmtutil, it runs in a directory holding no TeX files — `dir` — as kpathsea
534/// searches `.` first and a document's own copy of a format file (`./expl3-code.tex`) must not go
535/// into the format; the working directory is the process's, so it is `dir` for every thread until
536/// the build ends, whatever the outcome. As `tools/make_formats.sh` does, a build that logs an
537/// error or a fatal one is not kept. Written beside its final name and renamed, so a concurrent
538/// reader never sees half a dump.
539fn build(kind: &'static str, year: u32, dir: &Path) -> Result<(), String> {
540  let dest = dir.join(dump_paths::dump_filename(kind, year));
541  let partial = partial_dump(dir, kind, year);
542  let partial_name = partial.to_string_lossy().into_owned();
543  let init_file = if kind == "latex" {
544    "latex.ltx"
545  } else {
546    "plain.tex"
547  };
548  let previous = std::env::current_dir().map_err(|e| format!("no working directory: {e}"))?;
549  let workdir = dir.to_path_buf();
550  let joined = std::thread::Builder::new()
551    .name(format!("{kind}-format"))
552    .stack_size(256 * 1024 * 1024)
553    .spawn(move || -> Result<(), String> {
554      dump_paths::set_building_format(true);
555      std::env::set_current_dir(&workdir)
556        .map_err(|e| format!("cannot enter {}: {e}", workdir.display()))?;
557      let opts = latexml_core::common::Config {
558        bindings_dispatch: Some(std::rc::Rc::new(latexml_package::dispatch)),
559        extra_bindings_dispatch: Some(std::rc::Rc::new(latexml_contrib::dispatch)),
560        ..latexml_core::common::Config::default()
561      };
562      let mut converter = crate::converter::Converter::from_config(opts.clone());
563      let result = converter
564        .prepare_session(&opts)
565        .map_err(|e| format!("cannot prepare the {kind} format build: {e}"))
566        .and_then(|()| crate::ini_tex::dump_format(&mut converter, init_file, Some(&partial_name)))
567        .and_then(|_| {
568          let counts = latexml_core::common::error::snapshot_report_counts();
569          if counts.fatal || counts.error > 0 {
570            Err(format!(
571              "building the {kind} format logged {} error(s){}",
572              counts.error,
573              if counts.fatal { " and a fatal one" } else { "" }
574            ))
575          } else {
576            Ok(())
577          }
578        });
579      drop(converter);
580      latexml_core::reset_thread_engine();
581      result
582    })
583    .map_err(|e| format!("cannot start the {kind} format build: {e}"))?
584    .join();
585  let returned = std::env::set_current_dir(&previous);
586  let built = joined
587    .map_err(|_| format!("the {kind} format build panicked"))
588    .and_then(|built| built)
589    .and_then(|()| returned.map_err(|e| format!("cannot return to {}: {e}", previous.display())))
590    .and_then(|()| publish(kind, &partial, &dest));
591  if built.is_err() {
592    let _ = std::fs::remove_file(&partial);
593  }
594  built
595}
596
597/// Publishes the finished `kind` dump at `partial` as `dest`. The build's [`IN_FLIGHT`] record is
598/// cleared under the same lock as the rename: an interrupt in between would put the attempt note
599/// from before the build back beside the finished dump; one now waits, and finds nothing to undo.
600fn publish(kind: &str, partial: &Path, dest: &Path) -> Result<(), String> {
601  publish_read_only(partial)
602    .map_err(|e| format!("cannot set the {kind} dump's permissions: {e}"))?;
603  let mut slot = IN_FLIGHT.lock().unwrap_or_else(PoisonError::into_inner);
604  std::fs::rename(partial, dest)
605    .map_err(|e| format!("cannot move the {kind} dump to {}: {e}", dest.display()))?;
606  *slot = None;
607  Ok(())
608}

610#[cfg(test)]
611mod tests {
612  use super::*;
613
614  /// The tests share the process's [`IN_FLIGHT`] record: one at a time under `cargo test`.
615  static SERIAL: Mutex<()> = Mutex::new(());
616
617  /// An interrupted build is undone (issue 988): its attempt note goes back to what it held before
618  /// the build (none, or the count of earlier builds that did not succeed) and its partial dump is
619  /// removed, so the next process builds the format again instead of counting the interrupted build
620  /// towards marking it failed.
621  #[test]
622  fn an_interrupted_build_is_undone() {
623    let _serial = SERIAL.lock().unwrap_or_else(PoisonError::into_inner);
624    let dir = tempfile::tempdir().expect("create tempdir");
625    let dir = dir.path();
626    for before in [None, Some("1\tbuilding the latex format logged 3 error(s)")] {
627      if let Some(text) = before {
628        std::fs::write(kind_file(dir, ATTEMPTS, "latex"), text).expect("write the earlier note");
629      }
630      let count = previous_attempts(dir, "latex").0 + 1;
631      let before_build = attempts_text(dir, "latex");
632      note_attempt(
633        dir,
634        "latex",
635        count,
636        "the latex format build died with its process",
637      )
638      .expect("note the attempt");
639      let partial = partial_dump(dir, "latex", 2025);
640      std::fs::write(&partial, "half a dump").expect("write the partial dump");
641      set_in_flight(Some(InFlight {
642        attempts: kind_file(dir, ATTEMPTS, "latex"),
643        before:   before_build,
644        partial:  partial.clone(),
645      }));
646      assert!(cancel_in_flight_build());
647      assert_eq!(attempts_text(dir, "latex").as_deref(), before);
648      assert!(!partial.exists(), "the partial dump stayed");
649      assert!(!cancel_in_flight_build(), "a build is undone once");
650    }
651  }
652
653  /// A published dump is no longer the build's to undo: an interrupt after the rename used to put
654  /// the attempt note from before the build back beside the finished dump (CLIHARD 1b).
655  #[test]
656  fn a_published_build_is_not_undone() {
657    let _serial = SERIAL.lock().unwrap_or_else(PoisonError::into_inner);
658    let dir = tempfile::tempdir().expect("create tempdir");
659    let dir = dir.path();
660    let before = "1\tbuilding the latex format logged 3 error(s)";
661    std::fs::write(kind_file(dir, ATTEMPTS, "latex"), before).expect("write the earlier note");
662    note_attempt(
663      dir,
664      "latex",
665      2,
666      "the latex format build died with its process",
667    )
668    .expect("note the attempt");
669    let partial = partial_dump(dir, "latex", 2025);
670    std::fs::write(&partial, "a dump").expect("write the dump");
671    set_in_flight(Some(InFlight {
672      attempts: kind_file(dir, ATTEMPTS, "latex"),
673      before:   Some(before.to_string()),
674      partial:  partial.clone(),
675    }));
676    let dest = dir.join(dump_paths::dump_filename("latex", 2025));
677    publish("latex", &partial, &dest).expect("publish");
678    assert!(!cancel_in_flight_build(), "the published build was undone");
679    assert_eq!(
680      attempts_text(dir, "latex").as_deref(),
681      Some("2\tthe latex format build died with its process")
682    );
683    assert!(dest.is_file());
684  }
