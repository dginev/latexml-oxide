1//! A format dump for the TeX tree this process reads.
2//!
3//! A dump is the format: the state `latex.ltx` (and the L3 kernel it loads) or `plain.tex` leaves,
4//! built from one TeX tree's files. Packages come raw from the tree the process reads, so a dump
5//! built from another tree is the wrong format for them — `expl3.sty` refuses a format whose L3
6//! kernel date differs from its own ("Mismatched LaTeX support files", expl3.sty:64-78). The 2609
7//! release ran a dump built from `/usr/local/texlive/2025` with `/usr/share/texlive`'s packages, and
8//! 12,144 papers ended as Error (witnesses 2607.28725, 2609.40175).
9//!
10//! Every dump records the files it was built from, and the loaders pass over one this tree does not
11//! have (`latexml_engine::dump_paths::dump_matches_tree`). When no dump found matches, the format
12//! is built here once, as `--init` and fmtutil build it, into a per-user cache keyed by the engine
13//! build and the tree's files, and that dump is used. When it cannot be built, the format loads
14//! from the engine's own definitions, `LoadFormat`'s branch for a missing dump (Perl
15//! Package.pm:2755-2766).
16

28/// Marks, per format (`.failed.<kind>`), a cache directory where it could not be built, holding the
29/// reason: later processes take the base branch at once instead of building again. The mark lapses
30/// after [`FAILED_LAPSE`]; removing it retries at once.
31const FAILED: &str = ".failed";
32
33/// Counts, per format (`.attempts.<kind>`), the builds into a cache directory that have not
34/// succeeded in a row, with the last one's reason: written before a build, as one that dies with
35/// its process (killed, or a panic in a binary built to abort) records nothing after; removed by a
36/// success, or replaced by the [`FAILED`] mark.
37const ATTEMPTS: &str = ".attempts";
38
39/// Builds of a format that may fail in a row before it is marked [`FAILED`]: a format that does not
40/// build is not tried by every process, and one passing failure (a full disk, no thread) is not
41/// taken for it.
42const MAX_ATTEMPTS: u32 = 2;
43
44/// How long a [`FAILED`] mark holds: a failure that passes (the disk is cleared, the tree repaired)
45/// is not kept for good.
46const FAILED_LAPSE: Duration = Duration::from_secs(24 * 60 * 60);
47
48/// How long a process waits for another one building into the same directory — far longer than a
49/// build takes, even on a loaded host.
50const LOCK_WAIT: Duration = Duration::from_secs(300);
51
52/// Ensures each format has a dump built from this process's TeX tree, building any that has none
53/// into the cache. Once per process, before the first session loads a format; nothing to do without
54/// a TeX tree, under `LATEXML_NODUMP`, or while a format is being built (`--init`, or the thread
55/// building one here). A format that cannot be built is reported once, on stderr (and as a warning
56/// in each conversion's log, `converter::note_format_build_failure`), and loads from the engine's
57/// definitions. A build changes the process's working directory while it runs (see `build`), so it
58/// belongs before the process converts anything.
59pub fn ensure_format_dumps() {
60  if dump_paths::building_format() {
61    return;
62  }
63  static ENSURED: OnceLock<()> = OnceLock::new();
64  ENSURED.get_or_init(|| {
65    if let Err(reason) = ensure() {
66      eprintln!(
67        "[format_dumps] {reason}: the format loads from the engine's own definitions instead of a dump"
68      );
69      dump_paths::set_format_build_failure(reason);
70    }
71  });
72}

74fn ensure() -> Result<(), String> {
75  if std::env::var_os("LATEXML_NODUMP").is_some() || std::env::var_os("LATEXML_INI_MODE").is_some()
76  {
77    return Ok(());
78  }
79  let kinds: Vec<&'static str> = ["plain", "latex"]
80    .into_iter()
81    .filter(|kind| dump_paths::runtime_source_stamps(kind).is_some())
82    .collect();
83  if kinds.is_empty() {
84    return Ok(());
85  }
86  // A dump the configuration names, installs or keeps beside the sources is used before the cache's;
87  // the embedded one only after it, as reading it back costs every process a 4 MB dump.
88  let wanted: Vec<&'static str> = kinds
89    .into_iter()
90    .filter(|kind| !dump_paths::format_dump_on_disk(kind))
91    .collect();
92  if wanted.is_empty() {
93    return Ok(());
94  }
95  // Only the dump's file name carries the year.
96  let year = dump_paths::detect_ambient_texlive_year().unwrap_or(2000);
97  let dir = match cache_dir() {
98    Ok(dir) => dir,
99    Err(_)
100      if wanted
101        .iter()
102        .all(|kind| dump_paths::format_dump_available(kind)) =>
103    {
104      return Ok(());
105    },
106    Err(reason) => return Err(reason),
107  };
108  // Each kind uses the cache's dump on its own merits.
109  let use_cached = |kind: &str| match cached_dump(&dir, kind, year) {
110    Some((path, found_year)) => {
111      dump_paths::set_rebuilt_dump(kind, path, found_year);
112      true
113    },
114    None => false,
115  };
116  let missing: Vec<&'static str> = wanted
117    .into_iter()
118    .filter(|kind| !use_cached(kind) && !dump_paths::format_dump_available(kind))
119    .collect();
120  if missing.is_empty() {
121    return Ok(());
122  }
123  private_dir(&dir)?;
124  let mut reasons = Vec::new();
125  let missing: Vec<&'static str> = missing
126    .into_iter()
127    .filter(|kind| match failed_before(&dir, kind) {
128      Some(reason) => {
129        reasons.push(reason);
130        false
131      },
132      None => true,
133    })
134    .collect();
135  if !missing.is_empty() {
136    // What the builds so far have left, to tell whether one failed while this process waited. (A
137    // build that fails between this read and its release of the lock goes unseen: the window is
138    // the few instructions between its last write and its unlock.)
139    let seen: Vec<Option<String>> = missing
140      .iter()
141      .map(|kind| attempts_text(&dir, kind))
142      .collect();
143    // One process builds; the others wait for it and find the dump, or its failure.
144    let _lock = match lock_dir(&dir) {
145      Ok(lock) => lock,
146      Err(reason) => {
147        reasons.push(reason);
148        return Err(reasons.join("; "));
149      },
150    };
151    // No other process builds here while this one holds the lock: a partial dump is a killed
152    // build's.
153    remove_partial_dumps(&dir);
154    for (kind, seen) in missing.into_iter().zip(seen) {
155      // Built by the process this one waited for.
156      if use_cached(kind) {
157        continue;
158      }
159      if let Some(reason) = failed_before(&dir, kind) {
160        reasons.push(reason);
161        continue;
162      }
163      // A build that failed while this process waited is not tried again at once: a passing
164      // failure would likely recur, and a later process retries it.
165      let now = attempts_text(&dir, kind);
166      if now.is_some() && now != seen {
167        reasons.push(previous_attempts(&dir, kind).1);
168        continue;
169      }
170      if let Err(reason) = attempt(kind, year, &dir, &use_cached) {
171        reasons.push(reason);
172      }
173    }
174  }
175  if reasons.is_empty() {
176    Ok(())
177  } else {
178    Err(reasons.join("; "))
179  }
180}

182/// Builds the `kind` dump into `dir`, counting it until it succeeds and marking the format
183/// [`FAILED`] after [`MAX_ATTEMPTS`] builds in a row that did not.
184fn attempt(
185  kind: &'static str,
186  year: u32,
187  dir: &Path,
188  use_cached: &dyn Fn(&str) -> bool,
189) -> Result<(), String> {
190  let (failed, last_reason) = previous_attempts(dir, kind);
191  if failed >= MAX_ATTEMPTS {
192    return Err(mark_failed(dir, kind, &last_reason));
193  }
194  note_attempt(
195    dir,
196    kind,
197    failed + 1,
198    &format!("the {kind} format build died with its process"),
199  )?;
200  eprintln!(
201    "[format_dumps] no {kind} dump matches this TeX tree; building one into {}",
202    dir.display()
203  );
204  let built = build(kind, year, dir).and_then(|()| {
205    if use_cached(kind) {
206      Ok(())
207    } else {
208      Err(format!(
209        "the {kind} dump built into {} does not match this TeX tree",
210        dir.display()
211      ))
212    }
213  });
214  match built {
215    Ok(()) => {
216      let _ = std::fs::remove_file(kind_file(dir, ATTEMPTS, kind));
217      Ok(())
218    },
219    Err(reason) => {
220      let _ = note_attempt(dir, kind, failed + 1, &reason);
221      if failed + 1 >= MAX_ATTEMPTS {
222        Err(mark_failed(dir, kind, &reason))
223      } else {
224        Err(reason)
225      }
226    },
227  }
228}

230/// `dir`'s `marker` file (`.attempts`, `.failed`) for the `kind` format.
231fn kind_file(dir: &Path, marker: &str, kind: &str) -> PathBuf {
232  dir.join(format!("{marker}.{kind}"))
233}

235/// What the builds of the `kind` format into `dir` that have not succeeded have recorded.
236fn attempts_text(dir: &Path, kind: &str) -> Option<String> {
237  std::fs::read_to_string(kind_file(dir, ATTEMPTS, kind)).ok()
238}

240/// The builds of the `kind` format into `dir` that have not succeeded in a row so far, and the last
241/// one's reason.
242fn previous_attempts(dir: &Path, kind: &str) -> (u32, String) {
243  let Some(text) = attempts_text(dir, kind) else {
244    return (0, String::new());
245  };
246  let (count, reason) = text.split_once('\t').unwrap_or((&text, ""));
247  let reason = if reason.is_empty() {
248    format!("the {kind} format build did not succeed")
249  } else {
250    reason.to_string()
251  };
252  (count.trim().parse().unwrap_or(1), reason)
253}

255/// Records that the `count`th build of the `kind` format into `dir` in a row has not succeeded
256/// (yet), and why.
257fn note_attempt(dir: &Path, kind: &str, count: u32, reason: &str) -> Result<(), String> {
258  let path = kind_file(dir, ATTEMPTS, kind);
259  std::fs::write(&path, format!("{count}\t{reason}"))
260    .map_err(|e| format!("cannot write {}: {e}", path.display()))
261}

263/// Marks the `kind` format as one that could not be built into `dir`, and says so. The mark holds
264/// the reason in place of the count, so removing it gives the format fresh attempts.
265fn mark_failed(dir: &Path, kind: &str, reason: &str) -> String {
266  let path = kind_file(dir, FAILED, kind);
267  if std::fs::write(&path, reason).is_ok() {
268    let _ = std::fs::remove_file(kind_file(dir, ATTEMPTS, kind));
269  }
270  failure_message(&path, reason)
271}

273/// The failure a [`FAILED`] mark at `path` records, with how to retry.
274fn failure_message(path: &Path, reason: &str) -> String {
275  format!(
276    "no dump matches this TeX tree and none could be built ({reason}; remove {} to retry)",
277    path.display()
278  )
279}

281/// Where dumps built here go: `$LATEXML_FORMAT_CACHE`, else the user's cache directory
282/// (`$XDG_CACHE_HOME`, `~/.cache`, when absolute), else the temp dir under the user's id — one
283/// directory per engine build and TeX tree, so a process reuses what an earlier one built. The build
284/// is the executable's size and modification time beside the version: the embedded git revision is
285/// the one `build.rs` last saw, and a dump is the engine's as much as the tree's. Absolute, as the
286/// build runs in it.
287fn cache_dir() -> Result<PathBuf, String> {
288  let absolute_var = |name: &str| {
289    std::env::var_os(name)
290      .map(PathBuf::from)
291      .filter(|dir| dir.is_absolute())
292  };
293  let base = std::env::var_os("LATEXML_FORMAT_CACHE")
294    .map(PathBuf::from)
295    .or_else(|| absolute_var("XDG_CACHE_HOME").map(|dir| dir.join("latexml-oxide/formats")))
296    .or_else(|| absolute_var("HOME").map(|dir| dir.join(".cache/latexml-oxide/formats")))
297    .unwrap_or_else(|| std::env::temp_dir().join(format!("latexml-oxide-formats-{}", user_id())));
298  let base = std::path::absolute(&base)
299    .map_err(|e| format!("cannot place the format cache {}: {e}", base.display()))?;
300  let tree: String = ["plain", "latex"]
301    .into_iter()
302    .filter_map(dump_paths::runtime_source_stamps)
303    .flatten()
304    .map(|stamp| format!("{:08x}", stamp.checksum))
305    .collect::<Vec<_>>()
306    .join("-");
307  let build = std::env::current_exe()
308    .and_then(std::fs::metadata)
309    .map(|meta| {
310      let modified = meta
311        .modified()
312        .ok()
313        .and_then(|time| time.duration_since(std::time::UNIX_EPOCH).ok())
314        .map_or(0, |since| since.as_nanos());
315      format!("{:x}-{modified:x}", meta.len())
316    })
317    .unwrap_or_default();
318  Ok(base.join(format!("{}-{build}-{tree}", crate::identity::VERSION)))
319}

321/// The cache's `kind` dump for this tree and its year, if this user built it: a dump is loaded as
322/// the format, so one in a directory or file another user could have written is not used.
323fn cached_dump(dir: &Path, kind: &str, year: u32) -> Option<(PathBuf, u32)> {
324  if !trusted_dir(dir) {
325    return None;
326  }
327  dump_paths::resolve_versioned_in_dir(dir, kind, Some(year))
328    .filter(|(path, _)| private_to_user(path))
329}

331/// The failure an earlier build of the `kind` format into `dir` recorded, if its mark has not lapsed
332/// ([`FAILED_LAPSE`]); a lapsed mark is removed (no count is kept beside a mark, see `mark_failed`).
333fn failed_before(dir: &Path, kind: &str) -> Option<String> {
334  let path = kind_file(dir, FAILED, kind);
335  let reason = std::fs::read_to_string(&path).ok()?;
336  let lapsed = std::fs::metadata(&path)
337    .and_then(|meta| meta.modified())
338    .ok()
339    .and_then(|marked| marked.elapsed().ok())
340    .is_some_and(|age| age > FAILED_LAPSE);
341  if lapsed {
342    let _ = std::fs::remove_file(&path);
343    return None;
344  }
345  Some(failure_message(&path, &reason))
346}

348/// Takes `dir`'s build lock, waiting up to [`LOCK_WAIT`] for a process building there.
349fn lock_dir(dir: &Path) -> Result<std::fs::File, String> {
350  let file = std::fs::File::create(dir.join(".lock"))
351    .map_err(|e| format!("cannot lock {}: {e}", dir.display()))?;
352  let deadline = Instant::now() + LOCK_WAIT;
353  loop {
354    match file.try_lock() {
355      Ok(()) => return Ok(file),
356      Err(std::fs::TryLockError::WouldBlock) if Instant::now() < deadline => {
357        std::thread::sleep(Duration::from_millis(100));
358      },
359      Err(std::fs::TryLockError::WouldBlock) => {
360        return Err(format!(
361          "another process has been building into {} for {} s",
362          dir.display(),
363          LOCK_WAIT.as_secs()
364        ));
365      },
366      Err(std::fs::TryLockError::Error(e)) => {
367        return Err(format!("cannot lock {}: {e}", dir.display()));
368      },
369    }
370  }
371}

373/// Removes the partial dumps killed builds left in `dir`.
374fn remove_partial_dumps(dir: &Path) {
375  let Ok(entries) = std::fs::read_dir(dir) else {
376    return;
377  };
378  for entry in entries.flatten() {
379    if entry.file_name().to_string_lossy().ends_with(".partial") {
380      let _ = std::fs::remove_file(entry.path());
381    }
382  }
383}

385/// Builds the `kind` dump into `dir` as `--init` does, on a thread of its own: the engine's state is
386/// per thread, so the build starts from a fresh one, leaves the caller's untouched, and is released
387/// when it ends. Like fmtutil, it runs in a directory holding no TeX files — `dir` — as kpathsea
388/// searches `.` first and a document's own copy of a format file (`./expl3-code.tex`) must not go
389/// into the format; the working directory is the process's, so it is `dir` for every thread until
390/// the build ends, whatever the outcome. As `tools/make_formats.sh` does, a build that logs an
391/// error or a fatal one is not kept. Written beside its final name and renamed, so a concurrent
392/// reader never sees half a dump.
393fn build(kind: &'static str, year: u32, dir: &Path) -> Result<(), String> {
394  let dest = dir.join(dump_paths::dump_filename(kind, year));
395  let partial = dir.join(format!(
396    ".{}.{}.partial",
397    dump_paths::dump_filename(kind, year),
398    std::process::id()
399  ));
400  let partial_name = partial.to_string_lossy().into_owned();
401  let init_file = if kind == "latex" {
402    "latex.ltx"
403  } else {
404    "plain.tex"
405  };
406  let previous = std::env::current_dir().map_err(|e| format!("no working directory: {e}"))?;
407  let workdir = dir.to_path_buf();
408  let joined = std::thread::Builder::new()
409    .name(format!("{kind}-format"))
410    .stack_size(256 * 1024 * 1024)
411    .spawn(move || -> Result<(), String> {
412      dump_paths::set_building_format(true);
413      std::env::set_current_dir(&workdir)
414        .map_err(|e| format!("cannot enter {}: {e}", workdir.display()))?;
415      let opts = latexml_core::common::Config {
416        bindings_dispatch: Some(std::rc::Rc::new(latexml_package::dispatch)),
417        extra_bindings_dispatch: Some(std::rc::Rc::new(latexml_contrib::dispatch)),
418        ..latexml_core::common::Config::default()
419      };
420      let mut converter = crate::converter::Converter::from_config(opts.clone());
421      let result = converter
422        .prepare_session(&opts)
423        .map_err(|e| format!("cannot prepare the {kind} format build: {e}"))
424        .and_then(|()| crate::ini_tex::dump_format(&mut converter, init_file, Some(&partial_name)))
425        .and_then(|_| {
426          let counts = latexml_core::common::error::snapshot_report_counts();
427          if counts.fatal || counts.error > 0 {
428            Err(format!(
429              "building the {kind} format logged {} error(s){}",
430              counts.error,
431              if counts.fatal { " and a fatal one" } else { "" }
432            ))
433          } else {
434            Ok(())
435          }
436        });
437      drop(converter);
438      latexml_core::reset_thread_engine();
439      result
440    })
441    .map_err(|e| format!("cannot start the {kind} format build: {e}"))?
442    .join();
443  let returned = std::env::set_current_dir(&previous);
444  let built = joined
445    .map_err(|_| format!("the {kind} format build panicked"))
446    .and_then(|built| built)
447    .and_then(|()| returned.map_err(|e| format!("cannot return to {}: {e}", previous.display())))
448    .and_then(|()| {
449      publish_read_only(&partial)
450        .map_err(|e| format!("cannot set the {kind} dump's permissions: {e}"))?;
451      std::fs::rename(&partial, &dest)
452        .map_err(|e| format!("cannot move the {kind} dump to {}: {e}", dest.display()))
453    });
454  if built.is_err() {
455    let _ = std::fs::remove_file(&partial);
456  }
457  built
