1//! Post-processing pipeline API.
2//!
3//! Provides a public interface to the LaTeXML post-processing pipeline
4//! (Scan → Bibliography → CrossRef → Graphics → Split → MathML → XSLT → HTML5 fixups).
5//! Used by both the `latexml_oxide` binary and the `cortex_worker` binary.

20// Process-once cached env var (see WISDOM #56 — getenv hot-path race).
21static POST_AUDIT: Lazy<bool> = Lazy::new(|| std::env::var("LATEXML_POST_AUDIT").is_ok());

23/// Options for the post-processing pipeline.
24pub struct PostOptions<'a> {
25  pub pmml:                      bool,
26  pub cmml:                      bool,
27  pub keep_xmath:                bool,
28  pub stylesheet:                Option<&'a str>,
29  pub destination:               Option<&'a str>,
30  /// Base directory of the original LaTeX source (`--sourcedirectory`; Perl
31  /// Config.pm L147, passed to Post as `sourceDirectory` in LaTeXML.pm L429).
32  /// Used to locate graphics/resources the post-phase copies. When `None`,
33  /// the caller defaults it to the source file's own directory.
34  pub source_directory:          Option<&'a str>,
35  /// Root directory of the generated site (`--sitedirectory`; Perl Config.pm
36  /// L146, passed to Post as `siteDirectory` in LaTeXML.pm L430). Establishes
37  /// the base against which cross-document/resource URLs are made relative.
38  /// When `None`, Post defaults it to the destination's directory (Perl
39  /// Config.pm L466-469; `document.rs` mirrors that fallback).
40  pub site_directory:            Option<&'a str>,
41  /// Extra resource search paths (the `--path` flag): directories searched
42  /// for `--css`/`--javascript` files (and other post resources) to copy into
43  /// the destination, in addition to the document's own paths, the current
44  /// directory, and the binary's embedded resource table.
45  pub search_paths:              &'a [String],
46  pub nodefaultresources:        bool,
47  pub css_files:                 &'a [String],
48  pub js_files:                  &'a [String],
49  pub noinvisibletimes:          bool,
50  /// Remap styled alphanumerics to Unicode Plane-1 (Perl `--plane1`, default on).
51  pub plane1:                    bool,
52  /// Remap only the poorly-supported variants, to their simpler form
53  /// (Perl `--hackplane1`); implies `plane1`.
54  pub hackplane1:                bool,
55  pub mathtex:                   bool,
56  pub navigationtoc:             Option<&'a str>,
57  pub schemadocs:                bool,
58  pub split:                     bool,
59  pub split_xpath:               Option<String>,
60  pub split_naming:              Option<&'a str>,
61  pub xslt_parameters:           &'a [String],
62  /// If > 0, try the vector-SVG converters (mutool → pdftocairo) for PDF
63  /// graphics smaller than this many KB (vector-preservation path). Fall
64  /// back to the raster ImageMagick `convert`/`gs` → PNG path on failure
65  /// or timeout. Tracks upstream brucemiller/LaTeXML#902.
66  pub graphics_svg_threshold_kb: u32,
67  /// Convert `\includegraphics` figures to web images (Perl `graphicimages!`
68  /// → `dographics`, default on). When `false` (`--nographicimages`) the
69  /// Graphics post-phase is skipped entirely: the output keeps the raw
70  /// `<ltx:graphics>` references untouched — useful for faster runs or hosts
71  /// without the image tools.
72  pub graphicimages:             bool,
73  /// Timestamp string embedded in the generated page (Perl `--timestamp`,
74  /// XSLT `TIMESTAMP` param). `None` → no timestamp emitted (the deterministic
75  /// default; Perl instead defaults to the current time). Perl's
76  /// `--timestamp=0` "omit" case is normalized to `None` by the caller.
77  pub timestamp:                 Option<&'a str>,
78  /// Favicon resource for the generated site (Perl `--icon`, XSLT `ICON`
79  /// param): emitted as a `<link rel="icon">` and copied to the destination.
80  pub icon:                      Option<&'a str>,
81  /// Output extraction mode (Perl `LaTeXML::Util::Pack::whatsout`).
82  /// `Document` (default) → serialize the full post-processed
83  /// document; `Fragment` → embeddable HTML snippet via
84  /// `latexml_post::extract::get_embeddable`; `Math` → math subtree
85  /// via `get_math`. Applied to each `PostDocument` in the final
86  /// serialization loop.
87  pub whatsout:                  latexml_post::extract::Whatsout,
88}

90/// The built-in XSLT stylesheet for an output format (logical embedded-resource
91/// name; Perl `Config.pm` L543-551). Returns `None` for a format with no
92/// default (e.g. `xml`), so a caller keeps `--stylesheet` / no post.
93///
94/// SINGLE SOURCE OF TRUTH shared by the CLI (`bin/latexml_oxide.rs` picks
95/// `effective_stylesheet` when `--stylesheet` is unset) and the library
96/// (`crate::api::convert_to_html`), so the two can never disagree on which
97/// sheet an `html5`/`xhtml`/`epub` job gets.
98pub fn default_stylesheet(format: Option<&str>) -> Option<&'static str> {
99  match format {
100    Some("html5") => Some("resources/XSLT/LaTeXML-html5.xsl"),
101    Some("html") | Some("xhtml") => Some("resources/XSLT/LaTeXML-all-xhtml.xsl"),
102    Some("epub") | Some("epub3") => Some("resources/XSLT/LaTeXML-epub3.xsl"),
103    _ => None,
104  }
105}

107/// Emit a post-processing error: capture it into the active log buffer (so it
108/// reaches `cortex.log`) AND bump the Error counter so `Status:conversion`
109/// reflects post-phase failures (Perl `LaTeXML.pm` L633: `max(core, post)`).
110///
111/// Deliberately NOT the `Error!` macro: that expands to a too-many-errors
112/// `Fatal!(…)` → `return Err(error::Error)`, which only typechecks inside the
113/// digester's `Result<_, error::Error>` functions. The post-processing entry
114/// points return `String`/`Result<_, ()>`, so we log+count directly with no
115/// control-flow side effect.
116fn post_error(object: &str, message: &str) {
117  note_status(LogStatus::Error, None);
118  log::error!(target: &format!("post:{object}"), "{message}");
119}

121/// Result of [`run_post_processing_logged`]: the serialized HTML plus the
122/// post-phase log text and status code, mirroring Perl `LaTeXML.pm`'s
123/// `convert_post`, which flushes the log AFTER post and folds the post status
124/// into the final `max(core, post)` verdict (L631-634).
125pub struct PostOutcome {
126  /// Serialized post-processed output (same value as [`run_post_processing`]).
127  pub html:        String,
128  /// Log text captured during the post phase only — Graphics/MathML/XSLT
129  /// `Info!`/`Warn!`/`post_error` lines. Append to the core conversion log.
130  pub log:         String,
131  /// Status code read from the shared REPORT counter AFTER post. Because the
132  /// counter is NOT reset between core and post, this is already the combined
133  /// core+post code; callers still `max()` it with the core code as a
134  /// belt-and-suspenders guard for paths that force a fatal outside REPORT
135  /// (e.g. a `catch_unwind`-trapped panic).
136  pub status_code: usize,
137}

139/// Run post-processing while capturing its log into a fresh thread-local
140/// LOG_BUFFER. `Converter::convert()` already flushed the *core* log into its
141/// own response, leaving the buffer unbound — so without re-binding here every
142/// post-phase `Info!`/`Warn!`/`post_error` (incl. a silent EPS→PNG failure)
143/// reaches stderr only and never the persisted `--log`/`cortex.log`. Perl's
144/// single `flush_log()` after `convert_post` sweeps up the post log for ALL
145/// consumers; this is the Rust equivalent, shared by the `latexml_oxide` and
146/// `cortex_worker` binaries so their logs reach parity. See SYNC_STATUS task 5.
147pub fn run_post_processing_logged(xml: &str, opts: &PostOptions) -> PostOutcome {
148  run_post_processing_impl_logged(PostInput::Xml(xml), opts)
149}
150
151/// File-input variant of [`run_post_processing_logged`]: parses the XML from
152/// disk via libxml2's streaming reader rather than from an in-memory `String`.
153/// Used for already-converted `.xml` inputs, where slurping the file would keep
154/// a whole extra copy of a multi-hundred-MB document resident alongside the DOM.
155pub fn run_post_processing_from_file_logged(path: &str, opts: &PostOptions) -> PostOutcome {
156  run_post_processing_impl_logged(PostInput::File(path), opts)
157}
158
159fn run_post_processing_impl_logged(input: PostInput, opts: &PostOptions) -> PostOutcome {
160  latexml_core::util::logger::bind_log();
161  let html = run_post_processing_impl(input, opts);
162  let log = latexml_core::util::logger::flush_log();
163  let status_code = latexml_core::common::error::get_status_code();
164  PostOutcome { html, log, status_code }
165}

167/// Where the post-pipeline reads its already-converted LaTeXML XML from.
168#[derive(Clone, Copy)]
169enum PostInput<'a> {
170  /// XML already resident in memory (the TeX→XML conversion result). The common
171  /// path — the 2.8M-doc arXiv fleet and every TeX conversion feed this.
172  Xml(&'a str),
173  /// A path to an XML file, parsed via libxml2's streaming file reader
174  /// (`xmlReadIO`). Avoids vivifying a very large document as a Rust `String`
175  /// on top of the ~11× DOM it becomes.
176  File(&'a str),
177}

179/// Run the post-processing pipeline on in-memory XML output.
180///
181/// Executes: Split → Scan → MakeBibliography → CrossRef → Graphics → MathML → XSLT → HTML5 fixups.
182pub fn run_post_processing(xml: &str, opts: &PostOptions) -> String {
183  run_post_processing_impl(PostInput::Xml(xml), opts)
184}

188  run_post_processing_impl(PostInput::File(path), opts)

190
191fn run_post_processing_impl(input: PostInput, opts: &PostOptions) -> String {
192  let PostOptions {
193    pmml,
194    cmml,
195    keep_xmath,
196    stylesheet,
197    destination,
198    source_directory,
199    site_directory,
200    search_paths,
201    nodefaultresources,
202    css_files,
203    js_files,
204    noinvisibletimes,
205    plane1,
206    hackplane1,
207    mathtex,
208    navigationtoc,
209    schemadocs,
210    split,
211    ref split_xpath,
212    split_naming,
213    xslt_parameters,
214    graphics_svg_threshold_kb,
215    graphicimages,
216    timestamp,
217    icon,
218    whatsout,
219  } = *opts;
220
221  let mut doc_opts = PostDocumentOptions::default();
222  if let Some(dest) = destination {
223    doc_opts.destination = Some(dest.to_string());
224  }
225  if let Some(src_dir) = source_directory {
226    doc_opts.source_directory = Some(src_dir.to_string());
227    let mut sp = doc_opts.searchpaths.take().unwrap_or_default();
228    sp.push(src_dir.to_string());
229    doc_opts.searchpaths = Some(sp);
230  }
231  // --sitedirectory (Perl LaTeXML.pm L430 `siteDirectory`): the site root for
232  // relativizing cross-document/resource URLs. When unset, `PostDocument::new`
233  // falls back to the destination's directory (document.rs, Perl Config.pm L466).
234  if let Some(site_dir) = site_directory {
235    doc_opts.site_directory = Some(site_dir.to_string());
236  }
237  let audit = *POST_AUDIT;
238  let audit_start = |name: &str| -> Option<(String, std::time::Instant)> {
239    if audit {
240      Some((name.to_string(), std::time::Instant::now()))
241    } else {
242      None
243    }
244  };
245  let audit_end = |started: Option<(String, std::time::Instant)>| {
246    if let Some((name, t0)) = started {
247      let ms = t0.elapsed().as_millis();
248      Info!("audit", "phase", s!("{} took {}ms", name, ms));
249    }
250  };
251
252  telemetry::phase_enter(Phase::PostXmlParse);
253  let t_parse = audit_start("PostDocument parse");
254  // `raw_xml` holds the in-memory serialization — `Some` only for `Xml` input.
255  // `File` input streams from disk and never holds it, so `raw_xml` is `None`
256  // and the three raw-string features (empty-doc substitution, SVG extraction,
257  // ar5iv sniff) plus the on-failure echo are re-derived from the parsed doc.
258  //
259  // Perl LaTeXML.pm L330-336: an empty core result (e.g. after a Fatal) is still
260  // post-processed against a bare <document/> root ("important for utility
261  // features such as packing .zip archives"). Mirror that for empty in-memory
262  // input; a File is never empty (an empty file simply errors on parse below).
263  let raw_xml: Option<&str> = match input {
264    PostInput::Xml(xml) if xml.trim().is_empty() => Some("<document/>"),
265    PostInput::Xml(xml) => Some(xml),
266    PostInput::File(_) => None,
267  };
268  // Best-effort output when a phase bails: echo the original XML for in-memory
269  // input; for File input the source is on disk (and a failure here is itself
270  // the case we are hardening against), so return empty and rely on the logged
271  // Error + status code.
272  let fallback = || raw_xml.map(str::to_string).unwrap_or_default();
273
274  // Pass B re-parses each spilled page and must see the SAME searchpaths and
275  // site directory as this parse does.
276  let page_opts = doc_opts.clone();
277  let parsed = match input {
278    // `raw_xml` is `Some` for `Xml` input (the two arms above), so the default
279    // is dead — kept only to avoid an `unwrap`.
280    PostInput::Xml(_) => PostDocument::new_from_string(raw_xml.unwrap_or("<document/>"), doc_opts),
281    PostInput::File(path) => PostDocument::new_from_file(path, doc_opts),
282  };
283  let doc = match parsed {
284    Ok(d) => d,
285    Err(e) => {
286      post_error("parse", &format!("failed to parse XML: {e}"));
287      telemetry::phase_exit();
288      return fallback();
289    },
290  };
291  audit_end(t_parse);
292  telemetry::phase_exit();
293
294  // SVG extraction reads the pre-post XML before any in-tree mutation; do it
295  // once up-front so the regex-based fragment table is valid for every split
296  // sub-document below. In-memory input scans the string; File input locates
297  // pictures via a limit-safe `//ltx:picture` DOM query and serializes only
298  // those (never the whole file) — most large documents have none.
299  let t_svg = audit_start("SVG extraction");
300  let svg_fragments = match raw_xml {
301    Some(xml) => extract_svg_fragments(xml),
302    None => extract_svg_fragments_from_doc(&doc),
303  };
304  audit_end(t_svg);
305
306  // ar5iv sniff: the ar5iv package emits literal-intent MathML. In-memory input
307  // checks the raw string; file input checks the parsed doc's `<?latexml
308  // package="ar5iv"?>` PI. Computed here — before Split moves `doc` into `docs`.
309  let intent_literal = match raw_xml {
310    Some(xml) => xml.contains("package=\"ar5iv"),
311    None => doc
312      .processing_instructions()
313      .iter()
314      .any(|pi| pi.contains("package=\"ar5iv")),
315  };
316
317  // Perl-faithful pipeline order (latexmlpost L223-242):
318  //   Split → Scan → MakeBibliography → CrossRef → Graphics → ...
319  // Split runs FIRST so each downstream pass sees the per-page
320  // destination. With the previous order (Scan before Split), every
321  // entry's `location` was the root document and CrossRef built every
322  // ref as a within-page anchor — the user-visible TOC links pointed
323  // at `#Ch1` instead of `Ch1.html`.
324  //
325  // Phase 1: Split (only attributes time when --split is on)
326  let mut docs: Vec<PostDocument> = if split {
327    telemetry::phase_enter(Phase::Split);
328    let result = if let Some(xpath) = split_xpath {
329      let naming = match split_naming {
330        Some("id") | None => latexml_post::split::SplitNaming::Id,
331        Some("idrelative") => latexml_post::split::SplitNaming::IdRelative,
332        Some("label") => latexml_post::split::SplitNaming::Label,
333        Some("labelrelative") => latexml_post::split::SplitNaming::LabelRelative,
334        Some(other) => {
335          Warn!(
336            "post",
337            "split",
338            format!("Unknown splitnaming '{other}', using 'id'")
339          );
340          latexml_post::split::SplitNaming::Id
341        },
342      };
343      let mut splitter = latexml_post::split::Split::new(xpath, naming, false);
344      let split_nodes = splitter.to_process(&doc);
345      match splitter.process(doc, split_nodes) {
346        Ok(docs) => {
347          if docs.len() > 1 {
348            note_progress(&format!("Split into {} documents", docs.len()));
349          }
350          Ok(docs)
351        },
352        Err(e) => {
353          post_error("split", &format!("Split failed: {e}"));
354          Err(())
355        },
356      }
357    } else {
358      Ok(vec![doc])
359    };
360    telemetry::phase_exit();
361    match result {
362      Ok(ds) => ds,
363      Err(()) => return fallback(),
364    }
365  } else {
366    vec![doc]
367  };
368
369  // Helper: run one processor across every doc in a Vec, mirroring
370  // Perl Post.pm:50-65's `foreach $proc { @newdocs = ... }` loop.
371  // Each processor's per-doc `process` may return multiple docs (only
372  // Split actually does, and we've already run Split above), so the
373  // inner result is flattened back into `docs`.
374  fn run_phase<P: Processor + ?Sized>(
375    docs: Vec<PostDocument>,
376    proc: &mut P,
377    label: &'static str,
378  ) -> Result<Vec<PostDocument>, ()> {
379    let mut out: Vec<PostDocument> = Vec::with_capacity(docs.len());
380    for d in docs {
381      let nodes = proc.to_process(&d);
382      if nodes.is_empty() {
383        out.push(d);
384        continue;
385      }
386      match proc.process(d, nodes) {
387        Ok(processed) => out.extend(processed),
388        Err(e) => {
389          post_error(label, &format!("{label} failed: {e}"));
390          return Err(());
391        },
392      }
393    }
394    Ok(out)
395  }
396
397  // Phase 2: Scan — runs on EACH sub-document so its entries register
398  // the per-page `location` and `pageid`. Single shared ObjectDB so
399  // the later CrossRef pass can resolve cross-doc refs.
400  let mut scanner = latexml_post::scan::Scan::new(ObjectDB::new());
401  telemetry::phase_enter(Phase::PostScan);
402  let t_scan = audit_start("Scan");
403  docs = match run_phase(docs, &mut scanner, "Scan") {
404    Ok(d) => d,
405    Err(()) => {
406      telemetry::phase_exit();
407      return fallback();
408    },
409  };
410  audit_end(t_scan);
411  telemetry::phase_exit();
412
413  // ---- Pass A / Pass B boundary -------------------------------------------
414  //
415  // Scan is the ONLY phase that needs global knowledge: it visits every page
416  // and produces strings in the ObjectDB. Everything after it is page-local.
417  // The driver used to be phase-major (`for each phase { for each page }`),
418  // which keeps every page alive at every boundary — and a live page is not
419  // cheap: each is its own `xmlDoc` with its own dictionary, id table and
420  // lazily-built caches, measured at ~1.6 MB of per-document overhead. Across
421  // the 40,201 pages of a 614 MB core XML that alone grew RSS from 16 GB to
422  // 80 GB *during Scan*, wrote zero pages, and was killed by the ceiling.
423  //
424  // So: spill each scanned page to disk, free it, and render page-by-page
425  // below. Peak becomes the one-time split DOM plus ONE page. Spilling (rather
426  // than re-deriving pages from the source) keeps Scan's per-page
427  // `location`/`pageid` registration byte-identical — those semantics were
428  // themselves the fix for empty cross-document TOCs, so they are the wrong
429  // thing to restructure here.
430  let page_spill = match tempfile::Builder::new().prefix("latexml-post-").tempdir() {
431    Ok(dir) => dir,
432    Err(e) => {
433      post_error(
434        "post",
435        &format!("cannot create the page spill directory: {e}"),
436      );
437      return fallback();
438    },
439  };
440  // (path, destination, destination_directory) — the metadata that does NOT
441  // survive an XML round-trip and must be restored in pass B.
442  // Plus a flag per page: does it carry an index/glossary/bibliography
443  // placeholder? Recorded HERE, while the page is already parsed, so the two
444  // baton sweeps below visit only the handful of pages that have work instead
445  // of re-parsing every page twice (~80 k redundant parses on a 40 k-page
446  // document).
447  let mut spilled_pages: Vec<SpilledPage> = Vec::with_capacity(docs.len());
448  for (i, d) in docs.drain(..).enumerate() {
449    let path = page_spill.path().join(format!("page-{i:07}.xml"));
450    if let Err(e) = std::fs::write(&path, d.to_xml_string()) {
451      post_error("post", &format!("cannot spill page {i}: {e}"));
452      return fallback();
453    }
454    spilled_pages.push(SpilledPage {
455      needs_index: !d
456        .findnodes("//ltx:index[not(ltx:indexlist)] | //ltx:glossary[not(ltx:glossarylist)]")
457        .is_empty(),
458      needs_bib: !d.findnodes("//ltx:bibliography").is_empty(),
459      path,
460      destination: d.get_destination().map(String::from),
461      destination_directory: d.get_destination_directory().map(String::from),
462    });
463    // `d` drops here: one page's worth of DOM and caches released before the
464    // next is touched.
465  }
466
467  // Phase 2b: MakeIndex (Perl LaTeXML.pm L466-470)
468  //   Runs BEFORE MakeBibliography in Perl. Populates `<ltx:indexlist>`
469  //   inside `<ltx:index>` placeholders and `<ltx:glossarylist>` inside
470  //   `<ltx:glossary>` placeholders, using the `GLOSSARY:*` / `INDEX:*`
471  //   entries Scan registered in the ObjectDB. Without this pass, glossary
472  //   sections render empty in HTML (witness: tests/structure/glossary.tex).
473  let mut indexer = latexml_post::make_index::MakeIndex::new(scanner.db, false, false);
474
475  // Phase 3: MakeBibliography
476  //
477  // A raw `.bib` is converted by a recursive BibTeX session (Perl
478  // `MakeBibliography.pm::convertBibliography`), which needs this crate's
479  // model loader — so `latexml_post` declares the hook and we fill it in here,
480  // on the thread that is about to run the pipeline.
481  crate::bib_session::install();
482
483  // Phase 4: CrossRef — built only once the bibliography sweep has handed the
484  // ObjectDB baton on (see `sweep_pages`), so nothing holds two owners of it.
485  let build_crossref = |db| {
486    let mut crossref =
487      latexml_post::crossref::CrossRef::new(db, latexml_post::crossref::UrlStyle::File, true);
488    if let Some(navtoc) = navigationtoc {
489      crossref.set_navigation_toc(navtoc);
490    }
491    crossref
492  };
493
494  // Phase 5: Graphics (Perl `graphicimages!` → `dographics`, default on).
495  // `--nographicimages` skips the phase wholesale, leaving the raw
496  // `<ltx:graphics>` references in the output untouched.
497  let mut graphics_proc = graphicimages.then(|| {
498    latexml_post::graphics::Graphics::new(None, true)
499      .with_svg_threshold_kb(graphics_svg_threshold_kb)
500  });
501
502  // Phase 3: MathML + XSLT
503  let mut post = latexml_post::Post::new();
504  let mut processors: Vec<Box<dyn Processor>> = Vec::new();
505
506  // `intent_literal` was computed up-front (before Split consumed `doc`).
507
508  // Parallel P+C markup: when both formats are requested, the Content-MathML
509  // processor is a *secondary* of the Presentation primary, folded into one
510  // `<m:semantics>`/`<m:annotation-xml>` by `combine_parallel` — NOT a second
511  // independent pass (which left the content tree as an orphan `<apply>` sibling
512  // of `<m:semantics>`, rendered as stray text by browsers). Mirrors Perl
513  // `MathProcessor`'s primary→secondary parallel model.
514  if pmml {
515    let mut presentation = latexml_post::mathml::MathML::new_presentation()
516      .with_keep_xmath(keep_xmath)
517      .with_invisible_times(!noinvisibletimes)
518      .with_plane1(plane1, hackplane1)
519      .with_mathtex(mathtex)
520      .with_intent_literal(intent_literal);
521    if cmml {
522      presentation = presentation.with_secondaries(vec![Box::new(
523        latexml_post::mathml::MathML::new_content()
524          .with_keep_xmath(keep_xmath)
525          .with_invisible_times(!noinvisibletimes)
526          .with_plane1(plane1, hackplane1)
527          .secondary(),
528      )]);
529    }
530    processors.push(Box::new(presentation));
531  } else if cmml {
532    // Content-only output (no presentation primary).
533    processors.push(Box::new(
534      latexml_post::mathml::MathML::new_content()
535        .with_keep_xmath(keep_xmath)
536        .with_invisible_times(!noinvisibletimes)
537        .with_plane1(plane1, hackplane1),
538    ));
539  }
540  if let Some(xsl_path) = stylesheet {
541    let mut searchpaths = vec![".".to_string()];
542    if let Ok(exe) = std::env::current_exe()
543      && let Some(project_root) = exe
544        .parent()
545        .and_then(|p| p.parent())
546        .and_then(|p| p.parent())
547    {
548      searchpaths.insert(0, project_root.display().to_string());
549    }
550    // Prepend the user's `--path` directories so `--css`/`--javascript`
551    // resources are found there (and take priority over `.`/project root)
552    // when copy_param_resources searches for them.
553    for p in search_paths.iter().rev() {
554      searchpaths.insert(0, p.clone());
555    }
556    let mut xslt_params = rustc_hash::FxHashMap::default();
557
558    // When `--schemadocs` is on, auto-prepend the rustdoc-styled
559    // theme assets so callers don't need to repeat them on every
560    // invocation. Idempotent: skipped if the user has already
561    // listed the same basename via `--css` / `--javascript`.
562    let prepend = |list: &[String], extra: &str| -> Vec<String> {
563      if list
564        .iter()
565        .any(|p| std::path::Path::new(p).file_name().and_then(|s| s.to_str()) == Some(extra))
566      {
567        list.to_vec()
568      } else {
569        let mut out = Vec::with_capacity(list.len() + 1);
570        out.push(extra.to_string());
571        out.extend_from_slice(list);
572        out
573      }
574    };
575    let css_effective: Vec<String> = if schemadocs {
576      prepend(css_files, latexml_post::schema_docs::THEME_CSS_BASENAME)
577    } else {
578      css_files.to_vec()
579    };
580    let js_effective: Vec<String> = if schemadocs {
581      prepend(js_files, latexml_post::schema_docs::THEME_JS_BASENAME)
582    } else {
583      js_files.to_vec()
584    };
585    if !css_effective.is_empty() {
586      xslt_params.insert(
587        "CSS".to_string(),
588        format!("\"{}\"", css_effective.join("|")),
589      );
590    }
591    if !js_effective.is_empty() {
592      xslt_params.insert(
593        "JAVASCRIPT".to_string(),
594        format!("\"{}\"", js_effective.join("|")),
595      );
596    }
597    if let Some(navtoc) = navigationtoc {
598      xslt_params.insert("NAVIGATIONTOC".to_string(), format!("\"{}\"", navtoc));
599    }
600    if let Some(ts) = timestamp {
601      xslt_params.insert("TIMESTAMP".to_string(), format!("\"{}\"", ts));
602    }
603    if let Some(icon) = icon {
604      xslt_params.insert("ICON".to_string(), format!("\"{}\"", icon));
605    }
606    // Perl `LaTeXML.pm:562` unconditionally seeds `LATEXML_VERSION => '$LaTeXML::VERSION'`,
607    // which drives `LaTeXML-common.xsl`'s `LaTeXML_identifier` generator stamp
608    // (`<!--Generated by LaTeXML oxide (version X) ...-->`) and BookML's `utils.xsl`
609    // `b:version-leq($LATEXML_VERSION,…)`. We expose OUR own crate `X.Y.Z` (#320). Inserted
610    // BEFORE the user-override loop so `--xsltparameter LATEXML_VERSION=…` still wins (this
611    // is how Perl's test suite pins `version TEST` for version-independent goldens).
612    xslt_params.insert(
613      "LATEXML_VERSION".to_string(),
614      format!("\"{}\"", crate::core_interface::LATEXML_VERSION),
615    );
616    for param in xslt_parameters {
617      if let Some((key, value)) = param.split_once('=') {
618        xslt_params.insert(key.to_string(), format!("\"{}\"", value));
619      }
620    }
621    match latexml_post::xslt::XSLT::new(
622      xsl_path,
623      xslt_params,
624      nodefaultresources,
625      None,
626      searchpaths,
627    ) {
628      Ok(xslt) => processors.push(Box::new(xslt)),
629      Err(e) => post_error("xslt", &format!("XSLT error: {e}")),
630    }
631  }
632
633  // process_chain attributes per-processor inside latexml_post::Post::
634  // process_chain (MathmlPres / MathmlCont / Xslt). No outer phase wrap
635  // needed; the inner per-processor guards cover their own time.
636  //
637  // Perl-faithful: pass ALL split docs to ProcessChain at once. Perl's
638  // `Post::ProcessChain_internal` (Post.pm L41-67) seeds `@docs = ($doc)`
639  // and lets each processor return a (possibly multi-element) list which
640  // becomes the input to the next processor. We've already produced the
641  // post-Split list above; ProcessChain just needs to fan MathML/XSLT
642  // across it.
643  let is_html_out = stylesheet.is_some_and(|s| s.contains("html"));
644
645  // ---- Pass B: render one page at a time, then FREE it --------------------
646  //
647  // Document-major, not phase-major: each page goes through MakeIndex ->
648  // MakeBibliography -> CrossRef -> Graphics -> MathML -> XSLT -> write, and
649  // is dropped before the next is parsed. Every one of these phases is
650  // page-local; only Scan (pass A, above) needed the whole document, and it
651  // left its results in the ObjectDB as strings. Peak is therefore ONE page
652  // plus the index, whatever the page count.
653  //
654  // Failure semantics change deliberately: pages that finish are already on
655  // disk, so a later failure leaves a partial site rather than nothing. That
656  // matches how a resource Fatal already keeps partial core output.
657  let run_page_phase = |doc: PostDocument,
658                        proc: &mut dyn Processor,
659                        label: &'static str|
660   -> Result<Vec<PostDocument>, ()> {
661    let nodes = proc.to_process(&doc);
662    if nodes.is_empty() {
663      return Ok(vec![doc]);
664    }
665    proc.process(doc, nodes).map_err(|e| {
666      post_error(label, &format!("{label} failed: {e}"));
667    })
668  };
669
670  // The ObjectDB is a BATON: MakeIndex, MakeBibliography and CrossRef each take
671  // it by value in turn, so they cannot be alive simultaneously. Each therefore
672  // gets its own sweep over the spilled pages — still ONE page resident at a
673  // time. A page is only re-spilled when its phase actually had work
674  // (`to_process` non-empty), which for index/bibliography placeholders is a
675  // handful of pages out of tens of thousands, so the sweeps cost a parse each
676  // and almost no writes.
677  fn sweep_pages(
678    pages: &[SpilledPage],
679    opts: &PostDocumentOptions,
680    proc: &mut dyn Processor,
681    label: &'static str,
682    selector: fn(&SpilledPage) -> bool,
683  ) -> Result<(), ()> {
684    for page_meta in pages.iter().filter(|p| selector(p)) {
685      let (path, dest, dest_dir) = (
686        &page_meta.path,
687        &page_meta.destination,
688        &page_meta.destination_directory,
689      );
690      let path_str = path.to_string_lossy().into_owned();
691      let mut page = PostDocument::new_from_file(&path_str, opts.clone()).map_err(|e| {
692        post_error("post", &format!("cannot re-read a spilled page: {e}"));
693      })?;
694      page.destination = dest.clone();
695      page.destination_directory = dest_dir.clone();
696      let nodes = proc.to_process(&page);
697      if nodes.is_empty() {
698        continue; // untouched: the spill on disk is still current
699      }
700      let processed = proc.process(page, nodes).map_err(|e| {
701        post_error(label, &format!("{label} failed: {e}"));
702      })?;
703      // A phase may split one page into several (an index page beside its
704      // host); the extras are written by their own destination in pass B, so
705      // re-spill each under the original path only for the first and append
706      // the rest as additional spills is NOT needed here: MakeIndex and
707      // MakeBibliography fill placeholders in place and return the same page.
708      for d in processed.iter().take(1) {
709        if let Err(e) = std::fs::write(path, d.to_xml_string()) {
710          post_error("post", &format!("cannot re-spill a page: {e}"));
711          return Err(());
712        }
713      }
714    }
715    Ok(())
716  }
717
718  if sweep_pages(&spilled_pages, &page_opts, &mut indexer, "MakeIndex", |p| {
719    p.needs_index
720  })
721  .is_err()
722  {
723    return fallback();
724  }
725  let mut bibmaker = latexml_post::make_bibliography::MakeBibliography::new(indexer.db, false);
726  telemetry::phase_enter(Phase::Bibliography);
727  let t_bib = audit_start("MakeBibliography");
728  let bib_result = sweep_pages(
729    &spilled_pages,
730    &page_opts,
731    &mut bibmaker,
732    "MakeBibliography",
733    |p| p.needs_bib,
734  );
735  audit_end(t_bib);
736  telemetry::phase_exit();
737  if bib_result.is_err() {
738    return fallback();
739  }
740  let mut crossref = build_crossref(bibmaker.db);
741
742  let mut main_output: Option<String> = None;
743  let t_pages = audit_start("render_pages");
744  for SpilledPage {
745    path,
746    destination: dest,
747    destination_directory: dest_dir,
748    ..
749  } in spilled_pages
750  {
751    let path_str = path.to_string_lossy().into_owned();
752    let page = match PostDocument::new_from_file(&path_str, page_opts.clone()) {
753      Ok(mut d) => {
754        d.destination = dest;
755        d.destination_directory = dest_dir;
756        d
757      },
758      Err(e) => {
759        post_error("post", &format!("cannot re-read a spilled page: {e}"));
760        return fallback();
761      },
762    };
763    // Free the spill as soon as it is parsed: for a 40k-page document the
764    // spill area is the size of the core XML, and it need not outlive its page.
765    let _ = std::fs::remove_file(&path);
766
767    // CrossRef resolves this page's references out of the completed ObjectDB,
768    // and its navigation/TOC work reads the DB rather than sibling pages — so
769    // it is page-local once pass A has finished.
770    telemetry::phase_enter(Phase::Crossref);
771    let t_xref = audit_start("CrossRef");
772    let xref = run_page_phase(page, &mut crossref, "CrossRef");
773    audit_end(t_xref);
774    telemetry::phase_exit();
775    let mut staged = match xref {
776      Ok(out) => out,
777      Err(()) => return fallback(),
778    };
779    if let Some(gfx) = graphics_proc.as_mut() {
780      telemetry::phase_enter(Phase::Graphics);
781      let t_gfx = audit_start("Graphics");
782      let mut next = Vec::with_capacity(staged.len());
783      let mut failed = false;
784      for d in staged.drain(..) {
785        match run_page_phase(d, gfx, "Graphics") {
786          Ok(out) => next.extend(out),
787          Err(()) => {
788            failed = true;
789            break;
790          },
791        }
792      }
793      audit_end(t_gfx);
794      telemetry::phase_exit();
795      if failed {
796        return fallback();
797      }
798      staged = next;
799    }
800
801    let rendered = match post.process_chain(staged, &mut processors) {
802      Ok(r) => r,
803      Err(e) => {
804        post_error("convert", &format!("Post-processing failed: {e}"));
805        return fallback();
806      },
807    };
808
809    for doc in rendered {
810      let dest = doc.get_destination().map(String::from);
811      let output = latexml_post::extract::serialize_whatsout(&doc, whatsout);
812      let output = if is_html_out {
813        finalize_html5(output, &svg_fragments)
814      } else {
815        output
816      };
817      let output = if schemadocs && is_html_out {
818        latexml_post::schema_docs::process_page(&output)
819      } else {
820        output
821      };
822      if let Some(path) = dest.as_deref() {
823        if let Some(parent) = std::path::Path::new(path).parent()
824          && !parent.as_os_str().is_empty()
825        {
826          let _ = std::fs::create_dir_all(parent);
827        }
828        if let Err(e) = std::fs::write(path, &output) {
829          post_error("write", &format!("failed to write page {path}: {e}"));
830        }
831      }
832      if main_output.is_none() {
833        main_output = Some(output);
834      }
835      // `doc` drops here — the whole point.
836    }
837  }
838  audit_end(t_pages);
839  drop(page_spill);
840
841  main_output.unwrap_or_else(fallback)
842}

845/// lives, the metadata that does NOT survive an XML round-trip, and whether it
846/// carries the placeholders the two ObjectDB-baton sweeps look for (recorded in
847/// pass A, while the page is already parsed).
848struct SpilledPage {
849  path:                  std::path::PathBuf,
850  destination:           Option<String>,
851  destination_directory: Option<String>,
852  needs_index:           bool,
853  needs_bib:             bool,

856/// Re-derive SVG fragments from a parsed document (the file-input equivalent of
857/// [`extract_svg_fragments`]).
858///
859/// `//ltx:picture` is a *typed* descendant query, which libxml2 evaluates
860/// without materializing `descendant-or-self::node()` — so it stays under the
861/// 10M node-set ceiling even on a 600 MB document. Most large documents have no
862/// pictures at all (the fast, zero-serialization path); when they do, only the
863/// picture subtrees are serialized and handed to the shared string extractor,
864/// never the whole file.
865fn extract_svg_fragments_from_doc(doc: &PostDocument) -> Vec<(String, String)> {
866  let pictures = doc.findnodes("//ltx:picture");
867  if pictures.is_empty() {
868    return Vec::new();
869  }
870  let joined: String = pictures.iter().map(|p| doc.node_to_string(p)).collect();
871  extract_svg_fragments(&joined)
872}

874/// Apply HTML5 cleanup (XML prolog strip, void-element fixes) and inject SVG
875/// fragments into empty `ltx_picture` spans. Pulled out of `run_post_processing`
876/// so it can run on every split sub-document, not just the first.
877fn finalize_html5(output: String, svg_fragments: &[(String, String)]) -> String {
878  use std::sync::LazyLock;
879  // Cached at first call — regex compile is the slow part of `Regex::new`,
880  // and finalize_html5 runs on every (sub-)document in the post-pipeline.
881  static XML_PROLOG_RE: LazyLock<regex::Regex> =
882    LazyLock::new(|| regex::Regex::new(r"^<\?xml[^?]*\?>\s*").unwrap());
883  static NON_VOID_SELF_CLOSE_RE: LazyLock<regex::Regex> = LazyLock::new(|| {
884    regex::Regex::new(
885      r"<(span|div|p|a|td|th|tr|section|article|figure|figcaption|pre|code|em|strong|b|i|u|sub|sup|small|cite)(\s[^>]*)?/>",
886    ).unwrap()
887  });
888  static VOID_CLOSE_RE: LazyLock<regex::Regex> = LazyLock::new(|| {
889    regex::Regex::new(r"</(br|img|hr|input|meta|link|col|area|base|source|track|wbr|embed|param)>")
890      .unwrap()
891  });
892  static VOID_SELF_CLOSE_RE: LazyLock<regex::Regex> = LazyLock::new(|| {
893    regex::Regex::new(
894      r"<(br|img|hr|input|meta|link|col|area|base|source|track|wbr|embed|param)(\s[^>]*?)\s*/>",
895    )
896    .unwrap()
897  });
898
899  let _gp_html5 = telemetry::phase(Phase::Html5Fixups);
900  // Strip <?xml version...?> prolog: HTML5 must NOT have an XML declaration.
901  // libxml2's to_string() includes it by default; we strip it here.
902  let output = XML_PROLOG_RE.replace(&output, "").to_string();
903  let output = NON_VOID_SELF_CLOSE_RE
904    .replace_all(&output, "<$1$2></$1>")
905    .to_string();
906  let output = VOID_CLOSE_RE.replace_all(&output, "").to_string();
907  let mut output = VOID_SELF_CLOSE_RE
908    .replace_all(&output, "<$1$2>")
909    .to_string();
910  if !svg_fragments.is_empty() {
911    for (pic_id, svg_html) in svg_fragments {
912      let pattern = format!(
913        r#"<span id="{}" class="ltx_picture"([^>]*)></span>"#,
914        regex::escape(pic_id)
915      );
916      if let Ok(re) = regex::Regex::new(&pattern) {
917        output = re
918          .replace(&output, |caps: &regex::Captures| {
919            format!(
920              r#"<span id="{}" class="ltx_picture"{}>{}</span>"#,
921              pic_id, &caps[1], svg_html
922            )
923          })
924          .to_string();
925      }
926    }
927  }
928  output
929}

931/// Extract SVG fragments from intermediate LaTeXML XML.
932///
933/// Finds `<picture>` elements, converts their children to inline SVG HTML.
934/// Uses a lightweight regex+string approach (no libxml2) to avoid the
935/// use-after-free crash in PostDocument cleanup.
936///
937/// Returns (picture_id, svg_html) pairs for post-XSLT injection.
938fn extract_svg_fragments(xml: &str) -> Vec<(String, String)> {
939  use std::sync::LazyLock;
940  // Fast-fail: most documents have no `<picture>` elements (tikz / pgf
941  // is uncommon in the canvas). Skip the backtracking lazy-match
942  // regex (`(?s)...(.*?)`) when `<picture` doesn't appear as a literal
943  // substring. `str::contains` is a SIMD-accelerated byte search and
944  // takes microseconds even on ~MB inputs.
945  if !xml.contains("<picture") {
946    return Vec::new();
947  }
948  static PICTURE_RE: LazyLock<regex::Regex> =
949    LazyLock::new(|| regex::Regex::new(r#"(?s)<picture([^>]*)>(.*?)</picture>"#).unwrap());
950  static ID_RE: LazyLock<regex::Regex> =
951    LazyLock::new(|| regex::Regex::new(r#"xml:id="([^"]+)""#).unwrap());
952  static WIDTH_RE: LazyLock<regex::Regex> =
953    LazyLock::new(|| regex::Regex::new(r#"width="([^"]+)""#).unwrap());
954  static HEIGHT_RE: LazyLock<regex::Regex> =
955    LazyLock::new(|| regex::Regex::new(r#"height="([^"]+)""#).unwrap());
956  let mut fragments = Vec::new();
957  let picture_re = &*PICTURE_RE;
958  let id_re = &*ID_RE;
959  let width_re = &*WIDTH_RE;
960  let height_re = &*HEIGHT_RE;
961
962  for pic_caps in picture_re.captures_iter(xml) {
963    let attrs = &pic_caps[1];
964    let content = &pic_caps[2];
965    let id = id_re
966      .captures(attrs)
967      .map(|c| c[1].to_string())
968      .unwrap_or_default();
969    let width = width_re.captures(attrs).and_then(|c| parse_tex_dim(&c[1]));
970    let height = height_re.captures(attrs).and_then(|c| parse_tex_dim(&c[1]));
971
972    if id.is_empty() || content.trim().is_empty() {
973      continue;
974    }
975
976    let w = width.unwrap_or(100.0);
977    let h = height.unwrap_or(100.0);
978
979    // Build inline SVG: coordinate system has y-flip (TeX origin bottom-left, SVG top-left)
980    let mut svg_content = format!(
981      r#"<svg xmlns="http://www.w3.org/2000/svg" version="1.1" width="{w:.2}" height="{h:.2}" overflow="visible">"#,
982    );
983    svg_content.push_str(&format!(
984      r#"<g transform="translate(0,{h:.2}) scale(1,-1)">"#,
985    ));
986
987    // Convert ltx picture children to SVG elements
988    svg_content.push_str(&convert_picture_children_to_svg(content));
989
990    svg_content.push_str("</g></svg>");
991    fragments.push((id, svg_content));
992  }
993  fragments
994}

996/// Convert LaTeXML picture children (g, line, text, circle, etc.) to SVG elements.
997fn convert_picture_children_to_svg(content: &str) -> String {
998  use std::sync::LazyLock;
999  static G_RE: LazyLock<regex::Regex> =
1000    LazyLock::new(|| regex::Regex::new(r#"(?s)<g([^>]*)>(.*?)</g>"#).unwrap());
1001  static TRANSFORM_RE: LazyLock<regex::Regex> =
1002    LazyLock::new(|| regex::Regex::new(r#"transform="([^"]+)""#).unwrap());
1003  static LINE_RE: LazyLock<regex::Regex> =
1004    LazyLock::new(|| regex::Regex::new(r#"<line\s+points="([^"]+)"([^/]*)/?>"#).unwrap());
1005  static CIRCLE_RE: LazyLock<regex::Regex> =
1006    LazyLock::new(|| regex::Regex::new(r#"<circle([^/]*)/?>"#).unwrap());
1007  static ELLIPSE_RE: LazyLock<regex::Regex> =
1008    LazyLock::new(|| regex::Regex::new(r#"<ellipse([^/]*)/?>"#).unwrap());
1009  static RECT_RE: LazyLock<regex::Regex> =
1010    LazyLock::new(|| regex::Regex::new(r#"<rect([^/]*)/?>"#).unwrap());
1011  static POLYGON_RE: LazyLock<regex::Regex> =
1012    LazyLock::new(|| regex::Regex::new(r#"<polygon([^/]*)/?>"#).unwrap());
1013  static PATH_RE: LazyLock<regex::Regex> =
1014    LazyLock::new(|| regex::Regex::new(r#"<path([^/]*)/?>"#).unwrap());
1015  static BEZIER_RE: LazyLock<regex::Regex> =
1016    LazyLock::new(|| regex::Regex::new(r#"<bezier\s+points="([^"]+)"([^/]*)/?>"#).unwrap());
1017  static TEXT_RE: LazyLock<regex::Regex> =
1018    LazyLock::new(|| regex::Regex::new(r#"(?s)<text([^>]*)>(.*?)</text>"#).unwrap());
1019
1020  let mut svg = String::new();
1021
1022  for g_caps in G_RE.captures_iter(content) {
1023    let g_attrs = &g_caps[1];
1024    let g_content = &g_caps[2];
1025
1026    // Extract transform
1027    let transform = TRANSFORM_RE.captures(g_attrs).map(|c| c[1].to_string());
1028
1029    if let Some(t) = &transform {
1030      svg.push_str(&format!(r#"<g transform="{t}">"#));
1031    } else {
1032      svg.push_str("<g>");
1033    }
1034
1035    // <line points="x1,y1 x2,y2" stroke="..." stroke-width="..."/>
1036    for line_caps in LINE_RE.captures_iter(g_content) {
1037      let points = &line_caps[1];
1038      let rest_attrs = &line_caps[2];
1039      let coords: Vec<&str> = points.split_whitespace().collect();
1040      if coords.len() >= 2 {
1041        let p1: Vec<&str> = coords[0].split(',').collect();
1042        let p2: Vec<&str> = coords[1].split(',').collect();
1043        if p1.len() == 2 && p2.len() == 2 {
1044          svg.push_str(&format!(
1045            r#"<line x1="{}" y1="{}" x2="{}" y2="{}"{}/>"#,
1046            p1[0], p1[1], p2[0], p2[1], rest_attrs
1047          ));
1048        }
1049      }
1050    }
1051
1052    // <circle cx="..." cy="..." r="..." .../>
1053    for circle_caps in CIRCLE_RE.captures_iter(g_content) {
1054      svg.push_str(&format!("<circle{}/>", &circle_caps[1]));
1055    }
1056
1057    // <ellipse cx="..." cy="..." rx="..." ry="..." .../>
1058    for ellipse_caps in ELLIPSE_RE.captures_iter(g_content) {
1059      svg.push_str(&format!("<ellipse{}/>", &ellipse_caps[1]));
1060    }
1061
1062    // <rect x="..." y="..." width="..." height="..." .../>
1063    for rect_caps in RECT_RE.captures_iter(g_content) {
1064      svg.push_str(&format!("<rect{}/>", &rect_caps[1]));
1065    }
1066
1067    // <polygon points="..." .../>
1068    for polygon_caps in POLYGON_RE.captures_iter(g_content) {
1069      svg.push_str(&format!("<polygon{}/>", &polygon_caps[1]));
1070    }
1071
1072    // <path d="..." .../>
1073    for path_caps in PATH_RE.captures_iter(g_content) {
1074      svg.push_str(&format!("<path{}/>", &path_caps[1]));
1075    }
1076
1077    // <bezier points="x1,y1 x2,y2 x3,y3 x4,y4" .../>
1078    // Convert to SVG cubic bezier path
1079    for bez_caps in BEZIER_RE.captures_iter(g_content) {
1080      let points = &bez_caps[1];
1081      let rest = &bez_caps[2];
1082      let coords: Vec<&str> = points.split_whitespace().collect();
1083      if coords.len() >= 4 {
1084        // SVG cubic bezier: M x0,y0 C x1,y1 x2,y2 x3,y3
1085        let d = format!(
1086          "M {} C {} {} {}",
1087          coords[0], coords[1], coords[2], coords[3]
1088        );
1089        svg.push_str(&format!(r#"<path d="{d}"{rest} fill="none"/>"#));
1090      } else if coords.len() >= 3 {
1091        // Quadratic bezier: M x0,y0 Q x1,y1 x2,y2
1092        let d = format!("M {} Q {} {}", coords[0], coords[1], coords[2]);
1093        svg.push_str(&format!(r#"<path d="{d}"{rest} fill="none"/>"#));
1094      }
1095    }
1096
1097    // <arc .../> — arc segments (rarely used, stub for now)
1098    // <wedge .../> — filled wedges (rarely used, stub for now)
1099
1100    // <text>...</text> — wrap in SVG text with y-flip correction
1101    for text_caps in TEXT_RE.captures_iter(g_content) {
1102      let text_attrs = &text_caps[1];
1103      let text_content = &text_caps[2];
1104      svg.push_str(&format!(
1105        r#"<g transform="scale(1,-1)"><text{text_attrs}>{text_content}</text></g>"#,
1106      ));
1107    }
1108
1109    svg.push_str("</g>");
1110  }
1111
1112  // Also handle direct children not inside <g> (e.g. top-level <bezier>, <line>)
1113  // These appear directly inside <picture> without a <g> wrapper
1114  let direct_bezier_re =
1115    regex::Regex::new(r#"(?m)^\s*<bezier\s+points="([^"]+)"([^/]*)/?>"#).unwrap();
1116  for bez_caps in direct_bezier_re.captures_iter(content) {
1117    let points = &bez_caps[1];
1118    let rest = &bez_caps[2];
1119    let coords: Vec<&str> = points.split_whitespace().collect();
1120    if coords.len() >= 4 {
1121      let d = format!(
1122        "M {} C {} {} {}",
1123        coords[0], coords[1], coords[2], coords[3]
1124      );
1125      svg.push_str(&format!(r#"<path d="{d}"{rest} fill="none"/>"#));
1126    } else if coords.len() >= 3 {
1127      let d = format!("M {} Q {} {}", coords[0], coords[1], coords[2]);
1128      svg.push_str(&format!(r#"<path d="{d}"{rest} fill="none"/>"#));
1129    }
1130  }
1131
1132  svg
1133}

1135/// Parse a TeX dimension string (e.g. "100.0pt") to pixels.
1136fn parse_tex_dim(s: &str) -> Option<f64> {
1137  let s = s.trim();
1138  if let Some(rest) = s.strip_suffix("pt") {
1139    rest.parse::<f64>().ok().map(|v| v * 96.0 / 72.27)
1140  } else if let Some(rest) = s.strip_suffix("px") {
1141    rest.parse::<f64>().ok()
1142  } else {
1143    s.parse::<f64>().ok()
1144  }
1145}
