1//! Stable, high-level conversion API for using `latexml` as a **library**.
2//!
3//! Downstream Rust crates depend on `latexml` and call one function — no
4//! binary, no manual [`Config`] or
5//! binding-dispatch wiring:
6//!
7//! ```no_run
8//! let xml = latexml::api::convert_to_xml(r"\documentclass{article}\begin{document}Hi\end{document}")?;
9//! let html = latexml::api::convert_to_html(r"\documentclass{article}\begin{document}Hi\end{document}")?;
10//! # Ok::<(), String>(())
11//! ```
12//!
13//! ## What these encapsulate
14//! The engine is a **thread-local singleton**, so each call runs on its own
15//! worker thread with a large (256 MiB) stack — matching the `latexml_oxide`
16//! binary — so deeply nested math can't overflow the 8 MiB default stack, and
17//! the thread's `#[thread_local]` engine roots (~110 MiB) are released via
18//! [`reset_thread_engine`](latexml_core::reset_thread_engine) before the thread
19//! exits (those roots do **not** run destructors on a bare thread exit). The
20//! standard package + contrib binding-dispatch chain is wired for you.
21//!
22//! Like the `latexml_oxide` binary, these calls interpret a `.sty`/`.cls` that has no binding raw
23//! (Perl LaTeXML's `--includestyles`) and use the arXiv profile's runaway limits; a caller wanting
24//! Perl LaTeXML's binding-only default drives a [`Converter`] with its
25//! own [`Config`] (`include_styles: Some(false)`; OXIDIZED_DESIGN_DIVERGENCES #1080).
26//!
27//! ## Requirements at runtime
28//! Same host dependencies as the binary: a **TeX distribution** on `PATH` for
29//! packages/classes/fonts, and (only for figure-bearing HTML) the graphics
30//! tools. XML/XSLT/RelaxNG assets are embedded.
31//!
32//! For finer control (preloads, search paths, encoding, whatsin/out, split,
33//! …), drive [`crate::converter::Converter`] and [`crate::post`] directly; this
34//! module is the batteries-included entrypoint.
35

45/// Build the standard library `Config`: quiet, LaTeX mode, with the package +
46/// contrib binding-dispatch chain a normal conversion uses.
47fn library_config(format: OutputFormat) -> Config {
48  Config {
49    // Quiet by default: a library caller doesn't want progress notes on stderr.
50    // The per-conversion log is still captured internally and returned in the
51    // error path. Callers wanting logs can use `Converter` directly.
52    verbosity: -1,
53    format,
54    bindings_dispatch: Some(Rc::new(latexml_package::dispatch)),
55    extra_bindings_dispatch: Some(Rc::new(latexml_contrib::dispatch)),
56    // The CLI's defaults: raw `.sty`/`.cls` loading and ar5iv.sty's runaway limits (OD #1080).
57    include_styles: Some(true),
58    if_limit: Some(latexml_core::common::RUNAWAY_IF_LIMIT),
59    absorb_limit: Some(latexml_core::common::RUNAWAY_ABSORB_LIMIT),
60    ..Config::default()
61  }
62}

64/// Run `job` on a fresh 256 MiB-stack worker thread and free the thread-local
65/// engine before the thread exits (even if `job` panics). Mirrors
66/// `latexml_oxide::main`'s worker thread and `util::test`'s per-conversion
67/// `reset_thread_engine()`. Returns `Err` on spawn failure or worker panic
68/// rather than crashing the library caller.
69fn on_worker<T: Send + 'static>(
70  job: impl FnOnce() -> Result<T, String> + Send + 'static,
71) -> Result<T, String> {
72  let handle = std::thread::Builder::new()
73    .stack_size(256 * 1024 * 1024)
74    .spawn(move || {
75      // The CLI's pushback cap (ar5iv.sty's `pushbacklimit`), on the thread whose gullet converts.
76      if std::env::var_os("LATEXML_NO_DEFAULT_PUSHBACK_LIMIT").is_none() {
77        latexml_core::gullet::set_pushback_limit(Some(5_000_000));
78      }
79      // `#[thread_local]` engine roots don't Drop on thread exit; free the
80      // ~110 MiB explicitly (including on unwind) so repeated calls don't accumulate.
81      struct EngineResetGuard;
82      impl Drop for EngineResetGuard {
83        fn drop(&mut self) { latexml_core::reset_thread_engine(); }
84      }
85      let _guard = EngineResetGuard;
86      job()
87    })
88    .map_err(|e| format!("failed to spawn latexml worker thread: {e}"))?;
89  match handle.join() {
90    Ok(result) => result,
91    Err(payload) => {
92      let msg = payload
93        .downcast_ref::<&str>()
94        .copied()
95        .or_else(|| payload.downcast_ref::<String>().map(String::as_str))
96        .unwrap_or("unknown panic");
97      Err(format!("latexml worker thread panicked: {msg}"))
98    },
99  }
100}

102/// Extract the converted XML from a [`ConversionResponse`], preserving the
103/// captured conversion log in the `Err` message when conversion failed.
104fn core_result_xml(response: ConversionResponse, prefix: &str) -> Result<String, String> {
105  response.result.ok_or_else(|| {
106    if response.log.is_empty() {
107      prefix.to_string()
108    } else {
109      format!("{prefix}:\n{}", response.log)
110    }
111  })
112}

114/// Convert a TeX/LaTeX source string to LaTeXML **XML** (the intermediate
115/// representation, before any HTML post-processing).
116///
117/// Returns the serialized XML on success, or the captured conversion log on
118/// failure (a fatal error, or an engine that could not initialize).
119pub fn convert_to_xml(tex: &str) -> Result<String, String> {
120  let tex = tex.to_string();
121  on_worker(move || {
122    let opts = library_config(OutputFormat::XML);
123    let mut converter = Converter::from_config(opts.clone());
124    converter.note_main_source(&format!("literal:{tex}"));
125    converter
126      .prepare_session(&opts)
127      .map_err(|e| format!("could not prepare session: {e}"))?;
128    let response = converter.convert(format!("literal:{tex}"));
129    core_result_xml(response, "conversion failed")
130  })
131}

133/// Convert a TeX/LaTeX source string all the way to a standalone **HTML5**
134/// document (LaTeXML XML + the HTML post-processing pipeline, with
135/// Presentation MathML).
136///
137/// Figures are left as raw `<ltx:graphics>` references — image conversion
138/// writes files next to a destination, and this string-returning API has none;
139/// use [`crate::post`] with a `destination` when you need converted images.
140///
141/// The page links the bundled CSS/JS, and with no destination those files are
142/// copied into the working directory; [`convert_to_html_with`] and
143/// [`HtmlOptions`] choose otherwise.
144pub fn convert_to_html(tex: &str) -> Result<String, String> {
145  convert_to_html_with(tex, &HtmlOptions::default())
146}
147
148/// The settings [`convert_to_html_with`] takes. [`Default`] is what
149/// [`convert_to_html`] does. Start from [`HtmlOptions::default()`] and use its
150/// builder methods so future options remain backwards-compatible:
151///
152/// ```no_run
153/// use latexml::api::{HtmlOptions, convert_to_html_with};
154/// let options = HtmlOptions::default().with_default_resources(false);
155/// let html = convert_to_html_with(r"\documentclass{article}\begin{document}Hi\end{document}", &options)?;
156/// # Ok::<(), String>(())
157/// ```
158#[derive(Clone, Debug)]
159#[non_exhaustive]
160pub struct HtmlOptions {
161  /// Link and copy the bundled CSS/JS (`true`, Perl `defaultresources`). With no
162  /// destination the files land in the working directory; `false` is the
163  /// `--nodefaultresources` equivalent: no links to them, nothing written.
164  pub default_resources: bool,
165}
166
167impl Default for HtmlOptions {
168  fn default() -> Self { HtmlOptions { default_resources: true } }
169}
170
171impl HtmlOptions {
172  /// Set whether to link and copy the bundled CSS/JS resources (`--defaultresources` /
173  /// `--nodefaultresources`).
174  pub fn with_default_resources(mut self, default_resources: bool) -> Self {
175    self.default_resources = default_resources;
176    self
177  }
178}
179
180/// [`convert_to_html`] with the given [`HtmlOptions`].
181pub fn convert_to_html_with(tex: &str, options: &HtmlOptions) -> Result<String, String> {
182  let tex = tex.to_string();
183  let options = options.clone();
184  on_worker(move || {
185    let opts = library_config(OutputFormat::HTML5);
186    let mut converter = Converter::from_config(opts.clone());
187    converter.note_main_source(&format!("literal:{tex}"));
188    converter
189      .prepare_session(&opts)
190      .map_err(|e| format!("could not prepare session: {e}"))?;
191    let xml = core_result_xml(
192      converter.convert(format!("literal:{tex}")),
193      "conversion failed before post-processing",
194    )?;
195
196    let post_opts = html_post_options(options.default_resources);
197    post_result(post::run_post_processing_logged(&xml, &post_opts))
198  })
199}
200
201/// The post-processing [`convert_to_html_with`] runs: the CLI's `html5` defaults, no destination.
202fn html_post_options(default_resources: bool) -> post::PostOptions<'static> {
203  post::PostOptions {
204    pmml: true,
205    // Same per-format sheet the CLI uses (shared source of truth).
206    stylesheet: post::default_stylesheet(Some("html5")),
207    default_resources,
208    // No destination to write images to; keep the raw graphics references.
209    graphicimages: false,
210    ..post::PostOptions::default()
211  }
212}

214/// The HTML, or post's log when post-processing died (`Fatal:post:convert`, #782) and left no
215/// document, the CLI's test. The status also counts core's: a core Fatal's rescued document is
216/// post-processed and returned, as in Perl's convert (LaTeXML.pm L277-337).
217fn post_result(post: post::PostOutcome) -> Result<String, String> {
218  if post.html.is_empty() && post.status_code >= 3 {
219    Err(format!("post-processing failed:\n{}", post.log))
220  } else {
221    Ok(post.html)
222  }
223}

244  /// #782: a post Fatal (here a sheet that terminates) is an `Err` carrying the log, not `Ok("")`.
245  #[test]
246  fn a_post_fatal_is_an_err() {
247    let _ = latexml_core::util::logger::init(log::LevelFilter::Warn);
248    let dir = tempfile::tempdir().expect("tempdir");
249    let xsl = dir.path().join("terminate.xsl");
250    std::fs::write(
251      &xsl,
252      "<xsl:stylesheet version=\"1.0\" xmlns:xsl=\"http://www.w3.org/1999/XSL/Transform\">\
253       <xsl:template match=\"/\"><xsl:message terminate=\"yes\">STOP</xsl:message></xsl:template>\
254       </xsl:stylesheet>",
255    )
256    .unwrap();
257    let xsl = xsl.to_string_lossy().into_owned();
258    let xml = "<?xml version=\"1.0\"?><document xmlns=\"http://dlmf.nist.gov/LaTeXML\"><para><p>Hi</p></para></document>";
259    let result = on_worker(move || {
260      post_result(post::run_post_processing_logged(xml, &post::PostOptions {
261        stylesheet: Some(&xsl),
262        ..html_post_options(false)
263      }))
264    });
265    let log = result.expect_err("a post Fatal is an Err");
266    assert_eq!(log.matches("Fatal:post:convert").count(), 1, "{log}");
267  }

269  /// A core Fatal that keeps its partial document (too many errors) is still post-processed and
270  /// returned, as Perl's convert and the CLI do (LaTeXML.pm L277-337); the Fatal is in the core log.
271  #[test]
272  fn a_core_fatal_keeps_its_rescued_document() {
273    let tex = format!(
274      "\\documentclass{{article}}\\begin{{document}}Kept text.\n\n{}\\end{{document}}",
275      (0..101u8)
276        .map(|i| format!(
277          "\\nosuch{}{} ",
278          (b'a' + i / 26) as char,
279          (b'a' + i % 26) as char
280        ))
281        .collect::<String>()
282    );
283    let (core_log, result) = on_worker(move || {
284      let opts = library_config(OutputFormat::HTML5);
285      let mut converter = Converter::from_config(opts.clone());
286      converter.prepare_session(&opts).expect("session");
287      let response = converter.convert(format!("literal:{tex}"));
288      let xml = response.result.expect("the rescued document");
289      let post = post_result(post::run_post_processing_logged(
290        &xml,
291        &html_post_options(false),
292      ));
293      Ok(((response.status_code, response.log), post))
294    })
295    .expect("worker thread");
296    assert_eq!(core_log.0, 3, "{}", core_log.1);
297    assert!(
298      core_log.1.contains("101 errors; 1 fatal error;"),
299      "{}",
300      core_log.1
301    );
302    let html = result.expect("a core Fatal's rescued document is Ok");
303    assert!(html.contains("Kept text."), "{html}");
304  }
