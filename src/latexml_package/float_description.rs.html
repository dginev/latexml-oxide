1//! The accessible description of a float's image (acmart's `\Description[short]{long}`, oup-authoring-template's
2//! `\figalttext[short]{long}`): a hidden note in the float's caption, which the lone image takes as its `@alt` or
3//! `aria-describedby`. Shared by the class bindings whose classes carry such a command (`acmart_cls`,
4//! `oup_authoring_template_cls`).
5use crate::{engine::latex_constructs::out_of_line_size, prelude::*};
6
7/// A description's tokens with the characters text rejects outside math (`_ ^ & #`) as the text commands for them;
8/// math is `$…$` or `\(…\)`.
9fn description_text(tokens: &Tokens) -> Vec<Token> {
10  let mut in_math = false;
11  tokens
12    .unlist_ref()
13    .iter()
14    .map(|t| match t.get_catcode() {
15      Catcode::MATH => {
16        in_math = !in_math;
17        *t
18      },
19      Catcode::CS if *t == T_CS!("\\(") || *t == T_CS!("\\)") => {
20        in_math = *t == T_CS!("\\(");
21        *t
22      },
23      Catcode::SUB if !in_math => T_CS!("\\textunderscore"),
24      Catcode::SUPER if !in_math => T_CS!("\\textasciicircum"),
25      Catcode::ALIGN if !in_math => T_CS!("\\&"),
26      Catcode::PARAM if !in_math => T_CS!("\\#"),
27      _ => *t,
28    })
29    .collect()
30}

32/// Add `ids` to `node`'s `aria:describedby`, keeping whatever is already there.
33///
34/// `aria-describedby` is an id LIST, and a second `\Description` in the same
35/// float (or one alongside another that already wired the same host) would
36/// otherwise `set_attribute` straight over the first one's reference, leaving
37/// that description hidden in the DOM and announced by nothing — silently
38/// losing an annotation the author wrote. Existing ids keep their position,
39/// since order is announcement order.
40fn add_describedby(document: &mut Document, node: &mut Node, ids: &[String]) -> Result<()> {
41  let existing = node.get_attribute("aria:describedby").unwrap_or_default();
42  let mut refs: Vec<&str> = existing.split_whitespace().collect();
43  for id in ids {
44    if !refs.contains(&id.as_str()) {
45      refs.push(id);
46    }
47  }
48  document.set_attribute(node, "aria:describedby", &refs.join(" "))
49}

51/// The tokens of a description command's call, `\lx@acm@description[short]{long}`: the text of each argument with the
52/// characters text rejects outside math (`_ ^ & #`) as the text commands for them.
53pub fn float_description_tokens(short: Option<&Tokens>, long: &Tokens) -> Tokens {
54  let mut out = vec![T_CS!("\\lx@acm@description")];
55  if let Some(short) = short {
56    out.push(T_OTHER!("["));
57    out.push(T_BEGIN!());
58    out.extend(description_text(short));
59    out.push(T_END!());
60    out.push(T_OTHER!("]"));
61  }
62  out.push(T_BEGIN!());
63  out.extend(description_text(long));
64  out.push(T_END!());
65  Tokens::new(out)
66}

68/// Define the description's machinery for the class's `command` (which its diagnostics name): the WAI-ARIA namespace,
69/// the `acmlabel` counter of its ids and the `\lx@acm@description` constructor that places it.
70#[rustfmt::skip]
71pub fn define_float_description(command: &'static str) -> Result<()> {
72  // The command the diagnostics name: the class's own (`\Description`, `\figalttext`).
73  assign_value("float_description_command", Stored::from(command), Some(Scope::Global));
74  // Register WAI-ARIA namespace for accessible descriptions
75  RegisterDocumentNamespace!("aria", "http://www.w3.org/ns/wai-aria");
76
77  NewCounter!("acmlabel", "");
78  // `\Description[short]{long}` — acmart's accessible figure description
79  // (acmart.cls L895 gobbles both args; ACM requires the description, and HTML
80  // can carry it where PDF cannot, so we surface it rather than follow the
81  // class into discarding it).
82  //
83  // The MANDATORY long description is the real alt text, so it is what we
84  // emit. It is read `Undigested` — `ExpansionLevel::Off`, kept as
85  // `DigestedData::Postponed` tokens — so nothing inside it expands: the
86  // author never sees a defect there under pdflatex (the class gobbles it), so
87  // expanding it only manufactures errors. Witness arXiv:2607.21760, whose
88  // `\D1 … \D5` (a copy-paste slip from the adjacent `alt=` text, which has
89  // plain `D1 … D5`) raised `Error:undefined:\D` for content we then dropped.
90  //
91  // `LaTeXML-common.xsl` L404-421 maps any `aria:*` attribute to `aria-*`
92  // under HTML5, so setting them here needs no XSLT change. WHICH element
93  // they land on, and which slot each argument fills, is the
94  // `before_construct` table below.
95  //
96  // `[short]` and `{long}` are two DISTINCT authored fields, so they get two
97  // distinct elements — never concatenated into one. Merging them yields a
98  // run-on ("Fly 1 and Fly 2 look identical. Fly 1 and fly 2 comparison
99  // shows…") and destroys the distinction, so no consumer can tell which text
100  // the author wrote as the brief alternative. `aria-describedby` takes a
101  // SPACE-SEPARATED ID LIST and is announced in order, so where both are
102  // referenced it is short first, each still individually addressable.
103  //
104  // The old binding emitted the short one ALONE, which is why
105  // `t/complex/acm_aria` recorded "Fly 1 and Fly 2 look identical" and lost the
106  // sentence that actually describes the figure.
107  //
108  // Carried as `<ltx:note>` — the document builder places a note inside the
109  // float's `<caption>`, where the old binding put it. An `<ltx:text>` is
110  // INLINE, so the builder auto-opens a `<p>` for it, and a `<p>` at figure
111  // level is tagged `ltx_figure_panel`, which makes the post-processor spin up
112  // a whole `ltx_flex_figure`/`ltx_flex_cell` wrapper around two hidden spans
113  // (inert, since the caption partitions it and the real subfigures keep their
114  // width class, but pure noise in the output).
115  //
116  // The footnote decoration a note normally gets — the `†` mark and the
117  // `<role>: ` type prefix — is suppressed for these by a dedicated template in
118  // `LaTeXML-meta-xhtml.xsl` keyed on `ltx_acm_description`, so the referenced
119  // text stays clean for assistive technology.
120  //
121  // The short description needs its OWN id: it is referenced by
122  // `aria-describedby` whenever its text is not the one that became the
123  // image's `@alt`. It is derived in `properties` rather than written as
124  // `#id-short` in the template, which does NOT work: a `#name` hole runs to
125  // the end of the identifier, so `#id-short` names a property called
126  // `id-short` — absent, so the element silently emits NO xml:id at all —
127  // instead of `#id` followed by a literal `-short`. That produced a dangling
128  // `aria-describedby`, a reference resolving to nothing. Ids use a `-short`
129  // suffix rather than a dotted one so they need no escaping in a CSS
130  // selector.
131  // acmart.cls:895 `\Description` typesets nothing, so authors leave unescaped what LaTeX would reject in text
132  // (`\Description{Complaint_Loss}`, 2502.07049): outside math, `_ ^ & #` are the characters they show, and the
133  // description is digested as written otherwise.
134  DefConstructor!("\\lx@acm@description[]{}",
135    "^^?#1(<ltx:note xml:id='#shortid' class='ltx_nodisplay ltx_acm_description_short'>#1</ltx:note>)()\
136     <ltx:note xml:id='#id' class='ltx_nodisplay ltx_acm_description'>#2</ltx:note>",
137    sizer => sub[_whatsit] { Ok(out_of_line_size()) },
138    properties => {
139      let mut props = RefStepCounter!("acmlabel")?;
140      if let Some(id) = props.get("id") {
141        let short = format!("{id}-short");
142        props.insert("shortid", Stored::from(short));
143      }
144      // Perl acmart.cls.ltxml L79 sets these so the hidden note claims no
145      // layout space; carried over.
146      props.insert("width", Stored::from(Dimension!("0pt")));
147      props.insert("height", Stored::from(Dimension!("0pt")));
148      Ok(props)
149    },
150    // acmart's own documentation: "Unlike \caption, which is used alongside the
151    // image, \Description is intended to be used INSTEAD OF the image." So a
152    // `\Description` is a TEXT ALTERNATIVE — and the thing it is an alternative
153    // TO is the image, not the float. It therefore lands on the IMAGE:
154    //
155    //   `\Description[s]{l}`           → img @alt = s, aria-describedby → l
156    //   `\Description{l}`, l plain     → img @alt = l (it replaces the image)
157    //   `\Description{l}`, l w/ markup → img @alt untouched, describedby → l
158    //
159    // `@alt` (via `ltx:graphics/@description`, `LaTeXML-misc-xhtml.xsl` L167-171)
160    // rather than `aria-label`, because that is the attribute an `<img>` has for
161    // exactly this purpose and the one assistive technology expects there.
162    //
163    // NOT `aria:label` on the `<ltx:figure>`, which is what this binding did
164    // before: `aria-label` sets the accessible NAME, and a float's name is its
165    // caption, so labelling the figure with the description DISPLACED
166    // "Figure 1. caption text" and hid the caption from a screen reader
167    // (reviewer report, brucemiller/LaTeXML#430 r3674103638; the fix that
168    // review asks for is precisely "label + description should be attached to
169    // the first image in the figure … and the `<img>` tag eventually gets
170    // `@alt` and not `@aria-label`").
171    //
172    // `@alt` is a plain string and cannot carry markup, so the slot is chosen by
173    // the description's tokens as written (its reversion): a control sequence
174    // (or an active/`$`/`^`/`_` token) means real markup, so the block carries
175    // it. The argument itself is digested, as Perl's `{}` (acmart.cls.ltxml:78)
176    // and LaTeX do: read `Undigested`, the block printed its markup as raw TeX
177    // (`\sysname{}`, 2403.09168; 2401.04997, 2304.01062, 2312.11013).
178    //
179    // The block is always emitted, so the text stays addressable, but it is
180    // referenced only when it is not already the `@alt` — otherwise the same
181    // sentence would be announced twice. An unreferenced hidden block is inert:
182    // `display:none` content is announced only when something references it.
183    //
184    // TWO CASES HAVE NO IMAGE TO USE. An author's annotation is never dropped,
185    // so it goes to the next best host — the enclosing float — as
186    // `aria:describedby`, which supplements the name instead of replacing it,
187    // so the caption survives either way. Both `Warn!`, because the result is
188    // second-best and the author can do something about it:
189    //
190    //  * No `ltx:graphics` in the float at all — a figure built from tabular,
191    //    text or TikZ content (`t/complex/acm_aria` is one), a `table` float,
192    //    or an empty one. There is no image to be the alternative to.
193    //  * MORE than one. A `\Description` is scoped to the whole float, so on a
194    //    multi-panel figure it describes the ensemble; making it panel 1's
195    //    `@alt` would assert that one sentence is the alternative for one
196    //    panel, which is a claim the author never made. The review says "the
197    //    first image"; we narrow that to the case where "first" is also "only",
198    //    where it is unambiguous.
199    //
200    // We can only see the graphics ALREADY BUILT when `\Description` is
201    // constructed, so a `\Description` written BEFORE its `\includegraphics`
202    // falls into the first case. That is the safe direction to fail — the
203    // description is still announced, just not as the image's alternative — the
204    // warning names it as a possible cause, and acmart's own documentation puts
205    // `\Description` after the graphic.
206    //
207    // acmart's NEWER mechanism for the same purpose is `\includegraphics[alt=…]`
208    // (switched on by `\DocumentMetadata`), handled in `graphicx_sty.rs` and
209    // landing on the same `description` attribute. When an author uses BOTH —
210    // e.g. arXiv:2607.21760, which repeats the same paragraph in each — the
211    // explicit `alt=` WINS: it names one image, while `\Description` names the
212    // float, so the more specific statement stands. `\Description` then only
213    // adds its `aria-describedby` references.
214    before_construct => sub[document, whatsit] {
215      let Some(id) = whatsit.get_property("id").map(|v| v.to_string()) else {
216        return Ok(());
217      };
218      // Does the long description contain markup, or is it plain text?
219      let long_is_plain = whatsit
220        .get_arg(2)
221        .map(|d| d.revert())
222        .transpose()?
223        .is_some_and(|tks| {
224          tks.unlist_ref().iter().all(|t| {
225            matches!(
226              t.get_catcode(),
227              Catcode::LETTER | Catcode::OTHER | Catcode::SPACE | Catcode::EOL
228            )
229          })
230        });
231      // `\Description[]{…}` gives an empty optional argument, which the
232      // template's `?#1(…)` declines to emit — so there is no short note and
233      // no id to reference. Treat it as absent.
234      let short = whatsit
235        .get_arg(1)
236        .map(|d| d.to_attribute())
237        .filter(|s| !s.is_empty());
238      let short_id = whatsit.get_property("shortid").map(|v| v.to_string());
239      let Some(mut figure) = document.get_element() else {
240        return Ok(());
241      };
242      let mut graphics = document.findnodes(".//ltx:graphics", Some(&figure));
243      // Exactly one image: it IS the figure, so the description is its
244      // alternative. Zero or several: see the note above, keep it on the float.
245      let lone_graphic = (graphics.len() == 1).then(|| graphics.remove(0));
246      let mut described_by: Vec<String> = Vec::new();
247      match lone_graphic {
248        Some(mut graphic) => {
249          // `\includegraphics[alt=…]` already spoke for this image; it is the
250          // more specific statement, so it stands.
251          let alt = if graphic.get_attribute("description").is_some() {
252            None
253          } else if short.is_some() {
254            short.clone()
255          } else if long_is_plain {
256            whatsit.get_arg(2).map(|d| d.to_attribute())
257          } else {
258            None
259          };
260          match alt {
261            // Something of ours became the alternative. Whatever did NOT is
262            // still worth announcing, so reference it.
263            Some(text) => {
264              document.set_attribute(&mut graphic, "description", &text)?;
265              if short.is_some() {
266                described_by.push(id);
267              }
268            },
269            // Nothing of ours is the alternative — the author's own `alt=`
270            // holds it, or the lone description carries markup. Reference
271            // everything we emitted.
272            None => {
273              if short.is_some() && let Some(sid) = short_id {
274                described_by.push(sid);
275              }
276              described_by.push(id);
277            },
278          }
279          if !described_by.is_empty() {
280            add_describedby(document, &mut graphic, &described_by)?;
281          }
282        },
283        // Nowhere better to put it. Attach to the enclosing element rather than
284        // drop the author's annotation, and say so — the text is still
285        // announced, but not as the alternative for any one image, which is
286        // what a `\Description` is for.
287        None => {
288          if short.is_some() && let Some(sid) = short_id {
289            described_by.push(sid);
290          }
291          described_by.push(id);
292          add_describedby(document, &mut figure, &described_by)?;
293          let host = figure.get_name();
294          // The command the author wrote (`define_float_description`'s).
295          let command = lookup_string("float_description_command");
296          if graphics.is_empty() && host == "table" {
297            // NOT a fallback: a table float has no image BY CONSTRUCTION, and
298            // acmart asks for a `\Description` on every float — so the table
299            // itself is where the description belongs, and this is the author
300            // doing the right thing. Warning here demoted otherwise-clean
301            // papers: 27 of 45 sampled `no_problem -> warning` regressions in
302            // the 2026-07-30 sandbox-arxiv-2605 rerun were this one message.
303            // Report it (the attachment point is worth knowing), as INFO.
304            Info!("aria", &command,
305              &s!("attached to the enclosing <table> as aria:describedby — a \
306                   table has no image to describe, so this is where the \
307                   description belongs"));
308          } else {
309            let why = if !graphics.is_empty() {
310              s!("it holds more than one image, so which one is described is ambiguous")
311            } else if host == "figure" {
312              s!("the float has no image to describe (a {command} written \
313                  before its \\includegraphics, or a figure with no graphic)")
314            } else {
315              // Not a float at all — `\Description` outside a figure/table,
316              // where acmart's own `\@Description@present` bookkeeping expects it.
317              s!("it is outside any figure or table, so there is no image to describe")
318            };
319            Warn!("unexpected", &command,
320              &s!("attached to the enclosing <{host}> as aria:describedby, because \
321                   {why}. The text is still announced, but it is not any image's \
322                   alt text; put a {command} after the \\includegraphics it \
323                   describes, or use \\includegraphics[alt=...] per image"));
324          }
325        },
326      }
327    },
328    // A `\label` after the `\Description` names the float, as in LaTeX, where
329    // `\Description` typesets nothing (acmart.cls:895). `\label` takes the
330    // nearest element with an id from the current node's last child
331    // (`float_to_label`, Perl Document.pm floatToLabel), and that was the note,
332    // which moves into the caption only when the float closes (`arrange_panels`):
333    // 2403.09168's ten labels sat on hidden notes, so each `\ref` pointed at one
334    // (Perl 0.8.8 the same, KNOWN_PERL_ERRORS #535). So the caption already in the
335    // float before it — the last, the description's own — takes the notes now,
336    // where the float's close would put them.
337    after_construct => sub[document, whatsit] {
338      let ids: Vec<String> = ["id", "shortid"]
339        .iter()
340        .filter_map(|key| whatsit.get_property(key).map(|v| v.to_string()))
341        .collect();
342      let Some(float) = document.get_element() else {
343        return Ok(());
344      };
345      let children = float.get_child_elements();
346      let Some(mut caption) = children
347        .iter()
348        .rev()
349        .find(|c| c.get_name() == "caption").cloned() else {
350        return Ok(());
351      };
352      for mut note in children.into_iter().filter(|c| {
353        c.get_name() == "note" && c.get_attribute_ns("id", XML_NS).is_some_and(|id| ids.contains(&id))
354      }) {
355        note.unlink_node();
356        caption.add_child(&mut note)?;
357      }
358    }
359  );
360  Ok(())
