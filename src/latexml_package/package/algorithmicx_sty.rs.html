175/// Open the `ltx:listingline` of an algorithmicx `\item`, with its id and tags, or — inside a line still open
176/// (algpseudocodex's `varwidth` box, see `\lx@algorithmicx@@item`) — break that line.
177///
178/// The nearest listing bounds the walk, as algorithm2e's `line_reach` does: in a listing nested in a line (an
179/// `{algorithmic}` inside an algorithm2e float's line or inside another `{algorithmic}`'s `\State`) a line opens in
180/// that listing; walking on to the outer line put breaks and bare text in the inner listing (112 TooManyErrors papers
181/// of run 336: 1203.4481, 2410.01553, 2510.14887, 1604.06452).
182fn open_algorithmicx_line(document: &mut Document, props: &SymHashMap<Stored>) -> Result<()> {
183  let open_line = document.maybe_close_element("ltx:listingline")?.is_none() && {
184    let mut n = Some(document.get_node().clone());
185    let mut found = false;
186    while let Some(node) = n {
187      let qname = document::get_node_qname(&node);
188      if qname == pin!("ltx:listingline") {
189        found = true;
190        break;
191      }
192      if qname == pin!("ltx:listing") {
193        break;
194      }
195      n = node.get_parent();
196    }
197    found
198  };
199  if open_line {
200    document.insert_element("ltx:break", Vec::new(), None)?;
201  } else {
202    let mut attrs: HashMap<String, String> = HashMap::default();
203    if let Some(id) = props.get("id") {
204      attrs.insert("xml:id".into(), id.to_string());
205    }
206    document.open_element("ltx:listingline", Some(attrs), None)?;
207    if let Some(tags) = props.get("tags") {
208      let digested: Option<Digested> = tags.into();
209      if let Some(ref d) = digested {
210        document.absorb(d, None)?;
211      }
212    }
213  }
214  Ok(())
