1//! tcolorbox.sty — colored and framed text boxes
2//! Perl: tcolorbox.sty.ltxml
3use crate::prelude::*;
4
5#[rustfmt::skip]
6LoadDefinitions!({
7  // used in tcbbreakable.code.tex assuming it was defined
8  DefRegister!("\\doublecol@number" => Number::new(0));
9  // Ensure only unbreakable mode is possible.
10  // Perl: locked => 1 prevents raw TeX tcbbreakable.code.tex from overriding
11  // with the real breakable implementation (uses output routines → infinite loop).
12  DefMacro!("\\tcb@init@breakable", "\\tcb@init@unbreakable", locked => true);
13
14  // Perl 93f875a6: pre-define \tcb@use@autoparskip before raw TeX loading,
15  // as pgfkeys initialization may not complete and the \AtBeginDocument hook
16  // at tcolorbox.sty:1142 would call it undefined.
17  DefMacro!("\\tcb@use@autoparskip", "\\relax");
18
19  RequirePackage!("expl3");
20  RequirePackage!("xparse");
21
22  InputDefinitions!("tcolorbox", noltxml => true, extension => Some(Cow::Borrowed("sty")));
23
24  // Suppress tcolorbox's library version check. The Rust kpathsea binding
25  // may resolve a different tcolorbox.sty version than the library files,
26  // causing spurious "tcolorbox is not installed correctly" errors.
27  // Make the check a no-op — the versions are always compatible in practice.
28  DefMacro!("\\tcb@check@library@version", "", locked => true);
29
30  // \newtcblisting[init-options]{name}[N][default]{tcb-options} (tcolorbox
31  // `listings`/`minted` library) — a code-listing box. Its box styling is purely
32  // visual; what matters for the logical output is the code BODY, which must be
33  // captured verbatim and CLOSED at \end{name}. The raw library's body capture
34  // does not integrate with LaTeXML's verbatim reader, so the listing runs past
35  // its \end{name} and swallows following content (sections leak into
36  // <ltx:verbatim>). Delegate to listings' \lstnewenvironment (same
37  // name/[N][default] shape; the leading [init-options] and trailing tcb options
38  // are dropped), whose verbatim reader terminates correctly. `locked` so a later
39  // raw `\tcbuselibrary{listings}` can't clobber it.
40  //
41  // The signature MUST include the leading `[init-options]` optional (real
42  // tcolorbox `\NewDocumentCommand \__tcobox_new_tcolorbox:w { m +O{} m o +o +m }`,
43  // mirrored by tcblistings): omitting it made a call like
44  // `\newtcblisting[auto counter,...]{promptbox}[2][]{...}` read `[auto counter,...]`
45  // as the mandatory name, so `{promptbox}` was never defined and its verbatim body
46  // tokenized as normal LaTeX — every `_`/`^` in the listing then raising
47  // "Script can only appear in math mode" (Perl raw-loads the real macro and is
48  // clean). Witness: 2606.00555 (leading init-options). Prior witnesses use no
49  // leading optional and are unaffected: 2507.00833 (ar5iv #569/#570), 2402.13846 (#504).
50  DefMacro!(
51    "\\newtcblisting[]{}[][]{}",
52    "\\lstnewenvironment{#2}[#3][#4]{}{}",
53    locked => true
54  );
55});
