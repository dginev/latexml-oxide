5fn is_page_spec(s: &str) -> bool {
6  let s = s.trim();
7  if s.is_empty() {
8    return false;
9  }
10  let inner = if s.starts_with('{') && s.ends_with('}') && s.len() >= 2 {
11    &s[1..s.len() - 1]
12  } else {
13    s
14  };
15  let inner = inner.trim();
16  if inner == "-" || inner == "last" {
17    return true;
18  }
19  let without_last = inner.replace("last", "");
20  !without_last.is_empty()
21    && without_last
22      .chars()
23      .all(|c| c.is_ascii_digit() || c == '-' || c == ',' || c.is_ascii_whitespace())
24}

26fn parse_file_page_list(input: &str) -> Vec<(String, Option<String>)> {
27  let mut entries = Vec::new();
28  let mut current = String::new();
29  let mut depth = 0;
30  for ch in input.chars() {
31    match ch {
32      '{' => {
33        depth += 1;
34        current.push(ch);
35      },
36      '}' => {
37        if depth > 0 {
38          depth -= 1;
39        }
40        current.push(ch);
41      },
42      ',' if depth == 0 => {
43        let trimmed = current.trim();
44        if !trimmed.is_empty() {
45          entries.push(trimmed.to_string());
46        }
47        current.clear();
48      },
49      _ => {
50        current.push(ch);
51      },
52    }
53  }
54  let trimmed = current.trim();
55  if !trimmed.is_empty() {
56    entries.push(trimmed.to_string());
57  }
58
59  let mut pairs = Vec::new();
60  let mut current_file: Option<String> = None;
61
62  for entry in entries {
63    if let Some(file) = current_file.take() {
64      if is_page_spec(&entry) {
65        let ps = entry.trim();
66        let ps = if ps.starts_with('{') && ps.ends_with('}') && ps.len() >= 2 {
67          ps[1..ps.len() - 1].trim().to_string()
68        } else {
69          ps.to_string()
70        };
71        pairs.push((file, Some(ps)));
72      } else {
73        pairs.push((file, None));
74        current_file = Some(entry);
75      }
76    } else {
77      current_file = Some(entry);
78    }
79  }
80  if let Some(file) = current_file {
81    pairs.push((file, None));
82  }
83
84  pairs
85}

87/// The pages an `\includepdf` page list names, in order (pdfpages.sty's `pages` key: a comma list of `N`, `N-M`
88/// (descending when `M < N`), `N-`, `-M`, `-`, `last`, and `{}`, a blank page, which places nothing here). `None` for a
89/// list not understood, or one that needs the page count when `count` cannot read it.
90fn include_pdf_pages(spec: &str, count: impl FnOnce() -> Option<u32>) -> Option<Vec<u32>> {
91  let spec = spec.trim();
92  // (one group around the whole list is the list: `{1,3}`, not `{},1,{}`'s first and last blank pages)
93  let spec = match outer_group(spec) {
94    Some(inner) if !inner.trim().is_empty() => inner,
95    _ => spec,
96  };
97  let mut count = Some(count);
98  let mut known: Option<u32> = None;
99  let mut last = || -> Option<u32> {
100    if let Some(count) = count.take() {
101      known = count();
102    }
103    known
104  };
105  let mut bound = |text: &str, open: u32| -> Option<u32> {
106    match text.trim() {
107      "" if open == 0 => last(),
108      "" => Some(open),
109      "last" => last(),
110      number => number.parse().ok().filter(|n| *n > 0),
111    }
112  };
113  let mut pages = Vec::new();
114  for item in spec.split(',').map(str::trim) {
115    if item.is_empty() || item == "{}" {
116      continue;
117    }
118    match item.split_once('-') {
119      Some((from, to)) => {
120        let from = bound(from, 1)?;
121        let to = bound(to, 0)?;
122        if from <= to {
123          pages.extend(from..=to);
124        } else {
125          pages.extend((to..=from).rev());
126        }
127      },
128      None => pages.push(bound(item, 0)?),
129    }
130  }
131  Some(pages)
132}

134/// The pages of `pages` the file has (all, when its page `count` is not known), and those past its last.
135fn split_existing_pages(pages: Vec<u32>, count: Option<u32>) -> (Vec<u32>, Vec<u32>) {
136  match count {
137    Some(count) => pages.into_iter().partition(|page| *page <= count),
138    None => (pages, Vec::new()),
139  }
140}

142/// The text inside `text`'s braces when one group encloses it all.
143fn outer_group(text: &str) -> Option<&str> {
144  let inner = text.strip_prefix('{')?.strip_suffix('}')?;
145  let mut depth = 0i32;
146  for ch in inner.chars() {
147    match ch {
148      '{' => depth += 1,
149      '}' if depth == 0 => return None,
150      '}' => depth -= 1,
151      _ => {},
152    }
153  }
154  (depth == 0).then_some(inner)
155}

157/// The PDF `name` names, found as a graphic is (`\graphicspath`, the source directory).
158fn pdf_file(name: &str) -> Option<std::path::PathBuf> {
159  let found = image_candidates(name)
160    .split(',')
161    .find(|found| found.to_ascii_lowercase().ends_with(".pdf"))
162    .map(str::to_string)?;
163  let path = std::path::Path::new(&found);
164  let path = if path.is_relative() {
165    std::path::Path::new(&lookup_string("SOURCEDIRECTORY")).join(path)
166  } else {
167    path.to_path_buf()
168  };
169  path.is_file().then_some(path)
170}

173LoadDefinitions!({
174  // Perl: pdfpages.sty.ltxml
175  RequirePackage!("ifthen");
176  RequirePackage!("calc");
177  RequirePackage!("eso-pic");
178  RequirePackage!("graphicx");

180  // pdfpages.sty:205 `\includepdfset{keyvals}` accumulates default options
181  // for later `\includepdf`s; ours ignores them, so the
182  // defaults are absorbed (tutodoc-en/fr :1339; Perl pdfpages.sty.ltxml lacks
183  // it too). Guard: `perfect_kernel_batch56::includepdfset_is_absorbed`.
184  DefMacro!("\\includepdfset{}", "");
185  // `\includepdf[pages=<list>]{file}` places the file's pages, one a page (pdfpages.sty: `pages` defaults to 1, `-`
186  // is all of them, `{}` a blank page). Each page is the file's graphic at that page, `\includegraphics[page=N]`, as
187  // pdfpages itself includes it (pdfpages.sty `\AM@includegraphics`), with the graphicx keys given (`scale`, `angle`,
188  // `width`, `height`, `trim`, `clip`, `viewport`, `keepaspectratio`); an open range (`-`, `3-`, `last`) reads the
189  // page count from the file. Perl (pdfpages.sty.ltxml:30-36) and the former binding printed "See pages - of
190  // ⟨link⟩" with the PDF as a resource, which remains for a PDF not found (pdflatex's "Cannot find file") or a
191  // list not understood:
192  // OXIDIZED_DESIGN_DIVERGENCES #846. Guard `perfect_kernel_batch78j::includepdf_places_its_pages`.
193  DefMacro!("\\includepdf OptionalKeyVals Semiverbatim", sub[args] {
194    let mut it = args.into_iter();
195    let kv_arg = it.next().unwrap();
196    let file: Tokens = it.next().unwrap().into();
197    // (the name and the page list expanded, as pdfpages reads them: `\foreach \x {…}` loops a page `pages={\x}`,
198    // a name kept in a macro, 2605.05847, 2406.13026)
199    let name = Expand!(file).to_string().trim().to_string();
200    let mut spec = None;
201    let mut graphic_keys = Vec::new();
202    if let ArgWrap::KV(ref kv) = kv_arg {
203      for (key, value) in kv.get_pairs() {
204        match key.as_str() {
205          "pages" => {
206            spec = Some(match value.as_tokens() {
207              Ok(Some(tokens)) => Expand!(tokens.into_owned()).to_string(),
208              _ => value.to_string(),
209            })
210          },
211          "scale" | "angle" | "width" | "height" | "trim" | "clip" | "viewport" | "keepaspectratio" => {
212            graphic_keys.push(s!("{key}={{{}}}", value.to_string()))
213          },
214          _ => {},
215        }
216      }
217    }
218    let mut out = String::new();
219    let (mut count, mut file_name) = (None, name.clone());
220    let pages = pdf_file(&name).and_then(|path| {
221      count = read_pdf_page_count(&path);
222      if let Some(found) = path.file_name() {
223        file_name = found.to_string_lossy().into_owned();
224      }
225      include_pdf_pages(spec.as_deref().unwrap_or("1"), || count)
226    });
227    match pages {
228      Some(pages) => {
229        // A page past the file's last is an error and places nothing (pdfpages.sty:909-920 `\AM@checkpagenumber`, "Page 6
230        // of `five.pdf' does not exist"; pdflatex then fails to include it); a count not read keeps every page.
231        let (pages, missing) = split_existing_pages(pages, count);
232        for page in missing {
233          Error!("latex", "(pdfpages)", s!("Package pdfpages Error: Page {page} of `{file_name}' does not exist."));
234        }
235        for page in pages {
236          let mut keys = graphic_keys.clone();
237          keys.push(s!("page={page}"));
238          out.push_str(&s!("\\par\\includegraphics[{}]{{{name}}}\\par", keys.join(",")));
239        }
240      },
241      None => {
242        out.push_str("\\lx@includepdf@link");
243        if let Some(spec) = spec {
244          out.push_str(&s!("[{spec}]"));
245        }
246        out.push_str(&s!("{{{name}}}"));
247      },
248    }
249    Ok(mouth::tokenize_internal(TeXString::assembled(out)))
250  });
251  // Perl pdfpages.sty.ltxml L30-36: a link to the PDF, copied as a resource, "See [pages X of ]<ref>". The file name
252  // is Semiverbatim, as Perl graphicx.sty.ltxml:52 reads `\@includegraphicx`'s: Perl's `{}` tokenized
253  // `\includepdf{a_b.pdf}` under document catcodes and errored "Script _ can only appear in math mode" (latex4wp,
254  // `pandoc_template.pdf`; pdflatex reads the name as a file name). Guard:
255  // `binding_singletons_56::includepdf_file_name_is_semiverbatim`.
256  DefConstructor!("\\lx@includepdf@link[] Semiverbatim",
257    "<ltx:resource src='#src' type='application/pdf'/>See #pages<ltx:ref href='#src'>#src</ltx:ref>",
258    properties => sub[args] {
259      let pages = args[0].as_ref().map(|d| format!("pages {d} of ")).unwrap_or_default();
260      let src = args[1].as_ref().map(|d| d.to_string()).unwrap_or_default();
261      Ok(stored_map!("src" => src, "pages" => pages))
262    });
263
264  // pdfpages.sty:262 `\includepdfmerge[opts]{file-page-list}` iterates over a
265  // comma-separated list of filenames and optional page specs (e.g. `doc.pdf, 1-9, other.pdf`),
266  // delegating each file to `\includepdf` with its specific or default page range.
267  DefMacro!("\\includepdfmerge OptionalKeyVals {}", sub[args] {
268    let mut it = args.into_iter();
269    let kv_arg = it.next().unwrap();
270    let file_list_tokens: Tokens = it.next().unwrap().into();
271    let file_list_str = file_list_tokens.to_string();
272
273    let pairs = parse_file_page_list(&file_list_str);
274    if pairs.is_empty() {
275      return Ok(Tokens!());
276    }
277
278    // Extract any existing options from kv_arg (other than pages).
279    let mut base_opts: Vec<(String, String)> = Vec::new();
280    let default_pages = if let ArgWrap::KV(ref kv) = kv_arg {
281      for (k, v) in kv.get_pairs() {
282        if k != "pages" {
283          base_opts.push((k.clone(), v.to_string()));
284        }
285      }
286      kv.get_value("pages").map(|v| v.to_string())
287    } else {
288      None
289    };
290
291    let mut tex_buf = String::new();
292    for (file, page_spec) in pairs {
293      tex_buf.push_str("\\includepdf");
294      let active_pages = page_spec.or_else(|| default_pages.clone());
295      let mut file_opts = base_opts.clone();
296      if let Some(ps) = active_pages {
297        file_opts.push(("pages".to_string(), ps));
298      }
299      if !file_opts.is_empty() {
300        tex_buf.push('[');
301        let opt_parts: Vec<String> = file_opts
302          .into_iter()
303          .map(|(k, v)| format!("{k}={{{v}}}"))
304          .collect();
305        tex_buf.push_str(&opt_parts.join(","));
306        tex_buf.push(']');
307      }
308      tex_buf.push('{');
309      tex_buf.push_str(&file);
310      tex_buf.push('}');
311    }
312
313    Ok(mouth::tokenize(TeXString::assembled(tex_buf)))
314  });
315});

321  #[test]
322  fn include_pdf_pages_reads_pdfpages_lists() {
323    let pages = |spec: &str| include_pdf_pages(spec, || Some(5));
324    assert_eq!(pages("1"), Some(vec![1]));
325    assert_eq!(pages("-"), Some(vec![1, 2, 3, 4, 5]));
326    assert_eq!(pages("{2-4}"), Some(vec![2, 3, 4]));
327    assert_eq!(pages("3-"), Some(vec![3, 4, 5]));
328    assert_eq!(pages("-2"), Some(vec![1, 2]));
329    assert_eq!(pages("last"), Some(vec![5]));
330    assert_eq!(pages("last-3"), Some(vec![5, 4, 3]));
331    assert_eq!(pages("{},1,{},2,{}"), Some(vec![1, 2]));
332    assert_eq!(pages("{{},1,{},3}"), Some(vec![1, 3]));
333    assert_eq!(pages("x"), None);
334    assert_eq!(include_pdf_pages("-", || None), None);
335    assert_eq!(include_pdf_pages("2,4", || None), Some(vec![2, 4]));
336    // (`pages=4-7` of a 5-page file: pages 6 and 7 do not exist, an error each, pdfpages.sty:909-920)
337    assert_eq!(
338      split_existing_pages(pages("4-7").unwrap(), Some(5)),
339      (vec![4, 5], vec![6, 7])
340    );
341    assert_eq!(split_existing_pages(vec![2, 9], None), (vec![2, 9], vec![]));
342  }
