55/// The title-page fields PoS.cls and the PoS template's pos.sty v2 (`pos_sty`) share: `\ShortTitle{}`, the running
56/// head's title, and `\FullConference{name}`, the conference the title page names at its foot (pos.sty:362-369).
57#[rustfmt::skip]
58pub fn pos_title_page_fields() -> Result<()> {
59  DefMacro!("\\ShortTitle{}",     "\\lx@add@toctitle{#1}");
60  // \FullConference{name} — conference identification text. Round-34
61  // surpass-Perl: preserve as ltx:note frontmatter.
62  DefMacro!("\\FullConference{}",
63    "\\lx@add@frontmatter{ltx:note}[role=conference]{#1}");
64  Ok(())
65}
