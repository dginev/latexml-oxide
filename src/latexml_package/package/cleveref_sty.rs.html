153/// Port of raw cleveref's `\@crefname` core (`\newcommand\crefname[3]` →
154/// `\@crefname{cref}{type}{sg}{pl}{}`): define `\<prefix>@<type>@name` and
155/// `\<prefix>@<type>@name@plural` from the singular/plural arguments. `prefix` is
156/// `"cref"` (for `\crefname`) or `"Cref"` (for `\Crefname`). The names are stored as
157/// **raw** token bodies (never expanded), so markup such as `\textsc{lemma}` survives,
158/// matching cleveref's `\def`. The raw macro's `\toksdef`/`\expandafter` chains are not
159/// reproduced — the same clean approach `thmtools_sty.rs` uses for
160/// `\declaretheorem[refname=]`. Like that precedent, the cross-variant `\MakeUppercase`
161/// derivation (deriving `\Cref@…` from a lone `\crefname`) is not reproduced; provide
162/// `\Crefname` explicitly for the capitalised form.
163fn cref_define_name(
164  prefix: &str,
165  type_arg: Tokens,
166  singular: Tokens,
167  plural: Tokens,
168) -> Result<()> {
169  let ctype = do_expand(type_arg)?.to_string();
170  let ctype = ctype.trim();
171  def_macro(
172    T_CS!(s!("\\{}@{}@name", prefix, ctype)),
173    None,
174    singular,
175    None,
176  )?;
177  def_macro(
178    T_CS!(s!("\\{}@{}@name@plural", prefix, ctype)),
179    None,
180    plural,
181    None,
182  )?;
183  Ok(())
184}

186/// Resolve a cleveref type-name control sequence (`\cref@<type>@name`,
187/// `\Cref@<type>@name@plural`, …) to the tokens the `type_tag_formatter` emits.
188///
189/// When the primary CS is undefined and `theorem_fallback` is set, fall back to the
190/// theorem's stored heading `\lx@name@<type>`. This is a **surpass-Perl** divergence
191/// (OXIDIZED_DESIGN #131): real cleveref patches `\@ynthm`/`\@xnthm`/`\@othm` so that
192/// `\newtheorem{arch}{Architecture}` auto-registers "Architecture" as the cref name,
193/// but LaTeXML's `\newtheorem` (`define_new_theorem`) is a native primitive that never
194/// routes through those patches — so Perl and Rust alike leave `\cref@arch@name`
195/// undefined and `\cref{...}` renders bare "1" instead of "Architecture 1". LaTeXML
196/// already stores the heading as `\lx@name@<type>`, so the singular creftype/creftypecap
197/// formatters reuse it, matching the PDF. Only the *singular* names get the fallback:
198/// cleveref's theorem patches set only `cref@<type>@name@preamble` (never `@plural`).
199/// The heading is emitted verbatim; cleveref's first-letter `capitalize` case transform
200/// is not reproduced, so a lowercase-`\cref` under the default (non-`capitalize`) option
201/// keeps the heading's own case.
202///
203/// `\lx@name@<type>` is also set by `\floatname`/`\newfloat` (`float_sty.rs`), so custom
204/// floats get the same auto-naming — matching real cleveref; standard figure/table/equation
205/// keep their raw-cleveref primary name (fallback stays dormant). An explicit `\crefname`
206/// (now a real definition — see `cref_define_name`) populates the primary CS, so it wins
207/// over this fallback. Witness arXiv 2305.10391 (`\usepackage[capitalize,…]{cleveref}` +
208/// `\newtheorem{arch}{Architecture}`).
209fn cleverref_type_name(
210  type_arg: &str,
211  prefix: &str,
212  suffix: &str,
213  theorem_fallback: bool,
214) -> Tokens {
215  let ctype = cref_type(type_arg);
216  let cs = s!("\\{}@{}@{}", prefix, ctype, suffix);
217  if has_meaning(&T_CS!(&cs)) {
218    return Tokens!(T_CS!(&cs));
219  }
220  if theorem_fallback {
221    let name_cs = s!("\\lx@name@{}", ctype);
222    if has_meaning(&T_CS!(&name_cs)) {
223      return Tokens!(T_CS!(&name_cs));
224    }
225  }
226  Tokens!()
227}

229/// Perl: crefType($type) — resolve type alias
230fn cref_type(ctype: &str) -> String {
231  let alias_cs = s!("\\cref@{}@alias", ctype);
232  if has_meaning(&T_CS!(&alias_cs))
233    && let Ok(expanded) = do_expand(Tokens!(T_CS!(&alias_cs)))
234  {
235    return expanded.to_string();
236  }
237  ctype.to_string()
238}

240/// Perl: crefMulti($starred, $labels, $showtype, $capitalized)
241/// Generates tokens for \cref{label1,label2,...}
242/// Trim leading/trailing SPACE tokens from a label group.
243fn trim_space_tokens(tokens: Tokens) -> Tokens {
244  let mut v = tokens.unlist();
245  while v
246    .first()
247    .map(|t| t.get_catcode() == Catcode::SPACE)
248    .unwrap_or(false)
249  {
250    v.remove(0);
251  }
252  while v
253    .last()
254    .map(|t| t.get_catcode() == Catcode::SPACE)
255    .unwrap_or(false)
256  {
257    v.pop();
258  }
259  Tokens::new(v)
260}

262/// Split a Semiverbatim label argument on top-level OTHER commas, trimming
263/// surrounding spaces and dropping empties. Mirrors Perl cleveref `splitLabels`,
264/// which splits the **Tokens** object — NOT a stringified form. Critical for
265/// labels whose name contains a control word, e.g.
266/// `\cref{… the \SW moduli space …}` (`\SW` is a user `\newcommand`): the space
267/// after the control word `\SW` is consumed at tokenization, so the tokens are
268/// `… \SW moduli …` (no space token between). Stringifying and re-tokenizing
269/// (as the old code did via `\lx@cref{…}{label-string}`) rejoins them into a
270/// bogus `\SWmoduli` control sequence that then gets DIGESTED → `undefined`.
271/// Keeping the original tokens and splicing them straight into the `\lx@cref`
272/// invocation (as Perl's `Invocation(T_CS('\lx@cref'), …, $label_tokens)` does)
273/// avoids the round-trip entirely. Witness 1704.05859.
274fn split_label_tokens(tokens: Tokens) -> Vec<Tokens> {
275  let mut groups: Vec<Tokens> = Vec::new();
276  let mut cur: Vec<Token> = Vec::new();
277  for t in tokens.unlist() {
278    if t.get_catcode() == Catcode::OTHER && t.with_str(|s| s == ",") {
279      groups.push(Tokens::new(std::mem::take(&mut cur)));
280    } else {
281      cur.push(t);
282    }
283  }
284  groups.push(Tokens::new(cur));
285  groups
286    .into_iter()
287    .map(trim_space_tokens)
288    .filter(|g| !g.is_empty())
289    .collect()
290}

292fn cref_multi(
293  starred: bool,
294  label_tokens: Tokens,
295  showtype: bool,
296  capitalized: bool,
297) -> Result<Tokens> {
298  let labels = split_label_tokens(label_tokens);
299  let n = labels.len();
300  let mut out: Vec<Token> = Vec::new();
301
302  // Emit one `\lx@cref [*] {<show>} {<label tokens>}`. The `<show>` is a fixed
303  // internal string (no user macros) so tokenizing it is safe; the `<label>`
304  // is spliced from the ORIGINAL tokens (never re-tokenized), matching Perl's
305  // `Invocation(T_CS('\lx@cref'), $starred, T_OTHER(show), $label)`.
306  // `\lx@tilde` carries the inter-word `~` (HyperVerbatim in Perl).
307  let emit = |out: &mut Vec<Token>, show: &str, label: &Tokens| {
308    out.push(T_CS!("\\lx@cref"));
309    if starred {
310      out.push(T_OTHER!("*"));
311    }
312    out.push(T_BEGIN!());
313    out.extend(mouth::tokenize_internal(TeXString::assembled(show.to_string())).unlist());
314    out.push(T_END!());
315    out.push(T_BEGIN!());
316    out.extend(label.clone().unlist());
317    out.push(T_END!());
318  };
319
320  if n < 2 {
321    let show = if showtype {
322      if capitalized {
323        "creftypecap\\lx@tilde refnum"
324      } else {
325        "creftype\\lx@tilde refnum"
326      }
327    } else {
328      "refnum"
329    };
330    let empty = Tokens::new(Vec::new());
331    let label = labels.first().unwrap_or(&empty);
332    emit(&mut out, show, label);
333  } else {
334    let show = if showtype {
335      if capitalized {
336        "creftypepluralcap\\lx@tilde refnum"
337      } else {
338        "creftypeplural\\lx@tilde refnum"
339      }
340    } else {
341      "refnum"
342    };
343    emit(&mut out, show, &labels[0]);
344    if n == 2 {
345      out.push(T_CS!("\\crefpairconjunction"));
346      emit(&mut out, "refnum", &labels[1]);
347    } else {
348      for label in &labels[1..n - 1] {
349        out.push(T_CS!("\\crefmiddleconjunction"));
350        emit(&mut out, "refnum", label);
351      }
352      out.push(T_CS!("\\creflastconjunction"));
353      emit(&mut out, "refnum", &labels[n - 1]);
354    }
355  }
356  Ok(Tokens::new(out))
357}
