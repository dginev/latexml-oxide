530/// Whether document format `format` is text around its arguments, which a form at the target can hold: each command
531/// of its body typesets (not expandable, protected, or robust, `\protect\<cs>`, as `\textup`, `\ensuremath`), or is a
532/// macro of no arguments made only of such (`\S`, `~`). A macro that computes (`\StrCount`, `\nameCref`, `\cref`, a
533/// conditional) is not: cleveref typesets the format at the reference (cleveref.sty:1047-1071), where it reads what
534/// the target cannot know.
535fn cref_format_is_text(format: &Token) -> bool {
536  let Ok(Some(defn)) = lookup_definition(format) else {
537    return false;
538  };
539  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
540    return false;
541  };
542  // (the case changer cleveref puts first in a variant it derives, `\cref@othervariant`'s `\MakeUppercase`/
543  // `\MakeLowercase`, cleveref.sty:1283-1295, :1888-1904, is cleveref's own: the variant is the text it changes)
544  let body = body.unlist_ref();
545  let body = match body.split_first() {
546    Some((first, rest)) if [T_CS!("\\MakeUppercase"), T_CS!("\\MakeLowercase")].contains(first) => {
547      rest
548    },
549    _ => body,
550  };
551  body.iter().all(|token| cref_text_token(token, true))
552}

554/// One token of a format's body as [`cref_format_is_text`] reads it; `deep`, whether a macro of no arguments may be
555/// read through, once.
556fn cref_text_token(token: &Token, deep: bool) -> bool {
557  if !matches!(token.get_catcode(), Catcode::CS | Catcode::ACTIVE) {
558    return true;
559  }
560  // (a command let to a character is that character: `\egroup`, whose `}` a format may close a group with that it did
561  // not open, an Error where a reference prints it, as pdflatex's "Extra }, or forgotten \endgroup")
562  if matches!(lookup_meaning(token), Some(Stored::Token(_))) {
563    return true;
564  }
565  let Ok(Some(defn)) = lookup_definition(token) else {
566    return false;
567  };
568  if !defn.is_expandable() || defn.is_protected() {
569    return true;
570  }
571  if defn.get_conditional_type().is_some() {
572    return false;
573  }
574  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
575    return false;
576  };
577  let body = body.unlist_ref();
578  body.first() == Some(&T_CS!("\\protect"))
579    || (deep && defn.get_num_args() == 0 && body.iter().all(|token| cref_text_token(token, false)))
580}

602/// Removes the stray alignment tabs of a digested form (`&` outside an alignment, tex_tables.rs, an error where it is
603/// constructed); for each, whether it was in the cascade of a misplaced alignment command (a Warning there).
604fn remove_stray_tabs(digested: &Digested) -> Vec<bool> {
605  let is_tab = |boxed: &Digested| match boxed.data() {
606    DigestedData::Whatsit(whatsit) => {
607      let whatsit = whatsit.borrow();
608      (whatsit.get_definition().get_cs().get_catcode() == Catcode::ALIGN)
609        .then(|| whatsit.get_property("misplaced_cascade").is_some())
610    },
611    _ => None,
612  };
613  let mut removed = Vec::new();
614  match digested.data() {
615    DigestedData::List(list) => {
616      list.borrow_mut().boxes.retain(|boxed| match is_tab(boxed) {
617        Some(misplaced) => {
618          removed.push(misplaced);
619          false
620        },
621        None => true,
622      });
623      for boxed in list.borrow().boxes.iter() {
624        removed.extend(remove_stray_tabs(boxed));
625      }
626    },
627    DigestedData::Whatsit(whatsit) => {
628      let whatsit = whatsit.borrow();
629      for arg in whatsit.get_args().iter().flatten() {
630        removed.extend(remove_stray_tabs(arg));
631      }
632      if let Some(Stored::Digested(body)) = whatsit.get_property("body").as_deref() {
633        removed.extend(remove_stray_tabs(body));
634      }
635    },
636    _ => {},
637  }
638  removed
639}

898/// Port of raw cleveref's `\@crefname` core (`\newcommand\crefname[3]` →
899/// `\@crefname{cref}{type}{sg}{pl}{}`): define `\<prefix>@<type>@name` and
900/// `\<prefix>@<type>@name@plural` from the singular/plural arguments. `prefix` is
901/// `"cref"` (for `\crefname`) or `"Cref"` (for `\Crefname`). The names are stored as
902/// **raw** token bodies (never expanded), so markup such as `\textsc{lemma}` survives,
903/// matching cleveref's `\def`. The raw macro's `\toksdef`/`\expandafter` chains are not
904/// reproduced — the same clean approach `thmtools_sty.rs` uses for
905/// `\declaretheorem[refname=]`. The other variant, when undefined, is derived as
906/// cleveref.sty:1296-1325 derives it (`\cref@othervariant`, :1283-1295): `\Cref@<type>@name` is
907/// `\MakeUppercase` before the `\crefname` singular (its first token), and `\cref@…` from a lone
908/// `\Crefname` is `\MakeLowercase` before it, or the name itself under `capitalise`
909/// (`\crefname{appendixtable}{Appendix Table}{…}`: `\Cref` printed no name; 2404.11739).
910fn cref_define_name(
911  prefix: &str,
912  type_arg: Tokens,
913  singular: Tokens,
914  plural: Tokens,
915) -> Result<()> {
916  let ctype = do_expand(type_arg)?.to_string();
917  let ctype = ctype.trim();
918  def_macro(
919    T_CS!(s!("\\{}@{}@name", prefix, ctype)),
920    None,
921    singular.clone(),
922    None,
923  )?;
924  def_macro(
925    T_CS!(s!("\\{}@{}@name@plural", prefix, ctype)),
926    None,
927    plural.clone(),
928    None,
929  )?;
930  let other = if prefix == "cref" { "Cref" } else { "cref" };
931  if !has_meaning(&T_CS!(s!("\\{}@{}@name", other, ctype))) {
932    let change_case = if prefix == "cref" {
933      Some(T_CS!("\\MakeUppercase"))
934    } else if IsDefined!(&T_CS!("\\if@cref@capitalise"))
935      && Expand!(TokenizeInternal!(r"\if@cref@capitalise1\else0\fi"))
936        .to_string()
937        .trim()
938        == "1"
939    {
940      None
941    } else {
942      Some(T_CS!("\\MakeLowercase"))
943    };
944    // Both names change case, or neither, as the singular is empty or not (cleveref.sty:1309 `\expandafter\ifx\@tempa
945    // \@empty`, the singular alone): `\crefname{x}{}{xs}` derives the plural unchanged.
946    let change_case = change_case.filter(|_| !singular.unlist_ref().is_empty());
947    // cleveref applies the case change to the name's first token (`\expandafter\@tempd\@tempa`, :1311-1318): a letter
948    // is changed alone ("eq." → "Eq."), a macro is read whole by `\MakeUppercase` (`\crefname{proposition}
949    // {\nameProposition}…` → "PROPOSITION", 1704.08542). A command that takes its own argument (`\crefname{line}
950    // {\text{line}}…`, 2604.22361; 2206.03535) would be split from it by the kernel's grouped `\MakeUppercase` (latex.ltx
951    // :22385 `{{…}}`), an error in pdflatex too, but only where that variant is typeset; our tags typeset every variant,
952    // so a name led by a command is changed whole, braced.
953    let derive = |name: Tokens| -> Tokens {
954      match change_case {
955        Some(cs) => {
956          let mut tokens = vec![cs];
957          if name
958            .unlist_ref()
959            .first()
960            .is_some_and(|t| t.code.is_active_or_cs())
961          {
962            tokens.push(T_BEGIN!());
963            tokens.extend(name.unlist());
964            tokens.push(T_END!());
965          } else {
966            tokens.extend(name.unlist());
967          }
968          Tokens::new(tokens)
969        },
970        _ => name,
971      }
972    };
973    def_macro(
974      T_CS!(s!("\\{}@{}@name", other, ctype)),
975      None,
976      derive(singular),
977      None,
978    )?;
979    def_macro(
980      T_CS!(s!("\\{}@{}@name@plural", other, ctype)),
981      None,
982      derive(plural),
983      None,
984    )?;
985  }
986  Ok(())
987}

1126/// Split a Semiverbatim label argument on top-level OTHER commas, trimming
1127/// surrounding spaces and dropping empties. Mirrors Perl cleveref `splitLabels`,
1128/// which splits the **Tokens** object — NOT a stringified form. Critical for
1129/// labels whose name contains a control word, e.g.
1130/// `\cref{… the \SW moduli space …}` (`\SW` is a user `\newcommand`): the space
1131/// after the control word `\SW` is consumed at tokenization, so the tokens are
1132/// `… \SW moduli …` (no space token between). Stringifying and re-tokenizing
1133/// (as the old code did via `\lx@cref{…}{label-string}`) rejoins them into a
1134/// bogus `\SWmoduli` control sequence that then gets DIGESTED → `undefined`.
1135/// Keeping the original tokens and splicing them straight into the `\lx@cref`
1136/// invocation (as Perl's `Invocation(T_CS('\lx@cref'), …, $label_tokens)` does)
1137/// avoids the round-trip entirely. Witness 1704.05859.
1138///
1139/// A comma inside braces is part of its label: cleveref takes each label off its stack with the delimited
1140/// `\@cref@stack@top#1,#2\@nil` (cleveref.sty:388), which matches only a comma at brace depth 0, so
1141/// `\Cref{claim:copy_K_{2,4}}` names the one label `\label{claim:copy_K_{2,4}}` (2304.09155). Perl's `splitLabels`
1142/// (cleveref.sty.ltxml:42-45) splits the string at every comma (KPE #1173).
1143///
1144/// The list is split once expanded: cleveref pushes `\edef\@tempa{#2}` onto its stack (cleveref.sty:947), so a macro
1145/// holding a list names each label in it — 2108.04620's `\cref{\seq_use:Nn \g_cflist_pending {,}}` (12 dangling
1146/// references to `def:dist:set,def:norm` and the like as one label). Perl splits `ToString` of the unexpanded tokens (KPE #1180).
1147fn split_label_tokens(tokens: Tokens) -> Result<Vec<Tokens>> {
1148  let tokens = Expand!(tokens);
1149  let mut groups: Vec<Tokens> = Vec::new();
1150  let mut cur: Vec<Token> = Vec::new();
1151  let mut depth = 0usize;
1152  for t in tokens.unlist() {
1153    match t.get_catcode() {
1154      Catcode::BEGIN => depth += 1,
1155      Catcode::END => depth = depth.saturating_sub(1),
1156      _ => {},
1157    }
1158    if depth == 0 && t.get_catcode() == Catcode::OTHER && t.with_str(|s| s == ",") {
1159      groups.push(Tokens::new(std::mem::take(&mut cur)));
1160    } else {
1161      cur.push(t);
1162    }
1163  }
1164  groups.push(Tokens::new(cur));
1165  Ok(
1166    groups
1167      .into_iter()
1168      .map(trim_space_tokens)
1169      .filter(|g| !g.is_empty())
1170      .collect(),
1171  )
1172}

1228fn cref_multi(
1229  starred: bool,
1230  label_tokens: Tokens,
1231  showtype: bool,
1232  capitalized: bool,
1233  number: &str,
1234) -> Result<Tokens> {
1235  let labels = split_label_tokens(label_tokens)?;
1236  let n = labels.len();
1237  let mut out: Vec<Token> = Vec::new();
1238
1239  // Emit one `\lx@cref [*] {<show>} {<label tokens>}`. The `<show>` is a fixed
1240  // internal string (no user macros) so tokenizing it is safe; the `<label>`
1241  // is spliced from the ORIGINAL tokens (never re-tokenized), matching Perl's
1242  // `Invocation(T_CS('\lx@cref'), $starred, T_OTHER(show), $label)`.
1243  // `\lx@tilde` carries the inter-word `~` (HyperVerbatim in Perl).
1244  let emit = |out: &mut Vec<Token>, show: &str, label: &Tokens| {
1245    out.push(T_CS!("\\lx@cref"));
1246    if starred {
1247      out.push(T_OTHER!("*"));
1248    }
1249    out.push(T_BEGIN!());
1250    out.extend(mouth::tokenize_internal(TeXString::assembled(show.to_string())).unlist());
1251    out.push(T_END!());
1252    out.push(T_BEGIN!());
1253    out.extend(label.clone().unlist());
1254    out.push(T_END!());
1255  };
1256
1257  // The forms the document's formats give (`\lx@cref@docformat`), first when the document set any of the family:
1258  // the target prints its own when it has one, else cleveref's default (crossref.rs `generate_ref_aux`).
1259  let labelled = number == CREF_LABEL;
1260  let single_doc = labelled && cref_doc_family_set("single");
1261  let label_doc = labelled && (cref_doc_family_set("label") || cref_doc_family_set("single"));
1262  let multi_doc = labelled && showtype && cref_doc_family_set("multi");
1263  let label_multi_doc = labelled && !showtype && cref_doc_family_set("multi");
1264  let prefer = |doc: bool, role: &str, show: String| {
1265    if doc {
1266      s!("{role}\\lx@cref@or {show}")
1267    } else {
1268      show
1269    }
1270  };
1271  if n < 2 {
1272    let show = if showtype {
1273      if capitalized {
1274        prefer(
1275          single_doc,
1276          "crefformatcap",
1277          s!("creftypecap\\lx@tilde {number}"),
1278        )
1279      } else {
1280        prefer(single_doc, "crefformat", s!("creftype\\lx@tilde {number}"))
1281      }
1282    } else {
1283      prefer(label_doc, "labelcrefformat", number.to_string())
1284    };
1285    let empty = Tokens::new(Vec::new());
1286    let label = labels.first().unwrap_or(&empty);
1287    emit(&mut out, &show, label);
1288  } else {
1289    let show = if showtype {
1290      if capitalized {
1291        prefer(
1292          multi_doc,
1293          "crefmultifirstcap",
1294          s!("creftypepluralcap\\lx@tilde {number}"),
1295        )
1296      } else {
1297        prefer(
1298          multi_doc,
1299          "crefmultifirst",
1300          s!("creftypeplural\\lx@tilde {number}"),
1301        )
1302      }
1303    } else {
1304      prefer(label_multi_doc, "labelcrefmulti", number.to_string())
1305    };
1306    // A later position's show: its multi format's for `\cref`, the `\labelcref` list's for `\labelcref`.
1307    let later = |role: &str| {
1308      if showtype {
1309        prefer(multi_doc, role, number.to_string())
1310      } else {
1311        prefer(label_multi_doc, "labelcrefmulti", number.to_string())
1312      }
1313    };
1314    // A position's format holds its conjunction (cleveref.sty:1514-1519, `\crefpairconjunction` before the second
1315    // label's): the conjunction between is dropped where the next reference prints the document's format
1316    // (`\lx@cref@conjunction`, crossref.rs `settle_cref_conjunctions`).
1317    let conjunction = |out: &mut Vec<Token>, cs: &str| {
1318      if multi_doc {
1319        out.extend([
1320          T_CS!("\\lx@cref@conjunction"),
1321          T_BEGIN!(),
1322          T_CS!(cs),
1323          T_END!(),
1324        ]);
1325      } else {
1326        out.push(T_CS!(cs));
1327      }
1328    };
1329    emit(&mut out, &show, &labels[0]);
1330    if n == 2 {
1331      conjunction(&mut out, "\\crefpairconjunction");
1332      emit(&mut out, &later("crefmultisecond"), &labels[1]);
1333    } else {
1334      for label in &labels[1..n - 1] {
1335        conjunction(&mut out, "\\crefmiddleconjunction");
1336        emit(&mut out, &later("crefmultimiddle"), label);
1337      }
1338      conjunction(&mut out, "\\creflastconjunction");
1339      emit(&mut out, &later("crefmultilast"), &labels[n - 1]);
1340    }
1341  }
1342  Ok(Tokens::new(out))
1343}
