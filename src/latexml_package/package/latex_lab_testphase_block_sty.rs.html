4//! `\DocumentMetadata` loads the module (latex-lab-testphase-latest.sty:43); raw code then edits or

38fn is_name_char(c: u8) -> bool { c.is_ascii_alphabetic() || matches!(c, b'_' | b':' | b'@') }
39
40/// Skips spaces and `%` comments from `i`.
41fn skip_blank(s: &[u8], mut i: usize) -> usize {
42  while i < s.len() {
43    match s[i] {
44      b' ' | b'\t' | b'\r' | b'\n' => i += 1,
45      b'%' => {
46        while i < s.len() && s[i] != b'\n' {
47          i += 1;
48        }
49      },
50      _ => break,
51    }
52  }
53  i
54}

56/// Reads the control sequence at `i` (a `\`): its name and the index past it.
57fn read_cs(s: &[u8], i: usize) -> (&[u8], usize) {
58  let mut j = i + 1;
59  while j < s.len() && is_name_char(s[j]) {
60    j += 1;
61  }
62  if j == i + 1 && j < s.len() {
63    j += 1; // a control symbol, `\{` among them
64  }
65  (&s[i + 1..j], j)
66}
67
68/// The index past the balanced group opening at `i`, or `None`.
69fn skip_group(s: &[u8], i: usize) -> Option<usize> {
70  if s.get(i) != Some(&b'{') {
71    return None;
72  }
73  let (mut depth, mut j) = (0usize, i);
74  while j < s.len() {
75    match s[j] {
76      b'\\' => j = read_cs(s, j).1,
77      b'%' => j = skip_blank(s, j),
78      b'{' => {
79        depth += 1;
80        j += 1;
81      },
82      b'}' => {
83        depth -= 1;
84        j += 1;
85        if depth == 0 {
86          return Some(j);
87        }
88      },
89      _ => j += 1,
90    }
91  }
92  None
93}

95/// The declarations at brace depth 0 of the module's text, as TeX source: the `DECLARERS`
96/// statements (a template's code replaced by `\SetKnownTemplateKeys`), the variables it
97/// `\<type>_new:N`s (where not defined yet) and the `INERT_COMMANDS` it defines.
98fn top_level_declarations(text: &str) -> String {
99  let s = text.as_bytes();
100  let mut out = String::new();
101  let (mut depth, mut i) = (0usize, 0usize);
102  while i < s.len() {
103    match s[i] {
104      b'%' => i = skip_blank(s, i),
105      b'{' => {
106        depth += 1;
107        i += 1;
108      },
109      b'}' => {
110        depth = depth.saturating_sub(1);
111        i += 1;
112      },
113      b'\\' => {
114        let (name, next) = read_cs(s, i);
115        i = next;
116        if depth > 0 {
117          continue;
118        }
119        let name = std::str::from_utf8(name).unwrap_or_default();
120        if let Some(&(_, nargs)) = DECLARERS.iter().find(|(n, _)| *n == name) {
121          let mut args = Vec::with_capacity(nargs);
122          let mut j = i;
123          for _ in 0..nargs {
124            let start = skip_blank(s, j);
125            let Some(end) = skip_group(s, start) else {
126              break;
127            };
128            args.push(&text[start..end]);
129            j = end;
130          }
131          if args.len() < nargs {
132            continue;
133          }
134          out.push('\\');
135          out.push_str(name);
136          out.push_str(&args.concat());
137          if name == "DeclareTemplateCode" {
138            // the code itself is skipped; it absorbs the keys of its first argument
139            if let Some(end) = skip_group(s, skip_blank(s, j)) {
140              j = end;
141            }
142            let (ty, tmpl) = (args[0], args[1]);
143            out.push_str(&format!(r"{{\SetKnownTemplateKeys{ty}{tmpl}{{#1}}}}"));
144          }
145          out.push('\n');
146          i = j;
147        } else if name.ends_with("_new:N") {
148          let start = skip_blank(s, i);
149          if s.get(start) == Some(&b'\\') {
150            let (var, end) = read_cs(s, start);
151            let var = std::str::from_utf8(var).unwrap_or_default();
152            out.push_str(&format!("\\cs_if_exist:NF\\{var}{{\\{name}\\{var}}}\n"));
153            i = end;
154          }
155        } else if name.starts_with("cs_new") && name.ends_with(":Npn") {
156          let (cmd, _) = read_cs(s, skip_blank(s, i));
157          let cmd = std::str::from_utf8(cmd).unwrap_or_default();
158          if let Some((_, body)) = INERT_COMMANDS.iter().find(|(n, _)| *n == cmd) {
159            let body = if body.starts_with('#') {
160              (*body).to_string()
161            } else {
162              format!("{{{body}}}")
163            };
164            out.push_str(&format!("\\cs_new_protected:Npn\\{cmd}{body}\n"));
165          }
166        }
167      },
168      _ => i += 1,
169    }
170  }
171  out
172}
