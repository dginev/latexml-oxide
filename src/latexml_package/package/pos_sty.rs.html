1//! pos.sty (SISSA Medialab's PoS template, v2.1 2018/11/13 and v2.2 2021/09/21, loaded under `article`): the papers
2//! ship the package (it is not in TeX Live), which is read raw, its title page made by the frontmatter. pos.sty keeps
3//! that page in stores only its own `\maketitle` prints (:450-530), which the kernel's locked `\maketitle` refuses: the
4//! authors in expl3 sequences (:88-127 `\RenewDocumentCommand\author{som}`, which the kernel's locked `\author` refused
5//! too, so `\author*{Javier M. Duarte}` read `*` as the name and the name as an affiliation), the affiliations and
6//! e-mail addresses in token registers (:411-430), the conference in a box (:364-369). Every one is the frontmatter's
7//! here, defined before the copy is read, whose `\newcommand`s of the same names are then ignored as those of any name
8//! LaTeXML defines (and its `\NewDocumentCommand\onbehalf`, :249). Perl has no binding: it does not read the package,
9//! keeps an empty name and raises an error for each `\affiliation` and `\emailAdd` (KNOWN_PERL_ERRORS #1198).
10//! Witnesses 2409.20413, 2512.14592, 2602.22379 (OXIDIZED_DESIGN_DIVERGENCES #904).
11use crate::{
12  package::{jheppub_sty::jhep_contact_markup, pos_cls::pos_title_page_fields},
13  prelude::*,
14  shipped_class::{class_code, class_code_argument, shipped_file_text},
15};

17/// The text of the speaker's footnote in the paper's copy: the argument of the `\footnotetext` that `\printAuthors`
18/// sets under the speaker's mark (:228-230 `\renewcommand\thefootnote{$\speakerFlag$}\footnotetext{\speakerText}`),
19/// which v2.1 and early v2.2 copies write out ("Speaker"; 2107.12913's "Presenter"); `\speakerText` (:85, "Speaker",
20/// which a paper may renew) where the copy names it, or has none.
21fn speaker_text() -> String {
22  const MARK: &str = "\\renewcommand\\thefootnote{$\\speakerFlag$}";
23  shipped_file_text("pos", "sty")
24    .and_then(|text| {
25      let code = class_code(&text);
26      class_code_argument(&code[code.find(MARK)? + MARK.len()..], "\\footnotetext")
27    })
28    .unwrap_or_else(|| s!("\\speakerText"))
29}
30
31#[rustfmt::skip]
32LoadDefinitions!({
33  // :88-127 `\author*[marks]{name}`: one author per call (a name list in one is an author each, as jheppub's), its
34  // optional marks the labels of the affiliations it has (`\affiliation[a]{…}`), the star the speaker, whom the title
35  // page marks `\speakerFlag` (:84 `\ast`) and names in a footnote, "∗ Speaker" (:226-233): that footnote is the
36  // speaker's note, labelled `lx@pos@speaker`, which every starred author requests beside its marks (one footnote
37  // under every star; given once). A label of its own, as the star is a mark: an unlabelled note would be one of the
38  // notes the kernel deals out one per labelled author, in order, when they are as many — `\author[a]{Ann Lee}
39  // \author*[b]{Bob Kim}\onbehalf{…}` gave Ann Lee the "Speaker" note and Bob Kim the on-behalf line.
40  DefMacro!("\\author OptionalMatch:* []{}", sub[(star, marks, name)] {
41    let author = if star.is_some() { "\\lx@pos@speaker@author" } else { "\\lx@pos@author" };
42    Ok(Invocation!(T_CS!(author), vec![marks, Some(name)]))
43  });
44  DefMacro!("\\lx@pos@author[]{}", "\\lx@add@authors@append[annotations={#1}]{#2}");
45  DefMacro!("\\lx@pos@speaker@author[]{}",
46    "\\if\\relax\\detokenize{#1}\\relax\\expandafter\\@firstoftwo\\else\\expandafter\\@secondoftwo\\fi\
47     {\\lx@pos@author[lx@pos@speaker]{#2}}{\\lx@pos@author[#1,lx@pos@speaker]{#2}}\\lx@pos@speaker@note");
48  DefMacro!("\\lx@pos@speaker@note",
49    "\\lx@add@contact[role=note,label={lx@pos@speaker}]{\\lx@pos@speaker}\\global\\let\\lx@pos@speaker@note\\relax");
50  // `\affiliation[label]{…}` (:411-419), `\emailAdd{…}` (:421-430, the "E-mail:" line under the affiliations, each
51  // address after the first after a comma), `\note[mark]{…}` (:380-400, a footnote's text inside an `\author`) and
52  // `\dedicated{…}` (:359-360): jheppub.sty's commands, to the same effect.
53  jhep_contact_markup()?;
54  // `\ShortTitle{…}` and `\FullConference{…}` (:362-369): PoS.cls's.
55  pos_title_page_fields()?;
56  // `\onbehalf{for the X Collaboration}` (:249-260): the line `\printAuthors` sets after the names (:223), "Javier M.
57  // Duarte∗ on behalf of the ALICE, ATLAS, CMS, and LHCb Collaborations" (2409.20413; `\onbehalf*` sets it below
58  // them): a note of the author given last before it, after whose name it is printed (2308.05475, 2507.06616 give it
59  // after their author lists).
60  DefMacro!("\\onbehalf OptionalMatch:* {}", "\\lx@add@contact[role=note]{#2}");
61  // `\notes{…}` (:357): the text the title page prints under the abstract (:506), with no label
62  // (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #921).
63  DefMacro!("\\notes{}", "\\lx@add@pubnote[role=note,name={}]{#1}");
64  // :356 `\renewcommand{\abstract}[1]{\gdef\@abstract{#1}}`: the abstract is one argument, as jheppub.sty's.
65  AssignValue!("\\abstract:redefined@nargs" => Stored::Number(Number::new(1)), Some(Scope::Global));
66  RawTeX!(&s!("\\def\\lx@pos@speaker{{{}}}", speaker_text()));
67  if shipped_file_text("pos", "sty").is_some() {
68    InputDefinitions!("pos", noltxml => true, extension => Some(Cow::Borrowed("sty")));
69  } else {
70    // (no copy: the packages pos.sty:21-50 loads)
71    RequirePackage!("amsthm");
72    RequirePackage!("amsmath");
73    RequirePackage!("graphicx");
74    RequirePackage!("natbib", options => vec![s!("numbers"), s!("sort&compress")]);
75    RequirePackage!("caption", options => vec![s!("font=small"), s!("labelfont=bf")]);
76    RequirePackage!("xcolor");
77    RequirePackage!("hyperref");
78  }
79  RawTeX!(r"\providecommand{\speakerText}{Speaker}\providecommand{\speakerFlag}{\ast}");
80});
