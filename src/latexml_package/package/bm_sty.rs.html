5/// bm.sty's `\bm@define` (:307-316) `\xdef`s `\<name>` from its body, so a body that names
6/// `\<name>` itself (`\DeclareBoldMathCommand{\nabla}{\nabla}`) makes it bold over its meaning
7/// at that moment. Our body stays unexpanded, so that meaning is saved globally under a name of
8/// its own, `\lx@bm@saved@<n>@<name>` (a new `<n>` per definition, as each `\xdef` sees the
9/// meaning just before it: a second self-referential definition must not save the first's
10/// wrapper under the same name), and the body refers to that; a plain `\gdef` would recurse.
11fn bm_define(cs: Tokens, body: Tokens) -> Result<Tokens> {
12  let Some(target) = cs
13    .unlist()
14    .into_iter()
15    .find(|t| t.get_catcode() != Catcode::SPACE)
16  else {
17    Error!("expected", "\\bmdefine", "Missing the command to define");
18    return Ok(Tokens::new(Vec::new()));
19  };
20  let mut body = body.unlist();
21  let mut out = Vec::new();
22  if body.contains(&target) {
23    let n = lookup_int("lx@bm@saved@count") + 1;
24    assign_value("lx@bm@saved@count", n, Some(Scope::Global));
25    let saved = T_CS!(
26      &target.with_cs_name(|name| { s!("\\lx@bm@saved@{n}@{}", name.trim_start_matches('\\')) })
27    );
28    out.extend([T_CS!("\\global"), T_CS!("\\let"), saved, target]);
29    for token in body.iter_mut() {
30      if *token == target {
31        *token = saved;
32      }
33    }
34  }
35  out.extend([
36    T_CS!("\\gdef"),
37    target,
38    T_BEGIN!(),
39    T_CS!("\\bm"),
40    T_BEGIN!(),
41  ]);
42  out.extend(body);
43  out.extend([T_END!(), T_END!()]);
44  Ok(Tokens::new(out))
45}

47/// Whether `token` means `\left`, `\middle` or `\right` (bm.sty:173-174 `\let`s `\left` and `\right` to
48/// `\holdinginserts` while it scans, and tests a token's meaning against it, `\ifx\bm@previous\left`, :289).
49fn is_left_right(token: &Token) -> bool {
50  [T_CS!("\\left"), T_CS!("\\middle"), T_CS!("\\right")]
51    .iter()
52    .any(|command| token.defined_as(command))
53}

55/// The tokens of one TeX number after `\delimiter` (tex.web §440-445, at the token level), from `tokens[i]`: an
56/// optional sign, then `"` with hexadecimal digits, `'` with octal digits, `` ` `` with its character, decimal
57/// digits, or one control sequence (an internal quantity), and one optional space. Returns the index after them.
58fn take_number(tokens: &[Token], mut i: usize, out: &mut Vec<Token>) -> usize {
59  let text = |i: usize| tokens.get(i).map(ToString::to_string).unwrap_or_default();
60  while matches!(text(i).as_str(), "+" | "-" | " ") {
61    out.push(tokens[i]);
62    i += 1;
63  }
64  let digit: fn(char) -> bool = match text(i).as_str() {
65    "\"" => |c| c.is_ascii_digit() || ('A'..='F').contains(&c),
66    "'" => |c| ('0'..='7').contains(&c),
67    "`" => {
68      out.extend(tokens.get(i..i + 2).unwrap_or(&tokens[i..]));
69      return (i + 2).min(tokens.len());
70    },
71    _ if tokens
72      .get(i)
73      .is_some_and(|t| matches!(t.get_catcode(), Catcode::CS | Catcode::ACTIVE)) =>
74    {
75      out.push(tokens[i]);
76      return i + 1;
77    },
78    _ => |c| c.is_ascii_digit(),
79  };
80  if matches!(text(i).as_str(), "\"" | "'") {
81    out.push(tokens[i]);
82    i += 1;
83  }
84  while text(i).chars().next().is_some_and(digit) && text(i).chars().count() == 1 {
85    out.push(tokens[i]);
86    i += 1;
87  }
88  if tokens
89    .get(i)
90    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
91  {
92    out.push(tokens[i]);
93    i += 1;
94  }
95  i
96}

98/// A `_` or `^`: a script character, or a control sequence let to one (`\sb`, `\sp`).
99fn is_script(token: &Token) -> bool {
100  matches!(token.get_catcode(), Catcode::SUPER | Catcode::SUB)
101    || token.defined_as(&T_SUPER!())
102    || token.defined_as(&T_SUB!())

105/// The index of the first token at or after `i` that is not a blank.
106fn skip_blanks(tokens: &[Token], mut i: usize) -> usize {
107  while tokens
108    .get(i)
109    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
110  {
111    i += 1;
112  }
113  i

116/// The index of the `}` closing the group that opens at `tokens[i]`, if it closes.
117fn group_end(tokens: &[Token], i: usize) -> Option<usize> {
118  let mut depth = 0usize;
119  for (j, t) in tokens.iter().enumerate().skip(i) {
120    match t.get_catcode() {
121      Catcode::BEGIN => depth += 1,
122      Catcode::END => {
123        depth = depth.checked_sub(1)?;
124        if depth == 0 {
125          return Some(j);
126        }
127      },
128      _ => {},
129    }
130  }
131  None
132}

134/// The index after one undelimited argument from `tokens[i]` (tex.web §392): blanks, then one token or a balanced
135/// group.
136fn take_argument(tokens: &[Token], i: usize) -> Option<usize> {
137  let i = skip_blanks(tokens, i);
138  match tokens.get(i)?.get_catcode() {
139    Catcode::BEGIN => group_end(tokens, i).map(|end| end + 1),
140    Catcode::END => None,
141    _ => Some(i + 1),
142  }
143}

145/// The index after an optional `[…]` argument from `tokens[i]` (latex.ltx `\@ifnextchar[`, after blanks), or `i`
146/// when there is none.
147fn take_optional(tokens: &[Token], i: usize) -> Option<usize> {
148  let open = skip_blanks(tokens, i);
149  let is_other = |t: &Token, c: &str| t.get_catcode() == Catcode::OTHER && t.to_string() == c;
150  if !tokens.get(open).is_some_and(|t| is_other(t, "[")) {
151    return Some(i);
152  }
153  let mut depth = 0usize;
154  for (j, t) in tokens.iter().enumerate().skip(open + 1) {
155    match t.get_catcode() {
156      Catcode::BEGIN => depth += 1,
157      Catcode::END => depth = depth.checked_sub(1)?,
158      _ if depth == 0 && is_other(t, "]") => return Some(j + 1),
159      _ => {},
160    }
161  }
162  None
163}

165/// The argument of a script at `run[i]` and the index after it, or `None` when it cannot be told without expanding
166/// it. A braced group gives its content. bm.sty expands the argument as it scans it (`\bm@expand`, with `\protect`
167/// `\let` to `\@empty`, bm.sty:166-167, 286-287), so a command's own arguments go with it: a control sequence
168/// takes as many arguments as its definition has parameters (`\mathrm{L}`, `\text{L}`, `\frac{1}{2}`: a
169/// [`BRACED_ARGUMENT_TYPES`] parameter takes an argument, an optional one its `[…]` when present). An undefined
170/// command, `\bgroup`, a parameter of any other kind, or a parameterless macro before a `{` (whose expansion may
171/// take it, like a robust command's `\protect\cs␣`) cannot be told.
172fn script_argument(run: &[Token], i: usize) -> Option<(Vec<Token>, usize)> {
173  let i = skip_blanks(run, i);
174  let first = *run.get(i)?;
175  match first.get_catcode() {
176    Catcode::BEGIN => {
177      let end = group_end(run, i)?;
178      Some((run[i + 1..end].to_vec(), end + 1))
179    },
180    Catcode::END => None,
181    Catcode::CS | Catcode::ACTIVE => {
182      if first.defined_as(&T_CS!("\\bgroup")) || first.defined_as(&T_CS!("\\egroup")) {
183        return None;
184      }
185      let definition = lookup_definition(&first).ok().flatten()?;
186      let mut end = i + 1;
187      match definition.get_parameters() {
188        None => {
189          if definition.is_expandable()
190            && run
191              .get(skip_blanks(run, end))
192              .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
193          {
194            return None;
195          }
196        },
197        Some(parameters) => {
198          for parameter in parameters.get_parameters() {
199            if parameter.novalue {
200              continue;
201            }
202            let name = to_string(parameter.name);
203            end = if BRACED_ARGUMENT_TYPES.contains(&name.as_str()) {
204              take_argument(run, end)?
205            } else if parameter.optional && name != "OptionalMatch" {
206              take_optional(run, end)?
207            } else {
208              return None;
209            };
210          }
211        },
212      }
213      Some((run[i..end].to_vec(), end))
214    },
215    _ => Some((vec![first], i + 1)),
216  }
217}

219/// A run of `\bm`'s argument between its delimiter commands, bolded by `bold` (`\lx@bm@bold`). Its leading scripts
220/// attach to the delimiter before them, as bm.sty emits `_`/`^` where they stand (`\bm@add`): `\bm{\left\langle z
221/// \right\rangle_{L,R}}` (2107.08152) and `\bm{\left(\Sigma_i^e \right)_{(j,j)}}` (1703.10645) keep their subscript
222/// on the fence. bm.sty expands a script's argument before it bolds it (`\bm@expand`), so the argument stays whole
223/// ([`script_argument`]: `\right\rangle_\mathrm{L}` bolds `\mathrm{L}`) and is bolded. A prime becomes a bold
224/// `^{\prime}` (bm.sty:177 `\def'{\bm@prime\prime\relax}`), joined by a superscript right after it as
225/// `\active@math@prime` joins it. A script whose argument cannot be told keeps the rest of the run as its argument,
226/// with a Warning naming `cs`.
227fn bold_run(out: &mut Vec<Token>, cs: &str, bold: Token, run: Vec<Token>) {
228  let is_superscript = |t: &Token| t.get_catcode() == Catcode::SUPER || t.defined_as(&T_SUPER!());
229  let is_prime = |t: &Token| t.get_catcode() == Catcode::OTHER && t.to_string() == "'";
230  let undetermined = |script: &Token, rest: &[Token]| {
231    let head = rest
232      .iter()
233      .find(|t| t.get_catcode() != Catcode::SPACE)
234      .map(ToString::to_string)
235      .unwrap_or_default();
236    Warn!(
237      "unexpected",
238      cs,
239      s!(
240        "Cannot tell the argument of {script} (starting {head}) after a \\left, \\middle or \\right in the \
241         argument of {cs}; the rest of the argument is taken as it"
242      )
243    );
244  };
245  let mut i = 0;
246  loop {
247    while let Some(&t) = run.get(i)
248      && t.get_catcode() == Catcode::SPACE
249    {
250      out.push(t);
251      i += 1;
252    }
253    match run.get(i) {
254      Some(prime) if is_prime(prime) => {
255        let mut superscript = Vec::new();
256        while run.get(i).is_some_and(is_prime) {
257          superscript.push(T_CS!("\\prime"));
258          i += 1;
259        }
260        if let Some(script) = run.get(i).filter(|t| is_superscript(t)) {
261          match script_argument(&run, i + 1) {
262            Some((argument, next)) => {
263              superscript.extend(argument);
264              i = next;
265            },
266            None => {
267              undetermined(script, &run[i + 1..]);
268              superscript.extend(&run[i + 1..]);
269              i = run.len();
270            },
271          }
272        }
273        out.extend([T_SUPER!(), T_BEGIN!(), bold, T_BEGIN!()]);
274        out.extend(superscript);
275        out.extend([T_END!(), T_END!()]);
276      },
277      Some(&script) if is_script(&script) => {
278        out.push(script);
279        i += 1;
280        // Nothing after the script in the run: its argument follows `\bm`, where bm.sty leaves it.
281        if skip_blanks(&run, i) == run.len() {
282          return;
283        }
284        let (argument, next) = script_argument(&run, i).unwrap_or_else(|| {
285          undetermined(&script, &run[i..]);
286          (run[i..].to_vec(), run.len())
287        });
288        out.extend([bold, T_BEGIN!()]);
289        out.extend(argument);
290        out.push(T_END!());
291        i = next;
292      },
293      _ => break,
294    }
295  }
296  let rest = &run[i.min(run.len())..];
297  if !rest.is_empty() {
298    out.extend([bold, T_BEGIN!()]);
299    out.extend(rest);
300    out.push(T_END!());
301  }
302}

304/// How the package defining a bold-math command treats its argument.
305#[derive(Clone, Copy, PartialEq, Eq, Debug)]
306pub enum BoldArgument {
307  /// bm.sty's `\bm` scans its argument token by token and emits it at the level `\bm` is used at (`\bm@general`,
308  /// bm.sty:162-217): a `\left`, `\middle` or `\right` without its partner in the argument stands outside the
309  /// bolded runs, without a diagnostic.
310  Scanned,
311  /// The argument is set in a box or a group of its own: amsbsy's `\boldsymbol` (`\mathchoice{\hbox{$…$}}`,
312  /// amsbsy.sty:28-38), the `\bm` of the World Scientific classes (ws-ijmpe.cls:226-230 `\mbox{\boldmath$#1$}`) and
313  /// of interact.cls (:134-138), and RevTeX's `\bm` (revtex4-2.cls:7573-7576 `\REV@bm`: the first use is dropped
314  /// with "To use \bm, please load the bm package!", later ones are `\relax{…}`). A `\left` paired with its
315  /// `\right` inside the argument is the bounded constructor, as before; an unpaired one is TeX's error ("Missing
316  /// \right. inserted" at the box's end, "Extra \right." at the next box), set at the outer level as bm.sty would,
317  /// with a Warning.
318  Grouped,
319}

321/// Whether the `\left`, `\middle` and `\right` at an argument's top level pair up: each `\middle` and `\right`
322/// inside an open `\left`, each `\left` closed.
323fn delimiters_pair_up(commands: &[Token]) -> bool {
324  let mut open = 0usize;
325  for command in commands {
326    if command.defined_as(&T_CS!("\\left")) {
327      open += 1;
328    } else if open == 0 {
329      return false;
330    } else if command.defined_as(&T_CS!("\\right")) {
331      open -= 1;
332    }
333  }
334  open == 0
335}

337/// The expansion of a bold-math command `cs` (`\bm`, `\boldsymbol`) whose bounded bold constructor is `bold`
338/// (`\lx@bm@bold`, Perl's `\bm`, bm.sty.ltxml:20, and `\boldsymbol`, amsbsy.sty.ltxml:23, under their own names).
339/// bm.sty's `\bm` (`\bm@general`, bm.sty:162-217) scans its argument token by token and emits the bold result at the
340/// level `\bm` is used at, grouping only the argument's own `{…}` groups (`\bm@gr@up`): `\left` and `\right` are kept
341/// as they are and the delimiter after them becomes a bold `\delimiter` (`\bm@delimiter`, :416). Here the argument is
342/// bolded by the bounded `bold`, but a `\left`, `\middle` or `\right` at the argument's top level is taken out of
343/// it: since 65d `\left` opens a `math_left_group` (OXIDIZED_DESIGN #530) that an argument group would close before
344/// its `\right` (2204.13665, `\bm{\left\langle} #1 \bm{\right\rangle}`: "\bm Attempt to close a group that
345/// switched to mode math", 587 errors). The command and its delimiter stand between the bolded runs, followed by
346/// `\lx@bm@delimiter`, which makes the delimiter bold. Only an argument whose delimiters do not pair up is split so:
347/// a pair inside it stays in the bounded `bold`, whose group holds the `math_left_group` whole, so the fenced whole
348/// keeps its scripts and primes (`\bm{\left(A+B\right)^\mathsf{T}}`, `\bm{x \left( y \right)_\text{L}}`). For a
349/// [`BoldArgument::Grouped`] command the split comes with a Warning: TeX's box ends the `\left` there.
350/// OXIDIZED_DESIGN #544.
351pub fn bold_math(cs: &str, bold: Token, arg: Tokens, argument: BoldArgument) -> Tokens {
352  let tokens = arg.unlist();
353  let mut depth = 0usize;
354  let at_top_level = |depth: &mut usize, token: &Token| {
355    match token.get_catcode() {
356      Catcode::BEGIN => *depth += 1,
357      Catcode::END => *depth = depth.saturating_sub(1),
358      _ => {},
359    }
360    *depth == 0 && is_left_right(token)
361  };
362  let in_math = lookup_bool("IN_MATH");
363  // Perl's `requireMath` on `cs` (Package.pm:1069-1073), the warning naming `cs`.
364  requireMath!(cs);
365  // bm.sty's `\bm` emits a math shift at the level `\bm` is used at (`\bm@add{#1}`, bm.sty:281), where it ends the
366  // enclosing formula and the next begins one: the argument is set as bold runs between its top-level `$`s, which stand
367  // between them (bm's `\boldsymbol{e^+e^-$$\rightarrow$$J/\psi}`, 2506.20975: three formulas; rc_struct6 row 9;
368  // `\bm{\hat{m}$}` ends the formula after the bold m). Guard `perfect_kernel_batch73b::dollar_in_boldsymbol_is_text`.
369  if argument == BoldArgument::Scanned && in_math {
370    let mut level = 0usize;
371    let shifts: Vec<usize> = tokens
372      .iter()
373      .enumerate()
374      .filter_map(|(at, token)| {
375        match token.get_catcode() {
376          Catcode::BEGIN => level += 1,
377          Catcode::END => level = level.saturating_sub(1),
378          Catcode::MATH if level == 0 => return Some(at),
379          _ => {},
380        }
381        None
382      })
383      .collect();
384    if !shifts.is_empty() {
385      let mut out = Vec::new();
386      let mut from = 0;
387      for at in shifts.into_iter().chain(std::iter::once(tokens.len())) {
388        let segment = &tokens[from..at];
389        if skip_blanks(segment, 0) < segment.len() {
390          out.extend(bold_math(cs, bold, Tokens::new(segment.to_vec()), argument).unlist());
391        }
392        if let Some(&shift) = tokens.get(at) {
393          out.push(shift);
394        }
395        from = at + 1;
396      }
397      return Tokens::new(out);
398    }
399  }
400  let commands: Vec<Token> = tokens
401    .iter()
402    .filter(|t| at_top_level(&mut depth, t))
403    .copied()
404    .collect();
405  // A pair inside the argument stays in the bounded constructor, byte-identical to before: the `math_left_group`
406  // opens and closes within its group (`\bm{\left(A+B\right)^\mathsf{T}}` keeps its superscript on the fenced
407  // whole).
408  let split = in_math && !commands.is_empty() && !delimiters_pair_up(&commands);
409  if !split {
410    let mut tokens = tokens;
411    // bm.sty's `\bm` emits a script ending its argument as it stands (`\let_\relax` in `\bm@general`,
412    // bm.sty:162-217), so it takes its argument from after `\bm`'s closing brace: `\bm{G_}{(10)}` is a bold `G`
413    // subscripted by `(10)`, `M\bm_{max}` is `M_{max}` (arXiv 2010.13108, 1708.01493). Here the script follows the
414    // bounded `bold`, as `bold_run` leaves it on the split path; in the group it would read the `}` ("Missing
415    // sub/superscript argument"). A boxed or grouped command ([`BoldArgument::Grouped`]) keeps TeX's error, its box
416    // ending there.
417    let mut script = None;
418    if argument == BoldArgument::Scanned
419      && let Some(last) = tokens
420        .iter()
421        .rposition(|t| t.get_catcode() != Catcode::SPACE)
422      && is_script(&tokens[last])
423    {
424      script = Some(tokens[last]);
425      tokens.truncate(last);
426    }
427    let mut out = Vec::new();
428    // `\bm_`: the script alone, attaching to what precedes `\bm` as in TeX.
429    if script.is_none() || skip_blanks(&tokens, 0) < tokens.len() {
430      // A command that boxes its argument — amsbsy's `\boldsymbol`, `\hbox{$\m@th\displaystyle#1$}` (amsbsy.sty:28-36)
431      // — has a `$` there end that math and start text in the box, the next begin math again (tex.web §1193, §1138):
432      // `\lx@math@in@hbox` (tex_math.rs), local to `bold`'s bounded group. `$\boldsymbol{\theta=\pi/2$ and $\phi=0}$`
433      // (2508.01295; rc_struct6 row 9, KNOWN_PERL_ERRORS #907). Guard `perfect_kernel_batch73b::dollar_in_boldsymbol_is_text`.
434      out.extend([bold, T_BEGIN!()]);
435      if argument == BoldArgument::Grouped {
436        out.push(T_CS!("\\lx@math@in@hbox"));
437      }
438      out.extend(tokens);
439      out.push(T_END!());
440    }
441    out.extend(script);
442    return Tokens::new(out);
443  }
444  if argument == BoldArgument::Grouped {
445    Warn!(
446      "unexpected",
447      cs,
448      s!(
449        "\\left, \\middle or \\right without its partner in the argument of {cs}, whose box or group ends it in \
450         TeX (\"Missing \\right. inserted\"); set at the outer level"
451      )
452    );
453  }
454  depth = 0;
455  let mut out = Vec::new();
456  let mut run = Vec::new();
457  let mut i = 0;
458  while let Some(&token) = tokens.get(i) {
459    i += 1;
460    if !at_top_level(&mut depth, &token) {
461      run.push(token);
462      continue;
463    }
464    bold_run(&mut out, cs, bold, std::mem::take(&mut run));
465    out.push(token);
466    // tex.web §1160 `scan_delimiter` passes blanks and `\relax` by.
467    while let Some(&filler) = tokens.get(i)
468      && (filler.get_catcode() == Catcode::SPACE || filler.defined_as(&T_CS!("\\relax")))
469    {
470      out.push(filler);
471      i += 1;
472    }
473    // A brace (or nothing) where the delimiter belongs is the command's own "Missing delimiter".
474    let Some(&delimiter) = tokens.get(i) else {
475      break;
476    };
477    if matches!(delimiter.get_catcode(), Catcode::BEGIN | Catcode::END) {
478      continue;
479    }
480    out.push(delimiter);
481    i += 1;
482    if delimiter.defined_as(&T_CS!("\\delimiter")) {
483      i = take_number(&tokens, i, &mut out);
484    } else if delimiter.defined_as(&T_CS!("\\Udelimiter")) {
485      for _ in 0..3 {
486        i = take_number(&tokens, i, &mut out);
487      }
488    }
489    out.push(T_CS!("\\lx@bm@delimiter"));
490  }
491  bold_run(&mut out, cs, bold, std::mem::take(&mut run));
492  Tokens::new(out)
493}

495/// `\lx@bm@delimiter`: the delimiter `\left`, `\middle` or `\right` just emitted — the current node's last element,
496/// as Perl's `augmentDelimiterProperties` finds it (TeX_Math.pool.ltxml:800-823) — is made bold, bm.sty's
497/// `\bm@delimiter` (:416). The null delimiter (`XMHint`) has no glyph to bold.
498fn bold_delimiter(document: &mut Document) -> Result<()> {
499  let current = document.get_node().clone();
500  if let Some(mut delimiter) = current.get_last_element_child()
501    && delimiter.get_name() != "XMHint"
502  {
503    let bold = document.get_node_font(&delimiter).merge(Font {
504      forcebold: Some(true),
505      ..Font::default()
506    });
507    document.set_node_font(&mut delimiter, &bold)?;
508  }
509  Ok(())
510}

512/// Defines `\lx@bm@delimiter` (`bold_delimiter`), which the expansion of every [`bold_math`] command uses after a
513/// delimiter it sets at the outer level.
514pub fn define_bold_delimiter() -> Result<()> {
515  DefConstructor!("\\lx@bm@delimiter", "",
516    after_construct => sub[document, _whatsit] { bold_delimiter(document)?; },
517    reversion => "");
518  Ok(())
519}

521/// The `\bm` a class defines for itself, setting its argument in a box or group ([`BoldArgument::Grouped`]): the World
522/// Scientific classes (ws_journal_cls.rs), interact.cls (interact_cls.rs) and RevTeX (revtex4_support_sty.rs, Perl
523/// revtex4_support.sty.ltxml:167). Its bold constructor is bm.sty's `\lx@bm@bold`, reverting as `\bm`; a later
524/// `\usepackage{bm}` replaces the definition with bm.sty's.
525pub fn define_grouped_bm() -> Result<()> {
526  DefConstructor!("\\lx@bm@bold{}", "#1", bounded => true, font => { forcebold => true }, alias => "\\bm");
527  DefMacro!("\\bm{}", sub[(arg)] {
528    Ok(bold_math("\\bm", T_CS!("\\lx@bm@bold"), arg, BoldArgument::Grouped))
529  }, protected => true);
530  define_bold_delimiter()
531}
