3#[rustfmt::skip]
4LoadDefinitions!({
5  // STATUS — the whole scalerel family is NEUTRALIZED, not yet fully supported.
6  // This binding is deliberately STEP 1: *preserve the content* (the object survives,
7  // sized to text height) and stop the `Error:undefined` + broken layout — at the cost
8  // of the ACTUAL scale. The target height (`\scalerel`/`\scaleto`/`\stretchto`) and the
9  // numeric factor (`\scaleobj`/`\hstretch`/`\vstretch`) are DROPPED, so every scaled
10  // object renders at ~1em (text height) regardless of the requested size. Do NOT mistake
11  // this approximation for correct sizing.
12  // FUTURE (complete support): honour the real scale — measure the object box and the
13  // reference box (or read the factor), compute the ratio, and emit a CSS
14  // `transform: scale()` / SVG viewBox — which needs box measurement the engine does not
15  // yet expose. Tracked in docs/SYNC_STATUS.md (scalerel full box-measurement scaling).
16  //
17  // arXiv/html_feedback#6895: the `scalerel` package has no `.ltxml` binding in
18  // Perl OR Rust, and the raw `.sty` load leaves `\scalerel` undefined — so an
19  // inline icon built with `\scalerel*` (the `\orcidicon` of arXiv:2608.12272)
20  // raised `Error:undefined:\scalerel` and rendered its picture unscaled ("too
21  // big / multi-line"). Beyond Perl 0.8.8, which errors identically.
22  //
23  // `\scalerel*[maxwidth]{obj}{ref}` scales `obj` to the height of `ref`, aspect
24  // preserved (scalerel.sty L68-84). The dominant use is an inline object scaled
25  // to the surrounding *text* height, so — box-measurement scaling being
26  // unavailable in this engine — we wrap `obj` in an inline-block that CSS sizes
27  // to text height (`.ltx_scalerel`, `LaTeXML.css`). The `[maxwidth]` cap
28  // (default `99in`, i.e. unbounded) is accepted and dropped. The starred form
29  // yields just the scaled object; the plain `\scalerel{obj}{ref}` appends `ref`
30  // afterwards (scalerel.sty L84, `\scalerelplus`).
31  DefMacro!("\\scalerel", "\\@ifstar\\lx@scalerel@star\\lx@scalerel@plus");
32  DefMacro!("\\lx@scalerel@star []{}{}", "\\lx@scalerel@obj{#2}");
33  DefMacro!("\\lx@scalerel@plus []{}{}", "\\lx@scalerel@obj{#2}#3");
34  DefConstructor!("\\lx@scalerel@obj{}",
35    "<ltx:inline-block class='ltx_scalerel'>#1</ltx:inline-block>",
36    mode => "restricted_horizontal", enter_horizontal => true, bounded => true);
37  // `\stretchrel` stretches ignoring the aspect ratio; aspect-preserving is the
38  // safe default for an inline icon, so alias it to `\scalerel` (scalerel.sty L86).
39  Let!("\\stretchrel", "\\scalerel");
40  // Binding scalerel short-circuits the raw `.sty`, so EVERY command it defines is
41  // lost unless re-added here. Before this binding the raw load defined the whole
42  // family (as `\newcommand`s), so papers using them converted clean; covering only
43  // `\scalerel`/`\stretchrel` regressed `\scaleto`/`\scaleobj`/`\stretchto` &c. to
44  // `Error:undefined` (sandbox-arxiv-2605: 20+ papers, e.g. 2605.02053, 2605.03024,
45  // 2605.03521). Each scales/stretches its OBJECT to a height or by a factor we
46  // cannot measure box-wise, so — as with `\scalerel` — we wrap just the object in
47  // the text-height inline-block and drop the target (scalerel.sty L104/117/145-159).
48  //   \scaleto  [max]{obj}{ht}     obj=#2  (scalerel.sty L104)
49  //   \stretchto[min]{obj}{ht}     obj=#2  (scalerel.sty L117)
50  //   \scaleobj      {factor}{obj} obj=#2  (scalerel.sty L145)
51  //   \hstretch      {factor}{obj} obj=#2  (scalerel.sty L138)
52  //   \vstretch      {factor}{obj} obj=#2  (scalerel.sty L141)
53  DefMacro!("\\scaleto []{}{}",   "\\lx@scalerel@obj{#2}");
54  DefMacro!("\\stretchto []{}{}", "\\lx@scalerel@obj{#2}");
55  DefMacro!("\\scaleobj {}{}",    "\\lx@scalerel@obj{#2}");
56  DefMacro!("\\hstretch {}{}",    "\\lx@scalerel@obj{#2}");
57  DefMacro!("\\vstretch {}{}",    "\\lx@scalerel@obj{#2}");
58  // `\scaleleftright`/`\stretchleftright` place a scaled left and right object around
59  // a reference; both are built on `\scalerel`/`\stretchrel` (scalerel.sty L129-137),
60  // and the `.`-sentinel `\ifx` skips an empty side.
61  DefMacro!("\\scaleleftright []{}{}{}",
62    "\\ifx.#2#3\\else\\scalerel[#1]{#2}{#3}\\fi\\ifx.#4\\else\\scalerel*[#1]{#4}{#3}\\fi");
63  DefMacro!("\\stretchleftright []{}{}{}",
64    "\\ifx.#2#3\\else\\stretchrel[#1]{#2}{#3}\\fi\\ifx.#4\\else\\stretchrel*[#1]{#4}{#3}\\fi");
65});
