1use crate::prelude::*;
2
3#[rustfmt::skip]
4LoadDefinitions!({
5  // arXiv/html_feedback#6895: the `scalerel` package has no `.ltxml` binding in
6  // Perl OR Rust, and the raw `.sty` load leaves `\scalerel` undefined — so an
7  // inline icon built with `\scalerel*` (the `\orcidicon` of arXiv:2608.12272)
8  // raised `Error:undefined:\scalerel` and rendered its picture unscaled ("too
9  // big / multi-line"). Beyond Perl 0.8.8, which errors identically.
10  //
11  // `\scalerel*[maxwidth]{obj}{ref}` scales `obj` to the height of `ref`, aspect
12  // preserved (scalerel.sty L68-84). The dominant use is an inline object scaled
13  // to the surrounding *text* height, so — box-measurement scaling being
14  // unavailable in this engine — we wrap `obj` in an inline-block that CSS sizes
15  // to text height (`.ltx_scalerel`, `LaTeXML.css`). The `[maxwidth]` cap
16  // (default `99in`, i.e. unbounded) is accepted and dropped. The starred form
17  // yields just the scaled object; the plain `\scalerel{obj}{ref}` appends `ref`
18  // afterwards (scalerel.sty L84, `\scalerelplus`).
19  DefMacro!("\\scalerel", "\\@ifstar\\lx@scalerel@star\\lx@scalerel@plus");
20  DefMacro!("\\lx@scalerel@star []{}{}", "\\lx@scalerel@obj{#2}");
21  DefMacro!("\\lx@scalerel@plus []{}{}", "\\lx@scalerel@obj{#2}#3");
22  DefConstructor!("\\lx@scalerel@obj{}",
23    "<ltx:inline-block class='ltx_scalerel'>#1</ltx:inline-block>",
24    mode => "restricted_horizontal", enter_horizontal => true, bounded => true);
25  // `\stretchrel` stretches ignoring the aspect ratio; aspect-preserving is the
26  // safe default for an inline icon, so alias it to `\scalerel` (scalerel.sty L86).
27  Let!("\\stretchrel", "\\scalerel");
28});
