4use crate::{
5  package::bm_sty::{BoldArgument, bold_math, define_bold_delimiter},
6  prelude::*,
7};
8LoadDefinitions!({
9  RequirePackage!("amsgen");
10
11  // Perl amsbsy.sty.ltxml:23, the bounded constructor under a name of its own, reverting as `\boldsymbol`. amsbsy.sty:28-37
12  // `\DeclareRobustCommand{\boldsymbol}` sets its argument in a box per math style, so a `\left` without its `\right`
13  // there (`\boldsymbol{\left\langle} x \boldsymbol{\right\rangle}`) is TeX's error at the box's end: since 65d
14  // `\left` opens a group the bounded argument would close (OXIDIZED_DESIGN #530), so it is set at the outer level,
15  // as bm.sty's `\bm` sets it, with a Warning (bm_sty.rs `bold_math`, OXIDIZED_DESIGN #544).
16  DefConstructor!("\\lx@boldsymbol@bold{}", "#1", bounded => true, font => { forcebold => true },
17    alias => "\\boldsymbol");
18  DefMacro!("\\boldsymbol{}", sub[(arg)] {
19    Ok(bold_math("\\boldsymbol", T_CS!("\\lx@boldsymbol@bold"), arg, BoldArgument::Grouped))
20  }, protected => true);
21  define_bold_delimiter()?;
22
23  // I think the intent is that you use \pmb to get bold, but your current font doesn't supply
24  // a bold for those glyphs.  Since we're just forcing the glyph to be bold,
25  // \pmb is probably \boldsymbol. or... ??
26  Let!("\\pmb", "\\boldsymbol");
27
28  // amsbsy.sty:38-39, 58-69: `\binrel@{X}` lets `\binrel@@` be `\mathbin`, `\mathrel` or `\relax` by the spacing a
29  // box measures around X, and `\math@atom{X}{Y}` sets Y as X's class. No box measures math spacing here, so
30  // `\binrel@@` passes Y through ungrouped: Y keeps the class its own content has (the operator a paper restyles is
31  // still one), where an ordinary group would hide it. Papers call them for their own operators: the first error of
32  // 161 run-340 papers (witnesses 2103.00817, 0709.1048, 1310.2330).
33  RawTeX!(
34    r"\def\binrel@#1{\def\binrel@@##1{##1}}
35\let\binrel@@\relax
36\def\math@atom#1#2{\binrel@{#1}\binrel@@{#2}}"
37  );
38});
