245/// REVTeX's `\maketitle` switches the author block off (revtex4-2.cls:2670-2684, revtex4-1.cls:2649-2662,
246/// revtex4.cls:4709-4717); ours from a repeated block's first `\author` on, its new words counted (OD #1003: 2401.15572,
247/// 1801.07635, 1812.00931, 2510.13035, 2602.09975, 2307.07324, 2001.07542; lone setters kept: 1309.4768, 1106.2389, 2304.09163).
248pub(crate) fn switch_off_author_block_after_maketitle() -> Result<()> {
249  // (not AASTeX's, which loads this binding: aastex701.cls:13231-13266 is its own)
250  if !["revtex4", "revtex4-1", "revtex4-2"]
251    .into_iter()
252    .any(is_document_class)
253  {
254    return Ok(());
255  }
256  // (local `\let`s: a repeat's `\author` inside a group, `{\author{Ann}}\email{…}`, switches off only to its end)
257  DefMacro!(
258    "\\lx@revtex@repeated@author",
259    r"\let\and\relax\let\affiliation\lx@revtex@repeated@gobble\let\email\lx@revtex@repeated@gobble@url\let\thanks\lx@revtex@repeated@gobble\lx@revtex@repeated@gobble"
260  );
261  // (the repeat's names and contacts the content check counts: a new word the output lacks warns, 1206.0114, 1903.12442)
262  DefPrimitive!("\\lx@revtex@repeated@gobble{}", sub[(content)] {
263    fm_record_switched_off_call(&content);
264  });
265  // (`_` and `@` other as the live `\email` reads them, `ann_x@x.org` whole; `%` a comment as under `\@gobble`: revRB1c q2)
266  DefPrimitive!("\\lx@revtex@repeated@gobble@url Semiverbatim", sub[(content)] {
267    fm_record_switched_off_call(&content);
268  });
269  DefPrimitive!("\\lx@revtex@maketitle@setters", sub[_args] {
270    // (a document's own `\maketitle` or `\frontmatter@maketitle` wins, 2509.25344, 2111.12484; `\title`, `\date` stay
271    // the kernel's, 2606.04868; the class's `\let` lets a later `\renewcommand` through, so the locks lift, 2412.14256)
272    if !lookup_bool("\\maketitle:redefined") && lookup_definition(&T_CS!("\\frontmatter@maketitle"))?.is_none() {
273      let_i(&T_CS!("\\author"), &T_CS!("\\lx@revtex@repeated@author"), None);
274      for cs in ["\\affiliation", "\\author", "\\email", "\\thanks"] {
275        assign_value(&s!("{cs}:locked"), false, None);
276      }
277    }
278    Ok(())
279  });
280  AddToMacro!("\\lx@maketitle@body", "\\lx@revtex@maketitle@setters");
281  Ok(())
