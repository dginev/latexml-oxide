19  // same paper without it. Default false = authors shown (correct for preprint/final
20  // arXiv uploads). Witness 2605.17249.
21  DefConditional!("\\if@anonymous");

75  // {hide} environment. Perl (neurips.sty.ltxml L59) defines it
76  // UNCONDITIONALLY, but the raw neurips_2023.sty (L336-390) only runs
77  // `\NewEnviron{hide}{}` in the SUBMISSION branch —
78  // `\if@preprint … \else \if@neuripsfinal … \else <here> \fi \fi` — so in
79  // preprint/final mode `\hide` is left undefined. That matters: papers define
80  // their own `\newcommand{\hide}[1]{}` (a brace-gobbling "comment this out"
81  // helper) and use it as `\hide{ … }`. Defining `{hide}` unconditionally
82  // shadows that `\newcommand` (silently ignored as a redefinition), so `\hide{`
83  // is parsed as the environment opener and runs away to `\end{document}`
84  // looking for `\endhide` — swallowing the whole body (everything after the
85  // abstract). Gate on submission mode (neither preprint nor final) to match the
86  // real class. SURPASSES Perl, which drops the body identically here.

99  //
100  // \AtBeginDocument-defer + \@ifundefined-guard: defer until after the

102  // `\ifx\lemma\undefined \newtheorem{lemma} \newtheorem*{lemma*} \fi`)
103  // wins. Without deferral our unconditional defs run at .sty-load time,
104  // pre-define `\lemma`, and silently suppress the user's `\newtheorem*
105  // {lemma*}` branch. Witness 2305.11788 (neurips paper + mymath.sty).
106  RawTeX!(

108\@ifundefined{theorem}{\newtheorem{theorem}{Theorem}[section]}{}%
109\@ifundefined{lemma}{\newtheorem{lemma}[theorem]{Lemma}}{}%
110\@ifundefined{corollary}{\newtheorem{corollary}[theorem]{Corollary}}{}%
111\@ifundefined{proposition}{\newtheorem{proposition}[theorem]{Proposition}}{}%
112\@ifundefined{propo}{\newtheorem{propo}[theorem]{Proposition}}{}%
113\@ifundefined{definition}{\newtheorem{definition}[theorem]{Definition}}{}%
114\@ifundefined{remark}{\newtheorem{remark}[theorem]{Remark}}{}%
115\@ifundefined{example}{\newtheorem{example}[theorem]{Example}}{}%
116\@ifundefined{claim}{\newtheorem{claim}[theorem]{Claim}}{}%
117\@ifundefined{assumption}{\newtheorem{assumption}[theorem]{Assumption}}{}%
118\@ifundefined{question}{\newtheorem{question}[theorem]{Question}}{}%
