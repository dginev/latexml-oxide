1//! showexpl.sty: `LTXexample` and `\LTXinputExample` typeset LaTeX source as a listing AND run it as the result.
2//!
3//! The raw package writes the body to `\SX@codefile` through listings' write-file layer and reads it back twice
4//! (showexpl.sty:189-267); its `\lst@newenvironment` end-group parse failed in the raw-load path (a readBalanced
5//! `Expected opening '{'`), so `\SX@put@code@result` never registered. The binding reproduces the semantics
6//! instead: the body is captured raw, typeset by the listings display engine with the environment's keys (showexpl's
7//! keys are listings keys, :46-68: `caption=`/`label=` caption the code listing), and run as the result in
8//! showexpl's preset group (`\SX@@preset`, :88-114 verbatim; `\SX@resultInput`, :348-366) — a whole-document
9//! example's `\documentclass`, `\usepackage`, `document` environment, `\label`s do nothing (showexpl-test: the
10//! conversion ended at an example's `\end{document}`, 50 → 95 % recall with the preset). What is not modelled is
11//! recorded in OXIDIZED_DESIGN_DIVERGENCES #167. Perl has no binding and skips the package as a missing file.
12//!
13//! Witnesses (package loaded, CONVERR_7/CONVERR_3 → OK): 1604.00381, 1606.01035, 1706.03232, 1804.02704,
14//! 1804.07221, 1612.01022, 1905.12059, 1706.09226, 1701.01402, 1812.06820, 1801.01025, 1806.10927, 2001.08314,
15//! 2002.09910, 1901.08750.

25/// showexpl.sty:348-366 `\SX@resultInput`: the example's result is read back in a group with `%` a comment
26/// character, `@` other, a line end a space, and showexpl's preset (`\SX@@preset`, then the `preset=` key) in force,
27/// so a whole-document example's `\documentclass`, `\usepackage`, `document` environment, `\label`, `\index`, … do
28/// nothing (`\footnote` and `\cite` are locked here and still act).
29fn result_in_preset(input: Tokens, preset: Tokens) -> Vec<Token> {
30  let mut out = TokenizeInternal!(
31    r"\begingroup\MakePercentComment\makeatother\catcode`\^^M=5\relax\SX@@preset"
32  )
33  .unlist();
34  out.extend(preset.unlist());
35  // :97-103 nulls `\cref@old@label` under cleveref, which the cleveref binding's `\label` never calls.
36  out.extend(TokenizeInternal!(r"\@ifpackageloaded{cleveref}{\let\label\@gobble}{}").unlist());
37  out.extend(input.unlist());
38  out.extend(TokenizeInternal!(r"\par\endgroup").unlist());
39  out
40}

53  listings_sty::require_listings_aspects("writefile");
54  // {LTXexample}[keys] — showexpl's whole point: typeset the SOURCE as a
55  // listing AND its RESULT (real showexpl routes the body through listings'
56  // write-file layer into \jobname.tmp and \input's it back). Reproduce that
57  // semantic directly: capture the body raw, emit the code listing through
58  // the listings display engine, then re-tokenize the body so it EXECUTES
59  // as the result. `[pos=…]`-style keys arrange the two blocks on the page —
60  // presentation. (The TL doc corpus — koma/babel/gauss manuals — uses this
61  // env heavily; the earlier noop stub predates OXIDIZED_DESIGN #161, whose
62  // DefPlain fix unblocked raw showexpl parsing, and dropped example bodies
63  // entirely.)
64  // showexpl.sty:46-68: its keys are listings keys (`\lst@Key`); `preset` is read back below, the layout keys
65  // (`pos`, `width`, `hsep`, `rframe`, …) place the two blocks on the page, and `graphic` (a precompiled image of the
66  // result) is not modelled: the result is always typeset.
67  for key in ["pos", "width", "hsep", "vsep", "overhang", "rframe", "preset", "explpreset", "codefile",
68              "justification", "graphic"] {
69    DefKeyVal!("LST", key, "");
70  }
71  DefKeyVal!("LST", "scaled", "", "!");
72  for key in ["wide", "rangeaccept", "varwidth", "attachfile"] {
73    DefKeyVal!("LST", key, "", "true");
74  }
75  // showexpl.sty:88-114 (`\SX@@preset`, `\SX@eat@version`) and :346-347, verbatim.
76  RawTeX!(r"\newcommand*\SX@@preset{%
77  \renewcommand\documentclass[2][]{\SX@eat@version}%
78  \renewcommand\usepackage[2][]{\SX@eat@version}%
79  \renewenvironment{document}{}{}%
80  \renewcommand\cite[1][]{}%
81  \let\tableofcontens\relax \let\listoffigures\relax
82  \let\listoftables\relax \let\printindex\relax
83  \let\listfiles\relax \let\nofiles\relax
84  \let\index\@gobble
85  \expandafter\ifx\csname ver@cleveref.sty\endcsname\relax
86    \let\refstepcounter=\stepcounter
87    \let\label\@gobble
88  \else
89    \let\cref@old@refstepcounter=\stepcounter
90    \let\cref@old@label=\@gobble
91  \fi
92  \let\bibliography\@gobble
93  \let\pagestyle\@gobble \let\thispagestyle\@gobble
94  \renewcommand\marginpar[2][]{}%
95  \renewcommand\footnote[2][]{}%
96  \let\@footnotetext\@gobble
97}
98\newcommand*\SX@eat@version[1][]{}
99\providecommand*\MakePercentIgnore{\catcode`\%9\relax}
100\providecommand*\MakePercentComment{\catcode`\%14\relax}");
101  // {LTXexample}[keys] — showexpl's whole point: the SOURCE as a listing AND its RESULT. Real showexpl writes the
102  // body to `\SX@codefile` and reads it back twice (showexpl.sty:189-267): typeset as code with the keys
103  // (`\lstset{\SX@explpreset,#1}`), and run as the result in the preset group. The keys are activated as
104  // `lstlisting`'s are, so `caption=`/`label=` caption the code listing (showexpl puts one caption around the pair,
105  // :245/:261). The result was run without the preset: a whole-document example ended the conversion at its own
106  // `\end{document}` and its `\label`s leaked (showexpl-test, 50 % recall; RUST-ONLY).
107  {
108    let cs = T_CS!("\\begin{LTXexample}");
109    let params = parse_parameters("OptionalKeyVals:LST", &cs, true)?;
110    let expansion: Option<ExpansionBody> = Some(ExpansionBody::Closure(Rc::new(
111      move |args: Vec<ArgWrap>| {
112        let kv: Option<KeyVals> = args.into_iter().next().unwrap_or_default().into();
113        bgroup();
114        assign_value("current_environment", Stored::String(pin("LTXexample")), None);
115        lst_activate(kv.as_ref());
116        let text = listings_read_raw_lines("LTXexample");
117        let name = lst_get_tokens("name");
118        let code = lst_process_display_scoped(if name.is_empty() { None } else { Some(name) }, &text);
119        let preset = lst_get_tokens("preset");
120        egroup()?;
121        let mut out = lst_group_opener("LTXexample", kv.as_ref())?;
122        out.extend(code);
123        out.extend(result_in_preset(lst_run_body_via_input("LTXexample", &text)?, preset));
124        let mut end_tokens = vec![T_CS!("\\end"), T_BEGIN!()];
125        end_tokens.extend(ExplodeText!("LTXexample"));
126        end_tokens.push(T_END!());
127        unread_expansion(Tokens::new(end_tokens));
128        Ok(Tokens::new(out))
129      },
130    )));
131    def_macro(cs, params, expansion, None)?;
132    DefMacro!(T_CS!("\\end{LTXexample}"), None, Tokens!());
133  }
134  def_macro_noop("\\endLTXexample")?;
135  // \LTXinputExample[keys]{file} (showexpl.sty:396-398): the file as code, then run in the preset group (its
136  // `preset=` key is not read back here).
137  DefMacro!("\\LTXinputExample[]{}", "\\lstinputlisting[#1]{#2}\\lx@SX@resultinput{#2}");
138  DefMacro!("\\lx@SX@resultinput{}", sub[(file)] {
139    let mut input = vec![T_CS!("\\input"), T_BEGIN!()];

144  def_macro_noop("\\setupSXfiles")?;
145  def_macro_noop("\\setupLZfiles")?;

159    r"\newcommand*\SX@justification{\raggedright}",
