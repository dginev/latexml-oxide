433      if (artifact || actualtext.is_some() || alt.is_some())
434        && let Some(mut last_child) = document.get_node().get_last_child() {
435          let description = actualtext
436            .clone()
437            .or_else(|| alt.clone())
438            .unwrap_or_default();
439          document.set_attribute(&mut last_child, "description", &description)?;
440          if let (Some(alt_str), Some(_)) = (&alt, &actualtext) {
441            document.set_attribute(&mut last_child, "aria:description", alt_str)?;
442          }
443          if artifact {
444            document.set_attribute(&mut last_child, "aria:hidden", "true")?;
445          }
446        }
447    }
448  );
449  // The width it states is a float panel's (the panel grid rule, `declare_panel_width`): its `width`, not its height
450  // (1703.09474 Fig 13, 2503.09651 Fig 1), which scales its width where it states none (graphicx has no `totalwidth`
451  // key, review rev78x2 finding 7).
452  declare_panel_width("\\@includegraphicx", 2, &["width"], &[
453    "height",
454    "totalheight",
455  ]);
456});

458/// A file name's characters as written, whatever their catcodes (a subscript `_` is `_`, not its display rendering);
459/// a control sequence left by the expansion keeps its printed name.
460fn raw_name(tokens: &Tokens) -> String {
461  let mut name = String::new();
462  for t in tokens.unlist_ref() {
463    if t.get_catcode() == Catcode::CS {
464      name.push_str(&t.to_string());
465    } else {
466      with(t.get_sym(), |c| name.push_str(c));
467    }
468  }
469  name
470}

480/// The extension graphics.sty keys a file's rule by (`\Gin@rule@<ext>`, :213-231 `\Gin@getbase`): from the last `.` of
481/// the file's own name (`fig.v2.tex` → `.tex`, `x.pdf_tex` → `.pdf_tex`); none for a name without one.
482fn graphics_rule_extension(name: &str) -> Option<String> {
483  let base = name.rsplit('/').next().unwrap_or(name);
484  base
485    .rfind('.')
486    .filter(|at| *at > 0)
487    .map(|at| base[at..].to_string())
488}

491/// (`\graphicspath`, the source directory), else where `\input` finds it; `None` when neither does.

497    .map(str::to_string)
498    .or_else(|| find_file(name, None).map(|_| name.to_string()))
