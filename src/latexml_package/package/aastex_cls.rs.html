305/// Whether the paper ships an `aastex.cls` that is an AASTeX 6+ class under the old name — one that defines `\gridline`
306/// (aastex6.cls:85 `\ProvidesClass{aastex}`); TeX Live has none, so the file found is the paper's own.
307fn bundled_aastex_is_6_plus() -> bool {
308  find_file(
309    "aastex",
310    Some(FindFileOptions {
311      ext_type: Some("cls".into()),
312      ..FindFileOptions::default()
313    }),
314  )
315  .and_then(|path| std::fs::read(path).ok())
316  .is_some_and(|bytes| String::from_utf8_lossy(&bytes).contains("\\gridline"))
317}

319/// Leave `{plate}` and `{plate*}` (`\begin{plate}`, `\end{plate}`, `\plate`, `\endplate`; aas_support_sty.rs) undefined,
320/// for an AAS class that has no plate: AASTeX 6 and later, emulateapj.
321pub fn undefine_plate() {
322  for command in [
323    "\\plate",
324    "\\endplate",
325    "\\begin{plate}",
326    "\\end{plate}",
327    "\\plate*",
328    "\\endplate*",
329    "\\begin{plate*}",
330    "\\end{plate*}",
331  ] {
332    Let!(T_CS!(command), "\\@undefined");
333  }
