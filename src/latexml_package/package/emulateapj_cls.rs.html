99  DefPrimitive!("\\lx@eapj@head@extend", {
100    if let Some(Stored::Int(head)) = lookup_value("lx@eapj@head@rows") {
101      assign_value("lx@eapj@head@rows", Stored::None, Some(Scope::Global));
102      if let Some(alignment) = lookup_alignment()
103        && let Some(data) = alignment.alignment_cell() {
104          let mut al = data.borrow_mut();
105          // The rows since the head, without the last: the pseudo-row of this `\noalign`, the current row. They go
106          // back in place, so the row indices stay.
107          let mut rows = Vec::new();
108          while al.rows().len() > head as usize && let Some(row) = al.remove_row() {
109            rows.push(row);
110          }
111          rows.reverse();
112          let last = rows.len().saturating_sub(1);
113          for row in rows.iter_mut().take(last) {
114            // A row with content is a heading row; an empty one (the `\\` after the head) and a rule between them
115            // (`\cline`) are on the first page in no TeX table (a continuation head is not), so they are emptied
116            // into pseudo-rows without rules, which the alignment's normalization drops.
117            let empty = row.is_pseudo()
118              || row.get_columns().iter().all(|col| col.boxes.as_ref().is_none_or(Digested::is_skippable));
119            for col in row.get_columns_mut() {
120              if empty {
121                col.border.clear();
122                col.boxes = None;
123              } else {
124                col.thead_in_column = true;
125              }
126            }
127            if empty {
128              row.set_pseudo();
129            }
130          }
131          // With heading rows promoted, the head's own rule (`#1\\\hline`, a pseudo-row ending the head) goes under
132          // the last of them, above the first data row, where TeX's first page has it (the promoted rows are not on
133          // it; 1605.03270 Table 1, OD #663): it moves to the last row, this `\noalign`'s, whose rules the
134          // normalization gives the next row.
135          let promoted = rows[..last].iter().any(|row| !row.is_pseudo());
136          if promoted && let Some(current) = rows.last_mut() {
137            let mut rules = Vec::new();
138            while al.rows().back().is_some_and(|row| row.is_pseudo()) && let Some(mut rule) = al.remove_row() {
139              for (col, target) in rule.get_columns_mut().iter_mut().zip(current.get_columns_mut().iter_mut()) {

141                col.border.retain(|c| !matches!(c, 't' | 'T'));
142              }
143              rules.push(rule);
144            }
145            rules.reverse();
146            al.append_rows(rules);
147          }
148          al.append_rows(rows);
149        }
150    }
