1//! blkarray — block-array matrices.
2//!
3//! The raw `blkarray.sty` builds each matrix cell as inline math inside a raw
4//! `\halign`/`\ialign`, and its `block` delimiter machinery, digested inside
5//! display math, drives BOTH LaTeXML engines into the `\halign`-in-math runaway
6//! (Rust OOMs at the 4.5 GB cap in ~12 s; same-host Perl hangs ~90 s → rc=124;
7//! `pdflatex` renders fine). Full analysis + 4-line reproducer:
8//! `docs/archive/known_crashes/blkarray_halign_math/`. This binding SHADOWS the raw
9//! `.sty` (so it is never raw-loaded, even under `--includestyles`) and routes
10//! `blockarray`/`block` through the engine's well-behaved `array` alignment
11//! machinery instead. Surpass-Perl: upstream LaTeXML has no `blkarray.sty.ltxml`.
12//! Witnesses: arXiv:1811.10792 (ar5iv #594), arXiv:2310.17416 (ar5iv #473).
13//!
14//! **Faithfulness note (documented simplification).** In `blkarray`, a `block`'s
15//! column-spec delimiters (`(`/`[` in e.g. `block{c(cccccc)}`) wrap a SUB-REGION
16//! of the shared matrix — a construct LaTeXML's `array` cannot express (its
17//! `left=`/`right=` wrap the whole matrix). We render each `block` transparently:
18//! its rows flow into the single `blockarray` alignment (correct structure +
19//! content, and the label row/column are preserved), formatted by the block's own
20//! columns (`>{}`/`<{}`, OXIDIZED_DESIGN_DIVERGENCES #771), but the block's delimiter
21//! parentheses are DROPPED. This is chosen deliberately over the raw `.sty`'s OOM
22//! — a matrix without its outer parens beats losing the whole document section.
23use crate::prelude::*;
24
25LoadDefinitions!({
26  // `\begin{blockarray}[pos]{spec}` is defined as a MAGIC control sequence
27  // (`\begin{...}` takes the defined-CS fast path in latex_constructs.rs:3064,
28  // which does NOT inject a `\begingroup`). That is essential: a transparent
29  // `block` nested inside must not open a group that crosses the alignment's
30  // `\\` row boundaries. Route to the same machinery as `\array`
31  // (latex_constructs.rs `\@array@bindings`/`\@@array`/`\lx@begin@alignment`).
32  // blkarray's own blockarray column specs are plain (`c cccccc`, `cccc`) — the
33  // delimiters live on the block specs, which drop them (below) — so the spec passes to
34  // the AlignmentTemplate parser unchanged.
35  DefMacro!(
36    T_CS!("\\begin{blockarray}"),
37    "[]{}",
38    "\\let\\begin\\lx@blkarray@begin\\@array@bindings[#1]{#2}\\lx@blkarray@repeating\\@@array[#1]{#2}\\lx@begin@alignment"
39  );
40  // blkarray.sty:1099-1102, :1123 `\\let\\begin\\BA@begin`: inside a `{blockarray}` a `\\begin` of a block is expandable,
41  // its `\\noalign` reached by the look-ahead at a cell's head, where the kernel's `\\protected` `\\begin` stops.
42  RawTeX!(
43    r"\let\lx@blkarray@@begin\begin
44\def\lx@blkarray@begin#1{\ifcsname lx@blkarray@begin@#1\endcsname\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
45  {\csname begin{#1}\endcsname}{\lx@blkarray@@begin{#1}}}
46\let\lx@blkarray@begin@block\@empty
47\expandafter\let\csname lx@blkarray@begin@block*\endcsname\@empty"
48  );
49  // blkarray.sty:1206-1208 aligns with a repeating preamble, `\BA@upart##\BA@vpart&&\BA@upart##\BA@vpart\cr`, so
50  // a row may run past its spec's columns; such a cell gets no format (`\BA@col@use`'s `\csname` of a column the spec
51  // never defined is `\relax`, :838-839). Witness 2301.06399 (a 16-cell row under a 15-column spec).
52  DefPrimitive!("\\lx@blkarray@repeating", {
53    if let Some(alignment) = lookup_alignment()
54      && let Some(alignment) = alignment.alignment_cell()
55    {
56      alignment
57        .borrow_mut()
58        .get_template_mut()
59        .repeat_past_columns(Cell::default());
60    }
61  });
62  // blkarray.sty:749-750; papers set it (`\setlength\BA@colsep{4pt}`, 2301.06399).
63  RawTeX!(r"\newdimen\BA@colsep \BA@colsep=\tabcolsep");
64  // 68b (binding gaps G3): the public blkarray commands papers use inside a `{blockarray}`.
65  // blkarray.sty:934-947 `\BAmulticolumn{n}{spec}{text}`, blkarray's `\multicolumn` (1202.0500, 1204.2176,
66  // 1301.3189, 1311.0327, 1401.4921, 1407.4836).
67  Let!("\\BAmulticolumn", "\\multicolumn");
68  // :1677-1683 `\BAnewcolumntype[n]{c}{spec}`, a column type for blkarray's preambles (1302.2340, 1402.5950,
69  // 1408.0807, 1702.06723, 2005.02853: `\BAnewcolumntype{s}{>{$\small$}c}`): the rewrite the alignment template
70  // reads a column letter through, as array's `\newcolumntype` makes it (`\NC@rewrite@<c>`, alignment.rs).
71  RawTeX!(
72    r"\def\BAnewcolumntype{\@ifnextchar[{\lx@BA@nct}{\lx@BA@nct[0]}}
73\def\lx@BA@nct[#1]#2#3{\expandafter\@reargdef\csname NC@rewrite@\string#2\endcsname[#1]{#3}}"
74  );
75  // :1063 `\BAextrarowheight` (1312.6738, 1503.04543, 1503.08375 set it), the height blkarray adds to each row.
76  RawTeX!(r"\newdimen\BAextrarowheight");
77  // :1005-1012 `\BAnoalign{material}`/`\BAnoalign*{material}`: material between rows, which the star also lets
78  // the current block's delimiters span (1912.05110 `\BAnoalign{\vskip -7ex}`).
79  RawTeX!(
80    r"\def\BAnoalign{\@ifstar\lx@BA@noalign\lx@BA@noalign}\long\def\lx@BA@noalign#1{\noalign{#1}}"
81  );
82  // blkarray.sty:2057-2068: `\BAhline` rules every column (`\BAhhline{*{\BA@col@max}{-}}`), as `\hline` does.
83  Let!("\\BAhline", "\\hline");
84  DefMacro!("\\BAhhline Semiverbatim", sub[(spec)] {
85    let rules: String = ba_hhline_ruled_columns(&spec.to_string())
86      .into_iter()
87      .map(|(first, last)| s!("\\cline{{{first}-{last}}}"))
88      .collect();
89    Ok(Tokenize!(TeXString::assembled(rules)))
90  });
91  DefMacro!(
92    T_CS!("\\end{blockarray}"),
93    None,
94    "\\lx@end@alignment\\@end@array\\let\\begin\\lx@blkarray@@begin"
95  );

97  // A `block`'s own preamble formats its rows (blkarray.sty:868-922 `\BA@beginblock`, `\begin{block}` inside a
98  // `{blockarray}` by :1099-1102 `\BA@begin`): a `\noalign` that parses the block's columns, which its rows use
99  // (`\BA@col@use` keys each column's `>{}`/`<{}` parts by the block, :838-839) until `\BA@endblock`'s `\noalign`
100  // puts the enclosing preamble back (:885-892). The delimiters are not columns: `(`, `)`, `[`, `]`, `\{`, `\}`
101  // are blkarray's `\Left`/`\Right` (:1998-2003), which draw a parenthesis beside the block's sub-region, a shape
102  // the `array` alignment cannot express, so they are dropped. A block that changed no format (`{c(cc)}`) is the
103  // transparent block it was. `\begin{block}{*{2}{>{$\footnotesize}c<{$}}}` keeps its cells' `$`/`$` wrappers
104  // (1706.05431, 2207.06508, 2210.16910; OXIDIZED_DESIGN_DIVERGENCES #771). Magic CSes: no `\begingroup` around the
105  // block, so its boundary does not cross the alignment's rows.
106  // The `\noalign` is the hidden one, which adds no row of its own.
107  DefMacro!(
108    T_CS!("\\begin{block}"),
109    "{}",
110    r"\lx@hidden@noalign{\lx@blkarray@columntypes\lx@blkarray@delimiters\lx@blkarray@block{#1}}"
111  );
112  DefMacro!(
113    T_CS!("\\end{block}"),
114    None,
115    r"\lx@hidden@noalign{\lx@blkarray@endblock}"
116  );
117  DefMacro!(
118    T_CS!("\\begin{block*}"),
119    "{}",
120    r"\lx@hidden@noalign{\lx@blkarray@columntypes\lx@blkarray@delimiters\lx@blkarray@block{#1}}"
121  );
122  DefMacro!(
123    T_CS!("\\end{block*}"),
124    None,
125    r"\lx@hidden@noalign{\lx@blkarray@endblock}"
126  );
127  // blkarray's own column types for a block's preamble, local to the group it is read in: `>{}`/`<{}` (:1714-1725,
128  // :1818-1831, array.sty's without it) and the delimiters, which are no columns.
129  DefPrimitive!("\\lx@blkarray@columntypes", {
130    DefColumnType!(">{}", sub[(before)] {
131      let before = VecDeque::from(before.unlist());
132      with_building_template(|template| template.add_before_column(before));
133    });
134    DefColumnType!("<{}", sub[(after)] {
135      with_building_template(|template| template.add_after_column(after.unlist()));
136    });
137  });
138  RawTeX!(
139    r"\def\lx@blkarray@delimiters{\BAnewcolumntype{(}{}\BAnewcolumntype{)}{}\BAnewcolumntype{[}{}\BAnewcolumntype{]}{}\BAnewcolumntype{\{}{}\BAnewcolumntype{\}}{}\BAnewcolumntype[2]{\Left}{}\BAnewcolumntype[2]{\Right}{}}"
140  );
141  DefPrimitive!("\\lx@blkarray@block AlignmentTemplate", sub[(block)] {
142    let Some(alignment) = lookup_alignment() else { return Ok(Vec::new()) };
143    let Some(alignment) = alignment.alignment_cell() else { return Ok(Vec::new()) };
144    let mut alignment = alignment.borrow_mut();
145    let template = alignment.get_template_mut();
146    push_value("lx@blkarray@preambles", Stored::Template(Rc::new(template.clone())))?;
147    for (n, column) in block.get_columns().iter().enumerate() {
148      let mut column = column.clone();
149      column.mark_intercol_after();
150      match template.get_columns_mut().get_mut(n) {
151        Some(slot) => *slot = column,
152        None => template.get_columns_mut().push(column),
153      }
154    }
155  });
156  DefPrimitive!("\\lx@blkarray@endblock", {
157    if let Some(Stored::Template(outer)) = pop_value("lx@blkarray@preambles")?
158      && let Some(alignment) = lookup_alignment()
159      && let Some(alignment) = alignment.alignment_cell()
160    {
161      *alignment.borrow_mut().get_template_mut() = (*outer).clone();
162    }
163  });
164});

166/// The column runs `\BAhhline{spec}` rules (blkarray.sty:2074-2210), for one `\cline` each: `-` `=` `.` `"` rule a
167/// column and `~` leaves one bare, each moving to the next column unless the current one is still unused; `&` moves to
168/// the next; `*{n}{x}` repeats `x`; `|` `:` `#` `t` `b` draw vertical pieces between columns.
169fn ba_hhline_ruled_columns(spec: &str) -> Vec<(usize, usize)> {
170  let mut ruled: Vec<usize> = Vec::new();
171  let (mut column, mut unused) = (1, true);
172  for c in ba_hhline_repeats(spec).chars() {
173    match c {
174      '-' | '=' | '.' | '"' | '~' => {
175        if !unused {
176          column += 1;
177        }
178        unused = false;
179        if c != '~' {
180          ruled.push(column);
181        }
182      },
183      '&' => {
184        column += 1;
185        unused = true;
186      },
187      _ => {},
188    }
189  }
190  let mut runs: Vec<(usize, usize)> = Vec::new();
191  for column in ruled {
192    match runs.last_mut() {
193      Some((_, last)) if *last + 1 == column => *last = column,
194      _ => runs.push((column, column)),
195    }
196  }
197  runs
198}

200/// `spec` with each `*{n}{x}` written out `n` times (blkarray.sty:2198-2207; either argument may be one character).
201fn ba_hhline_repeats(spec: &str) -> String {
202  fn argument(chars: &[char], mut i: usize) -> Option<(String, usize)> {
203    while chars.get(i) == Some(&' ') {
204      i += 1;
205    }
206    match chars.get(i)? {
207      '{' => {
208        let mut depth = 0;
209        for (j, &c) in chars.iter().enumerate().skip(i) {
210          match c {
211            '{' => depth += 1,
212            '}' => {
213              depth -= 1;
214              if depth == 0 {
215                return Some((chars[i + 1..j].iter().collect(), j + 1));
216              }
217            },
218            _ => {},
219          }
220        }
221        None
222      },
223      &c => Some((c.to_string(), i + 1)),
224    }
225  }
226  let chars: Vec<char> = spec.chars().collect();
227  let mut out = String::new();
228  let mut i = 0;
229  while i < chars.len() {
230    if chars[i] == '*'
231      && let Some((count, next)) = argument(&chars, i + 1)
232      && let Some((body, next)) = argument(&chars, next)
233    {
234      out.push_str(&ba_hhline_repeats(&body).repeat(count.trim().parse().unwrap_or(0)));
235      i = next;
236    } else {
237      out.push(chars[i]);
238      i += 1;
239    }
240  }
241  out
