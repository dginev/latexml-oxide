3/// thm-restate.sty:35-49 `\thmt@collect@body`: the tokens up to the first `\end{<env>}` outside braces (the `\end` of
4/// another environment is body), the `\end{<env>}` left in the input.
5fn collect_body(env: &str) -> Result<Vec<Token>> {
6  let end_delim = Tokens!(T_CS!("\\end"));
7  let mut body: Vec<Token> = Vec::new();
8  loop {
9    let Some(text) = read_until(&end_delim)? else {
10      break; // end of input: the environment's own runaway
11    };
12    body.extend(text.unlist());
13    let name = read_arg(ExpansionLevel::Off)?;
14    let mut end = vec![T_CS!("\\end"), T_BEGIN!()];
15    end.extend(name.unlist_ref().iter().copied());
16    end.push(T_END!());
17    if name.to_string() == env {
18      unread_vec(end);
19      break;
20    }
21    body.extend(end);
22  }
23  Ok(body)
24}

36/// `<tokens>` under `\ifcsname the<counter of env>\endcsname`: thm-restate.sty:123-133 fakes `\the<env>`/`\c@<env>`
37/// only, so a wrapper environment's inner theorem (ltexpprt.sty:216-217 `\newenvironment{theorem}{\begin{@theorem}}`,
38/// no `\thetheorem`) steps its own counter and prints the next number (1911.08418, 2410.24104, 2503.21951).
39fn if_numbered(env: &str, inner: Vec<Token>) -> Vec<Token> {
40  let mut tokens = vec![T_CS!("\\ifcsname")];
41  tokens.extend(Explode!("the"));
42  tokens.extend([T_CS!("\\lx@counterfor"), T_BEGIN!()]);
43  tokens.extend(Explode!(env));
44  tokens.extend([T_END!(), T_CS!("\\endcsname")]);
45  tokens.extend(inner);
46  tokens.push(T_CS!("\\fi"));
47  tokens
48}

50/// What a theorem stored as `<name>` records right after its head: its number (`\the<counter>`, latex_constructs
51/// `\lx@the@@`, frozen; nothing where `\the<env>` is undefined, `if_numbered`), the one a starred restatement prints
52/// (thm-restate.sty:130-132 fakes `\the<env>` from the label `thmt@@<name>` that this theorem sets, :158-160)…
53fn renumber(name: &str, env: &str) -> Tokens {
54  let mut number = vec![
55    T_CS!("\\expandafter"),
56    T_CS!("\\protected@xdef"),
57    T_CS!("\\csname"),
58  ];
59  number.extend(Explode!(s!("lx@thm@restate@num@{name}")));
60  number.extend([
61    T_CS!("\\endcsname"),
62    T_BEGIN!(),
63    T_CS!("\\lx@the@@"),
64    T_BEGIN!(),
65  ]);
66  number.extend(Explode!(env));
67  number.extend([T_END!(), T_END!()]);
68  let mut tokens = if_numbered(env, number);
69  // …and the equation counter before its body, for the equations restated with it (:89-101, :146-152)
70  tokens.extend([T_CS!("\\expandafter"), T_CS!("\\xdef"), T_CS!("\\csname")]);
71  tokens.extend(Explode!(s!("lx@thm@restate@eq@{name}")));
72  tokens.extend([
73    T_CS!("\\endcsname"),
74    T_BEGIN!(),
75    T_CS!("\\the"),
76    T_CS!("\\c@equation"),
77    T_END!(),
78  ]);
79  Tokens::new(tokens)
80}

100/// `\lx@thm@restate@store{<end>}{<env>}{<name>}{<opt>}` and `…@store@star` (below): `star` for `{restatable*}`.
101fn store(star: bool, end: Tokens, env: Tokens, name: Tokens, opt: Tokens) -> Result<Tokens> {
102  // (the environment kept as `{restatable}` received it, for the restatement to expand when it runs — thm-restate.sty
103  // :153-163 stores `#2` in `\thmt@stored@<name>`: `{restatable}{\retheorem}{x}` with `\newcommand\retheorem{theorem}`,
104  // renewed to `rtheorem` before `\x*`, "Restated Theorem" (1410.0855, 1411.0871); its expansion now names the counter
105  // the number is frozen by; 1411.1619 1504.05476 1511.01379)
106  let raw_env = env.clone();
107  let (end, env, name) = (end.to_string(), Expand!(env).to_string(), name.to_string());
108  let body = collect_body(&end)?;
109  assign_value(
110    &stored("env", &name),
111    Stored::Tokens(raw_env),
112    Some(Scope::Global),
113  );
114  assign_value(
115    &stored("opt", &name),
116    Stored::Tokens(opt),
117    Some(Scope::Global),
118  );
119  assign_value(
120    &stored("body", &name),
121    Stored::Tokens(Tokens::new(body.clone())),
122    Some(Scope::Global),
123  );
124  let mut tokens = Vec::new();
125  if star {
126    // thm-restate.sty:207-213, :130-145: `{restatable*}` is not the one — its number, and a `\<name>*`'s until then,
127    // is the later `\<name>`'s that sets the label last (:158-160; each sets it, pdflatex keeps the last), as
128    // `continues=` takes its theorem's (the equation counter stored as for the one)
129    tokens.extend(renumber(&name, &env).unlist());
130    let mut number = vec![T_CS!("\\expandafter"), T_CS!("\\gdef"), T_CS!("\\csname")];
131    number.extend(Explode!(s!("lx@thm@restate@num@{name}")));
132    number.extend([T_CS!("\\endcsname"), T_BEGIN!()]);
133    number.extend(restate_ref(&name));
134    number.push(T_END!());
135    tokens.extend(if_numbered(&env, number));
136  } else {
137    tokens.extend(label(&name));
138    tokens.extend(renumber(&name, &env).unlist());
139  }
140  // `\<name>`: `\@ifstar{\lx@thm@restate*{<name>}}{\lx@thm@restate{<name>}}`
141  tokens.extend([T_CS!("\\expandafter"), T_CS!("\\gdef"), T_CS!("\\csname")]);
142  tokens.extend(Explode!(&name));
143  tokens.extend([
144    T_CS!("\\endcsname"),
145    T_BEGIN!(),
146    T_CS!("\\@ifstar"),
147    T_BEGIN!(),
148    T_CS!("\\lx@thm@restate"),
149    T_OTHER!("*"),
150    T_BEGIN!(),
151  ]);
152  tokens.extend(Explode!(&name));
153  tokens.extend([
154    T_END!(),
155    T_END!(),
156    T_BEGIN!(),
157    T_CS!("\\lx@thm@restate"),
158    T_BEGIN!(),
159  ]);
160  tokens.extend(Explode!(&name));
161  tokens.extend([T_END!(), T_END!(), T_END!()]);
162  tokens.extend(body);
163  Ok(Tokens::new(tokens))
164}
