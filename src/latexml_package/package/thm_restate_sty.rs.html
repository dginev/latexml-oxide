3/// thm-restate.sty:35-49 `\thmt@collect@body`: the tokens up to the first `\end{<env>}` outside braces (the `\end` of
4/// another environment is body), the `\end{<env>}` left in the input.
5fn collect_body(env: &str) -> Result<Vec<Token>> {
6  let end_delim = Tokens!(T_CS!("\\end"));
7  let mut body: Vec<Token> = Vec::new();
8  loop {
9    let Some(text) = read_until(&end_delim)? else {
10      break; // end of input: the environment's own runaway
11    };
12    body.extend(text.unlist());
13    let name = read_arg(ExpansionLevel::Off)?;
14    let mut end = vec![T_CS!("\\end"), T_BEGIN!()];
15    end.extend(name.unlist_ref().iter().copied());
16    end.push(T_END!());
17    if name.to_string() == env {
18      unread_vec(end);
19      break;
20    }
21    body.extend(end);
22  }
23  Ok(body)
24}

26/// The state key of a stored theorem's `part` (`env`, `opt`, `body`).
27fn stored(part: &str, name: &str) -> String { s!("thm_restate_{part}:{name}") }

29fn stored_tokens(part: &str, name: &str) -> Vec<Token> {
30  match lookup_value(&stored(part, name)) {
31    Some(Stored::Tokens(tokens)) => tokens.unlist(),
32    _ => Vec::new(),
33  }
34}

36/// What a theorem stored as `<name>` records right after its head: its number (`\the<counter>`, latex_constructs
37/// `\lx@the@@`, frozen), the one a starred restatement prints (thm-restate.sty:130-132 fakes `\the<env>` from the label
38/// `thmt@@<name>` that this theorem sets, :158-160)…
39fn renumber(name: &str, env: &str) -> Tokens {
40  let mut tokens = vec![
41    T_CS!("\\expandafter"),
42    T_CS!("\\protected@xdef"),
43    T_CS!("\\csname"),
44  ];
45  tokens.extend(Explode!(s!("lx@thm@restate@num@{name}")));
46  tokens.extend([
47    T_CS!("\\endcsname"),
48    T_BEGIN!(),
49    T_CS!("\\lx@the@@"),
50    T_BEGIN!(),
51  ]);
52  tokens.extend(Explode!(env));
53  tokens.extend([T_END!(), T_END!()]);
54  // …and the equation counter before its body, for the equations restated with it (:89-101, :146-152)
55  tokens.extend([T_CS!("\\expandafter"), T_CS!("\\xdef"), T_CS!("\\csname")]);
56  tokens.extend(Explode!(s!("lx@thm@restate@eq@{name}")));
57  tokens.extend([
58    T_CS!("\\endcsname"),
59    T_BEGIN!(),
60    T_CS!("\\the"),
61    T_CS!("\\c@equation"),
62    T_END!(),
63  ]);
64  Tokens::new(tokens)
65}

67/// `\label{restatable:<name>}`, the label of a stored theorem's numbered instance (thm-restate.sty:158-160
