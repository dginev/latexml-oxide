1//! OT2 font encoding: the Washington University Cyrillic fonts (wncyr, ot2enc.def), Cyrillic in the Latin slots.
2//!
3//! Each slot decodes to the glyph wncyr10 draws there (wncyr10.afm's names: `A` А, `C` Ц, `Q` Ч, `V` В, `W` Щ, `X` Ш,
4//! `Y` Ы, `J` Ј …), the letters as ot2enc.def names them (`\DeclareTextSymbol{\CYRNJE}{OT2}{0}` … `{\cyrhrdsn}{OT2}
5//! {127}`, Unicode from ot2enc.dfu): 32, 36, 38 and 64 are the accents of `\"`, `\U`, `\'` and `\u`, 61 the dotless i
6//! of `\.{\i}` (і); the fonts have no upper half. The transliteration's digraphs are the fonts' ligatures (`sh` ш,
7//! `ch` ч, `zh` ж, `shch` щ, `ya` я, `e1` э …: `ot2_ligature`). Perl has no OT2 map: the encoding fell back to OT1, so
8//! `\CYRZH` printed nothing and `{\fontencoding{OT2}\selectfont Moskva}` printed "Moskva" where pdflatex prints
9//! "Москва". OXIDIZED_DESIGN_DIVERGENCES #777.

170/// A run of characters of the ligature program, as the font typesets it: decoded back to its slots, ligated as TeX does
171/// (tex.web §1035-1040: the current character and the next form the ligature, which is current again), decoded again.
172fn ot2_ligature(run: &str, map: &[Option<char>], slots: &FxHashMap<char, u8>) -> String {
173  let mut codes = run.chars().filter_map(|c| slots.get(&c).copied());
174  let Some(mut current) = codes.next() else {
175    return run.to_string();
176  };
177  let mut out = String::new();
178  let emit = |code: u8, out: &mut String| {
179    if let Some(Some(c)) = map.get(usize::from(code)) {
180      out.push(*c);
181    }
182  };
183  for next in codes {
184    match LIGATURES
185      .iter()
186      .find(|(a, b, _)| *a == current && *b == next)
187    {
188      Some((_, _, ligature)) => current = *ligature,
189      None => {
190        emit(current, &mut out);
191        current = next;
192      },
193    }
194  }
195  emit(current, &mut out);
196  out
197}

199fn is_ot2(font: &Font) -> bool {
200  font
201    .get_encoding()
202    .is_some_and(|encoding| encoding == "OT2")
203}

205LoadDefinitions!({
206  let map = ot2_map();
207  DeclareFontMap!("OT2", map.clone());
208  let slots: FxHashMap<char, u8> = map
209    .iter()
210    .enumerate()
211    .filter_map(|(code, c)| Some(((*c)?, u8::try_from(code).ok()?)))
212    .collect();
213  let alphabet: String = LIGATURES
214    .iter()
215    .flat_map(|(a, b, _)| [*a, *b])
216    .filter(|code| !matches!(*code, b'-' | 123))
217    .filter_map(|code| map[usize::from(code)])
218    .collect::<std::collections::BTreeSet<char>>()
219    .into_iter()
220    .map(|c| regex::escape(&c.to_string()))
221    .collect();
222  DefLigature!(
223    &format!("[{alphabet}]{{2,}}"),
224    |caps: &regex::Captures| ot2_ligature(&caps[0], &map, &slots),
225    fontTest => sub[font] { is_ot2(font) }
226  );
227});
