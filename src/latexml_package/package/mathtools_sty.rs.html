930  // \DeclarePairedDelimiterX\cmd[nargs]{left}{right}{body} and the XPP form with pre- and postcode, as mathtools.sty
931  // :924-990 (`\DeclarePairedDelimiterXPP`, `\MT_delim_inner_generator:nnnnnnn`): in a group, `\delimsize` is the optional
932  // size (`\middle` starred, empty by default), then the precode, the left delimiter, the body, the right delimiter, the
933  // postcode. A starred delimiter is `\left`/`\right`; a sized one takes `\<size>l`/`\<size>r` before it
934  // (`\csname\MH_cs_to_str:N\delimsize l\endcsname`, `\relax` when undefined), and the delimiters themselves may use
935  // `\delimsize`. Perl (mathtools.sty.ltxml:677-692) puts the size itself before the delimiter and defines
936  // `\delimsize` outside any group: with `\delimsize` in the delimiters (`[\cbig]` where `\cbig#1` is
937  // `\mathopen{\hbox{$\big#1$}}`) the size read the delimiter's `\delimsize` as its argument and the closing `$` of
938  // its box was swallowed — 2609.17447, 100 such pairs, Fatal TooManyErrors.
939  fn declare_paired_delimiter_x(cs: Token, n: usize, pre: Vec<Token>, ldel: Vec<Token>, rdel: Vec<Token>,
940    post: Vec<Token>, body: Vec<Token>) -> Result<()> {
941    use latexml_core::definition::ExpansionBody;
942    let cmd = cs.to_string();
943    let inner_cs_name = s!("{}@inner", cmd);
944    let lsize_cs_name = s!("{}@lsize", cmd);
945    let rsize_cs_name = s!("{}@rsize", cmd);
946    let param_spec: String = (0..n.max(1)).map(|_| "{}").collect();
947    // The precode, the delimiters and the postcode are in the command the arguments are read by, as mathtools.sty:953-988
948    // generates it, so they may use them (`\DeclarePairedDelimiterXPP\expect[2]{\mathbb{E}_{#1}}[]{}{#2}`, 2310.04475;
949    // Perl's mathtools.sty.ltxml:702-708 puts them before the arguments: the `#` reached the stomach); the dispatcher
950    // gives each call its sizes.
951    let mut inner_body_toks = pre;
952    inner_body_toks.push(T_CS!(&lsize_cs_name));
953    inner_body_toks.extend(ldel);
954    inner_body_toks.extend(body);
955    inner_body_toks.push(T_CS!(&rsize_cs_name));
956    inner_body_toks.extend(rdel);
957    inner_body_toks.extend(post);
958    inner_body_toks.push(T_CS!("\\endgroup"));
959    def_macro(T_CS!(&inner_cs_name),
960      parse_parameters(&param_spec, &T_CS!(&inner_cs_name), true)?,
961      ExpansionBody::Tokens(Tokens::new(inner_body_toks)), None)?;
962    def_macro(cs, parse_parameters("OptionalMatch:* []", &cs, true)?,
963      Some(ExpansionBody::Closure(Rc::new(move |args| {
964        let is_star = !args[0].is_empty();
965        let size: Vec<Token> = if is_star { vec![T_CS!("\\middle")] } else { args[1].unlist_cow().to_vec() };
966        let size_cs = size.iter().find(|t| t.get_catcode() != Catcode::SPACE).copied();
967        // One side's sizing: `\left`/`\right`, or `\csname<size name><l|r>\endcsname` (`\MH_cs_to_str:N` is
968        // `\string` without its first character), or nothing.
969        let side = |left: bool| -> Vec<Token> {
970          if is_star {
971            vec![T_CS!(if left { "\\left" } else { "\\right" })]
972          } else if let Some(first) = size_cs {
973            let name = first.to_string();
974            let mut toks = vec![T_CS!("\\csname")];
975            toks.extend(name.chars().skip(1).chain([if left { 'l' } else { 'r' }]).map(|c| CharToken!(c)));
976            toks.push(T_CS!("\\endcsname"));
977            toks
978          } else {
979            vec![]
980          }
981        };
982        let mut toks: Vec<Token> = vec![T_CS!("\\begingroup"), T_CS!("\\def"), T_CS!("\\delimsize"), T_BEGIN!()];
983        toks.extend(size.iter().copied());
984        toks.push(T_END!());
985        toks.extend([T_CS!("\\def"), T_CS!(&lsize_cs_name), T_BEGIN!()]);
986        toks.extend(side(true));
987        toks.extend([T_END!(), T_CS!("\\def"), T_CS!(&rsize_cs_name), T_BEGIN!()]);
988        toks.extend(side(false));
989        toks.push(T_END!());
990        toks.push(T_CS!(&inner_cs_name));
991        Ok(Tokens::new(toks))
992      }))), None)?;
993    Ok(())
994  }

1176/// The `\lx@ams@cases` keys of a cases environment `name` whose preambles and delimiters are those `\newcases` and
1177/// `\MT_start_cases:nnnn` take (mathtools.sty:995-1019): its delimiters (a `.` or empty one is none), `\displaystyle`
1178/// when the left preamble sets it, and math conditions when the right preamble is a `$…$` cell.
1179fn cases_keys(
1180  name: &str,
1181  lpre: &Tokens,
1182  rpre: &Tokens,
1183  ldelim: Tokens,
1184  rdelim: Tokens,
1185) -> Vec<Token> {
1186  let display = lpre
1187    .unlist_ref()
1188    .iter()
1189    .any(|t| *t == T_CS!("\\displaystyle"));
1190  let math_conditions = rpre
1191    .unlist_ref()
1192    .iter()
1193    .any(|t| t.get_catcode() == Catcode::MATH);
1194  let mut keys: Vec<Token> = ExplodeText!(&format!("name={name},meaning=cases,"));
1195  for (key, delimiter, cs) in [
1196    ("left", ldelim, "\\lx@left"),
1197    ("right", rdelim, "\\lx@right"),
1198  ] {
1199    let text = delimiter.clone().untex();
1200    if !matches!(text.trim(), "" | ".") {
1201      keys.extend(ExplodeText!(&format!("{key}=")));
1202      keys.push(T_CS!(cs));
1203      keys.extend(delimiter.unlist());
1204      keys.extend(ExplodeText!(","));
1205    }
1206  }
1207  keys.extend(ExplodeText!("style="));
1208  keys.push(T_CS!(if display {
1209    "\\displaystyle"
1210  } else {
1211    "\\textstyle"
1212  }));
1213  keys.extend(ExplodeText!(&format!(
1214    ",conditionmode={}",
1215    if math_conditions { "math" } else { "text" }
1216  )));
1217  keys
1218}

1220/// The definitions `\newcases` makes (mathtools.sty:1015-1019): `\<name>` opens `\lx@ams@cases` with the keys its
1221/// arguments decide ([`cases_keys`]), `\end<name>` closes it.
1222fn mathtools_newcases(
1223  name: Tokens,
1224  lpre: &Tokens,
1225  rpre: &Tokens,
1226  ldelim: Tokens,
1227  rdelim: Tokens,
1228) -> Tokens {
1229  let name = name.untex();
1230  let name = name.trim();
1231  let keys = cases_keys(name, lpre, rpre, ldelim, rdelim);
1232  let mut toks = vec![T_CS!("\\expandafter"), T_CS!("\\def"), T_CS!("\\csname")];
1233  toks.extend(ExplodeText!(name));
1234  toks.extend([
1235    T_CS!("\\endcsname"),
1236    T_BEGIN!(),
1237    T_CS!("\\lx@ams@cases"),
1238    T_BEGIN!(),
1239  ]);
1240  toks.extend(keys);
1241  toks.extend([
1242    T_END!(),
1243    T_END!(),
1244    T_CS!("\\expandafter"),
1245    T_CS!("\\def"),
1246    T_CS!("\\csname"),
1247  ]);
1248  toks.extend(ExplodeText!(&format!("end{name}")));
1249  toks.extend([
1250    T_CS!("\\endcsname"),
1251    T_BEGIN!(),
1252    T_CS!("\\lx@end@ams@cases"),
1253    T_END!(),
1254  ]);
1255  Tokens::new(toks)
1256}
