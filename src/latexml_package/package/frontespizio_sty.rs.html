1//! frontespizio.sty — the Italian thesis title page.
2//!
3//! **Why a binding exists for this package** (the perfect-kernel mission
4//! writes no new bindings except where raw interpretation is structurally
5//! impossible): frontespizio's default `write` mode does not typeset the
6//! title page at all — its content macros are redirected to a separate
7//! `\jobname-frn.tex` (frontespizio.sty:207-229, `\front@write`), which the
8//! user compiles with a SECOND pdflatex run and re-includes as a graphic
9//! (:606-616 `\includegraphics{\jobname-frn}`). That external compilation is
10//! out of this engine's scope, exactly like shell-escape; raw-loaded, the
11//! `frontespizio` environment yields an empty `<titlepage>` and every one of
12//! the six shipped manuals loses its whole content (S3 recall 0/27; Perl,
13//! with no binding, loses it identically — a user-approved surpass shape).
14//!
15//! The package carries its own inline route: with `[nowrite,infront]` the
16//! content macros STORE their values (:260-309) and
17//! `\preparefrontpagestandard` / `\preparefrontpagesuftesi` (:311-538)
18//! typeset the page inside `\titlepage…\endtitlepage`. The binding forces
19//! that route at load time (the `infront` token registers, :158-171, are
20//! only allocated then), runs the font selection the package would have
21//! written to the external file (:151-156), and makes the environment's end
22//! call the shape's prepare macro. xcolor is loaded for the `suftesi`
23//! shape's `\color`/`Maroon` (:118-131 writes it to the external file).
24use crate::prelude::*;
25
26#[rustfmt::skip]
27LoadDefinitions!({
28  RequirePackage!("xcolor", options => vec!["svgnames".to_string()]);
29  // The document's own options (`suftesi`, `sans`, …) plus the inline
30  // regime, handed to the raw load's `\ProcessOptions`.
31  let mut opts: Vec<String> = lookup_vecdeque("opt@frontespizio.sty")
32    .map(|v| v.iter().map(|o| o.to_string()).collect())
33    .unwrap_or_default();
34  opts.retain(|o| o != "write" && o != "onlyinclude");
35  opts.push("nowrite".to_string());
36  opts.push("infront".to_string());
37  InputDefinitions!("frontespizio", noltxml => true, extension => Some(Cow::Borrowed("sty")),
38    handleoptions => true, options => opts);
39  // The environment, inline: begin selects the fonts (`\front@thefont` =
40  // `\fontoptionnormal`/`\fontoptionsans`, :151-156, otherwise written to
41  // the external file), end typesets the page through the shape's prepare
42  // macro (:311/:421), which opens and closes `titlepage` itself.
43  // `\Preambolo{…}` / `Preambolo*`: preamble material for the external
44  // title-page document, defined only in the write branch (:186-187, written
45  // out) and the empty branch (:541-542, gobbled) — not in the store branch
46  // the binding forces. Inline they are EXECUTED, so a `\newcommand` made
47  // there (examplec's `\compring`, used in `\Titolo`) exists, while the
48  // package loaders they carry (`\usepackage{fourier}`, `\setmainfont`)
49  // are gobbled: the external document's packages are not this document's.
50  // The environment form collects its body (environ's `\Collect@Body`, as
51  // the package itself does) and runs it AFTER the environment's group
52  // closes (`\aftergroup`), so the definitions are not local to it. The
53  // package loaders AND the page-layout call of that external document are
54  // neutralised: frontespizio writes geometry into `\jobname-frn.tex` itself
55  // (frontespizio.sty:120-127 `\front@geometry@standard`) and never loads it
56  // in the infront regime, so a `\geometry{…}` in `Preambolo*` (toptesi's
57  // toptesi-example-con-frontespizio.tex:73-75) configured only that
58  // document's pages; inline it was undefined and its argument leaked as
59  // body text (sweep-82 regression, `~/data/pk_agents/w23/regr82/toptesi`).
60  RawTeX!(r"\makeatletter
61\def\frontespizio{\front@thefont}
62\def\endfrontespizio{\csname preparefrontpage\front@shape\endcsname}
63\def\lx@front@ignorepkg{\@ifnextchar[{\lx@front@ignorepkg@}{\lx@front@ignorepkg@[]}}
64\def\lx@front@ignorepkg@[#1]#2{}
65\def\lx@front@preamble@begin{%
66  \let\lx@front@saved@usepackage\usepackage
67  \let\lx@front@saved@RequirePackage\RequirePackage
68  \let\lx@front@saved@setmainfont\setmainfont
69  \let\lx@front@saved@setsansfont\setsansfont
70  \let\lx@front@saved@geometry\geometry
71  \let\usepackage\lx@front@ignorepkg
72  \let\RequirePackage\lx@front@ignorepkg
73  \let\setmainfont\lx@front@ignorepkg
74  \let\setsansfont\lx@front@ignorepkg
75  \let\geometry\lx@front@ignorepkg}
76\def\lx@front@preamble@end{%
77  \let\usepackage\lx@front@saved@usepackage
78  \let\RequirePackage\lx@front@saved@RequirePackage
79  \let\setmainfont\lx@front@saved@setmainfont
80  \let\setsansfont\lx@front@saved@setsansfont
81  \let\geometry\lx@front@saved@geometry}
82\renewcommand{\Preambolo}[1]{\lx@front@preamble@begin#1\lx@front@preamble@end}
83\newtoks\lx@front@preamble@toks
84\def\lx@front@preamble@store#1{\global\lx@front@preamble@toks{#1}}
85\def\lx@front@preamble@run{\lx@front@preamble@begin\the\lx@front@preamble@toks\lx@front@preamble@end}
86\renewenvironment{Preambolo*}{\Collect@Body\lx@front@preamble@store}{\aftergroup\lx@front@preamble@run}
87\makeatother");
88});
