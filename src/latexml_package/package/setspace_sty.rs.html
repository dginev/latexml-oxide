9  // switch sets `\baselineskip` = leading × `\baselinestretch` (58c); the spacing commands are its
10  // 10pt stretches (the bindings' sizes are size10.clo's). `\singlespacing`'s `\vskip\baselineskip`
11  // correction is omitted. Perl's are no-ops (setspace.sty.ltxml). Witnesses 2605.18633, 2605.01923.

26  DefMacro!(

33  RawTeX!(
34    r"\g@addto@macro\lx@note@reset{\baselineskip\f@baselineskip\relax\baselineskip\setspace@singlespace\baselineskip}"
35  );
36  // setspace.sty:294-295: the package options run the spacing command at the package's end.
37  DeclareOption!("onehalfspacing", "\\AtEndOfPackage{\\onehalfspacing}");
38  DeclareOption!("doublespacing", "\\AtEndOfPackage{\\doublespacing}");
39  ProcessOptions!();
40  def_macro_noop("\\setdisplayskipstretch{}")?;

43  // {spacing}{}: the body-wrapping variant. Keep BOUND_MODE vertical so `$$`
44  // inside it still enters display math. Witness 2305.08368:
45  // `\begin{spacing}{1.25}` wrapping the whole body made `$$x_1=...$$` fall
46  // through the `$` handler's vertical-only check (tex_math.rs:447) → 199
47  // `Error:unexpected:_` cascades. Perl-faithful binding is `#body` only, but
48  // the default Package.pm mode is `restricted_horizontal`; in Rust we have
49  // to make `internal_vertical` explicit so paragraphs and display math
50  // survive the wrap.
51  //
52  // Every one ends with `\par` (setspace.sty:489-548, `\restore@spacing` :516-523), so its last
53  // paragraph closes before the end and the text after it starts a new one (a transparent `#body`
54  // left it open, 57l); `{spacing}` begins with `\par` (:525-526) and `{singlespace}` with
55  // `\vskip` (:489-495), which end the paragraph before them, while `{onehalfspace}` and
56  // `{doublespace}` begin with `\begingroup` alone (:534-548) and go on in it (KPE #320).
57  // Each sets its stretch in its group (setspace.sty:489-548: `\setstretch{#1}`,
58  // `{\setspace@singlespace}`, `\onehalfspacing`, `\doublespacing`).
59  DefEnvironment!("{spacing}{}", "#body", mode => "internal_vertical",
60    after_digest_begin => sub[whatsit] {
61      if let Some(stretch) = whatsit.get_arg(1) {
62        let mut tokens = vec![T_CS!("\\setstretch"), T_BEGIN!()];
63        tokens.extend(stretch.revert()?.unlist());
64        tokens.push(T_END!());
65        digest(Tokens::new(tokens))?;
66      }
67    },
68    before_digest_end => { leave_horizontal()?; },
69    // The begin code opens a `\begingroup` (a semi_simple_group) of its own: setspace.sty:525-547 `\begingroup`.
70    tex_group => { TexGroup::SemiSimple });
71  DefEnvironment!("{singlespace}", "#body", mode => "internal_vertical",
72    after_digest_begin => { Digest!("\\setstretch{\\setspace@singlespace}")?; },
73    before_digest_end => { leave_horizontal()?; });
74  // The paragraph these two close was opened outside their group, which a mode frame cannot end
75  // (ARCHITECTURE_THEMES 1), so the `\par` is the document's: the body goes on in the current
76  // paragraph and its `<p>` closes after it, where the text that follows starts a new one.
77  DefEnvironment!("{onehalfspace}", sub[document, _args, props] {
78    body_closing_its_paragraph(document, props)
79  }, after_digest_begin => { Digest!("\\onehalfspacing")?; },
80    // The begin code opens a `\begingroup` (a semi_simple_group) of its own: setspace.sty:525-547 `\begingroup`.
81    tex_group => { TexGroup::SemiSimple });
82  DefEnvironment!("{doublespace}", sub[document, _args, props] {
83    body_closing_its_paragraph(document, props)
84  }, after_digest_begin => { Digest!("\\doublespacing")?; },
85    // The begin code opens a `\begingroup` (a semi_simple_group) of its own: setspace.sty:525-547 `\begingroup`.
86    tex_group => { TexGroup::SemiSimple });
87  // Standalone-switch overrides: some papers (witness 2310.08233 IEEEtran)
88  // use `\singlespace` as a SWITCH inside an arg-grabbing context such as
89  // `\title{\singlespace ...}`. DefEnvironment binds `\singlespace` to the
90  // env-begin CS, which pushes a (restricted_)horizontal mode-switch group.
91  // That group cannot be cleanly popped at arg-close, leaking into the
92  // following `\@add@frontmatter@now`. Override the env-begin CS (and its
93  // sibling `\endsinglespace`) with plain noops so standalone use is inert.
94  // For `\begin{singlespace}...\end{singlespace}` blocks: \begin/\end track
95  // env scope themselves; the begin/end CSes are just hooks. Without the
96  // hooks the body still renders (it is just literal #body), so the env
97  // form remains correct.
98  def_macro_noop("\\singlespace")?;
99  def_macro_noop("\\endsinglespace")?;
100  // `\onehalfspace`/`\doublespace` as switches: setspace.sty:534-547 makes each a
101  // `\begingroup` and the spacing, ended by its `\end…` (`\restore@spacing`'s
102  // `\endgroup`, :516-523). The spacing is presentational and the group is not
103  // modelled: a real `\begingroup` there now keeps every word (65c, guard
104  // perfect_kernel_batch65c::unclosed_nested_environments_keep_their_text) but nests
105  // a later sibling environment inside the earlier one at `\end{document}`; the
106  // checked no-op keeps the author's sibling structure (content and semantics ahead
107  // of PDF fidelity). What the group does to TeX is reported, where TeX reports
108  // it: when the group a switch opened in ends with the switch still open.
109  // - A brace or box group: TeX's "Extra }, or forgotten \endgroup" — `{\doublespace
110  //   text}` (witness 2310.08233's shape), a tabular cell, `\footnote{…}`, and an
111  //   environment whose body is a box: its frame sets `\hsize` (a float's
112  //   `\@xfloat`/`\@endfloatbox`, `{minipage}`, subcaption's `{subfigure}`, wrapfig,
113  //   float.sty's `{algorithm}`), and its end's `}` meets the switch's group.
114  // - Another environment: its `\end`'s `\endgroup` closes the switch's group, not the
115  //   environment's, which stays open until the next `\end` — the enclosing
116  //   environment's: LaTeX "\begin{itemize} on input line N ended by \end{quote}",
117  //   or "… by \end{document}" (`{quote}`, an `\item`'s list, `{center}`). The closer
118  //   named is right when no group sits between the environment and the next one out;
119  //   with one between (`{\begin{itemize}…}`, a `\parbox` or `{minipage}` around it,
120  //   review 5 probes p6/p14/p15/p5) TeX says "Extra }" instead, and a plain group
121  //   around the environment is no error in TeX (RED
122  //   boxes-groups/plain_group_around_itemize_doublespace). The `\hsize` test is a
123  //   proxy: an author's `\hsize=` in a non-box environment, or `{list}`'s inner
124  //   frame, reads as a box (wording only, the count is right: probes p21, p4).
125  // - A plain `\begingroup … \endgroup`: the `\endgroup` closes the switch's group and
126  //   the plain group stays open: the switch re-arms on the enclosing group (Perl
127  //   `\aftergroup` order, stomach.rs `pop_stack_frame`), so a plain group inside
128  //   `{quote}` still makes the quote's error, and at the document level TeX only
129  //   logs "(\end occurred inside a group at level 1)": no error (Perl errs).
130  // An `\end…` closes the innermost switch of its own group (a frame-local stack, so
131  // of two switches one `\end…` leaves the outer open and named: review 6 probe h4). One met in a nested
132  // group closes nothing here, so the switch's group still reports. That errs toward
133  // flagging: a switch closed by `\end<name>` from inside a nested plain group still
134  // reports, where TeX's `\endgroup`s pair off cleanly (review 4 probe r9, pdflatex 0:
135  // RED boxes-groups/doublespace_end_in_a_nested_plain_group).
136  // One report per group, however many switches it holds (q6: 120 `\doublespace` in
137  // one `{quote}` are one LaTeX error); it names the innermost. A group digested twice
138  // (a `\section` title, for the heading and the toc) reports twice, like every error
139  // there (Perl and ours alike: RED sectioning-frontmatter/section_title_errors_reported_twice).
140  // At the document level the switch runs to `\end{document}` with no error, as in TeX.
141  // OXIDIZED_DESIGN #520.
142  for (name, spacing) in [
143    ("onehalfspace", "\\onehalfspacing"),
144    ("doublespace", "\\doublespacing"),
145  ] {
146    DefPrimitive!(T_CS!(s!("\\{name}")), None, {
147      // A switch read from a token list (`\footnote{…}`, a `\section` title) is placed
148      // where the document reads that list.
149      let at = get_location();
150      let at = if at.contains("Anonymous String") {
151        get_source_location().unwrap_or_default()
152      } else {
153        at
154      };
155      open_setspace_switch(name, at)?;
156      Digest!(T_CS!(spacing))?;
157    });
158    DefPrimitive!(T_CS!(s!("\\end{name}")), None, {
159      // A switch opened in this group: its `\endgroup` closes the innermost one, whatever
160      // its name (setspace.sty:516-523); the next one out names the report, the last one
161      // closes the check.
162      if is_value_bound("lx@setspace@open", Some(0)) {
163        let mut stack = lookup_string("lx@setspace@stack")
164          .split('\u{2}')
165          .filter(|entry| !entry.is_empty())
166          .map(str::to_string)
167          .collect::<Vec<_>>();
168        stack.pop();
169        let (open, at) = match stack.last().and_then(|top| top.split_once('\u{1}')) {
170          Some((open, at)) => (Stored::String(pin(open)), Stored::String(pin(at))),
171          None => (Stored::None, Stored::None),
172        };
173        assign_value("lx@setspace@open", open, Some(Scope::Local));
174        assign_value("lx@setspace@open@at", at, Some(Scope::Local));
175        assign_value(
176          "lx@setspace@stack",
177          Stored::String(pin(stack.join("\u{2}"))),
178          Some(Scope::Local),
179        );
180      }
181    });
182  }
183  // Run before the group pops (`beforeAfterGroup`), where the frame still says what
184  // it is. The tokens are digested from a string of their own, so the report is
185  // placed at the switch (where it opened), its object the group's closer.
186  DefPrimitive!(T_CS!("\\lx@setspace@check"), None, {
187    let name = lookup_string("lx@setspace@open");
188    if name.is_empty() || lookup_bool("lx@ending@document") {
189      return Ok(Vec::new());
190    }
191    let switch_at = lookup_string("lx@setspace@open@at");
192    // An environment's frame binds `current_environment` (the `{abstract}` binding
193    // captures its body in a frame of its own, `\lx@add@frontmatter@until`, so it
194    // reads as a box: "Extra }", where LaTeX reports "\begin{abstract} … ended by").
195    let environment =
196      is_value_bound("current_environment", Some(0)).then(|| lookup_string("current_environment"));
197    let extra_brace = || {
198      (
199        s!("}}"),
200        s!(
201          "Extra }}, or forgotten \\endgroup: \\{name}'s \\begingroup (setspace.sty:534-547) is still open"
202        ),
203      )
204    };
205    let (object, message) = match environment {
206      // A box environment: its frame sets `\hsize` (latex.ltx `\@xfloat`
207      // `\hsize\columnwidth`, `\@iiiminipage` `\hsize #4`), its end closes the box.
208      Some(_) if is_value_bound("\\hsize", Some(0)) => extra_brace(),
209      Some(environment) => {
210        // The environment stays open until the next `\end`: the enclosing one's.
211        let outer = match value_in_frame("current_environment", 1) {
212          Some(Stored::String(outer)) => to_string(outer),
213          _ => String::new(),
214        };
215        let closer = if outer.is_empty() {
216          s!("\\end{{document}}")
217        } else {
218          s!("\\end{{{outer}}}")
219        };
220        (
221          s!("\\end{{{environment}}}"),
222          s!(
223            "\\end{{{environment}}} closes \\{name}'s \\begingroup (setspace.sty:534-547) instead of \
224             {{{environment}}}'s group; {{{environment}}} stays open (LaTeX: \\begin{{{environment}}} … ended by {closer})"
225          ),
226        )
227      },
228      None if lookup_bool("groupNonBoxing") => {
229        // A plain `\begingroup`: its `\endgroup` closes the innermost switch's group, and
230        // the plain group stands open in its place: every switch of this group re-arms on
231        // the enclosing one, outermost first (review 7 probes k1, k3).
232        assign_value(
233          "lx@setspace@rearm",
234          Stored::String(pin(lookup_string("lx@setspace@stack"))),
235          Some(Scope::Global),
236        );
237        push_value(
238          "afterGroup",
239          Stored::Tokens(Tokens::new(vec![T_CS!("\\lx@setspace@rearm")])),
240        )?;
241        return Ok(Vec::new());
242      },
243      None => extra_brace(),
244    };
245    let at = if switch_at.is_empty() {
246      String::new()
247    } else {
248      s!("\n\t{switch_at} (the switch)")
249    };
250    emit_error(
251      "unexpected",
252      &object,
253      &s!("{message}{at}\n\tIn latexml_package/src/package/setspace_sty.rs\n"),
254    );
255  });
256  DefPrimitive!(T_CS!("\\lx@setspace@rearm"), None, {
257    let rearm = lookup_string("lx@setspace@rearm");
258    for entry in rearm.split('\u{2}') {
259      if let Some((name, at)) = entry.split_once('\u{1}') {
260        open_setspace_switch(name, at.to_string())?;
261      }
262    }
263  });
264});

266/// A setspace environment's body, then the `\par` its end runs (`\restore@spacing`, setspace.sty:516)
267/// as the closing of the current `<p>`.
268fn body_closing_its_paragraph(document: &mut Document, props: &SymHashMap<Stored>) -> Result<()> {
269  if let Some(Stored::Digested(body)) = props.get("body") {
270    document.absorb(body, None)?;
271  }
272  document.maybe_close_element("ltx:p")?;
273  Ok(())
274}

276/// A setspace switch (`\onehalfspace`, `\doublespace`) opened, or still open as a
277/// plain group it was opened in ends (`\lx@setspace@rearm`), in a group other than
278/// the document's: record it and where it opened (the innermost switch names the
279/// report), and arm the group's end check once per group (`\lx@setspace@check`,
280/// run before the group pops). The group stacks its open switches
281/// (`lx@setspace@stack`), the innermost closed by an `\end…` in the group.
282fn open_setspace_switch(name: &str, at: String) -> Result<()> {
283  let at_document_level = is_value_bound("current_environment", Some(0))
284    && lookup_string("current_environment") == "document";
285  if !at_document_level {
286    assign_value(
287      "lx@setspace@open",
288      Stored::String(pin(name)),
289      Some(Scope::Local),
290    );
291    assign_value(
292      "lx@setspace@open@at",
293      Stored::String(pin(&at)),
294      Some(Scope::Local),
295    );
296    let mut stack = if is_value_bound("lx@setspace@stack", Some(0)) {
297      lookup_string("lx@setspace@stack")
298    } else {
299      String::new()
300    };
301    if !stack.is_empty() {
302      stack.push('\u{2}');
303    }
304    stack.push_str(&s!("{name}\u{1}{at}"));
305    assign_value(
306      "lx@setspace@stack",
307      Stored::String(pin(&stack)),
308      Some(Scope::Local),
309    );
310    if !is_value_bound("lx@setspace@armed", Some(0)) {
311      assign_value("lx@setspace@armed", true, Some(Scope::Local));
312      unshift_value("beforeAfterGroup", vec![T_CS!("\\lx@setspace@check")]);
313    }
314  }
315  Ok(())
