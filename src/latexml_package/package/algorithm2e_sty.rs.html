5/// The `\Indp`/`\Indm` offset in levels (`\lx@algo@indentoffset`, algorithm2e.sty:1626-1629's `\leftskip`).

7  match LookupRegister!("\\lx@algo@indentoffset") {

24/// The levels a line ends with: the open blocks' (`\lx@algo@indentation`, one token a level), then a positive
25/// `\Indp`/`\Indm` offset's plain levels after their rules (a block's rule is `\moveright`ed, which `\leftskip` does not
26/// move, algorithm2e.sty:1780), or a negative one's removing the innermost; none at the least.
27fn algo_line_indentation() -> Vec<Token> {
28  let mut levels = match LookupRegister!("\\lx@algo@indentation") {
29    RegisterValue::Tokens(toks) => toks.unlist(),
30    _ => Vec::new(),
31  };
32  let offset = algo_indent_offset();
33  if offset >= 0 {
34    levels.extend((0..offset).map(|_| T_CS!("\\lx@algo@indent")));
35  } else {
36    let keep = levels
37      .len()
38      .saturating_sub(usize::try_from(-offset).unwrap_or(usize::MAX));
39    levels.truncate(keep);
40  }
41  levels
42}

44/// Does an `<ltx:listingline>` hold real content, or only its line-number
45/// `<ltx:tags>` and indentation `<ltx:rule>`s? A blank line produced by the
46/// `\lx@strippar` block terminator (three `\lx@algo@parx`) still carries the
47/// prepended indentation rules (`\lx@prepend@indentation`) and, because
48/// `\the\everypar` fires at endline, a stolen line-number tag — but no actual
49/// statement. Neither the number tag nor the indentation rules count as content,
50/// so such a line is dropped; a real line always has text or another element
51/// beyond them — a rule the statement draws (`\hrule`, `\rule{\linewidth}{0.8pt}`;
52/// 2301.00241, 2412.09565) is content, only the indentation's `\lx@algo@rule` is not.
53fn listingline_has_content(line: &Node) -> bool {
54  line.get_child_nodes().iter().any(|c| match c.get_type() {
55    Some(NodeType::ElementNode) => {
56      let name = c.get_name();
57      let indentation = name == "rule"
58        && c.get_attribute("width").as_deref() == Some("1px")
59        && c.get_attribute("height").as_deref() == Some("100%");
60      name != "tags" && !indentation
61    },
62    Some(NodeType::TextNode) => !c.get_content().trim().is_empty(),
63    _ => false,
64  })
65}

67/// Whether a closed line goes: it holds no content ([`listingline_has_content`]) and no `\label` names it.
68fn droppable_line(line: &Node) -> bool {

78/// Where the algorithm's line machinery stands: no open `ltx:listingline` at all (algorithm2e statements outside
79/// any listing), a line it can split, or a line it cannot reach.
80enum LineReach {
81  NoLine,
82  Line,
83  Barred,
84}

86/// A line is reachable only through inline wrappers and through the wrappers the document opened itself directly in
87/// the line (`_autoopened`: the `ltx:inline-block` bridging a list into the listingline, the `ltx:p` resuming after
88/// it). Anything else — a list item, an equation, a minipage or `\vbox`, a table cell, a footnote, a capture block —
89/// bars it: a `\\` or block macro there breaks that box's text (2004.03005, 1709.07249, 1410.4772).
90///
91/// The nearest listing bounds the walk: between the lines of a listing nested in a line (an `{algorithmic}` whose
92/// `\STATE` holds an algorithm2e `\For{…}{…}`, 2301.04312, 1904.07272, 2302.08012) a line opens in that listing, as
93/// TeX sets the block's body as more paragraphs of the inner list; walking on to the outer line barred it, and the body
94/// fell into the listing as bare text and breaks.
95fn line_reach(document: &Document) -> LineReach {
96  let mut chain: Vec<Node> = Vec::new();
97  let mut node = Some(document.get_node().clone());
98  let opened = |e: &Node| e.has_attribute("_autoopened");
99  let reachable = |chain: &[Node]| {
100    chain.iter().enumerate().all(|(k, e)| {
101      document::with_node_qname(e, |q| LINE_INLINE_WRAPPERS.contains(&q))
102        || (opened(e) && chain[k + 1..].iter().all(opened))
103    })
104  };
105  while let Some(n) = node {
106    if n.get_type() == Some(NodeType::ElementNode) {
107      if document::with_node_qname(&n, |q| q == "ltx:listingline") {
108        return if reachable(&chain) {
109          LineReach::Line
110        } else {
111          LineReach::Barred
112        };
113      }
114      if document::with_node_qname(&n, |q| q == "ltx:listing") {
115        return if reachable(&chain) {
116          LineReach::NoLine
117        } else {
118          LineReach::Barred
119        };
120      }
121      chain.push(n.clone());
122    }
123    node = n.get_parent();
124  }
125  LineReach::NoLine
126}

128/// `\lx@algo@@endline`: close the line (dropping it when empty); behind a barrier, a `<ltx:break/>` when
129/// `break_barred`; with no line, nothing.
130fn end_line(document: &mut Document, break_barred: bool) -> Result<()> {
131  match line_reach(document) {
132    // No listing (a caption's `\\`, 1412.0600): the break the kernel's `\\` makes there.
133    LineReach::NoLine => {
134      if break_barred
135        && !document.is_openable("ltx:listingline")
136        && !document.is_openable("ltx:listing")
137        && document.is_openable("ltx:break")
138      {
139        document.insert_element("ltx:break", Vec::new(), None)?;
140      }
141      return Ok(());
142    },
143    LineReach::Barred => {
144      if break_barred {
145        document.insert_element("ltx:break", Vec::new(), None)?;
146      }
147      return Ok(());
148    },
149    LineReach::Line => {},
150  }
151  document.close_element("ltx:listingline")?;
152  if let Some(line) = document.get_node().get_last_child()
153    && document::get_node_qname(&line) == pin!("ltx:listingline")
154    && droppable_line(&line)
155  {
156    document.remove_node(line);
157  }
158  Ok(())
159}

161/// Renumber the algorithm's numbered listinglines sequentially from 1, matching
162/// the pdflatex golden. `linesnumbered` numbers each body line via the engine's
163/// content-start `\everypar` hook, which fires `\nl` (steps `AlgoLine`). The count
164/// drifts off `1..N` for two reasons: `\lx@strippar`'s block terminator and other
165/// structural `\nl` fires over-step the counter, and empty listinglines that were
166/// numbered then dropped (by `\lx@algo@@endline`, or by their listing when it closes) leave a gap. This
167/// construction-time pass rewrites the SURVIVING numeric line-number tags to
168/// `1..N` in document order — algorithm2e always numbers off the sequential
169/// `AlgoLine` counter (auto via `linesnumbered`, or manual via `\nl`/`\lnl`), so
170/// `1..N` is always the intended visible sequence. Non-numeric tags (`\nlset`
171/// custom references) are left alone. (The underlying counter can still be
172/// over-stepped, so a `\ref` to a line may read high — a pre-existing residual.)
173fn renumber_algo_lines(document: &mut Document) {
174  let Some(float) = document.get_node().get_last_child() else {
175    return;
176  };
177  let Some(listing) = float
178    .get_child_elements()
179    .into_iter()
180    .find(|c| document::get_node_qname(c) == pin!("ltx:listing"))
181  else {
182    return;
183  };
184  // Overprint-collapse, FIRST-wins (batch 56eo, OXIDIZED_DESIGN #241): a line
185  // that fires numbering more than once — `\LinesNumbered` (auto `\nl` at
186  // content-start) PLUS an explicit `\nl` — accrues multiple `<ltx:tags>`, which
187  // the listingline model forbids (`tags?`, LaTeXML-block.rnc:189). Real
188  // algorithm2e `\llap`/`\rlap`-overprints the numbers (algorithm2e.sty:1647/1654)
189  // so only one is visible; keep the FIRST `<tags>` per line and remove the rest.
190  // First, not last: the listingline model requires tags first and 56eh's endline
191  // indentation-prepend seats the leading (first-fired) tags at child 0 (before
192  // the `<rule>`), so a later tags sits AFTER the rule; the visible NUMBER is
193  // corrected by the renumber-to-1..N below. Known limitation: an explicit
194  // `\nlset{custom}` coinciding with the auto `\nl` on one line loses the custom
195  // reference to the sequential number — real algorithm2e garbles that too
196  // (both overprint at the same kern), and keep-last would be schema-invalid.
197  // Done here (post-construct, stable DOM) — NOT in `\algocf@printnl@i`, where
198  // the rapid double-fire's mid-construction cursor state makes an in-place
199  // remove corrupt the tree. Perl keeps them all (SHARED schema bug); surpass.
200  for line in listing.get_child_elements() {
201    if document::get_node_qname(&line) != pin!("ltx:listingline") {
202      continue;
203    }
204    let tags: Vec<Node> = line
205      .get_child_elements()
206      .into_iter()
207      .filter(|c| document::get_node_qname(c) == pin!("ltx:tags"))
208      .collect();
209    for stale in tags.iter().skip(1) {
210      document.remove_node(stale.clone());
211    }
212  }
213  // Collect the numeric number-tag nodes in document order: the printed (role-less) ones, each with the reference forms
214  // its line took from its step (`\algocf@printnl@i`), which name the same number.
215  let mut numeric_tags: Vec<(Node, Vec<Node>)> = Vec::new();
216  for line in listing.get_child_elements() {
217    if document::get_node_qname(&line) != pin!("ltx:listingline") {
218      continue;
219    }
220    for tags in line.get_child_elements() {
221      if document::get_node_qname(&tags) != pin!("ltx:tags") {
222        continue;
223      }
224      let (printed, forms): (Vec<Node>, Vec<Node>) = tags
225        .get_child_elements()
226        .into_iter()
227        .filter(|tag| document::get_node_qname(tag) == pin!("ltx:tag"))
228        .partition(|tag| !tag.has_attribute("role"));
229      for tag in printed {
230        if tag.get_content().trim().parse::<i64>().is_ok() {
231          numeric_tags.push((tag, forms.clone()));
232        }
233      }
234    }
235  }
236  // Skip when already 1..N (correct sequence — do not touch).
237  let already_sequential = numeric_tags
238    .iter()
239    .enumerate()
240    .all(|(i, (t, _))| t.get_content().trim() == (i + 1).to_string());
241  if already_sequential {
242    return;
243  }
244  for (i, (tag, forms)) in numeric_tags.into_iter().enumerate() {
245    let old = tag.get_content().trim().to_string();
246    let new = (i + 1).to_string();
247    set_number_tag_text(&tag, &new);
248    // A reference form prints the line's number as the line does (`\theAlgoLine`), so it is renumbered with it.
249    for form in forms {
250      replace_number_in_form(&form, &old, &new);
251    }
252  }
253}

255/// The number `old` a line's reference form (`<ltx:tag role="refnum">3</ltx:tag>`, a cleveref form's "line 3") prints,
256/// as the form's last word, made `new` ([`renumber_algo_lines`]).
257fn replace_number_in_form(form: &Node, old: &str, new: &str) {
258  // (the form's last text, past any empty element after it: autoref's "line 3" ends in an empty `ltx:text`)
259  fn last_text(node: &Node) -> Option<Node> {
260    let mut child = node.get_last_child();
261    while let Some(current) = child {
262      match current.get_type() {
263        Some(NodeType::TextNode) if !current.get_content().trim().is_empty() => {
264          return Some(current);
265        },
266        Some(NodeType::ElementNode) => {
267          if let Some(text) = last_text(&current) {
268            return Some(text);
269          }
270        },
271        _ => {},
272      }
273      child = current.get_prev_sibling();
274    }
275    None
276  }
277  let Some(mut target) = last_text(form) else {
278    return;
279  };
280  let text = target.get_content();
281  if let Some(head) = text.strip_suffix(old)
282    && (head.is_empty() || head.ends_with(|c: char| c.is_whitespace() || c == '\u{a0}'))
283  {
284    let _ = target.set_content(&format!("{head}{new}"));
285  }
286}

288/// Rewrite a numeric line-number `<ltx:tag>`'s text to `n` while PRESERVING any
289/// font-markup wrapper. `\algocf@printnl` wraps the number in `\NlSty`
290/// (`\textnormal{\textbf{…}}`), so a numbered tag is `<ltx:tag><ltx:text
291/// font="bold">N</ltx:text></ltx:tag>`. A blunt `set_content` on the tag would
292/// replace that `<ltx:text>` wrapper with a bare text node, stripping the uniform
293/// bold and reintroducing the per-line font leak. Instead descend through
294/// single-element-child wrappers and set the text on the innermost node, keeping the
295/// `\NlSty` styling intact. Guard: `cluster_algorithm2e_uniform_line_number_font`.
296fn set_number_tag_text(tag: &Node, n: &str) {
297  let mut target = tag.clone();
298  loop {
299    let children = target.get_child_elements();
300    if children.len() == 1 {
301      target = children.into_iter().next().unwrap();
302    } else {
303      break;
304    }
305  }
306  let _ = target.set_content(n);
307}
