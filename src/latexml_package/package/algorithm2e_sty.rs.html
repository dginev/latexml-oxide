5/// Does an `<ltx:listingline>` hold real content, or only its line-number
6/// `<ltx:tags>` and indentation `<ltx:rule>`s? A blank line produced by the
7/// `\lx@strippar` block terminator (three `\lx@algo@parx`) still carries the
8/// prepended indentation rules (`\lx@prepend@indentation`) and, because
9/// `\the\everypar` fires at endline, a stolen line-number tag — but no actual
10/// statement. Neither the number tag nor the indentation rules count as content,
11/// so such a line is dropped; a real line always has text or a non-rule element
12/// beyond them.
13fn listingline_has_content(line: &Node) -> bool {
14  line.get_child_nodes().iter().any(|c| match c.get_type() {
15    Some(NodeType::ElementNode) => {
16      let name = c.get_name();
17      name != "tags" && name != "rule"
18    },
19    Some(NodeType::TextNode) => !c.get_content().trim().is_empty(),
20    _ => false,
21  })
22}

24/// Renumber the algorithm's numbered listinglines sequentially from 1, matching
25/// the pdflatex golden. `linesnumbered` numbers each body line via the engine's
26/// content-start `\everypar` hook, which fires `\nl` (steps `AlgoLine`). The count
27/// drifts off `1..N` for two reasons: `\lx@strippar`'s block terminator and other
28/// structural `\nl` fires over-step the counter, and empty listinglines that were
29/// numbered then dropped (see `\lx@algo@@endline`) leave a gap. This
30/// construction-time pass rewrites the SURVIVING numeric line-number tags to
31/// `1..N` in document order — algorithm2e always numbers off the sequential
32/// `AlgoLine` counter (auto via `linesnumbered`, or manual via `\nl`/`\lnl`), so
33/// `1..N` is always the intended visible sequence. Non-numeric tags (`\nlset`
34/// custom references) are left alone. (The underlying counter can still be
35/// over-stepped, so a `\ref` to a line may read high — a pre-existing residual.)
36fn renumber_algo_lines(document: &mut Document) {
37  let Some(float) = document.get_node().get_last_child() else {
38    return;
39  };
40  let Some(listing) = float
41    .get_child_elements()
42    .into_iter()
43    .find(|c| document::get_node_qname(c) == pin!("ltx:listing"))
44  else {
45    return;
46  };
47  // Collect the numeric number-tag nodes in document order.
48  let mut numeric_tags: Vec<Node> = Vec::new();
49  for line in listing.get_child_elements() {
50    if document::get_node_qname(&line) != pin!("ltx:listingline") {
51      continue;
52    }
53    for tags in line.get_child_elements() {
54      if document::get_node_qname(&tags) != pin!("ltx:tags") {
55        continue;
56      }
57      for tag in tags.get_child_elements() {
58        if document::get_node_qname(&tag) == pin!("ltx:tag")
59          && tag.get_content().trim().parse::<i64>().is_ok()
60        {
61          numeric_tags.push(tag);
62        }
63      }
64    }
65  }
66  // Skip when already 1..N (correct sequence — do not touch).
67  let already_sequential = numeric_tags
68    .iter()
69    .enumerate()
70    .all(|(i, t)| t.get_content().trim() == (i + 1).to_string());
71  if already_sequential {
72    return;
73  }
74  for (i, tag) in numeric_tags.into_iter().enumerate() {
75    set_number_tag_text(&tag, &(i + 1).to_string());
76  }
77}

79/// Rewrite a numeric line-number `<ltx:tag>`'s text to `n` while PRESERVING any
80/// font-markup wrapper. `\algocf@printnl` wraps the number in `\NlSty`
81/// (`\textnormal{\textbf{…}}`), so a numbered tag is `<ltx:tag><ltx:text
82/// font="bold">N</ltx:text></ltx:tag>`. A blunt `set_content` on the tag would
83/// replace that `<ltx:text>` wrapper with a bare text node, stripping the uniform
84/// bold and reintroducing the per-line font leak. Instead descend through
85/// single-element-child wrappers and set the text on the innermost node, keeping the
86/// `\NlSty` styling intact. Guard: `cluster_algorithm2e_uniform_line_number_font`.
87fn set_number_tag_text(tag: &Node, n: &str) {
88  let mut target = tag.clone();
89  loop {
90    let children = target.get_child_elements();
91    if children.len() == 1 {
92      target = children.into_iter().next().unwrap();
93    } else {
94      break;
95    }
96  }
97  let _ = target.set_content(n);
98}

383  // `leaveHorizontal_internal`: repack + mode reset, no `\par`), and the next line's
384  // first content re-enters hmode → fires `\everypar`. The binding therefore does NOT
385  // fire `\the\everypar` at endline. An empty paragraph never enters hmode → never
386  // numbered, so no phantom numbers; we still DROP a listingline left with no real
387  // content (only `<ltx:tags>` / indentation `<ltx:rule>`s) — `\lx@strippar`'s empty
388  // `\lx@algo@parx` block-terminator lines. Witnesses arXiv:2602.20153
389  // (`\Comment*[r]`), arXiv:2512.24601 (html_feedback #6080/#6236).
390  DefConstructor!("\\lx@algo@@endline", sub[document] {
391    document.close_element("ltx:listingline")?;
392    if let Some(line) = document.get_node().get_last_child()
393      && document::get_node_qname(&line) == pin!("ltx:listingline")
394      && !listingline_has_content(&line)
395    {
396      document.remove_node(line);
397    }
398  });
399  // Gentle vmode reset at the line seam: resume internal_vertical WITHOUT firing a
400  // `\par` (which would re-enter the line machinery), so the next line's first box is
401  // a fresh vmode→hmode entry that the engine's `\everypar` hook numbers.
402  DefPrimitive!("\\lx@algo@leave@hmode", {
403    ::latexml_core::stomach::leave_horizontal_internal();
404  });
405  // Indentation is now prepended at ENDLINE (see `\lx@prepend@indentation@`), so
406  // startline emits NO horizontal content — a line's first hmode entry is its real
407  // content, keeping the engine `\everypar` numbering correctly timed.
408  DefMacro!("\\lx@algo@startline", "\\lx@algo@@startline");
409  DefMacro!("\\lx@algo@endline", "\\lx@prepend@indentation\\lx@algo@@endline\\lx@algo@leave@hmode");
410
411  // Indentation prepending — Perl L197-208 (faithful DOM prepend at ENDLINE).
412  // `\lx@prepend@indentation@{#1}` floats up to the `<ltx:listingline>`, detaches
413  // its current children, absorbs the indentation `#1` (so it becomes the leading
414  // content), then re-appends the saved children — matching Perl's
415  // floatToElement/removeChild/absorb/appendChild.
416  //
417  // Emitting the indentation HERE (endline) rather than at `\lx@algo@startline` is
418  // required for the engine `\everypar` line numbering (see `\lx@algo@endline`):
419  // the indentation (`\hskip`s + a `<ltx:rule>`) is horizontal material, so emitting
420  // it at startline entered horizontal mode BEFORE the line's real content and fired
421  // `\everypar` too early — numbering a nested standalone `\tcp` comment that had not
422  // yet set `everyparnl=\relax`. At endline it is a pure DOM prepend (no hmode
423  // entry), so a line's FIRST hmode entry is its real content, at the correct time.
424  // The floatToElement reposition is also load-bearing structurally: it lifts the
425  // cursor OUT of any open inline box (the `_noautoclose` `<ltx:text>` an `\hbox`
426  // opens when a listing is wrapped in `\colorbox{…}{…}`) so the following
427  // `\lx@algo@@endline` closes cleanly. Witnesses 1911.01815, 1903.04631
428  // (algorithm2e inside `\colorbox`/`\hbox`).
429  DefMacro!("\\lx@prepend@indentation", "\\lx@prepend@indentation@{\\the\\lx@algo@indentation}");
430  DefConstructor!("\\lx@prepend@indentation@{}", sub[document, args] {
431    document.float_to_element("ltx:tags", false)?;
432    let mut line = document.get_node().clone();
433    let mut saved: Vec<Node> = line.get_child_nodes();
434    for child in saved.iter_mut() {
435      child.unlink();
436    }
437    if let Some(indent) = args.first().and_then(|a| a.as_ref()) {
438      document.absorb(indent, None)?;
439    }
440    for child in saved.iter_mut() {
441      line.add_child(child)?;
442    }
443  });
444
445  // Line numbering — Perl L210-221 (the ACTIVE \algocf@printnl; Perl L195's
446  // plain template is immediately overridden).
447  //
448  // The earlier Rust port used the plain L195 template
449  // (`<ltx:tags><ltx:tag>#1</ltx:tag></ltx:tags>`), which emits the line-number
450  // tags at the CURRENT cursor. That errors "ltx:tags isn't allowed in
451  // <ltx:text>" whenever `\nl` fires while an inline `<ltx:text>` is open — e.g.
452  // a `\SetKwInput{KwInit}{\nl initialize}` line, where the KwInput label
453  // wrapper opens an `<ltx:text>` before `\nl`. Perl's active definition first
454  // `floatToElement('ltx:tags')` to climb OUT of that `<ltx:text>` up to the
455  // enclosing `<ltx:listingline>` (which can contain tags), emits the tags
456  // there, then leaves the cursor restored so following content flows on. This
457  // is the same float used by `\lx@prepend@indentation@` above. We restore the
458  // saved node (rather than Perl's manual childNode remove/re-append prepend)
459  // so the KwInput label's content keeps its wrapper. Witness 2104.02680
460  // (`\SetKwInput{KwInit}{\nl initialize}`).
461  //
462  // The printed number is wrapped in `\NlSty` — real algorithm2e.sty L1638/L1644:
463  // `\NlSty{#1}` = `\textnormal{\textbf{\relsize{-2}#1}}` — so it renders uniform
464  // upright-bold regardless of the font active when `\nl` fires. Our numbering
465  // seam fires `\nl` at content-start (`\everypar`), where a `\For`/`\If`/`\While`
466  // line has already opened `\KwSty` bold; a BARE number would inherit that on
467  // keyword lines and stay plain elsewhere (non-uniform). `\NlSty`'s `\textnormal`
468  // resets shape/series, so every number is uniform bold. (Perl's `.ltxml` L210
469  // drops the wrap but recovers uniformity from its endline `\the\everypar` timing
470  // instead — we chose content-start timing, so we restore the real-sty wrap.)
471  // Matches the pdflatex golden. Guard: `cluster_algorithm2e_uniform_line_number_font`.
472  // Split into a macro (wraps `\NlSty` before digestion) feeding the constructor.
473  DefMacro!("\\algocf@printnl{}", "\\algocf@printnl@i{\\NlSty{#1}}");
474  DefConstructor!("\\algocf@printnl@i{}", sub[document, args] {
475    let num = args.first().and_then(|a| a.as_ref());
476    let savenode = document.float_to_element("ltx:tags", false)?;
477    document.open_element("ltx:tags", None, None)?;
478    match num {
479      Some(n) => { document.insert_element("ltx:tag", vec![n], None)?; },
480      None => { document.insert_element("ltx:tag", Vec::new(), None)?; },
481    }
482    document.close_element("ltx:tags")?;
483    if let Some(sn) = savenode { document.set_node(&sn); }
484  });
485
486  // Strip trailing pars — Perl L141-145
487  DefMacro!("\\lx@strippar{}", "#1\\lx@algo@parx\\lx@algo@parx\\lx@algo@parx");
