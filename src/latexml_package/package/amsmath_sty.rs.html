12/// Perl: amsAlignmentBindings($template, %properties) — amsmath.sty.ltxml lines 107-120
13/// Simple alignment bindings for ams environments (no equation rearrangement)
14fn ams_alignment_bindings(template: Template, mut xml_attributes: HashMap<String, String>) {
15  use crate::engine::tex_tables::{alignment_bindings, array_strut};
16  // Its `\label`s are amsmath's `\label@in@display`, made after the cell's group (`hold_raw_refstep_for_display`):
17  // an `{aligned}`/`{split}` in an `{equation*}` names the step held at its start (pdflatex 44), a `{multline}`'s none
18  // (`\math@cr@@@align`'s display tag only under `\iftag@`, amsmath.sty:1833-1843; review probe rev78g2 al.tex
19  // `{multline*}` `??`; repro `cross-references/amsmath_row_label_names_the_display_step.tex`).
20  let multline = xml_attributes.get("name").map(String::as_str) == Some("multline");
21  hold_raw_refstep_for_display(!multline);
22  let_display_label();
23  // amsmath's rows carry `\strut@`, `\strutbox`'s copy (amsmath.sty:714-718): an `aligned` in a table cell, where
24  // `\@array` zeroed `\baselineskip`, keeps its rows' height (KNOWN_PERL_ERRORS #445)
25  let mut properties = array_strut()
26    .map(|strut| stored_map!("strut" => strut))
27    .unwrap_or_default();
28  // amsmath's `\halign` preambles make each cell math with their own `$` (amsmath.sty `\aligned@a`, `\gather@`:
29  // `$\m@th\displaystyle##$`): a cell whose template `\omit` drops (`\multicolumn`) is text, as an array's is not
30  // (its `\multicolumn` re-inserts `$`, latex.ltx `\@arrayclassz`). `\lx@alignment@multicolumn` reads this.
31  properties.insert("template_math", Stored::Bool(true));
32  // Perl (amsmath.sty.ltxml L111-114):
33  //   my $cur_jot = LookupDimension('\jot');
34  //   if ($cur_jot && $cur_jot->valueOf != LookupDimension('\lx@default@jot')->valueOf)
35  //     { $properties{rowsep} = $cur_jot; }
36  //   alignmentBindings($template, 'math', attributes => {%properties});
37  // The rowsep is passed via the `attributes` arg, so it must land in
38  // `xml_attributes` (the container's XML attributes), NOT the alignment
39  // `properties` — only `xml_attributes` reach the openContainer callback. Use
40  // LookupDimension (lookup_dimension_cs) so a `\def`-ized `\jot` reads silently.
41  if !xml_attributes.contains_key("rowsep") {
42    let cur_jot = lookup_dimension_cs("\\jot", false).unwrap_or_default();
43    if cur_jot.value_of()
44      != lookup_dimension_cs("\\lx@default@jot", false)
45        .unwrap_or_default()
46        .value_of()
47    {
48      xml_attributes.insert(String::from("rowsep"), cur_jot.to_string());
49    }
50  }
51  alignment_bindings(template, String::from("math"), properties, xml_attributes);
52  // Perl: amsAlignmentBindings calls alignmentBindings('math') which sets
53  // Let(T_MATH, '\lx@dollar@in@mathmode'). Perl does NOT override it back.
54  // The \lx@dollar@in@mathmode handles nested math/text correctly using
55  // MATH_ALIGN_$_BEGUN boxing-level tracking.
56  let_i(
57    &T_CS!("\\\\"),
58    &T_CS!("\\lx@alignment@newline@noskip"),
59    None,
60  );
61}

63/// Perl: amsRearrangeableBindings — creates equationgroup/equation/_Capture_ alignment
64/// amsmath.sty.ltxml lines 121-147
65fn ams_rearrangeable_bindings(
66  template: Template,
67  mut xml_attributes: HashMap<String, String>,
68  redirect_label: bool,
69) -> Result<()> {
70  // A row's `\label` is amsmath's `\label@in@display` (amsmath.sty:1230-1236), made in the row's tag cell or not at
71  // all (:1240-1252), after the cell group a `\refstepcounter` was made in restored `\@currentlabel`
72  // (`hold_raw_refstep_for_display`). A `{gather}` makes the display tag on every row (:1621-1627): its labels name
73  // the step current where the display starts (`{gather*}` after a step: that step, a step in the cell the section's
74  // 7). The align family (`redirect_label`: `{align}`, `{flalign}`, `{alignat}`, starred or not) makes it only under
75  // `\iftag@` (:1833-1843): an untagged row's label is the next tagged row's (`\df@label` carried) or none (`??`),
76  // never a held step — it names none (kept on the section or the row as before; review probe rev78g2 al2.tex,
77  // repro `cross-references/amsmath_row_label_names_the_display_step.tex`; OXIDIZED_DESIGN_DIVERGENCES #813).
78  hold_raw_refstep_for_display(!redirect_label);
79  // amsmath.sty `\start@align`, `\start@gather`: the `\halign` is the display itself (`$$\halign`).
80  let mut properties = SymHashMap::default();
81  properties.insert(engine::tex_tables::ALIGNMENT_IS_DISPLAY, Stored::Bool(true));
82  // Perl (amsmath.sty.ltxml L123-125):
83  //   my $cur_jot = LookupDimension('\jot');
84  //   if ($cur_jot && $cur_jot->valueOf != LookupDimension('\lx@default@jot')->valueOf)
85  //     { $properties{attributes}{rowsep} = $cur_jot; }
86  // i.e. for the rearrangeable envs ({align},{gather}) `\jot`≠default sets the
87  // equationgroup's rowsep attribute. (Was missing entirely — Rust emitted no
88  // rowsep on align/gather, diverging from Perl.)
89  if !xml_attributes.contains_key("rowsep") {
90    let cur_jot = lookup_dimension_cs("\\jot", false).unwrap_or_default();
91    if cur_jot.value_of()
92      != lookup_dimension_cs("\\lx@default@jot", false)
93        .unwrap_or_default()
94        .value_of()
95    {
96      xml_attributes.insert(String::from("rowsep"), cur_jot.to_string());
97    }
98  }
99  // Digest-time group-id mint — same reasoning and ruling as the eqnarray
100  // site (`latex_constructs.rs`, OXIDIZED_DESIGN #91): Perl mints at absorb
101  // time and stamps every group with the LAST section's prefix.
102  let group_id: Option<String> = ref_step_id("@equationgroup")
103    .ok()
104    .and_then(|props| props.get("id").map(ToString::to_string));
105  // Create alignment with equationgroup/equation/_Capture_ hooks
106  let alignment = Alignment::new(AlignmentConfig {
107    template: Some(template),
108    open_container: Rc::new(move |document, mut props| {
109      if let Some(id) = &group_id {
110        props.insert(String::from("xml:id"), id.clone());
111      }
112      // Merge xml_attributes into props
113      // (attributes passed at creation time)
114      document
115        .open_element("ltx:equationgroup", Some(props), None)
116        .map(Some)
117    }),
118    close_container: Rc::new(|document| document.close_element("ltx:equationgroup")),
119    open_row: Rc::new(|document, mut props| {
120      // Read equation tags from state
121      if let Some(Stored::HashStored(eq_props)) = remove_value("EQUATIONROW_PROPS")
122        && let Some(id) = eq_props.get("id")
123      {
124        props.insert(String::from("xml:id"), Stored::from(id.to_string()));
125      }
126      // Extract tags (Digested) before converting to string props
127      let tags_digested = props.remove("tags");
128      let str_props: HashMap<String, String> =
129        props.into_iter().map(|(k, v)| (k, v.to_string())).collect();
130      document.open_element("ltx:equation", Some(str_props), None)?;
131      // If we have digested tags, absorb them into the opened element
132      if let Some(Stored::Digested(d)) = tags_digested {
133        document.absorb(&d, None)?;
134      }
135      Ok(())
136    }),
137    close_row: Rc::new(|document| document.close_element("ltx:equation")),
138    open_column: Rc::new(|document, props| {
139      document
140        .open_element("ltx:_Capture_", Some(props), None)
141        .map(Some)
142    }),
143    close_column: Rc::new(|document| document.close_element("ltx:_Capture_")),
144    is_math: true,
145    properties,
146    xml_attributes,
147  });
148  assign_alignment(alignment, None);
149  // NOTE: Perl's amsRearrangeableBindings does NOT set Let(T_MATH, '\lx@dollar@in@mathmode').
150  // That letdef is only in alignmentBindings (for {tabular}/{array}/{eqnarray}).
151  // For rearrangeable environments ({align},{gather}), the column template already contains
152  // literal $ tokens. The $ meaning is left as \lx@dollar@default — the template $ tokens
153  // toggle math/text mode via the default mechanism.
154  // Adding the letdef here would cause nested \begin{aligned} inside \begin{align} to hang:
155  // the after-template $ would run \lx@dollar@in@mathmode while inner frames are still on stack,
156  // mismatching MATH_ALIGN_$_BEGUN and opening restricted_horizontal mode instead of closing inline
157  // math.
158  let_i(
159    &T_CS!("\\\\"),
160    &T_CS!("\\lx@alignment@newline@noskip"),
161    None,
162  );
163  let_i(
164    &T_CS!("\\lx@alignment@row@before"),
165    &T_CS!("\\eqnarray@row@before"),
166    None,
167  );
168  let_i(
169    &T_CS!("\\lx@alignment@row@after"),
170    &T_CS!("\\eqnarray@row@after"),
171    None,
172  );
173  // Perl: Let('\intertext', '\@ams@intertext');
174  let_i(&T_CS!("\\intertext"), &T_CS!("\\@ams@intertext"), None);
175  // `\label` redirect — applied ONLY to multi-column rearrangeable envs
176  // ({align}/{alignat}/{flalign}), NOT single-column {gather}.
177  //
178  // Perl's `amsRearrangeableBindings` (amsmath.sty.ltxml L120-147) does NOT
179  // redirect `\label` at all; only `\@eqnarray@bindings` does. The redirect
180  // here is a Rust-only workaround: a multi-column cell whose ONLY content
181  // is a `\label` is "skippable" (Rust's `\hfil` contributes no width, unlike
182  // Perl's), so the plain `\lx@label` constructor never floats `labels=` onto
183  // the parent equation. Routing it through `\lx@hidden@noalign` processes
184  // the label at row level so `labels=` survives (witness split.tex's
185  // `\label{eq:before}\n&x`).
186  //
187  // But that same redirect BREAKS single-column {gather}: a `\label` before
188  // `\lefteqn` gets swallowed by the column-scan loop as `\lx@hidden@noalign`
189  // (never starting the column), so `\lefteqn` is expanded with
190  // `\if@in@firstcolumn` still TRUE and emits `\multicolumn{3}` into a 1-column
191  // gather -> "Extra alignment tab '&'" (driver 1906.11496). For gather we
192  // therefore match Perl exactly (no redirect): the plain `\label` starts the
193  // column, so `\lefteqn` takes the `\rlap` branch. gather has no `&`, so the
194  // skippable-label-only-cell case the redirect guards against does not arise
195  // (a bare `\label\\` row is dropped by both engines anyway).
196  if redirect_label {
197    let_i(
198      &T_CS!("\\lx@eqnarray@save@label"),
199      &T_CS!("\\lx@label"),
200      Some(Scope::Global),
201    );
202    let_i(&T_CS!("\\label"), &T_CS!("\\lx@eqnarray@label"), None);
203  }
204  let_display_label();
205  Ok(())
206}

208/// The display's `\label` is amsmath's `\label@in@display` (amsmath.sty:1230-1236): `\lx@ams@display@label`, which
209/// announces the label it makes (`mark_display_label`) and makes it with the `\label` it replaced
210/// (`\lx@ams@display@outer@label`: the kernel's, the `{align}` redirect, cleveref's). Expandable, so a label-only cell
211/// is scanned as before. Once per display: an inner one (`{aligned}`) keeps the outer's. Repros
212/// `cross-references/amsmath_row_label_names_the_display_step.tex`,
213/// `cross-references/amsmath_saved_kernel_label_names_the_cell_step.tex` (witness 2310.19138).
214fn let_display_label() {
215  let wrapped = matches!(
216    (lookup_meaning(&T_CS!("\\label")), lookup_meaning(&T_CS!("\\lx@ams@display@label"))),
217    (Some(Stored::Expandable(a)), Some(Stored::Expandable(b))) if Rc::ptr_eq(&a, &b)
218  );
219  if !wrapped {
220    let_i(
221      &T_CS!("\\lx@ams@display@outer@label"),
222      &T_CS!("\\label"),
223      None,
224    );
225    let_i(&T_CS!("\\label"), &T_CS!("\\lx@ams@display@label"), None);
226  }
227}

2360  /// No known reading (a closure without a parameter list, an ltcmd signature with a delimited or embellishment

2501          "m" => Some("{}"),
2502          "o" | "O" => Some("[]"),
2503          "s" => Some("OptionalMatch:* "),
2504          "+" | "!" | ">" | "=" => Some(""),
2505          _ => None,
2506        };
2507        spec = spec
2508          .take()
2509          .zip(parameter)
2510          .map(|(spec, parameter)| spec + parameter);
2511      }),
2512      _ => {},
2513    }
2514  }

2516    return Ok(UndelimitedArgument::Braced);
2517  }

2519    Some(spec) => parse_parameters(spec.trim(), &command, true)?,
2520    None => None,

2524    tex_errs: true,

2529/// The arguments of `command`'s call, read from what follows by `parameters` and reverted as the author wrote them
2530/// (`{x}`, `[a]`, `*`); `None` for a parameter list with a delimiter that is no argument (`\def\cs(#1){…}`), whose
2531/// reversion would drop it, or with an argument digested as it is read; `None` too, the arguments read so far put
2532/// back, when an undelimited argument would be a token that ends or structures the current list
2533/// ([`ends_the_list`]): the author's call is missing an argument there (`\hat\pair{a} & x`), and its braced reading
2534/// would take a cell's `&` or a row's `\\` into the group, so the call is left unbraced, as TeX leaves the tokens.
2535fn read_call_arguments(parameters: &Parameters) -> Result<Option<Vec<Token>>> {
2536  let parameters = parameters.get_parameters();
2537  if parameters
2538    .iter()
2539    .any(|parameter| parameter.novalue || parameter.predigest.is_some())
2540  {
2541    return Ok(None);
2542  }
2543  let mut read = Vec::new();
2544  for parameter in parameters {
2545    let undelimited =
2546      !parameter.optional && with(parameter.name, |name| BRACED_ARGUMENT_TYPES.contains(&name));
2547    if undelimited && next_ends_the_list()? {
2548      unread_vec(read);
2549      return Ok(None);
2550    }
2551    let argument = parameter.read(None)?;
2552    if let Some(reverted) = parameter.revert(argument.into())? {
2553      read.extend(reverted.unlist());
2554    }
2555  }
2556  Ok(Some(read))
2557}

2559/// Whether the next non-space token, which an undelimited argument would take, ends or structures the current list
2560/// ([`ends_the_list`]), or the input has ended ([`at_end_of_all_input`]); the end of a token list that is not the end
2561/// of the input (an argument digested on its own, `\frac{\hat\nv}{2}`) stands for its closing `}`, which a call reads
2562/// on, as at a `}`. The spaces before it are read and put back with it, so nothing is consumed.
2563fn next_ends_the_list() -> Result<bool> {
2564  let mut spaces = Vec::new();
2565  let next = loop {
2566    match peek_token()? {
2567      Some(space) if space.get_catcode() == Catcode::SPACE => {
2568        read_token()?;
2569        spaces.push(space);
2570      },
2571      next => break next,
2572    }
2573  };
2574  unread_vec(spaces);
2575  Ok(match next {
2576    Some(token) => ends_the_list(&token),
2577    None => at_end_of_all_input(),
2578  })
2579}

2581/// Whether `token` ends or structures the list being built, so no argument is taken from it: by meaning, a column
2582/// end (`&` of catcode 4 or a `\let` copy, `\cr`, `\crcr`, `\span`: [`gullet::is_column_end`]) and a parameter token
2583/// (catcode 6); by name, the control sequence `\par` (tex.web §392's `par_token`, whatever it means), and the macros
2584/// that end a row, an environment or a formula, whose meaning changes with the context: `\\`, `\end`, `\]`, `\)`.

2589      && token.with_str(|name| matches!(name, "\\par" | "\\\\" | "\\end" | "\\]" | "\\)")))

2603fn undelimited_argument(
2604  package: &str,
2605  reader: &str,
2606  head: Tokens,
2607  argument: Tokens,
2608  nests_accents: bool,
2609) -> Result<Tokens> {
2610  let mut tokens = head.unlist();
2611  let argument = argument.unlist();
2612  let reading = match argument.as_slice() {
2613    [single] if matches!(single.get_catcode(), Catcode::CS | Catcode::ACTIVE) => {
2614      classify_undelimited_argument(single, nests_accents)?
2615    },
2616    _ => UndelimitedArgument::Braced,
2617  };

2619    UndelimitedArgument::Braced => {
2620      tokens.push(T_BEGIN!());
2621      tokens.extend(argument);
2622      tokens.push(T_END!());
2623    },
2624    UndelimitedArgument::NestedAccent => {
2625      tokens.push(T_BEGIN!());
2626      tokens.extend(argument);
2627      tokens.extend([T_BEGIN!(), T_END!()]);

2638      if tex_errs && !lookup_bool(engine::tex_math::REREAD_COPY) {
2639        let given = argument[0].with_str(|name| name.to_string());

2641          (Some(arguments), _) => {
2642            let mut written = vec![argument[0]];
2643            written.extend(arguments.iter().copied());

2645              "read as {reader}{{{}}}",

2648          },

2650            s!("read as {reader}{{{given}…}}, {given} with its argument")
2651          },

2653          (None, CallReading::Item(_)) => {
2654            s!(
2655              "read as {reader}{{{given}…}}, {given} with the arguments it reads from what follows it"
2656            )
2657          },
2658          (None, _) => s!("read as {reader}{{{given}…}}, {given} with what follows it"),
2659        };
2660        let message = s!(
2661          "{} {reader} takes {given} alone as its argument, where TeX errs: {given} reads an argument of its own; \
2662           {reading}",
2663          possessive(package)
2664        );

2668          get_source_location().unwrap_or_default()
2669        } else {
2670          at
2671        };
2672        let at = if at.is_empty() { at } else { s!("\n\t{at}") };
2673        emit_warn(
2674          "malformed",
2675          reader,
2676          &s!(
2677            "{message}{at}\n\tIn {}:{}:{}\n",
2678            file!(),
2679            line!(),
2680            column!()
2681          ),
2682        );
2683      }
2684      match call {
2685        Some(arguments) => {
2686          tokens.push(T_BEGIN!());
2687          tokens.push(command);
2688          tokens.extend(arguments);
2689          tokens.push(T_END!());
2690        },
2691        None => tokens.push(command),
2692      }
2693    },
2694  }
2695  Ok(Tokens::new(tokens))
2696}

2724/// The font size of a delimiter `\bBigg@{<factor>}` sizes. amsmath makes the delimiter
2725/// `<factor>\big@size` tall (amsmath.sty:726-738), its four sizes being `\big` = `\bBigg@\@ne`,
2726/// `\Big` = `\bBigg@{1.5}`, `\bigg` = `\bBigg@\tw@`, `\Bigg` = `\bBigg@{2.5}` (:721-724). Those four
2727/// are the named sizes `\big` … `\Bigg` have here (Perl Font.pm:191, 1.2/1.6/2.1/2.6 times the
2728/// nominal size); any other factor is read between and beyond them along the same scale, so a
2729/// document's `\vast` = `\bBigg@{4}` is larger than `\Bigg` and `\bBigg@{0.8}` smaller than `\big`.
2730fn bbigg_font_size(factor: f64) -> f64 {
2731  const NAMED: [(f64, &str, f64); 4] = [
2732    (1.0, "big", 1.2),
2733    (1.5, "Big", 1.6),
2734    (2.0, "bigg", 2.1),
2735    (2.5, "Bigg", 2.6),
2736  ];
2737  if let Some((_, name, _)) = NAMED.iter().find(|(f, ..)| (f - factor).abs() < 1e-6) {
2738    return font::rationalize_font_size(name);
2739  }
2740  let ratio = if factor <= 1.0 {
2741    1.2 * factor.max(0.0)
2742  } else if factor >= 2.5 {
2743    2.6 + (factor - 2.5)
2744  } else {
2745    let i = NAMED.iter().rposition(|(f, ..)| *f <= factor).unwrap_or(0);
2746    let ((f0, _, r0), (f1, _, r1)) = (NAMED[i], NAMED[i + 1]);
2747    r0 + (r1 - r0) * (factor - f0) / (f1 - f0)
2748  };
2749  ratio * font::rationalize_font_size("normal")
2750}
