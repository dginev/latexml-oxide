6  // Bail if algorithmicx already defined \algorithmic (deeply incompatible)
7  if lookup_definition(&T_CS!("\\algorithmic"))?.is_some() {

16  Let!("\\lx@orig@algorithmic", "\\algorithmic");
17  DefMacro!("\\algorithmic", "\\lx@setup@algorithmic\\lx@orig@algorithmic");
18
19  DefPrimitive!("\\lx@setup@algorithmic", {
20    ResetCounter!("ALC@line");
21    // If not within an algorithm environment, step the counter for its id's
22    let in_algorithm = with_stacked_values_sym(pin!("current_environment"), |vals| {
23      vals.iter().any(|s| s.eq_text("algorithm"))
24    });
25    if !in_algorithm {
26      ref_step_id("algorithm")?;
27    }
28    Let!("\\list", "\\lx@algorithmic@beginlist");
29    Let!("\\endlist", "\\lx@algorithmic@endlist");
30    Let!("\\item", "\\lx@algorithmic@item");
31    Let!("\\hfill", "\\lx@algorithmic@hfill");
32    Let!("\\@centercr", "\\lx@algorithmic@centercr");
33  });

35  // algorithmic.sty's `\\` is `\@centercr` (algorithmic.sty:178 `\renewcommand{\\}{\@centercr}`): `\unskip\par`
36  // (latex.ltx:15402-15406), a new line under the same line number. A listing line is no paragraph, so the `\par` wrote
37  // nothing and, the `\unskip` having taken the space, the words either side met ("step on" and the formula on the
38  // next line, 2605.13790). The line end is a `<break/>` (Perl writes nothing; DIVERGENCES #422); `*` and an optional
39  // `[<len>]` are read as `\@xcentercr` reads them, and `\@nolnerr` stays for a `\\` with no line to end. The `\let`
40  // is `\@centercr`'s, so a raw `\centering` inside the environment breaks its lines the same way. Repro
41  // list-structure/algorithmic_line_break; guard `perfect_kernel_batch59::algorithmic_line_break`.
42  DefMacro!("\\lx@algorithmic@centercr",
43    r"\ifhmode\unskip\else\@nolnerr\fi\lx@algorithmic@break\@ifstar\lx@algorithmic@xcentercr\lx@algorithmic@xcentercr");
44  DefMacro!("\\lx@algorithmic@xcentercr", r"\@ifnextchar[\lx@algorithmic@icentercr\ignorespaces");
45  DefMacro!("\\lx@algorithmic@icentercr[]", r"\ignorespaces");
46  // Only where a line can break: `\\` between lines (no `\STATE` open, as when algpseudocode's `\State` is used
47  // here undefined, 2605.30102) is a `\par` that writes nothing, as before.
48  DefConstructor!("\\lx@algorithmic@break", sub[document] {
49    if let Some(context) = document.get_element()
50      && document::can_contain(&context, "ltx:break")
51    {
52      document.insert_element("ltx:break", Vec::new(), None)?;
53    }
54  });
55
56  DefConstructor!("\\lx@algorithmic@beginlist{}{}", "<ltx:listing>",
57    before_construct => sub[document] {
58      document.maybe_close_element("ltx:p")?;
59    },
60    after_digest => sub[_whatsit] {
61      Let!("\\list", "\\lx@algorithmic@beginlist@inner");

63    });
64
65  DefConstructor!("\\lx@algorithmic@endlist", "</ltx:listing>",

67      end_mode("internal_vertical")?;
68    },
69    before_construct => sub[document] {
70      document.maybe_close_element("ltx:listingline")?;
71    });
72
73  DefConstructor!("\\lx@algorithmic@beginlist@inner{}{}", "",
74    after_digest => sub[_whatsit] {

76    });

78  // An `\item[label]` prints its label in place of the line number (the counter is stepped by `\ALC@it`, which a
79  // bare `\item` follows): algorithmic.sty:155 `\REQUIRE` is `\item[\algorithmicrequire]` ("Require:", renamed
80  // "Input:" in 2201.01230), which Perl and earlier Rust tagged "0:" with the line counter; an `\item[]` prints none.
81  // The label goes braced to its own constructor, so a `]` in it stays (63j).
82  DefMacro!("\\lx@algorithmic@item OptionalUndigested", sub[(label)] {
83    let mut out = match label {
84      Some(label) => {
85        let mut out = vec![T_CS!("\\lx@algorithmic@item@label"), T_BEGIN!()];
86        out.extend(label.unlist());
87        out.push(T_END!());
88        out
89      },
90      None => vec![T_CS!("\\lx@algorithmic@item@@")],
91    };
92    out.extend(mouth::tokenize_internal("\\hskip\\ALC@tlm\\relax").unlist());
93    Ok(Tokens::new(out))
94  });
95
96  DefConstructor!("\\lx@algorithmic@labeltags{}", "<ltx:tags><ltx:tag>#1</ltx:tag></ltx:tags>");
97  DefConstructor!("\\lx@algorithmic@item@@",
98    "<ltx:listingline xml:id='#id' itemsep='#itemsep'>#tags",
99    properties => sub[_args] {
100      let id = digest(T_CS!("\\theALC@line@ID"))?.to_attribute();
101      let tags = Stored::from(digest(Invocation!("\\lx@make@tags",
102        vec![Some(Tokens!(T_OTHER!("ALC@line")))]))?);
103      Ok(stored_map!("id" => id, "tags" => tags))
104    },
105    before_construct => sub[document] {
106      document.maybe_close_element("ltx:listingline")?;
107    });
108  DefConstructor!("\\lx@algorithmic@item@label Undigested",
109    "<ltx:listingline xml:id='#id' itemsep='#itemsep'>#tags",
110    properties => sub[args] {
111      let id = digest(T_CS!("\\theALC@line@ID"))?.to_attribute();
112      let label = match args.first() {
113        Some(Some(label)) => Some(label.revert()?).filter(|label| !label.unlist_ref().is_empty()),
114        _ => None,
115      };
116      Ok(match label {
117        Some(label) => stored_map!("id" => id,
118          "tags" => Stored::from(digest(Invocation!("\\lx@algorithmic@labeltags", vec![Some(label)]))?)),
119        None => stored_map!("id" => id),
120      })
121    },
122    before_construct => sub[document] {
123      document.maybe_close_element("ltx:listingline")?;
124    });

127  NewCounter!("ALC@line", "algorithm", idprefix => "l");
128  DefMacro!("\\fnum@ALC@line", "\\ALC@lno");

131    "<ltx:text cssstyle='float:right'>");
