8  // Perl fancyvrb.sty.ltxml L22-25: hack internals to add css class ltx_verbatim
9  // to every typeset source line (fancyvrb applies \FancyVerbFormatLine per line
10  // for all Verbatim variants). NOTE: a package that redefines
11  // \FancyVerbFormatLine after requiring fancyvrb overwrites this hook — fvextra
12  // (L2249 `\def\FancyVerbFormatLine#1{#1}`) does exactly that, so fvextra_sty.rs
13  // re-installs the hook over its redefinition (issue #502).
14  Let!("\\lx@save@FancyVerbFormatLine", "\\FancyVerbFormatLine");
15  DefMacro!("\\FancyVerbFormatLine{}",
16    "\\lx@add@cssclass{ltx_verbatim}\\lx@save@FancyVerbFormatLine{#1}");
17
18  // === Issue #525 surpass: fancyvrb `frame=single` as a semantic framed box ===
19  // fancyvrb draws `frame=single` via raw \vrule/\hrule in the @Single hooks
20  // (fancyvrb.sty:776 selects them; the top/bottom rule is \FV@SingleFrameLine
21  // L869-904, the per-line side rules L948-963, the sep box \FV@SingleFrameSep
22  // L929-947). LaTeXML captures those as literal <ltx:rule> elements that never
23  // reconstruct into an HTML box — and the bottom \FV@SingleFrameSep box (only
24  // ink = two side vrules, no text) surfaces as a stray empty line below the last
25  // line. Same-host Perl 0.8.8 is byte-identical (SHARED-FAILURE); the lualatex
26  // PDF draws a proper frame. So: redefine the @Single hooks so `\FV@BeginListFrame`
27  // OPENS an `ltx_framed_rectangle` box and `\FV@EndListFrame` CLOSES it (both fire
28  // exactly once around the whole line set, inside \FV@List/\FV@EndList's single
29  // group), neutralizing the per-line side rules — which drops the raw rules AND
30  // the stray sep line in one move. framesep→padding, framerule→border-width,
31  // fvextra `\FancyVerbBackgroundColor`→background (its per-line \colorbox strip,
32  // collapsed to one wrapper background). rulecolor keeps the default black border.
33  // Only `frame=single` is remapped here (the case in issue #525); the `@Lines`
34  // topline/bottomline/lines variants still draw raw rules — a follow-up.
35  // OXIDIZED_DESIGN #111.
36  RawTeX!(concat!(
37    r"\def\FV@BeginListFrame@Single{",
38    r"\edef\lx@fv@sep{\the\dimexpr\FV@FrameSep\relax}",
39    r"\edef\lx@fv@rule{\the\dimexpr\FV@FrameRule\relax}",
40    r"\lx@fancyvrb@beginframe{\lx@fv@sep}{\lx@fv@rule}",
41    r"{\@ifundefined{FancyVerbBackgroundColor}{}{\FancyVerbBackgroundColor}}}", "\n",
42    r"\def\FV@EndListFrame@Single{\lx@fancyvrb@endframe}", "\n",
43    r"\let\FV@LeftListFrame@Single\relax", "\n",
44    r"\let\FV@RightListFrame@Single\relax", "\n",
45  ));
46  DefConstructor!("\\lx@fancyvrb@beginframe {}{}{}",
47    "<ltx:text framed='rectangle' class='ltx_framed_verbatim' cssstyle='#cssstyle' _noautoclose='1'>",
48    after_digest => sub[whatsit] {
49      let sep = whatsit.get_arg(1).map(|a| a.to_string()).unwrap_or_default();
50      let rule = whatsit.get_arg(2).map(|a| a.to_string()).unwrap_or_default();
51      let bg = whatsit.get_arg(3).map(|a| a.to_string()).unwrap_or_default();
52      let mut parts: Vec<String> = Vec::new();
53      // framerule → border width (skip a 0pt rule, e.g. fvextra's bg-only frame).
54      let rule = rule.trim();
55      if !rule.is_empty() && rule != "0.0pt" {
56        parts.push(s!("border-width:{rule}"));
57      }
58      // framesep → padding on all four sides.
59      let sep = sep.trim();
60      if !sep.is_empty() {
61        parts.push(s!("padding:{sep}"));
62      }
63      // fvextra backgroundcolor → a single wrapper background (name → hex).
64      let bg = bg.trim();
65      if !bg.is_empty() && bg != "none" {
66        parts.push(s!("background:{}", lookup_color(bg)));
67      }
68      // Only per-instance VALUES are inline here (border-width/padding/bg from
69      // the fancyvrb keys). The fixed responsive BEHAVIOR — cap at the viewport
70      // and scroll over-long lines within the box, so the print-`\linewidth`
71      // frame does not push its border off-screen on a phone — lives in the
72      // `ltx_framed_verbatim` class (LaTeXML.css delta + ar5iv-css mirror).
73      whatsit.set_property("cssstyle", parts.join(";"));
74      // The frame is a pure border/background wrapper — it must NOT carry the
75      // ambient `font` (typewriter). Clearing it keeps the box font-neutral so
76      // the inner verbatim lines carry their own font in the output; leaving it
77      // on the box makes the first line's font DEDUP against the box's font
78      // (document.rs open_text) resolve differently in the eager vs streamed
79      // build (the box's font is not yet the line's parent-font when streamed
80      // emits line 1), breaking the streaming==eager byte-identity invariant.
81      whatsit.set_property("font", Stored::None);
82      Ok(Vec::new())
83    });
84  DefConstructor!("\\lx@fancyvrb@endframe", "</ltx:text>");
85});
