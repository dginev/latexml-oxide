8/// Add `ids` to `node`'s `aria:describedby`, keeping whatever is already there.
9///
10/// `aria-describedby` is an id LIST, and a second `\Description` in the same
11/// float (or one alongside another that already wired the same host) would
12/// otherwise `set_attribute` straight over the first one's reference, leaving
13/// that description hidden in the DOM and announced by nothing — silently
14/// losing an annotation the author wrote. Existing ids keep their position,
15/// since order is announcement order.
16fn add_describedby(document: &mut Document, node: &mut Node, ids: &[String]) -> Result<()> {
17  let existing = node.get_attribute("aria:describedby").unwrap_or_default();
18  let mut refs: Vec<&str> = existing.split_whitespace().collect();
19  for id in ids {
20    if !refs.contains(&id.as_str()) {
21      refs.push(id);
22    }
23  }
24  document.set_attribute(node, "aria:describedby", &refs.join(" "))
25}

106  NewCounter!("acmlabel", "");
107  // `\Description[short]{long}` — acmart's accessible figure description
108  // (acmart.cls L895 gobbles both args; ACM requires the description, and HTML
109  // can carry it where PDF cannot, so we surface it rather than follow the
110  // class into discarding it).
111  //
112  // The MANDATORY long description is the real alt text, so it is what we
113  // emit. It is read `Undigested` — `ExpansionLevel::Off`, kept as
114  // `DigestedData::Postponed` tokens — so nothing inside it expands: the
115  // author never sees a defect there under pdflatex (the class gobbles it), so
116  // expanding it only manufactures errors. Witness arXiv:2607.21760, whose
117  // `\D1 … \D5` (a copy-paste slip from the adjacent `alt=` text, which has
118  // plain `D1 … D5`) raised `Error:undefined:\D` for content we then dropped.
119  //
120  // `LaTeXML-common.xsl` L404-421 maps any `aria:*` attribute to `aria-*`
121  // under HTML5, so setting them here needs no XSLT change. WHICH element
122  // they land on, and which slot each argument fills, is the
123  // `before_construct` table below.
124  //
125  // `[short]` and `{long}` are two DISTINCT authored fields, so they get two
126  // distinct elements — never concatenated into one. Merging them yields a
127  // run-on ("Fly 1 and Fly 2 look identical. Fly 1 and fly 2 comparison
128  // shows…") and destroys the distinction, so no consumer can tell which text
129  // the author wrote as the brief alternative. `aria-describedby` takes a
130  // SPACE-SEPARATED ID LIST and is announced in order, so where both are
131  // referenced it is short first, each still individually addressable.
132  //
133  // The old binding emitted the short one ALONE, which is why
134  // `t/complex/acm_aria` recorded "Fly 1 and Fly 2 look identical" and lost the
135  // sentence that actually describes the figure.
136  //
137  // Carried as `<ltx:note>` — the document builder places a note inside the
138  // float's `<caption>`, where the old binding put it. An `<ltx:text>` is
139  // INLINE, so the builder auto-opens a `<p>` for it, and a `<p>` at figure
140  // level is tagged `ltx_figure_panel`, which makes the post-processor spin up
141  // a whole `ltx_flex_figure`/`ltx_flex_cell` wrapper around two hidden spans
142  // (inert, since the caption partitions it and the real subfigures keep their
143  // width class, but pure noise in the output).
144  //
145  // The footnote decoration a note normally gets — the `†` mark and the
146  // `<role>: ` type prefix — is suppressed for these by a dedicated template in
147  // `LaTeXML-meta-xhtml.xsl` keyed on `ltx_acm_description`, so the referenced
148  // text stays clean for assistive technology.
149  //
150  // The short description needs its OWN id: it is referenced by
151  // `aria-describedby` whenever its text is not the one that became the
152  // image's `@alt`. It is derived in `properties` rather than written as
153  // `#id-short` in the template, which does NOT work: a `#name` hole runs to
154  // the end of the identifier, so `#id-short` names a property called
155  // `id-short` — absent, so the element silently emits NO xml:id at all —
156  // instead of `#id` followed by a literal `-short`. That produced a dangling
157  // `aria-describedby`, a reference resolving to nothing. Ids use a `-short`
158  // suffix rather than a dotted one so they need no escaping in a CSS
159  // selector.
160  DefConstructor!("\\Description[] Undigested",
161    "^^?#1(<ltx:note xml:id='#shortid' class='ltx_nodisplay ltx_acm_description_short'>#1</ltx:note>)()\
162     <ltx:note xml:id='#id' class='ltx_nodisplay ltx_acm_description'>#2</ltx:note>",
163    properties => {
164      let mut props = RefStepCounter!("acmlabel")?;
165      if let Some(id) = props.get("id") {
166        let short = format!("{id}-short");
167        props.insert("shortid", Stored::from(short));
168      }
169      // Perl acmart.cls.ltxml L79 sets these so the hidden note claims no
170      // layout space; carried over.
171      props.insert("width", Stored::from(Dimension!("0pt")));
172      props.insert("height", Stored::from(Dimension!("0pt")));
173      Ok(props)
174    },
175    // acmart's own documentation: "Unlike \caption, which is used alongside the
176    // image, \Description is intended to be used INSTEAD OF the image." So a
177    // `\Description` is a TEXT ALTERNATIVE — and the thing it is an alternative
178    // TO is the image, not the float. It therefore lands on the IMAGE:
179    //
180    //   `\Description[s]{l}`           → img @alt = s, aria-describedby → l
181    //   `\Description{l}`, l plain     → img @alt = l (it replaces the image)
182    //   `\Description{l}`, l w/ markup → img @alt untouched, describedby → l
183    //
184    // `@alt` (via `ltx:graphics/@description`, `LaTeXML-misc-xhtml.xsl` L167-171)
185    // rather than `aria-label`, because that is the attribute an `<img>` has for
186    // exactly this purpose and the one assistive technology expects there.
187    //
188    // NOT `aria:label` on the `<ltx:figure>`, which is what this binding did
189    // before: `aria-label` sets the accessible NAME, and a float's name is its
190    // caption, so labelling the figure with the description DISPLACED
191    // "Figure 1. caption text" and hid the caption from a screen reader
192    // (reviewer report, brucemiller/LaTeXML#430 r3674103638; the fix that
193    // review asks for is precisely "label + description should be attached to
194    // the first image in the figure … and the `<img>` tag eventually gets
195    // `@alt` and not `@aria-label`").
196    //
197    // The markup case is why the argument is read `Undigested`: `@alt` is a
198    // plain string and cannot carry markup, so we must inspect the tokens to
199    // choose the slot BEFORE expanding anything. A control sequence (or an
200    // active/`$`/`^`/`_` token) means real markup, so the block carries it.
201    //
202    // The block is always emitted, so the text stays addressable, but it is
203    // referenced only when it is not already the `@alt` — otherwise the same
204    // sentence would be announced twice. An unreferenced hidden block is inert:
205    // `display:none` content is announced only when something references it.
206    //
207    // TWO CASES HAVE NO IMAGE TO USE. An author's annotation is never dropped,
208    // so it goes to the next best host — the enclosing float — as
209    // `aria:describedby`, which supplements the name instead of replacing it,
210    // so the caption survives either way. Both `Warn!`, because the result is
211    // second-best and the author can do something about it:
212    //
213    //  * No `ltx:graphics` in the float at all — a figure built from tabular,
214    //    text or TikZ content (`t/complex/acm_aria` is one), a `table` float,
215    //    or an empty one. There is no image to be the alternative to.
216    //  * MORE than one. A `\Description` is scoped to the whole float, so on a
217    //    multi-panel figure it describes the ensemble; making it panel 1's
218    //    `@alt` would assert that one sentence is the alternative for one
219    //    panel, which is a claim the author never made. The review says "the
220    //    first image"; we narrow that to the case where "first" is also "only",
221    //    where it is unambiguous.
222    //
223    // We can only see the graphics ALREADY BUILT when `\Description` is
224    // constructed, so a `\Description` written BEFORE its `\includegraphics`
225    // falls into the first case. That is the safe direction to fail — the
226    // description is still announced, just not as the image's alternative — the
227    // warning names it as a possible cause, and acmart's own documentation puts
228    // `\Description` after the graphic.
229    //
230    // acmart's NEWER mechanism for the same purpose is `\includegraphics[alt=…]`
231    // (switched on by `\DocumentMetadata`), handled in `graphicx_sty.rs` and
232    // landing on the same `description` attribute. When an author uses BOTH —
233    // e.g. arXiv:2607.21760, which repeats the same paragraph in each — the
234    // explicit `alt=` WINS: it names one image, while `\Description` names the
235    // float, so the more specific statement stands. `\Description` then only
236    // adds its `aria-describedby` references.
237    before_construct => sub[document, whatsit] {
238      let Some(id) = whatsit.get_property("id").map(|v| v.to_string()) else {
239        return Ok(());
240      };
241      // Does the long description contain markup, or is it plain text?
242      let long_is_plain = whatsit
243        .get_arg(2)
244        .and_then(|d| d.raw_tokens())
245        .is_some_and(|tks| {
246          tks.unlist_ref().iter().all(|t| {
247            matches!(
248              t.get_catcode(),
249              Catcode::LETTER | Catcode::OTHER | Catcode::SPACE | Catcode::EOL
250            )
251          })
252        });
253      // `\Description[]{…}` gives an empty optional argument, which the
254      // template's `?#1(…)` declines to emit — so there is no short note and
255      // no id to reference. Treat it as absent.
256      let short = whatsit
257        .get_arg(1)
258        .map(|d| d.to_attribute())
259        .filter(|s| !s.is_empty());
260      let short_id = whatsit.get_property("shortid").map(|v| v.to_string());
261      let Some(mut figure) = document.get_element() else {
262        return Ok(());
263      };
264      let mut graphics = document.findnodes(".//ltx:graphics", Some(&figure));
265      // Exactly one image: it IS the figure, so the description is its
266      // alternative. Zero or several: see the note above, keep it on the float.
267      let lone_graphic = (graphics.len() == 1).then(|| graphics.remove(0));
268      let mut described_by: Vec<String> = Vec::new();
269      match lone_graphic {
270        Some(mut graphic) => {
271          // `\includegraphics[alt=…]` already spoke for this image; it is the
272          // more specific statement, so it stands.
273          let alt = if graphic.get_attribute("description").is_some() {
274            None
275          } else if short.is_some() {
276            short.clone()
277          } else if long_is_plain {
278            whatsit.get_arg(2).map(|d| d.to_attribute())
279          } else {
280            None
281          };
282          match alt {
283            // Something of ours became the alternative. Whatever did NOT is
284            // still worth announcing, so reference it.
285            Some(text) => {
286              document.set_attribute(&mut graphic, "description", &text)?;
287              if short.is_some() {
288                described_by.push(id);
289              }
290            },
291            // Nothing of ours is the alternative — the author's own `alt=`
292            // holds it, or the lone description carries markup. Reference
293            // everything we emitted.
294            None => {
295              if short.is_some() && let Some(sid) = short_id {
296                described_by.push(sid);
297              }
298              described_by.push(id);
299            },
300          }
301          if !described_by.is_empty() {
302            add_describedby(document, &mut graphic, &described_by)?;
303          }
304        },
305        // Nowhere better to put it. Attach to the enclosing element rather than
306        // drop the author's annotation, and say so — the text is still
307        // announced, but not as the alternative for any one image, which is
308        // what a `\Description` is for.
309        None => {
310          if short.is_some() && let Some(sid) = short_id {
311            described_by.push(sid);
312          }
313          described_by.push(id);
314          add_describedby(document, &mut figure, &described_by)?;
315          let host = figure.get_name();
316          if graphics.is_empty() && host == "table" {
317            // NOT a fallback: a table float has no image BY CONSTRUCTION, and
318            // acmart asks for a `\Description` on every float — so the table
319            // itself is where the description belongs, and this is the author
320            // doing the right thing. Warning here demoted otherwise-clean
321            // papers: 27 of 45 sampled `no_problem -> warning` regressions in
322            // the 2026-07-30 sandbox-arxiv-2605 rerun were this one message.
323            // Report it (the attachment point is worth knowing), as INFO.
324            Info!("aria", "\\Description",
325              &s!("attached to the enclosing <table> as aria:describedby — a \
326                   table has no image to describe, so this is where the \
327                   description belongs"));
328          } else {
329            let why = if !graphics.is_empty() {
330              s!("it holds more than one image, so which one is described is ambiguous")
331            } else if host == "figure" {
332              s!("the float has no image to describe (a \\Description written \
333                  before its \\includegraphics, or a figure with no graphic)")
334            } else {
335              // Not a float at all — `\Description` outside a figure/table,
336              // where acmart's own `\@Description@present` bookkeeping expects it.
337              s!("it is outside any figure or table, so there is no image to describe")
338            };
339            Warn!("unexpected", "\\Description",
340              &s!("attached to the enclosing <{host}> as aria:describedby, because \
341                   {why}. The text is still announced, but it is not any image's \
342                   alt text; put a \\Description after the \\includegraphics it \
343                   describes, or use \\includegraphics[alt=...] per image"));
344          }
345        },
346      }
347    }
348  );

448  //
449  // teaserfigure — real acmart DEFERS the teaser: `\newenvironment{teaserfigure}
450  // {\Collect@Body\@saveteaser}{}` (cls L2202) stashes the body into
451  // `\@teaserfigures`, and `\maketitle` renders it via `\@mkteasers` (cls L2240,
452  // L2899) as the last part of the top-matter box — so in the PDF the teaser
453  // appears AFTER the title+authors and BEFORE the abstract, not where the
454  // environment is written (typically before `\maketitle`). Perl LaTeXML has no
455  // teaserfigure binding at all, so any handling here is beyond-Perl.
456  //
457  // We DIGEST + CONSTRUCT the teaser in place (below) so `\label`/`\ref`,
458  // `\caption` numbering, `xml:id` and `inlist` all resolve exactly as for a
459  // normal float — the teaser in arXiv 2606.22880 is `\ref`-ed 6+ times, so the
460  // figure node must own the label. The prior inline result merely landed in the
461  // wrong PLACE (first `<document>` child, ahead of the title, because the env is
462  // written before `\maketitle`). The `\lx@relocate@teaser` DOCUMENT_REWRITE rule
463  // registered just below moves the finished node to just before the abstract
464  // (user-directed 2026-07-13 — timing/position of `\maketitle`, no frontmatter
465  // markup added over the figure).
466  DefEnvironment!("{teaserfigure}[]",
467    "<ltx:figure xml:id='#id' inlist='#inlist' class='ltx_teaserfigure' ?#1(placement='#1')>#tags#body</ltx:figure>",
468    before_digest => { before_float("figure", None); },
469    after_digest => sub[whatsit] { after_float(whatsit); },
470    mode => "internal_vertical"
471  );
472  // Relocate the constructed teaser to acmart's top-matter position: immediately
473  // before the abstract (PDF order: title, authors, teaser, abstract). Runs as a
474  // post-construction DOM rewrite so the figure keeps its id/label/number — only
475  // its position changes.
476  //
477  // We anchor on the ABSTRACT, not the teaser: the rewrite `replace` engine
478  // unbinds the matched node AND every FOLLOWING sibling (re-appending them after
479  // the closure), so matching the teaser — the first `<document>` child — would
480  // detach the whole frontmatter and leave the teaser stuck at the front. The
481  // abstract's PRECEDING siblings (the teaser, title, creators) stay bound, so we
482  // move the still-bound teaser to just before the re-attached abstract. The
483  // xpath predicate gates the rule to teaser-bearing documents only, so a plain
484  // acmart abstract is untouched.
485  DefRewrite!(
486    xpath => "//ltx:abstract[//ltx:figure[contains(@class,'ltx_teaserfigure')]]",
487    replace => sub[document, nodes] {
488      let abs = nodes.pop().unwrap();
489      // Re-attach the abstract at its original parent (the rewrite engine set the
490      // current node to that parent before calling us).
491      document.get_node_mut().add_child(abs)?;
492      // Move the still-bound teaser to just before the abstract.
493      let teasers =
494        document.findnodes("//ltx:figure[contains(@class,'ltx_teaserfigure')]", None);
495      if let Some(mut teaser) = teasers.into_iter().next() {
496        abs.add_prev_sibling(&mut teaser)?;
497      }
498    });

503    after_digest => sub[whatsit] { after_float(whatsit); },
504    mode => "internal_vertical"
