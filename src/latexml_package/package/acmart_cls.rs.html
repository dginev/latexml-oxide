137  RawTeX!(r"\def\@received{}\newcommand\lx@acm@received@history[2][]{\def\@tempa{#1}\ifx\@tempa\@empty
138  \ifx\@received\@empty\gdef\@received{Received #2}\else\g@addto@macro{\@received}{; revised #2}\fi

140  DefMacro!("\\received[]{}", sub[(label, date)] {
141    let mut history = vec![T_CS!("\\lx@acm@received@history")];
142    if let Some(label) = &label {
143      history.push(T_OTHER!("["));
144      history.extend(label.unlist_ref().iter().copied());
145      history.push(T_OTHER!("]"));
146    }
147    history.push(T_BEGIN!());
148    history.extend(date.unlist_ref().iter().copied());
149    history.push(T_END!());
150    let first = lookup_value("lx_acm_received_given").is_none();
151    assign_value("lx_acm_received_given", Stored::Bool(true), Some(Scope::Global));
152    let label = label.filter(|label| !label.is_empty());
153    let mut out = match (&label, first) {
154      (None, true) => TokenizeInternal!("\\lx@add@date[role=received]{").unlist(),
155      (None, false) => TokenizeInternal!(
156        "\\lx@add@frontmatter{ltx:date}[role=revised,name={revised~},accumulate=true]{"
157      )
158      .unlist(),
159      (Some(label), _) => {
160        // the role the label names: its words, lowercased and hyphenated (`accepted`, `published-online` for
161        // `Published~online`)
162        let words: String = label
163          .unlist_ref()
164          .iter()
165          .map(|t| match t.get_catcode() {
166            Catcode::LETTER | Catcode::OTHER => t.with_str(str::to_lowercase),
167            Catcode::SPACE => " ".to_string(),
168            Catcode::ACTIVE if t.with_str(|s| s == "~") => " ".to_string(),
169            Catcode::CS if *t == T_CS!("\\ ") => " ".to_string(),
170            _ => String::new(),
171          })
172          .collect();
173        let role = words
174          .split(|c: char| !c.is_alphanumeric())
175          .filter(|w| !w.is_empty())
176          .collect::<Vec<_>>()
177          .join("-");
178        let role = if role.is_empty() { "received".to_string() } else { role };
179        let mut out = mouth::tokenize_internal(TeXString::assembled(s!(
180          "\\lx@add@frontmatter{{ltx:date}}[role={role},accumulate=true,name={{"
181        )))
182        .unlist();
183        out.extend(label.unlist_ref().iter().copied());
184        out.extend(TokenizeInternal!("~}]{").unlist());
185        out
186      },
187    };
188    out.extend(date.unlist());
189    out.push(T_END!());
190    history.extend(out);
191    Ok(Tokens::new(history))

395  // Relocate the constructed teaser to acmart's top-matter position: immediately
396  // before the abstract (PDF order: title, authors, teaser, abstract). Runs as a
397  // post-construction DOM rewrite so the figure keeps its id/label/number — only
398  // its position changes.
399  //
400  // We anchor on the ABSTRACT, not the teaser: the rewrite `replace` engine
401  // unbinds the matched node AND every FOLLOWING sibling (re-appending them after
402  // the closure), so matching the teaser — the first `<document>` child — would
403  // detach the whole frontmatter and leave the teaser stuck at the front. The
404  // abstract's PRECEDING siblings (the teaser, title, creators) stay bound, so we
405  // move the still-bound teaser to just before the re-attached abstract. The
406  // xpath predicate gates the rule to teaser-bearing documents only, so a plain
407  // acmart abstract is untouched.
408  DefRewrite!(
409    xpath => "//ltx:abstract[//ltx:figure[contains(@class,'ltx_teaserfigure')]]",
410    replace => sub[document, nodes] {
411      let abs = nodes.pop().unwrap();
412      // Re-attach the abstract at its original parent (the rewrite engine set the
413      // current node to that parent before calling us).
414      document.get_node_mut().add_child(abs)?;
415      // Move the still-bound teaser to just before the abstract.
416      let teasers =
417        document.findnodes("//ltx:figure[contains(@class,'ltx_teaserfigure')]", None);
418      if let Some(mut teaser) = teasers.into_iter().next() {
419        abs.add_prev_sibling(&mut teaser)?;
420      }
421    });

1042/// A comma between names in an acmart `\author` argument: outside groups and math, or inside the one group that wraps
1043/// the whole argument (`{\bf A, B}`, `\textbf{A, B}`), as the author-line parse reads a wrapped line.
1044fn separates_names(tokens: &[Token]) -> bool {
1045  let start = tokens
1046    .iter()
1047    .position(|t| *t != T_SPACE!())
1048    .unwrap_or(tokens.len());
1049  let end = tokens
1050    .iter()
1051    .rposition(|t| *t != T_SPACE!())
1052    .map_or(start, |i| i + 1);
1053  let mut open = start
1054    + usize::from(
1055      tokens
1056        .get(start)
1057        .is_some_and(|t| t.get_catcode() == Catcode::CS),
1058    );
1059  let mut wrapped = false;
1060  if tokens
1061    .get(open)
1062    .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
1063  {
1064    let mut depth = 0usize;
1065    for (i, t) in tokens.iter().enumerate().take(end).skip(open) {
1066      match t.get_catcode() {
1067        Catcode::BEGIN => depth += 1,
1068        Catcode::END => depth = depth.saturating_sub(1),
1069        _ => {},
1070      }
1071      if depth == 0 {
1072        wrapped = i + 1 == end;
1073        break;
1074      }
1075    }
1076  }
1077  if !wrapped {
1078    open = start;
1079  }
1080  let (mut depth, mut math) = (0usize, false);
1081  let level = usize::from(wrapped);
1082  tokens[open..end].iter().any(|t| {
1083    match t.get_catcode() {
1084      Catcode::BEGIN => depth += 1,
1085      Catcode::END => depth = depth.saturating_sub(1),
1086      Catcode::MATH => math = !math,
1087      _ => {},
1088    }
1089    depth == level && !math && *t == T_OTHER!(",")
1090  })
1091}
