1441/// caption3.sty:1713-1748 `\caption@ifempty`: a caption text is empty when nothing is left once its spaces, a leading
1442/// `\ignorespaces` and its `\label{…}`, `\index{…}` and `\glossary{…}` are taken away (`\caption{\label{f3}}`,
1443/// pdflatex "Figure 3" with no separator); a group, even an empty one, is text.
1444fn caption_text_is_empty(text: &Tokens) -> bool {
1445  let tokens = text.unlist_ref();
1446  let mut i = 0;
1447  let skip_spaces = |i: &mut usize| {
1448    while tokens
1449      .get(*i)
1450      .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
1451    {
1452      *i += 1;
1453    }
1454  };
1455  loop {
1456    skip_spaces(&mut i);
1457    let Some(token) = tokens.get(i) else {
1458      return true;
1459    };
1460    i += 1;
1461    if *token == T_CS!("\\ignorespaces") {
1462      continue;
1463    }
1464    if *token == T_CS!("\\label")
1465      || *token == T_CS!("\\lx@caption@postlabel")
1466      || *token == T_CS!("\\lx@label@type")
1467      || *token == T_CS!("\\index")
1468      || *token == T_CS!("\\glossary")
1469    {
1470      // `\@gobbletwo`: the command and its argument, a group or one token.
1471      skip_spaces(&mut i);
1472      if tokens
1473        .get(i)
1474        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
1475      {
1476        let mut depth = 0;
1477        while let Some(t) = tokens.get(i) {
1478          i += 1;
1479          match t.get_catcode() {
1480            Catcode::BEGIN => depth += 1,
1481            Catcode::END => {
1482              depth -= 1;
1483              if depth == 0 {
1484                break;
1485              }
1486            },
1487            _ => {},
1488          }
1489        }
1490      } else {
1491        i += 1;
1492      }
1493      continue;
1494    }
1495    return false;
1496  }
1497}

1499/// A float's caption as caption formats it (caption3.sty:1661-1715), from its settings for the type: the type's name
1500/// (`name=`), its label format (`labelformat=`, the type's `\fnum@<type>` for `original`/`default`), its label
1501/// separator (`labelsep=`, none for an empty caption, :1680-1683), its fonts (`font=` for the whole caption,
1502/// `labelfont=` for the label, `textfont=` for the text) and its text format (`textformat=`). With none set the
1503/// caption is the class's (`\lx@format@title@@`), whose `\@makecaption` caption has replaced by its own
1504/// (`\lx@caption@makecaption`) — but a float.sty float's, which float.sty makes itself and caption makes instead
1505/// (caption.sty:856-868), takes caption's default separator for the class (`CAPTION_DEFAULT_LABELSEP`: acmart's journal
1506/// formats "Program 1. …", review rev78s2 finding 5), except after caption's kernel alone (`\lx@caption@kernel@only`:
1507/// floatrow, subfig's `caption=false`), which formats nothing. Sub-captions keep their packages' formats
1508/// (subcaption_sty.rs, subfig_sty.rs).
1509fn caption_format_tokens(ty: Tokens, text: Tokens) -> Result<Tokens> {
1510  let ty_name = ty.to_string().trim().to_string();
1511  let text_empty = caption_text_is_empty(&text);
1512  // A `\captionof` panel a binding hosts (`\lx@caption@hosted`, `\@captionof@`) is caption's `\caption` of that
1513  // type, so caption formats it as any other, after caption's kernel alone too: its default separator (caption3.sty:787
1514  // `\SetCaptionDefault{labelseparator}{colon}`), its `labelformat=` (`empty`: no label, 2605.30420) and its own fonts
1515  // in place of the sub-float package's (1608.05355, 1806.00503).
1516  let hosted = lookup_bool("lx@caption@hosted");
1517  let get = |key: &str| caption_setting(&ty_name, key);
1518  let format = get("labelformat").filter(|f| f != "default" && f != "original");
1519  let float_sty_caption =
1520    lookup_bool(&s!("float@style@{ty_name}")) && !is_defined("\\lx@caption@kernel@only");
1521  let separator = get("labelsep").filter(|s| s != "default").or_else(|| {
1522    (hosted || float_sty_caption).then(|| {
1523      let default = lookup_string(CAPTION_DEFAULT_LABELSEP);
1524      if default.is_empty() {
1525        "colon".to_string()
1526      } else {
1527        default
1528      }
1529    })
1530  });
1531  let (font, label_font, text_font) = (get("font"), get("labelfont"), get("textfont"));
1532  // caption3.sty:1680-1683: an empty text also takes the `simple` text format (`textformat=period` adds no ".").
1533  let text_format = get("textformat").filter(|f| f != "simple" && f != "default" && !text_empty);
1534  let name = lookup_tokens(&s!("CAPTION_{ty_name}_name@tokens"));
1535  let mut tokens = Vec::new();
1536  let mut line_break: Vec<Token> = Vec::new();
1537  // No type (a caption whose type an argument scan lost) has nothing to format, nor a type whose package makes its
1538  // captions itself (`lx@caption@own@<type>`: algorithm2e's `algorithm`, algorithm2e_sty.rs) — unless float.sty made
1539  // the type, whose captions caption makes (algorithm.sty's `\newfloat{algorithm}` beside `[algo2e]{algorithm2e}`:
1540  // "Algorithm 1 Leakage Removal", 2204.02306, 1812.03871, 2408.08621).
1541  let passthrough = ty_name.is_empty()
1542    || (lookup_bool(&s!("lx@caption@own@{ty_name}")) && !float_sty_caption)
1543    || (ty_name.starts_with("sub") && !hosted)
1544    || (format.is_none()
1545      && separator.is_none()
1546      && font.is_none()
1547      && label_font.is_none()
1548      && text_font.is_none()
1549      && text_format.is_none()
1550      && name.is_none()
1551      && !text_empty);
1552  let explode = |s: &str| ExplodeText!(s);
1553  let braced = |tokens: &mut Vec<Token>, inner: Vec<Token>| {
1554    tokens.push(T_BEGIN!());
1555    tokens.extend(inner);
1556    tokens.push(T_END!());
1557  };
1558  if !passthrough {
1559    tokens.push(T_CS!("\\begingroup"));
1560    // caption3.sty:1686 `\caption@applyfont`: caption's fonts, not subfigure.sty's `\subcapsize\subcaplabelfont` — nor
1561    // float.sty's `\@fs@cfont` (`\rmfamily` for the plain styles, float_sty.rs `\fnum@font@<type>`), as caption makes a
1562    // float.sty caption with its own `\caption@@make` (caption.sty:860-861): acmart sigchi-a's "Program 1" in the sans
1563    // body, not serif (review rev78y finding 5).
1564    if hosted || float_sty_caption {
1565      for font_cs in [s!("fnum@font@{ty_name}"), s!("format@title@font@{ty_name}")] {
1566        tokens.extend([T_CS!("\\let"), T_CS!(&s!("\\{font_cs}")), T_CS!("\\@empty")]);
1567      }
1568    }
1569    if let Some(name) = name {
1570      tokens.push(T_CS!("\\lx@caption@setname"));
1571      braced(&mut tokens, explode(&ty_name));
1572      braced(&mut tokens, name.unlist());
1573    }
1574    let font_tokens = font
1575      .as_deref()
1576      .map(caption_font_tokens)
1577      .transpose()?
1578      .unwrap_or_default();
1579    if font.is_some() || label_font.is_some() {
1580      let mut fonts = font_tokens.clone();
1581      fonts.extend(
1582        label_font
1583          .as_deref()
1584          .map(caption_font_tokens)
1585          .transpose()?
1586          .unwrap_or_default(),
1587      );
1588      tokens.push(T_CS!("\\lx@caption@addfont"));
1589      braced(&mut tokens, explode(&s!("fnum@font@{ty_name}")));
1590      braced(&mut tokens, fonts);
1591    }
1592    if font.is_some() || text_font.is_some() {
1593      let mut fonts = font_tokens;
1594      fonts.extend(
1595        text_font
1596          .as_deref()
1597          .map(caption_font_tokens)
1598          .transpose()?
1599          .unwrap_or_default(),
1600      );
1601      tokens.push(T_CS!("\\lx@caption@addfont"));
1602      braced(&mut tokens, explode(&s!("format@title@font@{ty_name}")));
1603      braced(&mut tokens, fonts);
1604    }
1605    if let Some(format) = format {
1606      if is_defined(&s!("\\caption@labelformat@{format}")) {
1607        tokens.push(T_CS!("\\lx@caption@fnum"));
1608        braced(&mut tokens, explode(&format));
1609        braced(&mut tokens, explode(&ty_name));
1610      } else {
1611        Error!(
1612          "undefined",
1613          &format,
1614          &s!("Undefined label format `{format}'")
1615        );
1616      }
1617    }
1618    // caption3.sty:1680-1683: an empty caption text (`caption_text_is_empty`) has no separator. A separator that
1619    // breaks the line (`\caption@LabelSeparatorWithNewline`, caption3.sty:769, :782: `newline`) is the text's first
1620    // line break, the tag's `close` being text.
1621    let separator_cs = if text_empty {
1622      Some(Vec::new())
1623    } else if let Some(separator) = separator {
1624      if is_defined(&s!("\\caption@lsep@{separator}")) {
1625        let crlist = do_expand(T_CS!("\\caption@labelsep@crlist"))?.to_string();
1626        if crlist.split(',').any(|item| item.trim() == separator) {
1627          line_break = vec![T_CS!(&s!("\\caption@lsep@{separator}"))];
1628          Some(Vec::new())
1629        } else {
1630          Some(vec![T_CS!(&s!("\\caption@lsep@{separator}"))])
1631        }
1632      } else {
1633        Error!(
1634          "undefined",
1635          &separator,
1636          &s!("Undefined label separator `{separator}'")
1637        );
1638        None
1639      }
1640    } else {
1641      None
1642    };
1643    if let Some(separator_cs) = separator_cs {
1644      tokens.push(T_CS!("\\lx@caption@settitle"));
1645      braced(&mut tokens, explode(&ty_name));
1646      braced(&mut tokens, separator_cs);
1647    }
1648  }
1649  let text = caption_text_format(text, text_format.filter(|_| !passthrough))?;
1650  tokens.push(T_CS!("\\lx@format@title@@"));
1651  braced(&mut tokens, ty.unlist());
1652  line_break.extend(text.unlist());
1653  braced(&mut tokens, line_break);
1654  if !passthrough {
1655    tokens.push(T_CS!("\\endgroup"));
1656  }
1657  Ok(Tokens::new(tokens))
1658}
