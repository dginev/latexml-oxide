8/// DEP-NEW (2026-05-19): data-drive helper for the ~94 `\<cs>{}{}{}`
9/// journal-abbreviation macros that all expand to
10/// `\@spires{<CODE>\%2C#1\%2C#3}{{\it <Name>}{\bf #1} (#2) #3}`.
11/// Body text differs per journal but the prototype and shape are
12/// identical. Replacing the 94 `DefMacro!` macro arms with this
13/// runtime helper collapses the repeated macro expansion to a
14/// single inlined function body.
15/// Per [[wisdom_data_drive_min_call_sites]]: 94 ≫ 5 threshold.
16fn def_jhep_journal(cs: &str, body: &str) -> Result<()> {
17  let proto = format!("{cs}{{}}{{}}{{}}");
18  let (cs_tok, params) = parse_prototype(&proto, true)?;
19  let body_toks = mouth::tokenize_internal(TeXString::assembled(body.to_string()));
20  def_macro(cs_tok, params, ExpansionBody::Tokens(body_toks), None)?;
21  Ok(())
22}

301  // Perl JHEP.cls.ltxml L133-136 — JHEP redefines `\href` as a 2-arg
302  // `Semiverbatim Semiverbatim` Constructor. The crucial difference from
303  // hyperref.sty's `\href HyperVerbatim {}` is that the SECOND arg is
304  // also `Semiverbatim`: catcode-neutralized so `^` / `_` in the body
305  // become OTHER tokens and do NOT fire script_handler when digested in
306  // math mode.
307  //
308  // Why this matters: JHEP defines journal-citation macros like
309  //   \am{}{}{} → \@spires{ANMAA\%2C#1\%2C#3}{...#3}
310  // where `\@spires{URL}{BODY}` expands to `\href{URL}{BODY}`. Papers
311  // call them inside math (`\beq … \am\mgr M^2S …`), grabbing `^` as
312  // the third arg. The `^` then ends up at the end of `\href`'s body
313  // — and `\href`'s body MUST treat it as Semiverbatim text, not as
314  // a SUPER catcode token that would fire script_handler. The earlier
315  // Rust port omitted this override, so `\href` stayed bound to
316  // hyperref's `HyperVerbatim {}` form and the trailing `^` errored.
317  //
318  // Witness: arXiv:2602.22473 (Pallis et al.) line 1019
319  //   \beq … -2\ld\am\mgr
320  //   M^2S, … \eeq

326      let url = args.first().and_then(|a| a.as_ref()).map(|t| t.to_string()).unwrap_or_default();
