279    let cs = T_CS!(cs);
280    let params = if params.is_empty() {
281      None
282    } else {
283      parse_parameters(params, &cs, true)?
284    };
285    def_macro(
286      cs,
287      params,
288      mouth::tokenize_internal(TeXString::assembled(body)),
289      None,
290    )?;
291    Ok(())
292  };
293  define(s!("\\the{sub}"), "", s!("\\alph{{{sub}}}"))?;

299  define(s!("\\p@{sub}"), "", s!("\\the{name}"))?;

306  let builder = s!("\\lx@subfloat@{name}@");

364/// `\subfloat` with an optional here).
365fn subfloat_tokens(
366  builder: &str,
367  list: Option<Tokens>,
368  caption: Option<Tokens>,
369  body: Tokens,
370) -> Tokens {
371  let mut tokens = vec![T_CS!(builder), T_BEGIN!(), T_CS!("\\lx@caption")];
372  let list = list.unwrap_or_default();
373  match caption {
374    Some(caption) => {
375      tokens.push(T_OTHER!("["));
376      tokens.extend(list.unlist());
377      tokens.extend([T_OTHER!("]"), T_BEGIN!()]);
378      tokens.extend(caption.unlist());
379    },
380    None => {
381      tokens.push(T_BEGIN!());
382      tokens.extend(list.unlist());
383    },
384  }
385  tokens.extend([T_END!(), T_END!(), T_BEGIN!()]);
386  tokens.extend(body.unlist());
387  tokens.push(T_END!());
388  Tokens::new(tokens)
389}
