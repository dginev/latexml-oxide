1//! tcblistingscore.code.tex — tcolorbox `listings@core` library (loaded by the
2//! `listings`, `listingsutf8` and `minted` libraries: tcblistings.code.tex:25,
3//! tcbminted.code.tex:30).
4//!
5//! The raw library loads as-is; only the listing-box DEFINITION family is then
6//! replaced (each entry documents why). Installing our definitions AFTER the raw
7//! load is what a real LaTeX run's order is: tcblistingscore.code.tex:318-378
8//! declares the family with `\NewDocumentCommand`, whose ltcmd
9//! `command-already-defined` check is an `\errmessage` (counted since batch
10//! 56g) — pre-defining them in tcolorbox_sty.rs raised seven of those on every
11//! `\tcbuselibrary{listings}`. Witnesses: codebox-doc-en, postit, keytheorems-doc.
12//! Guard: `perfect_kernel_batch56::tcbuselibrary_listings_installs_the_family_once`.

33  // \newtcblisting[init-options]{name}[N][default]{tcb-options} — a code-listing
34  // box. What matters for the logical output is the code BODY, captured verbatim
35  // and CLOSED at \end{name}, and the box's title (`tcb_store_pending_title`, the
36  // listing's title: OD #768). The raw library's body capture once ran past its
37  // \end{name} and swallowed following content; since the virtual-file round trip
38  // (`\tcbverbatimwrite` → `listings_read_raw_file`) tcolorbox's own definition
39  // (tcblistingscore.code.tex:301-322, `\newenvironment{name}…{\tcblisting{#6,
40  // options@for=name}}{\endtcblisting}`) closes too, but it draws the box as a
41  // picture whose `label=` lands on the nearest labelled ancestor — a section, else
42  // <document> (raw tcolorbox; SYNC silent-loss row "tcolorbox box label") — and the xparse family
43  // loses its begin-line reading (77m measurement, OD #768). So: delegate to
44  // listings' \lstnewenvironment (same name/[N][default] shape; the leading
45  // [init-options] and trailing tcb options feed `tcb_listing_startend`).
46  // The signature MUST include the leading `[init-options]` optional (real
47  // tcolorbox `\NewDocumentCommand \__tcobox_new_tcolorbox:w { m +O{} m o +o +m }`).
48  DefMacro!("\\newtcblisting[]{}[][]{}", sub[(init, name, n, default, opts)] {
49    let env_name = name.to_string().trim().to_string();
50    let (start, end) = tcb_listing_startend(&env_name, init.as_ref(), &opts);
51    // `[N]` alone = N MANDATORY arguments; only a present `[default]` makes the
52    // first one optional (tcblistingscore.code.tex:318-323 → `\newenvironment`
53    // shape). Emitting an empty `[]` default turned `\newtcblisting{DemoCode}[1]`
54    // into an optional-argument environment, so `\begin{DemoCode}{listing only}`
55    // read `#1` as empty and executed the displayed preamble code (sweep 39:
56    // calculatoritems, mathador, randintlist, tikz-decofonts, tikzbrickfigurines,
57    // OutilsGeomTikz, commalists-tools ×2, tikz2d-fr, tutodoc ×2 regressed).
58    // Guard: `perfect_kernel_batch56::newtcblisting_mandatory_argument_stays_mandatory`.
59    let mut signature = String::new();
60    if let Some(n) = n {
61      signature.push_str(&format!("[{}]", n.to_string().trim()));
62      if let Some(default) = default {
63        signature.push_str(&format!("[{default}]"));
64      }
65    }
66    Ok(Tokenize!(TeXString::assembled(format!(
67      "\\lstnewenvironment{{{env_name}}}{signature}{{{start}}}{{{end}}}"
68    ))))
69  }, locked => true);

71  // \NewTCBListing / \DeclareTCBListing / \RenewTCBListing / \ProvideTCBListing
72  // [init-options]{name}{xparse-sig}{options}: the xparse-signature flavor of
73  // \newtcblisting (tcbxparse library; tcblistingscore.code.tex:329-355 —
74  // `\__tcobox_new_TCBListing:w { m +O{} >{\TrimSpaces} m +m +m }`, so the
75  // user shape carries a LEADING optional, same as the plain `\newtcblisting`
76  // fix above). Map the signature to a \lstnewenvironment shape (mandatory
77  // count + one leading optional, see `tcb_xparse_listing`) and delegate like
78  // the plain form.
79  // Witnesses: keytheorems-doc L165 `\NewTCBListing{keythmscode}{ !O{} }{…}`
80  // (31 uses, no leading optional); atableau.tex L655
81  // `\NewTCBListing[use counter=example, …]{example}{ O{} s m }{…#1…#3…}`
82  // (leading optional — without absorbing it the three `{}` args grab `[`,
83  // `u`, `s` and the options body digests raw: misdefined:# storm).
84  // Known residual: the `s` star specifier is not expressible via
85  // \lstnewenvironment, so a starred `\begin{example}*` call mis-grabs `*`.
86  DefMacro!("\\NewTCBListing[]{}{}{}", sub[(init, name, sig, opts)] {
87    tcb_xparse_listing(name, sig, init.as_ref(), &opts)
88  });
89  DefMacro!("\\DeclareTCBListing[]{}{}{}", sub[(init, name, sig, opts)] {
90    tcb_xparse_listing(name, sig, init.as_ref(), &opts)
91  });
92  DefMacro!("\\RenewTCBListing[]{}{}{}", sub[(init, name, sig, opts)] {
93    tcb_xparse_listing(name, sig, init.as_ref(), &opts)
94  });
95  DefMacro!("\\ProvideTCBListing[]{}{}{}", sub[(init, name, sig, opts)] {
96    tcb_xparse_listing(name, sig, init.as_ref(), &opts)

99  // \newtcbinputlisting[init]{\cmd}[n][default]{options} — defines \cmd to
100  // INPUT a listing file into a listing box at each use
101  // (tcblistingscore.code.tex L355-378: \cmd = \tcbinputlisting{options}).
102  // Semantic core honored: `use counter from=<env>` steps the counter
103  // recorded for <env> (see tcb_listing_startend) and exposes
104  // \thetcbcounter; `listing file=F` — with the call's #-arguments
105  // substituted and expanded — is read (VFS first, then disk) and displayed
106  // as a listing. Witness: incgraph-doc.sty L128 `\newtcbinputlisting[use
107  // counter from=texexptitled]{\inputexamplelisting}[3][]{…listing
108  // file={#2}…}`.
109  // The options are a macro ARGUMENT in TeX (`\NewDocumentCommand`, L379),
110  // not a definition body, so they are read as `{}` and packed here (see
111  // `\lstnewenvironment`, listings_sty.rs).
112  DefPrimitive!("\\newtcbinputlisting []{}[Number][] {}", sub[(init, cmd, n, default, opts)] {
113    let opts = opts.pack_parameters()?;
114    let cmd_tok = cmd
115      .unlist_ref()
116      .iter()
117      .find(|t| t.code == Catcode::CS)
118      .copied();
119    let Some(cmd_tok) = cmd_tok else {
120      return Ok(Vec::new());
121    };
122    // The init options go through tcolorbox's own init processor under the command's name without
123    // its backslash (tcblistingscore.code.tex:374-377 `\__tcobox_set_backslash_removed:Nn`, then
124    // :359 `\__tcobox_process_newtcolorbox:nn`), as for a listing environment
125    // (`tcb_listing_startend`): `use counter=example` gives each use `\thetcbcounter` (the
126    // `\examplefile` template of 1912.06298, 2405.01480 …: `title={Example~\thetcbcounter: #2}`
127    // was "undefined:\thetcbcounter" once the title was typeset).
128    let box_name = cmd_tok.with_str(|s| s.trim_start_matches('\\').to_string());
129    let init_src = init.map(|t| t.to_string()).unwrap_or_default();
130    if !init_src.trim().is_empty() {
131      let _ = digest(mouth::tokenize_internal(TeXString::assembled(format!(
132        "\\tcb@proc@options@init{{{init_src}}}{{{box_name}}}"
133      ))));
134    }
135    let n: usize = n.value_of() as usize;
136    let mut param_spec = String::new();
137    if let Some(ref d) = default {
138      param_spec.push_str(&format!("[Default:{d}]"));
139      for _ in 1..n {
140        param_spec.push_str("{}");
141      }
142    } else {
143      for _ in 0..n {
144        param_spec.push_str("{}");
145      }
146    }
147    let params = if param_spec.is_empty() {
148      None
149    } else {
150      parse_parameters(&param_spec, &cmd_tok, true)?
151    };
152    // Same shape as \lstinputlisting (listings_sty.rs): a MACRO whose
153    // expansion opens the listing group (`bgroup()`) and yields the display
154    // tokens; the display's own trailer closes it. A primitive that unread
155    // the display, or a bare expansion without the bgroup, tripped
156    // "close a group that switched to mode internal_vertical".
157    let expansion: Option<ExpansionBody> = Some(ExpansionBody::Closure(Rc::new(
158      move |args: Vec<ArgWrap>| {
159              let sub_args: Vec<Option<Cow<Tokens>>> = args
160          .iter()
161          .map(|a| match a {
162            ArgWrap::None => None,
163            ArgWrap::Tokens(t) => Some(Cow::Borrowed(t)),
164            ArgWrap::Token(t) => Some(Cow::Owned(Tokens::new(vec![*t]))),
165            other => Some(Cow::Owned(Tokens::new(ExplodeText!(other.to_string())))),
166          })
167          .collect();
168        let opts_subst = opts.substitute_parameters(&sub_args);
169        let mut counter: Option<String> = None;
170        let mut file: Option<String> = None;
171        for (key, val) in split_keyval_source(&opts_subst.to_string()) {
172          let val = val.trim().trim_matches(['{', '}']).trim();
173          match key.trim() {
174            "use counter from" if !val.is_empty() => {
175              if let Some(Stored::String(sym)) =
176                lookup_value(&format!("tcb_env_counter_{val}"))
177              {
178                counter = Some(with(sym, |c| c.to_string()));
179              }
180            },
181            "listing file" if !val.is_empty() => file = Some(val.to_string()),
182            _ => {},
183          }
184        }
185        // (each use steps the box's counter, as `\tcb@proc@counter@autoanduse`'s `code=` does,
186        // tcolorbox.sty:2311)
187        if let Some(counter) = counter.filter(|_| {
188          lookup_meaning(&T_CS!(format!("\\tcb@cnt@{box_name}"))).is_none()
189        }) {
190          digest(Tokenize!(TeXString::assembled(format!(
191            "\\refstepcounter{{{counter}}}\\def\\thetcbcounter{{\\csname the{counter}\\endcsname}}"
192          ))))?;
193        } else {
194          digest(Tokenize!(TeXString::assembled(tcb_counter_step_code(&box_name))))?;
195        }
196        let Some(file) = file else {
197          return Ok(Tokens!());
198        };
199        let file = do_expand(Tokenize!(TeXString::assembled(file)))
200          .map(|t| t.to_string())
201          .unwrap_or_default();
202        let file = file.trim();
203        // Perl reads every listing file through listingsReadRawFile
204        // (`FindFile(…, noltxml => 1)`, listings.sty.ltxml:322-334): the session's
205        // virtual store first (a round-tripped `\jobname.<n>.listing`), then
206        // kpathsea and the source directory. A read relative to the working
207        // directory missed both a tex-tree file (commalists-tools-doc
208        // `\DemoCodeFile{commalists-tools.sty}`) and one beside the document
209        // (csvsimple-legacy `\xmllisting{namesort}`), and the listing came out
210        // empty with no diagnostic.
211        // (read before the box's options are resolved below, so a failed read parks no title,
212        // phantom or mode for an unrelated later listing)
213        let text = listings_read_raw_file(file)?.unwrap_or_default();
214        // The box's options resolved as the box resolves them (`tcb_resolve_listing_mode`, as for a
215        // listing environment): its title (typeset as the listing's — 1912.06298 and the other
216        // `\examplefile` papers: `title={Example~\thetcbcounter: #2}`), its phantom code (a per-use
217        // `label=` / `hypertarget=`, run at the listing's first line: `\examplefile[label={ex:header},
218        // …]` was a dangling `\ref`) and its mode, which `\tcbinputlisting` (tcblistingscore.code.tex
219        // :337-344 → `\tcb@inputlisting`, :163-177 inside the box, :59-160 outside) follows: the default
220        // `listing and text` typesets the file again after the listing (`\tcbuselistingtext` =
221        // `\input{\kvtcb@listingfile}`, :25), `text and listing` before it, `text only` alone
222        // (dropped with 0 errors before). Guards `perfect_kernel_batch77m::{
223        // newtcbinputlisting_keeps_its_title, newtcbinputlisting_keeps_its_label,
224        // newtcbinputlisting_follows_its_mode}`.
225        tcb_resolve_listing_mode(opts_subst)?;
226        let execute = lookup_bool("LISTINGS_EXECUTE_BODY");
227        AssignValue!("LISTINGS_EXECUTE_BODY" => false, Scope::Global);
228        let layout = lookup_string("tcb_pending_layout");
229        // (the text ends its paragraph, as the box it is typeset in does: tcolorbox.sty:360-366
230        // `before`/`after` `\par`)
231        let run_text = if execute {
232          let mut run = lst_run_body_via_input(&box_name, &text)?.unlist();
233          run.push(T_CS!("\\par"));
234          run
235        } else {
236          Vec::new()
237        };
238        if layout == "O" {
239          // No listing: the box holds its title bar and its text. With no listing to caption, the
240          // title is a paragraph before the text; the phantom code has no element of the box to name
241          // (a `label` there would name the enclosing section) and is dropped, as a `text only`
242          // listing environment drops it (`\lxtcbdropphantom`).
243          let title = match lookup_value("tcb_pending_title") {
244            Some(Stored::Tokens(title)) => title,
245            _ => Tokens!(),
246          };
247          AssignValue!("tcb_pending_phantom" => Stored::None, Scope::Global);
248          AssignValue!("tcb_pending_title" => Stored::None, Scope::Global);
249          AssignValue!("tcb_pending_remove_caption" => false, Scope::Global);
250          let mut out = vec![T_CS!("\\begingroup")];
251          if !title.is_empty() {
252            out.extend([T_CS!("\\par"), T_CS!("\\noindent")]);
253            out.extend(title.unlist());
254            out.push(T_CS!("\\par"));
255          }
256          out.extend(run_text);
257          out.push(T_CS!("\\endgroup"));
258          return Ok(Tokens::new(out));
259        }
260        bgroup();
261        // tcolorbox reads the file back `nolol` (tcblistings.code.tex:45, and `\lst@MakeCaption`
262        // gobbled, :31-33 `listing remove caption`): no list-of-listings entry, so the file name
263        // is no float's caption (guard `perfect_kernel_batch77m::newtcbinputlisting_keeps_its_title`).
264        assign_value("LST@nolol", Stored::String(pin("true")), None);
265        let display = lst_process_display(Some(Tokens::new(ExplodeText!(file))), &text);
266        if run_text.is_empty() {
267          return Ok(Tokens::new(display));
268        }
269        // the text inside the box's group, as `\tcbinputlisting`'s `\begingroup…\endgroup` holds it
270        let mut out = Vec::with_capacity(display.len() + run_text.len());
271        if layout == "OL" {
272          out.extend(run_text);
273          out.extend(display);
274        } else {
275          out.extend(display);
276          let closer = out.pop();
277          out.extend(run_text);
278          out.extend(closer);
279        }
280        Ok(Tokens::new(out))
281      },
282    )));
283    def_macro(cmd_tok, params, expansion, None)?;
284  }, locked => true);
285  DefMacro!("\\renewtcbinputlisting", "\\newtcbinputlisting", locked => true);
