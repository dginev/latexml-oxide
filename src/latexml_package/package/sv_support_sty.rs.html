5/// The Springer classes' bold vectors, the two forms their `\vec` takes (KNOWN_PERL_ERRORS #1143): `\lx@sv@vec@mathbf`
6/// is `\mbox{\boldmath$\mathbf{#1}$}` (svjour3.cls:631-636, svjour.cls:546-551 under `\if@mathematic`; svmult.cls:1015-1030
7/// and svproc.cls:389-404 by default), upright bold letters and bold italic Greek ("v + ω", CMBX10 and CMMIB10 in
8/// pdflatex); `\lx@sv@vec@boldmath` is `\mbox{\boldmath$#1$}` (llncs.cls:393-397 under `\if@custvec`, the svjour classes
9/// without `\if@mathematic`, svmult and svproc under `vecphys`), bold italic (CMMIB10). Each reverts as `\vec`; the class
10/// wraps the first in `\ensuremath` where it does. Their `\mathchoice` of four `\mbox`es, one per style, is the
11/// bold font of the current style here. Witnesses 2512.18788, hep-ex0701025 (svmult), 2609.05466 (svjour3).
12pub fn define_bold_vectors() -> Result<()> {
13  DefConstructor!("\\lx@sv@vec@mathbf{}", "#1", bounded => true, require_math => true,
14    font => { series => "bold", family => "serif", shape => "upright", forcebold => true }, alias => "\\vec");
15  DefConstructor!("\\lx@sv@vec@boldmath{}", "#1", bounded => true, require_math => true,
16    font => { forcebold => true }, alias => "\\vec");
17  Ok(())
18}

84  // A labelled piece's text before `\at` is the name it is the institute of (its `fuzzy:` label), and svjour3 prints
85  // it, the address on the line after (svjour3.cls:506 `\def\at{\\\let\and=\iand}`): kept so, the line before the
86  // address, when it prints a word no queued author's name prints ([`queued_author_name_prints`]: "Contact author:",
87  // 1711.01894; "Corresponding author", 2107.13864; a fuller name than the author line's, 1705.01399), which had been
88  // lost (Perl the same: sv_support.sty.ltxml:45-47; OXIDIZED_DESIGN_DIVERGENCES #952) — a piece printing only its
89  // author's name adds nothing the names do not hold.
90  // A list the piece opens (`\institute{\begin{itemize}\item[\Letter] Shiqiang~Du \\ \email{…} \at \item[1] Key
91  // Laboratory … \end{itemize}}`, 2305.05902, 2306.07927, 2312.14898, 2605.09054) is the list's structure — its
92  // `\begin{…}`, `\end{…}` and each `\item[…]` — and the words in it. The keep test reads the line as every name-shape
93  // test does ([`queued_author_name_prints`] through `visible_name_text`, where an environment's name and an item's
94  // label print no word: "itemize" had kept a line printing only a queued name, the rev98b review's pe1); only the
95  // words name the owner (`fuzzy:`; the `\begin` in the label had opened the list in the label's digestion, an
96  // Error); the structure the affiliation holds is kept explicitly: the
97  // list's `\begin` when the address closes it, with the item the address goes on in when the address opens with no
98  // `\item` of its own, and an `\end` the piece did not open is dropped (`\begin{itemize}\item Ann Able \at Univ X
99  // \and \item Bob Baker \at Univ Y \end{itemize}`: each piece its own affiliation, balanced).
100  DefMacro!("\\lx@sv@institute@ Until:\\at Until:\\done", sub[(name, address)] {
101    // (decided as `\institute` is read, the authors before it queued; an environment written whole there is no part of
102    // the line — `\begin{acknowledgements}…\end{acknowledgements} T. Ara\'ujo Lima \at …`, 2008.12867, whose text
103    // would be an `<ltx:acknowledgements>` inside a contact, malformed — and stays lost, as before)
104    let mut line = without_environments(name.unlist_ref());
105    let mut address = address.unlist();
106    let opens = line.iter().any(|t| *t == T_CS!("\\begin"));
107    let closes = closes_an_environment(&address);
108    // (an `\end` left in the line closes nothing the piece opened; a list the address does not close is none the
109    // affiliation holds, nor are its items)
110    let dropped: Vec<Range<usize>> = list_structure(&line)
111      .into_iter()
112      .filter(|(command, _)| *command == T_CS!("\\end") || !(opens && closes))
113      .map(|(_, range)| range)
114      .collect();
115    line = without_ranges(&line, &dropped);
116    if !opens && closes {
117      let ends: Vec<Range<usize>> = unopened_ends(&address);
118      address = without_ranges(&address, &ends);
119    }
120    let structure = list_structure(&line);
121    let ranges: Vec<Range<usize>> = structure.iter().map(|(_, range)| range.clone()).collect();
122    let words = without_ranges(&line, &ranges);
123    let mut named = Vec::new();
124    if !address.is_empty() {
125      if !queued_author_name_prints(&Tokens::new(line.clone())) {
126        // (without the spacing and a line break the piece opens with, the layout's: `\and\\ Name \at …`; and without
127        // its `\email` calls, which the label files — kept here they were filed twice, 2312.14898)
128        named = split_email_calls(&line)
129          .0
130          .into_iter()
131          .skip_while(|t| *t == T_SPACE!() || *t == T_CS!("\\\\"))
132          .collect();
133        while named.last() == Some(&T_SPACE!()) {
134          named.pop();
135        }
136        named.push(T_CS!("\\\\"));
137      } else {
138        for (command, range) in &structure {
139          if *command == T_CS!("\\begin") {
140            named.extend_from_slice(&line[range.clone()]);
141          }
142        }
143        let opens_item = address
144          .iter()
145          .find(|t| **t != T_SPACE!())
146          .is_some_and(|t| *t == T_CS!("\\item"));
147        if !opens_item
148          && let Some((_, range)) = structure
149            .iter()
150            .rev()
151            .find(|(command, _)| *command == T_CS!("\\item"))
152        {
153          named.extend_from_slice(&line[range.clone()]);
154        }
155      }
156    }
157    // (a list whose items spell two or more queued names whole before its one `\at` is a list of its authors, no one's:
158    // its owner label names no one (`fuzzy:` empty), so the affiliation is placed as an orphan is (#159: shared below
159    // the authors), and its `\email` calls, no label's now, are in it, each answered by the author it spells — the
160    // surname reading gave the list to the last name's alone, 2312.14898's Bissyandé. Names a comma or "and" parts,
161    // no list's, are each their own owner as before: `Paulo Rezeck and Hector Azp\'urua \at …`, 2202.12391; 1404.4148,
162    // 2108.10893, 2305.05654)
163    if !address.is_empty()
164      && !structure.is_empty()
165      && queued_author_names_spelled(&Tokens::new(line.clone()))? > 1
166    {
167      named.extend(split_email_calls(&words).1);
168      return Ok(Invocation!(
169        "\\lx@add@affiliation[label={fuzzy:}]{\\lx@sv@institute@marks#1\\lx@sv@institute@piece{#2}}",
170        vec![Some(Tokens::new(named)), Some(Tokens::new(address))]
171      ));
172    }
173    // (the name as written where it holds no list command)
174    let label = if dropped.is_empty() && structure.is_empty() {
175      name.unlist_ref().to_vec()
176    } else {
177      words
178    };
179    // (the label's `\email` calls, `Name (\email{…}) \at Address`, are its owner's e-mails, each filed under the label as
180    // its own call, before the affiliation as the line prints them: the old driver digested the label at the flush, its
181    // `\email` filing the contact, which the emitter, reading a label as text, never does — the address had been lost,
182    // no Warning (2206.00320, 1307.4424; OXIDIZED_DESIGN_DIVERGENCES #985). Not in the label any more, whose words are
183    // the owner's name)
184    let (label, calls) = split_email_calls(&label);
185    let label = Tokens::new(label);
186    let emails = if calls.is_empty() {
187      Vec::new()
188    } else {
189      cs_group_tokens("\\lx@sv@institute@emails", &[label.unlist_ref(), &calls])
190    };
191    Ok(Invocation!(
192      "\\ifx.#2.\\lx@sv@institute@unlabelled{#1}\\else#5\\lx@add@affiliation[label={fuzzy:#4}]{\\lx@sv@institute@marks#3\\lx@sv@institute@piece{#2}}\\fi",
193      vec![
194        Some(name),
195        Some(Tokens::new(address)),
196        Some(Tokens::new(named)),
197        Some(label),
198        Some(Tokens::new(emails))
199      ]
200    ))
201  });

241      let piece = Tokens::new(

246          .collect(),
247      );
248      Ok(Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![labelseq, Some(piece)]))
249    } else {
250      let mut content = vec![T_CS!("\\lx@sv@institute@marks")];
251      content.extend(piece.unlist());
252      Ok(Invocation!(T_CS!("\\lx@add@affiliation"), vec![labelseq, Some(Tokens::new(content))]))
253    }

634/// `tokens` without the environments written whole in them, each `\begin{…}` through its matching `\end{…}` and that
635/// command's argument; an environment opened and not closed there is kept (its `\end` comes after the piece:
636/// `\institute{\begin{itemize}\item[✉] Ann Able \at \item[1] Univ X \end{itemize}}`, 2305.05902, 2306.07927,
637/// 2312.14898, 2605.09054, whose `\begin` kept there balances the list).
638fn without_environments(tokens: &[Token]) -> Vec<Token> {
639  // the index after the `\end{…}` matching the `\begin` at `at`, if the tokens hold it
640  let environment_end = |at: usize| -> Option<usize> {
641    let mut depth = 0usize;
642    let mut i = at;
643    while i < tokens.len() {
644      if tokens[i] == T_CS!("\\begin") {
645        depth += 1;
646      } else if tokens[i] == T_CS!("\\end") {
647        depth -= 1;
648        if depth == 0 {
649          let mut j = i + 1;
650          while tokens.get(j) == Some(&T_SPACE!()) {
651            j += 1;
652          }
653          if tokens
654            .get(j)
655            .is_some_and(|u| u.get_catcode() == Catcode::BEGIN)
656          {
657            let mut level = 0usize;
658            while j < tokens.len() {
659              match tokens[j].get_catcode() {
660                Catcode::BEGIN => level += 1,
661                Catcode::END => {
662                  level -= 1;
663                  if level == 0 {
664                    return Some(j + 1);
665                  }
666                },
667                _ => {},
668              }
669              j += 1;
670            }
671            return None;
672          }
673          return Some(i + 1);
674        }
675      }
676      i += 1;
677    }
678    None
679  };
680  let mut out = Vec::with_capacity(tokens.len());
681  let mut i = 0;
682  while i < tokens.len() {
683    if tokens[i] == T_CS!("\\begin")
684      && let Some(end) = environment_end(i)
685    {
686      i = end;
687      continue;
688    }
689    out.push(tokens[i]);
690    i += 1;
691  }
692  out
693}

695/// The list structure in `tokens`, each command with the tokens it spans: `\begin{…}` and `\end{…}` with their
696/// argument, `\item` with its optional label (`\item [\textsuperscript{1}]`, spaces before it skipped as LaTeX's
697/// `\@ifnextchar` skips them).
698fn list_structure(tokens: &[Token]) -> Vec<(Token, Range<usize>)> {
699  // the index after the spaces from `at`
700  let past_spaces = |mut at: usize| {
701    while tokens.get(at) == Some(&T_SPACE!()) {
702      at += 1;
703    }
704    at
705  };
706  let mut structure = Vec::new();
707  let mut i = 0;
708  while i < tokens.len() {
709    let t = tokens[i];
710    let end = if t == T_CS!("\\begin") || t == T_CS!("\\end") {
711      let at = past_spaces(i + 1);
712      if tokens
713        .get(at)
714        .is_some_and(|u| u.get_catcode() == Catcode::BEGIN)
715      {
716        group_end(tokens, at)
717      } else {
718        i + 1
719      }
720    } else if t == T_CS!("\\item") {
721      let at = past_spaces(i + 1);
722      if tokens.get(at) == Some(&T_OTHER!("[")) {
723        // (the `]` outside braces: `\item[{[a]}]`)
724        let mut depth = 0usize;
725        let mut j = at + 1;
726        while j < tokens.len() && !(depth == 0 && tokens[j] == T_OTHER!("]")) {
727          match tokens[j].get_catcode() {
728            Catcode::BEGIN => depth += 1,
729            Catcode::END => depth = depth.saturating_sub(1),
730            _ => {},
731          }
732          j += 1;
733        }
734        (j + 1).min(tokens.len())
735      } else {
736        i + 1
737      }
738    } else {
739      i += 1;
740      continue;
741    };
742    structure.push((t, i..end));
743    i = end;
744  }
745  structure
746}

748/// The index after the brace group opening at `at`, or the end of `tokens` when it is not closed.
749fn group_end(tokens: &[Token], at: usize) -> usize {
750  let mut level = 0usize;
751  for (j, t) in tokens.iter().enumerate().skip(at) {
752    match t.get_catcode() {
753      Catcode::BEGIN => level += 1,
754      Catcode::END => {
755        level = level.saturating_sub(1);
756        if level == 0 {
757          return j + 1;
758        }
759      },
760      _ => {},
761    }
762  }
763  tokens.len()
764}

766/// The `\end{…}` commands in `tokens` that close no `\begin` of theirs: the environment a piece before opened.
767fn unopened_ends(tokens: &[Token]) -> Vec<Range<usize>> {
768  let mut depth = 0usize;
769  let mut ends = Vec::new();
770  for (command, range) in list_structure(tokens) {
771    if command == T_CS!("\\begin") {
772      depth += 1;
773    } else if command == T_CS!("\\end") {
774      if depth == 0 {
775        ends.push(range);
776      } else {
777        depth -= 1;
778      }
779    }
780  }
781  ends
782}

784/// Whether `tokens` close an environment opened before them (an `\end{…}` no `\begin` of theirs matches).
785fn closes_an_environment(tokens: &[Token]) -> bool { !unopened_ends(tokens).is_empty() }

795/// `cs` followed by each of `args` in braces.
796fn cs_group_tokens(cs: &str, args: &[&[Token]]) -> Vec<Token> {
797  let mut tokens = vec![T_CS!(cs)];
798  for arg in args {
799    tokens.push(T_BEGIN!());
800    tokens.extend_from_slice(arg);
801    tokens.push(T_END!());
802  }
803  tokens

806/// `tokens` without their `\email{…}` calls, and the calls.
807fn split_email_calls(tokens: &[Token]) -> (Vec<Token>, Vec<Token>) {
808  let mut rest = Vec::with_capacity(tokens.len());
809  let mut calls = Vec::new();
810  let mut i = 0;
811  while i < tokens.len() {
812    if tokens[i] == T_CS!("\\email") {
813      let mut at = i + 1;
814      while tokens.get(at) == Some(&T_SPACE!()) {
815        at += 1;
816      }
817      if tokens
818        .get(at)
819        .is_some_and(|u| u.get_catcode() == Catcode::BEGIN)
820      {
821        let end = group_end(tokens, at);
822        calls.extend_from_slice(&tokens[i..end]);
823        i = end;
824        continue;
825      }
826    }
827    rest.push(tokens[i]);
828    i += 1;
829  }
830  (rest, calls)
