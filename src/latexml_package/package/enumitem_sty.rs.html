242/// enumitem.sty:633-652 and :685-700: an enumerate's short label with each mark of `\enit@marklist` (`a` `A` `i` `I`
243/// `1`, and the document's `\SetEnumerateShortLabel`s, :641-652) replaced by its counter command, every occurrence
244/// outside braces (`\enit@replace`, a delimited macro, reads none inside), the marks in their list's order.
245fn enumerate_short_label(template: &Tokens) -> Result<Tokens> {
246  let marklist = match lookup_definition(&T_CS!("\\enit@marklist"))? {
247    Some(defn) => match defn.get_expansion() {
248      Some(ExpansionBody::Tokens(body)) => body.unlist_ref().to_vec(),
249      _ => Vec::new(),
250    },
251    None => Vec::new(),
252  };
253  let elt = T_CS!("\\enit@elt");
254  let mut label = template.unlist_ref().to_vec();
255  let mut rest = marklist.into_iter().peekable();
256  while let Some(t) = rest.next() {
257    if t != elt {
258      continue;
259    }
260    let mut group = || -> Vec<Token> {
261      let mut out = Vec::new();
262      let mut depth = 0i32;
263      for t in rest.by_ref() {
264        match t.get_catcode() {
265          Catcode::BEGIN if depth == 0 => {
266            depth = 1;
267            continue;
268          },
269          Catcode::BEGIN => depth += 1,
270          Catcode::END if depth == 1 => return out,
271          Catcode::END => depth -= 1,
272          _ => {},
273        }
274        out.push(t);
275      }
276      out
277    };
278    let mark = group();
279    let replacement = group();
280    if mark.is_empty() {
281      continue;
282    }
283    let mut replaced = Vec::with_capacity(label.len());
284    let mut depth = 0i32;
285    let mut i = 0;
286    while i < label.len() {
287      let t = label[i];
288      if depth == 0 && label[i..].starts_with(&mark) {
289        replaced.extend_from_slice(&replacement);
290        i += mark.len();
291        continue;
292      }
293      match t.get_catcode() {
294        Catcode::BEGIN => depth += 1,
295        Catcode::END => depth -= 1,
296        _ => {},
297      }
298      replaced.push(t);
299      i += 1;
300    }
301    label = replaced;
302  }
303  Ok(Tokens::new(label))
304}

306/// A `leftmargin` value we can safely place verbatim in a CSS length: a number
307/// followed by a unit CSS shares with TeX (`pt`/`em`/`ex`/`cm`/`mm`/`in`, plus
308/// CSS `px`/`rem`/`%`), or a bare `0`. Returns `None` for anything else — a raw
309/// TeX macro / `\dimexpr` is not valid CSS, so it must not reach the stylesheet.
310fn css_length(raw: &str) -> Option<String> {
311  let s = raw.trim();
312  for unit in ["pt", "px", "em", "ex", "rem", "cm", "mm", "in", "%"] {
313    if let Some(num) = s.strip_suffix(unit)
314      && num.parse::<f64>().is_ok()
315    {
316      return Some(s.to_string());
317    }
318  }
319  // A bare number (e.g. `0`) is a valid CSS length only when it is zero.
320  s.parse::<f64>()
321    .ok()
322    .filter(|n| *n == 0.0)
323    .map(|_| s.to_string())
324}

326/// enumitem.sty:910-936 `\enit@normlabel` (`\lx@enit@normlabel`, defined in TeX below): in a group,
327/// every counter command of the label list (`\enit@labellist`: the kernel's five and each
328/// `\AddEnumerateCounter`) and `\value` read their next argument and put `{<usecounter>}` for a `*`;
329/// the label is then expanded by `\protected@xdef`, braced (enumitem.sty:934, "as \ref is in the
330/// global scope"). So `\roman{*}`, `\csname greek\endcsname*` and a
331/// macro expanding to a counter command all star, and `**` stays text. Perl replaced every `*`
332/// (enumitem.sty.ltxml:114-119). Repros: list-structure/{enumitem_label_star_is_read_by_expansion,
333/// addenumeratecounter_reads_the_star}; witnesses 2605.17001, 2605.06065, moreenum-doc. Enumerate-type
334/// lists only (`enumerate_type` in `begin_enum_itemize`): an itemize label such as a `\tikz` bullet does
335/// not survive `\xdef` (list-structure/itemize_label_is_used_as_written).
336fn normalize_label(label: &Tokens, usecounter: &str) -> Result<Tokens> {
337  // `\lx@enit@label` starts as `\relax`, so a `\protected@xdef` that never assigns (an error
338  // recovery) keeps the label as written rather than the previous list's.
339  let mut toks = vec![
340    T_CS!("\\global"),
341    T_CS!("\\let"),
342    T_CS!("\\lx@enit@label"),
343    T_CS!("\\relax"),
344    T_CS!("\\lx@enit@normlabel"),
345    T_BEGIN!(),
346  ];
347  toks.extend(Explode!(usecounter));
348  toks.push(T_END!());
349  toks.push(T_BEGIN!());
350  toks.extend(label.unlist_ref().iter().copied());
351  toks.push(T_END!());
352  digest(Tokens::new(toks))?;
353  let defn = lookup_definition(&T_CS!("\\lx@enit@label"))?;
354  // An `\xdef` body is balanced in TeX; ours is not only after an error recovery inside the label
355  // (an unprotected `\tikz`, where pdflatex errors too), and an unbalanced `\the<ctr>` would end
356  // the document with `Fatal:Stomach:Misdefined`: keep the label as written then.
357  Ok(match defn.as_ref().and_then(|d| d.get_expansion()) {
358    Some(ExpansionBody::Tokens(body)) if body.is_balanced() => body.clone(),
359    _ => label.clone(),
360  })
361}

409/// Perl: store_enumitem_defaults($name, $kv) — enumitem.sty.ltxml L228-237
410fn store_enumitem_defaults(name: &str, kv: &KeyVals, scope: Option<Scope>) {
411  // Load existing keys directly inside the state/arena closure pair —
412  // the intermediate keys_str String is avoided; we split the interned
413  // &str and collect owned keys straight into the Vec.
414  let mut keys: Vec<String> = with_value(&s!("{name}@keys"), |v| match v {
415    Some(Stored::String(s)) => with(*s, |ks| {
416      ks.split(',')
417        .filter(|k| !k.is_empty())
418        .map(String::from)
419        .collect()
420    }),
421    _ => Vec::new(),
422  });
423
424  for (key, val) in kv.get_pairs() {
425    let val_key = s!("{name}@{key}");
426    match val {
427      ArgWrap::Tokens(t) => {
428        assign_value(&val_key, Stored::Tokens(t.clone()), scope);
429      },
430      ArgWrap::None => {
431        assign_value(&val_key, Stored::None, scope);
432      },
433      _ => {
434        assign_value(&val_key, Stored::String(pin(val.to_string())), scope);
435      },
436    }
437    if !keys.contains(key) {
438      keys.push(key.clone());
439    }
440  }
441  assign_value(
442    &s!("{name}@keys"),
443    Stored::String(pin(keys.join(","))),
444    scope,
445  );
446}

448/// Perl: merged_enumitem_keyvals($name, $level, $argkv) — enumitem.sty.ltxml L239-249
449///
450/// enumitem.sty:977-980 applies the stored keys of `list`, `list<\@listdepth>`, the list's
451/// name, then the name at its level; Perl's took the first and the last two.
452fn merged_enumitem_keyvals(
453  name: &str,
454  level: i64,
455  listdepth: i64,
456  argkv: Option<&KeyVals>,
457) -> rustc_hash::FxHashMap<String, ArgWrap> {
458  let mut hash = rustc_hash::FxHashMap::default();
459
460  let name = enumitem_setlist_name(name);
461  let default_names = [
462    enumitem_defaults_key("list", 0),
463    enumitem_defaults_key("list", listdepth),
464    enumitem_defaults_key(name, 0),
465    enumitem_defaults_key(name, level),
466  ];
467
468  for def_name in &default_names {
469    // with_value pulls the keys-string out of the Stored::String arm
470    // without cloning the envelope; the per-key inner lookup still
471    // needs to produce an owned ArgWrap, so we pay the clone there.
472    // Collect keys as owned Vec<String> inside state+arena closures
473    // so the split happens on the interned &str and the owned
474    // intermediary is smaller (Vec<String> of just the keys, not
475    // the whole comma-separated string plus allocs).
476    let keys: Vec<String> = with_value(&s!("{def_name}@keys"), |v| match v {
477      Some(Stored::String(s)) => with(*s, |ks| {
478        ks.split(',')
479          .filter(|k| !k.is_empty())
480          .map(String::from)
481          .collect()
482      }),
483      _ => Vec::new(),
484    });
485    if keys.is_empty() {
486      continue;
487    }
488    for key in &keys {
489      if !key.is_empty() {
490        let val_key = s!("{def_name}@{key}");
491        if let Some(val) = lookup_value(&val_key) {
492          let aw = match val {
493            Stored::Tokens(t) => ArgWrap::Tokens(t),
494            Stored::Number(n) => ArgWrap::Number(n),
495            Stored::None => ArgWrap::None,
496            Stored::String(s) => ArgWrap::Tokens(with(s, |t| {
497              mouth::tokenize_internal(TeXString::assembled(t.to_string()))
498            })),
499            _ => ArgWrap::None,
500          };
501          hash.insert(key.clone(), aw);
502        }
503      }
504    }
505  }
506
507  // Merge argument keyvals last (highest priority)
508  if let Some(kv) = argkv {
509    for (key, val) in kv.get_pairs() {
510      hash.insert(key.clone(), val.clone());
511    }
512  }
513
514  hash
515}
