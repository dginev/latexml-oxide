1//! tone.sty (tipa's tone letters, FUKUI Rei): the raw file, with its tone letters as Unicode's.
2//!
3//! `\tone{51}` (tone.sty:13-14) draws a contour from tipa's tone-letter parts — slots 128-145, each a segment raised to
4//! its level (`\@outputtone`, :37-58), not a character — and the stem (slot 191) at its right; `\rtone` (:16-17) puts
5//! the stem at the left, `\stone` (:19-20) draws short segments. Unicode writes the contour as its tone letters, one
6//! per level (UAX: Chao tone letters, U+02E5-02E9 with the stem at the right, U+A712-A716 at the left), which fonts
7//! join into the contour: `\tone{51}` ˥˩, `\rtone{15}` ꜖꜒. A level outside 1-5 keeps tone.sty's construction, whose
8//! tone-letter parts have no character, and says so. OXIDIZED_DESIGN_DIVERGENCES #781. Witnesses 2601.19932,
9//! 2602.10003 (`\usepackage[tone]{tipa}`, `\tone{…}`).
10use crate::prelude::*;
11
12/// The tone letters of `levels`, extra-low 1 to extra-high 5, each `stem + 5 - level`; `None` if a level is not one.
13fn tone_letters(levels: &str, stem: u32) -> Option<String> {
14  levels
15    .chars()
16    .filter(|c| !c.is_whitespace())
17    .map(|c| match c.to_digit(10) {
18      Some(level @ 1..=5) => char::from_u32(stem + 5 - level),
19      _ => None,
20    })
21    .collect()
22}
23
24LoadDefinitions!({
25  InputDefinitions!("tone", extension => Some("sty".into()), noltxml => true);
26  // tone.sty read again (`\input{tone.sty}`) keeps the commands saved the first time: saving these wrappers would
27  // make a level outside 1-5 expand to itself.
28  if IsDefined!(&T_CS!("\\lx@tone@raw@tone")) {
29    return Ok(());
30  }
31  for (cs, raw, stem) in [
32    ("\\tone", "\\lx@tone@raw@tone", 0x02E5),
33    ("\\stone", "\\lx@tone@raw@stone", 0x02E5),
34    ("\\rtone", "\\lx@tone@raw@rtone", 0xA712),
35  ] {
36    Let!(T_CS!(raw), T_CS!(cs));
37    DefMacro!(T_CS!(cs), "{}", sub[(levels)] {
38      let tokens = levels.unlist_cow().into_owned();
39      match tone_letters(&Tokens::new(tokens.clone()).to_string(), stem) {
40        Some(letters) if !letters.is_empty() => {
41          let mut out = vec![T_CS!("\\leavevmode")];
42          out.extend(letters.chars().map(|c| Token::new(c.to_string(), Catcode::OTHER)));
43          Tokens::new(out)
44        },
45        _ => {
46          let levels = Tokens::new(tokens.clone()).to_string();
47          if !already_reported(&s!("tone_letter_{cs}{levels}")) {
48            Warn!("missing_font_encoding", "T3",
49              s!("tone.sty's {cs}{{{levels}}} has no Chao tone letter (levels 1-5): its tone-letter parts have no \
50                  character and are dropped"));
51          }
52          let mut out = vec![T_CS!(raw), T_BEGIN!()];
53          out.extend(tokens);
54          out.push(T_END!());
55          Tokens::new(out)
56        },
57      }
58    });
59  }
60});
