1//! pdftexcmds.sty — pdfTeX utility commands
2//! Perl: pdftexcmds.sty.ltxml
3//! Everything is in pdfTeX.pool already; just require iftex.
4use crate::prelude::*;
5
6#[rustfmt::skip]
7LoadDefinitions!({
8  RequirePackage!("iftex");
9  // Real pdftexcmds.sty on a pdfTeX engine simply `\let`s each `\pdf@…`
10  // wrapper to the engine primitive (pdftexcmds.sty: "\let\pdf@strcmp
11  // \pdfstrcmp" etc.). Delegate the same way — the primitives in
12  // `latexml_engine/src/pdftex.rs` are REAL implementations (strcmp,
13  // md5, file size/date, PDF escapes), format-verified against live
14  // pdfTeX. The former hardcoded stubs here were actively wrong:
15  // `\pdf@strcmp{}{}` → "0" made EVERY string comparison "equal", so
16  // e.g. hep-paper.sty's `\ifnum\pdf@strcmp{\hep@bibliography}{false}=0`
17  // skipped loading its whole bibliography layer (`undefined
18  // \printbibliography` across the hep-* manuals, 2026-08-31 corpus).
19  // `\let`, not a macro that re-invokes the primitive BY NAME: a package that
20  // aliases the other way round — annotate-equations.sty:16 `\let\pdfstrcmp
21  // \pdf@strcmp` under its `\ifluatex` branch — would otherwise give
22  // `\pdfstrcmp` the body `\pdfstrcmp{#1}{#2}`, i.e. itself, and every
23  // `\ifnum\pdfstrcmp{#1}{south}=0` then expands without end
24  // (`Fatal:Timeout:TokenLimit`; latex-via-exemplos under the luatex identity,
25  // 300 s, batch 56ey). With the primitive's meaning aliased, re-aliasing is
26  // idempotent exactly as in pdfTeX.
27  Let!("\\pdf@strcmp", "\\pdfstrcmp");
28  Let!("\\pdf@mdfivesum", "\\pdfmdfivesum");
29  DefMacro!("\\pdf@filemdfivesum{}", "\\pdfmdfivesum file {#1}");
30  Let!("\\pdf@filesize", "\\pdffilesize");
31  Let!("\\pdf@filemoddate", "\\pdffilemoddate");
32  Let!("\\pdf@escapehex", "\\pdfescapehex");
33  Let!("\\pdf@unescapehex", "\\pdfunescapehex");
34  Let!("\\pdf@escapestring", "\\pdfescapestring");
35  Let!("\\pdf@escapename", "\\pdfescapename");
36  // \pdf@shellescape reports the shell-escape state; 0 (disabled) is the
37  // CORRECT value here — matches pdflatex run without -shell-escape, and we
38  // never execute \write18. Justified stub, not an approximation.
39  DefMacro!("\\pdf@shellescape", "0");
40  // \pdf@filedump{offset}{len}{file}: raw byte-dump probe (bmpsize sniffs
41  // image headers with it — TL bmpsize.sty L51-53, witnesses 2406.02536,
42  // 2406.03347). Graphics sizing has its own pipeline here; a hex dump of
43  // image bytes carries no document content. Justified noop.
44  def_macro_noop("\\pdf@filedump{}{}{}")?;
45  def_macro_noop("\\pdf@primitive{}")?;
46});
