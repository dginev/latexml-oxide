1use crate::prelude::*;
2
3#[rustfmt::skip]
4LoadDefinitions!({
5  // Perl: overpic.sty.ltxml requires graphicx + epic. epic (only needed for the
6  // `grid` option's `\grid`) has no Rust binding — a benign `missing_file`
7  // warning; the grid option is off by default, so overlay rendering is
8  // unaffected.
9  RequirePackage!("graphicx");
10  RequirePackage!("epic");
11
12  // DIVERGENCE from Perl (OXIDIZED_DESIGN #135): Perl's overpic.sty.ltxml emits an
13  // EMPTY `<ltx:picture tex='...'/>` and relies on the `PictureImages` post-
14  // processor to render the whole thing (graphic + overlays) as one LaTeX image.
15  // Two Rust realities make that produce nothing: `tex=` on `<ltx:picture>` is
16  // suppressed unconditionally (intentional divergence), and the LaTeXImages
17  // renderer is unwired — Rust renders `<ltx:picture>` as inline SVG from its
18  // CHILD elements. So instead we reproduce overpic.sty's OWN construction: box
19  // the graphic to size a `{picture}` to it, place the graphic at the origin, and
20  // let the body's `\put` overlays draw on top. This routes through Rust's
21  // working picture + `<ltx:graphics>` + `\put` machinery (the graphic resolves
22  // via the picture-nested-graphics SVG path, PR #675), so all of the ~37 arXiv
23  // papers that report a missing overpic figure now render.
24  //
25  // Faithful to overpic.sty's `\OVP@picture` + coordinate-mode keys. The default
26  // is `percent` (`\ExecuteOptions{percent}` → `rel=100`): the LARGER of
27  // width/total-height gets 100 coordinate units and `\unitlength = max(w,h)/100`
28  // (so `\put(50,50)` is the centre of a square image). The OVP keys are ported
29  // so a paper that overrides the mode positions its overlays correctly:
30  //   percent / rel=N / permil : `\unitlength = max(w,h)/N`, larger dim spans N.
31  //   abs                      : coordinates are `\unitlength`s (default 1pt or `unit=`).
32  //   unit=<dim>               : set `\unitlength` (used by `abs`).
33  //   grid / tics              : accepted; `grid` needs epic's `\grid` (unbound), so no-op.
34  // Rust measures a boxed `\includegraphics` (`\wd`/`\ht`/`\dp`) as pdfTeX does,
35  // which is what makes this port possible.
36  //
37  // Witnesses (arXiv/html_feedback): 2412.15262 (baseline, no `\put`),
38  // 2510.17772 (percent labels + trim/clip), 2401.13599 (abs/unit), 2409.06474
39  // (nested `\includegraphics` in `\put` + rotate/scale), 2405.00666 (128 envs).
40  RawTeX!(concat!(
41    r"\newsavebox\OVP@box", "\n",
42    // percent/rel mode: unitlength = max(w,h)/scale; the larger dim spans `scale`
43    // units. The `<\@ne` clamp keeps unitlength >= 1sp so a degenerate (missing /
44    // zero-size) image never triggers `Illegal \divide by 0` — common on arXiv.
45    r"\newcommand\OVP@calc@rel{%", "\n",
46    r"  \ifnum\@tempcnta>\@tempcntb ", "\n",
47    r"    \divide\@tempcnta by \OVP@scale \ifnum\@tempcnta<\@ne \@tempcnta=\@ne \fi ", "\n",
48    r"    \unitlength=\@tempcnta sp\relax \@tempcnta=\OVP@scale \divide\@tempcntb by \unitlength ", "\n",
49    r"  \else ", "\n",
50    r"    \divide\@tempcntb by \OVP@scale \ifnum\@tempcntb<\@ne \@tempcntb=\@ne \fi ", "\n",
51    r"    \unitlength=\@tempcntb sp\relax \@tempcntb=\OVP@scale \divide\@tempcnta by \unitlength ", "\n",
52    r"  \fi}", "\n",
53    // abs mode: coordinates are `\unitlength`s (set by `unit=`, else the 1pt
54    // register default). Clamp unitlength >= 1sp for the same divide-by-0 guard.
55    r"\newcommand\OVP@calc@abs{%", "\n",
56    r"  \ifdim\unitlength<\@ne sp \unitlength=1pt\relax \fi ", "\n",
57    r"  \divide\@tempcnta by \unitlength \divide\@tempcntb by \unitlength}", "\n",
58    // OVP key family (overpic.sty L25-48). `\@m` = 1000 (permil).
59    r"\define@key{OVP}{rel}{\def\OVP@scale{#1}\let\OVP@calc\OVP@calc@rel}", "\n",
60    r"\define@key{OVP}{percent}[]{\def\OVP@scale{100}\let\OVP@calc\OVP@calc@rel}", "\n",
61    r"\define@key{OVP}{permil}[]{\def\OVP@scale{\@m}\let\OVP@calc\OVP@calc@rel}", "\n",
62    r"\define@key{OVP}{abs}[]{\let\OVP@calc\OVP@calc@abs}", "\n",
63    r"\define@key{OVP}{unit}{\unitlength=\dimexpr#1\relax}", "\n",
64    r"\define@key{OVP}{grid}[true]{}", "\n",
65    r"\define@key{OVP}{tics}{}", "\n",
66    r"\newenvironment{overpic}[2][]{%", "\n",
67    r"  \sbox\OVP@box{\includegraphics[#1]{#2}}%", "\n",
68    // per-env default is percent (overpic.sty `\ExecuteOptions{percent}`); then
69    // read any OVP keys from the optarg. `\setkeys*` is relaxed — it ignores the
70    // graphicx keys (width/trim/clip/...) that also live in `#1`.
71    r"  \def\OVP@scale{100}\let\OVP@calc\OVP@calc@rel ", "\n",
72    r"  \setkeys*{OVP}{#1}%", "\n",
73    r"  \@tempcnta=\wd\OVP@box ", "\n",
74    r"  \@tempcntb=\ht\OVP@box \advance\@tempcntb\dp\OVP@box ", "\n",
75    r"  \OVP@calc ", "\n",
76    r"  \ifnum\@tempcnta<\@ne \@tempcnta=\@ne \fi \ifnum\@tempcntb<\@ne \@tempcntb=\@ne \fi ", "\n",
77    r"  \begin{picture}(\@tempcnta,\@tempcntb)%", "\n",
78    r"    \put(0,0){\makebox(0,0)[bl]{\usebox\OVP@box}}%", "\n",
79    r"}{%", "\n",
80    r"  \end{picture}%", "\n",
81    r"}",
82  ));
83
84  // Perl: {Overpic} (capital O) takes arbitrary TeX instead of an image — used in
85  // ~3 arXiv papers, not ported (none of the 44 overpic reports use it).
86});
