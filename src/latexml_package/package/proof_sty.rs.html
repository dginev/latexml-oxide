7/// proof.sty:160-166 `\@infer` sets the label and the conclusion as `\hbox{$#1$}`/`\hbox{$#2$}` and each premise in
8/// an `\halign` cell `$##$`: math in a box of its own, where an author's `$` is text (`\infer[$MP$]{A}{B}`, the
9/// rc_struct3 E1 sample; KNOWN_PERL_ERRORS #907). Each part is digested in `\lx@math@in@hbox@group` (tex_math.rs),
10/// scoped to the part; an empty part stays empty. Guard `perfect_kernel_batch69c::dollar_in_proof_and_accents_hbox_math_is_text`.
11fn in_hbox_math(tokens: Tokens) -> Tokens {
12  if tokens.is_empty() {
13    tokens
14  } else {
15    let mut out = vec![T_CS!("\\lx@math@in@hbox@group"), T_BEGIN!()];
16    out.extend(tokens.unlist());
17    out.push(T_END!());
18    Tokens::new(out)
19  }
20}

22/// One row of proof.sty's premises split at its `&`s, catcode 4 or active (`\lx@proof@split@and`): one premise as
23/// itself, several as the n-ary AND of `\lx@proof@logical@and`.
24fn split_premises(tokens: Tokens) -> Tokens {
25  let clauses: Vec<Tokens> = split_tokens(tokens, vec![T_ALIGN!().into(), T_ACTIVE!('&').into()])
26    .into_iter()
27    .map(in_hbox_math)
28    .collect();
29  match clauses.len() {
30    0 => Tokens!(),
31    1 => clauses.into_iter().next().unwrap(),
32    _ => i_apply(&[], Tokens!(T_CS!("\\lx@proof@logical@and")), clauses),
33  }
34}

36/// The premises without each row end's own `*` and `[<skip>]` (the `\\[1pt]` between 1709.05725's premise rows):
37/// amsmath's `\math@cr` reads them with `\@ifstar` and `\new@ifnextchar[` (amsmath.sty:1174-1180), and amsgen's
38/// `\@ifstar` is a `\new@ifnextchar` too (amsgen.sty:54-63), which skips no space: only a `*` right after the `\\` and
39/// a `[` right after that are the row end's. After a space (a token after the control symbol `\\`) they are the
40/// premise's own text: pdflatex prints `\infer{E}{s \\ [t] u}` and `{s \\ *[t] u}` in `{align*}` with their `[t]`,
41/// 0 errors. That is the rule of amsmath's displays, where proof.sty's papers put `\infer`; outside them (`{array}`,
42/// `{eqnarray}`, no amsmath) LaTeX's `\@xarraycr` skips spaces before `[` (latex.ltx:16585) and pdflatex reads
43/// `s \\ [t] u`'s `[t]` as a skip, where this keeps it as text. Read before the premises split into rows, whose pieces
44/// lose their leading spaces (`split_tokens`).
45fn drop_row_skips(tokens: Tokens) -> Tokens {
46  let tokens = tokens.unlist();
47  let mut out = Vec::with_capacity(tokens.len());
48  let mut level = 0usize;
49  let mut index = 0;
50  while index < tokens.len() {
51    let token = tokens[index];
52    index += 1;
53    match token.get_catcode() {
54      Catcode::BEGIN => level += 1,
55      Catcode::END => level = level.saturating_sub(1),
56      _ => {},
57    }
58    out.push(token);
59    if level == 0 && token == T_CS!("\\\\") {
60      if tokens.get(index) == Some(&T_OTHER!("*")) {
61        index += 1;
62      }
63      if tokens.get(index) == Some(&T_OTHER!("["))
64        && let Some(close) = tokens[index..].iter().position(|t| *t == T_OTHER!("]"))
65      {
66        index += close + 1;
67      }
68    }
69  }
70  Tokens::new(out)
71}
