3/// The journal options whose `sv<option>.clo` turns `\if@mathematic` off or sets `\vec` to `\ensuremath{\mathbf{#1}}`
4/// at the class's end, as every copy of it the census's svjour and svjour3 papers ship has it (svepj.clo: 46 of 53
5/// copies leave `\vec` the class's; svepjST.clo 5 of 5, svepj-spec.clo 3, svepjCONF.clo 3, svwebconf.clo 1 set it;
6/// 1504.01694's svepjST.clo:213-238, CMBX in its PDF): `(option, \if@mathematic, \vec is \mathbf)`, for a paper that does
7/// not ship the file.
8const JOURNAL_VECTORS: [(&str, bool, bool); 9] = [
9  ("epj", false, false),
10  ("epjST", false, true),
11  ("epj-spec", false, true),
12  ("epjCONF", false, true),
13  ("webconf", false, true),
14  ("epj-ND", false, false),
15  ("jog", false, false),
16  ("cmp", false, false),
17  ("arma", false, false),
18];

25  for option in [

41  // svjour.cls:78-81 inputs `sv<option>.clo` for any other option; `epjCONF` (EPJ Web of Conferences) is the one
42  // papers use. Its svepjCONF.clo:233-235 (2009/4/9 v1, the copy 1010.2389 ships) adds the session title,
43  // `\session-title{…}` (the parameter text is `-title#1`), which only its running head `\epjrunnhead` prints: print
44  // furniture, so stored and never typeset. The clo's layout (\maketitle, \abstract, \institutename) stays the
45  // binding's. Without it `\session` was undefined and "-title…" set as a paragraph (1010.2389, 1010.3788,
46  // 1202.0149).
47  DeclareOption!("epjCONF", {
48    RawTeX!(r"\def\session-title#1{\gdef\@session-title{#1}}
49\def\@session-title{Give the exact title of the conference}");

52  // svjour3.cls:46-56, :82-87 (1803.06644's copy, v3.2; svjour.cls:39-49, :66-71, the probe's 2003/04/15 copy): the
53  // theorem numbering options, which choose how the class's theorem set is numbered (below); `numbook` also numbers the
54  // equations, figures and tables within the sections.
55  RawTeX!(r"\let\if@envcntreset\iffalse\let\if@envcntsame\iffalse\let\if@envcntsect\iffalse\let\if@numbook\iffalse");
56  DeclareOption!("numbook", "\\let\\if@envcntsect\\iftrue\\let\\if@numbook\\iftrue");
57  DeclareOption!("envcountreset", "\\let\\if@envcntreset\\iftrue");
58  DeclareOption!("envcountsame", "\\let\\if@envcntsame\\iftrue");
59  DeclareOption!("envcountsect", "\\let\\if@envcntsect\\iftrue");
60  // Other options get passed to article
61  DeclareOption!(None, {
62    Digest!("\\PassOptionsToClass{\\CurrentOption}{article}")?;
63  });

65  ProcessOptions!();
66  LoadClass!("article");
67  // The page a float's panels are set in rows against (the panel grid rule, `declare_panel_page`) of the EPJ layout
68  // (svepj.clo:30 `\ExecuteOptions{final,twocolumn,10pt}`, :74 `\textwidth{180mm}`, :290 `\columnsep{4mm}`), the
69  // journal option papers use (72 of 97 svjour papers in the census), as pdflatex gives it: 512pt in two 250.309pt
70  // columns. The other journal styles' pages are not known here: their rows follow the registers.
71  if is_document_class("svjour") && given_class_options().iter().any(|option| option == "epj") {
72    declare_panel_page(250.30945, 512.0, true);
73  }
74  RequireResource!("ltx-svjour.css");
75  // svjour.cls:89-99 (0710.3184's), svjour2.cls:142-152, svjour3.cls:147-157 (every census copy, after the journal option
76  // file) `\DeclareMathSymbol{\Gamma}{\mathalpha}{letters}{"00}`…: the capital Greek letters are the math italic's,
77  // italic as a Latin letter unless a math alphabet is in force (Perl's binding leaves them upright, KNOWN_PERL_ERRORS
78  // #1147). Witnesses 2101.03721 (Ψ CMMI10 in its PDF), 0710.3184, 2609.05466.
79  declare_capital_greek("", Some(LettersClass::Variable))?;
80  // The class's theorem set (svjour3.cls:885-933, svjour.cls:756-792), numbered as the options say: `{theorem}`
81  // within the section ("1.1") under envcountsect, else on its own ("1") and reset by each section under
82  // envcountreset; the others on the theorem counter under envcountsame ("Theorem 1, Lemma 2"), within the section
83  // under envcountsect, else each on its own counter (reset by each section under envcountreset). sv_support's
84  // `\@ifundefined` set (svmult's, `{theorem}` within the section) then skips them. Perl's binding gave every class
85  // `{theorem}` within the section whatever the options (sv_support.sty.ltxml:216; svjour.cls.ltxml:24-26 "These could
86  // affect numbering"; 1604.08616: "Theorem 3.1" where the PDF prints 1): KNOWN_PERL_ERRORS #1100. A counter reset
87  // by each section is declared within it (`\lx@svjour@reset`) and printed alone, "1", so its ids carry the section,
88  // `S1.Thmtheorem1`, not an order-dependent `Thmtheorem1a`. The separator between a within-counter's number and its
89  // own, `\@thmcountersep`, is the class's (svjour3.cls:804, :887; svjour.cls:677, :757; every census copy of svjour,
90  // svjour2 and svjour3): empty, "." only under envcountsect (and numbook), so a paper's own
91  // `\newtheorem{x}{X}[section]` prints "11", "A1", as pdflatex (KNOWN_PERL_ERRORS #1111). Guards
92  // `perfect_kernel_batch78q::{svjour_theorem_counters, theorem_counter_separator_is_the_class_s}`.
93  RawTeX!(r"\if@numbook\renewcommand\thefigure{\thesection.\@arabic\c@figure}\renewcommand\thetable{\thesection.\@arabic\c@table}%
94\renewcommand\theequation{\thesection.\@arabic\c@equation}%
95\@addtoreset{figure}{section}\@addtoreset{table}{section}\@addtoreset{equation}{section}\fi
96\def\@thmcountersep{}\if@envcntsect\def\@thmcountersep{.}\fi
97\def\lx@svjour@reset#1#2{\newtheorem{#1}{#2}[section]\expandafter\gdef\csname the#1\endcsname{\arabic{#1}}}
98\if@envcntsect\newtheorem{theorem}{Theorem}[section]\else\if@envcntreset\lx@svjour@reset{theorem}{Theorem}\else
99  \newtheorem{theorem}{Theorem}\fi\fi
100\def\lx@svjour@theorem#1#2{\if@envcntsame\newtheorem{#1}[theorem]{#2}\else\if@envcntsect\newtheorem{#1}{#2}[section]\else
101  \if@envcntreset\lx@svjour@reset{#1}{#2}\else\newtheorem{#1}{#2}\fi\fi\fi}
102\lx@svjour@theorem{case}{Case}\lx@svjour@theorem{conjecture}{Conjecture}\lx@svjour@theorem{corollary}{Corollary}%
103\lx@svjour@theorem{definition}{Definition}\lx@svjour@theorem{example}{Example}\lx@svjour@theorem{exercise}{Exercise}%
104\lx@svjour@theorem{lemma}{Lemma}\lx@svjour@theorem{note}{Note}\lx@svjour@theorem{problem}{Problem}%
105\lx@svjour@theorem{property}{Property}\lx@svjour@theorem{proposition}{Proposition}\lx@svjour@theorem{question}{Question}%
106\lx@svjour@theorem{solution}{Solution}\lx@svjour@theorem{remark}{Remark}");
107  RequirePackage!("sv_support");
108  // svjour.cls:546-556 (svjour3.cls:631-641): `\vec`, upright bold letters (`\mbox{\boldmath$\mathbf{#1}$}`) or, without
109  // `\if@mathematic` (svjour.cls:38, svjour3.cls:45: true), bold italic (`\mbox{\boldmath$#1$}`), not the kernel's arrow
110  // accent (KNOWN_PERL_ERRORS #1143) — pdflatex prints "v + ω" in CMBX and CMMIB (2609.05466, 2509.03659, 2207.14807;
111  // 28 svjour3 and 25 svjour papers of the census use it). The journal file of an option (svjour.cls:78-81
112  // `\input{sv\CurrentOption.clo}`) may turn `\if@mathematic` off, and may set `\vec` itself at the class's end
113  // (`\AtEndOfClass{… \def\vec#1{\ensuremath{\mathbf{#1}}}}`, Greek left medium): the paper's copy says which; without
114  // one, the census's copies (`JOURNAL_VECTORS`). The older svjour copies set `\ensuremath{\mathbf{#1}}` under
115  // `\if@mathematic` (17 of 73: 0710.3184's), Greek left medium: the paper's copy says which.
116  RawTeX!(r"\let\if@mathematic\iftrue");
117  let plain_mathbf = |text: &str| crate::shipped_class::class_code_contains(text, "\\def\\vec#1{\\ensuremath{\\mathbf{#1}}}");
118  let class_mathbf = crate::shipped_class::shipped_class_text("svjour")?.is_some_and(|text| plain_mathbf(&text));
119  let mut journal_mathbf = false;
120  for option in crate::shipped_class::class_options()? {
121    let (mathematic, mathbf) = match crate::shipped_class::shipped_file_text(&s!("sv{option}"), "clo") {
122      Some(text) => (
123        !crate::shipped_class::class_code_contains(&text, "\\let\\if@mathematic\\iffalse"),
124        plain_mathbf(&text),
125      ),
126      None => JOURNAL_VECTORS
127        .iter()
128        .find(|(name, ..)| *name == option)
129        .map_or((true, false), |&(_, mathematic, mathbf)| (mathematic, mathbf)),
130    };
131    if !mathematic {
132      RawTeX!(r"\let\if@mathematic\iffalse");
133    }
134    journal_mathbf |= mathbf;
135  }
136  if journal_mathbf {
137    RawTeX!(r"\def\vec#1{\ensuremath{\mathbf{#1}}}");
138  } else {
139    let upright = if class_mathbf { r"\mathbf" } else { r"\lx@sv@vec@mathbf" };
140    raw_tex(&s!(
141      r"\if@mathematic\def\vec#1{{\ensuremath{{{upright}{{#1}}}}}}\else\def\vec#1{{\ensuremath{{\lx@sv@vec@boldmath{{#1}}}}}}\fi"
142    ))?;
143  }
144  // svjour.cls:61-62 and svjour3.cls:392-393 (v3.3): `\preprint{<number>}`, the report number `\makeheadbox` sets
145  // at the title page's top right (svjour.cls:489-492) — a publication note, as REVTeX's (1801.06113, 1605.08256,
146  // 2305.07696), with no label, as the class prints none (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #921).
147  DefMacro!("\\preprint{}", "\\lx@add@pubnote[role=preprint,name={}]{#1}");
148});
