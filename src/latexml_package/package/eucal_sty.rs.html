4LoadDefinitions!({
5  // Perl: eucal.sty.ltxml
6  // eucal basically redefines \mathcal to use an Euler script.
7  // There's really nothing to do for LaTeXML.
8  Let!("\\CMcal", "\\mathcal");
9  // eucal.sty:32 `\DeclareMathAlphabet\EuScript{U}{eus}{m}{n}`, the Euler script alphabet eucal's `\mathcal` is
10  // (as euscript_sty.rs has it); `\EuScript` was the first error of 229 run-340 papers (witnesses 2103.16394,
11  // 2103.05078). eucal.sty:37-40: option `mathscr` makes `\mathscr` the same alphabet.
12  Let!("\\EuScript", "\\mathcal");
13  DeclareOption!("psamsfonts", None);
14  DeclareOption!("mathcal", None);
15  DeclareOption!("mathscr", "\\providecommand{\\mathscr}{}\\renewcommand{\\mathscr}{\\EuScript}");
16  ProcessOptions!();
17});
