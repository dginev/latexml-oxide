1//! mdwtab.sty — alternative tabular environment by Mark Wooding (1996).
2//!
3//! mdwtab.sty completely reimplements `\tabular` with its own
4//! `\tab@*` preamble-parsing machinery (`\tab@right`,
5//! `\tab@restorehlstate`, `\tab@bgroup`, `\tab@multicol`, etc.).
6//! Raw-loading it would redefine our locked `\tabular` binding —
7//! the redefinition is silently ignored, but then mdwtab's own
8//! `\tab@*` helpers (referenced by every column type and `\@arstrut`
9//! handler) end up undefined when the user actually enters a
10//! tabular, cascading through `\omit`/`\@startsection` mode errors
11//! to a TooManyErrors fatal.
12//!
13//! Perl LaTeXML has no `mdwtab.sty.ltxml`; with its default
14//! `INCLUDE_STYLES=false` the raw `mdwtab.sty` is NOT loaded —
15//! Perl emits a "missing binding" warning and continues with the
16//! binding-aware `\tabular`. The user's tables then render
17//! through standard array/tabular machinery (no mdwtab-specific
18//! ornaments, but cleanly).
19//!
20//! In Rust we default to `INCLUDE_STYLES=true` (ar5iv preload sets
21//! it). This stub suppresses the raw-load so `\tabular` keeps
22//! pointing at our binding-aware constructor. Same pattern as
23//! `delarray_sty.rs` / `trace_sty.rs`.
24//!
25//! Witness: canvas-3 stage-26 0910.3293 (uses `\usepackage{mathenv}`
26//! which `\RequirePackage{mdwtab}`).
27use crate::prelude::*;
28
29LoadDefinitions!({
30  // mdwtab.sty:418-423 `\array` → `\tab@array`, whose `\tab@left` for the `c`, `t` and `b` positions is
31  // `\leavevmode\hbox\bgroup$\@arrayleft\vcenter\bgroup` (`\vtop`, `\vbox`; mdwtab.sty:505-529): the array enters
32  // math itself, so in text it is a formula of its own without the kernel's "Missing $ inserted" for a `[c]` array
33  // (sect10.rs `\lx@array@vcenter@check`, latex.ltx's `\@array`). Its cells stay math (`\tab@bmaths` = `$`).
34  // rc_struct3 E2 (est. 89 papers; arXiv 1807.04837): `\begin{center}\begin{array}{rl}…\end{array}\end{center}`.
35  // Guard: `perfect_kernel_batch69b::mdwtab_array_in_text_is_a_formula`.
36  DefMacro!(
37    "\\lx@array@text@open{}",
38    r"\ifmmode\let\lx@array@text@close\relax\else\lx@begin@inline@math\let\lx@array@text@close\lx@end@inline@math\fi"
39  );
40  // mdwtab.sty:765-790 `\hlx{letters}`: `v[dim]` = `\vgap`, which ENDS the
41  // current row (`\cr`) and adds vertical space; `h` = `\hline` (`hh`
42  // double); `b`, `/` are print-only. Parsed here and emitted as bare
43  // `\\`/`\hline` tokens — a TeX-level loop left its bookkeeping assignment
44  // as the next row's first token, which the alignment then took as cell
45  // content and the following `\hline` was "`\noalign` cannot be used here"
46  // (talkdoc `\hlx{hhv[1pt]}`). Guard:
47  // `perfect_kernel_batch54::mdwtab_hlx_ends_the_row_and_rules`.
48  DefMacro!("\\hlx{}", sub[(letters)] {
49    let spec = letters.to_string();
50    let mut out: Vec<Token> = Vec::new();
51    // At a table's very start (talkdoc's leading `\hlx{hhv[1pt]}`) no row is
52    // open, so no `\\` precedes the rules.
53    let mut row_open = match lookup_alignment() {
54      Some(a) => a.alignment_cell().map(|cell| cell.borrow().is_in_row()).unwrap_or(true),
55      None => true,
56    };
57    let mut chars = spec.chars().peekable();
58    while let Some(c) = chars.next() {
59      match c {
60        'v' | 'h' => {
61          if row_open {
62            out.push(T_CS!("\\\\"));
63            row_open = false;
64          }
65          if c == 'h' {
66            out.push(T_CS!("\\hline"));
67          }
68        },
69        '[' => {
70          for d in chars.by_ref() {
71            if d == ']' { break; }
72          }
73        },
74        _ => {},
75      }
76    }
77    Ok(Tokens::new(out))
78  });
79});
