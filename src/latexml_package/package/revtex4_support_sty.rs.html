896  let given = crate::shipped_class::class_options()?;

898  if post.is_empty() {

902      .collect();

904      .iter()
905      .enumerate()
906      .rfind(|(_, option)| named(option));
907    return Ok(match last {
908      Some((at, option)) => (
909        (if at < 9 { "aps" } else { "osa" }).to_string(),
910        (if matches!(*option, "aps" | "osa") {
911          ""
912        } else {
913          option
914        })
915        .to_string(),
916      ),
917      None => ("aps".to_string(), String::new()),
918    });
919  }
920  let societies = if post == "4-1" {
921    "aps aip apsrmp"
922  } else {
923    "aps aip aapm sor apsrmp"
924  };

926    .iter()
927    .find(|g| {
928      societies
929        .split_whitespace()
930        .any(|society| society == g.as_str())
931    })
932    .map_or("aps", String::as_str)
933    .to_string();
934  let journals = match (post, society.as_str()) {
935    ("4-1", "aps") => "pra prb prc prd pre prl prstab prstper rmp",

939    },
940    (_, "aip") => "jcp pop rsi jap apl apm cha pof bmf rse jmp adv sd jor cp",
941    _ => "",
942  };
943  let journal = given
944    .iter()
945    .find(|g| {
946      journals
947        .split_whitespace()
948        .any(|journal| journal == g.as_str())
949    })
950    .cloned()
951    .unwrap_or_default();
952  Ok((society, journal))
953}

1153  };

1157    if declared_grouped {
1158      s!("annotate={annotate},_grouped=true")
1159    } else {
1160      s!("annotate={annotate}")
1161    },
1162  )));

1164    T_CS!("\\lx@add@affiliation@marked")
1165  } else {
1166    T_CS!("\\lx@add@affiliation")
1167  };
