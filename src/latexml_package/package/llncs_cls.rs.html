732/// Does `tokens` step the footnote counter: a `\footnotemark` or `\footnote` given no `[number]`, an `\addtocounter`,
733/// `\setcounter`, `\stepcounter` or `\refstepcounter` of `{footnote}` — each by meaning, a `\let` copy too — or a
734/// `\thanks`, unless `thanks_gobbled` (llncs.cls:1028-1029's counting pass `\def\thanks##1{}`: the token spelled
735/// `\thanks` is gobbled there with its argument, a `\let` copy of it is not). A macro the document or a file it loaded
736/// (a raw-loaded package's too) defined is read in its body, as its call runs it (`\newcommand\fm{\footnotemark}`,
737/// `\DeclareRobustCommand`'s inner `\fm␣`), a few levels deep; past them, a command undefined here or a document macro
738/// with no token body is unsure, and read as a step (DG 2026-11-16: a wrong glyph is worse than none; so a tikz ORCID
739/// logo's `\draw`, defined only in its `tikzpicture`, withdraws 2009.11534's correct ⋆, row R LLNCSUNSURE). What runs
740/// nothing is no step: the tokens `\ifdefined` and `\ifx` test (`\ifdefined\Anonymous…\else…\fi`, 2509.14694), and
741/// `\undefined`, `\@undefined`. A given number steps nothing: `\samethanks`'s `\footnotemark[\value{footnote}]`
742/// (2007.09127, 2108.01436).
743fn steps_footnotes(tokens: &Tokens, thanks_gobbled: bool) -> Result<bool> {
744  let meaning = |cs: &str| lookup_definition(&T_CS!(cs));
745  let mut scan = StepScan {
746    thanks_gobbled,
747    tests: vec![(meaning("\\ifdefined")?, 1), (meaning("\\ifx")?, 2)],
748    thanks: meaning("\\thanks")?,
749    marks: vec![meaning("\\footnotemark")?, meaning("\\footnote")?],
750    counters: vec![
751      meaning("\\addtocounter")?,
752      meaning("\\setcounter")?,
753      meaning("\\stepcounter")?,
754      meaning("\\refstepcounter")?,
755    ],
756    read: Vec::new(),
757  };
758  scan.steps(tokens.unlist_ref(), 0)
759}

761/// How deep [`steps_footnotes`] reads a document's macros, and how many it reads.
762const STEP_SCAN_DEPTH: usize = 4;
763const STEP_SCAN_MACROS: usize = 32;
764
765/// [`steps_footnotes`]'s meanings and the macros it has read.
766struct StepScan {
767  thanks_gobbled: bool,
768  /// the conditionals testing tokens without running them, and how many each tests
769  tests:          Vec<(Option<Rc<dyn Definition>>, usize)>,
770  thanks:         Option<Rc<dyn Definition>>,
771  marks:          Vec<Option<Rc<dyn Definition>>>,
772  counters:       Vec<Option<Rc<dyn Definition>>>,
773  read:           Vec<Token>,
774}
775
776impl StepScan {
777  fn steps(&mut self, v: &[Token], depth: usize) -> Result<bool> {
778    let next = |from: usize| (from..v.len()).find(|&i| v[i].get_catcode() != Catcode::SPACE);
779    // (the end of the group opening at `open`)
780    let close = |open: usize| {
781      let mut depth = 0usize;
782      (open..v.len())
783        .find(|&i| {
784          match v[i].get_catcode() {
785            Catcode::BEGIN => depth += 1,
786            Catcode::END => depth = depth.saturating_sub(1),
787            _ => {},
788          }
789          depth == 0
790        })
791        .unwrap_or(v.len())
792    };
793    let mut at = 0;
794    while at < v.len() {
795      let t = v[at];
796      if !matches!(t.get_catcode(), Catcode::CS | Catcode::ACTIVE) {
797        at += 1;
798        continue;
799      }
800      let defn = lookup_definition(&t)?;
801      let is = |known: &Option<Rc<dyn Definition>>| match (&defn, known) {
802        (Some(a), Some(b)) => Rc::ptr_eq(a, b),
803        _ => false,
804      };
805      if let Some(&(_, tested)) = self.tests.iter().find(|(test, _)| is(test)) {
806        // (the tested tokens, past the spaces before each)
807        for _ in 0..tested {
808          at = next(at + 1).unwrap_or(v.len());
809        }
810      } else if t == T_CS!("\\undefined") || t == T_CS!("\\@undefined") {
811        // (undefined on purpose, to be tested: `\ifx\Anonymous\undefined`)
812      } else if t == T_CS!("\\thanks") && self.thanks_gobbled {
813        if let Some(arg) = next(at + 1).filter(|&arg| v[arg].get_catcode() == Catcode::BEGIN) {
814          at = close(arg);
815        }
816      } else if t == T_CS!("\\thanks") || is(&self.thanks) {
817        return Ok(true);
818      } else if t == T_CS!("\\footnotemark")
819        || t == T_CS!("\\footnote")
820        || self.marks.iter().any(is)
821      {
822        if next(at + 1).is_none_or(|arg| v[arg] != T_OTHER!("[")) {
823          return Ok(true);
824        }
825      } else if self.counters.iter().any(is) {
826        if let Some(arg) = next(at + 1).filter(|&arg| v[arg].get_catcode() == Catcode::BEGIN) {
827          let end = close(arg);
828          if Tokens::new(v[arg + 1..end.min(v.len())].to_vec())
829            .to_string()
830            .trim()
831            == "footnote"
832          {
833            return Ok(true);
834          }
835        }
836      } else if let Some(defn) = defn.as_ref().filter(|d| !d.get_origin().is_latexml_owned()) {
837        // (a document's or a loaded file's macro: its body, once)
838        if !self.read.contains(&t) {
839          if depth >= STEP_SCAN_DEPTH || self.read.len() >= STEP_SCAN_MACROS {
840            return Ok(true);
841          }
842          self.read.push(t);
843          match defn.get_expansion() {
844            Some(ExpansionBody::Tokens(body)) => {
845              if self.steps(body.unlist_ref(), depth + 1)? {
846                return Ok(true);
847              }
848            },
849            _ => return Ok(true),
850          }
851        }
852      } else if defn.is_none() {
853        return Ok(true);
854      }
855      at += 1;
856    }
857    Ok(false)
858  }
859}

861/// `body` without its top-level `\institute{…}` calls, and the calls.
862fn take_institutes(body: &Tokens) -> (Tokens, Vec<Token>) {
863  let v = body.unlist_ref();
864  let (mut names, mut institutes) = (Vec::with_capacity(v.len()), Vec::new());
865  let (mut at, mut depth) = (0usize, 0usize);
866  while at < v.len() {
867    let t = v[at];
868    match t.get_catcode() {
869      Catcode::BEGIN => depth += 1,
870      Catcode::END => depth = depth.saturating_sub(1),
871      _ => {},
872    }
873    if depth == 0 && t == T_CS!("\\institute") {
874      let mut arg = at + 1;
875      while v.get(arg) == Some(&T_SPACE!()) {
876        arg += 1;
877      }
878      if v
879        .get(arg)
880        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
881      {
882        let mut end = arg;
883        let mut inner = 0usize;
884        while end < v.len() {
885          match v[end].get_catcode() {
886            Catcode::BEGIN => inner += 1,
887            Catcode::END => {
888              inner -= 1;
889              if inner == 0 {
890                break;
891              }
892            },
893            _ => {},
894          }
895          end += 1;
896        }
897        if end < v.len() {
898          institutes.extend_from_slice(&v[at..=end]);
899          at = end + 1;
900          continue;
901        }
902      }
903    }
904    names.push(t);
905    at += 1;
906  }
907  (Tokens::new(names), institutes)
