161  // (\lx@llncs@affiliation -> \lx@add@affiliation, singular, no further split).
162  // Three shapes occur in the wild:
163  //  (a) Well-formed \and-separated institutions, each possibly with an internal
164  //      "\\" break between its name and its \email/\url (witness 2605.16562).
165  //      Faithful \and-split; \email inside an affiliation inherits that
166  //      affiliation's label, so it attaches to the matching \inst{N} authors.
167  //  (b) Single institution, no \and, "\\"-separating the name from a SHARED
168  //      \email covering every author (witness 2608.11332, arXiv/html_feedback
169  //      #6881). This is Perl's plain case: it must stay ONE affiliation, on all
170  //      the \inst{1} authors. Splitting the "\\" here (the pre-fix bug) put the
171  //      \email in its own "affiliation:2" contact that no author carries, so it
172  //      mislanded on a single creator and left an empty affiliation stub. Its
173  //      \email lines are each address its owner's (85b, below), not the brace
174  //      list whole on every author, which inheriting "affiliation:1" made it.
175  //  (c) Lazy: ONE block, no \and, hand-typed superscripts ("$^1$..\quad $^2$..")
176  //      (witness 2606.19939). Upstream makes a single affiliation = the whole
177  //      block attached to every \inst{1} author (duplicating it) while \inst{2}+
178  //      get nothing (reproduces identically in Perl LaTeXML 0.8.8). It is the
179  //      one institute of (b), which the frontmatter API's Cut (`cut_marked_contact`,
180  //      OD #954) cuts at its marks when two or more of them are requested by the
181  //      names: each entry labelled by its mark, which `\inst{N}` links to (Z18b: was
182  //      the binding's own cut, `affiliation_calls`, at the call; aa's sole
183  //      `\institute` is cut the same way, 2410.00577). A list whose marks no two
184  //      names request (`\institute{Univ A, Town \\ \textsuperscript{*}Equal
185  //      contribution}`) is not cut: the sole institute every author's, as
186  //      llncs.cls:868-886 prints it unnumbered under every name — it had gone to the
187  //      last author alone, dev20k 2603.18089, 2608.22772, 2207.13038, 2609.31726,
188  //      1905.07852; repro sectioning-frontmatter/llncs_institute_legend_line_shared.

191  // the `\and`s by meaning, as `\institutename`'s `\and` numbers them (Option C P2-3, OXIDIZED_DESIGN_DIVERGENCES #995):

615/// `body` without its top-level `\institute{…}` calls, and the calls.
616fn take_institutes(body: &Tokens) -> (Tokens, Vec<Token>) {
617  let v = body.unlist_ref();
618  let (mut names, mut institutes) = (Vec::with_capacity(v.len()), Vec::new());
619  let (mut at, mut depth) = (0usize, 0usize);
620  while at < v.len() {
621    let t = v[at];
622    match t.get_catcode() {
623      Catcode::BEGIN => depth += 1,
624      Catcode::END => depth = depth.saturating_sub(1),
625      _ => {},
626    }
627    if depth == 0 && t == T_CS!("\\institute") {
628      let mut arg = at + 1;
629      while v.get(arg) == Some(&T_SPACE!()) {
630        arg += 1;
631      }
632      if v
633        .get(arg)
634        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
635      {
636        let mut end = arg;
637        let mut inner = 0usize;
638        while end < v.len() {
639          match v[end].get_catcode() {
640            Catcode::BEGIN => inner += 1,
641            Catcode::END => {
642              inner -= 1;
643              if inner == 0 {
644                break;
645              }
646            },
647            _ => {},
648          }
649          end += 1;
650        }
651        if end < v.len() {
652          institutes.extend_from_slice(&v[at..=end]);
653          at = end + 1;
654          continue;
655        }
656      }
657    }
658    names.push(t);
659    at += 1;
660  }
661  (Tokens::new(names), institutes)
662}
