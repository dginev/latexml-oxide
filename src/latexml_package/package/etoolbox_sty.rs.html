3/// A token list as etoolbox's `\patchcmd` writes it out (etoolbox.sty:1358-1371): a macro's body as `\meaning`
4/// prints it (`params_doubled`: a nested parameter `##`), the search and replacement texts as `\detokenize` prints
5/// them with `#` an other character (:1355). A control sequence whose name is a word (or a letter or `@`, which the
6/// rescan under `\makeatletter`, :1376-1382, reads as letters) is followed by a space (tex.web §262 `print_cs`), so the text
7/// rescans to the same tokens whatever follows it.
8fn etb_patch_text(tokens: &Tokens, params_doubled: bool) -> String {
9  let mut out = String::new();
10  for t in tokens.unlist_ref() {
11    match t.get_catcode() {
12      Catcode::COMMENT => {},
13      Catcode::CS => t.with_str(|text| {
14        let name = text.strip_prefix('\\').unwrap_or(text);
15        out.push('\\');
16        out.push_str(name);
17        let mut chars = name.chars();
18        let word = match (chars.next(), chars.next()) {
19          (Some(_), Some(_)) => true,
20          (Some(c), None) => c.is_alphabetic() || c == '@',
21          _ => false,
22        };
23        if word {
24          out.push(' ');
25        }
26      }),
27      Catcode::PARAM => t.with_str(|text| {
28        out.push_str(text);
29        if params_doubled {
30          out.push_str(text);
31        }
32      }),
33      // A space token is one space in `\meaning`, whatever its source text (a line end keeps `\n`): review rev78p.
34      Catcode::SPACE => out.push(' '),
35      Catcode::ARG => {
36        out.push('#');
37        t.with_str(|text| out.push_str(text));
38      },
39      _ => t.with_str(|text| out.push_str(text)),
40    }
41  }
42  out
43}
