9/// Whether `digested` holds a table float of its own — the `{table}` a deluxetable opens (`{longrotatetable}`).
10fn holds_table_float(digested: &Digested) -> bool {
11  match digested.data() {
12    DigestedData::List(list) => list.borrow().boxes.iter().any(holds_table_float),
13    DigestedData::Whatsit(whatsit) => {
14      let whatsit = whatsit.borrow();
15      let cs = whatsit.get_definition().get_cs().to_string();
16      cs == "\\begin{table}"
17        || cs == "\\begin{table*}"
18        || whatsit
19          .get_body()
20          .ok()
21          .flatten()
22          .is_some_and(|body| holds_table_float(&body))
23        || whatsit.get_args().iter().flatten().any(holds_table_float)
24    },
25    _ => false,
26  }
27}

29/// The older AASTeX's affiliation marks (Perl aas_support.sty.ltxml): `\altaffilmark{1,2}` in an `\author` marks the

36  );
37  DefMacro!(
38    "\\altaffiltext Semiverbatim {}",
39    "\\lx@add@contact[role=altaffiliation,label={altaffil:#1}]{#2}"
40  );
41  Ok(())
42}

1421/// AASTeX's `\fig`-family dispatch, after its file argument and a peek at the next token (pushed back for the target
1422/// to read): a brace opens the panel form `panel`; anything else makes it a `\ref` (astro-ph/0003209).
1423fn aas_fig_dispatch(arg: Tokens, test: Token, panel: &str) -> Result<Tokens> {
1424  // Push order is reversed for stack semantics: last unread is first read.
1425  unread_one(test);
1426  unread_one(T_END!());
1427  unread_vec(arg.unlist());
1428  unread_one(T_BEGIN!());
1429  Ok(if test.get_catcode() != Catcode::BEGIN {
1430    Tokens!(T_CS!("\\ref"))
1431  } else {
1432    Tokens!(T_CS!(panel))
1433  })
