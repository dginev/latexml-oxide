5  // Perl: varioref.sty.ltxml
6  // INCOMPLETE IMPLEMENTATION (as in Perl)
7  // Perl varioref.sty.ltxml L24-29: all five CSes pass `locked => 1`
8  // so later packages (cleveref, revtex, etc.) or user \renewcommand
9  // calls can't silently override these stubs — the tests for whether
10  // varioref is loaded already read true once these are defined, so a
11  // quiet override would leave the references broken.
12  // The real macros read optionals Perl's signatures lack: `\vref*[text]{l}`,
13  // `\vpageref*[here][far]{l}`, `\vrefrange[here]{a}{b}`,
14  // `\vpagerefrange*[here]{a}{b}` (varioref.sty:803-966); Perl leaves the `[…]`
15  // as text and refs the label `[` (KPE #367). The bodies stay Perl's page-less
16  // `\ref`: the page text they would add is page-layout relative. `\vrefrange`
17  // prints `\reftextlabelrange`, as `\vrefrangedefaultformat` (varioref.sty:959). The
18  // star `\vrefrange` does not read (:966) stays Perl's.
19  DefMacro!("\\vref OptionalMatch:* [] Semiverbatim", "\\ref{#3}", locked => true);
20  DefMacro!("\\vpageref OptionalMatch:* [] [] Semiverbatim", "\\ref{#4}", locked => true);
21  DefMacro!("\\vrefrange OptionalMatch:* [] Semiverbatim Semiverbatim",
22    "\\reftextlabelrange{#3}{#4}", locked => true);
23  DefMacro!("\\vpagerefrange OptionalMatch:* [] Semiverbatim Semiverbatim",
24    "\\vref{#3}--\\vref{#4}", locked => true);
25
26  DefMacro!("\\vrefpagenum DefToken Semiverbatim", "\\def#1{\\ref{#2}}",
