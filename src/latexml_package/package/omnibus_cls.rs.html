12/// Push the digested body of a `{keyword}`/`{keywords}` env directly into
13/// the `frontmatter` map under `ltx:classification[scheme=keywords]`, the
14/// way Perl LaTeXML's after_digest_keywords does it
15/// (OmniBus.cls.ltxml `sub after_digest_keywords`). Avoids the
16/// raw_tex(`#body`) workaround that mistakenly tokenized `#` as PARAM
17/// and dumped the literal text "#body" into the output element.
18fn push_keyword_body_to_frontmatter(whatsit: &mut Whatsit) -> Result<Vec<Digested>> {
19  use latexml_core::{BoxOps, common::store::Stored};
20  if let Some(body) = whatsit.get_body()? {
21    let mut attrs: rustc_hash::FxHashMap<String, String> = rustc_hash::FxHashMap::default();
22    attrs.insert("scheme".to_string(), "keywords".to_string());
23    let entry = document::tag::TagData {
24      tag:     "ltx:classification".to_string(),
25      attr:    attrs,
26      content: vec![document::tag::TagContent::Box(body)],
27    };
28    with_value_mut("frontmatter", |val_opt| {
29      if let Some(Stored::HashTagData(frnt)) = val_opt {
30        frnt
31          .entry("ltx:classification".to_string())
32          .or_default()
33          .push(entry);
34      }
35    });
36  }
37  Ok(Vec::new())
38}

40/// True when `arg` (trimmed) is one or more ASCII digits and nothing else — the
41/// signal that a `\affil{n}` / `\affiliation{n}{…}` uses AGU's numbered
42/// superscript-reference convention rather than authblk/AAS affiliation text.
43fn is_all_ascii_digits(arg: &Tokens) -> bool {
44  let s = arg.to_string();
45  let s = s.trim();
46  !s.is_empty() && s.chars().all(|c| c.is_ascii_digit())
47}

49/// An `\authors` list with each `\affil{…}` whose argument is numbers (`1`, `1,2`) written `\inst{…}`: the request
50/// of those numbered affiliations, read as llncs's mark is (`\authors`, below).
51fn numbered_affil_marks(list: Tokens) -> Tokens {
52  let tokens = list.unlist();
53  let mut out = Vec::with_capacity(tokens.len());
54  let mut at = 0;
55  while at < tokens.len() {
56    let t = tokens[at];
57    at += 1;
58    if t == T_CS!("\\affil") {
59      let open = at
60        + tokens[at..]
61          .iter()
62          .take_while(|t| t.get_catcode() == Catcode::SPACE)
63          .count();
64      if tokens
65        .get(open)
66        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
67        && let Some(close) = tokens[open + 1..]
68          .iter()
69          .position(|t| t.get_catcode() == Catcode::END)
70          .map(|end| open + 1 + end)
71      {
72        let text = Tokens::new(tokens[open + 1..close].to_vec()).to_string();
73        let numbers = text.chars().any(|c| c.is_ascii_digit())
74          && text
75            .chars()
76            .all(|c| c.is_ascii_digit() || c == ',' || c == ' ');
77        if numbers {
78          out.push(T_CS!("\\inst"));
79          continue;
80        }
81      }
82    }
83    out.push(t);
84  }
85  Tokens::new(out)
86}

88/// OmniBus over the base class a stand-in binding loaded as its real file does (OXIDIZED_DESIGN_DIVERGENCES #736):
89/// provide-only ([`ProvideOnlyGuard`], as `\documentstyle` loads it over a class, sect01.rs), so it adds the frontmatter
90/// commands and environments the base class leaves undefined (`{keywords}`, `\email`, `\affiliation`, …) and replaces
91/// none of the base class's. A binding that once loaded OmniBus as its base (ceurart, atlasdoc, lmcs, pnas-new,
92/// czipreprint) keeps what OmniBus gave it where the real base class (scrartcl, amsart, extarticle) has nothing:
93/// ceurart's `{keywords}`, OmniBus's `ltx:classification`, which a trial without it lost (-10 to -15 words a paper;
94/// ceurart witnesses 2509.11496, 2608.23214).
95pub fn provide_over_base_class() -> Result<()> {
96  let _provide_only = ProvideOnlyGuard::new();
97  load_class("OmniBus", Vec::new(), Tokens!())
98}

259  // \authors{author list} — the plural byline macro used by journal classes
260  // (AGU's agujournal2019, ametsoc, …) as an alternative to \author. It holds
261  // ONE comma-separated list of authors, the last joined with "and", each name
262  // trailed by an `\affil{n}` superscript that cross-references a numbered
263  // `\affiliation{n}{text}` block (see \affil / \affiliation below): separate

266  // reading (`\lx@add@author@split`, as llncs's `\author`, OD #984), which parts it

822  // Perl L286-297: math/package autoloads — when a trigger CS is used and
823  // not yet defined, require the specified package/class and re-trigger.
824  // The Perl `DefAutoload` macro registers this semantic; we implement it
825  // inline. Perl encodes the target via the `.sty.ltxml` / `.cls.ltxml`
826  // suffix; we mirror that by carrying an explicit "sty" / "cls" kind
827  // alongside the bare name.
828  //
829  // Why the kind matters: `\thechapter` autoloads `book.cls.ltxml` in Perl,
830  // not `book.sty`. Routing through `require_package("book")` instead
831  // finds the obsolete `book.sty` shim (TL's 2.09-compat file) which
832  // immediately fires `\LoadClass{book}` from inside the body — past the
833  // preamble — and errors with "\LoadClass can only appear in the preamble".
834  // Witness: arXiv:2602.10407 (`\documentclass{saunders}` + `\chapter` in
835  // an `\include`'d chapter; saunders.cls is unbound so OmniBus takes over,
836  // and the chapter trigger hits `\thechapter`). Perl handles it cleanly
837  // by autoloading the class binding instead of the obsolete sty.
838  for (trigger, name, ext) in [
839    // env triggers: `\begin{align}` etc. In Rust, we only dispatch on the
840    // bare CS name of the trigger — works for control sequences like
841    // `\multline`, `\numberwithin`, `\mathfrak`, `\mathbb`, `\deluxetable`,
842    // `\curraddr`, `\subjclass`, `\thechapter`. For envs, autoload key is
843    // `\begin{env}` which is a CS token.
844    ("\\begin{align}",                "amsmath",      "sty"),
845    ("\\begin{subequations}",         "amsmath",      "sty"),
846    ("\\begin{split}",                "amsmath",      "sty"),
847    ("\\multline",                    "amsmath",      "sty"),
848    ("\\csname multline*\\endcsname", "amsmath",      "sty"),
849    ("\\numberwithin",                "amsmath",      "sty"),
850    ("\\mathfrak",                    "amsfonts",     "sty"),
851    ("\\mathbb",                      "amsfonts",     "sty"),
852    ("\\begin{deluxetable}",          "deluxetable",  "sty"),
853    ("\\curraddr",                    "ams_support",  "sty"),
854    ("\\subjclass",                   "ams_support",  "sty"),
855    ("\\thechapter",                  "book",         "cls"),
856  ] {
857    let cs = T_CS!(trigger);
858    if !IsDefined!(&cs) {
859      let cs_clone = cs;
860      let name_str = name.to_string();
861      let trigger_str = trigger.to_string();
862      let is_cls = ext == "cls";
863      def_macro(cs, None,
864        ExpansionBody::Closure(Rc::new(move |_args| {
865          // Mirrors Perl's DefAutoload → ClearAutoLoad in Package.pm:
866          // clear this autoload CS before loading, then re-emit the trigger as
867          // tokenized text. Re-tokenizing is important for `\begin{env}` triggers
868          // — amsmath defines `\split` (not `\begin{split}`), so the raw single-CS
869          // token would look undefined after clearing. Tokenizing expands into
870          // `\begin` + `{env}` which the standard `\begin{}` dispatcher resolves.
871          assign_meaning(
872            &cs_clone, Stored::None,
873            Some(Scope::Global));
874          if is_cls {
875            load_class(
876              &name_str, Vec::new(), Tokens::default())?;
877          } else if name_str == "amsmath" {
878            // A guess for a class we do not know: amsmath's environments, not its "Old form" error, which waits for a
879            // real load (`amsmath_sty::require_amsmath_beyond_original`; KNOWN_PERL_ERRORS #993).
880            amsmath_sty::require_amsmath_beyond_original(false, &[])?;
881          } else {
882            require_package(&name_str, RequireOptions::default())?;
883          }
884          Ok(mouth::tokenize_internal(TeXString::assembled(trigger_str.clone())))
885        })), None)?;
886    }
887  }
