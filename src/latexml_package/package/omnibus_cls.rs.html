8/// Push the digested body of a `{keyword}`/`{keywords}` env directly into
9/// the `frontmatter` map under `ltx:classification[scheme=keywords]`, the
10/// way Perl LaTeXML's after_digest_keywords does it
11/// (OmniBus.cls.ltxml `sub after_digest_keywords`). Avoids the
12/// raw_tex(`#body`) workaround that mistakenly tokenized `#` as PARAM
13/// and dumped the literal text "#body" into the output element.
14fn push_keyword_body_to_frontmatter(whatsit: &mut Whatsit) -> Result<Vec<Digested>> {
15  use latexml_core::{BoxOps, common::store::Stored};
16  if let Some(body) = whatsit.get_body()? {
17    let mut attrs: rustc_hash::FxHashMap<String, String> = rustc_hash::FxHashMap::default();
18    attrs.insert("scheme".to_string(), "keywords".to_string());
19    let entry = document::tag::TagData {
20      tag:     "ltx:classification".to_string(),
21      attr:    attrs,
22      content: vec![document::tag::TagContent::Box(body)],
23    };
24    with_value_mut("frontmatter", |val_opt| {
25      if let Some(Stored::HashTagData(frnt)) = val_opt {
26        frnt
27          .entry("ltx:classification".to_string())
28          .or_insert_with(Vec::new)
29          .push(entry);
30      }
31    });
32  }
33  Ok(Vec::new())
34}

36/// True when `arg` (trimmed) is one or more ASCII digits and nothing else — the
37/// signal that a `\affil{n}` / `\affiliation{n}{…}` uses AGU's numbered
38/// superscript-reference convention rather than authblk/AAS affiliation text.
39fn is_all_ascii_digits(arg: &Tokens) -> bool {
40  let s = arg.to_string();
41  let s = s.trim();
42  !s.is_empty() && s.chars().all(|c| c.is_ascii_digit())
43}

654  // Perl L286-297: math/package autoloads — when a trigger CS is used and
655  // not yet defined, require the specified package/class and re-trigger.
656  // The Perl `DefAutoload` macro registers this semantic; we implement it
657  // inline. Perl encodes the target via the `.sty.ltxml` / `.cls.ltxml`
658  // suffix; we mirror that by carrying an explicit "sty" / "cls" kind
659  // alongside the bare name.
660  //
661  // Why the kind matters: `\thechapter` autoloads `book.cls.ltxml` in Perl,
662  // not `book.sty`. Routing through `require_package("book")` instead
663  // finds the obsolete `book.sty` shim (TL's 2.09-compat file) which
664  // immediately fires `\LoadClass{book}` from inside the body — past the
665  // preamble — and errors with "\LoadClass can only appear in the preamble".
666  // Witness: arXiv:2602.10407 (`\documentclass{saunders}` + `\chapter` in
667  // an `\include`'d chapter; saunders.cls is unbound so OmniBus takes over,
668  // and the chapter trigger hits `\thechapter`). Perl handles it cleanly
669  // by autoloading the class binding instead of the obsolete sty.
670  for (trigger, name, ext) in [
671    // env triggers: `\begin{align}` etc. In Rust, we only dispatch on the
672    // bare CS name of the trigger — works for control sequences like
673    // `\multline`, `\numberwithin`, `\mathfrak`, `\mathbb`, `\deluxetable`,
674    // `\curraddr`, `\subjclass`, `\thechapter`. For envs, autoload key is
675    // `\begin{env}` which is a CS token.
676    ("\\begin{align}",                "amsmath",      "sty"),
677    ("\\begin{subequations}",         "amsmath",      "sty"),
678    ("\\begin{split}",                "amsmath",      "sty"),
679    ("\\multline",                    "amsmath",      "sty"),
680    ("\\csname multline*\\endcsname", "amsmath",      "sty"),
681    ("\\numberwithin",                "amsmath",      "sty"),
682    ("\\mathfrak",                    "amsfonts",     "sty"),
683    ("\\mathbb",                      "amsfonts",     "sty"),
684    ("\\begin{deluxetable}",          "deluxetable",  "sty"),
685    ("\\curraddr",                    "ams_support",  "sty"),
686    ("\\subjclass",                   "ams_support",  "sty"),
687    ("\\thechapter",                  "book",         "cls"),
688  ] {
689    let cs = T_CS!(trigger);
690    if !IsDefined!(&cs) {
691      let cs_clone = cs;
692      let name_str = name.to_string();
693      let trigger_str = trigger.to_string();
694      let is_cls = ext == "cls";
695      def_macro(cs, None,
696        ExpansionBody::Closure(Rc::new(move |_args| {
697          // Mirrors Perl's DefAutoload → ClearAutoLoad in Package.pm:
698          // clear this autoload CS before loading, then re-emit the trigger as
699          // tokenized text. Re-tokenizing is important for `\begin{env}` triggers
700          // — amsmath defines `\split` (not `\begin{split}`), so the raw single-CS
701          // token would look undefined after clearing. Tokenizing expands into
702          // `\begin` + `{env}` which the standard `\begin{}` dispatcher resolves.
703          assign_meaning(
704            &cs_clone, Stored::None,
705            Some(Scope::Global));
706          if is_cls {
707            load_class(
708              &name_str, Vec::new(), Tokens::default())?;
709          } else {
710            require_package(&name_str, RequireOptions::default())?;
711          }
712          Ok(mouth::tokenize_internal(TeXString::assembled(trigger_str.clone())))
713        })), None)?;
714    }
715  }
