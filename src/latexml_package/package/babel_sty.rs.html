1//! babel.sty — multilingual support
2//!
3//! Perl: babel.sty.ltxml (30 lines) — `InputDefinitions('babel', noltxml=>1)`.
4//! Our Rust port carries a thin orchestration layer on top of the raw babel
5//! load. With the @currname leakage fix in commit 56b0c35d2, babel's own
6//! option pipeline (and therefore its entire language-loading / shorthand /
7//! captions story) now works end-to-end. Only two small workarounds remain
8//! here: pre-allocating `\l@polutonikogreek` for older TeX Live builds that
9//! don't include it in the kernel dump, and setting DOCUMENT_LANGUAGE +
10//! `\bbl@main@language` globally (babel's own raw-load path may resolve
11//! main to a language whose .ldf happened to run last — not always the
12//! user's intended last option).
13use crate::prelude::*;
14
15#[rustfmt::skip]
16LoadDefinitions!({
17  // \l@polutonikogreek: allocate if not present in kernel dump (newer TeXLive
18  // includes it, older may not).
19  RawTeX!(r"\expandafter\ifx\csname l@polutonikogreek\endcsname\relax\newlanguage\l@polutonikogreek\fi");

21  // `activeacute` was historically an option to babel-spanish.ldf that
22  // activated `'` as an active accent. Some 1995-2010 papers wrote
23  // `\usepackage[<lang>,activeacute]{babel}` treating it as a language;
24  // modern babel doesn't recognize it as a language and `\InputIfFileExists
25  // {activeacute.ldf}` silently fails (no on-disk file in TL). Babel then
26  // proceeds and `\selectlanguage{...}` errors with "haven't defined the
27  // language 'activeacute' yet". We pre-register `\l@activeacute` plus the
28  // empty `<lang>` hooks so `\selectlanguage` resolves silently — actual
29  // active-acute-on-quote semantics are not reproduced (most affected
30  // papers only set this option as a side effect of preamble copy-paste).
31  RawTeX!(r"%
32    \expandafter\ifx\csname l@activeacute\endcsname\relax
33      \newlanguage\l@activeacute
34    \fi
35    \providecommand\captionsactiveacute{}%
36    \providecommand\extrasactiveacute{}%
37    \providecommand\noextrasactiveacute{}%
38    \providecommand\dateactiveacute{}");

40  // \bbl@opt@safe = \@empty inhibits some risky redefinitions in babel.
41  // Mirror Perl LaTeXML/lib/LaTeXML/Package/babel.def.ltxml: `Let('\bbl@opt@safe', '\@empty')`.
42  // Without this, babel.sty's option processing enters an infinite loop on
43  // some redefinition paths (verified: triggers token_limit:Timeout 100M).
44  RawTeX!(r"\let\bbl@opt@safe\@empty");

46  // Pre-define `\bbl@main@language` as `english` so papers that load babel
47  // transitively (e.g. via aastex62 → revtex4 → bibstyles → babel chain
48  // without any explicit `\usepackage[<lang>]{babel}`) don't error with
49  // "Token \bbl@main@language is not defined" when `\selectlanguage` or
50  // `\lx@babel@activate@mainlang` later expands it. The real `\ldf@finish`
51  // overrides this when a language .ldf actually runs. Witness 2301.13322
52  // (aastex62 + lipsum + blindtext, no explicit babel options).
53  RawTeX!(r"\providecommand\bbl@main@language{english}");
54
55  InputDefinitions!("babel", noltxml => true, extension => Some(Cow::Borrowed("sty")));
56
57  // Sets DOCUMENT_LANGUAGE and force-sets \bbl@main@language from
58  // \opt@babel.sty so babel's `\AtBeginDocument{\selectlanguage{\bbl@main
59  // @language}}` picks up the user's intended main language (the last
60  // option), not whichever .ldf's \ldf@finish happened to run last.
61  // Everything else (captions activation, active-char shorthands, port
62  // dispatching) is handled end-to-end by babel's own chain.
63  DefPrimitive!("\\lx@babel@activate@mainlang", {
64    let main = do_expand(T_CS!("\\bbl@main@language"))
65      .map(|t| t.to_string()).unwrap_or_default();
66    let opt_babel = do_expand(Tokenize!(r"\csname opt@babel.sty\endcsname"))
67      .map(|t| t.to_string()).unwrap_or_default();
68    // Modern babel accepts `main=<lang>` to pin the document main language
69    // (driver: 2109.00402 \usepackage[main=english]{babel}). Two cases:
70    //   - `main=<lang>`  → use <lang> directly (highest priority)
71    //   - bare positional option (no `=`) → treat as a language candidate
72    //   - `<key>=<value>` for any other key (e.g. `shorthands=off`,
73    //     `provide=*`) → SKIP, it's an option not a language. Driver:
74    //     2001.00747 `\usepackage[english, shorthands=off]{babel}` was
75    //     selecting "off" as the language because we promoted any value-
76    //     half of any key=value option.
77    let main_kv = opt_babel.split(',')
78      .map(str::trim)
79      .find_map(|s| s.strip_prefix("main=").map(str::trim).map(str::to_string));
80    // Filter: drop key=value options AND babel-language MODIFIERS (sub-options
81    // recognised by a single .ldf rather than a language by themselves).
82    // Modifiers we know about:
83    //   `es-*` (babel-spanish: es-tabla, es-cuadro, es-noindentfirst, …)
84    // Driver: 2102.11084 `\usepackage[spanish, es-tabla]{babel}` selected
85    // "es-tabla" as the language → "haven't defined the language 'es-tabla'"
86    // GenericError.
87    // ALSO drop babel's own bare KEYWORD options — `\DeclareOption{<kw>}` in
88    // babel.sty that are package switches, NOT languages (so babel consumes
89    // them via their `\ds@` handler and never treats them as a language to
90    // load/select). Without this, `\usepackage[english,strings]{babel}`
91    // selected the LAST bare option "strings" as the main language →
92    // `\selectlanguage{strings}` → "You haven't defined the language
93    // 'strings'". babel.sty L296/L336-379 (the no-`=` ones). Driver
94    // 2006.10240 (`[english,strings]`).
95    const BABEL_KEYWORD_OPTS: &[&str] = &[
96      "base", "showlanguages", "KeepShorthandsActive", "activeacute",
97      "activegrave", "debug", "noconfigs", "silent", "strings", "nocase",
98      "leqno", "fleqn",
99    ];
100    let is_lang_candidate = |s: &str| -> bool {
101      !s.is_empty() && s != "nil" && !s.contains('=')
102        && !s.starts_with("es-")
103        && !BABEL_KEYWORD_OPTS.contains(&s)
104    };
105    let pkg_last = main_kv.clone().unwrap_or_else(|| {
106      opt_babel.split(',').map(str::trim).rfind(|s| is_lang_candidate(s)).unwrap_or_default().to_string()
107    });
108    // Pick the active main-lang. The non-trivial case is when the user's
109    // OPTION NAME differs from the .ldf's CANONICAL language name —
110    // e.g. `\usepackage[russianb]{babel}` loads russianb.ldf which calls
111    // `\ldf@finish{russian}`. babel's `\select@language` later needs
112    // `\l@<canonical>` (russian), so we must use the canonical name
113    // when an alias is in play. But when our Rust binding for a `.ldf`
114    // (e.g. french_ldf) bypasses raw load, no `\main@language` runs,
115    // and `\bbl@main@language` retains whatever the FIRST option's
116    // raw-loaded .ldf set — which is wrong (user wants the LAST option).
117    //
118    // Heuristic: trust `main` (the `\bbl@main@language` value) only
119    // when it maps to the same ISO code as `pkg_last`. That captures
120    // the canonical-alias case (russianb → russian, ukrainianb →
121    // ukrainian, brazilian → brazil) while still falling through to
122    // `pkg_last` for the no-Rust-ldf-binding case (where `main`
123    // points at the first option, not the user's intended last one).
124    //
125    // Witnesses: 2312.08012 (russianb → russian alias, needed canonical),
126    // tests/babel/elsart_keyword_brace_form (`[english,french]` →
127    // french_ldf binding doesn't update main, but user wants french).
128    let same_alias_class = !main.is_empty() && !pkg_last.is_empty()
129      && main != pkg_last
130      && babel_support_sty::babel_language_to_iso(&main)
131         == babel_support_sty::babel_language_to_iso(&pkg_last)
132      && babel_support_sty::babel_language_to_iso(&main).is_some();
133    // A bare `\usepackage{babel}` whose language is a CLASS option
134    // (`\documentclass[ngerman]{scrlttr2}`, letter-copy-test): `\opt@babel.sty`
135    // is empty. babel took the main language from the class options
136    // (TL 2025 babel.sty:4197-4245) and loaded it last, prepending it to `\bbl@loaded`
137    // (:4129-4133); that `.ldf` is our binding, which never runs `\main@language`,
138    // so `\bbl@main@language` kept the english default: the document got
139    // `xml:lang="en"`, English captions, and blindtext's English text.
140    let loaded_first = if pkg_last.is_empty() {
141      // babel.sty:4129 `\let\bbl@loaded\@empty`: always defined here.
142      do_expand(T_CS!("\\bbl@loaded"))
143        .map(|t| t.to_string())
144        .unwrap_or_default()
145        .split(',')
146        .map(str::trim)
147        .find(|s| is_lang_candidate(s) && babel_support_sty::babel_language_to_iso(s).is_some())
148        .unwrap_or_default()
149        .to_string()
150    } else {
151      String::new()
152    };
153    let lang = if let Some(m) = main_kv {
154      m
155    } else if same_alias_class {
156      // canonical aliases — use the canonical name from \ldf@finish
157      main
158    } else if !pkg_last.is_empty() {
159      pkg_last
160    } else if !loaded_first.is_empty() {
161      loaded_first
162    } else if main != "nil" && !main.is_empty() {
163      main
164    } else {
165      return Ok(vec![]);
166    };
167    let iso = babel_support_sty::babel_language_to_iso(&lang);
168    if let Some(code) = iso {
169      assign_value("DOCUMENT_LANGUAGE",
170        Stored::from(code.to_string()), Some(Scope::Global));
171      merge_font(Font { language: Some(Cow::Owned(code.to_string())), ..Font::default() });
172    }
173    // Set \bbl@main@language so babel's AtBeginDocument
174    // \selectlanguage{\bbl@main@language} picks up the canonical name.
175    def_macro(T_CS!("\\bbl@main@language"), None,
176      Tokens!(Explode!(lang.clone())),
177      Some(ExpandableOptions { scope: Some(Scope::Global), ..ExpandableOptions::default() }))?;
178    // babel.sty L1136-1142 (\main@language) → L828 (\bbl@patterns):
179    // real babel assigns `\language=\l@<main>` DURING package load, so a
180    // preamble `\iflanguage{english}{..}{..}` under `[italian]` takes the
181    // FALSE branch (\language=\l@italian ≠ \l@english=0). We never set
182    // \language, so every non-English doc mis-branched preamble language
183    // tests (toptesi topfront manuals: 34 undefined-CS errors from an
184    // \iflanguage{english} block real LaTeX skips). Allocate \l@<lang> if
185    // missing, then assign.
186    unread(Tokenize!(TeXString::assembled(s!(
187      "\\ifcsname l@{lang}\\endcsname\\else\
188       \\expandafter\\newlanguage\\csname l@{lang}\\endcsname\\fi\
189       \\language=\\csname l@{lang}\\endcsname\\relax"
190    ))));
191  });
192  // Run mainlang at load time so DOCUMENT_LANGUAGE is set before
193  // \begin{document} opens (and base_schema's after_open reads it).
194  RawTeX!(r"\lx@babel@activate@mainlang");

196  // English-family caption/date/extras hook backfill. Modern babel's .ini
197  // path defines the per-variant `\captions<v>`/`\date<v>`/`\extras<v>`
198  // hooks only for the variant(s) whose .ini actually loaded (e.g.
199  // babel-british for `[british]`). A paper listing SEVERAL english
200  // variants — `\usepackage[british,USenglish]{babel}` — then invokes a
201  // variant whose .ini never ran (`\dateUSenglish`) or the `\captionsenglish`
202  // base, erroring at `\selectlanguage`/option dispatch. The classic .ldf
203  // loaders (`babel_lang_stubs::load_*`) also miss this because the .ini
204  // path bypasses them, and `english.sty`'s aliasing loop only runs for
205  // `\usepackage{english}`, not a direct `\usepackage[...]{babel}`. Backfill
206  // with `\@ifundefined` guards: never overrides a real definition; captions
207  // stay English (our HTML default); `\date<v>` aliases to `\dateenglish`
208  // (keeps `\today` faithful). Runs after babel's own option processing
209  // (InputDefinitions above), before the invocation in later preamble.
210  // Witness arXiv:1508.06150 (`\usepackage[british,USenglish]{babel}`); Perl rc=0.
211  // NB: no \makeatletter/\makeatother — RawTeX already digests with `@` as a
212  // letter; emitting \makeatother here would leave `@` catcode-12 globally and
213  // break babel's later `\l@<lang>`/`\bbl@…` parsing (manifested as a spurious
214  // "haven't defined the language" error).
215  RawTeX!(r"%
216    \@ifundefined{dateenglish}{\@namedef{dateenglish}{}}{}%
217    \@for\lx@bbl@engtmp:={english,USenglish,UKenglish,american,british,canadian,australian,newzealand}\do{%
218      \@ifundefined{captions\lx@bbl@engtmp}{\expandafter\let\csname captions\lx@bbl@engtmp\endcsname\@empty}{}%
219      \@ifundefined{extras\lx@bbl@engtmp}{\expandafter\let\csname extras\lx@bbl@engtmp\endcsname\@empty}{}%
220      \@ifundefined{noextras\lx@bbl@engtmp}{\expandafter\let\csname noextras\lx@bbl@engtmp\endcsname\@empty}{}%
221      \@ifundefined{date\lx@bbl@engtmp}{\expandafter\let\csname date\lx@bbl@engtmp\endcsname\dateenglish}{}%
222      \@ifundefined{l@\lx@bbl@engtmp}{\expandafter\let\csname l@\lx@bbl@engtmp\endcsname\l@english}{}}");
223  // ^ Also alias the hyphenation language register `\l@<variant>` to
224  // `\l@english` when undefined. The `.ini` path defines `\l@british` only for
225  // the variant whose `.ini` actually ran; a paper that loads several english
226  // variants then does `\selectlanguage{british}` → `\bbl@iflanguage{british}`
227  // tests `\ifx\csname l@british\endcsname\relax` and, finding it relax, errors
228  // "You haven't defined the language 'british' yet". english.ldf normally
229  // `\let`s every variant's `\l@` to `\l@english` (british uses English
230  // hyphenation); backfill the same. Witness 1508.06150.

232  // `\shorthandoff{<chars>}` / `\shorthandon{<chars>}`: babel shorthand
233  // toggles. Babel's raw implementation (babel.sty L1492-1496) iterates the
234  // argument switching each active-char shorthand, but ALSO fires
235  // `\PackageError{babel}{I can't switch '<c>' on or off--not a shorthand}`
236  // for any char that isn't a registered shorthand. Our language stubs
237  // (`install_lang_stub`) don't install the full shorthand machinery, so the
238  // raw impl errored on every call — which is why this was a no-op.
239  //
240  // But a pure no-op is WRONG: the *observable* effect of `\shorthandoff` is
241  // the CATCODE change (active → other), which genuinely affects tokenization,
242  // not just typesetting. French babel makes `;:!?` active; leaving them
243  // active when an author explicitly turned them off breaks downstream parsers
244  // that expect the literal chars — e.g. xy-pic `\xymatrix @!=8mm{…}` reads
245  // `@!=` expecting `!` at catcode 12; an active `!` derails the xymatrix so
246  // its `^`/`_` reach the math handler in text mode → 100+ `^`/`_` "can only
247  // appear in math mode" errors (witness arXiv:1804.10128, French + xy-pic).
248  //
249  // So apply the catcode change (the part our pipeline observes) and skip the
250  // error (the part the stub gap can't satisfy): `\shorthandoff` → OTHER(12),
251  // `\shorthandon` → ACTIVE(13). Local to the current group, mirroring TeX's
252  // local `\catcode`. Still correct for the defensive-call witness
253  // arXiv:1912.08056 (`\def\diag{\shorthandoff{;:!?}…}`) — catcode 12 is the
254  // intended effect, no error.
255  /// The single characters of a `\shorthandoff`/`\shorthandon` argument.
256  fn shorthand_chars(chars: Tokens) -> Vec<char> {
257    chars
258      .unlist()
259      .into_iter()
260      .filter(|tok| tok.get_catcode() != Catcode::SPACE)
261      .filter_map(|tok| {
262        tok.with_str(|s| {
263          let mut it = s.chars();
264          match (it.next(), it.next()) {
265            (Some(c), None) => Some(c),
266            _ => None,
267          }
268        })
269      })
270      .collect()
271  }
272  /// A meaning that is there and is not `\relax` (a `\csname` built once leaves `\relax`).
273  fn is_set(cs: &Token) -> bool {
274    match lookup_meaning(cs) {
275      None => false,
276      Some(meaning) => lookup_meaning(&T_CS!("\\relax")).is_none_or(|relax| meaning != relax),
277    }
278  }
279  // babel switches only its shorthands (babel.sty:1437-1440 `\bbl@ifunset{bbl@active@…}` — any
280  // other character is "not a shorthand", unchanged): off, an active character becomes other
281  // (catcode 12); on, a character with an active meaning becomes active again. Demoting every
282  // character broke `^` (a superscript, catcode 7) after `\shorthandoff*{~^}` (babel.dtx:3786-3794).
283  // babel's "not a shorthand" error (babel.def:1558, errbabel.def:93-97) is raised for a character
284  // that cannot be one: in the body, where our language bindings have made their shorthands active,
285  // one that is neither active nor of catcode 12 (a shorthand our bindings deactivated, or one a
286  // language switch turned off, is 12) and has no babel record (`\bbl@oricat@<c>`), such as `^`.
287  // In the preamble the bindings' shorthands are not active yet, so no character is judged there.
288  // Repro babel-lang/shorthandoff_star_reads_the_star (babel.dtx:3786-3794's `\shorthandoff*{~^}`);
289  // RED babel-lang/shorthandoff_quote_in_english_is_no_shorthand (a catcode-12 non-shorthand passes).
290  fn shorthand_off(chars: Tokens) -> Result<()> {
291    for c in shorthand_chars(chars) {
292      if lookup_catcode(c) == Some(Catcode::ACTIVE) {
293        assign_catcode(c, Catcode::OTHER, Some(Scope::Local));
294      } else {
295        report_not_a_shorthand(c)?;
296      }
297    }
298    Ok(())
299  }
300  /// errbabel.def:93-97 `not-a-shorthand-b`, through `\PackageError{babel}` (the
301  /// `latex:(babel)` diagnostic of make_message, as Perl's raw babel reports it).
302  fn report_not_a_shorthand(c: char) -> Result<()> {
303    let cs = T_CS!(s!("\\bbl@oricat@{c}"));
304    if lookup_bool("inPreamble")
305      || matches!(lookup_catcode(c), Some(Catcode::OTHER) | Some(Catcode::ACTIVE))
306      || is_set(&cs)
307    {
308      return Ok(());
309    }
310    Error!(
311      "latex",
312      "(babel)",
313      s!(
314        "Package babel Error: I can't switch '{c}' on or off--not a shorthand\nThis character is not a shorthand. Maybe you made\na typing mistake? I will ignore your instruction."
315      )
316    );
317    Ok(())
318  }
319  // babel.sty:1428-1430 `\shorthandoff` is `\@ifstar{\bbl@shorthandoff\tw@}{\bbl@shorthandoff\z@}`:
320  // the starred form restores each character's original catcode and meaning, as babel recorded
321  // them when it made the character active (`\bbl@oricat@<c>`, `\bbl@oridef@<c>`, :1449-1454,
322  // :1210). A shorthand our language bindings activated has no such record; its original catcode
323  // is LaTeX's 12 (`:;!?"`). Read without the star, `*` was the argument and `{:}` typeset.
324  DefPrimitive!("\\shorthandoff OptionalMatch:* {}", sub[(star, chars)] {
325    if star.is_none() {
326      shorthand_off(chars)?;
327      return Ok(Vec::new());
328    }
329    let mut restore: Vec<Token> = Vec::new();
330    for c in shorthand_chars(chars) {
331      let oricat = T_CS!(s!("\\bbl@oricat@{c}"));
332      if is_set(&oricat) {
333        restore.push(oricat);
334        let oridef = T_CS!(s!("\\bbl@oridef@{c}"));
335        if is_set(&oridef) {
336          restore.push(oridef);
337        }
338      } else if lookup_catcode(c) == Some(Catcode::ACTIVE) {
339        assign_catcode(c, Catcode::OTHER, Some(Scope::Local));
340      } else {
341        report_not_a_shorthand(c)?;
342      }
343    }
344    if !restore.is_empty() {
345      // The records are `\catcode…\relax` and `\let…` assignments: run them as TeX.
346      let _ = digest(Tokens::new(restore))?;
347    }
348  });
349  DefPrimitive!("\\shorthandon{}", sub[(chars)] {
350    for c in shorthand_chars(chars) {
351      if lookup_meaning(&T_ACTIVE!(c)).is_some() {
352        assign_catcode(c, Catcode::ACTIVE, Some(Scope::Local));
353      } else {
354        report_not_a_shorthand(c)?;
355      }
356    }
357  });
