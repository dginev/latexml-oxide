1//! wrapstuff.sty — text wrapped around a box (the expl3 successor of wrapfig).
2//!
3//! Raw, the package digests the environment body into a box register
4//! (wrapstuff.sty:312-316) and places it from the LaTeX2e paragraph hooks
5//! `para/begin`/`para/end` (:334, :539-552, :1905-1939), which neither
6//! LaTeXML models (`\everypar` is inert: TeX_Paragraph.pool.ltxml:42,
7//! tex_paragraph.rs:70): the box is built and never emitted — caption and
8//! content lost at 0 errors (wrapstuff-doc-en, latex-via-exemplos ex15;
9//! `~/data/pk_agents/w23/regr85/wrapstuff/NOTES.md`; raw, the content leaked
10//! inline as a bare box while the float structure and the caption were
11//! lost). The placement itself
12//! needs `\prevgraf`, `\parshape` and `\vsplit` arithmetic no flow model
13//! carries, so the environment is bound as the inline float wrapfig's
14//! `wrapfigure`/`wraptable` are (`wrapfig_sty.rs`; Perl binds wrapfig the same
15//! way, wrapfig.sty.ltxml:20-48): `type` (:2399) picks `ltx:table` /
16//! `ltx:figure` (else a plain `ltx:float`), `l`/`r` (:2404-2405; the default
17//! ratio `\c_one_fp`, :2461, is the right margin) the `float` side, `width`
18//! (:2395) the width. The raw package is NOT loaded (bindings outrank raw).
19use crate::{
20  engine::latex_constructs::{after_float, before_float},
21  prelude::*,
22};
23
24#[rustfmt::skip]
25LoadDefinitions!({
26  // wrapstuff.sty:2523 reads its options with `\ProcessKeyOptions`, which marks them processed: `\@curroptions` stays as it was.
27  key_options_processed()?;
28  // The package's two public commands (wrapstuff.sty:2516-2521):
29  // `\wrapstuffset{keys}` sets document-level defaults (`\keys_set:nn
30  // {wrapstuff}{#1}`) — all typographic here, consumed; `\wrapstuffclear` =
31  // `\par \__wstf_clear:` ends the wrapped paragraph, as wrapfig's `\WFclear`.
32  def_macro_noop("\\wrapstuffset{}")?;
33  DefMacro!("\\wrapstuffclear", "\\par");
34  DefEnvironment!("{wrapstuff} OptionalKeyVals",
35    "?#istable(<ltx:table xml:id='#id' inlist='#inlist' float='#float' ?#width(width='#width')>#tags#body</ltx:table>)\
36     (?#isfigure(<ltx:figure xml:id='#id' inlist='#inlist' float='#float' ?#width(width='#width')>#tags#body</ltx:figure>)\
37     (<ltx:float xml:id='#id' inlist='#inlist' float='#float' ?#width(width='#width')>#tags#body</ltx:float>))",
38    mode => "internal_vertical",
39    // The begin code opens a TeX group of its own: wrapstuff.sty:312 `\hbox_gset:Nw`.
40    tex_group => { TexGroup::Box },
41    after_digest_begin => sub[whatsit] {
42      let (ftype, side, width) = wrapstuff_keys(whatsit);
43      whatsit.set_property("float", Stored::String(pin(side)));
44      match ftype.as_str() {
45        "table" => { whatsit.set_property("istable", Stored::Bool(true)); },
46        "figure" => { whatsit.set_property("isfigure", Stored::Bool(true)); },
47        _ => {},
48      }
49      // Only a typed box is a captioned float (wrapstuff.sty:2399 `type`
50      // feeds `\@captype`); an untyped one is a plain wrapped box.
51      if !ftype.is_empty() {
52        whatsit.set_property("isfloat", Stored::Bool(true));
53        before_float(&ftype, None)?;
54        engine::latex_constructs::reset_float_box()?;
55      }
56      if let Some(w) = width {
57        set_wrap_width(whatsit, w);
58      }
59    },
60    after_digest => sub[whatsit] {
61      if whatsit.get_property("isfloat").is_some() {
62        after_float(whatsit);
63      }
64      Ok(Vec::new())
65    }
66  );
67});

69/// `(type, float side, width)` from the environment's keyvals.
70fn wrapstuff_keys(whatsit: &Whatsit) -> (String, &'static str, Option<Dimension>) {
71  let mut ftype = String::new();
72  let mut side = "right";
73  let mut width = None;
74  if let Some(arg) = whatsit.get_arg(1)
75    && let DigestedData::KeyVals(ref kvs) = *arg.data()
76  {
77    if let Some(t) = kvs.get_value("type") {
78      ftype = t.to_string().trim().to_string();
79    }
80    if kvs.get_value("l").is_some() {
81      side = "left";
82    }
83    if kvs.get_value("r").is_some() {
84      side = "right";
85    }
86    if let Some(w) = kvs.get_value("width") {
87      let spec = w.to_string();
88      if let Ok(d) = spec.trim().parse::<Dimension>() {
89        width = Some(d);
90      }
91    }
92  }
93  (ftype, side, width)
94}

96/// As wrapfig's: the box width becomes the text width inside and a
97/// percentage `width` on the element.
98fn set_wrap_width(whatsit: &mut Whatsit, width: Dimension) {
99  let v = width.value_of();
100  if v == 0 {
101    return;
102  }
103  let rv: RegisterValue = Dimension::new(v).into();
104  let _ = assign_register("\\hsize", rv.clone(), None, Vec::new());
105  let _ = assign_register("\\columnwidth", rv.clone(), None, Vec::new());
106  let _ = assign_register("\\linewidth", rv, None, Vec::new());
107  let Some(tw) = lookup_dimension("\\textwidth") else {
108    return;
109  };
110  let tw_sp = tw.value_of();
111  if tw_sp == 0 {
112    return;
113  }
114  let pct = (100 * v) / tw_sp;
115  whatsit.set_property("width", Stored::from(s!("{pct}%")));
116}
