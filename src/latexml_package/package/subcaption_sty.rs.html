8/// Perl subcaption.sty.ltxml L66/76/86/96: `properties => sub { (width => $_[2]) }`.
9/// Perl sets this on ALL four sub-float envs ({subfigure}, {subfigure*},
10/// {subtable}, {subtable*}); the Rust-only {subcaptionblock}/{subcaptionblock*}
11/// aliases inherit the same semantics. Records the sub-float's `{Dimension}`
12/// argument (the last) as the box's `width` property, so
13/// `arrange_panels_and_breaks` can compute the per-row layout from actual panel
14/// widths (without it, a panel reports its natural content width — the full
15/// float width — and every panel starts its own row; arXiv 2605.00347). Mirrors
16/// Perl storing the digested Dimension object.
17///
18/// The panel width: the environment's last argument, minipage's `{width}` (subcaption.sty:70-108 hands its
19/// `[pos][height][inner]{width}` to a minipage, whose `\setlength\hsize{#4}` reads it, latex.ltx `\@iiiminipage`).
20/// Bound as `[]{Dimension}` (Perl subcaption.sty.ltxml:60, :70), `\begin{subfigure}[c][0pt][c]{…}` read `[` as the
21/// width (2605.06598, 2605.21425, 2606.16001).
22pub(crate) fn subcaption_width_props(args: &[Option<Digested>]) -> Result<SymHashMap<Stored>> {
23  let mut props: SymHashMap<Stored> = SymHashMap::default();
24  if let Some(w) = args
25    .last()
26    .and_then(|a| a.as_ref())
27    .and_then(|a| Dimension::spec_to_f64(&a.to_string()).ok())
28    // #6903: a non-positive Dimension (e.g. `\subcaptionbox`'s `{0pt}` default,
29    // subcaption_sty L219) must NOT pin the panel width to 0 — that reads as a
30    // zero-width box in `arrange_panels` and the panels never share a row. Skip
31    // it so `panel_width` falls back to the panel's natural content width.
32    .filter(|w| *w > 0.0)
33  {
34    props.insert("width", Stored::Dimension(Dimension::new_f64(w)));
35  }
36  Ok(props)
37}

39/// A sub-float environment is subcaption's minipage (subcaption.sty:70-92 `\subcaption@iiminipage`, `\minipage[#1]#2{#3}`):
40/// its width argument (the fourth) sets the width registers inside it, `\linewidth` among them (`set_minipage_width`;
41/// KNOWN_PERL_ERRORS #1028, witness 2509.19464 A1.F6). `\subcaptionbox`, which opens one with its optional width, sets
42/// them only when given a width (caption.sty:460-489 `\parbox{#6}`); without, its content is measured at the outer
43/// width (:462-464 `\hbox{#4}`), so the `0pt` stand-in (`\@@subcaptionbox`) changes none (html_feedback #6903's
44/// `\subcaptionbox{}{\includegraphics[width=0.48\linewidth]{…}}`). Its box is a `\parbox` (caption.sty:473, :482), which
45/// sets `\hsize` and `\linewidth` only (latex.ltx `\@iiiparbox`), `\textwidth` and `\columnwidth` staying the outer ones
46/// (rev78d probe sfw case G); `\@@subcaptionbox` marks its sub-float `\lx@subcaption@parbox`.
47fn subfloat_minipage_width(whatsit: &mut Whatsit) -> Result<()> {
48  let parbox = lookup_bool("lx@subcaption@parbox");
49  if parbox {
50    // This sub-float's alone: one in its body is a minipage again.
51    assign_value("lx@subcaption@parbox", false, Some(Scope::Local));
52  }
53  if let Some(width) = whatsit.get_arg(4).map(|w| w.value_of()).filter(|w| *w > 0) {
54    if parbox {
55      let rv: RegisterValue = Dimension::new(width).into();
56      for register in ["\\hsize", "\\linewidth"] {
57        assign_register(register, rv.clone(), None, Vec::new())?;
58      }
59    } else {
60      set_minipage_width(Dimension::new(width))?;
61    }
62  }
63  Ok(())
64}

389  // two are `\subcaptionbox[{#1}]{#2}`; the sub-float follows `\@captype`, so a `\subfloat` in a
390  // table is a subtable. Perl's `\subfloat[][]{}` (subcaption.sty.ltxml:104, witness 2111.00007)
391  // read a lone optional as the list entry, so `\subfloat[Caption]{…}` came out uncaptioned (KPE
392  // #323; fixture svg_subfloat_2563). Without an optional subcaption sets a `\phantomcaption`
393  // (:293-300), as here (`\lx@subcaption@subfloat@phantom`; Perl printed an empty caption).
394  // `\columnwidth` stands in for the box's natural width (Perl L102-103).
395  // subcaption provides `\subfloat` only at `\begin{document}` (subcaption.sty:262-276
396  // `\caption@AtBeginDocument{…\providecommand*\subfloat{\subcaption@subfloat}}`), so a preamble's own
397  // `\newcommand{\subfloat}` (2511.09162) or subfig's stays; defined at load, the binding's took the author's away
398  // (Perl subcaption.sty.ltxml:104 the same, KNOWN_PERL_ERRORS #965). Guard:
399  // `perfect_kernel_batch73a::subcaption_subfloat_is_provided_at_begin_document`.
400  DefMacro!("\\lx@subcaption@subfloat",
401    "\\kernel@ifnextchar[\\lx@subcaption@subfloat@list\\lx@subcaption@subfloat@phantom");
402  RawTeX!(r"\AtBeginDocument{\providecommand*\subfloat{\lx@subcaption@subfloat}}");
403  DefMacro!("\\lx@subcaption@subfloat@list[]",
404    "\\kernel@ifnextchar[{\\lx@subcaption@subfloat@caption{#1}}{\\lx@subcaption@subfloat@@{}{#1}}");
405  DefMacro!("\\lx@subcaption@subfloat@caption{}[]", "\\lx@subcaption@subfloat@@{#1}{#2}");
406  DefMacro!("\\lx@subcaption@subfloat@@{}{}{}", sub[(list, caption, body)] {
407    subfloat_tokens(list, Some(caption), body)
408  });
409  // Without an optional, subcaption's `\subfloat` boxes the body with a `\phantomcaption`
410  // (subcaption.sty:293-300): the sub-float is numbered and tagged, but prints no caption and
411  // enters no list. `\subfloat[]{…}` (an empty caption) still prints its "(b)".
412  DefMacro!("\\lx@subcaption@subfloat@phantom{}", sub[(body)] {
413    subfloat_tokens(Tokens!(), None, body)
414  });
415
416  //======================================================================
417  // \subcaptionbox — delegates to sub<captype> environment
418  DefMacro!("\\subcaptionbox",
419    "\\expandafter\\@@subcaptionbox\\expandafter{\\@captype}"
420  );
421  // The sub-float is the box's `\parbox` (`subfloat_minipage_width`).
422  DefPrimitive!("\\lx@subcaption@parbox", {
423    assign_value("lx@subcaption@parbox", true, Some(Scope::Local));
424  });
425  DefMacro!("\\@@subcaptionbox{} []{} Optional:0pt []{}",
426    "\\begingroup\\lx@subcaption@parbox\\csname sub#1\\endcsname{#4}\
427     #6\
428     \\caption{#3}\
429     \\ifx.#2.\\else\\lx@subcaption@addinlist{#2}\\fi\
430     \\csname endsub#1\\endcsname\\endgroup"
431  );
432
433  //======================================================================
434  // Perl L116-117: \lx@subcaption@addinlist — `^ inlist='#1'` floats from the CURRENT node
435  // (the subfigure) to the first one that can take `inlist` (Document.pm:1080-1092). Setting
436  // it on the node's parent instead renamed the enclosing figure's list, so every figure
437  // holding a `\subfloat[entry]{…}` or `\subcaptionbox[entry]{…}{…}` left the List of Figures.
438  DefConstructor!("\\lx@subcaption@addinlist{}", "^ inlist='#1'");
439
440  //======================================================================
441  // \subref — delegates to \ref
442  DefMacro!("\\subref OptionalMatch:* Semiverbatim", "\\ref{#2}");
443
444});

446/// `{list}{caption}{body}` in a `sub<type>` environment. The type is `\@captype` less a leading
447/// `sub` (inside a `{subfigure}` it is `subfigure`), as Perl's `\subcaption` resolves it (L50-53);
448/// `figure` — Perl's only choice — outside a float, or when no `sub<type>` environment exists (a
449/// `\newfloat` type). No `caption` is subcaption's phantom one (subcaption.sty:293-300): a
450/// `\phantomcaption` where the caption would be.
451fn subfloat_tokens(list: Tokens, caption: Option<Tokens>, body: Tokens) -> Result<Tokens> {
452  let mut ctype = String::from("figure");
453  if has_meaning(&T_CS!("\\@captype")) {
454    let captype = do_expand(Tokens!(T_CS!("\\@captype")))?.to_string();
455    let captype = captype.trim();
456    let base = captype.strip_prefix("sub").unwrap_or(captype);
457    if !base.is_empty()
458      && (is_defined(&format!("\\sub{base}")) || is_defined(&format!("\\begin{{sub{base}}}")))
459    {
460      ctype = base.to_string();
461    }
462  }
463  let env = Tokens!(T_BEGIN!(), Explode!(s!("sub{}", ctype)), T_END!());
464  // The `{\columnwidth}` is the binding's, not a width the source states: subcaption's `\subfloat` boxes its body
465  // at its natural width (subcaption.sty:293-300 `\sbox\subcaption@box`), so the panel grid rule keeps nothing of it
466  // (`\lx@panel@unstated`; it made a `figure*`'s first `\subfloat` 0 wide against a two-column class's real column,
467  // review rev78x finding 1: 2602.06569 S4.F1/F2, 1902.05392 S3.F3), and its body's widths are the float's.
468  let mut tokens = vec![
469    T_CS!("\\begingroup"),
470    T_CS!("\\lx@panel@unstated"),
471    T_CS!("\\begin"),
472  ];
473  tokens.extend(env.clone().unlist());
474  tokens.extend([
475    T_BEGIN!(),
476    T_CS!("\\columnwidth"),
477    T_END!(),
478    T_CS!("\\lx@panel@stated"),
479  ]);
480  tokens.extend(body.unlist());
481  match caption {
482    Some(caption) => {
483      tokens.extend([T_CS!("\\caption"), T_BEGIN!()]);
484      tokens.extend(caption.unlist());
485      tokens.push(T_END!());
486    },
487    None => tokens.push(T_CS!("\\phantomcaption")),
488  }
489  if !list.is_empty() {
490    tokens.extend([T_CS!("\\lx@subcaption@addinlist"), T_BEGIN!()]);
491    tokens.extend(list.unlist());
492    tokens.push(T_END!());
493  }
494  tokens.push(T_CS!("\\end"));
495  tokens.extend(env.unlist());
496  tokens.push(T_CS!("\\endgroup"));
497  Ok(Tokens::new(tokens))
498}
