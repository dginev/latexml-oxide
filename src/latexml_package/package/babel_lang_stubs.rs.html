1//! babel-language stubs for `.ldf` files that aren't installed in
2//! minimal TeXLive environments.
3//!
4//! When babel-italian (or babel-spanish, babel-portuges, …) is
5//! missing on disk, babel.sty L4175's `\InputIfFileExists{<lang>.ldf}`
6//! fails and babel errors with "Unknown option '`<lang>`'. Either you
7//! misspelled it or the language definition file `<lang>`.ldf was not
8//! found".
9//!
10//! Each stub here:
11//!   - registers via `lib.rs` as `<lang>.ldf` so find_file resolves to our binding rather than the
12//!     missing file;
13//!   - allocates `\l@<lang>` and defines empty `\captions<lang>` / `\extras<lang>` /
14//!     `\noextras<lang>` / `\date<lang>` hooks so babel's `\selectlanguage` chain runs cleanly.
15//!
16//! The actual ISO-639 → xml:lang mapping happens at
17//! `\selectlanguage` time via
18//! `babel_support_sty::babel_language_to_iso`. We don't reproduce
19//! the per-language typographic conventions (punctuation, dates,
20//! etc.) — they're a typesetting-only concern, irrelevant for
21//! XML/HTML output.
22//!
23//! Cluster (recent stages): italian 7, spanish 5, brazil 5, vietnamese
24//! 3, portuguese 3, brazilian 2, polish 2, romanian 2, icelandic 2,
25//! czech 2, turkish 1, slovene 1, portuges 1, farsi 1, dutch 1,
26//! arabic 1 ≈ 38 papers.
27use crate::prelude::*;

29/// The stub is a FALLBACK, not a binding of the language: when the real
30/// `<lang>.ldf` is installed (full TeX Live), raw-load it and return `true`
31/// — the registry entry that routes `<lang>.ldf` here would otherwise
32/// SHADOW the installed file, skipping its `\DeclareOption{<modifier>}`
33/// (spanish.ldf:66-88 `mexico`, `es-noshorthands`) and
34/// `\bbl@declare@ttribute` (czech.ldf:328 `split`), so babel's leftover
35/// `\DeclareOption{<opt>}{unknown-package-option}` (babel.sty:4214) fired
36/// at `\ProcessOptions` (:4300) and `\languageattribute` missed the
37/// attribute (:1512-1538) — unamthesis, udepcolor-doc-ES, csbulletin; greek
38/// (no stub) already raw-loads clean. Only when the file is absent does the
39/// minimal hook set below stand in. Guard:
40/// `perfect_kernel_batch54::installed_ldf_outranks_the_language_stub`.
41fn install_lang_stub(lang: &str) -> Result<bool> {
42  if find_file(
43    lang,
44    Some(FindFileOptions {
45      ext_type: Some(Cow::Borrowed("ldf")),
46      ..Default::default()
47    }),
48  )
49  .is_some()
50  {
51    input_definitions(lang, InputDefinitionOptions {
52      noltxml: true,
53      extension: Some(Cow::Borrowed("ldf")),
54      ..Default::default()
55    })?;
56    return Ok(true);
57  }
58  install_lang_stub_hooks(lang)?;
59  Ok(false)
60}
61
62fn install_lang_stub_hooks(lang: &str) -> Result<()> {
63  // `\newlanguage` is a TeX macro of the form
64  // `\alloc@9\language\chardef\@cclvi`, which delegates to a 5-arg
65  // `\alloc@{}{}{}{}{}` macro. So a *raw* `\newlanguage\csname
66  // l@<lang>\endcsname` makes `\alloc@` grab `\csname` as the 5th
67  // argument (parameter-text token-grabbing does NOT expand `\csname`)
68  // and leaves `l@<lang>\endcsname` orphaned in the input stream — the
69  // unmatched `\endcsname` then cascades through every following
70  // package-load → 100 errors → fatal TooManyErrors abort.
71  //
72  // Force `\csname...\endcsname` to expand *first* with
73  // `\expandafter\newlanguage\csname...`, so `\newlanguage` receives
74  // the resolved `\l@<lang>` token directly.
75  //
76  // (The `\providecommand\captions{lang}{{}}` lines below are correct
77  // — Rust `format!` substitutes `{lang}` inline, giving
78  // `\providecommand\captionsbrazil{}` etc.)
79  let body = format!(
80    r"\expandafter\ifx\csname l@{lang}\endcsname\relax
81      \expandafter\newlanguage\csname l@{lang}\endcsname
82    \fi
83    \providecommand\captions{lang}{{}}%
84    \providecommand\extras{lang}{{}}%
85    \providecommand\noextras{lang}{{}}%
86    \providecommand\date{lang}{{}}",
87    lang = lang
88  );
89  raw_tex(&body)?;
90  Ok(())
91}

93pub fn load_italian() -> Result<()> {
94  let found = install_lang_stub("italian")?;
95  // italian.ldf:154-171 `\setISOcompliance` and the begin-document `\unit`.
96  // In pdflatex, verifica.cls:66-70 registers its
97  // `\AtBeginDocument{\@ifpackagewith{babel}{italian}{\setISOcompliance}…}`
98  // BEFORE italian.ldf's `\AtBeginDocument`, so FIFO execution sets
99  // `\it@ISOcompliance=1` before the ldf's hook tests it. In latexml-oxide the
100  // ldf's hook carries parameter tokens (`##1`) and is routed to
101  // `@at@begin@document@rawparam`, which fires before the L3 `begindocument`
102  // pool holding the class's hook (sect02.rs; the single-FIFO order is the
103  // K7 item). Bridge: `\setISOcompliance` binds `\unit` itself, so the order
104  // no longer matters. With the real ldf loaded (`found`) only that override
105  // is applied — its counter, hook and `\bbl@it@unit` stay the ldf's.
106  // Witness: verifica example1-5. Guard: `perfect_kernel_gemini::babel_italian_iso_compliance_unit`.
107  let set_iso = r"\def\setISOcompliance{%
108      \it@ISOcompliance=\@ne
109      \@ifpackageloaded{units}{}{%
110        \@ifpackageloaded{siunitx}{}{%
111          \@ifpackageloaded{SIunits}{}{%
112            \let\unit\bbl@it@unit}}}}";
113  if found {
114    raw_tex(&format!(
115      r"\ifdefined\bbl@it@unit\else\DeclareRobustCommand*{{\bbl@it@unit}}[1]{{\textormath{{\,\textup{{#1}}}}{{\,\mathrm{{#1}}}}}}\fi
116      \ifdefined\it@ISOcompliance {set_iso}\fi"
117    ))?;
118    return Ok(());
119  }
120  raw_tex(&format!(
121    r"\DeclareRobustCommand*{{\bbl@it@unit}}[1]{{\textormath{{\,\textup{{#1}}}}{{\,\mathrm{{#1}}}}}}%
122    \newcount\it@ISOcompliance \it@ISOcompliance=\z@
123    {set_iso}
124    \providecommand\IntelligentComma{{}}\providecommand\NoIntelligentComma{{}}
125    \AtBeginDocument{{\unless\ifnum\it@ISOcompliance=\z@
126      \@ifpackageloaded{{units}}{{}}{{%
127        \@ifpackageloaded{{siunitx}}{{}}{{%
128          \@ifpackageloaded{{SIunits}}{{}}{{%
129            \let\unit\bbl@it@unit}}}}}}\fi}}"
130  ))?;
131  Ok(())
132}
133// English-family stubs. babel-english.ldf uses `\@namedef{captions
134// \CurrentOption}` etc., so each variant gets its own
135// `\captions<variant>` / `\date<variant>`. When babel dispatches a
136// `\selectlanguage{american}` it expects `\captionsamerican` or the
137// `\captionsenglish` fallback. With incomplete raw-load these aren't
138// defined and the language-switch errors out (~17 papers in R-stages
139// for `\dateUSenglish`, 13 for `\captionsenglish`).
140//
141// We register the captions/extras/date hooks for the canonical
142// english variants (english, american, british, USenglish, UKenglish,
143// canadian, australian, newzealand) as no-ops — the variant captions
144// (chaptername etc.) just stay English in our HTML output, which is
145// already the project's default. Witness:
146// arXiv:1502.05791 (`\usepackage[british,american]{babel}`)
147// CONVERR_2 → expected OK.
148pub fn load_english() -> Result<()> { install_lang_stub("english").map(|_| ()) }
149pub fn load_american() -> Result<()> {
150  if install_lang_stub("american")? {
151    return Ok(());
152  }
153  install_lang_stub_hooks("USenglish")?;
154  install_lang_stub_hooks("english") // fallback chain
155}

194pub fn load_spanish() -> Result<()> {
195  if install_lang_stub("spanish")? {
196    return Ok(());
197  }
198  // babel-spanish-specific `\decimalpoint` — switches decimal separator
199  // from `,` (Spanish default) to `.`. We don't render locale-aware
200  // numerics; HTML uses `.` by default. No-op preserves intent.
201  // Driver 2511.19353 (`\usepackage[spanish]{babel}\decimalpoint`).
202  // Also `\decimalcomma` for the reverse direction.
203  //
204  // Spanish math-operator aliases — historical babel-spanish
205  // `\extrasspanish` hook adds the Spanish-language trig function
206  // names. Cataluña/Spain convention uses `sen` (seno), `tg`
207  // (tangente), `cotg` (cotangente), `cosec` (cosecante) etc. instead
208  // of the English/AMS \sin, \tan, \cot, \csc. We install them
209  // unconditionally rather than via the `\extras` hook — same
210  // outcome for our XML output and avoids the hook-timing complexity.
211  // Witness: arXiv:1909.12119 — `Error:undefined:\sen` /
212  // `\cotg` / `\tg` / `\arcsen` cluster on `\usepackage[spanish]{babel}`.
213  raw_tex(
214    r"\providecommand\decimalpoint{}\providecommand\decimalcomma{}
215    % spanish.ldf:680 \deactivatetilden (the ~n shorthand is not modelled; gaceta.cls:1612)
216    \providecommand\deactivatetilden{}\providecommand\activatetilden{}
217    \providecommand\sen{\mathop{\mathrm{sen}}\nolimits}
218    \providecommand\tg{\mathop{\mathrm{tg}}\nolimits}
219    \providecommand\cotg{\mathop{\mathrm{cotg}}\nolimits}
220    \providecommand\cosec{\mathop{\mathrm{cosec}}\nolimits}
221    \providecommand\arcsen{\mathop{\mathrm{arc\,sen}}\nolimits}
222    \providecommand\arctg{\mathop{\mathrm{arc\,tg}}\nolimits}
223    \providecommand\arccotg{\mathop{\mathrm{arc\,cotg}}\nolimits}",
224  )?;
225  Ok(())
226}

236pub fn load_vietnamese() -> Result<()> {
237  if install_lang_stub("vietnamese")? {
238    return Ok(());
239  }
240  // babel-vietnamese (vietnam.ldf) selects T5 font encoding and defines
241  // the Vietnamese precomposed-character command set (`\ecircumflex`,
242  // `\ocircumflex`, `\abreve`, `\ohorn`, `\uhorn`, hook-above `\h`, …).
243  // vietnam.ldf is NOT installed in TeX Live's base tree, so without
244  // this the commands stay undefined when a paper uses
245  // `\usepackage[vietnamese]{babel}` with Vietnamese author names.
246  // Surpass-Perl: Perl's babel can't find vietnam.ldf either; route
247  // through our t5enc binding (mirrors Perl `t5enc.def.ltxml`) — the
248  // same set vntex.sty pulls in. Witness 2003.07696
249  // (`\usepackage[english,vietnamese]{babel}`, author
250  // "Nguy\~\ecircumflex n Th\d{i} B\'ich Th\h{u}y").
251  t5enc_def::load_definitions()?;
252  Ok(())
253}
