1//! libertinust1math.sty — Libertine Unicode math font support.
2//!
3//! Same `\libus@undefine#1\let#1=\@undefined` pattern as fdsymbol.sty
4//! (see [`crate::package::fdsymbol_sty`] for the full rationale): the
5//! package re-declares every math symbol by first
6//! `\let\X\@undefined`-ing it and then `\DeclareMathSymbol{\X}{...}`-ing
7//! it back. Our kernel math chars (`\prime`, `\sum`, `\int`, ...) are
8//! locked at `DefMath!` time, so the re-DeclareMathSymbol is silently
9//! blocked and the previous `\let \@undefined` wins, cascading
10//! "undefined:\prime" errors for every formula that uses primes.
11//!
12//! Perl LaTeXML never raw-loads libertinust1math.sty (Perl-default
13//! `INCLUDE_STYLES=false`), so the kernel `\prime`/`\sum`/... stay
14//! intact. Match that behaviour: the binding re-declares only the capital
15//! Greek letters, whose shape the package sets (:484-497). Font choice is moot for
16//! XML/HTML output. Witness 2406.04255, 2406.04389, 2407.16071
17//! (~28 papers cumulative, up to 501 `\prime` errors per paper).
18use crate::prelude::*;
19
20LoadDefinitions!({
21  // libertinust1math.sty:59 reads its options with `\ProcessOptionsX`, in the order given, which marks them processed:
22  // `\@curroptions` stays as it was. The binding reads the Greek ones (:47-49, :62-65): `slantedGreek`, `uprightGreek`,
23  // `frenchmath` (upright) and `ISO` (slanted, after `frenchmath`).
24  DeclareOption!("slantedGreek", {
25    AssignValue!("libus_slanted_greek" => true);
26  });
27  DeclareOption!("uprightGreek", {
28    AssignValue!("libus_slanted_greek" => false);
29  });
30  DeclareOption!("frenchmath", {
31    AssignValue!("libus_french_math" => true);
32  });
33  DeclareOption!("ISO", {
34    AssignValue!("libus_iso" => true);
35  });
36  DeclareOption!(None, {});
37  ProcessKeyOptions!(*);
38  // Kernel math chars are authoritative: the binding re-declares none of them but the capital Greek letters (below).
39  //
40  // libertinust1math.sty:7 `\RequirePackage{xkeyval,etoolbox}` and :75 `\RequirePackage{amsmath}` (stubs load real
41  // dependencies): a document's `\DeclareMathOperator`/`\operatorname`/`\text` came from these (2110.09723,
42  // 2201.10869, 2208.02596).
43  RequirePackage!("xkeyval");
44  RequirePackage!("etoolbox");
45  RequirePackage!("amsmath");
46  // The package's own names papers reach for, which no kernel binding defines (a name a package defined first keeps
47  // its definition): :166 `\mathbcal` (`\DeclareMathAlphabet{\mathbcal}{U}{stixcal}{b}{n}`, the bold calligraphic
48  // letters) and :129 `\mathbfit` (the `bold-letters` alphabet, bold italic) — 2603.08217, 2605.03875; :1603
49  // `\widebar` (9 papers: 2311.14681, 2405.05597, 2411.03279) and :912 `\widecheck` (2602.14897), the wide accents;
50  // :1188-1189 `\lesssim` (2408.05538) and `\gtrsim` (2208.02596), :1443 `\blacksquare` (2412.05468), :643
51  // `\varkappa`, :1377 `\eqdef` (2506.09249), :1215 `\cupdot`, :872 `\bigcupdot` (2512.09624), :835 `\oiint`
52  // (2605.03888). Guard:
53  // `perfect_kernel_batch68j::libertinust1math_loads_amsmath_and_its_names`.
54  if !is_defined("\\mathbcal") {
55    DefConstructor!("\\mathbcal{}", "#1", bounded => true, require_math => true,
56      font => { family => "caligraphic", series => "bold", shape => "upright" });
57  }
58  if !is_defined("\\mathbfit") {
59    DefConstructor!("\\mathbfit{}", "#1", bounded => true, require_math => true,
60      font => { family => "serif", series => "bold", shape => "italic" });
61  }
62  if !is_defined("\\widebar") {
63    DefMath!("\\widebar Digested", "\u{00AF}", operator_role => "OVERACCENT");
64  }
65  if !is_defined("\\widecheck") {
66    DefMath!("\\widecheck Digested", "\u{02C7}", operator_role => "OVERACCENT");
67  }
68  if !is_defined("\\lesssim") {
69    DefMath!("\\lesssim", None, "\u{2272}", role => "RELOP", meaning => "less-than-or-similar-to");
70  }
71  if !is_defined("\\gtrsim") {
72    DefMath!("\\gtrsim", None, "\u{2273}", role => "RELOP", meaning => "greater-than-or-similar-to");
73  }
74  if !is_defined("\\blacksquare") {
75    DefMath!("\\blacksquare", None, "\u{25A0}");
76  }
77  if !is_defined("\\varkappa") {
78    DefMath!("\\varkappa", None, "\u{03F0}");
79  }
80  if !is_defined("\\eqdef") {
81    DefMath!("\\eqdef", None, "\u{225D}", role => "RELOP");
82  }
83  if !is_defined("\\cupdot") {
84    DefMath!("\\cupdot", None, "\u{228D}", role => "ADDOP");
85  }
86  if !is_defined("\\bigcupdot") {
87    DefMath!("\\bigcupdot", "\u{2A03}", role => "BIGOP", scriptpos => "mid", variablesize_op => true);
88  }
89  if !is_defined("\\oiint") {
90    DefMath!("\\oiint", None, "\u{222F}", role => "INTOP", meaning => "surface-integral", variablesize_op => true);
91  }
92  // :484-497 `\let\Gamma\upGamma` (:312, `\mathalpha` from `operators`): the capital Greek letters upright, over a
93  // class's italic ones (llncs.cls:379-389; KNOWN_PERL_ERRORS #1147), or under `slantedGreek` (:62-65: not with
94  // `frenchmath`, always with `ISO`) `\let\Gamma\itGamma` (:400, `\mathalpha` from `letters`), italic as a Latin letter
95  // unless a math alphabet is in force. Review rev79e finding 1 (witness 2606.09074).
96  let slanted = (lookup_bool("libus_slanted_greek") && !lookup_bool("libus_french_math"))
97    || lookup_bool("libus_iso");
98  declare_capital_greek("", slanted.then_some(LettersClass::Variable))?;
99});
