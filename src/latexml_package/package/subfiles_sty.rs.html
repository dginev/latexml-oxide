78/// `\subfiles@split` (subfiles.sty:111-116): the area of `file` up to its last `/` and the rest, given to `loader`
79/// (`\subimport` or `\subincludefrom`) as its two arguments. The name is unquoted first (2409.12749's
80/// `\subfile{"./counterExample"}`): `\subfiles@split` parses it as written, so pdflatex's area keeps the quote (`"./`) and
81/// a figure the subfile names fails with "Unbalanced quotes in file name"; a quoted name is one name, which TeX's own
82/// lookups read unquoted (`\unquote@name`, latex.ltx:9654-9666).
83fn subfile_import(loader: &str, file: &str) -> Tokens {
84  let unquoted = file.replace('"', "");
85  let file = unquoted.trim();
86  let (area, base) = file
87    .rfind('/')
88    .map_or(("", file), |slash| (&file[..=slash], &file[slash + 1..]));
89  let mut out = vec![T_CS!(loader), T_BEGIN!()];
90  out.extend(Explode!(area));
91  out.extend([T_END!(), T_BEGIN!()]);
92  out.extend(Explode!(base));
93  out.push(T_END!());
94  Tokens::new(out)
95}
