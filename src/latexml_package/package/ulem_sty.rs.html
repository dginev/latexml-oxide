92  // an unmatched brace, the group's own `}` closing it, and `\UL@swender`, :107, puts the group's `}` back after). An
93  // empty slot (papers write `\useunder{\uline}{\ul}{}`) defines nothing. Witnesses 2406.08270 (`\useunder` with
94  // `\uline{…}`), 2405.20343 and 2406.03441 (`{\ul 18.8262}`, `{\ul …}`: one character underlined while `\ul` was let
95  // to `\uline`). A `\ul{x}` outside any group, a fatal error in pdflatex, underlines up to the end of its paragraph.
96  DefMacro!("\\useunder{}{}{}", sub[(ucmd, decl, argcmd)] {
97    let mut out: Vec<Token> = Vec::new();

99    if !ucmd_v.is_empty() {
100      let target = ucmd_v[0];
101      let decl_v = decl.unlist();
102      if !decl_v.is_empty() {
103        out.extend([T_CS!("\\def"), decl_v[0], T_BEGIN!(), T_CS!("\\lx@ulem@declaration"), target, T_END!()]);
104      }

107        out.extend([T_CS!("\\let"), arg_v[0], target]);
108      }
109    }
