91  DefMacro!("\\useunder{}{}{}", sub[(ucmd, decl, argcmd)] {
92    let mut out: Vec<Token> = Vec::new();
93    let ucmd_v = ucmd.unlist();
94    if !ucmd_v.is_empty() {
95      let target = ucmd_v[0];
96      let decl_v = decl.unlist();
97      if !decl_v.is_empty() {
98        out.extend([T_CS!("\\def"), decl_v[0], T_BEGIN!(), T_CS!("\\lx@ulem@declaration"), target, T_END!()]);
99      }
100      let arg_v = argcmd.unlist();
101      if !arg_v.is_empty() {
102        out.extend([T_CS!("\\let"), arg_v[0], target]);
103      }
104    }
105    Ok(Tokens::new(out))
106  });
