188/// `<builder>{<caption>}{<body>}` for subfigure.sty's `\subfigure`/`\subtable` with its optional `[list]` and
189/// `[caption]` as read: `[list][caption]` is the kernel caption `\lx@caption[list]{caption}`, a lone `[caption]` (empty
190/// or not) `\lx@caption{caption}`, and no optional the sub-counter's step alone (`\@@add@caption@counters`, marked as
191/// a caption marks it): the panel is numbered but prints no caption (subfigure.sty:217, :251-262).
192fn subfloat_tokens(
193  builder: &str,
194  short: Option<Tokens>,
195  caption: Option<Tokens>,
196  body: Tokens,
197) -> Tokens {
198  let mut tokens = vec![T_CS!(builder), T_BEGIN!()];
199  match (short, caption) {
200    (short, Some(caption)) => {
201      tokens.extend([T_CS!("\\lx@caption"), T_OTHER!("[")]);
202      if let Some(short) = short {
203        tokens.extend(short.unlist());
204      }
205      tokens.extend([T_OTHER!("]"), T_BEGIN!()]);
206      tokens.extend(caption.unlist());
207      tokens.push(T_END!());
208    },
209    (Some(caption), None) => {
210      tokens.extend([T_CS!("\\lx@caption"), T_BEGIN!()]);
211      tokens.extend(caption.unlist());
212      tokens.push(T_END!());
213    },
214    (None, None) => tokens.extend([
215      T_CS!("\\lx@donecaptiontrue"),
216      T_CS!("\\@@add@caption@counters"),
217    ]),
218  }
219  tokens.extend([T_END!(), T_BEGIN!()]);
220  tokens.extend(body.unlist());
221  tokens.push(T_END!());
222  Tokens::new(tokens)
