1//! T3 font encoding: tipa's IPA fonts (tipa10 …, t3enc.def, FUKUI Rei).
2//!
3//! Each slot decodes to the glyph the tipa fonts draw there — tipa.mf's driver and its parts, whose `cmchar` names
4//! identify the glyph (tipadiac.mf accents and diacritics 0-31 and the marks at 34-38, 58-62, 92-95, 126-127,
5//! 152-159, 189-191; tipapnct.mf punctuation; tiparoml.mf letters; tipasym1-4.mf, tipagerm.mf, tipasc.mf IPA letters;
6//! tipatone.mf tone letters 128-151) — as the IPA letter Unicode has for it, by t3enc.def's command names
7//! (`\textschwa` 64, `\textesh` 83, `\textltailn` 241 …), cross-checked against tipauni.sty's Unicode table (TL
8//! tipauni 0.7a: `\textltailn` "272, the input letters `@` ə, `N` ŋ, `:` ː …). The ASCII slots are tipa's input
9//! convention (tipaman §3): digits, capitals and `@ : ; " |` are IPA letters, so `\textipa{[D@"lIm@]}` reads
10//! `[ðəˈlɪmə]`; `g` (103) is the single-storey ɡ (`\textscriptg`), the two-storey g is `\textg` (165). The fonts'
11//! ligature program (tipa10.tfm LIGTABLE, tipa.mf:66-67) makes `""` the secondary stress ˌ (127) and `||` the double
12//! pipe ǁ (123): `t3_ligature`.
13//!
14//! Where the glyph the font draws and its IPA meaning differ, the slot is the IPA meaning, as tipauni.sty maps it (user
15//! ruling 2026-10-20): `!` the alveolar click ǃ, `&` ɞ, `\textcorner` U+031A, `\textsubbar` U+0320; the open corner,
16//! which tipauni.sty does not map, stays ˹.
17//!
18//! An accent slot holds the combining mark it is set as (decoded alone on a no-break space, the Perl fontmaps' shape),
19//! or the spacing accent where T1 has one (0-12); [`t3_accent_mark`] gives the mark above or below a letter that
20//! t3enc.def's `\upper@accent`/`\lower@accent` set it as (t3enc_def.rs). Slots with no Unicode character — the tone
21//! letters' parts 128-145 and stem 191 (tone.sty assembles them; tone_sty.rs maps `\tone{51}` to ˥˩), the Celtic
22//! palatalization mark 32, the hooktop 35, the Viby and left-hooktop long letters 166, 167, 182, 183 — decode to
23//! nothing, and a text symbol at one warns where it is used (`\lx@unmapped@text@symbol`, sect08.rs).
24//!
25//! Perl has no T3 map: tipa's symbols printed nothing and `\textipa`'s input letters decoded as OT1 ("D@"lIm@" for
26//! ðəˈlɪmə). OXIDIZED_DESIGN_DIVERGENCES #781. Witnesses 1802.05444, 2405.01972 (`\textipa{\textltailn}` ɲ, `\!b` ɓ).

106/// Where t3enc.def sets an accent slot: above the letter (`\upper@accent`), below it (`\lower@accent`), or over it
107/// (`\textsuperimposetilde`'s `\ipaclap`, `\ipabar`'s bar).
108#[derive(Clone, Copy, Debug, PartialEq, Eq)]
109pub enum AccentPlace {
110  /// `\upper@accent`, t3enc.def:268-280.
111  Above,
112  /// `\lower@accent`, t3enc.def:282-294.
113  Below,
114  /// Superimposed, t3enc.def:330-354.
115  Over,
116}

118/// The combining mark of T3 slot `slot` set `place` on a letter, the glyph tipadiac.mf/tipatone.mf draw there as
119/// t3enc.def:368-516 sets it: the tilde (3) above is U+0303, below (`\~*`, `\textsubtilde`) U+0330; the round cap (16)
120/// above is the inverted breve U+0311, below (`\textsubarch`, `\u*`) U+032F; the stroke of `"` (34) below is the
121/// syllabicity mark U+0329 (`\textsyllabic`, `\s`). `None` where Unicode has no mark: the subscript W set above
122/// (`\textoverw`).
123pub fn t3_accent_mark(slot: i64, place: AccentPlace) -> Option<char> {
124  use AccentPlace::{Above, Below, Over};
125  let mark = match (slot, place) {
126    (0, Above) => '\u{0300}',
127    (0 | 14, Below) => '\u{0316}',
128    (1, Above) => '\u{0301}',
129    (1 | 15, Below) => '\u{0317}',
130    (2, Above) => '\u{0302}',
131    (2, Below) => '\u{032D}',
132    (3, Above) => '\u{0303}',
133    (3, Below) => '\u{0330}',
134    (4, Above) => '\u{0308}',
135    (4, Below) => '\u{0324}',
136    (5, Above) => '\u{030B}',
137    (6, Above) => '\u{030A}',

171    (34, Above) => '\u{030D}',
172    (38, Over) => '\u{0334}',

174    (62, Above) => '\u{0361}',

188/// tipa10.tfm's ligatures (tipa.mf:66-67), over the decoded characters: two primary stresses are the secondary stress,
189/// two pipes the double pipe. TeX reads the ligature again with the next character and has no ligature for it, so a
190/// third `"` follows the ˌ as ˈ.
191fn t3_ligature(run: &str) -> String {
192  let mut out = String::new();
193  let mut chars = run.chars().peekable();
194  while let Some(c) = chars.next() {
195    match (c, chars.peek()) {
196      ('\u{02C8}', Some('\u{02C8}')) => {
197        chars.next();
198        out.push('\u{02CC}');
199      },
200      ('\u{01C0}', Some('\u{01C0}')) => {
201        chars.next();
202        out.push('\u{01C1}');
203      },
204      _ => out.push(c),
205    }
206  }
207  out
208}

210fn is_t3(font: &Font) -> bool { font.get_encoding().is_some_and(|encoding| encoding == "T3") }
211
212LoadDefinitions!({
213  DeclareFontMap!("T3", t3_map());
214  // The marks t3enc.def:213-219 sets after the letter they modify — `\textrthook`, `\textpalhook`, `\textcorner`,
215  // drawn to hang off the letter before them (`t\textrthook` t̢, `k\textcorner` k̚: 2601.02906, 2609.04232) — are the
216  // bare combining marks, which attach to that letter, not the no-break space a standalone combining mark is
217  // decoded on; so are the tie bars typed `<` `>` (60, 62), set between the letters they join (`\textipa{t<s}` t͜s).
218  DeclareFontMapMultichar!("T3", {
219    36u8 => "\u{0322}",
220    37u8 => "\u{0321}",
221    60u8 => "\u{035C}",
222    62u8 => "\u{0361}",
223    94u8 => "\u{031A}",
224  });
225  DefLigature!(
226    "\u{02C8}{2,}|\u{01C0}{2,}",
227    |caps: &regex::Captures| t3_ligature(&caps[0]),
228    fontTest => sub[font] { is_t3(font) }
229  );
230});
