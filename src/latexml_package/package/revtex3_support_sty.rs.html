307/// `cs{<body>}`, the body of the environment `name` as the argument of `cs`: see [`read_environment_body`].
308fn environment_body_as_argument(cs: Token, name: &str) -> Result<Tokens> {
309  let mut call = vec![cs, T_BEGIN!()];
310  call.extend(read_environment_body(name)?.unlist());
311  call.push(T_END!());
312  Ok(Tokens::new(call))
313}

315/// The tokens of the environment `name`'s body, up to its `\end{name}` (a nested `\end{...}` is body), which is put
316/// back so that the environment ends as usual; without the spaces around it (the line ends after `\begin{title}` and
317/// before `\end{title}`), which REVTeX's environments ignore (revtex.sty v2.1:1340-1350 `\ignorespaces`, `\par`).
318fn read_environment_body(name: &str) -> Result<Tokens> {
319  let mut body: Vec<Token> = Vec::new();
320  while let Some(token) = read_token()? {
321    if token.get_catcode() == Catcode::BEGIN {
322      body.push(token);
323      body.extend(read_balanced(ExpansionLevel::Off, false, false)?.unlist());
324      body.push(T_END!());
325    } else if token == T_CS!("\\end") {
326      let env = read_arg(ExpansionLevel::Off)?;
327      if env.to_string() == name {
328        let mut end = vec![token, T_BEGIN!()];
329        end.extend(env.unlist());
330        end.push(T_END!());
331        unread_vec(end);
332        break;
333      }
334      body.push(token);
335      body.push(T_BEGIN!());
336      body.extend(env.unlist());
337      body.push(T_END!());
338    } else {
339      body.push(token);
340    }
341  }
342  while body
343    .last()
344    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
345  {
346    body.pop();
347  }
348  let lead = body
349    .iter()
350    .take_while(|t| t.get_catcode() == Catcode::SPACE)
351    .count();
352  body.drain(..lead);
353  Ok(Tokens::new(body))
