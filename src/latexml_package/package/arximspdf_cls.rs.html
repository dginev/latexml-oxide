1//! arximspdf.cls / arxstspdf.cls — the arXiv IMS journal classes (Annals of
2//! Probability/Statistics/Applied Probability, `aop`/`aos`/`aap`/`aoas`).
3//!
4//! Neither Perl LaTeXML nor (previously) Rust bound these self-contained ~3000-
5//! line classes, so `\documentclass{arximspdf}` papers cascaded into dozens of
6//! undefined-macro errors (the structured `\b*` bibliography, `{barticle}`/…,
7//! plus `\eqntext`/`\dvtx`/`{longlist}`). This binding loads `article` as the
8//! base and defines the IMS-specific macros semantically so the papers convert —
9//! a strict improvement over Perl, which fails outright (both engines lack the
10//! class; same-host confirmed). Witnesses: 0910.0069 (aop632) + the 16-paper
11//! aop/aos sandbox cluster.
12//!
13//! Bibliography note: arximspdf uses `natbib` + `\bibitem`/`thebibliography`, so
14//! a bibitem produces `ltx:bibitem > ltx:bibblock` (Flow.model). The structured
15//! `ltx:bib-*` vocabulary is schema-valid only inside `ltx:bibentry` (the BibTeX
16//! path), NOT inside `bibblock` — so the `\b*` field macros are PASSTHROUGH text
17//! (readable + schema-valid), not mapped to `ltx:bib-*`.

40  // `\sidewaystable`/`\sidewaysfigure` the class lets at `\begin{document}` (:1838-1848; 1304.4448, 1104.3398); the

48  RequirePackage!("amssymb");
49  // NB: arximspdf does NOT load amsmath — it defines \tfrac/\dfrac/\operatorname/
50  // {aligned} itself. We mirror that: load only amsopn (for \operatorname /
51  // \DeclareMathOperator) and define the rest directly. Loading full amsmath would
52  // override plain-TeX \matrix with its env-form, breaking the command-form
53  // \matrix{...\cr...} these papers use (Perl is parity-broken the same way and
54  // also omits amsmath).

61  // ---- math the class defines itself (arximspdf L1099-1100) ----------------
62  DefMacro!("\\tfrac{}{}", "{\\textstyle\\frac{#1}{#2}}");
63  DefMacro!("\\dfrac{}{}", "{\\displaystyle\\frac{#1}{#2}}");
64  DefMacro!("\\dvt",  "\\colon\\ ");
65  DefMacro!("\\dvtx", "\\colon\\;");
66  DefMacro!("\\divid{}{}", "\\frac{#1}{#2}");
67  DefMacro!("\\zs{}",  "#1");
68  DefMacro!("\\zss{}", "#1");
69  DefMacro!("\\eqntext{}", "#1");
70  // The structured `\b*` bibliography, `{pf}` proofs with `\qed`, `\tablewidth` table notes and `\bolds`: the
71  // markup the VTeX IMS classes share (ims_support_sty.rs).
72  RequirePackage!("ims_support");

74  // ---- misc list / sectioning / theorems the class adds --------------------
75  DefEnvironment!("{longlist}[]", "<ltx:enumerate>#body</ltx:enumerate>",
76    mode => "internal_vertical");
77  // arximspdf predefines a Theorem env (L1405), with its starred twin as every `\newtheorem` (vtexthm's, in
78  // ims_support with `\newproclaim`; `{hy*}` 1107.3284); its proof envs are in ims_support.
79  RawTeX!(r"\newtheorem{thm}{Theorem}");
80  // ---- frontmatter (standard LaTeXML frontmatter API; metadata preserved) --
81  // Standardize the IMS scaffolding onto the article frontmatter flow: the
82  // {frontmatter}/{aug}/{keyword} blocks are TRANSPARENT wrappers (no special
83  // mode), and the IMS metadata macros route to the same \lx@add@* sinks
84  // article/acmart use, so creators/title/keywords/pubnotes are emitted into the
85  // document frontmatter.
86  // `{frontmatter}` a plain group, as OmniBus's (omnibus_cls.rs): a DefEnvironment boxes its body in restricted
87  // horizontal mode, where a `$$` in the abstract is no display (TeX_Math.pool.ltxml:65).
88  Let!("\\frontmatter", "\\@empty");
89  Let!("\\endfrontmatter", "\\@empty");
90  DefEnvironment!("{aug}", "#body");
91  // `{keyword}[class=…]` and `\kwd` are the VTeX classes' shared markup (ims_support_sty.rs): arximspdf's entries
92  // are joined by ", " (arximspdf.cls:2438 of 1107.3284's copy), under its headings "Key words and phrases." and "AMS
93  // 2000 subject classifications." (:2408-2409; arxstspdf.cls:2356-2357 of 0903.0664's copy end them with ":"). The
94  // collecting `\lx@begin@keywords` started afresh at each block, so a paper's AMS block was lost to the keywords that
95  // follow it (1107.3284, 1309.4864).
96  let shipped = match shipped_class_text("arximspdf")? {
97    Some(text) => Some(text),
98    None => shipped_class_text("arxstspdf")?,
99  };
100  let colon = shipped
101    .as_deref()
102    .is_some_and(|text| class_code_contains(text, "{KWD}{Key words and phrases:}"));
103  RawTeX!(r"\def\sep{\unskip, }");
104  if colon {
105    RawTeX!(r"\setattribute{keyword}{KWD}{Key words and phrases:}\setattribute{keyword}{AMS}{AMS 2000 subject classifications:}");
106  } else {
107    RawTeX!(r"\setattribute{keyword}{KWD}{Key words and phrases.}\setattribute{keyword}{AMS}{AMS 2000 subject classifications.}");
108  }
109  // Author-name parts used inside \author{...}: pass their content through.
110  for m in ["fnms","snm","inits","degs","roles","suffix"] {
111    def_macro_identity(&format!("\\{m}{{}}"))?;
112  }
113  // \author[mark]{name} (IMS adds an optional thanks-mark) -> standard creator.
114  DefMacro!("\\author[]{}", "\\lx@add@creator[role=author]{#2}");
115  DefMacro!("\\affiliation{}", "\\lx@add@contact[role=affiliation]{#1}");
116  DefMacro!("\\address[]{}", "\\lx@add@contact[role=address]{#2}");
117  DefMacro!("\\email{}", "\\lx@add@contact[role=email,name={e-mail: }]{#1}");
118  DefMacro!("\\ead[]{}", "\\lx@add@contact[role=email,name={e-mail: }]{#2}");
119  // As the class prints them (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #921): arximspdf.cls:1911-1912 (math0702702's copy) sets the
120  // journal's name alone, then "Vol. V, No. N" (:366 and :383 the labels); :1913 "DOI: ", the role's.
121  DefMacro!("\\journal{}", "\\lx@add@pubnote[role=journal,name={}]{#1}");
122  DefMacro!("\\volume{}", "\\lx@add@pubnote[role=volume,name={Vol.~}]{#1}");
123  DefMacro!("\\issue{}", "\\lx@add@pubnote[role=number,name={No.~}]{#1}");
124  DefMacro!("\\pubyear{}", "\\lx@add@date[role=publication]{#1}");
125  DefMacro!("\\doi{}", "\\lx@add@pubnote[role=doi]{#1}");
126  DefMacro!("\\received{}", "\\lx@add@date[role=received]{#1}");
127  DefMacro!("\\revised{}", "\\lx@add@date[role=revised]{#1}");
128  DefMacro!("\\accepted{}", "\\lx@add@date[role=accepted]{#1}");
129  // History date parts + misc frontmatter: passthrough / drop.

131    def_macro_identity(&format!("\\{m}{{}}"))?;
132  }
133  DefMacro!("\\dochead{}", "");
134  DefMacro!("\\dedicated{}", "");
135  DefMacro!("\\runtitle{}", "");
136  DefMacro!("\\runauthor{}", "");

144  }
145  // `\printead*[text]{e1,e2}` prints the addresses `\ead[label=e1]` recorded (arximspdf.cls:599-603); `\ead`
146  // already adds each as a contact, so the reprint goes. A `"printead*"` entry in the list above was the

149  def_macro_noop("\\printead OptionalMatch:* []{}")?;
150  DefEnvironment!("{supplement}[]", "#body");

154  DefMacro!("\\endlocaldefs", "\\makeatother");

158  // The URL, attribute and script helpers of the arximspdf.cls versions (`\doiurl`, `\arxivurl`, `\surl`,
159  // `\setattribute`, `\tsup`, …) are the VTeX classes' shared markup (ims_support_sty.rs).
160  // What both classes define and papers call (run 347: 42 arxstspdf and 13 arximspdf papers; line numbers from
161  // 0808.2902's arxstspdf.cls and 0704.1778's arximspdf.cls), a copy without one leaving it to the paper
162  // (`DefinitionOrigin::Fallback`):
163  // - `\ccline{2-3,4-5}` (arxstspdf :1167-1169, arximspdf :1202-1204), a `\cline` for each range (33 papers: 1010.0304,
164  //   1010.5040, 1011.1079);
165  // - `\bhline` and its siblings `\thline`, `\fhline`, `\lhline` (arxstspdf :1828-1832), a table's rules spaced
166  //   around, `\bhline`'s thicker: `\hline`s;
167  // - `\freefootnotetext[mark]{text}` (arxstspdf :1917, arximspdf :1908), a footnote under the given mark, an empty one
168  //   for an unmarked note (1011.1160 "This conversation was conducted in spring 2008.", 1009.1507's disclaimer);
169  // - `\relateddois[label]{key}{text}` (1008.4373's arxstspdf.cls :2106-2115; 43 of the 68 copies, 99 of 371), a
170  //   title note naming the articles related to this one, its `\relateddoi{label}{doi}` each a link to the DOI
171  //   (1410.0163, 1301.7161, 1210.0385 "Discussed in …");
172  // - `\shref{url}{text}` (arxstspdf :3021-3024), a link whose text breaks at its `\\`s;
173  // - arximspdf's :2530-2531 `\write@toc@ignorecontentsline`/`\write@toc@restorecontentsline` (1205.5627) and
174  //   :2585-2601 `\maxcsec{…}`, `\maxcsubsec`, `\maxcsubsubsec`, the contents' number widths (1103.4825, 1309.3728); its
175  //   :533-540 `\setvaluelist`/`\getitemvalue` (1203.0565) are ims_support's, shared with arxbj.
176  {
177    let _class = OriginGuard::new(DefinitionOrigin::Fallback);
178    DefMacro!("\\ccline{}", sub[(ranges)] {
179      let ranges: Tokens = ranges;
180      let mut out: Vec<Token> = Vec::new();
181      for range in ranges.to_string().split(',') {
182        let range = range.trim();
183        if !range.is_empty() {
184          out.extend(mouth::tokenize_internal(TeXString::assembled(s!("\\cline{{{range}}}"))).unlist());
185        }
186      }
187      Ok(Tokens::new(out))
188    });
189    RawTeX!(r"\def\thline{\hline}\def\fhline{\hline}\def\bhline{\hline}\def\lhline{\hline}
190\def\freefootnotetext[#1]{\footnotetext[#1]}
191\@ifundefined{doi@base}{\def\doi@base{http://dx.doi.org/}}{}
192\def\relateddois{\@ifnextchar[{\lx@ims@relateddois}{\lx@ims@relateddois[]}}
193\def\lx@ims@relateddois[#1]#2#3{\lx@add@frontmatter{ltx:note}[role=relateddois]{\def\relateddoi##1##2{\href{\doi@base##2}{##2}}#3}}
194\DeclareRobustCommand\shref[2]{\href{#1}{\let\\\relax#2}}
195\def\write@toc@ignorecontentsline{\addtocontents{toc}{\protect\ignorecontentsline}}
196\def\write@toc@restorecontentsline{\addtocontents{toc}{\protect\restorecontentsline}}
197\def\maxcsec#1{}\def\maxcsubsec#1{}\def\maxcsubsubsec#1{}");
198  }
199});
