1236/// How many of the document's own macros [`holds_author_structure`] reads into from one call, and how many calls
1237/// [`blocks_as_printed`] expands in one body, at most.
1238const MEANING_READS: usize = 64;

1240/// Whether `token`, a macro of the document's own (`\newcommand`, `\def` in the document or its own files), holds author
1241/// structure by meaning — its definition, or a document macro it calls (recursively), spells a names block (a token
1242/// meaning `\IEEEauthorblockN`: the class's, `\authorblockN`, a `\let` copy), an `\and`-family separator (`\and`,
1243/// `\And`, `\AND`, or a token `\let` to one) or the grid's row break `\\` (or a `\let` copy). Read as the rehearsal reads a block's macros for `\let` copies of its
1244/// separators (`rehearsal::let_copies_of_bound_commands`): the definitions' tokens, no expansion.
1245fn holds_author_structure(token: &Token) -> bool {
1246  // (a copy means the command: `\authorblockN`, the kernel's `\And` and `\AND`, a document's `\let`)
1247  let structure = [T_CS!("\\IEEEauthorblockN"), T_CS!("\\and"), T_CS!("\\\\")];
1248  let document_macro = |t: &Token| -> Option<Tokens> {
1249    let defn = lookup_definition(t).ok().flatten()?;
1250    if !defn.is_expandable() || defn.get_origin() != definition::origin::DefinitionOrigin::Document
1251    {
1252      return None;
1253    }
1254    match defn.get_expansion() {
1255      Some(ExpansionBody::Tokens(body)) => Some(body.clone()),
1256      _ => None,
1257    }
1258  };
1259  let Some(body) = document_macro(token) else {
1260    return false;
1261  };
1262  let mut seen = rustc_hash::FxHashSet::<Token>::default();
1263  let mut pending: Vec<Token> = body.unlist();
1264  let mut macros = 1;
1265  while let Some(t) = pending.pop() {
1266    if !t.is_executable() || !seen.insert(t) {
1267      continue;
1268    }
1269    if structure.iter().any(|s| t == *s || t.defined_as(s)) {
1270      return true;
1271    }
1272    if macros < MEANING_READS
1273      && let Some(inner) = document_macro(&t)
1274    {
1275      macros += 1;
1276      pending.extend(inner.unlist());
1277    }
1278  }
1279  false
1280}

1422      .iter()
1423      .take_while(|t| **t == T_SPACE!())
1424      .count();
1425  if tokens.get(open)?.code != Catcode::BEGIN {
1426    return None;
1427  }
1428  let mut level = 0usize;
1429  let close = open
1430    + tokens[open..].iter().position(|t| {
1431      match t.code {
1432        Catcode::BEGIN => level += 1,
1433        Catcode::END => level -= 1,
1434        _ => {},
1435      }
1436      level == 0
1437    })?;

1439}

1441/// An author-block body with each affiliation block at the top level of another's argument read as conference mode
1442/// prints it, on rows of its own in the outer's column (IEEEtran.cls:4653-4665): one led by a mark, or after a line of
1443/// marks alone (its label), lifted out as a block of its own, the outer's text around it bare text again
1444/// (`wrap_bare_author_block_text`); an unmarked one spliced into the outer's lines (1710.07993, 2004.11080, 2107.11588,
1445/// 2605.02682, 1405.1593's `\thanks` wrapper; review revP4cR2 B1-B3; OXIDIZED_DESIGN_DIVERGENCES #1009).
1446fn unnested_affiliation_blocks(body: Tokens) -> Tokens {
1447  fn is_affiliation(t: &Token) -> bool { *t == T_CS!("\\IEEEauthorblockA") }
1448  fn is_break(t: &Token) -> bool { *t == T_CS!("\\\\") || *t == T_CS!("\\newline") }
1449  // `chunk` added to the outer's text, after the line break a spliced block's text owes the text after it
1450  fn append(text: &mut Vec<Token>, chunk: &[Token], owed: &mut bool) {
1451    let first = chunk.iter().position(|t| *t != T_SPACE!());
1452    if *owed && let Some(first) = first.filter(|&k| !is_break(&chunk[k])) {
1453      text.push(T_CS!("\\\\"));
1454      text.extend_from_slice(&chunk[first..]);
1455    } else {
1456      text.extend_from_slice(chunk);
1457    }
1458    *owed &= first.is_none();
1459  }
1460  // `tokens` at one level, each affiliation block's top-level inner blocks read; whether one was lifted out
1461  fn read(tokens: &[Token], out: &mut Vec<Token>) -> bool {
1462    let mut lifted_any = false;
1463    let mut i = 0;
1464    while i < tokens.len() {
1465      let Some((open, close)) =
1466        block_argument(tokens, i + 1).filter(|_| is_affiliation(&tokens[i]))
1467      else {
1468        out.push(tokens[i]);
1469        i += 1;
1470        continue;
1471      };
1472      let arg = &tokens[open + 1..close];
1473      // the outer's text and the blocks lifted out of it, in order (`true`: a block)
1474      let mut pieces: Vec<(bool, Vec<Token>)> = Vec::new();
1475      let mut text: Vec<Token> = Vec::new();
1476      let (mut from, mut j, mut owed, mut lifts) = (0, 0, false, false);
1477      while j < arg.len() {
1478        let Some((inner_open, inner_close)) =
1479          block_argument(arg, j + 1).filter(|_| is_affiliation(&arg[j]))
1480        else {
1481          // (a group skipped whole)
1482          j = block_argument(arg, j).map_or(j + 1, |(_, close)| close + 1);
1483          continue;
1484        };
1485        let lead = &arg[from..j];
1486        let inner = &arg[inner_open + 1..inner_close];
1487        // (the lead's last line, past its last top-level `\\`)
1488        let line = lead.len()
1489          - split_top_level_lines(lead)
1490            .last()
1491            .map_or(0, |last| last.len());
1492        let label = if prints_marks_alone(&lead[line..]) {
1493          Some(line)
1494        } else {
1495          leads_with_written_mark(inner).then_some(lead.len())
1496        };
1497        if let Some(label) = label {
1498          append(&mut text, &lead[..label], &mut owed);
1499          pieces.push((false, std::mem::take(&mut text)));
1500          let mut block = vec![arg[j], T_BEGIN!()];
1501          block.extend_from_slice(&lead[label..]);
1502          block.extend_from_slice(inner);
1503          block.push(T_END!());
1504          pieces.push((true, block));
1505          (owed, lifts) = (false, true);
1506        } else {
1507          // (spliced, on rows of its own: Lab Z under both names, `Dept X, Univ Y\\ \IEEEauthorblockA{Lab Z}`)
1508          append(&mut text, lead, &mut owed);
1509          while text.last() == Some(&T_SPACE!()) {
1510            text.pop();
1511          }
1512          if text.last().is_some_and(|t| !is_break(t)) {
1513            text.push(T_CS!("\\\\"));
1514          }
1515          text.extend_from_slice(inner);
1516          owed = true;
1517        }
1518        from = inner_close + 1;
1519        j = from;
1520      }
1521      let at = i;
1522      i = close + 1;
1523      if from == 0 {
1524        out.extend_from_slice(&tokens[at..i]);
1525        continue;
1526      }
1527      append(&mut text, &arg[from..], &mut owed);
1528      pieces.push((false, text));
1529      if !lifts {
1530        let mut block = vec![tokens[at], T_BEGIN!()];
1531        block.extend(pieces.into_iter().flat_map(|(_, piece)| piece));
1532        block.push(T_END!());
1533        lifted_any |= read(&block, out);
1534        continue;
1535      }
1536      lifted_any = true;
1537      for (block, piece) in pieces {
1538        if block {
1539          read(&piece, out);
1540        } else {
1541          out.extend(piece);
1542        }
1543      }
1544    }
1545    lifted_any
1546  }
1547  let tokens = body.unlist();
1548  let mut out = Vec::with_capacity(tokens.len());
1549  if read(&tokens, &mut out) {
1550    wrap_bare_author_block_text(Tokens::new(out), false)
1551  } else {
1552    Tokens::new(out)
1553  }
1554}

1556/// The grid's columns by meaning: `toks` cut at each separator the separator query reads at its own level
1557/// (`argument_separators`, OXIDIZED_DESIGN_DIVERGENCES #995: a written `\and`, a `\let` copy, `\And`/`\AND`, whatever it
1558/// means — IEEEtran's `\and` is its column whatever a document made it, as the spelling read it) whose own tokens stand
1559/// at the body's brace level (one inside a brace group, which the query counts at its level, stays in its cell, as
1560/// before). `None` where the query could not rehearse the body or a separator's tokens print something else too: the
1561/// caller cuts by `split_top_level`.
1562fn grid_columns(toks: &[Token]) -> Result<Option<Vec<Vec<Token>>>> {
1563  let Some(separators) = argument_separators(&Tokens::new(toks.to_vec()))? else {
1564    return Ok(None);
1565  };
1566  let mut level = Vec::with_capacity(toks.len());
1567  let mut depth = 0i32;
1568  for t in toks {
1569    level.push(depth);
1570    depth = brace_level(depth, t);
1571  }
1572  let mut cuts: Vec<(usize, usize)> = Vec::new();
1573  for separator in separators {
1574    let Some((start, end)) = separator.span else {
1575      return Ok(None);
1576    };
1577    if level.get(start) == Some(&0) {
1578      cuts.push((start, end));
1579    }
1580  }
1581  let mut columns = Vec::with_capacity(cuts.len() + 1);
1582  let mut from = 0;
1583  for (start, end) in cuts {
1584    columns.push(toks[from..start].to_vec());
1585    from = end;
1586  }
1587  columns.push(toks[from..].to_vec());
1588  Ok(Some(columns))
1589}

1591/// Transpose a genuine IEEEtran `\and`×`\\` author grid from column-major
1592/// (declaration order) to row-major (reading order).
1593///
1594/// html_feedback#6242-adjacent (arXiv:2403.16405): a conference `\author{}` lays
1595/// authors out as a 2-D grid — `\and` starts a new COLUMN, top-level `\\` a new ROW
1596/// within a column. Each `\IEEEauthorblockN` emits its creator in token order, so
1597/// LaTeXML (and Perl) linearize down each column = column-major, scrambling the
1598/// reading (row-major) order shown in the PDF and the arXiv metadata. Transpose so
1599/// creators come out row-major. Guarded TIGHTLY: only a REGULAR grid (≥2 columns,
1600/// every column the SAME number of rows ≥2) is reordered. A single-row `\and` list
1601/// (no top-level `\\`) and `\\` used only INSIDE `\IEEEauthorblockA{…\\…}` (nested,
1602/// depth>0) are left exactly as declared, and so is a grid with a cell naming no author (a column's affiliation rows
1603/// under its name, 2605.15804). Surpass-Perl divergence: OXIDIZED_DESIGN
1604/// #128 (shared upstream bug KNOWN_PERL_ERRORS #94). The grid's tokens are the body as its document macros print it
1605/// (`blocks_as_printed`), its columns the separators the query reads (`grid_columns`) and its rows its `\\`
1606/// (`split_top_level`, by meaning); the transposition stays here, on
1607/// tokens, until the reading holds the grid's events (each `\IEEEauthorblockN` is a ledger block of its own, the
1608/// separators between blocks unrecorded; P2-8a).
1609fn transpose_ieee_author_grid(body: Tokens) -> Result<Tokens> {
1610  let toks = body.unlist();
1611  let columns = match grid_columns(&toks)? {
1612    Some(columns) => columns,
1613    None => split_top_level(&toks, &T_CS!("\\and")),
1614  };
1615  if columns.len() < 2 {
1616    return Ok(Tokens::new(toks)); // not a multi-column grid
1617  }
1618  let grid: Vec<Vec<Vec<Token>>> = columns
1619    .iter()
1620    .map(|c| split_top_level(c, &T_CS!("\\\\")))
1621    .collect();
1622  let rows = grid[0].len();
1623  // Only a REGULAR grid transposes: same row count in every column, and ≥2 rows.
1624  if rows < 2 || grid.iter().any(|c| c.len() != rows) {
1625    return Ok(Tokens::new(toks));
1626  }
1627  // And a grid of authors: every cell that prints something (`prints_nothing`) names one. A row of a column's blocks
1628  // or text without a name is its author's, below the name as printed (`\IEEEauthorblockN{Ann Able} \textit{School
1629  // X}\\ \IEEEauthorblockA{Dept Y\\ City} \and …`, 2605.15804: transposed, every `\IEEEauthorblockA` of that row went
1630  // to the last author of the row above).
1631  let names = |cell: &Vec<Token>| {
1632    prints_nothing(&Tokens::new(cell.clone()))
1633      || cell.iter().any(|t| *t == T_CS!("\\IEEEauthorblockN"))
1634  };
1635  if !grid.iter().flatten().all(names) {
1636    return Ok(Tokens::new(toks));
1637  }
1638  let and = T_CS!("\\and");
1639  let mut out: Vec<Token> = Vec::new();
1640  for r in 0..rows {
1641    for col in &grid {
1642      let cell = &col[r];
1643      if cell.iter().all(|t| *t == T_SPACE!()) {
1644        continue;
1645      }
1646      if !out.is_empty() {
1647        out.push(and);
1648      }
1649      out.extend(cell.iter().copied());
1650    }
1651  }
1652  Ok(Tokens::new(out))
1653}

1655/// Wrap each top-level bare-text run in an IEEE author BLOCK body in `\IEEEauthorblockA{...}` so it attaches to the
1656/// current creator as an affiliation instead of digesting into the document body. A block macro
1657/// (`\IEEEauthorblockN/A`, `\authorblockN/A`) plus its balanced `{...}` argument is copied verbatim; `\and` passes
1658/// through untouched, a run that prints nothing (`prints_nothing`: spaces, line and paragraph breaks, `\vspace`) as its
1659/// spaces and braces, and a run of notes before an `\IEEEauthorblockA` as the document's notes. Without this, the
1660/// trailing bare email in `\author{\IEEEauthorblockN{…}\IEEEauthorblockA{…} \{…\}@host}` (the `\{`/`\}` are
1661/// catcode-12 literal-brace control symbols, so the email sits at brace-depth 0 = top level) is ordinary horizontal
1662/// text and `\lx@IEEE@author@blocks` digests it into the document body as the first paragraph. Real Perl LaTeXML
1663/// bundles the WHOLE author argument into `<personname>` (`\lx@add@authors`), so the tail never leaks —
1664/// GENUINE-RUST-ONLY. Witness arXiv 1901.07768 (`\{anuja, seth.gilbert, tankl\}@comp.nus.edu.sg`).
1665fn wrap_bare_author_block_text(body: Tokens, name_lines: bool) -> Tokens {
1666  let toks = body.unlist();
1667  let is_block = |t: &Token| *t == T_CS!("\\IEEEauthorblockN") || *t == T_CS!("\\IEEEauthorblockA");
1668  // (whether a names block leads with an ordinal: the template's convention, `\IEEEauthorblockN{1\textsuperscript{st} …}`)
1669  let ordinal_names = name_lines
1670    && toks.iter().enumerate().any(|(i, t)| {
1671      *t == T_CS!("\\IEEEauthorblockN")
1672        && block_argument(&toks, i + 1)
1673          .is_some_and(|(open, _)| leads_with_ordinal(&toks[open + 1..]))
1674    });
1675  fn wrap(
1676    run: &mut Vec<Token>,
1677    out: &mut Vec<Token>,
1678    before_affiliation: bool,
1679    ordinal_names: bool,
1680  ) {
1681    // the spans of the run's line breaks: `\\` with its `*` and `[…]`, `\newline`
1682    let mut breaks: Vec<(usize, usize)> = Vec::new();
1683    let mut i = 0;
1684    while i < run.len() {
1685      if run[i] == T_CS!("\\\\") || run[i] == T_CS!("\\newline") {
1686        let double_backslash = run[i] == T_CS!("\\\\");
1687        let past_spaces = |mut at: usize| {
1688          while run.get(at) == Some(&T_SPACE!()) {
1689            at += 1;
1690          }
1691          at
1692        };
1693        let mut end = i + 1;
1694        if double_backslash && run.get(past_spaces(end)) == Some(&T_OTHER!("*")) {
1695          end = past_spaces(end) + 1;
1696        }
1697        if double_backslash && run.get(past_spaces(end)) == Some(&T_OTHER!("[")) {
1698          let open = past_spaces(end);
1699          if let Some(close) = run[open..].iter().position(|t| *t == T_OTHER!("]")) {
1700            end = open + close + 1;
1701          }
1702        }
1703        breaks.push((i, end));
1704        i = end;
1705      } else {
1706        i += 1;
1707      }
1708    }
1709    let in_break = |at: usize| breaks.iter().any(|&(from, to)| (from..to).contains(&at));
1710    let printed: Vec<Token> = run
1711      .iter()
1712      .enumerate()
1713      .filter(|(at, _)| !in_break(*at))
1714      .map(|(_, t)| *t)
1715      .collect();
1716    if prints_nothing(&Tokens::new(printed.clone())) {
1717      // Layout, not a block: the line break between two blocks (`\IEEEauthorblockN{…}\\ \IEEEauthorblockA{…}`,
1718      // 2410.19527), a paragraph break or a `\vspace` (1401.2118, 1705.03529). Wrapped, it was an affiliation printing
1719      // nothing that took the block's authors before the shared `\IEEEauthorblockA` (`annotate=new`), which then
1720      // reached the last only. Its spaces and its braces pass through: a group it opens or closes around a block
1721      // (`\authorblockN{…}\\ {\small \authorblockA{…}}\\ \and`, cs0507001) stays balanced.
1722      out.extend(run.iter().filter(|t| {
1723        matches!(
1724          t.get_catcode(),
1725          Catcode::SPACE | Catcode::BEGIN | Catcode::END
1726        )
1727      }));
1728    } else if let Some(notes) =
1729      document_notes(&printed).filter(|notes| before_affiliation && !notes.is_empty())
1730    {
1731      // The authors' notes between the names and their shared block (`\thanks{…}`, 1508.00691) stay the document's
1732      // notes, as in a block: wrapped, they were such an affiliation printing nothing. Notes no `\IEEEauthorblockA`
1733      // follows are wrapped as before, so the block reader still reads them (the e-mails of
1734      // `\thanks{E-mail: \{lins, shahmohm, helgamal\}@ece.osu.edu.}` after the names, 0710.2705; the note a name's mark
1735      // requests, 1504.01072).
1736      out.extend(notes);
1737    } else {
1738      // Printed text → route to the creator as an affiliation, without the line break it ends with
1739      // (`\IEEEauthorblockN{Ann Able} \textit{School X}\\ \IEEEauthorblockA{…}`, 2605.15804: kept, it sent every
1740      // later block to the last creator).
1741      let mut end = run.len();
1742      loop {
1743        while end > 0 && run[end - 1] == T_SPACE!() {
1744          end -= 1;
1745        }
1746        match breaks.iter().find(|&&(_, to)| to == end) {
1747          Some(&(from, _)) => end = from,
1748          None => break,
1749        }
1750      }
1751      run.truncate(end);
1752      // (a run before the block's affiliation of lines each an ordinal-led name, where the names are, is name lines,
1753      // the template's: `\IEEEauthorblockN{1\textsuperscript{st} …} 3\textsuperscript{rd}Cy Cole \IEEEauthorblockA{…}`,
1754      // 1806.11431, read as an affiliation, took the block's authors; "2nd Floor, Bldg 5" no name, review revP4cR2 S1, S2)
1755      let names = (before_affiliation && ordinal_names)
1756        .then(|| {
1757          split_top_level_lines(run)
1758            .into_iter()
1759            .filter(|line| !prints_nothing(&Tokens::new(line.to_vec())))
1760            .collect::<Vec<_>>()
1761        })
1762        .filter(|lines| lines.iter().all(|line| ordinal_name_line(line)));
1763      if let Some(lines) = names {
1764        for line in lines {
1765          out.push(T_CS!("\\IEEEauthorblockN"));
1766          out.push(T_BEGIN!());
1767          out.extend_from_slice(line);
1768          out.push(T_END!());
1769        }
1770      } else {
1771        out.push(T_CS!("\\IEEEauthorblockA"));
1772        out.push(T_BEGIN!());
1773        out.append(run);
1774        out.push(T_END!());
1775      }
1776    }
1777    run.clear();
1778  }
1779  // A conditional around the blocks (`\ifdefined\hpcacameraready \IEEEauthorblockN{…}… \else … \fi`, 2407.02944)
1780  // keeps its own tokens as written — an `\if…` with its test, each `\else`/`\or`/`\fi` — and the bare text between
1781  // them is wrapped piece by piece: wrapped whole, the `\if` and its `\else`/`\fi` landed in different arguments. A
1782  // conditional the run holds whole (`\ifdefined\x a@b.c\fi` after a block) is wrapped with it, as any bare text.
1783  // `open_in_blocks` counts the conditionals a block's argument opened and left open (`\IEEEauthorblockA{… \if\x1
1784  // a@b.c} \vspace{-5mm} \else x@y.z \fi`, 2606.17718): the bare run's `\else`/`\fi` are theirs, so the run is wrapped
1785  // whole, as before the piecewise split (63x), and its own net depth closes them.
1786  fn flush(
1787    run: &mut Vec<Token>,
1788    out: &mut Vec<Token>,
1789    open_in_blocks: &mut i32,
1790    before_affiliation: bool,
1791    ordinal_names: bool,
1792  ) {
1793    let mut depth = 0i32;
1794    let mut whole = true;
1795    for t in run.iter() {
1796      match lookup_conditional(t) {
1797        Some(ConditionalType::If) => depth += 1,
1798        Some(ConditionalType::Fi) => depth -= 1,
1799        Some(ConditionalType::Else | ConditionalType::Or) if depth == 0 => whole = false,
1800        _ => {},
1801      }
1802      whole &= depth >= 0;
1803    }
1804    if *open_in_blocks > 0 {
1805      *open_in_blocks = (*open_in_blocks + depth).max(0);
1806      wrap(run, out, before_affiliation, ordinal_names);
1807      return;
1808    }
1809    if whole && depth == 0 {
1810      wrap(run, out, before_affiliation, ordinal_names);
1811      return;
1812    }
1813    let mut piece: Vec<Token> = Vec::new();
1814    let mut tokens = run.drain(..).peekable();
1815    while let Some(t) = tokens.next() {
1816      let Some(kind) = lookup_conditional(&t) else {
1817        piece.push(t);
1818        continue;
1819      };
1820      wrap(&mut piece, out, false, ordinal_names);
1821      out.push(t);
1822      if kind != ConditionalType::If {
1823        continue;
1824      }
1825      // the test the primitive reads (tex.web §498-§513): `\ifdefined` one token, `\ifx`/`\if`/`\ifcat` two,
1826      // `\ifcsname` through `\endcsname`, a number's or dimension's up to the space ending it; a `\newif` one none.
1827      // The common shapes only: a relation spaced from its operands (`\ifnum\value{x} > 0`) or an `\expandafter` before
1828      // the `\if` leaves part of the test to be wrapped.
1829      let name = t.with_str(|s| s.to_string());
1830      match name.as_str() {
1831        "\\ifdefined" | "\\ifx" | "\\if" | "\\ifcat" => {
1832          let n = if name == "\\ifdefined" { 1 } else { 2 };
1833          out.extend(tokens.by_ref().take(n));
1834        },
1835        "\\ifcsname" => {
1836          for u in tokens.by_ref() {
1837            out.push(u);
1838            if u == T_CS!("\\endcsname") {
1839              break;
1840            }
1841          }
1842        },
1843        "\\ifnum" | "\\ifdim" | "\\ifodd" | "\\ifcase" | "\\ifeof" | "\\ifvoid" | "\\ifhbox"
1844        | "\\ifvbox" => {
1845          while let Some(u) = tokens.next_if(|u| lookup_conditional(u).is_none()) {
1846            out.push(u);
1847            if u == T_SPACE!() {
1848              break;
1849            }
1850          }
1851        },
1852        _ => {},
1853      }
1854    }
1855    drop(tokens);
1856    wrap(&mut piece, out, before_affiliation, ordinal_names);
1857  }
1858  let mut out: Vec<Token> = Vec::new();
1859  let mut run: Vec<Token> = Vec::new();
1860  let mut open_in_blocks = 0i32;
1861  // An `\and` at the top level of an affiliation block's argument that a block macro follows starts the next author,
1862  // as pdflatex prints it (a brace missing after the first author's block: `\authorblockA{…\\ \and
1863  // \authorblockN{Hadas Kress-Gazit} \authorblockA{…}}}`, 2105.06537). The argument is closed before the `\and`, the
1864  // rest read as top-level blocks, and the argument's own closing brace — the first top-level `}` no run opened —
1865  // dropped (`unclosed`, `run_depth`): copied whole, the second author's blocks were the first author's too.
1866  let mut unclosed = 0usize;
1867  let mut run_depth = 0usize;
1868  let block_follows = |from: usize| {
1869    toks[from..]
1870      .iter()
1871      .find(|v| **v != T_SPACE!() && **v != T_CS!("\\par"))
1872      .is_some_and(is_block)
1873  };
1874  let mut i = 0;
1875  while i < toks.len() {
1876    let t = toks[i];
1877    if is_block(&t) {
1878      let affiliation = t == T_CS!("\\IEEEauthorblockA");
1879      flush(
1880        &mut run,
1881        &mut out,
1882        &mut open_in_blocks,
1883        affiliation,
1884        ordinal_names,
1885      );
1886      out.push(t);
1887      i += 1;
1888      // Copy the block's balanced `{...}` argument verbatim.
1889      if i < toks.len() && toks[i].get_catcode() == Catcode::BEGIN {
1890        let mut depth = 0;
1891        while i < toks.len() {
1892          let u = toks[i];
1893          if affiliation && depth == 1 && u == T_CS!("\\and") && block_follows(i + 1) {
1894            out.push(T_END!());
1895            unclosed += 1;
1896            break;
1897          }
1898          match u.get_catcode() {
1899            Catcode::BEGIN => depth += 1,
1900            Catcode::END => depth -= 1,

2024    }

2027      return false;
2028    }

2030      k -= 1;
2031    }

2033      && [
2034        T_CS!("\\thanks"),
2035        T_CS!("\\footnote"),
2036        T_CS!("\\IEEEauthorblockA"),

2038      ]
2039      .contains(&v[k - 1])
2040  };
2041  // (each reference mark's number and the tokens it spans; the numbers of the other marks that label a line)
2042  let mut marks: Vec<(usize, usize, u32)> = Vec::new();
2043  let mut line_numbers: Vec<u32> = Vec::new();

2045  while i < v.len() {
2046    let t = v[i];

2048      let at = skip_spaces(i + 1);

2051        .iter()
2052        .filter(|t| {
2053          !matches!(
2054            t.get_catcode(),
2055            Catcode::BEGIN | Catcode::END | Catcode::SPACE
2056          )
2057        })
2058        .map(ToString::to_string)
2059        .collect();

2061        marks.push((i, end, n));
2062      }
2063      i = end.max(i + 1);
2064    } else if t.get_catcode() == Catcode::SUPER || t == T_CS!("\\textsuperscript") {
2065      let end = operand_end(skip_spaces(i + 1));

2067        // (its numbers: `$^{1,2}$`'s 1 and 2)
2068        let operand: String = v[i + 1..end].iter().map(ToString::to_string).collect();
2069        line_numbers.extend(
2070          operand
2071            .split(|c: char| !c.is_ascii_digit())
2072            .filter_map(|n| n.parse::<u32>().ok()),
2073        );
2074      }
2075      i = end.max(i + 1);
2076    } else {
2077      i += 1;
2078    }
2079  }
2080  if !marks.iter().any(|(_, _, n)| line_numbers.contains(n)) {
2081    return body;
2082  }

2085      1 => vec![T_OTHER!("*")],
2086      2 => vec![T_CS!("\\dagger")],

2088      4 => vec![T_CS!("\\S")],
2089      5 => vec![T_CS!("\\P")],

2094      7..=9 => [one(n - 6), one(n - 6)].concat(),
2095      _ => Vec::new(),
2096    }
2097  };

2100  let mut in_math = false;
2101  for (start, end, n) in marks {
2102    for t in &v[from..start] {
2103      if t.get_catcode() == Catcode::MATH {
2104        in_math = !in_math;
2105      }
2106      out.push(*t);
2107    }
2108    from = end;
2109    let sym = symbol(n);
2110    // (a number of 10 or more prints its roman numeral, a lettered mark; 0 prints nothing)
2111    if sym.is_empty() {
2112      if n >= 10 {
2113        out.extend([T_CS!("\\textsuperscript"), T_BEGIN!()]);

2118    }

2120      out.push(T_MATH!());
2121    }
2122    out.extend([T_SUPER!(), T_BEGIN!()]);
2123    out.extend(sym);
2124    out.push(T_END!());
2125    if !in_math {
2126      out.push(T_MATH!());
2127    }
2128  }
2129  out.extend_from_slice(&v[from..]);
2130  Tokens::new(out)
2131}

2360/// The argument of the note whose command ends just before `at` (`\thanks[…]{…}`): its tokens, braces dropped, and
2361/// the position after it; past spaces and an optional `[…]`, a group or else one token.
2362fn note_argument(tokens: &[Token], mut at: usize) -> (Vec<Token>, usize) {

2365      *at += 1;
2366    }
2367  };

2370    while at < tokens.len() && tokens[at] != T_OTHER!("]") {
2371      at += 1;
2372    }

2374    skip_spaces(&mut at);
2375  }
2376  if at >= tokens.len() {
2377    return (Vec::new(), tokens.len());
2378  }
2379  if tokens[at].code != Catcode::BEGIN {

2383  for j in at..tokens.len() {
2384    match tokens[j].code {
2385      Catcode::BEGIN => level += 1,
2386      Catcode::END => {
2387        level -= 1;
2388        if level == 0 {
2389          return (tokens[at + 1..j].to_vec(), j + 1);
2390        }
2391      },
2392      _ => {},
2393    }
2394  }

2410    }
2411    while i < tokens.len() && tokens[i] == T_SPACE!() {
2412      out.push(tokens[i]);
2413      i += 1;
2414    }
2415    if i >= tokens.len() || tokens[i].code != Catcode::BEGIN {
2416      continue;
2417    }
2418    let start = i;
2419    let mut level = 0usize;
2420    while i < tokens.len() {
2421      match tokens[i].code {
2422        Catcode::BEGIN => level += 1,
2423        Catcode::END => {
2424          level -= 1;
2425          if level == 0 {
2426            break;
2427          }
2428        },
2429        _ => {},
2430      }
2431      i += 1;
2432    }
2433    if level != 0 || i >= tokens.len() {
2434      out.extend(tokens[start..].iter().copied());
2435      return Ok(Tokens::new(out));
2436    }
2437    out.push(T_BEGIN!());
2438    out.extend(without_stop_after_grade(tokens[start + 1..i].to_vec())?);
2439    out.push(T_END!());
2440    i += 1;
2441  }
2442  Ok(Tokens::new(out))
2443}
