1241/// How many of the document's own macros [`holds_author_structure`] reads into from one call, and how many calls
1242/// [`blocks_as_printed`] expands in one body, at most.
1243const MEANING_READS: usize = 64;

1245/// Whether `token`, a macro of the document's own (`\newcommand`, `\def` in the document or its own files), holds author
1246/// structure by meaning — its definition, or a document macro it calls (recursively), spells a names block (a token
1247/// meaning `\IEEEauthorblockN`: the class's, `\authorblockN`, a `\let` copy), an `\and`-family separator (`\and`,
1248/// `\And`, `\AND`, or a token `\let` to one) or the grid's row break `\\` (or a `\let` copy). Read as the rehearsal reads a block's macros for `\let` copies of its
1249/// separators (`rehearsal::let_copies_of_bound_commands`): the definitions' tokens, no expansion.
1250fn holds_author_structure(token: &Token) -> bool {
1251  // (a copy means the command: `\authorblockN`, the kernel's `\And` and `\AND`, a document's `\let`)
1252  let structure = [T_CS!("\\IEEEauthorblockN"), T_CS!("\\and"), T_CS!("\\\\")];
1253  let document_macro = |t: &Token| -> Option<Tokens> {
1254    let defn = lookup_definition(t).ok().flatten()?;
1255    if !defn.is_expandable() || defn.get_origin() != definition::origin::DefinitionOrigin::Document
1256    {
1257      return None;
1258    }
1259    match defn.get_expansion() {
1260      Some(ExpansionBody::Tokens(body)) => Some(body.clone()),
1261      _ => None,
1262    }
1263  };
1264  let Some(body) = document_macro(token) else {
1265    return false;
1266  };
1267  let mut seen = rustc_hash::FxHashSet::<Token>::default();
1268  let mut pending: Vec<Token> = body.unlist();
1269  let mut macros = 1;
1270  while let Some(t) = pending.pop() {
1271    if !t.is_executable() || !seen.insert(t) {
1272      continue;
1273    }
1274    if structure.iter().any(|s| t == *s || t.defined_as(s)) {
1275      return true;
1276    }
1277    if macros < MEANING_READS
1278      && let Some(inner) = document_macro(&t)
1279    {
1280      macros += 1;
1281      pending.extend(inner.unlist());
1282    }
1283  }
1284  false
1285}

1415  if tokens.iter().filter(|t| is_names(t)).count() < 2 {
1416    return Tokens::new(tokens);
1417  }
1418  let mut out = Vec::with_capacity(tokens.len());
1419  spliced(&tokens, false, &mut out);
1420  Tokens::new(out)
1421}

1446/// An author-block body with each affiliation block at the top level of another's argument read as conference mode
1447/// prints it, on rows of its own in the outer's column (IEEEtran.cls:4653-4665): one led by a mark, or after a line of
1448/// marks alone (its label), lifted out as a block of its own, the outer's text around it bare text again
1449/// (`wrap_bare_author_block_text`); an unmarked one spliced into the outer's lines (1710.07993, 2004.11080, 2107.11588,
1450/// 2605.02682, 1405.1593's `\thanks` wrapper; review revP4cR2 B1-B3; OXIDIZED_DESIGN_DIVERGENCES #1009).
1451fn unnested_affiliation_blocks(body: Tokens) -> Tokens {
1452  fn is_affiliation(t: &Token) -> bool { *t == T_CS!("\\IEEEauthorblockA") }
1453  fn is_break(t: &Token) -> bool { *t == T_CS!("\\\\") || *t == T_CS!("\\newline") }
1454  // `chunk` added to the outer's text, after the line break a spliced block's text owes the text after it
1455  fn append(text: &mut Vec<Token>, chunk: &[Token], owed: &mut bool) {
1456    let first = chunk.iter().position(|t| *t != T_SPACE!());
1457    if *owed && let Some(first) = first.filter(|&k| !is_break(&chunk[k])) {
1458      text.push(T_CS!("\\\\"));
1459      text.extend_from_slice(&chunk[first..]);
1460    } else {
1461      text.extend_from_slice(chunk);
1462    }
1463    *owed &= first.is_none();
1464  }
1465  // `tokens` at one level, each affiliation block's top-level inner blocks read; whether one was lifted out
1466  fn read(tokens: &[Token], out: &mut Vec<Token>) -> bool {
1467    let mut lifted_any = false;
1468    let mut i = 0;
1469    while i < tokens.len() {
1470      let Some((open, close)) =
1471        block_argument(tokens, i + 1).filter(|_| is_affiliation(&tokens[i]))
1472      else {
1473        out.push(tokens[i]);
1474        i += 1;
1475        continue;
1476      };
1477      let arg = &tokens[open + 1..close];
1478      // the outer's text and the blocks lifted out of it, in order (`true`: a block)
1479      let mut pieces: Vec<(bool, Vec<Token>)> = Vec::new();
1480      let mut text: Vec<Token> = Vec::new();
1481      let (mut from, mut j, mut owed, mut lifts) = (0, 0, false, false);
1482      while j < arg.len() {
1483        let Some((inner_open, inner_close)) =
1484          block_argument(arg, j + 1).filter(|_| is_affiliation(&arg[j]))
1485        else {
1486          // (a group skipped whole)
1487          j = block_argument(arg, j).map_or(j + 1, |(_, close)| close + 1);
1488          continue;
1489        };
1490        let lead = &arg[from..j];
1491        let inner = &arg[inner_open + 1..inner_close];
1492        // (the lead's last line, past its last top-level `\\`)
1493        let line = lead.len()
1494          - split_top_level_lines(lead)
1495            .last()
1496            .map_or(0, |last| last.len());
1497        let label = if prints_marks_alone(&lead[line..]) {
1498          Some(line)
1499        } else {
1500          leads_with_written_mark(inner).then_some(lead.len())
1501        };
1502        if let Some(label) = label {
1503          append(&mut text, &lead[..label], &mut owed);
1504          pieces.push((false, std::mem::take(&mut text)));
1505          let mut block = vec![arg[j], T_BEGIN!()];
1506          block.extend_from_slice(&lead[label..]);
1507          block.extend_from_slice(inner);
1508          block.push(T_END!());
1509          pieces.push((true, block));
1510          (owed, lifts) = (false, true);
1511        } else {
1512          // (spliced, on rows of its own: Lab Z under both names, `Dept X, Univ Y\\ \IEEEauthorblockA{Lab Z}`)
1513          append(&mut text, lead, &mut owed);
1514          while text.last() == Some(&T_SPACE!()) {
1515            text.pop();
1516          }
1517          if text.last().is_some_and(|t| !is_break(t)) {
1518            text.push(T_CS!("\\\\"));
1519          }
1520          text.extend_from_slice(inner);
1521          owed = true;
1522        }
1523        from = inner_close + 1;
1524        j = from;
1525      }
1526      let at = i;
1527      i = close + 1;
1528      if from == 0 {
1529        out.extend_from_slice(&tokens[at..i]);
1530        continue;
1531      }
1532      append(&mut text, &arg[from..], &mut owed);
1533      pieces.push((false, text));
1534      if !lifts {
1535        let mut block = vec![tokens[at], T_BEGIN!()];
1536        block.extend(pieces.into_iter().flat_map(|(_, piece)| piece));
1537        block.push(T_END!());
1538        lifted_any |= read(&block, out);
1539        continue;
1540      }
1541      lifted_any = true;
1542      for (block, piece) in pieces {
1543        if block {
1544          read(&piece, out);
1545        } else {
1546          out.extend(piece);
1547        }
1548      }
1549    }
1550    lifted_any
1551  }
1552  let tokens = body.unlist();
1553  let mut out = Vec::with_capacity(tokens.len());
1554  if read(&tokens, &mut out) {
1555    wrap_bare_author_block_text(Tokens::new(out), false)
1556  } else {
1557    Tokens::new(out)
1558  }
1559}

1561/// The grid's columns by meaning: `toks` cut at each separator the separator query reads at its own level
1562/// (`argument_separators`, OXIDIZED_DESIGN_DIVERGENCES #995: a written `\and`, a `\let` copy, `\And`/`\AND`, whatever it
1563/// means — IEEEtran's `\and` is its column whatever a document made it, as the spelling read it) whose own tokens stand
1564/// at the body's brace level (one inside a brace group, which the query counts at its level, stays in its cell, as
1565/// before). `None` where the query could not rehearse the body or a separator's tokens print something else too: the
1566/// caller cuts by `split_top_level`.
1567fn grid_columns(toks: &[Token]) -> Result<Option<Vec<Vec<Token>>>> {
1568  let Some(separators) = argument_separators(&Tokens::new(toks.to_vec()))? else {
1569    return Ok(None);
1570  };
1571  let mut level = Vec::with_capacity(toks.len());
1572  let mut depth = 0i32;
1573  for t in toks {
1574    level.push(depth);
1575    depth = brace_level(depth, t);
1576  }
1577  let mut cuts: Vec<(usize, usize)> = Vec::new();
1578  for separator in separators {
1579    let Some((start, end)) = separator.span else {
1580      return Ok(None);
1581    };
1582    if level.get(start) == Some(&0) {
1583      cuts.push((start, end));
1584    }
1585  }
1586  let mut columns = Vec::with_capacity(cuts.len() + 1);
1587  let mut from = 0;
1588  for (start, end) in cuts {
1589    columns.push(toks[from..start].to_vec());
1590    from = end;
1591  }
1592  columns.push(toks[from..].to_vec());
1593  Ok(Some(columns))
1594}

1596/// Transpose a genuine IEEEtran `\and`×`\\` author grid from column-major
1597/// (declaration order) to row-major (reading order).
1598///
1599/// html_feedback#6242-adjacent (arXiv:2403.16405): a conference `\author{}` lays
1600/// authors out as a 2-D grid — `\and` starts a new COLUMN, top-level `\\` a new ROW
1601/// within a column. Each `\IEEEauthorblockN` emits its creator in token order, so
1602/// LaTeXML (and Perl) linearize down each column = column-major, scrambling the
1603/// reading (row-major) order shown in the PDF and the arXiv metadata. Transpose so
1604/// creators come out row-major. Guarded TIGHTLY: only a REGULAR grid (≥2 columns,
1605/// every column the SAME number of rows ≥2) is reordered. A single-row `\and` list
1606/// (no top-level `\\`) and `\\` used only INSIDE `\IEEEauthorblockA{…\\…}` (nested,
1607/// depth>0) are left exactly as declared, and so is a grid with a cell naming no author (a column's affiliation rows
1608/// under its name, 2605.15804). Surpass-Perl divergence: OXIDIZED_DESIGN
1609/// #128 (shared upstream bug KNOWN_PERL_ERRORS #94). The grid's tokens are the body as its document macros print it
1610/// (`blocks_as_printed`), its columns the separators the query reads (`grid_columns`) and its rows its `\\`
1611/// (`split_top_level`, by meaning); the transposition stays here, on
1612/// tokens, until the reading holds the grid's events (each `\IEEEauthorblockN` is a ledger block of its own, the
1613/// separators between blocks unrecorded; P2-8a).
1614fn transpose_ieee_author_grid(body: Tokens) -> Result<Tokens> {
1615  let toks = body.unlist();
1616  let columns = match grid_columns(&toks)? {
1617    Some(columns) => columns,
1618    None => split_top_level(&toks, &T_CS!("\\and")),
1619  };
1620  if columns.len() < 2 {
1621    return Ok(Tokens::new(toks)); // not a multi-column grid
1622  }
1623  let grid: Vec<Vec<Vec<Token>>> = columns
1624    .iter()
1625    .map(|c| split_top_level(c, &T_CS!("\\\\")))
1626    .collect();
1627  let rows = grid[0].len();
1628  // Only a REGULAR grid transposes: same row count in every column, and ≥2 rows.
1629  if rows < 2 || grid.iter().any(|c| c.len() != rows) {
1630    return Ok(Tokens::new(toks));
1631  }
1632  // And a grid of authors: every cell that prints something (`prints_nothing`) names one. A row of a column's blocks
1633  // or text without a name is its author's, below the name as printed (`\IEEEauthorblockN{Ann Able} \textit{School
1634  // X}\\ \IEEEauthorblockA{Dept Y\\ City} \and …`, 2605.15804: transposed, every `\IEEEauthorblockA` of that row went
1635  // to the last author of the row above).
1636  let names = |cell: &Vec<Token>| {
1637    prints_nothing(&Tokens::new(cell.clone()))
1638      || cell.iter().any(|t| *t == T_CS!("\\IEEEauthorblockN"))
1639  };
1640  if !grid.iter().flatten().all(names) {
1641    return Ok(Tokens::new(toks));
1642  }
1643  let and = T_CS!("\\and");
1644  let mut out: Vec<Token> = Vec::new();
1645  for r in 0..rows {
1646    for col in &grid {
1647      let cell = &col[r];
1648      if cell.iter().all(|t| *t == T_SPACE!()) {
1649        continue;
1650      }
1651      if !out.is_empty() {
1652        out.push(and);
1653      }
1654      out.extend(cell.iter().copied());
1655    }
1656  }
1657  Ok(Tokens::new(out))
1658}

1660/// Wrap each top-level bare-text run in an IEEE author BLOCK body in `\IEEEauthorblockA{...}` so it attaches to the
1661/// current creator as an affiliation instead of digesting into the document body. A block macro
1662/// (`\IEEEauthorblockN/A`, `\authorblockN/A`) plus its balanced `{...}` argument is copied verbatim; `\and` passes
1663/// through untouched, a run that prints nothing (`prints_nothing`: spaces, line and paragraph breaks, `\vspace`) as its
1664/// spaces and braces, and a run of notes before an `\IEEEauthorblockA` as the document's notes. Without this, the
1665/// trailing bare email in `\author{\IEEEauthorblockN{…}\IEEEauthorblockA{…} \{…\}@host}` (the `\{`/`\}` are
1666/// catcode-12 literal-brace control symbols, so the email sits at brace-depth 0 = top level) is ordinary horizontal
1667/// text and `\lx@IEEE@author@blocks` digests it into the document body as the first paragraph. Real Perl LaTeXML
1668/// bundles the WHOLE author argument into `<personname>` (`\lx@add@authors`), so the tail never leaks —
1669/// GENUINE-RUST-ONLY. Witness arXiv 1901.07768 (`\{anuja, seth.gilbert, tankl\}@comp.nus.edu.sg`).
1670fn wrap_bare_author_block_text(body: Tokens, name_lines: bool) -> Tokens {
1671  let toks = body.unlist();
1672  let is_block = |t: &Token| *t == T_CS!("\\IEEEauthorblockN") || *t == T_CS!("\\IEEEauthorblockA");
1673  // (whether a names block leads with an ordinal: the template's convention, `\IEEEauthorblockN{1\textsuperscript{st} …}`)
1674  let ordinal_names = name_lines
1675    && toks.iter().enumerate().any(|(i, t)| {
1676      *t == T_CS!("\\IEEEauthorblockN")
1677        && block_argument(&toks, i + 1)
1678          .is_some_and(|(open, _)| leads_with_ordinal(&toks[open + 1..]))
1679    });
1680  fn wrap(
1681    run: &mut Vec<Token>,
1682    out: &mut Vec<Token>,
1683    before_affiliation: bool,
1684    ordinal_names: bool,
1685  ) {
1686    // the spans of the run's line breaks: `\\` with its `*` and `[…]`, `\newline`
1687    let mut breaks: Vec<(usize, usize)> = Vec::new();
1688    let mut i = 0;
1689    while i < run.len() {
1690      if run[i] == T_CS!("\\\\") || run[i] == T_CS!("\\newline") {
1691        let double_backslash = run[i] == T_CS!("\\\\");
1692        let past_spaces = |mut at: usize| {
1693          while run.get(at) == Some(&T_SPACE!()) {
1694            at += 1;
1695          }
1696          at
1697        };
1698        let mut end = i + 1;
1699        if double_backslash && run.get(past_spaces(end)) == Some(&T_OTHER!("*")) {
1700          end = past_spaces(end) + 1;
1701        }
1702        if double_backslash && run.get(past_spaces(end)) == Some(&T_OTHER!("[")) {
1703          let open = past_spaces(end);
1704          if let Some(close) = run[open..].iter().position(|t| *t == T_OTHER!("]")) {
1705            end = open + close + 1;
1706          }
1707        }
1708        breaks.push((i, end));
1709        i = end;
1710      } else {
1711        i += 1;
1712      }
1713    }
1714    let in_break = |at: usize| breaks.iter().any(|&(from, to)| (from..to).contains(&at));
1715    let printed: Vec<Token> = run
1716      .iter()
1717      .enumerate()
1718      .filter(|(at, _)| !in_break(*at))
1719      .map(|(_, t)| *t)
1720      .collect();
1721    if prints_nothing(&Tokens::new(printed.clone())) {
1722      // Layout, not a block: the line break between two blocks (`\IEEEauthorblockN{…}\\ \IEEEauthorblockA{…}`,
1723      // 2410.19527), a paragraph break or a `\vspace` (1401.2118, 1705.03529). Wrapped, it was an affiliation printing
1724      // nothing that took the block's authors before the shared `\IEEEauthorblockA` (`annotate=new`), which then
1725      // reached the last only. Its spaces and its braces pass through: a group it opens or closes around a block
1726      // (`\authorblockN{…}\\ {\small \authorblockA{…}}\\ \and`, cs0507001) stays balanced.
1727      out.extend(run.iter().filter(|t| {
1728        matches!(
1729          t.get_catcode(),
1730          Catcode::SPACE | Catcode::BEGIN | Catcode::END
1731        )
1732      }));
1733    } else if let Some(notes) =
1734      document_notes(&printed).filter(|notes| before_affiliation && !notes.is_empty())
1735    {
1736      // The authors' notes between the names and their shared block (`\thanks{…}`, 1508.00691) stay the document's
1737      // notes, as in a block: wrapped, they were such an affiliation printing nothing. Notes no `\IEEEauthorblockA`
1738      // follows are wrapped as before, so the block reader still reads them (the e-mails of
1739      // `\thanks{E-mail: \{lins, shahmohm, helgamal\}@ece.osu.edu.}` after the names, 0710.2705; the note a name's mark
1740      // requests, 1504.01072).
1741      out.extend(notes);
1742    } else {
1743      // Printed text → route to the creator as an affiliation, without the line break it ends with
1744      // (`\IEEEauthorblockN{Ann Able} \textit{School X}\\ \IEEEauthorblockA{…}`, 2605.15804: kept, it sent every
1745      // later block to the last creator).
1746      let mut end = run.len();
1747      loop {
1748        while end > 0 && run[end - 1] == T_SPACE!() {
1749          end -= 1;
1750        }
1751        match breaks.iter().find(|&&(_, to)| to == end) {
1752          Some(&(from, _)) => end = from,
1753          None => break,
1754        }
1755      }
1756      run.truncate(end);
1757      // (a run before the block's affiliation of lines each an ordinal-led name, where the names are, is name lines,
1758      // the template's: `\IEEEauthorblockN{1\textsuperscript{st} …} 3\textsuperscript{rd}Cy Cole \IEEEauthorblockA{…}`,
1759      // 1806.11431, read as an affiliation, took the block's authors; "2nd Floor, Bldg 5" no name, review revP4cR2 S1, S2)
1760      let names = (before_affiliation && ordinal_names)
1761        .then(|| {
1762          split_top_level_lines(run)
1763            .into_iter()
1764            .filter(|line| !prints_nothing(&Tokens::new(line.to_vec())))
1765            .collect::<Vec<_>>()
1766        })
1767        .filter(|lines| lines.iter().all(|line| ordinal_name_line(line)));
1768      if let Some(lines) = names {
1769        for line in lines {
1770          out.push(T_CS!("\\IEEEauthorblockN"));
1771          out.push(T_BEGIN!());
1772          out.extend_from_slice(line);
1773          out.push(T_END!());
1774        }
1775      } else {
1776        out.push(T_CS!("\\IEEEauthorblockA"));
1777        out.push(T_BEGIN!());
1778        out.append(run);
1779        out.push(T_END!());
1780      }
1781    }
1782    run.clear();
1783  }
1784  // A conditional around the blocks (`\ifdefined\hpcacameraready \IEEEauthorblockN{…}… \else … \fi`, 2407.02944)
1785  // keeps its own tokens as written — an `\if…` with its test, each `\else`/`\or`/`\fi` — and the bare text between
1786  // them is wrapped piece by piece: wrapped whole, the `\if` and its `\else`/`\fi` landed in different arguments. A
1787  // conditional the run holds whole (`\ifdefined\x a@b.c\fi` after a block) is wrapped with it, as any bare text.
1788  // `open_in_blocks` counts the conditionals a block's argument opened and left open (`\IEEEauthorblockA{… \if\x1
1789  // a@b.c} \vspace{-5mm} \else x@y.z \fi`, 2606.17718): the bare run's `\else`/`\fi` are theirs, so the run is wrapped
1790  // whole, as before the piecewise split (63x), and its own net depth closes them.
1791  fn flush(
1792    run: &mut Vec<Token>,
1793    out: &mut Vec<Token>,
1794    open_in_blocks: &mut i32,
1795    before_affiliation: bool,
1796    ordinal_names: bool,
1797  ) {
1798    let mut depth = 0i32;
1799    let mut whole = true;
1800    for t in run.iter() {
1801      match lookup_conditional(t) {
1802        Some(ConditionalType::If) => depth += 1,
1803        Some(ConditionalType::Fi) => depth -= 1,
1804        Some(ConditionalType::Else | ConditionalType::Or) if depth == 0 => whole = false,
1805        _ => {},
1806      }
1807      whole &= depth >= 0;
1808    }
1809    if *open_in_blocks > 0 {
1810      *open_in_blocks = (*open_in_blocks + depth).max(0);
1811      wrap(run, out, before_affiliation, ordinal_names);
1812      return;
1813    }
1814    if whole && depth == 0 {
1815      wrap(run, out, before_affiliation, ordinal_names);
1816      return;
1817    }
1818    let mut piece: Vec<Token> = Vec::new();
1819    let mut tokens = run.drain(..).peekable();
1820    while let Some(t) = tokens.next() {
1821      let Some(kind) = lookup_conditional(&t) else {
1822        piece.push(t);
1823        continue;
1824      };
1825      wrap(&mut piece, out, false, ordinal_names);
1826      out.push(t);
1827      if kind != ConditionalType::If {
1828        continue;
1829      }
1830      // the test the primitive reads (tex.web §498-§513): `\ifdefined` one token, `\ifx`/`\if`/`\ifcat` two,
1831      // `\ifcsname` through `\endcsname`, a number's or dimension's up to the space ending it; a `\newif` one none.
1832      // The common shapes only: a relation spaced from its operands (`\ifnum\value{x} > 0`) or an `\expandafter` before
1833      // the `\if` leaves part of the test to be wrapped.
1834      let name = t.with_str(|s| s.to_string());
1835      match name.as_str() {
1836        "\\ifdefined" | "\\ifx" | "\\if" | "\\ifcat" => {
1837          let n = if name == "\\ifdefined" { 1 } else { 2 };
1838          out.extend(tokens.by_ref().take(n));
1839        },
1840        "\\ifcsname" => {
1841          for u in tokens.by_ref() {
1842            out.push(u);
1843            if u == T_CS!("\\endcsname") {
1844              break;
1845            }
1846          }
1847        },
1848        "\\ifnum" | "\\ifdim" | "\\ifodd" | "\\ifcase" | "\\ifeof" | "\\ifvoid" | "\\ifhbox"
1849        | "\\ifvbox" => {
1850          while let Some(u) = tokens.next_if(|u| lookup_conditional(u).is_none()) {
1851            out.push(u);
1852            if u == T_SPACE!() {
1853              break;
1854            }
1855          }
1856        },
1857        _ => {},
1858      }
1859    }
1860    drop(tokens);
1861    wrap(&mut piece, out, before_affiliation, ordinal_names);
1862  }
1863  let mut out: Vec<Token> = Vec::new();
1864  let mut run: Vec<Token> = Vec::new();
1865  let mut open_in_blocks = 0i32;
1866  // An `\and` at the top level of an affiliation block's argument that a block macro follows starts the next author,
1867  // as pdflatex prints it (a brace missing after the first author's block: `\authorblockA{…\\ \and
1868  // \authorblockN{Hadas Kress-Gazit} \authorblockA{…}}}`, 2105.06537). The argument is closed before the `\and`, the
1869  // rest read as top-level blocks, and the argument's own closing brace — the first top-level `}` no run opened —
1870  // dropped (`unclosed`, `run_depth`): copied whole, the second author's blocks were the first author's too.
1871  let mut unclosed = 0usize;
1872  let mut run_depth = 0usize;
1873  let block_follows = |from: usize| {
1874    toks[from..]
1875      .iter()
1876      .find(|v| **v != T_SPACE!() && **v != T_CS!("\\par"))
1877      .is_some_and(is_block)
1878  };
1879  let mut i = 0;
1880  while i < toks.len() {
1881    let t = toks[i];
1882    if is_block(&t) {
1883      let affiliation = t == T_CS!("\\IEEEauthorblockA");
1884      flush(
1885        &mut run,
1886        &mut out,
1887        &mut open_in_blocks,
1888        affiliation,
1889        ordinal_names,
1890      );
1891      out.push(t);
1892      i += 1;
1893      // Copy the block's balanced `{...}` argument verbatim.

2404fn block_grade_stops(body: Tokens) -> Result<Tokens> {
2405  let tokens = body.unlist();
2406  let mut out: Vec<Token> = Vec::new();
2407  let mut i = 0;
2408  while i < tokens.len() {
2409    let t = tokens[i];
2410    out.push(t);
2411    i += 1;
2412    // (its old name too, read as this one by `with_block_names`: 1511.03982's grades)
2413    if t != T_CS!("\\IEEEauthorblockN") {
2414      continue;
2415    }
2416    while i < tokens.len() && tokens[i] == T_SPACE!() {
2417      out.push(tokens[i]);
2418      i += 1;
2419    }
2420    if i >= tokens.len() || tokens[i].code != Catcode::BEGIN {
2421      continue;
2422    }
2423    let start = i;
2424    let mut level = 0usize;
2425    while i < tokens.len() {
2426      match tokens[i].code {
2427        Catcode::BEGIN => level += 1,
2428        Catcode::END => {
2429          level -= 1;
2430          if level == 0 {
2431            break;
2432          }
2433        },
2434        _ => {},
2435      }
2436      i += 1;
2437    }
2438    if level != 0 || i >= tokens.len() {
2439      out.extend(tokens[start..].iter().copied());
2440      return Ok(Tokens::new(out));
2441    }
2442    out.push(T_BEGIN!());
2443    out.extend(without_stop_after_grade(tokens[start + 1..i].to_vec())?);
2444    out.push(T_END!());
2445    i += 1;
2446  }
2447  Ok(Tokens::new(out))
2448}

2460/// The IEEE journal template's block notes routed by kind (DG 2026-11-17, OXIDIZED_DESIGN_DIVERGENCES #1072): in a plain
2461/// `\author{…}` whose names print no mark, a `\thanks` sentence "<names> is/are with <affiliation>[. E-mail: <addresses>]"
2462/// becomes the affiliation and e-mail contacts of the authors it names, and a note "Corresponding author: <name>" that
2463/// author's correspondence, each placed after the name (as an `\IEEEmembership` grade is); every other note (funding,
2464/// "Manuscript received …") stays the title's. A note naming no author, or one ambiguously, or printing more than the
2465/// reading accounts for, stays whole (2003.11216, 2206.13704, 1207.6084). The class prints them as unmarked footnotes
2466/// (IEEEtran.cls:4756-4758), whose text names the authors.
2467fn route_author_thanks(body: Vec<Token>) -> Vec<Token> {
2468  let thanks = T_CS!("\\thanks");
2469  // (a body whose names print marks pairs its notes by them, the frontmatter reader's: left alone)
2470  let marked = body.iter().any(|t| {
2471    t.get_catcode() == Catcode::MATH
2472      || [
2473        T_CS!("\\IEEEauthorrefmark"),
2474        T_CS!("\\textsuperscript"),
2475        T_CS!("\\footnotemark"),
2476        T_CS!("\\footnote"),
2477        T_CS!("\\IEEEauthorblockN"),
2478      ]
2479      .contains(t)
2480  });
2481  if marked || !body.contains(&thanks) {
2482    return body;
2483  }
2484  // Each top-level `\thanks{…}`: (command, end after its group), and the names' units
2485  let mut notes: Vec<(usize, usize)> = Vec::new();
2486  let mut depth = 0;
2487  let mut i = 0;
2488  while i < body.len() {
2489    if depth == 0 && body[i] == thanks {
2490      let open = i
2491        + 1
2492        + body[i + 1..]
2493          .iter()
2494          .take_while(|t| **t == T_SPACE!())
2495          .count();
2496      if body
2497        .get(open)
2498        .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
2499      {
2500        return body;
2501      }
2502      let end = group_end(&body, open);
2503      notes.push((i, end));
2504      i = end;
2505      continue;
2506    }
2507    depth = brace_level(depth, &body[i]);
2508    i += 1;
2509  }
2510  let names = author_units(&body, &notes);
2511  if names.is_empty() {
2512    return body;
2513  }
2514  // (each note's replacement, and the contacts each name gains)
2515  let mut replaced: Vec<Option<Vec<Token>>> = vec![None; notes.len()];
2516  let mut gained: Vec<Vec<Token>> = vec![Vec::new(); names.len()];
2517  for (n, &(start, end)) in notes.iter().enumerate() {
2518    let open = start
2519      + 1
2520      + body[start + 1..]
2521        .iter()
2522        .take_while(|t| **t == T_SPACE!())
2523        .count();
2524    let note = &body[open + 1..end - 1];
2525    if let Some((kept, contacts)) = route_note(note, &names) {
2526      replaced[n] = Some(if kept.is_empty() {
2527        Vec::new()
2528      } else {
2529        let mut call = vec![thanks, T_BEGIN!()];
2530        call.extend(kept);
2531        call.push(T_END!());
2532        call
2533      });
2534      for (owner, contact) in contacts {
2535        gained[owner].extend(contact);
2536      }
2537    }
2538  }
2539  if replaced.iter().all(Option::is_none) {
2540    return body;
2541  }
2542  let mut out = Vec::with_capacity(body.len());
2543  for (at, token) in body.iter().enumerate() {
2544    for (unit, name) in names.iter().enumerate() {
2545      if name.insert_at == at {
2546        out.extend(gained[unit].iter().copied());
2547      }
2548    }
2549    if let Some(n) = notes.iter().position(|&(s, e)| s <= at && at < e)
2550      && let Some(call) = &replaced[n]
2551    {
2552      if notes[n].0 == at {
2553        out.extend(call.iter().copied());
2554      }
2555      continue;
2556    }
2557    out.push(*token);
2558  }
2559  for (unit, name) in names.iter().enumerate() {
2560    if name.insert_at >= body.len() {
2561      out.extend(gained[unit].iter().copied());
2562    }
2563  }
2564  out
2565}

2586/// The authors of a plain body (`notes` skipped): its top-level pieces between `,`, `\and`, `\\` and the word "and"; a
2587/// piece printing no letter (`~\IEEEmembership{Member,~IEEE}`) goes with the name before it.
2588fn author_units(body: &[Token], notes: &[(usize, usize)]) -> Vec<AuthorUnit> {
2589  let grade = T_CS!("\\IEEEmembership");
2590  let mut units: Vec<AuthorUnit> = Vec::new();
2591  let mut words = String::new();
2592  // (the piece's first note, before which its contacts go, and the end of what it printed last: a contact goes right
2593  // after its name and grade, never past a blank line or a `\\` to the next line's name, 2106.02811)
2594  let mut first_note: Option<usize> = None;
2595  let mut printed_end = 0;
2596  let mut depth = 0;
2597  let flush = |words: &mut String, end: usize, units: &mut Vec<AuthorUnit>| {
2598    let piece: Vec<String> = words.split_whitespace().map(str::to_string).collect();
2599    words.clear();
2600    if piece.iter().any(|w| w.chars().any(char::is_alphabetic)) {
2601      units.push(AuthorUnit {
2602        words:     piece,
2603        insert_at: end,
2604      });
2605    } else if let Some(last) = units.last_mut() {
2606      last.insert_at = end;
2607    }
2608  };
2609  let letter = |k: usize, c: &str| {
2610    body
2611      .get(k)
2612      .is_some_and(|t| t.get_catcode() == Catcode::LETTER && t.with_str(|s| s == c))
2613  };
2614  let spacing = |k: Option<&Token>| k.is_none_or(|t| *t == T_SPACE!() || *t == T_ACTIVE!('~'));
2615  let and_at = |at: usize| {
2616    // (a note before "and" prints nothing, so the space before the note is the word's: `Sidky,  \thanks{…}%` then
2617    // `and Xiaochuan Pan` reads " and ", 1105.4002)
2618    let mut prev = at;
2619    while let Some(&(start, _)) = notes.iter().find(|&&(_, end)| end == prev) {
2620      prev = start;
2621    }
2622    prev > 0
2623      && letter(at, "a")
2624      && letter(at + 1, "n")
2625      && letter(at + 2, "d")
2626      && spacing(body.get(prev - 1))
2627      && spacing(body.get(at + 3))
2628  };
2629  let mut at = 0;
2630  while at < body.len() {
2631    if let Some(&(_, end)) = notes.iter().find(|&&(s, _)| s == at) {
2632      // (a note before the piece's first letter, `Ann Able, \thanks{…} Bob Baker`, follows the author before: it is
2633      // not where the next one's contacts go, 2606.25681)
2634      if words.chars().any(char::is_alphabetic) {
2635        first_note.get_or_insert(at);
2636      }
2637      at = end;
2638      continue;
2639    }
2640    let t = &body[at];
2641    if depth == 0 && *t == grade {
2642      let open = at
2643        + 1
2644        + body[at + 1..]
2645          .iter()
2646          .take_while(|t| **t == T_SPACE!())
2647          .count();
2648      at = if body
2649        .get(open)
2650        .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
2651      {
2652        group_end(body, open)
2653      } else {
2654        open
2655      };
2656      printed_end = at;
2657      continue;
2658    }
2659    let word_and = depth == 0 && and_at(at);
2660    if depth == 0
2661      && (word_and || *t == T_OTHER!(",") || *t == T_CS!("\\and") || *t == T_CS!("\\\\"))
2662    {
2663      flush(
2664        &mut words,
2665        first_note.take().unwrap_or(at).min(printed_end),
2666        &mut units,
2667      );
2668      at += if word_and { 3 } else { 1 };
2669      continue;
2670    }
2671    depth = brace_level(depth, t);
2672    if *t != T_SPACE!() && *t != T_CS!("\\par") {
2673      printed_end = at + 1;
2674    }
2675    match t.get_catcode() {
2676      Catcode::LETTER | Catcode::OTHER => t.with_str(|s| words.push_str(s)),
2677      Catcode::SPACE | Catcode::ACTIVE => words.push(' '),
2678      _ => {},
2679    }
2680    at += 1;
2681  }
2682  flush(
2683    &mut words,
2684    first_note.unwrap_or(body.len()).min(printed_end),
2685    &mut units,
2686  );
2687  units
2688}

2690/// A note's printed text, spaces collapsed, each byte's token and brace depth beside it (`\{`, `\}`, `\_` their
2691/// characters, `~` and `\\` a space, other commands nothing).
2692fn note_text(note: &[Token]) -> (String, Vec<(usize, i32)>) {
2693  let mut text = String::new();
2694  let mut at_byte: Vec<(usize, i32)> = Vec::new();
2695  let mut depth = 0;
2696  for (at, t) in note.iter().enumerate() {
2697    let piece: Option<String> = match t.get_catcode() {
2698      Catcode::LETTER | Catcode::OTHER => Some(t.with_str(str::to_string)),
2699      Catcode::SPACE | Catcode::ACTIVE => Some(s!(" ")),
2700      Catcode::CS => t.with_str(|s| match s {
2701        "\\{" | "\\textbraceleft" | "\\lbrace" => Some("{".to_string()),
2702        "\\}" | "\\textbraceright" | "\\rbrace" => Some("}".to_string()),
2703        "\\_" | "\\textunderscore" => Some(s!("_")),
2704        "\\\\" | "\\newline" | "\\par" | "\\ " => Some(s!(" ")),
2705        _ => None,
2706      }),
2707      _ => None,
2708    };
2709    let inner = depth;
2710    depth = brace_level(depth, t);
2711    if let Some(piece) = piece {
2712      for c in piece.chars() {
2713        let c = if c.is_whitespace() { ' ' } else { c };
2714        if c == ' ' && (text.is_empty() || text.ends_with(' ')) {
2715          continue;
2716        }
2717        text.push(c);
2718        at_byte.extend(std::iter::repeat_n((at, inner), c.len_utf8()));
2719      }
2720    }
2721  }
2722  (text, at_byte)
2723}

2725/// The authors a list of names in a note's sentence names ("B. Cheng and Z. Li", "X. Wang", "Bin Cheng, Xiangke Wang,
2726/// and Zhongkui Li"), each one author's alone: its surname that author's last word, its first initial that author's
2727/// first word's (case aside). None where a name matches no author, or more than one.
2728fn named_authors(list: &str, names: &[AuthorUnit]) -> Option<Vec<usize>> {
2729  static SPLIT_RX: LazyLock<Regex> = LazyLock::new(|| Regex::new(r",?\s+and\s+|\s*,\s*").unwrap());
2730  let fold = |w: &str| -> String {
2731    w.chars()
2732      .filter(|c| c.is_alphabetic())
2733      .flat_map(char::to_lowercase)
2734      .collect()
2735  };
2736  let mut owners = Vec::new();
2737  for name in SPLIT_RX.split(list.trim()) {
2738    let words: Vec<&str> = name.split_whitespace().collect();
2739    let (surname, given) = words.split_last()?;
2740    let surname = fold(surname);
2741    let initial = given.first().and_then(|w| fold(w).chars().next());
2742    let matches: Vec<usize> = names
2743      .iter()
2744      .enumerate()
2745      .filter(|(_, unit)| {
2746        let (Some(last), Some(first)) = (unit.words.last(), unit.words.first()) else {
2747          return false;
2748        };
2749        unit.words.len() >= 2
2750          && fold(last) == surname
2751          && initial.is_none_or(|i| fold(first).starts_with(i))
2752      })
2753      .map(|(at, _)| at)
2754      .collect();
2755    let [owner] = matches[..] else {
2756      return None;
2757    };
2758    if owners.contains(&owner) {
2759      return None;
2760    }
2761    owners.push(owner);
2762  }
2763  Some(owners)
2764}

2766/// The tokens of `note` printing its text's bytes `from..to`, spaces trimmed, its groups whole: None where the span
2767/// starts inside a group.
2768fn note_span(
2769  note: &[Token],
2770  text: &str,
2771  at_byte: &[(usize, i32)],
2772  from: usize,
2773  to: usize,
2774) -> Option<Vec<Token>> {
2775  let piece = &text[from..to];
2776  let from = from + (piece.len() - piece.trim_start().len());
2777  let to = from + piece.trim().len();
2778  if from >= to {
2779    return Some(Vec::new());
2780  }
2781  let (start, depth) = at_byte[from];
2782  if depth != 0 {
2783    return None;
2784  }
2785  let mut end = at_byte[to - 1].0 + 1;
2786  let mut level = note[start..end].iter().fold(0, brace_level);
2787  while level > 0 && end < note.len() {
2788    level = brace_level(level, &note[end]);
2789    end += 1;
2790  }
2791  (level == 0).then(|| note[start..end].to_vec())
2792}

2794/// The addresses an e-mail line prints (`{bincheng,zhongkli}@pku.edu.cn` one per local part), None where it prints
2795/// more than addresses and the words between them.
2796fn email_line_addresses(line: &str) -> Option<Vec<String>> {
2797  static LIST_RX: LazyLock<Regex> = LazyLock::new(|| {
2798    Regex::new(r"\{([^{}@]+)\}\s*@\s*([A-Za-z0-9][A-Za-z0-9.\-]*[A-Za-z0-9])").unwrap()
2799  });
2800  static ONE_RX: LazyLock<Regex> = LazyLock::new(|| {
2801    Regex::new(r"[A-Za-z0-9][A-Za-z0-9._%+\-]*@[A-Za-z0-9][A-Za-z0-9.\-]*[A-Za-z0-9]").unwrap()
2802  });
2803  static REST_RX: LazyLock<Regex> =
2804    LazyLock::new(|| Regex::new(r"^(?:[\s,;.{}()]|\band\b)*$").unwrap());
2805  let mut addresses = Vec::new();
2806  let rest = LIST_RX.replace_all(line, |c: &regex::Captures| {
2807    for local in c[1].split(',') {
2808      addresses.push(s!("{}@{}", local.trim(), &c[2]));
2809    }
2810    " "
2811  });
2812  let rest = ONE_RX.replace_all(&rest, |c: &regex::Captures| {
2813    addresses.push(c[0].to_string());
2814    " "
2815  });
2816  (REST_RX.is_match(&rest)
2817    && !addresses.is_empty()
2818    && addresses.iter().all(|a| !a.starts_with('@')))
2819  .then_some(addresses)
2820}

2836/// The affiliation of a "<names> is/are with" sentence, read from `segment` (its text up to the next such sentence or
2837/// the note's end): its length, and the addresses of the e-mail tail after it. An affiliation is one sentence, ending
2838/// at ". " before a capital that ends no initial; after it, only an e-mail tail, labelled ("E-mail: …", "(e-mail: …)")
2839/// or a bare run of addresses (2608.15977, 1805.07098). None for anything else (a funding sentence: 1901.06512,
2840/// 2003.04031, 2008.12874, 2007.06695), so the note stays whole.
2841fn affiliation_and_addresses(segment: &str) -> Option<(usize, Vec<String>)> {
2842  let boundary = segment
2843    .match_indices(". ")
2844    .find(|&(at, _)| {
2845      !ends_initial(&segment[..at])
2846        && segment[at + 2..]
2847          .chars()
2848          .next()
2849          .is_some_and(char::is_uppercase)
2850    })
2851    .map(|(at, _)| at + 1);
2852  let (sentence, rest) = segment.split_at(boundary.unwrap_or(segment.len()));
2853  // (the sentence's own tail: its label, or its trailing run of addresses)
2854  let trim = |end: usize| sentence[..end].trim_end_matches([' ', ',', ';', '(']).len();
2855  let (end, mut addresses) = if let Some(label) = EMAIL_LABEL_RX.find(sentence) {
2856    (
2857      trim(label.start()),
2858      email_line_addresses(&sentence[label.end()..])?,
2859    )
2860  } else if let Some(run) = sentence.find('@').and_then(|first| {
2861    (0..first)
2862      .filter(|&p| {
2863        sentence.is_char_boundary(p)
2864          && (p == 0 || sentence[..p].ends_with(' ') || sentence[p..].starts_with(['(', '{']))
2865          && !sentence[p..].starts_with("and ")
2866      })
2867      .find_map(|p| Some((p, email_line_addresses(&sentence[p..])?)))
2868  }) {
2869    (trim(run.0), run.1)
2870  } else {
2871    (sentence.len(), Vec::new())
2872  };
2873  let rest = rest.trim();
2874  if !rest.is_empty() {
2875    if !addresses.is_empty() {
2876      return None;
2877    }
2878    addresses = match EMAIL_LABEL_RX.find(rest) {
2879      Some(label) if label.start() == 0 => email_line_addresses(&rest[label.end()..])?,
2880      _ => email_line_addresses(rest)?,
2881    };
2882  }
2883  // (a parenthesis the affiliation leaves open is another field's: "Town (Tel: 555-1234; e-mail: …)" keeps the note whole)
2884  let affiliation = &sentence[..end];
2885  if affiliation.matches('(').count() != affiliation.matches(')').count() {
2886    return None;
2887  }
2888  Some((end, addresses))
2889}

2894/// One `\thanks` of the author block routed (`route_author_thanks`): the text it keeps as the title's note, and each
2895/// contact with the author it goes to. None where it routes nothing.
2896fn route_note(note: &[Token], names: &[AuthorUnit]) -> Option<(Vec<Token>, Vec<RoutedContact>)> {
2897  const NAME: &str = r"\p{Lu}[\p{L}'’.\-]*(?: \p{Lu}[\p{L}'’.\-]*)*";
2898  static CLAUSE_RX: LazyLock<Regex> = LazyLock::new(|| {
2899    Regex::new(&s!(
2900      r"^((?:{NAME})(?:(?:,? and |, )(?:{NAME}))*) (?:is|are)(?: also| now| currently| presently)? with "
2901    ))
2902    .unwrap()
2903  });
2904  static CORRESPONDING_RX: LazyLock<Regex> =
2905    LazyLock::new(|| Regex::new(&s!(r"^(?i:corresponding author)(?::| is) ({NAME})\.?$")).unwrap());
2906  static ANOTHER_CLAUSE_RX: LazyLock<Regex> =
2907    LazyLock::new(|| Regex::new(r"\b(?:is|are|was|were)(?: \w+)? with\b").unwrap());
2908  // (a second sentence an initial hid: "Building C. This work …")
2909  static SENTENCE_RX: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\. \p{Lu}").unwrap());
2910  let contact = |role: &str, content: Vec<Token>| -> Vec<Token> {
2911    let mut call = vec![T_CS!("\\lx@add@contact"), T_OTHER!("[")];
2912    call.extend(ExplodeText!(s!("role={role}")));
2913    call.push(T_OTHER!("]"));
2914    call.push(T_BEGIN!());
2915    call.extend(content);
