1//! filehook.sty — hooks for input files
2//! Perl: filehook.sty.ltxml
3use crate::prelude::*;
4
5#[rustfmt::skip]
6LoadDefinitions!({
7  // Perl filehook.sty.ltxml L18-20: this comparison can't reliably work
8  // with latexml's subroutine macro definitions, so default it to *true*
9  // to avoid needless warnings. Perl adds `locked => 1` — critical here
10  // because the very next line (InputDefinitions) pulls in the raw
11  // filehook.sty which itself defines \filehook@cmp; without the lock,
12  // the raw-sty redefinition replaces our always-true stub.
13  DefMacro!("\\filehook@cmp{}{}", "\\@firstoftwo", locked => true);
14
15  // The raw filehook.sty:27 is a 1-line dispatcher:
16  //   \@ifl@t@r\fmtversion{2020/10/01}{\RequirePackage{filehook-2020}}
17  //                                   {\RequirePackage{filehook-2019}}
18  // and ALL the hook internals (`\filehook@every@atbegin`,
19  // `\filehook@prefixwarg`, `\filehook@appendwarg`, `\filehook@every@atend`
20  // — used by raw currfile.sty:68-73 — plus the `\AtBeginOfPackageFile`
21  // family) live in the versioned sub-file. Without this registration
22  // the version-suffix fallback (Perl Package.pm:2118-2121, mirrored in
23  // `content.rs` Step 3) strips `filehook-2020` → `filehook`, re-enters
24  // this binding (`Warning:recursion`) and the sub-file is never loaded,
25  // leaving every hook undefined. Perl shares that misfire
26  // (`Info:fallback:filehook-2020`). Registering the sub-files as
27  // INTERPRETABLE_SOURCES (Perl FindFile_aux L2107) makes the raw load
28  // find them directly — the same shape xparse.sty.ltxml uses for its
29  // `xparse-2018-04-12.sty` rollback file. Witnesses: TL doc corpus
30  // currfile/currfile (`\ifcurrfilename` swallowed the document),
31  // pythontex/pythontex (loads currfile); arXiv 2405.18977, 2406.01136,
32  // 2406.01832 (pbalance, formerly served by no-op hook stubs here).
33  AssignMapping!("INTERPRETABLE_SOURCES", "filehook-2020.sty" => 1);
34  AssignMapping!("INTERPRETABLE_SOURCES", "filehook-2019.sty" => 1);
35
36  InputDefinitions!("filehook", noltxml => true, extension => Some(Cow::Borrowed("sty")));
37});
