22/// Perl: $lst_charmapping — maps special chars to TeX control sequences.
23fn lst_char_mapping(ch: &str, upquote: bool) -> Option<&'static str> {
24  match ch {
25    "#" => Some("\\#"),
26    "$" => Some("\\textdollar"),
27    "&" => Some("\\&"),
28    "'" => Some(if upquote {
29      "\\textquotesingle"
30    } else {
31      "\\textquoteright"
32    }),
33    "*" => Some("\\textasteriskcentered"),
34    "<" => Some("\\textless"),
35    ">" => Some("\\textgreater"),
36    "\\" => Some("\\textbackslash"),
37    "^" => Some("\\textasciicircum"),
38    "_" => Some("\\textunderscore"),
39    "`" => Some(if upquote {
40      "\\textasciigrave"
41    } else {
42      "\\textquoteleft"
43    }),
44    "{" => Some("\\textbraceleft"),
45    "}" => Some("\\textbraceright"),
46    "%" => Some("\\%"),
47    "|" => Some("\\textbar"),
48    "~" => Some("\\textasciitilde"),
49    _ => None,
50  }
51}

53/// Perl: lstRescan — remap special chars via character mapping.
54fn lst_rescan(tokens: Option<Tokens>) -> Option<Tokens> {
55  let upquote = lst_get_boolean("upquote");
56  tokens.map(|toks| {
57    let remapped: Vec<Token> = toks
58      .unlist()
59      .iter()
60      .flat_map(|t| {
61        if t.get_catcode() == Catcode::OTHER {
62          let ch = t.to_string();
63          if let Some(cs) = lst_char_mapping(&ch, upquote) {
64            return vec![T_CS!(cs)];
65          }
66        }
67        vec![*t]
68      })
69      .collect();
70    Tokens::new(remapped)
71  })
72}

94/// Perl: listingsReadRawLines — read raw lines until \end{$environment}
95///
96/// OXIDIZED_DESIGN #61: the terminator is matched ANYWHERE in the line, not
97/// only at its start. Perl (`listings.sty.ltxml` L316) anchors the regex —
98/// `/^\s*\\end\{\Q$environment\E\}(.*?)$/` — so a line like
99/// `  </body></html> \end{lstlisting}` never terminates the environment and the
100/// reader runs to EOF, swallowing the rest of the document (its `\end{document}`
101/// included) with NO error at all. Witness 2605.11619: a complete 54 KB paper
102/// silently lost its Conclusion, `\bibliography` and appendix — 1.3 MB of HTML,
103/// zero `Error:`, zero references. Same-host Perl reports "No obvious problems"
104/// and loses the tail identically, so this is a shared upstream bug
105/// (KNOWN_PERL_ERRORS #51, candidate to upstream), not a Rust regression.
106///
107/// Real `listings` (ground truth: pdflatex) accepts it — it renders `hello world`
108/// as the final listing line and continues with the document. So the text BEFORE
109/// the terminator becomes the last line, and the text AFTER is unread, exactly
110/// as Perl already does for the trailing part.
111pub fn listings_read_raw_lines(environment: &str) -> String {
112  listings_read_raw_lines_with_outer(environment, None)
113}

115pub fn listings_read_raw_lines_with_outer(environment: &str, outer_env: Option<&str>) -> String {
116  let mut lines = Vec::new();
117  // Ignore the remainder of the `\begin{...}` line — real listings drops it —
118  // UNLESS the pushback holds a non-space token. An optional-argument probe
119  // (`\lstnewenvironment{x}[1][]` → `[Default:]` → `read_non_space`) crosses
120  // the newline after `\begin{x}` and unreads the body's first NON-SPACE
121  // character; `read_raw_line` then returns pushback + line remainder = the
122  // body's REAL first line, and discarding it swallowed line 1 of every such
123  // environment (Perl 0.8.8 shares the bug — same-host verified; ~148 TL doc
124  // manuals define example environments this way). Space-only pushback (the
125  // built-in lstlisting's `if_next` probe unreads the end-of-line SPACE) still
126  // means the first raw line is `\begin`-line leftover. OXIDIZED_DESIGN #162.
127  //
128  // That first line is taken from COLUMN 0 of the mouth's line, not as
129  // pushback + remainder: the probe's tokenization had dropped the
130  // indentation and every catcode-9 character, so under `\DocInput` (doc.sty
131  // `\MakePercentIgnore`) line 1 came back without its `% ` prefix while lines
132  // 2+ kept it, and `gobble=2` (lstdoc/forest-doc.sty:32) then ate two real
133  // characters of line 1 while the write-file tee re-input lines whose `%`
134  // commented out `\end{forest}` (forest-doc: 501 `readBalanced ran out of
135  // input` → Fatal). Guard:
136  // `perfect_kernel_batch56::forest_docinput_lstenv_writefile_gobbles_doc_percent`.
137  let begin = format!("\\begin{{{environment}}}");
138  // The listing's END already read as tokens: its body was tokenized ahead as a
139  // macro argument (`\parbox{3cm}{\begin{lstlisting}…\end{lstlisting}}`), so no
140  // line holds it any more and listings' line scan cannot find its end — TeX
141  // runs out ("Argument of \lst@next has an extra }", listings.sty:2142-2147
142  // dropping the begin line's text). The body is lost, so this is an Error.
143  let end = format!("\\end{{{environment}}}");
144  if pushback_text().contains(&end) {
145    emit_error(
146      "expected",
147      &end,
148      &latexml_core::generate_message!(format!(
149        "{end} was read ahead with the listing's body as a macro argument; listings reads its body \
150         line by line, so the body is lost"
151      )),
152    );
153  }
154  let mut first_line = if pushback_holds_nonspace() {
155    // A begin-line argument the environment's mapping could not consume
156    // leaves non-space pushback while the mouth is STILL on the `\begin`
157    // line; that line is not content — re-inputting it re-entered the
158    // environment without bound (tutodoc, simplebnf, istgame: MemoryBudget
159    // fatal, sweep #41). Guard:
160    // `perfect_kernel_batch56::tcb_listing_unmapped_begin_line_args_are_absorbed`.
161    // (Its arguments still reach a tcolorbox listing box: `{ o s }` + `\begin{env}*`.)
162    let line = read_raw_current_line_from_start();
163    let rest = line
164      .as_deref()
165      .and_then(|l| l.trim_start().strip_prefix(begin.as_str()));
166    tcb_begin_line_arguments(environment, rest);
167    line.filter(|l| !l.trim_start().starts_with(&begin))
168  } else {
169    // leftover of the \begin line — not content, EXCEPT a tcolorbox listing
170    // box's arguments written after `\begin{env}`.
171    let leftover = read_raw_line();
172    // A tcolorbox listing box reads its arguments from that line; anything else
173    // there is dropped, with listings' own warning (listings.sty:2142-2145
174    // `\lstenv@DroppedWarning`).
175    let tcb_line = matches!(
176      lookup_value("tcb_pending_mode_opts"),
177      Some(Stored::Tokens(_))
178    );
179    tcb_begin_line_arguments(environment, leftover.as_deref());
180    // An optional argument after a space (`\begin{lstlisting} [frame=single]`) is
181    // still the options to listings (`\@ifnextchar[` skips the space), so no text
182    // is dropped there; this reader leaves them in the line, unread (1606.07935).
183    let options_after_space = leftover
184      .as_deref()
185      .is_some_and(|l| l.starts_with(char::is_whitespace) && l.trim_start().starts_with('['));
186    // A `%` comment is not dropped text (TeX discards it reading the line): only
187    // what precedes it counts (`\begin{lstlisting}[language=C] % note`, review probe
188    // lst1). The line after a begin-line comment is KEPT as the listing's first line,
189    // where pdflatex drops it with this warning (content first; OXIDIZED_DESIGN #521).
190    if !tcb_line
191      && !options_after_space
192      && leftover
193        .as_deref()
194        .is_some_and(|l| !strip_tex_comment(l).trim().is_empty())
195    {
196      Warn!(
197        "latex",
198        "(Listings)",
199        "Package Listings Warning: Text dropped after begin of listing"
200      );
201    }
202    None
203  };
204  let mut end_patterns = vec![
205    format!(r"\\end\{{{}\}}", regex::escape(environment)),
206    format!(r"\\end{}", regex::escape(environment)),
207  ];
208  if let Some(outer) = outer_env
209    && !outer.is_empty()
210    && outer != environment
211  {
212    end_patterns.push(format!(r"\\end\{{{}\}}", regex::escape(outer)));
213  }
214  let end_re = Regex::new(&end_patterns.join("|")).unwrap();
215  while let Some(line) = first_line.take().or_else(read_raw_line) {
216    if let Some(m) = end_re.find(&line) {
217      let matched_str = m.as_str();
218      let is_outer = outer_env.is_some_and(|outer| matched_str == format!(r"\end{{{outer}}}"));
219      // Whitespace-only text before the terminator is the ordinary
220      // `\end{lstlisting}`-on-its-own-line case: contributes no listing line.
221      let before = &line[..m.start()];
222      if !before.trim().is_empty() {
223        lines.push(before.to_string());
224      }
225      if is_outer {
226        let rest = line[m.start()..].to_string();
227        if rest.trim().is_empty() {
228          unread_one(T_CR!());
229        } else if !restore_raw_line_rest(&rest)
230          && let Ok(mouth) = Mouth::new(&rest, None)
231        {
232          open_mouth(mouth, true);
233        }
234      } else {
235        let rest = line[m.end()..].to_string();
236        if rest.trim().is_empty() {
237          unread_one(T_CR!());
238        } else if !restore_raw_line_rest(&rest)
239          && let Ok(mouth) = Mouth::new(&rest, None)
240        {
241          open_mouth(mouth, true);
242        }
243      }
244      break;
245    }
246    lines.push(line);
247  }
248  lines.join("\n")
249}

251/// Read a balanced group from a token slice, starting at tokens[*pos].
252/// *pos should point to a T_BEGIN token. Returns content tokens (without outer braces).
253/// Advances *pos past the closing T_END.
254fn read_balanced_group(tokens: &[Token], pos: &mut usize) -> Vec<Token> {
255  let mut result = Vec::new();
256  if *pos >= tokens.len() || tokens[*pos].get_catcode() != Catcode::BEGIN {
257    return result;
258  }
259  *pos += 1; // skip T_BEGIN
260  let mut level = 1i32;
261  while *pos < tokens.len() && level > 0 {
262    match tokens[*pos].get_catcode() {
263      Catcode::BEGIN => {
264        level += 1;
265        result.push(tokens[*pos]);
266      },
267      Catcode::END => {
268        level -= 1;
269        if level > 0 {
270          result.push(tokens[*pos]);
271        }
272      },
273      _ => {
274        result.push(tokens[*pos]);
275      },
276    }
277    *pos += 1;
278  }
279  result
280}

282/// Perl: TokenizeBalanced — the TeX an escape holds (`escapechar`, `escapeinside`, `mathescape`, `texcl`), its groups
283/// balanced. listings executes it as TeX where it stands (lstmisc.sty `\lst@DefEsc`), reading it then, so a command
284/// in it may change the catcodes of what follows (`%B(col\verb._.idx(j))%`, 1106.5908: tokenized at once, the `_` was a
285/// subscript, "Script _ can only appear in math mode"). So the text is kept and read by a mouth of its own when the
286/// listing is digested (`\lx@lst@escape`), the balancing braces Perl adds put around it. Guard
287/// `perfect_kernel_batch73b::listings_escape_is_read_where_it_runs`.
288fn tokenize_balanced(text: &str) -> Vec<Token> {
289  let mut level: i32 = 0;
290  let mut opened: i32 = 0;
291  for t in mouth::tokenize(TeXString::assembled(text.to_string())).unlist() {
292    match t.get_catcode() {
293      Catcode::BEGIN => level += 1,
294      Catcode::END => level -= 1,
295      _ => {},
296    }
297    opened = opened.min(level);
298  }
299  let open = usize::try_from(-opened).unwrap_or(0);
300  let close = usize::try_from(level - opened).unwrap_or(0);
301  let balanced = s!("{}{text}{}", "{".repeat(open), "}".repeat(close));
302  // The conversion's count (a State value, as the escapes it names).
303  let id = lookup_int("lx@lst@escapes") + 1;
304  assign_value("lx@lst@escapes", Stored::Int(id), Some(Scope::Global));
305  assign_value(&s!("lx@lst@escape@{id}"), balanced, Some(Scope::Global));
306  let mut toks = vec![T_CS!("\\lx@lst@escape"), T_BEGIN!()];
307  toks.extend(Explode!(id));
308  toks.push(T_END!());
309  toks
310}

312/// Perl: listingsReadRawString — read until closing delimiter token.
313/// Handles mathescape: within $...$, content is read with normal catcodes
314/// and preserved as TeX (backslashes intact). Outside math, CS tokens have \ stripped.
315/// Returns UnTeX'd string representation.
316///
317/// Also returns whether the gullet counted the closing delimiter into the alignment ledger: a pre-tokenized `}` (an
318/// argument re-read, `\text{\lstinline{x}}`) keeps its END catcode and `read_token` counted it, a `}` read under the
319/// verbatim catcodes is OTHER and was not (see [`settle_verbatim_brace_ledger`]).
320pub fn listings_read_raw_string(
321  until: Option<&Token>,
322  saved_catcodes: &[(char, Catcode)],
323) -> (String, bool) {
324  let mathescape = lst_get_boolean("mathescape");
325  let mut inmath = false;
326  let mut tokens: Vec<Token> = Vec::new();
327
328  // A `{`-delimited inline listing (`\lstinline{…}`) ends at the BALANCED
329  // `}`: listings.sty:1968 `\lst@InlineG` tracks nesting, so
330  // `\lstinline{\renewcommand{\fnindent}{1.25em}}` (coolfn, tikz-shields
331  // `\mintinline{tex}{\usepackage[…]{…}}`) keeps its inner groups. Perl's
332  // reader stops at the first `}` (listings.sty.ltxml:281 "does NOT balance
333  // groups") and the trailing real `}` leaked to the stomach. A character
334  // delimiter (`|…|`) still matches its first occurrence.
335  let brace_delimited = until.is_some_and(|t| t.text == pin!("}"));
336  let mut depth: usize = 0;
337  let mut close_counted = false;
338  while let Ok(Some(token)) = read_token() {
339    if let Some(until_tok) = until {
340      // Perl `listings.sty.ltxml:291` matches by string only —
341      // `last if $until and $token->getString eq $until->getString;`.
342      // The verbatim catcode switch makes the `until` delimiter's
343      // catcode (e.g. END for `}`) differ from the body's reading
344      // catcode (OTHER), so a strict (text, code) match would never
345      // trigger and the body greedily consumes input. Match on
346      // interned text identity alone, mirroring Perl.
347      if token.text == until_tok.text {
348        if brace_delimited && depth > 0 && !inmath {
349          depth -= 1;
350        } else {
351          close_counted = token.get_catcode() == Catcode::END;
352          break;
353        }
354      } else if brace_delimited && !inmath && token.text == pin!("{") {
355        depth += 1;
356      }
357    }
358    // Check for mathescape $ toggle. Perl `listings.sty.ltxml:292-293`:
359    // entering math restores the saved (normal) catcode table, exiting
360    // returns to the verbatim (empty) one. This is what makes `{`/`}` balance
361    // and `\{`/`\}` read as CS tokens *only within* the mathescape'd `$…$`.
362    if mathescape && token.text == pin!("$") {
363      inmath = !inmath;
364      for &(c, cc) in saved_catcodes {
365        let target = if inmath { cc } else { Catcode::OTHER };
366        assign_catcode(c, target, Some(Scope::Local));
367      }
368      tokens.push(T_OTHER!("$"));
369      continue;
370    }
371    let cc = token.get_catcode();
372    if inmath && cc == Catcode::BEGIN {
373      // In math mode with {, read balanced group and preserve
374      tokens.push(T_BEGIN!());
375      if let Ok(balanced) = read_balanced(ExpansionLevel::Off, false, false) {
376        tokens.extend(balanced.unlist());
377      }
378      tokens.push(T_END!());
379    } else if !inmath && cc.is_active_or_cs() {
380      // Outside math: convert CS tokens to plain text, preserving the backslash
381      // In verbatim listing context, \end should appear as literal characters \ e n d
382      let name = token.to_string();
383      for c in name.chars() {
384        tokens.push(CharToken!(c));
385      }
386    } else {
387      tokens.push(token);
388    }
389  }
390  // Remove trailing spaces
391  while tokens
392    .last()
393    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
394  {
395    tokens.pop();
396  }
397  // UnTeX: convert tokens to string representation
398  let mut result = String::new();
399  for t in &tokens {
400    let cc = t.get_catcode();
401    if cc.is_active_or_cs() {
402      // Preserve CS with backslash
403      let s = t.to_string();
404      result.push_str(&s);
405      result.push(' '); // CS names need trailing space for re-tokenization
406    } else {
407      result.push_str(&t.to_string());
408    }
409  }
410  result.truncate(result.trim_end().len());
411  (result, close_counted)
412}

426/// Perl: listingsReadRawFile — read entire file contents as string.
427///
428/// A file that is nowhere, or that cannot be opened, is Perl's
429/// `Error('I/O', $filename, …, "Can't read listings file '$filename'", $!)`
430/// (listings.sty.ltxml:322-334); real listings stops with "Package Listings
431/// Error: File `x(.ext)' not found" (listings.sty:2074-2086; its `\read\m@ne`
432/// at :2090 is an Emergency stop in nonstop mode), and tcolorbox's
433/// `\tcbinputlisting` reaches it through `\lstinputlisting`
434/// (tcblistings.code.tex:42-53). It was a warning (crossreftools-doc:
435/// `labelstuff.tex`, `addcontentslineforlistoflabels.tex`, which the package
436/// does not ship). Guard: `package_leads_56::missing_listing_file_is_an_io_error`.
437pub fn listings_read_raw_file(file: &str) -> Result<Option<String>> {
438  // TeX's file name skips the blanks before it and ends at the first one after it (tex.web §526), so
439  // `\lstinputlisting{⏎ examples/simple-correct.c⏎ }` reads `examples/simple-correct.c` (2511.21509; pdflatex
440  // finds it). Perl keeps the end-of-line spaces in the name and cannot find the file.
441  let filename = file.trim().to_string();
442  // A file that exists only in the session's virtual file store — written by
443  // `\write` emulations such as tcolorbox's `\tcbverbatimwrite` (the
444  // `tcblisting` body goes to `\jobname.listing`, tcblistingscore.code.tex
445  // :275-282, and `\lstinputlisting` reads it back, tcblistings.code.tex
446  // :42-53), `filecontents`, or `\VerbatimOut` — is read from there, as
447  // `\input`/`\openin` already do (content.rs). `find_file` returned the bare
448  // key for such a file and the disk read came back empty, so every
449  // `listing only` box (sim-os-menus' terminal windows; ~33 low-recall
450  // manuals round-trip a listing file) rendered an EMPTY listing at 0
451  // errors — Perl, with no virtual store, reports "Can't read listings file".
452  if let Some(text) = vfs_read(&filename) {
453    return Ok(Some(if text.contains('\r') {
454      text.replace("\r\n", "\n").replace('\r', "\n")
455    } else {
456      text
457    }));
458  }
459  // Perl #2818 (41bd31e8): FindFile(..., noltxml => 1) — when reading a raw file
460  // for a listing (\lstinputlisting), never substitute an .ltxml binding for the
461  // real source text (Rust spells `noltxml` as `forbid_ltxml`).
462  if let Some(path) = find_file(
463    &filename,
464    Some(FindFileOptions {
465      forbid_ltxml: true,
466      ..Default::default()
467    }),
468  ) {
469    // `find_file` resolves an extensionless name to the virtual store's key
470    // (`snip` → `snip.tex`, content.rs `find_file_aux`), which only the store
471    // can read: tutodoc's `\tdoclatexinput{examples-…}` listings (a
472    // `filecontents` file, read by `\inputminted`) came out empty.
473    if let Some(text) = vfs_read(&path) {
474      return Ok(Some(if text.contains('\r') {
475        text.replace("\r\n", "\n").replace('\r', "\n")
476      } else {
477        text
478      }));
479    }
480    // Normalize line terminators to bare LF.
481    //
482    // DIVERGENCE (OXIDIZED_DESIGN #69) — Perl slurps the file verbatim, so a
483    // CRLF (or classic-Mac CR) source keeps its `\r`, and every downstream
484    // end-of-line test in the listings processor is written against `\n`: the
485    // `__NEWLINE__` close test for line comments, the blank-line test in
486    // `lst_process_start_line`, the line-skipping loops. A `\r` sitting before
487    // the `\n` defeats all of them, so a line comment never terminates and its
488    // style bleeds over every following line.
489    //
490    // TeX itself never sees the CR — its file reader strips the line
491    // terminator and appends `\endlinechar` — so normalizing here is what
492    // matches the engine we emulate, not a special case.
493    //
494    // Ground truth, arXiv 2412.04705 (arXiv/html_feedback#6735), whose Python
495    // sources are CRLF: pdflatex renders only the `#` line in comment green
496    // (measured: 9 green vs 69 black glyph groups), while BOTH LaTeXML engines
497    // paint the whole snippet green — Perl identically, so this is a shared

541/// Perl: lstUnGroup — strip outer {} if there's only a single group.
542fn lst_un_group(tokens: Option<Tokens>) -> Option<Tokens> {
543  tokens.map(|toks| {
544    let mut t = toks.unlist();
545    if t.len() >= 2
546      && t
547        .first()
548        .is_some_and(|tok| tok.get_catcode() == Catcode::BEGIN)
549      && t
550        .last()
551        .is_some_and(|tok| tok.get_catcode() == Catcode::END)
552    {
553      let mut groups = 0;
554      let mut level = 0;
555      for tok in t.iter() {
556        let cc = tok.get_catcode();
557        if cc == Catcode::END {
558          level -= 1;
559        } else if cc == Catcode::BEGIN {
560          if level == 0 {
561            groups += 1;
562          }
563          level += 1;
564        }
565      }
566      if groups == 1 {
567        t.pop();
568        t.remove(0);
569      }
570    }
571    Tokens::new(t)
572  })
573}

575/// Perl: lstSplit — split comma-separated, strip whitespace and TeX comments.
576fn lst_split(stuff: &Option<Tokens>) -> Vec<String> {
577  match stuff {
578    None => vec![],

581        .map(|t| t.to_string())
582        .unwrap_or_default();

585        std::sync::LazyLock::new(|| Regex::new(r"%.*?\n\s*").unwrap());

594    },

611/// Perl: lst_splitDelimiters — split a delimiter token list into (open, close) parts.
612/// The format is `{open}{close}` where the two TeX groups contain the delimiters.
613/// First group = open delimiter, second group = close delimiter.
614fn lst_split_delimiters(delims: &Tokens) -> (String, String) {
615  // Perl listings.sty.ltxml:629 lst_splitDelimiters
616  //   my @t  = grep { !Equals($_, T_BEGIN) } $delims->unlist;
617  //   my @t1 = ();
618  //   if (scalar(@t) == 2) { @t1 = ($t[0]); @t = ($t[1]); }
619  //   else {
620  //     while (@t && !Equals($t[0], T_END)) { push(@t1, shift(@t)); }
621  //     @t = grep { !Equals($_, T_END) } @t; }
622  //   return (Tokens(@t1), Tokens(@t));
623  // Two-token short-circuit is what handles bare `<>` (open='<', close='>'),
624  // which is how the XML language registers its `tag=**[s]<>` delimiter.
625  let toks: Vec<Token> = delims
626    .unlist_ref()
627    .iter()
628    .filter(|t| **t != T_BEGIN!())
629    .copied()
630    .collect();
631  let (open_toks, close_toks): (Vec<Token>, Vec<Token>) = if toks.len() == 2 {
632    (vec![toks[0]], vec![toks[1]])
633  } else {
634    let mut it = toks.into_iter();
635    let mut open = Vec::new();
636    let mut close = Vec::new();
637    let mut saw_end = false;
638    for t in it.by_ref() {
639      if t == T_END!() {
640        saw_end = true;
641        break;
642      }
643      open.push(t);
644    }
645    if saw_end {
646      for t in it {
647        if t == T_END!() {
648          continue;
649        }
650        close.push(t);
651      }
652    }
653    (open, close)
654  };
655  let open_str = lst_deslash(&Tokens::new(open_toks).to_string());
656  let close_str = lst_deslash(&Tokens::new(close_toks).to_string());
657  (open_str, close_str)
658}

724    Some(Stored::Dimension(d)) => d.value_of(),

732  // class group (`\@listingGroup{cls}{…}`) stays open across the lines it spans.

754      _ => boxes.push(item),

756  }

758  let whatsit_cs = |item: &Digested| match item.data() {
759    DigestedData::Whatsit(w) => Some(w.borrow().get_definition().get_cs_name().into_owned()),
760    _ => None,
761  };
762  let metric = |item: &Digested, key: &str| match item.data() {
763    DigestedData::Whatsit(w) => match w.borrow().get_property(key).as_deref() {
764      Some(Stored::Dimension(d)) => d.value_of(),
765      _ => 0,
766    },
767    _ => 0,
768  };
769  let font = whatsit.get_font()?.or_else(lookup_font).unwrap_or_default();
770  let (mut height, mut width) = (0, 0);

776          metric(&item, "lst_baselineskip"),

779        ));
780      },
781      Some("\\@lst@endline") => {

784        };

786        if breaklines && linewidth > 0 {
787          let mut options = SymHashMap::default();

789          let (w, ..) = font.compute_boxes_size(&content, options)?;
790          lines = lines.max((w.value_of() + linewidth - 1) / linewidth);

793        width = width.max(linewidth);
794      },
795      _ => {
796        if let Some((_, _, content)) = line.as_mut() {
797          content.push(item);
798        }
799      },
800    }
801  }
802  let skips = dimension("lst_aboveskip") + dimension("lst_belowskip");
803  Ok((
804    Dimension::new(width),
805    Dimension::new((skips + height).max(0)),
806    Dimension::new(0),
807  ))
808}

1540/// A marker item (`rangeprefix`, non-numeric) selects lines by their text, which is not modelled: the listing shows
1541/// whole, as before. Not modelled either: listings walks the intervals in order and only skips forward
1542/// (listings.sty:1564-1572), so `{5-6,2-3}` shows 5-7 where this shows the union; the `[interrange]` text between
1543/// intervals is not printed; and listings' PreSet hook (:1403-1408) clears `linerange` for each listing, so a global
1544/// `\lstset{linerange=...}` does not carry over as it does here. The shown lines are numbered by their source
1545/// line, not consecutively (RED singletons/listings_linerange_numbers_are_consecutive).
1546fn lst_line_intervals() -> Vec<(i64, i64)> {
1547  const LAST: i64 = 9_999_999;
1548  let range = lst_get_literal("linerange");
1549  // `\lstKV@OptArg[]{#1}`: an optional `[interrange]` comes first.
1550  let range = match range.trim().strip_prefix('[') {
1551    Some(rest) => rest.split_once(']').map_or("", |(_, items)| items),
1552    None => range.trim(),
1553  };
1554  let number = |text: &str| text.trim().parse::<i64>().ok();
1555  let mut lines = Vec::new();
1556  for item in range
1557    .split(',')
1558    .map(str::trim)
1559    .filter(|item| !item.is_empty())
1560  {
1561    let interval = match item.split_once('-') {
1562      None => number(item).map(|n| (n, n)),
1563      Some((first, last)) => {
1564        let first = if first.trim().is_empty() {
1565          Some(1)
1566        } else {
1567          number(first)
1568        };
1569        let last = if last.trim().is_empty() {
1570          Some(LAST)
1571        } else {
1572          number(last)
1573        };
1574        first.zip(last)
1575      },
1576    };
1577    match interval {
1578      Some(interval) => lines.push(interval),
1579      None => return vec![(1, LAST)],
1580    }
1581  }
1582  if lines.is_empty() {
1583    lines.push((lst_get_number("firstline"), lst_get_number("lastline")));
1584  }
1585  lines
1586}

1588/// Perl: lstProcess — main entry point for processing listing text.
1589fn lst_process(mode: &str, text: &str) -> Tokens {
1590  if text.is_empty() || !lst_get_boolean("print") {
1591    return Tokens!();
1592  }
1593
1594  let mut text = text.to_string();
1595  // Strip trailing whitespace if !showlines
1596  if !lst_get_boolean("showlines") {
1597    text = text.trim_end().to_string();
1598  }
1599
1600  // Establish line numbering parameters
1601  let firstnumber = lst_get_literal("firstnumber");
1602  let line0: i64 = match firstnumber.as_str() {
1603    "last" => lookup_value("LISTINGS_LAST_NUMBER")
1604      .map(|v| v.to_string().parse().unwrap_or(1))
1605      .unwrap_or(1),
1606    "auto" => {
1607      let name = lst_get_literal("name");
1608      if !name.is_empty() {
1609        let key = s!("LISTINGS_LAST_NUMBER_{name}");
1610        lookup_value(&key)
1611          .map(|v| v.to_string().parse().unwrap_or(1))
1612          .unwrap_or(1)
1613      } else {
1614        1
1615      }
1616    },
1617    _ => firstnumber.parse().unwrap_or(1),
1618  };
1619
1620  let stepnumber = lst_get_number("stepnumber");
1621  let _numpos = if stepnumber == 0 {
1622    "none".to_string()
1623  } else {
1624    lst_get_literal("numbers")
1625  };
1626
1627  // Build ID regex dynamically from character table in state
1628  // Perl: join('', sort keys %{$$characters{letter}})
1629  let letter_chars = build_char_class("letter");
1630  let digit_chars = build_char_class("digit");
1631  // Perl: LookupValue('LST@TEXCS') — set when texcs or moretexcs has been used
1632  let has_texcs = matches!(lookup_value("LST@TEXCS"), Some(Stored::Bool(true)));
1633  let id_re = if letter_chars.is_empty() {
1634    None
1635  } else {
1636    let id_pattern = if has_texcs {
1637      format!("\\\\?[{letter_chars}][{letter_chars}{digit_chars}]*")
1638    } else {
1639      format!("[{letter_chars}][{letter_chars}{digit_chars}]*")
1640    };
1641    Regex::new(&id_pattern).ok()
1642  };
1643
1644  // Build delimiter regexes from LST_DELIM_KEYS (Perl: $LaTeXML::DELIM_RE, $LaTeXML::ESCAPE_RE)
1645  let delim_keys: Vec<String> = match lookup_value("LST_DELIM_KEYS") {
1646    Some(Stored::Tokens(t)) => t.unlist_ref().iter().map(|tok| tok.to_string()).collect(),
1647    _ => vec![],
1648  };
1649  // Also check mathescape delimiter ($)
1650  let mut all_delim_keys = delim_keys;
1651  if let Some(Stored::String(_)) = lookup_value("LST_DELIM@$@open")
1652    && !all_delim_keys.iter().any(|k| k == "$")
1653  {
1654    all_delim_keys.push("$".to_string());
1655  }
1656  // Perl listings.sty.ltxml:1288
1657  //   local $LaTeXML::DELIM_RE = join('|', map { $$delimiters{$_}{open} } sort keys %$delimiters);
1658  // The lexical sort matters: leftmost-first regex alternation means short
1659  // prefixes that sort before longer ones win, e.g. `<` (XML tag) is matched
1660  // before `<!--` (XML comment) so a stray `<!--` is classified as a tag.
1661  all_delim_keys.sort();
1662
1663  let mut delim_opens: Vec<String> = Vec::new();
1664  let mut escape_opens: Vec<String> = Vec::new();
1665  for key in &all_delim_keys {
1666    let open_key = s!("LST_DELIM@{key}@open");
1667    if let Some(Stored::String(open_sym)) = lookup_value(&open_key) {
1668      let open_re = with(open_sym, |s| s.to_string());
1669      let escape_key = s!("LST_DELIM@{key}@escape");
1670      if matches!(lookup_value(&escape_key), Some(Stored::Bool(true))) {
1671        escape_opens.push(open_re);
1672      } else {
1673        delim_opens.push(open_re);
1674      }
1675    }
1676  }
1677  // An escape is tried before the other delimiters: listings chains each delimiter onto the meaning its first character
1678  // had (lstmisc.sty `\lst@DefDelimB`), and the escape (`\lst@DefEsc`, installed last) is tested first, so
1679  // `escapeinside={\#*}{*}` under Python's `#` comments escapes `#*\label{line:degree}*` (2104.08012; pdflatex resolves
1680  // `line \ref{line:degree}`), where the lexical order took `#` for a comment and printed the `\label` (KPE #1167).
1681  let delim_opens: Vec<String> = escape_opens.iter().cloned().chain(delim_opens).collect();
1682  let delim_re = if delim_opens.is_empty() {
1683    None
1684  } else {
1685    Regex::new(&format!("^({})", delim_opens.join("|"))).ok()
1686  };
1687  let escape_re = if escape_opens.is_empty() {
1688    None
1689  } else {
1690    Regex::new(&format!("^({})", escape_opens.join("|"))).ok()
1691  };
1692
1693  let space_token = if lst_get_boolean("showspaces") {
1694    T_CS!("\\@lst@visible@space")
1695  } else {
1696    T_CS!(" ")
1697  };
1698
1699  let case_sensitive = lst_get_boolean("sensitive");
1700  // Perl: if (!$CASE_SENSITIVE) { foreach word (keys %$words) { $$words{uc($word)} =
1701  // $$words{$word}; } }
1702  if !case_sensitive && let Some(Stored::Strings(word_list)) = lookup_value("LST_WORD_LIST") {
1703    for word_sym in word_list.iter() {
1704      let word = to_string(*word_sym);
1705      let upper = word.to_uppercase();
1706      if upper != word {
1707        let src_class_key = s!("LST_WORDS@{word}@class");
1708        let dst_class_key = s!("LST_WORDS@{upper}@class");
1709        if let Some(class_val) = lookup_value(&src_class_key) {
1710          assign_value(&dst_class_key, class_val, None);
1711        }
1712        let src_index_key = s!("LST_WORDS@{word}@index");
1713        let dst_index_key = s!("LST_WORDS@{upper}@index");
1714        if let Some(index_val) = lookup_value(&src_index_key) {
1715          assign_value(&dst_index_key, index_val, None);
1716        }
1717      }
1718    }
1719  }
1720
1721  let mut ctx = LstContext {
1722    listing: text,
1723    linenum: line0,
1724    colnum: 0,
1725    mode: mode.to_string(),
1726    linestart: None,
1727    emptyfrom: None,
1728    lsttokens: vec![T_BEGIN!()],
1729    id_re,
1730    delim_re,
1731    escape_re,
1732    quoted_re: Regex::new(r"^\\\\").unwrap(),
1733    space_token,
1734    case_sensitive,
1735    literate: build_literate_entries(),
1736    literate_re: build_literate_re(false),
1737    lines: lst_line_intervals(),
1738  };
1739
1740  // Add preamble tokens
1741  if let Some(Stored::Tokens(preamble)) = lookup_value("LISTINGS_PREAMBLE") {
1742    ctx.lsttokens.extend(preamble.unlist());
1743  }
1744  let basicstyle = lst_get_tokens("basicstyle");
1745  if !basicstyle.is_empty() {
1746    ctx.lsttokens.extend(basicstyle.unlist());
1747  }
1748
1749  // Perl: while ($listing && !&$linetest($linenum)) { skip lines before firstline }
1750  while !ctx.listing.is_empty() && !lst_linetest(&ctx) {
1751    if let Some(pos) = ctx.listing.find('\n') {
1752      ctx.listing = ctx.listing[pos + 1..].to_string();
1753    } else {
1754      ctx.listing = String::new();
1755    }
1756    ctx.linenum += 1;
1757  }
1758
1759  if mode != "inline" {
1760    // Register-level like real listings (`\global\c@lstnumber\lst@firstnumber`,
1761    // listings.sty L1516) — NOT user-level `\setcounter`, which counter
1762    // packages wrap (xassoccnt.sty L2553 → its expl3 body runs away into the
1763    // following `\@lst@startline` at construction time; 518 malformed errors
1764    // on xassoccnt_doc). Sibling of the `\lx@lst@stepnumber` newline emitter.
1765    ctx
1766      .lsttokens
1767      .extend(invoke(T_CS!("\\lx@lst@setnumber"), vec![Tokens::new(
1768        ExplodeText!(&ctx.linenum.to_string()),
1769      )]));
1770    lst_process_start_line(&mut ctx);
1771  }
1772
1773  lst_process_internal(&mut ctx, None, None);
1774
1775  if mode != "inline" {
1776    lst_process_end_line(&mut ctx);
1777  }
1778
1779  // Save line number for later use
1780  let name = lst_get_literal("name");
1781  assign_value(
1782    "LISTINGS_LAST_NUMBER",
1783    Stored::Int(ctx.linenum),
1784    Some(Scope::Global),
1785  );
1786  if !name.is_empty() {
1787    let key = s!("LISTINGS_LAST_NUMBER_{name}");
1788    assign_value(&key, Stored::Int(ctx.linenum), Some(Scope::Global));
1789  }
1790
1791  // Remove trailing empty lines.
1792  // Perl listings.sty.ltxml:1330 does the bare truncation:
1793  //   @lsttokens = @lsttokens[0 .. $emptyfrom - 1] if $emptyfrom;
1794  //
1795  // DIVERGENCE (OXIDIZED_DESIGN #68) — that truncation is UNBALANCED, in Perl
1796  // too. `$emptyfrom` is the index of the first trailing empty line's
1797  // `\@lst@startline`, but a delimited class left open by the last *content*
1798  // line (a string/comment the listing was cut off inside — exactly what
1799  // `lastline=N` on a longer file does) has its closing `}` tokens sitting in
1800  // that discarded tail. Dropping them emits a listing body with unclosed
1801  // groups; `\@@listings@block` then reads its arguments off the end of the
1802  // DOCUMENT. Perl reports that as "Missing argument {} for \@@listings@block",
1803  // we reported it as "readBalanced ran out of input" plus a cascade of
1804  // mode/sectioning errors — both engines lose the snippet.
1805  // Witness: arXiv 2412.04705 (arXiv/html_feedback#6735), whose
1806  // `\lstinputlisting[firstline=32,lastline=35]` over CRLF Python sources hit
1807  // this 7×. Measured discarded tail there:
1808  //   ["\@lst@startline", "{", "}", "}", "}", "}", "\@lst@endline"]
1809  // — three of those `}` close groups opened BEFORE the cut.
1810  //
1811  // So: truncate as Perl does, then re-close whatever the cut left open. The
1812  // discarded region is by construction only empty-line markup (it starts at a
1813  // line with colnum == 0), so nothing visible is lost by closing here.
1814  // `from > 0` mirrors Perl's `if $LaTeXML::emptyfrom` truthiness guard, which
1815  // skips index 0. It cannot be reached today (`lsttokens` is seeded with the
1816  // opening T_BEGIN, so the `linestart` that feeds `emptyfrom` is always >= 1),
1817  // but truncating to 0 would drop that T_BEGIN and leave the lone T_END pushed
1818  // below — the very imbalance this block exists to prevent.
1819  if let Some(from) = ctx.emptyfrom.filter(|from| *from > 0) {
1820    ctx.lsttokens.truncate(from);
1821    let mut depth: i64 = 0;
1822    for token in &ctx.lsttokens {
1823      match token.get_catcode() {
1824        Catcode::BEGIN => depth += 1,
1825        Catcode::END => depth -= 1,
1826        _ => {},
1827      }
1828    }
1829    // `depth` still counts the opening T_BEGIN this function seeded
1830    // `lsttokens` with; the T_END pushed below is its partner. Any depth
1831    // beyond that is a group whose closer was truncated away.
1832    for _ in 1..depth {
1833      ctx.lsttokens.push(T_END!());
1834    }
1835  }
1836
1837  ctx.lsttokens.push(T_END!());
1838  Tokens::new(ctx.lsttokens)
1839}

1842  let stepnumber = lst_get_number("stepnumber");

1844    "none".to_string()
1845  } else {

1853    let number_tokens = lst_do_number(ctx, is_empty);
1854    ctx.lsttokens.extend(number_tokens.unlist());
1855  }
1856  ctx.lsttokens.push(T_END!());
1857}

1863    }

1888/// Perl: lstProcess_internal — the recursive descent parser.
1889/// Order matches Perl: end_re, literate, delimiters, identifiers, newline, formfeed, whitespace,
1890/// quoted, default
1891fn lst_process_internal(ctx: &mut LstContext, end_re: Option<&Regex>, outer_class: Option<&str>) {
1892  // Perl listings.sty.ltxml:1411
1893  //   my $classname = ($outerclass ? undef : $$words{$lookup}{class} || 'identifiers');
1894  // When recursing inside a delimited class (string, tag, comment, …) inner
1895  // identifiers/keywords are emitted as plain tokens — only nested delimiters
1896  // (e.g. strings inside tags) re-wrap.
1897  let _ = outer_class;
1898  let mut prev_listing = String::new();
1899  // Precompile static regexes
1900  static NEWLINE_RE: std::sync::LazyLock<Regex> =
1901    std::sync::LazyLock::new(|| Regex::new(r"^\s*?\n").unwrap());
1902  let newline_re = &*NEWLINE_RE;
1903  static SPACE_RE: std::sync::LazyLock<Regex> =
1904    std::sync::LazyLock::new(|| Regex::new(r"^[\t ]+").unwrap());
1905  let space_re = &*SPACE_RE;
1906
1907  while !ctx.listing.is_empty() {
1908    // Loop guard — must make progress on every step
1909    if ctx.listing == prev_listing {
1910      emit_warn(
1911        "internal",
1912        "listings",
1913        &format!(
1914          "lstProcess_internal failed to make progress. Content: '{}'",
1915          &ctx.listing[..ctx.listing.len().min(60)]
1916        ),
1917      );
1918      ctx.listing.clear();
1919      break;
1920    }
1921    prev_listing = ctx.listing.clone();
1922
1923    // 1. Check end regex (close delimiter)
1924    // But first: if the end_re would match, check if quoted_re also matches at the same position.
1925    // If quoted_re matches, the end_re match is spurious (e.g. '' in doubled delimiters).
1926    // This replaces Perl's lookahead (?!...) in close_re which Rust regex doesn't support.
1927    if let Some(re) = end_re {
1928      let re_str = re.as_str();
1929      if re_str.contains("__NONWORD__") {
1930        let close = ctx.listing.is_empty()
1931          || ctx
1932            .listing
1933            .chars()
1934            .next()
1935            .is_none_or(|c| !c.is_alphanumeric() && c != '_');
1936        if close {
1937          break;
1938        }
1939      } else if re_str.contains("__NEWLINE__") {
1940        let close = ctx.listing.is_empty() || ctx.listing.starts_with('\n');
1941        if close {
1942          break;
1943        }
1944      } else if let Some(m) = re.find(&ctx.listing)
1945        && m.start() == 0
1946      {
1947        // Before closing, check if quoted_re matches (takes priority)
1948        let is_quoted = ctx.quoted_re.find(&ctx.listing).is_some();
1949        if !is_quoted {
1950          ctx.colnum += m.len() as i64;
1951          ctx.listing = ctx.listing[m.end()..].to_string();
1952          break;
1953        }
1954      }
1955    }
1956
1957    // 2. Literate expressions
1958    // Perl: use LITERATE_INNER_RE inside delimited contexts, LITERATE_RE at top level
1959    {
1960      let lit_re = if end_re.is_some() {
1961        // Inside a delimited context: only non-protected patterns
1962        build_literate_re(true)
1963      } else {
1964        ctx.literate_re.clone()
1965      };
1966      if let Some(ref re) = lit_re
1967        && let Some(m) = re.find(&ctx.listing)
1968      {
1969        let matched = m.as_str().to_string();
1970        ctx.listing = ctx.listing[m.end()..].to_string();
1971        ctx.colnum += matched.len() as i64;
1972        // Find the replacement tokens for this pattern
1973        let repl_key = s!("LST_LIT@{matched}");
1974        if let Some(Stored::Tokens(replacement)) = lookup_value(&repl_key) {
1975          ctx.lsttokens.push(T_CS!("\\@listingLiterate"));
1976          ctx.lsttokens.push(T_BEGIN!());
1977          ctx.lsttokens.extend(replacement.unlist());
1978          ctx.lsttokens.push(T_END!());
1979        }
1980        continue;
1981      }
1982    }
1983
1984    // 3. Delimiters — strings, comments, escapes, mathescape
1985    if let Some(ref delim_re) = ctx.delim_re.clone()
1986      && let Some(m) = delim_re.find(&ctx.listing)
1987    {
1988      let open = m.as_str().to_string();
1989      ctx.listing = ctx.listing[m.end()..].to_string();
1990      ctx.colnum += open.len() as i64;
1991
1992      // Look up delimiter info. The `invisible` flag is already applied
1993      // at delimiter-registration time (lst_add_delimiter sets empty
1994      // open_tex / close_tex when invisible, so lst_class_begin /
1995      // lst_class_end never emit the delim chars). No runtime check
1996      // needed here — Perl lstProcess_internal also has no separate
1997      // gate at this point.
1998      let class_key = s!("LST_DELIM@{open}@class");
1999      let close_key = s!("LST_DELIM@{open}@close");
2000      let classname = lookup_value(&class_key)
2001        .map(|v| v.to_string())
2002        .unwrap_or_default();
2003      let close_re_str = lookup_value(&close_key)
2004        .map(|v| v.to_string())
2005        .unwrap_or_default();
2006
2007      // Perl: lstProcessPush(lstClassBegin($classname))
2008      // Note: delimiter chars come from begin/end tokens in lstClassBegin/lstClassEnd
2009      ctx.lsttokens.extend(lst_class_begin(&classname));
2010
2011      // Check if this is an 'eval' class (mathescape, texcl, escapechar)
2012      let is_eval = lst_class_property(&classname, "eval").is_some_and(|v| v == "true" || v == "1");
2013
2014      if is_eval {
2015        // For eval classes: match until close, then tokenize the content as TeX
2016        // Perl: TokenizeBalanced($string) — close delimiter is NOT separately emitted,
2017        // because lstClassEnd already provides the closing tokens (e.g. $ for mathescape)
2018        if close_re_str == "__NEWLINE__" {
2019          // Sentinel: read until newline (for texcl line comments)
2020          let content = if let Some(pos) = ctx.listing.find('\n') {
2021            let c = ctx.listing[..pos].to_string();
2022            ctx.listing = ctx.listing[pos..].to_string();
2023            c
2024          } else {
2025            let c = ctx.listing.clone();
2026            ctx.listing.clear();
2027            c
2028          };
2029          let content_tokens = tokenize_balanced(&content);
2030          ctx.lsttokens.extend(content_tokens);
2031        } else {
2032          if let Ok(close_re) = Regex::new(&close_re_str)
2033            && let Some(cm) = close_re.find(&ctx.listing)
2034          {
2035            let content = ctx.listing[..cm.start()].to_string();
2036            ctx.listing = ctx.listing[cm.end()..].to_string();
2037            // Tokenize the content as real TeX (not raw listing)
2038            let content_tokens = tokenize_balanced(&content);
2039            ctx.lsttokens.extend(content_tokens);
2040            // Note: close delimiter is NOT pushed — lstClassEnd handles it
2041          }
2042        }
2043      } else {
2044        // For non-eval classes (strings, comments): recurse with limited delimiters
2045        let recursive_key = s!("LST_DELIM@{open}@recursive");
2046        let is_recursive = matches!(lookup_value(&recursive_key), Some(Stored::Bool(true)));
2047        if !close_re_str.is_empty() {
2048          // Sentinel close patterns (for zero-width assertions) use a dummy regex
2049          // that contains the sentinel as a flag — the actual check happens in step 1.
2050          let close_re_pattern = if close_re_str.starts_with("__") {
2051            // Create a regex that will never match normal text but carries the sentinel

4614      assign_value(table, stored, None);
4615    }
4616  }
4617  // Copy LST_LITERAL
4618  if let Some(stored) = lookup_value("LST_LITERAL") {
4619    assign_value("LST_LITERAL", stored, None);
4620  }
4621
4622  // Iterate over key-value pairs, in order
4623  for (key, val) in kv.get_pairs() {
4624    let val_tokens = lst_un_group(val.clone().owned_tokens());
4625    let cs = T_CS!(s!("\\lst@@{key}"));
4626    if has_meaning(&cs) {
4627      // Defaults for bare keys are already resolved during KeyVals parsing (add_value with
4628      // use_default). Digest: \lst@@KEY <value> \end
4629      let mut digest_tokens = vec![cs];
4630      if let Some(ref val_tks) = val_tokens {
4631        digest_tokens.extend(val_tks.unlist_ref().iter().cloned());
4632      }
4633      digest_tokens.push(T_CS!("\\end"));
4634      let _ = digest(Tokens::new(digest_tokens));
4635    }
4636    // Store LST@KEY => value
4637    let state_key = s!("LST@{key}");
4638    match val_tokens {
4639      Some(tks) => assign_value(&state_key, Stored::Tokens(tks), None),
4640      None => assign_value(&state_key, Stored::Tokens(Tokens!()), None),
4641    }
4642  }
4643}

4645/// Perl: lstActivateLanguage — load and activate a language definition.
4646/// Handles dialect lookup: first checks LSTDD@LANG for default dialect,
4647/// then looks up LST@LANGUAGE@LANG$DIALECT.
4648fn lst_activate_language(language: &str, dialect: Option<&str>) {
4649  let lang = language.to_uppercase().replace(char::is_whitespace, "");
4650  if lang.is_empty() {
4651    return;
4652  }
4653  // Determine dialect: explicit, or from default dialect state
4654  let dialect_str = match dialect {
4655    Some(d) if !d.is_empty() => d.to_uppercase().replace(char::is_whitespace, ""),
4656    _ => {
4657      let dd_key = s!("LSTDD@{lang}");
4658      lookup_value(&dd_key)
4659        .map(|v| {
4660          v.to_string()
4661            .to_uppercase()
4662            .replace(char::is_whitespace, "")
4663        })
4664        .unwrap_or_default()
4665    },
4666  };
4667  // Build the lookup key
4668  let name = if dialect_str.is_empty() {
4669    s!("LST@LANGUAGE@{lang}")
4670  } else {
4671    s!("LST@LANGUAGE@{lang}${dialect_str}")
4672  };
4673  // Try to find the language definition, also try without dialect
4674  let stored = lookup_value(&name).or_else(|| {
4675    if !dialect_str.is_empty() {
4676      lookup_value(&s!("LST@LANGUAGE@{lang}"))
4677    } else {
4678      None
4679    }
4680  });
4681  if let Some(stored) = stored {
4682    match stored {
4683      Stored::KeyVals(kv) => {
4684        // Stored as KeyVals — call lst_activate directly (matches Perl: lstActivate($values))
4685        lst_activate(Some(&kv));
4686      },
4687      Stored::Tokens(kv_tokens) => {
4688        // Stored as tokens — wrap in \lstset{...} and digest
4689        let _ = digest(Tokens::new(
4690          vec![T_CS!("\\lstset")]
4691            .into_iter()
4692            .chain(std::iter::once(T_BEGIN!()))
4693            .chain(kv_tokens.unlist().iter().cloned())
4694            .chain(std::iter::once(T_END!()))
4695            .collect(),
4696        ));
4697      },
4698      _ => {},
4699    }
4700  }
4701}

4721/// `line` up to its first unescaped `%` while `%` is a comment character (tex.web
4722/// §347: catcode 14 ends the line; `\%` is a control symbol). Under doc.sty's
4723/// `\MakePercentIgnore` (`\DocInput`) `%` is ignored, not a comment: the line is kept.
4724fn strip_tex_comment(line: &str) -> &str {
4725  if lookup_catcode('%') != Some(Catcode::COMMENT) {
4726    return line;
4727  }
4728  let mut escaped = false;
4729  for (at, c) in line.char_indices() {
4730    if c == '%' && !escaped {
4731      return &line[..at];
4732    }
4733    escaped = !escaped && c == '\\';
4734  }
4735  line
4736}

4757/// Write-file tee (lstmisc.sty aspect): when `\lst@BeginWriteFile`/
4758/// `\lst@BeginAlsoWriteFile` armed a target, append the captured listing
4759/// text to it. Returns false when the listing should NOT also display
4760/// (`\lst@BeginWriteFile`, the `\@gobble` output-box variant).
4761fn lst_writefile_tee(text: &str) -> bool {
4762  let file = match lookup_value("LST@WF@file") {
4763    Some(Stored::String(sym)) => with(sym, |s| s.to_string()),
4764    _ => String::new(),
4765  };
4766  if file.is_empty() {
4767    return true;
4768  }
4769  // Virtual store only — `\input`-back consumers (cnltx, incgraph) resolve
4770  // the written example via the VFS (find_file consults it first). A disk
4771  // write here landed in the process CWD, not the destination directory
4772  // (repo-root leakage from test runs). Appends within one begin/end span
4773  // (`\lst@WFAppend`); `lst_writefile_open` truncates on each fresh begin.
4774  // Real listings tees through `\lst@Append` (lstmisc.sty:47-64), AFTER the
4775  // per-line `gobble` was applied, so the written file holds the gobbled
4776  // text — what `\lst@sampleInput` (lstdoc.sty:187-192) re-inputs as code.
4777  let gobble = lst_get_number("gobble").max(0) as usize;
4778  let mut mirrored = vfs_read(&file).unwrap_or_default();
4779  for line in text.split('\n') {
4780    mirrored.extend(line.chars().skip(gobble));
4781    mirrored.push('\n');
4782  }
4783  vfs_store(&file, &mirrored);
4784  lookup_bool("LST@WF@also")
4785}
