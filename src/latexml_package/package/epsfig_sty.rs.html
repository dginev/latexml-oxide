29  // Perl epsfig.sty.ltxml L35-52: \psfig RequiredKeyVals:epsGin emits
30  // an ltx:graphics element. properties closure extracts 'file' or
31  // 'figure' key as the graphic path, removes those keys from the kv
32  // set, injects clip=true if epsfclip was toggled on, and serializes
33  // remaining key=value pairs as the options string. Prior Rust was a
34  // \psfig{} primitive that discarded everything, so every \psfig call
35  // produced no graphic node at all.
36  DefConstructor!("\\psfig RequiredKeyVals:epsGin",
37    "<ltx:graphics graphic='#graphic' candidates='#candidates' options='#options'/>",
38    enter_horizontal => true,
39    properties => sub[args] {
40      use DigestedData::*;
41      let file = if let Some(kv_arg) = args[0].as_ref() {
42        if let KeyVals(kv) = kv_arg.data() {
43          let raw = kv.get_value("file")
44            .or_else(|| kv.get_value("figure"))
45            .map(|v| v.to_string())
46            .unwrap_or_default();
47          raw.trim().to_string()
48        } else { String::new() }
49      } else { String::new() };
50      let candidates = util::image::image_candidates(&file);
51      // Serialize remaining keyvals as "k=v,k=v" (skip file/figure,
52      // inject clip=true if epsfclip was toggled on).
53      let clip_on = lookup_value("epsfclip")
54        .map(|v| v.to_string() != "0" && !v.to_string().is_empty())
55        .unwrap_or(false);
56      let mut opts = Vec::<String>::new();
57      if let Some(kv_arg) = args[0].as_ref()
58        && let KeyVals(kv) = kv_arg.data() {
59          let mut saw_clip = false;
60          for (k, v) in kv.get_pairs() {
61            if k == "file" || k == "figure" { continue; }
62            if k == "clip" { saw_clip = true; }
63            let v_str = v.to_string();
64            if v_str.is_empty() { opts.push(k.clone()); }
65            else { opts.push(format!("{}={}", k, v_str)); }
66          }
67          if clip_on && !saw_clip { opts.push("clip=true".into()); }
68        }
69      let options = opts.join(",");
70      Ok(stored_map!("graphic" => file, "candidates" => candidates, "options" => options))
71    },
72    // Perl `sizer => \&image_graphicx_sizer` (epsfig.sty.ltxml:37), as graphicx's `\includegraphics`: the box is the
73    // file's size under the `width`/`height` keys. Without it the box was the size of its arguments, 0 for the file
74    // name the properties take out, so a `\subfigure[…]{\psfig{file=…,width=0.4\textwidth}}` panel was 0 wide and joined
