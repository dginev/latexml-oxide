100      // are LITERAL url content. Demote `{`/`}` to OTHER so
101      // read_until_token doesn't engage its balanced-read branch on `{`
102      // and overshoot the closing delimiter. Real users write
103      // `\\path|{partial,| ... |partial}|` across multiple `\\urldef`
104      // calls, where `{`/`}` are literal chars in separate `|...|`
105      // paths. Driver 1906.08946 — without this demotion read_until('|')
106      // on `\\path|{...,|` reads PAST the closing `|` looking for a

115      // the surrounding `{\url www...pdf}` group's closing `}` must keep
116      // its END catcode so it still closes the group after the unread.
117      // Demoting it froze `}` as OTHER → the `{` group stayed open → the
118      // trailing `\endgroup` from `\url` hit the boxing group and errored
119      // (`Attempt to close non-boxing group`). Perl never demotes `{`/`}`
120      // (semiverbatim SPECIALS = ^_~&$#'). Witness 1503.07894 (bibitem
121      // `{\url www.maths...pdf}`).

218/// The string a `\url`/`\path` call as written prints (`\path|{a,b}@x.edu|`, `\url{a@x.edu}`): what stands between its
219/// delimiters, or the whole of `written` when it is no such call.
220fn url_printed_text(written: &str) -> String {
221  let Some(rest) = written.strip_prefix('\\') else {
222    return written.to_string();
223  };
224  let rest = rest
225    .trim_start_matches(|c: char| c.is_ascii_alphabetic())
226    .trim_start();
227  let mut chars = rest.chars();
228  let Some(open) = chars.next() else {
229    return written.to_string();
230  };
231  let close = if open == '{' { '}' } else { open };
232  let inner: &str = chars.as_str();
233  let mut depth = 0usize;
234  for (at, c) in inner.char_indices() {
235    if open == '{' && c == '{' {
236      depth += 1;
237    } else if c == close {
238      if depth == 0 {
239        return inner[..at].to_string();
240      }
241      depth -= 1;
242    }
243  }
244  written.to_string()
245}
