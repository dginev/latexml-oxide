299/// The records of an amsrefs database (`.ltb`) that `\ReadBibLoop` (amsrefs.sty:1348-1369) executes: it `\read`s
300/// the file a record at a time — a line, extended while its braces are open — and executes the records whose first
301/// token is `\bib`; a record that starts with `\endinput` ends the file. `%` comments are dropped as TeX reads them.
302fn amsrefs_database_records(text: &str) -> Vec<String> {
303  let mut records = Vec::new();
304  let mut record = String::new();
305  let mut depth: i64 = 0;
306  for line in text.lines() {
307    // Strip the comment: the first `%` not escaped by a backslash.
308    let mut code = String::new();
309    let mut chars = line.chars().peekable();
310    while let Some(c) = chars.next() {
311      match c {
312        '\\' => {
313          code.push(c);
314          if let Some(next) = chars.next() {
315            code.push(next);
316          }
317        },
318        '%' => break,
319        '{' => {
320          depth += 1;
321          code.push(c);
322        },
323        '}' => {
324          depth -= 1;
325          code.push(c);
326        },
327        _ => code.push(c),
328      }
329    }
330    record.push_str(&code);
331    record.push('\n');
332    if depth > 0 {
333      continue;
334    }
335    depth = 0;
336    let head = record.trim_start();
337    if head.starts_with("\\endinput") && !head[9..].starts_with(|c: char| c.is_ascii_alphabetic()) {
338      break;
339    }
340    if head.starts_with("\\bib") && !head[4..].starts_with(|c: char| c.is_ascii_alphabetic()) {
341      records.push(head.to_string());
342    }
343    record.clear();
344  }
345  records
346}

348/// The key of a database record `\bib{key}{type}{…}` (or `\bib*{key}…`).
349fn amsrefs_record_key(record: &str) -> Option<String> {
350  let rest = record.strip_prefix("\\bib")?.trim_start();
351  let rest = rest.strip_prefix('*').unwrap_or(rest).trim_start();
352  let rest = rest.strip_prefix('{')?;
353  rest.find('}').map(|end| rest[..end].trim().to_string())
