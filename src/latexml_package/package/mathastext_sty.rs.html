1//! mathastext.sty — math font substitution.
2//!
3//! Like fdsymbol.sty, mathastext L1075 runs `\let\lbrace\undefined`
4//! before `\DeclareMathDelimiter{\lbrace}`. Our `:locked` guard
5//! blocks the redef, leaving `\lbrace` (and `\rbrace`, etc.) UNDEFINED
6//! and cascading 500+ math-mode errors per paper.
7//!
8//! Perl LaTeXML has no mathastext binding and skips it via
9//! INCLUDE_STYLES=false. Match with a no-op binding — math-font
10//! substitution is moot for XML/HTML output.
11//!
12//! Witness 2410.03274 (\lbrace cascade with 505 errors).
13use crate::prelude::*;
14
15#[rustfmt::skip]
16LoadDefinitions!({
17  // Don't run mathastext's math-symbol slot reassignments: the kernel \lbrace / \rbrace / \langle / etc. remain
18  // authoritative. Its user commands choose fonts and math spacing, presentation the XML does not carry, so they
19  // take their arguments and do nothing (mathastext.sty 1.4e): `\MTsetmathskips{<char>}{<skip>}{<skip>}` (:964, 54
20  // papers; 1712.02869, 1804.04170), `\MTunsetmathskips` (:968); the font choices `\MTencoding`, `\MTfamily`,
21  // `\MTseries`, `\MTshape`, `\MTboldvariant`, `\MTlettershape` (:837-842, with their `\Mathastext…` names
22  // :843-848), `\MTgreekfont` (:857-858); the switches `\MTitgreek`, `\MTupgreek` (:849-850), `\MTmathactiveletters`,
23  // `\MTmathstandardletters`, `\MTicinmath`, `\MTnoicinmath` (:934-958), `\MT(easy)nonletters(donot)obeymathxx`
24  // (:490-549), `\MTexplicitbraces(donot)obeymathxx` (:1473-1477); `\MTWillUse[…]{…}{…}{…}{…}` (:2677). The
25  // versions are math versions: `\MTDeclareVersion[…]{<name>}{…}{…}{…}{…}` declares `<name>` (:2411-2420) and
26  // `\Mathastext[<name>]` through it (:2688-2697), and `\MTversion{<name>}` (`[<text version>]{<name>}`, or
27  // `*{<name>}`) selects it with `\mathversion` first (:2585-2588; `\MTversion{bold}` is bold math), its text fonts
28  // not modelled. Its `\Mathsf`/`\Mathtt` alphabets (:701-702, the text fonts' sans and typewriter) are
29  // `\mathsf`/`\mathtt`.
30  def_macro_noop("\\MTsetmathskips{}{}{}")?;
31  def_macro_noop("\\MTunsetmathskips{}")?;
32  for cs in ["\\MTencoding", "\\MTfamily", "\\MTseries", "\\MTshape", "\\MTboldvariant", "\\MTlettershape",
33    "\\MTgreekfont"] {
34    def_macro_noop(&s!("{cs}{{}}"))?;
35  }
36  for (alias, cs) in [("\\Mathastextencoding", "\\MTencoding"), ("\\Mathastextfamily", "\\MTfamily"),
37    ("\\Mathastextseries", "\\MTseries"), ("\\Mathastextshape", "\\MTshape"),
38    ("\\Mathastextboldvariant", "\\MTboldvariant"), ("\\Mathastextlettershape", "\\MTlettershape"),
39    ("\\Mathastextgreekfont", "\\MTgreekfont")] {
40    Let!(&T_CS!(alias), &T_CS!(cs));
41  }
42  for cs in ["\\MTitgreek", "\\MTupgreek", "\\MTmathactiveletters", "\\MTmathstandardletters", "\\MTicinmath",
43    "\\MTnoicinmath", "\\MTnonlettersobeymathxx", "\\MTnonlettersdonotobeymathxx", "\\MTeasynonlettersobeymathxx",
44    "\\MTeasynonlettersdonotobeymathxx", "\\MTexplicitbracesobeymathxx", "\\MTexplicitbracesdonotobeymathxx"] {
45    def_macro_noop(cs)?;
46  }
47  DefMacro!("\\MTDeclareVersion[]{}{}{}{}{}", "\\DeclareMathVersion{#2}");
48  Let!("\\MathastextDeclareVersion", "\\MTDeclareVersion");
49  DefMacro!("\\Mathastext[]", "\\if\\relax\\detokenize{#1}\\relax\\else\\DeclareMathVersion{#1}\\fi");
50  def_macro_noop("\\MTWillUse[]{}{}{}{}")?;
51  Let!("\\MathastextWillUse", "\\MTWillUse");
52  Let!("\\Mathastextwilluse", "\\MTWillUse");
53  DefMacro!("\\MTversion OptionalMatch:* []{}", "\\mathversion{#3}");
54  for alias in ["\\MathastextVersion", "\\Mathastextversion", "\\MTVersion", "\\mathastextversion"] {
55    Let!(&T_CS!(alias), &T_CS!("\\MTversion"));
56  }
57  Let!("\\Mathsf", "\\mathsf");
58  Let!("\\Mathtt", "\\mathtt");
59});
