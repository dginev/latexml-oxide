4LoadDefinitions!({
5  // Perl: german.sty.ltxml
6  // This should be essentially the same, right?
7  // (considering we don't do hyphenation, etc)
8  RequirePackage!("babel", options => vec!["german".to_string()]);
9
10  // Alias the `germanb` dialect's language number to `german` (kernel/dump
11  // `\l@german`), exactly as the real germanb.ldf does via
12  // `\let\l@germanb\l@german`. Without this, `\usepackage[…,germanb]{babel}`
13  // selects `germanb` as the main language and babel's
14  // `\selectlanguage{germanb}` → `\bbl@iflanguage{germanb}` errors "You haven't
15  // defined the language 'germanb' yet" — because this binding REPLACES the raw
16  // germanb.ldf load (which is where `\l@germanb` would otherwise come from).
17  // Witness: arXiv:1010.4065 (`\usepackage[english,germanb]{babel}`).
18  RawTeX!(r"\expandafter\ifx\csname l@germanb\endcsname\relax
19    \expandafter\ifx\csname l@german\endcsname\relax
20      \expandafter\newlanguage\csname l@germanb\endcsname
21    \else
22      \expandafter\let\csname l@germanb\expandafter\endcsname\csname l@german\endcsname
23    \fi
24  \fi");
25
26  // German caption strings (from germanb.ldf). \providecommand so raw
27  // babel/germanb.ldf processing (if any) doesn't overwrite.
28  RawTeX!(r"\providecommand\captionsgerman{%
29    \def\prefacename{Vorwort}\def\refname{Literatur}%
30    \def\abstractname{Zusammenfassung}\def\bibname{Literaturverzeichnis}%
31    \def\chaptername{Kapitel}\def\appendixname{Anhang}%
32    \def\contentsname{Inhaltsverzeichnis}%
33    \def\listfigurename{Abbildungsverzeichnis}%
34    \def\listtablename{Tabellenverzeichnis}%
35    \def\indexname{Index}\def\figurename{Abbildung}%
36    \def\tablename{Tabelle}\def\partname{Teil}%
37    \def\pagename{Seite}\def\seename{siehe}%
38    \def\alsoname{siehe auch}\def\proofname{Beweis}}");
39  // germanb.ldf:124-130: the German date (an empty stub left `\today` in English).
40  // Guard: `perfect_kernel_batch58::today_follows_the_babel_language`.
41  RawTeX!(r#"\providecommand\month@german{\ifcase\month\or
42    Januar\or Februar\or M\"arz\or April\or Mai\or Juni\or
43    Juli\or August\or September\or Oktober\or November\or Dezember\fi}"#);
44  RawTeX!(r"\providecommand\dategerman{\def\today{\number\day.~\month@german\space\number\year}}");
45  RawTeX!(r"\providecommand\captionsngerman{\captionsgerman}");
46  RawTeX!(r"\providecommand\datengerman{\dategerman}");
47  // germanb.ldf language hooks — see the matching note in ngerman_sty.rs.
48  RawTeX!(r#"\providecommand\extrasgerman{\@ifundefined{languageshorthands}{}{\languageshorthands{german}\bbl@activate{"}}}"#);
49  RawTeX!(r#"\providecommand\noextrasgerman{\@ifundefined{bbl@deactivate}{}{\bbl@deactivate{"}}}"#);
50  RawTeX!(r"\providecommand\extrasngerman{\extrasgerman}");
51  RawTeX!(r"\providecommand\noextrasngerman{\noextrasgerman}");
52
53  // The `"` shorthand dispatch (`\lx@german@dq@dispatch`), `\mdqon` and
54  // `\mdqoff` live in babel_support_sty.rs: they belong to babel's German
55  // (every `\usepackage[ngerman]{babel}` document, not only german.sty).
56  // germanb.ldf L173: `\def\dq{"}` — `\dq` yields a literal double-quote (the
57  // saved catcode-12 `"`, since `"` itself becomes the active shorthand). Map to
58  // `\textquotedbl`, LaTeXML's literal double-quote. (An earlier note recorded
59  // `\dq` as "actively undefined after this binding runs" — that was while the
60  // binding truncated past L57, before `\bbl@allowhyphens` below was added; the
61  // truncation is resolved, so this now sticks.)
62  RawTeX!(r"\providecommand\dq{\textquotedbl}");
63  // germanb.ldf helper stubs — no-op in Rust (no hyphenation / ligature phase).
64  RawTeX!(r"\providecommand\bbl@allowhyphens{}");
65  RawTeX!(r"\providecommand\bbl@ss{\ss}\providecommand\bbl@SS{SS}");
66  RawTeX!(r"\providecommand\bbl@sz{\ss}\providecommand\bbl@SZ{SZ}");
67  // german.sty:314-319, 662-671: the user-level switches. `\germanTeX` is
68  // what the kernel's first aid (latex2e-first-aid-for-external-files.ltx:160-166,
69  // `file/german.sty/after`) runs once our binding reports the file's version
70  // — and what documents written for german.sty call themselves (a0poster
71  // a0/a0_eng, adrconv, akletter, cryst … 11 TL manuals errored
72  // `undefined:\germanTeX`). `\umlautlow`/`\umlauthigh` choose the accent
73  // placement of `\"` — a font matter with no XML counterpart.
74  RawTeX!(r"\providecommand\umlautlow{}\providecommand\umlauthigh{}");
75  RawTeX!(r"\providecommand\germanTeX{\mdqon\selectlanguage{german}}");
76  // german.sty:373-377 `\def@dqmacro{c}{text}{math}` / `\let@dqmacro` — the
77  // `"c` shorthand installers a derivative patches (germkorr.sty:141-144);
78  // recorded under german's own `\@grmn@@<c>dq` name.
79  RawTeX!(
80    r"\def\grmn@dq@macro#1#2{\expandafter#1\csname @grmn@@\string #2dq\endcsname}
81\def\def@dqmacro#1#2#3{\grmn@dq@macro\def{#1}{{#2}{#3}}}
82\def\let@dqmacro#1#2{\begingroup\edef\x{\endgroup\let\grmn@dq@macro\noexpand{#1}\grmn@dq@macro\noexpand{#2}}\x}"
83  );
84});
