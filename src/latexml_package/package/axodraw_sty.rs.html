1//! axodraw.sty — J.A.M. Vermaseren's Feynman-diagram package, version 1
2//! ("Made by J.A.M. Vermaseren 14-feb-1994").
3//!
4//! axodraw v1 is neither in TeX Live (which has its successor `axodraw2`) nor,
5//! mostly, shipped with the papers that load it: run 336 had 1,507 arXiv
6//! papers with `\Text` undefined (149 as their first error; witnesses
7//! hep-th0506235, hep-ph0512261, hep-ph0305218, 0710.3255, 0809.0800). Its
8//! drawing layer is PostScript: every command is `\put(\axoxoff,\axoyoff)`
9//! around a `\special{"…}` that calls a procedure the package installs with
10//! `\special{! …}` (dvips header code), so a raw load draws nothing even where
11//! the file is present. This binding keeps v1's TeX layer — the same command
12//! signatures, `\Set*` state macros and `\put` placement — and replaces each
13//! `\special` with `\lx@axo@draw{procedure}{arguments}`, which evaluates the
14//! procedure's geometry here, transcribed from the v1 PostScript, into the
15//! picture model (`ltx:line`, `ltx:bezier`, `ltx:arc`, `ltx:circle`,
16//! `ltx:polygon`, `ltx:rect`, `ltx:ellipse`) that `{picture}` and the
17//! pstricks binding emit and the SVG post-processor renders.
18//!
19//! A paper's own copy of axodraw.sty (about one zip in 830 ships one) is
20//! answered by this binding too, as any binding outranks a raw file: raw, its
21//! drawing is lost all the same. axodraw2 (TeX Live) and axodraw4j
22//! (JaxoDraw's 2008 keyval forerunner of axodraw2, shipped by papers:
23//! 1301.5845, 1701.08134) are other packages, read from their files
24//! (axodraw2_sty.rs keeps the versioned-name fallback off axodraw2).
25//!
26//! An argument is what v1 splices into PostScript, so it may be a PostScript
27//! expression over a dvips-header constant (`{20 Tdensity mul}`, 0901.1195,
28//! 1701.01554): `ps_value` evaluates the arithmetic.
29//!
30//! The command set is the union of the v1 variants arXiv papers ship (all
31//! share one PostScript geometry): the original 1994 file (hep-ph9702393,
32//! hep-ph0101185) and the colordvi-aware one adding `\SetColor`, `\CCirc`,
33//! `\CBox(c)`, `\COval`, `\CText`, `\C2Text` and the triangles (hep-ph0101235).
34//!
35//! Units, v1 semantics (both sides deliberate in v1, and relied on by papers:
36//! 0809.0800 writes `\unitlength 0.6pt\SetScale{0.6}`): graphics coordinates
37//! are PostScript points scaled by `\SetScale` and ignore `\unitlength`;
38//! `\Text`/`\rText` place TeX text at `\unitlength` multiples, unscaled. v1
39//! draws in big points; this reads them as TeX points, the correction
40//! axodraw2 made (axodraw2.sty:3080 `/gs{gsave 1.00375 div dup scale}`), so a
41//! label set at the same coordinates as a line meets it.

54/// PostScript's current transformation matrix: user `(x,y)` maps to
55/// `(a·x + c·y + e, b·x + d·y + f)` in picture points.
56#[derive(Clone, Copy, Debug)]
57struct Ctm {
58  a: f64,
59  b: f64,
60  c: f64,
61  d: f64,
62  e: f64,
63  f: f64,
64}
65
66impl Ctm {
67  /// v1's `gs` (`gsave dup scale`): every procedure draws in user space
68  /// scaled by `\axoscale`.
69  fn scaled(s: f64) -> Self {
70    Ctm {
71      a: s,
72      b: 0.0,
73      c: 0.0,
74      d: s,
75      e: 0.0,
76      f: 0.0,
77    }
78  }
79
80  /// PostScript `tx ty translate`.
81  fn translate(self, tx: f64, ty: f64) -> Self {
82    let (e, f) = self.apply((tx, ty));
83    Ctm { e, f, ..self }
84  }
85
86  /// PostScript `angle rotate` (degrees, anticlockwise).
87  fn rotate(self, degrees: f64) -> Self {
88    let (s, c) = degrees.to_radians().sin_cos();
89    Ctm {
90      a: self.a * c + self.c * s,
91      b: self.b * c + self.d * s,
92      c: self.c * c - self.a * s,
93      d: self.d * c - self.b * s,
94      ..self
95    }
96  }
97
98  fn apply(&self, (x, y): Pt) -> Pt {
99    (
100      self.a * x + self.c * y + self.e,
101      self.b * x + self.d * y + self.f,
102    )
103  }
104}

106/// PostScript `round`: the nearest integer, a half rounding up.
107fn ps_round(x: f64) -> f64 { (x + 0.5).floor() }
108/// PostScript `dy dx atan`, in degrees.
109fn ps_atan(dy: f64, dx: f64) -> f64 { dy.atan2(dx).to_degrees() }
110fn cosd(degrees: f64) -> f64 { degrees.to_radians().cos() }
111fn sind(degrees: f64) -> f64 { degrees.to_radians().sin() }

144/// v1's dash fitting (`/dashline`, `/dashcarc`): the `[dsize dsize]` pattern
145/// is stretched so a path of length `r` holds an odd number `2n+1` of dashes
146/// and gaps, beginning and ending with a dash.
147fn fit_dash(r: f64, dsize: f64) -> f64 {
148  let n = ps_round(r / (dsize * 2.0) - 0.5).max(0.0);
149  r / (n * 2.0 + 1.0)
150}

152/// A v1 argument as the number the PostScript procedure receives: the
153/// arguments are spliced into PostScript, so beside a number one may be a
154/// PostScript expression — `{6.283 #3 mul 360 div … Tdensity mul}`
155/// (0901.1195, 1701.01554), naming constants a dvips header defined
156/// (`\special{! /Tdensity {0.10} def}`, recorded by `\special` as
157/// `dvips_header:<name>`). Evaluated on an operand stack with the arithmetic
158/// operators; `None` for anything else, a stack not left with one value, or a
159/// value that is not finite (`1 0 div`, `-1 sqrt`, or the text `inf`, which
160/// PostScript would reject: dvips fails the page).
161fn ps_value(text: &str) -> Option<f64> { ps_expression(text).filter(|v| v.is_finite()) }
162
163fn ps_expression(text: &str) -> Option<f64> {
164  let text = text.trim();
165  if let Ok(v) = text.parse::<f64>() {
166    return Some(v);
167  }
168  let mut stack: Vec<f64> = Vec::new();
169  for word in text.split_whitespace() {
170    if let Ok(v) = word.parse::<f64>() {
171      stack.push(v);
172      continue;
173    }
174    match word {
175      "add" | "sub" | "mul" | "div" | "exch" | "atan" => {
176        let b = stack.pop()?;
177        let a = stack.pop()?;
178        match word {
179          "add" => stack.push(a + b),
180          "sub" => stack.push(a - b),
181          "mul" => stack.push(a * b),
182          "div" => stack.push(a / b),
183          "atan" => stack.push(ps_atan(a, b).rem_euclid(360.0)),
184          _ => stack.extend([b, a]),
185        }
186      },
187      "neg" | "abs" | "sqrt" | "sin" | "cos" | "dup" => {
188        let a = stack.pop()?;
189        match word {
190          "neg" => stack.push(-a),
191          "abs" => stack.push(a.abs()),
192          "sqrt" => stack.push(a.sqrt()),
193          "sin" => stack.push(sind(a)),
194          "cos" => stack.push(cosd(a)),
195          _ => stack.extend([a, a]),
196        }
197      },
198      name => {
199        let value = lookup_string(&format!("dvips_header:{name}"));
200        stack.push(value.trim().parse::<f64>().ok()?);
201      },
202    }
203  }
204  match stack.as_slice() {
205    [v] => Some(*v),
206    _ => None,
207  }
208}

210/// The most pieces one procedure call draws: winding counts and axis
211/// subdivisions beyond it (`{20000000}` windings) are refused with the "Cannot
212/// draw" warning, as the PostScript would run for hours, not converted for as
213/// long. Real diagrams stay below a few hundred.
214const MAX_PIECES: f64 = 10_000.0;
215
216/// A count of pieces (windings, half wiggles, axis marks) a procedure may draw.
217fn drawable_count(count: f64) -> bool { count.abs() <= MAX_PIECES }
218
219/// The number of passes of PostScript's `first 1 limit {…} for` — none when
220/// `limit < first` (a NaN or `-inf` limit too) — or `None` for an infinite pass
221/// count or one beyond [`MAX_PIECES`].
222fn ps_for_count(first: f64, limit: f64) -> Option<usize> {
223  let passes = if limit >= first {
224    (limit - first).floor() + 1.0
225  } else {
226    0.0
227  };
228  (passes.is_finite() && drawable_count(passes)).then_some(passes as usize)
229}

231/// The value of a v1 state macro (`\axowidth` is `0.5 `, …) as a number; one
232/// PostScript cannot read (a direct `\def\axoscale{0,5}`; `\SetScale` and
233/// `\SetWidth` refuse such values themselves) is reported, once, and read as 0.
234fn axo_number(cs: &str) -> Result<f64> {
235  let text = do_expand(Tokens::new(vec![T_CS!(cs)]))?.to_string();
236  Ok(ps_value(&text).unwrap_or_else(|| {
237    if !already_reported(&format!("axodraw_value_{cs}_{}", text.trim())) {
238      Warn!(
239        "unexpected",
240        "axodraw",
241        format!("Cannot read {cs} = '{}' as a number; using 0", text.trim())
242      );
243    }
244    0.0
245  }))
246}

248/// A v1 color name (a colordvi/dvipsnam name: `Red`, `Blue`, …) as a color
249/// attribute; blank — v1 without colordvi, `\def\axocolor{ }` — is black.
250/// The names are dvips's whichever way they are known: plainly (colordvi,
251/// `color` with `usenames`, or its on-demand dvipsnam load), or as color's
252/// `named` model when `color` loaded dvipsnam.def for its `dvips` option
253/// without `usenames` (hep-ph0612301: `\documentclass[dvips]{article}`).
254fn axo_color(name: &str) -> String {
255  let name = name.trim();
256  if name.is_empty() {
257    return BLACK.to_string();
258  }
259  let named = format!("named_{name}");
260  if lookup_value(&color_key(name)).is_none() && lookup_value(&color_key(&named)).is_some() {
261    lookup_color(&named)
262  } else {
263    lookup_color(name)
264  }

295/// The elements one procedure call draws, in the coordinates of its `\put`.
296struct Drawing {
297  pen:        Pen,
298  base:       Ctm,
299  elements:   Vec<Attrs>,
300  /// An element was not drawn: its numbers were not finite.
301  degenerate: bool,
302}

304impl Drawing {
305  fn new(pen: Pen) -> Self {
306    let base = Ctm::scaled(pen.scale);
307    Drawing {
308      pen,
309      base,
310      elements: Vec::new(),
311      degenerate: false,
312    }
313  }

315  /// A user point: v1 adds `\axoxo`/`\axoyo` before the procedure runs.
316  fn at(&self, x: f64, y: f64) -> Pt { (x + self.pen.offset.0, y + self.pen.offset.1) }
317
318  /// Add an element computed from `numbers`. A degenerate argument (a zero
319  /// amplitude, coincident curve points) makes the PostScript divide by
320  /// zero, failing the dvips page; only the element it breaks is dropped,
321  /// and the drawing reported.
322  fn push(&mut self, qname: &str, numbers: &[f64], mut attrs: Attrs) {
323    if numbers.iter().any(|v| !v.is_finite()) {
324      self.degenerate = true;
325      return;
326    }
327    attrs.insert("_qname".to_string(), qname.to_string());
328    self.elements.push(attrs);
329  }

331  fn stroke(&self, color: &str, width: f64) -> Attrs {
332    map!(
333      "stroke" => color.to_string(),
334      "stroke-width" => px(width * self.pen.scale)
335    )
336  }

338  /// A dash pattern `[d d]`; a dash size of zero (`\DashLine(…){0}`, which
339  /// PostScript's `setdash` rejects) leaves the line solid.
340  fn dashed(&self, attrs: &mut Attrs, dash: Option<f64>) {
341    if let Some(d) = dash.filter(|d| d.is_finite() && *d > 0.0) {
342      let d = px(d * self.pen.scale);
343      attrs.insert("stroke-dasharray".to_string(), format!("{d} {d}"));
344    }
345  }

347  /// `newpath … lineto … stroke` with the current line width.
348  fn line(&mut self, ctm: Ctm, points: &[Pt], dash: Option<f64>) {
349    let width = self.pen.width;
350    self.line_width(ctm, points, dash, width);
351  }

353  fn line_width(&mut self, ctm: Ctm, points: &[Pt], dash: Option<f64>, width: f64) {
354    let mapped: Vec<Pt> = points.iter().map(|p| ctm.apply(*p)).collect();
355    let mut attrs = self.stroke(&self.pen.color.clone(), width);
356    attrs.insert("points".to_string(), px_points(&mapped));
357    self.dashed(&mut attrs, dash);
358    self.push("ltx:line", &flatten(&mapped), attrs);
359  }

361  /// One stroked `curveto` segment.
362  fn curve(&mut self, ctm: Ctm, points: [Pt; 4]) {
363    let mapped: Vec<Pt> = points.iter().map(|p| ctm.apply(*p)).collect();
364    let mut attrs = self.stroke(&self.pen.color.clone(), self.pen.width);
365    attrs.insert("points".to_string(), px_points(&mapped));
366    self.push("ltx:bezier", &flatten(&mapped), attrs);
367  }

369  /// `x y r a1 a2 arc stroke` under a translation: PostScript raises `a2` by
370  /// 360 until it is at least `a1`, so the arc runs anticlockwise from `a1`;
371  /// a sweep of 360 or more is the full circle.
372  fn arc(&mut self, ctm: Ctm, r: f64, a1: f64, a2: f64, dash: Option<f64>) {
373    let sweep = if a2 >= a1 {
374      a2 - a1
375    } else {
376      (a2 - a1).rem_euclid(360.0)
377    };
378    let (cx, cy) = ctm.apply((0.0, 0.0));
379    let mut attrs = self.stroke(&self.pen.color.clone(), self.pen.width);
380    attrs.insert("x".to_string(), px(cx));
381    attrs.insert("y".to_string(), px(cy));
382    attrs.insert("r".to_string(), px(r * self.pen.scale));
383    self.dashed(&mut attrs, dash);
384    if sweep >= 360.0 {
385      attrs.insert("fill".to_string(), "none".to_string());
386      self.push("ltx:circle", &[cx, cy, r, sweep], attrs);
387    } else if sweep > 0.0 {
388      attrs.insert("angle1".to_string(), format!("{a1}"));
389      attrs.insert("angle2".to_string(), format!("{}", a1 + sweep));
390      self.push("ltx:arc", &[cx, cy, r, sweep], attrs);
391    }
392  }

437    self.push("ltx:rect", &[x1, y1, x2, y2], attrs);

503  /// The frame of a straight procedure: origin at `p1`, x along `p1→p2`;
504  /// returns it with the length `dr`.
505  fn segment_frame(&self, p1: Pt, p2: Pt) -> (Ctm, f64) {
506    let (dx, dy) = (p2.0 - p1.0, p2.1 - p1.1);

508    (self.base.translate(p1.0, p1.1).rotate(ps_atan(dy, dx)), dr)

511  /// `/gluon`: a head and a tail of one winding each and `num-1` windings of
512  /// body, as one path of `curveto`s.
513  fn gluon(&mut self, p1: Pt, p2: Pt, ampi: f64, num: f64) {
514    let (f, dr) = self.segment_frame(p1, p2);
515    let num = ps_round(num - 0.5);
516    let inc = dr / (num * 2.0 + 2.0);
517    let amp8 = (ampi * 0.9).abs();
518    let mut x1 = inc * 2.0;
519    let mut current = (0.0, 0.0);
520    let mut curveto = |d: &mut Drawing, c1: Pt, c2: Pt, end: Pt| {
521      d.curve(f, [current, c1, c2, end]);
522      current = end;
523    };
524    curveto(
525      self,
526      (inc * 0.1, ampi * 0.5),
527      (inc * 0.5, ampi),
528      (inc * 1.4, ampi),
529    );
530    curveto(self, (x1 + amp8, ampi), (x1 + amp8, -ampi), (x1, -ampi));
531    // `2 1 num {…} for`, on an integer counter
532    for _ in 0..ps_for_count(2.0, num).unwrap_or(0) {
533      curveto(
534        self,
535        (x1 - amp8, -ampi),
536        (x1 - amp8, ampi),
537        (x1 + inc, ampi),
538      );
539      x1 += inc * 2.0;
540      curveto(self, (x1 + amp8, ampi), (x1 + amp8, -ampi), (x1, -ampi));
541    }
542    curveto(
543      self,
544      (x1 - amp8, -ampi),
545      (x1 - amp8, ampi),
546      (x1 + inc * 0.6, ampi),
547    );
548    curveto(
549      self,
550      (x1 + inc * 1.5, ampi),
551      (dr - inc * 0.1, ampi * 0.5),
552      (dr, 0.0),
553    );
554  }

556  /// `/photon` (`zigzag = false`): `2·num - 0.5` (rounded) half wiggles, each
557  /// a separately stroked `curveto` of alternating sign; `/zigzag`: the same
558  /// half periods as two `lineto`s.
559  fn photon(&mut self, p1: Pt, p2: Pt, ampi: f64, num: f64, zigzag: bool) {
560    let (mut f, dr) = self.segment_frame(p1, p2);
561    let num = ps_round(num * 2.0 - 0.5);
562    let x2 = dr / num;
563    let mut sign = 1.0;
564    // `1 1 num {…} for`, on an integer counter
565    for _ in 0..ps_for_count(1.0, num).unwrap_or(0) {
566      if zigzag {
567        self.line(f, &[(0.0, 0.0), (x2 / 2.0, ampi * sign), (x2, 0.0)], None);
568      } else {
569        // v1 `/pi 3.141592` (π to 6.5e-7, immaterial at any drawing size)
570        let a = 4.0 / 3.0 * x2 / PI;
571        let h = 4.0 / 3.0 * ampi * sign;
572        self.curve(f, [(0.0, 0.0), (a, h), (x2 - a, h), (x2, 0.0)]);
573      }
574      sign = -sign;
575      f = f.translate(x2, 0.0);
576    }
577  }

579  /// `/photonarc`: `2·num` (rounded) half wiggles on the arc, each one
580  /// `curveto` in a frame rotated by `phi` per step.
581  fn photon_arc(&mut self, c: Pt, radius: f64, arcstart: f64, arcend: f64, ampli: f64, num: f64) {
582    let num = ps_round(num * 2.0);
583    let arcend = if arcend < arcstart {
584      arcend + 360.0
585    } else {
586      arcend
587    };
588    let phi = (arcend - arcstart) / num;
589    let mut f = self.base.translate(c.0, c.1).rotate(arcstart);
590    let (cp, sp) = (cosd(phi), sind(phi));
591    let (cp2, sp2) = (cosd(phi / 2.0), sind(phi / 2.0));
592    let mut ampli = ampli;
593    // `1 1 num {…} for`, on an integer counter
594    for _ in 0..ps_for_count(1.0, num).unwrap_or(0) {
595      let beta = radius * phi / (180.0 * ampli);
596      let tt = (sp - cp * beta) / (cp + sp * beta);
597      let amp1 = ((radius + ampli) * 8.0 * (beta * cp2 - sp2)
598        - (beta * (4.0 + cp) + (tt * cp * 3.0 - sp * 4.0)) * radius)
599        / ((beta - tt) * 3.0);
600      let x1 = ((radius + ampli) * 8.0 * cp2 - (1.0 + cp) * radius) / 3.0 - amp1;
601      self.curve(f, [
602        (radius, 0.0),
603        (x1, (x1 - radius) * beta),
604        (amp1, (amp1 - radius * cp) * tt + radius * sp),
605        (radius * cp, radius * sp),
606      ]);
607      ampli = -ampli;
608      f = f.rotate(phi);
609    }
610  }

612  /// `/gluearc`: the straight gluon laid along the arc — a point `x` along
613  /// the arc and `a` from the centre maps to `(a·cos(x·const), a·sin(x·const))`,
614  /// with the control radii corrected (`amp4`…`amp7`) for the bending.
615  fn gluon_arc(&mut self, c: Pt, radius: f64, arcstart: f64, arcend: f64, ampi: f64, num: f64) {
616    let f = self.base.translate(c.0, c.1).rotate(arcstart);
617    let darc = arcend - arcstart;
618    // v1 `3.141592` (π)
619    let dr = darc / 180.0 * PI * radius;
620    let konst = darc / dr;
621    let num = ps_round(num - 0.5);
622    let inc = dr / (num * 2.0 + 2.0);
623    let amp8 = ampi * 0.9;
624    let amp1 = radius + ampi;
625    let amp2 = radius - ampi;
626    let amp3 = radius + ampi / 2.0;
627    let amp4 = amp1 / cosd((inc + amp8) * konst);
628    let amp5 = amp2 / cosd(amp8 * konst);
629    let amp6 = amp1 / cosd((inc * 0.6 + amp8) * konst);
630    let amp7 = amp1 / cosd(inc * 0.9 * konst);
631    let amp8 = amp8.abs();
632    let p = |x: f64, a: f64| -> Pt { (cosd(x * konst) * a, sind(x * konst) * a) };
633    let mut x1 = inc * 2.0;
634    let mut current = (radius, 0.0);
635    let mut curveto = |d: &mut Drawing, c1: Pt, c2: Pt, end: Pt| {
636      d.curve(f, [current, c1, c2, end]);
637      current = end;
638    };
639    curveto(
640      self,
641      p(inc * 0.1, amp3),
642      p(inc * 0.5, amp7),
643      p(inc * 1.4, amp1),
644    );
645    curveto(self, p(x1 + amp8, amp6), p(x1 + amp8, amp5), p(x1, amp2));
646    // `2 1 num {…} for`, on an integer counter
647    for _ in 0..ps_for_count(2.0, num).unwrap_or(0) {
648      curveto(
649        self,
650        p(x1 - amp8, amp5),
651        p(x1 - amp8, amp4),
652        p(x1 + inc, amp1),
653      );
654      x1 += inc * 2.0;
655      curveto(self, p(x1 + amp8, amp4), p(x1 + amp8, amp5), p(x1, amp2));
656    }
657    curveto(
658      self,
659      p(x1 - amp8, amp5),
660      p(x1 - amp8, amp6),
661      p(x1 + inc * 0.6, amp1),
662    );
663    curveto(
664      self,
665      p(x1 + inc * 1.5, amp7),
666      p(dr - inc * 0.1, amp3),
667      p(dr, radius),
668    );
669  }

671  /// `/makecurve` (`/docurve`): the cubic through consecutive points whose
672  /// tangents are fitted from the neighbours, the end tangents extrapolated.
673  fn smooth_curve(points: &[Pt]) -> Vec<[Pt; 4]> {
674    let docurve = |p0: Pt, p1: Pt, p2: Pt, p3: Pt| -> [Pt; 4] {
675      let ((x0, y0), (x1, y1), (x2, y2), (x3, y3)) = (p0, p1, p2, p3);
676      let c1 = (
677        (x1 * 2.0 + x2) / 3.0,
678        ((y1 - y0) / (x1 - x0) * (x2 - x0) + (y2 - y0) / (x2 - x0) * (x1 - x0) + y1 + y0 * 2.0)
679          / 3.0,
680      );
681      let c2 = (
682        (x1 + x2 * 2.0) / 3.0,
683        ((y2 - y3) / (x2 - x3) * (x1 - x3) + (y1 - y3) / (x1 - x3) * (x2 - x3) + y2 + y3 * 2.0)
684          / 3.0,
685      );
686      [p1, c1, c2, p2]
687    };
688    let n = points.len();
689    let mut segments = Vec::new();
690    if n < 3 {
691      return segments;
692    }
693    let ((x1, y1), (x2, y2), (x3, y3)) = (points[0], points[1], points[2]);
694    let x0 = x1 * 2.0 - x2;
695    let y0 =
696      ((y3 - y2) / (x3 - x2) - (y2 - y1) / (x2 - x1)) * 2.0 * ((x2 - x1) * (x2 - x1) / (x3 - x1))
697        + y1 * 2.0
698        - y2;
699    segments.push(docurve((x0, y0), points[0], points[1], points[2]));
700    for w in points.windows(4) {
701      segments.push(docurve(w[0], w[1], w[2], w[3]));
702    }
703    let ((x0, y0), (x1, y1), (x2, y2)) = (points[n - 3], points[n - 2], points[n - 1]);
704    let x3 = x2 * 2.0 - x1;
705    let y3 =
706      ((y2 - y1) / (x2 - x1) - (y1 - y0) / (x1 - x0)) * 2.0 * ((x2 - x1) * (x2 - x1) / (x2 - x0))
707        + y2 * 2.0
708        - y1;
709    segments.push(docurve(
710      points[n - 3],
711      points[n - 2],
712      points[n - 1],
713      (x3, y3),
714    ));
715    segments
716  }

718  /// `/logaxis` (`linear = false`) and `/linaxis`: the axis line and its
719  /// hash marks, the minor ones at 0.6 of the line width. `false` (nothing
720  /// drawn) when a step is not finite (`perdec` 0: `x1 = rr/0`) or a mark
721  /// count is beyond [`MAX_PIECES`]; the PostScript `for` loops are counted
722  /// up front and run on an integer counter, so no float step can stall.
723  /// Guard: `perfect_kernel_batch64w::axodraw_v1_unbounded_arguments`.
724  #[allow(clippy::too_many_arguments)]
725  fn axis(
726    &mut self,
727    p1: Pt,
728    p2: Pt,
729    count: f64,
730    per: f64,
731    hashsize: f64,
732    offset: f64,
733    width: f64,
734    linear: bool,
735  ) -> bool {
736    let (f, rr) = self.segment_frame(p1, p2);
737    let mut major = Vec::new();
738    let mut minor = Vec::new();
739    if linear {
740      let (numdec, perdec) = (count, per);
741      // `/x1 rr numdec perdec mul div def /y1 rr numdec div def
742      // /offset offset x1 mul def`
743      let x1 = rr / (numdec * perdec);
744      let y1 = rr / numdec;
745      let offset = offset * x1;
746      if !(x1.is_finite() && y1.is_finite() && offset.is_finite()) {
747        return false;
748      }
749      // `0 1 numdec {…} for`, `offset cvi 1 numdec perdec mul offset add {…} for`
750      let start = offset.trunc();
751      let (Some(n_major), Some(n_minor)) = (
752        ps_for_count(0.0, numdec),
753        ps_for_count(start, numdec * perdec + offset),
754      ) else {
755        return false;
756      };
757      major.extend((0..n_major).map(|i| i as f64 * y1 - offset));
758      minor.extend((0..n_minor).map(|i| (start + i as f64) * x1 - offset));
759    } else {
760      let nlogs = count;
761      let mut offset = if offset != 0.0 {
762        offset.log10()
763      } else {
764        offset
765      };
766      offset -= offset.trunc();
767      let lsize = rr / nlogs;
768      if !(lsize.is_finite() && offset.is_finite()) {
769        return false;
770      }
771      // `0 1 nlogs {…} for`, eight minor marks a decade
772      let Some(n) = ps_for_count(0.0, nlogs).filter(|n| drawable_count(*n as f64 * 9.0)) else {
773        return false;
774      };
775      for i in 0..n {
776        let k = i as f64;
777        if k >= offset {
778          major.push((k - offset) * lsize);
779        }
780        for j in 2..=9 {
781          let xx = f64::from(j).log10() + k;
782          if xx >= offset {
783            minor.push((xx - offset) * lsize);
784          }
785        }
786      }
787    }
788    self.line_width(f, &[(0.0, 0.0), (rr, 0.0)], None, width);
789    for (marks, length, w) in [(major, 1.2, width), (minor, 0.8, width * 0.6)] {
790      for v in marks.into_iter().filter(|v| *v >= 0.0 && *v <= rr) {
791        self.line_width(f, &[(v, 0.0), (v, hashsize * length)], None, w);
792      }
793    }
794    true
795  }

797  /// Evaluate one v1 procedure on its `\special` arguments; `false` for an
798  /// unknown procedure or a malformed argument list.
799  fn draw(&mut self, procedure: &str, args: &[String]) -> bool {
800    let n = |i: usize| -> Option<f64> { args.get(i).and_then(|a| ps_value(a)) };
801    let nums = |count: usize| -> Option<Vec<f64>> {
802      if args.len() < count {
803        return None;
804      }
805      (0..count).map(n).collect()
806    };
807    let color = |i: usize| -> String { axo_color(args.get(i).map(String::as_str).unwrap_or("")) };
808    let ink = self.pen.color.clone();
809    match procedure {
810      "line" | "dashline" | "arrowline" | "longarrow" | "dasharrowline" => {
811        let dashed = procedure.starts_with("dash");
812        let Some(v) = nums(if dashed { 5 } else { 4 }) else {
813          return false;
814        };
815        let (p1, p2) = (self.at(v[0], v[1]), self.at(v[2], v[3]));
816        let (f, dr) = self.segment_frame(p1, p2);
817        match procedure {
818          "line" => self.line(self.base, &[p1, p2], None),
819          "dashline" => self.line(self.base, &[p1, p2], Some(fit_dash(dr, v[4]))),
820          "arrowline" => {
821            self.line(self.base, &[p1, p2], None);
822            self.arrow_right(f.translate(dr / 2.0, 0.0));
823          },
824          "longarrow" => {
825            let w = self.pen.width;
826            self.line(f, &[(0.0, 0.0), (dr - w * 3.0, 0.0)], None);
827            self.arrow_right(f.translate(dr - w * 1.5, 0.0));
828          },
829          _ => {
830            let half = dr / 2.0;
831            let dash = Some(fit_dash(half, v[4]));
832            self.line(f, &[(0.0, 0.0), (half, 0.0)], dash);
833            let g = f.translate(half, 0.0);
834            self.line(g, &[(0.0, 0.0), (half, 0.0)], dash);
835            self.arrow_right(g);
836          },
837        }
838      },
839      "gluon" | "photon" | "zigzag" => {
840        let Some(v) = nums(6) else { return false };
841        if !drawable_count(v[5] * 2.0) {
842          return false;
843        }
844        let (p1, p2) = (self.at(v[0], v[1]), self.at(v[2], v[3]));
845        if procedure == "gluon" {
846          self.gluon(p1, p2, v[4], v[5]);
847        } else {
848          self.photon(p1, p2, v[4], v[5], procedure == "zigzag");
849        }
850      },
851      "photonarc" | "gluearc" => {
852        let Some(v) = nums(7) else { return false };
853        if !drawable_count(v[6] * 2.0) {
854          return false;
855        }
856        let c = self.at(v[0], v[1]);
857        if procedure == "photonarc" {
858          self.photon_arc(c, v[2], v[3], v[4], v[5], v[6]);
859        } else {
860          self.gluon_arc(c, v[2], v[3], v[4], v[5], v[6]);
861        }
862      },
863      "carc" | "dashcarc" | "arrowarc" | "longarrowarc" | "dasharrowarc" | "arrowarcn"
864      | "longarrowarcn" | "dasharrowarcn" => {
865        let dashed = procedure.starts_with("dash");
866        let Some(v) = nums(if dashed { 6 } else { 5 }) else {
867          return false;
868        };
869        let c = self.at(v[0], v[1]);
870        let f = self.base.translate(c.0, c.1);
871        let (radius, a1, a2) = (v[2], v[3], v[4]);
872        // `/dashcarc`: the arc length, its sweep raised by 360 once.
873        let dash_for = |from: f64, to: f64| -> Option<f64> {
874          let sweep = if to - from < 0.0 {
875            to - from + 360.0
876          } else {
877            to - from
878          };
879          Some(fit_dash(sweep * PI / 180.0 * radius, v[5]))
880        };
881        // `540 width 1 add mul 3.14159 div radius div`: the angle the
882        // arrowhead of a `\LongArrowArc` takes off the arc.
883        let arcmid = 540.0 * (self.pen.width + 1.0) / PI / radius;
884        match procedure {
885          "carc" => self.arc(f, radius, a1, a2, None),
886          "dashcarc" => self.arc(f, radius, a1, a2, dash_for(a1, a2)),
887          "arrowarc" => {
888            self.arc(f, radius, a1, a2, None);
889            let a2 = if a1 > a2 { a2 + 360.0 } else { a2 };
890            self.arrow_vertical(f.rotate((a1 + a2) / 2.0).translate(radius, 0.0), true);
891          },
892          "longarrowarc" => {
893            let a2 = if a1 > a2 { a2 + 360.0 } else { a2 };
894            self.arc(f, radius, a1, a2 - arcmid, None);
895            self.arrow_vertical(f.rotate(a2 - arcmid / 2.0).translate(radius, 0.0), true);
896          },
897          "dasharrowarc" => {
898            let a2 = if a2 < a1 { a2 + 360.0 } else { a2 };
899            let mid = (a1 + a2) / 2.0;
900            self.arc(f, radius, a1, mid, dash_for(a1, mid));
901            self.arc(f, radius, mid, a2, dash_for(mid, a2));
902            self.arrow_vertical(f.rotate(mid).translate(radius, 0.0), true);
903          },
904          "arrowarcn" => {
905            self.arcn(f, radius, a1, a2, None);
906            let a1 = if a1 < a2 { a1 + 360.0 } else { a1 };
907            self.arrow_vertical(f.rotate((a1 + a2) / 2.0).translate(radius, 0.0), false);
908          },
909          "longarrowarcn" => {
910            let a1 = if a1 < a2 { a1 + 360.0 } else { a1 };
911            self.arcn(f, radius, a1, a2 + arcmid, None);
912            self.arrow_vertical(f.rotate(a2 + arcmid / 2.0).translate(radius, 0.0), false);
913          },
914          _ => {
915            let a1 = if a1 < a2 { a1 + 360.0 } else { a1 };
916            let mid = (a1 + a2) / 2.0;
917            self.arc(f, radius, mid, a1, dash_for(mid, a1));
918            self.arc(f, radius, a2, mid, dash_for(a2, mid));
919            self.arrow_vertical(f.rotate(mid).translate(radius, 0.0), false);
920          },
921        }
922      },
923      "vertex" | "bcirc" | "gcirc" | "ccirc" => {
924        let Some(v) = nums(3) else { return false };
925        let c = self.at(v[0], v[1]);
926        let f = self.base.translate(c.0, c.1);
927        match procedure {
928          "vertex" => self.circle(f, v[2], Some(&ink), None),
929          "bcirc" => self.circle(f, v[2], Some(WHITE), Some(&ink)),
930          "gcirc" => {
931            let Some(g) = n(3) else { return false };
932            self.circle(f, v[2], Some(&gray(g)), Some(BLACK));
933          },
934          _ => self.circle(f, v[2], Some(&color(4)), Some(&color(3))),
935        }
936      },
937      "box" | "bbox" | "gbox" | "cbox" | "boxc" | "bboxc" | "gboxc" | "cboxc" => {
938        let Some(v) = nums(4) else { return false };
939        let (p1, p2) = if procedure.ends_with("boxc") {
940          // `\Boxc(x,y)(w,h)`: centred on (x,y).
941          let (hw, hh) = (v[2] / 2.0, v[3] / 2.0);
942          (self.at(v[0] - hw, v[1] - hh), self.at(v[0] + hw, v[1] + hh))
943        } else {
944          (self.at(v[0], v[1]), self.at(v[2], v[3]))
945        };
946        match procedure.trim_end_matches('c') {
947          "box" => self.rect(p1, p2, None, Some(&ink)),
948          "bbox" => self.rect(p1, p2, Some(WHITE), Some(&ink)),
949          "gbox" => {
950            let Some(g) = n(4) else { return false };
951            self.rect(p1, p2, Some(&gray(g)), Some(BLACK));
952          },
953          _ => self.rect(p1, p2, Some(&color(5)), Some(&color(4))),
954        }
955      },
956      "triangle" | "btriangle" | "gtriangle" | "ctriangle" => {
957        let Some(v) = nums(6) else { return false };
958        let points = [
959          self.at(v[0], v[1]),
960          self.at(v[2], v[3]),
961          self.at(v[4], v[5]),
962        ];
963        let base = self.base;
964        match procedure {
965          "triangle" => self.polygon(base, &points, None, Some(&ink)),
966          "btriangle" => self.polygon(base, &points, Some(WHITE), Some(&ink)),
967          "gtriangle" => {
968            let Some(g) = n(6) else { return false };
969            self.polygon(base, &points, Some(&gray(g)), Some(BLACK));
970          },
971          _ => self.polygon(base, &points, Some(&color(7)), Some(&color(6))),
972        }
973      },
974      "oval" | "goval" | "coval" => {
975        let Some(v) = nums(5) else { return false };
976        let c = self.at(v[0], v[1]);
977        match procedure {
978          "oval" => self.ellipse(c, v[2], v[3], v[4], None, Some(&ink)),
979          "goval" => {
980            let Some(g) = n(5) else { return false };
981            self.ellipse(c, v[2], v[3], v[4], Some(&gray(g)), Some(BLACK));
982          },
983          _ => self.ellipse(c, v[2], v[3], v[4], Some(&color(6)), Some(&color(5))),
984        }
985      },
986      "curve" | "dashcurve" => {
987        // `\Curve{(x1,y1)(x2,y2)…}` (v1 `\kromme` reads the pairs): the
988        // numbers in order, the dash size first for `\DashCurve`.
989        let joined = args.join(",");
990        let Some(v): Option<Vec<f64>> = joined
991          .split([',', '(', ')'])
992          .map(str::trim)
993          .filter(|s| !s.is_empty())
994          .map(ps_value)
995          .collect()
996        else {
997          return false;
998        };
999        let (dash, coords) = if procedure == "dashcurve" {
1000          match v.split_first() {
1001            Some((d, rest)) => (Some(*d), rest),
1002            None => return false,
1003          }
1004        } else {
1005          (None, &v[..])
1006        };
1007        let points: Vec<Pt> = coords
1008          .as_chunks::<2>()
1009          .0
1010          .iter()
1011          .map(|[x, y]| self.at(*x, *y))
1012          .collect();
1013        if points.len() == 2 {
1014          self.line(self.base, &points, None);
1015        }
1016        let segments = Drawing::smooth_curve(&points);
1017        match dash {
1018          // `/centerdash`: `[ddsize]` with `ddsize` fitting the path length
1019          // in whole periods. The curve is flattened into one polyline so
1020          // the pattern runs continuously along it.
1021          Some(dsize) => {
1022            let mut flat = Vec::new();
1023            for seg in &segments {
1024              for k in 0..=16 {
1025                let t = f64::from(k) / 16.0;
1026                let u = 1.0 - t;
1027                let b = |i: usize| -> f64 {
1028                  let (p0, p1, p2, p3) = (seg[0], seg[1], seg[2], seg[3]);
1029                  let pick = |p: Pt| if i == 0 { p.0 } else { p.1 };
1030                  u * u * u * pick(p0)
1031                    + 3.0 * u * u * t * pick(p1)
1032                    + 3.0 * u * t * t * pick(p2)
1033                    + t * t * t * pick(p3)
1034                };
1035                if k > 0 || flat.is_empty() {
1036                  flat.push((b(0), b(1)));
1037                }
1038              }
1039            }
1040            let length: f64 = flat
1041              .windows(2)
1042              .map(|w| ((w[1].0 - w[0].0).powi(2) + (w[1].1 - w[0].1).powi(2)).sqrt())
1043              .sum();
1044            // v1 `cvi`s the period count; one shorter than a period would
1045            // divide by zero (a dvips error): one period.
1046            let periods = (length / dsize / 2.0).trunc().max(1.0);
1047            let ddsize = length / (periods * 2.0);
1048            if flat.len() > 1 {
1049              self.line(self.base, &flat, Some(ddsize));
1050            }
1051          },
1052          None => {
1053            for seg in segments {
1054              self.curve(self.base, seg);
1055            }
1056          },
1057        }
1058      },
1059      "logaxis" | "linaxis" => {
1060        let linear = procedure == "linaxis";
1061        let Some(v) = nums(if linear { 9 } else { 8 }) else {
1062          return false;
1063        };
1064        let (p1, p2) = (self.at(v[0], v[1]), self.at(v[2], v[3]));
1065        if linear {
1066          return self.axis(p1, p2, v[4], v[5], v[6], v[7], v[8], true);
1067        }
1068        return self.axis(p1, p2, v[4], 0.0, v[5], v[6], v[7], false);
1069      },
1070      "btext" | "gtext" | "ctext" | "b2text" | "g2text" | "c2text" => {
1071        // `x,y[,gray | ,color,background],textwidth`: the box `/btext` and
1072        // `/b2text` fit around the text — `fsize` wider than it, `4/3·fsize`
1073        // (two lines: `7/3·fsize`) high; the text is set by `\lx@axo@labels`.
1074        let Some(hsize) = args.last().and_then(|a| ps_value(a)) else {
1075          return false;
1076        };
1077        let Some(v) = nums(2) else { return false };
1078        let fsize = self.pen.fsize;
1079        let half_w = (hsize + fsize) / 2.0;
1080        let half_h = if procedure.contains('2') {
1081          fsize * 7.0 / 6.0
1082        } else {
1083          fsize * 2.0 / 3.0
1084        };
1085        let (p1, p2) = (
1086          self.at(v[0] - half_w, v[1] - half_h),
1087          self.at(v[0] + half_w, v[1] + half_h),
1088        );
1089        match procedure.chars().next() {
1090          Some('b') => self.rect(p1, p2, Some(WHITE), Some(&ink)),
1091          Some('g') => {
1092            let Some(g) = n(2) else { return false };
1093            self.rect(p1, p2, Some(&gray(g)), Some(BLACK));
1094          },
1095          _ => self.rect(p1, p2, Some(&color(3)), Some(&color(2))),
1096        }
1097      },
1098      _ => return false,
1099    }
1100    true
1101  }

1114/// A point of the graphics (user, scaled) space as `\put` coordinates: v1's
1115/// `\put(\axoxoff,\axoyoff)` origin, in `\unitlength`s, plus the point in
1116/// points.
1117fn graphics_put(x: f64, y: f64) -> Result<(f64, f64)> {
1118  let pen = Pen::current()?;
1119  let unit = unitlength_sp()? as f64 / UNITY_F64;
1120  let unit = if unit == 0.0 { 1.0 } else { unit };
1121  let to_units = |v: f64, offset: f64| (v + offset) * pen.scale / unit;
1122  Ok((
1123    axo_number("\\axoxoff")? + to_units(x, pen.offset.0),
1124    axo_number("\\axoyoff")? + to_units(y, pen.offset.1),
1125  ))
1126}

1128/// `\put(X,Y){\makebox(0,0)[pos]{<text at size>}}`, rotated by `angle`: a
1129/// PostScript-font text of v1 (`/ptext`, `/btext`) set in TeX at its scaled
1130/// size, as axodraw2 does (axodraw2.sty:2885-2900 `\PText`).
1131fn ps_label(x: f64, y: f64, angle: &str, pos: &str, text: Tokens) -> Result<Tokens> {
1132  let (px_, py_) = graphics_put(x, y)?;
1133  let pen = Pen::current()?;
1134  let size = pen.fsize * pen.scale;
1135  let mut body = mouth::tokenize_internal(TeXString::assembled(format!(
1136    "\\makebox(0,0)[{pos}]{{\\fontsize{{{size}}}{{{size}}}\\selectfont "
1137  )))
1138  .unlist();
1139  body.extend(text.unlist());
1140  body.push(T_END!());
1141  let angle = angle.trim();
1142  let mut out =
1143    mouth::tokenize_internal(TeXString::assembled(format!("\\put({px_},{py_}){{"))).unlist();
1144  if angle.is_empty() || angle.parse::<f64>().ok() == Some(0.0) {
1145    out.extend(body);
1146  } else {
1147    out.extend(
1148      mouth::tokenize_internal(TeXString::assembled(format!("\\rotatebox{{{angle}}}{{"))).unlist(),
1149    );
1150    out.extend(body);
1151    out.push(T_END!());
1152  }
1153  out.push(T_END!());
1154  Ok(Tokens::new(out))
1155}
