2028/// The TeX that sets `\Hy@driver` as hyperref.sty's `\Hy@AtEndOfPackage` (:4246-4407) leaves it. In PDF mode (`\ifpdf`,
2029/// which follows `\pdfoutput`) the driver is forced to `hpdftex` (`hluatex` under the `luatex` profile, hyperref's
2030/// `\pdfextension` test; :4251-4265), whatever option was given; under XeTeX to `hxetex` (:4280-4292). In DVI mode the
2031/// last driver option given wins, kvoptions' `\ProcessKeyvalOptions{Hyp}` (:4066) reading the class's global options
2032/// before the package's own; with none, `\Hy@defaultdriver` = `hdvips` (:4339, :4371).
2033fn hyperref_driver_tex() -> String {
2034  let mut given: Vec<String> = Vec::new();
2035  for key in ["class_options", "opt@hyperref.sty"] {
2036    for item in lookup_vecdeque(key).unwrap_or_default() {
2037      match item {
2038        Stored::String(s) => given.push(with(s, |s| s.trim().to_string())),
2039        Stored::Strings(ss) => given.extend(ss.iter().map(|s| with(*s, |s| s.trim().to_string()))),
2040        _ => {},
2041      }
2042    }
2043  }
2044  let dvi_driver = given
2045    .iter()
2046    .rev()
2047    .find_map(|option| {
2048      HYPERREF_DVI_DRIVER_OPTIONS
2049        .iter()
2050        .find(|(name, _)| name == option)
2051        .map(|(_, driver)| *driver)
2052    })
2053    .unwrap_or("hdvips");
2054  s!(
2055    "\\ifpdf\\ifluatex\\def\\Hy@driver{{hluatex}}\\else\\def\\Hy@driver{{hpdftex}}\\fi\\else\\ifxetex\\def\\Hy@driver{{hxetex}}\\else\\def\\Hy@driver{{{dvi_driver}}}\\fi\\fi"
2056  )

2186/// The PDF string escapes a document writes in its text: `\ddd` the PDFDocEncoding code (`\374` ü), and hyperref's
2187/// UTF-16 forms, `\8d\ddd` and `\9ddd\ddd` (puenc.def `\81\122` Œ). Anything else stays as it is written.
2188fn decode_pdf_escapes(text: &str) -> String {
2189  if !text.contains('\\') {
2190    return text.to_string();
2191  }
2192  let chars: Vec<char> = text.chars().collect();
2193  let octal = |from: usize, count: usize| -> Option<u32> {
2194    let digits = chars.get(from..from + count)?;
2195    digits
2196      .iter()
2197      .try_fold(0, |value, digit| Some(value * 8 + digit.to_digit(8)?))
2198  };
2199  let low_byte =
2200    |from: usize| -> Option<u32> { (chars.get(from) == Some(&'\\')).then(|| octal(from + 1, 3))? };
2201  let mut out = String::with_capacity(text.len());
2202  let mut i = 0;
2203  while i < chars.len() {
2204    if chars[i] == '\\' {
2205      let decoded = match chars.get(i + 1) {
2206        Some('8') => octal(i + 2, 1)
2207          .zip(low_byte(i + 3))
2208          .map(|(high, low)| (char::from_u32(high << 8 | low), 7)),
2209        Some('9') => octal(i + 2, 3)
2210          .zip(low_byte(i + 5))
2211          .map(|(high, low)| (char::from_u32(high << 8 | low), 9)),
2212        _ => octal(i + 1, 3).filter(|code| *code <= 0o377).map(|code| {
2213          let glyph = match code {
2214            0x80..=0xA0 => Some(PDF_DOC_0X80[code as usize - 0x80]),
2215            _ => char::from_u32(code),
2216          };
2217          (glyph, 4)
2218        }),
2219      };
2220      if let Some((Some(glyph), width)) = decoded {
2221        out.push(glyph);
2222        i += width;
2223        continue;
2224      }
2225    }
2226    out.push(chars[i]);
2227    i += 1;
2228  }
2229  out
2230}

2240/// The body of `\hyperdef`/`\hypertarget`: anchor the construct's own `text`, never the words before it.
2241/// hyperref's default `\Hy@nestingfalse` (hyperref.sty:323) is a point destination, then the text
2242/// (`\hyper@@anchor{#1}{\relax}#2`, :4805-4810); case 2's wrapping is LaTeXML's convention, the
2243/// nesting-on shape (:4834-4845), kept where the anchor may hold the text.
2244///   1. Guard 1 (see [`localized_anchor`]): empty text is a pure hyperlink destination with nothing
2245///      to wrap — localizing would only capture unrelated surrounding content (or the open note of
2246///      the "linked footnote" idiom, OXIDIZED_DESIGN #104) — so a bare anchor at the insertion point.
2247///   2. Where the insertion point admits `ltx:anchor` (running text) and the text is all horizontal
2248///      material, the anchor is inserted there with the text as its content.
2249///   3. Where it admits the anchor but the text holds non-horizontal material — a display
2250///      (`A \hypertarget{d}{\[x=1\]} b.`), a list, a tabular, a footnote, a `\par` — the anchor is a
2251///      point destination before the text, hyperref's order, since such material cannot sit inside
2252///      `ltx:anchor`. Wrapping a display's `ltx:Math` put the anchor inside `ltx:equation`
2253///      (philexmanual, schema-invalid; Perl the same).
2254///   4. Otherwise (vertical context: `\hypertarget{x}{\section{…}}`, a `\parbox`, an `\item`, a
2255///      `quote`, a `p{}` cell) the text is absorbed and the anchor localized by Perl's walk, which lands
2256///      it in the section title; for a block (a display, a table, a figure's content) it finds nothing
2257///      wrappable and the destination goes in the paragraph after the block — one block late.

2260  // an empty name): an `ltx:anchor` without an id is no destination.
2261  if id.is_empty() {
2262    if let Some(text) = text {
2263      document.absorb(text, None)?;
2264    }
2265    return Ok(());
2266  }
2267  let text = match text {
2268    Some(text) if !text.is_empty()? => text,
2269    _ => return insert_bare_anchor(document, id),
2270  };
2271  // The element at the insertion point: mid-paragraph the insertion point is the running text
2272  // node, whose `ltx:p` is what must admit the anchor.
2273  let admits_anchor = document
2274    .get_element()
2275    .is_some_and(|element| can_contain_qsym(get_node_qname(&element), pin!("ltx:anchor")));
2276  if admits_anchor && is_horizontal_material(text) {
2277    document.insert_element("ltx:anchor", vec![text], Some(string_map!("xml:id" => id)))?;
2278    Ok(())
2279  } else if admits_anchor {
2280    insert_bare_anchor(document, id)?;
2281    document.absorb(text, None)?;
2282    Ok(())
2283  } else {
2284    document.absorb(text, None)?;

2286  }

2305// Perl: hyperref.sty.ltxml `localized_anchor`. DFS walks the current node's
2306// subtree and wraps the first descendant that ltx:anchor is allowed to
2307// contain. The traversal mirrors Perl's pop+unshift order: candidates are
2308// popped from the back and child nodes are unshifted to the front, so we
2309// visit the rightmost element of each level before descending.
2310//
2311// Surpass-Perl (OXIDIZED_DESIGN #104, issue #526): the base Perl walk both
2312// (a) has no content-empty short-circuit and (b) will wrap a still-OPEN node,
2313// so an empty `\hypertarget{id}{}` at the head of a floating `ltx:note` (the
2314// "linked footnote" idiom, `\footnotetext{\hypertarget{..}{}..}`) selects and
2315// prematurely closes the open note, emptying it and orphaning the footnote text
2316// (`malformed:ltx:anchor`/`malformed:ltx:note`). Perl fails identically. Two
2317// general guards fix the whole class — no per-constructor special-case:
2318//   1. empty localizable content ⇒ a pure destination, emit a bare anchor (checked by the
2319//      caller, `anchor_own_text`, before any walk);
2320//   2. never wrap an open node; if nothing wrappable is found, emit a bare anchor.
2321//   3. wrap a node only where its parent may hold the `ltx:anchor` that replaces it — the condition
2322//      Perl's `wrapNodes` (Document.pm:1972-1995) assumes without checking: an `ltx:Math` is
2323//      anchor content, but its `ltx:equation` holds no anchor (philexmanual's
2324//      `\parbox{…}{\centering\hypertarget{compo}{\[…\]}}`, philex.sty:136).
2325fn localized_anchor(document: &mut Document, id: &str) -> CoreResult<()> {
2326  let parent_admits_anchor = |node: &Node| {
2327    node
2328      .get_parent()
2329      .is_some_and(|parent| can_contain_qsym(get_node_qname(&parent), pin!("ltx:anchor")))
2330  };
2331  let mut candidates: Vec<Node> = vec![document.get_node().clone()];
2332  let mut found: Option<Node> = None;
2333  while let Some(candidate) = candidates.pop() {
2334    match candidate.get_type() {
2335      Some(NodeType::ElementNode) => {
2336        let qname = get_node_qname(&candidate);
2337        // Guard 2. Wrap a candidate only when ltx:anchor may legally contain it
2338        // AND it is not still open. Wrapping an OPEN node (e.g. a floating
2339        // ltx:note whose body is mid-digestion) would close it prematurely and
2340        // orphan the rest of its content.
2341        if can_contain_qsym(pin!("ltx:anchor"), qname) && !document.is_open(&candidate) {
2342          if parent_admits_anchor(&candidate) {
2343            found = Some(candidate);
2344            break;
2345          }
2346          // Guard 3: anchor content whose parent holds no anchor — a display's `ltx:Math`, an
2347          // `ltx:graphics` or `ltx:tabular` in a block — is one unit. Its insides (math `XMText`, table
2348          // cells) are no place for its destination: the math pass renames an id inside `XMText`, and a
2349          // cell is not the table.
2350          continue;
2351        }
2352        // Generated material is never a destination: the numbering (`ltx:tags` — a figure's `typerefnum`
2353        // tag would hide it) and a math fork's presentation alternative (`ltx:MathBranch`, an `align`'s
2354        // cells: the destination would land in its last cell).
2355        if qname == pin!("ltx:tags") || qname == pin!("ltx:MathBranch") {
2356          continue;
2357        }
2358        // Perl: unshift(@candidates, $candidate->childNodes); pushes the child
2359        // list to the front, so the rightmost child is popped next.
2360        let children = candidate.get_child_nodes();
2361        for child in children.into_iter().rev() {
2362          candidates.insert(0, child);
2363        }
2364      },
2365      // Perl: any non-element node short-circuits; the candidate (text node)
2366      // is then wrapped, which works because anchor.model = Inline allows text —
2367      // where its parent may hold the anchor (guard 3).
2368      _ if parent_admits_anchor(&candidate) => {
2369        found = Some(candidate);
2370        break;
2371      },
2372      _ => {},
2373    }
2374  }
2375  // Wrap the located target in an anchor; if the walk found nothing wrappable
2376  // (or wrap_nodes declines it), drop a bare destination anchor instead — one
2377  // fallback for both "no target" and "target refused".
2378  let anchor = match found {
2379    Some(target) => document.wrap_nodes("ltx:anchor", vec![target])?,
2380    None => None,
2381  };
2382  match anchor {
2383    Some(mut anchor) => {
2384      document.set_attribute(&mut anchor, "xml:id", id)?;
2385      if document.is_open(&anchor) {
2386        document.close_node(&anchor)?;
2387      }
2388    },
2389    None => insert_bare_anchor(document, id)?,
2390  }
2391  Ok(())
2392}

2507    // genuinely-leftover control sequences (e.g. `\def`) as CS tokens, which the
2508    // filter below stringifies.
2509    DefMacro!(T_CS!("\\%"),              None, T_OTHER!("%"), scope => Some(Scope::Local));
2510    DefMacro!(T_CS!("\\#"),              None, T_OTHER!("#"), scope => Some(Scope::Local));
2511    DefMacro!(T_CS!("\\&"),              None, T_OTHER!("&"), scope => Some(Scope::Local));
2512    DefMacro!(T_CS!("\\textunderscore"), None, T_OTHER!("_"), scope => Some(Scope::Local));
2513    let_i(&T_CS!("\\_"), &T_CS!("\\textunderscore"), None);
2514    DefMacro!(T_CS!("\\hyper@tilde"),    None, T_OTHER!("~"), scope => Some(Scope::Local));
2515    let_i(&T_CS!("\\~"), &T_CS!("\\hyper@tilde"), None);
2516    let_i(&T_CS!("\\textasciitilde"), &T_CS!("\\hyper@tilde"), None);
2517    DefMacro!(T_CS!("\\^"), None, T_OTHER!("^"), scope => Some(Scope::Local));
2518    let_i(&T_CS!("\\textasciicircum"), &T_CS!("\\^"), None);
2519    let_i(&T_CS!("\\\\"), &T_CS!("\\@backslashchar"), None);
2520    DefMacro!(T_CS!("\\textbackslash"), None, T_OTHER!("\\"), scope => Some(Scope::Local));
2521    let (open,close,url) = if open.get_catcode() == Catcode::BEGIN {
2522      ( T_OTHER!("{"), T_OTHER!("}"),
2523        read_balanced(ExpansionLevel::Partial,false,false)?.unwrap_or_default()) // Expand as we go!
2524    } else {
2525      ( T_OTHER!("{"), T_OTHER!("}"),
2526        Tokens!(open.as_other()) )
2527    };
2528    end_semiverbatim()?;
2529    let toks : Vec<Token> = url.unlist().into_iter()
2530      .filter(|t| t.get_catcode() != Catcode::SPACE)
2531      // url.sty `\meaning`-stringifies leftover control sequences: recatcode any
2532      // surviving CS (e.g. `\def`) to `other` so it is LITERAL href text, not
2533      // digested/executed (#723). The escapes above already resolved during the
2534      // read. (Perl "let CS's through!" to digestion — the divergence we surpass.)
2535      .map(|t| t.as_other())
2536      .collect();
2537    let mut url_wrapped = vec![T_CS!("\\UrlFont"), T_CS!("\\UrlLeft")];
2538    url_wrapped.extend(toks.clone());
2539    url_wrapped.push(T_CS!("\\UrlRight"));
2540    let mut invocation_tokens = Invocation!(T_CS!("\\lx@hyper@url@"),vec![
2541        Tokens!(cmd.as_other()),
2542        Tokens!(open),
2543        Tokens!(close),
2544        Tokens::new(toks),
2545        Tokens::new(url_wrapped)]).unlist();
2546    invocation_tokens.push(T_CS!("\\endgroup"));
2547    Tokens::new(invocation_tokens)
2548  });
2549
2550  // RE-define from url w. Robust, as hyperref.sty:4801
2551  // `\DeclareRobustCommand*{\url}` (see url_sty.rs).
2552  if url {
2553    DefMacro!("\\url", "\\begingroup\\lx@hyper@url\\url", locked => true, protected => true);
2554  }
2555
2556  // Perl hyperref.sty.ltxml L187-194: bounded + enterHorizontal both
2557  // present. enter_horizontal => true was missing in the Rust port —
2558  // a `\url{...}` between paragraphs at top level opened <ltx:ref>
2559  // outside any <ltx:p>, producing invalid block-level structure.
2560  DefConstructor!("\\lx@hyper@url@ Undigested {}{} Semiverbatim {}",// Allow this to work in Math!
2561    "?#isMath(<ltx:XMWrap class='#class' href='#href'>#5</ltx:XMWrap>)(<ltx:ref href='#href' class='#class'>#5</ltx:ref>)",
2562    bounded   => true, enter_horizontal => true,
2563    properties => sub[args] {
2564      unref!(args => cmd, _open, _close, url, _formattedurl);
2565      let ltx_cmd = s!("ltx_{}", LEADING_BACKSLASH_RE.replace(&cmd.to_string(),""));
2566      Ok(stored_map!(
2567        "href" => compose_url(&lookup_string("BASE_URL"), &url.to_string(), None),
2568        "class"=> ltx_cmd
2569      ))
2570    },
2571    sizer     => "#5",
2572    reversion => "#1#2#4#3");
2573  Ok(())
2574}

2576/// What `\autoref` prints for a label only another document holds, from the `\r@<label>` xr imported
2577/// (`record_external_label`): hyperref's name of the anchor's type, `\<type>autorefname` or `\<type>name` (a trailing
2578/// `*` stripped next), before an active `~`, then the label's text and `\null` (hyperref.sty:8211-8278
2579/// `\HyRef@autosetref`, `\HyRef@testreftype`, the type being the anchor up to its first `.`: `theorem.3`, `figure.caption.1`).
2580/// A name that is no name noun is not run, as at a target of this document (`\lx@autorefnum@@`); a label with no anchor
2581/// (the other document without hyperref) prints its text alone, as pdflatex does with "No autoref name". Witness
2582/// 2402.05535 (`\autoref*` to its `srds-` labels).
2583fn external_autoref_text(label: &str) -> Result<Option<Digested>> {
2584  let Some(text) = external_label_tex(label, "label") else {
2585    return Ok(None);
2586  };
2587  // (`\null`, the empty box after the number, prints nothing: `\relax` here, so it leaves no empty element, and it still
2588  // ends a name's `~#1\null` argument, a delimiter being matched by the token)
2589  let mut tokens = vec![
2590    T_CS!("\\begingroup"),
2591    T_CS!("\\let"),
2592    T_CS!("\\null"),
2593    T_CS!("\\relax"),
2594  ];
2595  let anchor = external_label_field(label, "anchor").unwrap_or_default();
2596  let ty = anchor.split('.').next().unwrap_or_default();
2597  if !ty.is_empty() {
2598    let unstarred = ty.strip_suffix('*').unwrap_or(ty);
2599    let name = [
2600      s!("\\{ty}autorefname"),
2601      s!("\\{ty}name"),
2602      s!("\\{unstarred}autorefname"),
2603      s!("\\{unstarred}name"),
2604    ]
2605    .into_iter()
2606    .find(|cs| is_defined(cs));
2607    match name {
2608      Some(cs) if cs.ends_with("autorefname") || is_name_noun_cs(&cs) => {
2609        tokens.extend([T_CS!(cs), T_ACTIVE!('~')])
2610      },
2611      Some(_) => tokens.push(T_ACTIVE!('~')),
2612      None => {},
2613    }
2614  }
2615  tokens.extend(text.unlist());
2616  tokens.extend([T_CS!("\\null"), T_CS!("\\endgroup")]);
2617  Ok(Some(digest(Tokens::new(tokens))?))
2618}
