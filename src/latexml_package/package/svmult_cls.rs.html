434/// The Greek letter the Symbol font (Adobe Symbol encoding, `\usefont{U}{psy}{m}{n}`) has at the ASCII letter
435/// `letter`, if it is one.
436fn symbol_font_greek(letter: &str) -> Option<char> {
437  let mut chars = letter.chars();
438  let (Some(c), None) = (chars.next(), chars.next()) else {
439    return None;
440  };
441  const LOWER: &str = "αβχδεφγηιϕκλμνοπθρστυϖωξψζ";
442  const UPPER: &str = "ΑΒΧΔΕΦΓΗΙϑΚΛΜΝΟΠΘΡΣΤΥςΩΞΨΖ";
443  match c {
444    'a'..='z' => LOWER.chars().nth(c as usize - 'a' as usize),
445    'A'..='Z' => UPPER.chars().nth(c as usize - 'A' as usize),
446    _ => None,
447  }
