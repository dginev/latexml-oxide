5  // \DeclareMathOperator*{cs}{text} — Perl L23-29: `DefMathI($cs, undef,
6  // Invocation(T_CS('\operatorname'), $star, $text), …)`, a token presentation, so the DefMath is
7  // the wrapped kind (Package.pm:1656-1660, `def_math_wrapped`): `\cs` expands to
8  // `\cs@wrapper{\cs@presentation}` and the body stays the tokens as written — unexpanded until use,
9  // with their catcodes — inside `\operatorname`'s upright operator font. The name is the control
10  // sequence's (it differs from the presentation, Package.pm:1628-1632), the wrapper's `XMWrap`
11  // carries it with the role and scriptpos onto the parsed operator, and the formula's `tex=` reads
12  // `\operatorname{…}` (the wrapper reverts as its argument). An earlier string round trip had to
13  // untex `{\rm Aut}` (0806.2705, 0808.0535), expand expl3 names at definition (numerica.sty:50-51)
14  // under `\protected@edef` (pm-isomath.sty:185, euclideangeometry-man) and keep a body naming an
15  // undefined control sequence verbatim (iidef.sty:147, thucoursework ithw); the tokens need none
16  // of it. Guards `perfect_kernel_batch56::{declaremathoperator_keeps_protected_macros,
17  // declaremathoperator_body_stays_lazy, declaremathoperator_keeps_the_following_letters_italic}`;
18  // golden tests/parse/declared_operators.tex.
19  // The name is the argument's first token; what follows it is the parameter text TeX gives the command
20  // (amsopn.sty:50-53 `\@declmathop` → `\DeclareRobustCommand{#2}`, whose `\def` reads it: `\DeclareMathOperator{\N^1}{…}`
21  // makes `\N^1` the operator, 2301.04262), not text after the command (Perl `{Token}`: a Warning, and its `^` a script
22  // outside math; KNOWN_PERL_ERRORS #891). Guard `perfect_kernel_batch69a::declaremathoperator_parameter_text`.
23  DefPrimitive!("\\DeclareMathOperator OptionalMatch:* {} {}", sub[(star, name, text)] {
24    let mut name: Vec<Token> =
25      name.unlist().into_iter().skip_while(|t| t.get_catcode() == Catcode::SPACE).collect();
26    if name.is_empty() {
27      return Ok(Vec::new());
28    }
29    let cs = name.remove(0);
30    // …but only for a command already defined (and not `\relax`): `\declare@robustcommand`'s
31    // `\ifx\@undefined#1\else\ifx\relax#1\else … \fi\fi` (latex.ltx:1386-1389) typesets the parameter text of an
32    // undefined one (`\DeclareMathOperator{\dash_exp}{-exp}`, math0312019; pdflatex errs and leaves the command
33    // undefined), so there it is read after the command, as Perl's `{Token}` reads it: the Warning, then the script's
34    // Error (KNOWN_PERL_ERRORS #891). Guard `perfect_kernel_batch69a::declaremathoperator_undefined_parameter_text`.
35    if !name.is_empty()
36      && (lookup_meaning(&cs).is_none() || x_equals(&cs, &T_CS!("\\relax")))
37    {
38      if let Some(first) = name.iter().find(|t| t.get_catcode() != Catcode::SPACE) {
39        Warn!(
40          "unexpected",
41          first.stringify(),
42          s!(
43            "Unexpected text after the value in an argument of \\DeclareMathOperator ('{}'); it is read after \\DeclareMathOperator",
44            Tokens::new(name.clone())
45          )
46        );
47      }
48      unread_vec(std::mem::take(&mut name));
49    }
50    let has_star = star.is_some();
51    let mut presentation = vec![T_CS!("\\operatorname")];
52    if has_star {
53      presentation.push(T_OTHER!("*"));
54    }
55    presentation.push(T_BEGIN!());
56    presentation.extend(text.unlist());
57    presentation.push(T_END!());
58    // Perl L26-29: scriptpos => ($star ? \&doScriptpos : 'post') — the starred form's is `mid` in
59    // display, read with the wrapper's properties at digestion (dialect.rs); revert_as => 'context'
60    // (Perl passes it; the wrapped DefMath does not read it).
61    let opts = MathPrimitiveOptions {
62      role: Some(if has_star { "OPERATOR" } else { "OPFUNCTION" }.to_string()),
63      font: Some(fontmap!(family => "serif", series => "medium", shape => "upright").into()),
64      scriptpos: if has_star { None } else { Some("post".to_string()) },
65      dynamic_scriptpos: has_star,
66      revert_as: Some(Cow::Borrowed("context")),
67      ..Default::default()};
68    def_math(cs, None, Tokens::new(presentation), opts)?;
69    if !name.is_empty() {
70      let operator = T_CS!(s!("\\lx@amsopn@{}", cs.to_string().trim_start_matches('\\')));
71      let_i(&operator, &cs, None);
72      let mut definition = vec![T_CS!("\\def"), cs];
73      definition.extend(name);
74      definition.extend([T_BEGIN!(), operator, T_END!()]);
75      digest(Tokens::new(definition))?;
76    }
77  });
