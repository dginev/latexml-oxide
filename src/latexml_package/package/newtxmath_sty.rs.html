7/// A Greek letter in italic whatever the alphabet around it, as `def_math_upright_greek` sets an upright one.
8fn def_math_italic_greek(cs: &str, present: &str) -> Result<()> {
9  let (cs_tok, params) = parse_prototype(cs, true)?;
10  let opts = MathPrimitiveOptions {
11    font: Some(FontDirective::from(
12      fontmap!(shape => "italic", forceshape => true),
13    )),
14    ..MathPrimitiveOptions::default()
15  };
16  def_math(cs_tok, params, present.to_string(), opts)?;
17  Ok(())
18}

61/// newtxmath.sty:246-247, :251 (newpxmath.sty:173-175): the Greek options its binding reads, recorded for
62/// `newtx_capital_greek` after the options are processed. Any other option is not the binding's concern.
63pub fn declare_newtx_greek_options() -> Result<()> {
64  // In newtxmath.sty's order (:246 before :247; newpxmath.sty:173 before :174), which `\ProcessOptions` follows.
65  DeclareOption!("uprightGreek", {
66    AssignValue!("newtx_slanted_greek" => false);
67  });
68  DeclareOption!("slantedGreek", {
69    AssignValue!("newtx_slanted_greek" => true);
70  });
71  DeclareOption!("frenchmath", {
72    AssignValue!("newtx_french_math" => true);
73  });
74  DeclareOption!(None, {});
75  Ok(())
76}

78/// newtxmath.sty:354 and :1015-1039 (newpxmath.sty:212, :595-619): `frenchmath` implies `uprightGreek`; under
79/// `slantedGreek` the capital Greek letters are the `letters` italics, italic as a Latin letter unless a math alphabet
80/// is in force (KNOWN_PERL_ERRORS #1147), else the upright ones of `lettersA` — over a class's italic ones (llncs).
81pub fn newtx_capital_greek() -> Result<()> {
82  let slanted = lookup_bool("newtx_slanted_greek") && !lookup_bool("newtx_french_math");
83  declare_capital_greek("", slanted.then_some(LettersClass::Variable))
84}
