276/// A keyval value without one balanced outer brace pair, as xkeyval reads it (`{0.5\textwidth}`; not `{a}{b}`).
277pub(crate) fn unbraced(value: &str) -> &str {
278  let value = value.trim();
279  if !value.starts_with('{') || !value.ends_with('}') {
280    return value;
281  }
282  let mut depth = 0usize;
283  for (at, c) in value.char_indices() {
284    match c {
285      '{' => depth += 1,
286      '}' => {
287        depth = depth.saturating_sub(1);
288        if depth == 0 && at + 1 < value.len() {
289          return value;
290        }
291      },
292      _ => {},
293    }
294  }
295  value[1..value.len() - 1].trim()
296}

298/// The items of a keyval list at brace depth 0.
299pub(crate) fn top_level_items(list: &str) -> Vec<String> {
300  let (mut items, mut current, mut depth) = (Vec::new(), String::new(), 0usize);
301  for c in list.chars() {
302    match c {
303      '{' => depth += 1,
304      '}' => depth = depth.saturating_sub(1),
305      ',' if depth == 0 => {
306        items.push(std::mem::take(&mut current));
307        continue;
308      },
309      _ => {},
310    }
311    current.push(c);
312  }
313  items.push(current);
314  items
315    .into_iter()
316    .map(|i| i.trim().to_string())
317    .filter(|i| !i.is_empty())
318    .collect()
