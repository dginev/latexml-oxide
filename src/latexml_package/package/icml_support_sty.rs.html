15/// icml2016.sty … icml2025.sty, which this binding stands in for (Perl has no binding for them; its versioned-name
16/// fallback answers them with icml.sty.ltxml, so the distributed pair loads whatever the copy says, KNOWN_PERL_ERRORS
17/// #684): every copy as ICML distributes it loads `algorithm` and `algorithmic` (icml2016.sty:70-71, icml2017.sty:72-73,
18/// icml2025.sty:95-96), but a paper using `algpseudocode` edits `algorithmic` out of its copy (icml2024.sty:94
19/// "removed due to conflict with algpseudocode", 2402.05403) — loaded, it bails algorithmicx (algorithmicx.sty.ltxml:9-12)
20/// and leaves `\State`/`\For` undefined (1706.03643, 1706.04918, 1706.04933, 1706.04964, 2002.06473, 2106.05074). The
21/// paper's copy decides the algorithm packages; the distributed pair without one.
22pub fn load_icml_style() -> Result<()> { icml_support(true) }

24#[rustfmt::skip]
25fn icml_support(follow_paper_copy: bool) -> Result<()> {
26  // The style as requested (`icml2026`, also through the versioned fallback: `fallback_request`), whose year decides
27  // what differs between versions.
28  let fallback_request = lookup_string("fallback_request");
29  let request = match fallback_request.rsplit('/').next() {
30    Some(name) if name.starts_with("icml") => fallback_request.clone(),
31    _ => do_expand(T_CS!("\\@currname"))?.to_string(),
32  };
33  let version = request.rsplit('/').next().unwrap_or_default().to_string();
34  let year = version.strip_prefix("icml").and_then(|year| year.parse::<u32>().ok());
35  // The paper's copy of the style (`None` without one, and for icml.sty's own pair), whose edits of the notice
36  // strings and the packages it loads are the paper's.
37  let copy = if follow_paper_copy { shipped_copy_text("sty")? } else { None };
38  RequirePackage!("times");
39  RequirePackage!("fancyhdr");
40  RequirePackage!("color");
41  // ICML 2024/2025 templates use xcolor's \colorlet for callout colors;
42  // load xcolor eagerly. Pre-load with [dvipsnames, table] so that
43  // user `\usepackage[dvipsnames, table]{xcolor}` doesn't silently
44  // option-clash and leave colortbl/dvipsnam.def unloaded. Witness
45  // 2405.18180 (icml2025).
46  RequirePackage!("xcolor", options => vec!["dvipsnames".to_string(), "table".to_string()]);
47  if !(follow_paper_copy && require_shipped_algorithm_packages("sty")?) {
48    RequirePackage!("algorithm");
49    RequirePackage!("algorithmic");
50  }
51  RequirePackage!("natbib");
52  // icml2026.sty:124-127 (and the 2025 copies that add them) `\RequirePackage{eso-pic}`, `{forloop}`, `{url}`,
53  // `{caption}`: required as the paper's copy loads them. Its `\printAffiliationsAndNotice` runs `\forloop` over the
54  // affiliations, which a paper's own redefinition copies (2601.04575).
55  if follow_paper_copy {
56    require_shipped_packages("sty", &["eso-pic", "url", "caption"])?;
57  }
58  // forloop has no binding, which the copy's dependency scan would load; 2026's style loads it (icml2026.sty:125), and
59  // with the affiliation counters (icml2017.sty … icml2026.sty:434, :518; icml2016.sty has none) a paper's own copy
60  // of the loop runs. A copy decides both; without one, the version.
61  let copy_or = |code: &str, by_version: bool| copy.as_deref().map_or(by_version, |text| class_code_contains(text, code));
62  let copy_loads_or = |package: &str, by_version: bool| copy.as_deref().map_or(by_version, |text| class_text_loads(text, package));
63  if copy_loads_or("forloop", year == Some(2026)) {
64    RequirePackage!("forloop");
65  }
66  if copy_or(r"\newcounter{@affiliationcounter}", year.is_some_and(|year| year >= 2017)) {
67    // (icml2019.sty, read raw by its binding, has defined them already)
68    RawTeX!(r"\@ifundefined{c@@affiliationcounter}{\newcounter{@affiliationcounter}\newcounter{@affilnum}}{}");
69  }
70
71  // Citations
72  DefMacro!("\\yrcite Semiverbatim", "\\citeyearpar{#1}");
73  DefMacro!("\\cite Semiverbatim", "\\citep{#1}");
74
75  // icml2018.sty L709: `\long\def\comment#1{}` — a review-annotation macro that
76  // gobbles its argument (authors write `\comment{\ref{…}}` to hide draft
77  // notes). Our binding intercepts icml2016/2017/2018 but had omitted it, so a
78  // paper using `\comment{…}` (via arxiv.sty → `\RequirePackage{icml2018}`) hit
79  // `undefined:\comment` where Perl (which loads icml2018) is clean. Gobbling
80  // is the intended, source-faithful behavior, the binding's declared drop (the title-page check wants none of it,
81  // OXIDIZED_DESIGN_DIVERGENCES #953 "106t"). Witness 1803.00942.
82  DefMacro!("\\comment{}", "", drops => [1]);
83
84  // Frontmatter
85  Let!("\\icmltitle", "\\title");
86  // Perl gobbles \icmltitlerunning; surpass: it's the running-head
87  // variant of the title, genuine author metadata.
88  DefMacro!("\\icmltitlerunning{}",
89    "\\lx@add@frontmatter{ltx:toctitle}{#1}");
90  // \icmlsetsymbol{name}{symbol} — creates `\<name>` macro expanding
91  // to <symbol> for use in author lists. Real icml2024.sty L100-ish:
92  //   `\def\icmlsetsymbol#1#2{\expandafter\def\csname #1\endcsname{#2}}`
93  // Previous stub gobbled both args without defining the CS, breaking
94  // `\icmlauthor{...}{equal,affil,icmlWorkDone}` which later references
95  // `\icmlWorkDone` in `\printAffiliationsAndNotice`. Witness 2310.06430.
96  DefMacro!("\\icmlsetsymbol{}{}",
97    "\\expandafter\\def\\csname #1\\endcsname{#2}");
98
99  DefEnvironment!("{icmlauthorlist}", "#body");
100
101  // \icmlauthor{author}{labels}   One per author; labels used to connect to affiliation
102  DefMacro!("\\icmlauthor{} Semiverbatim", "\\lx@add@creator[role=author,annotations={#2}]{#1}");
103  // icml20XX.sty: \newcommand{\icmlCorrespondingAuthor}{\textsuperscript{$\dagger$}Corresponding author}
104  // (0-arg marker). icml2025.sty L530: \newcommand{\icmlCorr}{\textsuperscript{\dag}Corresponding author }
105  // (shorter alias, used as `\printAffiliationsAndNotice{\icmlCorr}`). Our binding
106  // intercepts the raw .sty so the `\newcommand`s never run → undefined where Perl
107  // (which loads the raw .sty) defines them. Witnesses 2403.01475, 2507.11588.
108  //
109  // These are 0-arg footnote-style markers carrying no author identity, so — unlike the
110  // 2-arg `\icmlcorrespondingauthor` below (name+email → `\lx@add@contact`) — they stay
111  // note-style metadata (role=corresponding-author) via `\lx@add@frontmatter` rather
112  // than emitting a raw inline `\textsuperscript` footnote marker.
113  DefMacro!("\\icmlCorrespondingAuthor",
114    "\\lx@add@frontmatter{ltx:note}[role=corresponding-author]{Corresponding author}");
115  DefMacro!("\\icmlCorr",
116    "\\lx@add@frontmatter{ltx:note}[role=corresponding-author]{Corresponding author}");
117  // \icmlaffiliation{label}{address}  provides affiliation for author with same label
118  DefMacro!("\\icmlaffiliation Semiverbatim {}", "\\lx@add@contact[role=affiliation,label={#1}]{#2}");
119  DefMacro!("\\icmladdress OptionalSemiverbatim {}", "\\lx@add@contact[role=address,label={#1}]{#2}"); // deprecated
120  // \icmlcorrespondingauthor{author}{email}
121  // Attempt to use author name as a matching label to connect to author
122  // (should be a fuzzy match?)
123  DefMacro!("\\icmlcorrespondingauthor{}{}",
124    "\\lx@add@contact[label={\\ifx.#1.\\else fuzzy:#1\\fi},role=email,name={Correspondence to: }]{#2}");
125
126  // \printAffiliationsAndNotice / \printAffiliationsAndWorkNotice emit
127  // a re-iteration of the affiliation list + a free-form notice. Since
128  // \icmladdress already feeds frontmatter, the affiliation list is
129  // captured separately; preserve the notice arg as a ltx:note so the
130  // author-supplied "Work done while at X" string survives.
131  // Witness: 2502.18679 (icml2025.sty L564).
132  DefMacro!("\\printAffiliationsAndNotice{}",
133    "\\lx@add@frontmatter{ltx:note}[role=affiliationnotice]{#1}");
134  DefMacro!("\\printAffiliationsAndWorkNotice{}",
135    "\\lx@add@frontmatter{ltx:note}[role=affiliationnotice]{#1}");
136  // ICML 2024: simpler `\printAffiliations` no-arg form (newer template
137  // emits the affiliation list inline without trailing notice). Witness
138  // 2310.18127.
139  def_macro_noop("\\printAffiliations")?;
140  DefMacro!("\\icmlEqualContribution", "Equal contribution");
141  // ICML 2023 (icml2023.sty L526): per-paper "equal advising" marker.
142  // Witness 2306.01153.
143  DefMacro!("\\icmlEqualAdvising", "Equal advising");
144  // ICML 2025: extended marker for joint first + senior authorship.
145  // Witness: 2503.15703 (icml2025.sty L534).
146  DefMacro!("\\icmlEqualContributionAndSenior",
147    "\\textsuperscript{*}Equal contribution, \
148     \\textsuperscript{\\char`\u{2020}}Equal senior authorship");
149  // ICML 2024 introduced `\icmlEqualSeniorContribution` — senior-only
150  // joint authorship marker (no joint-first). Witness 2305.xxxxx in
151  // wp4 (2 papers in stage 1).
152  DefMacro!("\\icmlEqualSeniorContribution",
153    "\\textsuperscript{\\char`\u{2020}}Equal senior contribution");
154  // Some paper-bundled icml*.sty templates add author-status markers
155  // beyond the kernel `\icmlEqualContribution`. The most common is
156  // `\icmlIntershipWork`, an internship-affiliation annotation passed
157  // to `\printAffiliationsAndNotice{...}` in icml2024 papers.
158  // Witness 2401.00604.
159  DefMacro!("\\icmlIntershipWork",
160    "\\textsuperscript{*}Work done during an internship");
161  // `\icmlInternship` — paper-bundled internship marker (correct spelling,
162  // distinct from the `\icmlIntershipWork` typo above). icml2019.sty L502
163  // `\newcommand{\icmlInternship}{\textsuperscript{*}This work has been done
164  // during the internship at <institution>.}`. Papers pass it into
165  // `\printAffiliationsAndNotice{\icmlInternship}`; our binding intercepts
166  // icml20xx (so the paper's bundled def never runs) and — unlike Perl,
167  // whose `\printAffiliationsAndNotice{}` gobbles its arg — preserves the
168  // notice arg as a frontmatter note, expanding the inner CS. Provide a
169  // generic fallback (institution unknowable from the binding) so the note
170  // survives error-free. Witness 1902.02603 (icml2019.sty L502).
171  DefMacro!("\\icmlInternship",
172    "\\textsuperscript{*}Work done during an internship");
173  // `\airesident` — paper-bundled AI-residency marker, same family as
174  // `\icmlInternship`. icml2019.sty L503 `\newcommand{\airesident}{
175  // \textsuperscript{$\dagger$}This work was completed as part of the
176  // <program> AI Residency}`. Passed into
177  // `\printAffiliationsAndNotice{\icmlEqualContribution\airesident}`; our
178  // binding intercepts icml20xx so the bundled def never runs, and the
179  // preserved notice arg expands the undefined CS. Generic fallback
180  // (program unknowable from the binding). Witness 1902.09574.
181  DefMacro!("\\airesident",
182    "\\textsuperscript{\\char`\u{2020}}Work completed as part of an AI Residency");
183  // \icmlOutsideContribution — paper-bundled marker noting that the
184  // contribution was made outside the author's primary affiliation.
185  // Witness 2310.14751.
186  DefMacro!("\\icmlOutsideContribution",
187    "\\textsuperscript{*}Work done outside of primary affiliation");
188  // \icmlEqualLast — paper-bundled co-last-authors marker (icml2024.sty
189  // L525). Witness 2402.02526.
190  DefMacro!("\\icmlEqualLast",
191    "\\textsuperscript{*}Co last authors");
192  // \icmlIntern — paper-bundled internship-affiliation marker.
193  // Witness 2312.05253 (icml2024.sty L534).
194  DefMacro!("\\icmlIntern",
195    "\\textsuperscript{*}Work done while interning");
196  // \icmlEqualwork — paper-bundled joint-work marker (icml2021 papers
197  // routinely redefine this in their bundled icml2021.sty to a custom
198  // institutional note). Provide a generic fallback so the canonical
199  // binding fires when the paper's bundled .sty is masked by ours.
200  // Witness 2111.13293.
201  DefMacro!("\\icmlEqualwork",
202    "\\textsuperscript{*}Joint work");
203  // \icmlProjectLead — paper-bundled project-lead marker.
204  // Witness 2402.04924 (icml2024.sty L535).
205  DefMacro!("\\icmlProjectLead",
206    "\\textsuperscript{\\char`\u{2020}}Project lead");
207  // Every icml20xx.sty sends the keywords only to the PDF metadata (icml2018.sty:530-541 … icml2026.sty:546-550
208  // `\ifdefined\nohyperref\else\ifdefined\hypersetup\hypersetup{pdfkeywords={#1}}\fi\fi`; the printed "Keywords:" line is
209  // commented out in each), which hyperref_sty.rs writes as RDFa (`dcterms:subject`), not a displayed keywords block
210  // (Perl `\lx@add@keywords`, icml_support.sty.ltxml:55; user ruling 2026-10-09; 2602.13503).
211  DefMacro!("\\icmlkeywords{}",
212    "\\ifdefined\\nohyperref\\else\\ifdefined\\hypersetup\\hypersetup{pdfkeywords={#1}}\\fi\\fi");
213  // \iclrfinalcopy — some icml*-bundled papers also use ICLR's
214  // `\iclrfinalcopy` macro (bundled icml2023.sty L: `\def\iclrfinalcopy
215  // {\iclrfinaltrue}`). Stub both as no-op. Witness 2206.06661.
216  DefConditional!("\\ificlrfinal");
217  def_macro_noop("\\iclrfinalcopy")?;
218
219  // `\toptitlebar`/`\bottomtitlebar` — the decorative rules drawn above and
220  // below the title block. Real icml20XX.sty (e.g. icml2019.sty L410-411):
221  //   \def\toptitlebar{\hrule height1pt \vskip .25in}
222  //   \def\bottomtitlebar{\vskip .22in \hrule height1pt \vskip .3in}
223  // No-arg macros (the `\toptitlebar{\Large\bf #1}` in some bundled `arxiv.sty`
224  // is `\toptitlebar` followed by a separate title group). Our binding
225  // intercepts the paper-bundled icml20XX.sty so those defs never run; Perl
226  // ships no icml2019 binding and raw-loads the .sty, reaching L410. Supply
227  // them directly. Witness 1905.03711 (article + arxiv.sty →
228  // `\RequirePackage[accepted]{icml2019}`).
229  DefMacro!("\\toptitlebar", "\\hrule height1pt \\vskip .25in");
230  DefMacro!("\\bottomtitlebar", "\\vskip .22in \\hrule height1pt \\vskip .3in");
231
232  // Random extra bits
233  def_macro_noop("\\abovestrut{}")?;
234  def_macro_noop("\\belowstrut{}")?;
235  def_macro_noop("\\abovespace")?;
236  def_macro_noop("\\aroundspace")?;
237  def_macro_noop("\\belowspace")?;
238  def_macro_noop("\\icmlruler{}")?;
239
240  // The notice under the first column and its options (icml2016.sty … icml2025.sty:103-133, icml2026.sty:129-177),
241  // which papers print from their own `\printAffiliationsAndNotice` (`…#1\\[2pt]\Notice@String`, 32 papers;
242  // 2601.23143), and icml2026.sty:504-516's check that the notice was printed, a flag the template's own redefinition
243  // sets (`\global\icml@noticeprintedtrue`, example_paper.tex; 95 papers, 2601.22575): the review notice, `accepted`'s
244  // proceedings line (`\ICML@appearing`) and 2026's `preprint` one, and the `\ificmlshowauthors` switch the two
245  // options turn on. The strings are the paper's copy's where it ships one — authors edit them (a workshop's line:
246  // 2606.28747 "3rd AI for Math Workshop", 2604.14572, 2507.02987, 2506.09099; `accepted` setting no notice,
247  // 2106.00042) — else the version's as ICML distributed it; a version whose copies disagree (icml2020's PMLR 108 in
248  // 2006.00731 and 2006.00866, 119 in 2006.00706, empty in 2006.00937 and 2009.02479) or that is unknown has an empty
249  // line, so an accepted paper prints neither the review notice nor an invented proceedings line.
250  let copy_definition = |name: &str| {
251    copy
252      .as_deref()
253      .and_then(|text| class_code_argument(text, &s!("\\newcommand{{\\{name}}}")))
254  };
255  // An option's notice: the `\renewcommand{\Notice@String}{…}` of the copy's `\DeclareOption{option}` (`None` inside
256  // when that option sets none; `None` outside when the copy declares no such option), or `by_version` without a copy.
257  let option_notice = |option: &str, by_version: Option<&str>| match copy.as_deref() {
258    Some(text) => class_code_argument(text, &s!("\\DeclareOption{{{option}}}"))
259      .map(|body| class_code_argument(&body, r"\renewcommand{\Notice@String}")),
260    None => by_version.map(|notice| Some(notice.to_string())),
261  };
262  let appearing = copy_definition("ICML@appearing").unwrap_or_else(|| {
263    let proceedings = match year {
264      Some(2015) => r"\textit{Proceedings of the $\mathit{32}^{nd}$ International Conference on Machine Learning}, Lille, France, 2015. JMLR: W\&CP volume 37.  Copyright 2015 by the author(s).",
265      Some(2016) => r"\textit{Proceedings of the $\mathit{33}^{rd}$ International Conference on Machine Learning}, New York, NY, USA, 2016. JMLR: W\&CP volume 48.  Copyright 2016 by the author(s).",
266      Some(2017) => r"\textit{Proceedings of the $\mathit{34}^{th}$ International Conference on Machine Learning}, Sydney, Australia, 2017. JMLR: W\&CP.  Copyright 2017 by the author(s).",
267      Some(2018) => r"\textit{Proceedings of the $\mathit{35}^{th}$ International Conference on Machine Learning}, Stockholm, Sweden, PMLR 80, 2018. Copyright 2018 by the author(s).",
268      Some(2019) => r"\textit{Proceedings of the $\mathit{36}^{th}$ International Conference on Machine Learning}, Long Beach, California, PMLR 97, 2019. Copyright 2019 by the author(s).",
269      Some(2021) => r"\textit{Proceedings of the $\mathit{38}^{th}$ International Conference on Machine Learning}, PMLR 139, 2021. Copyright 2021 by the author(s).",
270      Some(2022) => r"\textit{Proceedings of the $\mathit{39}^{th}$ International Conference on Machine Learning}, Baltimore, Maryland, USA, PMLR 162, 2022. Copyright 2022 by the author(s).",
271      Some(2023) => r"\textit{Proceedings of the $\mathit{40}^{th}$ International Conference on Machine Learning}, Honolulu, Hawaii, USA. PMLR 202, 2023. Copyright 2023 by the author(s).",
272      Some(2024) => r"\textit{Proceedings of the $\mathit{41}^{st}$ International Conference on Machine Learning}, Vienna, Austria. PMLR 235, 2024. Copyright 2024 by the author(s).",
273      Some(2025) => r"\textit{Proceedings of the $\mathit{42}^{nd}$ International Conference on Machine Learning}, Vancouver, Canada. PMLR 267, 2025. Copyright 2025 by the author(s).",
274      Some(2026) => r"\textit{Proceedings of the $\mathit{43}^{rd}$ International Conference on Machine Learning}, Seoul, South Korea. PMLR 306, 2026. Copyright 2026 by the author(s).",
275      _ => "",
276    };
277    proceedings.to_string()
278  });
279  let review = copy_definition("Notice@String").unwrap_or_else(|| {
280    r"Preliminary work.  Under review by the International Conference on Machine Learning (ICML)\@.  Do not distribute."
281      .to_string()
282  });
283  let preprint = copy_definition("ICML@preprint").unwrap_or_else(|| r"\textit{Preprint. \today.}".to_string());
284  RawTeX!(&s!("\\providecommand{{\\Notice@String}}{{{review}}}
285\\providecommand{{\\ICML@appearing}}{{{appearing}}}
286\\providecommand{{\\ICML@preprint}}{{{preprint}}}
287\\newif\\ificmlshowauthors \\icmlshowauthorsfalse
288\\newif\\ificml@noticeprinted \\icml@noticeprintedfalse"));
289  let declare_notice_option = |option: &str, flag: &str, by_version: Option<&str>| -> Result<()> {
290    if let Some(notice) = option_notice(option, by_version) {
291      let set_notice = notice.map(|notice| s!("\\renewcommand{{\\Notice@String}}{{{notice}}}")).unwrap_or_default();
292      let code = TokenizeInternal!(TeXString::assembled(s!("{set_notice}\\gdef\\{flag}{{1}}")));
293      DeclareOption!(option, code);
294    }
295    Ok(())
296  };
297  declare_notice_option("accepted", "isaccepted", Some(r"\ICML@appearing"))?;
298  declare_notice_option("preprint", "ispreprint", (year == Some(2026)).then_some(r"\ICML@preprint"))?;
299  DeclareOption!("nohyperref", r"\gdef\nohyperref{1}");
300  DeclareOption!(None, {});
301  ProcessOptions!();
302  RawTeX!(r"\ifdefined\isaccepted \icmlshowauthorstrue \fi
303\ifdefined\ispreprint \icmlshowauthorstrue \fi");
304  Ok(())
