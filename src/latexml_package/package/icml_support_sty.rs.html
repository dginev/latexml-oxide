15/// icml2016.sty … icml2025.sty, which this binding stands in for (Perl has no binding for them; its versioned-name
16/// fallback answers them with icml.sty.ltxml, so the distributed pair loads whatever the copy says, KNOWN_PERL_ERRORS
17/// #684): every copy as ICML distributes it loads `algorithm` and `algorithmic` (icml2016.sty:70-71, icml2017.sty:72-73,
18/// icml2025.sty:95-96), but a paper using `algpseudocode` edits `algorithmic` out of its copy (icml2024.sty:94
19/// "removed due to conflict with algpseudocode", 2402.05403) — loaded, it bails algorithmicx (algorithmicx.sty.ltxml:9-12)
20/// and leaves `\State`/`\For` undefined (1706.03643, 1706.04918, 1706.04933, 1706.04964, 2002.06473, 2106.05074). The
21/// paper's copy decides the algorithm packages; the distributed pair without one.
22pub fn load_icml_style() -> Result<()> { icml_support(true) }

24#[rustfmt::skip]
25fn icml_support(follow_paper_copy: bool) -> Result<()> {
26  // The style as requested (`icml2026`, also through the versioned fallback: `fallback_request`), whose year decides
27  // what differs between versions.
28  let fallback_request = lookup_string("fallback_request");
29  let request = match fallback_request.rsplit('/').next() {
30    Some(name) if name.starts_with("icml") => fallback_request.clone(),
31    _ => do_expand(T_CS!("\\@currname"))?.to_string(),
32  };
33  let version = request.rsplit('/').next().unwrap_or_default().to_string();
34  let year = version.strip_prefix("icml").and_then(|year| year.parse::<u32>().ok());
35  // The paper's copy of the style (`None` without one, and for icml.sty's own pair), whose edits of the notice
36  // strings and the packages it loads are the paper's.
37  let copy = if follow_paper_copy { shipped_copy_text("sty")? } else { None };
38  RequirePackage!("times");
39  RequirePackage!("fancyhdr");
40  RequirePackage!("color");
41  // ICML 2024/2025 templates use xcolor's \colorlet for callout colors;
42  // load xcolor eagerly. Pre-load with [dvipsnames, table] so that
43  // user `\usepackage[dvipsnames, table]{xcolor}` doesn't silently
44  // option-clash and leave colortbl/dvipsnam.def unloaded. Witness
45  // 2405.18180 (icml2025).
46  RequirePackage!("xcolor", options => vec!["dvipsnames".to_string(), "table".to_string()]);
47  if !(follow_paper_copy && require_shipped_algorithm_packages("sty")?) {
48    RequirePackage!("algorithm");
49    RequirePackage!("algorithmic");
50  }
51  RequirePackage!("natbib");
52  // icml2026.sty:124-127 (and the 2025 copies that add them) `\RequirePackage{eso-pic}`, `{forloop}`, `{url}`,
53  // `{caption}`: required as the paper's copy loads them. Its `\printAffiliationsAndNotice` runs `\forloop` over the
54  // affiliations, which a paper's own redefinition copies (2601.04575).
55  if follow_paper_copy {
56    require_shipped_packages("sty", &["eso-pic", "url", "caption"])?;
57  }
58  // forloop has no binding, which the copy's dependency scan would load; 2026's style loads it (icml2026.sty:125), and
59  // with the affiliation counters (icml2017.sty … icml2026.sty:434, :518; icml2016.sty has none) a paper's own copy
60  // of the loop runs. A copy decides both; without one, the version.
61  let copy_or = |code: &str, by_version: bool| copy.as_deref().map_or(by_version, |text| class_code_contains(text, code));
62  let copy_loads_or = |package: &str, by_version: bool| copy.as_deref().map_or(by_version, |text| class_text_loads(text, package));
63  if copy_loads_or("forloop", year == Some(2026)) {
64    RequirePackage!("forloop");
65  }
66  if copy_or(r"\newcounter{@affiliationcounter}", year.is_some_and(|year| year >= 2017)) {
67    // (icml2019.sty, read raw by its binding, has defined them already)
68    RawTeX!(r"\@ifundefined{c@@affiliationcounter}{\newcounter{@affiliationcounter}\newcounter{@affilnum}}{}");
69  }
70
71  // Citations
72  DefMacro!("\\yrcite Semiverbatim", "\\citeyearpar{#1}");
73  DefMacro!("\\cite Semiverbatim", "\\citep{#1}");
74
75  // icml2018.sty L709: `\long\def\comment#1{}` — a review-annotation macro that
76  // gobbles its argument (authors write `\comment{\ref{…}}` to hide draft
77  // notes). Our binding intercepts icml2016/2017/2018 but had omitted it, so a
78  // paper using `\comment{…}` (via arxiv.sty → `\RequirePackage{icml2018}`) hit
79  // `undefined:\comment` where Perl (which loads icml2018) is clean. Gobbling
80  // is the intended, source-faithful behavior, the binding's declared drop (the title-page check wants none of it,
81  // OXIDIZED_DESIGN_DIVERGENCES #953 "106t"). Witness 1803.00942.
82  DefMacro!("\\comment{}", "", drops => [1]);
83
84  // Frontmatter
85  Let!("\\icmltitle", "\\title");
86  // Perl gobbles \icmltitlerunning; surpass: it's the running-head
87  // variant of the title, genuine author metadata.
88  DefMacro!("\\icmltitlerunning{}",
89    "\\lx@add@frontmatter{ltx:toctitle}{#1}");
90  // icml2026.sty:465-467 (icml2024.sty:489-491): only `\@icmlsymbol<name>`, which `\icmlauthor` prints below; a
91  // `\<name>` too made `\icmlsetsymbol{dagger}{$\dagger$}` a self-calling `\dagger` (2606.08980). 2310.06430's
92  // `\icmlWorkDone` is its copy's `\newcommand` (icml2024.sty:529), the copy read as the file this binding stands in for.
93  DefMacro!("\\icmlsetsymbol{}{}", "\\expandafter\\gdef\\csname @icmlsymbol#1\\endcsname{#2}");
94
95  DefEnvironment!("{icmlauthorlist}", "#body");
96
97  // \icmlauthor{author}{labels}   One per author; labels used to connect to affiliation
98  // A label `\icmlsetsymbol` named prints its symbol after the name (icml2026.sty:446-458 `\@pa`; icml2017.sty on):
99  // `\icmlauthor{Ann Able}{equal,a}` is "Ann Able∗ 1", the name's mark the notice's `\icmlEqualContribution` explains
100  // (2602.04557), a mark-only note of the creator (as acmart's `\authornotemark`), so the name stays the one
101  // `\icmlcorrespondingauthor` spells (1904.07793); a symbol that prints no one mark (`printed_label_pieces`) prints in
102  // the name. The other labels request their affiliations.
103  DefMacro!("\\icmlauthor{} Semiverbatim", sub[(name, labels)] {
104    let mut name = name.unlist();
105    let mut marks = Vec::new();
106    for label in labels.to_string().split(',').map(str::trim).filter(|label| !label.is_empty()) {
107      let symbol = T_CS!(s!("\\@icmlsymbol{label}"));
108      let Some(meaning) = lookup_meaning(&symbol) else {
109        continue;
110      };
111      let glyph = match meaning {
112        Stored::Expandable(d) => match d.get_expansion() {
113          Some(ExpansionBody::Tokens(body)) => match printed_label_pieces(body)?.as_slice() {
114            // (a glyph re-tokenized below must carry no TeX special: `\icmlsetsymbol{equal}{\%}` commented out the rest
115            // and lost a creator, revINVDAGb B1; such a symbol prints in the name instead)
116            [mark] if !mark.chars().any(|c| "#%{}\\~^_&$".contains(c)) => Some(mark.clone()),
117            _ => None,
118          },
119          _ => None,
120        },
121        _ => None,
122      };
123      match glyph {
124        Some(glyph) => marks.extend(
125          TokenizeInternal!(TeXString::assembled(s!(
126            "\\lx@annotate@frontmatter{{ltx:creator}}{{ltx:note}}[role=footnotemark,mark={{{glyph}}}]{{{glyph}}}"
127          )))
128          .unlist(),
129        ),
130        None => name.extend(Invocation!(T_CS!("\\textsuperscript"), vec![Some(Tokens::new(vec![symbol]))]).unlist()),
131      }
132    }
133    let mut call = Invocation!("\\lx@add@creator[role=author,annotations={#1}]{#2}", vec![
134      Some(labels),
135      Some(Tokens::new(name))
136    ])
137    .unlist();
138    call.extend(marks);
139    Ok(Tokens::new(call))
140  });
141  // icml20XX.sty: \newcommand{\icmlCorrespondingAuthor}{\textsuperscript{$\dagger$}Corresponding author}
142  // (0-arg marker). icml2025.sty L530: \newcommand{\icmlCorr}{\textsuperscript{\dag}Corresponding author }
143  // (shorter alias, used as `\printAffiliationsAndNotice{\icmlCorr}`). Our binding
144  // intercepts the raw .sty so the `\newcommand`s never run → undefined where Perl
145  // (which loads the raw .sty) defines them. Witnesses 2403.01475, 2507.11588.
146  //
147  // These are 0-arg footnote-style markers carrying no author identity, so — unlike the
148  // 2-arg `\icmlcorrespondingauthor` below (name+email → `\lx@add@contact`) — they stay
149  // note-style metadata (role=corresponding-author) via `\lx@add@frontmatter` rather
150  // than emitting a raw inline `\textsuperscript` footnote marker.
151  DefMacro!("\\icmlCorrespondingAuthor",
152    "\\lx@add@frontmatter{ltx:note}[role=corresponding-author]{Corresponding author}");
153  DefMacro!("\\icmlCorr",
154    "\\lx@add@frontmatter{ltx:note}[role=corresponding-author]{Corresponding author}");
155  // \icmlaffiliation{label}{address}  provides affiliation for author with same label
156  DefMacro!("\\icmlaffiliation Semiverbatim {}", "\\lx@add@contact[role=affiliation,label={#1}]{#2}");
157  DefMacro!("\\icmladdress OptionalSemiverbatim {}", "\\lx@add@contact[role=address,label={#1}]{#2}"); // deprecated
158  // \icmlcorrespondingauthor{author}{email}
159  // Attempt to use author name as a matching label to connect to author
160  // (should be a fuzzy match?)
161  // It also builds the "Correspondence to:" line its notice prints (icml2026.sty:490-499, :529-530; icml2017.sty on,
162  // or the paper's copy), which documents extend (2605.04712, 2605.21190, 2606.01172
163  // `\g@addto@macro\icmlcorrespondingauthor@text`; OD #1036): printed by `\printAffiliationsAndNotice` below.
164  if copy_or(r"\icmlcorrespondingauthor@text", year.is_some_and(|year| year >= 2017)) {
165    DefMacro!("\\icmlcorrespondingauthor{}{}",
166      r"\lx@add@contact[label={\ifx.#1.\else fuzzy:#1\fi},role=email,name={Correspondence to: }]{#2}%
167\ificmlshowauthors
168  \ifdefined\icmlcorrespondingauthor@text
169    \g@addto@macro\icmlcorrespondingauthor@text{, #1 \textless{}#2\textgreater{}}%
170  \else
171    \gdef\icmlcorrespondingauthor@text{#1 \textless{}#2\textgreater{}}%
172  \fi
173\else
174  \gdef\icmlcorrespondingauthor@text{Anonymous Author \textless{}anon.email@domain.com\textgreater{}}%
175\fi
176\global\let\lx@icml@corr@built\icmlcorrespondingauthor@text");
177  } else {
178    DefMacro!("\\icmlcorrespondingauthor{}{}",
179      "\\lx@add@contact[label={\\ifx.#1.\\else fuzzy:#1\\fi},role=email,name={Correspondence to: }]{#2}");
180  }

182  // \printAffiliationsAndNotice / \printAffiliationsAndWorkNotice emit
183  // a re-iteration of the affiliation list + a free-form notice. Since
184  // \icmladdress already feeds frontmatter, the affiliation list is
185  // captured separately; preserve the notice arg as a ltx:note so the
186  // author-supplied "Work done while at X" string survives.
187  // Witness: 2502.18679 (icml2025.sty L564).
188  // The notice's "Correspondence to:" line (icml2026.sty:529-530) is the contacts here, but for what a document added to
189  // the line (2605.04712, 2605.21190, 2606.01172 `\g@addto@macro\icmlcorrespondingauthor@text{. Code at …}`): a note.
190  DefMacro!("\\printAffiliationsAndNotice{}",
191    "\\lx@add@frontmatter{ltx:note}[role=affiliationnotice]{#1}\\lx@icml@corr@added");
192  DefMacro!("\\printAffiliationsAndWorkNotice{}",
193    "\\lx@add@frontmatter{ltx:note}[role=affiliationnotice]{#1}\\lx@icml@corr@added");
194  // (the line past what this binding's `\icmlcorrespondingauthor` built, `\lx@icml@corr@built`, without the
195  // punctuation it starts with; the whole line, as printed, when the document replaced it — or set it without calling
196  // `\icmlcorrespondingauthor`, which leaves `\lx@icml@corr@built` undefined)
197  DefMacro!("\\lx@icml@corr@added", sub[_args] {
198    let body = |cs: &str| match lookup_meaning(&T_CS!(cs)) {
199      Some(Stored::Expandable(d)) => match d.get_expansion() {
200        Some(ExpansionBody::Tokens(body)) => Some(body.unlist_ref().to_vec()),
201        _ => Some(Vec::new()),
202      },
203      _ => None,
204    };
205    let Some(text) = body(r"\icmlcorrespondingauthor@text") else { return Ok(Tokens::default()) };
206    let built = body(r"\lx@icml@corr@built");
207    let (note, added) = match built.and_then(|built| text.strip_prefix(built.as_slice())) {
208      Some(added) => (r"\lx@add@pubnote[role=note,name={}]", added),
209      None => (r"\lx@add@pubnote[role=note,name={Correspondence to: }]", text.as_slice()),
210    };
211    let punctuation = [T_OTHER!("."), T_OTHER!(","), T_OTHER!(";"), T_OTHER!(":")];
212    let lead = added
213      .iter()
214      .take_while(|t| t.get_catcode() == Catcode::SPACE || punctuation.contains(t))
215      .count();
216    let added = &added[lead..];
217    if added.is_empty() {
218      return Ok(Tokens::default());
219    }
220    let mut out = mouth::tokenize_internal(note).unlist();
221    out.push(T_BEGIN!());
222    out.extend_from_slice(added);
223    out.push(T_END!());
224    Ok(Tokens::new(out))
225  });

236  // ICML 2025: extended marker for joint first + senior authorship.
237  // Witness: 2503.15703 (icml2025.sty L534).
238  // The dagger of this and the markers below is `\textdagger`, as the papers' own `\dag`: a `\char` of U+2020 is no

240  DefMacro!("\\icmlEqualContributionAndSenior",
241    "\\textsuperscript{*}Equal contribution, \
242     \\textsuperscript{\\textdagger}Equal senior authorship");

334  // The notice under the first column and its options (icml2016.sty … icml2025.sty:103-133, icml2026.sty:129-177),
335  // which papers print from their own `\printAffiliationsAndNotice` (`…#1\\[2pt]\Notice@String`, 32 papers;
336  // 2601.23143), and icml2026.sty:504-516's check that the notice was printed, a flag the template's own redefinition
337  // sets (`\global\icml@noticeprintedtrue`, example_paper.tex; 95 papers, 2601.22575): the review notice, `accepted`'s
338  // proceedings line (`\ICML@appearing`) and 2026's `preprint` one, and the `\ificmlshowauthors` switch the two
339  // options turn on. The strings are the paper's copy's where it ships one — authors edit them (a workshop's line:
340  // 2606.28747 "3rd AI for Math Workshop", 2604.14572, 2507.02987, 2506.09099; `accepted` setting no notice,
341  // 2106.00042) — else the version's as ICML distributed it; a version whose copies disagree (icml2020's PMLR 108 in
342  // 2006.00731 and 2006.00866, 119 in 2006.00706, empty in 2006.00937 and 2009.02479) or that is unknown has an empty
343  // line, so an accepted paper prints neither the review notice nor an invented proceedings line.
344  let copy_definition = |name: &str| {
345    copy
346      .as_deref()
347      .and_then(|text| class_code_argument(text, &s!("\\newcommand{{\\{name}}}")))
348  };
349  // An option's notice: the `\renewcommand{\Notice@String}{…}` of the copy's `\DeclareOption{option}` (`None` inside
350  // when that option sets none; `None` outside when the copy declares no such option), or `by_version` without a copy.
351  let option_notice = |option: &str, by_version: Option<&str>| match copy.as_deref() {
352    Some(text) => class_code_argument(text, &s!("\\DeclareOption{{{option}}}"))
353      .map(|body| class_code_argument(&body, r"\renewcommand{\Notice@String}")),
354    None => by_version.map(|notice| Some(notice.to_string())),
355  };
356  let appearing = copy_definition("ICML@appearing").unwrap_or_else(|| {
357    let proceedings = match year {
358      Some(2015) => r"\textit{Proceedings of the $\mathit{32}^{nd}$ International Conference on Machine Learning}, Lille, France, 2015. JMLR: W\&CP volume 37.  Copyright 2015 by the author(s).",

361      Some(2018) => r"\textit{Proceedings of the $\mathit{35}^{th}$ International Conference on Machine Learning}, Stockholm, Sweden, PMLR 80, 2018. Copyright 2018 by the author(s).",
362      Some(2019) => r"\textit{Proceedings of the $\mathit{36}^{th}$ International Conference on Machine Learning}, Long Beach, California, PMLR 97, 2019. Copyright 2019 by the author(s).",
