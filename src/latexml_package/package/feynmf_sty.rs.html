13/// feynmf/feynmp `{fmfgraph}`/`{fmfgraph*}` environments: a `(width,height)`
14/// pair (feynmf.sty:180 `\def\fmfgraph(#1,#2)`, :193 `fmfgraph*`) followed by
15/// the diagram body. The real package emits a Metafont/MetaPost diagram. For
16/// HTML rendering we drop the graphics body (neither runs in our pipeline) but
17/// preserve the env so the surrounding equation/figure context still parses
18/// cleanly. Witness 2309.07343 (15 errors all from {fmfgraph*} undefined).
19/// The pair is `Match:( Until:, Until:)`, as `\put` reads its: the former
20/// `{}{}` took `(` and `3` of `(30,20)`, so the note read
21/// "(Feynman diagram, (x3)". Guard:
22/// `binding_singletons_56::fmfgraph_reads_its_size_pair`.
23pub(crate) fn feynmf_graph_environments() -> Result<()> {
24  RawTeX!(
25    r"\def\lx@fmf@keep#1{\expandafter\fmf@keep\lx@fmf@size{#1}}
26\def\lx@fmf@keepstar#1{\expandafter\fmf@keepstar\lx@fmf@size{#1}}"
27  );
28  // In the graph `\fmfkeep{<name>}` keeps it for `\fmfreuse{<name>}` (feynmf.sty:182/:196, feynmp.sty:156):
29  // `\fmf@keep`/`\fmf@keepstar` (feynmf.sty:230-237) record the graph `\begin{fmfgraph}(w,h)\end{fmfgraph}`,
30  // which the raw `\fmfreuse` (:239-250) sets again. Outside a graph it stays the package's
31  // `\fmf@nokeep` error (:226-229), which every kept graph raised (witnesses hep-lat0210017,
32  // hep-th9805098, 1411.4662, hep-ph0606300).
33  DefEnvironment!("{fmfgraph} Match:( Until:, Until:)",
34    "<ltx:note role='feynman-diagram'>(Feynman diagram, #2x#3)</ltx:note>",
35    after_digest_begin => sub[whatsit] { fmf_keep_in_graph(whatsit, "\\lx@fmf@keep") },
36    mode => "internal_vertical");
37  DefEnvironment!("{fmfgraph*} Match:( Until:, Until:)",
38    "<ltx:note role='feynman-diagram'>(Feynman diagram, #2x#3)</ltx:note>",
39    after_digest_begin => sub[whatsit] { fmf_keep_in_graph(whatsit, "\\lx@fmf@keepstar") },
40    mode => "internal_vertical");
41  // {fmffile}{name} - wraps a Feynman-diagram session. Render as no-op
42  // env (the diagrams inside are rendered by {fmfgraph}/{fmfgraph*}).
43  // The argument is the METAFONT/MetaPost file's name (feynmp.sty:97-109 `\def\thefmffile{#1}`, `\openout`), never
44  // typeset: `{sde_feynman}` raised "Script _ can only appear in math mode" (1906.06341, 2512.17695; OXIDIZED_DESIGN
45  // #551). Guard `perfect_kernel_batch65m::file_arguments_are_not_typeset`.
46  // feynmf.sty:97 `\def\thefmffile{#1}` for the environment, which `\fmfkeep` records and `\fmfreuse` reads
47  // back (`\fmf@k:f:<name>`, :233/:240); left `\relax` (:142), the reuse defined `\thefmffile` as itself.
48  DefEnvironment!("{fmffile} Semiverbatim", "#body",
49    after_digest_begin => sub[whatsit] {
50      let name = whatsit.get_arg(1).map(|a| a.to_string()).unwrap_or_default();
51      def_macro(T_CS!("\\thefmffile"), None, Tokens::new(ExplodeText!(name.trim())), None)?;
52    },
53    mode => "internal_vertical");
54  Ok(())
55}

57/// Diagram-content macros used inside `{fmfgraph}`/`{fmfgraph*}` (shared by the
58/// `feynmf` and `feynmp` packages — feynmp is the MetaPost/PDF variant with the
59/// SAME user macros). We don't render the diagrams (no MetaFont/MetaPost in our
60/// pipeline), so — like Perl, which raw-loads the real `.sty` — we absorb the
61/// macros' args as no-ops so the surrounding context parses cleanly instead of
62/// digesting their bodies (e.g. a `label=$$` would otherwise cascade into
63/// `expected:$` display-math errors; witness 1003.1620 feynmp, Rust 28 / Perl 0).
64pub(crate) fn feynmf_diagram_stubs() -> Result<()> {
65  def_macro_noop("\\fmf{}{}")?;
66  def_macro_noop("\\fmfv{}{}")?;
67  def_macro_noop("\\fmfset{}{}")?;
68  def_macro_noop("\\fmflabel{}{}")?;
69  // Vertex-placement and decoration macros (1-arg vertex lists / 0-arg).
70  def_macro_noop("\\fmfleft{}")?;
71  def_macro_noop("\\fmfright{}")?;
72  def_macro_noop("\\fmftop{}")?;
73  def_macro_noop("\\fmfbottom{}")?;
74  def_macro_noop("\\fmfsurround{}")?;
75  def_macro_noop("\\fmfdot{}")?;
76  def_macro_noop("\\fmfblob{}{}")?;
77  def_macro_noop("\\fmffreeze")?;
78  def_macro_noop("\\fmfcmd{}")?;
79  def_macro_noop("\\fmfpen{}")?;
80  Ok(())
81}

83/// The graph's `\fmfkeep` (feynmf.sty:182 `\def\fmfkeep##1{\fmf@keep{#1}{#2}{##1}}`): `keeper` (`\lx@fmf@keep` or
84/// `\lx@fmf@keepstar`, which pass the graph's `{<width>}{<height>}`, `\lx@fmf@size`), local to the graph's group.
85fn fmf_keep_in_graph(whatsit: &mut Whatsit, keeper: &str) -> Result<()> {
86  let mut size = vec![T_BEGIN!()];
87  if let Some(width) = whatsit.get_arg(2) {
88    size.extend(width.revert()?.unlist());
89  }
90  size.extend([T_END!(), T_BEGIN!()]);
91  if let Some(height) = whatsit.get_arg(3) {
92    size.extend(height.revert()?.unlist());
93  }
94  size.push(T_END!());
95  def_macro(T_CS!("\\lx@fmf@size"), None, Tokens::new(size), None)?;
96  let_i(&T_CS!("\\fmfkeep"), &T_CS!(keeper), None);
97  Ok(())
98}
