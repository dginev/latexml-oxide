1//! delarray.sty — array delimiter package.
2//!
3//! delarray.sty (D. Carlisle, 1991-1994, tools bundle) lets `array` take a
4//! delimiter pair around its column spec: `\begin{array}({cc})…\end{array}`,
5//! `\begin{array}[t]\{{lL}.`, `|{cc}|`. It hooks the kernel entry chain
6//! (delarray.sty:43-58): `\@tabarray` → `\@@array[pos]` peeks with
7//! `\@ifnextchar\bgroup` — a brace means the plain `\@array[pos]{cols}`,
8//! anything else is `\@del@array[pos]<left>{cols}<right>`, which sets
9//! `\left<left>` … `\right<right>` around the array (`c` position; `t`/`b`
10//! lower a `\vcenter`ed box by the same visual amount).
11//!
12//! Perl LaTeXML has no `delarray.sty.ltxml`; both engines replace `\array`
13//! with their own `\array[]{}` (Perl latex_constructs.pool.ltxml:3755, Rust
14//! latex_constructs.rs `\array[]{}`) that reads the template as `#2` and never
15//! routes through `\@tabarray`, so the raw hook is dead code and the delimiter
16//! `(` is read as the column spec — every `&` then reports "Extra alignment
17//! tab" (memoir manual: 33 errors from memoir.cls:5468's
18//! `\RequirePackage{delarray}`; SHARED with Perl). The binding re-enters our
19//! `\array` machinery with the same peek and wraps the alignment in
20//! `\left…\right` (the `t`/`b` lowering is presentational).
21//!
22//! Earlier witnesses (stub era, raw `\@@array` clobbering ours): canvas-3
23//! 0809.4328, 0810.2088, 0810.2091, 0811.2514, 0811.4484, 0812.1967,
24//! 0901.2107, 0901.3167.
25use crate::prelude::*;
26
27LoadDefinitions!({
28  RequirePackage!("array");
29  // delarray.sty:43-44 — `\@tabarray`/`\@@array[pos]` peek for a brace. Outside math the array is its own formula,
30  // as the kernel `\array` is (sect10.rs `\lx@array@text@open`; memoir.cls:5468 loads delarray), its `\right`
31  // inside it.
32  DefMacro!(
33    "\\array",
34    r"\@ifnextchar[{\lx@delarray@pos}{\lx@delarray@pos[c]}"
35  );
36  DefMacro!(
37    "\\lx@delarray@pos[]",
38    r"\lx@array@text@open{#1}\@ifnextchar\bgroup{\let\@arrayright\relax\lx@delarray@plain[#1]}{\lx@delarray@del[#1]}"
39  );
40  DefMacro!(
41    "\\lx@delarray@plain[]{}",
42    r"\@array@bindings[#1]{#2}\@@array[#1]{#2}\lx@begin@alignment"
43  );
44  // delarray.sty:45-58 `\@del@array[pos]<left>{cols}<right>`.
45  DefMacro!(
46    "\\lx@delarray@del[] Token {} Token",
47    r"\def\@arrayright{\right#4}\left#2\lx@delarray@plain[#1]{#3}"
48  );
49  DefMacro!(
50    "\\endarray",
51    None,
52    r"\lx@end@alignment\@end@array\@arrayright\lx@array@text@close"
53  );
54  DefMacro!("\\@arrayright", None, r"\relax");
55  // array.sty's `\tabular` enters the same `\@tabarray` (array.sty:291, :297-310 `\@tabular` … `\@tabarray`), so delarray's
56  // peek delimits a `tabular` too: `\begin{tabular}[t]\{{l}.` (1304.5109, 1106.2670). The kernel's tabular reads its
57  // `[pos]{cols}` in `\lx@@tabular@args` (sect10.rs, locked as there): a brace is the plain template; else the
58  // delimiters go around the tabular in a formula, as array.sty's `$`-built tabular has them (`\@arrayleft` before
59  // the `\vcenter`, `\@arrayright` after it), `\endtabular` closing both. A plain tabular forgets an enclosing
60  // delimited one's right delimiter and formula. Perl has no delarray binding and its `\tabular` never reaches
61  // `\@tabarray`, so the delimiter is read as the template (KNOWN_PERL_ERRORS #1096, OXIDIZED_DESIGN_DIVERGENCES #858).
62  // Repro alignment-bindings/delarray_delimits_a_tabular.
63  DefMacro!(
64    "\\lx@@tabular@args[]",
65    r"\@ifnextchar\bgroup{\let\@arrayright\relax\let\lx@array@text@close\relax\lx@delarray@tabular[#1]}{\lx@delarray@tabular@del[#1]}",
66    locked => true
67  );
68  DefMacro!(
69    "\\lx@delarray@tabular[]{}",
70    r"\@tabular@bindings{#2}[vattach=#1]\@@tabular[#1]{#2}\lx@begin@alignment\@tabular@before"
71  );
72  DefMacro!(
73    "\\lx@delarray@tabular@del[] Token {} Token",
74    r"\def\@arrayright{\right#4}\lx@delarray@tabular@open\left#2\lx@delarray@tabular[#1]{#3}"
75  );
76  // array.sty:297-301: the tabular is set in math of its own (`\leavevmode\hbox\bgroup$`), at any position — no
77  // `\vcenter` outside math, so not the array's "Missing $ inserted" (`\lx@array@text@open`).
78  DefMacro!(
79    "\\lx@delarray@tabular@open",
80    r"\ifmmode\let\lx@array@text@close\relax\else\leavevmode\lx@begin@inline@math\let\lx@array@text@close\lx@end@inline@math\fi"
81  );
82  DefMacro!(
83    "\\endtabular",
84    r"\@tabular@after\lx@end@alignment\@end@tabular\lx@raw@array@close\@arrayright\lx@array@text@close",
85    locked => true
86  );
87});
