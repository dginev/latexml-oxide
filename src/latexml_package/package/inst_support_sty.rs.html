5/// inst_support: an `\inst` after a comma is the name's before it. It owns the `\author` of OmniBus, which makes `\alignauthor` a no-op
6/// (OmniBus.cls.ltxml:76) that parts the columns of what the class stands in for (sig-alternate.cls:283, sigchi.cls), as nothing
7/// printed tells: its column break.

9  name: "inst_support",
10  binding: "inst_support.sty",
11  inst_after_comma: true,
12  cite: "Perl inst_support.sty.ltxml:33-35; OD #807 (2401.16957)",
13  block_breaks: &[("\\alignauthor", FmBreakKind::Column)],

18LoadDefinitions!({
19  // (this binding's frontmatter conventions, first: OXIDIZED_DESIGN_DIVERGENCES #837)
20  declare_frontmatter_conventions(&FRONTMATTER_CONVENTIONS);
21  // Perl: inst_support.sty.ltxml (PR #2767)
22  // This bit of code supports the \inst style institution markup
23  // used by several document classes and styles [aa, llncs, sv]
24  // HOWEVER, the care support is now built into Engine/Base_Utility
25  // so this shouldn't be needed.
26
27  // In inst style, the \author (used once for each author)
28  // gets some "labels" which are used to connect to \affiliation
29  // (also one per affiliation); the affiliation with the matching label
30  // is attached to the author.
31
32  // \author[marks]{author}
33  // Perl inst_support.sty.ltxml L33 defines `\author{}` — a SINGLE-arg macro
34  // that DROPS the optional `[marks]` its own comment documents, and whose
35  // `\lx@clear@creators` wipes prior authors on every call. A class using the
36  // inst convention of one `\author[label]{name}` PER author (ifacconf:
37  // `\author[First]{Eryn Vaid}` ×4) then reads the literal `[` as the name and
38  // keeps only the last — Perl and Rust both emit a single `[` personname
39  // (shared upstream bug; witness arXiv:2605.00004, whose pdflatex PDF lists all
40  // four authors). Surpass-Perl (OXIDIZED_DESIGN #53):
41  //   * accept the optional `[marks]` so `[` is never mistaken for the name,
42  //   * take the name from #2, the whole list one `\lx@add@author@split` call the reading cuts at its `\and`s (the
43  //     `Sep(And)` events the rehearsal digests: a macro-held one too, which Perl's split by spelling missed; P2-1,
44  //     OD #993), then each piece at its commas, " and " and " \& " (`R. Braun\inst{1} and W. B. Burton\inst{2}`,
45  //     astro-ph9810433; 63e),
46  //   * ACCUMULATE across calls — drop the per-call `\lx@clear@creators`, which
47  //     is a no-op on the first call anyway, so single-`\author` classes are
48  //     unaffected. (llncs, sv and aa define their own `\author` — aa's the kernel's
49  //     replacing one aa.cls keeps, OXIDIZED_DESIGN #922. Restoring Perl's clear
50  //     here (92b) changed 162 of 1,305 OmniBus papers with two or more `\author`
51  //     calls, most of them stacking classes reached through OmniBus — achemso,
52  //     autart, pasj00, lipics, the AMS `-l` classes, oup, a subfile of a revtex
53  //     main —, so the clear stays dropped until each binding says its class's:
54  //     OXIDIZED_DESIGN #920.)
55  // The `[marks]` (author↔affiliation label) are dropped, matching Perl's own
56  // handling; wiring them to the affiliation annotation is a separate follow-up.
57  DefMacro!("\\author[]{}",
58    "\\lx@add@author@split{#2}\\lx@author@trailing");
59  // \institute{list}: one `labelseq=affiliation` contact call per institute (Perl L36-38 splits by spelling at
60  // `\and`/`\And`), cut at the kernel's `\and` by meaning as well — a macro holding only the separator, a `\let` copy
61  // where the document renewed `\and` — which the spelling missed, so a piece holding an `\email` is one call of its
62  // own, not one call the emitter writes whole for every author (Option C P2-2, OXIDIZED_DESIGN_DIVERGENCES #995;
63  // `\lx@splitting@by@meaning`, the cut eptcs's cells share). A separator the spelling cut keeps its cut whatever it
64  // means (a renewed `\and`); one a macro holds together with institutes stays in its piece, which the reading cuts.
65  DefMacro!("\\institute{}",
66    "\\lx@clear@frontmatter{ltx:contact}[role=affiliation]\\lx@splitting@by@meaning{\\lx@add@contact[role=affiliation,labelseq=affiliation]}{\\and\\And}{#1}");
67  DefMacro!("\\inst{}", "\\lx@request@frontmatter@annotation[affiliation]{#1}");
68
69  // \and variants — Perl L41-45
70  Let!("\\at", "\\and"); // Actually this is different than \and, but...
71  Let!("\\iand", "\\and");
72  Let!("\\nand", "\\and");
73  Let!("\\lastand", "\\and");
74  Let!("\\AND", "\\and");
75});
