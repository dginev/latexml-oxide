89/// Perl `save_psfrag` (L34-39): appends to the saved fragments, NOT globally, the leading control sequence deferred.
90fn save_psfrag(stuff: Vec<Token>) {
91  let mut saved = lookup_tokens("saved_psfragments")
92    .map(Tokens::unlist)
93    .unwrap_or_default();
94  saved.push(T_CS!("\\noexpand"));
95  saved.extend(stuff);
96  assign_value("saved_psfragments", Tokens::new(saved), None);
97}

99/// The hooks' `afterDigest` (Perl L82-87, L105-110, L127-132) for a graphic `whatsit` of any including command
100/// (`\includegraphics`, `\epsfbox` and its `\let` copies, `\psfig`). An EPS graphic (`eps_file`) is where psfrag
101/// places the saved fragments (psfrag.sty:46-62 `\Ginclude@eps`/`\epsfbox`) and, under a scan, the `(\\tex` strings
102/// of the file (psfrag.sty:152-209 `\pfg@scan`); they are not rendered (PSFR), so the graphic is one Warning naming
103/// each of them. The scan stays on until `\psfragscanoff` (psfrag.sty:24), where Perl resets it per graphic
104/// (OXIDIZED_DESIGN_DIVERGENCES #1017).
105pub fn psfrag_graphic(whatsit: &Whatsit) {
106  let Some(saved) = lookup_tokens("saved_psfragments") else {
107    return;
108  };
109  let tags = fragment_tags(&saved);
110  let scan = lookup_bool("psfrag_scan") || lookup_bool("psfrag_scan_all");
111  if tags.is_empty() && !scan {
112    return;
113  }
114  let mut graphic = whatsit.get_property_string("graphic");
115  if graphic.is_empty() {
116    graphic = whatsit.get_property_string("path");
117  }
118  let Some(eps) = eps_file(&whatsit.get_property_string("candidates"), graphic.trim()) else {
119    return;
120  };
121  let quote = |text: &String| s!("'{text}'");
122  let mut dropped: Vec<String> = tags.iter().map(quote).collect();
123  let mut count = tags.len();
124  if scan {
125    let content = eps.unwrap_or_else(|file| std::fs::read(file).unwrap_or_default());
126    let embedded = embedded_fragments(&content);
127    if !embedded.is_empty() {
128      count += embedded.len();
129      let embedded: Vec<String> = embedded.iter().map(quote).collect();
130      dropped.push(s!("the file's own \\tex labels {}", embedded.join(", ")));
131    }
132  }
133  let (noun, verb) = match count {
134    0 => return,
135    1 => ("replacement", "is"),
136    _ => ("replacements", "are"),
137  };
138  let dropped = dropped.join(", ");
139  Warn!(
140    "missing_feature",
141    "psfrag",
142    s!("The psfrag {noun} of {dropped} in the graphic '{graphic}' {verb} not rendered")
143  );
144}

146/// The `(\\tex…)` strings `\pfg@scan` reads from an EPS file (psfrag.sty:152-209), each as TeX reads it: the
147/// PostScript string from `\tex` to its closing parenthesis, its `\\`, `\(` and `\)` escapes undone.
148fn embedded_fragments(content: &[u8]) -> Vec<String> {
149  let mut found = Vec::new();
150  let mut at = 0;
151  while let Some(start) = content[at..].windows(6).position(|w| w == br"(\\tex") {
152    let mut text = Vec::new();
153    let mut depth = 0usize;
154    let mut bytes = content[at + start + 1..].iter();
155    while let Some(&b) = bytes.next() {
156      match b {
157        b'\\' => text.extend(bytes.next()),
158        b'(' => {
159          depth += 1;
160          text.push(b);
161        },
162        b')' if depth == 0 => break,
163        b')' => {
164          depth -= 1;
165          text.push(b);
166        },
167        _ => text.push(b),
168      }
169    }
170    at += start + 6;
171    found.push(String::from_utf8_lossy(&text).into_owned());
172  }
173  found
174}

176/// The EPS file the graphic includes: the first candidate that is PostScript ([`is_postscript`]) (`Err`, its path),
177/// else a `{filecontents}` file of the graphic's name or that name with dvips.def's `.eps`, `.ps` (`Ok`, its content).
178fn eps_file(
179  candidates: &str,
180  graphic: &str,
181) -> Option<std::result::Result<Vec<u8>, std::path::PathBuf>> {
182  let source = std::path::PathBuf::from(lookup_string("SOURCEDIRECTORY"));
183  let found = candidates
184    .split(',')
185    .map(str::trim)
186    .filter(|c| !c.is_empty())
187    .map(|c| source.join(c))
188    .find(|path| {
189      let mut header = [0u8; 4];
190      let read =
191        std::fs::File::open(path).and_then(|mut f| std::io::Read::read(&mut f, &mut header));
192      is_postscript(
193        path.to_string_lossy().as_ref(),
194        &header[..read.unwrap_or(0)],
195      )
196    });
197  if let Some(path) = found {
198    return Some(Err(path));
199  }
200  [graphic.to_string(), s!("{graphic}.eps"), s!("{graphic}.ps")]
201    .iter()
202    .find_map(|name| vfs_read(name).filter(|text| is_postscript(name, text.as_bytes())))
203    .map(|text| Ok(text.into_bytes()))
204}

206/// Whether psfrag places fragments in the file `name` with this `header`: a PostScript header (`%!`, or the DOS EPS
207/// `C5D0D3C6`), as Perl's `psfrag_requirepicture` asks `image_type` (Util/Image.pm:74-84) — dvips's catch-all
208/// `\Gin@rule@*` makes any file EPS, and the header is what it can place them in (`.epsi`, `.epsf`: hep-ph0610439) —
209/// or a graphics rule of type `eps` for its extension, which a compressed file needs (`\DeclareGraphicsRule{.eps.gz}`,
210/// dvips.def:203-210; psfrag scans its compressed bytes, finding no `\tex` label).
211fn is_postscript(name: &str, header: &[u8]) -> bool {
212  const DVIPS_EPS_RULES: [&str; 8] = [
213    ".ps", ".eps", ".mps", ".pz", ".eps.Z", ".ps.Z", ".ps.gz", ".eps.gz",
214  ];
215  if header.starts_with(b"%!") || header.starts_with(b"\xc5\xd0\xd3\xc6") {
216    return true;
217  }
218  let base = name.rsplit('/').next().unwrap_or(name);
219  base.match_indices('.').any(|(at, _)| {
220    let ext = &base[at..];
221    match lookup_definition(&T_CS!(s!("\\Gin@rule@{ext}"))) {
222      Ok(Some(rule)) => matches!(rule.get_expansion(), Some(ExpansionBody::Tokens(body)) if rule_type(body) == "eps"),
223      _ => DVIPS_EPS_RULES.contains(&ext),
224    }
225  })
226}

228/// The `<type>` of a graphics rule's body `{<type>}{<read-file>}{<command>}` (graphics.sty:150-154).
229fn rule_type(body: &Tokens) -> String {
230  let tokens = body.unlist_ref();
231  match tokens.first() {
232    Some(t) if t.get_catcode() == Catcode::BEGIN => {
233      let inner: Vec<Token> = tokens[1..]
234        .iter()
235        .take_while(|t| t.get_catcode() != Catcode::END)
236        .cloned()
237        .collect();
238      Tokens::new(inner).untex().trim().to_string()
239    },
240    _ => String::new(),
241  }
242}

244/// The tag of each saved `\lx@delayed@psfrag` invocation: its first braced argument, after an optional `*`.
245fn fragment_tags(saved: &Tokens) -> Vec<String> {
246  let tokens = saved.unlist_ref();
247  let mut tags = Vec::new();
248  for (i, t) in tokens.iter().enumerate() {
249    if t.get_catcode() != Catcode::CS || t.with_str(|s| s != "\\lx@delayed@psfrag") {
250      continue;
251    }
252    let mut rest = tokens[i + 1..]
253      .iter()
254      .skip_while(|t| t.get_catcode() == Catcode::OTHER && t.with_str(|s| s == "*"));
255    if rest
256      .next()
257      .is_none_or(|t| t.get_catcode() != Catcode::BEGIN)
258    {
259      continue;
260    }
261    let mut depth = 0usize;
262    let tag: Vec<Token> = rest
263      .take_while(|t| match t.get_catcode() {
264        Catcode::BEGIN => {
265          depth += 1;
266          true
267        },
268        Catcode::END if depth == 0 => false,
269        Catcode::END => {
270          depth -= 1;
271          true
272        },
273        _ => true,
274      })
275      .cloned()
276      .collect();
277    tags.push(Tokens::new(tag).untex());
278  }
279  tags
