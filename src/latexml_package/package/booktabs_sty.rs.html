150/// How a booktabs rule width reads as a TeX `<dimen>` (tex.web §448-458).
151enum WidthShape {
152  /// A factor and a unit (`.5pt`, `2.5\arrayrulewidth`), or a register or macro (`\heavyrulewidth`).
153  Dimension,
154  /// A number with no unit (`0`): "Illegal unit of measure (pt inserted)".
155  NoUnit,
156  /// No number (`lr`, a typo for `(lr)`): "Missing number, treated as zero".
157  NoNumber,
158}
159
160fn width_shape(width: &str) -> WidthShape {
161  static DIMENSION: Lazy<Regex> = Lazy::new(|| {
162    Regex::new(
163      // TeX's unit keywords ignore case (tex.web §407 scan_keyword: `\toprule[1PT]`).
164      r"(?i)^\s*[-+\s]*(?:(?:\d+[.,]?\d*|[.,]\d+)\s*(?:true\s*)?(?:pt|pc|in|bp|cm|mm|dd|cc|sp|em|ex|px|nd|nc)|(?:\d+[.,]?\d*|[.,]\d+)?\s*\\[A-Za-z@])",
165    )
166    .unwrap()
167  });
168  static NUMBER: Lazy<Regex> = Lazy::new(|| Regex::new(r"^\s*[-+\s]*(?:\d|[.,]\d)").unwrap());
169  if DIMENSION.is_match(width) {
170    WidthShape::Dimension
171  } else if NUMBER.is_match(width) {
172    WidthShape::NoUnit
173  } else {
174    WidthShape::NoNumber
175  }
