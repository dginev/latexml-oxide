1//! The markup the VTeX IMS journal classes share: arximspdf.cls / arxstspdf.cls (Annals of Probability/Statistics)
2//! and arxbj.cls (Bernoulli) define the same structured bibliography (`{barticle}`/… with `\b*` fields; arxbj.cls:
3//! 2599-2897), the same `{pf}` proof and automatic `\qed` (arximspdf.cls:1417-1434 = arxbj.cls:1428-1445) and the same
4//! `\tablewidth` table notes (arximspdf.cls:1729-1790, arxbj.cls:1665-1809). Their bindings load this beside their own
5//! options, math and frontmatter.
6//!
7//! Bibliography note: these classes use `natbib` + `\bibitem`/`thebibliography`, so a bibitem produces `ltx:bibitem >
8//! ltx:bibblock` (Flow.model). The structured `ltx:bib-*` vocabulary is schema-valid only inside `ltx:bibentry` (the
9//! BibTeX path), NOT inside `bibblock`, so the `\b*` field macros are PASSTHROUGH text (readable + schema-valid), not
10//! mapped to `ltx:bib-*`.

143/// The `{keyword}[class=<class>]` block with `\kwd[<sep>]{<text>}` entries that the VTeX classes share: arxbj.cls
144/// (2490-2541 of 1106.2467's copy), arximspdf.cls (2404-2450 of 1107.3284's) and imsart.sty (1518-1567 of 1410.4462's
145/// copy, 2334-2393 of 2605.27137's). The entries are joined by `\sep` (none before the first, an explicit `[<sep>]` in
146/// its place: `\kwd[Primary ]{60F05}\kwd[; secondary ]{62G20}`), headed by the class's `\keyword@<class>`
147/// (`\setattribute{keyword}{KWD}{Keywords:}`, imsart.sty's `\setpkgattr{keyword}{MSC2020}{…}`; the caller sets the
148/// headings and `\sep`, or the class file it reads after does). The blocks add up, as the classes keep each (arxbj's and
149/// imsart's `\keyword@box`, arximspdf's `\keyword@toks`): a keywords block per `{keyword}`, and a block of a subject
150/// classification class — `AMS`, `MSC`, `MSC2010`, `MSC2020` (imsart.sty:1521-1525, :2338-2341) — the paper's
151/// classification under that scheme. Each `\kwd` had started the list afresh (elsart's `{keyword}`: arxbj 1106.2467,
152/// 0811.0643 lost every entry; OmniBus's `\kwd`, KNOWN_PERL_ERRORS #1196: imsart 1410.4462, 2407.13970, 2605.27137 kept
153/// only their last keyword) or a second block replaced the first (arximspdf's: the AMS codes of 1107.3284 and
154/// 1309.4864), and imsart.sty's own blocks set a box nothing typesets (the MSC codes of the three imsart papers).
155/// The classification's scheme is its printed heading, edition included, without the closing ":" or "." its name
156/// keeps (`\lx@ims@scheme`): the HTML shows the scheme as the classification's heading, its name only when it has none
157/// (LaTeXML-structure-xhtml.xsl:239-240, as Perl's), so the class key alone showed "MSC" where pdflatex prints "MSC 2010
158/// subject classifications:" (1410.4462) or "MSC2020 subject classifications." (2605.27137, 2407.13970), as the AMS
159/// classes' `\subjclass` gives "2020 Mathematics Subject Classification". A class without a heading for the key
160/// (arxbj's `class=MSC`) keeps the key. Needs `keyval`.
161pub fn vtex_keyword_markup() -> Result<()> {
162  DefMacro!(T_CS!("\\begin{keyword}"), None, "\\begingroup\\keyword");
163  DefMacro!(T_CS!("\\end{keyword}"), None, "\\endkeyword\\endgroup");
164  DefMacro!("\\lx@ims@scheme", {
165    let body = |cs: &str| -> Result<Vec<Token>> {
166      Ok(match lookup_definition(&T_CS!(cs))? {
167        Some(defn) => match defn.get_expansion() {
168          Some(ExpansionBody::Tokens(body)) => body.clone().unlist(),
169          _ => Vec::new(),
170        },
171        None => Vec::new(),
172      })
173    };
174    let class = body("\\keyword@class")?;
175    let mut heading = body(&s!("\\keyword@{}", Tokens::new(class.clone())))?;
176    // the heading's last character, inside the groups that close it (`\ims@href{\msc@url}{…}:`, `{\it …:}`)
177    let last = heading
178      .iter()
179      .rposition(|t| !matches!(t.get_catcode(), Catcode::SPACE | Catcode::END));
180    if let Some(i) = last
181      && heading[i].get_catcode() == Catcode::OTHER
182      && heading[i].with_str(|s| s == ":" || s == ".")
183    {
184      heading.remove(i);
185    }
186    while heading
187      .last()
188      .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
189    {
190      heading.pop();
191    }
192    if heading.iter().all(|t| t.get_catcode() == Catcode::SPACE) {
193      heading = class;
194    }
195    Ok(Tokens::new(heading))
196  });
197  RawTeX!(
198    r"\def\keyword{\@ifnextchar[\lx@ims@keyword{\lx@ims@keyword[class=KWD]}}
199\define@key{lx@ims@keyword}{class}{\def\keyword@class{#1}}
200\@namedef{lx@ims@scheme@AMS}{}\@namedef{lx@ims@scheme@MSC}{}\@namedef{lx@ims@scheme@MSC2010}{}\@namedef{lx@ims@scheme@MSC2020}{}
201\def\lx@ims@keyword[#1]{\def\keyword@class{KWD}\setkeys{lx@ims@keyword}{#1}\let\kwd@sep\@empty
202\@ifundefined{lx@ims@scheme@\keyword@class}
203{\lx@add@frontmatter@until{ltx:keywords}[accumulate=true,name={\csname keyword@\keyword@class\endcsname~}]{\lx@end@keywords}}
204{\lx@add@frontmatter@until{ltx:classification}[scheme={\lx@ims@scheme},name={\csname keyword@\keyword@class\endcsname~}]{\lx@end@keywords}}}
205\def\endkeyword{\lx@end@keywords}
206\def\sep{\unskip, }
207\DeclareRobustCommand*\kwd{\@ifnextchar[\@kwd{\@kwd[\kwd@sep]}}
208\def\@kwd[#1]#2{\unskip#1{#2}\let\kwd@sep\sep}"
209  );
210  Ok(())
211}

213/// `\href{<base><arg>}{<arg>}`, or `\href{<arg>}{<arg>}` when the argument is already a full URL (it contains `full`:
214/// `http://` for `\doiurl`/`\arxivurl`, `://` for `\surl`), dropping the `\\` break hints.
215fn class_url_link(base: &'static str, full: &'static str, arg: Tokens) -> Result<Tokens> {
216  let text: Vec<Token> = arg
217    .unlist()
218    .into_iter()
219    .filter(|t| *t != T_CS!("\\\\"))
220    .collect();
221  let full = Tokens::new(text.clone()).untex().contains(full);
222  let mut out = vec![T_CS!("\\href"), T_BEGIN!()];
223  if !full {
224    out.extend(Tokenize!(base).unlist());
225  }
226  out.extend(text.iter().copied());
227  out.extend([T_END!(), T_BEGIN!()]);
228  out.extend(text);
229  out.push(T_END!());
230  Ok(Tokens::new(out))
