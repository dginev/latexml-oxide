1//! lua-widow-control.sty — LuaTeX widow/orphan control. Page-layout only, and
2//! its engine half is Lua: `\__lwc_enable:` and friends are created by the Lua
3//! module loaded at lua-widow-control.sty:120-123, which no reader here runs, so
4//! under the luatex profile the raw package leaves them undefined for the
5//! `.code` keys (:232-238) and hooks (:174) that call them (homework-demo-*,
6//! jwjournal-demo-cn, abntexto ×2). The user surface (`\lwcsetup`,
7//! `\lwcenable`/`\lwcdisable`, `\iflwc`, `\lwcemergencystretch`,
8//! `\lwcdisablecmd`) is kept with no layout effect. Outside the luatex profile
9//! the package's own `no-luatex` critical (:68-70) is reported instead.
10//! Guard: `perfect_kernel_batch56::lua_widow_control_surface`.
11use crate::prelude::*;
12
13#[rustfmt::skip]
14LoadDefinitions!({
15  // lua-widow-control.sty:300-303 reads its options with `\ProcessKeyOptions` (or l3keys2e's), which marks them
16  // processed: `\@curroptions` stays as it was.
17  key_options_processed()?;
18  if !lookup_bool("LUATEX_PROFILE") {
19    Error!("latex", "lua-widow-control",
20      "Package lua-widow-control Error: LuaTeX is REQUIRED for lua-widow-control (lua-widow-control.sty:68)");
21    return Ok(());
22  }
23  RequirePackage!("expl3");
24  RawTeX!(r"\newif\iflwc
25\ExplSyntaxOn
26\cs_set_protected:Npn \__lwc_enable: { \lwctrue }
27\cs_set_protected:Npn \__lwc_disable: { \lwcfalse }
28\prg_set_conditional:Npnn \__lwc_if_enabled: { p, T, F, TF }
29  { \iflwc \prg_return_true: \else \prg_return_false: \fi }
30\cs_set_eq:NN \lwcenable \__lwc_enable:
31\cs_set_eq:NN \lwcdisable \__lwc_disable:
32\cs_set_protected:Npn \lwcsetup #1 { }
33\cs_set_protected:Npn \lwcemergencystretch { }
34\cs_set_protected:Npn \lwcdisablecmd #1 { }
35\ExplSyntaxOff
36\lwctrue");
37});
