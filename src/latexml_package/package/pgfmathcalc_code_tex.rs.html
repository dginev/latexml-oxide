12  // Perl L24-32: \pgfmathsetmacro — evaluates expression and defines macro.
13  // Perl's DefMacroI call explicitly passes `scope => 'local'`, so the
14  // new CS vanishes when the enclosing group closes — essential for
15  // tikz/pgf loops that reuse the same CS (e.g. \foreach \n …). The
16  // Rust port had dropped the scope, leaking redefinitions upward past
17  // the begingroup/endgroup wrapper. Restore Scope::Local to match Perl.
18  DefPrimitive!("\\pgfmathsetmacro{}{}", sub[(cs, expression)] {
19    // Perl L26-28: the parse runs in the group (pgfmath.code.tex.ltxml's `sub pgfmathparse`, as `\pgfmathparse`'s).
20    bgroup();
21    let parsed = pgfmath_code_tex::pgfmathparse(expression);
22    egroup()?;
23    let (result, _units) = parsed?;
24    let result_tokens = mouth::tokenize_internal(TeXString::assembled(result));
25    def_macro(
26      cs.unlist().into_iter().next().unwrap_or_else(|| T_CS!("\\pgfmathresult")),
27      None,
28      result_tokens,
29      Some(ExpandableOptions { scope: Some(Scope::Local), ..Default::default() }))?;
30    Ok(Vec::new())
31  });

33  // pgfmathcalc.code.tex:366-468 `\pgfmathpointintersectionoflineandarc`:
34  // pgf bisects the arc's parametric angle until the angle from #2 to the
35  // arc point EQUALS (`\ifdim\x pt=\q pt`, :447) the angle from #2 to #1 in
36  // its own fixed-point trig — the loop's only exit. Our trig is float
37  // (pgfmath_code_tex.rs `\pgfmathcos@`/`sin@`/`atantwo@`), so the two
38  // never meet exactly and a rounded-rectangle corner border query (a
39  // self-loop wire — tikz-cd `\ar[loop]`, zx-calculus `\zxLoopAboveDots`;
40  // callout nodes, arXiv 2201.09268) spun 2 boxes per iteration into the
41  // cycle fatal. The intersection is closed form: the ray from #2 toward #1
42  // meets the ellipse (center #3, radii #6) at a quadratic's roots; the root
43  // on the arc (nearest to it otherwise — pgf's "best estimate" `\n`) is the
44  // parametric angle. Same point set as the bisection's limit, no loop.
45  // Guard: `perfect_kernel_batch56::line_and_arc_intersection_is_closed_form`.
46  RawTeX!(r"\def\pgfmathpointintersectionoflineandarc#1#2#3#4#5#6{%
47  \pgf@process{%
48    \pgfmathanglebetweenpoints{#2}{#1}%
49    \let\x\pgfmathresult
50    \pgfmath@in@{and }{#6}%
51    \ifpgfmath@in@\pgf@polar@#6\@@\else\pgf@polar@#6 and #6\@@\fi
52    \edef\xarcradius{\the\pgf@x}%
53    \edef\yarcradius{\the\pgf@y}%
54    \pgfmathsetmacro\s{#4}%
55    \pgfmathsetmacro\e{#5}%
56    \pgf@process{#2}\edef\lx@lineandarc@q{\the\pgf@x,\the\pgf@y}%
57    \pgf@process{#3}\edef\lx@lineandarc@c{\the\pgf@x,\the\pgf@y}%
58    \lx@pgf@lineandarc
59    \pgfpointadd{#3}{\pgfpointpolar{\n}{\xarcradius and \yarcradius}}%
60  }%
61}");
62  DefPrimitive!("\\lx@pgf@lineandarc", sub[_args] {
63    fn num(s: &str) -> f64 { s.trim().trim_end_matches("pt").trim().parse::<f64>().unwrap_or(0.0) }
64    fn pair(s: &str) -> (f64, f64) {
65      let mut it = s.split(',');
66      (num(it.next().unwrap_or("0")), num(it.next().unwrap_or("0")))
67    }
68    let expand = |cs: &str| -> Result<String> { Ok(do_expand(Tokens!(T_CS!(cs)))?.to_string()) };
69    let x = num(&expand("\\x")?);
70    let s = num(&expand("\\s")?);
71    let e = num(&expand("\\e")?);
72    let rx = num(&expand("\\xarcradius")?).abs().max(1e-6);
73    let ry = num(&expand("\\yarcradius")?).abs().max(1e-6);
74    let (qx, qy) = pair(&expand("\\lx@lineandarc@q")?);
75    let (cx, cy) = pair(&expand("\\lx@lineandarc@c")?);
76    // pgfmathcalc.code.tex:395-406: the arc's angles mod 360, straddling zero
77    // when the end comes before the start.
78    let norm = |t: f64| t.rem_euclid(360.0);
79    let (ss, ee) = (norm(s), norm(e));
80    let on_arc = |t: f64| if ee < ss { t >= ss || t <= ee } else { ss <= t && t <= ee };
81    let arc_distance = |t: f64| {
82      if on_arc(t) { 0.0 } else {
83        let d = |a: f64| { let m = (t - a).rem_euclid(360.0); m.min(360.0 - m) };
84        d(ss).min(d(ee))
85      }
86    };
87    let (dx, dy) = (x.to_radians().cos(), x.to_radians().sin());
88    let (u, v, a, b) = ((qx - cx) / rx, (qy - cy) / ry, dx / rx, dy / ry);
89    let (qa, qb, qc) = (a * a + b * b, 2.0 * (u * a + v * b), u * u + v * v - 1.0);
90    let disc = qb * qb - 4.0 * qa * qc;
91    // The default `\n`: the middle of the arc (:412-413). A ray that MISSES
92    // the arc (the shapes' `\anchorborder` never sends one — it takes the
93    // straight edges through `\pgfpointintersectionoflines`) gets the true
94    // off-arc root nearest the arc where pgf's bisection would clamp inside
95    // [#4,#5]; both are "meaningless points" per pgf's own comment (:356).
96    let mut n = (s + e) / 2.0;
97    if disc >= 0.0 && qa > 0.0 {
98      let sq = disc.sqrt();
99      let mut best: Option<(bool, f64, f64)> = None; // (forward ray, distance to the arc, angle)
100      for lam in [(-qb + sq) / (2.0 * qa), (-qb - sq) / (2.0 * qa)] {
101        let (px, py) = (qx + lam * dx, qy + lam * dy);
102        let t = norm(((py - cy) / ry).atan2((px - cx) / rx).to_degrees());
103        let cand = (lam >= 0.0, arc_distance(t), t);
104        let better = match best {
105          None => true,
106          Some((fwd, dist, _)) => (cand.0 && !fwd) || (cand.0 == fwd && cand.1 < dist),
107        };
108        if better { best = Some(cand); }
109      }
110      if let Some((_, _, t)) = best { n = t; }
111    }
112    let text = format!("{n:.5}");
113    let text = text.trim_end_matches('0').trim_end_matches('.').to_string();
114    def_macro(
115      T_CS!("\\n"),
116      None,
117      mouth::tokenize_internal(TeXString::assembled(text)),
118      Some(ExpandableOptions { scope: Some(Scope::Local), ..Default::default() }))?;
119    Ok(Vec::new())
120  });
