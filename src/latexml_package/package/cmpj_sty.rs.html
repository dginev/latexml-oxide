1//! cmpj.sty (Condensed Matter Physics, Lviv), with cmpj2.sty and cmpj3.sty — Rust-only bindings (Perl has none and
2//! reads the package raw only under `--includestyles`, KNOWN_PERL_ERRORS #1233).
3use crate::prelude::*;
4
5LoadDefinitions!({
6  load_cmpj_style("cmpj")?;
7});
8
9/// The real style `name` (the paper's copy first), raw, but `\refaddr`'s address marks print as TeX's second run does.
10/// `\@@refadx` prints `\ref{key}` only once `r@key` is defined from the .aux, a bold "?" before (cmpj.sty:446-456,
11/// cmpj2.sty:560-570, cmpj3.sty:659-669); LaTeXML resolves `\ref` itself and never defines `r@` (`\newlabel` is a no-op,
12/// latex_constructs.pool.ltxml:5748), so the names showed "?" (1212.6367, 1212.6137, 1603.02437, 2112.09940).
13pub fn load_cmpj_style(name: &str) -> Result<()> {
14  // (read as the package itself, with the document's options, so its stores reach the frontmatter after it:
15  // `\lx@package@loaded@raw`)
16  let options: Vec<String> = lookup_vecdeque(&s!("opt@{name}.sty"))
17    .map(|v| v.iter().map(|o| o.to_string()).collect())
18    .unwrap_or_default();
19  InputDefinitions!(name, noltxml => true, extension => Some(Cow::Borrowed("sty")), handleoptions => true,
20    options => options);
21  DefMacro!(
22    r"\@@refadx{}",
23    r"\let\@@refada\@empty\@@refad{\@for\@@refadb:=#1\do{\@@refada\def\@@refada{,\penalty\@m}\edef\@@refadb{\expandafter\@firstofone\@@refadb\@empty}\hbox{\ref{\@@refadb}}}}"
24  );
25  Ok(())
26}
