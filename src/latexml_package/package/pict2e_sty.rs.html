1//! pict2e.sty — extended LaTeX picture environment.
2//!
3//! pict2e enhances the standard LaTeX picture environment with arbitrary-
4//! slope `\line` / `\vector`, smooth curves, quadratic/cubic Bezier
5//! handling and a PostScript-like path interface. LaTeX picture is rendered
6//! as XML/SVG in our pipeline regardless of driver, so the driver-dispatch
7//! chain (p2e-pdftex.def, p2e-dvips.def, …) has no XML meaning and is not
8//! raw-loaded; the user-facing commands are bound here.
9//!
10//! Under `\pdfoutput=0` (pdftex.rs:11, and Perl) pict2e.cfg would pick the
11//! dvips driver and pict2e.sty:160 `\ifnum\pIIe@mode>\z@` would define the
12//! path interface `\moveto`/`\lineto`/`\curveto`/`\circlearc`/`\closepath`/
13//! `\strokepath`/`\fillpath` (pict2e.sty:742-774). An earlier stub skipped
14//! them ("No suitable driver" was a stale reading), so fancyqr-doc and
15//! curve2e-manual (curve2e.sty:19 requires pict2e, :92 `\let\originalmoveto
16//! \moveto`) failed on `undefined:\moveto` while same-host Perl (raw load)
17//! converted with an empty picture. Here the path is accumulated per subpath
18//! and stroked/filled as `ltx:line` polylines (curves and arcs sampled), which
19//! renders what Perl drops. Witness 2503.14673 (pict2e error blocking 1 paper).

22/// The graph being built: what `\pIIe@GRAPH` holds. A global token register (pict2e.sty:186 `\global\pIIe@GRAPH`, :203
23/// emptied by a stroke or a fill alone), so a path left unstroked is stroked with the next one, in another picture too, as
24/// pdflatex prints it (`0 0 m 1 1 l 2 2 m 3 3 l S`), and it is state a rollback puts back: a rehearsal's path, stroked once
25/// more at the body's `\strokepath`, gave one more line on and off alike.
26#[derive(Clone, Default)]
27struct Graph {
28  /// One point list per subpath (`\moveto` opens one), in `\unitlength` multiples. `\strokepath`/`\fillpath` consume it.
29  path:     Vec<Vec<(f64, f64)>>,
30  /// The points `\pIIe@add@nums`/`\pIIe@add@CP` put on the graph ahead of the operator `\pIIe@addtoGraph` adds
31  /// (pict2e.sty:206-217), in `\unitlength` multiples.
32  operands: Vec<(f64, f64)>,
33}
34
35static GRAPH: DocumentState<Graph> = DocumentState::new("pict2e_graph");

37/// `\unitlength` in sp, exact, as `{picture}`'s coordinates are scaled by it.
38fn unitlength_sp() -> Result<i64> { latexml_engine::base_utilities::unitlength_sp() }
39
40/// An angle argument (`\pIIe@arc … {start}{end}`): a plain or macro-held number.
41fn pic_number(tokens: Tokens) -> Result<f64> {
42  let text = Expand!(tokens).to_string();
43  Ok(text.trim().parse().unwrap_or(0.0))
44}
45
46/// A driver-level `\pIIe@*` argument is a real dimension (pict2e.sty:267-308
47/// `\pIIe@add@CP{#1}{#2}`): convert to `\unitlength` multiples.
48fn pic_dim(d: Dimension) -> Result<f64> {
49  let unit = unitlength_sp()?;
50  Ok(if unit == 0 {
51    0.0
52  } else {
53    d.value_of() as f64 / unit as f64
54  })
55}

57fn path_push(point: (f64, f64), new_subpath: bool) {
58  GRAPH.update(|graph| {
59    let p = &mut graph.path;
60    if new_subpath || p.is_empty() {
61      p.push(Vec::new());
62    }
63    p.last_mut().expect("subpath").push(point);
64  });
65}
66
67fn path_current() -> Option<(f64, f64)> {
68  GRAPH.read(|graph| graph.path.last().and_then(|s| s.last().copied()))
69}
70
71/// Cubic Bezier from the current point, sampled into the current subpath.
72fn path_curveto(c1: (f64, f64), c2: (f64, f64), end: (f64, f64)) {
73  let (x0, y0) = path_current().unwrap_or((0.0, 0.0));
74  for i in 1..=8 {
75    let t = f64::from(i) / 8.0;
76    let u = 1.0 - t;
77    let x = u * u * u * x0 + 3.0 * u * u * t * c1.0 + 3.0 * u * t * t * c2.0 + t * t * t * end.0;
78    let y = u * u * u * y0 + 3.0 * u * u * t * c1.1 + 3.0 * u * t * t * c2.1 + t * t * t * end.1;
79    path_push((x, y), false);
80  }
81}

83/// `\pIIe@arc[mode]{cx}{cy}{r}{start}{end}` (pict2e.sty:615-660): mode 0 = `\lineto` the arc's start from the
84/// current point, 1 = `\moveto` it (a new subpath), 2 = the arc continues the current subpath (`\ifcase #1
85/// \pIIe@lineto \or \pIIe@moveto \or \fi`, :643-646; the pict2e manual's `\circlearc[N]`). `\arc*`'s sector is the
86/// mode 0 arc after a `\moveto` to the centre (:670-676). Angles in degrees, sampled every ≤10°.
87fn path_arc(mode: i64, (cx, cy, r): (f64, f64, f64), start: f64, end: f64) {
88  let sweep = end - start;
89  let steps = ((sweep.abs() / 10.0).ceil() as usize).max(1);
90  for i in 0..=steps {
91    let a = (start + sweep * (i as f64) / (steps as f64)).to_radians();
92    let point = (cx + r * a.cos(), cy + r * a.sin());
93    if i == 0 && mode == 2 && path_current().is_some() {
94      continue;
95    }
96    path_push(point, i == 0 && mode == 1);
97  }
98}

100fn path_close() {
101  GRAPH.update(|graph| {
102    if let Some(sub) = graph.path.last_mut()
103      && let Some(first) = sub.first().copied()
104    {
105      sub.push(first);
106    }
107  });
108}
109
110/// Emit one `\lx@pic@polyline{}{closed}(x,y)…` per subpath and clear the path. A closed polyline returns to its
111/// start itself, so a subpath already closed by `\pIIe@closepath` (`\arc*`'s sector) drops its closing point.
112fn path_flush(closed: &str) -> Result<Tokens> {
113  let mut subpaths = GRAPH.update(|graph| std::mem::take(&mut graph.path));
114  let mut out = Vec::new();
115  if closed != "0" {
116    for sub in &mut subpaths {
117      if sub.len() > 2 && sub.first() == sub.last() {
118        sub.pop();
119      }
120    }
121  }
122  for sub in subpaths.iter().filter(|s| s.len() > 1) {
123    let mut pairs = String::new();
124    for (x, y) in sub {
125      pairs.push_str(&format!("({x},{y})"));
126    }
127    out.extend(
128      Invocation!(T_CS!("\\lx@pic@polyline"), vec![
129        Tokens!(),
130        Tokens::new(ExplodeText!(closed))
131      ])
132      .unlist(),
133    );
134    out.extend(ExplodeText!(&pairs));
135  }
136  Ok(Tokens::new(out))
137}
