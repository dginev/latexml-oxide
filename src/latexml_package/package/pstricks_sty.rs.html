1//! pstricks.sty — PSTricks graphics package
2//! We raw-load the real pstricks.sty to set up its internal state
3//! (\ifpst@useCalc, \ifpst@psfonts, the `\psset` keys, …), then
4//! pstricks_support (pstricks_support_sty.rs) overrides the drawing objects
5//! with `ltx:picture` constructors and defines `{pspicture}` and the
6//! placement commands, as Perl's does; this file the framing commands.
7//! Perl: pstricks.sty.ltxml (44L) + pstricks_support.sty.ltxml (1057L)
8use crate::prelude::*;

10/// Whether a framed box is drawn inside a `{pspicture}` (set locally by its
11/// `before_digest`), where the frame is a picture group.
12fn ps_frame_properties() -> Result<SymHashMap<Stored>> {
13  Ok(stored_map!("inpicture" => Stored::Bool(lookup_bool("lx_in_pspicture"))))
14}

46  // Perl pstricks_support.sty.ltxml L849-861: `\newpsobject{name}{oldname}{keyval}`
47  // dynamically defines `\<name>` to forward to `\<oldname>` with the saved
48  // `<keyval>` baked into the optional argument. The paper's drawing object is
49  // then drawn by the resolved `\<oldname>` (typically `\psline`, `\psdots`).
50  // Perl stores `oldname` and `keyval` in two LookupValue keys so the
51  // generated forwarder can read them at call time, and additionally merges
52  // a user-supplied `[opt]` into the saved `keyval`.
53  //
54  // Witness: physics/9710028 uses
55  //   \newpsobject{PST@Border}{psline}{linewidth=.0015,linestyle=solid}
56  // then later calls `\PST@Border(...)`. With the prior no-op stub
57  // `\PST@Border` stayed undefined and Rust errored; Perl recovered.
58  //
59  // The key list is kept as TOKENS: Perl's `Explode(ToString(...))` (:850,
60  // :857) spelled a macro value out as characters, and pstricks.tex:1462-1466
61  // stores `#3` itself (`\addbefore@par{#3}`). egameps.sty:62-63
62  // `\newpsobject{branchline}{psline}{linecolor=\@branchcolor,…}` then set
63  // `linecolor` to the characters `\@branchcolor` (egameps 101 errors and a
64  // Fatal once `\psline` passes its keys to `\psset`).
65  DefPrimitive!("\\newpsobject{}{}{}", sub[(newname, oldname, keyval)] {
66    let newcs    = s!("\\{}", newname.to_string());
67    let oldcs    = s!("\\{}", oldname.to_string());
68    let new_tok  = T_CS!(newcs);
69    let params   = parse_parameters("OptionalMatch:* []", &new_tok, true)?;
70    // Generated forwarder closure: read OptionalMatch:* and []; emit
71    //   \<old>(*)([combined-key])
72    // combined-key = saved-key + ',' + user-key (Perl L855).
73    let oldcs_owned = oldcs;
74    let key_owned   = keyval.unlist_ref().clone();
75    let body_closure: ExpansionBody = ExpansionBody::Closure(Rc::new(move |args| {
76      let star = args.first().map(|a| !a.is_none()).unwrap_or(false);
77      let usr: Vec<Token> = args.get(1)
78        .and_then(|a| match a.as_tokens() { Ok(Some(t)) => Some(t.unlist_ref().clone()), _ => None })
79        .unwrap_or_default();
80      let mut combined = key_owned.clone();
81      if !combined.is_empty() && !usr.is_empty() { combined.push(T_OTHER!(",")); }
82      combined.extend(usr);
83      let mut out = vec![T_CS!(oldcs_owned.clone())];
84      if star { out.push(T_OTHER!("*")); }
85      if !combined.is_empty() {
86        out.push(T_OTHER!("["));
87        out.extend(combined);
88        out.push(T_OTHER!("]"));
89      }
90      // Perl L856 — emit the suffix only; the paren-coords tuple is
91      // read by the resolved \psline (or sibling) object's own
92      // `PSCoordList` (pstricks_support_sty.rs).
93      Ok(Tokens::new(out))
94    }));
95    def_macro(new_tok, params, Some(body_closure), None)?;
96  });

99  // {pscs@<name>}{…}`): the raw `style` key (pstricks.tex:633-636) consults

101  // noop stub here hid every `\newpsstyle` from the real `\psset` (a stub over
102  // a pure value computer is a bug; pst-calendar-doc 15→101 once `\rput`
103  // bodies were digested in batch 56ao; Perl's own `\newpsstyle` binding stores

106  // PSCoordList-emulator for objects that are not drawn (`\psaxes`,
107  // pst_plot_sty.rs); the pstricks objects themselves read Perl's
108  // `PSCoordList` (pstricks_support_sty.rs). Without it the `(x,y)(x,y)...`
109  // tuples leak as raw text into the document — opening an `<ltx:p>` that
110  // doesn't auto-close before subsequent block content (witness:
111  // hep-ph0102192 minipage-in-figure failure). Recursive `\@ifnextchar`
112  // idiom: peek for `(`; consume one tuple; recurse.
113  RawTeX!("\\def\\lx@psgobble@parens{\\@ifnextchar({\\lx@psgobble@one}{}}");
114  RawTeX!("\\def\\lx@psgobble@one(#1){\\lx@psgobble@parens}");
115  // `\lx@psgobble@shape`: consume an OPTIONAL leading `{<arrows>}` brace group
116  // (e.g. `\psline{->}…`), THEN the trailing `(x,y)(x,y)…` coordinate tuples.
117  // The pstricks open-curve commands take an OPTIONAL arrow spec before the
118  // coordinates; the previous signatures declared it as a MANDATORY `{}` arg,
119  // so when arrows were absent (`\pscurve[opts](x,y)…`) the `{}` swallowed the
120  // first `(` and `\lx@psgobble@parens` then saw a digit, stopped, and left the
121  // remaining coordinates as stray picture text. Following an open `\put{…}`
122  // `<ltx:text>`, that stray text trapped every later block in an un-closeable
123  // `<ltx:text>` (witness 1112.2096). Peeking for the leading brace makes the
124  // arrow spec optional without over-gobbling any trailing document braces.
125  RawTeX!("\\def\\lx@psgobble@shape{\\@ifnextchar\\bgroup{\\lx@psgobble@arrows}{\\lx@psgobble@parens}}");
126  RawTeX!("\\def\\lx@psgobble@arrows#1{\\lx@psgobble@parens}");

128  // Box commands
129  // Framed boxes (Perl pstricks_support.sty.ltxml:955-980, `DefPSConstructor`):
130  // the `[<params>]` are set locally (only when given, Perl :501) and the BODY
131  // is framed. The former `#2` expansion printed the params and dropped the
132  // body (ffslides.cls:262-275 `\btext` via `\newpsobject{btextbox}{psframebox}`;
133  // pst-poker-doc, sesamath-doc-fr). Inside a `{pspicture}` the frame is a
134  // `framed` `ltx:g`, as in Perl. In running text it is an inline framed
135  // `ltx:text`, as pdflatex draws it: Perl's `ltx:g` there auto-opens an
136  // `ltx:picture` that swallows the rest of the paragraph, and the unsized SVG
137  // overprints it (DIVERGENCES #297). Fill and stroke colours are not modelled
138  // (no PS parameter state), hence no `fillframe` (its SVG filter is not ported).
139  // Guard: `perfect_kernel_batch56::psframebox_keeps_its_body`.
140  RawTeX!(r"\def\lx@ps@set#1{\if\relax\detokenize{#1}\relax\else\psset{#1}\fi}");
141  DefMacro!("\\psframebox OptionalMatch:* []{}", "{\\lx@ps@set{#2}\\lx@ps@mathflag\\lx@ps@framebox{#3}{\\lx@ps@mathbox{#3}}}");
142  DefMacro!("\\psdblframebox OptionalMatch:* []{}", "{\\lx@ps@set{#2}\\lx@ps@mathflag\\lx@ps@dblframebox{#3}{\\lx@ps@mathbox{#3}}}");
143  DefMacro!("\\psshadowbox OptionalMatch:* []{}", "{\\lx@ps@set{#2}\\lx@ps@mathflag\\lx@ps@shadowbox{#3}{\\lx@ps@mathbox{#3}}}");
144  DefMacro!("\\pscirclebox OptionalMatch:* []{}", "{\\lx@ps@set{#2}\\lx@ps@mathflag\\lx@ps@circlebox{#3}{\\lx@ps@mathbox{#3}}}");
145  DefMacro!("\\psovalbox OptionalMatch:* []{}", "{\\lx@ps@set{#2}\\lx@ps@mathflag\\lx@ps@ovalbox{#3}{\\lx@ps@mathbox{#3}}}");
