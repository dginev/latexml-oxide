1use crate::prelude::*;

3// fvextra.sty — extends fancyvrb (breaklines, breakanywhere, improved line
4// numbering, math-mode verbatim, …). Perl LaTeXML ships no fvextra.sty.ltxml
5// (it raw-loads the real file only under `--includestyles`; by default the
6// package is simply missing there). We raw-load unconditionally — house
7// idiom — so every environment/key a document declares
8// (`\DefineVerbatimEnvironment{Prompt}{Verbatim}{breaklines,…}`) is defined.
9//
10// AFTER loading, we neutralise ONLY fvextra's char-by-char break scanner:
11// `breakanywhere`/`breakbefore`/`breakafter` do `\let\FancyVerbBreakStart
12// \FV@Break` at key-SET time — a recursive scanner that measures every
13// character by boxing line-prefixes (`\sbox{\FV@LineBox}{\FV@BProcessLine
14// {#1}}`). In our engine that recurses through
15// `predigest_box_contents_in_mode` and grows the gullet pushback unboundedly
16// until the 650000 `PushbackLimit` Fatal fires (display path; 121/185 fatal
17// papers in sandbox-arxiv-2605, witness 2605.01024) or hangs the inline
18// `\Verb` path to Fatal:Timeout:TokenLimit — where Perl converts cleanly.
19// Aliasing the TARGET `\FV@Break` to `\relax` makes every later key-set
20// propagate `\relax`, and BOTH consumers gate on
21// `\ifx\FancyVerbBreakStart\relax`, taking their plain paths.
22//
23// The `breaklines` line-processor `\FV@ListProcessLine@Break` itself is
24// left INTACT: with the scanner relaxed it typesets an over-wide line as a
25// `\parbox[t]{\FV@LineWidth}` with ragged-right and breakable spaces —
26// plain TeX paragraph machinery our engine wraps natively, so the measured
27// height budget counts the same wrapped lines pdflatex produces and the
28// content stays inside the drawn frame (2605.00468 prompt boxes poked
29// 4-60px past the right border when this was over-neutralised to the
30// `@NoBreak` processor, which hboxes each SOURCE line unbroken).
31#[rustfmt::skip]
32LoadDefinitions!({
33  InputDefinitions!("fvextra", noltxml => true, extension => Some(Cow::Borrowed("sty")));
34
35  RawTeX!(r"\let\FV@Break\relax");
36
37  // fvextra L2249 `\def\FancyVerbFormatLine#1{#1}` OVERWRITES the ltx_verbatim
38  // css-class hook that fancyvrb_sty.rs installs on `\FancyVerbFormatLine` (via
39  // `\lx@add@cssclass{ltx_verbatim}`). fvextra runs its `\def` AFTER
40  // `\RequirePackage{fancyvrb}`, so a Verbatim loaded once fvextra is present
41  // lost `class="ltx_verbatim"` — and with it `white-space:pre` — collapsing to
42  // ordinary typewriter text (issue #502). Re-install the hook over fvextra's
43  // redefinition (with the scanner relaxed above, `\FancyVerbFormatLine` is
44  // just the per-line formatter, so wrapping it is safe).
45  Let!("\\lx@save@FancyVerbFormatLine", "\\FancyVerbFormatLine");
46  DefMacro!("\\FancyVerbFormatLine{}",
47    "\\lx@add@cssclass{ltx_verbatim}\\lx@save@FancyVerbFormatLine{#1}");
48
49  // Issue #525: define fvextra's `backgroundcolor`/`bgcolor` keys if the host
50  // fvextra predates them (the TL fvextra loaded above may be older — it then
51  // errors `keyval: backgroundcolor undefined` on the reporter's document).
52  // Faithful port of fvextra.sty L2435-2444: store the colour name in
53  // \FancyVerbBackgroundColor, which fancyvrb_sty.rs's `frame` box reads as its
54  // background. Guarded on the key's absence so a newer host fvextra's own
55  // definition (with its per-line \colorbox machinery) is left untouched.
56  RawTeX!(concat!(
57    r"\@ifundefined{KV@FV@backgroundcolor}{",
58    r"\define@key{FV}{backgroundcolor}{\def\FancyVerbBackgroundColor{#1}}",
59    r"\fvset{backgroundcolor=none}",
60    r"\define@key{FV}{bgcolor}{\fvset{backgroundcolor=#1}}",
61    r"}{}",
62  ));
63
64  // A newer host fvextra paints `backgroundcolor` per line via `\FV@BGColor@List`
65  // (`fvextra.sty` L2547: a `\colorbox` `\rlap`+`\hspace{\linewidth}` strip
66  // behind each line, plus over-deep struts to close the vertical seams). LaTeXML
67  // captures that as nested `backgroundcolor` `<ltx:text>` boxes with runs of
68  // padding spaces — version-dependent noise. `fancyvrb_sty.rs`'s `frame` box
69  // already carries the background (reading `\FancyVerbBackgroundColor`), so
70  // neutralize the per-line strip to a pass-through: the background then comes
71  // from the single wrapper on BOTH fvextra versions, keeping the output clean
72  // and stable. Guarded on the macro's presence (the older fvextra has none).
73  RawTeX!(r"\@ifundefined{FV@BGColor@List}{}{\long\def\FV@BGColor@List#1{#1}}");
74});
