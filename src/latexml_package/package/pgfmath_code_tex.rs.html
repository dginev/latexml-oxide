20/// Whether TeX would stop on "Dimension too large" holding `points`: past 2^30−1 sp after its rounding of the

27/// `<number><unit>` as pgf's scan into `\pgfmath@x` reads it, in the fpu too: past `\maxdimen` "Dimension too large"
28/// (tex.web §460) and `\maxdimen` kept.
29fn scanned_dimension(number: f64, unit: &str) -> f64 {
30  let points = pgfmath_convert(number, unit);
31  if !exceeds_max_dimen(points) {
32    return points;
33  }
34  report_pgfmath_error(
35    "pgfmath",
36    "overflow",
37    &format!("Dimension too large: {points}. Input was: {number}{unit}"),
38  );
39  MAX_DIMEN_POINTS.copysign(points)
40}

44/// Format a RAW pgfmath result, as set by pgf's internal `@`-suffixed
45/// functions (e.g. `\pgfmathmod@`, the trig used by
46/// `\pgfmathanglebetweenpoints`). Real pgf's `\pgfmath@tonumber`
47/// (`\expandafter\Pgf@geT\the<dimen>`) converts a dimension (e.g. `0.0pt`) to
48/// `"0.0"` preserving the decimal dot. Delimited parameter scanners like
49/// `\pgfplotscolormap@floor@unforgiving#1.#2\relax` require the dot.
50/// Perl pgfmath.code.tex.ltxml:85-90 strips `.0`, which latexml-oxide diverges
51/// from (surpass, OXIDIZED_DESIGN) to match real pgf.
52///
53/// Note: previously 2201.09268's bisection in `\pgfmathpointintersectionoflineandarc`
54/// needed bare integers, but batch 56ac replaced that bisection with a closed-form
55/// binding (KNOWN_PERL_ERRORS #210).
56fn pgfmath_result_str(value: f64) -> String {
57  if value.is_nan() || value.is_infinite() {
58    return "0.0".to_string();
59  }
60  let clamped = value.clamp(-MAX_PGF_NUMBER, MAX_PGF_NUMBER);
61  if clamped == (clamped as i64) as f64 {
62    return format!("{}.0", clamped as i64);
63  }
64  let mut s = format!("{:.5}", clamped);
65  // Strip trailing zeros after the decimal point, preserving at least one digit after the dot.
66  if s.contains('.') {
67    while s.ends_with('0') && !s.ends_with(".0") {
68      s.pop();
69    }
70  }
71  s
72}

74/// pgf's `@` arithmetic is TeX dimension-register arithmetic (pgfmathfunctions.basic.code.tex:16-25 `add`:
75/// `\pgfmath@x=#1pt\relax \pgfmath@y=#2pt\relax \advance\pgfmath@x by\pgfmath@y`; `subtract`, `neg` alike): an
76/// operand is scanned to scaled points as `#1pt` is (tex.web §102 `round_decimals`), the arithmetic is on integers,
77/// and the result is printed by `\the` (`\pgfmath@returnone`, §103 `print_scaled`). Done in f64 and rounded to five
78/// decimals, as Perl's binding does (pgfmath.code.tex.ltxml:85-90, :155-156), the cloud shape's border-anchor binary
79/// search (pgflibraryshapes.symbols.code.tex:1236-1292, which ends on `\ifdim\p pt=\s pt`) could round its midpoint up
80/// to its end forever: Fatal on 2606.26406 and 2508.07407 (KNOWN_PERL_ERRORS #461).
81/// A value past TeX's largest dimension is clamped to it, as TeX's `arith_error` clamps a scanned `#1pt` (§460) — and
82/// so the integer arithmetic below cannot overflow.
83fn pgfmath_sp(text: &Tokens) -> i64 {
84  fixpoint(
85    parse_pgf_number(text).clamp(-MAX_PGF_NUMBER, MAX_PGF_NUMBER),
86    None,
87  )
88}

90/// pgf's `divide` (pgfmathfunctions.basic.code.tex:66-100): a whole divisor divides the scaled points as TeX's
91/// `\divide` does, truncating toward zero (§1240 → §106 `x_over_n`), so an interval halved to 1sp has its midpoint at
92/// its start; any other divisor is pgf's long division, its quotient on the same grid. A zero divisor leaves the
93/// dividend ([`pgfmath_divide`]).
94fn pgfmath_divide_sp(x: i64, y: i64) -> i64 {
95  if y == 0 {
96    fixpoint(pgfmath_divide(x as f64 / 65536.0, 0.0), None)
97  } else if y % 65536 == 0 {
98    x / (y / 65536)
99  } else {
100    fixpoint(x as f64 / y as f64, None)
101  }
102}

122/// Return tokens that \def\pgfmathresult{<string>}
123///
124/// pgfmathparser.code.tex:392-396 keeps a `"…"` string operand as the REAL
125/// tokens it was written with, so a ternary like braids.sty:276
126/// `\pgfmathparse{\value{c} < \n ? "\noexpand\setcounter{c}{\n}" : ""}` followed
127/// by `\pgfmathresult` EXECUTES the `\setcounter`. Our evaluator carries the
128/// string through Rust, so a result holding a control sequence is re-tokenized
129/// (at the caller's `@` catcode — braids.sty runs under `\makeatletter`) instead
130/// of exploded into dead `\`+letters. Numbers explode as before. Perl's pgfmath
131/// cannot parse the ternary at all (SHARED). Witness: braids/braids.
132/// Guard: `perfect_kernel_batch56::pgfmath_string_result_keeps_control_sequences`.
133fn pgfmath_result_tokens_str(s: &str) -> Vec<Token> {
134  let mut toks = vec![T_CS!("\\def"), T_CS!("\\pgfmathresult"), T_BEGIN!()];
135  // pgfmathparser.code.tex:35-40 `\pgfmath@catcodes` re-catcodes only `= , | &`,
136  // so a string operand's LETTERS stay catcode 11: tikz's path dispatcher
137  // `\tikz@handle` (tikz.code.tex:2134-2163) `\ifx`-matches the letter `a` of
138  // an `arc[…]` result against a catcode-11 `{a}` — exploded OTHER letters never
139  // dispatch and the path gives up (zx-calculus curved wires,
140  // tikzlibraryzx-calculus.code.tex:56-63; Perl's `ExplodeText` shares the
141  // gap). Guard: `perfect_kernel_batch56::pgfmath_string_result_keeps_letter_catcodes`.
142  if s.contains('\\') || s.chars().any(|c| c.is_ascii_alphabetic()) {
143    // A control word with `@` in the string came from expanding an internal macro (a string
144    // operand is expanded first: `\,` is `\lx@thinspace` here, robust and kept in LaTeX), so
145    // it is read back with `@` a letter even in the document body; read as `\lx` and
146    // `@thinspace` it was an undefined `\lx` (wheelchart.tex's `5>10?"#1":"\,\%"` labels,
147    // 29 errors).
148    let internal_name = s.split('\\').skip(1).any(|rest| {
149      let name: &str = rest
150        .split(|c: char| !(c.is_ascii_alphabetic() || c == '@'))
151        .next()
152        .unwrap_or("");
153      // Only a defined `@`-name: a user's `"\LaTeX@home"` stays `\LaTeX` then `@home`.
154      name.contains('@')
155        && lookup_definition(&T_CS!(&s!("\\{name}")))
156          .ok()
157          .flatten()
158          .is_some()
159    });
160    let at_letter = internal_name || matches!(lookup_catcode('@'), Some(Catcode::LETTER));
161    let src = TeXString::assembled(s.to_string());
162    let retok = if at_letter {
163      TokenizeInternal!(src)
164    } else {
165      Tokenize!(src)
166    };
167    toks.extend(retok.unlist());
168  } else {
169    toks.extend(Explode!(s));
170  }
171  toks.push(T_END!());
172  toks
173}

175/// pgf's `mod` (pgfmathfunctions.basic.code.tex:263-278): truncated, as TeX's `\divide` truncates, and a zero divisor
176/// leaves the dividend (`mod(x, 0)` is `x`, :271). Its `Mod` (:286-297) is that `mod` plus the divisor when it is
177/// negative, so a multiple of the divisor is 0 (`Mod(-5,5)`: 1811.01404's `style\st`, style5 for style0). Perl's
178/// floor mod (pgfmath.code.tex.ltxml:196-200) makes it the divisor, and its second `\pgfmathmod@` (:206) defines
179/// `mod` as that floor mod (KNOWN_PERL_ERRORS #944). Guard `perfect_kernel_batch71a::pgfmath_mod_of_a_multiple_is_zero`.
180fn pgfmath_mod(a: f64, b: f64, never_negative: bool) -> f64 {
181  let truncated = if b == 0.0 {
182    pgfmath_divide(a, b)
183  } else {
184    a % b
185  } + 0.0;
186  if never_negative && truncated < 0.0 {
187    truncated + b
188  } else {
189    truncated
190  }
191}

292/// Parse a pgfNumber argument: expand macros, then convert to f64
293/// Perl: pgfNumber parameter type reads and evaluates the number
294fn parse_pgf_number(arg: &Tokens) -> f64 {
295  // First try direct string (avoids expansion overhead for literal numbers)
296  let s = arg.to_string();
297  let s = strip_leading_double_negation(s.trim());
298  if s == "." {
299    return 0.0;
300  }
301  if let Ok(v) = s.parse::<f64>() {
302    return v;
303  }
304  // If direct parse fails (e.g. unexpanded macro), expand and retry
305  match do_expand(arg.clone()) {
306    Ok(expanded) => {
307      let s = expanded.to_string();
308      let s = strip_leading_double_negation(s.trim());
309      if s == "." {
310        return 0.0;
311      }
312      s.parse::<f64>().unwrap_or(0.0)
313    },
314    _ => 0.0,
315  }
316}

318// Perl #2711 uses `s/^\-\-//g`, which strips every leading `--` pair
319// (e.g. `----5` → `5`). `strip_prefix` only removes one pair.
320fn strip_leading_double_negation(mut s: &str) -> &str {
321  while let Some(rest) = s.strip_prefix("--") {
322    s = rest;
323  }
324  s
325}

330/// Try to parse input as a simple number and format per Perl rules.
331/// Returns None if input is not a simple number.
332fn try_simple_number(input: &str) -> Option<String> {
333  let bytes = input.as_bytes();
334  let mut pos = 0;
335
336  // Optional sign
337  let sign: Option<char> = if pos < bytes.len() && (bytes[pos] == b'+' || bytes[pos] == b'-') {
338    let c = bytes[pos] as char;
339    pos += 1;
340    Some(c)
341  } else {
342    None
343  };
344
345  // Required: at least one digit
346  let int_start = pos;
347  while pos < bytes.len() && bytes[pos].is_ascii_digit() {
348    pos += 1;
349  }
350  if pos == int_start {
351    return None;
352  }
353  let integer = &input[int_start..pos];
354
355  // Optional decimal part: \.[\d.]*
356  let decimals: Option<&str> = if pos < bytes.len() && bytes[pos] == b'.' {
357    let dec_start = pos;
358    pos += 1;
359    while pos < bytes.len() && (bytes[pos].is_ascii_digit() || bytes[pos] == b'.') {
360      pos += 1;
361    }
362    Some(&input[dec_start..pos])
363  } else {
364    None
365  };
366
367  // Must be at end of string
368  if pos != bytes.len() {
369    return None;
370  }
371
372  // Format result (Perl L332-344)
373  let mut result = input.to_string();
374  if let Some(s) = sign {
375    if s == '+' {
376      result = match decimals {
377        Some(d) => format!("{}{}", integer, d),
378        None => integer.to_string(),
379      };
380    } else {
381      // sign == '-'
382      if !result.contains(|c: char| ('1'..='9').contains(&c)) {
383        result = "0.0".to_string();
384      } else if decimals.is_none() {
385        result = format!("{}.0", result);
386      }
387      // pgf negates in `\pgfmath@x` (Perl .ltxml:342-344 errs on `-16383.99999` too), not in the fpu.
388      if let Ok(numval) = result.parse::<f64>()
389        && numval < -MAX_PGF_NUMBER
390        && !fpu_active()
391      {
392        if exceeds_max_dimen(numval) {
393          report_pgfmath_error(
394            "pgfmath",
395            "overflow",
396            &format!("Dimension too large: {result}. Input was: {input}"),
397          );
398        }
399        result = format!("{}", -MAX_PGF_NUMBER);
400      }
401    }
402  }
403  Some(result)
404}

457/// The e-TeX expression `text` opens (`\dimexpr`, `\glueexpr`, `\numexpr`, `\muexpr`), read as TeX reads pgf's
458/// `\pgfmath@dimen\dimexpr…` (pgfmathparser.code.tex:571-600): its value (points, the integer, or the mu) and the
459/// bytes of `text` before the text TeX left (1711.06128, 2206.04438).
460fn pgfmath_read_expr(primitive: &str, text: &str) -> Option<(f64, usize)> {
461  let tokens = mouth::tokenize_internal(TeXString::assembled(text.to_string()));
462  let integer = primitive == "\\numexpr";
463  let mu = primitive == "\\muexpr";
464  let (value, left) = Mouth::new("", None)
465    .and_then(|mouth| {
466      reading_from_mouth(mouth, move || {
467        unread(tokens);
468        let value = if integer {
469          read_number()?.value_of() as f64
470        } else if mu {
471          read_mu_glue()?.value_of() as f64 / 65536.0
472        } else {
473          read_dimension()?.value_of() as f64 / 65536.0
474        };
475        let mut left = Vec::new();
476        while let Some(token) = read_token()? {
477          left.push(token);
478        }
479        Ok((value, Tokens::new(left).to_string()))
480      })
481    })
482    .ok()?;
483  // Where the text left starts in `text`, matched from the end, spaces aside.
484  let text = text.trim_end();
485  let mut chars = text
486    .char_indices()
487    .rev()
488    .filter(|(_, c)| !c.is_whitespace());
489  let mut start = text.len();
490  for wanted in left.chars().rev().filter(|c| !c.is_whitespace()) {
491    let (at, c) = chars.next()?;
492    if c != wanted {
493      return None;
494    }
495    start = at;
496  }
497  Some((value, start))
498}

595/// Apply a built-in pgf math function (Perl L556-647)
596/// pgf's `min`/`max` list fold (pgfmathfunctions.misc.code.tex:292-336): every
597/// argument against the sentinel; a lone comma-less argument is scanned one
598/// token at a time by `\pgfmathmin@@#1{}` — its decimal digits.
599fn pgfmath_fold_list(args: &[f64], sentinel: f64, pick: fn(f64, f64) -> f64) -> f64 {
600  if args.len() == 1 {
601    let text = format!("{}", args[0].trunc().abs() as i64);
602    return text
603      .bytes()
604      .map(|b| f64::from(b - b'0'))
605      .fold(sentinel, pick);
606  }
607  args.iter().copied().fold(sentinel, pick)
608}

782/// Apply a user-defined pgf math function by calling TeX
783/// Perl: sub pgfmath_apply (L447-459)
784///
785/// An argument in integer form (`arg_ints`) is passed as an integer (`7`, not `7.0`), and the
786/// result's own form is returned with it: a body whose `\pgfmathresult` is an integer literal
787/// (`\pgfmathparse{#1>0}`, `\def\pgfmathresult{1}`) yields an integer, as pgf leaves it.
788fn pgfmath_apply_user(name: &str, args: &[f64], arg_ints: &[bool]) -> UserApply {
789  let cs_name = format!("\\pgfmath{}@", name);
790  let cs_tok = T_CS!(&cs_name);
791  // Check if the function is defined
792  if !matches!(lookup_definition(&cs_tok), Ok(Some(_))) {
793    return UserApply::Undefined;
794  }
795  // Build invocation tokens: \pgfmath{name}@{arg1}{arg2}...
796  let mut tokens: Vec<Token> = vec![cs_tok];
797  for (k, arg) in args.iter().enumerate() {
798    let s = if arg_ints.get(k).copied().unwrap_or(false) && *arg == (*arg as i64) as f64 {
799      format!("{}", *arg as i64)
800    } else {
801      pgfmath_result_str(*arg)
802    };
803    tokens.push(T_BEGIN!());
804    tokens.extend(Explode!(s));
805    tokens.push(T_END!());
806  }
807  // Reset the global `\ifpgfmathunitsdeclared` flag BEFORE digesting the body
808  // so that, afterwards, the flag reflects ONLY whether THIS function body
809  // parsed a unit'd value. A user function body may run a nested
810  // `\pgfmathparse` (e.g. pgfplots' `pgfplotsbarwidthgeneric` body parses the
811  // `<bar width>pt` dimension); the outer parser must inherit that via
812  // `absorb_units_flag` so it doesn't clobber the flag back to false on exit.
813  // Without the reset a prior parse's stale flag would leak in for bodies that
814  // set `\pgfmathresult` directly (e.g. `\def\pgfmathresult{42}`). Mirrors
815  // Perl, where `\ifpgfmathunitsdeclared` is a persistent global threaded
816  // through nested evaluations. Fixes `bar shift={\pgfplotbarwidth}` under
817  // `symbolic x coords` (2110.14597).
818  let _ = digest(Tokens::from(vec![T_CS!("\\pgfmathunitsdeclaredfalse")]));
819  // Digest the invocation (sets \pgfmathresult) inside the parse's group
820  // (`pgfmath_grammar::evaluate`): what the body defines locally — a tikzmath function's
821  // parameters `\cx=#1` — is seen by later calls of the same parse and gone after it. Ungrouped
822  // (Perl pgfmath.code.tex.ltxml:452-455 too, KPE #421), 2605.28612's indexed variable
823  // `\CircleCenterX1` read the leaked parameter: `0`, `1` → `01`, then 11, 111 cm.
824  let _ = digest(Tokens::from(tokens));
825  // Read back \pgfmathresult via expansion
826  let result = do_expand(Tokens::from(vec![T_CS!("\\pgfmathresult")]));
827  let Ok(result_tokens) = result else {
828    return UserApply::NotNumeric(String::new());
829  };
830  let s = result_tokens.to_string();
831  let s = s.trim();
832  match s.parse::<f64>() {
833    Ok(value) => UserApply::Value(value, s.parse::<i64>().is_ok()),
834    Err(_) => UserApply::NotNumeric(s.to_string()),
835  }
836}

838/// The outcome of calling a pgfmath function through its TeX body.
839enum UserApply {
840  /// The result and whether it is in integer form.
841  Value(f64, bool),
842  /// The body left a `\pgfmathresult` that is not a number.
843  NotNumeric(String),
844  /// No `\pgfmath<name>@` is defined.
845  Undefined,
846}

848/// [`pgfmath_apply_user`], its body run once, an unusable result reported (Perl
849/// pgfmath.code.tex.ltxml:457-459 for an undefined operator) and read as 0.
850fn pgfmath_apply_user_reported(name: &str, args: &[f64], arg_ints: &[bool]) -> (f64, bool) {
851  let message = match pgfmath_apply_user(name, args, arg_ints) {
852    UserApply::Value(value, int) => return (value, int),
853    UserApply::NotNumeric(text) => {

1071  /// The number TeX reads from a text value followed by `pt`, and whether it reads all it should: `1` and `1.5`, a unit
1072  /// of its own (`1pt` and `1cm`, whose `pt` left over prints in pgf's null font), but `1e2` is 1 and an illegal unit,
1073  /// `a` is a missing number.
1074  fn text_as_number(text: &str) -> (f64, bool) {
1075    let sign = text
1076      .bytes()
1077      .take_while(|b| matches!(b, b'+' | b'-'))
1078      .count();
1079    let mut point = false;
1080    let digits = text[sign..]
1081      .bytes()
1082      .take_while(|b| b.is_ascii_digit() || (*b == b'.' && !std::mem::replace(&mut point, true)))
1083      .count();
1084    let (number, unit) = text.split_at(sign + digits);
1085    let negative = number.bytes().filter(|b| *b == b'-').count() % 2 == 1;
1086    let Ok(value) = number[sign..].parse::<f64>() else {
1087      return (0.0, false);
1088    };
1089    let value = if negative { -value } else { value };

1099  /// The bytes of the token `text` opens with: a character, or a control sequence (one spelled `\csname…\endcsname`
1100  /// by `pgfmath_input_text` whole).
1101  fn token_width(text: &str) -> usize {
1102    if text.starts_with("\\csname ")
1103      && let Some((_, len)) = cs_at(text)
1104    {
1105      return len;
1106    }
1107    match text.strip_prefix('\\') {
1108      Some(name) => {
1109        let letters = name.bytes().take_while(u8::is_ascii_alphabetic).count();
1110        1 + if letters > 0 {
1111          letters
1112        } else {
1113          name.chars().next().map_or(0, char::len_utf8)
1114        }
1115      },
1116      None => text.chars().next().map_or(0, char::len_utf8),
1117    }

1120  /// The tokens of a value, as elements: `\pgfmatharray@@`'s undelimited argument skips spaces, and takes a group whole.
1121  fn text_tokens(text: &str) -> Vec<ArrayElement> {
1122    let mut tokens = Vec::new();
1123    let mut rest = text.trim_start();
1124    while let Some(c) = rest.chars().next() {
1125      let width = if c == '{' {
1126        matching_brace(rest).map_or(1, |end| end + 1)
1127      } else {
1128        token_width(rest)
1129      };
1130      let (token, tail) = rest.split_at(width);
1131      tokens.push(ArrayElement::Text(
1132        token
1133          .strip_prefix('{')
1134          .and_then(|t| t.strip_suffix('}'))
1135          .unwrap_or(token)
1136          .to_string(),
1137      ));
1138      rest = tail.trim_start();
1139    }

1141  }

1143  /// What `\pgfmath@collect@`'s `\pgfmath@bgroup@strip` leaves of a string in a list of two or more
1144  /// (pgfmathparser.code.tex:963, :984-1009): its re-read `\pgfmath@bgroup@@@strip#1#2#3` takes two undelimited arguments,
1145  /// which skip the spaces before them, so a space before the first token or the second goes and any later one stays
1146  /// (`"a b"` is `ab`, `"a ? b"` is `a? b`, `"ab c"` stays). A string that opens with a group is kept whole.
1147  fn collected_string(text: &str) -> String {
1148    let rest = text.trim_start();
1149    if rest.starts_with('{') {
1150      return text.to_string();
1151    }
1152    let first = token_width(rest);
1153    let (head, tail) = rest.split_at(first);
1154    // (a control word keeps the space that ends it, which is no token)
1155    let word = head.len() > 2 && head.starts_with('\\');
1156    if word {
1157      rest.to_string()
1158    } else {
1159      format!("{head}{}", tail.trim_start())
1160    }
1161  }

1163  /// The elements of the array `{…}` the input starts at (each a string, a nested array, or a formula), consumed.
1164  fn array_literal(i: &mut In) -> ModalResult<Vec<ArrayElement>> {
1165    let text = i.input;
1166    let Some(end) = matching_brace(text) else {
1167      return fail();
1168    };
1169    let inner = &text[1..end];
1170    let mut elements = Vec::new();
1171    if !inner.trim().is_empty() {
1172      let pieces = super::split_top_commas(inner).unwrap_or_default();
1173      let collected = pieces.len() > 1;
1174      for piece in pieces {
1175        let piece = piece.trim();
1176        let quoted = piece
1177          .strip_prefix('"')
1178          .and_then(|rest| rest.strip_suffix('"'))
1179          .filter(|text| !text.contains('"'));
1180        if let Some(string) = quoted {
1181          elements.push(ArrayElement::Text(if collected {
1182            collected_string(string)
1183          } else {
1184            string.to_string()
1185          }));
1186        } else if piece.is_empty() {
1187          elements.push(ArrayElement::Text(String::new()));
1188        } else if piece.starts_with('{') && matching_brace(piece) == Some(piece.len() - 1) {
1189          i.input = piece;
1190          let mut inner = array_literal(i)?;
1191          // (a list's strip takes a lone string's braces too: `{{"a b"},"d"}[0]` is `ab`)
1192          elements.push(match inner.as_mut_slice() {
1193            [ArrayElement::Text(text)] if collected => ArrayElement::Text(collected_string(text)),
1194            _ => ArrayElement::Array(inner),
1195          });
1196        } else {
1197          i.input = piece;
1198          let value = formula(i);
1199          let number = value.unwrap_or(0.0);
1200          elements.push(match i.state.text_result.take() {
1201            Some(string) => ArrayElement::Text(string),
1202            None => ArrayElement::Number(number, i.state.int_result),
1203          });
1204        }
1205      }
1206    }
1207    i.input = &text[end + 1..];
1208    Ok(elements)
1209  }

1259  /// `{…}` and its accesses `[i][j]…` (pgfmathparser.code.tex:943-949 `\pgfmath@startindex@`, which gives an array
1260  /// element back in a group of its own: `{{1,2},{3,4}}[1]` is `{{3}{4}}`). Live pgf for each form, repro
1261  /// graphics-tikz/pgfmath_array_primary_and_strings.
1262  fn array_primary(i: &mut In) -> ModalResult<f64> {
1263    let mut value = ArrayElement::Array(array_literal(i)?);
1264    let mut indexed = false;
1265    loop {
1266      let spaces = i.input.bytes().take_while(u8::is_ascii_whitespace).count();
1267      if i.input.as_bytes().get(spaces) != Some(&b'[') {
1268        break;
1269      }
1270      bump(i, spaces + 1);
1271      let index = formula(i)?;
1272      number_wanted(i);
1273      eat(i, b']');
1274      value = match value {
1275        ArrayElement::Array(elements) => list_element(elements, index),
1276        // an index into what is no array: pgf reads its one token as the array
1277        other => array_element(vec![other], index),
1278      };
1279      indexed = true;
1280    }
1281    // (a whole formula's braces go before pgf reads it, pgfmathparser.code.tex:139: `{"a"}` is `"a"`)
1282    if !indexed
1283      && let ArrayElement::Array(elements) = &mut value
1284      && matches!(elements.as_slice(), [ArrayElement::Text(_)])
1285    {
1286      value = elements.remove(0);
1287    }
1288    let array = indexed && matches!(value, ArrayElement::Array(_));
1289    let number = array_value(i, value);
1290    if array {
1291      i.state.text_result = i.state.text_result.take().map(|text| format!("{{{text}}}"));
1292    }
1293    Ok(number)
1294  }

1354  /// formula := logic ('?' logic ':' logic)*
1355  ///
1356  /// pgf's precedences (pgfmathparser.code.tex:898-916): `?` 100 and `:` 101 are the loosest,
1357  /// then `&&`/`||` 200, the comparisons 250, `+`/`-` 500; so chained ternaries group to the
1358  /// left, `a?b:c?d:e` is `(a?b:c)?d:e`. The ternary yields its chosen branch as that branch
1359  /// left the result (an integer literal stays `7`, arithmetic gives a real); a missing `:`
1360  /// branch is 0.
1361  pub(super) fn formula(i: &mut In) -> ModalResult<f64> {
1362    let mut condition = logic(i)?;
1363    loop {
1364      sb(i);
1365      if !eat(i, b'?') {
1366        return Ok(condition);
1367      }
1368      number_wanted(i);
1369      let then_val = logic(i)?;
1370      let then_int = i.state.int_result;
1371      let then_text = i.state.text_result.take();
1372      sb(i);
1373      let (else_val, else_int) = if eat(i, b':') {
1374        (logic(i)?, i.state.int_result)
1375      } else {
1376        (0.0, false)
1377      };
1378      let else_text = i.state.text_result.take();
1379      i.state.text_result = if condition != 0.0 {
1380        then_text
1381      } else {
1382        else_text
1383      };
1384      let (value, int) = if condition != 0.0 {
1385        (then_val, then_int)
1386      } else {
1387        (else_val, else_int)
1388      };
1389      i.state.int_result = int;
1390      condition = value;
1391    }
1392  }

1395  fn logic(i: &mut In) -> ModalResult<f64> {

1398      sb(i);
1399      let op = match i.input.as_bytes() {
1400        [b'&', b'&', ..] => "&&",
1401        [b'|', b'|', ..] => "||",
1402        _ => break,

1412        },
1413        Err(_) => {
1414          i.reset(&cp);
1415          break;
1416        },
1417      }
1418    }

1420  }

1450      && let two @ (b"==" | b"!=" | b">=" | b"<=") = &[b[0], b[1]]

1466  /// expr := term (('+'|'-') term)*
1467  fn expr(i: &mut In) -> ModalResult<f64> {
1468    let mut result = term(i)?;
1469    loop {
1470      sb(i);
1471      match first(i) {
1472        Some(b'+') => {
1473          bump(i, 1);
1474          number_wanted(i);
1475          result += term(i)?;
1476          number_wanted(i);
1477          // Arithmetic yields a real: `2+(1<2)` → "3.0", NOT the integer a bare
1478          // comparison produces (pgfmath ground truth). A comparison anywhere in
1479          // an operand set int_result; combining it arithmetically clears that.
1480          i.state.int_result = false;
1481        },
1482        Some(b'-') => {
1483          bump(i, 1);
1484          number_wanted(i);
1485          result -= term(i)?;
1486          number_wanted(i);
1487          i.state.int_result = false;
1488        },
1489        _ => break,
1490      }
1491    }
1492    Ok(result)
1493  }

1495  /// term := factor (('*'|'/') factor | 'r' power_tail)* — postfix `r` (deg, 600) is below `*`/`/` (700), `!` (800)
1496  /// and `^` (900): pgf's operator stack holds it until the term ends, so it applies to the product so far and to the
1497  /// powers and factorials after it (`4/2r` (4/2)r, `3*2r^2` ((3*2)^2)r; pgfmathparser.code.tex:765-790, 900-919).
1498  fn term(i: &mut In) -> ModalResult<f64> {
1499    let mut result = factor(i)?;
1500    let mut degrees = 0;
1501    loop {
1502      sb(i);
1503      match first(i) {
1504        Some(b'r') if !second(i).map(|c| c.is_ascii_alphabetic()).unwrap_or(false) => {
1505          bump(i, 1);
1506          degrees += 1;
1507          result = power_tail(i, result)?;
1508          i.state.int_result = false;
1509        },
1510        Some(b'*') => {
1511          bump(i, 1);
1512          number_wanted(i);
1513          result *= factor(i)?;
1514          number_wanted(i);
1515          // See expr(): arithmetic demotes a comparison's integer to a real
1516          // (`2*(1<2)` → "2.0", not "2").
1517          i.state.int_result = false;
1518        },
1519        Some(b'/') => {
1520          bump(i, 1);
1521          number_wanted(i);
1522          // Infix `/` is pgf's `divide`: a zero divisor leaves the dividend.
1523          result = pgfmath_divide(result, factor(i)?);
1524          number_wanted(i);
1525          i.state.int_result = false;
1526        },
1527        _ => break,
1528      }
1529    }
1530    for _ in 0..degrees {
1531      result = result.to_degrees();
1532    }
1533    Ok(result)
1534  }

1543  /// on the stack for the `^` chain after it, so it applies to the whole power (`2^2!` 24.0, `2!^2` (2^2)! 24.0).
1544  fn power_tail(i: &mut In, mut result: f64) -> ModalResult<f64> {
1545    let mut factorials = 0;
1546    loop {
1547      sb(i);
1548      match first(i) {
1549        Some(b'^') => {
1550          bump(i, 1);
1551          number_wanted(i);
1552          let exp = simplefactor(i)?;
1553          number_wanted(i);
1554          i.state.int_result = false; // exponentiation yields a real
1555          result = result.powf(exp);
1556        },
1557        Some(b'!') if second(i) != Some(b'=') => {
1558          bump(i, 1);
1559          factorials += 1;
1560          i.state.int_result = false;
1561        },
1562        _ => break,
1563      }
1564    }
1565    for _ in 0..factorials {
1566      result = pgfmath_factorial(result as i64);
1567    }
1568    Ok(result)
1569  }

1571  /// simplefactor — the atom: parens, prefix ops, numbers (with unit/register
1572  /// suffix), registers, identifiers (functions/constants), lone dot.
1573  fn simplefactor(i: &mut In) -> ModalResult<f64> {
1574    sb(i);
1575    // A value is a real unless it is an integer literal, a comparison, or an integer function
1576    // (pgf keeps `7` and `(7)` as typed; `-7`, `\pgf@x`, `2pt`, `max(1,2)` print `.0`).
1577    i.state.int_result = false;
1578    match first(i) {
1579      None => return fail(),
1580      Some(b'{') => return array_primary(i),
1581      Some(b'(') => {
1582        bump(i, 1);
1583        let val = formula(i)?;
1584        sb(i);
1585        eat(i, b')'); // forgive unclosed at end (Perl)
1586        return Ok(val);
1587      },
1588      Some(b'-') => {
1589        // Unary minus is pgf's `neg` at precedence 500 (pgfmathfunctions.basic.code.tex:46-49), below `r` (600),
1590        // `*` and `/` (700), `!` (800) and `^` (900): it negates the whole term after it (`-2^2` -4.0, `-3!` -6.0,
1591        // `8/-2*2` -2.0, `2^-1*3` 0.125 in pdflatex; witness 2311.02367's `exp(-(\y)^2)`).
1592        bump(i, 1);
1593        let value = -term(i)?;
1594        number_wanted(i);
1595        i.state.int_result = false;
1596        return Ok(value);
1597      },
1598      Some(b'+') => {
1599        bump(i, 1);
1600        return simplefactor(i);
1601      },
1602      Some(b'!') => {
1603        if second(i) == Some(b'=') {
1604          return fail();
1605        }
1606        bump(i, 1);
1607        let val = simplefactor(i)?;
1608        i.state.int_result = true; // `!0` → `1` (live pgf), like `not(0)`
1609        return Ok(if val == 0.0 { 1.0 } else { 0.0 });
1610      },
1611      _ => {},
1612    }
1613
1614    if let Some((num, literal)) = try_number(i) {
1615      sb(i);
1616      if let Some(unit) = try_unit(i) {
1617        i.state.units_declared = true;
1618        return Ok(super::scanned_dimension(num, &unit));
1619      }
1620      if first(i) == Some(b'\\') {
1621        let cp = i.checkpoint();
1622        if let Some(operand) = try_cs_register(i) {
1623          return Ok(operand.after(i, num));
1624        }
1625        i.reset(&cp);
1626      }
1627      // An integer literal keeps its form: decimal digits, or a hex/binary literal (`0x1F` 31); one opening with `0`
1628      // is octal (`010+1` is 9.0, pgfmathparser.code.tex:355-358, 486-497).
1629      return Ok(match literal {
1630        Literal::Based => {
1631          i.state.int_result = true;
1632          num
1633        },
1634        Literal::Decimal(digits) if digits.bytes().all(|b| b.is_ascii_digit()) => {
1635          i.state.int_result = true;
1636          match digits.strip_prefix('0') {
1637            Some(octal) if !octal.is_empty() => {
1638              i64::from_str_radix(octal, 8).map_or(num, |v| v as f64)
1639            },
1640            _ => num,
1641          }
1642        },
1643        _ => num,
1644      });
1645    }

1647    if first(i) == Some(b'\\')
1648      && let Some(operand) = try_cs_register(i)

1651      sb(i);
1652      if first(i) == Some(b'\\') {
1653        let cp = i.checkpoint();
1654        if let Some(operand) = try_cs_register(i) {
1655          return Ok(operand.after(i, value));
1656        }
1657        i.reset(&cp);
1658      }
1659      return Ok(value);
1660    }

1663      let name = read_identifier(i);
1664      if matches!(name.as_str(), "width" | "height" | "depth")
1665        && let Some(text) = quoted_argument(i)

1667        return Ok(super::pgfmath_sizer(&name, text));

1670        return function_call(i, &name);
1671      }
1672      if is_builtin_constant(&name) {
1673        set_int_result_for_fn(i, &name);
1674        return Ok(pgfmath_apply_fn(&name, &[]));
1675      }
1676      if is_user_function(&name) {
1677        // A user function whose TeX body may declare units — inherit them
1678        // afterwards (see `absorb_units_flag`). Builtin functions above must
1679        // NOT absorb: they compute purely in Rust and never touch the global
1680        // `\ifpgfmathunitsdeclared`, so reading it could only pick up a STALE
1681        // true from an earlier evaluation and wrongly mark e.g. `sin(30)` as
1682        // unit'd. (Real `\pgfmathparse` resets the global at parse-start, which
1683        // is why #249's recdescent — which absorbed for builtins too — never
1684        // tripped; the golden corpus runs evals back-to-back without that
1685        // reset and exposes the staleness.)
1686        let v = function_call(i, &name)?;
1687        absorb_units_flag(i);
1688        return Ok(v);
1689      }
1690      if is_user_constant(&name) {
1691        // A 0-arg user pseudo-constant whose body may parse a unit'd value
1692        // (e.g. `pgfplotsbarwidthgeneric`). Inherit its units flag — see
1693        // `absorb_units_flag`. Builtin constants above are pure and need no
1694        // absorb.
1695        open_parse_group(i);
1696        let (v, int) = super::pgfmath_apply_user_reported(&name, &[], &[]);
1697        i.state.int_result = int;
1698        absorb_units_flag(i);
1699        return Ok(v);
1700      }
1701      return fail();
1702    }
1703
1704    if first(i) == Some(b'.') {
1705      bump(i, 1);
1706      return Ok(0.0);
1707    }
1708    fail()
1709  }

1755  /// FUNCTION '(' formula (',' formula)* ')' | FUNCTION simplefactor
1756  fn function_call(i: &mut In, name: &str) -> ModalResult<f64> {
1757    sb(i);
1758    // `array({e0,e1,…}, index)` — the one builtin whose FIRST argument is a
1759    // brace-delimited expression list, not a formula (pgfmathfunctions.code
1760    // `\pgfmatharray@`). Parse the elements in place with the shared parser,
1761    // then select by the (0-based) index; out-of-bounds yields 0 like an
1762    // empty result. Witness colorblind.sty L433-449
1763    // `round(255*array({\colorspec},0))`.
1764    // `dim({…})`, the number of its elements (pgfmathfunctions.misc.code.tex:269-290): an array's, or 1 for any other
1765    // value (tikzpingus.sty's lists; `dim(5)`).
1766    if name == "dim" && first(i) == Some(b'(') {
1767      bump(i, 1);
1768      let count = if brace_follows(i) {
1769        array_literal(i)?.len()
1770      } else {
1771        formula(i)?;
1772        i.state.text_result = None;
1773        1
1774      };
1775      sb(i);
1776      eat(i, b')');
1777      i.state.int_result = true;
1778      return Ok(count as f64);
1779    }
1780    if name == "array" && first(i) == Some(b'(') {
1781      bump(i, 1);
1782      if brace_follows(i) {
1783        let elements = array_literal(i)?;
1784        sb(i);
1785        let index = if eat(i, b',') { formula(i)? } else { 0.0 };
1786        number_wanted(i);
1787        sb(i);
1788        eat(i, b')');
1789        let element = list_element(elements, index);
1790        return Ok(array_value(i, element));
1791      }
1792      sb(i);
1793      // Brace-less form: pgfmath's own macro layers can strip the list
1794      // braces before the expression reaches this evaluator, leaving
1795      // `array(3,7,5,1)` — the LAST value is the index, the rest the
1796      // elements.
1797      let mut elems: Vec<f64> = Vec::new();
1798      // Each element's integer form: `array` returns the chosen element as it is.
1799      let mut elem_ints: Vec<bool> = Vec::new();
1800      while let Ok(v) = formula(i) {
1801        elems.push(v);
1802        elem_ints.push(i.state.int_result);
1803        sb(i);
1804        if !eat(i, b',') {
1805          break;
1806        }
1807      }
1808      elem_ints.pop();
1809      let idx = elems.pop().unwrap_or(0.0);
1810      sb(i);
1811      eat(i, b')');
1812      let idx = idx.round();
1813      let (val, int) = if idx >= 0.0 {
1814        (
1815          elems.get(idx as usize).copied().unwrap_or(0.0),
1816          elem_ints.get(idx as usize).copied().unwrap_or(false),
1817        )
1818      } else {
1819        (0.0, false)
1820      };
1821      i.state.int_result = int;
1822      return Ok(val);
1823    }
1824    if first(i) == Some(b'(') {
1825      bump(i, 1);
1826      let mut args = vec![formula(i)?];
1827      number_wanted(i);
1828      // Each argument's integer form: `ifthenelse` returns its chosen argument as it is
1829      // (pgfmathfunctions.comparison.code.tex:77-86), text too, so its branches want no number.
1830      let mut arg_ints = vec![i.state.int_result];
1831      let mut branch_texts = Vec::new();
1832      loop {
1833        sb(i);
1834        if eat(i, b',') {
1835          match formula(i) {
1836            Ok(arg) => {
1837              if name == "ifthenelse" {
1838                branch_texts.push(i.state.text_result.take());
1839              } else {
1840                number_wanted(i);
1841              }
1842              args.push(arg);
1843              arg_ints.push(i.state.int_result);
1844            },
1845            _ => {
1846              break;
1847            },
1848          }
1849        } else {
1850          break;
1851        }
1852      }
1853      sb(i);
1854      eat(i, b')');
1855      if !is_builtin_function(name) {
1856        open_parse_group(i);
1857        let (value, int) = super::pgfmath_apply_user_reported(name, &args, &arg_ints);
1858        i.state.int_result = int;
1859        return Ok(value);
1860      }
1861      if name == "ifthenelse" && args.len() == 3 {
1862        let chosen = if args[0] != 0.0 { 1 } else { 2 };
1863        i.state.int_result = arg_ints[chosen];
1864        i.state.text_result = branch_texts.swap_remove(chosen - 1);
1865      } else {
1866        set_int_result_for_fn(i, name);
1867      }
1868      return Ok(pgfmath_apply_fn(name, &args));
1869    }
1870    let arg = simplefactor(i)?;
1871    number_wanted(i);
1872    if !is_builtin_function(name) {
1873      open_parse_group(i);
1874      let (value, int) = super::pgfmath_apply_user_reported(name, &[arg], &[i.state.int_result]);
1875      i.state.int_result = int;
1876      return Ok(value);
1877    }
1878    set_int_result_for_fn(i, name);
1879    Ok(pgfmath_apply_fn(name, &[arg]))
1880  }

1929  /// pgf's number (pgfmathparser.code.tex:426-540): digits and points, with the spaces between them skipped as pgf's
1930  /// argument reading skips them (`-4 .5` is -4.5, 1612.03043); then an exponent `e`/`E` and its integer, or after a
1931  /// lone `0` a base prefix `x`/`X`, `b`/`B` and its digits, unless the letter opens a unit (`0bp`, 1106.3771,
1932  /// 1104.5553, 1503.00895). Returns the value and the literal as pgf collected it.
1933  fn try_number(i: &mut In) -> Option<(f64, Literal)> {
1934    sb(i);
1935    let b = i.input.as_bytes();
1936    let skip = |mut at: usize| {
1937      while b.get(at) == Some(&b' ') {
1938        at += 1;
1939      }
1940      at
1941    };
1942    let mut pos = 0;
1943    let mut number = String::new();
1944    loop {
1945      let at = skip(pos);
1946      match b.get(at) {
1947        Some(c) if c.is_ascii_digit() || *c == b'.' => {
1948          number.push(*c as char);
1949          pos = at + 1;
1950        },
1951        _ => break,
1952      }
1953    }
1954    if !number.bytes().any(|c| c.is_ascii_digit()) {
1955      return None;
1956    }
1957    let at = skip(pos);
1958    let next = b.get(at).copied();
1959    let radix = match next {
1960      Some(b'x' | b'X') => 16,
1961      Some(b'b' | b'B') => 2,
1962      _ => 0,
1963    };
1964    if number == "0" && radix != 0 && !opens_unit(&i.input[at..]) {
1965      let mut digits = String::new();
1966      let mut end = at + 1;
1967      loop {
1968        let at = skip(end);
1969        match b.get(at) {
1970          Some(c) if (*c as char).is_digit(radix) => {
1971            digits.push(*c as char);
1972            end = at + 1;
1973          },
1974          _ => break,
1975        }
1976      }
1977      let value = i64::from_str_radix(&digits, radix).ok()? as f64;
1978      bump(i, end);
1979      return Some((value, Literal::Based));
1980    }
1981    if matches!(next, Some(b'e' | b'E')) {
1982      // `\pgfmath@count` reads the exponent as TeX reads an integer: spaces, a sign, digits.
1983      let mut end = skip(at + 1);
1984      let negative = b.get(end) == Some(&b'-');
1985      if matches!(b.get(end), Some(b'+' | b'-')) {
1986        end = skip(end + 1);
1987      }
1988      let digits = b[end.min(b.len())..]
1989        .iter()
1990        .take_while(|c| c.is_ascii_digit())
1991        .count();
1992      if digits > 0 {
1993        let exponent = &i.input[end..end + digits];
1994        let sign = if negative { "-" } else { "" };
1995        let value = format!("{number}e{sign}{exponent}")
1996          .parse::<f64>()
1997          .unwrap_or(0.0);
1998        bump(i, end + digits);
1999        return Some((value, Literal::Scientific));
2000      }
2001    }
2002    let value = number.parse::<f64>().unwrap_or(0.0);
2003    bump(i, pos);
2004    Some((value, Literal::Decimal(number)))
2005  }

2036  /// `perfect_kernel_batch56::pgfmath_reads_expl3_register_names`, row `pgfmath_csname_register_operands`.

2048    }

2062    }

2068  /// the `\csname…\endcsname` that `pgfmath_input_text` spelled it in; `None` for neither.

2102    let number = rest[..digits].parse().ok()?;

2105  }

2174/// Perl: L383-393 of pgfmath.code.tex.ltxml

2177  // an overflow — must NOT serialize as the literal "NaN"/"inf": that string
2178  // poisons every downstream `\pgfmathresult` dimension read. Mirror the
2179  // sibling `pgfmath_result_str` guard (L21) and degrade to 0; the public path
2180  // keeps the `.0`. NB `clamp` below does NOT rescue NaN (`NaN.clamp(..) ==
2181  // NaN`), so this must come first. Review M4.
2182  if result.is_nan() || result.is_infinite() {
2183    return "0.0".to_string();

2187  let result = if fpu_active() {
2188    result
2189  } else {
2190    if exceeds_max_dimen(result) {
2191      report_pgfmath_error(
2192        "pgfmath",
2193        "overflow",

2196    }
2197    result.clamp(-MAX_PGF_NUMBER, MAX_PGF_NUMBER)
2198  };

2200  // Integer result
2201  if result == (result as i64) as f64 {
2202    let i = result as i64;
2203    // Perl: $result .= '.0' unless $input =~ /^int\(/;

2205      return format!("{}", i);
2206    }

2214      s.pop();
2215    }
2216  }

2218}

2283/// The parser's text of an expanded argument: its string, keeping the space TeX's mouth skipped after a control word
2284/// where a letter follows (`\mydim r` is `\mydim` then `r`, not `\mydimr` as Perl's pgfmath.code.tex.ltxml:328 reads;
2285/// KNOWN_PERL_ERRORS #1226). A name its text would not read back as one token (`\pal@data1H`, `\l_x_dim`) is spelled
2286/// `\csname…\endcsname`, so every reader of the text gets the token TeX holds (`pgfmath_grammar::cs_at`).
2287fn pgfmath_input_text(tokens: Tokens) -> String {
2288  let mut spaced = Vec::with_capacity(tokens.unlist_ref().len());
2289  let mut after_control_word = false;
2290  for t in tokens.unlist() {
2291    let code = t.get_catcode();
2292    if after_control_word && code == Catcode::LETTER {
2293      spaced.push(T_SPACE!());
2294    }
2295    if code == Catcode::CS
2296      && let Some(name) = t.with_str(|cs| {
2297        let name = cs.strip_prefix('\\').unwrap_or(cs);
2298        (name.chars().nth(1).is_some()
2299          && !name.bytes().all(|c| c.is_ascii_alphabetic() || c == b'@'))
2300        .then(|| name.to_string())
2301      })
2302    {
2303      spaced.extend([
2304        T_CS!("\\csname"),
2305        T_SPACE!(),
2306        T_OTHER!(&name),
2307        T_CS!("\\endcsname"),
2308      ]);
2309      after_control_word = true;
2310      continue;
2311    }
2312    // A control word as tex.web's print_cs (§262) reads one: a name of two or more characters, or a single letter.
2313    after_control_word = code == Catcode::CS
2314      && t.with_str(|name| {
2315        let mut chars = name.strip_prefix('\\').unwrap_or(name).chars();
2316        match (chars.next(), chars.next()) {
2317          (Some(c), None) => lookup_catcode(c) == Some(Catcode::LETTER),
2318          (first, _) => first.is_some(),
2319        }
2320      });
2321    spaced.push(t);
2322  }
2323  Tokens::new(spaced).to_string()
2324}

2326/// Main pgfmathparse evaluation function
2327/// Perl: sub pgfmathparse (L316-394)
2328/// Returns (result_string, units_declared)
2329pub(crate) fn pgfmathparse_eval_with_units(raw_input: &str) -> (String, bool) {
2330  // Normalize whitespace
2331  let input: String = raw_input.split_whitespace().collect::<Vec<_>>().join(" ");
2332  let mut input = input.trim();
2333  // `\pgfmath@bgroup@strip` takes a whole formula's braces (pgfmathparser.code.tex:139); the group under the last pair
2334  // the grammar reads, an array or a grouping (`{{"a","b"}}` is `{a}{b}`)
2335  while let Some(inner) = input
2336    .strip_prefix('{')
2337    .and_then(|rest| rest.strip_suffix('}'))
2338    && pgfmath_grammar::matching_brace(input) == Some(input.len() - 1)
2339    && inner.trim_start().starts_with('{')
2340    && pgfmath_grammar::matching_brace(inner.trim()) == Some(inner.trim().len() - 1)
2341  {
2342    input = inner.trim();
2343  }
2344
2345  // 0. String-valued ternary / ifthenelse — Perl pgfmath uses untyped scalars so the `?:` operator
2346  //    and `ifthenelse(...)` can return string values like `"black"` or `"pgreen!50"` for color
2347  //    names. Our parser is f64-only, so detect the pattern <test> ? "a" : "b"     OR
2348  //    ifthenelse(<test>, "a", "b") early, evaluate the test as a number, and return the chosen
2349  //    string literal. Drivers: 2601.14798 (color jitter heatmap), Stage-20 v6 ~1579 errors from
2350  //    \pgfmathsetmacro{\clr}{...}.
2351  if let Some(result) = try_string_ternary(input) {
2352    return (result, false);
2353  }
2354
2355  // 0b. Bare string literal (arXiv-fork e0dd3e33/aeb24f2d: string results
2356  //     pass through untouched, including the empty string) — PGF strings
2357  //     like `\pgfmathparse{"south east"}` feed anchors/colors verbatim.
2358  {
2359    let trimmed = input.trim();
2360    if trimmed.len() >= 2 && trimmed.starts_with('"') && trimmed.ends_with('"') {
2361      let inner = &trimmed[1..trimmed.len() - 1];
2362      if !inner.contains('"') {
2363        return (inner.to_string(), false);
2364      }
2365    }
2366  }
2367
2368  // 1. Simple number check (Perl L332-345)
2369  if let Some(result) = try_simple_number(input) {
2370    return (result, false);
2371  }
2372
2373  // 2. Unit expression check (Perl L352-354): /^([+-]?[\d\.]+)(UNIT)$/ Handled by the parser below,
2374  //    since it's a subset of the grammar.
2375
2376  // 3. Parse with the winnow grammar (replaces both Perl eval and RecDescent; partial parses still
2377  //    return what was read, as before).
2378  let (parsed, state) = pgfmath_grammar::evaluate(input);
2379  if parsed.is_ok()
2380    && let Some(text) = state.text_result
2381  {
2382    return (text, false);
2383  }
2384  if let Ok(result) = parsed {
2385    // arXiv-fork c52d3cf6: a top-level comparison returns an integer string.
2386    if state.int_result && result == (result as i64) as f64 {
2387      return (format!("{}", result as i64), state.units_declared);
2388    }
2389    return (format_parse_result(result, input), state.units_declared);
2390  }
2391
2392  // 4. Fallback
2393  ("0.0".to_string(), false)
2394}

2400/// Detect string-valued pgfmath ternaries `<test> ? "a" : "b"` or
2401/// `ifthenelse(<test>, "a", "b")` and return the chosen literal.
2402/// Returns None if the pattern doesn't match or contains no string args.
2403fn try_string_ternary(input: &str) -> Option<String> {
2404  // Strip ifthenelse(...) wrapper if present
2405  let inner = if let Some(rest) = input.strip_prefix("ifthenelse(") {
2406    let body = rest.strip_suffix(')')?;
2407    // Re-shape as ternary: split on top-level commas (depth 0 of nested parens/quotes)
2408    let parts = split_top_commas(body)?;
2409    if parts.len() != 3 {
2410      return None;
2411    }
2412    format!(
2413      "{} ? {} : {}",
2414      parts[0].trim(),
2415      parts[1].trim(),
2416      parts[2].trim()
2417    )
2418  } else {
2419    input.to_string()
2420  };
2421  // Find top-level `?` and `:` (skipping inside quotes and parens)
2422  let (cond, then_part, else_part) = split_ternary(&inner)?;
2423  // Branches must include at least one quoted string for this path to make sense
2424  let then_is_str = then_part.trim().starts_with('"');
2425  let else_is_str = else_part.trim().starts_with('"');
2426  if !then_is_str && !else_is_str {
2427    return None;
2428  }
2429  // Evaluate the test as a number via the regular f64 grammar
2430  let test_val = pgfmath_grammar::evaluate(cond.trim()).0.ok()?;
2431  let chosen = if test_val != 0.0 {
2432    then_part
2433  } else {
2434    else_part
2435  };
2436  // Strip outer quotes if present; a branch reading an array (`{"a","b"}[0]`) is a formula; otherwise return as-is
2437  let trimmed = chosen.trim();
2438  if trimmed.starts_with('{') {
2439    return Some(pgfmathparse_eval_with_units(trimmed).0);
2440  }
2441  let unq = if trimmed.starts_with('"') && trimmed.ends_with('"') && trimmed.len() >= 2 {
2442    &trimmed[1..trimmed.len() - 1]
2443  } else {
2444    trimmed
2445  };
2446  Some(unq.to_string())
2447}

2449/// Split a string on commas at the top level (depth 0), skipping
2450/// nested parens/brackets and quoted substrings.
2451fn split_top_commas(s: &str) -> Option<Vec<&str>> {
2452  let bytes = s.as_bytes();
2453  let mut parts = Vec::new();
2454  let mut start = 0usize;
2455  let mut depth: i32 = 0;
2456  let mut in_quote = false;
2457  for (i, &b) in bytes.iter().enumerate() {
2458    match b {
2459      b'"' => in_quote = !in_quote,
2460      b'(' | b'[' | b'{' if !in_quote => depth += 1,
2461      b')' | b']' | b'}' if !in_quote => depth -= 1,
2462      b',' if !in_quote && depth == 0 => {
2463        parts.push(&s[start..i]);
2464        start = i + 1;
2465      },
2466      _ => {},
2467    }
2468  }
2469  parts.push(&s[start..]);
2470  Some(parts)
2471}

2473/// Split `cond ? then : else` at the top-level `?` and `:`.
2474/// Skips inside quotes and parens. Returns (cond, then, else).
2475fn split_ternary(s: &str) -> Option<(&str, &str, &str)> {
2476  let bytes = s.as_bytes();
2477  let mut depth: i32 = 0;
2478  let mut in_quote = false;
2479  let mut q_pos: Option<usize> = None;
2480  for (i, &b) in bytes.iter().enumerate() {
2481    match b {
2482      b'"' => in_quote = !in_quote,
2483      b'(' | b'[' | b'{' if !in_quote => depth += 1,
2484      b')' | b']' | b'}' if !in_quote => depth -= 1,
2485      b'?' if !in_quote && depth == 0 => {
2486        q_pos = Some(i);
2487        break;
2488      },
2489      _ => {},
2490    }
2491  }
2492  let qp = q_pos?;
2493  // Now find the matching `:` after qp at depth 0
2494  let mut depth: i32 = 0;
2495  let mut in_quote = false;
2496  for (j, &b) in bytes.iter().enumerate().skip(qp + 1) {
2497    match b {
2498      b'"' => in_quote = !in_quote,
2499      b'(' | b'[' | b'{' if !in_quote => depth += 1,
2500      b')' | b']' | b'}' if !in_quote => depth -= 1,
2501      b':' if !in_quote && depth == 0 => {
2502        return Some((&s[..qp], &s[qp + 1..j], &s[j + 1..]));
2503      },
2504      _ => {},
2505    }
2506  }
2507  None
2508}

2713    // '+'-prefixed (relative/glue) value through native glue reading, which CAN
2714    // read box dimension registers like `\wd<box>` that the pgfmath expression
2715    // parser cannot (pgfmath returns 0 for `\wd`/`\ht`/`\dp`). The value is very

2718    // `\pgf@decorate@width` expands to `+.5\wd\pgf@lib@dec@text@box`. Testing the
2719    // *raw* first token sees the macro (not '+'), so we fell to pgfmathparse,
2720    // `\wd<box>` → 0, the decoration never advanced along the path, and
2721    // `text along path` (decorations.text) looped forever placing boxes → RSS
2722    // runaway. Expanding first lets the '+' surface (pdflatex ground truth:
2723    // `\pgfmathsetlength\d{+.5\wd0}` = 3.75pt). Witnesses: arXiv 1709.07916 /
2724    // 1912.13052 / 2004.14791 / 1312.6499 (STABILITY_WITNESSES Cluster E).

2735      // assigns `#1#2\unskip` (pgfmathcalc.code.tex:30-38), so the scan ends with the
2736      // argument. Read from the live input, the optional space and `plus` keyword scan after
2737      // the unit expanded the next token — `\ifdim\pgfdecoratedremainingdistance<\pgf@x`
2738      // (pgfmoduledecorations.code.tex:1020-1035) then compared the stale `\pgf@x`, and text

2740      // takes a dimension (no `plus`/`minus`), a skip register glue. Perl reads the live input
2741      // too (pgfmath.code.tex.ltxml:412-416).

2746      // `\nullfont` (KNOWN_PERL_ERRORS #437); the register is assigned after it, and the group closes on an error.

2750        assign_font(Rc::new(font), Some(Scope::Local));
2751      }
2752      let read = Mouth::new("", None).and_then(|mouth| reading_from_mouth(mouth, move || {
2753        unread(Tokens::new(toks));
2754        let value: RegisterValue = match register_type {
2755          Some(RegisterType::Glue) => read_glue()?.into(),
2756          Some(RegisterType::MuGlue) => read_mu_glue()?.into(),
2757          Some(RegisterType::MuDimension) => read_mu_dimension()?.into(),
2758          _ => read_dimension()?.into(),
2759        };

2762          rest.push(token);
2763        }
2764        // pgf's `\unskip` after `#2` removes a trailing space (pgf's decoration states pass
2765        // `+.5\pgfdecorationsegmentlength ` with one).
2766        while rest.last().is_some_and(|t| t.get_catcode() == Catcode::SPACE) {
2767          rest.pop();
2768        }
2769        Ok((value, rest))
2770      }));
2771      endgroup()?;
2772      let (value, rest): (RegisterValue, Vec<Token>) = read?;
2773      let cs = register.to_string();

2775      if !rest.is_empty() {

2778    } else {

2780      // `\pgfmath@selectfont`'s (`pgfmath_convert`), and it sets both units flags globally.

2783
2784      // Perl L419-422: check \ifpgfmathmathunitsdeclared for mu units
2785      let cs = register.to_string();
