7LoadDefinitions!({
8  DefEnvironment!("{wrapfigure} [Number] Position [Dimension] {Dimension}",
9    "<ltx:figure xml:id='#id' inlist='#inlist' float='#float' ?#width(width='#width')>#tags#body</ltx:figure>",
10    mode => "internal_vertical",
11    after_digest_begin => sub[whatsit] {
12      // INTENTIONAL DIVERGENCE from Perl wrapfig.sty.ltxml (which captures the
13      // mandatory {Dimension} wrap width as arg 4 but then DISCARDS it): emit it
14      // as the figure @width (→ base-styling `width:`) so the float — image AND
15      // caption — is capped to the declared wrap width instead of expanding to
16      // fit a single-line caption (which leaves a small figure in an enormous
17      // box). Mirrors the {minipage} width idiom. See OXIDIZED_DESIGN.
18      set_wrap_float(whatsit, 2, 4);
19    },
20    // The begin code opens a TeX group of its own: wrapfig.sty:58 `\vtop\bgroup`.
21    tex_group => { TexGroup::Box },
22    before_digest => { before_float("figure", None) },
23    after_digest => sub[whatsit] { after_float(whatsit); Ok(Vec::new()) }
24  );

35    tex_group => { TexGroup::Box },

80/// The placement (argument `dir_arg`, `r` or `l`) and the wrap width (argument `width_arg`) of a wrapped float.
81fn set_wrap_float(whatsit: &mut Whatsit, dir_arg: usize, width_arg: usize) {
82  let dir = whatsit
83    .get_arg(dir_arg)
84    .map(|a| a.to_string())
85    .unwrap_or_default();
86  // Perl wrapfig.sty.ltxml L29/L43: `eq 'r'` / `eq 'l'` — exact
87  // case-sensitive match. Prior Rust also matched `R`/`L` (uppercase);
88  // drop per feedback_case_folding_parity.
89  let float_val = match dir.trim() {
90    "r" => "right",
91    "l" => "left",
92    _ => "",
93  };
94  if !float_val.is_empty() {
95    whatsit.set_property("float", Stored::String(pin(float_val)));
96  }
97  set_wrap_width(whatsit, width_arg);
98}

100// Emit the wrapfig/wraptable mandatory {Dimension} (the declared wrap
101// width) as the float's `width` property so it renders as a CSS `width:`,
102// capping the float (image + caption) to that width. Perl's wrapfig binding
103// discards this arg; applying it is an intentional ar5iv-rendering divergence.
104//
105// Emit it as a PERCENTAGE of \textwidth, not the resolved absolute pt. A
106// wrapfigure width is a layout reservation -- the fraction of the line the float
107// occupies (authors write it proportionally, e.g. 0.48\linewidth). In print the
108// column is fixed so absolute == relative; in responsive HTML the main column is
109// far wider than the print \textwidth, so an absolute pt renders far narrower
110// than the intended fraction (0.48\linewidth -> ~165pt -> ~26% of an 832px
111// column instead of ~48%). A percentage preserves the proportion and scales.
112// This also makes wrapfig consistent with the floatfig/floatflt sibling
113// packages, which already emit `100 * dimen / \textwidth . '%'` (Perl toPercent).
114fn set_wrap_width(whatsit: &mut Whatsit, width_arg: usize) {
115  let Some(width_arg) = whatsit.get_arg(width_arg) else {
116    return;
117  };
118  let v = width_arg.value_of();
119  if v == 0 {
120    return;
121  }
122  // Cap the float's INNER line width to the declared wrap width, so a body
123  // `\includegraphics[width=\linewidth]` / `\rule{\linewidth}` / nested box is
124  // sized to the float and not to the full page. Real LaTeX wrapfig does this
125  // (`\hsize<width>` then `\@parboxrestore`'s `\linewidth\hsize`); Perl's
126  // binding discards the width arg entirely, so its wrapfig image renders at the
127  // full page width and overflows the narrow float, colliding with the wrapped
128  // text (SHARED-FAILURE). Witness arXiv 2603.23669 Fig. 3: a 0.296\textwidth
129  // wrapfigure whose 512px tikz image overran the 29% float. `\textwidth` is
130  // left alone (the @width percentage below is relative to the OUTER textwidth,
131  // and wrapfig itself never rescales \textwidth). Runs in after_digest_begin,
132  // before the body digests, so the reduced \linewidth reaches the image.
133  // OXIDIZED_DESIGN #29 (extends the width-emission divergence).
134  let rv: RegisterValue = Dimension::new(v).into();
135  let _ = assign_register("\\hsize", rv.clone(), None, Vec::new());
136  let _ = assign_register("\\columnwidth", rv.clone(), None, Vec::new());
137  let _ = assign_register("\\linewidth", rv, None, Vec::new());
138  let Some(tw) = lookup_dimension("\\textwidth") else {
139    return;
140  };
141  let tw_sp = tw.value_of();
142  if tw_sp == 0 {
143    return;
144  }
145  let pct = (100 * v) / tw_sp;
146  whatsit.set_property("width", Stored::from(s!("{pct}%")));
147}
