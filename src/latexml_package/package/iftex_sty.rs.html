97/// XeTeX's inter-character primitives, for the packages that DECLARE themselves
98/// XeTeX-only: `\RequireXeTeX` installs them (ucharclasses.sty:987, then its
99/// `\XeTeXcharclass` loops over whole Unicode blocks). With them undefined, the
100/// loop `\XeTeXcharclass \count \class` degraded into a spurious assignment and
101/// never ended (latexbangla: 95k warnings, then a 6.3 GB `alloc_failed`; the
102/// sweep-57 cluster). They read their real arguments — `<number> [=] <number>`,
103/// `<number> <number> [=] {<tokens>}`, an integer parameter — and change nothing:
104/// the classes drive font switching the Unicode-native engine does not need.
105/// Installed on request, or by latexml.sty's `xetex` profile (run at the format
106/// seam, `latex.rs`), because amsmath, mathastext, minted2 and others probe
107/// `\XeTeXcharclass` to detect XeTeX (ordinary pdfLaTeX documents must not take
108/// those branches). `\newXeTeXintercharclass` is latex.ltx:22025-22031's
109/// allocator, over a fresh counter (latex.ltx `\countdef`s register 257, which
110/// the pdfTeX-persona format may already use).
111pub fn define_xetex_interchar() -> Result<()> {
112  // The optional `=` is scanned with expansion (tex.web §405), as for every register.
113  DefPrimitive!("\\lx@XeTeXcharclass Number SkipKeyword:= Number", sub[(_c, _k)] {});
114  DefPrimitive!("\\lx@XeTeXinterchartoks Number Number SkipKeyword:= {}", sub[(_a, _b, _t)] {});
115  DefRegister!("\\lx@XeTeXinterchartokenstate" => Number::new(0));
116  RawTeX!(
117    r"\def\lx@xetex@interchar{\ifx\XeTeXcharclass\@undefined
118  \let\XeTeXcharclass\lx@XeTeXcharclass
119  \let\XeTeXinterchartoks\lx@XeTeXinterchartoks
120  \let\XeTeXinterchartokenstate\lx@XeTeXinterchartokenstate
121  \chardef\e@alloc@intercharclass@top=4095
122  \newcount\xe@alloc@intercharclass
123  \def\newXeTeXintercharclass{\e@alloc\XeTeXcharclass\chardef\xe@alloc@intercharclass\m@ne\e@alloc@intercharclass@top}%
124  \fi}"
125  );
126  Ok(())
