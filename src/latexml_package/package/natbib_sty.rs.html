27/// Run natbib's style option `option` as its `\DeclareOption` code does (natbib.sty:238-263), the options its
28/// `\ExecuteOptions` names in their order.
29fn natbib_style_option(option: &str) {
30  match option {
31    "numbers" => {
32      set_citation_style(&[("numbers", None)]);
33      for sub in ["square", "comma", "nobibstyle"] {
34        natbib_style_option(sub);
35      }
36    },
37    "super" => {
38      set_citation_style(&[("super", None)]);
39      assign_value("CITE_OPEN", Stored::Tokens(Tokens!()), None);
40      assign_value("CITE_CLOSE", Stored::Tokens(Tokens!()), None);
41      natbib_style_option("nobibstyle");
42    },
43    "authoryear" => {
44      set_citation_style(&[("authoryear", None)]);
45      for sub in ["round", "semicolon", "bibstyle"] {
46        natbib_style_option(sub);
47      }
48    },
49    "round" | "square" | "angle" | "curly" | "comma" | "semicolon" => {
50      set_citation_style(&[(option, None)]);
51      natbib_style_option("nobibstyle");
52    },
53    "colon" => natbib_style_option("semicolon"),
54    "sort" => assign_value("CITE_SORTCITES", true, None),
55    "compress" => assign_value("CITE_COMPRESS", true, None),
56    "sort&compress" => {
57      natbib_style_option("sort");
58      natbib_style_option("compress");
59    },
60    "nobibstyle" => {
61      Let!("\\bibstyle", "\\@gobble");
62    },
63    "bibstyle" => {
64      Let!("\\bibstyle", "\\@citestyle");
65    },
66    _ => {},
67  }
68}

199// Helper: handle the [pre][post] optional arg swap (natbib.sty.ltxml:113 `if (!$post) { ($pre, $post) = (undef, $pre) }`,
200// then :114-115 an empty one dropped). Only an ABSENT post-note swaps: a Perl Tokens is a true object even when empty,
201// so `\citep[e.g.][]{k}` keeps "e.g." the pre-note, "(e.g. Rigden, 2017)" as natbib.sty prints it (`\NAT@@citetp`,
202// natbib.sty:688-690). The reader gives `[]` as `Some(empty)` (gullet.rs `read_optional`).
203fn swap_pre_post(pre: Option<Tokens>, post: Option<Tokens>) -> (Option<Tokens>, Option<Tokens>) {
204  if post.is_none() {
205    (None, pre.filter(|t| !t.is_empty()))
206  } else {
207    (
208      pre.filter(|t| !t.is_empty()),
209      post.filter(|t| !t.is_empty()),
210    )
211  }
212}

214/// `cs` re-invoked over a natbib citation's arguments as they were read: a present note is `[…]` even when empty, which
215/// natbib's commands tell from an absent one (`swap_pre_post`), and a post-note given without a pre-note has an empty
216/// one before it. `Invocation!` reverts an empty optional argument to nothing, as Perl's `Optional` reversion does
217/// (Base_ParameterTypes.pool.ltxml:55-62): `\citealt[e.g.][]{k}` became `\citet[e.g.]{k}`, its pre-note the post-note
218/// (KNOWN_PERL_ERRORS #868; natbib.sty:702-730 passes both to `\NAT@citetp`). The bindings that delegate to natbib
219/// (apacite's `\cite<pre>[post]`, nmbib's `\citeall`) re-emit their notes through it too.
220pub fn natbib_invocation(
221  cs: &str,
222  star: Option<Tokens>,
223  pre: Option<Tokens>,
224  post: Option<Tokens>,
225  keys: Tokens,
226) -> Tokens {
227  let mut tokens = vec![T_CS!(cs)];
228  if is_some_and_nonempty(&star) {
229    tokens.push(T_OTHER!("*"));
230  }
231  if pre.is_some() || post.is_some() {
232    tokens.push(T_OTHER!("["));
233    tokens.extend(pre.map(|p| p.revert()).unwrap_or_default());
234    tokens.push(T_OTHER!("]"));
235  }
236  if let Some(post) = post {
237    tokens.push(T_OTHER!("["));
238    tokens.extend(post.revert());
239    tokens.push(T_OTHER!("]"));
240  }
241  tokens.push(T_BEGIN!());
242  tokens.extend(keys.revert());
243  tokens.push(T_END!());
244  Tokens::new(tokens)
245}

247/// A label piece whose math shifts are balanced within each brace group (see `\NAT@wrout`): a group (or the piece)
248/// left in math is closed before its `}` (or at the piece's end), or, when its last token is that unpaired `$` (a
249/// BibTeX name split as `{Sasaki$}`), the `$` is dropped, so that no `$$` is formed.
250fn balance_math_shifts(piece: Option<Tokens>) -> Option<Tokens> {
251  piece.map(|tokens| {
252    let mut balanced = Vec::with_capacity(tokens.len() + 1);
253    // One parity per open brace group, the piece's own at the bottom.
254    let mut open_math = vec![false];
255    let close = |balanced: &mut Vec<Token>| {
256      if balanced
257        .last()
258        .is_some_and(|t: &Token| t.get_catcode() == Catcode::MATH)
259      {
260        balanced.pop();
261      } else {
262        balanced.push(T_MATH!());
263      }
264    };
265    for token in tokens.unlist() {
266      match token.get_catcode() {
267        Catcode::MATH => *open_math.last_mut().unwrap() ^= true,
268        Catcode::BEGIN => open_math.push(false),
269        Catcode::END if open_math.len() > 1 => {
270          if open_math.pop() == Some(true) {
271            close(&mut balanced);
272          }
273        },
274        _ => {},
275      }
276      balanced.push(token);
277    }
278    if open_math.pop() == Some(true) {
279      close(&mut balanced);
280    }
281    Tokens::new(balanced)
282  })
283}

285/// The binding's command for the citation form natbib's flags select (natbib.sty:685-749): `\NAT@ctype` 0 a
286/// citation, 1 the author, 2 the year, 3 the alias; `\ifNAT@par` parenthesised, `\ifNAT@swa` the citation's
287/// bracketed (`p`) form, `\ifNAT@full` the starred one. The saved `\lx@natbib@…` copies are used, so a document's
288/// `\citealp` redefined over `\NAT@citetp` does not call itself. The caller's `\begingroup` is closed after it, as
289/// natbib's formatters close it (:355-362).
290fn cite_by_flags(
291  pre: Option<Tokens>,
292  post: Option<Tokens>,
293  keys: Option<Tokens>,
294) -> Result<Tokens> {
295  let same = |cs: &str, other: &str| {
296    let meaning = lookup_meaning(&T_CS!(cs));
297    meaning.is_some() && meaning == lookup_meaning(&T_CS!(other))
298  };
299  let par = same("\\ifNAT@par", "\\iftrue");
300  let swa = same("\\ifNAT@swa", "\\iftrue");
301  let star = same("\\ifNAT@full", "\\iftrue").then(|| Tokens!(T_OTHER!("*")));
302  let ctype = ["\\z@", "\\@ne", "\\tw@", "\\thr@@"]
303    .iter()
304    .position(|cs| same("\\NAT@ctype", cs))
305    .unwrap_or(0);
306  let (command, star) = match (ctype, par, swa) {
307    (1, ..) => ("\\lx@natbib@citeauthor", star),
308    (2, true, _) => ("\\lx@natbib@citeyearpar", None),
309    (2, false, _) => ("\\lx@natbib@citeyear", None),
310    (3, true, _) => ("\\lx@natbib@citepalias", None),
311    (3, false, _) => ("\\lx@natbib@citetalias", None),
312    (_, true, true) => ("\\lx@natbib@citep", star),
313    (_, true, false) => ("\\lx@natbib@citet", star),
314    (_, false, true) => ("\\lx@natbib@citealp", star),
315    (_, false, false) => ("\\lx@natbib@citealt", star),
316  };
317  let mut result = natbib_invocation(command, star, pre, post, keys.unwrap_or_default()).unlist();
318  result.push(T_CS!("\\endgroup"));
319  Ok(Tokens::new(result))
320}

355    });
356  }
357  Ok(())

360/// The keys of a merging citation list, read as natbib.sty:1135-1160 `\NAT@star@cite` reads each one (`\@for` splits the
361/// list at its commas outside braces): a `*` prefix and the `[pre][post]` notes after it dropped. natbib prints the notes
362/// in the entry's bibliography text (`\nat@keyopt@open`, `\nat@keyopt@shut`; not done here, KPE #1081): 1509.05768's
363/// `\cite{[See][and references therein.]pra78}`, 1206.6986's `[][{, Chap.~2.}]Peresbook`.
364fn merged_citation_keys(keys: Tokens) -> Tokens {
365  let is = |t: &Token, c: &str| t.get_catcode() == Catcode::OTHER && t.to_string() == c;
366  let mut pieces: Vec<Vec<Token>> = vec![Vec::new()];
367  let mut depth = 0i32;
368  for token in keys.unlist() {
369    match token.get_catcode() {
370      Catcode::BEGIN => depth += 1,
371      Catcode::END => depth -= 1,
372      _ => {},
373    }
374    if depth == 0 && is(&token, ",") {
375      pieces.push(Vec::new());
376    } else if let Some(piece) = pieces.last_mut() {
377      piece.push(token);
378    }
379  }
380  let mut out = Vec::new();
381  for (index, piece) in pieces.into_iter().enumerate() {
382    let mut key = piece
383      .into_iter()
384      .skip_while(|t| t.get_catcode() == Catcode::SPACE)
385      .peekable();
386    if key.peek().is_some_and(|t| is(t, "*")) {
387      key.next();
388    }
389    let mut key: Vec<Token> = key
390      .skip_while(|t| t.get_catcode() == Catcode::SPACE)
391      .collect();
392    for _ in 0..2 {
393      if !key.first().is_some_and(|t| is(t, "[")) {
394        break;
395      }
396      let mut depth = 0i32;
397      let close = key.iter().position(|t| {
398        match t.get_catcode() {
399          Catcode::BEGIN => depth += 1,
400          Catcode::END => depth -= 1,
401          _ => {},
402        }
403        depth == 0 && is(t, "]")
404      });
405      let Some(close) = close else { break };
406      key.drain(..=close);
407      let spaces = key
408        .iter()
409        .take_while(|t| t.get_catcode() == Catcode::SPACE)
410        .count();
411      key.drain(..spaces);
412    }
413    if index > 0 {
414      out.push(T_OTHER!(","));
415    }
416    out.extend(key);
417  }
418  Tokens::new(out)
419}

421fn natbib_cite(
422  star: Option<Tokens>,
423  pre_raw: Option<Tokens>,
424  post_raw: Option<Tokens>,
425  keys: Tokens,
426  parenthetical: bool,
427) -> Result<Tokens> {
428  let style = cite_style();
429  let open = cite_open();
430  let close = cite_close();
431  let ns = cite_ns();
432  let ay = cite_ay();
433
434  let (pre, post) = swap_pre_post(pre_raw, post_raw);
435  let is_star = is_some_and_nonempty(&star);
436  let author = if is_star { "FullAuthors" } else { "Authors" };
437
438  if style == "numbers" {
439    // Invocation(\@@cite, Tokens(Explode('cite')),
440    //   Tokens(open, (pre? (pre, T_SPACE)),
441    //     Invocation(\@@bibref, Tokens(Explode("Number")), keys, undef, undef),
442    //     (post? (ns, T_SPACE, post)), close))
443    let bibref = Invocation!(T_CS!("\\@@bibref"), vec![
444      Tokens::new(Explode!("Number")),
445      keys,
446      Tokens!(),
447      Tokens!()
448    ]);
449    let mut body = open.unlist();
450    if let Some(p) = pre {
451      body.extend(p.unlist());
452      body.push(T_SPACE!());
453    }
454    body.extend(bibref.unlist());
455    if let Some(p) = post {
456      body.extend(ns.unlist());
457      body.push(T_SPACE!());
458      body.extend(p.unlist());
459    }
460    body.extend(close.unlist());
461    Ok(Invocation!(T_CS!("\\lx@@cite"), vec![
462      Tokens::new(Explode!("cite")),
463      Tokens::new(body)
464    ]))
465  } else if style == "super" {
466    // superscript style
467    let bibref = Invocation!(T_CS!("\\@@bibref"), vec![
468      Tokens::new(Explode!("Number")),
469      keys,
470      Tokens!(),
471      Tokens!()
472    ]);
473    let inside = super_postnote_inside();
474    let mut sup = bibref.unlist();
475    if inside && let Some(p) = &post {
476      sup.extend([T_OTHER!(","), T_SPACE!()]);
477      sup.extend(p.clone().unlist());
478    }
479    let sup_arg = Invocation!(T_CS!("\\textsuperscript"), vec![Tokens::new(sup)]);
480    let mut body = Vec::new();
481    if let Some(p) = pre {
482      body.extend(p.unlist());
483      body.push(T_SPACE!());
484    }
485    body.extend(sup_arg.unlist());
486    if let Some(p) = post.filter(|_| !inside) {
487      body.push(T_SPACE!());
488      body.extend(p.unlist());
489    }
490    Ok(Invocation!(T_CS!("\\lx@@cite"), vec![
491      Tokens::new(Explode!("cite")),
492      Tokens::new(body)
493    ]))
494  } else {
495    // authoryear
496    if parenthetical {
497      // parenthetical, with its pre/post notes
498      let show = s!("{author}Phrase1Year");
499      let mut ay_space = ay.unlist();
500      ay_space.push(T_SPACE!());
501      let phrase1 = Invocation!(T_CS!("\\@@citephrase"), vec![Tokens::new(ay_space)]);
502      let bibref = Invocation!(T_CS!("\\@@bibref"), vec![
503        Tokens::new(Explode!(show)),
504        keys,
505        phrase1,
506        Tokens!()
507      ]);
508      let mut body = open.unlist();
509      if let Some(p) = pre {
510        body.extend(p.unlist());
511        body.push(T_SPACE!());
512      }
513      body.extend(bibref.unlist());
514      if let Some(p) = post {

1129  // `RequiredKeyVals:natbib` (L407-408), whose unknown key is an Info there and
1130  // a warning under the Rust Warn promotion: a package option passed here

1132  // split as natbib splits it; `<word>=<value>` carries its value.
1133  DefPrimitive!("\\setcitestyle{}", sub[(list)] {
1134    let owned: Vec<(String, Option<Tokens>)> = split_tokens(list, vec![T_OTHER!(",").into()])
1135      .into_iter()
1136      .filter_map(|item| {
1137        let toks = item.unlist();
1138        match toks.iter().position(|t| *t == T_OTHER!("=")) {
1139          Some(i) => {
1140            let key = Tokens::new(toks[..i].to_vec()).to_string().trim().to_string();
1141            Some((key, Some(Tokens::new(toks[i + 1..].to_vec()))))
1142          },
1143          None => {
1144            let key = Tokens::new(toks).to_string().trim().to_string();
1145            (!key.is_empty()).then_some((key, None))
1146          },
1147        }
1148      })
1149      .collect();
1150    // natbib.sty's `\@for` tests each word against its own list and ignores
1151    // any other; Perl's fall-through (natbib.sty.ltxml:402-404) instead resets
1152    // the style to authoryear, so `numbers,sort&compress` lost `numbers`.
1153    // Only natbib's words reach the style (OXIDIZED_DESIGN #278).
1154    const WORDS: [&str; 16] = [
1155      "round", "square", "angle", "curly", "semicolon", "colon", "comma", "authoryear",
1156      "numbers", "super", "open", "close", "aysep", "yysep", "notesep", "citesep",
1157    ];
1158    let pairs: Vec<(&str, Option<Tokens>)> = owned
1159      .iter()
1160      .filter(|(k, _)| WORDS.contains(&k.as_str()))
1161      .map(|(k, v)| (k.as_str(), v.clone()))
1162      .collect();
1163    set_citation_style(&pairs);
1164    // A class's own `\setcitestyle` that reads the ordering words as the options do (acmart.cls:306-345,
1165    // `natbib@setcitestyle@orders`): natbib's own ignores them.
1166    if lookup_bool("natbib@setcitestyle@orders") {
1167      for (key, _) in &owned {
1168        match key.as_str() {
1169          "sort" | "compress" | "sort&compress" => natbib_style_option(key),
1170          "nosort" => assign_value("CITE_SORTCITES", false, None),
1171          "nocompress" => assign_value("CITE_COMPRESS", false, None),
1172          _ => {},
1173        }
1174      }

1833// Helper: peel a brace-delimited argument from a token list
1834fn nat_peel_arg(mut tokens: Vec<Token>) -> (Option<Tokens>, Vec<Token>) {
1835  if tokens.is_empty() {
1836    return (None, tokens);
1837  }
1838  if tokens[0].get_catcode() != Catcode::BEGIN {
1839    // Not braced — single token
1840    let t = tokens.remove(0);
1841    return (Some(Tokens::new(vec![t])), tokens);
1842  }
1843  // Braced group
1844  tokens.remove(0); // remove opening brace
1845  let mut arg = Vec::new();
1846  let mut level = 1i32;
1847  while let Some(t) = tokens.first().cloned() {
1848    tokens.remove(0);
1849    let cc = t.get_catcode();
1850    if cc == Catcode::BEGIN {
1851      level += 1;
1852    }
1853    if cc == Catcode::END {
1854      level -= 1;
1855      if level == 0 {
1856        break;
1857      }
1858    }
1859    arg.push(t);
1860  }
1861  let result = if arg.is_empty() {
1862    None
1863  } else {
1864    Some(Tokens::new(arg))
1865  };
1866  (result, tokens)
1867}
