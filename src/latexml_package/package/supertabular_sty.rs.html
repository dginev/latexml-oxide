122  DefMacro!("\\@supertabular@docaption", sub[_args] {
123    let cap = lookup_value("SUPERTABULAR_CAPTION");
124    let toccap = lookup_value("SUPERTABULAR_TOCCAPTION");
125    if let Some(Stored::Tokens(ref c)) = cap {
126      let mut result: Vec<Token> = vec![T_CS!("\\@caption"), T_BEGIN!()];
127      result.extend(Explode!("table"));
128      result.push(T_END!());
129      // The short caption is the caption itself when none is given (supertabular.sty's
130      // `\@dblarg`; KNOWN_PERL_ERRORS #468): a `#1[#2]#3` `\@caption` finds its `[`. Braced,
131      // as `\@xdblarg`'s `[{#2}]`, so a `]` in it ("Dimensions [mm]") does not end it.
132      result.push(T_OTHER!("["));
133      result.push(T_BEGIN!());
134      match toccap {
135        Some(Stored::Tokens(ref tc)) if !tc.is_empty() => result.extend_from_slice(tc.unlist_ref()),
136        _ => result.extend_from_slice(c.unlist_ref()),
137      }
138      result.push(T_END!());
139      result.push(T_OTHER!("]"));
140      result.push(T_BEGIN!());
141      result.extend_from_slice(c.unlist_ref());
142      result.push(T_END!());
143      result
144    } else {
145      Vec::new()
146    }
147  });
