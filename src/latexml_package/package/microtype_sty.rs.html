10  // All microtypography macros are no-ops.
11  // Perl uses `[]` optional args on the `Set*`/`DisableLigatures` family
12  // (context selectors like `[font name]`); Rust was using `OptionalMatch:*`
13  // which consumes a leading `*` not an optional bracket group. Fix to
14  // match Perl signatures so user-side `\SetProtrusion[ctx]{set}{list}`
15  // parses correctly instead of stranding the `[ctx]` as literal text.
16  def_macro_noop("\\microtypesetup{}")?;
17  def_macro_noop("\\DeclareMicrotypeSet OptionalMatch:* []{}{}")?;
18  def_macro_noop("\\DeclareMicrotypeSetDefault[]{}")?;
19  def_macro_noop("\\DeclareMicrotypeAlias{}{}")?;
20  def_macro_noop("\\SetProtrusion[]{}{}")?;
21  def_macro_noop("\\SetTracking[]{}{}")?;
22  def_macro_noop("\\SetExpansion[]{}{}")?;
23  def_macro_noop("\\DisableLigatures[]{}")?;
24  def_macro_noop("\\SetExtraKerning[]{}{}")?;
25  def_macro_noop("\\SetExtraSpacing[]{}{}")?;
26  // \textls passes through #3 (the body)
27  DefMacro!("\\textls OptionalMatch:* []{}", "#3");
28  def_macro_noop("\\lsstyle")?;
29  DefMacro!("\\lslig{}", "#1");
30
31  // Perl L32-46 — additional no-op microtype commands
32  def_macro_noop("\\UseMicrotypeSet[]{}")?;
33  def_macro_noop("\\DeclareCharacterInheritance[]{}{}")?;
34  def_macro_noop("\\DeclareMicrotypeVariants OptionalMatch:* {}")?;
35  def_macro_noop("\\LoadMicrotypeFile{}")?;
36  // microtype provides BOTH `\microtypecontext{settings}` (a scoped
37  // settings *declaration* — no body, applies to the rest of the current
38  // group) AND a `{microtypecontext}` environment. Our `DefEnvironment`
39  // defines the bare `\microtypecontext` CS as the env-begin, which
40  // clobbers the declaration form — so a bare
41  // `\begingroup\microtypecontext{expansion=sloppy}…\endgroup` (common
42  // around `\bibliography`) treated `\microtypecontext` as an unclosed
43  // env-begin, opening a restricted_horizontal mode-switch group that
44  // the later `\endgroup` couldn't close (`unexpected:\endgroup`).
45  // Define the environment FIRST, then the no-op declaration macro, so
46  // `\microtypecontext{…}` resolves to the harmless declaration while
47  // `\begin{microtypecontext}` still finds the environment (env lookup
48  // is independent of the `\microtypecontext` CS). Witness 2007.06927.
49  // No `{microtypecontext}` ENVIRONMENT: real microtype.sty:80 only defines
50  // the `\microtypecontext{…}` declaration, so `\endmicrotypecontext` stays
51  // undefined until a `\begin{microtypecontext}` resolves it on the fly —
52  // synthslant.sty:302 probes `\ifcsdef{endmicrotypecontext}` and calls it
53  // when TRUE (a live env-end here → "Attempt to end mode
54  // restricted_horizontal", 101 errors in synthslant-gauge). The generic
55  // `\begin`/`\end` path still handles the env form through the no-op
56  // declaration (2007.06927 `\begingroup\microtypecontext{…}\endgroup`).
57  def_macro_noop("\\microtypecontext{}")?;
58  DefMacro!("\\textmicrotypecontext{}{}", "#2");
59  def_macro_noop("\\DeclareMicrotypeBabelHook{}{}")?;
60  // microtype.sty:36-37 `\def\MT@MT{microtype}` — the marker typog.sty:68
61  // tests with `\ifdefined\MT@MT` (else "package microtype not (pre-)loaded"
62  // under `trackingttspacing`; typog-example) — and the package's `\ifMT@*`
63  // switches (microtype.sty:194-205,233) a client may test.
64  RawTeX!(
65    r"\def\MT@MT{microtype}
66\newif\ifMT@protrusion \newif\ifMT@expansion \newif\ifMT@auto \newif\ifMT@selected
67\newif\ifMT@noligatures \newif\ifMT@draft \newif\ifMT@disable \newif\ifMT@spacing
68\newif\ifMT@kerning \newif\ifMT@tracking \newif\ifMT@babel \newif\ifMT@document \newif\ifMT@fontspec"
69  );
70  // microtype.sty:3162-3195: with `babel` and `kerning` both on (default off)
71  // and babel loaded, microtype's setup switches off French babel's active
72  // `:;!?` (Turkish: `:!=`) at `\begin{document}`, since its own kerning does
73  // that job. Code written in the preamble relies on it: cahierprof.sty:366-388
74  // freezes a `\tikzmath{…; …}` whose `;` must stay other-catcode when its
75  // `\AtBeginDocument` hook runs (:671-676), or tikzmath picks the active-`;`
76  // delimiter and the statements leak (cahierprof-doc, -exemple; SHARED with
77  // Perl's stub). Registered at load, so it runs before a later package's hook.
78  if let Some(opts) = lookup_vecdeque("opt@microtype.sty") {
79    for opt in opts.iter() {
80      let opt = opt.to_string();
81      let (key, value) = opt
82        .split_once('=')
83        .map_or((opt.trim(), "true"), |(k, v)| (k.trim(), v.trim()));
84      if matches!(key, "babel" | "kerning") {
85        RawTeX!(&s!(
86          "\\MT@{key}{}",
87          if value == "false" { "false" } else { "true" }
88        ));
89      }
90    }
91  }
92  DefMacro!("\\lx@MT@babel@shorthandoff", {
93    // `\ifMT@babel`/`\ifMT@kerning` true and babel loaded; a switch that is gone
94    // (microtype loaded inside a group: fullwidth) counts as off.
95    let on = |cs: &str| x_equals(&T_CS!(cs), &T_CS!("\\iftrue"));
96    if !on("\\ifMT@babel") || !on("\\ifMT@kerning") || !package_loaded("babel") {
97      return Ok(Tokens::default());
98    }
99    let listed = |name: &str| -> Result<bool> {
100      let opts = do_expand(TokenizeInternal!(
101        r"\csname opt@babel.sty\endcsname,\@classoptionslist"
102      ))?
103      .to_string();
104      Ok(opts.split(',').any(|o| o.trim() == name))
105    };
106    let mut out = Tokens::default();
107    if ["french", "frenchb", "francais", "canadien", "acadian"]
108      .iter()
109      .map(|n| listed(n))
110      .collect::<Result<Vec<_>>>()?
111      .contains(&true)
112    {
113      out = Tokenize!(r"\shorthandoff{:;!?}");
114    } else if listed("turkish")? {
115      out = Tokenize!(r"\shorthandoff{:!=}");
116    }
117    Ok(out)
118  });
119  RawTeX!(r"\AtBeginDocument{\lx@MT@babel@shorthandoff}");
