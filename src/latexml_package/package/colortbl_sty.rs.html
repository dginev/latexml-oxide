303/// The global copy of `arc`'s colour a preamble freezes (array.sty:54 `\xdef`), one per distinct body.
304fn frozen_arc(arc: &Token) -> Token {
305  let body = match lookup_meaning(arc) {
306    Some(Stored::Expandable(defn)) => match defn.get_expansion() {
307      Some(ExpansionBody::Tokens(body)) => Some(body.to_string()),
308      _ => None,
309    },
310    _ => None,
311  };
312  let key = body.map(|body| s!("lx@CT@arc@frozen@{body}"));
313  if let Some(Stored::Token(frozen)) = key.as_deref().and_then(lookup_value) {
314    return frozen;
315  }
316  let n = lookup_int("lx@CT@arc@frozen") + 1;
317  assign_value("lx@CT@arc@frozen", n, Some(Scope::Global));
318  let frozen = T_CS!(s!("\\lx@CT@arc@{n}"));
319  let_i(&frozen, arc, Some(Scope::Global));
320  if let Some(key) = key {
321    assign_value(&key, Stored::Token(frozen), Some(Scope::Global));
322  }
323  frozen
324}

326/// Gives the rule just drawn, the current list's last box, the colour `arc` gives as its `border_color`.
327/// Relies on the `|` template putting this token right after its own `\vrule` (`\vrule\lx@CT@paint…`): it reads only the last box.
328fn paint_rule(arc: &Token) -> Result<Vec<Digested>> {
329  if let Some(color) = arc_color(arc)?
330    && let Some(mut rule) = pop_box_list()
331  {
332    if rule.get_property_bool("isVerticalRule") {
333      rule.set_property("border_color", Stored::String(pin(&color)));
334    }
335    push_box_list(rule);
336  }
337  Ok(Vec::new())
338}

340/// The colour `arc` gives (colortbl.sty:161-192 `\CT@arc@`, or a preamble's frozen copy); black and `\relax` none.
341fn arc_color(arc: &Token) -> Result<Option<String>> {
342  if arc.defined_as(&T_RELAX!()) {
343    return Ok(None);
344  }
345  bgroup();
346  digest(Tokens!(*arc))?;
347  let color = lookup_font()
348    .and_then(|font| font.color)
349    .map(|color| color.to_attribute())
350    .filter(|color| *color != common::color::BLACK.to_attribute());
351  egroup()?;
352  Ok(color)
