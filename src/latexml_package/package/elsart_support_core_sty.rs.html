346  // This is a *DefMacro* (token expansion), NOT a DefConstructor — so the
347  // body tokenises to LITERAL TEXT (`<`, `>` are catcode-OTHER): `\note{X}`
348  // expands to the characters `<ltx:note>` + X + `</ltx:note>`, NOT a real
349  // <ltx:note> element. Perl's own `# ?` flags it as questionable, but it is
350  // the ground truth and crucially is ERROR-FREE: a block argument such as
351  // `\note{\begin{remark}…\end{remark}}` (a `\newtheorem`-based environment)
352  // renders the remark as a normal <ltx:theorem> bracketed by stray literal

390/// The separator slots of the `stm/affiliation` keys, in [`stm_affiliation`]'s order.
391const STM_ORGANIZATION: usize = 0;
392const STM_ADDRESS_LINE: usize = 1;
393const STM_CITY: usize = 2;
394const STM_POSTAL_CODE: usize = 3;
395const STM_STATE: usize = 4;
396const STM_COUNTRY: usize = 5;
397
398/// `\affiliation`'s argument printed as elsarticle.cls:387-440 and cas-common.sty:1118-1160 print their
399/// `stm/affiliation` keys (the value commands at elsarticle.cls:330-356, cas-common.sty:1061-1090): `organization`
400/// (`o`, `or`, and in elsarticle `organisation`), `addressline` (`a`, `ad`), `city` (`c`, `ci`), `postcode` (`p`,
401/// `pc`) and `state` (`s`, `st`) as the value, its separator — `,` unless `<key>sep` (`op`, `ap`, `cp`, `pp`, `sp`,
402/// `orp`, …) gave another — and a space; `country` (`cy`) as the value and its own separator (`cyp`, none by
403/// default); a key the set does not know as its value, or, given none, its own text, and a space. So
404/// `organization={X}, city={Y}, citysep={}, postcode={1}` is "X, Y 1" (2609.01851), and a braced `{ARTORG Center,}`
405/// part is kept (2609.04891). Three departures from the PDF (OXIDIZED_DESIGN_DIVERGENCES #472): an argument with no
406/// key at all (`Univ, City, Country`) is kept as written, its commas with it, where the class reads each part as an
407/// unknown key and drops them; the separator the class leaves after a last part that is not a country ("Univ A,") is
408/// not kept; a key's own text is typeset, not printed as its string. Perl (PR #2767) joined the known values with
409/// ", " and dropped the rest (KNOWN_PERL_ERRORS #545). `elsarticle` selects that class's set, which cas-common lacks
410/// two keys of (`organisation`, `oraganisationsep`).
411pub fn stm_affiliation(raw: Tokens, elsarticle: bool) -> Result<Tokens> {
412  let items = top_level_items(raw.unlist_ref(), ',');
413  if items
414    .iter()
415    .all(|item| top_level_items(item, '=').len() < 2)
416  {
417    return Ok(raw);
418  }
419  // `\__reset_affiliation:` (cas-common.sty:1164-1172) before each `\affiliation`
420  let mut separators: [Vec<Token>; 6] = [
421    vec![T_OTHER!(",")],
422    vec![T_OTHER!(",")],
423    vec![T_OTHER!(",")],
424    vec![T_OTHER!(",")],
425    vec![T_OTHER!(",")],
426    Vec::new(),
427  ];
428  // Each value is set down with its separator's macro (`\ca_aff_city[\l_city_punc_tl]{…}`), which the class expands
429  // only when it assembles the affiliation (cas-common.sty:1262 `\csxdef`): a `citysep={}` after the `city` still
430  // applies. So the parts are kept with their separator slot, and the separators read at the end.
431  let mut parts: Vec<(Vec<Token>, Option<usize>)> = Vec::new();
432  for item in items {
433    let (key, value) = match top_level_items(&item, '=').as_slice() {
434      [key] => (unbraced(key), None),
435      [key, value @ ..] => (unbraced(key), Some(unbraced(&value.join(&T_OTHER!("="))))),
436      [] => continue,
437    };
438    if key.is_empty() && value.is_none() {
439      continue;
440    }
441    let name = Tokens::new(key.clone()).to_string();
442    let separator_key = match name.as_str() {
443      "op" | "oraganizationsep" | "orp" => Some(STM_ORGANIZATION),
444      "oraganisationsep" if elsarticle => Some(STM_ORGANIZATION),
445      "ap" | "addresslinesep" | "adp" => Some(STM_ADDRESS_LINE),
446      "cp" | "citysep" | "cip" => Some(STM_CITY),
447      "pp" | "postcodesep" | "pcp" => Some(STM_POSTAL_CODE),
448      "sp" | "statesep" | "stp" => Some(STM_STATE),
449      "cyp" => Some(STM_COUNTRY),
450      _ => None,
451    };
452    if let Some(slot) = separator_key {
453      separators[slot] = value.unwrap_or_default();
454      continue;
455    }
456    let value_key = match name.as_str() {
457      "o" | "or" | "organization" => Some(STM_ORGANIZATION),
458      "organisation" if elsarticle => Some(STM_ORGANIZATION),
459      "a" | "ad" | "addressline" => Some(STM_ADDRESS_LINE),
460      "c" | "ci" | "city" => Some(STM_CITY),
461      "p" | "pc" | "postcode" => Some(STM_POSTAL_CODE),
462      "s" | "st" | "state" => Some(STM_STATE),
463      "cy" | "country" => Some(STM_COUNTRY),
464      _ => None,
465    };
466    match value_key {
467      Some(slot) => parts.push((value.unwrap_or_default(), Some(slot))),
468      // the unknown key's value, or its own text (`\l_keys_key_tl`)
469      None => parts.push((value.filter(|v| !v.is_empty()).unwrap_or(key), None)),
470    }
471  }
472  // A separator separates: none follows a last part that is not the country, where the class, expecting a `country`
473  // there, prints one with nothing after it ("Univ A," for a lone `organization`); the country's own (`cyp`) stays.
474  let mut out: Vec<Token> = Vec::new();
475  let count = parts.len();
476  for (i, (value, slot)) in parts.into_iter().enumerate() {
477    out.extend(value);
478    if i + 1 == count && slot != Some(STM_COUNTRY) {
479      break;
480    }
481    if let Some(slot) = slot {
482      out.extend(separators[slot].iter().copied());
483    }
484    if slot != Some(STM_COUNTRY) {
485      out.push(T_SPACE!());
486    }
487  }
488  while out
489    .last()
490    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
491  {
492    out.pop();
493  }
494  Ok(Tokens::new(out))
495}

497/// `tokens` split at each brace-depth-0 character `c`, each part without its surrounding spaces.
498fn top_level_items(tokens: &[Token], c: char) -> Vec<Vec<Token>> {
499  let delimiter = pin(c.to_string());
500  let mut parts: Vec<Vec<Token>> = vec![Vec::new()];
501  let mut depth: i32 = 0;
502  for t in tokens {
503    match t.get_catcode() {
504      Catcode::BEGIN => depth += 1,
505      Catcode::END => depth -= 1,
506      Catcode::OTHER if depth == 0 && t.text == delimiter => {
507        parts.push(Vec::new());
508        continue;
509      },
510      _ => {},
511    }
512    parts.last_mut().unwrap().push(*t);
513  }
514  parts.into_iter().map(|part| trimmed(&part)).collect()
515}
516
517/// `tokens` without leading or trailing spaces.
518fn trimmed(tokens: &[Token]) -> Vec<Token> {
519  let start = tokens
520    .iter()
521    .position(|t| t.get_catcode() != Catcode::SPACE)
522    .unwrap_or(tokens.len());
523  let end = tokens
524    .iter()
525    .rposition(|t| t.get_catcode() != Catcode::SPACE)
526    .map_or(start, |e| e + 1);
527  tokens[start..end].to_vec()
528}

530/// `tokens` trimmed, and without the braces of a group that is all of it (`{ARTORG Center,}`), as the key-value
531/// parser strips one level.
532fn unbraced(tokens: &[Token]) -> Vec<Token> {
533  let tokens = trimmed(tokens);
534  if tokens.len() >= 2
535    && tokens[0].get_catcode() == Catcode::BEGIN
536    && tokens[tokens.len() - 1].get_catcode() == Catcode::END
537  {
538    // the first brace must close at the very end
539    let mut depth = 0;
540    for (i, t) in tokens.iter().enumerate() {
541      match t.get_catcode() {
542        Catcode::BEGIN => depth += 1,
543        Catcode::END => {
544          depth -= 1;
545          if depth == 0 && i + 1 < tokens.len() {
546            return tokens;
547          }
548        },
549        _ => {},
550      }
551    }
552    return tokens[1..tokens.len() - 1].to_vec();
553  }
554  tokens
555}

557/// elsart.cls's `{keyword}` markup: the keywords, then the `\PACS`, `\MSC`, `\JEL` and `\UK` classifications that
558/// cut them, `\sep` between the entries (Perl elsart_support_core.sty.ltxml:130-153). autart.cls carries elsart.cls's
559/// (autart.cls:1248-1258 of 1106.2518's copy), so its binding defines the same (autart_cls.rs).
560#[rustfmt::skip]
561pub fn define_keyword_markup() -> Result<()> {
562  // keyword environment and macros with XUntil pattern
563  // Perl L135-152: \begin{keyword}/\end{keyword} use DefMacroI with begingroup/endgroup,
564  // NOT DefEnvironment!, to properly scope the XUntil delimiter reading.
565  DefMacro!(T_CS!("\\begin{keyword}"), None, "\\begingroup\\@keyword");
566  DefMacro!(T_CS!("\\end{keyword}"), None, "\\@keyword@cut\\endgroup");
567  // Perl elsart_support_core.sty.ltxml L138-141: `\keyword{...}` (1-arg
568  // form) reads a balanced group and wraps with `\@keyword <arg>
569  // \@keyword@cut` so XUntil terminates at the inserted token.
570  //
571  // History — DO NOT switch this back to `read_balanced(_, _, false)`:
572  // commit 09bc60c2 ("\keyword absorbs trailing unbalanced }") tried to
573  // mirror Perl's `$gullet->readBalanced` with no args (require_open=
574  // false → level starts at 1) to absorb the legacy unbalanced-`}`
575  // idiom from arXiv:1710.03688 / hep-ph0702114, where the abstract
576  // ends:
577  //     \keyword{Keyword1; ...} \vskip ...\noindent{...e3}}
578  // (a trailing literal `}` past the keyword arg). For that input
579  // alone, `require_open=false` correctly consumes `{...} }` and stops.
580  //
581  // BUT for the common balanced form — `\keyword{Higgs; Boson}` with
582  // no trailing extra `}` (the test fixture
583  // `tests/babel/elsart_keyword_brace_form.tex` and the vast majority
584  // of real-world usage) — `require_open=false` reads past the
585  // matching `}`, walks through `\end{abstract}` (raw `\end` is a CS
586  // and CC_END count is unchanged across that expansion), then
587  // `\end{frontmatter}`, … all the way to EOF. The reader's pushback
588  // grows unboundedly on the way; in CI we observed a single
589  // `memory allocation of 21743271936 bytes failed` (21 GB) → SIGABRT
590  // before the test could even report. The cost of one paper's
591  // 1-error reduction is the entire test suite OOM-aborting at
592  // `81_babel::elsart_keyword_brace_form_test`.
593  //
594  // RESOLVED 2026-05-30 via the design note's first suggestion: keep the
595  // strict balanced read of the keyword argument, then peek past spaces and
596  // gobble an OPTIONAL trailing `}` (catcode END). This reproduces Perl's
597  // lenient `readBalanced` *result* (it absorbs the legacy unbalanced `}`)
598  // for the trailing-`}` idiom — abstract bodies that end
599  //     \keyword{Kw1; Kw2; …}   <comments/spaces>   }   \end{abstract}
600  // (the stray `}` is the orphaned close of a commented-out brace group;
601  // witness 1601.01227, also 1710.03688 / hep-ph0702114) — WITHOUT the
602  // speculative read-to-EOF that OOM'd the balanced fixture: for the common
603  // `\keyword{Higgs; Boson}\end{abstract}` form the token after the strict
604  // read is `\end` (a CS, not `}`), so nothing is gobbled. The peek is
605  // bounded to one token, never crossing `\end`.
606  // Skip leading spaces and blank-line `\par`s, then gobble a single trailing
607  // `}` (catcode END) if one is there — Perl's `readBalanced` reads straight
608  // through the `\par` from a blank line before the stray brace (witness
609  // 1705.01354 separates `\keyword{…}` and the orphaned `}` by a blank line).
610  // The skipped whitespace/`\par` tokens are buffered and FULLY RESTORED when
611  // no trailing `}` follows, so the common `\keyword{…}\end{abstract}` form
612  // (and `\keyword{…}<blank>\end{abstract}`) is left byte-for-byte untouched.
613  // Bounded: stops at the first non-space, non-`\par` token; never crosses
614  // `\end`.
615  // Perl's `readBalanced` (L140) reads tokens up to the next UNMATCHED `}`
616  // (tracking brace depth), absorbing that stray `}` as its terminator —
617  // regardless of what non-brace material (spaces, `\par`, `\vskip <dimen>`,
618  // `\noindent`, …) sits between `\keyword{…}` and the orphaned brace. The
619  // earlier version only skipped spaces/`\par`s, so it stopped at the first
620  // real token (e.g. `\vskip`) and left the stray `}` to hit the abstract's
621  // mode-switch frame (`ltx:para`-free `}` → "close a group that switched to
622  // mode internal_vertical"; witness 1604.00855:
623  // `\keyword{…} \vskip 0.5\baselineskip}`).
624  //
625  // Walk forward tracking brace depth: an unmatched `}` (depth 0) is the
626  // stray terminator — DROP it, re-inject everything else (so the `\vskip`
627  // etc. still render, content-preserving). Bound the scan at `\end` (depth
628  // 0) — the env terminator — and a hard token cap, so the common
629  // `\keyword{Higgs; Boson}\end{abstract}` form (no stray `}`) reads one
630  // token (`\end`), restores it, and is left untouched (this is also what
631  // averts the read-to-EOF OOM a naive `readBalanced` hit). Witnesses:
632  // 1604.00855, 1601.01227, 1705.01354, 1710.03688, hep-ph0702114.
633  DefPrimitive!("\\lx@elsart@gobble@optbrace", {
634    let mut skipped: Vec<Token> = Vec::new();
635    let mut depth: i32 = 0;
636    let mut count: usize = 0;
637    while let Some(tok) = read_token()? {
638      count += 1;
639      if count > 4096 {
640        skipped.push(tok); // safety cap: restore everything, gobble nothing
641        break;
642      }
643      let cc = tok.get_catcode();
644      if cc == Catcode::BEGIN {
645        depth += 1;
646        skipped.push(tok);
647      } else if cc == Catcode::END {
648        if depth == 0 {
649          break; // unmatched stray `}` — drop it (absorbed, like readBalanced)
650        }
651        depth -= 1;
652        skipped.push(tok);
653      } else if depth == 0 && tok == T_CS!("\\end") {
654        skipped.push(tok); // env end reached before any stray `}` — restore all
655        break;
656      } else {
657        skipped.push(tok);
658      }
659    }
660    // Re-inject everything we buffered, in order. The only token ever removed
661    // is the stray `}` (which is never pushed onto `skipped`).
662    for tok in skipped.into_iter().rev() {
663      unread_one(tok);
664    }
665  });
666  DefMacro!("\\keyword{}", "\\@keyword #1 \\@keyword@cut\\lx@elsart@gobble@optbrace");
667  DefMacro!("\\endkeyword", "\\@keyword@cut");
668  DefMacro!("\\PACS", "\\@keyword@cut\\@PACS");
669  DefMacro!("\\MSC[]", "\\@keyword@cut\\@MSC{#1}");
670  DefMacro!("\\JEL", "\\@keyword@cut\\@JEL");
671  DefMacro!("\\UK", "\\@keyword@cut\\@UK");
672
673  // Perl L148-152: @keyword reads until @keyword@cut delimiter using XUntil.
674  // XUntil expands tokens while reading, so \end{keyword} → \@keyword@cut is found.
675  DefConstructor!("\\@keyword@cut", "");
676  DefMacro!("\\@keyword XUntil:\\@keyword@cut", "\\lx@add@keywords{#1}");
677  DefMacro!("\\@PACS XUntil:\\@keyword@cut", "\\lx@add@classification[scheme=PACS,name={PACS:~}]{#1}");
678  DefMacro!("\\@MSC{} XUntil:\\@keyword@cut", "\\lx@add@classification[scheme={#1 MSC},name={MSC:~}]{#2}");
679  DefMacro!("\\@JEL XUntil:\\@keyword@cut", "\\lx@add@classification[scheme=JEL,name={JEL:~}]{#1}");
680  DefMacro!("\\@UK XUntil:\\@keyword@cut", "\\lx@add@classification[scheme=UK,name={UK:~}]{#1}");
681  Ok(())
682}

801pub(crate) fn els_outside_mark(kind: &str) -> String {
802  if lookup_bool(ELS_NAMES_PRINTED) {
803    return lookup_string(&s!("{ELS_PRINTED_MARKS}@{kind}"));
804  }
805  lookup_string(&s!("{ELS_OUTSIDE}@{kind}@{}", lookup_int(ELS_NAMES_LEVEL)))
806}

808/// The global flag elsarticle sets: its `\cortext`/`\fntext` label their notes by key and route (elsarticle_cls.rs).

811/// The deferred label (`\lx@label@ref`) of the outside mark of kind `kind` the first name shows, where its own sets none.
812fn els_first_mark_key(kind: &str) -> String { s!("els@first@{kind}") }

814/// Record, as the first name's deferred labels ([`els_first_mark_key`], read at the flush), the outside marks read last
815/// at the names' level ([`els_outside_mark`]): elsarticle prints the class's `\@corref`/`\@fnmark` beside the first
816/// name only when the names print (elsarticle.cls:159-165, :188-190, :256-264), so a mark read after its call still
817/// moves it, until the names print (OD #1067). `\corref` reads its key whole: a key list prints no mark
818/// ([`els_mark_list`]). And the labels of the notes the outside `\corref` names and named before (`els_note_shadowed`).
819fn els_settle_marks() {
820  if reading_author_fields() || lookup_bool(ELS_NAMES_PRINTED) {
821    return;
822  }
823  let cor = els_outside_mark("cor");
824  let cor_label = if cor.trim().is_empty() || els_mark_list(&cor) {
825    String::new()
826  } else {
827    els_route_label(&cor, "cor")
828  };
829  set_label_ref(&els_first_mark_key("cor"), &cor_label);
830  let fn_labels: Vec<String> = els_outside_mark("fn")
831    .split(',')
832    .map(str::trim)
833    .filter(|key| !key.is_empty())
834    .map(|key| els_route_label(key, "fn"))
835    .collect();
836  set_label_ref(&els_first_mark_key("fn"), &fn_labels.join(","));
837  let settled = lookup_string(ELS_SETTLED_COR);
838  if settled != cor {
839    assign_value(ELS_SETTLED_COR, Stored::String(pin(&cor)), Scope::Global);
840    els_settle_note_labels(&settled);
841    els_settle_note_labels(&cor);
842  }
843}

845/// Settle what `key`, just read as a note's, an address's or a `\corref`'s key, decides: its notes' labels and the first
846/// name's marks ([`els_settle_marks`]).
847pub(crate) fn els_settle_key(key: &str) {
848  if reading_author_fields() {
849    return;
850  }
851  els_settle_note_labels(key);

855/// Where the class prints the names (`\end{frontmatter}`, `\maketitle`; elsarticle.cls:1303, :805): each name's
856/// `\@corref`/`\@fnmark` is read there and cleared after the first (:256-265), so the first name's marks are frozen as
857/// they stand, and a mark read later moves no name (revELSARTc r09, r13, r14; a second `frontmatter` reprints the
858/// names, which one creator list does not model, r02). The key lists read so far are warned of.
859pub(crate) fn els_names_printed() {
860  if reading_author_fields() || lookup_bool(ELS_NAMES_PRINTED) {
861    return;
862  }
863  els_settle_marks();
864  for kind in ["cor", "fn"] {
865    let printed = els_outside_mark(kind);
866    assign_value(
867      &s!("{ELS_PRINTED_MARKS}@{kind}"),
868      Stored::String(pin(&printed)),
869      Scope::Global,
870    );
871  }
872  assign_value(ELS_NAMES_PRINTED, true, Scope::Global);
873  els_warn_pending_lists();
874}

883/// Whether `key`, a `\corref`'s or `\cormark`'s argument, is a key list under elsarticle: the class looks the argument up
884/// whole (`\elsRef{#1}`, elsarticle.cls:174-176, :188-190), so one with a comma that no `\cortext`, `\fntext` or
885/// address labels prints no mark and names no note.
886pub(crate) fn els_mark_list(key: &str) -> bool {
887  let key = key.trim();
888  lookup_bool(ELS_NOTE_LABELS)
889    && key.contains(',')
890    && ![ELS_CORTEXT_KEYS, ELS_FNTEXT_KEYS, ELS_PLACE_KEYS]
891      .iter()
892      .any(|list| recorded_els_key(list, key))
893}

895/// Note `\<cs>{key}` read (`cs` "corref" or "cormark"): a key list ([`els_mark_list`]) is warned of once the keys it
896/// could name are read, where the names print ([`els_names_printed`]) or at the document's end.
897fn els_read_mark_key(cs: &str, key: &str) {
898  if !lookup_bool(ELS_NOTE_LABELS) || !key.contains(',') {
899    return;
900  }
901  if lookup_bool(ELS_NAMES_PRINTED) {
902    els_warn_mark_list(cs, key);
903  } else {
904    let mut pending = lookup_string(ELS_PENDING_LISTS);
905    pending.push_str(&s!("{cs}\u{1f}{}\u{1e}", key.trim()));
906    assign_value(
907      ELS_PENDING_LISTS,
908      Stored::String(pin(&pending)),
909      Scope::Global,
910    );
911  }
912}

914/// Warn of the key lists read and not yet warned of ([`els_read_mark_key`]).
915pub(crate) fn els_warn_pending_lists() {
916  let pending = lookup_string(ELS_PENDING_LISTS);
917  if pending.is_empty() {
918    return;
919  }
920  assign_value(ELS_PENDING_LISTS, Stored::String(pin("")), Scope::Global);
921  for entry in pending.split('\u{1e}').filter(|entry| !entry.is_empty()) {
922    if let Some((cs, key)) = entry.split_once('\u{1f}') {
923      els_warn_mark_list(cs, key);
924    }
925  }
926}

928/// The Warning for `\<cs>{key}` when it is a key list ([`els_mark_list`]).
929fn els_warn_mark_list(cs: &str, key: &str) {
930  if !els_mark_list(key) {
931    return;
932  }
933  let call = if cs == "cormark" {
934    s!("\\cormark[{}]", key.trim())
935  } else {
936    s!("\\{cs}{{{}}}", key.trim())
937  };
938  Warn!(
939    "frontmatter",
940    "elsarticle_mark_list",
941    s!(
942      "elsarticle: {call} names a key list, for which the class prints no mark; its notes are the block's"
943    )
944  );
945}

963/// The keys `<braced>{<keys>}` and `<bracketed>[<keys>]` name in `tokens`, an `\author`'s argument (`\corref` and
964/// `\cormark`, `\fnref` and `\fnmark`), each comma-separated list split.
965pub(crate) fn keys_in(tokens: &[Token], braced: &Token, bracketed: &Token) -> Vec<String> {
966  ref_spans(tokens, braced, bracketed)
967    .into_iter()
968    .flat_map(|(_, keys, _)| keys)
969    .collect()
970}

972/// Each `<braced>{<keys>}` and `<bracketed>[<keys>]` in `tokens`: its token range, its keys and its argument whole
973/// (trimmed).
974pub(crate) fn ref_spans(
975  tokens: &[Token],
976  braced: &Token,
977  bracketed: &Token,
978) -> Vec<(std::ops::Range<usize>, Vec<String>, String)> {
979  let mut spans = Vec::new();
980  for (at, t) in tokens.iter().enumerate() {
981    let (open, close) = if t == braced {
982      (Catcode::BEGIN, Catcode::END)
983    } else if t == bracketed {
984      (Catcode::OTHER, Catcode::OTHER)
985    } else {
986      continue;
987    };
988    let mut i = at + 1;
989    while tokens
990      .get(i)
991      .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
992    {
993      i += 1;
994    }
995    if !tokens
996      .get(i)
997      .is_some_and(|t| t.get_catcode() == open && (open == Catcode::BEGIN || t.to_string() == "["))
998    {
999      continue;
1000    }
1001    let start = i + 1;
1002    let mut depth = 0;
1003    let mut end = start;
1004    while let Some(t) = tokens.get(end) {
1005      depth += t.group_step();
1006      let closed = if close == Catcode::END {
1007        depth < 0
1008      } else {
1009        depth == 0 && t.to_string() == "]"
1010      };
1011      if closed {
1012        break;
1013      }
1014      end += 1;
1015    }
1016    let key = Tokens::new(tokens[start..end.min(tokens.len())].to_vec()).to_string();
1017    let keys = key
1018      .split(',')
1019      .map(str::trim)
1020      .filter(|k| !k.is_empty())
1021      .map(String::from)
1022      .collect();
1023    spans.push((
1024      at..(end + 1).min(tokens.len()),
1025      keys,
1026      key.trim().to_string(),
1027    ));
1028  }
1029  spans
1030}

1032/// The label a route of kind `kind` (`""` an `\author[…]` key, `"cor"`, `"fn"`) gives `key`: the form the key was first
1033/// read by ([`els_route`]), so that all its routes meet.
1034fn els_route_label(key: &str, kind: &str) -> String {
1035  let key = key.trim();
1036  match els_route(key).as_deref().unwrap_or(kind) {
1037    "" => key.to_string(),
1038    route => s!("{route}:{}", els_label_key(key)),
1039  }
