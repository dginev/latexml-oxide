260/// The separator slots of the `stm/affiliation` keys, in [`stm_affiliation`]'s order.
261const STM_ORGANIZATION: usize = 0;
262const STM_ADDRESS_LINE: usize = 1;
263const STM_CITY: usize = 2;
264const STM_POSTAL_CODE: usize = 3;
265const STM_STATE: usize = 4;
266const STM_COUNTRY: usize = 5;
267
268/// `\affiliation`'s argument printed as elsarticle.cls:387-440 and cas-common.sty:1118-1160 print their
269/// `stm/affiliation` keys (the value commands at elsarticle.cls:330-356, cas-common.sty:1061-1090): `organization`
270/// (`o`, `or`, and in elsarticle `organisation`), `addressline` (`a`, `ad`), `city` (`c`, `ci`), `postcode` (`p`,
271/// `pc`) and `state` (`s`, `st`) as the value, its separator — `,` unless `<key>sep` (`op`, `ap`, `cp`, `pp`, `sp`,
272/// `orp`, …) gave another — and a space; `country` (`cy`) as the value and its own separator (`cyp`, none by
273/// default); a key the set does not know as its value, or, given none, its own text, and a space. So
274/// `organization={X}, city={Y}, citysep={}, postcode={1}` is "X, Y 1" (2609.01851), and a braced `{ARTORG Center,}`
275/// part is kept (2609.04891). Three departures from the PDF (OXIDIZED_DESIGN_DIVERGENCES #472): an argument with no
276/// key at all (`Univ, City, Country`) is kept as written, its commas with it, where the class reads each part as an
277/// unknown key and drops them; the separator the class leaves after a last part that is not a country ("Univ A,") is
278/// not kept; a key's own text is typeset, not printed as its string. Perl (PR #2767) joined the known values with
279/// ", " and dropped the rest (KNOWN_PERL_ERRORS #545). `elsarticle` selects that class's set, which cas-common lacks
280/// two keys of (`organisation`, `oraganisationsep`).
281pub fn stm_affiliation(raw: Tokens, elsarticle: bool) -> Result<Tokens> {
282  let items = top_level_items(raw.unlist_ref(), ',');
283  if items
284    .iter()
285    .all(|item| top_level_items(item, '=').len() < 2)
286  {
287    return Ok(raw);
288  }
289  // `\__reset_affiliation:` (cas-common.sty:1164-1172) before each `\affiliation`
290  let mut separators: [Vec<Token>; 6] = [
291    vec![T_OTHER!(",")],
292    vec![T_OTHER!(",")],
293    vec![T_OTHER!(",")],
294    vec![T_OTHER!(",")],
295    vec![T_OTHER!(",")],
296    Vec::new(),
297  ];
298  // Each value is set down with its separator's macro (`\ca_aff_city[\l_city_punc_tl]{…}`), which the class expands
299  // only when it assembles the affiliation (cas-common.sty:1262 `\csxdef`): a `citysep={}` after the `city` still
300  // applies. So the parts are kept with their separator slot, and the separators read at the end.
301  let mut parts: Vec<(Vec<Token>, Option<usize>)> = Vec::new();
302  for item in items {
303    let (key, value) = match top_level_items(&item, '=').as_slice() {
304      [key] => (unbraced(key), None),
305      [key, value @ ..] => (unbraced(key), Some(unbraced(&value.join(&T_OTHER!("="))))),
306      [] => continue,
307    };
308    if key.is_empty() && value.is_none() {
309      continue;
310    }
311    let name = Tokens::new(key.clone()).to_string();
312    let separator_key = match name.as_str() {
313      "op" | "oraganizationsep" | "orp" => Some(STM_ORGANIZATION),
314      "oraganisationsep" if elsarticle => Some(STM_ORGANIZATION),
315      "ap" | "addresslinesep" | "adp" => Some(STM_ADDRESS_LINE),
316      "cp" | "citysep" | "cip" => Some(STM_CITY),
317      "pp" | "postcodesep" | "pcp" => Some(STM_POSTAL_CODE),
318      "sp" | "statesep" | "stp" => Some(STM_STATE),
319      "cyp" => Some(STM_COUNTRY),
320      _ => None,
321    };
322    if let Some(slot) = separator_key {
323      separators[slot] = value.unwrap_or_default();
324      continue;
325    }
326    let value_key = match name.as_str() {
327      "o" | "or" | "organization" => Some(STM_ORGANIZATION),
328      "organisation" if elsarticle => Some(STM_ORGANIZATION),
329      "a" | "ad" | "addressline" => Some(STM_ADDRESS_LINE),
330      "c" | "ci" | "city" => Some(STM_CITY),
331      "p" | "pc" | "postcode" => Some(STM_POSTAL_CODE),
332      "s" | "st" | "state" => Some(STM_STATE),
333      "cy" | "country" => Some(STM_COUNTRY),
334      _ => None,
335    };
336    match value_key {
337      Some(slot) => parts.push((value.unwrap_or_default(), Some(slot))),
338      // the unknown key's value, or its own text (`\l_keys_key_tl`)
339      None => parts.push((value.filter(|v| !v.is_empty()).unwrap_or(key), None)),
340    }
341  }
342  // A separator separates: none follows a last part that is not the country, where the class, expecting a `country`
343  // there, prints one with nothing after it ("Univ A," for a lone `organization`); the country's own (`cyp`) stays.
344  let mut out: Vec<Token> = Vec::new();
345  let count = parts.len();
346  for (i, (value, slot)) in parts.into_iter().enumerate() {
347    out.extend(value);
348    if i + 1 == count && slot != Some(STM_COUNTRY) {
349      break;
350    }
351    if let Some(slot) = slot {
352      out.extend(separators[slot].iter().copied());
353    }
354    if slot != Some(STM_COUNTRY) {
355      out.push(T_SPACE!());
356    }
357  }
358  while out
359    .last()
360    .is_some_and(|t| t.get_catcode() == Catcode::SPACE)
361  {
362    out.pop();
363  }
364  Ok(Tokens::new(out))
365}

367/// `tokens` split at each brace-depth-0 character `c`, each part without its surrounding spaces.
368fn top_level_items(tokens: &[Token], c: char) -> Vec<Vec<Token>> {
369  let delimiter = pin(c.to_string());
370  let mut parts: Vec<Vec<Token>> = vec![Vec::new()];
371  let mut depth: i32 = 0;
372  for t in tokens {
373    match t.get_catcode() {
374      Catcode::BEGIN => depth += 1,
375      Catcode::END => depth -= 1,
376      Catcode::OTHER if depth == 0 && t.text == delimiter => {
377        parts.push(Vec::new());
378        continue;
379      },
380      _ => {},
381    }
382    parts.last_mut().unwrap().push(*t);
383  }
384  parts.into_iter().map(|part| trimmed(&part)).collect()
385}
386
387/// `tokens` without leading or trailing spaces.
388fn trimmed(tokens: &[Token]) -> Vec<Token> {
389  let start = tokens
390    .iter()
391    .position(|t| t.get_catcode() != Catcode::SPACE)
392    .unwrap_or(tokens.len());
393  let end = tokens
394    .iter()
395    .rposition(|t| t.get_catcode() != Catcode::SPACE)
396    .map_or(start, |e| e + 1);
397  tokens[start..end].to_vec()
398}
399
400/// `tokens` trimmed, and without the braces of a group that is all of it (`{ARTORG Center,}`), as the key-value
401/// parser strips one level.
402fn unbraced(tokens: &[Token]) -> Vec<Token> {
403  let tokens = trimmed(tokens);
404  if tokens.len() >= 2
405    && tokens[0].get_catcode() == Catcode::BEGIN
406    && tokens[tokens.len() - 1].get_catcode() == Catcode::END
407  {
408    // the first brace must close at the very end
409    let mut depth = 0;
410    for (i, t) in tokens.iter().enumerate() {
411      match t.get_catcode() {
412        Catcode::BEGIN => depth += 1,
413        Catcode::END => {
414          depth -= 1;
415          if depth == 0 && i + 1 < tokens.len() {
416            return tokens;
417          }
418        },
419        _ => {},
420      }
421    }
422    return tokens[1..tokens.len() - 1].to_vec();
423  }
424  tokens
425}

427/// elsart.cls's `{keyword}` markup: the keywords, then the `\PACS`, `\MSC`, `\JEL` and `\UK` classifications that
428/// cut them, `\sep` between the entries (Perl elsart_support_core.sty.ltxml:130-153). autart.cls carries elsart.cls's
429/// (autart.cls:1248-1258 of 1106.2518's copy), so its binding defines the same (autart_cls.rs).
430#[rustfmt::skip]
431pub fn define_keyword_markup() -> Result<()> {
432  // keyword environment and macros with XUntil pattern
433  // Perl L135-152: \begin{keyword}/\end{keyword} use DefMacroI with begingroup/endgroup,
434  // NOT DefEnvironment!, to properly scope the XUntil delimiter reading.
435  DefMacro!(T_CS!("\\begin{keyword}"), None, "\\begingroup\\@keyword");
436  DefMacro!(T_CS!("\\end{keyword}"), None, "\\@keyword@cut\\endgroup");
437  // Perl elsart_support_core.sty.ltxml L138-141: `\keyword{...}` (1-arg
438  // form) reads a balanced group and wraps with `\@keyword <arg>
439  // \@keyword@cut` so XUntil terminates at the inserted token.
440  //
441  // History — DO NOT switch this back to `read_balanced(_, _, false)`:
442  // commit 09bc60c2 ("\keyword absorbs trailing unbalanced }") tried to
443  // mirror Perl's `$gullet->readBalanced` with no args (require_open=
444  // false → level starts at 1) to absorb the legacy unbalanced-`}`
445  // idiom from arXiv:1710.03688 / hep-ph0702114, where the abstract
446  // ends:
447  //     \keyword{Keyword1; ...} \vskip ...\noindent{...e3}}
448  // (a trailing literal `}` past the keyword arg). For that input
449  // alone, `require_open=false` correctly consumes `{...} }` and stops.
450  //
451  // BUT for the common balanced form — `\keyword{Higgs; Boson}` with
452  // no trailing extra `}` (the test fixture
453  // `tests/babel/elsart_keyword_brace_form.tex` and the vast majority
454  // of real-world usage) — `require_open=false` reads past the
455  // matching `}`, walks through `\end{abstract}` (raw `\end` is a CS
456  // and CC_END count is unchanged across that expansion), then
457  // `\end{frontmatter}`, … all the way to EOF. The reader's pushback
458  // grows unboundedly on the way; in CI we observed a single
459  // `memory allocation of 21743271936 bytes failed` (21 GB) → SIGABRT
460  // before the test could even report. The cost of one paper's
461  // 1-error reduction is the entire test suite OOM-aborting at
462  // `81_babel::elsart_keyword_brace_form_test`.
463  //
464  // RESOLVED 2026-05-30 via the design note's first suggestion: keep the
465  // strict balanced read of the keyword argument, then peek past spaces and
466  // gobble an OPTIONAL trailing `}` (catcode END). This reproduces Perl's
467  // lenient `readBalanced` *result* (it absorbs the legacy unbalanced `}`)
468  // for the trailing-`}` idiom — abstract bodies that end
469  //     \keyword{Kw1; Kw2; …}   <comments/spaces>   }   \end{abstract}
470  // (the stray `}` is the orphaned close of a commented-out brace group;
471  // witness 1601.01227, also 1710.03688 / hep-ph0702114) — WITHOUT the
472  // speculative read-to-EOF that OOM'd the balanced fixture: for the common
473  // `\keyword{Higgs; Boson}\end{abstract}` form the token after the strict
474  // read is `\end` (a CS, not `}`), so nothing is gobbled. The peek is
475  // bounded to one token, never crossing `\end`.
476  // Skip leading spaces and blank-line `\par`s, then gobble a single trailing
477  // `}` (catcode END) if one is there — Perl's `readBalanced` reads straight
478  // through the `\par` from a blank line before the stray brace (witness
479  // 1705.01354 separates `\keyword{…}` and the orphaned `}` by a blank line).
480  // The skipped whitespace/`\par` tokens are buffered and FULLY RESTORED when
481  // no trailing `}` follows, so the common `\keyword{…}\end{abstract}` form
482  // (and `\keyword{…}<blank>\end{abstract}`) is left byte-for-byte untouched.
483  // Bounded: stops at the first non-space, non-`\par` token; never crosses
484  // `\end`.
485  // Perl's `readBalanced` (L140) reads tokens up to the next UNMATCHED `}`
486  // (tracking brace depth), absorbing that stray `}` as its terminator —
487  // regardless of what non-brace material (spaces, `\par`, `\vskip <dimen>`,
488  // `\noindent`, …) sits between `\keyword{…}` and the orphaned brace. The
489  // earlier version only skipped spaces/`\par`s, so it stopped at the first
490  // real token (e.g. `\vskip`) and left the stray `}` to hit the abstract's
491  // mode-switch frame (`ltx:para`-free `}` → "close a group that switched to
492  // mode internal_vertical"; witness 1604.00855:
493  // `\keyword{…} \vskip 0.5\baselineskip}`).
494  //
495  // Walk forward tracking brace depth: an unmatched `}` (depth 0) is the
496  // stray terminator — DROP it, re-inject everything else (so the `\vskip`
497  // etc. still render, content-preserving). Bound the scan at `\end` (depth
498  // 0) — the env terminator — and a hard token cap, so the common
499  // `\keyword{Higgs; Boson}\end{abstract}` form (no stray `}`) reads one
500  // token (`\end`), restores it, and is left untouched (this is also what
501  // averts the read-to-EOF OOM a naive `readBalanced` hit). Witnesses:
502  // 1604.00855, 1601.01227, 1705.01354, 1710.03688, hep-ph0702114.
503  DefPrimitive!("\\lx@elsart@gobble@optbrace", {
504    let mut skipped: Vec<Token> = Vec::new();
505    let mut depth: i32 = 0;
506    let mut count: usize = 0;
507    while let Some(tok) = read_token()? {
508      count += 1;
509      if count > 4096 {
510        skipped.push(tok); // safety cap: restore everything, gobble nothing
511        break;
512      }
513      let cc = tok.get_catcode();
514      if cc == Catcode::BEGIN {
515        depth += 1;
516        skipped.push(tok);
517      } else if cc == Catcode::END {
518        if depth == 0 {
519          break; // unmatched stray `}` — drop it (absorbed, like readBalanced)
520        }
521        depth -= 1;
522        skipped.push(tok);
523      } else if depth == 0 && tok == T_CS!("\\end") {
524        skipped.push(tok); // env end reached before any stray `}` — restore all
525        break;
526      } else {
527        skipped.push(tok);
528      }
529    }
530    // Re-inject everything we buffered, in order. The only token ever removed
531    // is the stray `}` (which is never pushed onto `skipped`).
532    for tok in skipped.into_iter().rev() {
533      unread_one(tok);
534    }
535  });
536  DefMacro!("\\keyword{}", "\\@keyword #1 \\@keyword@cut\\lx@elsart@gobble@optbrace");
537  DefMacro!("\\endkeyword", "\\@keyword@cut");
538  DefMacro!("\\PACS", "\\@keyword@cut\\@PACS");
539  DefMacro!("\\MSC[]", "\\@keyword@cut\\@MSC{#1}");
540  DefMacro!("\\JEL", "\\@keyword@cut\\@JEL");
541  DefMacro!("\\UK", "\\@keyword@cut\\@UK");
542
543  // Perl L148-152: @keyword reads until @keyword@cut delimiter using XUntil.
544  // XUntil expands tokens while reading, so \end{keyword} → \@keyword@cut is found.
545  DefConstructor!("\\@keyword@cut", "");
546  DefMacro!("\\@keyword XUntil:\\@keyword@cut", "\\lx@add@keywords{#1}");
547  DefMacro!("\\@PACS XUntil:\\@keyword@cut", "\\lx@add@classification[scheme=PACS,name={PACS:~}]{#1}");
548  DefMacro!("\\@MSC{} XUntil:\\@keyword@cut", "\\lx@add@classification[scheme={#1 MSC},name={MSC:~}]{#2}");
549  DefMacro!("\\@JEL XUntil:\\@keyword@cut", "\\lx@add@classification[scheme=JEL,name={JEL:~}]{#1}");
550  DefMacro!("\\@UK XUntil:\\@keyword@cut", "\\lx@add@classification[scheme=UK,name={UK:~}]{#1}");
551  Ok(())
552}
