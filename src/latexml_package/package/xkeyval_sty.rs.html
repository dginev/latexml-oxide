1481// Helper: Setup the XKV@documentclass and XKV@classoptionslist macros
1482fn xkeyval_setup_document_class() {
1483  let filelist = do_expand(Tokens!(T_CS!("\\@filelist")))
1484    .map(|t| t.to_string())
1485    .unwrap_or_default();
1486  let clsext = do_expand(Tokens!(T_CS!("\\@clsextension")))
1487    .map(|t| t.to_string())
1488    .unwrap_or_default();
1489
1490  // Try to find the document class in @filelist (Perl approach)
1491  for file in filelist.split(',') {
1492    let file = file.trim();
1493    if file.is_empty() {
1494      continue;
1495    }
1496    let (_area, _base, ext) = pathname::split(file);
1497    // Perl xkeyval.sty.ltxml L254: `if ($ext eq $clsext)` — case-sensitive.
1498    if ext == clsext {
1499      let opt_cs = T_CS!(s!("\\opt@{file}"));
1500      if lookup_meaning(&opt_cs).is_some() {
1501        let file_tks = Tokens::new(Explode!(file));
1502        let _ = def_macro(
1503          T_CS!("\\XKV@documentclass"),
1504          None,
1505          Some(ExpansionBody::Tokens(file_tks)),
1506          None,
1507        );
1508        let_i(
1509          &T_CS!("\\XKV@classoptionslist"),
1510          &T_CS!("\\@classoptionslist"),
1511          None,
1512        );
1513        return;
1514      }
1515    }
1516  }
1517  // Fallback: check if \@classoptionslist is defined (non-\relax) even without @filelist.
1518  // In Rust, compiled bindings don't call \@addtofilelist, so @filelist may be empty,
1519  // but \@classoptionslist is set by input_definitions when loading a .cls.
1520  let classoptlist = do_expand(Tokens!(T_CS!("\\@classoptionslist")))
1521    .map(|t| t.to_string())
1522    .unwrap_or_default();
1523  if !classoptlist.is_empty() {
1524    // We have class options but couldn't find the class in @filelist.
1525    // Still set up XKV@classoptionslist from \@classoptionslist.
1526    let_i(
1527      &T_CS!("\\XKV@classoptionslist"),
1528      &T_CS!("\\@classoptionslist"),
1529      None,
1530    );
1531    // Determine document class name from stored value
1532    let doc_class = match lookup_value("document_class_filename") {
1533      Some(Stored::String(sym)) => with(sym, |s| s.to_string()),
1534      _ => String::new(),
1535    };
1536    let _ = def_macro(
1537      T_CS!("\\XKV@documentclass"),
1538      None,
1539      Some(ExpansionBody::Tokens(Tokens::new(Explode!(doc_class)))),
1540      None,
1541    );
1542    return;
1543  }
1544  // oops, we did not have a documentclass
1545  // Perl xkeyval.sty.ltxml L260: `Error('undefined', 'xkeyval', ...)`.
1546  // Was Warn! pre-fix — severity downgrade vs Perl. Use Error! to match.
1547  // IIFE wraps because the enclosing fn returns `()` and the Error!
1548  // macro's Fatal-cap path uses `return Err(...)`.
1549  let _ = (|| -> Result<()> {
1550    Error!(
1551      "undefined",
1552      "xkeyval",
1553      "Package xkeyval loaded before \\documentclass"
1554    );
1555    Ok(())
1556  })();
1557  let _ = def_macro(
1558    T_CS!("\\XKV@documentclass"),
1559    None,
1560    Some(ExpansionBody::Tokens(Tokens!())),
1561    None,
1562  );
1563  let _ = def_macro(
1564    T_CS!("\\XKV@classoptionslist"),
1565    None,
1566    Some(ExpansionBody::Tokens(Tokens!())),
1567    None,
1568  );
