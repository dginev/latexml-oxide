75/// Does the epsf this document loads reset `\epsfxsize`/`\epsfysize` globally after each `\epsfbox`? The versions
76/// conflict: TL's epsf.sty/epsf.tex (2.7.4, :505-506) `\global \epsfxsize = 0pt`, a 1991 copy (Rokicki/Knuth, 22 of the
77/// 43 copies shipped in the `\epsfsize` and epsf-control populations: hep-ph9404329, astro-ph9605026, 0801.4912)
78/// `\epsfxsize=0pt\epsfysize=0pt` in the current group. The copy TeX would read decides: the file the binding answers
79/// (`binding_request`), `epsf.sty` for a package load and for `\input epsf.sty` — which leaves `\@currname` unset, so it
80/// read only `epsf.tex`, while latex reads a shipped 1991 `./epsf.sty` (0906.1361, astro-ph0006390, astro-ph0109126,
81/// astro-ph0410051, astro-ph9701125) —, `epsf.tex` for `\input epsf`; followed to `epsf.tex` when the file is a wrapper
82/// with no `\epsfbox` of its own (`\input epsf.tex`: astro-ph0111238, cs0501064); global, as TL's, when there is none.
83/// A wrapper that loads epsfig instead (psfrag's epsf.sty, `\RequirePackage{epsfig}`: astro-ph0012205, astro-ph9611186,
84/// astro-ph9911225) has epsfig's `\epsfbox`, whose reset is local (epsfig.sty:100-101 `\epsfysize\z@ \epsfxsize\z@`
85/// after its `\egroup`). A `\def` is read with the spaces TL's copy puts in it (`\def \epsfbox #1{`, epsf.tex:294), so
86/// TL's epsf.sty, found when the paper ships only an `epsf.tex`, is the copy that decides (review rev78s2).
87fn epsf_resets_globally() -> Result<bool> {
88  let currname = Expand!(Tokens!(T_CS!("\\@currname"))).to_string();
89  let currext = Expand!(Tokens!(T_CS!("\\@currext"))).to_string();
90  let requested_sty = match binding_request() {
91    Some(request) => request == "epsf.sty",
92    None => currname == "epsf" && currext == "sty",
93  };
94  let exts: &[&str] = if requested_sty {
95    &["sty", "tex"]
96  } else {
97    &["tex"]
98  };
99  for ext in exts {
100    if let Some(text) = shipped_file_text("epsf", ext) {
101      let code = class_code(&text);
102      if code
103        .split("\\def")
104        .skip(1)
105        .any(|rest| rest.trim_start().starts_with("\\epsfbox"))
106      {
107        return Ok(code.match_indices("\\global").any(|(at, global)| {
108          code[at + global.len()..]
109            .trim_start()
110            .starts_with("\\epsfxsize")
111        }));
112      }
113      if ["\\RequirePackage", "\\usepackage"].iter().any(|load| {
114        code.match_indices(load).any(|(at, cs)| {
115          let rest = code[at + cs.len()..].trim_start();
116          let rest = rest
117            .strip_prefix('[')
118            .and_then(|r| r.split_once(']'))
119            .map_or(rest, |(_, r)| r.trim_start());
120          rest.starts_with("{epsfig}")
121        })
122      }) {
123        return Ok(false);
124      }
125    }
126  }
127  Ok(true)
128}

130/// The `<ltx:graphics>` properties of `\epsfbox[<bb>]{<file>}` (Perl epsf.sty.ltxml L45-62): the candidates of the
131/// file, and options assembled from the clip switch (the value `clip_key`, which `\epsfclipon` sets), the bounding box,
132/// `\epsfxsize` and `\epsfysize`. Shared with epsfig's `\epsfbox` (epsfig.sty:61-100).
133pub(crate) fn epsfbox_properties(bb: &str, path: &str, clip_key: &str) -> SymHashMap<Stored> {
134  let bb = bb.trim();
135  let path = path.trim().to_string();
136  let candidates = util::image::image_candidates(&path);
137  let clip = lookup_value(clip_key)
138    .map(|v| v.to_string() != "0" && !v.to_string().is_empty())
139    .unwrap_or(false);
140  let mut options = String::new();
141  if clip {
142    options = if bb.is_empty() {
143      "clip".to_string()
144    } else {
145      format!("viewport={}, clip", bb)
146    };
147  }
148  let reg_dim = |name: &str| -> Option<Dimension> {
149    match lookup_register(name, Vec::new()) {
150      Ok(Some(RegisterValue::Dimension(d))) => Some(d),
151      _ => None,
152    }
153  };
154  if let Some(w) = reg_dim("\\epsfxsize")
155    && w.value_of() > 0
156  {
157    if !options.is_empty() {
158      options.push(',');
159    }
160    options.push_str(&format!("width={}", w.to_attribute()));
161  }
162  if let Some(h) = reg_dim("\\epsfysize")
163    && h.value_of() > 0
164  {
165    if !options.is_empty() {
166      options.push(',');
167    }
168    options.push_str(&format!("height={}", h.to_attribute()));
169  }
170  stored_map!("graphic" => path, "candidates" => candidates, "options" => options)
171}
