97  DefMacro!("\\lx@thmt@declaretheorem@i{}{}[]", sub[(first, names, last)] {
98    let mut options = first.unwrap_or_default().unlist();
99    let last = last.unwrap_or_default().unlist();
100    if !options.is_empty() && !last.is_empty() {
101      options.push(T_OTHER!(","));
102    }
103    options.extend(last);
104    let comma = T_OTHER!(",");
105    let mut out = Vec::new();
106    for name in names.unlist().split(|t| *t == comma) {
107      let name: Vec<Token> = name.iter().skip_while(|t| t.get_catcode() == Catcode::SPACE).cloned().collect();
108      let end = name.iter().rposition(|t| t.get_catcode() != Catcode::SPACE).map_or(0, |i| i + 1);
109      if end == 0 {
110        continue;
111      }
112      out.push(T_CS!("\\lx@thmt@declaretheorem"));
113      out.push(T_OTHER!("["));
114      out.extend(options.iter().cloned());
115      out.push(T_OTHER!("]"));
116      out.push(T_BEGIN!());
117      out.extend(name[..end].iter().cloned());
118      out.push(T_END!());
119    }
