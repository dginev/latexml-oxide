3/// The prefix of a raw body line up to its first *unescaped* `%` — comment.sty
4/// reads its body with `%` active as a TeX comment, so an `\end{name}` sitting
5/// behind a `%` (e.g. `% […] \end{comment}`) does NOT close the environment.
6/// `%` and `\` are ASCII, so byte scanning is UTF-8-safe (the cut index always
7/// lands on a char boundary).
8fn strip_tex_comment(line: &str) -> &str {
9  let bytes = line.as_bytes();
10  for i in 0..bytes.len() {
11    if bytes[i] == b'%' {
12      let mut bs = 0;
13      while i > bs && bytes[i - 1 - bs] == b'\\' {
14        bs += 1;
15      }
16      if bs % 2 == 0 {
17        return &line[..i];
18      }
19    }
20  }
21  line
22}

24LoadDefinitions!({
25  //**********************************************************************
26  // Define \name and \begin{name} to start an ignored section
27  // until \endname or \end{name}, respectively
28  let define_excluded: PrimitiveClosure = Rc::new(|mut args: Vec<ArgWrap>| {
29    let name = args.remove(0).owned_tokens().unwrap();
30    let begin_mark = s!("\\begin{{{name}}}");
31    DefConstructor!(T_CS!(begin_mark), None, None,
32    after_digest => {
33      // Detect `\end{name}` MID-LINE (allowing spaces, `\end {name}`), but only
34      // in the code part of the line — an `\end` hidden behind a `%` comment
35      // does NOT close the environment (comment.sty reads its body with `%`
36      // active). Perl comment.sty.ltxml L30 matched only a whole line,
37      // `/^\s*\Q$endmark\E\s*$/`, and our prior port did the same
38      // (`line.trim() == end_mark`), so a comment ending `…text.\end{name}`
39      // overran to EOF and silently swallowed everything after it — a document's
40      // bibliography included. Surpass-Perl divergence (OXIDIZED_DESIGN #133):
41      // both LaTeXML engines lose it, and pdflatex keeps it for the witness
42      // arXiv:2606.11493 (`\begin{comment}…\(G(h_1)=0\).\end{comment}` swallowed
43      // a 31-`\bibitem` thebibliography). The `%`-guard keeps the tokenize
44      // `comment` fixture's `% […] \end{Excluded}` from closing early. Guard
45      // `comment_midline_end_keeps_bibliography`.
46      let end_re = Regex::new(&format!("\\\\end\\s*\\{{{name}\\}}")).unwrap();
47      let mut nlines = 0;
48      read_raw_line();    // IGNORE 1st line (after the \begin{$name} !!!
49      while let Some(line) = read_raw_line() {
50        let code = strip_tex_comment(&line);
51        if let Some(m) = end_re.find(code) {
52          // comment.sty expects `\end{name}` at the line's end; trailing content
53          // on that line is dropped, as verbatim.sty's `\verbatim@` does too.
54          if !code[m.end()..].trim().is_empty() {
55            Info!("unexpected", "stuff",
56              s!("Characters dropped after '\\end{{{name}}}'"));
57          }
58          break;
59        }
60        nlines += 1;
61      }
62      note_progress(&s!("[Skipped {name} ({nlines} lines)]"));
63      Ok(Vec::new())
64    });
65    Ok(Vec::new())
66  });
67
68  // I don't understand Rust closures enough to figure out how to clone one, so instantiating it
69  // twice instead, via a macro
70  let define_included: PrimitiveBody = PrimitiveBody::Closure(Rc::new(|mut args: Vec<ArgWrap>| {
71    args.reverse(); // we'll be using .pop() from the front
72    let name = args
73      .pop()
74      .unwrap()
75      .owned_tokens()
76      .expect("expecting a Tokens argument")
77      .to_string();
78    let mut before_tokens = match args.pop() {
79      Some(arg) => arg.unlist(),
80      None => Vec::new(),
81    };
82    before_tokens.push(T_CS!("\\ignorespaces"));
83    let mut after_tokens = match args.pop() {
84      Some(arg) => arg.unlist(),
85      None => Vec::new(),
86    };
87    after_tokens.push(T_CS!("\\ignorespaces"));
88    // Note that we define the `magic' environment control sequences,
89    // but DO NOT do any of the normal environ things, like \begingroup \endgroup!
90    DefMacro!(T_CS!(s!("\\begin{{{name}}}")), None, {
91      read_raw_line(); // IGNORE 1st line (after the \begin{$name} !!!
92      before_tokens.clone()
93    });
94    DefMacro!(
95      T_CS!(s!("\\end{{{name}}}")),
96      None,
97      Tokens::new(after_tokens)
98    );
99    Ok(Vec::new())
100  }));
101
102  define_excluded(vec![ArgWrap::Tokens(Tokenize!("comment"))])?;
103
104  DefPrimitive!("\\includecomment{}", Some(define_included.clone()));
105  DefPrimitive!(
106    "\\excludecomment{}",
107    Some(PrimitiveBody::Closure(define_excluded))
108  );
109  DefPrimitive!("\\specialcomment{}{}{}", Some(define_included));
110  DefPrimitive!("\\processcomment{}{}{}{}", None);
111  DefMacro!("\\csarg{}{}", r"\expandafter#1\csname#2\endcsname");
112});
