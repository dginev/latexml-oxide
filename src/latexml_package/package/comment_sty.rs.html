5  // Define \name and \begin{name} to start an ignored section
6  // until \endname or \end{name}, respectively
7  let define_excluded: PrimitiveClosure = Rc::new(|mut args: Vec<ArgWrap>| {
8    let name = args.remove(0).owned_tokens().unwrap();
9    let name_str = name.to_string();
10    let begin_mark = format!("\\begin{{{name_str}}}");
11    let begin_mark_let = begin_mark.clone();
12    let name_clone = name_str.clone();
13    DefConstructor!(T_CS!(begin_mark), None, None,
14    after_digest => {
15      // comment.sty:186-193 reads the body LINE by line with every special
16      // made innocent and compares each whole line against `\end{name}`
17      // (`\ProcessCommentLine#1^^M{\def\test{#1}\csarg\ifx{End…Test}\test`):
18      // only a line that IS `\end{name}` ends the comment — TeX's line
19      // reader strips trailing spaces (tex.web §362), so `\end{name}   ` does,
20      // while a trailing `%` (innocent, so literal), text
21      // before it, or text after it do not, and the comment then runs to the
22      // end of the file where pdflatex stops with "File ended while scanning
23      // use of \next" (verified on each shape, 2026-09-05). The earlier
24      // mid-line detection (OXIDIZED_DESIGN #133, since retracted) showed
25      // MORE than pdflatex; we now report TeX's error at EOF instead.
26      // An end line indented by spaces or tabs ends it too, as Perl's `/^\s*\Q$endmark\E\s*$/` does
27      // (comment.sty.ltxml:30), where pdflatex runs on to the end of the file: a TAB indent ended it in LaTeX until
28      // 2024-11, which put TAB in `\dospecials`, and the acmart papers that indent `\end{CCSXML}` lost their whole body
29      // to it (2401.14656, 2405.12964, 2411.10188, 2501.06699, 2504.19287; user ruling 2026-10-09: accept both).
30      // The terminator is handed to the CURRENT `\end` macro (K3, #199).
31      // Guards: `00_tokenize::comment_test` (golden), `06_cluster_bibliography::
32      // comment_midline_end_runs_to_eof_like_pdflatex`,
33      // `perfect_kernel_gemini::comment_self_terminating_hands_to_end`.
34      let end_line = format!("\\end{{{name_clone}}}");
35      let mut nlines = 0;
36      let mut ended = false;
37      read_raw_line();    // IGNORE 1st line (after the \begin{$name} !!!
38      while let Some(line) = read_raw_line() {
39        if line.trim() == end_line {
40          let mut end_tokens = vec![T_CS!("\\end"), T_BEGIN!()];
41          end_tokens.extend(ExplodeText!(&name_clone));
42          end_tokens.push(T_END!());
43          unread_expansion(Tokens::new(end_tokens));
44          ended = true;
45          break;
46        }
47        nlines += 1;
48      }
49      if !ended {
50        Error!("unexpected", "EOF",
51          s!("File ended while scanning use of \\next (no whole-line \\end{{{name_clone}}} closes this {name_clone})"));
52      }
53      note_progress(&s!("[Skipped {name_clone} ({nlines} lines)]"));
54      Ok(Vec::new())
55    });
56    DefMacro!(T_CS!(format!("\\end{{{name_str}}}")), None, Tokens!());
57    // (an empty name, `\excludecomment{}` or `\includecomment{}` written by mistake — 2508.09827:687 — names no command
58    // of the document, comment.sty's being `\csname\endcsname`: `\<name>` would be read as `\\`, every later line break
59    // reading the rest of its line raw, and `\end<name>` would be `\end`)
60    if !name_str.is_empty() {
61      DefMacro!(T_CS!(format!("\\end{name_str}")), None, Tokens!());
62    }
63    // comment.sty's own name for the environment (`\csarg\def{#1}`, comment.sty:241-248): a kernel-style `\begin`
64    // that looks up only `\<name>` — arabtex's alatex.sty:113 — finds it, where Perl's binding (only `\begin{<name>}`)
65    // left "Environment comment undefined" and typeset the comment (KNOWN_PERL_ERRORS #1148; 2104.08110's commented-out
66    // figure, 4 missing images).
67    // Guard: `perfect_kernel_batch79k::a_comment_under_a_kernel_begin_is_skipped`.
68    if !name_str.is_empty() {
69      let_i(
70        &T_CS!(format!("\\{name_str}")),
71        &T_CS!(begin_mark_let),
72        None,
73      );
74    }
75    Ok(Vec::new())
76  });
77
78  // I don't understand Rust closures enough to figure out how to clone one, so instantiating it
79  // twice instead, via a macro
80  let define_included: PrimitiveBody = PrimitiveBody::Closure(Rc::new(|mut args: Vec<ArgWrap>| {
81    args.reverse(); // we'll be using .pop() from the front
82    let name = args
83      .pop()
84      .unwrap()
85      .owned_tokens()
86      .expect("expecting a Tokens argument")
87      .to_string();
88    let mut before_tokens = match args.pop() {
89      Some(arg) => arg.unlist(),
90      None => Vec::new(),
91    };
92    before_tokens.push(T_CS!("\\ignorespaces"));
93    let mut after_tokens = match args.pop() {
94      Some(arg) => arg.unlist(),
95      None => Vec::new(),
96    };
97    after_tokens.push(T_CS!("\\ignorespaces"));
98    // Note that we define the `magic' environment control sequences,
99    // but DO NOT do any of the normal environ things, like \begingroup \endgroup!
100    DefMacro!(T_CS!(s!("\\begin{{{name}}}")), None, {
101      read_raw_line(); // IGNORE 1st line (after the \begin{$name} !!!
102      before_tokens.clone()
103    });
104    DefMacro!(
105      T_CS!(s!("\\end{{{name}}}")),
106      None,
107      Tokens::new(after_tokens)
108    );
109    // comment.sty's own names (`\csarg\def{#1}`, comment.sty:234-240), as for an excluded comment above, and none for
110    // an empty name
111    if !name.is_empty() {
112      let_i(
113        &T_CS!(s!("\\{name}")),
114        &T_CS!(s!("\\begin{{{name}}}")),
115        None,
116      );
117      let_i(
118        &T_CS!(s!("\\end{name}")),
119        &T_CS!(s!("\\end{{{name}}}")),
120        None,
121      );
122    }
123    Ok(Vec::new())
124  }));
125
126  define_excluded(vec![ArgWrap::Tokens(Tokenize!("comment"))])?;
127
128  DefPrimitive!("\\includecomment{}", Some(define_included.clone()));
129  DefPrimitive!(
130    "\\excludecomment{}",
131    Some(PrimitiveBody::Closure(define_excluded))
132  );
133  DefPrimitive!("\\specialcomment{}{}{}", Some(define_included));
134  DefMacro!("\\csarg{}{}", r"\expandafter#1\csname#2\endcsname");
135  // comment.sty's own machinery (:170-233, :283-289), for what builds a comment type of its own on it (lwarp.sty:1349-1363
136  // `\CommentEndDef`, `\ProcessComment`, `\SetUpCutFile`: tikz-trackschematic and its snippets), and the two types the
137  // binding leaves to it, as the file has them (:266-272 `\generalcomment`, `\processcomment`; Perl's binding has none of
138  // this, "\CommentEndDef undefined"). `\excludecomment`, `\includecomment`, `\specialcomment` stay the binding's.
139  RawTeX!(
140    r"\def\makeinnocent#1{\catcode`#1=12 }
141\newwrite\CommentStream
142\def\DefaultCutFileName{\def\CommentCutFile{comment.cut}}
143\DefaultCutFileName
144\def\ProcessComment#1{\def\CurrentComment{#1}%
145  \begingroup
146  \let\do\makeinnocent \dospecials
147  \makeinnocent\^^L%
148  \endlinechar`\^^M\relax \catcode`\^^M=12\relax \xComment}
149{\catcode`\^^M=12 \endlinechar=-1 %
150 \gdef\xComment#1^^M{\ProcessCommentLine}
151 \gdef\ProcessCommentLine#1^^M{\def\test{#1}
152      \csarg\ifx{End\CurrentComment Test}\test
153          \edef\next{\endgroup\noexpand\EndOfComment{\CurrentComment}}%
154      \else \ThisComment{#1}\let\next\ProcessCommentLine
155      \fi \next}
156}
157\def\SetUpCutFile{\immediate\openout\CommentStream=\CommentCutFile
158  \PrepareCutFile}
159\def\PrepareCutFile{}
160\long\def\WriteCommentLine#1{\immediate\write\CommentStream{\unexpanded{#1}}}
161\let\ThisComment\WriteCommentLine
162\def\ProcessCutFile{\message{Straight input of \CommentCutFile.}%
163  \input{\CommentCutFile}\relax}
164\def\CloseAndInputCutFile{\FinalizeCutFile
165  \immediate\closeout\CommentStream
166  \ProcessCutFile}
167\def\FinalizeCutFile{}
168\long\def\generalcomment#1#2#3{\message{General comment '#1'}%
169  \csarg\def{#1}{\endgroup #2 \relax \SetUpCutFile \ProcessComment{#1}}%
170  \csarg\def{After#1Comment}{\CloseAndInputCutFile #3}%
171  \CommentEndDef{#1}}
172\long\def\processcomment#1#2#3#4{\message{Lines-Processing comment '#1'}%
173  \csarg\def{#1}{\endgroup \SetUpCutFile #2\relax
174    \ProcessComment{#1}}%
175  \csarg\def{After#1Comment}{#3\CloseAndInputCutFile #4}%
176  \CommentEndDef{#1}}
177\def\EndOfComment#1{\csname After#1Comment\endcsname
178  \begingroup\def\@currenvir{#1}\end{#1}}
179\def\CommentEndDef#1{{\escapechar=-1\relax
180  \csarg\xdef{End#1Test}{\string\\end\string\{#1\string\}}}}"
181  );
