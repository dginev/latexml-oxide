8/// DEP-NEW (2026-05-19): data-drive helper for the 74 IOP-physics
9/// journal abbreviation macros that all expand to
10/// `\textit{<name>}` (some with trailing italic-correction `\/`).
11/// Replacing the per-call `DefMacro!` macro arms with this runtime
12/// helper collapses the repeated macro expansion to one inlined fn.
13/// Per [[wisdom_data_drive_min_call_sites]]: 74 ≫ 5 threshold.
14/// iopart.cls ends each with a space (`\newcommand{\jpa}{{\it J. Phys. A: Math. Theor.} }`, 2603.00348's
15/// :972-1125), so `\jpa \textbf{40}` reads "J. Phys. A: Math. Theor. 40"; Perl's dropped it (KNOWN_PERL_ERRORS #634).
16fn def_iop_journal(cs: &str, name: &str) -> Result<()> {
17  let (cs_tok, params) = parse_prototype(cs, true)?;
18  let body_toks =
19    mouth::tokenize_internal(TeXString::assembled(format!("\\textit{{{name}}}\\space")));
20  def_macro(cs_tok, params, ExpansionBody::Tokens(body_toks), None)?;
21  Ok(())
22}

24/// The IOP classes' address and e-mail blocks — iopart.cls:240-248, 363-364 and jpconf.cls:104-110 (IOP's conference
25/// class, which copies them: `\eads`, `\ead`, `\mailto`, `\address`) — as frontmatter: each `\address` an affiliation,
26/// each `\ead` an e-mail placed with its author. For iopart_support and the jpconf binding (`jpconf_cls`, which reads
27/// the paper's copy raw and then these, where the raw class set them as body lists: 0705.0202, 1011.0523).
28#[rustfmt::skip]
29pub fn iop_address_and_email() -> Result<()> {
30  // iopart.cls:240-248: "Use once for each address" — each `\address` prints its own block, so each adds its
31  // affiliation, linked by its superscript marks when the authors carry them. Perl's `\lx@add@affiliations`
32  // (iopart_support.sty.ltxml:53) dequeues the affiliations before it, so only the last `\address` survived
33  // (KNOWN_PERL_ERRORS #511; 21 of the 41 2609 iopart papers call it more than once, 2609.01831 lost three of four).
34  // (its one leading mark is the mark of the `\ead` blocks printed right after it, `\lx@address@mark@set`)
35  DefMacro!("\\address{}", "\\lx@address@mark@set{#1}\\lx@add@affiliation@marked{#1}");
36  // iopart.cls:363 `\def\ead#1{\vspace*{5pt}\address{E-mail: \mailto{#1}}}`, `\mailto` = `{\tt #1}` (:364; 2301.12119,
37  // 2501.12730): the address prints as an address block of its own right after the one before it, no name's. Perl's
38  // `\lx@add@email` gives it to the last author (2311.12551's benedikt.maier@cern.ch on T. K. Årrestad, author 7 of 7;
39  // in the 162 census iopart `\ead` papers A/B'd for batch 78f, int47a put 83 addresses, in 81 papers, on an author
40  // another name spells). Rust-only (OXIDIZED_DESIGN_DIVERGENCES #806, #808, #809): placed as jheppub's `\emailAdd`
41  // is — with the author it spells, else the n-th author's while the addresses are the authors' in order, else with
42  // the one author requesting the mark of the `\address` it follows (`mark=address`: 2407.14305's `rockfish3141@` under
43  // `$^1$ Waseda`, Hidetomo Hoshino's; 2311.07087 — in a document that writes an `\ead` between its addresses;
44  // 1209.0173, 0910.0753, 1111.3344 write them all after), else the group's only author's (one `\author` per call:
45  // 2311.17496), else — the one address left on the author block — the ONE author of its group (and of those requesting
46  // its address's mark) marked as the corresponding author by a note naming them so or a mark whose legend says so
47  // (`$^{1,*}$` under `\address{$^*$ Author to whom any correspondence should be addressed.}`, 2504.15015; two
48  // legend-less `$^{*}$`, 2111.00753, mark no one), else the author block's (`\lx@add@contact@placed[owner=sequence]`).
49  // An `\ead{a@x, b@y}` or `\ead{a@x and b@y}` list is an address each (2111.00753; 0710.2104, 1003.0073), and a
50  // `\mailto` in it is the address font iopart.cls:364 gives it
51  // (`\def\mailto#1{{\tt #1}}`: `\ead{\mailto{edvard@physto.se}, \mailto{chris.clarkson@uct.ac.za}}`, 0811.0981,
52  // where the `\let\mailto\ead` below made each a contact nested in the first).
53  DefMacro!("\\ead Semiverbatim", "\\lx@add@contact@list{\\lx@iop@ead}{#1}");
54  DefMacro!("\\lx@iop@ead{}",
55    "\\lx@add@contact@placed[role=email,owner=sequence,mark=address]{}{\\lx@address@urlfont{\\let\\mailto\\@firstofone#1}}");
56  // Contact — Perl L57-59
57  Let!("\\mailto", "\\ead");
58  DefMacro!("\\eads{}", "#1");
59  Ok(())
60}
