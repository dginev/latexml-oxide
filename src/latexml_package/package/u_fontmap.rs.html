1//! The `U` ("unknown") font encoding: glyphs come from the FAMILY. A raw
2//! `\DeclareSymbolFont{AMSa}{U}{msa}{m}{n}` (oz.sty:34, amsfonts users) then
3//! decodes through the per-family map `U_msa_fontmap` (`font_decode`; AMSb's `msb` through
4//! `U_msb_fontmap`, which amsfonts' `\newsymbol` reaches), which
5//! Perl (no `u.fontmap.ltxml`) leaves undecoded. The Euler Roman family `eur` (amsfonts' ueur.fd, eurm/eurb) decodes
6//! through the math italic layout, OML, at the 111 slots it shares with cmmi: its TFMs say "TEX MATH ITALIC SUBSET",
7//! tftopl gives no slot outside cmmi10's, and rendered they draw cmmi's glyphs upright (program.sty:834-835's
8//! `\powersetsymbol` ℘ at "7D, older sn-jnl.cls copies; statex, beilstein.cls, mismath, mathgreeks, sansmath raw);
9//! the 17 slots it lacks, 37 and 38 (ϱ and ς, "Euler's missing variants", sansmath.sty:76), 40-47 (harpoons, hooks,
10//! triangles), 63 (star), 91-95 (flat, natural, sharp, smile, frown) and 126 (the vector accent), stay undecodable
11//! (`U_eur_fontmap`), and are recorded as gaps of the font (`U_eur_fontmap_lacks`): a kernel name declared there —
12//! `\star` at "3F, `\vec` at "7E — is U+FFFD with the font's Warning, as pdflatex prints nothing there
13//! (OXIDIZED_DESIGN_DIVERGENCES #27). So are msbm's five gaps, "5F and "62-"65 (`U_msb_fontmap_lacks`: amssymb's
14//! `AMSa` `\varpropto`, `\Subset`, `\Supset`, `\Cup`, `\Cap` slots, which a name declared in `AMSb` would otherwise
15//! decode by name); msam has none. Every family's gaps are generated by tftopl over each size its .fd loads
16//! (`U_FONT_GAPS`, tools/perfect_kernel/gen_reference_math_slots.py `--u-font-gaps`); the maps' other empty slots are
17//! glyphs with no single character (msbm's negated pairs, msam's `\dabar@`), no gaps.
18//! The encoding has no map of its own, bbold loaded or not (its blackboard map is its own family's, `U_blackboard` and
19//! `U_bbold`, bbold_sty.rs; OXIDIZED_DESIGN_DIVERGENCES #914): a U symbol font of another family
20//! (shuffle.sty, stmaryrd's `stmry`) stays undecoded — U+FFFD, and one Warning per symbol font
21//! (OXIDIZED_DESIGN_DIVERGENCES #27) — but for a kernel or AMS name declared at its fontmath.ltx or
22//! amssymb slot (sect08.rs `reference_symbol`, OXIDIZED_DESIGN_DIVERGENCES #910) — and the text switch
23//! `\fontencoding{U}` falls back to OT1 while `\f@encoding` still names `U` (`\lx@fontencoding`, tex_fonts.rs;
24//! OXIDIZED_DESIGN_DIVERGENCES #645). Guards:
25//! `perfect_kernel_batch54::declare_math_delimiter_defines_the_symbol`,
26//! `perfect_kernel_batch68v::u_symbol_font_without_a_map_stays_undecoded`,
27//! `perfect_kernel_batch89c::euler_roman_decodes_as_math_italic`.
28//! Witnesses: 2306.04672 (`\shuffle`); 2205.13841, 2212.02709 (program.sty's `U/eur` via sn-jnl.cls copies; neither
29//! prints `\powerset`, so no census paper shows the ℘).
30use crate::prelude::*;

32LoadDefinitions!({
33  DeclareFontMap!("U", amsa_fontmap::amsa_table(), "msa");
34  DeclareFontMap!("U", amsb_fontmap::amsb_table(), "msb");
35  if let Some(oml) = load_font_map("OML") {
36    let eur: Rc<[Option<char>]> = oml
37      .iter()
38      .enumerate()
39      .map(|(slot, glyph)| {
40        if font_gaps("eur").contains(&slot) {
41          None
42        } else {
43          *glyph
44        }
45      })
46      .collect();
47    DeclareFontMap!("U", eur, "eur");
48  }
49  // The slots each family's fonts lack, as `\DeclareMathSymbol`/`\DeclareMathAccent` read them: a gap of the font
50  // (pdflatex: "Missing character"), not a glyph the map has no character for, so no reference name decodes it
51  // (sect08.rs `symbol_font_lacks_slot`).
52  for (family, gaps) in U_FONT_GAPS {
53    let lacks: HashMap<String, String> = gaps
54      .iter()
55      .map(|slot| (slot.to_string(), String::new()))
56      .collect();
57    assign_value(&s!("U_{family}_fontmap_lacks"), lacks, Some(Scope::Global));
58  }
59});

61/// The gaps of the mapped `U` family `family` ([`U_FONT_GAPS`]).
62fn font_gaps(family: &str) -> &'static [usize] {
63  U_FONT_GAPS
64    .iter()
65    .find(|(name, _)| *name == family)
66    .map_or(&[], |(_, gaps)| gaps)
67}
