3/// The columns an `\hhline` specification rules (hhline.sty:61-96 `\HH@loop`, its `*{n}{…}` repeats expanded first
4/// by `\HH@xexpast`, :49-60): each `-` (a rule), `=` (a double rule) and `~` (none) is a column, in order from the
5/// first; `|`, `:`, `#` (vertical rules where they cross) and `t`, `b` (the halves of a double rule at a corner) mark
6/// the rule between or on the columns, and spaces are skipped. Under colortbl a `>{<code>}` sets the rules' colours
7/// (colortbl.sty:364-369), not modelled here. Returns the ruled columns, the doubly ruled ones, and what `\HH@loop`
8/// ignores with its warning (a braced group, read by `\HH@let` as one argument, once).
9fn hhline_columns(spec: &str, colortbl: bool) -> (Vec<usize>, Vec<usize>, Vec<String>) {
10  let mut chars: Vec<char> = Vec::new();
11  expand_repeats(&spec.chars().collect::<Vec<char>>(), &mut chars);
12  let (mut ruled, mut doubled, mut ignored) = (Vec::new(), Vec::new(), Vec::new());
13  let mut column = 0;
14  let mut at = 0;
15  // Past the item at `at`: a braced group whole, else one character.
16  let skip_item = |at: usize| -> usize {
17    if chars.get(at) != Some(&'{') {
18      return at + 1;
19    }
20    let mut depth = 0;
21    for (i, &c) in chars.iter().enumerate().skip(at) {
22      match c {
23        '{' => depth += 1,
24        '}' => {
25          depth -= 1;
26          if depth == 0 {
27            return i + 1;
28          }
29        },
30        _ => {},
31      }
32    }
33    chars.len()
34  };
35  while at < chars.len() {
36    let c = chars[at];
37    match c {
38      '-' | '=' | '~' => {
39        column += 1;
40        if c != '~' {
41          ruled.push(column);
42        }
43        if c == '=' {
44          doubled.push(column);
45        }
46      },
47      '|' | ':' | '#' | 't' | 'b' => {},
48      c if c.is_whitespace() => {},
49      '>' if colortbl => {
50        // `>` and its code, whatever spaces stand between (`\HH@loop` reads `##1##2`).
51        let mut next = at + 1;
52        while chars.get(next).is_some_and(|c| c.is_whitespace()) {
53          next += 1;
54        }
55        at = skip_item(next);
56        continue;
57      },
58      '{' => {
59        ignored.push("begin-group character {".to_string());
60        at = skip_item(at);
61        continue;
62      },
63      c => ignored.push(s!("the character {c}")),
64    }
65    at += 1;
66  }
67  (ruled, doubled, ignored)
68}

70/// `\HH@xexpast`'s `*<n><spec>` (as latex.ltx's `\@xexpast` for a column preamble): `<n>` and `<spec>` are each one
71/// character or one braced group, the repeat expanded in place, nested repeats too.
72fn expand_repeats(spec: &[char], out: &mut Vec<char>) {
73  // One undelimited argument at `at`: a braced group's inside or one character, and where it ends.
74  fn argument(spec: &[char], mut at: usize) -> (Vec<char>, usize) {
75    while at < spec.len() && spec[at].is_whitespace() {
76      at += 1;
77    }
78    if at >= spec.len() {
79      return (Vec::new(), at);
80    }
81    if spec[at] != '{' {
82      return (vec![spec[at]], at + 1);
83    }
84    let mut depth = 0;
85    for (i, &c) in spec.iter().enumerate().skip(at) {
86      match c {
87        '{' => depth += 1,
88        '}' => {
89          depth -= 1;
90          if depth == 0 {
91            return (spec[at + 1..i].to_vec(), i + 1);
92          }
93        },
94        _ => {},
95      }
96    }
97    (spec[at + 1..].to_vec(), spec.len())
98  }
99  let mut at = 0;
100  while at < spec.len() {
101    if spec[at] == '*' {
102      let (count, next) = argument(spec, at + 1);
103      let (body, next) = argument(spec, next);
104      let count: usize = count.iter().collect::<String>().trim().parse().unwrap_or(0);
105      for _ in 0..count {
106        expand_repeats(&body, out);
107      }
108      at = next;
109    } else {
110      out.push(spec[at]);
111      at += 1;
112    }
113  }
114}

116#[rustfmt::skip]
117LoadDefinitions!({
118  // hhline.sty:61-96: `\hhline{<spec>}` rules the columns its specification names — `-` a rule, `=` a double rule,
119  // `~` no rule, the vertical and corner marks `|`, `:`, `#`, `t`, `b` between them, `*{n}{…}` a repeat — as a row of
120  // its own under the one above. Here the ruled columns are `\cline`s of the table model (sect10.rs `\lx@cline`), a
121  // doubly ruled column twice (a doubled rule, as `\hline\hline`). Perl's binding (hhline.sty.ltxml:24) maps every
122  // `\hhline` to a full-width `\hline` (KNOWN_PERL_ERRORS #1024: `\hhline{~~--}` ruled every column, `=` a single rule;
123  // about 6,000 papers). What `\HH@loop` does not know is ignored with hhline's warning (:89-91). Guard
124  // `perfect_kernel_batch78c::hhline_rules_the_columns_it_names`.
125  DefMacro!("\\hhline Semiverbatim", sub[(spec)] {
126    let spec = spec.to_string();
127    let (ruled, doubled, ignored) = hhline_columns(&spec, package_loaded("colortbl"));
128    for meaning in ignored {
129      Warn!("unexpected", "\\hhline", s!("Package hhline Warning: {meaning} ignored in \\hhline argument"));
130    }
131    // One `\cline{a-b}` per run of adjacent columns, so the rule reverts as LaTeX writes it (a formula's `tex`).
132    let mut code = String::new();
133    for columns in [&ruled, &doubled] {
134      let mut runs: Vec<(usize, usize)> = Vec::new();
135      for &column in columns.iter() {
136        match runs.last_mut() {
137          Some((_, last)) if *last + 1 == column => *last = column,
138          _ => runs.push((column, column)),
139        }
140      }
141      for (first, last) in runs {
142        code.push_str(&s!("\\noalign{{\\lx@cline{{{first}-{last}}}}}"));
143      }
144    }
145    Ok(TokenizeInternal!(TeXString::assembled(code)))
146  });
147});
