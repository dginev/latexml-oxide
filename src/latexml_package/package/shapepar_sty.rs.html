3// shapepar.sty: paragraphs typeset in a shape. Raw, `\shapepar`'s automatic
4// scaling (shapepar.sty:206-219) measures the shape's area from the text's
5// line geometry (`\SH@measline`, :229) and `\loop`s `\multiply … 256` until
6// the area is ≥ 1pt — an area LaTeXML's no-layout model leaves at 0, so the
7// loop never ends (arsclassica/ArsClassica `\heartpar`, `MemoryBudget`;
8// Perl loops too; pdflatex clean). A shaped paragraph is visual layout:
9// the binding keeps the text as a plain paragraph and accepts the shape
10// specifications. Ready-made wrappers per shapepar.sty:999-1133 (`\diamondpar`
11// keeps its `$\diamondsuit$` ornaments). Guard:
12// `perfect_kernel_batch54::shapepar_paragraphs_are_plain_paragraphs`.
13#[rustfmt::skip]
14LoadDefinitions!({
15  // `\shapepar[<scale>]{<shape spec>} <text> \par` (:151), `\Shapepar` (:148)
16  def_macro_noop("\\shapepar[]{}")?;
17  def_macro_noop("\\Shapepar[]{}")?;
18  def_macro_noop("\\cutout{}{}")?;
19  DefMacro!("\\cutoutsepstretch", ".5");
20  for shape in ["\\diamondshape", "\\squareshape", "\\heartshape", "\\circleshape",
21                "\\nutshape", "\\hexagonshape", "\\CDlabshape", "\\starshape"] {
22    def_macro_noop(shape)?;
23  }
24  def_macro_noop("\\rectangleshape{}{}")?;
25  DefMacro!("\\diamondpar{}", "$\\diamondsuit$ #1 $\\diamondsuit$\\par");
26  DefMacro!("\\squarepar{}", "#1\\par");
27  DefMacro!("\\heartpar{}", "#1\\par");
28  DefMacro!("\\circlepar{}", "#1\\par");
29  DefMacro!("\\nutpar{}", "#1\\par");
30  DefMacro!("\\hexagonpar{}", "#1\\par");
31  DefMacro!("\\starpar{}", "#1\\par");
32});
