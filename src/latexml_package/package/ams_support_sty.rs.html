10/// amsart.cls/amsbook.cls/amsproc.cls:258-296 (the three classes' option blocks are the same): each size option sets
11/// `\@mainsize`, `\@ptsize` and `\@typesizes`. Declared by the class bindings, as the classes declare them, so a class
12/// loaded with a size (`\LoadClass[12pt]{amsart}`) sets it; ams_support declares them as no-ops (Perl
13/// ams_support.sty.ltxml:31-39), as a package's `ProcessOptions` would run the document's size option again after the
14/// class settled its own (and over a non-AMS class's `\@ptsize`).
15#[rustfmt::skip]
16pub fn declare_ams_size_options() -> Result<()> {
17  // amsart.cls:169-170
18  raw_tex(r"\providecommand{\@mainsize}{10}\providecommand{\@ptsize}{0}")?;
19  ams_size_default()?;
20  DeclareOption!("8pt", "\\def\\@mainsize{8}\\def\\@ptsize{8}\\def\\@typesizes{\\or{5}{6}\\or{5}{6}\\or{5}{6}\\or{6}{7}\\or{7}{8}\\or{8}{10}\\or{9}{11}\\or{10}{12}\\or{\\@xipt}{13}\\or{\\@xiipt}{14}\\or{\\@xivpt}{17}}");
21  DeclareOption!("9pt", "\\def\\@mainsize{9}\\def\\@ptsize{9}\\def\\@typesizes{\\or{5}{6}\\or{5}{6}\\or{6}{7}\\or{7}{8}\\or{8}{10}\\or{9}{11}\\or{10}{12}\\or{\\@xipt}{13}\\or{\\@xiipt}{14}\\or{\\@xivpt}{17}\\or{\\@xviipt}{20}}");
22  DeclareOption!("10pt", "\\def\\@mainsize{10}\\def\\@ptsize{0}\\def\\@typesizes{\\or{5}{6}\\or{6}{7}\\or{7}{8}\\or{8}{10}\\or{9}{11}\\or{10}{12}\\or{\\@xipt}{13}\\or{\\@xiipt}{14}\\or{\\@xivpt}{17}\\or{\\@xviipt}{20}\\or{\\@xxpt}{24}}");
23  DeclareOption!("11pt", "\\def\\@mainsize{11}\\def\\@ptsize{1}\\def\\@typesizes{\\or{6}{7}\\or{7}{8}\\or{8}{10}\\or{9}{11}\\or{10}{12}\\or{\\@xipt}{13}\\or{\\@xiipt}{14}\\or{\\@xivpt}{17}\\or{\\@xviipt}{20}\\or{\\@xxpt}{24}\\or{\\@xxvpt}{30}}");
24  DeclareOption!("12pt", "\\def\\@mainsize{12}\\def\\@ptsize{2}\\def\\@typesizes{\\or{7}{8}\\or{8}{10}\\or{9}{11}\\or{10}{12}\\or{\\@xipt}{13}\\or{\\@xiipt}{14}\\or{\\@xivpt}{17}\\or{\\@xviipt}{20}\\or{\\@xxpt}{24}\\or{\\@xxvpt}{30}\\or{\\@xxvpt}{30}}");
25  Ok(())
26}

491/// amsart's author storage (amscls/amsart.cls:460-477, amsbook.cls the same): `\authors`,
492/// `\shortauthors` and `\addresses` are parameterless macros, empty until `\author[#1]{#2}` adds
493/// its names, joined by `\and`; a document's `\maketitle` or running head prints them. Perl's
494/// ams_support reads an argument (ams_support.sty.ltxml:82-84), gobbling the next token: "Written
495/// by \authors\ (…)" lost its text, and 2605.03453's `\authors` inside a tabular swallowed its
496/// `\end` (KNOWN_PERL_ERRORS #366). amsart's `\maketitle` keeps `\and` (amsart.cls:599-621), which
497/// the kernel title code clears as article's does, so the class empties that step
498/// (`\lx@maketitle@clear@and`): a document's own `\renewcommand\and` survives the title too.
499/// `\addresses` stays empty: LaTeXML builds the addresses from `\address`. The one-argument setters
500/// stay for the classes that call them (aas_support gobbles 0709.4236's `\shortauthors{…&…}`, an
501/// aastex paper); here a setter-style call typesets its text, as in pdflatex. Not modelled: amsart's
502/// `\maketitle` rewrites `\shortauthors` for the running head (`\andify`, or the short title when
503/// empty, amsart.cls:604-606).
504pub fn amsart_author_storage() -> Result<()> {
505  RequirePackage!("amsgen");
506  Let!("\\authors", "\\@empty");
507  Let!("\\shortauthors", "\\@empty");
508  Let!("\\addresses", "\\@empty");
509  RawTeX!(
510    r"\def\lx@ams@addto@authors#1#2{%
511  \ifx\@empty\authors\gdef\authors{#2}\else\g@addto@macro\authors{\and#2}\fi
512  \@ifnotempty{#1}{\ifx\@empty\shortauthors\gdef\shortauthors{#1}\else\g@addto@macro\shortauthors{\and#1}\fi}}
513\let\lx@maketitle@clear@and\relax"
514  );
515  // amsart's list joiners over that storage (amsart.cls:580-598, :803-807; its `\newcommand`s as
516  // `\long\def`, the class having no earlier definition to guard): a derived
517  // class prints `\authors` through them (resphilosophica.cls:323 `\author@andify\authors`:
518  // "A, B, and C"), where ams_support's inert `\author@andify` left the names run together.
519  RawTeX!(
520    r"\long\def\xandlist#1#2#3#4{\@andlista{{#1}{#2}{#3}}#4\and\and}
521\def\@andlista#1#2\and#3\and{\@andlistc{#2}\@ifnotempty{#3}{%
522  \@andlistb#1{#3}}}
523\def\@andlistb#1#2#3#4#5\and{%
524  \@ifempty{#5}{%
525    \@andlistc{#2#4}%
526  }{%
527    \@andlistc{#1#4}\@andlistb{#1}{#3}{#3}{#5}%
528  }}
529\let\@andlistc\@iden
530\long\def\nxandlist#1#2#3#4{%
531  \def\@andlistc##1{\toks@\@xp{\the\toks@##1}}%
532  \toks@{\toks@\@emptytoks \@andlista{{#1}{#2}{#3}}}%
533  \the\@xp\toks@#4\and\and
534  \edef#4{\the\toks@}%
535  \let\@andlistc\@iden}
536\def\@@and{and}
537\long\def\andify{%
538  \nxandlist{\unskip, }{\unskip{} \@@and~}{\unskip, \@@and~}}
539\def\author@andify{%
540  \nxandlist {\unskip ,\penalty-1 \space\ignorespaces}%
541    {\unskip {} \@@and~}%
542    {\unskip ,\penalty-2 \space \@@and~}%
543}"
544  );
545  Ok(())
546}

756/// amsart.cls:620-629 `\@cleartopmattertags`: once its `\maketitle` has run, amsart lets `\keywords`, `\subjclass`,
757/// `\dedicatory` (and the other topmatter commands) be `\relax`, so pdflatex typesets a later `\keywords{…}` as the
758/// text it is and never reads it as an argument. The binding goes on reading it, for its frontmatter markup (ruling
759/// 2026-10-25), except where that short argument would run away — a `\par` (tex.web §392 `par_token`) or the end of the
760/// look-ahead's input before its closing brace, which pdflatex never meets: there the call is pdflatex's `\relax`, and the
761/// group is read as text (2204.07665: `\keywords{\small{ … }` a brace short after a first `\maketitle`; "Paragraph
762/// ended before `\keywords` was complete", the two labels inside it lost). After the title our cleanup has let
763/// `\maketitle` be `\relax` too, as amsart's does. Guard `perfect_kernel_batch80a::amsart_post_title_keywords_one_brace_short`.
764fn post_title_topmatter(setter: &str) -> Result<Tokens> {
765  let title_done = x_equals(&T_CS!("\\maketitle"), &T_CS!("\\relax"));
766  if title_done && short_argument_runs_away() {
767    return Ok(Tokens!());
768  }
769  Ok(Tokens!(T_CS!(setter)))
770}

772/// Whether the argument ahead — an optional `[…]`, then a braced group — meets a `\par` before its group closes, or
773/// is still open where the look-ahead ends (a bounded peek: nothing is read).
774fn short_argument_runs_away() -> bool {
775  let mut tokens = peek_upcoming_tokens(120, 48_000, 12_000)
776    .filter(|t| t.get_catcode() != Catcode::SPACE)
777    .peekable();
778  let par = T_CS!("\\par");
779  if tokens.peek() == Some(&T_OTHER!("[")) {
780    // the optional argument, to its `]` outside braces
781    let mut depth = 0usize;
782    loop {
783      match tokens.next() {
784        None => return false,
785        Some(t) if t == par => return false,
786        Some(t) if t.get_catcode() == Catcode::BEGIN => depth += 1,
787        Some(t) if t.get_catcode() == Catcode::END => depth = depth.saturating_sub(1),
788        Some(t) if depth == 0 && t == T_OTHER!("]") => break,
789        Some(_) => {},
790      }
791    }
792  }
793  if !tokens
794    .next()
795    .is_some_and(|t| t.get_catcode() == Catcode::BEGIN)
796  {
797    return false;
798  }
799  let mut depth = 1usize;
800  for t in tokens {
801    if t == par {
802      return true;
803    }
804    match t.get_catcode() {
805      Catcode::BEGIN => depth += 1,
806      Catcode::END => {
807        depth -= 1;
808        if depth == 0 {
809          return false;
810        }
811      },
812      _ => {},
813    }
814  }
815  true
816}
