3/// Whether the float in effect is an aastex `{longrotatetable}` itself, not a float nested in it: its frame
4/// (`lx@lrt@frame`, set at its start by aas_support_sty.rs) is the current float's (`lx@float@frame`, each float's
5/// own, `begin_float`).
6pub(crate) fn in_longrotatetable_body() -> bool {
7  match (lookup_value("lx@lrt@frame"), lookup_value("lx@float@frame")) {
8    (Some(lrt), Some(float)) => {
9      !matches!(lrt, Stored::None) && lrt.to_string() == float.to_string()
10    },
11    _ => false,
12  }
13}

15/// What a table-note mark shows, the key that pairs a `\tablenotetext` with its `\tablenotemark`s (latex_constructs
16/// `relocate_footnote` matches their `mark`s, ignoring spaces) and the mark the note prints: the characters and
17/// symbols as set, in text or math, a command's shown arguments and a formula's body read for theirs (`$\dagger$`,
18/// `^\dagger` and `\dag` are "†", `\rm{b}` and `\textrm{b}` are "b", `\hyperlink{tn5}{e}` is "e" and
19/// `\raisebox{1pt}{g}` "g": a value, a key-value list or a semiverbatim name sets no character), its spaces collapsed
20/// (`c, d` prints "c, d", as TeX sets it, and pairs with `c,d`). The one function serves both commands, so a mark
21/// pairs by what it shows whatever its source. Math's `*` and `\ast` are TeX's one asterisk (plain.tex
22/// `\mathcode`\*="2203`, `\mathchardef\ast="2203`), set here as U+2217: an asterisk, as a text `*` is.
23fn shown_mark(digested: &Digested) -> String {
24  /// Parameter types whose argument is a value, a name or a key-value list, which sets no character.
25  const UNSHOWN: &[&str] = &[
26    "Number",
27    "NumExpr",
28    "Dimension",
29    "DimExpr",
30    "Glue",
31    "GlueExpr",
32    "MuDimension",
33    "MuGlue",
34    "MuExpr",
35    "Length",
36    "Float",
37    "TempboxaDimension",
38    "GraphixDimension",
39    "GraphixDimensions",
40    "SetlengthDimension",
41    "SetlengthGlue",
42    "Keyword",
43    "CSName",
44    "CSNameQuiet",
45    "Token",
46    "XToken",
47    "DefToken",
48    "Undigested",
49    "OptionalUndigested",
50    "TeXFileName",
51    "DirectoryList",
52    "PackageOptions",
53  ];
54  fn shows(parameter: &Parameter) -> bool {
55    let unshown = parameter.semiverbatim.is_some()
56      || with(parameter.name, |name| {
57        UNSHOWN.contains(&name) || name.contains("KeyVals")
58      });
59    !unshown
60      && parameter
61        .inner
62        .as_ref()
63        .is_none_or(|inner| inner.get_parameters().into_iter().any(shows))
64  }
65  fn shown(digested: &Digested, out: &mut String) {
66    match digested.data() {
67      DigestedData::List(list) => list.borrow().boxes.iter().for_each(|item| shown(item, out)),
68      DigestedData::RegisterValue(_) | DigestedData::KeyVals(_) => {},
69      DigestedData::Whatsit(whatsit) => {
70        let whatsit = whatsit.borrow();
71        let body = whatsit.get_body().ok().flatten();
72        let args = whatsit.get_args();
73        if args.iter().all(Option::is_none) && body.is_none() {
74          out.push_str(&digested.to_attribute());
75        } else {
76          // One argument per valued parameter (`Parameters::read_arguments_and_digest`); without the parameters
77          // every argument is read.
78          let definition = whatsit.get_definition();
79          let parameters: Vec<&Parameter> = definition
80            .get_parameters()
81            .map(|parameters| {
82              parameters
83                .get_parameters()
84                .into_iter()
85                .filter(|p| !p.novalue)
86                .collect()
87            })
88            .unwrap_or_default();
89          let aligned = parameters.len() == args.len();
90          for (index, arg) in args.iter().enumerate() {
91            if let Some(arg) = arg
92              && (!aligned || shows(parameters[index]))
93            {
94              shown(arg, out);
95            }
96          }
97          if let Some(body) = body {
98            shown(&body, out);
99          }
100        }
101      },
102      _ => out.push_str(&digested.to_attribute()),
103    }
104  }
105  let mut text = String::new();
106  shown(digested, &mut text);
107  text
108    .split_whitespace()
109    .collect::<Vec<_>>()
110    .join(" ")
111    .chars()
112    .map(|c| if c == '\u{2217}' { '*' } else { c })
113    .collect()
114}
