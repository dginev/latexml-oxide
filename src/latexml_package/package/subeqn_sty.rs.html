1//! subeqn.sty — sub-equation numbering
2//! Perl: subeqn.sty.ltxml
3use crate::prelude::*;
4
5#[rustfmt::skip]
6LoadDefinitions!({
7  // Perl subeqn.sty.ltxml L21-22 locks both `\subequations` and
8  // `\endsubequations`. arXiv papers commonly load subeqn.sty alongside
9  // amsmath (which also defines these names) — the lock keeps our
10  // \lx@equationgroup@subnumbering@* trampolines from being shadowed.
11  DefMacro!("\\subequations",    "\\lx@equationgroup@subnumbering@begin",
12    locked => true);
13  DefMacro!("\\endsubequations", "\\lx@equationgroup@subnumbering@end",
14    locked => true);
15  // subeqn.sty:51-57 `subeqnarray` = `\subequations` around `eqnarray`, with
16  // an optional leading `\label{main}` taken by `\@lab@subeqnarray`. The
17  // eqnarray half is the subeqnarray binding's (subeqnarray_sty.rs) — Perl
18  // subeqn.sty.ltxml omits the environment (subeqn-sample; pdflatex clean).
19  // Guard: `perfect_kernel_batch56::subeqn_subeqnarray_environment`.
20  DefMacro!("\\subeqnarray",
21    "\\lx@equationgroup@subnumbering@begin\\@ifnextchar\\label{\\lx@subeqn@lab}{\\lx@subeqn@eqnarray}",
22    locked => true);
23  DefMacro!("\\lx@subeqn@lab{}{}", "#1{#2}\\lx@subeqn@eqnarray");
24  DefMacro!("\\lx@subeqn@eqnarray",
25    "\\@eqnarray@bindings\\@@eqnarray\
26     \\@equationgroup@numbering{numbered=1,preset=1,deferretract=1,grouped=1,aligned=1}\
27     \\lx@begin@alignment");
28  DefMacro!("\\endsubeqnarray",
29    "\\cr\\lx@end@alignment\\end@eqnarray\\lx@equationgroup@subnumbering@end",
30    locked => true);
31});
