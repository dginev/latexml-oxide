334fn longtable_bindings(template: Template) -> Result<()> {
335  let mut props = SymHashMap::default();
336  props.insert("guess_headers", Stored::Bool(false));
337  // longtable.sty:133 `\let\\\LT@tabularcr`, which opens with no brace (`\noalign\\` is "Missing { inserted")
338  props.insert("row_end", Stored::Token(T_CS!("\\LT@tabularcr")));
339  tabular_bindings(template, props, HashMap::default())?;
340  let_i(
341    &T_CS!("\\endfirsthead"),
342    &T_CS!("\\lx@longtable@endfirsthead"),
343    None,
344  );
345  let_i(&T_CS!("\\endhead"), &T_CS!("\\lx@longtable@endhead"), None);
346  let_i(&T_CS!("\\endfoot"), &T_CS!("\\lx@longtable@endfoot"), None);
347  let_i(
348    &T_CS!("\\endlastfoot"),
349    &T_CS!("\\lx@longtable@endlastfoot"),
350    None,
351  );
352  let_i(&T_CS!("\\caption"), &T_CS!("\\lx@longtable@caption"), None);
353  let_i(&T_CS!("\\label"), &T_CS!("\\lx@longtable@label"), None);
354  // caption.sty:1142-1147 (`\caption@AtBeginLongtable`): inside a longtable `\captionsetup` is a `\noalign`, so one
355  // between rows does not open a row (2310.02059: `\captionsetup{…}` before the `\hline` of a head).
356  if lookup_definition(&T_CS!("\\captionsetup"))?.is_some() {
357    let_i(
358      &T_CS!("\\lx@longtable@saved@captionsetup"),
359      &T_CS!("\\captionsetup"),
360      None,
361    );
362    let_i(
363      &T_CS!("\\captionsetup"),
364      &T_CS!("\\lx@longtable@captionsetup"),
365      None,
366    );
367  }
368  let_i(&T_CS!("\\kill"), &T_CS!("\\lx@longtable@kill"), None);
369
370  assign_value("LONGTABLE_LABEL", Stored::None, Some(Scope::Global));
371  assign_value("LONGTABLE_LABEL_TYPE", Stored::None, Some(Scope::Global));
372  assign_value("LONGTABLE_CAPTIONS", Stored::None, Some(Scope::Global));
373  assign_value("LONGTABLE_HEAD_CAPTIONS", Stored::None, Some(Scope::Global));
374  assign_value("LONGTABLE_FOOT_CAPTIONS", Stored::None, Some(Scope::Global));
375  assign_value("LONGTABLE_HEAD", Stored::None, Some(Scope::Global));
376  assign_value("LONGTABLE_FOOT", Stored::None, Some(Scope::Global));
377
378  // properties happen too late!!! - do RefStepCounter now. The kernel's plain step
379  // (`\@kernel@refstepcounter`, longtable.sty:115): caption types a longtable and prepares its
380  // step only at its caption (`\lx@longtable@caption@`).
381  let props = ref_step_counter("table", false)?;
382  assign_value(
383    "LONGTABLE_PROPERTIES",
384    Stored::HashStored(props),
385    Some(Scope::Global),
386  );
387
388  Ok(())
389}

391/// Ask the current alignment for its row's disposition at the row's end.
392fn dispose_current_row(disposition: RowDisposition) {
393  if let Some(alignment) = lookup_alignment()
394    && let Some(cell) = alignment.alignment_cell()
395  {
396    cell.borrow_mut().set_row_disposition(disposition);
397  }
398}

400/// A caption's row goes with the caption (#534). A note in it, a `\footnote` on the caption (`\caption{…}\footnote{…}\\`),
401/// is set after the caption's text, its mark where longtable's caption cell has it (longtable.sty:476 `\LT@makecaption`;
402/// 2201.11330, lost with the row). Any other content goes with the row, counted by a Warning: text before the caption
403/// (`\begin{longtable*}{lc}[t]`'s stray `[t]`, 1009.4206) or after it (1903.09068's "."; a nested tabular, RED
404/// longtable_caption_then_a_nested_tabular_red, GA4C). The caption's cell is a `\multicolumn` (`\LT@mcol`), so the
405/// column template's own text (`r@{.}`'s ".", 1111.5166) is not printed and not counted.
406fn caption_row_notes(row: &Row) {
407  // (the notes the cells hold, inside a `p` column's box too; the text of what else shows)
408  fn shown(d: &Digested, out: &mut Vec<Digested>, text: &mut String) {
409    match d.data() {
410      DigestedData::List(list) => list.borrow().boxes.iter().for_each(|b| shown(b, out, text)),
411      DigestedData::Whatsit(w) => {
412        let w_ref = w.borrow();
413        let cs_name = w_ref.get_definition().get_cs_name().to_string();
414        if cs_name.starts_with("\\lx@note") {
415          out.push(d.clone());
416        } else if cs_name == "\\lx@tabular@p@"
417          && let Some(Some(body)) = w_ref.get_args().get(2)
418        {
419          shown(body, out, text);
420        } else if typesets_content(d) {
421          text.push_str(&d.to_string());
422        }
423      },
424      _ if typesets_content(d) => text.push_str(&d.to_string()),
425      _ => {},
426    }
427  }
428  let mut notes = Vec::new();
429  let mut lost = false;
430  for column in row.get_columns() {
431    if let Some(boxes) = &column.boxes {
432      let mut text = String::new();
433      shown(boxes, &mut notes, &mut text);
434      let template: String = [&column.before, &column.after]
435        .into_iter()
436        .flatten()
437        .flat_map(|tokens| tokens.unlist_ref())
438        .filter(|t| matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER))
439        .map(|t| t.with_str(str::to_owned))
440        .collect();
441      let text: String = text.split_whitespace().collect();
442      lost |= !text.is_empty() && text != template;
443    }
444  }
445  if lost {
446    Warn!(
447      "unexpected",
448      "\\caption",
449      "a longtable \\caption's row holds content besides the caption; it is dropped with the row"
450    );
451  }
452  if !notes.is_empty()
453    && let Some(Stored::VecDigested(mut captions)) = lookup_value("LONGTABLE_CAPTIONS")
454    && let Some(caption) = captions.pop()
455  {
456    let mut boxes = vec![caption];
457    boxes.extend(notes);
458    captions.push(List::new(boxes).into());
459    assign_value(
460      "LONGTABLE_CAPTIONS",
461      Stored::VecDigested(captions),
462      Some(Scope::Global),
463    );
464  }
465}
