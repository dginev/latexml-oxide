1//! glossaries.sty — strict translation of Perl `glossaries.sty.ltxml` (126L).
2//!
3//! Perl raw-loads real TL glossaries.sty (~8700 lines) via
4//! `InputDefinitions(noltxml=>1)`, then layers surgical overrides:
5//!   - wrap `\@gls@link` output in `<ltx:glossaryref>`
6//!   - hook `\@newglossaryentryposthook` to emit `<ltx:glossarydefinition>`
7//!   - redefine `\printglossary` to emit `<ltx:glossary>`
8//!
9//! This Rust file mirrors the Perl 1:1.

18/// The State key of the glossary entries referenced so far (`\gls`, `\glsadd`; global).
19const GLO_REFERENCED: &str = "lx@glossaries@referenced";
20/// The State key set once a glossary is printed (`\printglossary`; global).
21const GLO_PRINTED: &str = "lx@glossaries@printed";
22
23/// The label an entry's held definition diagnostics answer to (logger.rs `defer_to_label`).
24fn glossary_entry_label(key: &str) -> String { s!("glossary:{key}") }
25
26/// An entry referenced (`\gls`, `\glsadd`): a glossary printed or yet to be prints its definition, whose held
27/// diagnostics are then raised (`\lx@glossaries@newentry`).
28fn glossary_entry_referenced(key: &str) -> Result<()> {
29  if lookup_bool(GLO_PRINTED) {
30    util::logger::note_label_use(&glossary_entry_label(key), "glossary");
31  } else {
32    push_value(GLO_REFERENCED, Stored::from(key.to_string()))?;
33  }
34  Ok(())
35}

352        && let DigestedData::KeyVals(ref kvs) = *kv_digested.data() {
353          // Sort by role (Perl: `sort keys %$hash`).
354          let mut pairs: Vec<(String, ArgWrap)> = kvs.get_pairs()
355            .map(|(k, v)| (k.clone(), v.clone())).collect();
356          pairs.sort_by(|a, b| a.0.cmp(&b.0));
357          for (role, val) in pairs {
358            let val_str = val.to_string();

360            // Insert <ltx:glossaryphrase key=key role=role>val</ltx:glossaryphrase>.
361            // Perl inserts the DIGESTED value (glossaries.sty.ltxml:96,
362            // `insertElement('ltx:glossaryphrase', $value, …)` after
363            // KeyVals::beDigested), so a `$\alpha$` name is math and an
364            // `\emph` description is emphasis; absorbing `val_str` wrote the
365            // TeX source (`\emph{in}`) as text. The string is the fallback.
366            let digested = kvs.get_value_digested(&role).cloned();
367            // A key outside the fixed fields is a recorded `\\glsaddkey` display key (`\\lx@glossaries@userkv`):
368            // classed, so the glossary list prints it and no other phrase role.
369            let mut attributes = string_map!("key" => key.clone(), "role" => role.clone());
370            if !GLO_FIELDS.iter().any(|(_, field)| *field == role) {
371              attributes.insert("class".to_string(), "ltx_glossary_userkey".to_string());
372            }
373            document.open_element("ltx:glossaryphrase", Some(attributes), None)?;
374            // The `sort` and `parent` values are data, compared as strings (makeindex's sort key, the parent's
375            // label): their source text, not set in the font (OT1 prints the sanitized `\\` and `"` as `“` `”`). Repro
376            // index-bib/glossary_sort_key_is_its_string.
377            match digested {
378              Some(value) if matches!(role.as_str(), "sort" | "parent") => {
379                document.absorb_string(&value.untex()?, &NO_PROPERTIES)?;
380              },
381              Some(value) => document.absorb(&value, None)?,
382              None => {
383                document.absorb_string(&val_str, &NO_PROPERTIES)?;
384              },
385            }
386            document.close_element("ltx:glossaryphrase")?;
387          }
388        }
389      document.close_element("ltx:glossarydefinition")?;
390    },
