7  // Perl: import.sty.ltxml — `AssignValue(SEARCHPATHS => …)`, local-by-default,
8  // reverted by the `{…}` group each `\import`/`\subimport` wraps its body in.
9  // SEARCHPATHS is now a group-scoped value in Rust too (state::get/set with
10  // `Scope::Local`), so no explicit save/restore stack is needed — the group
11  // handles the revert, faithful to Perl. Witnesses: arXiv:2604.09744,
12  // 2603.04457 (sibling `\subimport{Chapter/}{File}` calls — the second must
13  // NOT concat Chapter/ onto the first call's Chapter/).
14
15  // Mark the `{…}` each `\import`/`\subimport` opens as a LaTeXML subfile scope.
16  // OXIDIZED_DESIGN #65 (#311): the group is a LaTeXML artifact — the real
17  // import.sty never groups the input (`\@import` restores `\input@path`/
18  // `\Ginput@path` by plain `\def` AFTER the `\input`, at the caller's level;
19  // "input files must have balanced grouping", L42). Naming the region lets
20  // `require_package` give a package loaded in there the outermost-level lifetime
21  // real LaTeX would (content.rs `is_scope_active(subfile_scope_here())`). The
22  // marker is `Scope::Local`, so the region ends with the group. (This primitive
23  // formerly also saved SEARCHPATHS; that is now handled by group-scoping the
24  // value itself — see `\lx@set@path`/`\lx@append@path`.) Mirror of
25  // `standalone_sty.rs`'s inline `activate_scope(subfile_scope_here())`.
26  DefPrimitive!("\\lx@activate@subfile@scope", {
27    activate_scope(subfile_scope_here());
28  });
29
30  // Perl import.sty.ltxml L20-29: \lx@set@path OptionalMatch:* {}
31  //   path = ToString(Expand(#2)); if relative, resolve vs SOURCEDIRECTORY.
32  //   If * → replace SEARCHPATHS with [canonical(path)]
33  //   else → prepend canonical(path) to existing SEARCHPATHS.
34  DefPrimitive!("\\lx@set@path OptionalMatch:* {}", sub[(star, path_tks)] {
35    let raw = Expand!(path_tks).to_string();
36    let mut path = raw.trim().to_string();
37    if path.is_empty() { return Ok(Vec::new()); }
38    if !pathname::is_absolute(&path) {
39      let source_dir = lookup_string("SOURCEDIRECTORY");
40      if !source_dir.is_empty() {
41        path = pathname::concat(&source_dir, &path);
42      }
43    }
44    let canonical = pathname::canonical(&path);
45    // LOCAL: reverted by the enclosing `\import`/`\subimport` `{…}` group,
46    // matching Perl's default-local `AssignValue(SEARCHPATHS…)`.
47    if star.is_some() {
48      set_search_paths_local(vec![canonical]);
49    } else {
50      let mut new_paths = vec![canonical];
51      new_paths.extend(get_search_paths());
52      set_search_paths_local(new_paths);
53    }
54  });
55
56  // Perl import.sty.ltxml L31-42: \lx@append@path OptionalMatch:* {}
57  //   If SEARCHPATHS has entries, concat the first with path:
58  //   new_lead = concat(lead_path, path); star → [new_lead], else → [new_lead, ...rest].
59  //   If SEARCHPATHS is empty, this is a no-op (matches Perl's early-return).
60  //   DIVERGENCE (OXIDIZED_DESIGN #137): an absolute `path` is used verbatim
61  //   rather than concatenated, to match real LaTeX — see the body.
62  DefPrimitive!("\\lx@append@path OptionalMatch:* {}", sub[(star, path_tks)] {
63    let raw = Expand!(path_tks).to_string();
64    let path = raw.trim().to_string();
65    if path.is_empty() { return Ok(Vec::new()); }
66    let mut paths = get_search_paths();
67    if paths.is_empty() { return Ok(Vec::new()); }
68    let lead = paths.remove(0);
69    // OXIDIZED_DESIGN #137 — surpass Perl to match real LaTeX (pdflatex,
70    // verified): an ABSOLUTE directory arg is used verbatim, not concatenated
71    // onto the lead search path (which yields an unresolvable `<lead>//abs/…`).
72    // Perl's \lx@append@path (import.sty.ltxml L31-42) ALWAYS concats, so both
73    // engines fail `\subimport*{/abs/}{file}` where pdflatex succeeds — issue
74    // #697. \lx@set@path already special-cases absolute; mirror it here.
75    // Relative args are unchanged (the common `\subimport*{sub/}{file}` case).
76    let new_lead = if pathname::is_absolute(&path) {
77      pathname::canonical(&path)
78    } else {
79      pathname::concat(&lead, &path)
80    };
81    // LOCAL (see `\lx@set@path`): the `{…}` group reverts it.
82    if star.is_some() {
83      set_search_paths_local(vec![new_lead]);
84    } else {
85      let mut new_paths = vec![new_lead];
86      new_paths.extend(paths);
87      set_search_paths_local(new_paths);
88    }
89  });
90
91  // Each `\import`/`\subimport` wraps its body in a `{…}` group (with
92  // `\lx@activate@subfile@scope` naming the subfile scope). The path change is LOCAL, so the
93  // group reverts it at `}` — each sibling starts from the BASE search paths,
94  // exactly as Perl's default-local `AssignValue(SEARCHPATHS…)` does. Without
95  // group-local paths, two sibling `\subimport{Chapter/}{Abstract}` +
96  // `\subimport{Chapter/}{Poster}` would concat Chapter/ onto the first call's
97  // still-mutated lead → "Chapter/Chapter/Poster". Witnesses 2604.09744, 2603.04457.
98  //
99  // KNOWN_PERL_ERRORS #56: `\includefrom`/`\subincludefrom` take TWO arguments
100  // after the star — `\includefrom{dir/}{file}` — but Perl's prototypes declare
101  // only one while their bodies reference `#3` (import.sty.ltxml L45/L47). The
102  // undeclared `#3` expands to nothing, so `\include{}` includes nothing and the
103  // file is dropped in silence: no error, no warning, no content. Real
104  // `import.sty` takes both for all four (L57/L58 route `\includefrom` /
105  // `\subincludefrom` through the same `\@doimport` as `\import`/`\subimport`;
106  // `\@sub@import` L65 consumes the directory as #3 and `\@import` L82 the file
107  // name as #7), so the arity below is the real package's, not Perl's typo.
108  DefMacro!("\\import OptionalMatch:* {}{}",
109    "{\\lx@activate@subfile@scope\\lx@set@path #1{#2} \\input{#3}}");
110  DefMacro!("\\includefrom OptionalMatch:* {}{}",
111    "{\\lx@activate@subfile@scope\\lx@set@path #1{#2} \\include{#3}}");
112  DefMacro!("\\subimport OptionalMatch:* {}{}",
113    "{\\lx@activate@subfile@scope\\lx@append@path #1{#2} \\input{#3}}");
114  DefMacro!("\\subincludefrom OptionalMatch:* {}{}",
115    "{\\lx@activate@subfile@scope\\lx@append@path #1{#2} \\include{#3}}");
116  Let!("\\inputfrom", "\\import");
117  Let!("\\subinputfrom", "\\subimport");
118});
