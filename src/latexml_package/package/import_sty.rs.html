5/// import.sty:147-151 `\import@path@fix`: a non-empty directory ends in exactly one `/` (`\@ensure@one@trailing@slash`);
6/// an empty one stays empty.
7fn import_path_fix(path: &str) -> String {
8  let path = path.trim();
9  if path.is_empty() {
10    String::new()
11  } else {
12    s!("{}/", path.trim_end_matches('/'))
13  }
14}

16/// The current `\import@path` (import.sty:142, empty outside any import).
17fn import_path() -> String { lookup_string("lx@import@path") }

19/// import.sty:82-85 `\@import`: `\import@path` becomes `path`, and `\input@path` gets it in front of the enclosing
20/// entries (TeX resolves each entry against the working directory, here the source directory). The `*` forms are
21/// the same (import.sty:60-62 "Consume now-meaningless `*`"); Perl's `*` searched only the import's directory, which
22/// loses the source directory the joined name is relative to (`\subimport*{nested/}{outer.tex}` reads
23/// `nested/outer.tex`).
24fn enter_import_path(path: String) {
25  assign_value("lx@import@path", Stored::String(pin(&path)), None);
26  let mut new_paths = Vec::new();
27  if !path.is_empty() {
28    let dir = if pathname::is_absolute(&path) {
29      path.clone()
30    } else {
31      let source_dir = lookup_string("SOURCEDIRECTORY");
32      if source_dir.is_empty() {
33        path.clone()
34      } else {
35        pathname::concat(&source_dir, &path)
36      }
37    };
38    new_paths.push(pathname::canonical(&dir));
39  }
40  new_paths.extend(get_search_paths());
41  set_search_paths_local(new_paths);
42}

47  // The paths are saved before and restored after each import's input
48  // (`\lx@import@save`/`\lx@import@restore` below), so sibling
49  // `\subimport{Chapter/}{File}` calls each start from the base paths (the
50  // second must NOT concat Chapter/ onto the first call's Chapter/). Witnesses:
51  // arXiv:2604.09744, 2603.04457.
52  //
53  // The import's file is found as import.sty finds it (:82-88): `\import@path` is the directory, reset by `\import`
54  // (:49) and extended by `\subimport` (:55, `\import@path#3`), and the file read is `\import@path #7`, with
55  // `\input@path` = `{\import@path}` + the enclosing import's entries. A nested `\import{intro}{intro}` inside
56  // `\import{chapters/}{chapters}` reads `intro/intro`, found under the enclosing entry as
57  // `chapters/intro/intro.tex`, and a `\subimport{}{statements/x.tex}` there reads `intro/statements/x.tex`
58  // (2408.09583: its chapters, then its statement files). Perl's `\input{#3}` against a search path made of the
59  // first entry and the new directory (import.sty.ltxml:20-47) found neither (KPE #1171).

62  // directory unless absolute.
63  DefPrimitive!("\\lx@set@path OptionalMatch:* {}", sub[(_star, path_tks)] {
64    let path = import_path_fix(&Expand!(path_tks).to_string());
65    enter_import_path(path);
66  });

68  // Perl import.sty.ltxml L31-42: \lx@append@path OptionalMatch:* {} — `\subimport`'s directory, appended to the
69  // current `\import@path`.
70  //   DIVERGENCE (OXIDIZED_DESIGN #137): an absolute `path` is used verbatim
71  //   rather than concatenated, to match real LaTeX — see the body.
72  DefPrimitive!("\\lx@append@path OptionalMatch:* {}", sub[(_star, path_tks)] {
73    let raw = Expand!(path_tks).to_string();
74    let raw = raw.trim();
75    // OXIDIZED_DESIGN #137 — surpass Perl to match real LaTeX (pdflatex,
76    // verified): an ABSOLUTE directory arg is used verbatim, not concatenated
77    // onto the current import path (which yields an unresolvable `<lead>//abs/…`).
78    // Perl's \lx@append@path (import.sty.ltxml L31-42) ALWAYS concats, so both
79    // engines fail `\subimport*{/abs/}{file}` where pdflatex succeeds — issue
80    // #697. \lx@set@path already special-cases absolute; mirror it here.
81    let path = if pathname::is_absolute(raw) {
82      import_path_fix(raw)
83    } else {
84      import_path_fix(&s!("{}{raw}", import_path()))
85    };
86    enter_import_path(path);
87  });

90  // bare name does, the bare name, as import.sty before v6 read it (`#1{#7}` with `\import@path` first on
91  // `\input@path`; TeX tries the working directory first): an archive whose `\import{pictures/}{cantor.pdf_tex}`
92  // ships `cantor.pdf_tex` at its top (1409.7023) or whose `\subimport{SM_quarkgluon/}{quarkgluon_main}` ships the
93  // chapter beside the main file (1605.04692) compiled so on arXiv; TL 2025's v6.2 stops ("File not found").
94  DefMacro!("\\lx@import@input{}{}", sub[(command, file)] {
95    let mut toks = command.unlist();
96    toks.push(T_BEGIN!());
97    let path = import_path();
98    let name = Expand!(file.clone()).to_string();
99    let name = name.trim();

101      && (find_file(&s!("{path}{name}"), None).is_some() || find_file(name, None).is_none())
102    {
103      toks.extend(ExplodeText!(path));
104    }
105    toks.extend(file.unlist());
106    toks.push(T_END!());
107    Ok(Tokens::new(toks))

110  // No group around the input (import.sty:65-92): `\@sub@import` closes its own
111  // group before `\@import` runs the `\input`/`\include` at the caller's level,
112  // then restores `\input@path`/`\Ginput@path` by plain `\def`. Perl's binding
113  // (import.sty.ltxml L44-47) wraps the input in `{…}`, so every definition the
114  // imported file makes — its `\newcommand`s, and the packages it loads, whose
115  // loaded-flags are global — was popped at the `}`: `\subimport{}{macros}` left
116  // the paper's macros undefined, and a package loaded there (hyperref → etoolbox)
117  // was "already loaded" but gone when biblatex asked for `\newbool` (arXiv
118  // 2605.20598, `Fatal:TooManyErrors`). The search paths and `\import@path` are
119  // saved before and restored after the input instead (import.sty:92-95), so each
120  // sibling `\subimport{Chapter/}{…}` still starts from the base paths (witnesses
121  // 2604.09744, 2603.04457).
122  // OXIDIZED_DESIGN #280.
123  //
124  // KNOWN_PERL_ERRORS #56: `\includefrom`/`\subincludefrom` take TWO arguments
125  // after the star — `\includefrom{dir/}{file}` — but Perl's prototypes declare
126  // only one while their bodies reference `#3` (import.sty.ltxml L45/L47). The
127  // undeclared `#3` expands to nothing, so `\include{}` includes nothing and the
128  // file is dropped in silence: no error, no warning, no content. Real
129  // `import.sty` takes both for all four (L57/L58 route `\includefrom` /
130  // `\subincludefrom` through the same `\@doimport` as `\import`/`\subimport`;
131  // `\@sub@import` L65 consumes the directory as #3 and `\@import` L82 the file
132  // name as #7), so the arity below is the real package's, not Perl's typo.
133  DefPrimitive!("\\lx@import@save", {
134    let saved = lookup_value("SEARCHPATHS").unwrap_or(Stored::None);
135    let mut stack = match lookup_value("lx@import@saved@paths") {
136      Some(Stored::VecDequeStored(stack)) => stack,
137      _ => VecDeque::new(),
138    };
139    stack.push_front(saved);
140    stack.push_front(Stored::String(pin(import_path())));
141    assign_value("lx@import@saved@paths", Stored::VecDequeStored(stack), Some(Scope::Global));
142  });
143  DefPrimitive!("\\lx@import@restore", {
144    if let Some(Stored::VecDequeStored(mut stack)) = lookup_value("lx@import@saved@paths")
145      && let Some(saved_import_path) = stack.pop_front()
146      && let Some(saved) = stack.pop_front()
147    {
148      assign_value("lx@import@saved@paths", Stored::VecDequeStored(stack), Some(Scope::Global));
149      assign_value("SEARCHPATHS", saved, Some(Scope::Local));
150      assign_value("lx@import@path", saved_import_path, None);
151    }
152  });
153  DefMacro!("\\import OptionalMatch:* {}{}",
154    "\\lx@import@save\\lx@set@path #1{#2} \\lx@import@input{\\input}{#3}\\lx@import@restore");
155  DefMacro!("\\includefrom OptionalMatch:* {}{}",
156    "\\lx@import@save\\lx@set@path #1{#2} \\lx@import@input{\\include}{#3}\\lx@import@restore");
157  DefMacro!("\\subimport OptionalMatch:* {}{}",
158    "\\lx@import@save\\lx@append@path #1{#2} \\lx@import@input{\\input}{#3}\\lx@import@restore");
159  DefMacro!("\\subincludefrom OptionalMatch:* {}{}",
160    "\\lx@import@save\\lx@append@path #1{#2} \\lx@import@input{\\include}{#3}\\lx@import@restore");
161  Let!("\\inputfrom", "\\import");
162  Let!("\\subinputfrom", "\\subimport");
