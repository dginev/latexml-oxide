3/// The current value of a token register, empty when the register is undefined or not a register.
4fn toks_register(cs: &str) -> Tokens {
5  match lookup_register_quiet(cs) {
6    Some(RegisterValue::Tokens(t)) => t,
7    _ => Tokens!(),
8  }
9}

11/// Whether `\th@<style>` is still the binding's own style (`\lx@theorem@newtheoremstyle`), not one a class or document
12/// put in its place: ifacconf.cls:603-624 `\gdef`s `\th@plain`/`\th@definition` (upright bodies, the head in its own
13/// `\theorem@headerfont`; 1911.07501, 1911.10645, 2211.14033, 2512.06733).
14fn is_binding_style(style: &str) -> bool {
15  let style_cs = T_CS!(s!("\\th@{style}"));
16  let own_cs = T_CS!(s!("\\lx@theorem@th@{style}"));
17  has_meaning(&own_cs) && x_equals(&style_cs, &own_cs)
18}

89  // theorem.sty:16-48 `\@xnthm`/`\@ynthm`/`\@othm`: a theorem's environment begins with
90  //   `\th@<style> \the\theorem@bodyfont \@thm{<counter>}{<name>}`
91  // with the style and the body font taken when `\newtheorem` runs. A body font may be code that
92  // takes the `\@thm{…}{…}` call as its delimited arguments (ccthm.sty:100-107
93  // `\def\theorempreamblenn\@thm#1#2{…\the\everytheorem\@thm{#1}{#2}}`), so the binding keeps that
94  // sequence: it is the theorem's body font (`define_new_theorem`'s per-theorem `bodyfont`, run at
95  // the body's start). A binding style (`\lx@theorem@newtheoremstyle`) sets the binding's style
96  // registers here and contributes its font; a style the document or class defines
97  // (`\def\th@mine{…}`, thcc.sty's `\th@cc`, ifacconf.cls's own `\gdef\th@plain`) runs itself at
98  // the body's start, as in theorem.sty.
99  DefPrimitive!("\\newtheorem OptionalMatch:* [] {}[]{}[]", sub[(flag, _style_opt, thmset, otherthmset, typ, reset)] {
100    let style = toks_register("\\theorem@style").to_string();
101    let style_cs = T_CS!(s!("\\th@{style}"));
102    // A document's style may set its body font in its `\@begintheorem` instead (arXivclass.cls:1808-1817
103    // `\def\@begintheorem##1##2{\normalfont\TheoremTextFont\item[…]}`; 2505.07298), which the binding's
104    // environment never calls: the binding's current style font stands before it.
105    // The sequence begins by making `\@thm` a no-op for the theorem's group (see below).
106    let mut body: Vec<Token> = vec![T_CS!("\\let"), T_CS!("\\@thm"), T_CS!("\\@gobbletwo")];
107    let binding_style = is_binding_style(&style);
108    if binding_style {
109      use_theorem_style(&style);
110    }
111    body.extend(toks_register("\\thm@bodyfont").unlist());
112    if !binding_style && has_meaning(&style_cs) {
113      body.push(style_cs);
114    }
115    body.extend(toks_register("\\theorem@bodyfont").unlist());
116    let otherthmset = otherthmset.filter(|t| !t.is_empty());
117    body.push(T_CS!("\\@thm"));
118    body.push(T_BEGIN!());
119    body.extend(otherthmset.clone().unwrap_or_else(|| thmset.clone()).unlist());
120    body.push(T_END!());
121    body.push(T_BEGIN!());
122    body.extend(typ.clone().unlist());
123    body.push(T_END!());
124    define_new_theorem(
125      flag.filter(|f| !f.is_empty()),
126      thmset,
127      otherthmset,
128      if typ.is_empty() { None } else { Some(typ) },
129      reset.filter(|t| !t.is_empty()),
130      Some(Tokens::new(body)),
131    )?;
132    // As the kernel's `\newtheorem`.
133    assign_register("\\thm@prework",
134      RegisterValue::Tokens(Tokens!()), None, vec![])?;
135    assign_register("\\thm@postwork",
136      RegisterValue::Tokens(Tokens!()), None, vec![])?;
137  });
138  // theorem.sty:114-119 `\@thm{<counter>}{<name>}` steps the counter and sets the head; the binding's
139  // theorem environment has done both by the time its body font runs, so the literal `\@thm{…}{…}`
140  // that ends the sequence above (the delimiter of ccthm.sty's `\theorempreamble\@thm#1#2`) is
141  // gobbled by the `\let\@thm\@gobbletwo` the sequence starts with. The no-op is local to the
142  // theorem's group: a raw `\@thm` elsewhere — a theorem.sty-based class's
143  // `\newenvironment{myclaim}{\th@plain\@thm{claim}{Claim}}{\@endtheorem}` — keeps the kernel's,
144  // with its head and counter step (theorem.sty:114's `\gdef\@thm` differs from it only in the skips).
145  // Guard `perfect_kernel_batch67p::theorem_raw_thm_keeps_heads`.

147  // \lx@theorem@newtheoremstyle{name}{bodyfont}{headstyle}{swap}
148  DefPrimitive!("\\lx@theorem@newtheoremstyle{}{}{}{}", sub[(name, bodyfont, headstyle, swap)] {
149    let name_str = name.to_string();
150    let swap_val = swap.eq_text("S");
151    save_theorem_style(&name_str, vec![
152      ("\\thm@bodyfont".into(), Stored::Tokens(bodyfont)),
153      ("\\thm@headstyling".into(), Stored::Tokens(headstyle)),
154      ("thm@swap".into(), Stored::Bool(swap_val)),
155    ]);
156    let name_for_closure = name_str.clone();
157    let style_cs = T_CS!(s!("\\th@{name_str}"));
158    DefMacro!(
159      style_cs,
160      None,
161      Some(ExpansionBody::Closure(Rc::new(move |_args| {
162        use_theorem_style(&name_for_closure);
163        Ok(Tokens!())
164      })))
165    );
166    let_i(&T_CS!(s!("\\lx@theorem@th@{name_str}")), &style_cs, None);
167  });
