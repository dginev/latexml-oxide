1use crate::prelude::*;

3/// The AMS environments empheq's `overload` option gives an optional `[<empheq options>]` (empheq.sty:961-1066, at
4/// `\AtBeginDocument`; `multline` under the default `newmultline`). `alignat` reads its column count first.
5const OVERLOADED: [&str; 12] = [
6  "equation",
7  "equation*",
8  "gather",
9  "gather*",
10  "align",
11  "align*",
12  "flalign",
13  "flalign*",
14  "multline",
15  "multline*",
16  "alignat",
17  "alignat*",
18];
19
20#[rustfmt::skip]
21LoadDefinitions!({
22  // empheq.sty:37-55: `overload` (and `overload2`, which adds `\[…\]`) gives the AMS environments empheq's options;
23  // the rest go to mathtools.
24  DeclareOption!("overload", { assign_value("empheq_overload", true, None); });
25  DeclareOption!("overload2", { assign_value("empheq_overload", true, None); });
26  DeclareOption!("oldmultline", None);
27  DeclareOption!("newmultline", None);
28  DeclareOption!("ntheorem", None);
29  DeclareOption!(None, {
30    Digest!("\\PassOptionsToPackage{\\CurrentOption}{mathtools}")?;
31  });
32  ProcessOptions!();
33  RequirePackage!("amsmath");
34  RequirePackage!("mathtools");
35  RequirePackage!("keyval");
36
37  DefMacro!("\\empheqset{}", None);
38
39  // The options of `{empheq}` and of an overloaded AMS environment (empheq.sty:785-838 `\EQ_options_name:`): `left`
40  // and `right` are math set before and after the display, which stays as written — the first and the last cell of
41  // the alignment receive them (`\begin{align}[left={F(x)=\empheqlbrace}]`, 1410.0933, 1906.02657, 2210.09120); the
42  // boxes are styling, not kept. Perl's binding (empheq.sty.ltxml) reads and drops them all, the left-hand side too.
43  RawTeX!(r"\def\lx@empheq@left{}\def\lx@empheq@right{}%
44    \define@key{lx@EmphEq}{left}{\def\lx@empheq@left{#1}}%
45    \define@key{lx@EmphEq}{right}{\def\lx@empheq@right{#1}}%
46    \define@key{lx@EmphEq}{box}{}\define@key{lx@EmphEq}{innerbox}{}%
47    \define@key{lx@EmphEq}{outerbox}{}\define@key{lx@EmphEq}{marginbox}{}%
48    \def\lx@empheq@setkeys#1{\def\lx@empheq@left{}\def\lx@empheq@right{}\setkeys{lx@EmphEq}{#1}}%
49    \def\lx@empheq@go#1#2#3{\csname lx@empheq@AmS@#2\endcsname#3#1}%
50    \def\lx@empheq@ov#1#2[#3]{\lx@empheq@setkeys{#3}\expandafter\lx@empheq@go\expandafter{\lx@empheq@left}{#1}{#2}}%
51    \def\lx@empheq@peek#1#2{\@ifnextchar[{\lx@empheq@ov{#1}{#2}}{\lx@empheq@ov{#1}{#2}[]}}%
52    \def\lx@empheq@peekarg#1#2{\lx@empheq@peek{#1}{{#2}}}");
53
54  // The AMS environments as amsmath left them, which `{empheq}` runs (and, under `overload`, the environments of the
55  // same name): `\lx@empheq@AmS@<env>`/`\lx@empheq@AmS@end<env>`, after `\<env>`/`\end<env>`.
56  let overload = lookup_bool("empheq_overload");
57  for env in OVERLOADED {
58    let (begin, end) = (s!("\\{env}"), s!("\\end{env}"));
59    if !is_defined(&begin) || !is_defined(&end) {
60      continue;
61    }
62    Let!(T_CS!(s!("\\lx@empheq@AmS@{env}")), T_CS!(begin.clone()));
63    Let!(T_CS!(s!("\\lx@empheq@AmS@end{env}")), T_CS!(end.clone()));
64    if overload {
65      // empheq.sty:964-975 `\EQ_renew_ams_environment:n`: `\renewenvironment{<env>}[1][]`; alignat's count first
66      // (:1013-1029). An environment that is a constructor (`\begin{equation}`, which `\begin` runs before
67      // `\equation`) takes the option there, and keeps its own end (no `right`). The definitions keep their lock.
68      let constructor_cs = s!("\\begin{{{env}}}");
69      let constructor = is_defined(&constructor_cs);
70      let (begin, target) = if constructor {
71        Let!(T_CS!(s!("\\lx@empheq@AmS@begin@{env}")), T_CS!(constructor_cs.clone()));
72        (constructor_cs, s!("begin@{env}"))
73      } else {
74        (begin, env.to_string())
75      };
76      let peek = if env.starts_with("alignat") {
77        s!("\\lx@empheq@peekarg{{{target}}}")
78      } else {
79        s!("\\lx@empheq@peek{{{target}}}{{}}")
80      };
81      let lock = |cs: &str| Some(ExpandableOptions { locked: lookup_bool(&s!("{cs}:locked")), ..Default::default() });
82      let begin_lock = lock(&begin);
83      def_macro(T_CS!(begin), None, TokenizeInternal!(TeXString::assembled(peek)), begin_lock)?;
84      if !constructor {
85        let end_lock = lock(&end);
86        let right = TokenizeInternal!(TeXString::assembled(s!("\\lx@empheq@right\\csname lx@empheq@AmS@end{env}\\endcsname")));
87        def_macro(T_CS!(end), None, right, end_lock)?;
88      }
89    }
90  }
91
92  // Just pass on to the ams environment
93  // Usually \begin{empheq}{amsenv} ==> \begin{amsenv}
94  // but note specialcase: \begin{empheq}{alignat=2} ==> \begin{alignat}{2} !!!
95  // The options' `left` opens the environment's first cell, `right` closes its last (above).
96  DefMacro!("\\empheq[]{}",
97    "\\lx@empheq@setkeys{#1}\\lx@empheq#2==\\end");
98
99  RawTeX!(r"\def\lx@empheq #1=#2=#3\end{%
100    \expandafter\def\expandafter\endempheq\expandafter{\expandafter\lx@empheq@right
101      \csname lx@empheq@AmS@end#1\endcsname}%
102    \if.#2.\def\lx@empheq@args{}\else\def\lx@empheq@args{{#2}}\fi
103    \expandafter\lx@empheq@go\expandafter{\lx@empheq@left}{#1}{\lx@empheq@args}}");
104
105  // empheq.sty:839-882 `\DeclareLeftDelimiter`/`\DeclareRightDelimiter`: `\empheq<delim>` and `\empheqbig<delim>`
106  // (`\empheqlbrace`, `\empheqbigrangle`, …), the delimiter grown to the display's height (`\left#2…\right.`, its
107  // phantom and kerns presentational), then the optional material.
108  RawTeX!(r"\newcommand*\DeclareLeftDelimiter[2][]{%
109      \@namedef{empheq\expandafter\@gobble\string#2}{\left#2\right.#1}%
110      \@namedef{empheqbig\expandafter\@gobble\string#2}{\left#2\right.#1}}%
111    \newcommand*\DeclareRightDelimiter[2][]{%
112      \@namedef{empheq\expandafter\@gobble\string#2}{#1\left.\right#2}%
113      \@namedef{empheqbig\expandafter\@gobble\string#2}{#1\left.\right#2}}%
114    \DeclareLeftDelimiter{\lbrace}\DeclareRightDelimiter{\rbrace}%
115    \DeclareLeftDelimiter{\lbrack}\DeclareRightDelimiter{\rbrack}%
116    \DeclareLeftDelimiter{\langle}\DeclareRightDelimiter{\rangle}%
117    \DeclareLeftDelimiter{\lvert}\DeclareRightDelimiter{\rvert}%
118    \DeclareLeftDelimiter{\lVert}\DeclareRightDelimiter{\rVert}%
119    \DeclareLeftDelimiter{\lfloor}\DeclareRightDelimiter{\rfloor}%
120    \DeclareLeftDelimiter{\lceil}\DeclareRightDelimiter{\rceil}%
121    \DeclareLeftDelimiter{\lparen}\DeclareRightDelimiter{\rparen}");
122});
