3/// This binding's frontmatter conventions (the observer's, OXIDIZED_DESIGN_DIVERGENCES #837), declared at its load
4/// (`declare_frontmatter_conventions`, user ruling 2026-10-28: each binding declares its own):
5/// the `\emails` line under the affiliations (ijcai19.sty:127-130), a class's address list: placed by spelling, else
6/// by author order, else the block (`owner=sequence`, OD #908; 2105.02711, 1905.13566, 2308.08563).
7pub const FRONTMATTER_CONVENTIONS: FmConventions = FmConventions {
8  name: "ijcai",
9  binding: "ijcai.sty",
10  email_owner: FmEmailOwner::SpelledElseOrder,
11  cite: "ijcai.sty.ltxml:50-53 (`\\emails`, `labelseq=author`); OD #908 (2105.02711, 1905.13566, 2308.08563)",
12  block_breaks: &[
13    ("\\affiliations", FmBreakKind::Section),
14    ("\\emails", FmBreakKind::Section),
15  ],
16  ..FM_KERNEL_CONVENTIONS
17};
18
19#[rustfmt::skip]
20LoadDefinitions!({
21  // (this binding's frontmatter conventions, first: OXIDIZED_DESIGN_DIVERGENCES #837)
22  declare_frontmatter_conventions(&FRONTMATTER_CONVENTIONS);
23  // Perl: ijcai.sty.ltxml (PR #2767)
24  // borrow a few of the cite-related definitions from natbib
25  RequirePackage!("natbib");
26
27  // \author{} : used once, separates multiple authors by \and (\And for last), followed by \\ !!!
28  // Bizarrely, \affiliations and \emails should be WITHIN \author!
29  // \affiliations{} gives \\ separated set of affiliations, corresponding to author in order
30  // \emails{} gives a ", " separated set of emails, corresponding to author in order
31  // If the association isn't "obvious" (one-to-one, or only a single affil?),
32  // the author is expected to use $^{1,2}$ superscript markers!!
33  Let!("\\AND",      "\\and");
34  Let!("\\And",      "\\and");
35  Let!("\\leftcite", "\\cite");
36  // ijcai24.sty:62 `\pubnote` sets its text in the running head, no label (FIDELITY_CHARTER §4, user ruling 2026-10-30; OXIDIZED_DESIGN_DIVERGENCES #921) (2406.14201).
37  DefMacro!("\\pubnote{}", "\\lx@add@pubnote[role=note,name={}]{#1}");
38
39  // These are used as separators WITHIN \author, so...
40  def_macro_noop("\\affiliations")?;
41  def_macro_noop("\\emails")?;
42
43  // `\author{}` → the shared sectioned-author splitter. The `\lx@ijcai@*`
44  // machinery it drives now lives in the engine (base_utilities.rs), so a
45  // raw-loaded ijcai derivative (e.g. ttm.sty) that never loads this binding
46  // still gets the same split via the `\lx@add@authors` marker-branch.
47  // Affiliations are `\\`-separated; the `\emails` list is placed as a class's address list (OD #908).
48  DefMacro!("\\author{}", "\\lx@ijcai@authorsplit#1\\affiliations\\done");
49});
