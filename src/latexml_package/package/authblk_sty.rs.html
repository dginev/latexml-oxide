114/// authblk's `\author` as the frontmatter call (Perl authblk.sty.ltxml, PR #2767).
115fn author_call(label: Option<Tokens>, author: Tokens) -> Result<Tokens> {
116  if let Some(label) = label {
117    // Use label attachment; a list of names under one label (`\author[1]{Ann Able, Bob Baker}`; deepseek.cls's
118    // `\author[*]{…}` block, 2401.14196) is one author each, every one carrying the label (63e)
119    return Ok(Invocation!(
120      "\\lx@add@authors@append[annotations={#1}]{#2}",
121      vec![Some(label), Some(author)]
122    ));
123  }
124  // (the kernel `\and`s the names hold, by meaning, asked once and only where a comma or the word "and" does not decide:
125  // `separated_by_and`)
126  let mut kernel_and: Option<bool> = None;
127  let mut separated = |author: &Tokens| -> Result<bool> {
128    if let Some(found) = kernel_and {
129      return Ok(found);
130    }
131    let found = separated_by_and(author)?;
132    kernel_and = Some(found);
133    Ok(found)
134  };
135  if let Some(rows) = lines_of_names_before_affil(&author, &mut separated)? {
136    // (lines under the names that read as more names, the block's institution in the `\affil` after them: names)
137    Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(
138      rows
139    )]))
140  } else if names_a_list(&author, &mut separated)?
141    || author_marks_end_name(&author, &[T_CS!("\\affil")])
142  {
143    // A comma anywhere in the names (a comma inside a group too: `\textbf{Ann Able, Bob Baker}`, whose split reads into
144    // the wrapper; one in a note or a mark, `\thanks{Dept. X, Univ Y}`, `$^{1,2}$`, leaves the name whole), a kernel
145    // `\and` at the names' own level by meaning (`separated_by_and`: written, `\And`, held in a macro of the document's,
146    // a `\let` copy, a `\csname and\endcsname`; Option C P2-7, where a spelling test missed the macro-held one), the
147    // word "and" between two names outside every group, as pdflatex prints them (`\author{Ajay Chandra$^{1}$\orcidlink{…}
148    // and Harprit Singh$^2$\orcidlink{…}}`, 2405.16615), a separator the kernel's author
149    // read parts the block at (`has_separated_names`: `Ann Able\thanks{…} \qquad Bob Baker\thanks{…}`, 2610.06709;
150    // `\hspace{2.5em}`, 2610.18435; `\quad\quad`, 2610.28573, one name holding them all with every note after it, each
151    // note's author lost; 2610.39888's `\quad` came here only by the commas in its notes; a line under a name, `Ann
152    // Able\thanks{…}\\ Dept. of Physics`, 2609.17614, 2303.15964, name and line one name), or superscript marks ending
153    // the name, only notes, spacing or an `\affil` after them (`\author{Cat$^{1}$}`, `\author{Ann$^1$\affil{$^1$Univ X}}`
154    // under titling, 2605.03796; `author_marks_end_name`: its mark requests the affiliation labelled by it on every
155    // path, as the list's names' marks do; on `\lx@add@creator` it requested nothing, and an affiliation a list author
156    // took from it was lost to it, 77n r2 review — a name whose marks other text follows, `Ann Able$^{a}$ (ann@x.org)`,
157    // stays on `\lx@add@creator`: the list path shows an unanswered mark after the name's last text, 77n r3 review
158    // probes m2, m8, m9) → a multi-author list, split into separate creators by the reading (the same comma /
159    // " and " name split the DEFAULT `\author`
160    // already applies) and added after the authors before it, as authblk adds every `\author` to its list
161    // (authblk.sty:114-145): `\lx@add@authors@list`. authblk's own no-`\and` branch keeps the whole comma list as ONE
162    // `<personname>A, B, C</personname>`; matching the default fixes that inconsistency (html_feedback#6255,
163    // googledeepmind). A single name that legitimately contains a comma ("Smith, Jr.") is mis-split, but that is the
164    // pre-existing #52 comma tradeoff the default `\author` already makes, not a new divergence. OXIDIZED_DESIGN #52(h).
165    // Replacing the authors before (`\lx@add@authors`) dropped them: `\author{Ann Able\thanks{Dept. X, Univ
166    // Y}}\affil{X}\author{Bob Baker\thanks{Dept. Z, Univ W}}` lost Ann Able, whose note's comma sent her here too (the
167    // split reads one name there: a note's comma is no separator); AIAA new-aiaa.cls papers kept 1 of 3 authors
168    // (2610.21324, 2008.05535). Perl's `\and` branch (authblk.sty.ltxml:52-53, `\lx@add@authors`) replaces them too
169    // (KNOWN_PERL_ERRORS #992).
170    Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(
171      author
172    )]))
173  } else {
174    // (a name whose marks text follows, `Bob Baker$^{1}$\textsuperscript{equal contrib.}`, `Bob Baker$^{1}$
175    // (bob@x.org)`, kept as it prints, its marks still requesting the affiliations they pick, as on the list path:
176    // `author_label_marks`, 77n r5 review probes k1, k2, k5)
177    let requests = author_label_marks(&author, &[T_CS!("\\affil")]);
178    let mut name = author.unlist();
179    for marks in requests {
180      name.extend(
181        Invocation!(
182          "\\lx@request@frontmatter@annotation[affiliation]{#1}",
183          vec![Some(marks)]
184        )
185        .unlist(),
186      );
187    }
188    Ok(Invocation!(T_CS!("\\lx@add@creator"), vec![
189      None,
190      Some(Tokens::new(name))
191    ]))
192  }
193}

196/// aside): a comma, the word "and" or a kernel `\and` between names, a separator the kernel's author read parts at.
197/// Shared with the meta-class binding (latexml_contrib `meta_class.rs`, 2605.17829).
198pub fn author_names_a_list(author: &Tokens) -> Result<bool> {
199  names_a_list(author, &mut separated_by_and)
200}

202fn names_a_list(
203  author: &Tokens,
204  separated: &mut impl FnMut(&Tokens) -> Result<bool>,
205) -> Result<bool> {
206  Ok(
207    holds_a_comma(author)
208      || has_top_level_and_word(author)
209      || separated(author)?
210      || has_separated_names(author),
211  )

214/// Whether `author` parts its names or lines with a separator the kernel's author read splits at
215/// (`has_author_block_separator`: `\quad`, `\qquad`, `\hspace`, `\hfill`, a run of spacing, `\AND`, `\\`), not one
216/// inside a note's argument or a mark, and not the space between one person's notes (`Ann Able\thanks{…}\hspace{0.4em}\thanks{…}`,
217/// the shape of 2610.04108). A group or a font command around the whole of it is transparent, as for
218/// [`has_top_level_and_word`]. Perl's authblk checks only `\and`/`\And` (authblk.sty.ltxml:52-55);
219/// OXIDIZED_DESIGN_DIVERGENCES #52(h).
220///
221/// Lines under the names that read as more names and that nothing tells from an institution are no such separator
222/// (`lines_under_names_read_as_names`: `Ann Able\\ Bob Baker`, `Ann Able\quad Bob Baker\\ Cat Cole`, `Ann Able\\ Google
223/// DeepMind`): the kernel's read would give the line to the names as their affiliation, silently, which is wrong as often
224/// as it is right, so the argument stays one name, the merge the document's Error counts (`flag_merged_creators`) —
225/// unless an `\affil` follows, [`lines_of_names_before_affil`]. So does a piece that is a macro printing a link alone
226/// (`holds_a_macro_printing_a_link`, 2601.07891): the read would make it a person, silently.
227fn has_separated_names(author: &Tokens) -> bool {
228  (has_author_block_separator(author)
229    && !lines_under_names_read_as_names(author)
230    && !holds_a_macro_printing_a_link(author))
231    || whole_wrapper_content(author).is_some_and(|inner| has_separated_names(&inner))
232}

234/// `author` as the row of names it prints when its lines under the names read as more names
235/// (`lines_under_names_read_as_names`) and an `\affil` comes next: the block's institution is printed there, so a line
236/// of the author font under the names that reads as a name is one (`Aditya Golatkar$^{*}$\hspace{1.6em}…\\[0.35em]
237/// Baishakhi Ray\hspace{1.6em}…\\[0.35em] Stefano Soatto}\affil{AWS Agentic AI}`, 2610.38214: a creator, 12 of 12). Not
238/// otherwise, nothing else saying which: eight real papers print `\\ Waymo LLC`, `\\ Physical Intelligence`, `\\ UC San
239/// Diego`, `\\ World Agents`, `\\ Welo Data` under rows of names with no `\affil` (2404.19531, 2505.23705, 2601.06328,
240/// 2610.37518, 2610.27304). Only an argument that reaches the kernel read by its separators alone, as
241/// [`has_separated_names`] reads it.
242fn lines_of_names_before_affil(
243  author: &Tokens,
244  separated: &mut impl FnMut(&Tokens) -> Result<bool>,
245) -> Result<Option<Tokens>> {
246  let bare = !holds_a_comma(author) && !has_top_level_and_word(author);
247  Ok(
248    (bare
249      && has_author_block_separator(author)
250      && lines_under_names_read_as_names(author)
251      && !separated(author)?
252      && next_is_affil())
253    .then(|| line_breaks_as_name_separators(author)),
254  )
255}

257/// Whether `author` holds a comma anywhere, a group's or a note's too (what the list path's split reads: a note's comma
258/// is no separator there, [`author_call`]).
259fn holds_a_comma(author: &Tokens) -> bool { author.unlist_ref().contains(&T_OTHER!(",")) }

261/// Whether `author` parts its names at an `\and`: one at its own level meaning the kernel's (`argument_separators`,
262/// OXIDIZED_DESIGN_DIVERGENCES #995) — written, held in a macro of the document's (`\newcommand\aut{Ann Able \and Bob
263/// Baker}`), a `\let` copy, as authblk.sty's `\author` (:114-145) leaves the kernel's `\and` to part the names in
264/// `\@author`. A separator of another meaning (the document's renewed `\and`, amsmath's `\And`, a class binding's
265/// stand-in) is no kernel column (trap T8 of the P2 plan); a binding's `\AND` `\let` to the kernel's `\and` (inst_support,
266/// the OmniBus path) is one, which the spelling missed. Where the query reads nothing (`None`: inside a rehearsal, over
267/// its allowance, a group left open, an argument too long for the field) the spelling decides, as before Option C's
268/// P2-7: a token meaning what `\and` or `\And` means, at any depth.
269fn separated_by_and(author: &Tokens) -> Result<bool> {
270  Ok(match argument_separators(author)? {
271    Some(separators) => separators
272      .iter()
273      .any(|found| found.meaning == FmSepMeaning::Kernel),
274    None => author
275      .unlist_ref()
276      .iter()
277      .any(|t| t.defined_as(&T_CS!("\\and")) || t.defined_as(&T_CS!("\\And"))),
278  })
279}

281/// Whether an `\affil` is what the input holds next, past spaces and paragraph breaks.
282fn next_is_affil() -> bool {

284    .find(|t| t.get_catcode() != Catcode::SPACE && *t != T_CS!("\\par"))
285    == Some(T_CS!("\\affil"))

288/// What a group or a font command around the whole of `author` holds, spaces around it aside: `{…}`, `\textbf{…}`,
289/// `\emph{…}`; not `\textsuperscript{…}`, a mark.
290fn whole_wrapper_content(author: &Tokens) -> Option<Tokens> {
291  let v = trimmed_spaces(author.unlist_ref());
292  // (the group a wrapper opens: `{…}`, `\cmd{…}`, when it closes on the last token)
293  let open = match v.first().map(Token::get_catcode) {
294    Some(Catcode::BEGIN) => 0,
295    Some(Catcode::CS)
296      if v[0].with_str(|cs| {
297        (cs.starts_with("\\text") && cs != "\\textsuperscript") || cs == "\\emph"
298      }) && v.get(1).is_some_and(|t| t.get_catcode() == Catcode::BEGIN) =>
299    {
300      1
301    },
302    _ => return None,
303  };
304  let mut depth = 0i32;
305  let close = (open..v.len()).find(|&k| {
306    match v[k].get_catcode() {
307      Catcode::BEGIN => depth += 1,
308      Catcode::END => depth -= 1,
309      _ => {},
310    }
311    depth == 0
312  });
313  (close == Some(v.len() - 1) && v.len() > open + 2)
314    .then(|| Tokens::new(v[open + 1..v.len() - 1].to_vec()))
315}

317/// `v` without the spaces at its ends.
318fn trimmed_spaces(v: &[Token]) -> &[Token] {
319  let start = v.iter().position(|t| *t != T_SPACE!()).unwrap_or(v.len());
320  let end = v
321    .iter()
322    .rposition(|t| *t != T_SPACE!())
323    .map_or(start, |e| e + 1);
324  &v[start..end]
325}

327/// Whether `author` holds the word "and" at its top level, between spaces or ties (not inside a note's argument or a
328/// group: `\thanks{Ann Able and Bob Baker are with X}`). A group or a font command around the whole of it is
329/// transparent: `{Nikolai N. Leonenko\textsuperscript{a}\thanks{…} ~and Andrey Pepelyshev\textsuperscript{a}}`
330/// (2506.23337), `\textbf{Ann Able and Bob Baker}` hold it at their top level.
331fn has_top_level_and_word(author: &Tokens) -> bool {
332  if let Some(inner) = whole_wrapper_content(author) {
333    return has_top_level_and_word(&inner);
334  }
335  let v = trimmed_spaces(author.unlist_ref());
336  let gap = |t: Option<&Token>| t.is_some_and(|t| *t == T_SPACE!() || *t == T_ACTIVE!('~'));
337  let mut depth = 0i32;
338  (0..v.len()).any(|i| {
339    match v[i].get_catcode() {
340      Catcode::BEGIN => depth += 1,
341      Catcode::END => depth -= 1,
342      _ => {},
343    }
344    depth == 0
345      && v[i..].starts_with(&[T_LETTER!("a"), T_LETTER!("n"), T_LETTER!("d")])
346      && gap(i.checked_sub(1).and_then(|b| v.get(b)))
347      && gap(v.get(i + 3))
348  })
349}

351/// authblk's `\author[mark]{name}` and `\affil[mark]{affiliation}` (authblk.sty:114-167) as frontmatter calls; also
352/// run by a class that copies authblk's code over them (lni.cls:384-464, `lni_cls.rs` in latexml_contrib), so the
353/// copy's accumulators — read only by its refused `\maketitle` — do not take the author block.
354#[rustfmt::skip]
355pub fn define_author_and_affil() -> Result<()> {
356  // Perl: authblk.sty.ltxml (PR #2767)
357  DefMacro!("\\author[]{}", sub[(label, author)] {
358    // authblk's bookkeeping after the author (`\lx@AB@author@state`, below).
359    let mut state = vec![T_CS!("\\lx@AB@author@state"), T_BEGIN!()];
360    if let Some(label) = label.as_ref() {
361      note_labels(NAME_LABELS, label)?;
362      // (each piece printing one mark, by its spelling in the call, which the writer requests it by: `count_lost_labels`;
363      // not one of several words, `\author[1 2]`, which the writer reads apart, 1909.01989)
364      for (_, piece) in split_tokens_delimited(label.clone(), vec![T_OTHER!(",").into()]) {
365        let spelling = piece.to_string();
366        if spelling.trim().contains(char::is_whitespace) {
367          continue;
368        }
369        if let [mark] = printed_label_pieces(&piece)?.as_slice() {
370          let mut kept = lookup_string(NAME_LABEL_MARKS);
371          kept.push_str(&s!("{spelling}\u{2}{mark}\u{1}"));
372          assign_value(NAME_LABEL_MARKS, Stored::String(pin(&kept)), Some(Scope::Global));
373        }
374      }
375      state.extend(label.unlist_ref().iter().copied());
376    }
377    state.push(T_END!());
378    state.push(T_BEGIN!());
379    state.extend(author.unlist_ref().iter().copied());
380    state.push(T_END!());
381    // What a package appends to `\author` (`\lx@author@after`, sect05.rs; titling's `\theauthor`).
382    state.extend([T_CS!("\\lx@author@after"), T_BEGIN!()]);
383    state.extend(author.unlist_ref().iter().copied());
384    state.push(T_END!());
385    let state = Tokens::new(state);
386    Ok(Tokens::new([author_call(label, author)?.unlist(), state.unlist()].concat()))
387  });
388
389  // Robust, as authblk.sty:148's `\newcommand\affil[2][]` is (`\@protected@testopt`): an `\affil` inside
390  // `\author` survives titling's `\protected@xdef\theauthor{\@author}` (witness 2605.03796, whose body was lost).
391  // An unlabelled `\affil` is every author's given since the last one (`annotate=run:N`, [`affil_run`]): its note is
392  // `\AB@thenote`, which only the next `\author` after an `\affil` steps (authblk.sty:115-116), so consecutive `\affil`s
393  // (`\author{Ann Able and Bob Baker}\affil{Univ X}\affil{Univ Y}`) are all printed under the same authors; an author
394  // of the group with an affiliation already (a `\\` line under the name) is passed over, as the run does. One that
395  // opens with a hand-written mark a queued author carries too (`\author{Ann Able$^{1}$ and Bob Baker$^2$}\affil{$^1$Univ
396  // X}\affil{$^2$Univ Y}`; 1507.08653, 2510.01944) goes to the authors showing its mark, as a marked affiliation list's
397  // lines do (`\lx@add@affiliation@marked`) — beyond authblk, whose `\AB@thenote` prints it under the whole block
398  // (OXIDIZED_DESIGN #765). A footnote symbol (`$^*$Equal contribution`) or a mark no author carries (a typo, an author
399  // without marks after marked ones) keeps authblk's run: a label no one requests would go to a nameless author.
400  // LOCKED, as the kernel's `\title`/`\author` are (sect05.rs): a document's, class's or package's own `\affil` is
401  // refused, and its body still runs first for its side effects, its output discarded (`\lx@replay@class@setter`,
402  // OXIDIZED_DESIGN #482), so the affiliation stays the binding's contact. The JOSS template's copy of authblk's own
403  // `\affil` code (paper_joss.tex:162-182, a `\renewcommand{\affil}[2][]` that differs from authblk.sty:148-167 only in
404  // typography) put every affiliation in `\AB@authors`, which nothing prints here (2504.08081, 2310.11002, 2107.08407,
405  // 2507.13341, 2604.08855); and so did the raw classes that copy it after `\RequirePackage{authblk}`, oasics.cls:610
406  // (1709.00722) and eurocg25.cls (2503.02336), dagman.cls:696 the same (1701.09007; lipics-v2016.cls:611 is read as
407  // its binding's stand-in copy, which declines it unrecorded: 1603.00928). A redefinition that takes arguments is refused at
408  // every origin (state.rs `install_definition`, OXIDIZED_DESIGN #500); a binding that loads authblk and defines its
409  // own `\affil` loads unlocked and keeps it (lni_cls.rs). OXIDIZED_DESIGN_DIVERGENCES #959.
410  DefMacro!("\\affil OptionalSemiverbatim {}", sub[(label, affil)] {
411    let label = label.unwrap_or_default();
412    if label.is_empty() {
413      assign_value(LINE_UNLABELLED, true, Some(Scope::Global));
414    } else {
415      // (read as authblk.sty:148 reads `\affil[#1]`, not verbatim: `\affil[$^1$]` prints 1, as `\author[$^{1,2}$]`'s
416      // pieces do, 2106.15893)
417      note_labels(LINE_LABELS, &Tokenize!(TeXString::assembled(label.clone().untex_string().to_string())))?;
418    }
419    let replay = Invocation!(T_CS!("\\lx@replay@class@setter"), vec![
420      Some(Tokens::new(Explode!("affil"))),
421      Some(label.clone()),
422      Some(affil.clone())
423    ]);
424    // (a mark glued to a footnote symbol labels the line by its number, as a name's asks for it apart)
425    let affil = glued_line_marks_as_marks(affil, true);
426    let run = affil_run();
427    // (the label printed as written before the line, `\AB@affilnote`, authblk.sty:142: the mark a legend's note shows,
428    // `_labelprints`, OXIDIZED_DESIGN_DIVERGENCES #961)
429    let call = if !label.is_empty() {
430      Invocation!("\\lx@add@contact[role=affiliation,annotate=1,label={#1},_labelprints=true]{#2}", vec![
431        Some(label.clone()),
432        Some(affil)
433      ])
434    } else if opens_with_a_mark(&affil) && queued_authors_carry_entry_mark(&affil)? {
435      Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![None, Some(affil)])
436    } else if let Some((unmarked, marked)) = unmarked_before_carried_marks(&affil)? {
437      let mut calls = Vec::new();
438      for piece in unmarked {
439        calls.extend(
440          Invocation!("\\lx@add@contact[role=affiliation,label=addresses:block]{#1}", vec![Some(piece)])
441            .unlist(),
442        );
443      }
444      calls.extend(
445        Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![None, Some(marked)]).unlist(),
446      );
447      Tokens::new(calls)
448    } else if run > 0 {
449      Invocation!("\\lx@add@contact[role=affiliation,annotate=run:#1]{#2}", vec![
450        Some(Tokens::new(Explode!(run.to_string()))),
451        Some(affil)
452      ])
453    } else {
454      // (no author queued as it expands — inside `\author{Ann \affil{Univ X}}`, read at the flush, or after `\maketitle`
455      // (2605.01689): the run uncapped, as Perl's `annotate=new` (authblk.sty.ltxml:57) reaches them; a cap of 0
456      // attached it to no one, 77n r3 review)
457      Invocation!("\\lx@add@contact[role=affiliation,annotate=run]{#1}", vec![Some(affil)])
458    };
459    Ok(Tokens::new([
460      replay.unlist(),
461      call.unlist(),
462      Invocation!(T_CS!("\\lx@AB@affil@state"), vec![Some(label)]).unlist(),
463    ]
464    .concat()))
465  }, robust => true, locked => true);
466  // (at `\maketitle`, the names' labels no `\affil` prints: `count_lost_labels`)
467  DefPrimitive!("\\lx@AB@lost@labels", { count_lost_labels(); });
468  AddToMacro!("\\lx@maketitle@begin", "\\lx@AB@lost@labels");
469  Ok(())
470}

472/// The labels of the names (`\author[1,*,\S]`'s pieces) and of the lines (`\affil[*]`'s) given so far, each piece once,
473/// `\u{1}` apart; and whether an `\affil` came unlabelled, numbered by authblk's counter (`\AB@thenote`).
474const NAME_LABELS: &str = "lx_AB_name_labels";
475/// The names' label pieces that print one mark (`§`, `\S`, `$\dagger$`, a document's macro printing one), each its
476/// spelling and the mark it prints, `\u{2}` between them and `\u{1}` after each.
477const NAME_LABEL_MARKS: &str = "lx_AB_name_label_marks";
478const LINE_LABELS: &str = "lx_AB_line_labels";
479const LINE_UNLABELLED: &str = "lx_AB_line_unlabelled";
480
481/// Note each comma piece of `label` under `key` ([`NAME_LABELS`], [`LINE_LABELS`]) by the text it prints, both sides read
482/// alike (`printed_label_pieces`: `\author[\affA]` with `\newcommand\affA{1}` is `\affil[1]`'s 1, `$\dag$` and `\dag`
483/// one †): one that is letters and digits, or symbols (`**`, `§`); a mixed piece (`1$^*$`, 2412.03338), or one read by
484/// its spelling holding more than one command, is no mark compared here.
485fn note_labels(key: &str, label: &Tokens) -> Result<()> {
486  let mut kept = lookup_string(key);
487  for piece in printed_label_pieces(label)? {
488    let symbol = |c: char| {
489      c == '*' || c == '#' || (!c.is_ascii() && !c.is_alphanumeric() && !c.is_whitespace())
490    };
491    let single = match piece.strip_prefix('\\') {
492      None => piece.chars().all(char::is_alphanumeric) || piece.chars().all(symbol),
493      Some(name) => {
494        !name.is_empty() && (name.chars().all(char::is_alphabetic) || name.chars().count() == 1)
495      },
496    };
497    if single && !kept.split('\u{1}').any(|k| k == piece) {
498      if !kept.is_empty() {
499        kept.push('\u{1}');
500      }
501      kept.push_str(&piece);
502    }
503  }
504  assign_value(key, Stored::String(pin(&kept)), Some(Scope::Global));
505  Ok(())
506}

508/// authblk prints each name's labels (`\AB@authnote`, authblk.sty:120-133) and each line's before it
509/// (`\AB@affilnote`, :142); a name's label no line prints is declared to the writer, which shows the mark it prints on the
510/// name (`declare_unexplained_name_labels`, `Policy::labels_print`: 2606.00446's `\author[1,*,\S]`, whose `§` only the
511/// `\date` line explains, "Ann Able1,*,§"), but one no piece prints alone (`$^{1,2}$`'s, a piece printing two marks) the
512/// frontmatter places nowhere, counted (FIDELITY_CHARTER: no silent loss). Not where an `\affil` is unlabelled, as its
513/// number (`\AB@thenote`) may answer one; read at `\maketitle`, before the flush, then cleared.
514fn count_lost_labels() {
515  let names = lookup_string(NAME_LABELS);
516  let lines = lookup_string(LINE_LABELS);
517  let marks = lookup_string(NAME_LABEL_MARKS);
518  let mut shown: Vec<(String, String)> = Vec::new();
519  if !names.is_empty() && !lookup_bool(LINE_UNLABELLED) {
520    let mut lost: Vec<&str> = Vec::new();
521    for label in names
522      .split('\u{1}')
523      .filter(|label| !lines.split('\u{1}').any(|line| line == *label))
524    {
525      let spellings: Vec<(String, String)> = marks
526        .split('\u{1}')
527        .filter_map(|entry| entry.split_once('\u{2}'))
528        .filter(|(_, mark)| mark == &label)
529        .map(|(spelling, mark)| (spelling.to_string(), mark.to_string()))
530        .collect();
531      if spellings.is_empty() {
532        lost.push(label);
533      }
534      shown.extend(spellings);
535    }
536    if !lost.is_empty() {
537      Warn!(
538        "frontmatter",
539        "unanswered_mark_lost",
540        s!(
541          "authblk prints {} label(s) of the names no \\affil explains that no name can show: {}",
542          lost.len(),
543          lost.join(", ")
544        )
545      );
546    }
547  }
548  declare_unexplained_name_labels(&shown);
549  for key in [NAME_LABELS, NAME_LABEL_MARKS, LINE_LABELS] {
550    assign_value(key, Stored::String(pin("")), Some(Scope::Global));
551  }
552  assign_value(LINE_UNLABELLED, false, Some(Scope::Global));
553}

555/// The number of creators an unlabelled `\affil` is printed under at most: the ones queued since the last `\affil`
556/// before them, which share its `\AB@thenote` (authblk.sty:115-116 step it at the first `\author` after an `\affil`)
557/// — the same group as the `\affil` before when no author came between (`\author{Ann}\author{Bob}\affil{X}\affil{Y}`:
558/// both under both). Counted here, as every `\affil` passes: a mark-picked one too, whose affiliation reaches its
559/// authors only once their marks are matched, so the run alone would take its authors for new ones
560/// (`\author{Ann$^1$}\affil{$^1$X}\author{Bob}\affil{$^2$Y}` printed Y under Ann as well, 77n r2 review); 0 before
561/// any author.
562fn affil_run() -> usize {
563  let count = queued_creator_count();
564  let (before, run) = match lookup_value("lx_AB_affil_run") {
565    Some(Stored::String(last)) => with(last, |s| {
566      let (before, run) = s.split_once('|').unwrap_or_default();
567      (before.parse().unwrap_or(0), run.parse().unwrap_or(0))
568    }),
569    _ => (0, 0),
570  };
571  // (a flush that cleared the queue since starts the count anew)
572  let run = match count.checked_sub(before) {
573    Some(0) => run,
574    Some(new) => new,
575    None => count,
576  };
577  assign_value(
578    "lx_AB_affil_run",
579    Stored::String(pin(s!("{count}|{run}"))),
580    Some(Scope::Global),
581  );
582  run
583}

585/// An affiliation whose first pieces, parted by `\quad`, no mark opens and a later piece a mark a queued author carries
586/// opens: those pieces, and the rest. `\textsuperscript{}~Univ. of California \quad \textsuperscript{2}~Northwestern
587/// University \quad …` (2505.14999): an empty mark is no mark (`drop_empty_marks`, user 2026-10-20), so the first piece is
588/// no one's by a mark, and it is every author's by nothing: the block's, as an unmarked line among marked ones is (user
589/// ruling R12), not `Luo$^2$`'s, who shows the mark of his own.
590fn unmarked_before_carried_marks(affil: &Tokens) -> Result<Option<(Vec<Tokens>, Tokens)>> {
591  let pieces = split_tokens_delimited(affil.clone(), vec![
592    T_CS!("\\quad").into(),
593    T_CS!("\\qquad").into(),
594  ]);
595  let mut first = None;
596  for (at, (_, piece)) in pieces.iter().enumerate() {
597    if opens_with_a_mark(piece) && queued_authors_carry_entry_mark(piece)? {
598      first = Some(at);
599      break;
600    }
601  }
602  let Some(first) = first.filter(|first| *first > 0) else {
603    return Ok(None);
604  };
605  // (no mark on them, an empty one aside: a legend or an entry a mark leads is the list's, `$^{*}$Corresponding author \\
606  // $^{1}$Univ X \quad $^{3}$Univ Z`, 2505.00989)
607  if pieces[..first]
608    .iter()
609    .any(|(_, piece)| author_mark_position(&drop_empty_marks(piece.clone())).is_some())
610  {
611    return Ok(None);
612  }
613  let mut marked: Vec<Token> = Vec::new();
614  for (at, (delimiter, piece)) in pieces.iter().enumerate().skip(first) {
615    if at > first {
616      marked.extend(delimiter.iter().copied());
617    }
618    marked.extend(piece.unlist_ref().iter().copied());
619  }
620  // (each alone, its empty mark no mark of a marked list, what leads it no text)
621  let unmarked = pieces[..first]
622    .iter()
623    .map(|(_, piece)| {
624      let piece = drop_empty_marks(piece.clone());
625      let v = piece.unlist_ref();
626      Tokens::new(v[leading_unprinted_end(v)..].to_vec())
627    })
628    .filter(|piece| !piece.is_empty() && !prints_nothing(piece))
629    .collect();
630  Ok(Some((unmarked, Tokens::new(marked))))
631}

633/// Whether an affiliation opens with a superscript mark, `$^{1}$` or `\textsuperscript{1}`, after what prints nothing
634/// before it: spaces, spacing and declarations (`leading_unprinted_end`, as an author block's lines are read: 2605.11124's
635/// `\affil{\vspace{-1.05em}$^2$…}` went to every author since the last `\affil`, 77n r5 review).
636fn opens_with_a_mark(affil: &Tokens) -> bool {
637  let v = affil.unlist_ref();
638  let mut i = leading_unprinted_end(v);
639  while v.get(i) == Some(&T_SPACE!()) {
640    i += 1;
641  }
642  match v.get(i) {
643    Some(t) if *t == T_CS!("\\textsuperscript") => true,
644    Some(t) if t.get_catcode() == Catcode::MATH => v.get(i + 1) == Some(&T_SUPER!()),
645    _ => false,
646  }
