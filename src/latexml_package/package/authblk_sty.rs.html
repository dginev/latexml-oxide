113/// authblk's `\author` as the frontmatter call (Perl authblk.sty.ltxml, PR #2767).
114fn author_call(label: Option<Tokens>, author: Tokens) -> Result<Tokens> {
115  if let Some(label) = label {
116    // Use label attachment; a list of names under one label (`\author[1]{Ann Able, Bob Baker}`; deepseek.cls's
117    // `\author[*]{…}` block, 2401.14196) is one author each, every one carrying the label (63e)
118    return Ok(Invocation!(
119      "\\lx@add@authors@append[annotations={#1}]{#2}",
120      vec![Some(label), Some(author)]
121    ));
122  }
123  // (the kernel `\and`s the names hold, by meaning, asked once and only where a comma or the word "and" does not decide:
124  // `separated_by_and`)
125  let mut kernel_and: Option<bool> = None;
126  let mut separated = |author: &Tokens| -> Result<bool> {
127    if let Some(found) = kernel_and {
128      return Ok(found);
129    }
130    let found = separated_by_and(author)?;
131    kernel_and = Some(found);
132    Ok(found)
133  };
134  if let Some(rows) = lines_of_names_before_affil(&author, &mut separated)? {
135    // (lines under the names that read as more names, the block's institution in the `\affil` after them: names)
136    Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(
137      rows
138    )]))
139  } else if names_a_list(&author, &mut separated)?
140    || author_marks_end_name(&author, &[T_CS!("\\affil")])
141  {
142    // A comma anywhere in the names (a comma inside a group too: `\textbf{Ann Able, Bob Baker}`, whose split reads into
143    // the wrapper; one in a note or a mark, `\thanks{Dept. X, Univ Y}`, `$^{1,2}$`, leaves the name whole), a kernel
144    // `\and` at the names' own level by meaning (`separated_by_and`: written, `\And`, held in a macro of the document's,
145    // a `\let` copy, a `\csname and\endcsname`; Option C P2-7, where a spelling test missed the macro-held one), the
146    // word "and" between two names outside every group, as pdflatex prints them (`\author{Ajay Chandra$^{1}$\orcidlink{…}
147    // and Harprit Singh$^2$\orcidlink{…}}`, 2405.16615), a separator the kernel's author
148    // read parts the block at (`has_separated_names`: `Ann Able\thanks{…} \qquad Bob Baker\thanks{…}`, 2610.06709;
149    // `\hspace{2.5em}`, 2610.18435; `\quad\quad`, 2610.28573, one name holding them all with every note after it, each
150    // note's author lost; 2610.39888's `\quad` came here only by the commas in its notes; a line under a name, `Ann
151    // Able\thanks{…}\\ Dept. of Physics`, 2609.17614, 2303.15964, name and line one name), or superscript marks ending
152    // the name, only notes, spacing or an `\affil` after them (`\author{Cat$^{1}$}`, `\author{Ann$^1$\affil{$^1$Univ X}}`
153    // under titling, 2605.03796; `author_marks_end_name`: its mark requests the affiliation labelled by it on every
154    // path, as the list's names' marks do; on `\lx@add@creator` it requested nothing, and an affiliation a list author
155    // took from it was lost to it, 77n r2 review — a name whose marks other text follows, `Ann Able$^{a}$ (ann@x.org)`,
156    // stays on `\lx@add@creator`: the list path shows an unanswered mark after the name's last text, 77n r3 review
157    // probes m2, m8, m9) → a multi-author list, split into separate creators by the reading (the same comma /
158    // " and " name split the DEFAULT `\author`
159    // already applies) and added after the authors before it, as authblk adds every `\author` to its list
160    // (authblk.sty:114-145): `\lx@add@authors@list`. authblk's own no-`\and` branch keeps the whole comma list as ONE
161    // `<personname>A, B, C</personname>`; matching the default fixes that inconsistency (html_feedback#6255,
162    // googledeepmind). A single name that legitimately contains a comma ("Smith, Jr.") is mis-split, but that is the
163    // pre-existing #52 comma tradeoff the default `\author` already makes, not a new divergence. OXIDIZED_DESIGN #52(h).
164    // Replacing the authors before (`\lx@add@authors`) dropped them: `\author{Ann Able\thanks{Dept. X, Univ
165    // Y}}\affil{X}\author{Bob Baker\thanks{Dept. Z, Univ W}}` lost Ann Able, whose note's comma sent her here too (the
166    // split reads one name there: a note's comma is no separator); AIAA new-aiaa.cls papers kept 1 of 3 authors
167    // (2610.21324, 2008.05535). Perl's `\and` branch (authblk.sty.ltxml:52-53, `\lx@add@authors`) replaces them too
168    // (KNOWN_PERL_ERRORS #992).
169    Ok(Invocation!(T_CS!("\\lx@add@authors@list"), vec![Some(
170      author
171    )]))
172  } else {
173    // (a name whose marks text follows, `Bob Baker$^{1}$\textsuperscript{equal contrib.}`, `Bob Baker$^{1}$
174    // (bob@x.org)`, kept as it prints, its marks still requesting the affiliations they pick, as on the list path:
175    // `author_label_marks`, 77n r5 review probes k1, k2, k5)
176    let requests = author_label_marks(&author, &[T_CS!("\\affil")]);
177    let mut name = author.unlist();
178    for marks in requests {
179      name.extend(
180        Invocation!(
181          "\\lx@request@frontmatter@annotation[affiliation]{#1}",
182          vec![Some(marks)]
183        )
184        .unlist(),
185      );
186    }
187    Ok(Invocation!(T_CS!("\\lx@add@creator"), vec![
188      None,
189      Some(Tokens::new(name))
190    ]))
191  }
192}

196/// Shared with the meta-class binding (latexml_contrib `meta_class.rs`, 2605.17829).
197pub fn author_names_a_list(author: &Tokens) -> Result<bool> {
198  names_a_list(author, &mut separated_by_and)

201fn names_a_list(
202  author: &Tokens,
203  separated: &mut impl FnMut(&Tokens) -> Result<bool>,
204) -> Result<bool> {
205  Ok(
206    holds_a_comma(author)
207      || has_top_level_and_word(author)
208      || separated(author)?
209      || has_separated_names(author),
210  )

213/// Whether `author` parts its names or lines with a separator the kernel's author read splits at
214/// (`has_author_block_separator`: `\quad`, `\qquad`, `\hspace`, `\hfill`, a run of spacing, `\AND`, `\\`), not one
215/// inside a note's argument or a mark, and not the space between one person's notes (`Ann Able\thanks{…}\hspace{0.4em}\thanks{…}`,
216/// the shape of 2610.04108). A group or a font command around the whole of it is transparent, as for
217/// [`has_top_level_and_word`]. Perl's authblk checks only `\and`/`\And` (authblk.sty.ltxml:52-55);
218/// OXIDIZED_DESIGN_DIVERGENCES #52(h).
219///
220/// Lines under the names that read as more names and that nothing tells from an institution are no such separator
221/// (`lines_under_names_read_as_names`: `Ann Able\\ Bob Baker`, `Ann Able\quad Bob Baker\\ Cat Cole`, `Ann Able\\ Google
222/// DeepMind`): the kernel's read would give the line to the names as their affiliation, silently, which is wrong as often
223/// as it is right, so the argument stays one name, the merge the document's Error counts (`flag_merged_creators`) —
224/// unless an `\affil` follows, [`lines_of_names_before_affil`]. So does a piece that is a macro printing a link alone
225/// (`holds_a_macro_printing_a_link`, 2601.07891): the read would make it a person, silently.
226fn has_separated_names(author: &Tokens) -> bool {
227  (has_author_block_separator(author)
228    && !lines_under_names_read_as_names(author)
229    && !holds_a_macro_printing_a_link(author))
230    || whole_wrapper_content(author).is_some_and(|inner| has_separated_names(&inner))
231}

233/// `author` as the row of names it prints when its lines under the names read as more names
234/// (`lines_under_names_read_as_names`) and an `\affil` comes next: the block's institution is printed there, so a line
235/// of the author font under the names that reads as a name is one (`Aditya Golatkar$^{*}$\hspace{1.6em}…\\[0.35em]
236/// Baishakhi Ray\hspace{1.6em}…\\[0.35em] Stefano Soatto}\affil{AWS Agentic AI}`, 2610.38214: a creator, 12 of 12). Not
237/// otherwise, nothing else saying which: eight real papers print `\\ Waymo LLC`, `\\ Physical Intelligence`, `\\ UC San
238/// Diego`, `\\ World Agents`, `\\ Welo Data` under rows of names with no `\affil` (2404.19531, 2505.23705, 2601.06328,
239/// 2610.37518, 2610.27304). Only an argument that reaches the kernel read by its separators alone, as
240/// [`has_separated_names`] reads it.
241fn lines_of_names_before_affil(
242  author: &Tokens,
243  separated: &mut impl FnMut(&Tokens) -> Result<bool>,
244) -> Result<Option<Tokens>> {
245  let bare = !holds_a_comma(author) && !has_top_level_and_word(author);
246  Ok(
247    (bare
248      && has_author_block_separator(author)
249      && lines_under_names_read_as_names(author)
250      && !separated(author)?
251      && next_is_affil())
252    .then(|| line_breaks_as_name_separators(author)),
253  )
254}

256/// Whether `author` holds a comma anywhere, a group's or a note's too (what the list path's split reads: a note's comma
257/// is no separator there, [`author_call`]).
258fn holds_a_comma(author: &Tokens) -> bool { author.unlist_ref().contains(&T_OTHER!(",")) }
259
260/// Whether `author` parts its names at an `\and`: one at its own level meaning the kernel's (`argument_separators`,
261/// OXIDIZED_DESIGN_DIVERGENCES #995) — written, held in a macro of the document's (`\newcommand\aut{Ann Able \and Bob
262/// Baker}`), a `\let` copy, as authblk.sty's `\author` (:114-145) leaves the kernel's `\and` to part the names in
263/// `\@author`. A separator of another meaning (the document's renewed `\and`, amsmath's `\And`, a class binding's
264/// stand-in) is no kernel column (trap T8 of the P2 plan); a binding's `\AND` `\let` to the kernel's `\and` (inst_support,
265/// the OmniBus path) is one, which the spelling missed. Where the query reads nothing (`None`: inside a rehearsal, over
266/// its allowance, a group left open, an argument too long for the field) the spelling decides, as before Option C's
267/// P2-7: a token meaning what `\and` or `\And` means, at any depth.
268fn separated_by_and(author: &Tokens) -> Result<bool> {
269  Ok(match argument_separators(author)? {
270    Some(separators) => separators
271      .iter()
272      .any(|found| found.meaning == FmSepMeaning::Kernel),
273    None => author
274      .unlist_ref()
275      .iter()
276      .any(|t| t.defined_as(&T_CS!("\\and")) || t.defined_as(&T_CS!("\\And"))),
277  })
278}

280/// Whether an `\affil` is what the input holds next, past spaces and paragraph breaks.
281fn next_is_affil() -> bool {

283    .find(|t| t.get_catcode() != Catcode::SPACE && *t != T_CS!("\\par"))
284    == Some(T_CS!("\\affil"))

287/// What a group or a font command around the whole of `author` holds, spaces around it aside: `{…}`, `\textbf{…}`,
288/// `\emph{…}`; not `\textsuperscript{…}`, a mark.
289fn whole_wrapper_content(author: &Tokens) -> Option<Tokens> {
290  let v = trimmed_spaces(author.unlist_ref());
291  // (the group a wrapper opens: `{…}`, `\cmd{…}`, when it closes on the last token)
292  let open = match v.first().map(Token::get_catcode) {
293    Some(Catcode::BEGIN) => 0,
294    Some(Catcode::CS)
295      if v[0].with_str(|cs| {
296        (cs.starts_with("\\text") && cs != "\\textsuperscript") || cs == "\\emph"
297      }) && v.get(1).is_some_and(|t| t.get_catcode() == Catcode::BEGIN) =>
298    {
299      1
300    },
301    _ => return None,
302  };
303  let mut depth = 0i32;
304  let close = (open..v.len()).find(|&k| {
305    match v[k].get_catcode() {
306      Catcode::BEGIN => depth += 1,
307      Catcode::END => depth -= 1,
308      _ => {},
309    }
310    depth == 0
311  });
312  (close == Some(v.len() - 1) && v.len() > open + 2)
313    .then(|| Tokens::new(v[open + 1..v.len() - 1].to_vec()))
314}

316/// `v` without the spaces at its ends.
317fn trimmed_spaces(v: &[Token]) -> &[Token] {
318  let start = v.iter().position(|t| *t != T_SPACE!()).unwrap_or(v.len());
319  let end = v
320    .iter()
321    .rposition(|t| *t != T_SPACE!())
322    .map_or(start, |e| e + 1);
323  &v[start..end]
324}

326/// Whether `author` holds the word "and" at its top level, between spaces or ties (not inside a note's argument or a
327/// group: `\thanks{Ann Able and Bob Baker are with X}`). A group or a font command around the whole of it is
328/// transparent: `{Nikolai N. Leonenko\textsuperscript{a}\thanks{…} ~and Andrey Pepelyshev\textsuperscript{a}}`
329/// (2506.23337), `\textbf{Ann Able and Bob Baker}` hold it at their top level.
330fn has_top_level_and_word(author: &Tokens) -> bool {
331  if let Some(inner) = whole_wrapper_content(author) {
332    return has_top_level_and_word(&inner);
333  }
334  let v = trimmed_spaces(author.unlist_ref());
335  let gap = |t: Option<&Token>| t.is_some_and(|t| *t == T_SPACE!() || *t == T_ACTIVE!('~'));
336  let mut depth = 0i32;
337  (0..v.len()).any(|i| {
338    match v[i].get_catcode() {
339      Catcode::BEGIN => depth += 1,
340      Catcode::END => depth -= 1,
341      _ => {},
342    }
343    depth == 0
344      && v[i..].starts_with(&[T_LETTER!("a"), T_LETTER!("n"), T_LETTER!("d")])
345      && gap(i.checked_sub(1).and_then(|b| v.get(b)))
346      && gap(v.get(i + 3))
347  })
348}

350/// authblk's `\author[mark]{name}` and `\affil[mark]{affiliation}` (authblk.sty:114-167) as frontmatter calls; also
351/// run by a class that copies authblk's code over them (lni.cls:384-464, `lni_cls.rs` in latexml_contrib), so the
352/// copy's accumulators — read only by its refused `\maketitle` — do not take the author block.
353#[rustfmt::skip]
354pub fn define_author_and_affil() -> Result<()> {
355  // Perl: authblk.sty.ltxml (PR #2767)
356  DefMacro!("\\author[]{}", sub[(label, author)] {
357    // authblk's bookkeeping after the author (`\lx@AB@author@state`, below).
358    let mut state = vec![T_CS!("\\lx@AB@author@state"), T_BEGIN!()];
359    if let Some(label) = label.as_ref() {
360      state.extend(label.unlist_ref().iter().copied());
361    }
362    state.push(T_END!());
363    state.push(T_BEGIN!());
364    state.extend(author.unlist_ref().iter().copied());
365    state.push(T_END!());
366    // What a package appends to `\author` (`\lx@author@after`, sect05.rs; titling's `\theauthor`).
367    state.extend([T_CS!("\\lx@author@after"), T_BEGIN!()]);
368    state.extend(author.unlist_ref().iter().copied());
369    state.push(T_END!());
370    let state = Tokens::new(state);
371    Ok(Tokens::new([author_call(label, author)?.unlist(), state.unlist()].concat()))
372  });
373
374  // Robust, as authblk.sty:148's `\newcommand\affil[2][]` is (`\@protected@testopt`): an `\affil` inside
375  // `\author` survives titling's `\protected@xdef\theauthor{\@author}` (witness 2605.03796, whose body was lost).
376  // An unlabelled `\affil` is every author's given since the last one (`annotate=run:N`, [`affil_run`]): its note is
377  // `\AB@thenote`, which only the next `\author` after an `\affil` steps (authblk.sty:115-116), so consecutive `\affil`s
378  // (`\author{Ann Able and Bob Baker}\affil{Univ X}\affil{Univ Y}`) are all printed under the same authors; an author
379  // of the group with an affiliation already (a `\\` line under the name) is passed over, as the run does. One that
380  // opens with a hand-written mark a queued author carries too (`\author{Ann Able$^{1}$ and Bob Baker$^2$}\affil{$^1$Univ
381  // X}\affil{$^2$Univ Y}`; 1507.08653, 2510.01944) goes to the authors showing its mark, as a marked affiliation list's
382  // lines do (`\lx@add@affiliation@marked`) — beyond authblk, whose `\AB@thenote` prints it under the whole block
383  // (OXIDIZED_DESIGN #765). A footnote symbol (`$^*$Equal contribution`) or a mark no author carries (a typo, an author
384  // without marks after marked ones) keeps authblk's run: a label no one requests would go to a nameless author.
385  // LOCKED, as the kernel's `\title`/`\author` are (sect05.rs): a document's, class's or package's own `\affil` is
386  // refused, and its body still runs first for its side effects, its output discarded (`\lx@replay@class@setter`,
387  // OXIDIZED_DESIGN #482), so the affiliation stays the binding's contact. The JOSS template's copy of authblk's own
388  // `\affil` code (paper_joss.tex:162-182, a `\renewcommand{\affil}[2][]` that differs from authblk.sty:148-167 only in
389  // typography) put every affiliation in `\AB@authors`, which nothing prints here (2504.08081, 2310.11002, 2107.08407,
390  // 2507.13341, 2604.08855); and so did the raw classes that copy it after `\RequirePackage{authblk}`, oasics.cls:610
391  // (1709.00722) and eurocg25.cls (2503.02336), dagman.cls:696 the same (1701.09007; lipics-v2016.cls:611 is read as
392  // its binding's stand-in copy, which declines it unrecorded: 1603.00928). A redefinition that takes arguments is refused at
393  // every origin (state.rs `install_definition`, OXIDIZED_DESIGN #500); a binding that loads authblk and defines its
394  // own `\affil` loads unlocked and keeps it (lni_cls.rs). OXIDIZED_DESIGN_DIVERGENCES #959.
395  DefMacro!("\\affil OptionalSemiverbatim {}", sub[(label, affil)] {
396    let label = label.unwrap_or_default();
397    let replay = Invocation!(T_CS!("\\lx@replay@class@setter"), vec![
398      Some(Tokens::new(Explode!("affil"))),
399      Some(label.clone()),
400      Some(affil.clone())
401    ]);
402    // (a mark glued to a footnote symbol labels the line by its number, as a name's asks for it apart)
403    let affil = glued_line_marks_as_marks(affil, true);
404    let run = affil_run();
405    // (the label printed as written before the line, `\AB@affilnote`, authblk.sty:142: the mark a legend's note shows,
406    // `_labelprints`, OXIDIZED_DESIGN_DIVERGENCES #961)
407    let call = if !label.is_empty() {
408      Invocation!("\\lx@add@contact[role=affiliation,annotate=1,label={#1},_labelprints=true]{#2}", vec![
409        Some(label.clone()),
410        Some(affil)
411      ])
412    } else if opens_with_a_mark(&affil) && queued_authors_carry_entry_mark(&affil)? {
413      Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![None, Some(affil)])
414    } else if let Some((unmarked, marked)) = unmarked_before_carried_marks(&affil)? {
415      let mut calls = Vec::new();
416      for piece in unmarked {
417        calls.extend(
418          Invocation!("\\lx@add@contact[role=affiliation,label=addresses:block]{#1}", vec![Some(piece)])
419            .unlist(),
420        );
421      }
422      calls.extend(
423        Invocation!(T_CS!("\\lx@add@affiliation@marked"), vec![None, Some(marked)]).unlist(),
424      );
425      Tokens::new(calls)
426    } else if run > 0 {
427      Invocation!("\\lx@add@contact[role=affiliation,annotate=run:#1]{#2}", vec![
428        Some(Tokens::new(Explode!(run.to_string()))),
429        Some(affil)
430      ])
431    } else {
432      // (no author queued as it expands — inside `\author{Ann \affil{Univ X}}`, read at the flush, or after `\maketitle`
433      // (2605.01689): the run uncapped, as Perl's `annotate=new` (authblk.sty.ltxml:57) reaches them; a cap of 0
434      // attached it to no one, 77n r3 review)
435      Invocation!("\\lx@add@contact[role=affiliation,annotate=run]{#1}", vec![Some(affil)])
436    };
437    Ok(Tokens::new([
438      replay.unlist(),
439      call.unlist(),
440      Invocation!(T_CS!("\\lx@AB@affil@state"), vec![Some(label)]).unlist(),
441    ]
442    .concat()))
443  }, robust => true, locked => true);
444  Ok(())
445}

447/// The number of creators an unlabelled `\affil` is printed under at most: the ones queued since the last `\affil`
448/// before them, which share its `\AB@thenote` (authblk.sty:115-116 step it at the first `\author` after an `\affil`)
449/// — the same group as the `\affil` before when no author came between (`\author{Ann}\author{Bob}\affil{X}\affil{Y}`:
450/// both under both). Counted here, as every `\affil` passes: a mark-picked one too, whose affiliation reaches its
451/// authors only once their marks are matched, so the run alone would take its authors for new ones
452/// (`\author{Ann$^1$}\affil{$^1$X}\author{Bob}\affil{$^2$Y}` printed Y under Ann as well, 77n r2 review); 0 before
453/// any author.
454fn affil_run() -> usize {
455  let count = queued_creator_count();
456  let (before, run) = match lookup_value("lx_AB_affil_run") {
457    Some(Stored::String(last)) => with(last, |s| {
458      let (before, run) = s.split_once('|').unwrap_or_default();
459      (before.parse().unwrap_or(0), run.parse().unwrap_or(0))
460    }),
461    _ => (0, 0),
462  };
463  // (a flush that cleared the queue since starts the count anew)
464  let run = match count.checked_sub(before) {
465    Some(0) => run,
466    Some(new) => new,
467    None => count,
468  };
469  assign_value(
470    "lx_AB_affil_run",
471    Stored::String(pin(s!("{count}|{run}"))),
472    Some(Scope::Global),
473  );
474  run
475}

477/// An affiliation whose first pieces, parted by `\quad`, no mark opens and a later piece a mark a queued author carries
478/// opens: those pieces, and the rest. `\textsuperscript{}~Univ. of California \quad \textsuperscript{2}~Northwestern
479/// University \quad …` (2505.14999): an empty mark is no mark (`drop_empty_marks`, user 2026-10-20), so the first piece is
480/// no one's by a mark, and it is every author's by nothing: the block's, as an unmarked line among marked ones is (user
481/// ruling R12), not `Luo$^2$`'s, who shows the mark of his own.
482fn unmarked_before_carried_marks(affil: &Tokens) -> Result<Option<(Vec<Tokens>, Tokens)>> {
483  let pieces = split_tokens_delimited(affil.clone(), vec![
484    T_CS!("\\quad").into(),
485    T_CS!("\\qquad").into(),
486  ]);
487  let mut first = None;
488  for (at, (_, piece)) in pieces.iter().enumerate() {
489    if opens_with_a_mark(piece) && queued_authors_carry_entry_mark(piece)? {
490      first = Some(at);
491      break;
492    }
493  }
494  let Some(first) = first.filter(|first| *first > 0) else {
495    return Ok(None);
496  };
497  // (no mark on them, an empty one aside: a legend or an entry a mark leads is the list's, `$^{*}$Corresponding author \\
498  // $^{1}$Univ X \quad $^{3}$Univ Z`, 2505.00989)
499  if pieces[..first]
500    .iter()
501    .any(|(_, piece)| author_mark_position(&drop_empty_marks(piece.clone())).is_some())
502  {
503    return Ok(None);
504  }
505  let mut marked: Vec<Token> = Vec::new();
506  for (at, (delimiter, piece)) in pieces.iter().enumerate().skip(first) {
507    if at > first {
508      marked.extend(delimiter.iter().copied());
509    }
510    marked.extend(piece.unlist_ref().iter().copied());
511  }
512  // (each alone, its empty mark no mark of a marked list, what leads it no text)
513  let unmarked = pieces[..first]
514    .iter()
515    .map(|(_, piece)| {
516      let piece = drop_empty_marks(piece.clone());
517      let v = piece.unlist_ref();
518      Tokens::new(v[leading_unprinted_end(v)..].to_vec())
519    })
520    .filter(|piece| !piece.is_empty() && !prints_nothing(piece))
521    .collect();
522  Ok(Some((unmarked, Tokens::new(marked))))
523}

525/// Whether an affiliation opens with a superscript mark, `$^{1}$` or `\textsuperscript{1}`, after what prints nothing
526/// before it: spaces, spacing and declarations (`leading_unprinted_end`, as an author block's lines are read: 2605.11124's
527/// `\affil{\vspace{-1.05em}$^2$…}` went to every author since the last `\affil`, 77n r5 review).
528fn opens_with_a_mark(affil: &Tokens) -> bool {
529  let v = affil.unlist_ref();
530  let mut i = leading_unprinted_end(v);
531  while v.get(i) == Some(&T_SPACE!()) {
532    i += 1;
533  }
534  match v.get(i) {
535    Some(t) if *t == T_CS!("\\textsuperscript") => true,
536    Some(t) if t.get_catcode() == Catcode::MATH => v.get(i + 1) == Some(&T_SUPER!()),
537    _ => false,
538  }
539}
