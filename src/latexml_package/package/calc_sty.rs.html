8// calc.sty expression parser — ported from Perl calc.sty.ltxml
9// Grammar:
10//   <expression> -> <term> ((+|-) <term>)*
11//   <term>       -> <value> ((*|/) <value>)*  (where multiply/divide factor is float-aware)
12//   <value>      -> <literal> | \real{...} | \ratio{...}{...}
13//                 | \minof{...}{...} | \maxof{...}{...}
14//                 | \widthof{...} | ( <expression> )
15
16/// Internal representation for calc values: either an integer-based RegisterValue
17/// or a floating-point factor (from \real or \ratio).
18enum CalcValue {
19  Reg(RegisterValue),
20  Flt(f64),
21}
22
23impl CalcValue {
24  fn into_regval(self, expr_type: &str) -> RegisterValue {
25    match self {
26      CalcValue::Reg(rv) => rv,
27      CalcValue::Flt(f) => {
28        if expr_type == "Number" {
29          RegisterValue::Number(Number::new(f as i64))
30        } else {
31          RegisterValue::Glue(Glue::new(f as i64))
32        }
33      },
34    }
35  }
36}

38/// calc.sty:51-56 `\calc@assign@generic`: evaluate `expression` as `kind` and
39/// assign it to the register `variable` names (silently nothing when it names
40/// none, as `\setlength` here). What calc could not parse (after it reported
41/// the first token) stays in the input, read after the assignment.
42fn calc_assign(variable: ArgWrap, expression: Tokens, kind: &str) -> Result<()> {
43  if let ArgWrap::RegisterDefinition(dbox) = variable {
44    let (rtoken, params) = *dbox;
45    if let Some(defn) = rtoken.to_register() {
46      let stated = expression.clone();
47      let (value, tail) = evaluate(kind, expression)?;
48      // What a length's assignment states, for a float's width read through the register (the panel grid rule,
49      // `note_setlength`).
50      if kind != "Number" && params.is_empty() {
51        engine::latex_constructs::note_setlength(
52          &defn.get_cs_name(),
53          &stated,
54          value.clone().value_of(),
55        );
56      }
57      defn.set_value(value, None, params);
58      unread_vec(tail);
59    }
60  }
61  Ok(())
62}

64/// Evaluate the calc expression `tokens` as `expr_type` — calc.sty:51-53
65/// `\calc@open(#4!`, the whole argument — and return the value with whatever
66/// followed the expression. A token that is not an operator, `)`, `\relax` or
67/// the end is reported by [`calc_post_scan`] and consumed; the rest is the
68/// returned tail, which calc leaves in the input (OXIDIZED_DESIGN #317).
69fn evaluate(expr_type: &str, tokens: Tokens) -> Result<(RegisterValue, Vec<Token>)> {
70  read_braced(tokens, || {
71    let value = read_expression_body(expr_type)?;
72    calc_post_scan()?;
73    Ok(value)
74  })
75}

77/// A nested calc expression (`\ratio`, `\minof`/`\maxof` operands, a

79/// and has no input to return to.
80fn read_expression(expr_type: &str, tokens: Tokens) -> Result<RegisterValue> {
81  Ok(evaluate(expr_type, tokens)?.0)

84/// calc.sty:144-160 `\calc@post@scan`: after a term calc expects `+ - * / )`,
85/// `\relax` (skipped) or the expression's end; anything else goes to
86/// calc.sty:281-284 `\calc@error`, which reports "`x' invalid at this point"
87/// and consumes it (its `#1`). `A\hspace{1em\foo}B` with `\def\foo{x}` and
88/// calc: pdflatex reports the `x`.
89///
90/// A package may take calc's error over: picture.sty:79-120 `\let`s
91/// `\calc@error` to a handler that accepts the `\unitlength` after a LENGTH
92/// (`\setlength\dimen@{#1\unitlength}`, so a picture coordinate may be `1cm`),
93/// gobbling the rest of the expression to calc's `!`, and passes any other
94/// token to calc's own (`\PcOrg@calc@error`, :88-107). That handler runs on
95/// calc.sty's internals, which this binding does not have, so its decision is
96/// taken here: under a redefined `\calc@error` a `\unitlength` ends the
97/// expression silently (circledsteps' `\oval(\csteps@XLength,…)` under
98/// picture.sty: 2605.09094, 2605.10684); anything else is reported.
99fn calc_post_scan() -> Result<()> {
100  while let Some(token) = read_x_token(None, false, None)? {
101    // An undefined control sequence was reported and stubbed as it was
102    // expanded; TeX discards it (tex.web §370) and calc never sees it:
103    // `\hspace{6\@p@t}` with USG.cls's `\@p@t` missing (2605.25073, 10 → 242
104    // errors in the first cut), `\setlength{\itemsep}{\bibitemsep}` (2605.08378).
105    if token.get_catcode() == Catcode::SPACE
106      || token.defined_as(&TOKEN_RELAX)
107      || is_error_stub(&token)
108    {
109      continue;
110    }
111    let own = x_equals(&T_CS!("\\calc@error"), &T_CS!("\\lx@calc@error"));
112    if !own && x_equals(&token, &T_CS!("\\unitlength")) {
113      take_rest_of_mouth();
114    } else {
115      report_invalid(&token)?;
116    }
117    break;
118  }
119  Ok(())
120}

123fn report_invalid(token: &Token) -> Result<()> {
124  Error!(
125    "unexpected",
126    token.stringify(),
127    s!("`{}' invalid at this point", Tokens!(*token)),
128    "calc expected to see one of: + - * / )"
129  );
130  Ok(())
131}

133/// The calc `<expression> -> <term> ((+|-) <term>)*` loop, reading from the
134/// CURRENT gullet context. [`read_expression`] wraps it with a mouth over a
135/// captured token list; the base-dimension seam (installed below) calls it
136/// directly on the live stream to evaluate an inline `( … ) * \real{…}` width.
137fn read_expression_body(expr_type: &str) -> Result<RegisterValue> {
138  let mut result = read_term(expr_type)?;
139  while let Some(op) = read_keyword(&["+", "-"])? {
140    let term2 = read_term(expr_type)?;
141    result = if op == "+" {
142      result.add(term2)
143    } else {
144      result.subtract(term2)
145    };
146  }
147  Ok(result)
148}

150fn read_term(expr_type: &str) -> Result<RegisterValue> {
151  skip_spaces()?;
152  let mut factor = read_value(expr_type)?.into_regval(expr_type);
153  while let Some(op) = read_keyword(&["*", "/"])? {
154    let factor2 = read_value("Number")?;
155    factor = match factor2 {
156      CalcValue::Flt(f) => apply_float_op(&factor, &op, f),
157      CalcValue::Reg(rv) => {
158        if op == "*" {
159          factor.multiply(rv)
160        } else {
161          factor.divide(rv)
162        }
163      },
164    };
165  }
166  Ok(factor)
167}

169/// Apply a float multiply/divide to a RegisterValue, preserving Glue components
170fn apply_float_op(val: &RegisterValue, op: &str, f: f64) -> RegisterValue {
171  match val {
172    RegisterValue::Glue(g) => {
173      if op == "*" {
174        RegisterValue::Glue(Glue {
175          skip:  (g.skip as f64 * f) as i64,
176          plus:  g.plus.map(|p| (p as f64 * f) as i64),
177          pfill: g.pfill,
178          minus: g.minus.map(|m| (m as f64 * f) as i64),
179          mfill: g.mfill,
180        })
181      } else {
182        let div = if f == 0.0 { 1.0 } else { f };
183        RegisterValue::Glue(Glue {
184          skip:  (g.skip as f64 / div) as i64,
185          plus:  g.plus.map(|p| (p as f64 / div) as i64),
186          pfill: g.pfill,
187          minus: g.minus.map(|m| (m as f64 / div) as i64),
188          mfill: g.mfill,
189        })
190      }
191    },
192    RegisterValue::Number(n) => {
193      let v = n.value_of() as f64;
194      if op == "*" {
195        RegisterValue::Number(Number::new((v * f) as i64))
196      } else {
197        let div = if f == 0.0 { 1.0 } else { f };
198        RegisterValue::Number(Number::new((v / div) as i64))
199      }
200    },
201    RegisterValue::Dimension(d) => {
202      let v = d.value_of() as f64;
203      if op == "*" {
204        RegisterValue::Dimension(Dimension::new((v * f) as i64))
205      } else {
206        let div = if f == 0.0 { 1.0 } else { f };
207        RegisterValue::Dimension(Dimension::new((v / div) as i64))
208      }
209    },
210    _ => val.clone(),
211  }
212}

334/// box mode. The mode matters: mhchem's `\mhchem@minispace` (mhchem.sty:
335/// 2898-2904) runs `\setlength{…}{…\widthof{${}^8_8$}…}` INSIDE an open
336/// `\ensuremath` (:2781, prescripts `\ce{^{227}Th}`), and a bare `digest`
337/// inherits the ambient `IN_MATH`, so the argument's own `$` tried to close
338/// the ENCLOSING math frame (`\lx@end@inline@math … Attempt to end mode
339/// math`, ×54 in mhchem/mhchem). Witness: mhchem/mhchem 67 → 13; guard
340/// `perfect_kernel_batch50::calc_widthof_in_math_measures_own_box`.
341fn digest_measured_box(arg: Tokens) -> Result<Digested> {
342  with_own_adjust_queue(|| {
343    begin_mode("restricted_horizontal")?;
344    let result = digest(arg);
345    end_mode("restricted_horizontal")?;
346    result
347  })

350fn read_value(expr_type: &str) -> Result<CalcValue> {
351  skip_spaces()?;
352  let peek = read_x_token(None, false, None)?;
353  // A braced group is calc.sty:88-89 `\calc@pre@scan#1`'s undelimited argument
354  // after `\romannumeral` expanded the first token: one level of braces goes
355  // and the content is tested as written (`\ifx(#1`, `\ifx\widthof#1`), then
356  // read — `\def\a{{\a}}` ends in "Missing number", as in TeX, not a loop.
357  // mhchem.sty:1020 `\__mhchem_arrow_options_minLength:n` = `{2em}` reaches
358  // `\makebox[#7]` (:1239) and `\__mhchem_arrow_base:` as `{2em}-1em`.
359  let peek = match peek {
360    Some(open) if open.get_catcode() == Catcode::BEGIN => {
361      let group = read_balanced(ExpansionLevel::Off, false, false)?;
362      unread(group);
363      read_token()?
364    },
365    other => other,
366  };
367  let peek = match peek {
368    Some(t) => t,
369    None => {
370      if expr_type == "Number" {
371        return Ok(CalcValue::Reg(RegisterValue::Number(Number::new(0))));
372      } else {
373        return Ok(CalcValue::Reg(RegisterValue::Glue(Glue::new(0))));
374      }
375    },
376  };
377  // \widthof{...} — Perl calc.sty.ltxml L139-143
378  if peek == T_CS!("\\widthof") {
379    let arg = read_arg(ExpansionLevel::Off)?;
380    let box_result = digest_measured_box(arg)?;
381    if expr_type == "Number" {
382      Error!(
383        "unexpected",
384        "\\widthof",
385        format!("\\widthof not expected here (reading {expr_type})")
386      );
387    }
388    let width = box_result
389      .get_width(None)?
390      .unwrap_or_else(|| RegisterValue::Dimension(Dimension::new(0)));
391    return Ok(CalcValue::Reg(width));
392  }
393  // \heightof{...} — Perl calc.sty.ltxml L144-148
394  if peek == T_CS!("\\heightof") {
395    let arg = read_arg(ExpansionLevel::Off)?;
396    let box_result = digest_measured_box(arg)?;
397    if expr_type == "Number" {
398      Error!(
399        "unexpected",
400        "\\heightof",
401        format!("\\heightof not expected here (reading {expr_type})")
402      );
403    }
404    let height = box_result
405      .get_height()
406      .unwrap_or_else(|| RegisterValue::Dimension(Dimension::new(0)));
407    return Ok(CalcValue::Reg(height));
408  }
409  // \depthof{...} — Perl calc.sty.ltxml L149-153
410  if peek == T_CS!("\\depthof") {
411    let arg = read_arg(ExpansionLevel::Off)?;
412    let box_result = digest_measured_box(arg)?;
413    if expr_type == "Number" {
414      Error!(
415        "unexpected",
416        "\\depthof",
417        format!("\\depthof not expected here (reading {expr_type})")
418      );
419    }
420    let depth = box_result
421      .get_depth()
422      .unwrap_or_else(|| RegisterValue::Dimension(Dimension::new(0)));
423    return Ok(CalcValue::Reg(depth));
424  }
425  // \totalheightof{...} — Perl calc.sty.ltxml L154-158
426  if peek == T_CS!("\\totalheightof") {
427    let arg = read_arg(ExpansionLevel::Off)?;
428    let box_result = digest_measured_box(arg)?;
429    if expr_type == "Number" {
430      Error!(
431        "unexpected",
432        "\\totalheightof",
433        format!("\\totalheightof not expected here (reading {expr_type})")
434      );
435    }
436    let height = box_result
437      .get_height()
438      .unwrap_or_else(|| RegisterValue::Dimension(Dimension::new(0)));
439    let depth = box_result
440      .get_depth()
441      .unwrap_or_else(|| RegisterValue::Dimension(Dimension::new(0)));
442    return Ok(CalcValue::Reg(height.add(depth)));
443  }
444  // \real{<decimal>} — returns a Float factor for multiplication
445  if peek == T_CS!("\\real") {
446    let arg = read_arg(ExpansionLevel::Off)?;
447    let reader_mouth = Mouth::new("", None)?;
448    let float = reading_from_mouth(reader_mouth, move || {
449      unread(arg);
450      let f = read_float()?;
451      // Round as fixpoint, matching Perl calc.sty.ltxml line 164
452      Ok(kround(f.value_f64() * UNITY as f64) as f64 / UNITY as f64)
453    })?;
454    return Ok(CalcValue::Flt(float));
455  }
456  // \ratio{<dimen expr>}{<dimen expr>} — returns a Float factor
457  if peek == T_CS!("\\ratio") {
458    let arg_x = read_arg(ExpansionLevel::Off)?;
459    let arg_y = read_arg(ExpansionLevel::Off)?;
460    let x = read_expression("Glue", arg_x)?;
461    let y = read_expression("Glue", arg_y)?;
462    let y_val = y.value_of() as f64;
463    let y_pt = if y_val == 0.0 { 1.0 } else { y_val };
464    return Ok(CalcValue::Flt(x.value_of() as f64 / y_pt));
465  }
466  // \minof{<expr>}{<expr>}
467  if peek == T_CS!("\\minof") {
468    let arg_x = read_arg(ExpansionLevel::Off)?;
469    let arg_y = read_arg(ExpansionLevel::Off)?;
470    let x = read_expression(expr_type, arg_x)?;
471    let y = read_expression(expr_type, arg_y)?;
472    return Ok(CalcValue::Reg(x.smaller(y)));
473  }
474  // \maxof{<expr>}{<expr>}
475  if peek == T_CS!("\\maxof") {
476    let arg_x = read_arg(ExpansionLevel::Off)?;
477    let arg_y = read_arg(ExpansionLevel::Off)?;
478    let x = read_expression(expr_type, arg_x)?;
479    let y = read_expression(expr_type, arg_y)?;
480    return Ok(CalcValue::Reg(x.larger(y)));
481  }
482  // Parenthesized subexpression: ( <expression> ) — calc.sty:106-108
483  // `\calc@open(` opens a group that its matching `)` (`\calc@close`) ends, so
484  // the expression is read in place and parentheses nest. Perl (calc.sty.ltxml
485  // :183-184) and this port read up to the FIRST `)`: hexgame.sty:77
486  // `1.5\halfhexwidth*((\value{modulocounter})-1)` lost its `-1)`, silently while
487  // an argument's rest was dropped.
488  if peek == T_OTHER!("(") {
489    let value = read_expression_body(expr_type)?;
490    skip_spaces()?;
491    match read_x_token(None, false, None)? {
492      Some(close) if close == T_OTHER!(")") => {},
493      // Not a `)`: `calc_post_scan` reports it where the expression ends.
494      Some(other) => unread_one(other),
495      None => {},
496    }
497    return Ok(CalcValue::Reg(value));
498  }
499  // Else: literal value — put back token and read normally. calc.sty:110 reads it by assigning to a register
500  // (`\calc@numeric` = `\afterassignment\calc@post@scan \global\calc@A`), whose `<optional equals>` (tex.web
501  // §1224) takes one `=`: `\resizebox{=0.93\textwidth}{!}{…}` (2606.05909).
502  if peek != T_OTHER!("=") {
503    unread_one(peek);
504  }
505  if expr_type == "Number" {
506    Ok(CalcValue::Reg(RegisterValue::Number(read_number()?)))
507  } else {
508    Ok(CalcValue::Reg(RegisterValue::Glue(read_glue()?)))
509  }
510}

512/// Which box dimension `\widthof`/`\heightof`/`\depthof`/`\totalheightof` reports.
513#[derive(Clone, Copy)]
514enum BoxDim {
515  Width,
516  Height,
517  Depth,
518  TotalHeight,
519}
520
521/// Digest a `\widthof{…}`-family box argument and return the requested size.
522///
523/// The LaTeXML analogue of calc.sty's `\let\widthof\wd` (calc.sty L124-137):
524/// inside its expression scanner real calc typesets the argument into a box and
525/// takes `\wd`/`\ht`/`\dp` of it. Called from the base dimension reader's
526/// internal-dimension seam (installed below) so `\makebox[\widthof{X}]`,
527/// `\rule{\widthof{X}}{…}`, `\hspace{\widthof{X}}`, … — whose length is read by
528/// the base gullet dimension reader, NOT calc's own `read_value` scanner (which
529/// keeps its inline digest for `\setlength`) — resolve to the real size.
530/// Surpass-perl: OXIDIZED_DESIGN #115.
531fn measure_box_dim(toks: Tokens, kind: BoxDim) -> RegisterValue {
532  let zero = || RegisterValue::Dimension(Dimension::new(0));
533  // Measured in its own box mode, like `digest_measured_box` above: from a
534  // `<dimen>` read inside `align*` (mathtools `\mathmakebox[\widthof{$x$}]`,
535  // optidef) a bare digest's `$` closed the ENCLOSING math.
536  let box_result = match digest_measured_box(toks) {
537    Ok(b) => b,
538    Err(_) => return zero(),
539  };
540  match kind {
541    BoxDim::Width => box_result
542      .get_width(None)
543      .ok()
544      .flatten()
545      .unwrap_or_else(zero),
546    BoxDim::Height => box_result.get_height().unwrap_or_else(zero),
547    BoxDim::Depth => box_result.get_depth().unwrap_or_else(zero),
548    BoxDim::TotalHeight => {
549      let h = box_result.get_height().unwrap_or_else(zero);
550      let d = box_result.get_depth().unwrap_or_else(zero);
551      h.add(d)
552    },
553  }
554}

602  set_internal_dimension_fn(Rc::new(|tok: &Token| {
603    // calc infix expression opening with `(` — the base dimension reader
604    // (read_dimension/read_glue) cannot parse it and would warn "Missing number
605    // (Dimension), treated as zero". The canonical trigger is Pandoc's default
606    // relative-width table column `p{(\columnwidth - N\tabcolsep) * \real{X}}`:
607    // an unparsed width collapses the p{} column to a zero-width cell whose text
608    // wraps one character per line ("a river of characters").
609    //
610    // Real LaTeX defers this length: `array.sty`'s `\@startpbox` (L189-191) keeps
611    // the `p{}` width as *tokens* and only later assigns `\setlength\hsize{#1}`,
612    // and calc.sty patches `\setlength` to run its scanner — so the expression is
613    // evaluated at box-setup time, never eagerly. LaTeXML has no such deferral: it
614    // must yield the width *value* to emit `<td>`'s `width=` attribute, so it reads
615    // the width eagerly here. We evaluate with calc's OWN parser (`read_value` ->
616    // `read_expression_body`, the same grammar `\setlength` uses), so the result
617    // matches calc — only the invocation point differs. The parser reads the whole
618    // expression off the LIVE stream and stops at the first non-(+|-|*|/) token, so
619    // it never over-consumes; scoped to a leading `(` (the exact Pandoc idiom), so
620    // nothing that parsed before is disturbed. `read_dimension` has already peeled
621    // any leading sign, so `-(…)` negates the evaluated result correctly.
622    //
623    // Surpass-perl (Perl 0.8.8 emits the same 0pt): OXIDIZED_DESIGN #141.
624    // arXiv/html_feedback#6909; witness 2606.08266.
625    if *tok == T_OTHER!("(") {
626      unread_one(*tok);
627      // "Glue" mirrors calc's `\setlength`; the cell width is a pure `<dimen>`, so
628      // coerce down (dropping any stretch/shrink) exactly as `read_internal_dimension`
629      // does for a register value.
630      let value: Dimension = read_expression_body("Glue")?.into();
631      return Ok(Some(RegisterValue::Dimension(value)));
632    }
633    let kind = if *tok == T_CS!("\\widthof") {
634      BoxDim::Width
635    } else if *tok == T_CS!("\\heightof") {
636      BoxDim::Height
637    } else if *tok == T_CS!("\\depthof") {
638      BoxDim::Depth
639    } else if *tok == T_CS!("\\totalheightof") {
640      BoxDim::TotalHeight
641    } else {
642      return Ok(None);
643    };
644    let toks = read_arg(ExpansionLevel::Off)?;
645    Ok(Some(measure_box_dim(toks, kind)))
646  }));
