1//! IEEEtrantools.sty — the standalone IEEE alignment package (IEEEeqnarray,
2//! IEEEeqnarraybox, …) for use with ANY class.
3//!
4//! Perl LaTeXML ships NO `IEEEtrantools.sty.ltxml` — it only binds the same
5//! machinery inside `IEEEtran.cls.ltxml`. So a plain `article` +
6//! `\usepackage{IEEEtrantools}` falls through to raw-loading IEEEtrantools.sty,
7//! whose raw `\halign`-based IEEEeqnarray breaks LaTeXML's alignment model in
8//! BOTH engines: an IEEEeqnarray row that begins with an empty cell (a leading
9//! `&`, e.g. `\nonumber\\ & & +\beta\ldots`) raises
10//! `\halign Attempt to end mode restricted_horizontal` and cascades
11//! `_`/`^ can only appear in math mode`, mangling the equation. Perl fails the
12//! same way. Witness: /home/deyan/Downloads/ieee_eqn_bug (main_arXiv.tex L554);
13//! minimal reproducer docs/reproducers/ieeeeqnarray_leading_empty_cell.tex.
14//!
15//! This binding maps the IEEEeqnarray family onto LaTeXML's NATIVE alignment
16//! (`\eqnarray`'s, over the columns of the specification), which handles
17//! leading-empty cells correctly — so we surpass Perl here. The IEEEtran class
18//! binding (`ieeetran_cls.rs`, Perl IEEEtran.cls.ltxml L242-332) shares it:
19//! [`define_ieeeeqnarray`].

434/// The IEEEeqnarraybox family — a variant of `\array` (Perl IEEEtran.cls.ltxml L315-332) over the IEEEtrantools
435/// column types (IEEEtrantools.sty:2194 `\@IEEEbuildpreamble`). `\ifmmode` dispatches to the math (m) or text (t)
436/// form.
437pub fn define_eqnarraybox() -> Result<()> {
438  RawTeX!(
439    r"\def\IEEEeqnarraybox{\ifmmode\def\@tempa{\let\endIEEEeqnarraybox\endIEEEeqnarrayboxm\IEEEeqnarrayboxm}\else\def\@tempa{\let\endIEEEeqnarraybox\endIEEEeqnarrayboxt\IEEEeqnarrayboxt}\fi\@tempa}"
440  );
441  // IEEEtrantools.sty:2166-2173: `[<decl>][<pos>][<width>]{<cols>}` (Perl takes `{<cols>}` alone, so a box written
442  // with its options read `[` as the column specification).
443  // The math form in text (IEEEtrantools.sty:2143, `\@IEEEeqnarrayboxHBOXSWfalse`): TeX sets its `\vbox` of math cells in
444  // the paragraph, where the alignment here needs math around it — opened as the text form opens it (:2201
445  // `\hbox\bgroup$`), and closed at the end by `\lx@IEEE@box@end`, which each box sets at the level its end reads it.
446  // Before, a text-mode `{IEEEeqnarrayboxm}` opened none, and its Math took the following text (repro
447  // alignment-bindings/ieeeeqnarrayboxm_in_text_keeps_its_math; guard
448  // `perfect_kernel_batch77v::ieeeeqnarrayboxm_in_text_keeps_its_math`).
449  DefMacro!("\\IEEEeqnarrayboxm OptionalMatch:* [][][] {}", sub[(star, decl, pos, width, cols)] {
450    let mut out = TokenizeInternal!(
451      r"\ifmmode\let\lx@IEEE@box@end\relax\else\lx@begin@inline@math\let\lx@IEEE@box@end\lx@end@inline@math\fi"
452    ).unlist();
453    out.extend(eqnarraybox_begin(star.is_some(), [decl, pos, width], &cols, false).unlist());
454    Ok(Tokens::new(out))
455  });
456  DefMacro!(
457    "\\endIEEEeqnarrayboxm",
458    "\\lx@end@alignment\\@end@array\\lx@IEEE@box@end"
459  );
460  DefMacro!("\\IEEEeqnarrayboxt OptionalMatch:* [][][] {}", sub[(star, decl, pos, width, cols)] {
461    Ok(eqnarraybox_begin(star.is_some(), [decl, pos, width], &cols, true))
462  });
463  DefMacro!(
464    "\\endIEEEeqnarrayboxt",
465    "\\lx@end@alignment\\@end@array\\lx@end@inline@math"
466  );
467  // The array over the specification's column types, its cells math already: a math column's `$` comes out, a text
468  // column's goes around it (`math_cell_column`). `[<pos>]` (`t`, `c`, `b`) attaches the box as an array's.
469  DefPrimitive!("\\@IEEEeqnarraybox@bindings [] {}", sub[(pos, cols)] {
470    let columns: Vec<Cell> = ieee_columns(&Expand!(cols))?
471      .into_iter()
472      .map(|(before, after)| {
473        let (before, after) = math_cell_column(before, after);
474        Cell { before: Some(before), after: Some(after), ..Cell::default() }
475      })
476      .collect();
477    set_column_count(columns.len());
478    let template = Template::new(TemplateConfig {
479      columns: Some(columns),
480      ..TemplateConfig::default()
481    });
482    // Not `omit_is_text`: the rule rows' `\omit` spans (`\IEEEeqnarrayrulerow`) keep their leader cells (67x guards).
483    math_array_bindings(pos.as_ref(), template, false)?;
484    AssignValue!("IEEEeqnarray@mathcells" => true);
485  });
486  // `{<star>}{<options>{<cols>}}`, as written, for the reversion.
487  DefConstructor!("\\@@IEEE@array Undigested Undigested DigestedBody", "#3",
488    before_digest => { bgroup(); },
489    reversion => "\\begin{IEEEeqnarraybox#1}#2#3\\end{IEEEeqnarraybox#1}");
490  // IEEEtrantools.sty:2145-2159: the starred environments are the boxes without the `\jot` row padding (2609.32652
491  // `{IEEEeqnarraybox*}`): the same alignment here, the star kept for the reversion.
492  RawTeX!(
493    r"\expandafter\def\csname IEEEeqnarraybox*\endcsname{\IEEEeqnarraybox*}
494\expandafter\def\csname endIEEEeqnarraybox*\endcsname{\endIEEEeqnarraybox}
495\expandafter\def\csname IEEEeqnarrayboxm*\endcsname{\IEEEeqnarrayboxm*}
496\expandafter\let\csname endIEEEeqnarrayboxm*\endcsname\endIEEEeqnarrayboxm
497\expandafter\def\csname IEEEeqnarrayboxt*\endcsname{\IEEEeqnarrayboxt*}
498\expandafter\let\csname endIEEEeqnarrayboxt*\endcsname\endIEEEeqnarrayboxt"
499  );
500  Ok(())
501}

509/// The alignment an IEEEeqnarraybox opens over the column specification `cols`; `\@@IEEE@array` keeps `cols` whole
510/// for the reversion.
511fn eqnarraybox_begin(
512  star: bool,
513  options: [Option<Tokens>; 3],
514  cols: &Tokens,
515  text: bool,
516) -> Tokens {
517  let mut out = Vec::new();
518  if text {
519    out.push(T_CS!("\\lx@begin@inline@math"));
520  }
521  out.push(T_CS!("\\@IEEEeqnarraybox@bindings"));
522  if let [_, Some(pos), _] = &options {
523    out.push(T_OTHER!("["));
524    out.extend_from_slice(pos.unlist_ref());
525    out.push(T_OTHER!("]"));
526  }
527  out.push(T_BEGIN!());
528  out.extend_from_slice(cols.unlist_ref());
529  out.push(T_END!());
530  out.push(T_CS!("\\@@IEEE@array"));
531  out.push(T_BEGIN!());
532  if star {
533    out.push(T_OTHER!("*"));
534  }
535  out.push(T_END!());
536  out.push(T_BEGIN!());
537  // The options up to the last one given, an earlier one omitted written `[]` as IEEEtrantools reads it.
538  let given = options
539    .iter()
540    .rposition(Option::is_some)
541    .map_or(0, |last| last + 1);
542  for option in &options[..given] {
543    out.push(T_OTHER!("["));
544    if let Some(option) = option {
545      out.extend_from_slice(option.unlist_ref());
546    }
547    out.push(T_OTHER!("]"));
548  }
549  out.push(T_BEGIN!());
550  out.extend_from_slice(cols.unlist_ref());
551  out.push(T_END!());
552  out.push(T_END!());
553  out.push(T_CS!("\\lx@begin@alignment"));
554  Tokens::new(out)
555}

557/// The material around each column's cells — `(\@IEEEeqnarraycolPRE<type>, \@IEEEeqnarraycolPOST<type>)` — of a
558/// column specification, fully expanded as IEEEtrantools.sty:2280 `\edef`s it (macro containers) and read as
559/// `\@@IEEEbuildpreamble` (:2289-2310) does: a letter, or a braced name starting with one (`{vv}`), is a column of
560/// that type (`\@IEEEgetcoltype`, :2333-2355); a digit or one of the punctuation glues `! , : ; ' " . / ? * + -`
561/// (:2373-2422) spaces the columns and makes none, so `{,c/c/c,}` is three centred columns (the glues are not
562/// modelled). Anything else is the package's error: punctuation no glue names (:2418), a control sequence (:2352).
563fn ieee_columns(expanded: &Tokens) -> Result<Vec<(Tokens, Tokens)>> {
564  let mut columns = Vec::new();
565  let mut tokens = expanded.unlist_ref().iter();
566  while let Some(token) = tokens.next() {
567    match token.get_catcode() {
568      Catcode::LETTER => columns.push(ieee_column_type(&token.to_string())?),
569      Catcode::BEGIN => {
570        let mut depth = 1usize;
571        let mut name = String::new();
572        let mut starts_with_letter = None;
573        for inner in tokens.by_ref() {
574          match inner.get_catcode() {
575            Catcode::BEGIN => depth += 1,
576            Catcode::END => {
577              depth -= 1;
578              if depth == 0 {
579                break;
580              }
581            },
582            _ => {},
583          }
584          starts_with_letter.get_or_insert(inner.get_catcode() == Catcode::LETTER);
585          name.push_str(&inner.to_string());
586        }
587        if starts_with_letter == Some(true) {
588          columns.push(ieee_column_type(&name)?);
589        }
590      },
591      Catcode::SPACE | Catcode::END => {},
592      Catcode::OTHER => {
593        if !with(token.text, |c| {
594          c.len() == 1 && "!,:;'\"./?*+-0123456789".contains(c)
595        }) {
596          let what = token.to_string();
597          Error!(
598            "unexpected",
599            what,
600            s!(
601              "Invalid predefined inter-column glue type \"{what}\" in IEEEeqnarray column specifications. Using a default value of 0pt instead"
602            ),
603            "Only !,:;'\"./?*+ and - are valid predefined glue types in the IEEEeqnarray column specifications."
604          );
605        }
606      },
607      _ => {
608        let what = token.to_string();
609        Error!(
610          "unexpected",
611          what,
612          s!("Invalid character \"{what}\" in IEEEeqnarray column specifications"),
613          "Only letters, numerals and certain other symbols are allowed as IEEEeqnarray column specifiers."
614        );
615      },
616    }
617  }
618  Ok(columns)
619}

621/// The material around a cell of the column type `name` (IEEEtrantools.sty:1485 `\IEEEeqnarraydefcol`); an undefined
622/// type is the package's error and its default centred math column (:2362-2367 `\@IEEEgetcurcol`).
623fn ieee_column_type(name: &str) -> Result<(Tokens, Tokens)> {
624  let defined = |name: &str| -> Result<bool> {
625    Ok(lookup_definition(&T_CS!(s!("\\@IEEEeqnarraycolDEF{name}")))?.is_some())
626  };
627  let name = if defined(name)? {
628    name
629  } else {
630    Error!(
631      "unexpected",
632      name,
633      s!(
634        "Invalid column type \"{name}\" in IEEEeqnarray column specifications. Using a default centering column instead"
635      ),
636      "You must define IEEEeqnarray column types before use."
637    );
638    "@IEEEdefault"
639  };
640  let body = |part: &str| -> Result<Tokens> {
641    Ok(
642      match lookup_definition(&T_CS!(s!("\\@IEEEeqnarraycol{part}{name}")))? {
643        Some(defn) => match defn.get_expansion() {
644          Some(ExpansionBody::Tokens(body)) => body.clone(),
645          _ => Tokens!(),
646        },
647        None => Tokens!(),
648      },
649    )
650  };
651  Ok((body("PRE")?, body("POST")?))
652}

654/// A column's material for cells that are math already (the IEEEeqnarraybox's array, whose `$` leaves math,
655/// `\lx@dollar@in@mathmode`): the `$`s a math column enters math with come out, and a text column (`s t u`, the
656/// rules, `x`) is put in `$…$`, outside its `\hfil`s.
657fn math_cell_column(before: Tokens, after: Tokens) -> (Tokens, Tokens) {
658  let is_shift = |t: &Token| t.get_catcode() == Catcode::MATH;
659  if before
660    .unlist_ref()
661    .iter()
662    .chain(after.unlist_ref())
663    .any(is_shift)
664  {
665    let strip = |tokens: Tokens| {
666      Tokens::new(
667        tokens
668          .unlist()
669          .into_iter()
670          .filter(|t| !is_shift(t))
671          .collect(),
672      )
673    };
674    return (strip(before), strip(after));
675  }
676  let (mut lead, before) = split_hfil(before, true);
677  let (mut after, trail) = split_hfil(after, false);
678  lead.push(T_MATH!());
679  lead.extend(before);
680  after.push(T_MATH!());
681  after.extend(trail);
682  (Tokens::new(lead), Tokens::new(after))
683}

685/// `tokens` split at its leading `\hfil`s (`(hfils, rest)`) or at its trailing ones (`(rest, hfils)`).
686fn split_hfil(tokens: Tokens, leading: bool) -> (Vec<Token>, Vec<Token>) {
687  let hfil = T_CS!("\\hfil");
688  let mut tokens = tokens.unlist();
689  if leading {
690    let n = tokens.iter().take_while(|t| **t == hfil).count();
691    let rest = tokens.split_off(n);
692    (tokens, rest)
693  } else {
694    let n = tokens.iter().rev().take_while(|t| **t == hfil).count();
695    let hfils = tokens.split_off(tokens.len() - n);
696    (tokens, hfils)
697  }
698}

700/// The QED symbols and `{IEEEproof}` (IEEEtran.cls:5533-5556, IEEEtrantools.sty:2726-2747; Perl IEEEtran.cls.ltxml
701/// L194-222), shared with the IEEEtran class binding (`ieeetran_cls.rs`), whose class carries the same code.

705  // Sized as IEEEtran.cls:5533's `\mbox{\rule[0pt]{1.3ex}{1.3ex}}`: a size-less whatsit alone in an alignment cell

775  // same with the QED as the display's equation number (1903.06134, 2201.03502). Perl's `\IEEEQEDhere`
776  // (IEEEtran.cls.ltxml:193-196) took the top of the QED@stack an amsthm proof shares; Perl has no `\IEEEQEDhereeqn`.

807        .findnodes("ancestor-or-self::ltx:equation[1]", Some(&node))
808        .into_iter()
809        .next()
810      else {
811        return Ok(());
812      };
813      let tags = document.findnodes("ltx:tags", Some(&equation));
814      let Some(mut qed_tags) = tags.last().cloned() else {
815        return Ok(());
816      };
817      let Some(mut qed_tag) = document.findnodes("ltx:tag", Some(&qed_tags)).into_iter().next() else {
818        return Ok(());
819      };
820      // The QED's material as marked elements: its element children, or one `ltx:text` holding them when text nodes are
821      // among them.
822      let children = qed_tag.get_child_nodes();
823      let mut material: Vec<Node> = if children.iter().all(|child| child.get_type() == Some(NodeType::ElementNode)) {
824        children
825      } else {
826        let mut wrapper = document.open_element_at(&mut qed_tag, "ltx:text", None, None)?;
827        for mut child in children {
828          child.unlink_node();
829          if child.get_type() == Some(NodeType::TextNode) {
830            wrapper.append_text(&child.get_content()).ok();
831          } else {
832            wrapper.add_child(&mut child).ok();
833          }
834        }
835        document.close_element_at(&mut wrapper)?;
836        let _ = wrapper.remove_attribute("font");
837        vec![wrapper]
838      };
839      for element in material.iter_mut() {
840        element.set_attribute("_qed", "1").ok();
841      }
842      // The equation's own displayed tag, when it has one before ours.
843      let Some(number_tags) = tags.first().filter(|first| **first != qed_tags) else {
844        return Ok(());
845      };
846      let Some(mut number_tag) = document
847        .findnodes("ltx:tag[not(@role)]", Some(number_tags))
848        .into_iter()
849        .next()
850      else {
851        return Ok(());
852      };
853      let before = number_tag
854        .get_child_nodes()
855        .into_iter()
856        .find(|child| child.get_attribute("_qed").is_none());
857      for element in material.iter_mut() {
858        element.unlink_node();
859        match &before {
860          Some(before) => {
861            before.clone().add_prev_sibling(element).ok();
862          },
863          None => {
864            number_tag.add_child(element).ok();
865          },
866        }
867      }
868      qed_tags.unlink_node();
869    });
870  Ok(())
871}

969/// Whether the current row will be left unnumbered (IEEEtrantools' `\if@eqnsw` false): its own `retract`, or the
970/// numbering's on its preset number, unless it says `noretract`.
971fn ieee_row_retracts() -> bool {
972  with_value("EQUATION_NUMBERING", |numbering| {
973    with_value("EQUATIONROW_TAGS", |tags| {
974      let set = |v: Option<&Stored>| matches!(v, Some(&Stored::Bool(true)));
975      match (numbering, tags) {
976        (Some(Stored::HashStored(numbering)), Some(Stored::HashStored(tags))) => {
977          !set(tags.get("noretract"))
978            && (set(tags.get("retract"))
979              || (set(numbering.get("retract"))
980                && set(numbering.get("preset"))
981                && set(tags.get("preset"))))
982        },
983        _ => false,
984      }
985    })
986  })
