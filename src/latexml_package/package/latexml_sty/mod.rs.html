1386/// LuaTeX `scan_direction`: the argument of a direction primitive is one
1387/// expanded token — a direction primitive itself (the internal form,
1388/// `\pagedir\bodydir`) — or a three-letter keyword (`TLT`, `TRT`, `LTL`,
1389/// `RTT`); leading spaces are skipped as `scan_keyword` does. Nothing is kept:
1390/// TLT is the only direction this engine models.
1391fn scan_direction() -> Result<()> {
1392  const DIRECTION_PRIMITIVES: [&str; 6] = [
1393    "\\pagedir",
1394    "\\bodydir",
1395    "\\pardir",
1396    "\\textdir",
1397    "\\mathdir",
1398    "\\linedir",
1399  ];
1400  let mut first = None;
1401  while let Some(t) = read_x_token(None, false, None)? {
1402    if t.get_catcode() == Catcode::SPACE {
1403      continue;
1404    }
1405    first = Some(t);
1406    break;
1407  }
1408  let Some(first) = first else {
1409    return Ok(());
1410  };
1411  if first.get_catcode() == Catcode::CS && first.with_str(|s| DIRECTION_PRIMITIVES.contains(&s)) {
1412    return Ok(());
1413  }
1414  // The keyword form: `first` is its first letter; take the remaining two.
1415  // `scan_keyword` matches letters by character, not catcode (`\the\bodydir`
1416  // expands to OTHER-catcode `TLT`).
1417  let is_keyword_char = |t: &Token| {
1418    matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER)
1419      && t.with_str(|s| s.chars().all(|c| c.is_ascii_alphabetic()))
1420  };
1421  if !is_keyword_char(&first) {
1422    unread(Tokens::new(vec![first]));
1423    return Ok(());
1424  }
1425  for _ in 0..2 {
1426    match read_x_token(None, false, None)? {
1427      Some(t) if is_keyword_char(&t) => {},
1428      Some(t) => {
1429        unread(Tokens::new(vec![t]));
1430        break;
1431      },
1432      None => break,
1433    }
1434  }
1435  Ok(())
1436}
