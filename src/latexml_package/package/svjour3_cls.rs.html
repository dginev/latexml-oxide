11  // (svjour3.cls:42, :1414 `\columnsep{1.5cc}`), as pdflatex gives it: 495pt in two 238.964pt columns, where the

17  // What editions of svjour3.cls papers ship add (run 347: 42 svjour3 papers), a copy without one leaving it to the
18  // paper (`DefinitionOrigin::Fallback`):
19  // - the economics edition's `\JEL{codes}` (6 copies, v3.2 and v3.3, 1402.2599's :314-317 "JEL Classification", its
20  //   codes separated by `\and`), as `\subclass`: 1402.2599, 2110.08612, 2401.08594, 2404.18137, 2409.15978;
21  // - the 2023 edition's back-matter statements (3 copies, 2411.13175's :1133-1159, 1315-1317): `{funding}`/`{Funding}`
22  //   "Funding", `{dataavailability}`/`{data}` "Data Availability", `{conflict}`/`{Conflict}` "Conflict of interest",
23  //   each a run-in bold heading over its text (2307.15129, 2411.13175, 2001.01655, 2403.11828).
24  {
25    let _class = OriginGuard::new(DefinitionOrigin::Fallback);
26    DefMacro!("\\JELname", "JEL Classification");
27    DefMacro!(
28      "\\JEL{}",
29      "\\lx@add@classification[scheme=JEL,name={\\JELname: }]{\\def\\and{\\unskip\\ $\\cdot$ }#1}"
30    );
31    RawTeX!(r"\def\fundname{Funding}
32\def\dataname{Data Availability}
33\def\conflictname{Conflict of interest}
34\def\funding{\par\small\rmfamily\trivlist\if!\fundname!\item[]\else\item[\hskip\labelsep{\bfseries\fundname}]\fi}
35\def\endfunding{\endtrivlist}
36\newenvironment{Funding}{\begin{funding}}{\end{funding}}
37\def\dataavailability{\par\small\rmfamily\trivlist\if!\dataname!\item[]\else\item[\hskip\labelsep{\bfseries\dataname}]\fi}
