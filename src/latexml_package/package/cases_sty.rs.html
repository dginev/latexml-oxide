3/// Perl: cases.sty.ltxml — numcases and subnumcases environments
4/// Perl: numcasesBindings($lhs) — creates 3-column alignment for numcases
5fn numcases_bindings(lhs: Tokens) -> Result<()> {
6  use latexml_core::alignment::cell::Cell;
7  let col1 = Cell {
8    before: Some(Tokens!(
9      T_CS!("\\hfil"),
10      T_MATH!(),
11      T_CS!("\\lx@hidden@bgroup"),
12      T_CS!("\\displaystyle")
13    )),
14    after: Some(Tokens!(T_CS!("\\lx@hidden@egroup"), T_MATH!())),
15    ..Cell::default()
16  };
17  // cases.sty:228-233: `amsstyle` sets the right-hand side in `\textstyle` (`\numc@dispstyle`) and the explanation in
18  // math (`\numc@expstyle` = `$`, the column `\numc@expstyle\quad ##\unskip\numc@expstyle`, :81).
19  let amsstyle = lookup_bool("numcases_amsstyle");
20  let rhs_style = if amsstyle {
21    T_CS!("\\textstyle")
22  } else {
23    T_CS!("\\displaystyle")
24  };
25  let col2 = Cell {
26    before: Some(Tokens!(T_MATH!(), T_CS!("\\lx@hidden@bgroup"), rhs_style)),
27    after: Some(Tokens!(
28      T_CS!("\\lx@hidden@egroup"),
29      T_MATH!(),
30      T_CS!("\\hfil")
31    )),
32    ..Cell::default()
33  };
34  let col3 = if amsstyle {
35    Cell {
36      before: Some(Tokens!(T_MATH!(), T_CS!("\\lx@hidden@bgroup"))),
37      after: Some(Tokens!(
38        T_CS!("\\lx@hidden@egroup"),
39        T_MATH!(),
40        T_CS!("\\hfil")
41      )),
42      ..Cell::default()
43    }
44  } else {
45    Cell {
46      before: Some(Tokens!(T_CS!("\\lx@hidden@bgroup"))),
47      after: Some(Tokens!(T_CS!("\\lx@hidden@egroup"), T_CS!("\\hfil"))),
48      ..Cell::default()
49    }
50  };
51  use latexml_core::alignment::template::TemplateConfig;
52  let mut template = Template::new(TemplateConfig::default());
53  template.add_column(col1);
54  template.add_column(col2);
55  template.add_column(col3);
56
57  let mut attrs = string_map! { "class" => "ltx_eqn_numcases" };
58  // Perl (cases.sty.ltxml L82): colsep => LookupDimension('\arraycolsep')->multiply(2)
59  // LookupDimension (lookup_dimension_cs), not lookup_register, so a `\def`-ized
60  // `\arraycolsep` reads its body silently, matching Perl — lookup_register would
61  // emit a spurious `expected:register` warning (same class as the eqnarray fix).
62  let colsep_dim = lookup_dimension_cs("\\arraycolsep", false).unwrap_or_default();
63  let doubled = Dimension::new(colsep_dim.value_of() * 2);
64  attrs.insert(String::from("colsep"), doubled.to_attribute());
65  // Perl (cases.sty.ltxml L83-85): \jot ≠ default → rowsep on the equationgroup.
66  let cur_jot = lookup_dimension_cs("\\jot", false).unwrap_or_default();
67  if cur_jot.value_of()
68    != lookup_dimension_cs("\\lx@default@jot", false)
69      .unwrap_or_default()
70      .value_of()
71  {
72    attrs.insert(String::from("rowsep"), cur_jot.to_string());
73  }
74
75  // Not a display alignment: cases.sty:73-75 sets the `\halign` inside `\setbox\tw@\vbox\bgroup`, so material a
76  // document's `\halign` macro typesets before it is kept (pdflatex prints it above the cases, no error).
77  let numcases_properties = SymHashMap::default();
78  let alignment = Alignment::new(AlignmentConfig {
79    template:        Some(template),
80    open_container:  Rc::new(move |document, mut props| {
81      if let Ok(id_props) = ref_step_id("@equationgroup")
82        && let Some(id) = id_props.get("id")
83      {
84        props.insert(String::from("xml:id"), id.to_string());
85      }
86      // Perl: %attributes has class => 'ltx_eqn_numcases' which overrides
87      // the openContainer's default class. Use 'ltx_eqn_numcases' directly.
88      props
89        .entry(String::from("class"))
90        .or_insert_with(|| String::from("ltx_eqn_numcases"));
91      document
92        .open_element("ltx:equationgroup", Some(props), None)
93        .map(Some)
94    }),
95    close_container: Rc::new(|document| document.close_element("ltx:equationgroup")),
96    open_row:        Rc::new(|document, mut props| {
97      if let Some(Stored::HashStored(eq_props)) = remove_value("EQUATIONROW_PROPS")
98        && let Some(id) = eq_props.get("id")
99      {
100        props.insert(String::from("xml:id"), Stored::from(id.to_string()));
101      }
102      let tags_digested = props.remove("tags");
103      let str_props: HashMap<String, String> =
104        props.into_iter().map(|(k, v)| (k, v.to_string())).collect();
105      document.open_element("ltx:equation", Some(str_props), None)?;
106      if let Some(Stored::Digested(d)) = tags_digested {
107        document.absorb(&d, None)?;
108      }
109      Ok(())
110    }),
111    close_row:       Rc::new(|document| document.close_element("ltx:equation")),
112    open_column:     Rc::new(|document, props| {
113      document
114        .open_element("ltx:_Capture_", Some(props), None)
115        .map(Some)
116    }),
117    close_column:    Rc::new(|document| document.close_element("ltx:_Capture_")),
118    is_math:         true,
119    properties:      numcases_properties,
120    xml_attributes:  attrs,
121  });
122  assign_alignment(alignment, None);
123  let_i(&T_MATH!(), &T_CS!("\\lx@dollar@in@mathmode"), None);
124  // cases.sty sets the left-hand side once, in `\box\z@`, before its `\let\\\@eqncr` (cases.sty:73): a `\\` in it
125  // keeps its outer meaning and ends no row. We repeat it in each row, so `\lx@numcases@inlhs` marks the first cell as
126  // the left-hand side (a local flag, gone with the cell's group) and `\@numcases@newline` there is the outer `\\`
127  // (`\begin{numcases}{M_0=0\Rightarrow \\}`, 2202.06960: each row's left-hand side ended the row and re-inserted
128  // itself — the loop detector's Fatal, then an out-of-memory one). The mark goes before the left-hand side, where no
129  // argument it leaves open can take it (`\begin{numcases}` with no `{…}`: `#1` = `\mathrm`, 2412.05950).
130  let_i(&T_CS!("\\lx@numcases@outer@cr"), &T_CS!("\\\\"), None);
131  let mut lhs_expansion = vec![T_CS!("\\lx@numcases@inlhs")];
132  lhs_expansion.extend(lhs.unlist());
133  lhs_expansion.push(T_ALIGN!());
134  def_macro(
135    T_CS!("\\@numcases@LHS"),
136    None,
137    Tokens::new(lhs_expansion),
138    None,
139  )?;
140  Let!("\\\\", "\\@numcases@newline");
141  let_i(
142    &T_CS!("\\lx@alignment@row@before"),
143    &T_CS!("\\eqnarray@row@before"),
144    None,
145  );
146  let_i(
147    &T_CS!("\\lx@alignment@row@after"),
148    &T_CS!("\\eqnarray@row@after"),
149    None,
150  );
151  Ok(())
152}

185  // Sub-numbering: step parent equation, save counter, reset, redefine \theequation
186  // Perl: cases.sty.ltxml L57-64
187  DefPrimitive!("\\lx@numcases@subnumbering@begin", sub[_args] {
188    use latexml_core::binding::counter::dialect::reset_counter;
189    use latexml_core::mouth;
190    // Step the equation counter and get properties (id, refnum)
191    let eqn_props = ref_step_counter_by_meaning("equation")?;
192    // Expand \theequation to get the parent equation number text (e.g. "3")
193    let eqnum_toks = do_expand(T_CS!("\\theequation"))?;
194    let eqnum_str = eqnum_toks.to_string();
195    // Save current equation counter value
196    let saved = lookup_register("\\c@equation", Vec::new())?.map_or(0, |rv| {
197      match rv {
198        RegisterValue::Number(n) => n.0,
199        _ => 0,
200      }
201    });
202    assign_value("SAVED_EQUATION_NUMBER", Stored::Number(Number::new(saved)), Some(Scope::Global));
203    // Reset equation counter to 0 for sub-lettering
204    reset_counter(&T_OTHER!("equation"))?;
205    // Redefine \theequation to parent_number + \alph{equation} (e.g. "3a", "3b")
206    let new_theequation = format!("{}\\alph{{equation}}", eqnum_str);
207    def_macro(T_CS!("\\theequation"), None, mouth::tokenize_internal(TeXString::assembled(new_theequation)), None)?;
208    // Redefine \theequation@ID for xml:id generation (e.g. "S0.E3.\@equation@ID")
209    let id_str = eqn_props.iter().find_map(|(k, v)| {
210      if with(*k, |ks| ks == "id") { Some(v.to_string()) } else { None }
211    }).unwrap_or_default();
212    if !id_str.is_empty() {
213      let new_id_macro = format!("{}.\\@equation@ID", id_str);
214      def_macro(T_CS!("\\theequation@ID"), None, mouth::tokenize_internal(TeXString::assembled(new_id_macro)), None)?;
215    }
216  });
217  DefPrimitive!("\\lx@numcases@subnumbering@end", sub[_args] {
218    // Restore saved equation counter
219    if let Some(Stored::Number(n)) = lookup_value("SAVED_EQUATION_NUMBER") {
220      let _ = assign_register("\\c@equation", RegisterValue::Number(n), Some(Scope::Global), Vec::new());
221    }
222  });
223
224  // Perl cases.sty.ltxml:111 `DefMacro('\@numcases@newline[]', …)` reads the `[` test as an ordinary argument, so a
225  // row whose first cell is empty (`\\` + newline + `& $x$`) gave its `&` to the alignment before the row ended
226  // ("\lx@hidden@egroup Attempt to close a group that switched to mode math"). cases.sty:73 `\let\\\@eqncr`, whose
227  // `{\ifnum0=`}\fi` keeps the row's tabs from the `*`/`[` test (latex.ltx:15776-15777): read as
228  // `\lx@alignment@newline` reads it, then the left-hand side (KNOWN_PERL_ERRORS #1154; 2510.01717, 1709.02454).
229  DefMacro!("\\@numcases@newline", {
230    if lookup_bool("numcases_in_lhs") {
231      Tokens!(T_CS!("\\lx@numcases@outer@cr"))
232    } else {
233      engine::tex_tables::alignment_newline_then(true, &[T_CS!("\\@numcases@LHS")])?
234    }
235  });
236  DefPrimitive!("\\lx@numcases@inlhs", { assign_value("numcases_in_lhs", true, None); });
237  DefMacro!("\\@numcases@cr", "\\lx@alignment@cr\\@numcases@LHS");
238
239  DefConstructor!("\\@@numcases SkipSpaces DigestedBody", "#1",
240    before_digest => sub { bgroup(); },
241    after_construct => sub[document, _whatsit] {
242      let node = document.get_node();
243      if let Some(equationgroup) = node.get_last_child() {
244        rearrange_numcases(document, &equationgroup)?;
245      }
246    });
247
248  DefPrimitive!("\\end@numcases", sub[_args] { egroup()?; });

251fn rearrange_numcases(document: &mut Document, equationgroup: &Node) -> Result<()> {
252  use crate::engine::base_xmath::equationgroup_join_cols;
253  let equations: Vec<Node> = document.findnodes("ltx:equation", Some(equationgroup));
254  for equation in equations {
255    let cells: Vec<Node> = document.findnodes("ltx:_Capture_", Some(&equation));
256    if cells.is_empty() {
257      continue;
258    }
259    let cell1cont: Vec<Node> = cells[0].get_child_elements();
260    if cells.len() == 1
261      && cell1cont.len() == 1
262      && cell1cont[0]
263        .get_attribute("class")
264        .unwrap_or_default()
265        .contains("ltx_intertext")
266    {
267      let mut cell1cont0 = cell1cont[0].clone();
268      cell1cont0.unlink_node();
269      let mut eq = equation.clone();
270      let _ = eq.add_prev_sibling(&mut cell1cont0);
271      eq.unlink_node();
272    } else if cells.len() == 1 && cell1cont.is_empty() {
273      let mut eq = equation.clone();
274      eq.unlink_node();
275    } else {
276      equationgroup_join_cols(document, 3, &mut equation.clone())?;
277    }
278  }
279  Ok(())
