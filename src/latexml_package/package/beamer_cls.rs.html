27/// Follow beamerbasecolor's inheritance: an empty/absent fg (bg) falls back
28/// to the first parent that yields one.
29fn beamer_resolve(name: &str, field: &str, depth: usize) -> Option<String> {
30  if depth > 16 {
31    return None;
32  }
33  if let Some(v) = beamer_color_lookup(name, field)
34    && !v.is_empty()
35  {
36    return Some(v);
37  }
38  if let Some(parents) = beamer_color_lookup(name, "parent") {
39    for parent in parents.split(',') {
40      let parent = parent.trim().trim_matches(['{', '}']).trim();
41      if !parent.is_empty()
42        && let Some(v) = beamer_resolve(parent, field, depth + 1)
43      {
44        return Some(v);
45      }
46    }
47  }
48  None
49}

51/// Quiet probe: is `name` a defined color? (State `color_<name>` from
52/// color/xcolor, or raw xcolor storage `\color@<name>` — the
53/// `wisdom_xcolor_internal_storage_interop` shape.)
54fn beamer_color_known(name: &str) -> bool {
55  // One spelling per name (K10): the key and the `\color@…` macro are
56  // spelled as `color_sty::color_key`/`def_color` spell them.
57  let key = color_sty::color_key(name);
58  let name = &key["color_".len()..];
59  lookup_value(&key).is_some() || lookup_meaning(&T_CS!(s!("\\color@{name}"))).is_some()
60}

62/// A color EXPR (`A!30!B`, `-A`, `A`) is registrable when every base name it
63/// references is already defined; otherwise defer (a later \setbeamercolor
64/// or \usebeamercolor retries).
65fn beamer_expr_defined(expr: &str) -> bool {
66  let expr = expr.trim();
67  if expr.is_empty() {
68    return false;
69  }
70  for segment in expr.split('!') {
71    let segment = segment.trim().trim_start_matches('-').trim();
72    if segment.is_empty() || segment.chars().all(|c| c.is_ascii_digit() || c == '.') {
73      continue; // mix percentage
74    }
75    if !beamer_color_known(segment) {
76      return false;
77    }
78  }
79  true
80}

82/// (Re)register the xcolor names `<name>.fg` / `<name>.bg` from the stored
83/// palette, when their expressions resolve. Global, like a theme's palette.
84fn beamer_register_color(name: &str) -> Result<()> {
85  for field in ["fg", "bg"] {
86    if let Some(expr) = beamer_resolve(name, field, 0)
87      && beamer_expr_defined(&expr)
88    {
89      digest(Tokenize!(TeXString::assembled(format!(
90        "\\xglobal\\colorlet{{{name}.{field}}}{{{expr}}}"
91      ))))?;
92    }
93  }
94  Ok(())
95}

97/// Split a beamer color-option list at top-level commas into (key, value).
98fn beamer_color_opts(opts: &str) -> Vec<(String, String)> {
99  let mut out = Vec::new();
100  let mut depth = 0usize;
101  let mut current = String::new();
102  let mut parts = Vec::new();
103  for c in opts.chars() {
104    match c {
105      '{' => {
106        depth += 1;
107        current.push(c);
108      },
109      '}' => {
110        depth = depth.saturating_sub(1);
111        current.push(c);
112      },
113      ',' if depth == 0 => {
114        parts.push(std::mem::take(&mut current));
115      },
116      _ => current.push(c),
117    }
118  }
119  parts.push(current);
120  for part in parts {
121    let part = part.trim();
122    if part.is_empty() {
123      continue;
124    }
125    match part.split_once('=') {
126      Some((k, v)) => out.push((
127        k.trim().to_string(),
128        v.trim().trim_matches(['{', '}']).trim().to_string(),
129      )),
130      None => out.push((part.to_string(), String::new())),
131    }
132  }
133  out
134}

136/// Does a beamer `<…>` specification apply in presentation mode? Parts are
137/// separated by `|`; a part `mode:overlays` or bare `mode` names a mode
138/// (beamer, presentation, handout, trans, article, second, all); a part with
139/// no mode name is an overlay spec and applies. False only when every part
140/// names a mode other than beamer/presentation/all.
141fn beamer_mode_spec_applies(spec: &str) -> bool {
142  const MODES: &[&str] = &[
143    "beamer",
144    "presentation",
145    "handout",
146    "trans",
147    "article",
148    "second",
149    "all",
150  ];
151  let mut saw_part = false;
152  for part in spec.split('|') {
153    let part = part.trim();
154    if part.is_empty() {
155      continue;
156    }
157    saw_part = true;
158    let mode = part.split(':').next().unwrap_or("").trim();
159    if !MODES.contains(&mode) || matches!(mode, "beamer" | "presentation" | "all") {
160      return true;
161    }
162  }
163  !saw_part
164}

166/// Halve adjacent `Catcode::PARAM` tokens (`##` -> `#`), mirroring TeX's
167/// `\def` replacement-text scanning when `\beamer@doifinframe` is stored
168/// (beamerbaseframe.sty:527). Single `#` tokens are preserved intact for
169/// backwards compatibility with inline macro definitions in frames.
170/// In real beamer (beamerbaseframe.sty:524-529), the collected frame body is
171/// processed through two nested `\def` levels:
172/// 1. `\loop ... \repeat` defines `\def\iterate{...}` enclosing the body.
173/// 2. `\iterate` executes `\def\beamer@doifinframe{\begin{beamer@frameslide} #1 \end{beamer@frameslide}}`.
174///
175/// Each `\def` level collapses double-PARAM `##` to `#`. Thus `####` in the
176/// source becomes `##` in step 1, and `#` in step 2.
177fn halve_frame_hashes(tokens: Vec<Token>) -> Vec<Token> {
178  fn halve_once(toks: Vec<Token>) -> Vec<Token> {
179    let mut halved = Vec::with_capacity(toks.len());
180    let mut i = 0;
181    while i < toks.len() {
182      let t = toks[i];
183      if t.get_catcode() == Catcode::PARAM {
184        if i + 1 < toks.len() && toks[i + 1].get_catcode() == Catcode::PARAM {
185          // Double PARAM: collapse `##` to `#`
186          halved.push(t);
187          i += 2;
188        } else {
189          // Single PARAM: keep as is for backward compatibility
190          halved.push(t);
191          i += 1;
192        }
193      } else {
194        halved.push(t);
195        i += 1;
196      }
197    }
198    halved
199  }
200
201  // Two \def levels: \loop's \def\iterate and \def\beamer@doifinframe
202  halve_once(halve_once(tokens))
203}

205/// Collect raw tokens from the input stream up to the `\end` that closes the
206/// frame: a generic environment depth (any `\begin{X}` opens a level, any
207/// `\end{X}` closes one), the way LaTeX's `\end`/`\@checkend` pairs
208/// environments by nesting, not by name. Counting only the literal name
209/// `frame` ran away when a wrapper environment produced the close by
210/// expansion — beamerthemeTorinoTh.sty:97 `\newenvironment{tframe}
211/// {\begin{frame}[t]}{\end{frame}}`: the body's `\end{tframe}` was swallowed,
212/// the scan reached EOF and re-injected the rest of the document (beamer2thesis
213/// 4→83 errors, sweep 45). The closing tokens are returned verbatim, so a
214/// wrapper's `\end{tframe}` still runs its own end code.
215/// Guard: `perfect_kernel_batch56::beamer_wrapper_frame_environment_terminates`.
216fn collect_frame_body() -> Result<(Vec<Token>, Option<Vec<Token>>)> {
217  let end_cs = T_CS!("\\end");
218  let begin_cs = T_CS!("\\begin");
219  let mut body_tokens: Vec<Token> = Vec::new();
220  let mut env_depth: usize = 0;
221
222  while let Some(tok) = read_token()? {
223    if tok == begin_cs || tok == end_cs {
224      let is_begin = tok == begin_cs;
225      let mut lookahead = Vec::new();
226      let mut named = false;
227
228      while let Some(space_tok) = read_token()? {
229        lookahead.push(space_tok);
230        if space_tok.get_catcode() != Catcode::SPACE {
231          break;
232        }
233      }
234
235      if let Some(&first_non_space) = lookahead.last()
236        && first_non_space.get_catcode() == Catcode::BEGIN
237      {
238        while let Some(name_tok) = read_token()? {
239          lookahead.push(name_tok);
240          if name_tok.get_catcode() == Catcode::END {
241            named = true;
242            break;
243          }
244        }
245      }
246
247      if named {
248        if is_begin {
249          env_depth += 1;
250        } else if env_depth == 0 {
251          let mut end_tokens = Vec::with_capacity(1 + lookahead.len());
252          end_tokens.push(tok);
253          end_tokens.extend(lookahead);
254          return Ok((body_tokens, Some(end_tokens)));
255        } else {
256          env_depth -= 1;
257        }
258      }
259      body_tokens.push(tok);
260      body_tokens.extend(lookahead);
261    } else {
262      body_tokens.push(tok);
263    }
264  }
265
266  Ok((body_tokens, None))
267}
