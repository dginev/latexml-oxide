145/// The lines of a shipped mn2e.cls that define the journal abbreviations (the names of
146/// [`MN2E_JOURNAL_ABBREVIATIONS`]), each as the copy writes it — one-line definitions, braces balanced — preceded by the
147/// one-line definitions of the internal (`@`) macros they are built on, which copies name differently (`\ref@jnl` and
148/// its `\jnl@style`, 0711.3034; `\reff@jnl`, 1209.3072). A line filter, not TeX: the bytes are read as Latin-1; lines
149/// between a line-leading `\iffalse` and `\fi` are skipped; a definition is taken only whole on its line (a multi-line
150/// one is left out), and a line is taken whole (a second definition on it comes along); a `%`-commented line is not a
151/// definition.
152fn shipped_journal_abbreviations(text: &str) -> String {
153  static NAME_RX: std::sync::LazyLock<Regex> =
154    std::sync::LazyLock::new(|| Regex::new(r"\\(?:newcommand|def|let)\\([A-Za-z@]+)").unwrap());
155  let name_rx = &*NAME_RX;
156  let wanted: Vec<String> = name_rx
157    .captures_iter(MN2E_JOURNAL_ABBREVIATIONS)
158    .filter_map(|c| c.get(1).map(|m| m.as_str().to_string()))
159    .filter(|name| !name.contains('@'))
160    .collect();
161  static LINE_RX: std::sync::LazyLock<Regex> = std::sync::LazyLock::new(|| {
162    Regex::new(r"^\\(?:newcommand\*?|def|let)\s*\\([A-Za-z@]+)").unwrap()
163  });
164  let line_rx = &*LINE_RX;
165  static INTERNAL_RX: std::sync::LazyLock<Regex> =
166    std::sync::LazyLock::new(|| Regex::new(r"\\([A-Za-z]*@[A-Za-z@]*)").unwrap());
167  let internal_rx = &*INTERNAL_RX;
168  // Each one-line definition in the copy, by the name it defines.
169  let mut in_iffalse = false;
170  let definitions: Vec<(String, &str)> = text
171    .lines()
172    .map(str::trim)
173    .filter(|line| {
174      if line.starts_with("\\iffalse") {
175        in_iffalse = true;
176      } else if in_iffalse && line.starts_with("\\fi") {
177        in_iffalse = false;
178        return false;
179      }
180      !in_iffalse
181    })
182    .filter_map(|line| {
183      let name = line_rx.captures(line)?.get(1)?.as_str().to_string();
184      let bare = line.replace("\\{", "").replace("\\}", "");
185      (bare.matches('{').count() == bare.matches('}').count()).then_some((name, line))
186    })
187    .collect();
188  let mut lines: Vec<&str> = Vec::new();
189  let mut helpers: Vec<&str> = Vec::new();
190  for (name, line) in &definitions {
191    if wanted.contains(name) {
192      lines.push(line);
193    }
194  }
195  // The internal macros those lines use, and the ones theirs use.
196  let mut used: Vec<String> = Vec::new();
197  let mut pending: Vec<String> = lines
198    .iter()
199    .flat_map(|line| {
200      internal_rx
201        .captures_iter(line)
202        .filter_map(|c| c.get(1).map(|m| m.as_str().to_string()))
203    })
204    .collect();
205  while let Some(name) = pending.pop() {
206    if used.contains(&name) {
207      continue;
208    }
209    if let Some((_, line)) = definitions.iter().find(|(defined, _)| *defined == name) {
210      helpers.insert(0, line);
211      pending.extend(
212        internal_rx
213          .captures_iter(line)
214          .filter_map(|c| c.get(1).map(|m| m.as_str().to_string()))
215          .filter(|used_name| *used_name != name),
216      );
217    }
218    used.push(name);
219  }
220  let mut out = String::new();
221  for line in helpers.into_iter().chain(lines) {
222    out.push_str(line);
223    out.push('\n');
224  }
225  out
226}

228/// The page a float's panels are set in rows against (the panel grid rule, `declare_panel_page`) of mn2e.cls and
229/// mn.cls (mn.sty), as pdflatex gives it: 504pt (42pc) in two 240pt columns 2pc apart (mn2e.cls:15-22, :470-484;
230/// mn.sty:73-79, :895-903), one column under `onecolumn` or `referee` (`\ds@referee`, mn2e.cls:57-60, whose text is
231/// 39pc = 468pt), the options as given. The registers' 345pt read as the page set a `figure` against 345pt and a
232/// `figure*` against the same line (review rev79c2 finding 1).
233pub(crate) fn declare_mn_page() {
234  let referee = given_class_options()
235    .iter()
236    .any(|option| option == "referee");
237  let text = if referee { 468.0 } else { 504.0 };
238  let twocolumn = !matches!(
239    last_class_option(&["onecolumn", "twocolumn", "referee"]),
240    Some("onecolumn" | "referee")
241  );
242  declare_panel_page((text - 24.0) / 2.0, text, twocolumn);
243}
