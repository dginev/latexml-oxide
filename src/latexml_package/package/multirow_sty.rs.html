9  // (:173-190), which tables also set themselves (run 340, witnesses 1906.02927, 2311.04185, hep-ph0507175).

11  // \multirow: structural split into DefMacro wrapper + internal DefPrimitive
12  // setup (2-layer pattern). Perl's single DefPrimitive digests all args;
13  // Rust separates the alignment-cell state mutation (primitive, below)
14  // from the \hbox-wrapped content flow (DefMacro at :52). The split lets
15  // content flow naturally through alignment cell boxes and enables the
16  // text-mode \hbox wrap used by 1004.2626 Table 6. (Not WISDOM #41 —
17  // that entry is about math-mode ParameterType adaptations.)
18  //
19  // \lx@multirow@setup: internal primitive that sets rowspan/vattach on current cell.
20  // Separated from content so that content flows naturally through alignment cell boxes.
21  DefPrimitive!("\\lx@multirow@setup{Float}[]{}", sub[(nrows, attachment, _width)] {
22    if let Some(alignment) = lookup_alignment()
23      && let Some(data) = alignment.alignment_cell() {
24        let mut data_lock = data.borrow_mut();
25        if let Some(colspec) = data_lock.current_column() {
26          let rowspan_f = nrows.value_f64();
27          // A negative count spans the rows above (multirow.sty:172), done where the alignment is normalized
28          // (latexml_core alignment/normalize.rs `normalize_upward_spans`, which warns when the cells above are not
29          // empty); Perl's binding warns and spans one row (multirow.sty.ltxml:28, example arXiv:2210.15835
30          // `\multirow{-8.2}{*}{\includegraphics…}`; KNOWN_PERL_ERRORS #1074). Witness 1509.00049.
31          // (a fraction of a row is the box's raise, which spans the whole rows: one Warning at most, the span's)
32          let rowspan = if rowspan_f < 0.0 {
33            colspec.rowspan_up = Some(-rowspan_f as usize);
34            1usize
35          } else if rowspan_f != rowspan_f.floor() {
36            Warn!("unsupported", "multirow",
37              "Fractional row sizes for \\multirow are not yet supported.");
38            rowspan_f as usize
39          } else {
40            rowspan_f as usize
41          };
42          colspec.rowspan = Some(rowspan);
43          // Only set vattach when optional [] is explicitly provided and non-empty
44          if let Some(ref att) = attachment {
45            let att_str = att.to_string();
46            if !att_str.trim().is_empty() {
47              colspec.vattach = Some(translate_attachment(att).to_string());
48            }
49          }
50        }
51      }
52    Ok(())
53  });
54  // \multirow[vpos]{nrows}[bigstruts]{width}[fixup]{content}
55  //
56  // Perl multirow.sty.ltxml L19 defines `\multirow` as DefPrimitive
57  // whose sub body reads the Alignment column, computes rowspan, and
58  // digests the content inline. Rust uses DefMacro instead so that
59  // the content can be wrapped in `\hbox{\multirowsetup #6}` at gullet
60  // expansion time — the \hbox forces text mode so nested `$…$` cleanly
61  // switches into math; without it, inside array-cell (outer math)
62  // context the inner `$` toggles math OFF, landing content in text
63  // mode with script errors (arxiv 1004.2626 Table 6 was the witness).
64  //
65  // Intentional DefPrimitive → DefMacro kind divergence (WISDOM #44).
66  // The rowspan/colspec computation moved into `\lx@multirow@setup`,
67  // a paired DefPrimitive that runs after gullet expansion. Observable
68  // XML remains identical to the Perl port for well-formed input, and
69  // strictly better (no script-mode bleed) for malformed input.
70  // Inside the hbox, `\\` is still bound to the surrounding tabular's
71  // `\lx@alignment@newline` which fires `\lx@begin@alignment` — invalid
72  // in restricted_horizontal mode. multirow's content allows `\\` as a
73  // soft line break (the package stacks rows visually). Rebind `\\` to
74  // `\lx@newline` (horizontal-mode break) at the start of the hbox so
75  // nested `\\` survives. Witness: arXiv:1504.01713 line 694
76  // `\multirow{6}{17pt}{$\alpha_\mathrm{exp}$\\$\alpha_\mathrm{C}$}`.
77  // Gate the `\hbox{...content}` body on the actual presence of a
78  // brace-arg. The Perl multirow.sty.ltxml (Bruce's local version)
79  // `$content = Tokens() unless $content` neutralizes a missing 6th
80  // arg before reaching the `\hbox` digest. Our DefMacro substitutes
81  // `#6` unconditionally, so a malformed `\multirow{3}{*}` (next
82  // non-space is `&`) wraps the `&` token inside `\hbox{...&}` and
83  // triggers a cascade of "Stray alignment" + mode-frame errors
84  // (501 in 0903.4199, 4969 in 0908.2482). Use `\@ifnextchar\bgroup`
85  // to peek for an actual `{`-arg and only emit `\hbox{...}` when
86  // content is present; otherwise emit `\lx@multirow@setup` alone
87  // and leave `&` (or whatever comes next) for the outer tabular.
88  // `[<bigstruts>]` is a count with an optional `t`/`b` prefix (multirow.sty
89  // :158-166 `\multirow@piii`, `[b2]`) and `[<vmove>]` a length added to the
90  // `\newlength` `\multirow@dima` (:197); both unused here, so read untyped and as
91  // a skip — a `{Number}`/`{Dimension}` would put `b2` or a `plus` part back in
92  // the input (OXIDIZED_DESIGN #317).
93  // A width other than `*` sets the text as a paragraph box (multirow.sty:174-177 `\\@parboxrestore`): its interline
94  // values are the document's, though `\\@array` zeroed them in the cell (KNOWN_PERL_ERRORS #445).
95  DefMacro!("\\multirow[]{Float}[]{}[Glue]",
96    "\\lx@multirow@setup{#2}[#1]{#4}\\@ifnextchar\\bgroup{\\lx@multirow@hbox{#4}}{}");
97  // The text is set between two struts, `\hbox{\strut#6\strut}` / `\strut#6\strut\par` (multirow.sty:174-177): a
98  // one-argument author macro as the whole text (`\multirow{5}{*}{\instructblip}`, 2507.07731) takes the closing
99  // `\strut`, not the box's `}` (Perl multirow.sty.ltxml `\hbox{\multirowsetup #6}` reads the brace: KNOWN_PERL_ERRORS
100  // #888). The closing strut is `\lx@multirow@strut`, which typesets nothing: the kernel's `\strut` is a `\vrule` in
101  // math, which the formula's `tex=` spelled out (`\vrule height=10.49995pt,…`, 2501.19081's `array`); the opening one
102  // delimits nothing and is left out. Guard `perfect_kernel_batch69a::multirow_text_is_strut_delimited`.
103  DefMacro!("\\lx@multirow@strut", None);
104  DefMacro!("\\lx@multirow@hbox{}{}",
105    "\\hbox{\\let\\\\\\lx@newline\\if*#1\\else\\lx@restore@interline\\fi\\multirowsetup #2\\lx@multirow@strut}");
106

109  // runtime uses them errors. Define defensively. Witness 2406.16171.
110  DefRegister!("\\multirow@dima" => Dimension!("0pt"));
111  DefRegister!("\\multirow@dimb" => Dimension!("0pt"));
112  DefRegister!("\\multirow@cntb" => Number(0));
113  DefConditional!("\\ifmultirow@prefixt");
