3// Perl titlesec.sty.ltxml L35-40: titlesec "shape" option → CSS class.
4// Rust inlines the same four-entry map at each lookup site so both
5// primitives reference a single source of truth.
6fn titlesec_shape_class(shape: &str) -> Option<&'static str> {
7  match shape {
8    "runin" => Some("ltx_runin"),
9    "frame" => Some("ltx_framed ltx_framed_rectangle"),
10    "rightmargin" => Some("ltx_align_right"),
11    "leftmargin" => Some("ltx_align_left"),
12    _ => None,
13  }
14}

16/// Whether `tokens` hold a parameter reference: a `#` not escaped as `##`, which an inner
17/// definition in a title format may use (tex.web §476-479).
18fn has_parameter_reference(tokens: &[Token]) -> bool {
19  let mut tokens = tokens.iter().peekable();
20  while let Some(t) = tokens.next() {
21    if t.get_catcode() == Catcode::PARAM {
22      if tokens
23        .peek()
24        .is_some_and(|next| next.get_catcode() == Catcode::PARAM)
25      {
26        tokens.next();
27      } else {
28        return true;
29      }
30    }
31  }
32  false
33}

388/// The heading `\ttl@straightclass{<name>}` sets: the kernel `\@startsection{<name>}{<level>}…` (under its own name
389/// `\lx@startsection`, which a raw class that re-lets `\@startsection` does not reach; sect04.rs) (which
390/// reads the star, `[<toc>]` and title), at the kernel level of a standard sectioning command, else at
391/// the `\ttll@<name>` level `\titleclass` recorded, shifted by one in a chapterless class as
392/// `\titleclass` defines the command (titlesec's own level 0 is `\part` there, LaTeX's `\chapter`).
393fn ttl_class_heading(name: &str) -> Tokens {
394  let name = name.trim().trim_start_matches('\\');
395  let level = match name {
396    "part" => Some(-1),
397    "chapter" => Some(0),
398    "section" => Some(1),
399    "subsection" => Some(2),
400    "subsubsection" => Some(3),
401    "paragraph" => Some(4),
402    "subparagraph" => Some(5),
403    _ => None,
404  };
405  let level = level.unwrap_or_else(|| {
406    let shift = if has_chapter_counter() && class_defines_chapter() {
407      0
408    } else {
409      1
410    };
411    do_expand(Tokenize!(TeXString::assembled(s!(
412      "\\csname ttll@{name}\\endcsname"
413    ))))
414    .map(|t| t.to_string().trim().parse::<i64>().unwrap_or(0))
415    .unwrap_or(0)
416      + shift
417  });
418  mouth::tokenize_internal(TeXString::assembled(s!(
419    "\\lx@startsection{{{name}}}{{{level}}}{{}}{{}}{{}}{{}}"
420  )))
