3/// Run a chunk of our own expl3-syntax TeX under the expl3 catcode regime.
4///
5/// **Every `raw_tex` chunk in this file that names an expl3 CS must go through
6/// here.** Real expl3 code always lives between `\ExplSyntaxOn` and
7/// `\ExplSyntaxOff`, which make `:` and `_` LETTER so a name like
8/// `\c_sys_shell_escape_int` is a SINGLE control sequence. `raw_tex` tokenizes
9/// with the **ambient** catcodes, and after the expl3 load the ambient
10/// (document) regime has `_` = SUB (8) — so an unwrapped chunk mis-tokenizes:
11/// the CS terminates at the first `_`, and `\edef\c_sys_shell_escape_int{0}`
12/// becomes `\edef\c` with parameter text `_sys_shell_escape_int` and body `0`,
13/// silently rebinding LaTeX's cedilla accent `\c` (`Fran\c cois` → "Fran0cois",
14/// no error at all, where Perl renders "François"). That was issue 421 —
15/// witness arXiv 2605.11579; see the NOTE at the deleted `\c_sys_*` block.
16///
17/// Catcode-INDEPENDENT alternatives, when they fit, are safer still: `T_CS!` /
18/// `Let!` / `parse_prototype` (`def_macro_noop` &c.) build the CS name as a
19/// string and never reach the tokenizer.
20///
21/// We cannot delegate to `\ExplSyntaxOn`/`\ExplSyntaxOff` here: our expl3
22/// kernel's `\ExplSyntaxOff` is incomplete (the reason xparse_sty.rs, l3keys2e_sty.rs and
23/// mhsetup_sty.rs restore the syntax catcodes after their raw loads, `with_expl_syntax_restored` —
24/// docs/archive/EXPL3_CATCODE_GAP_2026-06-08.md). Instead this
25/// saves the ambient catcodes of `:`/`_`, installs the expl3 LETTER regime for
26/// the duration, and restores the saved values — including on the error path,
27/// so a failing chunk cannot leak the LETTER regime into the document.
28fn with_expl_catcodes<T>(body: impl FnOnce() -> Result<T>) -> Result<T> {
29  // (char, catcode to fall back to if the ambient table has no entry — the
30  // document/plain-TeX default for that char).
31  const EXPL_LETTERS: [(char, Catcode); 2] = [(':', Catcode::OTHER), ('_', Catcode::SUB)];
32  let saved = EXPL_LETTERS.map(|(c, dflt)| lookup_catcode(c).unwrap_or(dflt));
33  for (c, _) in EXPL_LETTERS {
34    assign_catcode(c, Catcode::LETTER, Some(Scope::Global));
35  }
36  let result = body();
37  for ((c, _), cc) in EXPL_LETTERS.into_iter().zip(saved) {
38    assign_catcode(c, cc, Some(Scope::Global));
39  }
40  result
41}

43/// The characters whose catcodes `\ExplSyntaxOn` changes (the ones `\ExplSyntaxOff` restores, expl3-code.tex:247-261),
44/// each with the `unwrap_or` fallback used when no saved value is found.
45const EXPL_SYNTAX_CHARS: [(char, Catcode); 8] = [
46  ('\t', Catcode::SPACE),
47  (' ', Catcode::SPACE),
48  ('"', Catcode::OTHER),
49  (':', Catcode::OTHER),
50  ('^', Catcode::SUPER),
51  ('_', Catcode::SUB),
52  ('|', Catcode::OTHER),
53  ('~', Catcode::ACTIVE),
54];

56/// Run the raw load of an expl3-syntax file and give the catcodes and `\endlinechar` it changes back the values they
57/// had before it, as `\ExplSyntaxOff` restores what `\ExplSyntaxOn` saved (expl3-code.tex:247-261) — our kernel's
58/// `\ExplSyntaxOff` is incomplete (docs/archive/EXPL3_CATCODE_GAP_2026-06-08.md). A document's own catcode survives
59/// the load: gb4e's active `_` (`\catcode`\_=13`) stayed active in pdflatex where a fixed `_`=SUB reset after
60/// tikzmark's xparse load made every later `_` a subscript in text (2408.09634). Shared by expl3_sty.rs,
61/// xparse_sty.rs, l3keys2e_sty.rs and mhsetup_sty.rs.
62pub(crate) fn with_expl_syntax_restored<T>(load: impl FnOnce() -> Result<T>) -> Result<T> {
63  let saved = EXPL_SYNTAX_CHARS.map(|(c, dflt)| lookup_catcode(c).unwrap_or(dflt));
64  let endlinechar =
65    lookup_register("\\endlinechar", Vec::new())?.map_or(13, |value| value.value_of());
66  let result = load();
67  for ((c, _), cc) in EXPL_SYNTAX_CHARS.into_iter().zip(saved) {
68    assign_catcode(c, cc, Some(Scope::Global));
69  }
70  raw_tex(&s!("\\endlinechar={endlinechar}\\relax"))?;
71  result
72}

77  //   LoadPool('LaTeX');

82  //   \expandafter\ifx\csname tex_let:D\endcsname\relax
83  //     \expandafter\@firstofone\else\expandafter\@gobble\fi
84  //     {\input expl3-code.tex }%
85  // which detects the dump-loaded `\tex_let:D` PA-alias and skips
86  // re-loading expl3-code.tex. So this 3-line wrapper does the right
87  // thing: load lua portion, then load .sty (which short-circuits).
88  LoadPool!("LaTeX");
89  InputDefinitions!("expl3", extension => Some(Cow::Borrowed("lua")), notex => true);

91  // Mirror expl3.sty's TeX-level guard so we know whether the .sty load
92  // about to run will actually `\input expl3-code.tex` (cascade-prone
93  // raw load) or short-circuit it (dump path). The guard inside
94  // expl3.sty is `\ifx\csname tex_let:D\endcsname\relax {\input ...}`,
95  // so an undefined `\tex_let:D` here ⇒ raw load will fire.
96  let raw_load_will_run = lookup_meaning(&T_CS!("\\tex_let:D")).is_none();
97
98  // In degraded no-dump mode the raw expl3.sty load re-runs expl3-code.tex,
99  // whose group-end codepoint block (L33074-33180) trips known raw-load-only
100  // cascades that the dump avoids and that Perl's own raw-load does not produce
101  // (Perl converts `\usepackage{fvextra}` with 0 errors). They leave expl3's
102  // runtime usable — the document still converts correctly — but their benign
103  // error/fatal records would otherwise count against the CONVERSION, so a
104  // perfectly good no-dump run reports "Conversion failed: 1 fatal error" (issue
105  // #651: a bare `\usepackage{fvextra}`, whose output is a healthy
106  // `<p>text</p>`). Snapshot the report across this raw load and restore it —
107  // and mute the spurious stderr lines — so only the DOCUMENT's own diagnostics
108  // reach the conversion verdict. Dump mode short-circuits the re-load
109  // (`raw_load_will_run == false`), so this is scoped strictly to the degraded
110  // fallback; canvas/parity always run on the dump.
111  // Perl `local $LaTeXML::IGNORE_ERRORS = 1`: the scope drops each record
112  // WHOLE (line and count together), so the log can never show `Error:` lines
113  // the summary does not count. (It used to snapshot/restore the counts while
114  // the unconditional Error/Fatal lines still reached the log — a log/status
115  // inconsistency, diagnostics audit 2026-09-20.)
116  // The `undefined`/`unexpected` tallies are direct `note_status` calls gated
117  // by the state-level suppressors (as the dump build sets them, ini_tex.rs),
118  // so those ride along: a summary naming an undefined macro with no
119  // `Error:undefined:` line is the very inconsistency this scope exists to
120  // prevent.
121  use latexml_core::common::error::IgnoreDiagnosticsScope;
122  let ignore_scope = raw_load_will_run.then(IgnoreDiagnosticsScope::new);
123  if raw_load_will_run {
124    assign_value("SUPPRESS_UNDEFINED_ERRORS", true, None);
125    assign_value("SUPPRESS_UNEXPECTED_ERRORS", true, None);
126  }
127
128  let _ = with_expl_syntax_restored(|| Ok(input_definitions("expl3", NewDefault!(InputDefinitionOptions,
129    noltxml => true, extension => Some(Cow::Borrowed("sty"))))));
130
131  if raw_load_will_run {
132    assign_value("SUPPRESS_UNDEFINED_ERRORS", false, None);
133    assign_value("SUPPRESS_UNEXPECTED_ERRORS", false, None);
134  }
135  drop(ignore_scope);

137  // Post-load fixup for `\__kernel_msg_info:nnxx`. xparse-2018-04-12.sty
138  // (line 101, 112, 218, 222) calls `\__kernel_msg_info:nnxx { xparse }
139  // { define-command }` etc. for every `\NewDocumentCommand`, but
140  // expl3-code.tex defines only the `:nnee` variant — `:nnxx` is a
141  // deprecated argument-spec letter (`x` = e-expanded, replaced by
142  // `e` in modern expl3) that xparse-2018-04-12 expects but expl3
143  // never auto-generates. Without this stub the CS is undefined →
144  // generate_error_stub installs an `<ltx:ERROR>` Constructor and
145  // EVERY `\NewDocumentCommand` invocation leaks the error element
146  // plus the unused message-body args into document text.
147  //
148  // We define `\__kernel_msg_info:nnxx` as a 4-arg no-op, matching
149  // Perl LaTeXML's effective end-state (`\msg_info:nnxx` is a
150  // log-only path; we have no log channel so a no-op is the closest
151  // equivalent).
152  //
153  // The historical "\cs_end: cascade" that this stub also masked was
154  // root-caused and fixed in latexml_core/src/binding/content.rs:
155  // \@pushfilename now runs BEFORE \@currname/\@currext are set,
156  // matching latex.ltx:15518-15519. With that fix the prior need to
157  // also stub `\g__file_record_seq` is gone.
158  //
159  // GATE: only install when the raw .sty actually re-loaded
160  // expl3-code.tex. On the dump path the guard short-circuits the
161  // re-load and the dump already provides the right state.
162  if raw_load_will_run {
163    with_expl_catcodes(|| raw_tex(r"\protected\gdef\__kernel_msg_info:nnxx#1#2#3#4{}"))?;
164  }

166  // expl3 case-folding override.
167  //
168  // The kernel `\__kernel_codepoint_case:nn` walks per-codepoint case maps
169  // built from `c__codepoint_<case>_<cp>_tl` constants. Those are populated
170  // by reading UnicodeData.txt / CaseFolding.txt / SpecialCasing.txt during
171  // expl3-code.tex's group-end block at L33074-33180. Our raw expl3 load
172  // currently fails to open those files (the `ior_open` chain trips on a
173  // file_input dispatch issue tracked separately), leaving the codepoint
174  // tables empty — so `\str_lowercase:n {Hello}` returns "Hello" unchanged.
175  //
176  // Override the kernel function with a Rust impl using `char::to_lowercase`
177  // and `char::to_uppercase` from std. Returns a triple `{cp1}{cp2}{cp3}`
178  // matching expl3's expected return contract — first slot is the primary
179  // result codepoint, slots 2/3 hold combining chars for compound mappings
180  // (e.g. "ß" → "SS" upper has slot1=S, slot2=S; we model only single-cp
181  // mappings here, leaving slots 2/3 blank). For ASCII this is exact; for
182  // non-Latin scripts that map to multi-char sequences (Latin extended,
183  // Greek, etc.) Rust's std char::to_lowercase yields the right primary cp.
184  DefMacro!(T_CS!("\\__kernel_codepoint_case:nn"), "{}{}", sub[(case_type, cp_str)] {
185    let case = case_type.to_string().to_lowercase();
186    let cp_text = cp_str.to_string();
187    let cp_n: u32 = cp_text.trim().parse().unwrap_or(0);
188    let result_cp = if cp_n == 0 {
189      0u32
190    } else if let Some(c) = char::from_u32(cp_n) {
191      let folded: String = match case.as_str() {
192        "lowercase" | "casefold" => c.to_lowercase().collect(),
193        "uppercase" | "titlecase" => c.to_uppercase().collect(),
194        _ => c.to_string(),
195      };
196      folded.chars().next().map(|fc| fc as u32).unwrap_or(cp_n)
197    } else {
198      cp_n
199    };
200    Ok(Tokenize!(TeXString::assembled(format!("{{{}}}{{}}{{}}", result_cp))))
201  });
