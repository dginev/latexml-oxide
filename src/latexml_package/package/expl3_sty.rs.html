3/// Run a chunk of our own expl3-syntax TeX under the expl3 catcode regime.
4///
5/// **Every `raw_tex` chunk in this file that names an expl3 CS must go through
6/// here.** Real expl3 code always lives between `\ExplSyntaxOn` and
7/// `\ExplSyntaxOff`, which make `:` and `_` LETTER so a name like
8/// `\c_sys_shell_escape_int` is a SINGLE control sequence. `raw_tex` tokenizes
9/// with the **ambient** catcodes, and after the expl3 load the ambient
10/// (document) regime has `_` = SUB (8) — so an unwrapped chunk mis-tokenizes:
11/// the CS terminates at the first `_`, and `\edef\c_sys_shell_escape_int{0}`
12/// becomes `\edef\c` with parameter text `_sys_shell_escape_int` and body `0`,
13/// silently rebinding LaTeX's cedilla accent `\c` (`Fran\c cois` → "Fran0cois",
14/// no error at all, where Perl renders "François"). That was issue 421 —
15/// witness arXiv 2605.11579; see the NOTE at the deleted `\c_sys_*` block.
16///
17/// Catcode-INDEPENDENT alternatives, when they fit, are safer still: `T_CS!` /
18/// `Let!` / `parse_prototype` (`def_macro_noop` &c.) build the CS name as a
19/// string and never reach the tokenizer.
20///
21/// We cannot delegate to `\ExplSyntaxOn`/`\ExplSyntaxOff` here: our expl3
22/// kernel's `\ExplSyntaxOff` is incomplete (the reason xparse_sty.rs and
23/// siunitx_sty.rs hardcode a `~`/`:`/`_` restore after their raw loads —
24/// docs/parity/diagnostics/EXPL3_CATCODE_GAP_2026-06-08.md). Instead this
25/// saves the ambient catcodes of `:`/`_`, installs the expl3 LETTER regime for
26/// the duration, and restores the saved values — including on the error path,
27/// so a failing chunk cannot leak the LETTER regime into the document.
28fn with_expl_catcodes<T>(body: impl FnOnce() -> Result<T>) -> Result<T> {
29  // (char, catcode to fall back to if the ambient table has no entry — the
30  // document/plain-TeX default for that char).
31  const EXPL_LETTERS: [(char, Catcode); 2] = [(':', Catcode::OTHER), ('_', Catcode::SUB)];
32  let saved = EXPL_LETTERS.map(|(c, dflt)| lookup_catcode(c).unwrap_or(dflt));
33  for (c, _) in EXPL_LETTERS {
34    assign_catcode(c, Catcode::LETTER, Some(Scope::Global));
35  }
36  let result = body();
37  for ((c, _), cc) in EXPL_LETTERS.into_iter().zip(saved) {
38    assign_catcode(c, cc, Some(Scope::Global));
39  }
40  result
41}

44LoadDefinitions!({
45  // Strict-Perl translation of LaTeXML/lib/LaTeXML/Package/expl3.sty.ltxml:
46  //   LoadPool('LaTeX');
47  //   InputDefinitions('expl3', type => 'lua');
48  //   InputDefinitions('expl3', type => 'sty', noltxml => 1);
49  //
50  // The raw expl3.sty file has a TeX-level guard
51  //   \expandafter\ifx\csname tex_let:D\endcsname\relax
52  //     \expandafter\@firstofone\else\expandafter\@gobble\fi
53  //     {\input expl3-code.tex }%
54  // which detects the dump-loaded `\tex_let:D` PA-alias and skips
55  // re-loading expl3-code.tex. So this 3-line wrapper does the right
56  // thing: load lua portion, then load .sty (which short-circuits).
57  LoadPool!("LaTeX");
58  InputDefinitions!("expl3", extension => Some(Cow::Borrowed("lua")), notex => true);
59
60  // Mirror expl3.sty's TeX-level guard so we know whether the .sty load
61  // about to run will actually `\input expl3-code.tex` (cascade-prone
62  // raw load) or short-circuit it (dump path). The guard inside
63  // expl3.sty is `\ifx\csname tex_let:D\endcsname\relax {\input ...}`,
64  // so an undefined `\tex_let:D` here ⇒ raw load will fire.
65  let raw_load_will_run = lookup_meaning(&T_CS!("\\tex_let:D")).is_none();
66
67  // In degraded no-dump mode the raw expl3.sty load re-runs expl3-code.tex,
68  // whose group-end codepoint block (L33074-33180) trips known raw-load-only
69  // cascades that the dump avoids and that Perl's own raw-load does not produce
70  // (Perl converts `\usepackage{fvextra}` with 0 errors). They leave expl3's
71  // runtime usable — the document still converts correctly — but their benign
72  // error/fatal records would otherwise count against the CONVERSION, so a
73  // perfectly good no-dump run reports "Conversion failed: 1 fatal error" (issue
74  // #651: a bare `\usepackage{fvextra}`, whose output is a healthy
75  // `<p>text</p>`). Snapshot the report across this raw load and restore it —
76  // and mute the spurious stderr lines — so only the DOCUMENT's own diagnostics
77  // reach the conversion verdict. Dump mode short-circuits the re-load
78  // (`raw_load_will_run == false`), so this is scoped strictly to the degraded
79  // fallback; canvas/parity always run on the dump.
80  use latexml_core::common::error::{REPORT, set_suppress_log_output};
81  let report_snapshot = raw_load_will_run.then(|| REPORT.borrow().clone());
82  let prev_suppress = raw_load_will_run.then(|| set_suppress_log_output(true));
83
84  let _ = input_definitions("expl3", NewDefault!(InputDefinitionOptions,
85    noltxml => true, extension => Some(Cow::Borrowed("sty"))));
86
87  if let Some(snapshot) = report_snapshot {
88    *REPORT.borrow_mut() = snapshot;
89  }
90  if let Some(prev) = prev_suppress {
91    set_suppress_log_output(prev);
92  }

94  // Post-load fixup for `\__kernel_msg_info:nnxx`. xparse-2018-04-12.sty
95  // (line 101, 112, 218, 222) calls `\__kernel_msg_info:nnxx { xparse }
96  // { define-command }` etc. for every `\NewDocumentCommand`, but
97  // expl3-code.tex defines only the `:nnee` variant — `:nnxx` is a
98  // deprecated argument-spec letter (`x` = e-expanded, replaced by
99  // `e` in modern expl3) that xparse-2018-04-12 expects but expl3
100  // never auto-generates. Without this stub the CS is undefined →
101  // generate_error_stub installs an `<ltx:ERROR>` Constructor and
102  // EVERY `\NewDocumentCommand` invocation leaks the error element
103  // plus the unused message-body args into document text.
104  //
105  // We define `\__kernel_msg_info:nnxx` as a 4-arg no-op, matching
106  // Perl LaTeXML's effective end-state (`\msg_info:nnxx` is a
107  // log-only path; we have no log channel so a no-op is the closest
108  // equivalent).
109  //
110  // The historical "\cs_end: cascade" that this stub also masked was
111  // root-caused and fixed in latexml_core/src/binding/content.rs:
112  // \@pushfilename now runs BEFORE \@currname/\@currext are set,
113  // matching latex.ltx:15518-15519. With that fix the prior need to
114  // also stub `\g__file_record_seq` is gone.
115  //
116  // GATE: only install when the raw .sty actually re-loaded
117  // expl3-code.tex. On the dump path the guard short-circuits the
118  // re-load and the dump already provides the right state.
119  if raw_load_will_run {
120    with_expl_catcodes(|| raw_tex(r"\protected\gdef\__kernel_msg_info:nnxx#1#2#3#4{}"))?;
121  }

123  // expl3 case-folding override.
124  //
125  // The kernel `\__kernel_codepoint_case:nn` walks per-codepoint case maps
126  // built from `c__codepoint_<case>_<cp>_tl` constants. Those are populated
127  // by reading UnicodeData.txt / CaseFolding.txt / SpecialCasing.txt during
128  // expl3-code.tex's group-end block at L33074-33180. Our raw expl3 load
129  // currently fails to open those files (the `ior_open` chain trips on a
130  // file_input dispatch issue tracked separately), leaving the codepoint
131  // tables empty — so `\str_lowercase:n {Hello}` returns "Hello" unchanged.
132  //
133  // Override the kernel function with a Rust impl using `char::to_lowercase`
134  // and `char::to_uppercase` from std. Returns a triple `{cp1}{cp2}{cp3}`
135  // matching expl3's expected return contract — first slot is the primary
136  // result codepoint, slots 2/3 hold combining chars for compound mappings
137  // (e.g. "ß" → "SS" upper has slot1=S, slot2=S; we model only single-cp
138  // mappings here, leaving slots 2/3 blank). For ASCII this is exact; for
139  // non-Latin scripts that map to multi-char sequences (Latin extended,
140  // Greek, etc.) Rust's std char::to_lowercase yields the right primary cp.
141  DefMacro!(T_CS!("\\__kernel_codepoint_case:nn"), "{}{}", sub[(case_type, cp_str)] {
142    let case = case_type.to_string().to_lowercase();
143    let cp_text = cp_str.to_string();
144    let cp_n: u32 = cp_text.trim().parse().unwrap_or(0);
145    let result_cp = if cp_n == 0 {
146      0u32
147    } else if let Some(c) = char::from_u32(cp_n) {
148      let folded: String = match case.as_str() {
149        "lowercase" | "casefold" => c.to_lowercase().collect(),
150        "uppercase" | "titlecase" => c.to_uppercase().collect(),
151        _ => c.to_string(),
152      };
153      folded.chars().next().map(|fc| fc as u32).unwrap_or(cp_n)
154    } else {
155      cp_n
156    };
157    Ok(Tokenize!(TeXString::assembled(format!("{{{}}}{{}}{{}}", result_cp))))
158  });

160  // `\c_sys_jobname_str` is one of the system-info constants bound by
161  // `\g__sys_everyjob_tl` at job start (via `\everyjob`), which our engine
162  // never fires (matching Perl's gap). When packages like `duckuments.sty`
163  // then do
164  //   `\str_if_eq_p:Vn \c_sys_jobname_str { example-image-duck }`
165  // the V-expansion triggers `\if_int_compare:w` cascades on Rust
166  // (Perl emits one undefined error and recovers; Rust's recovery
167  // re-fires per scan, surfacing 21+ relational-token cascades).
168  //
169  // A plain `\Let` alias to `\jobname`, rather than the full `\str_const:Ne`
170  // machinery — those expl3 constructors themselves require a working
171  // `\c_sys_jobname_str` at definition time. The sibling date/time int
172  // constants need NO such patch-up: they are already defined, with live
173  // values, by the time a package body runs (see the NOTE below).
174  //
175  // Driver: 2406.14142 (duckuments cascade, 21 errors → 4 expected
176  // (matching Perl's residual undefined-CS count)).
177  Let!("\\c_sys_jobname_str", "\\jobname");
178
179  // expl3 historical-alias: `\hbox_unpack_clear:N` was deprecated
180  // around 2018 in favor of `\hbox_unpack_drop:N` (both call `\unhbox`
181  // — read out the box's contents AND clear/drop the box itself).
182  // Modern l3kernel no longer ships the alias, so our dump doesn't
183  // contain it, but third-party expl3 packages (mmacells.sty,
184  // letgut-lstlang.sty) still call the deprecated name. Add the
185  // alias post-dump so those packages load cleanly. Witness:
186  // arXiv:2002.07146 (uses `\usepackage{mmacells}`).
187  // (See also `\hbox_unpack:N` in dump = `\unhcopy` which is the
188  // non-clearing version.)
189  Let!("\\hbox_unpack_clear:N", "\\hbox_unpack_drop:N");
190
191  // NOTE (issue 421): there used to be a `RawTeX!` block here `\edef`-ing the
192  // seven `\c_sys_{minute,hour,day,month,year}_int` / `\c_sys_timestamp_str` /
193  // `\c_sys_shell_escape_int` constants, on the belief that they stay undefined
194  // because we never fire `\everyjob`. **Do not re-add it.** Both halves of that
195  // belief were wrong, measured on the current engine:
196  //
197  //  * They ARE defined, at package-load time and with live values — probing
198  //    `\number\csname c_sys_year_int\endcsname` right after `\usepackage{xparse}`
199  //    gives the real year, and `c_sys_minute_int` advances between runs. Only
200  //    `\c_sys_jobname_str` needed the `Let!` above. (That is the dump path. On
201  //    the `LATEXML_NODUMP=1` raw-load branch they come back undefined — but the
202  //    block did not define them there either, so nothing changed; that branch
203  //    dies earlier anyway, on the expl3-code codepoint group.)
204  //  * The block never actually ran. Written as raw TeX, it was tokenized with
205  //    the AMBIENT catcodes, and after the expl3 load the document regime has
206  //    `_` = SUB — so `\edef\c_sys_minute_int{0}` parsed as `\edef\c` with
207  //    parameter text `_sys_minute_int`. It defined none of the constants and
208  //    instead REBOUND LaTeX's cedilla accent `\c` (`\meaning\c` =
209  //    `macro:_sys_shell_escape_int->0`), so every later `Fran\c cois` rendered
210  //    "Fran0cois" — silently, 0 errors, where Perl renders "François".
211  //    Witness arXiv 2605.11579; guard
212  //    `expl3_load_does_not_clobber_cedilla_accent`.
213  //
214  // Fixing only the tokenization would have been worse than deleting: the
215  // constants would then overwrite expl3's live clock values with frozen
216  // dummies (and a hardcoded year). Perl's expl3.sty.ltxml has no such block
217  // either — it is three lines. Any FUTURE expl3-syntax raw TeX added to this
218  // file must go through `with_expl_catcodes`.
219
220  // expl3 l3regex is handled NATIVELY by the real expl3 VM (loaded from
221  // expl3-code.tex: \__regex_compile:n / \__regex_build:n / \__regex_match:n,
222  // etc.), which now runs correctly under our gullet — so \regex_match,
223  // \regex_count, \regex_replace_all, \regex_extract_once and their \seq_*
224  // results all work faithfully. A former Rust-`regex`-crate SHIM bypassed the
225  // VM (its \if_int_compare:w-driven expansion used to stall); it was REMOVED
226  // 2026-06-20 after intervening gullet fixes made the real VM pass — verified
227  // on its original cascade witness 2406.14142 (21 errors -> 0) and the
228  // expl3/regex_match + expl3/regex_native test fixtures. (Shim in git history.)
229
230  assign_catcode(':', Catcode::OTHER, Some(Scope::Global));
231  assign_catcode('_', Catcode::SUB, Some(Scope::Global));
232  // Also restore `~` to ACTIVE (13). `\usepackage{expl3}` in real LaTeX
233  // leaves expl3 syntax OFF — `~` back to its document catcode (13/active),
234  // not the expl3 `~`=10 (SPACE). Our raw expl3-code.tex load leaves `~` at
235  // 10 because Rust's `\ExplSyntaxOff` is incomplete; the `:`/`_` restores
236  // above already turn expl3-letters off, but `~` was missed. Without this, a
237  // LATER `\usepackage[english]{babel}` runs `\initiate@active@char{~}` with
238  // `~` at catcode 10 → an `expected:<relationaltoken>` cascade. Minimal
239  // repro `\usepackage{expl3}\usepackage[english]{babel}` (2 errors → 0).
240  // `~` is not an expl3 LETTER char (unlike `:`/`_`), so this is glossary-safe.
241  // Complements the per-package restore in xparse_sty.rs / siunitx_sty.rs.
242  assign_catcode('~', Catcode::ACTIVE, Some(Scope::Global));
243});
