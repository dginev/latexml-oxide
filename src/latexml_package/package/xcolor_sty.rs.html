12/// Perl: sub delta { my ($v, $n) = @_; ($v <= ($n+1)/2 ? $v/($n+1) : ($v+1)/($n+1)) }
13fn delta(v: f64, n: f64) -> f64 {
14  if v <= (n + 1.0) / 2.0 {
15    v / (n + 1.0)
16  } else {
17    (v + 1.0) / (n + 1.0)
18  }
19}

24/// Perl: sub rangeReduction — perverse rotation of value back into [0..1]
25fn range_reduction(value: f64) -> f64 {
26  if value > 1.0 {
27    if value > 1.00001 {
28      value - (value as i64) as f64
29    } else {
30      1.0
31    }
32  } else if value < 0.0 {
33    if value < -0.0001 {
34      value - (value as i64) as f64 + 1.0
35    } else {
36      0.0
37    }
38  } else {
39    value
40  }
41}

43/// Convert from extended model to core model.
44/// Perl: RGB→rgb, HTML→rgb, Hsb→hsb, HSB→hsb, Gray→gray
45fn convert_extended_to_core(model: &str, spec: &str) -> Color {
46  // NOTE: lowercase core models (rgb/cmy/cmyk/hsb/gray) use 0..1 components, while
47  // uppercase extended models (RGB, HTML, HSB, Gray, ...) use 0..255 / 0..240 etc.
48  // Match CASE-SENSITIVELY: do NOT lowercase `model` before dispatching.
49  match model {
50    "rgb" | "cmy" | "cmyk" | "hsb" | "gray" => color_from_model_spec(model, spec),
51    _ => convert_ext_model(model, spec),
52  }
53}

55fn convert_ext_model(model: &str, spec: &str) -> Color {
56  let comps: Vec<f64> = spec
57    .split(|c: char| c == ',' || c.is_whitespace())
58    .filter(|s| !s.is_empty())
59    .filter_map(|s| s.trim().parse::<f64>().ok())
60    .collect();
61  match model {
62    "RGB" => {
63      let l = 255.0; // default \rangeRGB
64      Color::Rgb(
65        delta(comps.first().copied().unwrap_or(0.0), l),
66        delta(comps.get(1).copied().unwrap_or(0.0), l),
67        delta(comps.get(2).copied().unwrap_or(0.0), l),
68      )
69    },
70    "HTML" => {
71      // RRGGBB hex string
72      let hex = spec.trim();
73      if hex.len() >= 6 {
74        let r = u8::from_str_radix(&hex[0..2], 16).unwrap_or(0);
75        let g = u8::from_str_radix(&hex[2..4], 16).unwrap_or(0);
76        let b = u8::from_str_radix(&hex[4..6], 16).unwrap_or(0);
77        Color::Rgb(
78          delta(r as f64, 255.0),
79          delta(g as f64, 255.0),
80          delta(b as f64, 255.0),
81        )
82      } else {
83        BLACK
84      }
85    },
86    "Hsb" => {
87      let h_range = 360.0; // default \rangeHsb
88      Color::Hsb(
89        comps.first().copied().unwrap_or(0.0) / h_range,
90        comps.get(1).copied().unwrap_or(0.0),
91        comps.get(2).copied().unwrap_or(0.0),
92      )
93    },
94    "HSB" => {
95      let m = 240.0;
96      Color::Hsb(
97        delta(comps.first().copied().unwrap_or(0.0), m),
98        delta(comps.get(1).copied().unwrap_or(0.0), m),
99        delta(comps.get(2).copied().unwrap_or(0.0), m),
100      )
101    },
102    // xcolor.sty.ltxml:131-142: `\rangetHsb`'s tuned hue to hsb's.
103    "tHsb" => {
104      let h_range = 360.0; // default \rangeHsb
105      Color::Hsb(
106        thsb_to_hsb(comps.first().copied().unwrap_or(0.0), h_range) / h_range,
107        comps.get(1).copied().unwrap_or(0.0),
108        comps.get(2).copied().unwrap_or(0.0),
109      )
110    },
111    "Gray" => {
112      let n = 15.0;
113      Color::Gray(delta(comps.first().copied().unwrap_or(0.0), n))
114    },
115    "wave" => {
116      let lambda = comps.first().copied().unwrap_or(500.0);
117      let h = if lambda < 440.0 {
118        4.0 + ((lambda - 440.0) / (-60.0)).clamp(0.0, 1.0)
119      } else if lambda < 490.0 {
120        4.0 - ((lambda - 440.0) / 50.0).clamp(0.0, 1.0)
121      } else if lambda < 510.0 {
122        2.0 + ((lambda - 510.0) / (-20.0)).clamp(0.0, 1.0)
123      } else if lambda < 580.0 {
124        2.0 - ((lambda - 510.0) / 70.0).clamp(0.0, 1.0)
125      } else if lambda < 645.0 {
126        ((lambda - 645.0) / (-65.0)).clamp(0.0, 1.0)
127      } else {
128        0.0
129      };
130      let bb = if lambda < 420.0 {
131        (0.3 + 0.7 * (lambda - 380.0) / 40.0).clamp(0.0, 1.0)
132      } else if lambda < 700.0 {
133        1.0
134      } else {
135        (0.3 + 0.7 * (lambda - 780.0) / (-80.0)).clamp(0.0, 1.0)
136      };
137      Color::Hsb(h / 6.0, 1.0, bb).to_rgb()
138    },
139    _ => {
140      // A model neither core nor one of xcolor.sty.ltxml's `DefColorModel`s (:85-168: RGB HTML Hsb HSB tHsb Gray wave) is
141      // Perl's Color.pm:61 Error, unless raw code defined it, as xcolor.sty:565 tests a model
142      // (`\@ifundefined{XC@mod@<model>}`): colorspace.sty's `\spc@mix@*` set `\XC@mod@<name>` for its
143      // `\definecolorspace` models and :59 `\XC@mod@&spot` (the colorspace manual; Perl errs, KNOWN_PERL_ERRORS #958).
144      // That one is known, and its colour is lost: a counted Warning, the recovery black. xcolor.sty's own `natural`,
145      // `ignore` and `ps` (:140, :564, :1131) are its internals, on which pdflatex errs too: they stay the Error.
146      if !matches!(model, "natural" | "ignore" | "ps")
147        && is_defined_token(&T_CS!(s!("\\XC@mod@{model}")))
148      {
149        report_unconverted_model(model);
150      } else {
151        report_unrecognized_model(model);
152      }
153      color_from_model_spec(model, spec)
154    },
155  }
156}

158/// Perl: LookupXColor — lookup name with complement support via '-' prefix
159fn lookup_xcolor(name: &str) -> Color {
160  if name.is_empty() {
161    return WHITE;
162  }
163  if name == "." {
164    // Current color
165    return match lookup_value("color_.") {
166      Some(Stored::String(sym)) => with(sym, Color::from_stored).unwrap_or(BLACK),
167      _ => BLACK,
168    };
169  }
170  // Handle complement prefix
171  let stripped = name.trim_start_matches('-');
172  let dash_count = name.len() - stripped.len();
173  let color = lookup_color_obj(stripped);
174  if dash_count % 2 == 1 {
175    color.complement()
176  } else {
177    color
178  }
179}

181/// Perl: DecodeColor — decode color expressions
182/// Handles: names, mix expressions (!pct!color), complements, extended expressions (model:...)
183fn decode_color(expression: &str) -> Color {
184  let expression = expression.trim();
185  if expression.is_empty() {
186    return WHITE;
187  }
188
189  // Check for extended expression: core_model(,div)?:expr1,dec1;...
190  if let Some(colon_pos) = expression.find(':') {
191    let before_colon = &expression[..colon_pos];
192    // Check if before colon is a core model (possibly with ,div)
193    let (model_part, div_part) = if let Some(comma_pos) = before_colon.find(',') {
194      (
195        &before_colon[..comma_pos],
196        before_colon[comma_pos + 1..].trim(),
197      )
198    } else {
199      (before_colon, "")
200    };
201    let model_trimmed = model_part.trim();
202    if matches!(model_trimmed, "rgb" | "cmy" | "cmyk" | "hsb" | "gray") {
203      let exprs_str = &expression[colon_pos + 1..];
204      return decode_extended_color(model_trimmed, div_part, exprs_str);
205    }
206  }
207
208  // Standard color expression: prefix name mix_expr postfix
209  // Handle functional expressions (>wheel, >twheel)
210  let (base_expr, func_expr) = if let Some(gt_pos) = expression.find('>') {
211    (&expression[..gt_pos], Some(&expression[gt_pos..]))
212  } else {
213    (expression, None)
214  };
215
216  // Parse the base: optional - prefix, name, optional !mix
217  let (name_part, mix_part, postfix) = parse_standard_expr(base_expr);
218
219  // Perl L328-331: lookup name (WITHOUT complement — complement applied after mix)
220  // Count and strip dash prefixes for later complement application
221  let stripped_name = name_part.trim_start_matches('-');
222  let dash_count = name_part.len() - stripped_name.len();
223
224  let mut color = if let Some(pf) = &postfix {
225    if pf.starts_with("!![") {
226      let n_str = pf.trim_start_matches("!![").trim_end_matches(']');
227      let n: usize = n_str.parse().unwrap_or(0);
228      index_color_series(stripped_name, n)
229    } else {
230      lookup_color_obj(stripped_name)
231    }
232  } else {
233    lookup_color_obj(stripped_name)
234  };
235
236  // Apply mix expressions: !pct!name...
237  if !mix_part.is_empty() {
238    color = apply_mix_expr(color, &mix_part);
239  }
240  // Then the `\blendcolors` blend (xcolor.sty `\XC@blendcolors`: the blend is
241  // a mix applied to the RESOLVED color). Perl (xcolor.sty.ltxml:337) string-
242  // concatenated it onto the local mix — `black!75` + `!60!white` became
243  // `black!75!60!white`, "mix with color 60" (iodhbwm via ydoc-desc.sty:125;
244  // SHARED, pdflatex clean). Guard:
245  // `perfect_kernel_batch56::xcolor_blend_applies_after_the_local_mix`.
246  let blend = match lookup_value("color_blend") {
247    Some(Stored::String(blend_sym)) => with(blend_sym, |b| b.to_string()),
248    _ => String::new(),
249  };
250  if !blend.is_empty() {
251    color = apply_mix_expr(color, &blend);
252  }
253
254  // Perl L343: complement applied AFTER mix, not before
255  if dash_count % 2 == 1 {
256    color = color.complement();
257  }
258
259  // Handle postfix stepping: !!+ or !!++
260  if let Some(pf) = &postfix
261    && pf.starts_with("!!")
262    && pf.contains('+')
263  {
264    let plus_count = pf.chars().filter(|c| *c == '+').count();
265    step_color_series(&name_part, plus_count);
266  }
267
268  // Apply function expressions (>wheel, >twheel)
269  if let Some(func) = func_expr {
270    color = apply_func_expr(color, func);
271  }
272
273  color
274}

276fn parse_standard_expr(expr: &str) -> (String, String, Option<String>) {
277  // Split off postfix !!... if present
278  let (main, postfix) = if let Some(pp) = expr.find("!!") {
279    (&expr[..pp], Some(expr[pp..].to_string()))
280  } else {
281    (expr, None)
282  };
283
284  // Split name from mix expression at first !
285  if let Some(bang_pos) = main.find('!') {
286    let name = main[..bang_pos].to_string();
287    let mix = main[bang_pos..].to_string();
288    (name, mix, postfix)
289  } else {
290    (main.to_string(), String::new(), postfix)
291  }
292}

294/// Apply mix expressions: !pct!name!pct!name...
295fn apply_mix_expr(mut color: Color, mix_str: &str) -> Color {
296  let mut remaining = mix_str;
297  while remaining.starts_with('!') {
298    remaining = &remaining[1..]; // skip leading !
299    // Read percentage
300    let (pct_str, rest) = split_at_bang(remaining);
301    let pct_str_clean = pct_str.replace("--", "");
302    let pct: f64 = if pct_str_clean.is_empty() || pct_str_clean == "." {
303      if pct_str_clean.is_empty() { 100.0 } else { 0.0 }
304    } else {
305      pct_str_clean.parse().unwrap_or(100.0)
306    };
307
308    if rest.is_empty() {
309      // No second color specified, mix with white. xcolor/pgf allow
310      // pct > 100 (extrapolation past the base = darker than the base);
311      // Color::mix clamps the resulting components to [0,1].
312      color = color.mix(&WHITE, pct / 100.0);
313      break;
314    }
315    // Skip the ! before the name
316    let after_bang = if let Some(stripped) = rest.strip_prefix('!') {
317      stripped
318    } else {
319      rest
320    };
321    // Read the name (up to next !)
322    let (name, next_rest) = split_at_bang(after_bang);
323    let other = if name.is_empty() {
324      WHITE
325    } else {
326      lookup_xcolor(name)
327    };
328    color = color.mix(&other, pct / 100.0);
329    remaining = next_rest;
330  }
331  color
332}

334fn split_at_bang(s: &str) -> (&str, &str) {
335  if let Some(pos) = s.find('!') {
336    (&s[..pos], &s[pos..])
337  } else {
338    (s, "")
339  }

342/// Decode extended color expression: model(,div)?:expr1,dec1;...
343fn decode_extended_color(model: &str, div_str: &str, exprs_str: &str) -> Color {
344  let mut color = BLACK.convert(model);
345  let mut dectot: f64 = 0.0;
346  let mut palette: Vec<(Color, f64)> = Vec::new();
347
348  for part in exprs_str.split(';') {
349    let part = part.trim();
350    if part.is_empty() {
351      continue;
352    }
353    // Split at last comma to get (expr, dec)
354    if let Some(comma_pos) = part.rfind(',') {
355      let expr_part = part[..comma_pos].trim();
356      let dec_str = part[comma_pos + 1..].trim().replace("--", "");
357      if dec_str.is_empty() || dec_str == "." {
358        continue;
359      }
360      let dec: f64 = dec_str.parse().unwrap_or(0.0);
361      if dec == 0.0 {
362        continue;
363      }
364      dectot += dec;
365      palette.push((decode_color(expr_part), dec));
366    }
367  }
368
369  let div: f64 = if !div_str.is_empty() {
370    div_str.trim().parse().unwrap_or(dectot)
371  } else {
372    dectot
373  };
374
375  if div == 0.0 {
376    return color;
377  }
378
379  for (c, dec) in &palette {
380    let converted = c.convert(model);
381    color = color.add(&converted.scale(*dec / div));
382  }
383  color
384}

386/// Apply function expressions: >wheel,angle or >wheel,angle,full or >twheel,...
387fn apply_func_expr(mut color: Color, func_str: &str) -> Color {
388  let mut remaining = func_str;
389  while remaining.starts_with('>') {
390    remaining = &remaining[1..];
391    let (func_part, rest) = if let Some(gt) = remaining.find('>') {
392      (&remaining[..gt], &remaining[gt..])
393    } else {
394      (remaining, "")
395    };
396    let parts: Vec<&str> = func_part.split(',').collect();
397    if parts.len() >= 2 {
398      let func = parts[0].trim();
399      let angle: f64 = parts[1].trim().parse().unwrap_or(0.0);
400      let full: Option<f64> = parts.get(2).and_then(|s| s.trim().parse().ok());
401      let is_twheel = func == "twheel";
402      // Convert to hsb (internal [0,1] range)
403      let hsb = color.to_hsb();
404      let (h, s, b_val) = if let Color::Hsb(h, s, b) = hsb {
405        (h, s, b)
406      } else {
407        (0.0, 0.0, 0.0)
408      };
409      let h_range = 360.0; // \rangeHsb default
410      let circle = if let Some(f) = full { h_range / f } else { 1.0 };
411      // Perl: Color($model, $h_scaled + $angle * $circle, $s, $b)
412      // h is in [0,1] internally, scale to [0, h_range] for computation
413      let h_scaled = h * h_range;
414      let new_h_scaled = h_scaled + angle * circle;
415      // Convert back to [0,1] — for Hsb, just divide by h_range
416      // For tHsb, apply piecewise-linear mapping first
417      let new_h = if is_twheel {
418        thsb_to_hsb(new_h_scaled, h_range) / h_range
419      } else {
420        new_h_scaled / h_range
421      };
422      color = Color::Hsb(new_h, s, b_val);
423    }
424    remaining = rest;
425  }
426  color
427}

429/// Convert tHsb hue to hsb hue using piecewise-linear mapping.
430/// Perl: \rangetHsb = '60,30;120,60;180,120;210,180;240,240'
431/// Input: tHsb h in [0, h_range], output: hsb h in [0, h_range]
432fn thsb_to_hsb(h: f64, h_range: f64) -> f64 {
433  // Piecewise linear map: tHsb→hsb breakpoints (x→y)
434  // Perl: "60,30;120,60;180,120;210,180;240,240" + ";360,360"
435  let breakpoints: &[(f64, f64)] = &[
436    (60.0, 30.0),
437    (120.0, 60.0),
438    (180.0, 120.0),
439    (210.0, 180.0),
440    (240.0, 240.0),
441    (h_range, h_range),
442  ];
443  let (mut x0, mut y0) = (0.0, 0.0);
444  for &(xn, yn) in breakpoints {
445    if h <= xn {
446      return y0 + (yn - y0) / (xn - x0) * (h - x0);
447    }
448    x0 = xn;
449    y0 = yn;
450  }
451  h // fallback: identity for h > h_range
452}

454/// Step the color series
455fn step_color_series(name: &str, n: usize) {
456  let color_key = color_sty::color_key(name);
457  let step_key = s!("color_series_{name}_step");
458  if let (Some(Stored::String(c_sym)), Some(Stored::String(s_sym))) =
459    (lookup_value(&color_key), lookup_value(&step_key))
460  {
461    let color = Color::from_stored(&to_string(c_sym)).unwrap_or(BLACK);
462    let step = Color::from_stored(&to_string(s_sym)).unwrap_or(BLACK);
463    let comps = color.components();
464    let step_comps = step.components();
465    let new_comps: Vec<f64> = comps
466      .iter()
467      .zip(step_comps.iter())
468      .map(|(c, s)| range_reduction(c + n as f64 * s))
469      .collect();
470    let new_color = from_model_components(color.model(), &new_comps);
471    def_color(name, &new_color, Some(Scope::Global)).ok();
472  }
473}

475/// Index into color series (but don't step)
476fn index_color_series(name: &str, p: usize) -> Color {
477  let base_key = s!("color_series_{name}_base");
478  let step_key = s!("color_series_{name}_step");
479  match (lookup_value(&base_key), lookup_value(&step_key)) {
480    (Some(Stored::String(b_sym)), Some(Stored::String(s_sym))) => {
481      let base = Color::from_stored(&to_string(b_sym)).unwrap_or(BLACK);
482      let step = Color::from_stored(&to_string(s_sym)).unwrap_or(BLACK);
483      let comps = base.components();
484      let step_comps = step.components();
485      let new_comps: Vec<f64> = comps
486        .iter()
487        .zip(step_comps.iter())
488        .map(|(c, s)| range_reduction(c + p as f64 * s))
489        .collect();
490      from_model_components(base.model(), &new_comps)
491    },
492    _ => BLACK,
493  }
494}

496/// xcolor.sty:104-120 `\XC@edef`/`\XC@mdef`: a colour name, model or
497/// expression is expanded with each of its separators (`! : - + ; " >`, and a
498/// model list's `/`) that is currently active standing for itself. A table that makes `!` an active `\itshape`
499/// (`\catcode`!=13 \def!{\itshape}`) otherwise expands `red!100!black` into
500/// font switches (arXiv 2605.30133: 101 "Can't find color named", Fatal; Perl's
501/// binding expands unprotected too, xcolor.sty.ltxml:561).
502fn xc_expand<T: Into<Tokens>>(tokens: T) -> Result<Tokens> {
503  let active: Vec<char> = ['!', ':', '-', '+', ';', '"', '>', '/']
504    .into_iter()
505    .filter(|&c| lookup_catcode(c) == Some(Catcode::ACTIVE))
506    .collect();
507  if active.is_empty() {
508    return do_expand(tokens);
509  }
510  push_frame();
511  for c in active {
512    let_i(&T_ACTIVE!(c), &T_OTHER!(c.to_string().as_str()), None);
513  }
514  let expanded = do_expand(tokens);
515  pop_frame()?;
516  expanded
517}

519/// Perl xcolor.sty.ltxml L403-409: if the optional `[type]` argument
520/// equals "ps", emit an Info and return false. `\colorlet`/`\definecolorset`
521/// then skip the definition as Perl does; `\XC@definecolor`/`\providecolor`
522/// instead register the model-white fallback real xcolor keeps
523/// (xcolor.sty:531-533). Non-"ps" types (empty, "named", "ful", ...) all
524/// pass through.
525fn check_no_postscript(type_opt: Option<Tokens>, macro_name: &str) -> Result<bool> {
526  if let Some(t) = type_opt {
527    let s = xc_expand(t)?.to_string();
528    if s == "ps" {
529      Info!(
530        "ignored",
531        macro_name,
532        s!("Ignoring definition of postscript color in {macro_name}")
533      );
534      return Ok(false);
535    }
536  }
537  Ok(true)
538}

540/// xcolor.sty:535-537 — when a non-`natural` target model is selected
541/// (`\ifconvertcolorsD`), a color being DEFINED is converted to
542/// `\XC@tgt@mod{<its model>}` before it is stored. `\XC@tgt@mod` is the
543/// identity for `natural` and a constant otherwise (xcolor.sty:141/145), so
544/// expanding it on the color's own model yields the model to store. See the
545/// `\selectcolormodel` definition for the pgf witness.
546fn convert_to_target_model(color: Color) -> Result<Color> {
547  if if_condition(&T_CS!("\\ifconvertcolorsD"))? != Some(true) {
548    return Ok(color);
549  }
550  let target = do_expand(mouth::tokenize_internal(TeXString::assembled(s!(
551    "\\XC@tgt@mod{{{}}}",
552    color.model()
553  ))))?
554  .to_string();
555  Ok(color.convert(target.trim()))

558/// Perl: ParseXColor($models, $specs, $tomodel)
559pub(crate) fn parse_xcolor(models: Option<&str>, specs: &str, tomodel: Option<&str>) -> Color {
560  // Perl ParseXColor (xcolor.sty.ltxml L218-241) has NO entry trim at all;
561  // keep only whitespace normalization here.
562  let specs = specs.trim();
563  let color = if let Some(models_str) = models {
564    let models_str = models_str.trim();
565    if models_str.is_empty() {
566      decode_color(specs)
567    } else {
568      // Check for tomodel prefix: "tomodel:model/model/..."
569      let (effective_tomodel, models_str) = if let Some(colon_pos) = models_str.find(':') {
570        let tm = &models_str[..colon_pos];
571        (Some(tm.to_string()), &models_str[colon_pos + 1..])
572      } else {
573        (tomodel.map(|s| s.to_string()), models_str)
574      };
575      // Split by /
576      let model_list: Vec<&str> = models_str.split('/').collect();
577      let spec_list: Vec<&str> = specs.split('/').collect();
578      if model_list.len() != spec_list.len() {
579        // Perl xcolor.sty.ltxml L228-231:
580        // `Error('unexpected', $specs, ..., "Length of color model_list must
581        //  be same as spec_list.", "models is '$models'; specs is '$specs'");
582        // return Black();`
583        let _ = (|| -> Result<()> {
584          Error!(
585            "unexpected",
586            specs,
587            format!(
588              "Length of color model_list must be same as spec_list (models is '{models_str}'; specs is '{specs}')"
589            )
590          );
591          Ok(())
592        })();
593        return BLACK;
594      }
595      // Choose first model (target model matching is TODO)
596      let model = model_list[0].trim();
597      // Perl L240-242: balanced-pair trim only (see trim_color_spec).
598      let spec = trim_color_spec(spec_list[0]);
599      let mut c = if model == "named" {
600        lookup_color_obj(spec)
601      } else {
602        convert_extended_to_core(model, spec)
603      };
604      if let Some(tm) = effective_tomodel {
605        c = c.convert(&tm);
606      }
607      c
608    }
609  } else {
610    decode_color(specs)
611  };
612  if let Some(tm) = tomodel {
613    color.convert(tm)
614  } else {
615    color
616  }

1853  for pair in id_pairs.split(',') {
1854    let pair = pair.trim();
1855    if pair.is_empty() {
1856      continue;
1857    }
1858    let (name, from) = if let Some(eq_pos) = pair.find('=') {
1859      (pair[..eq_pos].trim(), pair[eq_pos + 1..].trim())
1860    } else {
1861      (pair, pair)

1871      assign_value(&color_sty::color_key(name), stored, None);
