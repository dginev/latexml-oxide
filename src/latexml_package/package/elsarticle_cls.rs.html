505/// The contact of an elsarticle note text (`\fntext`, `\cortext`) its key's names carry, with the mark they print.
506fn marked_note_text(
507  prefix: &str,
508  role: &str,
509  mark: Option<String>,
510  key: Option<Tokens>,
511  text: Tokens,
512) -> Tokens {
513  let key = key.map(|key| key.to_string()).unwrap_or_default();
514  // (a plain key's label is settled as the entries are read, `els_settle_note_labels`: its own here, so every label
515  // the notes request is defined)
516  let label = if els_plain_key(&key) && !reading_author_fields() {
517    els_settle_key(&key);
518    s!("{LABEL_REF}{{{}}}", els_note_ref(prefix, &key))
519  } else {
520    els_note_label(prefix, &key)
521  };
522  let mark = mark
523    .map(|mark| s!("\\lx@fm@entry@attr{{_fnmark}}{{{mark}}}"))
524    .unwrap_or_default();
525  let mut call = mouth::tokenize_internal(TeXString::assembled(s!(
526    "\\lx@add@contact[label={{{label}}},role={role}]{{{mark}"
527  )))
528  .unlist();
529  call.extend(text.unlist());
530  call.push(T_END!());
531  Tokens::new(call)
532}

534/// The entry an elsarticle key names, of the ones whose `\elsLabel` writes it ([`els_key_entry`]).
535#[derive(Clone, Copy, PartialEq, Eq)]
536enum ElsEntry {
537  Place,
538  Footnote,
539  Correspondence,
540}
541
542/// The entry `key` names in elsarticle's one key space: the one whose label the .aux writes last — an `\address`'s or
543/// `\affiliation`'s (at shipout) over an `\fntext`'s over a `\cortext`'s (elsarticle.cls:126-131, :766-778, :1376-1399).
544fn els_key_entry(key: &str) -> Option<ElsEntry> {
545  if recorded_els_key(ELS_PLACE_KEYS, key) {
546    Some(ElsEntry::Place)
547  } else if recorded_els_key(ELS_FNTEXT_KEYS, key) {
548    Some(ElsEntry::Footnote)
549  } else if recorded_els_key(ELS_CORTEXT_KEYS, key) {
550    Some(ElsEntry::Correspondence)
551  } else {
552    None
553  }
554}
555
556/// Whether the `\cortext` (`kind` "cor") or `\fntext` ("fn") keyed `key` is no name's: its key names another entry, and
557/// for a `\cortext` no `\corref` names it (`\corref` prints its ∗ whatever the key's number, elsarticle.cls:188-190).
558fn els_note_shadowed(kind: &str, key: &str) -> bool {
559  let key = key.trim();
560  if key.is_empty() {
561    return false;
562  }
563  match (kind, els_key_entry(key)) {
564    (_, Some(ElsEntry::Place)) => true,
565    ("cor", Some(ElsEntry::Footnote)) => {
566      !recorded_els_key(ELS_CORREF_KEYS, key) && els_outside_mark("cor").trim() != key
567    },
568    _ => false,
569  }
570}
571
572/// A `\cortext`'s or `\fntext`'s label: its key's, `cor:<key>`/`fn:<key>`, or, where the key names another entry, none a
573/// name reaches — no `:`, so Perl's prefix-stripped fallback (`relocate_annotations`) finds no name listing the key.
574fn els_note_label(kind: &str, key: &str) -> String {
575  if els_note_shadowed(kind, key) {
576    s!("{kind}@{}", key.trim())
577  } else {
578    s!("{kind}:{}", els_label_key(key))
579  }
580}

582/// Whether `key` reads back as written in a deferred label ([`els_note_ref`]): no space, brace or control sequence.
583fn els_plain_key(key: &str) -> bool {
584  let key = key.trim();
585  !key.is_empty() && !key.contains(|c: char| c.is_whitespace() || "\\{}%#".contains(c))
586}

588/// The deferred label (`\lx@label@ref`) of the note of kind `kind` (`cor`, `fn`) keyed `key`.
589fn els_note_ref(kind: &str, key: &str) -> String { s!("els@note@{kind}@{}", key.trim()) }

591/// Record the label ([`els_note_label`]) of each note keyed `key` as its deferred label, read at the flush: the entry a
592/// key names is the one whose `\elsLabel` the .aux wrote last (elsarticle.cls:126-131), whatever the order the document
593/// gives them in, so a note read before the entry its key names is decided as one read after it (1711.08299; OD
594/// #1067). Settled for each key as it is read (`els_settle_key`).
595pub(super) fn els_settle_note_labels(key: &str) {
596  if !lookup_bool(ELS_NOTE_LABELS) || reading_author_fields() || !els_plain_key(key) {
597    return;
598  }
599  for (kind, list) in [("cor", ELS_CORTEXT_KEYS), ("fn", ELS_FNTEXT_KEYS)] {
600    if recorded_els_key(list, key) {
601      set_label_ref(&els_note_ref(kind, key), &els_note_label(kind, key));
602    }
603  }
604}
