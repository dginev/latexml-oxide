140/// jheppub.sty's markup of an author's contacts, which the PoS template's pos.sty v2 shares (the same commands to the
141/// same effect, pos.sty:359-430: `pos_sty`): the labelled affiliations, the footnote text without its mark given
142/// inside an `\author`, the e-mail list printed under the affiliations, and the dedication.
143#[rustfmt::skip]
144pub fn jhep_contact_markup() -> Result<()> {
145  // (its label printed before the line, jheppub.sty `\item[$^{#1}$]`: one that prints, `_labelprints`, the mark a
146  // legend's note shows, OXIDIZED_DESIGN_DIVERGENCES #961; jcappub's `\affiliation[*]{Corresponding author}`,
147  // 2204.11919)
148  DefMacro!("\\affiliation OptionalSemiverbatim {}",
149    "\\lx@add@contact[role=affiliation,label={#1},_labelprints=true]{#2}");
150  // jheppub.sty:84-89 `\newcommand\note[2][]`: a footnote's text without its mark, for use inside `\author`; the
151  // optional argument is the mark it prints the text under (the footnote counter then not stepped), else the next
152  // footnote number. Perl's jheppub.sty.ltxml:39 `\note{}` takes one argument, so `\note[*]{Corresponding authors.}`
153  // read `[` as the note and the rest into the name ("Jia-Jun Wu*]Corresponding authors" and a note "[", 2606.26596;
154  // 1510.03434, 2404.00867; KNOWN_PERL_ERRORS #1033). The note is placed as an unmarked one is (inside an `\author`
155  // that author's, given apart from the names the author's before it: 2404.00867's `\note[*]{Corresponding author:
156  // steinh@princeton.edu}` after `\author[*\,b,c]{Paul J. Steinhardt,}`); its mark (`_mark`) is the legend the
157  // corresponding-author step of an address list reads (`marked_corresponding_author`, OXIDIZED_DESIGN_DIVERGENCES
158  // #809).
159  DefMacro!("\\note[]{}", "\\lx@add@contact[role=note,_mark={#1}]{#2}");
160  // jheppub.sty:127-131 `\emailAdd` lists the addresses, each `\href{mailto:#1}{\tt #1}` (2301.03278), which
161  // `\maketitle` prints as one "E-mail:" line under the affiliations, attributing none. Perl's jheppub.sty.ltxml
162  // gives the n-th address to the n-th author (`labelseq=author`), which named the wrong author where the list is not
163  // in author order (2011.05971, 2212.09776, 2407.18115; jcappub 1806.05015, 2310.18996, 2407.07862; jinstpub
164  // 2311.16063's bross@fnal.gov on Andrea Bersani, 2307.12991, 1910.06438, 2311.00870, 2605.30709). Rust-only
165  // (OXIDIZED_DESIGN_DIVERGENCES #805, #801, #809): an address goes to the author it spells, else to the n-th author
166  // while the list is one address per author in their order (no address spells another author than the one at its
167  // place), else to the group's only author, else — the one address left on the author block — to the ONE author
168  // marked as the corresponding author, by a note naming them so or a mark whose legend says so (1901.08773's and
169  // 2510.11030's `\note{Corresponding author.}`; 1802.03677's note, whose number 1 two authors carry, leaves its
170  // address to the block; a `*` with no legend counts only where the address itself says "(Corresponding author)",
171  // 2308.06532),
172  // else to the author block (`\lx@add@contact@placed[owner=sequence]`, `sequenced_contact_labels`). An
173  // `\emailAdd{a@x, b@y}` list is an address each (2406.03543, 2505.23122; jinstpub 2211.05026, 2409.03065,
174  // 2112.00370). The family's rule: jcappub and jinstpub take it with this binding.
175  DefMacro!("\\emailAdd Semiverbatim", "\\lx@add@contact@list{\\lx@jhep@email}{#1}");
176  DefMacro!("\\lx@jhep@email{}", "\\lx@add@contact@placed[role=email,owner=sequence]{}{\\lx@address@urlfont{#1}}");
177  DefMacro!("\\dedicated{}", "\\lx@add@pubnote[role=dedication,name={}]{#1}");
178  Ok(())
