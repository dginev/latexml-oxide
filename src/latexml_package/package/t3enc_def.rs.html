1//! t3enc.def (tipa's T3 encoding, FUKUI Rei): the raw file, with its accent machinery set as Unicode combining marks.
2//!
3//! t3enc.def sets every tipa diacritic by stacking boxes: `\upper@accent`/`\lower@accent` (t3enc.def:268-294, the
4//! same as exaccent.sty's) put `\hbox{#2\char#4}` — slot `#4` of the font `#2` selects (`\tipaencoding`, or the current
5//! font for `\empty`) — above or below `\hbox{#5}` in a `\vbox`/`\vtop`, `\ipaclap` (:330-333) overlaps two boxes and
6//! `\ipabar` (:343-354) rules a bar across one. Read raw, the accent box came apart from its letter or was lost
7//! (`\textsubring{n}` printed "n", `\textbarb` a two-row table). Here the slot's glyph is the combining mark the tipa
8//! font draws there (t3_fontmap.rs `t3_accent_mark`: `\textsubring{n}` n̥, `\~*{a}` a̰, `\textroundcap{a}` ȃ,
9//! `\t{ts}` t͡s), applied as the kernel applies `\accent` (`\lx@applyaccent`); a box accent (`\Upperaccent`'s `b`
10//! form: tipa.sty:477 `\sups`) stays raw. In T3 the kernel accents `\'`, `\~`, `\r` … are t3enc.def's own (:489-516,
11//! with their `*` forms below the letter), as LaTeX's encoding-dispatched accents are. OXIDIZED_DESIGN_DIVERGENCES
12//! #781. Witnesses 1802.05444, 2405.01972.
13use crate::{
14  package::t3_fontmap::{AccentPlace, t3_accent_mark},
15  prelude::*,
16};
17
18/// The state value carrying the mark `\lx@tipa@accent@mark` decoded to `\lx@tipa@accent@apply`: the combining mark and
19/// the slot's standalone glyph, tab-separated; empty when the slot has none.
20const MARK_KEY: &str = "lx@tipa@accent@mark";
21
22/// The kernel accents t3enc.def:489-516 declares for T3, and tipa.sty:391's `\|` (the math ∥ elsewhere).
23const T3_ACCENTS: [&str; 17] = [
24  "\\`", "\\'", "\\^", "\\~", "\\\"", "\\H", "\\r", "\\v", "\\u", "\\=", "\\.", "\\b", "\\c",
25  "\\d", "\\t", "\\k", "\\|",
26];
27
28/// The macro-made T3 symbols (t3enc.def:523-560) Unicode has a letter for: barred b, d and g, the crossed two and
29/// crossed inverted glottal stop, and the kerned ligatures ʣ ʥ ʨ (tipauni.sty's `\textdzlig` "02A3 …). The others keep
30/// t3enc.def's construction (`\ipabar` → a stroke overlay below; `\textctdctzlig` the two letters).
31const T3_MADE_LETTERS: [(&str, &str); 8] = [
32  ("\\T3\\textbarb", "\u{0180}"),
33  ("\\T3\\textbard", "\u{0111}"),
34  ("\\T3\\textbarg", "\u{01E5}"),
35  ("\\T3\\textcrtwo", "\u{01BB}"),
36  ("\\T3\\textcrinvglotstop", "\u{01BE}"),
37  ("\\T3\\textdzlig", "\u{02A3}"),
38  ("\\T3\\textdctzlig", "\u{02A5}"),
39  ("\\T3\\texttctclig", "\u{02A8}"),
40];

42fn current_encoding() -> String {
43  lookup_font()
44    .and_then(|font| font.get_encoding().map(|encoding| encoding.to_string()))
45    .unwrap_or_else(|| "OT1".to_string())
46}

48/// The mark of slot `slot` set `place` in the current font: T3's from its glyph table, another encoding's (an
49/// exaccent-style `\upperaccent` in T1 text) from the kernel's accent combiners of the slot's standalone glyph.
50fn accent_mark(slot: i64, place: AccentPlace) -> Option<(char, String)> {
51  let encoding = current_encoding();
52  let standalone = u8::try_from(slot)
53    .ok()
54    .and_then(|code| font::decode_str(code, Some(encoding.clone()), false))
55    .map(|glyph| with(glyph, |g| g.to_string()))
56    .unwrap_or_default();
57  let mark = if encoding == "T3" {
58    t3_accent_mark(slot, place)
59  } else {
60    let combiner = match place {
61      AccentPlace::Below => "accent_combiner_below",
62      AccentPlace::Above | AccentPlace::Over => "accent_combiner_above",
63    };
64    lookup_mapping(combiner, &standalone)
65      .map(|mark| mark.to_string())
66      .and_then(|mark| mark.chars().next())
67  }?;
68  let standalone = if standalone.is_empty() {
69    mark.to_string()
70  } else {
71    standalone
72  };
73  Some((mark, standalone))
74}
75
76LoadDefinitions!({
77  InputDefinitions!("t3enc", extension => Some("def".into()), noltxml => true);
78
79  // Run inside `\begingroup <font switch> … \endgroup`: the slot is decoded in the font `#2` selected, as
80  // t3enc.def:269 `\setbox\@ne\hbox{#2\char#4}` sets it, and handed on past the group.
81  DefPrimitive!("\\lx@tipa@accent@mark{}{}", sub[(place, slot)] {
82    let place = match place.to_string().as_str() {
83      "below" => AccentPlace::Below,
84      "over" => AccentPlace::Over,
85      _ => AccentPlace::Above,
86    };
87    let (slot, _) = read_braced(Tokens::new(slot.revert()), read_number)?;
88    let slot = slot.value_of();
89    let value = match accent_mark(slot, place) {
90      Some((mark, standalone)) => s!("{mark}\t{standalone}"),
91      None => {
92        let encoding = current_encoding();
93        if !already_reported(&s!("tipa_accent_{encoding}_{slot}_{place:?}")) {
94          Warn!("missing_font_encoding", encoding,
95            s!("Font encoding {encoding} has no combining mark for the accent in slot {slot} set {place:?}: \
96                the letter is printed without it"));
97        }
98        String::new()
99      },
100    };
101    assign_value(MARK_KEY, value, Some(Scope::Global));
102  });
103  // The letter with the mark decoded just before: the kernel's `\accent` composition (tex_character.rs
104  // `apply_accent`, precomposed where Unicode has the letter: `\'a` á). Read when it is expanded, before the letter is
105  // digested, so a nested accent (`\textdotbreve`, t3enc.def:452-454) decodes its own.
106  DefMacro!("\\lx@tipa@accent@apply{}", sub[(letter)] {
107    let value = lookup_string(MARK_KEY);
108    assign_value(MARK_KEY, String::new(), Some(Scope::Global));
109    let letter = letter.unlist();
110    match value.split_once('\t') {
111      Some((mark, standalone)) => {
112        let mut tokens = vec![
113          T_CS!("\\lx@applyaccent"), T_CS!("\\lx@tipa@accent@apply"),
114          Token::new(mark, Catcode::OTHER), T_OTHER!(standalone), T_BEGIN!(),
115        ];
116        tokens.extend(letter);
117        tokens.push(T_END!());
118        Tokens::new(tokens)
119      },
120      None => Tokens::new(letter),
121    }
122  }, protected => true);
123
124  // t3enc.def:268-294: the `a` form's accent is slot #4 of the font #2 selects; the `b` form's is material, set raw.
125  Let!("\\lx@tipa@raw@upper@accent", "\\upper@accent");
126  Let!("\\lx@tipa@raw@lower@accent", "\\lower@accent");
127  for (cs, raw, place) in [
128    ("\\upper@accent", "\\lx@tipa@raw@upper@accent", "above"),
129    ("\\lower@accent", "\\lx@tipa@raw@lower@accent", "below"),
130  ] {
131    DefMacro!(T_CS!(cs), "{}{}[]{}{}", sub[(form, font, kern, accent, letter)] {
132      let form = form.unlist_cow().into_owned();
133      let font = font.unlist_cow().into_owned();
134      let accent = accent.unlist_cow().into_owned();
135      let letter = letter.unlist_cow().into_owned();
136      let mut tokens = Vec::new();
137      if matches!(form.as_slice(), [t] if t.to_string() == "a") {
138        tokens.push(T_CS!("\\begingroup"));
139        tokens.extend(font);
140        tokens.push(T_CS!("\\lx@tipa@accent@mark"));
141        tokens.extend([T_BEGIN!()]);
142        tokens.extend(Explode!(place));
143        tokens.extend([T_END!(), T_BEGIN!()]);
144        tokens.extend(accent);
145        tokens.extend([T_END!(), T_CS!("\\endgroup"), T_CS!("\\lx@tipa@accent@apply"), T_BEGIN!()]);
146        tokens.extend(letter);
147        tokens.push(T_END!());
148      } else {
149        tokens.push(T_CS!(raw));
150        tokens.extend(form);
151        tokens.extend(font);
152        tokens.push(T_OTHER!("["));
153        tokens.extend(kern.unlist_cow().into_owned());
154        tokens.push(T_OTHER!("]"));
155        tokens.push(T_BEGIN!());
156        tokens.extend(accent);
157        tokens.extend([T_END!(), T_BEGIN!()]);
158        tokens.extend(letter);
159        tokens.push(T_END!());
160      }
161      Tokens::new(tokens)
162    });
163  }
164  // t3enc.def:431-432: the tilde of slot 38 over the letter (`\ipaclap`), U+0334.
165  DefMacro!(
166    T_CS!("\\T3\\textsuperimposetilde"),
167    "{}",
168    "\\begingroup\\tipaencoding\\lx@tipa@accent@mark{over}{38}\\endgroup\\lx@tipa@accent@apply{#1}"
169  );
170  // t3enc.def:343-354: a bar ruled across the symbol, the short stroke overlay U+0335 — over a character run
171  // (`\textbarc`'s `{\tipaencoding c}`). Anything else keeps t3enc.def's construction: `\textdoublebaresh`
172  // (:535-537) bars an `\hb@xt@` box of another `\ipabar`, which the overlay printed as TeX source (rev77x).
173  // `\providecommand`ed: read again, t3enc.def keeps this binding's, and the raw one saved first stays.
174  if !IsDefined!(&T_CS!("\\lx@tipa@raw@ipabar")) {
175    Let!("\\lx@tipa@raw@ipabar", "\\ipabar");
176  }
177  DefMacro!("\\ipabar{}{}{}{}{}", sub[(symbol, height, width, left, right)] {
178    let symbol = symbol.unlist();
179    let character_run = symbol.iter().all(|t| {
180      matches!(t.get_catcode(), Catcode::LETTER | Catcode::OTHER | Catcode::BEGIN | Catcode::END)
181        || *t == T_CS!("\\tipaencoding")
182    });
183    let mut tokens = if character_run {
184      vec![
185        T_CS!("\\lx@applyaccent"), T_CS!("\\ipabar"), T_OTHER!("\u{0335}"), T_OTHER!("\u{00A0}\u{0335}"),
186        T_BEGIN!(),
187      ]
188    } else {
189      vec![T_CS!("\\lx@tipa@raw@ipabar"), T_BEGIN!()]
190    };
191    tokens.extend(symbol);
192    tokens.push(T_END!());
193    if !character_run {
194      for arg in [height, width, left, right] {
195        tokens.push(T_BEGIN!());
196        tokens.extend(arg.unlist());
197        tokens.push(T_END!());
198      }
199    }
200    Tokens::new(tokens)
201  });
202  for (cs, glyph) in T3_MADE_LETTERS {
203    def_primitive(
204      T_CS!(cs),
205      None,
206      Some(PrimitiveBody::String(pin(glyph))),
207      PrimitiveOptions::default(),
208    )?;
209  }
210
211  // t3enc.def read again (`\input{t3enc.def}` after tipa.sty) re-declares its T3 commands, and the kernel ones saved
212  // the first time stay saved: saving again would save these wrappers, which then expand to themselves (rev77x;
213  // guard `perfect_kernel_batch77x::t3enc_def_loaded_twice`).
214  if IsDefined!(&T_CS!("\\lx@tipa@native@'")) {
215    return Ok(());
216  }
217
218  // In T3 text the kernel accents are t3enc.def's (latex.ltx's text commands dispatch on `\cf@encoding`; ours are
219  // native and take the encoding's command only here), and `\|` is tipa.sty's diacritic shorthand (`\|[{t}` t̪), once
220  // tipa.sty has declared it: elsewhere, and in math, they are what they were. The command stays an expandable macro
221  // that expands to the native one (a `\csname` or `\edef` sees what it saw before, OXIDIZED_DESIGN #170); the T3 one
222  // is reached through a protected step, as LaTeX's dispatcher is robust.
223  for name in T3_ACCENTS {
224    let cs = T_CS!(name);
225    let native = T_CS!(s!("\\lx@tipa@native@{}", &name[1..]));
226    let t3 = T_CS!(s!("\\T3{name}"));
227    let t3_step = T_CS!(s!("\\lx@tipa@T3@{}", &name[1..]));
228    Let!(native, cs);
229    DefMacro!(t3_step, "", sub[_args] { Tokens!(t3) }, protected => true);
230    DefMacro!(cs, "", sub[_args] {
231      if !lookup_bool("IN_MATH") && current_encoding() == "T3" && IsDefined!(&t3) {
232        Tokens!(t3_step)
233      } else {
234        Tokens!(native)
235      }
236    });
237  }
238});
