11/// The class file in the paper's source that the family binding `binding` interprets: the class the document asked for
12/// when the loader answered it with this binding, else the binding's own name; `None` when the source does not have it.
13/// The request is the fallback's (`fallback_request`, when its `fallback_target` is this binding: a renamed `JINST_mod`
14/// or `myJINST`, 1307.5525, 1407.3938, and a path-prefixed request loaded as its basename's binding,
15/// `JINST-Sample-files/JINST`, whose class is in a subdirectory: 1504.01965, 1410.4420, 1805.09245; `load_class`), or
16/// the load's own `\@currname` when it names the binding with a directory (the raw load of a basename-keyed package
17/// binding resolves its directory the same way, `input_definitions`).
18pub fn requested_shipped_class(binding: &str) -> Result<Option<String>> {
19  let is_binding = |name: &str| {
20    let name = name.rsplit(['/', '\\']).next().unwrap_or_default();
21    let name = strip_cls(name);
22    name.eq_ignore_ascii_case(binding)
23  };
24  let fallback_request = lookup_string("fallback_request");
25  let currname = Expand!(Tokens!(T_CS!("\\@currname"))).to_string();
26  let class = if !fallback_request.is_empty() && is_binding(&lookup_string("fallback_target")) {
27    strip_cls(&fallback_request)
28  } else if is_binding(&currname) {
29    strip_cls(&currname)
30  } else {
31    binding
32  };
33  Ok(
34    find_file(
35      class,
36      Some(FindFileOptions {
37        ext_type: Some(Cow::Borrowed("cls")),
38        forbid_ltxml: true,
39        ..Default::default()
40      }),
41    )
42    .map(|_| class.to_string()),
43  )
44}

46/// A stand-in binding's answer when the source lacks the class it reads raw ([`requested_shipped_class`] is `None`):
47/// OmniBus, with the loader's missing-file status and `missing_file` Warning for a class it has no binding for
48/// (`load_class`; Perl Package.pm:2717-2728), which the binding's presence kept the loader from raising — pdflatex
49/// stops on the missing file. Not again for a request the prefix fallback answered with this binding: the loader
50/// warned for it. an.cls (astro-ph0611016), cimento, compositio, jinst, the apa family.
51pub fn omnibus_without_copy(binding: &str) -> Result<()> {
52  let fallback_target = lookup_string("fallback_target");
53  let answered = !lookup_string("fallback_request").is_empty()
54    && strip_cls(
55      fallback_target
56        .rsplit(['/', '\\'])
57        .next()
58        .unwrap_or_default(),
59    )
60    .eq_ignore_ascii_case(binding);
61  if !answered {
62    let name = Expand!(Tokens!(T_CS!("\\@currname"))).to_string();
63    let name = if name.is_empty() {
64      binding.to_string()
65    } else {
66      name
67    };
68    note_status(LogStatus::Missing, Some(&s!("{name}.cls")));
69    Warn!(
70      "missing_file",
71      &name,
72      s!("Can't find binding for class {name} (using OmniBus)"),
73      "Anticipate undefined macros or environments"
74    );
75  }
76  LoadClass!("OmniBus");
77  Ok(())
78}

102/// The text of the file `file` of type `ext` in the paper's source (not a binding), its bytes read as Latin-1
103/// characters, or the virtual file a `{filecontents}` wrote. A binding that serves every version of a generic file
104/// reads the copy TeX would load where the versions conflict (epsf's size reset, `epsf_sty`).
105pub fn shipped_file_text(file: &str, ext: &str) -> Option<String> {
106  find_file(
107    file,
108    Some(FindFileOptions {
109      ext_type: Some(Cow::Owned(ext.to_string())),
110      forbid_ltxml: true,
111      ..Default::default()
112    }),
113  )
114  .and_then(|path| {
115    vfs_read(&path).or_else(|| {
116      std::fs::read(path)
117        .ok()
118        .map(|bytes| bytes.iter().map(|&b| b as char).collect::<String>())
119    })
120  })
121}

123/// Does `text` (a class file's) load `package` with `\RequirePackage`/`\usepackage` outside a comment, alone or in a
124/// list (`\RequirePackage{graphicx,epstopdf}`), its options spanning lines or not (aa.cls's
125/// `\RequirePackage[colorlinks=true,⏎ linkcolor=blue, …]{hyperref}`)? A load inside a conditional or a hook
126/// (`\IfFileExists`, `\AtBeginDocument`) counts; `\RequirePackageWithOptions` does not.
127pub fn class_text_loads(text: &str, package: &str) -> bool {
128  let code = class_code(text);
129  ["\\RequirePackage", "\\usepackage"].iter().any(|command| {
130    code.match_indices(command).any(|(at, _)| {
131      let rest = code[at + command.len()..].trim_start();
132      let rest = match rest.strip_prefix('[') {
133        Some(options) => options
134          .split_once(']')
135          .map(|(_, after)| after)
136          .unwrap_or_default(),
137        None => rest,
138      };
139      rest
140        .trim_start()
141        .strip_prefix('{')
142        .and_then(|list| list.split_once('}'))
143        .is_some_and(|(list, _)| list.split(',').any(|name| name.trim() == package))
144    })
145  })
146}

216/// A stand-in binding reads the paper's copy of the package it answers (`stand_in_request`) raw, as pdflatex loads it
217/// (acl_sty, aaai_sty, iclr_sty); `false` when the source has none, which the raw load reports as the loader does.
218/// The read is the loader's: after it, the step the loader takes after any package it reads raw
219/// (`\lx@package@loaded@raw`, which a binding's own read skips) — usenix2019_v3.sty's `\abstract` heading stays out of
220/// the creators (2301.04841).
221pub fn input_shipped_package() -> Result<bool> {
222  let request = stand_in_request()?;
223  let shipped = !request.is_empty() && shipped_file_text(&request, "sty").is_some();
224  InputDefinitions!(&request, noltxml => true, extension => Some(Cow::Borrowed("sty")));
225  if shipped && IsDefined!(&T_CS!("\\lx@package@loaded@raw")) {
226    Digest!(mouth::tokenize_internal(TeXString::assembled(s!(
227      "\\lx@package@loaded@raw{{{request}}}"
228    ))))?;
229  }
230  Ok(shipped)
231}

239/// A binding's raw read of its copy (`read`), then the marks the copy's `\maketitle` sets, by the body TeX records
240/// (DG 2026-11-17; revMTREDUCE B1, S2): the one the lock refused as `read` ran, else at `\begin{document}` the one the
241/// copy's own `\AtBeginDocument` code defines (acl.sty:147-150) or its stand-in copy's ([`declare_stand_in_copy_title_marks`]).
242pub fn read_copy_title_marks(
243  lines: &'static [TitleMarkLine],
244  read: impl FnOnce() -> Result<bool>,
245) -> Result<bool> {
246  let before = refused_maketitle();
247  // (runs ahead of the copy's own `\AtBeginDocument` code, registered as it is read)
248  RawTeX!(r"\AtBeginDocument{\global\let\lx@title@copy@before\lx@dropped@maketitle}");
249  let shipped = read()?;
250  match refused_maketitle() {
251    Some(now) if !before.is_some_and(|before| Rc::ptr_eq(&before, &now)) => {
252      declare_body_title_marks(&now, lines, BoundMaketitle::Refused)?;
253    },
254    _ if shipped => declare_stand_in_copy_title_marks(lines)?,
255    _ => {},
256  }
257  Ok(shipped)
258}

260/// A binding declares the title marks of its stand-in copy's `\maketitle`, read after it (`content::read_stand_in_copy`:
261/// jmlr2e, neurips, ijcai), or of one the copy's `\AtBeginDocument` code defines: at `\begin{document}`, by the body TeX
262/// records ([`read_copy_title_marks`]). The lines wait in the state, under the copy's file, with the number its stand-in
263/// copy gets.
264pub fn declare_stand_in_copy_title_marks(lines: &'static [TitleMarkLine]) -> Result<()> {
265  // (the copy's record is its file's basename, `content::new_copy_provides_record`: a request from a subdirectory,
266  // `\usepackage{sub/neurips_2022}`, names it by that too — revMTREDUCEc B1, 2210.05974, 2510.04543)
267  let request = stand_in_request()?;
268  let file = s!(
269    "{}.sty",
270    request.rsplit(['/', '\\']).next().unwrap_or(&request)
271  );
272  let mut record = next_copy_provides_id().to_string();
273  for (line, marks) in lines {
274    record.push_str(&s!("\n{line}\t{marks}"));
275  }
276  assign_value(
277    &s!("{COPY_TITLE_LINES}{file}"),
278    Stored::String(pin(&record)),
279    Some(Scope::Global),
280  );
281  DefMacro!("\\lx@title@copy@marks{}", sub[(file)] {
282    declare_copy_title_marks_at_begin_document(&file.to_string())?;
283    Ok(Tokens::new(Vec::new()))
284  });
285  RawTeX!(&s!(r"\AtBeginDocument{{\lx@title@copy@marks{{{file}}}}}"));
286  Ok(())
287}

289/// The state key of a copy's title-mark lines ([`declare_stand_in_copy_title_marks`]), its file after it.
290const COPY_TITLE_LINES: &str = "lx_title_copy_lines:";

292/// `\lx@title@copy@marks{<file>}` at `\begin{document}` ([`declare_stand_in_copy_title_marks`]): the `\maketitle` the
293/// copy's `\AtBeginDocument` code defined, which the lock refused there (since `\lx@title@copy@before`), else its stand-in
294/// copy's.
295fn declare_copy_title_marks_at_begin_document(file: &str) -> Result<()> {
296  let record = lookup_string(&s!("{COPY_TITLE_LINES}{file}"));
297  let mut rows = record.lines();
298  let from = rows
299    .next()
300    .and_then(|id| id.parse::<u32>().ok())
301    .unwrap_or(1);
302  let lines: Vec<(&str, &str)> = rows.filter_map(|row| row.split_once('\t')).collect();
303  let before = lookup_definition(&T_CS!("\\lx@title@copy@before"))?;
304  match refused_maketitle() {
305    Some(now) if !before.is_some_and(|before| Rc::ptr_eq(&before, &now)) => {
306      declare_body_title_marks(&now, &lines, BoundMaketitle::Refused)
307    },
308    _ => match copy_provides_maketitle_of(file, from) {
309      Some((id, body)) => match body_title_marks(&body, &lines) {
310        Some(marks) => declare_package_title_marks(&marks, BoundMaketitle::Copy(id)),
311        None => Ok(()),
312      },
313      None => Ok(()),
314    },
315  }
316}

325/// Declare the title marks the refused `\maketitle` `definition` sets ([`body_title_marks`]), if any.
326fn declare_body_title_marks(

384/// The document's class options (`\@classoptionslist`) in the order given, for a binding that reads them as its class
385/// does rather than in the order they are declared (REVTeX's journal, `revtex4_support_sty::selected_journal`).
386pub fn class_options() -> Result<Vec<String>> {
387  let Some(defn) = lookup_definition(&T_CS!("\\@classoptionslist"))? else {
388    return Ok(Vec::new());
389  };
390  let Some(ExpansionBody::Tokens(body)) = defn.get_expansion() else {
391    return Ok(Vec::new());
392  };
393  Ok(
394    body
395      .to_string()
396      .split(',')
397      .map(|given| given.trim().to_string())
398      .filter(|given| !given.is_empty())
399      .collect(),
400  )
401}

403/// The braced argument of the first `call` in `text` (a class file's) outside a comment that is followed by one, its
404/// braces balanced: the default a copy gives a setting command (sn-jnl.cls's `\def\interntxt#1{…}\interntxt{Work done
405/// as an intern at Westlake University.}`, 2606.24824's copy :598).
406pub fn class_code_argument(text: &str, call: &str) -> Option<String> {
407  let code = class_code(text);
408  code.match_indices(call).find_map(|(at, _)| {
409    let rest = code[at + call.len()..].strip_prefix('{')?;
410    let mut depth = 0usize;
411    for (i, c) in rest.char_indices() {
412      match c {
413        '{' => depth += 1,
414        '}' if depth == 0 => return Some(rest[..i].to_string()),
415        '}' => depth -= 1,
416        _ => {},
417      }
418    }
419    None
420  })
421}

423/// `text` without its comments (`%` to the end of the line). A line ends where TeX Live's TeX ends it, at a carriage
424/// return, a line feed or the pair: a copy saved with classic Mac line ends (23 of the census's 164 sn-jnl.cls:
425/// 2609.04047's, 2106.13602's), which pdflatex reads line by line (2609.04047's "Table A1"), was read here as one
426/// comment, and none of the code a binding looks for in it was found.
427pub fn class_code(text: &str) -> String {
428  text
429    .split("\r\n")
430    .flat_map(|chunk| chunk.split(['\n', '\r']))
431    .map(|line| line.split('%').next().unwrap_or_default())
432    .collect::<Vec<_>>()
433    .join("\n")
434}

436/// `name` without a `.cls` extension, in any case.
437fn strip_cls(name: &str) -> &str {
438  match name.len().checked_sub(4) {
439    Some(cut) if name.is_char_boundary(cut) && name[cut..].eq_ignore_ascii_case(".cls") => {
440      &name[..cut]
441    },
442    _ => name,
443  }
444}

446/// A file a binding reads raw — a TeX Live class or package, or an author's class the paper ships — with the binding's
447/// adjustments, each placed where the file defines what it adjusts ([`read_raw`]). One declaration per binding, so the
448/// read, the no-copy fallback and the order of the adjustments cannot drift between bindings.
449pub struct RawRead<'a> {
450  /// The file's name, without its extension (`pasa`; a family binding's, chosen as it loads: `compositio`).
451  pub name:     &'a str,
452  /// Its extension (`cls`, `sty`).
453  pub ext:      &'static str,
454  /// An author's class, not in TeX Live: read from the paper's copy ([`requested_shipped_class`]); without one the
455  /// document gets OmniBus and the missing-file warning, as for any class the paper names and lacks.
456  pub shipped:  bool,
457  /// `(cs, definition)`: held while the file is read — `definition` (TeX) replaces the meaning of `cs`, locked against
458  /// the file's own definition — and the meaning before returns after it (pasa.cls's era `\@popfilename`).
459  pub held:     &'static [(&'static str, &'static str)],
460  /// Meanings the file overrides with copies of an older kernel's: left to the file while it is read (its own code may
461  /// need them), and the meaning before restored after it (pasa.cls's era `\@pushfilename`).
462  pub restored: &'static [&'static str],
463  /// `(setter, body)`: a store the file defines (`\aaabstract`), redefined after the read to `body`, a frontmatter API
464  /// call with `#1` the setter's argument (the body grammar of K11's `STORE_SETTERS`) — only when the file defines it.
465  pub setters:  &'static [(&'static str, &'static str)],
466  /// TeX run last, for what the declarations above do not cover.
467  pub after:    &'static str,
468}
469
470impl<'a> RawRead<'a> {
471  /// The plain declaration of `name`.`ext`: read raw, nothing held, no setter, nothing after.
472  pub const fn new(name: &'a str, ext: &'static str) -> Self {
473    RawRead {
474      name,
475      ext,
476      shipped: false,
477      held: &[],
478      restored: &[],
479      setters: &[],
480      after: "",
481    }
482  }
483}
484
485/// Read the file `spec` declares, raw, with its adjustments in their order: the held meanings in place and locked, the
486/// file, the held and restored meanings back to what they were before it, the setters the file defined redefined, then
487/// `after`. `false` when a shipped class is not in the paper's source: the document then has OmniBus
488/// ([`omnibus_without_copy`]) and the binding stops there.
489pub fn read_raw(spec: &RawRead) -> Result<bool> {
490  let file = if spec.shipped {
491    let Some(class) = requested_shipped_class(spec.name)? else {
492      omnibus_without_copy(spec.name)?;
493      return Ok(false);
494    };
495    class
496  } else {
497    spec.name.to_string()
498  };
499  let kept = |cs: &str| s!("\\lx@raw@kept@{}", &cs[1..]);
500  for cs in spec.held.iter().map(|(cs, _)| cs).chain(spec.restored) {
501    Let!(T_CS!(&kept(cs)), T_CS!(*cs));
502  }
503  for (cs, definition) in spec.held {
504    RawTeX!(*definition);
505    AssignValue!(&s!("{cs}:locked") => true, Some(Scope::Global));
506  }
507  InputDefinitions!(&file, noltxml => true, extension => Some(Cow::Owned(spec.ext.to_string())));
508  for (cs, _) in spec.held {
509    AssignValue!(&s!("{cs}:locked") => false, Some(Scope::Global));
510  }
511  for cs in spec.held.iter().map(|(cs, _)| cs).chain(spec.restored) {
512    Let!(T_CS!(*cs), T_CS!(&kept(cs)), Scope::Global);
513  }
514  for (setter, body) in spec.setters {
515    if IsDefined!(&T_CS!(*setter)) {
516      RawTeX!(&s!("\\long\\def{setter}#1{{{body}}}"));
517    }
518  }
519  if !spec.after.is_empty() {
520    RawTeX!(spec.after);
521  }
522  Ok(true)
523}
