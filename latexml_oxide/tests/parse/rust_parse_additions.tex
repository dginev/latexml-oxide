% Rust-only addition to parse/functions.tex (b53811d5c0); Perl MathParser.pm:1291 ApplyDelimited
% A known function applied to a paren comma-list spreads the items: f@(a,b), not f@(vector@(a,b));
% an unknown keeps its vector wrap (OXIDIZED_DESIGN_MATH #18)
\section{Multi-argument calls flatten}

\[ \max(a,b) \]

% Rust-only addition to parse/relations.tex (7d847c113f; the 2026-06-22 `distribute_list_relation` surpass)

% presentation keeps the list as the relation's left side; a list RIGHT of a relation stays list@(0<x,y)
\section{Comma-list left of a relation (distributed XMDual)}

\[ x,y \le z \]
