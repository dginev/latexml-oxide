% brucemiller/LaTeXML#2563: loading `svg` after `subcaption` must not break
% `\subfloat`. LaTeXML's svg.sty.ltxml:19 does RequirePackage('subfig'), and
% subfig's RawTeX tail defines `\lx@subfloat@figure` only inside
% `\@ifundefined{c@subfigure}{\newsubfloat{figure}}{}` — a guard on the COUNTER.
% subcaption (loaded first) already defined `c@subfigure`, so Perl 0.8.8 skips
% \newsubfloat, leaves `\lx@subfloat@figure` undefined, and `\subfloat` dumps its
% args as literal text: `<p>[This is a caption.]This is a figure.</p>`.
%
% The real svg.sty (:66-73) loads no subfig (KPE #322), so here `\subfloat` is
% subcaption's own, which reads a lone optional as the caption
% (subcaption.sty:282-285): a figure panel captioned "This is a caption.".
\documentclass{article}
\usepackage{subcaption}
\usepackage{svg}
\begin{document}
\begin{figure}
    \subfloat[This is a caption.]{This is a figure.}
\end{figure}
\end{document}
