\usepackage[style=numeric]{biblatex}
\DeclareSourcemap{
  \maps[datatype=bibtex]{
    \map{
      \step[fieldsource=edition, match=\regexp{^(\d+)\.\s+(ed.|edition)$}, replace=\regexp{$1}]
    }
    \map[overwrite]{
      \step[fieldsource=shortjournal, final]
      \step[fieldset=journal, origfieldval]
    }
  }
}
