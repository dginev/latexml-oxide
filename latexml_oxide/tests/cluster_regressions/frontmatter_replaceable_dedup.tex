% Guard: a re-added title (subtitle/toctitle/date) keeps only ONE entry — a later one
% REPLACES the earlier — while creators accumulate, and so do abstracts: a second
% {abstract} digested in place is printed where it stands (pdflatex prints both, the
% vendored Perl keeps both). Upstream added %ReplaceableFrontmatterTags (bb048fc1, #2573)
% and removed it (23f3acfa, #2767); we keep it for titles, dates and keywords (which
% bindings re-add as their list grows), not for abstracts.
% OXIDIZED_DESIGN #154. Witnesses arXiv 2002.09766 (appendix re-\icmltitle duplicated
% <title>), 2511.21969 (nested {abstract}), 1006.5288 (amsart: the second abstract
% replaced the first, losing its keywords and MSC). Uses \lx@add@frontmatter@now
% directly so the double-add is deterministic (class \maketitle is once-only).
\documentclass{article}
\makeatletter
\begin{document}
\lx@add@frontmatter@now{ltx:title}{First Title}
\lx@add@frontmatter@now{ltx:title}{Second Title}
\lx@add@frontmatter@now{ltx:creator}{AuthorA}
\lx@add@frontmatter@now{ltx:creator}{AuthorB}
\begin{abstract}First abstract.\end{abstract}
\begin{abstract}Second abstract.\end{abstract}
\lx@frontmatterhere
Body paragraph.
\end{document}
