% Declarations scoped to labels on nodes that stay on the streaming spine (a top-level theorem, a table) and
% on a labelled equation: the scope resolves in the eager and the streaming path alike (57aq review: the
% streaming label index held only spilled content's labels, so these errored and lost their roles).
% Perl: x ID inside the theorem, y inside the table, z inside the equation; UNKNOWN outside. And a bare `\label`
% before the first unit lands on the root, whose id no segment records: its scope `f` is FUNCTION everywhere, in
% every spilled segment too (57aq re-review; 1703.09326 shape).
\documentclass{article}
\usepackage{latexml}
\newtheorem{thm}{Theorem}
\begin{document}

\lxDeclare[scope=label:sec:intro,role=FUNCTION]{$f$}%

para two $x$ and more words here.

para three $x$ and more words here.
\end{thm}
